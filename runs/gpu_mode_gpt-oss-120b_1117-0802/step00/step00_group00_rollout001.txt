@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    total_batches, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # grid: (grid_n, grid_m, total_batches)
    pid_n = tl.program_id(0)   # column tile
    pid_m = tl.program_id(1)   # row tile
    batch_id = tl.program_id(2)

    # decode batch index into (b_idx,d_idx)
    b_idx = batch_id // hidden_dim
    d_idx = batch_id % hidden_dim

    # compute offsets for M,N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute pointer to start of this batch slice
    a_base = a_ptr + b_idx * batch_stride_a + d_idx
    b_base = b_ptr + b_idx * batch_stride_b + d_idx
    c_base = c_ptr + batch_id * batch_stride_c

    # using tl.arange for K offset will be updated within loop
    # we will compute a_ptrs for each K tile inside loop

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Offsets for k tile
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for K loaded elements
        k_valid = offs_k < K

        # Compute pointers
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & (offs_n[None, :] < N), other=0.0)

        # dot product: a (M,K) dot b (K,N)
        acc += tl.dot(a, b)

    # Write out
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # LayerNorm (norm) weight and bias
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    # use torch layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask application
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute for kernel
    left_perm = left.permute(0, 3, 1, 2)  # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)

    # Cast to float16 for kernel
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    B, H, N, _ = left_fp16.shape
    total_batches = B * H

    # allocate output
    out_perm = torch.empty((B, H, N, N), device=device, dtype=torch.float32)

    # Determine block sizes:
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # adjust block sizes if N smaller than block dimension: ensure at least 1
    # but block sizes can be static

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), total_batches)
    # launch kernel
    tri_mul_matmul_kernel[grid](
        left_fp16, right_fp16, out_perm,
        left_fp16.stride(0), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(3), right_fp16.stride(2),
        out_perm.stride(0), out_perm.stride(2), out_perm.stride(3),
        N, N, N,
        total_batches, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # permute back
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # multiply by out_gate
    out = out * out_gate

    # final linear
    to_out_weight = weights['to_out.weight']
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul "outgoing" implementation.

The heavy O(N³) operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is accelerated with a custom Triton kernel that computes a batch of
matrix multiplications `left @ rightᵀ` for every hidden dimension slice.
All other operations (LayerNorm, Linear projections, gating, final
normalisation) are performed with PyTorch for simplicity.
The kernel works on `float16` inputs and accumulates in `float32`,
producing a `float32` output that matches the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    # strides / batch offsets
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    # dimensions
    M, N, K,
    hidden_dim,                # hidden dimension size (to split combined batch)
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many batches where:
       A: (batch, M, K)   -> here (B*H, N, N)
       B: (batch, N, K)   -> stored as (K,N) in memory (i.e. we treat it as Bᵀ)
       C: (batch, M, N)
    """
    pid_n = tl.program_id(0)      # column tile
    pid_m = tl.program_id(1)      # row tile
    pid_batch = tl.program_id(2)  # combined batch index (B * hidden)

    # decode batch index into original batch and hidden slice
    batch_idx = pid_batch // hidden_dim
    hidden_idx = pid_batch % hidden_dim

    # pointers to the start of this batch slice (including hidden offset)
    a_base = a_ptr + batch_idx * batch_stride_a + hidden_idx
    b_base = b_ptr + batch_idx * batch_stride_b + hidden_idx
    c_base = c_ptr + pid_batch * batch_stride_c

    # tile offsets within M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for valid K entries
        k_mask = offs_k < K

        # pointers for the current K‑tile
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles with appropriate masks
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # dot product (FP16 inputs, FP32 accumulation)
        acc += tl.dot(a, b)

    # store the result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    store_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operator, fully GPU‑accelerated.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask: [B, N, N] (bool or float, broadcastable)
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Apply mask if provided (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)        # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating mechanisms (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    #   left_perm, right_perm : (B, hidden, N, N) with hidden stride == 1
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to float16 for the kernel (still keep fp32 versions for later steps)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    total_batches = B * hidden_dim

    # Output tensor for the kernel (float32)
    out_perm = torch.empty(
        (B, hidden_dim, N, N), device=device, dtype=torch.float32
    )

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # columns
        triton.cdiv(N, BLOCK_M),   # rows
        total_batches,             # batch*hidden
    )

    # launch kernel
    _tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_perm,
        # strides
        left_fp16.stride(0), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(3), right_fp16.stride(2),
        out_perm.stride(0), out_perm.stride(2), out_perm.stride(3),
        # dimensions
        N, N, N,
        hidden_dim,
        # block sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Rearrange back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Final normalisation, gating and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
