def custom_kernel(data):
    import torch, triton, triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # device
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32

    # get weight and bias tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input x
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    # torch.nn.functional.linear expects weight (out_features, in_features)
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if provided
    if mask_tensor is None:
        mask = torch.ones_like(input_tensor[...,0])  # shape B,N,N
    else:
        mask = mask_tensor
    # broadcast mask
    mask_ = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_
    right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to bring hidden_dim to second axis
    # left: (B,N,N,hidden_dim) -> (B,hidden_dim,N,N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # right_T: swap second and third axes to get (B, N, N, hidden_dim) -> (B,N,N,hidden_dim) with swapped axes
    right_T = right_fp16.permute(0, 2, 1, 3).contiguous()
    right_perm = right_T.permute(0, 3, 1, 2).contiguous()  # (B, hidden_dim, N, N)

    B, H, N, _ = left_perm.shape  # B batch, H = hidden_dim

    # Flatten batch*hidden_dim
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Launch kernel
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), left_flat.shape[0])
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape result back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # shape (B,N,N,hidden_dim)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out
```

"""
TriMul "outgoing" forward implementation using Triton for the heavy
pairwise multiplication over the sequence dimension. The kernel computes,
for each batch and hidden channel, a batched matrix multiplication:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where left and right are the projected and gated pairwise tensors.
All remaining linear, gating and normalization operations are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for tensors of shape (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,
    stride_Bb, stride_Br, stride_Bc,
    stride_Cb, stride_Cr, stride_Cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index (flattened B*H)
    pid_m = tl.program_id(0)          # block row in M dimension
    pid_n = tl.program_id(1)          # block col in N dimension

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this tile (k=0)
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
    b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Br + offs_n[None, :] * stride_Bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over tiles of dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k

        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Br

    # write result back
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Mask (if provided)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(batch_size, seq_len, seq_len, device=device, dtype=dtype)
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1).to(dtype)           # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy pairwise multiplication (batched GEMM) using Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to bring hidden dimension to the second axis
    # shape becomes (B, hidden_dim, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # transpose the second/third axes of right and then bring hidden dimension forward
    right_T = right_fp16.permute(0, 2, 1, 3).contiguous()
    right_perm = right_T.permute(0, 3, 1, 2).contiguous()

    # Flatten batch & hidden dim → (B*hidden_dim, N, N)
    left_flat = left_perm.view(-1, seq_len, seq_len)
    right_flat = right_perm.view(-1, seq_len, seq_len)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides for the flattened tensors
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Tile sizes (tuned for typical GPUs; powers of two and multiples of 32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(seq_len, BLOCK_M),   # blocks in M dimension
        triton.cdiv(seq_len, BLOCK_N),   # blocks in N dimension
        left_flat.shape[0]               # batch*hidden_dim
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len, seq_len, seq_len,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = out_flat.view(batch_size, hidden_dim, seq_len, seq_len)
    out = out.permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight → [B, N, N, dim]

    return out
