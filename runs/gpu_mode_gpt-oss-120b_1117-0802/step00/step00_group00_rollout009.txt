def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # LayerNorm on dim.
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # x = x.to(torch.float32) (already likely float32)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Mask
    mask_ = mask.unsqueeze(-1).type_as(x)
    left = left * mask_
    right = right * mask_
    left = left * left_gate
    right = right * right_gate

    # Cast left/right to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Gather strides
    stride_A_b = left_fp16.stride(0)
    stride_A_i = left_fp16.stride(1)
    stride_A_k = left_fp16.stride(2)
    stride_A_d = left_fp16.stride(3)  # should be 1

    stride_B_b = right_fp16.stride(0)
    stride_B_i = right_fp16.stride(1)
    stride_B_k = right_fp16.stride(2)
    stride_B_d = right_fp16.stride(3)

    stride_C_b = out_fp32.stride(0)
    stride_C_i = out_fp32.stride(1)
    stride_C_j = out_fp32.stride(2)
    stride_C_d = out_fp32.stride(3)

    # Block size selection heuristics
    # for N up to 1024, choose BLOCK_M = BLOCK_N = 64, BLOCK_K = 32 (common)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # numbers of blocks
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    # Launch kernel
    tri_mul_kernel[grid](

        stride_A_b, stride_A_i, stride_A_k,
        stride_B_b, stride_B_k, stride_B_i,  # careful ordering for transposed

def custom_kernel(data):
    """
    Triangle Multiplicative Update (outgoing) forward pass 
    using a fused Triton kernel for the N×N×N reduction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ----- LayerNorm on input -----
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ----- Linear projections -----
    # All linear layers have bias=False.
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Gating tensors -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Apply mask (if provided) -----
    if mask is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Prepare tensors for Triton kernel (fp16 for speed) -----
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output tensor (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_ab = left_fp16.stride(0)   # batch stride
    stride_ai = left_fp16.stride(1)   # i (row) stride
    stride_ak = left_fp16.stride(2)   # k (col) stride

    stride_bb = right_fp16.stride(0)  # batch stride (same as left)
    stride_bk = right_fp16.stride(2)  # k stride (row in transposed view)
    stride_bn = right_fp16.stride(1)  # j stride (col in transposed view)

    stride_cb = out.stride(0)
    stride_ci = out.stride(1)  # i stride
    stride_cj = out.stride(2)  # j stride

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_ci, stride_cj,
        N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Output post‑processing -----
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out

        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape
    N = seq_len
    H = hidden_dim

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------- Mask (optional) ----------
    if mask is None:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    left = left * mask
    right = right * mask

    # ---------- Apply gating ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton kernel ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((bs, N, N, H), dtype=torch.float32, device=device)

    # Strides (in number of elements)
    stride_ab = left_fp16.stride(0)   # batch stride
    stride_ai = left_fp16.stride(1)   # i (row) stride
    stride_ak = left_fp16.stride(2)   # k (col) stride

    stride_bb = right_fp16.stride(0)  # batch stride (same as left)
    stride_bk = right_fp16.stride(2)  # k stride (row in transposed view)
    stride_bn = right_fp16.stride(1)  # j stride (col in transposed view)

    stride_cb = out.stride(0)         # batch stride
    stride_ci = out.stride(1)         # i stride
    stride_cj = out.stride(2)         # j stride

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs * H,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_ci,
        stride_cj,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
