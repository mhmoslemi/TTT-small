@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_z = tl.program_id(2)  # batch
    
    # offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # compute base pointers for this batch
    a_batch_ptr = a_ptr + pid_z * batch_stride_a
    b_batch_ptr = b_ptr + pid_z * batch_stride_b
    c_batch_ptr = c_ptr + pid_z * batch_stride_c
    
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for this k-tile
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K
        
        # pointers for A and B
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # Convert to float32 if loaded as float16
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        # dot product
        acc += tl.dot(a, b)
    
    # write output
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # ensure device
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask handling
    if config.get("nomask", False):
        mask = torch.ones_like(mask_tensor, dtype=x.dtype, device=device).unsqueeze(-1)
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=device).unsqueeze(-1)
    left = left * mask
    right = right * mask
    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # apply gates
    left = left * left_gate
    right = right * right_gate
    # contraction via Triton
    # Rearrange for batch*hidden dim GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    total_batch = B * H
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)
    out_flat = torch.empty_like(left_flat)  # float32
    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), total_batch)
    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides
        N, 1,       # stride_am, stride_ak
        1, N,       # stride_bk, stride_bn
        N, 1,       # stride_cm, stride_cn
        N * N, N * N, N * N,  # batch strides a, b, c
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],

    )

"""
TriMul forward implementation using Triton for the O(N³) contraction.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Project to a hidden dimension with two linear layers (left/right).
3. Apply an optional mask and per‑channel sigmoid gates.
4. Compute the contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that performs a batched matrix
   multiplication for each (batch, hidden) pair:  A @ Bᵀ.
5. Apply a second layer‑norm, an output gate, and the final linear
   projection back to the original dimension.

The heavy O(N³) part is handled by a tiled Triton GEMM‑like kernel
which works on the flattened batch × hidden dimension, keeping memory
accesses coalesced. All other operations remain in PyTorch for
readability.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul of A (M×K) and B (N×K) → C (M×N)
# The kernel computes C = A @ Bᵀ, i.e.  Σ_k A[i,k] * B[j,k].
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_am, stride_ak,                    # A: row‑stride, col‑stride
    stride_bk, stride_bn,                    # B (transposed view): K‑stride, N‑stride
    stride_cm, stride_cn,                    # C: row‑stride, col‑stride
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides
    M, N, K,                                 # matrix sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_z = tl.program_id(2)   # batch index (batch * hidden)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for this batch
    a_batch = a_ptr + pid_z * batch_stride_a
    b_batch = b_ptr + pid_z * batch_stride_b
    c_batch = c_ptr + pid_z * batch_stride_c

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A[i, k] slice
        a_ptrs = a_batch + (offs_m[:, None] * stride_am +
                            cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B_T[k, j] slice (B[j, k] in memory)
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk +
                            offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot product over K
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm +
                        offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton‑accelerated contraction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict: contains "dim", "hidden_dim", and "nomask" flags

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask handling ----------
    if config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction ----------
    # bring hidden dimension to the batch axis for a simple GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B_, H_, N_, _ = left_perm.shape
    total_batch = B_ * H_

    # flatten batch*hidden → (total_batch, N, N)
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)
    out_flat = torch.empty_like(left_flat)   # float32 output

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batch,
    )
    # stride arguments: A (M stride, K stride), B (K stride, N stride), C (M stride, N stride)
    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, 1,          # stride_am, stride_ak
        1, N,          # stride_bk, stride_bn (transposed view)
        N, 1,          # stride_cm, stride_cn
        N * N, N * N, N * N,   # batch strides for A, B, C
        N, N, N,               # M, N, K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Post‑processing ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
