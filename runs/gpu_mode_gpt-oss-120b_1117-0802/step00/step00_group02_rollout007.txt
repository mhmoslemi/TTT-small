        bs, seq_len, _, dim = input_tensor.shape
        hidden_dim = config["hidden_dim"]

        x = input_tensor
        mask = mask

        norm_weight = weights["norm.weight"]
        norm_bias = weights["norm.bias"]

        )

        left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
        right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
        # Gates
        left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
        out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

        left_proj = left_proj.contiguous()

            BLOCK=BLOCK,

        right = right_proj

"""
TriMul (outgoing) forward pass with a lightweight Triton kernel.

The heavy O(N³) contraction is performed by a batch GEMM (torch.bmm) after
projecting the input into a hidden dimension.  A tiny Triton kernel fuses the
masking and gating (element‐wise multiplication) for the two projected tensors,
working on blocks of (batch, i, j) and slices of the hidden dimension.
All other operations (LayerNorm, Linear, GELU, final projection) use PyTorch
which already runs on cuBLAS/cuDNN.  The result matches the reference
implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: in‑place fusion of mask, left/right gates onto the projected
# tensors.  It processes a block of (b,i,j) entries (flattened) and a slice of
# the hidden dimension (VEC elements).  The kernel is deliberately lightweight
# to keep register pressure low.
# --------------------------------------------------------------------------- #
@triton.jit
def apply_gate_mask_inplace_kernel(
    left_ptr,      # *float32  [B, N, N, H] (output, in‑place)
    left_gate_ptr,  # *float32  [B, N, N, H]
    right_ptr,     # *float32  [B, N, N, H] (output, in‑place)
    right_gate_ptr,# *float32  [B, N, N, H]
    mask_ptr,      # *float32  [B, N, N]  (scalar per pair)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK: tl.constexpr,   # number of (b,i,j) elements per program
    VEC: tl.constexpr,     # slice of hidden dimension (e.g. 32)
):
    pid_x = tl.program_id(0)            # block index over flattened (B,N,N)
    pid_y = tl.program_id(1)            # hidden‑dim slice index (0 .. H/VEC)

    total = B * N * N
    # ---- indices of the (B,N,N) elements handled by this program ----
    offs = pid_x * BLOCK + tl.arange(0, BLOCK)   # shape (BLOCK,)
    mask_valid = offs < total

    # ---- hidden‑dim slice handled by this program ----
    h_start = pid_y * VEC
    hi = tl.arange(0, VEC) + h_start               # shape (VEC,)

    # base address for the hidden slice of each (b,i,j) element
    # each element occupies H contiguous entries in memory
    base = offs * H + h_start                       # shape (BLOCK,)

    # ---- load scalar mask (broadcast later) ----
    mask_val = tl.load(mask_ptr + offs, mask=mask_valid, other=0.0)   # (BLOCK,)
    mask_vec = mask_val[:, None]   # (BLOCK,1)

    # ---- load slices -------------------------------------------------
    left = tl.load(left_ptr + base[:, None] + hi[None, :],
                   mask=mask_valid[:, None], other=0.0)
    left_gate = tl.load(left_gate_ptr + base[:, None] + hi[None, :],
                        mask=mask_valid[:, None], other=0.0)
    right = tl.load(right_ptr + base[:, None] + hi[None, :],
                    mask=mask_valid[:, None], other=0.0)
    right_gate = tl.load(right_gate_ptr + base[:, None] + hi[None, :],
                         mask=mask_valid[:, None], other=0.0)

    # ---- fuse mask & gates -------------------------------------------
    left_out = left * left_gate * mask_vec
    right_out = right * right_gate * mask_vec

    # ---- write back (in‑place) ----------------------------------------
    tl.store(left_ptr + base[:, None] + hi[None, :], left_out,
             mask=mask_valid[:, None])
    tl.store(right_ptr + base[:, None] + hi[None, :], right_out,
             mask=mask_valid[:, None])


# --------------------------------------------------------------------------- #
# Entry point expected by the autograder.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor    [B, N, N, C]   (float32, on CUDA)
        - mask         : torch.Tensor    [B, N, N]      (bool/int/float, on CUDA)
        - weights      : dict of torch.Tensor (all float32, on CUDA)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias) and gate computations
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Ensure contiguity for the Triton kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_float = mask.to(dtype=input_tensor.dtype).contiguous()

    # ------------------------------------------------------------------- #
    # 3) Fuse mask and gates with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------- #
    total_elements = B * N * N
    BLOCK = 1024                     # # (b,i,j) per program
    VEC = 32                         # slice of hidden_dim per program
    grid = (triton.cdiv(total_elements, BLOCK), triton.cdiv(hidden_dim, VEC))
    apply_gate_mask_inplace_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_float,
        B,
        N,
        hidden_dim,
        BLOCK=BLOCK,
        VEC=VEC,
    )
    # After the kernel, `left_proj` and `right_proj` now hold the masked+gated tensors.
    left = left_proj
    right = right_proj

    # ------------------------------------------------------------------- #
    # 4) Pairwise contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batch GEMM on (B * hidden_dim) matrices.
    # ------------------------------------------------------------------- #
    # Move hidden dimension to the batch axis for a batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------- #
    # 5) Final Normalisation, gating, and projection back to `dim`
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
