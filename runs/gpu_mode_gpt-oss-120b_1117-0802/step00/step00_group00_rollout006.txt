"""
TriMul (outgoing) forward pass with a Triton kernel for the cubic reduction.
The kernel computes, for each batch b and channel d:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the projected and gated tensors of shape
[B, N, N, hidden_dim].  The rest of the module (LayerNorms, Linear
projections, gating and final projection) is performed with PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM over the third index k for every hidden channel."""
    pid_i = tl.program_id(0)          # tile index along the i dimension
    pid_j = tl.program_id(1)          # tile index along the j dimension
    pid_bc = tl.program_id(2)         # combined batch * channel index

    # ------------------------------------------------------------------
    # Decode batch and hidden channel from pid_bc
    # ------------------------------------------------------------------
    b = pid_bc // D                   # batch index
    d = pid_bc % D                    # hidden channel index

    # ------------------------------------------------------------------
    # Compute tile start indices
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    i = i_base + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    j = j_base + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # ------------------------------------------------------------------
    # Offsets that are constant for the whole tile
    # ------------------------------------------------------------------
    batch_off_l = b * stride_l_b + d * stride_l_d
    batch_off_r = b * stride_r_b + d * stride_r_d
    batch_off_o = b * stride_o_b + d * stride_o_d

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in blocks
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        k = k_base + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            left_ptr
            + batch_off_l
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load right tile transposed so shape is (BLOCK_K, BLOCK_N)
        b_ptrs = (
            right_ptr
            + batch_off_r
            + k[:, None] * stride_r_k
            + j[None, :] * stride_r_j
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate matrix multiplication over the k‑block
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_off_o
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_ij)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]
            - mask_tensor  : [B, N, N] (binary or 0/1)
            - weights_dict : model parameters (see keys below)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # Triton kernel for the cubic reduction
    # ------------------------------------------------------------------
    out0 = torch.empty_like(left)  # [B,N,N,hidden_dim]

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out0,
        B,
        N,
        hidden_dim,
        left.stride(0),
        left.stride(1),
        left.stride(2),
        left.stride(3),
        right.stride(0),
        right.stride(1),
        right.stride(2),
        right.stride(3),
        out0.stride(0),
        out0.stride(1),
        out0.stride(2),
        out0.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out0,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out.to(torch.float32)
