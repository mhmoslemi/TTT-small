@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, D,
                   stride_la_b, stride_la_i, stride_la_k, stride_la_d,
                   stride_ra_b, stride_ra_i, stride_ra_k, stride_ra_d,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)
    # compute batch and channel indices
    b = pid_bd // D
    d = pid_bd % D

    # Compute offsets indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute valid mask
        cur_mask = mask_i[:, None] & mask_j[None, :] & mask_k[None, :]  # but we may need to mask k inside loads.

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b*stride_la_b + d*stride_la_d + \
                 offs_i[:, None] * stride_la_i + offs_k[None, :] * stride_la_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile: shape (BLOCK_K, BLOCK_N) (k dimension first)
        b_ptrs = right_ptr + b*stride_ra_b + d*stride_ra_d + \
                 offs_k[:, None] * stride_ra_k + offs_j[None, :] * stride_ra_i

        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b_tile)  # shape (BLOCK_M,BLOCK_N)

    # write back
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_d + \
               offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, D,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_d,
                   stride_r_b, stride_r_i, stride_r_k, stride_r_d,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_d,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    d = pid_bd % D

    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The kernel computes the core tensor contraction:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

where left/right are the projected and gated pairwise tensors.
The contraction is a batched matrix multiplication for each hidden
channel `d`.  The Triton kernel processes a (i, j) tile for a fixed
(batch, hidden) pair, iterating over the k‑dimension in blocks.
All other operations (LayerNorms, linear projections, gating) are
performed with PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = sum_k left[..., i, k, d] * right[..., j, k, d].

    Grid: (grid_i, grid_j, grid_bd) where
        grid_i = ceil(N / BLOCK_M)
        grid_j = ceil(N / BLOCK_N)
        grid_bd = B * D (batch * hidden channel)
    """
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile column
    pid_bd = tl.program_id(2)         # combined batch & hidden

    # Decode batch and hidden index
    b = pid_bd // D
    d = pid_bd % D

    # Offsets for the i (row) and j (col) dimensions of this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b * stride_l_b + d * stride_l_d + \
                 offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile from right (k then j): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + b * stride_r_b + d * stride_r_d + \
                 offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b_tile)

    # Store the result tile
    out_ptrs = out_ptr + b * stride_o_b + d * stride_o_d + \
               offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM over the third dimension using Triton.
    left/right: [B, N, N, D] (contiguous, float32)
    Returns out of same shape.
    """
    B, N, _, D = left.shape
    # Ensure contiguous layout for stride calculations
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides in number of elements (not bytes)
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    # Block sizes – tuned for B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tile dimension
        triton.cdiv(N, BLOCK_N),   # j‑tile dimension
        B * D                      # batch * hidden
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, D,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a Triton kernel
    for the O(N³) contraction.
    Arguments:
        data: Tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---- LayerNorm over the feature dimension ----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---- Linear projections (no bias) ----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---- Apply mask (if any) ----
    if mask is not None:
        m = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ---- Gating networks ----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---- Core Tensor Contraction via Triton ----
    out = _tri_mul_gemm(left, right)   # shape [B,N,N,hidden_dim]

    # ---- Output LayerNorm + gating ----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # ---- Final linear projection back to original dim ----
    out = F.linear(out, weights["to_out.weight"])
    return out
