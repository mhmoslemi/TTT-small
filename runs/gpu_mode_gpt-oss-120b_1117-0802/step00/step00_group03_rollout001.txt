@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    MASK_PTR, MASK_ENABLED: tl.constexpr,  # mask pointer optional
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute program ids:
    batch_idx = tl.program_id(0)  # combined batch dimension: b * hidden_dim
    pid_m = tl.program_id(1)  # i block index
    pid_n = tl.program_id(2)  # j block index

    # derive b, d from batch_idx:
    # batch_idx = b * hidden_dim + d
    b = batch_idx // hidden_dim
    d = batch_idx % hidden_dim

    # offsets for output tile:
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Use mask for out-of-bounds:
    m_mask = offs_m < N
    n_mask = offs_n < N

    # compute base pointer for A left (i,k):
    # left_ptr + b*stride_left_b + d*stride_left_d
    a_ptr = left_ptr + b * stride_left_b + d * stride_left_d
    # compute base pointer for B right (j,k) but we need B transposed shape (k,j)
    # right pointer base: right_ptr + b * stride_right_b + d * stride_right_d
    b_ptr = right_ptr + b * stride_right_b + d * stride_right_d

    # compute base pointer for output:
    out_ptr = out_ptr + b * stride_out_b + d * stride_out_d

    # Initialize accumulator to zero FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k block
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for k dimension:
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N) but we need B[k,j] => offset k*stride_right_k + j*stride_right_j
        b_ptrs = b_ptr + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0).to(tl.float32)

        # Compute dot:
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=m_mask[:, None] & n_mask[None, :])  # maybe store fp16 for speed? But final output expects fp32; we can store directly fp32.
    # We'll store fp32:
    tl.store(out_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])

"""
Optimized Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

The kernel computes the core contraction:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
where left/right are linear projections of a layer‑normalized 4‑D
input tensor. The rest of the TriMul forward pass (layer‑norms,
gates, and final projection) is performed in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------- Triton kernel ---------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,    # * const half*
    right_ptr,   # * const half*
    out_ptr,     # * float*
    B, N, D,     # shape parameters (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM where A has shape (B*D, N, N) and B is transposed.
    We treat each (batch, d) pair as an independent GEMM.
    """
    # Indexing
    batch_idx = tl.program_id(0)   # enumerates b*D + d
    pid_m = tl.program_id(1)      # output row block i
    pid_n = tl.program_id(2)      # output col block j

    # Derive batch (b) and hidden index (d)
    b = batch_idx // D
    d = batch_idx % D

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Guard masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b,d) slice
    a_base = left_ptr + b * stride_left_b + d * stride_left_d
    b_base = right_ptr + b * stride_right_b + d * stride_right_d
    c_base = out_ptr + b * stride_out_b + d * stride_out_d

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile : left[b,i,k,d]
        a_ptrs = a_base + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile : right[b,j,k,d] but we need (k,j) for GEMM
        # Access pattern: offset = k*stride_right_k + j*stride_right_j
        b_ptrs = b_base + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_base + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------- Host entry point --------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton kernel for the
    N^3 contraction. All other operations (layer‑norms, linear projections,
    gates, final projection) are carried out with PyTorch.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict mapping weight names to torch.Tensors
            - config: dict containing "dim" and "hidden_dim" (ints) and optional flags

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    assert C == dim, "Input last dimension must match config dim"

    # ---------- LayerNorm on input ----------
    # Using torch.nn.functional.layer_norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    # If nomask flag is set, ignore mask (treat as all ones)
    if config.get("nomask", False):
        mask_tensor = torch.ones((B, N, N), dtype=left.dtype, device=device)
    else:
        mask_tensor = mask.to(left.dtype)
    mask_exp = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]

    left = left * mask_exp
    right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the heavy contraction ----------
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()

    # ---------- Allocate output buffer ----------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ---------- Compute strides (in elements) ----------
    sB_left, sI_left, sK_left, sD_left = left_half.stride()
    sB_right, sJ_right, sK_right, sD_right = right_half.stride()
    sB_out, sI_out, sJ_out, sD_out = out.stride()

    # ---------- Triton launch parameters ----------
    # Reasonable block sizes for typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    total_batch = B * hidden_dim
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_half,
        right_half,
        out,
        B,
        N,
        hidden_dim,
        sB_left,
        sI_left,
        sK_left,
        sD_left,
        sB_right,
        sJ_right,
        sK_right,
        sD_right,
        sB_out,
        sI_out,
        sJ_out,
        sD_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel has completed before using the result
    torch.cuda.synchronize()

    # ---------- Post‑contraction processing ----------
    # LayerNorm on hidden dimension
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out_gated = out_norm * out_gate

    # Final linear projection back to dim
    result = F.linear(out_gated, weights["to_out.weight"])

    return result

    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    assert C == dim, "Input channel dimension does not match config['dim']"

    # ---------- Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if config.get("nomask", False):
        mask_tensor = torch.ones((B, N, N), dtype=left.dtype, device=device)
    else:
        mask_tensor = mask.to(left.dtype)
    mask_exp = mask_tensor.unsqueeze(-1)   # [B, N, N, 1]

    left = left * mask_exp
    right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the contraction ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ---------- Output buffer ----------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ---------- Strides (in elements) ----------
    sB_l, sI_l, sK_l, sD_l = left_fp16.stride()
    sB_r, sJ_r, sK_r, sD_r = right_fp16.stride()
    sB_o, sI_o, sJ_o, sD_o = out.stride()

    # ---------- Triton launch ----------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    total_batch = B * hidden_dim
    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        sB_l,
        sI_l,
        sK_l,
        sD_l,
        sB_r,
        sJ_r,
        sK_r,
        sD_r,
        sB_o,
        sI_o,
        sJ_o,
        sD_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ---------- Post‑contraction LayerNorm ----------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Output gate ----------
    out_gated = out_norm * out_gate

    # ---------- Final projection ----------
    result = F.linear(out_gated, weights["to_out.weight"])

    return result
