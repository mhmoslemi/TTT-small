"""
TriMul forward implementation with a small Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension.
2. Linear projections `left = x @ W_left`, `right = x @ W_right`.
3. (Optional) apply a binary mask to `left` and `right`.
4. Compute sigmoid gates `left_gate`, `right_gate`, `out_gate`.
   Apply the first two gates point‑wise to `left` and `right`.
5. Contract over the third dimension (k) using a batched GEMM:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is performed as a batch of `hidden_dim` independent matrix
   multiplications with `torch.bmm`.
6. Apply a second LayerNorm to the contracted result.
7. Fuse the element‑wise multiplication with `out_gate` using a
   Triton kernel.
8. Final linear projection back to the original `dim`.

Only the element‑wise multiplication with `out_gate` is performed in
Triton; the rest uses highly‑optimised PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(x_ptr, y_ptr, out_ptr, numel, BLOCK: tl.constexpr):
    """Element‑wise multiplication: out = x * y (both 1‑D flattened)."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < numel
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, x * y, mask=mask)

def custom_kernel(data):
    """
    Triton‑augmented forward pass of the TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of named torch.Tensor parameters
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # Ensure all weights are on the correct device & dtype
    norm_weight = norm_weight.to(device, dtype)
    norm_bias   = norm_bias.to(device, dtype)
    left_proj_weight   = left_proj_weight.to(device, dtype)
    right_proj_weight  = right_proj_weight.to(device, dtype)
    left_gate_weight   = left_gate_weight.to(device, dtype)
    right_gate_weight  = right_gate_weight.to(device, dtype)
    out_gate_weight    = out_gate_weight.to(device, dtype)
    to_out_norm_weight = to_out_norm_weight.to(device, dtype)
    to_out_norm_bias   = to_out_norm_bias.to(device, dtype)
    to_out_weight      = to_out_weight.to(device, dtype)

    # -----------------------------------------------------------------
    # 1) Input layer‑norm
    # -----------------------------------------------------------------
    # Input shape: [B, N, N, dim]
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3) Optional mask
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> cast to float and broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gates
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Contract over the third dimension (k) using batched GEMM
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # Prepare tensors for batched GEMM:
    # left: [B, N, N, H] -> A: [B, H, N, N] (i, k)
    # right: [B, N, N, H] -> B: [B, H, N, N] (j, k) need transpose k ↔ j
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    # Collapse batch and hidden_dim into a single batch dimension
    left_flat = left_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N) # [B*H, N, N]

    # Batched matrix multiplication: (i,k) @ (k,j) -> (i,j)
    out_flat = torch.bmm(left_flat, right_flat)       # [B*H, N, N]

    # Restore original shape: [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 7) Element‑wise multiply with out_gate (Triton)
    # -----------------------------------------------------------------
    out_norm = out_norm.contiguous()
    out_gate = out_gate.contiguous()
    out_mul = torch.empty_like(out_norm)

    numel = out_norm.numel()
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)

    elementwise_mul_kernel[grid](
        out_norm,
        out_gate,
        out_mul,
        numel,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # -----------------------------------------------------------------
    final_out = F.linear(out_mul, to_out_weight)  # [B, N, N, dim]

    return final_out.to(torch.float32)
