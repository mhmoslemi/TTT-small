```
@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr, weight_ptr, bias_ptr,
    stride_b0, stride_i, stride_j, stride_c,
    eps, DIM: tl.constexpr, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    total = tl.num_programs(0) # not available; but we compute.

    # Compute coordinates:
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # Compute base pointer offset for the start of dimension
    offset = b * stride_b0 + i * stride_i + j * stride_j

    # First pass: compute sum
    sum_val = tl.zeros([1], dtype=tl.float32)
    for start in range(0, DIM, BLOCK):
        cur_off = tl.arange(0, BLOCK) + start
        mask = cur_off < DIM
        x = tl.load(inp_ptr + offset + cur_off * stride_c, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / DIM

    # Second pass: compute variance
    var_sum = tl.zeros([1], dtype=tl.float32)
    for start in range(0, DIM, BLOCK):
        cur_off = tl.arange(0, BLOCK) + start
        mask = cur_off < DIM
        x = tl.load(inp_ptr + offset + cur_off * stride_c, mask=mask, other=0.0)
        diff = x - mean
        var_sum += tl.sum(diff * diff, axis=0)

    var = var_sum / DIM

    inv_std = 1.0 / tl.sqrt(var + eps)

    # Third pass: write normalized values
    for start in range(0, DIM, BLOCK):
        cur_off = tl.arange(0, BLOCK) + start
        mask = cur_off < DIM
        x = tl.load(inp_ptr + offset + cur_off * stride_c, mask=mask, other=0.0)
        w = tl.load(weight_ptr + cur_off, mask=mask, other=0.0)
        b = tl.load(bias_ptr + cur_off, mask=mask, other=0.0)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(out_ptr + offset + cur_off * stride_c, y, mask=mask)

"""
TriMul (outgoing) forward pass with Triton‑fused LayerNorm.

The kernel `layernorm_kernel` computes a per‑position LayerNorm over the
last dimension (C) of a 4‑D tensor [B, N, N, C].  It is used twice:
* on the input tensor before any linear projection,
* on the intermediate tensor after the triangle‑multiplicative update.

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed efficiently by reshaping the tensors and invoking a batched
matrix‑multiply (`torch.bmm`).  All linear layers and sigmoid gates are
handled with PyTorch functional calls.

The implementation respects the required signature:
    custom_kernel(data) → output
where `data` = (input, mask, weights, config).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,               # pointers
    weight_ptr, bias_ptr,           # layer‑norm scale & bias
    stride_b, stride_i, stride_j, stride_c,  # input strides
    eps,                            # epsilon (runtime scalar)
    DIM: tl.constexpr, BLOCK: tl.constexpr, N: tl.constexpr,
):
    """LayerNorm over the last dimension of a [B, N, N, DIM] tensor.

    Each program instance processes one (b, i, j) location.
    """
    pid = tl.program_id(0)

    # ---- decode (b, i, j) from pid ---------------------------------
    seq_sq = N * N
    b = pid // seq_sq
    rem = pid % seq_sq
    i = rem // N
    j = rem % N

    # base offset for the start of the C‑dimension
    base = b * stride_b + i * stride_i + j * stride_j

    # ---------- first pass: compute mean -------------------------------
    acc = tl.zeros([1], dtype=tl.float32)
    for off in range(0, DIM, BLOCK):
        cur = tl.arange(0, BLOCK) + off
        mask = cur < DIM
        x = tl.load(inp_ptr + base + cur * stride_c, mask=mask, other=0.0, dtype=tl.float32)
        acc = acc + tl.sum(x, axis=0)
    mean = acc / DIM

    # ---------- second pass: compute variance -------------------------
    acc = tl.zeros([1], dtype=tl.float32)
    for off in range(0, DIM, BLOCK):
        cur = tl.arange(0, BLOCK) + off
        mask = cur < DIM
        x = tl.load(inp_ptr + base + cur * stride_c, mask=mask, other=0.0, dtype=tl.float32)
        diff = x - mean
        acc = acc + tl.sum(diff * diff, axis=0)
    var = acc / DIM
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- third pass: write normalized values --------------------
    for off in range(0, DIM, BLOCK):
        cur = tl.arange(0, BLOCK) + off
        mask = cur < DIM
        x = tl.load(inp_ptr + base + cur * stride_c, mask=mask, other=0.0, dtype=tl.float32)
        w = tl.load(weight_ptr + cur, mask=mask, other=0.0, dtype=tl.float32)
        b_ = tl.load(bias_ptr + cur, mask=mask, other=0.0, dtype=tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b_
        tl.store(out_ptr + base + cur * stride_c, y, mask=mask)


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm using the Triton kernel defined above.
    Input shape: [B, N, N, C]; weight & bias: [C].
    Returns a tensor of the same shape and dtype as `x`.
    """
    B, N, N2, C = x.shape
    assert N == N2, "Expect square pairwise dimension"
    stride_b, stride_i, stride_j, stride_c = x.stride()
    out = torch.empty_like(x)

    # Choose a block size that divides nicely (multiple of 32 for warp efficiency)
    BLOCK = 64 if C > 64 else 32

    grid = (B * N * N,)
    layernorm_kernel[grid](
        x,
        out,
        weight,
        bias,
        stride_b,
        stride_i,
        stride_j,
        stride_c,
        eps,
        DIM=C,
        BLOCK=BLOCK,
        N=N,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]   (float32, on CUDA)
            - mask_tensor  : [B, N, N]        (float32 or bool)
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : contains "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Extract weight tensors (all stored without bias except LayerNorms)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure everything is float32 and contiguous on the same device
    device = input_tensor.device
    for name in [
        "norm.weight", "norm.bias", "left_proj.weight", "right_proj.weight",
        "left_gate.weight", "right_gate.weight", "out_gate.weight",
        "to_out_norm.weight", "to_out_norm.bias", "to_out.weight",
    ]:
        w = weights[name]
        if w.dtype != torch.float32 or not w.is_contiguous():
            weights[name] = w.to(dtype=torch.float32, device=device, copy=True)

    # Refresh local references (now guaranteed float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (Triton)
    # -----------------------------------------------------------------
    x_norm = triton_layernorm(input_tensor, norm_weight, norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)          # left_proj
    right = F.linear(x_norm, right_proj_weight)        # right_proj

    # -----------------------------------------------------------------
    # 3) Apply mask if provided (mask has shape [B, N, N])
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid of linear maps)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # -----------------------------------------------------------------
    # 5) Modulate projections by gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6) Triangle multiplicative update: batched GEMM over the "k" axis
    # -----------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to [B*H, N, N] for batched matrix multiply
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # out_flat: [B*H, N, N] = left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Second LayerNorm (Triton) over hidden dimension
    # -----------------------------------------------------------------
    out_norm = triton_layernorm(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 8) Apply the output gate
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate

    # -----------------------------------------------------------------
    # 9) Final linear projection back to original dimension
    # -----------------------------------------------------------------
    output = F.linear(out_norm, to_out_weight)   # shape [B, N, N, dim]

    return output
