Define kernel:

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)  # combined batch*D index
    pid_m = tl.program_id(1)  # i block
    pid_n = tl.program_id(2)  # j block

    # compute batch and hidden index from pid_z:
    # pid_z = b * D + d
    # b = pid_z // D; d = pid_z % D
    b = pid_z // D
    d = pid_z % D

    # compute start indices
    i0 = pid_m * BLOCK_M

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # get dtype maybe input is float32 already.
    # Normalize input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5
    )
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if provided (mask shape [bs, seq_len, seq_len])
    # mask broadcast to [bs, seq_len, seq_len, 1]
    mask_ = mask.unsqueeze(-1)
    left = left * mask_
    right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast left and right to half for faster kernel
    left_hp = left.to(torch.float16)
    right_hp = right.to(torch.float16)

    # Allocate output tensor
    out = torch.empty_like(left_hp, dtype=torch.float32, device=left_hp.device)

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides
    stride_l_b = left_hp.stride(0)
    stride_l_i = left_hp.stride(1)
    stride_l_k = left_hp.stride(2)
    stride_l_d = left_hp.stride(3)

    stride_r_b = right_hp.stride(0)
    stride_r_j = right_hp.stride(1)
    stride_r_k = right_hp.stride(2)
    stride_r_d = right_hp.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_d = out.stride(3)

    # Grid
    grid = (bs * hidden_dim, triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))

    tri_mul_kernel[grid](
        left_hp,
        right_hp,
        out,
        bs, seq_len, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_j, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Now out is float32
    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_z // D
    d = pid_z % D

    # Compute start coordinates
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Bounds for i and j
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left tile (M,K)
        left_ptrs = left_ptr + b*stride_l_b + i[:, None]*stride_l_i + k[None, :] * stride_l_k + d * stride_l_d
        left_mask = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0, dtype=tl.float16).to(tl.float32)

        # Load right tile (K,N) (actually K rows, N cols)
        # Right has dimensions (B, N_j, N_k, D)
        right_ptrs = right_ptr + b*stride_r_b + k[:, None] * stride_r_k + j[None, :] * stride_r_j + d * stride_r_d
        right_mask = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0, dtype=tl.float16).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # Store result if within bounds
    out_ptrs = out_ptr + b*stride_o_b + i[:, None]*stride_o_i + j[None, :] * stride_o_j + d * stride_o_d
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel for the
core contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, Linear projections, gating) are performed in
PyTorch. The contraction is executed as a batched GEMM (B × hidden_dim
independent matrix multiplies) using block‑tiling (64×64 tiles) and a
float‑16 kernel with float‑32 accumulation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_z = tl.program_id(0)   # combined batch * hidden_dim
    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # decode batch index and hidden‑dim index
    b = pid_z // D
    d = pid_z % D

    # tile origin
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # ranges inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left tile (M × K) ----------
        # address: left[b, i, k, d]
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + d * stride_l_d
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # -------- right tile (K × N) ----------
        # address: right[b, j, k, d]  (we load as (k, j) to get right^T)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
            + d * stride_r_d
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0,
                        dtype=tl.float16).to(tl.float32)

        # accumulate dot product
        acc += tl.dot(left, right)

    # -------- write output (M × N) ----------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + d * stride_o_d
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Execute the TriMul (outgoing) operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (bool or float)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Layer‑norm on the input
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply mask (mask is broadcasted on the channel dim)
    mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton
    # Cast to half for the kernel (weights already FP32)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output (FP32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                      device=left_h.device)

    # block sizes (tuned for the B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for the tensors
    s_l_b, s_l_i, s_l_k, s_l_d = left_h.stride()
    s_r_b, s_r_j, s_r_k, s_r_d = right_h.stride()
    s_o_b, s_o_i, s_o_j, s_o_d = out.stride()

    # launch grid: (batch*hidden, i‑tiles, j‑tiles)
    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_d,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_d,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Final LayerNorm, gating and projection to original dimension
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate  # gating

    # final linear projection (no bias)
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
