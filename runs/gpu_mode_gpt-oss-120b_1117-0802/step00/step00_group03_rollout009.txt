```
def trmul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)
    # compute tile coordinates
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets for M,N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointer for this batch
    batch_offset_a = batch * stride_ab
    batch_offset_b = batch * stride_bb
    batch_offset_c = batch * stride_cb

    # offsets within tile for K loop
    offs_k = tl.arange(0, BLOCK_K)

    # pointers at start of tile
    a_ptrs = a_ptr + batch_offset_a + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + batch_offset_b + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a and b tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # convert to fp16 then to fp32 for dot product
        a = a.to(tl.float16)
        b = b.to(tl.float16)
        acc += tl.dot(a, b).to(tl.float32)
        # increment pointers for next k tile
        a_ptrs += BLOCK_K * stride_ak
```

```
def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    import torch.nn.functional as F
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True) # if missing default True

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask before gating (if any)
    if not nomask:
        mask_exp = mask.unsqueeze(-1).type_as(x)  # cast to same dtype
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Permute & reshape for batched matrix multiplication
    B, N, _, _ = left_f16.shape
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # allocate out_mat
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=left_mat.device)

    # determine block sizes (tune for N)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # For very small N maybe reduce block size to avoid wasted threads
    if N <= 64:
        BLOCK_M = BLOCK_N = BLOCK_K = 32
    # For large N maybe increase block?
    # Compute strides
    stride_am = left_mat.stride(1)   # K dimension
    stride_ak = left_mat.stride(2)   # 1
    stride_ab = left_mat.stride(0)   # batch stride (M*K)
    stride_bk = right_mat.stride(2)  # 1 (since original col stride)
    stride_bn = right_mat.stride(1)  # N (row stride)
    stride_bb = right_mat.stride(0)  # batch stride (K*N)
    stride_cm = out_mat.stride(1)    # N
    stride_cn = out_mat.stride(2)    # 1
    stride_cb = out_mat.stride(0)    # batch stride

    # Launch kernel
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), B * hidden_dim)
    trmul_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out_perm = out_mat.view(B, hidden_dim, N, N)
    out = out_perm.permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)
    out_f32 = out.to(torch.float32)

    # Final layer norm and linear
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out_norm = out_norm * out_gate  # out_gate is float32
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
```

```
@triton.jit
def trmul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)
    # Compute tile indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch * stride_ab
    b_base = b_ptr + batch * stride_bb
    c_base = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N) - note we treat B as transposed
        b = tl.load(
            b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Convert to fp16 for dot
        a_f16 = a.to(tl.float16)
        b_f16 = b.to(tl.float16)
        acc += tl.dot(a_f16, b_f16).to(tl.float32)

    # Store result
    c = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c, acc.to(tl.float16), mask=mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trmul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B^T for a batch of matrices using tiling.
    A,B have shape (batch, M, K) and (batch, N, K) respectively.
    """
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets within M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch * stride_ab
    b_base = b_ptr + batch * stride_bb
    c_base = c_ptr + batch * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles of A and B (B is accessed as transposed)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # dot product (M x K) @ (K x N) -> (M x N)
        acc += tl.dot(a, b).to(tl.float32)

    # Write back result in fp16
    c = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c, acc.to(tl.float16), mask=mask)

    Triton-accelerated implementation of the TriMul forward pass (outgoing version).
    The heavy contraction `einsum('... i k d, ... j k d -> ... i j d')` is performed
    by a tiled matrix‑multiply kernel that computes C = A @ Bᵀ for each hidden
    dimension independently.

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------- LayerNorm -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------- Linear projections -------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- Optional masking -------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------- Gates -------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------- Apply gates -------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------- Cast to FP16 for the contraction -------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Release intermediate FP32 tensors early
    del left, right, left_gate, right_gate

    B, N, _, _ = left_f16.shape

    # Bring hidden dimension to leading axis for batched GEMM
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(-1, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=left_mat.device)

    # ------------------- Triton GEMM (A @ Bᵀ) -------------------
    # Simple heuristic for block sizes – good for all provided test shapes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    if N <= 64:

    stride_bn = right_mat.stride(1)     # N  (row stride of original matrix, used as col stride after transpose)

    stride_cn = out_mat.stride(2)       # 1
    stride_cb = out_mat.stride(0)       # batch* N * N

        stride_ab, stride_bb, stride_cb,

    out_perm = out_mat.view(B, hidden_dim, N, N)

"""
TriMul forward (outgoing) with Triton acceleration.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication C = A @ Bᵀ for each hidden
dimension independently.  A and B have shape (B·hidden, N, N) after a
permute+view, and a tiled Triton kernel computes the product.  All other
operations (LayerNorm, Linear, sigmoids, gating) are done with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.

    A : (batch, M, K)   (row‑major)
    B : (batch, N, K)   (row‑major, accessed transposed)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)          # tile over M×N
    batch = tl.program_id(1)        # batch*hidden dimension

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch entry
    a_base = a_ptr + batch * stride_ab
    b_base = b_ptr + batch * stride_bb
    c_base = c_ptr + batch * stride_cb

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M×K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # B tile (K×N) – we access B as transposed
        b = tl.load(
            b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,

        acc += tl.dot(a, b).to(tl.float32)

    c = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c, acc.to(tl.float16), mask=store_mask)
