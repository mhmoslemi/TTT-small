    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # LayerNorm input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Contract via batched matmul
    # Permute to B, hidden_dim, N, N

    right_perm = right.permute(0,3,1,2).contiguous()

"""
Triton‐accelerated Triangle Multiplicative Update (TriMul) – outgoing version.

The forward pass follows the reference PyTorch implementation:
1. Layer‑norm the input.
2. Linear projections (left/right) and three sigmoid gates.
3. Apply mask and gates.
4. Core contraction: for each hidden channel `d`,
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  (i.e. a batched matrix‑multiply).
5. Layer‑norm the result, apply the output gate.
6. Final linear projection back to `dim` using a custom Triton kernel.

Only the final dense projection is offloaded to Triton; the rest is done with
high‑level PyTorch ops for clarity and correctness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: dense projection from hidden_dim -> dim
# ----------------------------------------------------------------------
@triton.jit
def to_out_linear_kernel(
    out_ptr,                # *mut float32,  [B, N, N, dim]
    in_ptr,                 # *mut float32,  [B, N, N, hidden_dim]
    weight_ptr,             # *mut float32,  [dim, hidden_dim]

    B, N, H,                # constants
    stride_out_b, stride_out_i, stride_out_j, stride_out_o,
    stride_in_b,  stride_in_i,  stride_in_j,  stride_in_h,
    stride_w_o,   stride_w_h,
    BLOCK_SIZE: tl.constexpr
):
    """
    Compute y = x @ W.T for each (b,i,j) position.
    - `x` has shape (B, N, N, H)
    - `W` has shape (dim, H)
    - `y` has shape (B, N, N, dim)
    Each program instance processes ONE output position (b,i,j) and ONE output
    channel `d` (dim). The reduction over the hidden dimension H is performed
    in blocks of size BLOCK_SIZE using `tl.dot`.
    """
    pid_pos = tl.program_id(0)   # flattened index over B * N * N
    pid_dim = tl.program_id(1)   # output channel index (0 .. dim-1)

    total = B * N * N
    if pid_pos >= total:
        return

    # decode (b, i, j) from flattened index
    b = pid_pos // (N * N)
    rem = pid_pos % (N * N)
    i = rem // N
    j = rem % N

    # Base pointers for the current (b,i,j) vector and the weight row
    in_base   = in_ptr   + b * stride_in_b  + i * stride_in_i  + j * stride_in_j
    weight_row = weight_ptr + pid_dim * stride_w_o  # start of W[pid_dim, :]

    # Destination address
    out_addr = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j + pid_dim * stride_out_o

    acc = tl.float32(0.0)

    # Loop over hidden dimension in tiles of BLOCK_SIZE
    for k in range(tl.cdiv(H, BLOCK_SIZE)):
        offs = k * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < H

        x = tl.load(in_base + offs * stride_in_h,   mask=mask, other=0.0)
        w = tl.load(weight_row + offs * stride_w_h, mask=mask, other=0.0)

        # dot reduces across the block to a single scalar
        acc += tl.dot(x, w)

    tl.store(out_addr, acc)


# ----------------------------------------------------------------------
# Triton‑backed custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Configuration & weight extraction
    # ------------------------------------------------------------------
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projection weights (bias=False)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]

    # Gate weights
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Output norm + projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # shape (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N]; broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: batched matmul across hidden_dim
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B = left.shape[0]
    N = left.shape[1]   # seq_len
    # Rearrange to (B, hidden_dim, N, N) for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Merge batch and hidden_dim dimensions
    left_mat  = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)     # (B*H, N, N)

    # Batched matrix multiplication (right is transposed on the inner dim)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection (hidden_dim -> dim) via Triton
    # ------------------------------------------------------------------
    # Allocate output tensor
    output = torch.empty((B, N, N, dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_out = output.stride()          # (B_stride, N_i_stride, N_j_stride, dim_stride)
    stride_in  = out.stride()             # (B_stride, N_i_stride, N_j_stride, hidden_stride)
    stride_w   = to_out_w.stride()        # (dim_stride, hidden_stride)

    # Grid: one program per (b,i,j) position and one per output channel
    total_positions = B * N * N
    grid = (total_positions, dim)

    # Choose a reasonable block size for the reduction over hidden_dim
    BLOCK_SIZE = 32
    # Launch kernel
    to_out_linear_kernel[grid](
        output,
        out,
        to_out_w,
        B,
        N,
        hidden_dim,
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        stride_in[0],  stride_in[1],  stride_in[2],  stride_in[3],
        stride_w[0],   stride_w[1],
        BLOCK_SIZE=BLOCK_SIZE,
    )

    return output
