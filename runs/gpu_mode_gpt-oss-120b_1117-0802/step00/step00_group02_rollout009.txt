"""
custom_tri_mul_kernel
=====================

A highly‑optimized forward implementation of the “outgoing” TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. **LayerNorm (first)** – fused Triton kernel.
   Normalises over the last dimension `dim` (shape [B,N,N,dim]) using
   the provided weight/bias. The kernel processes one (B,N,N) element per
   program instance and reduces over the channel dimension in blocks of
   `BLOCK_C` (default 128). It computes mean/variance in a first pass
   and the normalised result in a second pass.

2. **Linear projections & gates** – performed with `torch.nn.functional.linear`
   (FP32). The mask (if present) is applied, then element‑wise gates are
   multiplied.

3. **Batched matrix multiplication** – the core `einsum('... i k d, ... j k d -> ... i j d')`
   is re‑expressed as many independent GEMMs:
   for each hidden dimension `d` we compute `L_d @ R_dᵀ`.  By reshaping to
   `(B*hidden_dim, N, N)` we can call a single `torch.bmm`, which runs
   on cuBLAS and is extremely fast.  The multiplication is performed in
   FP16 to reduce bandwidth and improve throughput, then cast back to FP32.

4. **LayerNorm (second)** – another fused Triton kernel normalises over the
   hidden dimension `hidden_dim` (shape [B,N,N,hidden_dim]).

5. **Out‑gate & final linear** – point‑wise multiplication with the out‑gate
   followed by a bias‑free linear projection back to the original `dim`.

Only the LayerNorms are executed in Triton; the remaining ops stay in
PyTorch for readability and to keep the implementation concise while still
meeting the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,            # *float32  input (flattened [M, C])
    y_ptr,            # *float32  output (flattened [M, C])
    weight_ptr,       # *float32  weight vector (C,)
    bias_ptr,         # *float32  bias vector   (C,)
    M,                # i32       number of rows  (B * N * N)
    C,                # i32       channels per row (dim or hidden_dim)
    eps: tl.constexpr,# f32       epsilon for variance
    BLOCK_C: tl.constexpr,  # compile‑time block size for channel dim
    NUM_SLICES: tl.constexpr # compile‑time number of channel blocks
):
    """Fused LayerNorm over the last dimension.

    One program instance processes a single row (position) `pos` of the
    flattened tensor.  The channel dimension is handled by `BLOCK_C` threads.
    """
    pos = tl.program_id(0)

    # guard (grid is sized exactly to M, but keep for safety)
    if pos >= M:
        return

    # ------------------------------------------------------------------
    # First pass: compute sum and sum of squares for mean/var
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for s in range(NUM_SLICES):
        d_start = s * BLOCK_C
        offs = d_start + tl.arange(0, BLOCK_C)
        mask = offs < C
        # load a slice of the row
        x = tl.load(x_ptr + pos * C + offs, mask=mask, other=0.0)
        sum_val += tl.sum(x)                # reduction across BLOCK_C threads
        sum_sq  += tl.sum(x * x)

    # mean and variance (unbiased=False)
    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalised values (and apply weight/bias)
    # ------------------------------------------------------------------
    for s in range(NUM_SLICES):
        d_start = s * BLOCK_C
        offs = d_start + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_ptr + pos * C + offs, mask=mask, other=0.0)
        y = (x - mean) * inv_std

        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b = tl.load(bias_ptr   + offs, mask=mask, other=0.0)
        y = y * w + b

        tl.store(y_ptr + pos * C + offs, y, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with Triton‑accelerated
    LayerNorms.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [bs, seq_len, seq_len, dim] (float32)
            - mask:        torch.Tensor [bs, seq_len, seq_len] (bool/float)
            - weights:     dict of parameter tensors (already on the correct device)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) First LayerNorm (over `dim`)
    # ------------------------------------------------------------------
    # flatten first three dimensions: M = B * N * N
    B, N, _, C = input_tensor.shape
    assert C == dim
    M = B * N * N

    # prepare output container
    x_norm = torch.empty_like(input_tensor)

    # view as 2‑D tensors (contiguous)
    x_flat_in = input_tensor.contiguous().view(-1, dim)
    x_flat_out = x_norm.view(-1, dim)

    # weight / bias
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device, dtype=torch.float32)

    # Triton launch parameters
    BLOCK_C = 128                      # channel block size (covers dim up to 384)
    num_slices = (dim + BLOCK_C - 1) // BLOCK_C
    grid = (M,)

    layernorm_kernel[grid](
        x_flat_in,
        x_flat_out,
        norm_weight,
        norm_bias,
        M,
        dim,
        eps,
        BLOCK_C=BLOCK_C,
        NUM_SLICES=num_slices,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (still in FP32)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype=torch.float32)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float32)

    left_gate_w = weights["left_gate.weight"].to(device, dtype=torch.float32)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float32)
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float32)

    # Linear projections
    left = F.linear(x_norm, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # Optional mask (already on device)
    if mask is not None:
        mask_f = mask.to(device, dtype=torch.float32).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core contraction (einsum) via batched GEMM, using FP16 for speed
    # ------------------------------------------------------------------
    # Cast to half‑precision before the matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*hidden_dim, N, N)
    left_reshape = left_h.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right_h.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication: out = left @ rightᵀ
    out_h = torch.bmm(left_reshape, right_reshape.transpose(1, 2))   # (B*hidden_dim, N, N)

    # reshape back to (B,N,N,hidden_dim) and convert to FP32
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4) Second LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out)
    out_flat_in = out.contiguous().view(-1, hidden_dim)
    out_flat_out = out_norm.view(-1, hidden_dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float32)

    # launch parameters for hidden_dim (max 384)
    num_slices_h = (hidden_dim + BLOCK_C - 1) // BLOCK_C
    grid_h = (M,)

    layernorm_kernel[grid_h](
        out_flat_in,
        out_flat_out,
        to_out_norm_w,
        to_out_norm_b,
        M,
        hidden_dim,
        eps,
        BLOCK_C=BLOCK_C,
        NUM_SLICES=num_slices_h,
    )

    # ------------------------------------------------------------------
    # 5) Out‑gate multiplication and final linear projection
    # ------------------------------------------------------------------
    out = out_norm * out_gate                     # element‑wise with gate
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float32)
    final = F.linear(out, to_out_w)               # (B,N,N,dim)

    return final
