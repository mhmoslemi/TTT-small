    proj_ptr, gate_ptr, mask_ptr,
    out_ptr,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_d,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    B, N, hidden,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    # each thread processes a tile of (i,j,hidden)
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_M + tl.arange(0, BLOCK_M)

def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_d,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    B, N, hidden,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < hidden

    mask_ij = mask_i[:, None] & mask_j[None, :]
    mask_ijd = mask_ij[:, :, None] & mask_d[None, None, :]

    # compute pointer offsets
    proj_off = pid_b * stride_proj_b + offs_i[:, None, None] * stride_proj_i + offs_j[None, :, None] * stride_proj_j + offs_d[None, None, :] * stride_proj_d
    gate_off = pid_b * stride_gate_b + offs_i[:, None, None] * stride_gate_i + offs_j[None, :, None] * stride_gate_j + offs_d[None, None, :] * stride_gate_d
    mask_off = pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j

    proj_val = tl.load(proj_ptr + proj_off, mask=mask_ijd, other=0.0)
    gate_val = tl.load(gate_ptr + gate_off, mask=mask_ijd, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask_ij, other=0.0)

    # broadcast mask_val to hidden dim
    mask_val = mask_val[:, :, None]

    out_val = proj_val * gate_val * mask_val

    # write back in-place
    tl.store(out_ptr + proj_off, out_val, mask=mask_ijd)

Algorithm
---------
1. Layer‑norm on the [B, N, N, C] input.
2. Linear projections → left/right (shape [B,N,N,H]) and three gating tensors.
3. Fuse mask, left_gate/right_gate with a Triton kernel (in‑place on left/right).
4. Convert left/right to bfloat16 and perform a batched matrix multiplication
   for each hidden slice:
        out_{b,i,j,:} = Σ_k left_{b,i,k,:} * right_{b,j,k,:}
   implemented via torch.bmm on a reshaped view (very fast cuBLAS).
5. Layer‑norm over the hidden dimension, multiply by out_gate.
6. Final linear projection back to the original feature dimension.

At least one Triton kernel (step 3) is used to demonstrate custom fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    proj_ptr,      # left or right tensor (B,N,N,H)
    gate_ptr,      # corresponding gate tensor (B,N,N,H)
    mask_ptr,      # mask tensor (B,N,N,1)
    out_ptr,       # output tensor (same shape as proj, written in‑place)
    # strides for proj
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    # strides for gate
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # strides for mask (no hidden dim)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for out (identical to proj, kept for API symmetry)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,                     # size parameters
    BLOCK_I: tl.constexpr,       # tile size for the first sequence dimension
    BLOCK_J: tl.constexpr,       # tile size for the second sequence dimension
    BLOCK_H: tl.constexpr,       # tile size for the hidden dimension
):
    """In‑place: proj = proj * gate * mask (broadcast mask over hidden dim)."""
    pid_b = tl.program_id(0)          # batch axis
    pid_i = tl.program_id(1)          # i‑axis
    pid_j = tl.program_id(2)          # j‑axis

    # Offsets for each tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H
    mask_ij = mask_i[:, None] & mask_j[None, :]
    mask_ijh = mask_ij[:, :, None] & mask_h[None, None, :]

    # Compute byte offsets (pointer arithmetic)
    proj_off = (
        pid_b * stride_proj_b
        + offs_i[:, None, None] * stride_proj_i
        + offs_j[None, :, None] * stride_proj_j
        + offs_h[None, None, :] * stride_proj_h
    )
    gate_off = (
        pid_b * stride_gate_b
        + offs_i[:, None, None] * stride_gate_i
        + offs_j[None, :, None] * stride_gate_j
        + offs_h[None, None, :] * stride_gate_h
    )
    mask_off = (
        pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    out_off = (
        pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )

    # Load tensors
    proj = tl.load(proj_ptr + proj_off, mask=mask_ijh, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask_ijh, other=0.0)
    mask = tl.load(mask_ptr + mask_off, mask=mask_ij, other=0.0)
    mask = mask[:, :, None]               # broadcast over hidden dim

    # Fuse operations
    out = proj * gate * mask

    # Store result (in‑place if out_ptr == proj_ptr)
    tl.store(out_ptr + out_off, out, mask=mask_ijh)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input, mask, weights, config)
            input   – Tensor [B, N, N, C] (float32)
            mask    – Tensor [B, N, N]   (bool / int / float)
            weights – Dict of model weights with keys:
                      "norm.weight", "norm.bias",
                      "left_proj.weight", "right_proj.weight",
                      "left_gate.weight", "right_gate.weight",
                      "out_gate.weight",
                      "to_out_norm.weight", "to_out_norm.bias",
                      "to_out.weight"
            config  – Dict containing at least "dim" and "hidden_dim".

    Returns:
        Tensor [B, N, N, C] (float32) – processed output.
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # Load weights (all assumed float32)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------------------------------------------------------------------
    # Step 1 – Input LayerNorm
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ----------------------------------------------------------------------
    # Step 2 – Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right = F.linear(x, right_proj_weight)       # [B,N,N,H]

    # ----------------------------------------------------------------------
    # Step 3 – Gates (sigmoid of linear projections)
    # ----------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ----------------------------------------------------------------------
    # Step 4 – Apply mask and gates using Triton (in‑place on left/right)
    # ----------------------------------------------------------------------
    # Prepare mask broadcasted to hidden dimension
    mask = mask_tensor.to(dtype).unsqueeze(-1)    # [B,N,N,1]
    # Ensure tensors are contiguous for stride calculations
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    # Triton launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32  # hidden dim tile

    grid = (B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    # Fuse left side
    mask_gate_kernel[grid](
        left, left_gate, mask, left,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        B, N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H
    )
    # Fuse right side
    mask_gate_kernel[grid](
        right, right_gate, mask, right,
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        B, N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H
    )
    # Gates are no longer needed → free memory early
    del left_gate, right_gate, mask

    torch.cuda.synchronize()

    # ----------------------------------------------------------------------
    # Step 5 – Batched matrix multiplication (B,N,N,H) × (B,N,N,H)^T over k
    # ----------------------------------------------------------------------
    # Cast to bfloat16 for speed
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # Reshape to (B*H, N, N) to use torch.bmm
    left_flat = left_bf16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_bf16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Matrix multiplication over the k‑dimension (N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]

    # Restore original layout [B, N, N, H] and convert back to float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ----------------------------------------------------------------------
    # Step 6 – Output LayerNorm and final gating
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate   # element‑wise scaling by the output gate

    # ----------------------------------------------------------------------
    # Step 7 – Final linear projection back to original dimensionality
    # ----------------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # shape [B, N, N, C]

    return output
