def layernorm_kernel(x_ptr, y_ptr, gamma_ptr, beta_ptr, M, C, eps,
                     BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid*BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, BLOCK_C)
    m_mask = offs_m < M
    c_mask = offs_c < C

    # sum over C

    ss = tl.sum(x*x, axis=1)

    tl.store(y_ptr + offs_m[:,None]*C + offs_c[None,:], y.to(tl.bfloat16), mask=mask)

"""
Fused Triton TriMul (outgoing) forward.

Pipeline:
1) Triton LayerNorm -> bf16 normalized input.
2) PyTorch bf16 gate GEMMs + sigmoid (left/right/out gates).
3) Triton fused projection + gate/mask multiply, written in transposed layout:
   left/right  : [B, hidden_dim, seq, seq]
4) Triton batched GEMM over hidden_dim:
   out[h, i, j] = sum_k left[h, i, k] * right[h, j, k]
   stored as [B, hidden_dim, seq, seq]
5) Triton fused final LayerNorm (over hidden_dim) + out_gate + output projection,
   returned as float32 [B, seq, seq, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _cdiv(a: int, b: int) -> int:
    return (a + b - 1) // b


def _as_dtype(t: torch.Tensor, dtype: torch.dtype) -> torch.Tensor:
    if t.requires_grad:
        t = t.detach()
    if t.dtype != dtype:
        t = t.to(dtype)
    if not t.is_contiguous():
        t = t.contiguous()
    return t


@triton.jit
def _ln_kernel(
    x_ptr,
    y_ptr,
    gamma_ptr,
    beta_ptr,
    M,
    C,
    inv_C,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, BLOCK_C)

    m_mask = offs_m < M
    c_mask = offs_c < C
    load_mask = m_mask[:, None] & c_mask[None, :]

    x = tl.load(
        x_ptr + offs_m[:, None] * C + offs_c[None, :],
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(x, axis=1) * inv_C
    xc = x - mean[:, None]
    xc = tl.where(c_mask[None, :], xc, 0.0)

    var = tl.sum(xc * xc, axis=1) * inv_C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    gamma = tl.load(gamma_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)

    y = xc * rstd[:, None] * gamma[None, :] + beta[None, :]
    tl.store(
        y_ptr + offs_m[:, None] * C + offs_c[None, :],
        y.to(tl.bfloat16),
        mask=load_mask,
    )


@triton.jit
def _proj_kernel(
    x_ptr,
    w_ptr,
    gate_ptr,
    mask_ptr,
    out_ptr,
    SEQ,
    C,
    H,
    HAS_MASK: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    # grid: (H_blocks, K_blocks, B * SEQ)
    pid_h = tl.program_id(0)
    pid_k = tl.program_id(1)
    pid_row = tl.program_id(2)

    b = pid_row // SEQ
    i = pid_row % SEQ

    k0 = pid_k * BLOCK_K
    h0 = pid_h * BLOCK_H

    offs_k = k0 + tl.arange(0, BLOCK_K)
    offs_h = h0 + tl.arange(0, BLOCK_H)
    k_mask = offs_k < SEQ
    h_mask = offs_h < H

    base = (b * SEQ + i) * SEQ * C

    acc = tl.zeros((BLOCK_K, BLOCK_H), dtype=tl.float32)

    for c0 in range(0, C, BLOCK_C):
        c_offs = c0 + tl.arange(0, BLOCK_C)
        c_mask = c_offs < C

        a_ptrs = x_ptr + base + offs_k[:, None] * C + c_offs[None, :]
        a = tl.load(
            a_ptrs,
            mask=k_mask[:, None] & c_mask[None, :],
            other=0.0,
        )

        w_ptrs = w_ptr + offs_h[None, :] * C + c_offs[:, None]
        w = tl.load(
            w_ptrs,
            mask=h_mask[None, :] & c_mask[:, None],
            other=0.0,
        )

        acc = tl.dot(a, w, acc)

    gate_base = (b * SEQ + i) * SEQ * H
    gate_ptrs = gate_base + offs_k[:, None] * H + offs_h[None, :]
    gate = tl.load(
        gate_ptrs,
        mask=k_mask[:, None] & h_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    val = acc * gate

    if HAS_MASK:
        m = tl.load(
            mask_ptr + (b * SEQ + i) * SEQ + offs_k,
            mask=k_mask,
            other=0.0,
        ).to(tl.float32)
        val = val * m[:, None]

    # out layout: [B, H, SEQ, SEQ], store [H_block, K_block]
    out_ptrs = (
        out_ptr
        + (b * H + offs_h)[:, None] * (SEQ * SEQ)
        + i * SEQ
        + offs_k[None, :]
    )
    tl.store(
        out_ptrs,
        tl.trans(val).to(tl.bfloat16),
        mask=h_mask[:, None] & k_mask[None, :],
    )


@triton.jit
def _core_kernel(
    l_ptr,
    r_ptr,
    out_ptr,
    SEQ,
    H,
    NUM_I: tl.constexpr,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # grid: (I_tiles * J_tiles, B * H)
    pid_tile = tl.program_id(0)
    bh = tl.program_id(1)

    b = bh // H
    h = bh % H

    ib = pid_tile % NUM_I
    jb = pid_tile // NUM_I

    offs_i = ib * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = jb * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    i_mask = offs_i < SEQ
    j_mask = offs_j < SEQ

    base = (b * H + h) * SEQ * SEQ

    a_ptrs = l_ptr + base + offs_i[:, None] * SEQ + offs_k[None, :]
    b_ptrs = r_ptr + base + offs_j[:, None] * SEQ + offs_k[None, :]

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k0 in range(0, SEQ, BLOCK_K):
        k_mask = (k0 + offs_k) < SEQ

        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        bmat = tl.load(b_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        acc = tl.dot(a, tl.trans(bmat), acc)

        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K

    # out layout: [B, H, SEQ, SEQ]
    out_ptrs = out_ptr + base + offs_i[:, None] * SEQ + offs_j[None, :]
    tl.store(
        out_ptrs,
        acc.to(tl.bfloat16),
        mask=i_mask[:, None] & j_mask[None, :],
    )


@triton.jit
def _final_kernel(
    out_t_ptr,
    gate_ptr,
    ln_w_ptr,
    ln_b_ptr,
    w_out_ptr,
    y_ptr,
    SEQ,
    H,
    C,
    inv_H,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    # grid: (C_blocks, M_blocks, B)
    pid_c = tl.program_id(0)
    pid_m = tl.program_id(1)
    b = tl.program_id(2)

    M = SEQ * SEQ
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < M

    base_out = b * H * M

    s1 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    s2 = tl.zeros((BLOCK_M,), dtype=tl.float32)

    # First pass: LayerNorm statistics over hidden_dim.
    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        out_ptrs = base_out + offs_h[:, None] * M + offs_m[None, :]
        out_f = tl.load(
            out_t_ptr + out_ptrs,
            mask=h_mask[:, None] & m_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        s1 += tl.sum(out_f, axis=0)
        s2 += tl.sum(out_f * out_f, axis=0)

    mean = s1 * inv_H
    var = s2 * inv_H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    c0 = pid_c * BLOCK_C
    offs_c = c0 + tl.arange(0, BLOCK_C)
    c_mask = offs_c < C

    acc = tl.zeros((BLOCK_M, BLOCK_C), dtype=tl.float32)

    # Second pass: normalize, gate, and project back to dim.
    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        out_ptrs = base_out + offs_h[:, None] * M + offs_m[None, :]
        out_b = tl.load(
            out_t_ptr + out_ptrs,
            mask=h_mask[:, None] & m_mask[None, :],
            other=0.0,
        )
        out_mh_f = tl.trans(out_b).to(tl.float32)

        gate_ptrs = b * M * H + offs_m[:, None] * H + offs_h[None, :]
        gate_b = tl.load(
            gate_ptr + gate_ptrs,
            mask=m_mask[:, None] & h_mask[None, :],
            other=0.0,
        )

        ln_w = tl.load(ln_w_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)
        ln_b = tl.load(ln_b_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)

        val = (out_mh_f - mean[:, None]) * rstd[:, None] * ln_w[None, :] + ln_b[None, :]
        val = val * gate_b.to(tl.float32)

        # to_out.weight is [C, H]; load as [H, C]
        w_ptrs = offs_c[None, :] * H + offs_h[:, None]
        w_b = tl.load(
            w_out_ptr + w_ptrs,
            mask=c_mask[None, :] & h_mask[:, None],
            other=0.0,
        )

        acc = tl.dot(val.to(tl.bfloat16), w_b, acc)

    # y layout: [B, SEQ, SEQ, C]
    y_ptrs = (b * M + offs_m[:, None]) * C + offs_c[None, :]
    tl.store(
        y_ptr + y_ptrs,
        acc.to(tl.float32),
        mask=m_mask[:, None] & c_mask[None, :],
    )


def custom_kernel(data):
    """
    data:
        (input, mask, weights, config) or (input, weights, config)
    input:
        [B, N, N, C]
    mask:
        [B, N, N] or None
    Returns:
        [B, N, N, C] float32
    """
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    B = int(x.shape[0])
    N = int(x.shape[1])
    C = int(x.shape[-1])
    if config is not None and config.get("hidden_dim", None) is not None:
        H = int(config["hidden_dim"])
    else:
        H = int(weights["left_proj.weight"].shape[0])

    device = x.device

    if x.dtype not in (torch.float32, torch.float16, torch.bfloat16):
        x = x.to(torch.float32)
    x = x.contiguous()

    use_mask = False
    mask_f = None
    if mask is not None and not bool(config.get("nomask", False)):
        mask = mask
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1)
        if mask.dim() == 2:
            if B == 1:
                mask = mask.unsqueeze(0)
            else:
                mask = mask.unsqueeze(0).expand(B, N, N)
        if mask.dim() != 3 or mask.shape[0] != B or mask.shape[1:] != (N, N):
            # If shape is unexpected, ignore mask rather than crash silently
            # in the very unlikely case.
            use_mask = False
        else:
            mask_f = _as_dtype(mask, torch.float32)
            use_mask = True

    # 1) LayerNorm to bf16
    norm_w = _as_dtype(weights["norm.weight"], torch.float32)
    norm_b = _as_dtype(weights["norm.bias"], torch.float32)

    M_rows = B * N * N
    x_norm = torch.empty((B, N, N, C), dtype=torch.bfloat16, device=device)

    block_c_ln = _next_pow2(C)
    block_m_ln = 16 if block_c_ln > 256 else 32
    _ln_kernel[(_cdiv(M_rows, block_m_ln),)](
        x,
        x_norm,
        norm_w,
        norm_b,
        M_rows,
        C,
        1.0 / float(C),
        1e-5,
        BLOCK_M=block_m_ln,
        BLOCK_C=block_c_ln,
        num_warps=4,
        num_stages=2,
    )
    del x

    x_norm_2d = x_norm.view(M_rows, C)

    def compute_gate(weight_name: str) -> torch.Tensor:
        w = _as_dtype(weights[weight_name], torch.bfloat16)
        g = F.linear(x_norm_2d, w)
        g = torch.sigmoid(g).to(torch.bfloat16)
        return g

    # 2) Left projection + left_gate + mask, stored transposed
    block_h_proj = 128
    block_k_proj = 128
    block_c_proj = 128
    num_h_blocks = _cdiv(H, block_h_proj)
    num_k_blocks = _cdiv(N, block_k_proj)
    proj_grid = (num_h_blocks, num_k_blocks, B * N)

    gate_l = compute_gate("left_gate.weight")
    L = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    w_l = _as_dtype(weights["left_proj.weight"], torch.bfloat16)
    _proj_kernel[proj_grid](
        x_norm,
        w_l,
        gate_l,
        mask_f if use_mask else x_norm,
        L,
        N,
        C,
        H,
        HAS_MASK=use_mask,
        BLOCK_K=block_k_proj,
        BLOCK_H=block_h_proj,
        BLOCK_C=block_c_proj,
        num_warps=8,
        num_stages=3,
    )
    del gate_l

    # 3) Right projection + right_gate + mask, stored transposed
    gate_r = compute_gate("right_gate.weight")
    R = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    w_r = _as_dtype(weights["right_proj.weight"], torch.bfloat16)
    _proj_kernel[proj_grid](
        x_norm,
        w_r,
        gate_r,
        mask_f if use_mask else x_norm,
        R,
        N,
        C,
        H,
        HAS_MASK=use_mask,
        BLOCK_K=block_k_proj,
        BLOCK_H=block_h_proj,
        BLOCK_C=block_c_proj,
        num_warps=8,
        num_stages=3,
    )
    del gate_r

    # 4) Core: for each h, out[h] = L[h] @ R[h].T
    out_t = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    block_i = 128
    block_j = 128
    block_k_core = 128
    num_i = _cdiv(N, block_i)
    num_j = _cdiv(N, block_j)
    core_grid = (num_i * num_j, B * H)

    _core_kernel[core_grid](
        L,
        R,
        out_t,
        N,
        H,
        NUM_I=num_i,
        BLOCK_I=block_i,
        BLOCK_J=block_j,
        BLOCK_K=block_k_core,
        num_warps=8,
        num_stages=4,
    )
    del L, R

    # 5) Final LayerNorm + out_gate + to_out
    gate_o = compute_gate("out_gate.weight")
    del x_norm_2d
    del x_norm

    ln_w = _as_dtype(weights["to_out_norm.weight"], torch.float32)
    ln_b = _as_dtype(weights["to_out_norm.bias"], torch.float32)
    w_out = _as_dtype(weights["to_out.weight"], torch.bfloat16)

    y = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    block_m_final = 64
    block_h_final = 128
    block_c_final = 128
    num_m_final = _cdiv(N * N, block_m_final)
    num_c_final = _cdiv(C, block_c_final)
    final_grid = (num_c_final, num_m_final, B)

    _final_kernel[final_grid](
        out_t,
        gate_o,
        ln_w,
        ln_b,
        w_out,
        y,
        N,
        H,
        C,
        1.0 / float(H),
        1e-5,
        BLOCK_M=block_m_final,
        BLOCK_H=block_h_final,
        BLOCK_C=block_c_final,
        num_warps=8,
        num_stages=2,
    )

    return y
