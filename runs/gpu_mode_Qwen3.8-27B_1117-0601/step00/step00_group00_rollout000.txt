@triton.jit
def _ln_kernel(
    x_ptr, out_ptr, w_ptr, b_ptr,
    M, C,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr, eps: tl.constexpr
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(M, BLOCK_M)
    b = pid // num_m
    pid_m = pid % num_m
    m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = m < M
    c = tl.arange(0, BLOCK_C)
    c_mask = c < C
    mask2d = m_mask[:, None] & c_mask[None, :]
    ptrs = x_ptr + (b * M + m)[:, None] * C + c[None, :]
    x = tl.load(ptrs, mask=mask2d, other=0.0).to(tl.float32)
    inv_c = 1.0 / C
    mean = tl.sum(x, axis=1) * inv_c
    diff = x - mean[:, None]
    diff = tl.where(c_mask[None, :], diff, 0.0)
    var = tl.sum(diff * diff, axis=1) * inv_c
    rstd = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    beta = tl.load(b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    y = diff * rstd[:, None] * w[None, :] + beta[None, :]
    y = tl.where(c_mask[None, :], y, 0.0)
    out_ptrs = out_ptr + (b * M + m)[:, None] * C + c[None, :]
    tl.store(out_ptrs, y.to(tl.bfloat16), mask=mask2d)

```python
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


def _next_pow2(x: int) -> int:
    x = max(1, int(x))
    return 1 << (x - 1).bit_length()


def _pow2_floor(x: int) -> int:
    x = max(1, int(x))
    return 1 << (x - 1).bit_length()


@triton.jit
def _ln_kernel(
    x_ptr,
    out_ptr,
    w_ptr,
    b_ptr,
    M,
    C,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    eps: tl.constexpr,
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(M, BLOCK_M)
    b = pid // num_m
    pid_m = pid % num_m

    m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    c = tl.arange(0, BLOCK_C)

    m_mask = m < M
    c_mask = c < C
    load_mask = m_mask[:, None] & c_mask[None, :]

    x_off = (b * M + m)[:, None] * C + c[None, :]
    x = tl.load(x_ptr + x_off, mask=load_mask, other=0.0).to(tl.float32)

    inv_c = 1.0 / C
    mean = tl.sum(x, axis=1) * inv_c
    diff = x - mean[:, None]
    diff = tl.where(c_mask[None, :], diff, 0.0)

    var = tl.sum(diff * diff, axis=1) * inv_c
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + c, mask=c_mask, other=0.0).to(tl.float32)
    beta = tl.load(b_ptr + c, mask=c_mask, other=0.0).to(tl.float32)

    y = diff * rstd[:, None] * w[None, :] + beta[None, :]
    y = tl.where(c_mask[None, :], y, 0.0)

    out_off = (b * M + m)[:, None] * C + c[None, :]
    tl.store(out_ptr + out_off, y.to(tl.bfloat16), mask=load_mask)


@triton.jit
def _proj_lr_kernel(
    x_ptr,
    mask_ptr,
    wl_ptr,
    wgl_ptr,
    wr_ptr,
    wgr_ptr,
    L_ptr,
    R_ptr,
    M,
    H,
    C,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_h = tl.program_id(2)

    m = pid_m * BM + tl.arange(0, BM)
    h = pid_h * BH + tl.arange(0, BH)

    m_mask = m < M
    h_mask = h < H

    x_base = x_ptr + (pid_b * M) * C
    L_base = L_ptr + (pid_b * H + pid_h * BH) * M
    R_base = R_ptr + (pid_b * H + pid_h * BH) * M

    acc_pl = tl.zeros((BM, BH), dtype=tl.float32)
    acc_gl = tl.zeros((BM, BH), dtype=tl.float32)
    acc_pr = tl.zeros((BM, BH), dtype=tl.float32)
    acc_gr = tl.zeros((BM, BH), dtype=tl.float32)

    for k0 in range(0, C, BK):
        k = k0 + tl.arange(0, BK)
        k_mask = k < C

        x_tile_mask = m_mask[:, None] & k_mask[None, :]
        w_tile_mask = k_mask[:, None] & h_mask[None, :]

        x_tile = tl.load(
            x_base + m[:, None] * C + k[None, :],
            mask=x_tile_mask,
            other=0.0,
        )
        wl_tile = tl.load(
            wl_ptr + k[:, None] + h[None, :] * C,
            mask=w_tile_mask,
            other=0.0,
        )
        wgl_tile = tl.load(
            wgl_ptr + k[:, None] + h[None, :] * C,
            mask=w_tile_mask,
            other=0.0,
        )
        wr_tile = tl.load(
            wr_ptr + k[:, None] + h[None, :] * C,
            mask=w_tile_mask,
            other=0.0,
        )
        wgr_tile = tl.load(
            wgr_ptr + k[:, None] + h[None, :] * C,
            mask=w_tile_mask,
            other=0.0,
        )

        acc_pl += tl.dot(x_tile, wl_tile)
        acc_gl += tl.dot(x_tile, wgl_tile)
        acc_pr += tl.dot(x_tile, wr_tile)
        acc_gr += tl.dot(x_tile, wgr_tile)

    if HAS_MASK:
        mv = tl.load(mask_ptr + pid_b * M + m, mask=m_mask, other=0.0).to(tl.float32)
    else:
        mv = m_mask.to(tl.float32)

    gl = tl.minimum(tl.maximum(acc_gl, -20.0), 20.0)
    gl = 1.0 / (1.0 + tl.exp(-gl))

    gr = tl.minimum(tl.maximum(acc_gr, -20.0), 20.0)
    gr = 1.0 / (1.0 + tl.exp(-gr))

    lv = acc_pl * mv[:, None] * gl
    rv = acc_pr * mv[:, None] * gr

    store_mask = h_mask[:, None] & m_mask[None, :]
    l_ptrs = L_base + h[:, None] * M + m[None, :]
    r_ptrs = R_base + h[:, None] * M + m[None, :]

    tl.store(l_ptrs, tl.trans(lv.to(tl.bfloat16)), mask=store_mask)
    tl.store(r_ptrs, tl.trans(rv.to(tl.bfloat16)), mask=store_mask)


@triton.jit
def _core_kernel(
    L_ptr,
    R_ptr,
    out_ptr,
    N,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BK: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    base = pid_bh * (N * N)

    i = pid_i * BM + tl.arange(0, BM)
    j = pid_j * BN + tl.arange(0, BN)

    i_mask = i < N
    j_mask = j < N

    acc = tl.zeros((BM, BN), dtype=tl.float32)

    for k0 in range(0, N, BK):
        k = k0 + tl.arange(0, BK)
        k_mask = k < N

        a_ptrs = L_ptr + base + i[:, None] * N + k[None, :]
        b_ptrs = R_ptr + base + j[:, None] * N + k[None, :]

        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        acc += tl.dot(a, tl.trans(b))

    out_ptrs = out_ptr + base + i[:, None] * N + j[None, :]
    tl.store(out_ptrs, acc.to(tl.bfloat16), mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _final_kernel(
    raw_ptr,
    x_ptr,
    wog_ptr,
    wout_ptr,
    tnw_ptr,
    tnb_ptr,
    out_ptr,
    M,
    C,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BK: tl.constexpr,
    BC: tl.constexpr,
    eps: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)

    m = pid_m * BM + tl.arange(0, BM)
    h = tl.arange(0, BH)

    m_mask = m < M

    raw_base = raw_ptr + pid_b * (BH * M)
    raw_ptrs = raw_base + h[:, None] * M + m[None, :]
    raw_mask = tl.broadcast_to(m_mask[None, :], (BH, BM))
    raw = tl.load(raw_ptrs, mask=raw_mask, other=0.0).to(tl.float32)

    inv_h = 1.0 / BH
    mean = tl.sum(raw, axis=0) * inv_h
    diff = raw - mean[None, :]
    var = tl.sum(diff * diff, axis=0) * inv_h
    rstd = 1.0 / tl.sqrt(var + eps)

    tnw = tl.load(tnw_ptr + h).to(tl.float32)
    tnb = tl.load(tnb_ptr + h).to(tl.float32)
    norm = diff * rstd[None, :] * tnw[:, None] + tnb[:, None]

    gate_acc = tl.zeros((BM, BH), dtype=tl.float32)
    x_base = x_ptr + pid_b * M * C

    for k0 in range(0, C, BK):
        k = k0 + tl.arange(0, BK)
        k_mask = k < C

        x_tile = tl.load(
            x_base + m[:, None] * C + k[None, :],
            mask=m_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        wog_tile = tl.load(
            wog_ptr + k[:, None] + h[None, :] * C,
            mask=k_mask[:, None],
            other=0.0,
        )
        gate_acc += tl.dot(x_tile, wog_tile)

    gate = tl.minimum(tl.maximum(gate_acc, -20.0), 20.0)
    gate = 1.0 / (1.0 + tl.exp(-gate))

    xhat = (tl.trans(norm) * gate).to(tl.bfloat16)

    out_base = out_ptr + pid_b * M * C
    for c0 in range(0, C, BC):
        c = c0 + tl.arange(0, BC)
        c_mask = c < C

        wout_tile = tl.load(
            wout_ptr + c[None, :] * BH + h[:, None],
            mask=tl.broadcast_to(c_mask[None, :], (BH, BC)),
            other=0.0,
        )
        y = tl.dot(xhat, wout_tile)

        out_ptrs = out_base + m[:, None] * C + c[None, :]
        tl.store(
            out_ptrs,
            y.to(tl.float32),
            mask=m_mask[:, None] & c_mask[None, :],
        )


def _bf16(t: torch.Tensor) -> torch.Tensor:
    return t.to(torch.bfloat16, copy=False)


def _unpack_data(data):
    if len(data) == 4:
        x, mask, weights, config = data
    elif len(data) == 3:
        a, b, c = data
        if isinstance(a, (tuple, list)) and len(a) >= 2:
            x, mask = a
            weights, config = b, c
        else:
            x, weights, config = a, b, c
            mask = None
            if isinstance(config, dict) and "mask" in config:
                mask = config["mask"]
    elif len(data) == 2:
        a, b = data
        if isinstance(a, (tuple, list)) and len(a) >= 2:
            x, mask = a
            weights = b
            config = weights.get("config", {}) if isinstance(weights, dict) else {}
        else:
            raise ValueError("Unsupported input data format.")
    else:
        raise ValueError("Unsupported input data format.")

    return x, mask, weights, config


def custom_kernel(data):
    """
    Fused Triton implementation of AlphaFold3-style outgoing TriMul.

    Pipeline:
      1) LayerNorm(input) -> bf16.
      2) One Triton GEMM epilogue computes and writes:
           L = mask * left_proj(x_norm) * sigmoid(left_gate(x_norm))
           R = mask * right_proj(x_norm) * sigmoid(right_gate(x_norm))
         in head-major layout [B, hidden_dim, N*N].
      3) Triton batched matmul:
           raw[b, h, i, j] = sum_k L[b, h, i, k] * R[b, h, j, k].
      4) Fused final kernel:
           - recomputes out_gate = sigmoid(x_norm @ W_out_gate^T)
           - LayerNorm(raw) over hidden_dim
           - multiply by out_gate
           - project to final dim with Triton dot.

    Mixed precision: activations/projections are bf16 with fp32 accumulation.
    The final output is cast to fp32.
    """
    x, mask, weights, config = _unpack_data(data)

    B, N, _, C = x.shape
    M = N * N
    H = int(config.get("hidden_dim", weights["to_out_norm.weight"].shape[0]))
    device = x.device

    if not (
        x.stride() == (N * N * C, N * C, C, 1)
    ):
        x = x.contiguous()

    if mask is not None:
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1)
        mask = mask.reshape(B, M).contiguous()
        has_mask = True
        mask_ptr = mask
    else:
        has_mask = False
        mask_ptr = x

    norm_weight = weights["norm.weight"].float()
    norm_bias = weights["norm.bias"].float()

    left_proj_w = _bf16(weights["left_proj.weight"])
    left_gate_w = _bf16(weights["left_gate.weight"])
    right_proj_w = _bf16(weights["right_proj.weight"])
    right_gate_w = _bf16(weights["right_gate.weight"])
    out_gate_w = _bf16(weights["out_gate.weight"])
    to_out_w = _bf16(weights["to_out.weight"])

    tn_weight = weights["to_out_norm.weight"].float()
    tn_bias = weights["to_out_norm.bias"].float()

    # 1) LayerNorm -> bf16, flattened as [B, N*N, C].
    x_norm = torch.empty((B, M, C), dtype=torch.bfloat16, device=device)
    LN_BLOCK_C = _next_pow2(C)
    LN_BLOCK_M = max(1, _pow2_floor(8192 // LN_BLOCK_C))

    _ln_kernel[(B * triton.cdiv(M, LN_BLOCK_M),)](
        x,
        x_norm,
        norm_weight,
        norm_bias,
        M,
        C,
        BLOCK_M=LN_BLOCK_M,
        BLOCK_C=LN_BLOCK_C,
        eps=1e-5,
        num_warps=4,
        num_stages=2,
    )

    # 2) Fused left/right projection + gates + mask, head-major output.
    L = torch.empty((B, H, M), dtype=torch.bfloat16, device=device)
    R = torch.empty((B, H, M), dtype=torch.bfloat16, device=device)

    PROJ_BM = 64
    PROJ_BH = 128
    if C == 128:
        PROJ_BK = 128
        PROJ_STAGES = 1
    else:
        PROJ_BK = 64
        PROJ_STAGES = 2

    _proj_lr_kernel[
        (B, triton.cdiv(M, PROJ_BM), triton.cdiv(H, PROJ_BH))
    ](
        x_norm,
        mask_ptr,
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        L,
        R,
        M,
        H,
        C,
        BM=PROJ_BM,
        BH=PROJ_BH,
        BK=PROJ_BK,
        HAS_MASK=has_mask,
        num_warps=8,
        num_stages=PROJ_STAGES,
    )

    # 3) Core matrix multiplication: L_h @ R_h^T for every (b, h).
    raw_out = torch.empty((B, H, M), dtype=torch.bfloat16, device=device)
    CORE_BM = 128
    CORE_BN = 128
    CORE_BK = 128

    _core_kernel[
        (B * H, triton.cdiv(N, CORE_BM), triton.cdiv(N, CORE_BN))
    ](
        L,
        R,
        raw_out,
        N,
        BM=CORE_BM,
        BN=CORE_BN,
        BK=CORE_BK,
        num_warps=8,
        num_stages=3,
    )

    # 4) Final fused: LayerNorm(raw) * out_gate -> to_out.
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    if H == 128:
        FINAL_BM = 64
        FINAL_BH = 128
        FINAL_BC = 128
        if C == 128:
            FINAL_BK = 128
            FINAL_STAGES = 1
        else:
            FINAL_BK = 64
            FINAL_STAGES = 3

        _final_kernel[
            (B, triton.cdiv(M, FINAL_BM))
        ](
            raw_out,
            x_norm,
            out_gate_w,
            to_out_w,
            tn_weight,
            tn_bias,
            out,
            M,
            C,
            BM=FINAL_BM,
            BH=FINAL_BH,
            BK=FINAL_BK,
            BC=FINAL_BC,
            eps=1e-5,
            num_warps=8,
            num_stages=FINAL_STAGES,
        )
    else:
        # Correct path for non-128 hidden dimensions. The provided test matrix
        # uses hidden_dim=128, for which the fused kernel above is used.
        x_flat = x_norm.reshape(B * M, C)
        out_gate = (x_flat @ out_gate_w.t()).float().sigmoid()

        raw_flat = raw_out.permute(0, 2, 1).reshape(B * M, H)
        normed = F.layer_norm(
            raw_flat.float(),
            (H,),
            weight=tn_weight,
            bias=tn_bias,
            eps=1e-5,
        )
        normed = (normed * out_gate).to(torch.bfloat16)

        out_flat = (normed @ to_out_w.t()).float()
        out = out_flat.view(B, N, N, C)

    return out
```
