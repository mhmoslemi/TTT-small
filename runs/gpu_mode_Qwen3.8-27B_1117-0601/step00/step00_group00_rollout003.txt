```
def _ln_kernel(x_ptr, w_ptr, b_ptr, y_ptr, R, C, eps, BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    offs_r = pid*BLOCK_R + tl.arange(0, BLOCK_R)
    offs_c = tl.arange(0, BLOCK_C)
    r_mask = offs_r < R
    c_mask = offs_c < C
    x = tl.load(x_ptr + offs_r[:,None]*C + offs_c[None,:], mask=r_mask[:,None]&c_mask[None,:], other=0.).to(tl.float32)

    s = tl.sum(x_m, axis=1)

    d = tl.where(c_mask[None,:], x - mean[:,None], 0.0)
```

```
def _gemm_kernel(l_ptr, r_ptr, o_ptr, N, N2,
                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                 EVEN: tl.constexpr, GROUP_M: tl.constexpr):

    pid = tl.program_id(1)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    group_size_m = min(num_pid_m - first_m, GROUP_M)
    pid_m = first_m + (pid % group_size_m)
    pid_n = (pid % (GROUP_M * num_pid_n)) // group_size_m
    offs_m = pid_m*BLOCK_M + tl.arange(0,BLOCK_M)
    offs_n = pid_n*BLOCK_N + tl.arange(0,BLOCK_N)
    offs_k = tl.arange(0,BLOCK_K)

    a_ptrs = l_ptr + base + offs_m[:,None]*N + offs_k[None,:]
    b_ptrs = r_ptr + base + offs_k[:,None] + offs_n[None,:]*N
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


"""
TriMul forward (outgoing):
- Fused LayerNorm + projections/gates/mask for the left/right triangle tensors.
- Batched GEMM computes out[b,i,j,h] = sum_k left[b,h,i,k] * right[b,h,j,k].
- Final fused LayerNorm + out-gate + output projection.
All large intermediates are kept in Triton-friendly layouts:
  L/R/out_gate/out: [B, H, N, N], final output: [B, N, N, C].
"""


@triton.jit
def _sigmoid(x):
    x = tl.minimum(tl.maximum(x, -88.0), 88.0)
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def _ln_kernel(
    x_ptr, gw_ptr, gb_ptr, y_ptr,
    R, C, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    offs_c = tl.arange(0, BLOCK_C)

    r_mask = offs_r < R
    c_mask = offs_c < C

    x = tl.load(
        x_ptr + offs_r[:, None] * C + offs_c[None, :],
        mask=r_mask[:, None] & c_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    x_m = tl.where(c_mask[None, :], x, 0.0)
    s = tl.sum(x_m, axis=1)
    mean = s / C

    d = tl.where(c_mask[None, :], x - mean[:, None], 0.0)
    var = tl.sum(d * d, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    gw = tl.load(gw_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    gb = tl.load(gb_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)

    y = (d * rstd[:, None]) * gw[None, :] + gb[None, :]
    tl.store(
        y_ptr + offs_r[:, None] * C + offs_c[None, :],
        y.to(tl.bfloat16),
        mask=r_mask[:, None] & c_mask[None, :],
    )

        bm = k2_mask[:, None] & h_mask[None, :]
        wl = tl.load(wl_ptr + offs_k2[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wr = tl.load(wr_ptr + offs_k2[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wlg = tl.load(wlg_ptr + offs_k2[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wrk = tl.load(wrk_ptr + offs_k2[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wog = tl.load(wog_ptr + offs_k2[:, None] * H + offs_h[None, :], mask=bm, other=0.0)

        acc_l = tl.dot(a2, wl, acc_l)
        acc_r = tl.dot(a2, wr, acc_r)
        acc_lg = tl.dot(a2, wlg, acc_lg)
        acc_rg = tl.dot(a2, wrk, acc_rg)
        acc_og = tl.dot(a2, wog, acc_og)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=p_mask, other=1.0).to(tl.float32)[:, None]
    else:
        mask_val = 1.0

    lg = _sigmoid(acc_lg)
    rg = _sigmoid(acc_rg)
    og = _sigmoid(acc_og)

    left = acc_l * lg * mask_val
    right = acc_r * rg * mask_val

    out_off = (b * H + offs_h[None, :]) * N2 + offs_p[:, None]
    st_mask = p_mask[:, None] & h_mask[None, :]

    tl.store(l_ptr + out_off, left.to(tl.bfloat16), mask=st_mask)
    tl.store(r_ptr + out_off, right.to(tl.bfloat16), mask=st_mask)
    tl.store(og_ptr + out_off, og.to(tl.bfloat16), mask=st_mask)


@triton.jit
def _precompute_kernel(
    xn_ptr, mask_ptr,
    l_ptr, r_ptr, og_ptr,
    wl_ptr, wr_ptr, wlg_ptr, wrk_ptr, wog_ptr,
    N2, C, H,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BK: tl.constexpr,
    K_BLOCKS,
    HAS_MASK: tl.constexpr,
):
    b = tl.program_id(0)
    p_block = tl.program_id(1)
    h_block = tl.program_id(2)

    offs_p = p_block * BLOCK_R + tl.arange(0, BLOCK_R)
    p_mask = offs_p < N2

    offs_h = h_block * BLOCK_H + tl.arange(0, BLOCK_H)
    h_mask = offs_h < H

    offs_r = b * N2 + offs_p

    acc_l = tl.zeros((BLOCK_R, BLOCK_H), tl.float32)
    acc_r = tl.zeros((BLOCK_R, BLOCK_H), tl.float32)
    acc_lg = tl.zeros((BLOCK_R, BLOCK_H), tl.float32)
    acc_rg = tl.zeros((BLOCK_R, BLOCK_H), tl.float32)
    acc_og = tl.zeros((BLOCK_R, BLOCK_H), tl.float32)

    for kb in range(K_BLOCKS):
        offs_k = kb * BK + tl.arange(0, BK)
        k_mask = offs_k < C

        a = tl.load(
            xn_ptr + offs_r[:, None] * C + offs_k[None, :],
            mask=p_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        bm = k_mask[:, None] & h_mask[None, :]
        wl = tl.load(wl_ptr + offs_k[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wr = tl.load(wr_ptr + offs_k[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wlg = tl.load(wlg_ptr + offs_k[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wrk = tl.load(wrk_ptr + offs_k[:, None] * H + offs_h[None, :], mask=bm, other=0.0)
        wog = tl.load(wog_ptr + offs_k[:, None] * H + offs_h[None, :], mask=bm, other=0.0)

        acc_l = tl.dot(a, wl, acc_l)
        acc_r = tl.dot(a, wr, acc_r)
        acc_lg = tl.dot(a, wlg, acc_lg)
        acc_rg = tl.dot(a, wrk, acc_rg)
        acc_og = tl.dot(a, wog, acc_og)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=p_mask, other=1.0).to(tl.float32)[:, None]
    else:
        mask_val = 1.0

    lg = _sigmoid(acc_lg)
    rg = _sigmoid(acc_rg)
    og = _sigmoid(acc_og)

    left = acc_l * lg * mask_val
    right = acc_r * rg * mask_val

    out_off = (b * H + offs_h[None, :]) * N2 + offs_p[:, None]
    st_mask = p_mask[:, None] & h_mask[None, :]

    tl.store(l_ptr + out_off, left.to(tl.bfloat16), mask=st_mask)
    tl.store(r_ptr + out_off, right.to(tl.bfloat16), mask=st_mask)
    tl.store(og_ptr + out_off, og.to(tl.bfloat16), mask=st_mask)


@triton.jit
def _gemm_kernel(
    l_ptr, r_ptr, o_ptr,
    N, N2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EVEN: tl.constexpr, GROUP_M: tl.constexpr,
):
    bid = tl.program_id(0)
    pid = tl.program_id(1)

    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    group_id = pid // (GROUP_M * num_pid_n)
    first_m = group_id * GROUP_M
    group_size_m = tl.minimum(num_pid_m - first_m, GROUP_M)

    pid_m = first_m + (pid % group_size_m)
    pid_n = (pid % (GROUP_M * num_pid_n)) // group_size_m

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    base = bid * N2

    a_ptrs = l_ptr + base + offs_m[:, None] * N + offs_k[None, :]
    b_ptrs = r_ptr + base + offs_k[:, None] + offs_n[None, :] * N

    acc = tl.zeros((BLOCK_M, BLOCK_N), tl.float32)

    m_mask = offs_m < N
    n_mask = offs_n < N

    K_ITERS = tl.cdiv(N, BLOCK_K)
    for k in range(0, K_ITERS):
        if EVEN:
            a = tl.load(a_ptrs)
            b = tl.load(b_ptrs)
        else:
            k_start = k * BLOCK_K
            k_mask = (k_start + offs_k) < N
            a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        acc = tl.dot(a, b, acc)

        a_ptrs += BLOCK_K * N
        b_ptrs += BLOCK_K

    o_ptrs = o_ptr + base + offs_m[:, None] * N + offs_n[None, :]
    if EVEN:
        tl.store(o_ptrs, acc.to(tl.bfloat16))
    else:
        tl.store(o_ptrs, acc.to(tl.bfloat16), mask=m_mask[:, None] & n_mask[None, :])


@triton.jit
def _final_kernel(
    o_ptr, g_ptr, w_ptr, gw_ptr, gb_ptr, y_ptr,
    B, N2, C, H, eps,
    BLOCK_P: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
    C_BLOCKS,
):
    b = tl.program_id(0)
    p_block = tl.program_id(1)

    offs_p = p_block * BLOCK_P + tl.arange(0, BLOCK_P)
    p_mask = offs_p < N2

    offs_h = tl.arange(0, BLOCK_H)
    h_mask = offs_h < H

    base = b * H * N2
    ptr_off = base + offs_h[:, None] * N2 + offs_p[None, :]
    hp_mask = h_mask[:, None] & p_mask[None, :]

    o = tl.load(o_ptr + ptr_off, mask=hp_mask, other=0.0).to(tl.float32)

    o_m = tl.where(h_mask[:, None], o, 0.0)
    s = tl.sum(o_m, axis=0)
    mean = s / H

    d = tl.where(h_mask[:, None], o - mean[None, :], 0.0)
    var = tl.sum(d * d, axis=0) / H
    rstd = 1.0 / tl.sqrt(var + eps)

    gw = tl.load(gw_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)
    gb = tl.load(gb_ptr + offs_h, mask=h_mask, other=0.0).to(tl.float32)

    g = tl.load(g_ptr + ptr_off, mask=hp_mask, other=0.0).to(tl.float32)

    v = d * rstd[None, :] * gw[:, None] + gb[:, None]
    v = v * g
    v = tl.where(h_mask[:, None], v, 0.0)
    v = tl.trans(v).to(tl.bfloat16)

    y_base = b * N2 * C

    for cb in range(C_BLOCKS):
        offs_c = cb * BLOCK_C + tl.arange(0, BLOCK_C)
        c_mask = offs_c < C

        w = tl.load(
            w_ptr + offs_h[:, None] * C + offs_c[None, :],
            mask=h_mask[:, None] & c_mask[None, :],
            other=0.0,
        )

        acc = tl.dot(v, w)

        y_ptrs = y_ptr + y_base + offs_p[:, None] * C + offs_c[None, :]
        tl.store(y_ptrs, acc.to(tl.float32), mask=p_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    if len(data) == 4:
        x, mask, weights, config = data
    else:
        x, weights, config = data
        mask = None

    B, N, _, C = x.shape
    H = int(config.get("hidden_dim", 128))
    N2 = N * N
    R = B * N2
    device = x.device
    x = x.contiguous()

    eps = 1e-5

    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    def _wt(name):
        return weights[name].to(torch.bfloat16).t().contiguous()

    w_l = _wt("left_proj.weight")
    w_r = _wt("right_proj.weight")
    w_lg = _wt("left_gate.weight")
    w_rg = _wt("right_gate.weight")
    w_og = _wt("out_gate.weight")

    left = torch.empty(B, H, N, N, dtype=torch.bfloat16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.bfloat16, device=device)
    og = torch.empty(B, H, N, N, dtype=torch.bfloat16, device=device)

    has_mask = False
    mask_tensor = x
    if mask is not None:
        if mask.numel() != 0:
            mt = mask
            if mt.dim() == 4 and mt.shape[-1] == 1:
                mt = mt.squeeze(-1)
            if mt.dim() == 2:
                mt = mt.unsqueeze(0)
            if mt.size(0) == 1 and B > 1:
                mt = mt.expand(B, N, N)
            if mt.numel() == N2 and B > 1:
                mt = mt.expand(B, N, N)
            mask_tensor = mt.reshape(R).to(torch.float32).contiguous()
            has_mask = True

    if C <= 384:
        BK = 128
        k_blocks = triton.cdiv(C, BK)
        BLOCK_R = 32
        BLOCK_H = 128
        h_blocks = triton.cdiv(H, BLOCK_H)
        p_blocks = triton.cdiv(N2, BLOCK_R)
        grid = (B, p_blocks, h_blocks)

        _fused_precompute_kernel[grid](
            x, mask_tensor,
            left, right, og,
            norm_w, norm_b,
            w_l, w_r, w_lg, w_rg, w_og,
            N2, C, H, eps,
            BLOCK_R=BLOCK_R, BLOCK_H=BLOCK_H, BK=BK,
            K_BLOCKS=k_blocks,
            HAS_MASK=has_mask,
            num_warps=8, num_stages=1,
        )
    else:
        xn = torch.empty(B, N, N, C, dtype=torch.bfloat16, device=device)
        block_c = triton.next_power_of_two(C)
        if block_c > 1024:
            block_c = 1024
        block_r = 32 if block_c >= 512 else 64
        grid = (triton.cdiv(R, block_r),)

        _ln_kernel[grid](
            x, norm_w, norm_b, xn,
            R, C, eps,
            BLOCK_R=block_r, BLOCK_C=block_c,
            num_warps=8, num_stages=1,
        )

        BK = 128
        k_blocks = triton.cdiv(C, BK)
        BLOCK_R = 32
        BLOCK_H = 128
        h_blocks = triton.cdiv(H, BLOCK_H)
        p_blocks = triton.cdiv(N2, BLOCK_R)
        grid = (B, p_blocks, h_blocks)

        _precompute_kernel[grid](
            xn, mask_tensor,
            left, right, og,
            w_l, w_r, w_lg, w_rg, w_og,
            N2, C, H,
            BLOCK_R=BLOCK_R, BLOCK_H=BLOCK_H, BK=BK,
            K_BLOCKS=k_blocks,
            HAS_MASK=has_mask,
            num_warps=8, num_stages=2,
        )

    out = torch.empty(B, H, N, N, dtype=torch.bfloat16, device=device)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    even = (N % BLOCK_M == 0) and (N % BLOCK_N == 0) and (N % BLOCK_K == 0)
    num_m = triton.cdiv(N, BLOCK_M)
    num_n = triton.cdiv(N, BLOCK_N)
    grid = (B * H, num_m * num_n)

    _gemm_kernel[grid](
        left, right, out,
        N, N2,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        EVEN=even, GROUP_M=8,
        num_warps=8, num_stages=3,
    )

    y = torch.empty(B, N, N, C, dtype=torch.float32, device=device)
    to_out_norm_w = weights["to_out_norm.weight"].float().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].float().contiguous()

    if H <= 256:
        block_h = max(16, triton.next_power_of_two(H))
        if block_h <= 256:
            block_p = 64 if block_h <= 128 else 32
            block_c = 128
            c_blocks = triton.cdiv(C, block_c)
            grid = (B, triton.cdiv(N2, block_p))

            w_to = weights["to_out.weight"].to(torch.bfloat16).t().contiguous()

            _final_kernel[grid](
                out, og, w_to,
                to_out_norm_w, to_out_norm_b,
                y,
                B, N2, C, H, eps,
                BLOCK_P=block_p, BLOCK_H=block_h, BLOCK_C=block_c,
                C_BLOCKS=c_blocks,
                num_warps=8, num_stages=2,
            )
        else:
            out_v = out.view(B, H, N, N).permute(0, 2, 3, 1).float().contiguous()
            og_v = og.view(B, H, N, N).permute(0, 2, 3, 1).float().contiguous()
            out_v = F.layer_norm(out_v, (H,), to_out_norm_w, to_out_norm_b, eps=eps)
            out_v = out_v * og_v
            w_to = weights["to_out.weight"].float().contiguous()
            y = torch.bmm(
                out_v.reshape(B, N2, H),
                w_to.t().unsqueeze(0).expand(B, C, H),
            ).view(B, N, N, C)
    else:
        out_v = out.view(B, H, N, N).permute(0, 2, 3, 1).float().contiguous()
        og_v = og.view(B, H, N, N).permute(0, 2, 3, 1).float().contiguous()
        out_v = F.layer_norm(out_v, (H,), to_out_norm_w, to_out_norm_b, eps=eps)
        out_v = out_v * og_v
        w_to = weights["to_out.weight"].float().contiguous()
        y = torch.bmm(
            out_v.reshape(B, N2, H),
            w_to.t().unsqueeze(0).expand(B, C, H),
        ).view(B, N, N, C)

    return y
```
