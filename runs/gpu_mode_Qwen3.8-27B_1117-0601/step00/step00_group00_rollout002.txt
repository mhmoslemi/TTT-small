```python
import torch
import triton
import triton.language as tl


"""
Fused TriMul forward.

Pipeline:
1. Triton kernel:
   - LayerNorm on the pair head [B, N, N, C]
   - fused projections for left/right, left/right gates, and out gate
   - apply mask + multiplication with sigmoid gates for left/right
   - direct emission of left/right in [B, H, N, N] layout for fast batched GEMM
   - emit out_gate as [B, N, N, H]
2. cuBLAS batched matmul computes the N^3 TriMul:
   out_h = left_h @ right_h^T for every batch b and head h
3. Triton kernel:
   - reads the batched-GEMM result, applies LayerNorm over H,
     multiplies by out_gate, and projects to C.
"""

_ONES_MASK_CACHE = {}


def _get_ones_mask(B: int, N: int, device: torch.device) -> torch.Tensor:
    key = (B, N, device)
    m = _ONES_MASK_CACHE.get(key)
    if m is None:
        m = torch.ones((B, N, N), dtype=torch.float32, device=device)
        _ONES_MASK_CACHE[key] = m
    return m


@triton.jit
def fused_pre_tri_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    lt_ptr,
    rt_ptr,
    og_ptr,
    B,
    N,
    C,
    H: tl.constexpr,
    P: tl.constexpr,
    BLOCK_C: tl.constexpr,
    C_BLOCKS: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    b = pid0 // N
    i = pid0 % N

    N2 = N * N
    k0 = pid1 * P
    offs_k = k0 + tl.arange(0, P)
    k_valid = offs_k < N
    row_valid = k_valid[:, None]

    h_off = tl.arange(0, H)
    c_off_base = tl.arange(0, BLOCK_C)

    # Row pointer for x: [B, N, N, C]
    base_x = (b * N2 + i * N) * C
    row_ptrs = x_ptr + base_x + offs_k[:, None] * C

    # First pass: LayerNorm statistics over C.
    sum_x = tl.zeros((P,), dtype=tl.float32)
    sum_xx = tl.zeros((P,), dtype=tl.float32)

    for cb in tl.static_range(C_BLOCKS):
        c_off = cb * BLOCK_C + c_off_base
        c_mask = c_off[None, :] < C
        x_chunk = tl.load(
            row_ptrs + c_off[None, :],
            mask=row_valid & c_mask,
            other=0.0,
        ).to(tl.float32)

        sum_x += tl.sum(x_chunk, axis=1)
        sum_xx += tl.sum(x_chunk * x_chunk, axis=1)

    mean = sum_x / C
    var = sum_xx / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    # Second pass: build normalized bf16 activations and do all projections.
    acc_l = tl.zeros((P, H), dtype=tl.float32)
    acc_r = tl.zeros((P, H), dtype=tl.float32)
    acc_lg = tl.zeros((P, H), dtype=tl.float32)
    acc_rg = tl.zeros((P, H), dtype=tl.float32)
    acc_og = tl.zeros((P, H), dtype=tl.float32)

    for cb in tl.static_range(C_BLOCKS):
        c_off = cb * BLOCK_C + c_off_base
        c_mask_1 = c_off < C
        c_mask_2d = c_mask_1[None, :]
        w_mask_2d = (c_mask_1[:, None] & (h_off[None, :] < H))

        x_chunk = tl.load(
            row_ptrs + c_off[None, :],
            mask=row_valid & c_mask_2d,
            other=0.0,
        ).to(tl.float32)

        nw_chunk = tl.load(norm_w_ptr + c_off, mask=c_mask_1, other=0.0).to(tl.float32)
        nb_chunk = tl.load(norm_b_ptr + c_off, mask=c_mask_1, other=0.0).to(tl.float32)

        xnorm = (x_chunk - mean[:, None]) * rstd[:, None] * nw_chunk[None, :] + nb_chunk[None, :]
        xbf = xnorm.to(tl.bfloat16)

        wl_chunk = tl.load(
            wl_ptr + cb * BLOCK_C * H + c_off[:, None] * H + h_off[None, :],
            mask=w_mask_2d,
            other=0.0,
        ).to(tl.bfloat16)
        wr_chunk = tl.load(
            wr_ptr + cb * BLOCK_C * H + c_off[:, None] * H + h_off[None, :],
            mask=w_mask_2d,
            other=0.0,
        ).to(tl.bfloat16)
        wlg_chunk = tl.load(
            wlg_ptr + cb * BLOCK_C * H + c_off[:, None] * H + h_off[None, :],
            mask=w_mask_2d,
            other=0.0,
        ).to(tl.bfloat16)
        wrg_chunk = tl.load(
            wrg_ptr + cb * BLOCK_C * H + c_off[:, None] * H + h_off[None, :],
            mask=w_mask_2d,
            other=0.0,
        ).to(tl.bfloat16)
        wog_chunk = tl.load(
            wog_ptr + cb * BLOCK_C * H + c_off[:, None] * H + h_off[None, :],
            mask=w_mask_2d,
            other=0.0,
        ).to(tl.bfloat16)

        acc_l += tl.dot(xbf, wl_chunk)
        acc_r += tl.dot(xbf, wr_chunk)
        acc_lg += tl.dot(xbf, wlg_chunk)
        acc_rg += tl.dot(xbf, wrg_chunk)
        acc_og += tl.dot(xbf, wog_chunk)

    # Apply mask and gates.
    mask_base = b * N2 + i * N
    mask_row = tl.load(
        mask_ptr + mask_base + offs_k,
        mask=k_valid,
        other=0.0,
    ).to(tl.float32)
    mask_2d = mask_row[:, None]

    lg = tl.sigmoid(acc_lg)
    rg = tl.sigmoid(acc_rg)
    og = tl.sigmoid(acc_og)

    left = acc_l * mask_2d * lg
    right = acc_r * mask_2d * rg

    left_bf = left.to(tl.bfloat16)
    right_bf = right.to(tl.bfloat16)
    og_bf = og.to(tl.bfloat16)

    # Store out_gate as [B, N, N, H]
    p = b * N2 + i * N + offs_k
    og_ptrs = og_ptr + p[:, None] * H + h_off[None, :]
    tl.store(og_ptrs, og_bf, mask=(k_valid[:, None] & (h_off[None, :] < H)))

    # Store left/right in [B, H, N, N] layout: fixed (b,h) -> i,k contiguous.
    lt_base = lt_ptr + b * H * N2 + i * N + k0
    rt_base = rt_ptr + b * H * N2 + i * N + k0

    left_t = tl.trans(left_bf)
    right_t = tl.trans(right_bf)

    lt_ptrs = lt_base + h_off[:, None] * N2 + offs_k[None, :]
    rt_ptrs = rt_base + h_off[:, None] * N2 + offs_k[None, :]

    store_mask_t = (h_off[:, None] < H) & k_valid[None, :]
    tl.store(lt_ptrs, left_t, mask=store_mask_t)
    tl.store(rt_ptrs, right_t, mask=store_mask_t)


@triton.jit
def fused_final_tri_kernel(
    cb_ptr,
    gate_ptr,
    w_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    B,
    N,
    C,
    H: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_C: tl.constexpr,
    C_BLOCKS: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    N2 = N * N
    M = B * N2

    j0 = pid_j * BLOCK_J
    offs_j = j0 + tl.arange(0, BLOCK_J)
    j_valid = offs_j < N

    p = pid_b * N2 + pid_i * N + offs_j
    p_valid = p < M

    offs_h = tl.arange(0, H)

    # Load C_b shape [B, H, N, N] as tile [H, BLOCK_J] for fixed (b, i, j-block).
    core_ptrs = cb_ptr + (pid_b * H + offs_h[:, None]) * N2 + (
        pid_i * N + offs_j[None, :]
    )
    core = tl.load(core_ptrs, mask=(j_valid[None, :] & (offs_h[:, None] < H)), other=0.0).to(
        tl.float32
    )

    # LayerNorm over H for each (i, j).
    mean = tl.sum(core, axis=0) / H
    sumsq = tl.sum(core * core, axis=0)
    var = sumsq / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    core_t = tl.trans(core)  # [BLOCK_J, H]

    gate = tl.load(
        gate_ptr + p[:, None] * H + offs_h[None, :],
        mask=(p_valid[:, None] & (offs_h[None, :] < H)),
        other=0.0,
    ).to(tl.float32)

    nw = tl.load(norm_w_ptr + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)
    nb = tl.load(norm_b_ptr + offs_h, mask=offs_h < H, other=0.0).to(tl.float32)

    a = (core_t - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
    a = (a * gate).to(tl.bfloat16)

    offs_c = tl.arange(0, BLOCK_C)
    out_base = out_ptr + p[:, None] * C

    for cb in tl.static_range(C_BLOCKS):
        c_off = cb * BLOCK_C + offs_c
        c_mask = c_off < C
        c_mask_2d = c_mask[None, :]

        w_block = tl.load(
            w_ptr + offs_h[:, None] * C + c_off[None, :],
            mask=((offs_h[:, None] < H) & c_mask_2d),
            other=0.0,
        ).to(tl.bfloat16)

        acc = tl.dot(a, w_block)

        tl.store(
            out_base + c_off[None, :],
            acc,
            mask=(p_valid[:, None] & c_mask_2d),
        )


def _as_bf16_t_contig(w: torch.Tensor) -> torch.Tensor:
    return w.t().contiguous().to(torch.bfloat16)


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    C = int(config["dim"])
    H = int(config["hidden_dim"])

    x = input_tensor.contiguous()
    B, N, N2, _ = x.shape
    device = x.device

    if mask is None:
        mask_f = _get_ones_mask(B, N, device)
    else:
        mask_f = mask.contiguous().to(torch.float32)

    # Norm weights/biases in fp32 for stable normalization.
    norm_w = weights["norm.weight"].contiguous().to(torch.float32)
    norm_b = weights["norm.bias"].contiguous().to(torch.float32)

    # Projection/gate weights as [C, H] bf16.
    wl = _as_bf16_t_contig(weights["left_proj.weight"])
    wr = _as_bf16_t_contig(weights["right_proj.weight"])
    wlg = _as_bf16_t_contig(weights["left_gate.weight"])
    wrg = _as_bf16_t_contig(weights["right_gate.weight"])
    wog = _as_bf16_t_contig(weights["out_gate.weight"])

    # Outputs from fused pre-kernel.
    Lt = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    Rt = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    out_gate = torch.empty((B, N, N, H), dtype=torch.bfloat16, device=device)

    # Launch fused pre-processing.
    P = 32
    BLOCK_C = 128
    C_BLOCKS = (C + BLOCK_C - 1) // BLOCK_C
    grid_pre = (B * N, triton.cdiv(N, P))

    fused_pre_tri_kernel[grid_pre](
        x,
        mask_f,
        norm_w,
        norm_b,
        wl,
        wr,
        wlg,
        wrg,
        wog,
        Lt,
        Rt,
        out_gate,
        B,
        N,
        C,
        H=H,
        P=P,
        BLOCK_C=BLOCK_C,
        C_BLOCKS=C_BLOCKS,
        num_warps=4,
        num_stages=2,
    )

    # Core TriMul: for each (b,h), C[ b,h ] = L[ b,h ] @ R[ b,h ]^T.
    BH = B * H
    L_bmm = Lt.view(BH, N, N)
    R_bmm = Rt.view(BH, N, N)
    Cb = torch.bmm(L_bmm, R_bmm.transpose(1, 2))  # [B*H, N, N], bf16

    # Free large left/right before allocating final output.
    del Lt, Rt

    # Final fused LayerNorm + gate + projection.
    to_out_w = _as_bf16_t_contig(weights["to_out.weight"])  # [H, C]
    to_out_norm_w = weights["to_out_norm.weight"].contiguous().to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].contiguous().to(torch.float32)

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    BLOCK_J = 64
    FINAL_BLOCK_C = 128
    FINAL_C_BLOCKS = (C + FINAL_BLOCK_C - 1) // FINAL_BLOCK_C
    grid_final = (B, N, triton.cdiv(N, BLOCK_J))

    fused_final_tri_kernel[grid_final](
        Cb,
        out_gate,
        to_out_w,
        to_out_norm_w,
        to_out_norm_b,
        out,
        B,
        N,
        C,
        H=H,
        BLOCK_J=BLOCK_J,
        BLOCK_C=FINAL_BLOCK_C,
        C_BLOCKS=FINAL_C_BLOCKS,
        num_warps=4,
        num_stages=2,
    )

    return out
```
