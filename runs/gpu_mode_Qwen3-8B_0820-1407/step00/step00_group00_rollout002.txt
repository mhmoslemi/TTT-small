import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data):
    """
    Implements the "outgoing" version of the Triangle Multiplicative Update (TriMul)
    operator for protein structure prediction models like AlphaFold3, Chai, and Protenix.

    This kernel computes the TriMul operation using Triton for optimized performance.
    The operation involves several linear projections and element-wise operations,
    followed by a matrix multiplication over the sequence dimension.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: [bs, seq_len, seq_len, dim], input tensor
            - mask: [bs, seq_len, seq_len], mask tensor (optional)
            - weights: Dictionary containing weights for all operations
            - config: Dictionary containing hyperparameters (dim, hidden_dim)

    Returns:
        output: [bs, seq_len, seq_len, dim], computed TriMul output tensor
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen = input_tensor.size(2)
    bs = input_tensor.size(0)

    # Ensure all tensors are on the same device
    device = input_tensor.device
    dtype = torch.float32

    # Access weights
    norm_weight = weights["norm.weight"].to(device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device, dtype=dtype)
    left_proj_weight = weights["left_proj.weight"].to(device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=dtype)
    left_gate_weight = weights["left_gate.weight"].to(device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=dtype)
    to_out_weight = weights["to_out.weight"].to(device, dtype=dtype)

    # Reshape input tensor to [bs * seq_len, seq_len, dim]
    batched_input = input_tensor.reshape(bs * seqlen, seqlen, dim)

    # Apply LayerNorm (only on the last dimension)
    batched_input_norm = batched_input.to(dtype) * norm_weight + norm_bias

    # Linear projections and gates
    left = tl.dot(batched_input_norm, left_proj_weight.T)
    right = tl.dot(batched_input_norm, right_proj_weight.T)

    left_gate = tl.dot(batched_input_norm, left_gate_weight.T).sigmoid()
    right_gate = tl.dot(batched_input_norm, right_gate_weight.T).sigmoid()
    out_gate = tl.dot(batched_input_norm, out_gate_weight.T).sigmoid()

    # Apply masks
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Perform the TriMul operation: (i, j) = sum_k left[i, k] * right[j, k]
    # We reshape the tensors to [bs * seq_len, seq_len, hidden_dim] and compute the matrix multiplication
    left = left.reshape(bs * seqlen, seqlen, hidden_dim)
    right = right.reshape(bs * seqlen, seqlen, hidden_dim)
    out = torch.einsum('ijk, ijk -> ijk', left, right)

    # Apply the output normalization and gate
    out = out * out_gate

    # Apply the final linear projection
    out = tl.dot(out.reshape(bs * seqlen, seqlen * hidden_dim), to_out_weight.T)

    # Reshape back
    out = out.reshape(bs, seqlen, seqlen, dim)

    # Apply the output normalization
    out = out * to_out_norm_weight + to_out_norm_bias

    # Ensure final output is in float32 for consistency
    return out.to(torch.float32)
