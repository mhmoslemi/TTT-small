import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Implements the "outgoing" version of the Triangle Multiplicative Update (TriMul) module
    for protein structure prediction models. This kernel accelerates the key computational step
    using Triton's tensor operations and memory efficiency strategies, optimizing for NVIDIA L40S
    architecture. The main operation is a fast matrix multiplication with attention masking.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access weights and configuration parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize output
    output = torch.zeros((bs, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32)

    # Precompute masks
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]

    # Define Triton kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr, mask_ptr, output_ptr,
        seq_len: tl.constexpr, dim: tl.constexpr, hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr, BETA: tl.constexpr
    ):
        pid = tl.program_id(0)
        pid_m = pid // seq_len
        pid_n = pid % seq_len

        # Compute positions for this thread
        m_idx = pid_m * seq_len + tl.arange(0, seq_len)
        n_idx = pid_n * seq_len + tl.arange(0, seq_len)

        # Load mask
        mask = tl.load(mask_ptr + m_idx[:, None] * seq_len + n_idx[None, :], mask=tl.arange(0, seq_len) < seq_len, other=0.0)
        mask = mask.to(tl.float32)

        # Load projected left, right, left_gate, right_gate, out_gate
        left_proj = tl.load(left_proj_ptr + m_idx[:, None] * seq_len + tl.arange(0, seq_len)[:, None] * hidden_dim + tl.arange(0, hidden_dim),
                            mask=tl.arange(0, seq_len)[:, None] < seq_len, other=0.0)
        right_proj = tl.load(right_proj_ptr + n_idx[None, :] * seq_len + tl.arange(0, seq_len)[None, :] * hidden_dim + tl.arange(0, hidden_dim),
                            mask=tl.arange(0, seq_len)[None, :] < seq_len, other=0.0)
        left_gate = tl.load(left_gate_ptr + m_idx[:, None] * seq_len + tl.arange(0, seq_len)[:, None] * hidden_dim + tl.arange(0, hidden_dim),
                            mask=tl.arange(0, seq_len)[:, None] < seq_len, other=0.0)
        right_gate = tl.load(right_gate_ptr + n_idx[None, :] * seq_len + tl.arange(0, seq_len)[None, :] * hidden_dim + tl.arange(0, hidden_dim),
                            mask=tl.arange(0, seq_len)[None, :] < seq_len, other=0.0)
        out_gate = tl.load(out_gate_ptr + m_idx[:, None] * seq_len + tl.arange(0, seq_len)[:, None] * hidden_dim + tl.arange(0, hidden_dim),
                           mask=tl.arange(0, seq_len)[:, None] < seq_len, other=0.0)

        # Apply gates
        left = left_proj * left_gate
        right = right_proj * right_gate

        # Apply mask
        left = left * mask
        right = right * mask

        # Compute out
        out = tl.dot(left, right)

        # Apply ToOutNorm
        out_norm = tl.load(to_out_norm_ptr + tl.arange(0, hidden_dim), other=0.0)
        out = out * out_norm

        # Apply out_gate
        out = out * out_gate

        # Store output
        tl.store(output_ptr + m_idx[:, None] * seq_len + n_idx[None, :] * seq_len + tl.arange(0, hidden_dim),
                 out, mask=tl.arange(0, seq_len)[:, None] < seq_len)

    # Launch Triton kernel
    BLOCK = 128
    BETA = 0.0
    grid = (seq_len * seq_len + (BLOCK - 1) // BLOCK, )
    tri_mul_kernel[grid](
        input_tensor.data_ptr(),
        left_proj_weight.data_ptr(),
        right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(),
        right_gate_weight.data_ptr(),
        out_gate_weight.data_ptr(),
        to_out_norm_weight.data_ptr(),
        to_out_weight.data_ptr(),
        mask.data_ptr(),
        output.data_ptr(),
        seq_len=seq_len,
        dim=dim,
        hidden_dim=hidden_dim,
        BLOCK=BLOCK,
        BETA=BETA
    )

    # Apply final normalization and output gate
    # These are handled internally in the kernel above, so we don't need to repeat them

    return output
