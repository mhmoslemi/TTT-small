import torch
import triton
import triton.language as tl
import math
import numpy as np

def custom_kernel(data):
    """
    This custom kernel implements the outgoing version of the Triangle Multiplicative Update (TriMul) module
    from AlphaFold3 and related models, optimized for performance using Triton. It computes the pairwise
    product over a sequence of dimension N for each batch using efficient memory access and parallelism.
    The kernel avoids unnecessary memory copies and uses fused operations to reduce overhead.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define all the weights from the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Convert mask to device and extend dimension
    mask = mask.to(input_tensor.device).unsqueeze(-1).bool()

    # Step 1: Apply LayerNorm to input
    # We compute layer norm using the formula: x = (x - mean) / sqrt(var + eps) * gamma + beta
    eps = 1e-5
    mean = torch.sum(input_tensor * mask.type_as(input_tensor), dim=(1, 2), keepdim=True) / \
           torch.sum(mask.type_as(input_tensor), dim=(1, 2), keepdim=True)
    var = torch.sum((input_tensor - mean) * (input_tensor - mean) * mask.type_as(input_tensor), dim=(1, 2), keepdim=True) / \
          torch.sum(mask.type_as(input_tensor), dim=(1, 2), keepdim=True) + eps
    input_norm = (input_tensor - mean) / torch.sqrt(var) * norm_weight + norm_bias

    # Step 2: Apply linear projections (left and right)
    left_proj = torch.matmul(input_norm, left_proj_weight.t())
    right_proj = torch.matmul(input_norm, right_proj_weight.t())

    # Step 3: Apply mask (element-wise multiplication)
    left_masked = left_proj * mask.type_as(left_proj)
    right_masked = right_proj * mask.type_as(right_proj)

    # Step 4: Apply gate (sigmoid) to the gates
    left_gate = torch.matmul(input_norm, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_norm, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_norm, out_gate_weight.t()).sigmoid()

    # Step 5: Apply gate to the projections
    left = left_masked * left_gate
    right = right_masked * right_gate

    # Step 6: Compute the TriMul output using einsum-like operation:
    # out = einsum('...i k d, ...j k d -> ...i j d', left, right)
    # Here we implement this in a fused way using Triton kernels

    # We need to create a Triton kernel for the main operation (TriMul)
    # We'll perform the computation in blocks for performance
    # We'll use the sequence dimension as the most expensive dimension (N)
    # We'll optimize for large N (up to 1024)

    # Kernel to perform the TriMul operation
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len, dim,
        BS: tl.constexpr, BLOCK: tl.constexpr,
        BLOCK_SEQ: tl.constexpr, BLOCK_DIM: tl.constexpr,
    ):
        # Get program id
        pid = tl.program_id(0)
        # Get block position
        block_idx = pid % (seq_len // BLOCK_SEQ)
        # Get which dimension we're computing
        dim_idx = pid // (seq_len // BLOCK_SEQ)

        # Compute the offset for the current block
        offset = tl.arange(0, BLOCK_SEQ)
        offset += block_idx * BLOCK_SEQ

        # Compute the offset in the input tensors
        left_offsets = offset[:, None] * seq_len + tl.arange(0, BLOCK_DIM)
        right_offsets = offset[:, None] * seq_len + tl.arange(0, BLOCK_DIM)
        left_offsets = left_offsets[:, None, :]  # [BLOCK_SEQ, 1, BLOCK_DIM]
        right_offsets = right_offsets[:, None, :]  # [BLOCK_SEQ, 1, BLOCK_DIM]

        # Load left and right values
        left = tl.load(left_ptr + left_offsets, mask=(offset < seq_len)[:, None, :], other=0.0)
        right = tl.load(right_ptr + right_offsets, mask=(offset < seq_len)[:, None, :], other=0.0)

        # Compute the product
        acc = tl.zeros((BLOCK_SEQ, BLOCK_DIM), dtype=tl.float32)
        for k in range(seq_len):
            # Load from left
            left_k = tl.load(left_ptr + left_offsets[:, :, k], mask=(offset < seq_len), other=0.0)
            # Load from right
            right_k = tl.load(right_ptr + right_offsets[:, :, k], mask=(offset < seq_len), other=0.0)
            # Compute the dot product and accumulate
            acc += tl.dot(left_k, right_k)

        # Apply the gate and save
        out = acc * tl.load(out_gate_ptr + dim_idx * seq_len + offset, mask=(offset < seq_len), other=0.0)
        tl.store(out_ptr + dim_idx * seq_len + offset, out, mask=(offset < seq_len))

    # Calculate grid size
    num_blocks = (seq_len * dim) // (BLOCK_SEQ * BLOCK_DIM) + 1

    # Allocate output tensor
    out = torch.zeros((bs, seq_len, seq_len, dim), device=input_tensor.device)

    # Launch Triton kernel
    num_warps = 4
    kernel_grid = (num_blocks,)
    tri_mul_kernel[lambda x: triton.cdiv(seq_len * dim, BLOCK_SEQ * BLOCK_DIM), num_warps](
        left_masked, right_masked, out, seq_len, dim, BS=bs, BLOCK=BLOCK_SEQ, BLOCK_SEQ=BLOCK_SEQ, BLOCK_DIM=BLOCK_DIM
    )

    # Step 7: Apply LayerNorm to the output
    # Compute layer norm for the output
    out_mean = torch.sum(out * mask.type_as(out), dim=(1, 2), keepdim=True) / \
              torch.sum(mask.type_as(out), dim=(1, 2), keepdim=True)
    out_var = torch.sum((out - out_mean) * (out - out_mean) * mask.type_as(out), dim=(1, 2), keepdim=True) / \
              torch.sum(mask.type_as(out), dim=(1, 2), keepdim=True) + eps
    out_norm = (out - out_mean) / torch.sqrt(out_var) * to_out_norm_weight + to_out_norm_bias

    # Step 8: Apply final linear transformation
    output = torch.matmul(out_norm, to_out_weight.t())

    return output
