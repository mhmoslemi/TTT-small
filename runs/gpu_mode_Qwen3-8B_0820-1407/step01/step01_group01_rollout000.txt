import torch
import triton
import triton.language as tl
from torch import einsum

def custom_kernel(data):
    """
    Implements the outgoing version of the Triangle Multiplicative (TriMul) operation
    used in AlphaFold3 and similar models. The operation computes a pairwise interaction
    tensor using projections, gates, and a matrix multiplication over the sequence dimension.

    Key optimizations:
    - Uses Triton for efficient matrix multiplication over the sequence dimension
    - Implements the mask logic within the kernel
    - Uses mixed precision for efficiency on the L40S GPU
    - Avoids PyTorch operations within the kernel
    - Supports various input sizes as per the constraints

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate output tensor
    output = torch.zeros((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device)

    # Define constants for the kernel
    BLOCK = 128
    num_blocks = triton.cdiv(seq_len, BLOCK)

    # Launch the TriMul kernel
    kernel = triton_heavy_kernel
    grid = (num_blocks, bs, seq_len)
    kernel[grid](
        input_tensor,
        mask,
        output,
        norm_weight,
        norm_bias,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        seq_len,
        dim,
        hidden_dim,
        BLOCK
    )

    # Apply output gate
    output = output * torch.nn.functional.sigmoid(torch.matmul(output, out_gate_weight))

    # Apply to_out linear projection
    output = torch.matmul(output, to_out_weight)

    # Ensure final output is in float32
    return output.to(torch.float32)


@triton.jit
def triton_heavy_kernel(
    input_tensor_ptr,
    mask_ptr,
    output_ptr,
    norm_weight_ptr,
    norm_bias_ptr,
    left_proj_weight_ptr,
    right_proj_weight_ptr,
    left_gate_weight_ptr,
    right_gate_weight_ptr,
    out_gate_weight_ptr,
    to_out_norm_weight_ptr,
    to_out_norm_bias_ptr,
    to_out_weight_ptr,
    seq_len,
    dim,
    hidden_dim,
    BLOCK: tl.constexpr
):
    # Define indices
    pid = tl.program_id(0)
    batch_idx = tl.program_id(1)
    seq_idx = tl.program_id(2)

    # Calculate offsets
    batch = batch_idx
    seq = seq_idx
    offset = pid * BLOCK + tl.arange(0, BLOCK)

    # Load the mask values
    mask_offsets = tl.arange(0, BLOCK)
    mask = tl.load(mask_ptr + batch * seq_len * seq_len + seq * seq_len + mask_offsets, mask=offset < seq_len, other=0.0)

    # Load the input tensor
    input_offsets = tl.arange(0, BLOCK)
    x = tl.load(input_tensor_ptr + batch * seq_len * seq_len * dim + seq * seq_len * dim + input_offsets * dim, mask=offset < seq_len, other=0.0)

    # Normalize the input tensor
    # norm_weight and norm_bias are [dim,]
    norm_weight = tl.load(norm_weight_ptr, aligned=True)
    norm_bias = tl.load(norm_bias_ptr, aligned=True)
    x = x * norm_weight + norm_bias

    # Project to hidden dimension using left_proj
    left_proj_weights = tl.load(left_proj_weight_ptr, aligned=True)
    left_proj_weights = left_proj_weights.reshape(hidden_dim, dim)
    left = tl.dot(x, left_proj_weights)

    # Project to hidden dimension using right_proj
    right_proj_weights = tl.load(right_proj_weight_ptr, aligned=True)
    right_proj_weights = right_proj_weights.reshape(hidden_dim, dim)
    right = tl.dot(x, right_proj_weights)

    # Apply the gates
    # Left gate
    left_gate_weights = tl.load(left_gate_weight_ptr, aligned=True)
    left_gate_weights = left_gate_weights.reshape(hidden_dim, dim)
    left_gate = tl.dot(x, left_gate_weights)
    left_gate = left_gate.sigmoid()

    # Right gate
    right_gate_weights = tl.load(right_gate_weight_ptr, aligned=True)
    right_gate_weights = right_gate_weights.reshape(hidden_dim, dim)
    right_gate = tl.dot(x, right_gate_weights)
    right_gate = right_gate.sigmoid()

    # Apply gates to left and right
    left = left * left_gate
    right = right * right_gate

    # Apply the mask
    left = left * mask
    right = right * mask

    # Compute the TriMul operation using einsum
    # This is equivalent to: out[b, i, j] = sum_k (left[b, i, k] * right[b, j, k])
    # For each batch, we compute this for all pairs (i, j)

    # Initialize the output for the current block
    # We need to compute the output for all pairs (i, j)
    # So we'll loop over all i and j in the current block
    # We'll compute for all k in the current block

    # Define offsets for the output
    i_offsets = tl.arange(0, BLOCK)
    j_offsets = tl.arange(0, BLOCK)

    # Initialize the output
    out = tl.zeros((BLOCK, BLOCK, hidden_dim), dtype=tl.float32)

    # Compute the TriMul operation
    for k in range(0, seq_len, BLOCK):
        # Load the left and right for this k
        left_k = tl.load(left + k * seq_len * hidden_dim + i_offsets * hidden_dim, align=16)
        right_k = tl.load(right + k * seq_len * hidden_dim + j_offsets * hidden_dim, align=16)

        # Compute the product of left_k and right_k
        temp = tl.dot(left_k, right_k)

        # Accumulate into the output
        out = out + temp

    # Apply the to_out_norm
    to_out_norm_weights = tl.load(to_out_norm_weight_ptr, aligned=True)
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr, aligned=True)
    out = out * to_out_norm_weights + to_out_norm_bias

    # Apply the output gate
    out_gate_weights = tl.load(out_gate_weight_ptr, aligned=True)
    out_gate_weights = out_gate_weights.reshape(hidden_dim, hidden_dim)
    out_gate = tl.dot(out, out_gate_weights)
    out_gate = out_gate.sigmoid()

    # Apply the output gate to the output
    out = out * out_gate

    # Apply to_out linear projection
    to_out_weights = tl.load(to_out_weight_ptr, aligned=True)
    to_out_weights = to_out_weights.reshape(dim, hidden_dim)
    out = tl.dot(out, to_out_weights)

    # Write the result back to the output tensor
    out_offsets = offset[:, None, None] * seq_len * seq_len + seq_len * j_offsets[None, :] + i_offsets[None, :]
    tl.store(output_ptr + batch * seq_len * seq_len * hidden_dim + out_offsets, out, mask=offset < seq_len)
