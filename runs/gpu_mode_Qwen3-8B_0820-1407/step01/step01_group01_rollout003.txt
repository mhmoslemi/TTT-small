import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim_output = dim

    # Define Triton kernel to perform the TriMul operation
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_w_ptr, right_proj_w_ptr,
        left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
        to_out_norm_w_ptr, to_out_norm_b_ptr, to_out_w_ptr,
        out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Get the block index for the batch dimension
        pid = tl.program_id(0)
        # Get the block index for the sequence dimension (i.e., the row index of the output)
        pid_seq = tl.program_id(1)
        # Get the block index for the sequence dimension (i.e., the column index of the output)
        pid_seq2 = tl.program_id(2)
        # Get the indices within the block
        offs_b = pid
        offs_i = pid_seq * BLOCK + tl.arange(0, BLOCK)
        offs_j = pid_seq2 * BLOCK + tl.arange(0, BLOCK)
        # Create mask for valid indices
        m = (offs_i < seq_len) & (offs_j < seq_len)

        # Load weights
        left_proj_w = tl.load(left_proj_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        right_proj_w = tl.load(right_proj_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        left_gate_w = tl.load(left_gate_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        right_gate_w = tl.load(right_gate_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        out_gate_w = tl.load(out_gate_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        to_out_norm_w = tl.load(to_out_norm_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        to_out_norm_b = tl.load(to_out_norm_b_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)
        to_out_w = tl.load(to_out_w_ptr + offs_i[:, None] * hidden_dim + offs_j[None, :], mask=m, other=0.0)

        # Load input tensor (batch, seq_len, seq_len, dim) -> (B, i, j, d)
        input = tl.load(input_ptr + offs_b * seq_len * seq_len * dim + offs_i[:, None] * seq_len * dim + offs_j[None, :] * dim + tl.arange(0, hidden_dim), mask=m, other=0.0)

        # Apply projections
        left = input @ left_proj_w
        right = input @ right_proj_w

        # Apply masks
        mask = tl.load(mask_ptr + offs_b * seq_len * seq_len + offs_i[:, None] * seq_len + offs_j[None, :], mask=m, other=0.0)
        mask = mask[:, None]
        left = left * mask
        right = right * mask

        # Apply gates
        left_gate = input @ left_gate_w
        right_gate = input @ right_gate_w
        left_gate = left_gate.sigmoid()
        right_gate = right_gate.sigmoid()
        left = left * left_gate
        right = right * right_gate

        # Compute the output using einsum equivalent
        # out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This is a tensor of size (batch_size, seq_len, seq_len, hidden_dim)
        # We'll compute this as a sum over k
        out = tl.zeros((BLOCK, BLOCK, hidden_dim), dtype=tl.float32)
        for k in range(seq_len):
            left_k = left + k * dim
            right_k = right + k * dim
            acc = tl.dot(left_k, right_k)
            out += acc

        # Apply LayerNorm
        out = out - to_out_norm_b
        out = out * to_out_norm_w
        out = out * out_gate

        # Apply final projection
        out = out @ to_out_w
        # Store the result
        tl.store(out_ptr + offs_b * seq_len * seq_len * dim + offs_i[:, None] * seq_len * dim + offs_j[None, :] * dim + tl.arange(0, dim), out, mask=m)

    # Define the kernel parameters
    kernel_params = {
        "input_ptr": input_tensor,
        "mask_ptr": mask,
        "left_proj_w_ptr": weights["left_proj.weight"],
        "right_proj_w_ptr": weights["right_proj.weight"],
        "left_gate_w_ptr": weights["left_gate.weight"],
        "right_gate_w_ptr": weights["right_gate.weight"],
        "out_gate_w_ptr": weights["out_gate.weight"],
        "to_out_norm_w_ptr": weights["to_out_norm.weight"],
        "to_out_norm_b_ptr": weights["to_out_norm.bias"],
        "to_out_w_ptr": weights["to_out.weight"],
        "out_ptr": torch.empty((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32),
        "batch_size": batch_size,
        "seq_len": seq_len,
        "dim": dim,
        "hidden_dim": hidden_dim,
    }

    # Launch the kernel
    BLOCK = 64
    grid = (triton.cdiv(batch_size, 1), triton.cdiv(seq_len, BLOCK), triton.cdiv(seq_len, BLOCK))
    tri_mul_kernel[grid](**kernel_params)

    return kernel_params["out_ptr"]
