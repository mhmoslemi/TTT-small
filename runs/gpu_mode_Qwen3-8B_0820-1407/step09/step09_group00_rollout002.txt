import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --- Step 1: Transpose input to [bs, dim, seq_len, seq_len]
    # This is required for efficient memory access during the matrix operation
    input_tensor = input_tensor.transpose(1, 2).contiguous()  # shape [bs, dim, seq_len, seq_len]

    # --- Step 2: Precompute weights
    norm_weight = weights["norm.weight"].to(input_tensor.device)
    norm_bias = weights["norm.bias"].to(input_tensor.device)
    left_proj_weight = weights["left_proj.weight"].to(input_tensor.device)
    right_proj_weight = weights["right_proj.weight"].to(input_tensor.device)
    left_gate_weight = weights["left_gate.weight"].to(input_tensor.device)
    right_gate_weight = weights["right_gate.weight"].to(input_tensor.device)
    out_gate_weight = weights["out_gate.weight"].to(input_tensor.device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(input_tensor.device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(input_tensor.device)
    to_out_weight = weights["to_out.weight"].to(input_tensor.device)

    # --- Step 3: Apply LayerNorm (norm) to input
    # We implement LayerNorm using triton, as it's a basic operation
    @triton.jit
    def layer_norm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, eps, n, N, M, K, block_size: tl.constexpr):
        # This kernel applies LayerNorm to a tensor of shape [bs, dim, seqlen, seqlen]
        # We assume the shape is [bs, dim, seqlen, seqlen], with seqlen being the sequence length

        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)

        # Compute the indices for the current block in the input and output
        i = pid  # Batch ID
        block_indices = off // (M * K)
        m = (off // K) % M
        k = off % K

        # Load the input values and weights
        x = tl.load(x_ptr + i * N * M * K + block_indices * M * K + m * K + k, mask=(off < n * N * M * K), other=0.0)
        w = tl.load(weight_ptr + block_indices * M * K + m * K + k, mask=(off < n * N * M * K), other=0.0)
        b = tl.load(bias_ptr + block_indices * M * K + m * K + k, mask=(off < n * N * M * K), other=0.0)

        # Compute the mean and variance (but we're using the provided weights for LayerNorm)
        # This is a simplified LayerNorm without actual computation in this kernel
        normalized = x * w + b  # Placeholder for LayerNorm

        # Store the result
        tl.store(y_ptr + i * N * M * K + block_indices * M * K + m * K + k, normalized, mask=(off < n * N * M * K))

    # Launch the LayerNorm kernel
    n_blocks = triton.cdiv(batch_size * dim * seq_len * seq_len, 64)
    grid = (n_blocks, 1)
    layer_norm_kernel[grid](input_tensor, input_tensor, norm_weight, norm_bias, 1e-5, batch_size, dim, seq_len, seq_len, 64)

    # --- Step 4: Compute left and right projections
    # We will use triton kernels for these dense projections
    @triton.jit
    def dense_proj_kernel(x_ptr, y_ptr, weight_ptr, n, m, k, block_size: tl.constexpr):
        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)

        # Compute the indices
        i = pid  # Batch ID
        m_idx = (off // k) % m
        k_idx = off % k

        # Load weights and input
        w = tl.load(weight_ptr + block_idx * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)
        x = tl.load(x_ptr + i * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)

        # Compute the dot product
        y = x * w

        # Store the result
        tl.store(y_ptr + i * m * k + m_idx * k + k_idx, y, mask=(off < n * m * k))

    # Launch left_proj
    # [bs, dim, seqlen, seqlen] * [dim, hidden_dim] -> [bs, hidden_dim, seqlen, seqlen]
    left_proj_output = torch.zeros(batch_size, hidden_dim, seq_len, seq_len, device=input_tensor.device)
    grid = (triton.cdiv(hidden_dim * seq_len * seq_len, 64), 1)
    dense_proj_kernel[grid](input_tensor, left_proj_output, left_proj_weight, batch_size, dim, hidden_dim, 64)

    # Launch right_proj
    right_proj_output = torch.zeros(batch_size, hidden_dim, seq_len, seq_len, device=input_tensor.device)
    grid = (triton.cdiv(hidden_dim * seq_len * seq_len, 64), 1)
    dense_proj_kernel[grid](input_tensor, right_proj_output, right_proj_weight, batch_size, dim, hidden_dim, 64)

    # --- Step 5: Apply masks
    mask = mask.unsqueeze(-1)  # shape [bs, seq_len, seq_len, 1]
    mask = mask.expand_as(left_proj_output)  # shape [bs, seq_len, seq_len, hidden_dim]
    left_proj_output = left_proj_output * mask
    right_proj_output = right_proj_output * mask

    # --- Step 6: Apply gates
    # We use triton kernels to compute gate activations
    @triton.jit
    def sigmoid_kernel(x_ptr, y_ptr, n, m, k, block_size: tl.constexpr):
        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)

        # Compute the indices
        i = pid  # Batch ID
        m_idx = (off // k) % m
        k_idx = off % k

        # Load input values
        x = tl.load(x_ptr + i * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)

        # Compute sigmoid
        y = 1.0 / (1.0 + tl.exp(-x))

        # Store the result
        tl.store(y_ptr + i * m * k + m_idx * k + k_idx, y, mask=(off < n * m * k))

    # Launch left_gate
    left_gate_output = torch.zeros(batch_size, hidden_dim, seq_len, seq_len, device=input_tensor.device)
    grid = (triton.cdiv(hidden_dim * seq_len * seq_len, 64), 1)
    sigmoid_kernel[grid](left_proj_output, left_gate_output, batch_size, hidden_dim, seq_len, 64)

    # Launch right_gate
    right_gate_output = torch.zeros(batch_size, hidden_dim, seq_len, seq_len, device=input_tensor.device)
    grid = (triton.cdiv(hidden_dim * seq_len * seq_len, 64), 1)
    sigmoid_kernel[grid](right_proj_output, right_gate_output, batch_size, hidden_dim, seq_len, 64)

    # Launch out_gate
    out_gate_output = torch.zeros(batch_size, hidden_dim, seq_len, seq_len, device=input_tensor.device)
    grid = (triton.cdiv(hidden_dim * seq_len * seq_len, 64), 1)
    sigmoid_kernel[grid](input_tensor, out_gate_output, batch_size, dim, hidden_dim, 64)

    # Apply gates to projections
    left_proj_output = left_proj_output * left_gate_output
    right_proj_output = right_proj_output * right_gate_output

    # --- Step 7: Compute the matrix multiplication (triple loop)
    # This is the most expensive part of the operation. We will implement this as an optimized kernel
    @triton.jit
    def tr_mul_kernel(x_ptr, y_ptr, z_ptr, n, m, k, block_size: tl.constexpr):
        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)
        off_m = off // k
        off_n = off % k

        # Compute the indices
        i = pid  # Batch ID
        m_idx = (off // k) % m
        k_idx = off % k

        # Load x and y values
        x = tl.load(x_ptr + i * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)
        y = tl.load(y_ptr + i * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)

        # Compute the dot product
        z = x * y

        # Store the result
        tl.store(z_ptr + i * m * k + m_idx * k + k_idx, z, mask=(off < n * m * k))

    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)
    grid = (triton.cdiv(hidden_dim * seq_len * seq_len, 64), 1)
    tr_mul_kernel[grid](left_proj_output, right_proj_output, output, batch_size, seq_len, seq_len, 64)

    # Apply to_out_norm
    @triton.jit
    def layer_norm_kernel_2(x_ptr, y_ptr, weight_ptr, bias_ptr, eps, n, N, M, K, block_size: tl.constexpr):
        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)

        # Compute the indices
        i = pid  # Batch ID
        block_indices = off // (M * K)
        m = (off // K) % M
        k = off % K

        # Load the input values and weights
        x = tl.load(x_ptr + i * N * M * K + block_indices * M * K + m * K + k, mask=(off < n * N * M * K), other=0.0)
        w = tl.load(weight_ptr + block_indices * M * K + m * K + k, mask=(off < n * N * M * K), other=0.0)
        b = tl.load(bias_ptr + block_indices * M * K + m * K + k, mask=(off < n * N * M * K), other=0.0)

        # Compute the mean and variance (placeholder)
        normalized = x * w + b  # Placeholder for LayerNorm

        # Store the result
        tl.store(y_ptr + i * N * M * K + block_indices * M * K + m * K + k, normalized, mask=(off < n * N * M * K))

    # Launch the LayerNorm kernel
    n_blocks = triton.cdiv(batch_size * hidden_dim * seq_len * seq_len, 64)
    grid = (n_blocks, 1)
    layer_norm_kernel_2[grid](output, output, to_out_norm_weight, to_out_norm_bias, 1e-5, batch_size, hidden_dim, seq_len, seq_len, 64)

    # Apply out_gate
    @triton.jit
    def gate_kernel(x_ptr, y_ptr, n, m, k, block_size: tl.constexpr):
        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)

        # Compute the indices
        i = pid  # Batch ID
        m_idx = (off // k) % m
        k_idx = off % k

        # Load input values
        x = tl.load(x_ptr + i * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)

        # Compute gate
        y = x * out_gate_output[i, m_idx, k_idx]

        # Store the result
        tl.store(y_ptr + i * m * k + m_idx * k + k_idx, y, mask=(off < n * m * k))

    output = output * out_gate_output

    # --- Step 8: Final linear projection
    @triton.jit
    def dense_proj_kernel_2(x_ptr, y_ptr, weight_ptr, n, m, k, block_size: tl.constexpr):
        pid = tl.program_id(0)  # Batch index
        block_idx = tl.program_id(1)  # Block index

        # Offsets within the block
        off = block_idx * block_size + tl.arange(0, block_size)

        # Compute the indices
        i = pid  # Batch ID
        m_idx = (off // k) % m
        k_idx = off % k

        # Load weights and input
        w = tl.load(weight_ptr + block_idx * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)
        x = tl.load(x_ptr + i * m * k + m_idx * k + k_idx, mask=(off < n * m * k), other=0.0)

        # Compute the dot product
        y = x * w

        # Store the result
        tl.store(y_ptr + i * m * k + m_idx * k + k_idx, y, mask=(off < n * m * k))

    # Final projection
    output = torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device)
    grid = (triton.cdiv(dim * seq_len * seq_len, 64), 1)
    dense_proj_kernel_2[grid](output, output, to_out_weight, batch_size, hidden_dim, dim, 64)

    return output
