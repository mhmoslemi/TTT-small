import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seq_len = input_tensor.shape[0], input_tensor.shape[1]
    output_shape = (bs, seq_len, seq_len, dim)

    # Convert masks to appropriate dtype
    mask = mask.to(torch.float32)

    # Define kernel
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        mask_ptr, norm_weight, norm_bias,
        left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
    ):
        pid = tl.program_id(0)
        m = pid // (seq_len // BLOCK_M)
        n = pid % (seq_len // BLOCK_M)

        offs_m = m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Load masks
        mask = tl.load(mask_ptr + offs_m[:, None], mask=(offs_m < seq_len)[:, None], other=0.0)
        mask = mask * tl.load(mask_ptr + offs_n[None, :], mask=(offs_n < seq_len), other=0.0)

        # Load left and right projections
        left = tl.load(left_ptr + offs_m[:, None, None] * (seq_len * dim) + offs_n[None, :, None] * dim)
        right = tl.load(right_ptr + offs_m[:, None, None] * (seq_len * dim) + offs_n[None, :, None] * dim)

        # Apply masks to left and right
        left *= mask
        right *= mask

        # Load gates
        left_gate = tl.load(left_gate_weight + offs_m[:, None] * hidden_dim + offs_n[None, :] * 1)
        right_gate = tl.load(right_gate_weight + offs_m[:, None] * hidden_dim + offs_n[None, :] * 1)
        out_gate = tl.load(out_gate_weight + offs_m[:, None] * hidden_dim + offs_n[None, :] * 1)

        # Apply gates
        left *= left_gate
        right *= right_gate

        # Compute output using triton.matmul
        out = tl.matmul(left, right, allow_tf32=True)

        # Apply out gate
        out *= out_gate

        # Apply to_out_norm
        out = tl.dot(out, to_out_norm_weight + tl.arange(0, hidden_dim) * hidden_dim)
        out += tl.load(to_out_norm_bias + tl.arange(0, hidden_dim) * hidden_dim)

        # Apply to_out
        out = tl.dot(out, to_out_weight + tl.arange(0, dim) * dim)
        out += tl.load(to_out_norm_bias + tl.arange(0, dim) * dim)

        # Store output
        tl.store(out_ptr + offs_m[:, None] * seq_len + offs_n[None, :], out)

    # Compute grid
    num_blocks = triton.cdiv(seq_len, BLOCK_M) * triton.cdiv(seq_len, BLOCK_N)
    grid = (num_blocks,)

    # Launch kernel
    triton_trimul_kernel[grid](input_tensor, input_tensor, input_tensor,
                              mask, norm_weight, norm_bias,
                              left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight,
                              to_out_norm_weight, to_out_norm_bias, to_out_weight,
                              bs, seq_len, dim, hidden_dim,
                              BLOCK=128, BLOCK_M=128, BLOCK_N=128)

    return input_tensor
