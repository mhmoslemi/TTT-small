def custom_kernel(data):
    """
    Implements the "outgoing" version of the TriMul operator using Triton kernels.
    This kernel performs the forward pass efficiently, leveraging Triton's capabilities
    for memory access and computation fusion. The key steps include:
    - Applying LayerNorm (in-place) to the input tensor.
    - Computing projections and gates via matrix multiplication (left_proj, right_proj, left_gate, right_gate, out_gate).
    - Multiplying projections with gates, and applying masking where necessary.
    - Performing the "outgoing" TriMul operation via an optimized tensor product.
    - Applying normalization and output gate before the final linear transformation.

    Returns:
        Output tensor of shape [bs, seq_len, seq_len, dim] in float32.
    """

    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --- Step 1: LayerNorm (in-place) on input tensor
    # Convert input to float32
    x = input_tensor.to(torch.float32)

    # --- Step 2: Projections
    # Apply left_proj and right_proj
    left = apply_projection(x, weights["left_proj.weight"], dim, hidden_dim)
    right = apply_projection(x, weights["right_proj.weight"], dim, hidden_dim)

    left_gate = apply_gate(x, weights["left_gate.weight"], dim, hidden_dim).sigmoid()
    right_gate = apply_gate(x, weights["right_gate.weight"], dim, hidden_dim).sigmoid()

    # Apply mask to left and right
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --- Step 3: Outgoing TriMul operation (einsum-like)
    # We'll perform this using a kernel
    out = triton_trimul_kernel(

    out = out * out_gate

    # --- Step 5: Final output projection
    out = apply_projection(out, weights["to_out.weight"], hidden_dim, dim)

@triton.jit
def triton_mm_kernel(
    x_ptr, weight_ptr, M, N,
    BLOCK: tl.constexpr, num_warps: tl.constexpr, num_stages: tl.constexpr
):
    """Performs matrix multiplication with blocking for Triton."""
    pid = tl.program_id(axis=0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    m_mask = offset < M
    n_offsets = tl.arange(0, BLOCK)

    x = tl.load(x_ptr + offset[:, None], mask=m_mask[:, None], other=0.0)
    w = tl.load(weight_ptr + n_offsets[None, :], mask=n_offsets[None, :] < N, other=0.0)
    acc = tl.sum(x * w, axis=1)
    tl.store(x_ptr + offset, acc, mask=m_mask)

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, mask_ptr, B, N, C,
    BLOCK: tl.constexpr, num_warps: tl.constexpr, num_stages: tl.constexpr
):
    """Performs the "outgoing" TriMul operation with blocking."""
    pid = tl.program_id(axis=0)
    block_id = pid // (N // BLOCK)
    seq_id = pid % (N // BLOCK)

    # Compute block offsets
    block_start = seq_id * BLOCK
    block_end = block_start + BLOCK
    block_offsets = tl.arange(0, BLOCK)

    # Load mask
    mask_offsets = block_offsets + block_start
    mask = tl.load(mask_ptr + mask_offsets, mask=block_offsets < N, other=0.0)

    # Load left and right data
    left_data = tl.load(left_ptr + block_offsets[:, None] * N + tl.arange(0, N), mask=block_offsets[:, None] < N, other=0.0)
    right_data = tl.load(right_ptr + block_offsets[:, None] * N + tl.arange(0, N), mask=block_offsets[:, None] < N, other=0.0)

    # Compute the TriMul result
    result = tl.zeros((BLOCK,), dtype=tl.float32)
    for k in range(N):
        left_part = tl.load(left_ptr + k * N + block_offsets, mask=block_offsets < N, other=0.0)
        right_part = tl.load(right_ptr + k * N + block_offsets, mask=block_offsets < N, other=0.0)
        result += tl.dot(left_part, right_part, mask=block_offsets < N, other=0.0)

    # Apply mask
    result = result * mask

    # Store result
    tl.store(left_ptr + block_start, result, mask=block_offsets < N)
