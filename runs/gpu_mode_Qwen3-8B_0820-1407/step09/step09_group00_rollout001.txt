import torch
import triton
import triton.language as tl

# The custom kernel for the TriMul operation
@triton.jit
def tri_mul_kernel(
    input_ptr, mask_ptr, 
    left_proj_w_ptr, right_proj_w_ptr, 
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr, 
    to_out_norm_w_ptr, to_out_norm_b_ptr, to_out_w_ptr, 
    output_ptr,
    batch_size, seq_len, dim, hidden_dim, 
    BLOCK: tl.constexpr, 
):
    # Compute the grid indices for this thread
    pid = tl.program_id(0)

    # Each thread processes a (i, j) pair in the output tensor
    i = pid // seq_len
    j = pid % seq_len

    # Compute the offset for the current thread
    offset_i = i * seq_len + j
    offset_i = tl.arange(0, seq_len) + offset_i

    # Load the mask
    mask = tl.load(mask_ptr + offset_i, mask=offset_i < seq_len, other=0.0)

    # Load the left and right projections
    left = tl.load(left_proj_w_ptr + offset_i, mask=offset_i < seq_len, other=0.0)
    right = tl.load(right_proj_w_ptr + offset_i, mask=offset_i < seq_len, other=0.0)

    # Compute gate activations
    left_gate = tl.load(left_gate_w_ptr + offset_i, mask=offset_i < seq_len, other=0.0)
    right_gate = tl.load(right_gate_w_ptr + offset_i, mask=offset_i < seq_len, other=0.0)
    out_gate = tl.load(out_gate_w_ptr + offset_i, mask=offset_i < seq_len, other=0.0)

    # Apply gates
    left = left * mask
    right = right * mask
    left = left * left_gate
    right = right * right_gate

    # Compute the TriMul sum over k
    acc = tl.zeros((dim,), dtype=tl.float32)
    for k in range(seq_len):
        left_k = tl.load(left_proj_w_ptr + offset_i + k * seq_len)
        right_k = tl.load(right_proj_w_ptr + offset_i + k * seq_len)
        dot = tl.dot(left_k, right_k, mask=mask, other=0.0)
        acc += dot

    # Apply output gate
    acc = acc * out_gate

    # Apply normalization
    norm_w = tl.load(to_out_norm_w_ptr + offset_i)
    norm_b = tl.load(to_out_norm_b_ptr + offset_i)
    acc = acc * norm_w + norm_b

    # Apply final projection
    to_out_w = tl.load(to_out_w_ptr + offset_i)
    acc = acc * to_out_w

    # Store the result
    tl.store(output_ptr + offset_i, acc)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Ensure input and mask are correct shapes
    assert input_tensor.shape == (batch_size, seq_len, seq_len, dim)
    assert mask.shape == (batch_size, seq_len, seq_len)

    # Convert mask to float32 and add dummy dimension
    mask = mask.unsqueeze(-1).float()

    # Initialize output tensor
    output = torch.empty((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32)

    # Define block size for Triton kernel
    BLOCK = 128

    # Launch the Triton kernel
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)
    tri_mul_kernel[grid](
        input_tensor, mask, 
        weights["left_proj.weight"], weights["right_proj.weight"], 
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"], 
        weights["to_out_norm.weight"], weights["to_out_norm.bias"], weights["to_out.weight"], 
        output,
        batch_size, seq_len, dim, hidden_dim, 
        BLOCK=BLOCK
    )

    return output
