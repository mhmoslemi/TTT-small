import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Convert mask to device if needed
    mask = mask.to(input_tensor.device)

    # Define the kernel
    @triton.jit
    def trit_mul_kernel(out_ptr, left_ptr, right_ptr, mask_ptr, dim, hidden_dim, seq_len,
                        num_seqs, batch_size, BLOCK: tl.constexpr):
        # Compute the program ID
        pid = tl.program_id(0)
        # Compute offset for the current sequence
        seq_idx = pid % num_seqs
        # Compute the offset for the current batch
        batch_idx = pid // num_seqs
        # Compute the indices for the current block
        offs_i = tl.arange(0, BLOCK)
        offs_j = tl.arange(0, BLOCK)
        # Compute the indices for the k dimension
        offs_k = tl.arange(0, seq_len)
        
        # Calculate the indices in the tensor
        i_offsets = seq_idx * seq_len + offs_i
        j_offsets = seq_idx * seq_len + offs_j
        k_offsets = seq_idx * seq_len + offs_k
        
        # Load the mask values
        mask = tl.load(mask_ptr + i_offsets[:, None] * seq_len + j_offsets[None, :], mask=tl.arange(0, BLOCK)[:, None] < seq_len, other=0.0)

        # Initialize the result
        out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)

        # Loop over k
        for k in range(0, seq_len, BLOCK):
            # Load the k-th block of left and right
            left_val = tl.load(left_ptr + batch_idx * seq_len * seq_len * hidden_dim + seq_idx * seq_len * hidden_dim + i_offsets[:, None] * hidden_dim + k * hidden_dim + offs_k[None, :], mask=tl.arange(0, BLOCK)[:, None] < seq_len, other=0.0)
            right_val = tl.load(right_ptr + batch_idx * seq_len * seq_len * hidden_dim + seq_idx * seq_len * hidden_dim + j_offsets[None, :] * hidden_dim + k * hidden_dim + offs_k[None, :], mask=tl.arange(0, BLOCK)[None, :] < seq_len, other=0.0)

            # Multiply and accumulate
            out = tl.dot(left_val, right_val) * mask + out

        # Store the result
        tl.store(out_ptr + batch_idx * seq_len * seq_len * dim + seq_idx * seq_len * dim + i_offsets[:, None] * dim + j_offsets[None, :], out)

    # Prepare the output tensor
    output = torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device)

    # Launch the trit_mul_kernel
    num_seqs = seq_len
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)  # Adjust BLOCK for optimal performance
    trit_mul_kernel[grid](output, input_tensor, input_tensor, mask, dim, hidden_dim, seq_len, num_seqs, batch_size, BLOCK=128)

    return output
