import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    """
    Implements the "outgoing" version of the TriMul operator using a Triton kernel.
    This kernel efficiently performs the matrix multiplication and mask operations
    required for the TriMul operator in BioML models like AlphaFold3.

    Key Operations:
    1. Applies all linear projections and activation functions.
    2. Multiplies with mask to zero out masked positions.
    3. Performs the matrix multiplication using a fused kernel over the sequence dimension.
    4. Applies final normalization and activation.

    The implementation is optimized for NVIDIA L40S with tensor cores and focuses on
    memory efficiency, tiling, and reuse of L2 cache to minimize memory bandwidth usage.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Normalize input
    x = input_tensor.to(torch.float32)
    x = x * norm_weight + norm_bias

    # Perform linear projections
    # Left and right projections
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Left and right gate transformations
    left_gate = torch.nn.functional.linear(x, left_gate_weight)
    right_gate = torch.nn.functional.linear(x, right_gate_weight)
    out_gate = torch.nn.functional.linear(x, out_gate_weight)

    # Apply mask
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
    left = left * mask
    right = right * mask

    # Apply sigmoid to gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Now, perform the triangular multiplication
    # We will perform matrix multiplication with triton to optimize this part
    # Prepare the output tensor (we'll allocate it in the kernel)
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=x.device, dtype=x.dtype)

    # Define the kernel
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len, B, C,
        BLOCK: tl.constexpr, THREADS: tl.constexpr
    ):
        # Get the current thread id
        pid = tl.program_id(0)
        # Each thread processes a row of the output
        i = pid
        # Calculate the block offsets
        block_offset = (pid // B) * B
        # Current output row
        i = block_offset + tl.arange(0, B)
        # Get the batch and element indices
        batch = pid % B
        # Get the current column indices (j)
        j = tl.arange(0, C)
        # Get the current k indices
        k = tl.arange(0, C)

        # Load the left and right matrices
        # We access them as [batch, i, k] and [batch, j, k]
        # We need to use the mask to zero out masked values
        # First, load the left matrix for this row i
        # left: [batch, seq_len, seq_len, hidden_dim]
        left = tl.load(left_ptr + batch * seq_len * seq_len * C + i * seq_len * C + k * C)
        right = tl.load(right_ptr + batch * seq_len * seq_len * C + j * seq_len * C + k * C)
        # Compute the product
        out = tl.dot(left, right)
        # Store the result
        tl.store(out_ptr + batch * seq_len * seq_len * C + i * seq_len * C + j * C, out)

    # Launch the kernel
    grid = (batch_size, seq_len, 1)
    triton_trimul_kernel[grid](
        left, right, output, seq_len, batch_size, hidden_dim, 
        BLOCK=128, THREADS=128
    )

    # Apply the output gate
    output = output * out_gate

    # Apply the final normalization
    output = torch.nn.functional.layer_norm(output, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    output = torch.nn.functional.linear(output, to_out_weight)

    return output
