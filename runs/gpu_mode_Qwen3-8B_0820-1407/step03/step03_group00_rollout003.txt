import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any
import math

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimized Triton implementation of the outgoing version of the TriMul operator.

    Args:
        input_tensor: Input tensor of shape [bs, seq_len, seq_len, dim]
        mask: Mask tensor of shape [bs, seq_len, seq_len]
        weights: Dictionary containing model weights
        config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define kernel parameters
    BLOCK = 128
    num_blocks = triton.cdiv(seq_len * seq_len, BLOCK)

    # Access the weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate the output tensor
    out = torch.empty_like(input_tensor)

    # Initialize the output tensor with zeros
    out.fill_(0)

    # Perform the TriMul operation using Triton
    @triton.jit
    def triton_trimul_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr, out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Determine the block index
        pid = tl.program_id(axis=0)
        block_idx = pid

        # Compute the block position
        b = (block_idx // (seq_len * seq_len // BLOCK)) % batch_size
        i = ((block_idx // seq_len) % seq_len)
        j = (block_idx % seq_len)

        # Load the batch
        batch_idx = b * seq_len * seq_len + i * seq_len + j
        input_block = tl.load(input_ptr + batch_idx * dim, mask=tl.arange(0, dim) < dim, other=0.0)
        mask_block = tl.load(mask_ptr + batch_idx, mask=tl.arange(0, 1) < 1, other=0.0)

        # LayerNorm weight and bias
        norm_weight = tl.load(norm_weight, mask=tl.arange(0, dim) < dim, other=0.0)
        norm_bias = tl.load(norm_bias, mask=tl.arange(0, dim) < dim, other=0.0)
        input_block = (input_block - norm_bias) * norm_weight

        # Left and right projections
        left_proj = tl.load(left_proj_ptr, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        left_proj = input_block @ left_proj
        right_proj = tl.load(right_proj_ptr, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        right_proj = input_block @ right_proj

        # Mask values
        mask_scalar = mask_block
        left_proj = left_proj * mask_scalar
        right_proj = right_proj * mask_scalar

        # Gates
        left_gate = tl.load(left_gate_ptr, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        left_gate = input_block @ left_gate
        left_gate = left_gate.sigmoid()

        right_gate = tl.load(right_gate_ptr, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        right_gate = input_block @ right_gate
        right_gate = right_gate.sigmoid()

        out_gate = tl.load(out_gate_ptr, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        out_gate = input_block @ out_gate
        out_gate = out_gate.sigmoid()

        # Apply gates
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

        # Compute TriMul
        # out[b, i, j, :] = sum_k (left[b, i, k, :] * right[b, j, k, :])
        # We will compute this as a dot product between left[i, k, :] and right[j, k, :]
        # So we transpose right to have [k, j, :], and compute with left[i, k, :]

        # Initialize output
        output_block = tl.zeros((dim,), dtype=tl.float32)

        # Accumulate the result
        for k in range(seq_len):
            k_idx = k * batch_size * seq_len * seq_len + b * seq_len * seq_len + i * seq_len + j
            left_k = tl.load(left_proj_ptr + k_idx * dim, mask=tl.arange(0, dim) < dim, other=0.0)
            right_k = tl.load(right_proj_ptr + k_idx * dim, mask=tl.arange(0, dim) < dim, other=0.0)
            output_block += tl.dot(left_k, right_k)

        # Apply the output gate and normalization
        output_block = output_block * out_gate
        to_out_norm_weight = tl.load(to_out_norm_weight, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        to_out_norm_bias = tl.load(to_out_norm_bias, mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
        output_block = to_out_norm_weight * output_block + to_out_norm_bias

        # Apply the final linear transformation
        to_out_weight = tl.load(to_out_weight, mask=tl.arange(0, dim) < dim, other=0.0)
        output_block = output_block @ to_out_weight

        # Store the result
        out_idx = b * seq_len * seq_len + i * seq_len + j
        tl.store(out_ptr + out_idx * dim, output_block, mask=tl.arange(0, dim) < dim)

    # Prepare the input and output pointers
    input_ptr = input_tensor.data_ptr()
    mask_ptr = mask.data_ptr()
    left_proj_ptr = left_proj_weight.data_ptr()
    right_proj_ptr = right_proj_weight.data_ptr()
    left_gate_ptr = left_gate_weight.data_ptr()
    right_gate_ptr = right_gate_weight.data_ptr()
    out_gate_ptr = out_gate_weight.data_ptr()
    to_out_norm_ptr = to_out_norm_weight.data_ptr()
    to_out_ptr = to_out_weight.data_ptr()
    out_ptr = out.data_ptr()

    # Launch the kernel
    grid = (num_blocks,)
    triton_trimul_kernel[grid](
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr, out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK=BLOCK
    )

    return out
