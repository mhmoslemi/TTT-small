import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Optimized Triton kernel for the outgoing version of the TriMul operator.
    This implementation uses Triton's native operations and avoids PyTorch operations
    within the kernel for performance. The kernel handles the core multiplication
    and summation over the sequence dimension efficiently.
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    seqlen = N
    batch_size = B

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Convert to appropriate data types
    input_tensor = input_tensor.to(torch.float32)

    # Perform norm
    x = input_tensor
    x = x * norm_weight + norm_bias

    # Compute left, right, left_gate, right_gate, out_gate
    left = x @ left_proj_weight.T
    right = x @ right_proj_weight.T
    left_gate = x @ left_gate_weight.T
    right_gate = x @ right_gate_weight.T
    out_gate = x @ out_gate_weight.T

    # Apply masks
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare output tensor
    out_shape = (batch_size, seqlen, seqlen, hidden_dim)
    out = torch.zeros(out_shape, device=input_tensor.device, dtype=torch.float32)

    # Triton kernel for the TriMul operation
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, C, 
        BLOCK: tl.constexpr, 
        SEQLEN: tl.constexpr
    ):
        # Compute the thread indices
        pid = tl.program_id(0)
        i = pid
        offs_i = tl.arange(0, BLOCK)
        offs_j = tl.arange(0, BLOCK)
        # Compute the index for the current block
        block_i = i * BLOCK
        block_j = i * BLOCK

        # Load the blocks of left and right
        # Each block is of size BLOCK x C
        # Load the left and right matrices
        left = tl.load(left_ptr + tl.arange(0, BLOCK)[:, None] * C + tl.arange(0, C)[None, :], 
                      mask=(tl.arange(0, BLOCK)[:, None] < SEQLEN) & (tl.arange(0, C)[None, :] < C), 
                      other=0.0)
        right = tl.load(right_ptr + tl.arange(0, BLOCK)[:, None] * C + tl.arange(0, C)[None, :], 
                      mask=(tl.arange(0, BLOCK)[:, None] < SEQLEN) & (tl.arange(0, C)[None, :] < C), 
                      other=0.0)

        # Compute the output matrix
        out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)
        for k in range(0, C, BLOCK):
            # Load the current block of left and right
            left_block = tl.load(left_ptr + (tl.arange(0, BLOCK)[:, None] * C + k + tl.arange(0, C)[None, :]), 
                               mask=(tl.arange(0, BLOCK)[:, None] < SEQLEN) & (tl.arange(0, C)[None, :] < C), 
                               other=0.0)
            right_block = tl.load(right_ptr + (tl.arange(0, BLOCK)[:, None] * C + k + tl.arange(0, C)[None, :]), 
                                 mask=(tl.arange(0, BLOCK)[:, None] < SEQLEN) & (tl.arange(0, C)[None, :] < C), 
                                 other=0.0)
            # Compute the dot product
            out += tl.dot(left_block, right_block)
        # Store the output back
        tl.store(out_ptr + i * SEQLEN * SEQLEN * C + tl.arange(0, BLOCK)[:, None] * SEQLEN + tl.arange(0, BLOCK)[None, :], 
                 out, 
                 mask=(tl.arange(0, BLOCK)[:, None] < SEQLEN) & (tl.arange(0, BLOCK)[None, :] < SEQLEN))

    # Launch the Triton kernel
    BLOCK = 128
    grid = (triton.cdiv(seqlen * seqlen, BLOCK * BLOCK), )
    # Compute the pointers for the left, right, and output tensors
    # Reshape the left and right tensors to be [B, seqlen, seqlen, C] and [B, seqlen, seqlen, C]
    left_reshaped = left.reshape(B, seqlen, seqlen, hidden_dim)
    right_reshaped = right.reshape(B, seqlen, seqlen, hidden_dim)

    # Convert left and right to contiguous tensors
    left_contiguous = left_reshaped.contiguous()
    right_contiguous = right_reshaped.contiguous()

    # Prepare for the kernel
    left_ptr = left_contiguous.data_ptr()
    right_ptr = right_contiguous.data_ptr()
    out_ptr = out.data_ptr()

    # Launch the kernel
    tri_mul_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, hidden_dim,
        BLOCK=BLOCK,
        SEQLEN=seqlen
    )

    # Apply the output gate and norm
    out = out * out_gate
    out = out @ to_out_norm_weight.T + to_out_norm_bias
    out = out @ to_out_weight.T
    out = out.to(torch.float32)

    return out
