import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple, Dict

def custom_kernel(data: Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]) -> Tensor:
    """
    Implement the "outgoing" version of the TriMul operation using a fully optimized Triton kernel.
    This kernel handles the forward pass of the TriMul module for the given input and weights.

    The TriMul operation involves:
    1. Applying LayerNorm to the input
    2. Projection to hidden dimensions using linear projections
    3. Masking operations
    4. Applying gate activations (sigmoid)
    5. Performing a matrix multiplication between left and right projections
    6. Applying normalization and final gate activation

    Parameters:
        data: Tuple of input tensor, mask tensor, weights dictionary, and config dictionary

    Returns:
        output: Tensor of shape [bs, seq_len, seq_len, dim], the processed output of the TriMul operation
    """

    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access all weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Convert to Triton types
    input_tensor = input_tensor.contiguous()
    out = torch.zeros(bs, seq_len, seq_len, hidden_dim, device=input_tensor.device, dtype=input_tensor.dtype)

    # Define the Triton kernel
    @triton.jit
    def tritongrid_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        out_ptr,
        norm_weight_ptr, norm_bias_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        N, D, H, B,
        BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr
    ):
        # Coordinates of the current thread
        pid = tl.program_id(0)
        pid_n = pid % N
        pid_d = pid // N % D
        pid_h = pid // (N * D)

        # Offsets for the current block
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
        offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

        # Mask indices (mask is applied in the kernel)
        mask_vals = tl.load(mask_ptr + pid_n * N + offs_n, mask=offs_n < N, other=0.0)
        mask_vals = mask_vals.to(tl.float32)

        # Load norm weights
        norm_weight = tl.load(norm_weight_ptr + offs_d, mask=offs_d < D, other=0.0)
        norm_bias = tl.load(norm_bias_ptr + offs_d, mask=offs_d < D, other=0.0)

        # Load weights
        left_proj = tl.load(left_proj_ptr + offs_d * D + offs_h, mask=offs_d < D, other=0.0)
        right_proj = tl.load(right_proj_ptr + offs_d * D + offs_h, mask=offs_d < D, other=0.0)
        left_gate = tl.load(left_gate_ptr + offs_d * D + offs_h, mask=offs_d < D, other=0.0)
        right_gate = tl.load(right_gate_ptr + offs_d * D + offs_h, mask=offs_d < D, other=0.0)
        out_gate = tl.load(out_gate_ptr + offs_d * D + offs_h, mask=offs_d < D, other=0.0)

        # Compute the norm
        norm_weight = norm_weight + norm_bias
        norm_weight = norm_weight.to(input_tensor.dtype)
        norm_weight = norm_weight / (D ** 0.5)

        # Compute the projections
        left_proj = left_proj * norm_weight
        right_proj = right_proj * norm_weight

        # Apply mask
        left_proj = left_proj * mask_vals
        right_proj = right_proj * mask_vals

        # Apply gates
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

        # Compute the TriMul operation
        acc = tl.zeros((BLOCK_N, BLOCK_H), dtype=tl.float32)
        for k in range(N):
            left = tl.load(input_ptr + pid_n * N * D + k * D + offs_d * D + offs_h, mask=offs_d < D, other=0.0)
            right = tl.load(input_ptr + pid_n * N * D + k * D + offs_d * D + offs_h, mask=offs_d < D, other=0.0)
            acc += tl.dot(left, right, acc.dtype)

        # Apply final gate
        acc = acc * out_gate

        # Apply to_out_norm and to_out
        to_out_norm_weight = tl.load(to_out_norm_weight_ptr + offs_d, mask=offs_d < D, other=0.0)
        to_out_norm_bias = tl.load(to_out_norm_bias_ptr + offs_d, mask=offs_d < D, other=0.0)
        to_out_weight = tl.load(to_out_weight_ptr + offs_d * D + offs_h, mask=offs_d < D, other=0.0)

        # Normalize
        to_out_norm_weight = to_out_norm_weight + to_out_norm_bias
        to_out_norm_weight = to_out_norm_weight / (D ** 0.5)
        acc = acc * to_out_norm_weight

        # Multiply by to_out_weight
        acc = acc * to_out_weight

        # Store the result
        tl.store(out_ptr + pid_n * N * D + pid_d * D + pid_h, acc)

    # Compute grid size
    num_blocks = (bs * seq_len * seq_len * dim) // (BLOCK_N * BLOCK_D * BLOCK_H)
    grid = (num_blocks,)
    BLOCK_N, BLOCK_D, BLOCK_H = 128, 64, 128

    # Launch kernel
    tritongrid_kernel[grid](
        input_tensor.data_ptr(), mask.data_ptr(), left_proj_weight.data_ptr(), right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(), right_gate_weight.data_ptr(), out_gate_weight.data_ptr(),
        out.data_ptr(),
        norm_weight.data_ptr(), norm_bias.data_ptr(),
        to_out_norm_weight.data_ptr(), to_out_norm_bias.data_ptr(), to_out_weight.data_ptr(),
        seq_len, dim, hidden_dim, bs,
        BLOCK_N, BLOCK_D, BLOCK_H
    )

    return out
