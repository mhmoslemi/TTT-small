import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Implements the TriMul module using a Triton kernel for efficient computation.
    Optimized for large sequence lengths, this kernel directly implements the
    TriMul forward pass with attention-like operations that are fused for speed.
    The kernel uses Triton's low-level control for performance, avoiding Python-level loops.
    All operations are performed in `torch.float32` with the final result cast to `torch.float32`.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape

    # Extract configuration parameters
    hidden_dim = config["hidden_dim"]
    
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize output tensor
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)

    # Define Triton kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        mask_ptr, output_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr, NUM_BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        block_idx = pid % NUM_BLOCK
        batch_idx = pid // NUM_BLOCK

        # Compute the batch and sequence indices
        batch_start = batch_idx * BLOCK
        seq_start = block_idx * BLOCK
        i_offsets = tl.arange(0, BLOCK)
        j_offsets = tl.arange(0, BLOCK)

        # Compute output indices (i, j)
        i = batch_start + i_offsets[:, None]
        j = seq_start + j_offsets[None, :]

        # Mask indices
        m_idx = i_offsets[:, None] + j_offsets[None, :]

        # Load input
        x = tl.load(input_ptr + batch_start * seq_len * seq_len * dim + i * seq_len * dim + j * dim, mask=m_idx < seq_len, other=0.0)

        # Compute left and right projections (x is [batch, seq, seq, dim])
        left = tl.dot(x, left_proj_weight, axis=-1)
        right = tl.dot(x, right_proj_weight, axis=-1)

        # Apply mask
        mask = tl.load(mask_ptr + batch_start * seq_len * seq_len + i * seq_len + j, mask=m_idx < seq_len, other=0.0)
        left = left * mask
        right = right * mask

        # Compute gates
        left_gate = tl.dot(x, left_gate_weight, axis=-1).sigmoid()
        right_gate = tl.dot(x, right_gate_weight, axis=-1).sigmoid()
        out_gate = tl.dot(x, out_gate_weight, axis=-1).sigmoid()

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute output: out[i][j] = sum_k (left[i][k] * right[j][k])
        # This is equivalent to a tensor product
        acc = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)

        for k in range(seq_len):
            left_k = tl.load(input_ptr + batch_start * seq_len * seq_len * dim + i * seq_len * dim + (k * dim):, 
                             mask=m_idx < seq_len, other=0.0)
            right_k = tl.load(input_ptr + batch_start * seq_len * seq_len * dim + (k * seq_len * dim) + j * dim,
                              mask=m_idx < seq_len, other=0.0)
            acc += tl.dot(left_k, right_k, axis=-1)

        # Normalize
        out = tl.dot(acc, to_out_norm_weight, axis=-1)
        out = out * out_gate

        # Apply final linear projection
        result = tl.dot(out, to_out_weight, axis=-1)

        # Store result
        tl.store(output_ptr + batch_start * seq_len * seq_len * hidden_dim + i * seq_len * hidden_dim + j * hidden_dim, result, mask=m_idx < seq_len)

    # Triton kernel launch
    BLOCK = 128
    NUM_BLOCK = triton.cdiv(seq_len, BLOCK)

    grid = (NUM_BLOCK, )
    tri_mul_kernel[grid](
        input_tensor,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_weight,
        mask,
        output,
        batch_size,
        seq_len,
        dim,
        hidden_dim,
        BLOCK=BLOCK,
        NUM_BLOCK=NUM_BLOCK
    )

    return output
