import triton
import triton.language as tl
import torch
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Ensure all data is on GPU and in the right precision
    input_tensor = input_tensor.to(torch.float32).contiguous()
    mask = mask.to(torch.float32).unsqueeze(-1).contiguous()

    # Access weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Allocate output tensor
    output = torch.empty((batch_size, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=torch.float32)

    # Normalize input
    x = input_tensor
    x = x * norm_weight + norm_bias

    # Linear projections
    left = torch.matmul(x, left_proj_weight.t())
    right = torch.matmul(x, right_proj_weight.t())

    # Apply mask
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = torch.matmul(x, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(x, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(x, out_gate_weight.t()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Perform the TriMul operation using Triton kernel
    def triton_trimul_kernel(x, y, z, n, m, k, dim, BLOCK: tl.constexpr):
        pid = tl.program_id(axis=0)
        num_pid_m = tl.cdiv(n, BLOCK)
        num_pid_n = tl.cdiv(m, BLOCK)
        pid_m = pid // num_pid_n
        pid_n = pid % num_pid_n

        offs_m = pid_m * BLOCK + tl.arange(0, BLOCK)
        offs_n = pid_n * BLOCK + tl.arange(0, BLOCK)

        m_mask = offs_m < n
        n_mask = offs_n < m

        x_offsets = tl.load(x + (offs_m[:, None] * k + offs_n[None, :] * 1), m_mask[:, None] & n_mask[None, :])
        y_offsets = tl.load(y + (offs_m[:, None] * k + offs_n[None, :] * 1), m_mask[:, None] & n_mask[None, :])

        z_offsets = tl.dot(x_offsets, y_offsets)
        tl.store(z + (offs_m[:, None] * m + offs_n[None, :] * 1), z_offsets, m_mask[:, None] & n_mask[None, :])

    # Compute the TriMul operation via Triton kernel
    BLOCK = 128
    grid = (triton.cdiv(seq_len, BLOCK) * triton.cdiv(seq_len, BLOCK),)
    triton_trimul_kernel[grid](left, right, output, seq_len, seq_len, hidden_dim, dim, BLOCK=BLOCK)

    # Apply LayerNorm and output gate
    output = output * to_out_norm_weight + to_out_norm_bias
    output = output * out_gate

    # Apply final output projection
    output = torch.matmul(output, to_out_weight.t())

    return output
