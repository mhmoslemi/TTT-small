import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define parameters
    B = batch_size
    N = seq_len
    C = dim

    # Constants for the kernel
    BLOCK = 64  # Tune this for performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # BLOCK_M = 128
    # BLOCK_N = 128
    # BLOCK_K = 128

    # Convert input tensor and mask to device
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32).unsqueeze(-1)

    # Access weights directly
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Output tensor
    output = torch.zeros((B, N, N, hidden_dim), device=input_tensor.device)

    # Apply LayerNorm to input
    # Using fused computation to avoid memory copy
    x = input_tensor
    x = x * norm_weight + norm_bias

    # Compute projections
    left = x @ left_proj_weight.transpose(0, 1)
    right = x @ right_proj_weight.transpose(0, 1)

    # Apply mask
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate = x @ left_gate_weight.transpose(0, 1)
    right_gate = x @ right_gate_weight.transpose(0, 1)
    out_gate = x @ out_gate_weight.transpose(0, 1)

    # Apply sigmoid gates (using approximate implementation for performance)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute TriMul output
    @triton.jit
    def tri_mul_kernel(out_ptr, left_ptr, right_ptr, B, N, H, 
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        # Map the thread to a position
        pid = tl.program_id(0)
        pid_m = pid // (N // BLOCK_M)
        pid_n = pid % (N // BLOCK_M)
        pid_k = pid % (N // BLOCK_K)

        # Compute offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left and right tensors
        left = tl.load(left_ptr + offs_m[:, None] * N + offs_k[None, :], 
                      mask=(offs_m < N) & (offs_k < N), 
                      other=0.0)
        right = tl.load(right_ptr + offs_n[None, :] * N + offs_k[:, None], 
                      mask=(offs_n < N) & (offs_k < N), 
                      other=0.0)

        # Compute output
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        for k in range(0, N, BLOCK_K):
            a = left[:, k:k + BLOCK_K]
            b = right[k:k + BLOCK_K, :]
            acc += tl.dot(a, b)
        tl.store(out_ptr + offs_m[:, None] * N + offs_n[None, :], acc)

    # Launch the kernel
    grid = (triton.cdiv(N * N, BLOCK_M * BLOCK_N),)
    tri_mul_kernel[grid](output, left, right, B, N, hidden_dim, 
                         BLOCK_M, BLOCK_N, BLOCK_K)

    # Apply LayerNorm and gate to output
    output = output @ to_out_norm_weight.transpose(0, 1) + to_out_norm_bias
    output = output * out_gate

    # Apply final linear projection
    output = output @ to_out_weight.transpose(0, 1)

    # Ensure output is in float32
    output = output.to(torch.float32)

    return output
