import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple, Dict, Any
import math

def custom_kernel(data: Tuple[Tensor, Tensor, Dict[str, Tensor], Dict[str, Any]]) -> Tensor:
    """
    Implements the outgoing version of the TriMul operator using Triton for high performance.

    This kernel performs the following steps:
    1. Normalization of input using LayerNorm
    2. Projection of input to hidden dimension
    3. Applying gate functions and masks
    4. Performing the triple matrix multiplication using einsum-like operation
    5. Applying normalization and gate functions again
    6. Final output projection

    The kernel is optimized for the given constraints and hardware (L40S).

    Args:
        data: Tuple of (input: Tensor, mask: Tensor, weights: Dict[str, Tensor], config: Dict)
    
    Returns:
        Output tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Define constants
    MAX_SEQ_LEN = 1024
    BLOCK = 64
    N_BLOCK = 16
    L_BLOCK = 16

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output tensor
    output = torch.zeros((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=input_tensor.dtype)

    # Define kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w,
        to_out_norm_w, to_out_norm_b, to_out_w,
        out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr,
        N_BLOCK: tl.constexpr,
        L_BLOCK: tl.constexpr
    ):
        # Grid dimensions
        pid = tl.program_id(0)
        batch_id = pid // (seq_len * seq_len)
        pid_in_batch = pid % (seq_len * seq_len)
        i = pid_in_batch // seq_len
        j = pid_in_batch % seq_len

        # Load input and mask
        input = tl.load(input_ptr + tl.arange(0, N_BLOCK)[:, None] * seq_len * dim + tl.arange(0, L_BLOCK)[None, :], 
                      mask=tl.arange(0, N_BLOCK)[:, None] * seq_len * dim + tl.arange(0, L_BLOCK)[None, :] < seq_len * seq_len * dim,
                      other=0.0)
        mask_val = tl.load(mask_ptr + tl.arange(0, N_BLOCK)[:, None] * seq_len + tl.arange(0, L_BLOCK)[None, :],
                           mask=tl.arange(0, N_BLOCK)[:, None] * seq_len + tl.arange(0, L_BLOCK)[None, :] < seq_len * seq_len,
                           other=0.0)
        
        # Apply mask
        input = input * mask_val

        # Projection and gate operations
        left_proj = tl.einsum("ikd, kmd -> ijd", input, left_proj_w)
        left_proj = left_proj * mask_val
        right_proj = tl.einsum("ikd, kmd -> ijd", input, right_proj_w)
        right_proj = right_proj * mask_val

        left_gate = tl.einsum("ikd, kmd -> ijd", input, left_gate_w)
        left_gate = left_gate.sigmoid() * mask_val
        right_gate = tl.einsum("ikd, kmd -> ijd", input, right_gate_w)
        right_gate = right_gate.sigmoid() * mask_val
        out_gate = tl.einsum("ikd, kmd -> ijd", input, out_gate_w)
        out_gate = out_gate.sigmoid() * mask_val

        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

        # TriMul operation
        out = tl.einsum("ikd, jkd -> ijd", left_proj, right_proj)

        # Apply LayerNorm
        out = tl.einsum("ijd, j -> ijd", out, to_out_norm_w) + to_out_norm_b
        out = out * out_gate

        # Final output
        out = tl.einsum("ijd, kmd -> ijd", out, to_out_w)
        tl.store(out_ptr + i * seq_len * dim + j * dim, out)

    # Launch kernel
    grid = (seq_len * seq_len,)
    tri_mul_kernel[grid](
        input_tensor, mask, left_proj_weight, right_proj_weight, 
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK=BLOCK, N_BLOCK=N_BLOCK, L_BLOCK=L_BLOCK
    )

    # Ensure output is in float32
    return output.to(torch.float32)
