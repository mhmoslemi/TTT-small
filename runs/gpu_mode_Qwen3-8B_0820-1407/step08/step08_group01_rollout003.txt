import torch
import triton
import triton.language as tl
from torch import einsum

def custom_kernel(data):
    """
    Implements the outgoing version of the TriMul operator as described in the AlphaFold3 paper.
    The kernel computes an efficient, fused version of the TriMul operation using Triton.

    Key Features:
    - Uses Triton for optimized tensor operations.
    - Applies projections, gate activations, and einsum-style matrix multiplication.
    - Applies a mask to ensure only allowed indices are considered.
    - Uses mixed precision to optimize performance.
    - Output is cast back to float32.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict, config: Dict)

    Returns:
        output: Tensor of shape [bs, seq_len, seq_len, dim]
    """

    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Load configuration
    block_m = 128
    block_n = 128
    block_k = 128

    # Normalize input
    norm_weight = norm_weight.to(input_tensor.device)
    norm_bias = norm_bias.to(input_tensor.device)
    x_norm = (input_tensor * norm_weight) + norm_bias

    # Function to project and apply mask
    def project_and_mask(weight, input):
        weight = weight.to(input.device)
        proj = einsum("b i j d, b j d -> b i d", input, weight)
        mask = mask.unsqueeze(-1)
        return proj * mask

    # Projection operations
    left = project_and_mask(left_proj_weight, x_norm)
    right = project_and_mask(right_proj_weight, x_norm)

    # Apply gate activations
    left_gate = einsum("b i j d, b j d -> b i d", x_norm, left_gate_weight)
    left_gate = left_gate.sigmoid()
    left = left * left_gate

    right_gate = einsum("b i j d, b j d -> b i d", x_norm, right_gate_weight)
    right_gate = right_gate.sigmoid()
    right = right * right_gate

    # Apply out gate to the output
    out_gate = einsum("b i j d, b j d -> b i d", x_norm, out_gate_weight)
    out_gate = out_gate.sigmoid()

    # Define Triton kernel
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, output_ptr,
        batch_size, seq_len, dim,
        BLOCK: tl.constexpr,
    ):
        pid = tl.program_id(0)
        block_idx = pid

        # Compute the offset in the output
        out_row = block_idx * BLOCK + tl.arange(0, BLOCK)
        out_col = tl.arange(0, BLOCK)

        # Initialize the output
        out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)

        # Get the current batch index
        batch_idx = block_idx // (seq_len * seq_len)
        block_idx = block_idx % (seq_len * seq_len)

        # Compute i and j indices for the output
        i = block_idx // seq_len
        j = block_idx % seq_len

        # Loop over all k indices
        for k in range(seq_len):
            # Load left and right data
            left = tl.load(left_ptr + (i * seq_len + k) * dim + tl.arange(0, BLOCK), mask=(i * seq_len + k) < seq_len * seq_len)
            right = tl.load(right_ptr + (j * seq_len + k) * dim + tl.arange(0, BLOCK), mask=(j * seq_len + k) < seq_len * seq_len)

            # Compute dot product
            dot = tl.dot(left, right)

            # Add to output
            out += dot

        # Store the output
        tl.store(output_ptr + (i * seq_len + j) * dim + tl.arange(0, BLOCK), out)

    # Launch Triton kernel
    output_shape = (batch_size, seq_len, seq_len, dim)
    output = torch.zeros(output_shape, device=input_tensor.device, dtype=torch.float32)

    # Define kernel parameters
    grid = (triton.cdiv(seq_len * seq_len, block_m),)
    tri_mul_kernel[grid](
        left, right, output,
        batch_size, seq_len, dim,
        BLOCK=block_m,
    )

    # Apply norms and gate
    output = einsum("b i j d, b d -> b i j d", output, to_out_norm_weight)
    output = output + to_out_norm_bias
    output = output * out_gate

    # Apply final projection
    output = einsum("b i j d, b d -> b i j d", output, to_out_weight)

    return output
