    # We'll use the same offset for all output indices (i, j)
    # We'll iterate over k for the inner dimension
    offset = pid * BLOCK
    i = tl.arange(0, BLOCK)
    j = tl.arange(0, BLOCK)

    # Load the mask
    mask = tl.load(mask_ptr + offset + i[:, None] * seq_len + j[None, :], mask=(i < seq_len) & (j < seq_len), other=0.0)

    i_idx = i < seq_len
    j_idx = j < seq_len
    i = i[i_idx]
    j = j[j_idx]

    # Then, we transpose them to (hidden_dim, dim)

    # We'll iterate over k in the sequence and multiply the projections
    # Then apply the gate functions and accumulate the result

        # Load the x values for this k
        # This will depend on how the input is stored and how we access it
        # We'll make the following assumption for this implementation:
        # x is stored in (bs, seq_len, seq_len, dim), and we access it as (i, k), (j, k)

        # Load left and right values
        # This is a simplified version and may need more precise access depending on the actual structure
        # It will be adjusted based on the actual data format

        # right = x[j, k, :, :] * mask[j, k]

        x_j = tl.load(x_ptr + (offset + j[:, None] * seq_len + k[None, :]) * dim + tl.arange(0, dim)[None, :], 
                      shape=(j_len, dim))

        # Apply the mask
        mask_i = tl.load(mask_ptr + offset + i[:, None] * seq_len + k[None, :],

        mask_j = tl.load(mask_ptr + offset + j[:, None] * seq_len + k[None, :],
