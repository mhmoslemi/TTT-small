import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Convert input_tensor and mask to device
    device = input_tensor.device
    # Convert weights to device and correct data type
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # Convert mask to device
    mask = mask.to(device)

    # Apply LayerNorm (output shape is same as input)
    x = input_tensor
    x = x.to(dtype=tl.float32)

    # Apply LayerNorm: x = (x - mean) / sqrt(var + eps) * weight + bias
    # We compute it manually because we cannot use torch.nn.LayerNorm in Triton
    # Here we use the same logic as PyTorch's LayerNorm
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, keepdim=True, unbiased=False)
    x = (x - mean) * tl.rsqrt(var + 1e-5)
    x = x * norm_weight + norm_bias

    # Perform matrix multiplication for left and right projections
    left = x @ left_proj_weight
    right = x @ right_proj_weight

    # Apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
        left *= mask
        right *= mask

    # Compute left_gate, right_gate, and out_gate
    left_gate = x @ left_gate_weight
    right_gate = x @ right_gate_weight
    out_gate = x @ out_gate_weight

    left_gate = tl.sigmoid(left_gate)
    right_gate = tl.sigmoid(right_gate)
    out_gate = tl.sigmoid(out_gate)

    # Apply gates and convert to bfloat16 for better memory efficiency
    left = left * left_gate
    right = right * right_gate

    # Convert left and right to bfloat16 for better memory efficiency
    left = left.to(tl.bfloat16)
    right = right.to(tl.bfloat16)

    # Perform the einsum-like operation (i, k, d) * (j, k, d) -> (i, j, d)
    # This is equivalent to computing (left @ right.T) where each row is sum over k
    # We use the "Triton-style" implementation to avoid using einsum directly
    # The output shape is [batch_size, seq_len, seq_len, hidden_dim]

    # Precompute strides
    stride_i = seq_len * seq_len * hidden_dim
    stride_j = seq_len * hidden_dim
    stride_k = hidden_dim
    stride_d = 1

    # Initialize output
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=device, dtype=tl.float32)

    # Launch the kernel for the einsum-like operation
    @triton.jit
    def tritondot_kernel(i, j, k, d, BLOCK: tl.constexpr):
        # Initialize the accumulator
        acc = tl.zeros([BLOCK], dtype=tl.float32)

        # Define the indices
        pid = tl.program_id(0)
        pid_m = pid // (seq_len * seq_len)
        pid_n = (pid // seq_len) % seq_len
        pid_k = pid % seq_len

        # Compute the offset for the current block
        offs_i = pid_m * seq_len + pid_n
        offs_j = pid_n
        offs_k = pid_k

        # Load the left and right values
        left_val = left[offs_i, offs_k, d]
        right_val = right[offs_j, offs_k, d]

        # Do the dot operation
        acc += left_val * right_val

        # Store the result
        output[offs_i, offs_j, d] = acc

    # Launch the kernel
    num_blocks = seq_len * seq_len * seq_len
    grid = (num_blocks,)
    tritondot_kernel[grid](0, 0, 0, 0, seq_len)

    # Apply LayerNorm to the output
    out = output
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, keepdim=True, unbiased=False)
    out = (out - mean) * tl.rsqrt(var + 1e-5)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply the out_gate
    out = out * out_gate

    # Final output after to_out linear transformation
    out = out @ to_out_weight

    # Ensure the final output is in float32
    return out.to(torch.float32)
