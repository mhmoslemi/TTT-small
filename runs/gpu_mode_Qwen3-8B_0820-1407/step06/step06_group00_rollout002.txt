import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Implements the 'outgoing' version of the TriMul operator using Triton for optimized computation.
    The kernel computes a pairwise product across the sequence dimension using a fusion of projections,
    gates, and dense operations, with masking applied for invalid positions.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']

    # Extract weights from the provided dictionary
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # Convert input to float32
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.unsqueeze(-1).to(torch.float32)

    # Initialize output tensors
    output = torch.zeros((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device)

    # Launch Triton kernel for TriMul operation
    num_warps = 8
    BLOCK = 128
    num_blocks = triton.cdiv(seq_len, BLOCK)

    # Triton kernel for TriMul
    @triton.jit
    def triton_trimul_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        output_ptr, bs, seq_len, dim, hidden_dim, mask_value: tl.constexpr,
        BLOCK: tl.constexpr,
        stride_input_bs: tl.constexpr, stride_input_seq: tl.constexpr, stride_input_seq2: tl.constexpr,
        stride_input_dim: tl.constexpr,
        stride_mask_bs: tl.constexpr, stride_mask_seq: tl.constexpr, stride_mask_seq2: tl.constexpr,
        stride_left_proj_bs: tl.constexpr, stride_left_proj_seq: tl.constexpr, stride_left_proj_seq2: tl.constexpr, stride_left_proj_dim: tl.constexpr,
        stride_right_proj_bs: tl.constexpr, stride_right_proj_seq: tl.constexpr, stride_right_proj_seq2: tl.constexpr, stride_right_proj_dim: tl.constexpr,
        stride_left_gate_bs: tl.constexpr, stride_left_gate_seq: tl.constexpr, stride_left_gate_seq2: tl.constexpr, stride_left_gate_dim: tl.constexpr,
        stride_right_gate_bs: tl.constexpr, stride_right_gate_seq: tl.constexpr, stride_right_gate_seq2: tl.constexpr, stride_right_gate_dim: tl.constexpr,
        stride_out_gate_bs: tl.constexpr, stride_out_gate_seq: tl.constexpr, stride_out_gate_seq2: tl.constexpr, stride_out_gate_dim: tl.constexpr,
        stride_to_out_norm_bs: tl.constexpr, stride_to_out_norm_seq: tl.constexpr, stride_to_out_norm_seq2: tl.constexpr, stride_to_out_norm_dim: tl.constexpr,
        stride_to_out_bs: tl.constexpr, stride_to_out_seq: tl.constexpr, stride_to_out_seq2: tl.constexpr, stride_to_out_dim: tl.constexpr,
        stride_output_bs: tl.constexpr, stride_output_seq: tl.constexpr, stride_output_seq2: tl.constexpr, stride_output_dim: tl.constexpr
    ):
        # Determine the batch, sequence indices
        pid = tl.program_id(0)
        batch_idx = pid // (seq_len * seq_len)
        seq_idx = (pid // seq_len) % seq_len
        seq2_idx = pid % seq_len

        # Compute offsets in the tensor
        offset_input_bs = batch_idx * stride_input_bs
        offset_input_seq = seq_idx * stride_input_seq
        offset_input_seq2 = seq2_idx * stride_input_seq2
        offset_input = offset_input_bs + offset_input_seq + offset_input_seq2

        # Load input and compute norm
        input_val = tl.load(input_ptr + offset_input)
        norm_input = tl.dot(input_val, norm_weight) + norm_bias

        # Compute projections
        left_proj_val = tl.dot(norm_input, left_proj_weight)
        right_proj_val = tl.dot(norm_input, right_proj_weight)

        # Apply mask
        masked_left_proj_val = left_proj_val * mask
        masked_right_proj_val = right_proj_val * mask

        # Compute gates
        left_gate_val = tl.dot(norm_input, left_gate_weight).sigmoid()
        right_gate_val = tl.dot(norm_input, right_gate_weight).sigmoid()
        out_gate_val = tl.dot(norm_input, out_gate_weight).sigmoid()

        # Apply gates
        masked_left_proj_val = masked_left_proj_val * left_gate_val
        masked_right_proj_val = masked_right_proj_val * right_gate_val

        # Compute the TriMul output
        # For all k in [0, seq_len), sum over k
        for k in range(0, seq_len):
            k_offset = k * stride_input_seq2
            left_val = tl.load(input_ptr + offset_input_bs + offset_input_seq + k_offset)
            right_val = tl.load(input_ptr + offset_input_bs + k_offset + offset_input_seq2)
            # Compute element-wise product and sum
            product = left_val * right_val
            out_val = tl.add(output_ptr + offset_input_bs + offset_input_seq + offset_input_seq2, product)

        # Apply norms and out gate
        norm_output_val = tl.dot(out_val, to_out_norm_weight) + to_out_norm_bias
        norm_output_val = norm_output_val * out_gate_val
        output_val = tl.dot(norm_output_val, to_out_weight)
        tl.store(output_ptr + offset_input_bs + offset_input_seq + offset_input_seq2, output_val)

    # Configure grid for the kernel
    grid = (seq_len * seq_len, )
    triton_trimul_kernel[grid](
        input_tensor, mask, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_weight,
        output, bs, seq_len, dim, hidden_dim, mask_value=1.0,
        BLOCK=BLOCK,
        stride_input_bs=bs * seq_len * seq_len * dim,
        stride_input_seq=seq_len * seq_len * dim,
        stride_input_seq2=seq_len * dim,
        stride_input_dim=dim,
        stride_mask_bs=bs * seq_len * seq_len,
        stride_mask_seq=seq_len * seq_len,
        stride_mask_seq2=seq_len,
        stride_left_proj_bs=bs * seq_len * seq_len * hidden_dim,
        stride_left_proj_seq=seq_len * seq_len * hidden_dim,
        stride_left_proj_seq2=seq_len * hidden_dim,
        stride_left_proj_dim=hidden_dim,
        stride_right_proj_bs=bs * seq_len * seq_len * hidden_dim,
        stride_right_proj_seq=seq_len * seq_len * hidden_dim,
        stride_right_proj_seq2=seq_len * hidden_dim,
        stride_right_proj_dim=hidden_dim,
        stride_left_gate_bs=bs * seq_len * seq_len * hidden_dim,
        stride_left_gate_seq=seq_len * seq_len * hidden_dim,
        stride_left_gate_seq2=seq_len * hidden_dim,
        stride_left_gate_dim=hidden_dim,
        stride_right_gate_bs=bs * seq_len * seq_len * hidden_dim,
        stride_right_gate_seq=seq_len * seq_len * hidden_dim,
        stride_right_gate_seq2=seq_len * hidden_dim,
        stride_right_gate_dim=hidden_dim,
        stride_out_gate_bs=bs * seq_len * seq_len * hidden_dim,
        stride_out_gate_seq=seq_len * seq_len * hidden_dim,
        stride_out_gate_seq2=seq_len * hidden_dim,
        stride_out_gate_dim=hidden_dim,
        stride_to_out_norm_bs=bs * seq_len * seq_len * hidden_dim,
        stride_to_out_norm_seq=seq_len * seq_len * hidden_dim,
        stride_to_out_norm_seq2=seq_len * hidden_dim,
        stride_to_out_norm_dim=hidden_dim,
        stride_to_out_bs=bs * seq_len * seq_len * dim,
        stride_to_out_seq=seq_len * seq_len * dim,
        stride_to_out_seq2=seq_len * dim,
        stride_to_out_dim=dim,
        stride_output_bs=bs * seq_len * seq_len * hidden_dim,
        stride_output_seq=seq_len * seq_len * hidden_dim,
        stride_output_seq2=seq_len * hidden_dim,
        stride_output_dim=hidden_dim
    )

    return output
