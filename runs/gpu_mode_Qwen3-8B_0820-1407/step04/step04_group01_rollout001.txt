import torch
import triton
import triton.language as tl
from task import input_t, output_t

# --- Summary ---
# This Triton kernel implements the "outgoing" version of the TriMul module used in AlphaFold3,
# Chai, and Protenix. It performs all operations needed in the forward pass using fused and optimized
# operations, focusing on minimizing the N^3 computational complexity through tiling and blocking.
# We aim to keep the kernel as efficient as possible for large sequence lengths (up to 1024).

# --- Kernel Definition ---
@triton.jit
def trimul_kernel(
    x_ptr, mask_ptr, left_proj_weight_ptr, right_proj_weight_ptr, left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
    to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr, out_ptr,
    batch_size, seq_len, dim, hidden_dim, BLOCK: tl.constexpr,
    mask_value: tl.constexpr, use_mask: tl.constexpr, dtype: tl.constexpr
):
    # Get the current program ID and compute the block indices
    pid = tl.program_id(0)
    pid_m = pid // (seq_len // BLOCK)  # Batch and sequence axis
    pid_n = pid % (seq_len // BLOCK)

    # Compute the block offsets
    m_offsets = pid_m * seq_len + tl.arange(0, BLOCK)
    n_offsets = pid_n * seq_len + tl.arange(0, BLOCK)
    k_offsets = tl.arange(0, seq_len)

    # Load the weights
    left_proj_weights = tl.load(left_proj_weight_ptr, shape=(hidden_dim, dim))
    right_proj_weights = tl.load(right_proj_weight_ptr, shape=(hidden_dim, dim))
    left_gate_weights = tl.load(left_gate_weight_ptr, shape=(hidden_dim, dim))
    right_gate_weights = tl.load(right_gate_weight_ptr, shape=(hidden_dim, dim))
    out_gate_weights = tl.load(out_gate_weight_ptr, shape=(hidden_dim, dim))
    to_out_norm_weights = tl.load(to_out_norm_weight_ptr, shape=(hidden_dim, hidden_dim))
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr, shape=(hidden_dim,))
    to_out_weights = tl.load(to_out_weight_ptr, shape=(dim, hidden_dim))

    # Allocate output buffer
    out = tl.zeros((BLOCK, BLOCK, hidden_dim), dtype=dtype)

    for k in range(0, seq_len, BLOCK):
        # Load weights
        left_proj_weights_k = left_proj_weights[k_offsets, :]
        right_proj_weights_k = right_proj_weights[k_offsets, :]

        # Load input data
        x = tl.load(x_ptr + m_offsets[:, None, None] * (seq_len * dim) + k_offsets[None, :, None] * dim + n_offsets[None, None, :], 
                    shape=(BLOCK, BLOCK, seq_len, dim), 
                    dtype=dtype)
        x = x.reshape((BLOCK, BLOCK, seq_len, dim))

        # Apply mask
        if use_mask:
            mask = tl.load(mask_ptr + m_offsets[:, None, None] * (seq_len * seq_len) + n_offsets[None, None, :], 
                          shape=(BLOCK, BLOCK, seq_len), 
                          dtype=tl.int32)
            x = x * (mask > 0).astype(dtype)

        # Project left and right
        left = tl.dot(x, left_proj_weights_k)
        right = tl.dot(x, right_proj_weights_k)

        # Apply gate functions
        left_gate = tl.dot(x, left_gate_weights)
        right_gate = tl.dot(x, right_gate_weights)
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Perform TriMul
        for i in range(BLOCK):
            for j in range(BLOCK):
                for d in range(hidden_dim):
                    out[i, j, d] = 0.0
                    for k_idx in range(seq_len):
                        out[i, j, d] += left[i, k_idx, d] * right[j, k_idx, d]

    # Apply normalization and gate
    out = tl.dot(out, to_out_norm_weights) + to_out_norm_bias
    out_gate = tl.dot(x, out_gate_weights)
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))
    out = out * out_gate

    # Store output
    out = out.reshape((BLOCK, BLOCK, hidden_dim))
    tl.store(out_ptr + m_offsets[:, None, None] * (seq_len * hidden_dim) + n_offsets[None, None, :], out)

def custom_kernel(data: input_t) -> output_t:
    """
    Triton implementation of the TriMul module for the "outgoing" version used in AlphaFold3,
    Chai, and Protenix. This kernel supports large sequence lengths and efficient blocking.
    """
    input_tensor, mask, weights, config = data
    batch_size = input_tensor.size(0)
    seq_len = input_tensor.size(1)
    dim = input_tensor.size(3)
    hidden_dim = config["hidden_dim"]
    dtype = input_tensor.dtype

    # Extract weights from the input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate output buffer
    output = torch.empty_like(input_tensor)

    # Launch the kernel
    block_size = 128
    grid = (triton.cdiv(seq_len, block_size),)
    trimul_kernel[grid](
        input_tensor, mask, left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight, output,
        batch_size, seq_len, dim, hidden_dim, block_size,
        mask_value=0, use_mask=bool(mask is not None), dtype=dtype
    )

    return output
