import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # Apply LayerNorm
    input_tensor = torch.nn.functional.layer_norm(
        input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias
    )

    # Compute projections
    left = torch.nn.functional.linear(input_tensor, left_proj_weight)
    right = torch.nn.functional.linear(input_tensor, right_proj_weight)

    # Apply mask
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate = torch.nn.functional.linear(input_tensor, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(input_tensor, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(input_tensor, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Perform TriMul
    # This is the main operation: out[i, j] = sum_k left[i, k] * right[j, k]
    # We will implement this as a Triton kernel

    # Prepare for kernel
    # Input shape: [bs, seq_len, seq_len, hidden_dim]
    # Output shape: [bs, seq_len, seq_len, hidden_dim]

    # Define kernel
    @triton.jit
    def tri_mul_kernel(out_ptr, left_ptr, right_ptr, mask_ptr, seq_len: tl.constexpr, hidden_dim: tl.constexpr, bs: tl.constexpr):
        # Define block size
        BLOCK = 128

        # Compute the index in the batch
        pid = tl.program_id(0)
        # Compute the index in the output
        idx = tl.program_id(1)
        # Compute the index in the hidden dimension
        hid = tl.program_id(2)

        # Offset for the current block
        off_idx = idx * BLOCK + tl.arange(0, BLOCK)
        off_hid = hid * hidden_dim + tl.arange(0, hidden_dim)

        # Mask
        mask = tl.load(mask_ptr + off_idx, mask=(off_idx < seq_len), other=0.0)

        # Load left and right tensors
        left = tl.load(left_ptr + off_idx[:, None, None] * seq_len * hidden_dim + off_hid[None, :, None], mask=(off_idx[:, None, None] < seq_len) & (off_hid[None, :, None] < hidden_dim), other=0.0)
        right = tl.load(right_ptr + off_idx[None, :, None] * seq_len * hidden_dim + off_hid[None, None, :], mask=(off_idx[None, :, None] < seq_len) & (off_hid[None, None, :] < hidden_dim), other=0.0)

        # Compute the output
        acc = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)
        for k in range(seq_len):
            a = left[:, :, k]
            b = right[k, :, :]
            acc += tl.dot(a, b)

        # Apply mask
        acc = acc * mask

        # Store the result
        tl.store(out_ptr + off_idx[:, None] * seq_len * hidden_dim + off_hid[None, :], acc, mask=(off_idx[:, None] < seq_len) & (off_hid[None, :] < hidden_dim))

    # Allocate output tensor
    out = torch.zeros((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=input_tensor.dtype)

    # Launch kernel
    grid = (bs, seq_len, hidden_dim)
    tri_mul_kernel[grid](out, left, right, mask, seq_len, hidden_dim, bs)

    # Apply LayerNorm and gate
    out = torch.nn.functional.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate

    # Final linear transformation
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
