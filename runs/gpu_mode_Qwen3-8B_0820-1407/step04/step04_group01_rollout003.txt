import torch
import triton
import triton.language as tl
from triton import language as tl

# Define the TriMul custom kernel for the outgoing version of the TriMul operator
@triton.jit
def tri_mul_kernel(
    input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr, 
    to_out_norm_ptr, to_out_ptr, 
    out_ptr,
    batch_size, seq_len, dim, hidden_dim,
    BLOCK: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute the 3D grid for batch, sequence, sequence
    # Use 1D grid for the sequence, sequence, and dim
    pid = tl.program_id(0)
    n_seq = seq_len
    n_dim = dim
    # Calculate the number of blocks per dimension
    num_block_seq = tl.cdiv(n_seq, BLOCK)
    num_block_dim = tl.cdiv(n_dim, BLOCK)
    # Calculate the local sequence and dim indices
    seq_idx = pid % n_seq
    dim_idx = pid // n_seq
    # Ensure the current block is within the valid range
    if seq_idx >= n_seq or dim_idx >= n_dim:
        return
    # Load the mask for the current position
    mask = tl.load(mask_ptr + seq_idx + n_seq * dim_idx, mask=seq_idx < n_seq and dim_idx < n_dim, other=0.0)
    # Load the input tensor
    input_val = tl.load(input_ptr + seq_idx + n_seq * dim_idx, mask=seq_idx < n_seq and dim_idx < n_dim, other=0.0)
    # Initialize output tensor to zero
    out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)
    # Iterate over all k in the sequence dimension
    for k in range(seq_len):
        # Load left and right projections
        left_val = tl.load(left_proj_ptr + seq_idx + n_seq * k + seq_len * n_seq * dim_idx, mask=seq_idx < n_seq and k < seq_len and dim_idx < n_dim, other=0.0)
        right_val = tl.load(right_proj_ptr + seq_idx + n_seq * k + seq_len * n_seq * dim_idx, mask=seq_idx < n_seq and k < seq_len and dim_idx < n_dim, other=0.0)
        # Apply mask to left and right projections
        left_val = left_val * mask
        right_val = right_val * mask
        # Compute the product of left and right projections
        product = tl.dot(left_val, right_val)
        # Accumulate the product to the out tensor
        out += product
    # Apply the gate functions
    left_gate = tl.load(left_gate_ptr + seq_idx + n_seq * dim_idx)
    right_gate = tl.load(right_gate_ptr + seq_idx + n_seq * dim_idx)
    out_gate = tl.load(out_gate_ptr + seq_idx + n_seq * dim_idx)
    # Apply the gate operations
    out = out * left_gate
    out = out * right_gate
    out = out * out_gate
    # Apply the normalization and final linear transformation
    norm_val = tl.load(to_out_norm_ptr + seq_idx + n_seq * dim_idx)
    out = out * norm_val
    out = tl.dot(out, to_out_ptr)
    # Store the result in the output tensor
    tl.store(out_ptr + seq_idx + n_seq * dim_idx, out)

def custom_kernel(data):
    """
    Custom kernel for TriMul operation in the outgoing version used by AlphaFold3.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, seq_len, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Prepare inputs and outputs
    out = torch.zeros_like(input_tensor)

    # Define the block sizes
    BLOCK = 128
    BLOCK_K = BLOCK

    # Define the Triton kernel launch configuration
    # 1D grid, where each element is a (seq, seq) pair in the same position
    # and dim.
    grid = (bs * seq_len * seq_len * dim,)

    # Launch the kernel
    tri_mul_kernel[grid](
        input_tensor, mask, weights["left_proj.weight"], weights["right_proj.weight"], 
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"],
        weights["to_out_norm.weight"], weights["to_out.weight"],
        out,
        bs, seq_len, dim, hidden_dim,
        BLOCK=BLOCK, BLOCK_K=BLOCK_K,
    )

    return out
