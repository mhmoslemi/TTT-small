def TriMulKernel(
    input_ptr, mask_ptr, output_ptr,
    batch_size, seq_len, dim, hidden_dim,
    BLOCK: tl.constexpr
):

    pid = tl.program_id(0)
    i = pid // (seq_len * seq_len // BLOCK)
    j = pid % (seq_len * seq_len // BLOCK)

    # Compute the actual indices in the batch
    b = i // (seq_len * seq_len // BLOCK)
    i = i % (seq_len * seq_len // BLOCK)
    j = j % (seq_len * seq_len // BLOCK)

    # Initialize the output for this block
    out = tl.zeros((BLOCK, ), dtype=tl.float32)

    # Iterate over the diagonal elements
    for k in range(seq_len):
        # Load input for left and right projections
        left = tl.load(input_ptr + b * seq_len * seq_len * dim + k * seq_len * dim + i * dim,

                       other=0.0)

                        other=0.0)

        product = left * right

    # Store the result
    tl.store(output_ptr + b * seq_len * seq_len * dim + i * seq_len * dim + j * dim,
