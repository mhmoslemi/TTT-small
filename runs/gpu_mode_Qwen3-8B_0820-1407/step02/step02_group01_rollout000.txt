import triton
import triton.language as tl
import torch
import math
from typing import Tuple, Dict, Any

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton implementation of the TriMul operator from AlphaFold3, optimized for large sequence lengths.

    This kernel computes the TriMul operation by applying several linear projections, gates, and pairwise multiplicative updates
    efficiently using Triton's tiling and parallelism features. It is designed to handle 4D tensor operations of
    [bs, seq_len, seq_len, dim] shape and includes support for optional masks.

    Key features:
    - Efficient tiling for large sequence lengths
    - Support for optional masking
    - Mixed precision computations for performance
    - Ensures final output is in float32
    - Designed for NVIDIA L40S (Ada Lovelace) with tensor cores

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)

    Returns:
        output: [bs, seq_len, seq_len, dim] tensor with TriMul results in float32
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Configuration for the kernel
    BLOCK = 64  # Block size for tiling
    NUM_BLOCK_M = triton.cdiv(seq_len, BLOCK)
    NUM_BLOCK_N = triton.cdiv(seq_len, BLOCK)
    NUM_BLOCK_K = triton.cdiv(seq_len, BLOCK)

    # Extract weights from the dictionary
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)

    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Create output tensor
    output = torch.empty((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=torch.float32)

    # Launch kernel for each batch
    for batch in range(bs):
        # Launch kernel for this batch
        kernel[triton.cdiv(seq_len, BLOCK), num_blocks=(NUM_BLOCK_M, NUM_BLOCK_N, NUM_BLOCK_K)](input_tensor[batch], mask[batch], 
                                                                                              left_proj_weight, right_proj_weight, left_gate_weight, 
                                                                                              right_gate_weight, out_gate_weight, to_out_norm_weight, 
                                                                                              to_out_norm_bias, to_out_weight, norm_weight, norm_bias,
                                                                                              output[batch], seq_len, BLOCK)

    return output

@triton.jit
def kernel(
    input_ptr, mask_ptr, 
    left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight, 
    to_out_norm_weight, to_out_norm_bias, to_out_weight, norm_weight, norm_bias,
    output_ptr, seq_len, BLOCK: tl.constexpr
):
    # Indexing
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # Compute the range for the current block
    offset_m = pid_m * BLOCK + tl.arange(0, BLOCK)
    offset_n = pid_n * BLOCK + tl.arange(0, BLOCK)
    offset_k = tl.arange(0, seq_len)

    # Load the input tensor
    input = tl.load(input_ptr + offset_m[:, None] * seq_len * seq_len * hidden_dim + offset_k[None, :] * seq_len * hidden_dim + offset_n[None, :] * hidden_dim, 
                   mask=(offset_m < seq_len) & (offset_n < seq_len) & (offset_k < seq_len), 
                   other=0.0)
    
    # Apply layer norm
    mean = tl.sum(input, axis=[1, 2, 3]) / (seq_len * seq_len * hidden_dim)
    var = tl.sum((input - mean[None, None, None, :]) ** 2, axis=[1, 2, 3]) / (seq_len * seq_len * hidden_dim)
    x_norm = (input - mean[None, None, None, :]) / tl.sqrt(var[None, None, None, :] + 1e-5)
    x_norm = x_norm * norm_weight + norm_bias
    
    # Apply left and right projections
    left = tl.dot(x_norm, left_proj_weight.T, axis=[2, 0])
    right = tl.dot(x_norm, right_proj_weight.T, axis=[2, 0])
    
    # Apply gate operations
    left_gate = tl.dot(x_norm, left_gate_weight.T, axis=[2, 0]).sigmoid()
    right_gate = tl.dot(x_norm, right_gate_weight.T, axis=[2, 0]).sigmoid()
    out_gate = tl.dot(x_norm, out_gate_weight.T, axis=[2, 0]).sigmoid()
    
    # Apply masks
    mask = tl.load(mask_ptr + offset_m[:, None] * seq_len + offset_n[None, :], mask=(offset_m < seq_len) & (offset_n < seq_len), other=0.0)
    left = left * mask[:, :, None]
    right = right * mask[:, :, None]
    
    # Apply gate operations
    left = left * left_gate
    right = right * right_gate

    # Compute the TriMul operation
    out = tl.dot(left, right, axis=[2, 1])  # (i, j, d) = sum_k (left[i, k, d] * right[j, k, d])
    
    # Apply the output gate and normalization
    out = out * out_gate
    out = out * to_out_norm_weight + to_out_norm_bias
    out = tl.dot(out, to_out_weight.T, axis=[2, 0])

    # Store the result
    tl.store(output_ptr + offset_m[:, None] * seq_len * hidden_dim + offset_n[None, :] * hidden_dim, out, mask=(offset_m < seq_len) & (offset_n < seq_len))
