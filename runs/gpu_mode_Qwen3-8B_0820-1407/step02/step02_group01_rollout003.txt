import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict

def custom_kernel(data):
    """
    Implementation of the TriMul module as a Triton kernel. The TriMul operation
    is implemented as a tensor contraction using a tiling strategy to optimize
    performance on NVIDIA L40S GPUs.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --- Extract weights ---
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --- Device and dtype ---
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --- Precompute dimensions ---
    n = seq_len
    b = bs
    d = dim
    h = hidden_dim

    # --- Allocate output tensor ---
    out = torch.empty((b, n, n, h), device=device, dtype=dtype)

    # --- Define the Triton kernel ---
    @triton.jit
    def triton_kernel(
        x_input_ptr, x_mask_ptr,
        left_proj_weight_ptr, right_proj_weight_ptr,
        left_gate_weight_ptr, right_gate_weight_ptr,
        out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr,
        to_out_weight_ptr,
        out_ptr,
        b: tl.constexpr, n: tl.constexpr, h: tl.constexpr, d: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Compute the position in the batch, output sequence length, and channel
        pid = tl.program_id(0)
        pid_b = pid // (n * n)
        pid_n = (pid % (n * n)) // n
        pid_m = (pid % (n * n)) % n

        # Load the mask
        mask = tl.load(x_mask_ptr + pid_b * n * n + pid_n * n + pid_m, dtype=tl.float32)

        # Initialize output
        out = tl.zeros((n, n, h), dtype=tl.float32)

        # Load the weights for this batch
        for k in range(tl.cdiv(d, BLOCK)):
            # Load left projection weights
            left_proj_weights = tl.load(
                left_proj_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )

            # Load right projection weights
            right_proj_weights = tl.load(
                right_proj_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )

            # Compute left and right projections
            left = tl.dot(x_input_ptr + pid_b * n * n * d + pid_n * n * d + k * d, left_proj_weights)
            right = tl.dot(x_input_ptr + pid_b * n * n * d + pid_m * n * d + k * d, right_proj_weights)

            # Apply gates
            left_gate_weights = tl.load(
                left_gate_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )
            right_gate_weights = tl.load(
                right_gate_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )
            out_gate_weights = tl.load(
                out_gate_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )

            left_gate = tl.sigmoid(left_gate_weights)
            right_gate = tl.sigmoid(right_gate_weights)
            out_gate = tl.sigmoid(out_gate_weights)

            # Apply gates to left and right
            left = left * left_gate
            right = right * right_gate

            # Compute the output using the einsum-like operation
            for i in range(tl.cdiv(n, BLOCK)):
                for j in range(tl.cdiv(n, BLOCK)):
                    # Compute the product of left and right across k
                    for k in range(BLOCK):
                        tmp = tl.load(x_input_ptr + pid_b * n * n * d + pid_n * n * d + k * d, dtype=tl.float32)
                        tmp = tmp * left_gate_weights[k]
                        tmp = tmp * right_gate_weights[k]
                        tmp = tmp * mask

                        # Accumulate to output
                        out[i][j][k] += tmp

            # Apply the output gate
            out = out * out_gate

            # Apply the output normalization
            out_norm_weights = tl.load(
                to_out_norm_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )
            out_norm_bias = tl.load(
                to_out_norm_bias_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )
            out = out * out_norm_weights + out_norm_bias

            # Apply the final output weights
            final_weights = tl.load(
                to_out_weight_ptr + k * h * d + tl.arange(0, h) * d + tl.arange(0, BLOCK),
                mask=tl.arange(0, h) < h,
                other=0.0
            )
            out = out * final_weights

            # Store the result in the output tensor
            out_ptr[pid_b * n * n * h + pid_n * n * h + pid_m * h + k] = out

    # --- Launch the Triton kernel ---
    # Define the BLOCK size based on the configuration
    BLOCK = 128  # Adjust based on your testing for optimal performance

    # Get the input and mask pointers
    x_input_ptr = torch.tensor(input_tensor.view(-1), device=device, dtype=torch.int32)
    x_mask_ptr = torch.tensor(mask.view(-1), device=device, dtype=torch.int32)

    # Get the weight pointers
    left_proj_weight_ptr = torch.tensor(left_proj_weight.view(-1), device=device, dtype=torch.int32)
    right_proj_weight_ptr = torch.tensor(right_proj_weight.view(-1), device=device, dtype=torch.int32)
    left_gate_weight_ptr = torch.tensor(left_gate_weight.view(-1), device=device, dtype=torch.int32)
    right_gate_weight_ptr = torch.tensor(right_gate_weight.view(-1), device=device, dtype=torch.int32)
    out_gate_weight_ptr = torch.tensor(out_gate_weight.view(-1), device=device, dtype=torch.int32)
    to_out_norm_weight_ptr = torch.tensor(to_out_norm_weight.view(-1), device=device, dtype=torch.int32)
    to_out_norm_bias_ptr = torch.tensor(to_out_norm_bias.view(-1), device=device, dtype=torch.int32)
    to_out_weight_ptr = torch.tensor(to_out_weight.view(-1), device=device, dtype=torch.int32)

    # Launch the kernel
    grid = (b * n * n, )
    triton_kernel[grid](
        x_input_ptr, x_mask_ptr,
        left_proj_weight_ptr, right_proj_weight_ptr,
        left_gate_weight_ptr, right_gate_weight_ptr,
        out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr,
        to_out_weight_ptr,
        out,
        b, n, h, d,
        BLOCK=BLOCK
    )

    return out
