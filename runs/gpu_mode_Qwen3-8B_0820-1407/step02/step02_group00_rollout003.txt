import torch
import triton
import triton.language as tl
import math
from torch import Tensor

def custom_kernel(data: tuple) -> Tensor:
    """
    Triton implementation of the Triangle Multiplicative Update (TriMul) operator.
    This kernel computes the outgoing version of the TriMul operation over a 4D tensor.
    The implementation supports various model configurations such as different sequence lengths,
    batch sizes, and hidden dimensions. The kernel is optimized for large sequence dimensions
    and leverages Triton's parallelism and memory management capabilities for performance.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim] computed using the TriMul operator
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights and load them into device memory
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Move weights to device
    norm_weight = norm_weight.to(input_tensor.device)
    norm_bias = norm_bias.to(input_tensor.device)
    left_proj_weight = left_proj_weight.to(input_tensor.device)
    right_proj_weight = right_proj_weight.to(input_tensor.device)
    left_gate_weight = left_gate_weight.to(input_tensor.device)
    right_gate_weight = right_gate_weight.to(input_tensor.device)
    out_gate_weight = out_gate_weight.to(input_tensor.device)
    to_out_norm_weight = to_out_norm_weight.to(input_tensor.device)
    to_out_norm_bias = to_out_norm_bias.to(input_tensor.device)
    to_out_weight = to_out_weight.to(input_tensor.device)

    # Compute the norm (LayerNorm)
    x = input_tensor.to(torch.float32)
    x_norm = x * norm_weight + norm_bias

    # Compute left and right projections
    left = x_norm @ left_proj_weight.transpose(0, 1)
    right = x_norm @ right_proj_weight.transpose(0, 1)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    # Apply mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    # Compute gate activations
    left_gate = x_norm @ left_gate_weight.transpose(0, 1)
    right_gate = x_norm @ right_gate_weight.transpose(0, 1)
    out_gate = x_norm @ out_gate_weight.transpose(0, 1)

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply gate activations
    left = left * left_gate
    right = right * right_gate

    # Compute the TriMul operation using Triton kernel

    # Define kernel
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seq_len, dim,
        BLOCK: tl.constexpr
    ):
        # Create block index
        pid = tl.program_id(0)
        batch_idx = pid // (seq_len * seq_len)
        seq_idx = (pid // seq_len) % seq_len
        seq_idx2 = pid % seq_len

        off = tl.arange(0, BLOCK)

        # Load the left and right tensors
        left = tl.load(left_ptr + batch_idx * seq_len * seq_len * dim + seq_idx * seq_len * dim + off, mask=off < dim)
        right = tl.load(right_ptr + batch_idx * seq_len * seq_len * dim + seq_idx2 * seq_len * dim + off, mask=off < dim)

        # Compute the dot product
        out = tl.dot(left, right)

        # Store the output
        tl.store(out_ptr + batch_idx * seq_len * seq_len * dim + seq_idx * seq_len * dim + seq_idx2 * dim + off, out, mask=off < dim)

    # Allocate output tensor
    out = torch.empty((batch_size, seq_len, seq_len, dim), dtype=torch.float32, device=input_tensor.device)

    # Launch Triton kernel
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)
    triton_trimul_kernel[grid](
        left, right, out,
        batch_size, seq_len, dim,
        BLOCK=128
    )

    # Apply the output gate and the final linear layer
    out = out * out_gate
    out = out @ to_out_weight.transpose(0, 1)
    out = out @ to_out_norm_weight.transpose(0, 1) + to_out_norm_bias

    # Ensure output is in float32
    out = out.to(torch.float32)

    return out
