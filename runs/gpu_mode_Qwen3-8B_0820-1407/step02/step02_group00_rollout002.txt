import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton kernel implementation of the TriMul operator.
    This kernel computes the TriMul operation using efficient tiling and memory access patterns.
    It supports different input sizes and configurations as specified in the problem description.
    """
    # Extract input and parameters
    input_tensor, mask, weights, config = data
    bs, seqlen, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Retrieve weights from the provided dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Output tensor
    output = torch.zeros((bs, seqlen, seqlen, dim), device=input_tensor.device)

    # Define the Triton kernel
    @triton.jit
    def tri_mul_kernel(
        x_ptr, y_ptr, mask_ptr, z_ptr,
        dim, hidden_dim,
        seqlen, bs,
        n_block: tl.constexpr,
        m_block: tl.constexpr,
        k_block: tl.constexpr,
    ):
        # Compute the block offsets
        pid = tl.program_id(axis=0)
        pid_m = tl.program_id(axis=1)
        pid_n = tl.program_id(axis=2)
        pid_k = tl.program_id(axis=3)

        # Compute the block indices
        block_m = pid_m * m_block + tl.arange(0, m_block)
        block_n = pid_n * n_block + tl.arange(0, n_block)
        block_k = pid_k * k_block + tl.arange(0, k_block)

        # Compute the global indices
        m_offsets = block_m + tl.arange(0, n_block)
        n_offsets = block_n + tl.arange(0, m_block)
        k_offsets = block_k + tl.arange(0, k_block)

        # Load the x and y data
        x = tl.load(x_ptr + m_offsets[:, None] * seqlen * dim + k_offsets[None, :] * dim, mask=block_m < seqlen, other=0.0)
        y = tl.load(y_ptr + k_offsets[:, None] * seqlen * dim + n_offsets[None, :] * dim, mask=block_k < seqlen, other=0.0)

        # Apply the mask
        mask = tl.load(mask_ptr + m_offsets[:, None] * seqlen + n_offsets[None, :], mask=block_m < seqlen, other=0.0)

        # Apply the mask to x and y
        x = x * mask
        y = y * mask

        # Compute the dot product
        acc = tl.zeros((m_block, n_block), dtype=tl.float32)
        for k in range(k_block, seqlen, k_block):
            x_k = tl.load(x_ptr + m_offsets[:, None] * seqlen * dim + k * dim + tl.arange(0, m_block) * 1, mask=block_m < seqlen, other=0.0)
            y_k = tl.load(y_ptr + k * seqlen * dim + n_offsets[None, :] * dim + tl.arange(0, n_block) * 1, mask=block_k < seqlen, other=0.0)
            acc += tl.dot(x_k, y_k)

        # Store the result
        tl.store(z_ptr + m_offsets[:, None] * seqlen * dim + n_offsets[None, :], acc, mask=block_m < seqlen)

    # Launch the kernel
    grid = (triton.cdiv(seqlen, n_block), triton.cdiv(seqlen, m_block), triton.cdiv(seqlen, k_block), triton.cdiv(seqlen, k_block))
    tri_mul_kernel[grid](
        input_tensor, input_tensor, mask,
        output,
        dim, hidden_dim,
        seqlen, bs,
        n_block=64,
        m_block=64,
        k_block=64
    )

    # Apply the to_out_norm and to_out operations
    output = output.to(torch.float32)
    output = output.to(out_gate_weight.device)
    output = torch.nn.functional.linear(output, to_out_norm_weight, to_out_norm_bias)
    output = torch.nn.functional.linear(output, to_out_weight)
    output = output.to(torch.float32)

    return output
