import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Convert mask to float for element-wise multiplication
    mask = mask.unsqueeze(-1).to(input_tensor.dtype)

    # Define constants for the kernel
    SEQLEN = seq_len
    BS = batch_size
    DIM = dim
    HIDDEN_DIM = hidden_dim
    BLOCK = 128  # Block size for tiling
    MAX_BLOCKS = 128  # Max number of blocks to process in a single kernel
    
    # Load weights directly into Triton buffers
    norm_weight = weights["norm.weight"].to(input_tensor.device)
    norm_bias = weights["norm.bias"].to(input_tensor.device)
    left_proj_weight = weights["left_proj.weight"].to(input_tensor.device)
    right_proj_weight = weights["right_proj.weight"].to(input_tensor.device)
    left_gate_weight = weights["left_gate.weight"].to(input_tensor.device)
    right_gate_weight = weights["right_gate.weight"].to(input_tensor.device)
    out_gate_weight = weights["out_gate.weight"].to(input_tensor.device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(input_tensor.device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(input_tensor.device)
    to_out_weight = weights["to_out.weight"].to(input_tensor.device)

    # Define Triton kernel to perform the TriMul operation
    @triton.jit
    def tritondot(
        input_ptr, mask_ptr, left_proj_w_ptr, right_proj_w_ptr, left_gate_w_ptr,
        right_gate_w_ptr, out_gate_w_ptr, to_out_norm_w_ptr, to_out_norm_b_ptr,
        to_out_w_ptr,
        output_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr,
    ):
        # Position within the batch
        b = tl.program_id(0)
        if b >= batch_size:
            return

        # Position within the block
        pid = tl.program_id(1)
        block_id = pid // (seq_len // BLOCK)
        m = pid % (seq_len // BLOCK)

        # Load input and mask
        block_start = m * BLOCK
        block_size = min(BLOCK, seq_len - block_start)
        input_block = tl.load(input_ptr + b * seq_len * seq_len * dim + block_start * seq_len * dim, shape=(block_size, seq_len, dim))
        mask_block = tl.load(mask_ptr + b * seq_len * seq_len, shape=(block_size, seq_len))

        # Normalize input
        norm_weight_block = tl.load(norm_weight + (block_start // seq_len) * dim + tl.arange(0, dim), shape=(dim,))
        norm_bias_block = tl.load(norm_bias, shape=(dim,))  # Shared across all blocks
        input_block = input_block * norm_weight_block + norm_bias_block

        # Apply projections and gates
        left_proj_weight_block = tl.load(left_proj_weight + block_start * hidden_dim + tl.arange(0, hidden_dim), shape=(hidden_dim,))
        right_proj_weight_block = tl.load(right_proj_weight + block_start * hidden_dim + tl.arange(0, hidden_dim), shape=(hidden_dim,))
        left_gate_weight_block = tl.load(left_gate_weight + block_start * hidden_dim + tl.arange(0, hidden_dim), shape=(hidden_dim,))
        right_gate_weight_block = tl.load(right_gate_weight + block_start * hidden_dim + tl.arange(0, hidden_dim), shape=(hidden_dim,))
        out_gate_weight_block = tl.load(out_gate_weight + block_start * hidden_dim + tl.arange(0, hidden_dim), shape=(hidden_dim,))

        # Compute left and right projections
        left = input_block @ left_proj_weight_block
        right = input_block @ right_proj_weight_block

        # Apply mask to left and right
        left = left * mask_block
        right = right * mask_block

        # Compute gates
        left_gate = input_block @ left_gate_weight_block
        right_gate = input_block @ right_gate_weight_block
        out_gate = input_block @ out_gate_weight_block

        # Apply sigmoid to gates
        left_gate = tl.sigmoid(left_gate)
        right_gate = tl.sigmoid(right_gate)
        out_gate = tl.sigmoid(out_gate)

        # Multiply by gates
        left = left * left_gate
        right = right * right_gate

        # Compute TriMul output
        out = tl.dot(left, right, axis=1)

        # Normalize output
        to_out_norm_weight_block = tl.load(to_out_norm_weight + (block_start // seq_len) * hidden_dim + tl.arange(0, hidden_dim), shape=(hidden_dim,))
        to_out_norm_bias_block = tl.load(to_out_norm_bias, shape=(hidden_dim,))
        out = (out * to_out_norm_weight_block) + to_out_norm_bias_block

        # Multiply by out_gate
        out = out * out_gate

        # Apply final projection
        to_out_weight_block = tl.load(to_out_weight + block_start * dim + tl.arange(0, dim), shape=(dim,))
        out = out @ to_out_weight_block

        # Store output
        out_ptr = output_ptr + b * seq_len * seq_len * dim + block_start * seq_len * dim
        tl.store(out_ptr, out)

    # Launch kernel
    grid = (batch_size, triton.cdiv(seq_len, BLOCK))
    tritondot[grid](
        input_tensor,
        mask,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device),
        batch_size,
        seq_len,
        dim,
        hidden_dim,
        BLOCK=BLOCK
    )

    return torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device)
