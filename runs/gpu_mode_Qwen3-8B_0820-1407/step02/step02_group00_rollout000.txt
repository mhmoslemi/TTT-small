import torch
import triton
import triton.language as tl
from functools import reduce
import math

def custom_kernel(data):
    """
    Triton-optimized custom kernel for the TriMul operation.
    
    This kernel performs the TriMul operation by applying projections,
    gates, and a pairwise multiplication sum over the sequence dimension.
    It is optimized for high performance on NVIDIA L40S GPUs with a focus on
    memory efficiency, fusion, and tiling strategies that reuse data through L2.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    seqlen = input_tensor.shape[1]
    batch_size = input_tensor.shape[0]

    # --- Extract weights from the config dictionary
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # --- Apply LayerNorm to input
    input_norm = input_tensor.to(torch.float32)
    input_norm = input_norm * norm_weight + norm_bias

    # --- Project input to hidden_dim
    left = input_norm @ left_proj_weight.t()  # [bs, seq_len, seq_len, hidden_dim]
    right = input_norm @ right_proj_weight.t()  # [bs, seq_len, seq_len, hidden_dim]

    # --- Apply mask
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
    left = left * mask
    right = right * mask

    # --- Gate operations
    left_gate = input_norm @ left_gate_weight.t()
    left_gate = left_gate.sigmoid()
    left = left * left_gate

    right_gate = input_norm @ right_gate_weight.t()
    right_gate = right_gate.sigmoid()
    right = right * right_gate

    out_gate = input_norm @ out_gate_weight.t()
    out_gate = out_gate.sigmoid()

    # --- Perform the TriMul operation: sum over k of left[i,k] * right[j,k]
    # This is a batched operation with dimensions [bs, seq_len, seq_len, hidden_dim]
    # The output is [bs, seq_len, seq_len, hidden_dim]

    # --- Optimized TriMul kernel
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seq_len, hidden_dim,
        stride_left_batch, stride_left_seq, stride_left_seq_k, stride_left_dim,
        stride_right_batch, stride_right_seq, stride_right_seq_k, stride_right_dim,
        stride_out_batch, stride_out_seq, stride_out_seq_k, stride_out_dim,
        BLOCK: tl.constexpr,  # Block size for the sequence dimension
        BLOCK_DIM: tl.constexpr,  # Block size for the hidden dimension
    ):
        # Compute the thread index
        pid = tl.program_id(0)
        # Grid dimension: number of batches x number of sequence pairs
        grid = tl.program_id(1)
        # Compute the batch index and sequence indices
        b = grid // (seq_len * seq_len)
        i = (grid // seq_len) % seq_len
        j = grid % seq_len

        # Compute the offsets for the local block of elements
        # These offsets are in the range of a single block
        off_k = tl.arange(0, BLOCK_DIM)
        off_m = tl.arange(0, BLOCK)
        off_n = tl.arange(0, BLOCK)

        # Compute the index in the left tensor for this block
        # left_ptr: shape [bs, seq_len, seq_len, hidden_dim]
        # (b, i, k, d)
        left_idx = ((b * stride_left_batch) + (i * stride_left_seq) + off_k * stride_left_seq_k + off_m * stride_left_dim)
        # right_ptr: shape [bs, seq_len, seq_len, hidden_dim]
        # (b, j, k, d)
        right_idx = ((b * stride_right_batch) + (j * stride_right_seq) + off_k * stride_right_seq_k + off_n * stride_right_dim)

        # Load left and right tensors
        left_block = tl.load(left_ptr + left_idx, mask=off_k < hidden_dim, other=0.0)
        right_block = tl.load(right_ptr + right_idx, mask=off_k < hidden_dim, other=0.0)

        # Compute the product
        product = tl.dot(left_block, right_block)

        # Store the result in the output tensor
        # out_idx: (b, i, j, d)
        out_idx = (b * stride_out_batch) + (i * stride_out_seq) + (j * stride_out_seq_k) + off_n * stride_out_dim
        tl.store(out_ptr + out_idx, product)

    # --- Launch the TriMul kernel
    # Prepare the strides
    stride_left_batch = left.stride(0)
    stride_left_seq = left.stride(1)
    stride_left_seq_k = left.stride(2)
    stride_left_dim = left.stride(3)

    stride_right_batch = right.stride(0)
    stride_right_seq = right.stride(1)
    stride_right_seq_k = right.stride(2)
    stride_right_dim = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_seq = out.stride(1)
    stride_out_seq_k = out.stride(2)
    stride_out_dim = out.stride(3)

    # Prepare the grid size
    grid = (batch_size * seq_len * seq_len,)

    # Use a reasonable block size for the hidden dimension and sequence dimension
    # We use 64 for both to allow good tiling and overlap of memory operations
    BLOCK_DIM = 64
    BLOCK = 128

    # Launch the kernel
    tri_mul_kernel[grid](
        left, right, out,
        batch_size, seq_len, hidden_dim,
        stride_left_batch, stride_left_seq, stride_left_seq_k, stride_left_dim,
        stride_right_batch, stride_right_seq, stride_right_seq_k, stride_right_dim,
        stride_out_batch, stride_out_seq, stride_out_seq_k, stride_out_dim,
        BLOCK, BLOCK_DIM
    )

    # --- Normalize the output
    out = out.to(torch.float32)
    out = out * to_out_norm_weight + to_out_norm_bias
    out = out * out_gate

    # --- Final projection
    output = out @ to_out_weight.t()

    return output
