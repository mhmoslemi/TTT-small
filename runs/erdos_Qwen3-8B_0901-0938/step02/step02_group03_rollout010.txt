import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    dx = 2.0 / n_points

            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Scale to satisfy the integral constraint
        h = h / np.sum(h) * (n_points / 2)

        # Define the objective function
        def objective(h_vec):
            h = h_vec.reshape(-1)
            # Compute cross-correlation
            corr = signal.correlate(h, 1 - h, mode='full')
            # Find the maximum overlap
            c5 = np.max(corr) * dx
            return c5

        # Define constraints
        def constraint(h_vec):
            return np.sum(h_vec) - (n_points / 2)

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]

        # Run optimization
        result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:

            c5 = objective(h_opt)

                best_c5 = c5
                best_h = h_opt
