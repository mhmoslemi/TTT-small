    n_points = 100  # Start with a small grid for faster computation

    def evaluate_overlap(h):
        dx = 2.0 / n_points

        h_shifted = np.roll(h_array, shift=1)
        overlap = np.correlate(h_array, 1 - h_shifted, mode="full")

        return np.sum(h) - n_points / 2
