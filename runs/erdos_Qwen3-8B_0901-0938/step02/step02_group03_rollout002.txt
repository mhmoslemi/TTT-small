import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is pre-loaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Compute the overlap integrals
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    # Start optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Check if we have enough time to do more iterations or refine
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time remains, try a global optimization to escape local minima
        result_global = optimize.differential_evolution(
            objective,
            bounds,
            constraints=cons,
            maxiter=50,
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1.0),
            recombination='blend',
            disp=False
        )
        result = result_global
    
    best_h = result.x
    best_c5 = result.fun
    n_points = len(best_h)
    
    return best_h, best_c5, n_points
