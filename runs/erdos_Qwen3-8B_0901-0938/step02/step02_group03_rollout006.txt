import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for efficiency
    
    # Try different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess
        h_initial = np.array(initial_h_values[:n_points])  # Use initial values up to n_points
        h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')  # Pad if needed
        h_initial = h_initial[:n_points]  # Ensure length matches
        
        # Define the objective function
        def objective(h):
            h = h.reshape(-1)
            dx = 2.0 / n_points
            h = np.clip(h, 0, 1)  # Ensure h is in [0, 1]
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define the constraint: integral of h must be 1
        def constraint(h):
            h = h.reshape(-1)
            return np.sum(h) - n_points / 2
        
        # Set up optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint}
        
        # Run optimization
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            h = result.x
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s:
            break
    
    # Evaluate the best solution
    if best_h is not None:
        best_h = best_h[:n_points]
        best_h = np.clip(best_h, 0, 1)
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
