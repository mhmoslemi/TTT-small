```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 samples for speed
    max_n_points = 500  # Max points to consider

    # Initial guess
    h_initial = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values
    h_initial = np.resize(h_initial, n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    def objective(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Optimization with simulated annealing
    result = optimize.minimize(
        objective,
        x0=h_initial,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-5, 'disp': False}
    )

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)

    # Try increasing n_points if time allows
    for n_points in range(150, max_n_points, 50):
        if time.time() - start_time > budget_s - 10:
            break
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)

        def objective_high_res(h):
            nonlocal best_c5
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            return c5

        result = optimize.minimize(
            objective_high_res,
            x0=h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 50, 'ftol': 1e-5, 'disp': False}
        )

        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)

    # Final check with a small perturbation
    perturbation = np.random.rand(n_points) * 0.05 - 0.025
    best_h = np.clip(best_h + perturbation, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)

    return best_h, best_c5, n_points
```
