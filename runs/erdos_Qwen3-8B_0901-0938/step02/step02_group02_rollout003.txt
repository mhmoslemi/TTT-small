import scipy.optimize as optimize
import random
from scipy.signal import correlate
from scipy.optimize import minimize
import time

evaluate_erdos_solution = lambda h: max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is provided

def run(seed=42, budget_s=600, **kwargs):

        def objective(h):
            h = np.clip(h, 0, 1)
            c5 = evaluate_erdos_solution(h)

        def constraint(h):
            return np.sum(h) - n / 2
