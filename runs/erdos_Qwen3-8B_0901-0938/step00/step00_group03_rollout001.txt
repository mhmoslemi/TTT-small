```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assume this is a pre-loaded initial solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    best_n_points = n_points
    
    # Try smaller n_points if possible
    for n_points in range(50, 1000, 50):
        if time.time() - start_time > budget_s - 10:
            break
        # Generate random initial guess
        h = np.random.rand(n_points) * 2 - 1  # Random in [-1, 1], then scale to [0, 1]
        h = (h + 1) / 2
        h = h * (n_points / 2)  # Ensure sum(h) == n_points / 2
        h = h / (n_points / 2)
        dx = 2.0 / n_points
        # Optimize using gradient descent
        for _ in range(100):
            # Compute current C5
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n_points = n_points
            # Compute gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Compute the overlap integral for each shift
                for k in range(1, n_points):
                    if i + k < n_points:
                        overlap = h[i] * (1 - h[i + k])
                        grad[i] += overlap
                    if i - k >= 0:
                        overlap = h[i] * (1 - h[i - k])
                        grad[i] += overlap
            # Update h
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            # Ensure sum(h) == n_points / 2
            h = h * (n_points / 2) / np.sum(h)
            h = h / (n_points / 2)
    
    # Final check with the best solution
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, best_n_points)
```
