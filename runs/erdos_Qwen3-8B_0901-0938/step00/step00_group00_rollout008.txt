```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start with the initial construction
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    dx_initial = 2.0 / n_points_initial
    c5_initial = evaluate_erdos_solution(initial_h, dx_initial)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = initial_h
        best_n_points = n_points_initial

    # Try increasing n_points in steps
    for n_points in [n_points_initial + 100, n_points_initial + 200, n_points_initial + 300, n_points_initial + 400]:
        if time.time() - start_time > budget_s - 10:
            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Ensure sum(h) == n_points / 2

        # Define the optimization problem
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h_var) == n_points / 2]
        objective = cp.Minimize(0)  # We'll compute the C5 value separately

        # Solve the problem using a local solver
        try:
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.SCS, max_iter=1000, eps=1e-6)
            h_opt = h_var.value
            h_opt = np.clip(h_opt, 0, 1)
            c5 = evaluate_erdos_solution(h_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points
        except:
            pass

    # Try a few random perturbations of the best solution
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        perturbation = np.random.rand(len(best_h)) * 0.1 - 0.05
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        dx = 2.0 / len(h_perturbed)
        c5 = evaluate_erdos_solution(h_perturbed, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed
            best_n_points = len(h_perturbed)

    return (best_h, best_c5, best_n_points)
```
