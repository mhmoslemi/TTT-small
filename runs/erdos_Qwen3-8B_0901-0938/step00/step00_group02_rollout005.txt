import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 45  # Start with the initial n_points
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the max overlap integral
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate and return the best solution
    def evaluate_and_return(h_values):
        c5 = compute_c5(h_values)
        return (h_values, c5, len(h_values))
    
    # Initial evaluation
    best_h = h
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    
    # Try increasing n_points if possible
    for n_points in range(45, 100):
        dx = 2.0 / n_points
        # Initialize h with the previous best
        h = np.zeros(n_points)
        # Distribute the integral to 1
        h[:int(n_points / 2)] = 1.0
        # Add some noise to avoid local optima
        h += np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points
    
    # If we still haven't improved, try optimization
    if best_c5 > 0.380876:
        # Try optimization with a smaller n_points
        n_points = 100
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:int(n_points / 2)] = 1.0
        h += np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        
        # Optimization problem: minimize max_k ∫ h(x)(1 - h(x+k)) dx
        # We'll use a heuristic optimization approach
        def objective(h_values):
            h = np.array(h_values)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        best_h = result.x
        best_c5 = compute_c5(best_h)
        best_n_points = n_points
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, best_n_points)
