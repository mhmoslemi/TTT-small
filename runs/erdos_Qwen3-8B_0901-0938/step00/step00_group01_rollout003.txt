```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Initial guess
    initial_h = initial_h_values.copy()
    initial_h = np.resize(initial_h, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.resize(h, n_points)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, np.zeros(n_points)])
        corr = signal.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    # Function to optimize h
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        obj = cp.Minimize(evaluate_c5(h))
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver='ECOS', max_iter=1000, eps=1e-6)
        if result == cp.OPTIMAL:
            return h.value
        else:
            return h_initial
    
    # Optimization loop
    for _ in range(10):  # Run multiple optimizations
        h = optimize_h(initial_h)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
        # Adjust initial guess for next iteration
        initial_h = h + np.random.normal(0, 0.05, n_points)
        initial_h = np.clip(initial_h, 0, 1)
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break
    
    # Try increasing n_points if time allows
    if elapsed < budget_s - 10:
        n_points = 200
        initial_h = initial_h_values.copy()
        initial_h = np.resize(initial_h, n_points)
        initial_h = np.clip(initial_h, 0, 1)
        h = optimize_h(initial_h)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Final check
    if best_h is None:
        best_h = initial_h
    
    return best_h, best_c5, n_points
```
