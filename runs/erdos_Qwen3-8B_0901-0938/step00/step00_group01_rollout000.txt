def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points

    h = np.clip(h, 0, 1)

    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        # Normalize to satisfy the constraint
        h = h * (n_points / 2) / sum_h
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        result = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(result) * dx
        return max_overlap
    
    # Optimization function to minimize C5
    def objective_func(h_vec):
        h = h_vec.reshape(-1)
        return compute_c5(h)

    def optimize_h(h_initial, max_iter=1000, learning_rate=0.01, tol=1e-6):
        h = h_initial.copy()
        for i in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            if current_c5 < tol:
                break
            
            # Compute gradient (approximate using finite differences)
            grad = np.zeros_like(h)

                c5_plus = compute_c5(h_plus)

                h_minus[idx] -= 1e-5
                c5_minus = compute_c5(h_minus)
                grad[idx] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update h with gradient descent
            h -= learning_rate * grad
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Enforce integral constraint
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-6:
                h = h * (n_points / 2) / sum_h
            
            # Check time budget
            if time.time() - start_time > budget_s - 10:
                break
        
        return h
    
    # Run optimization
    best_h = optimize_h(h)
    best_c5 = compute_c5(best_h)

    # Final check
    final_c5 = compute_c5(best_h)
