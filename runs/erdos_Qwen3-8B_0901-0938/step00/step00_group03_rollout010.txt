```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values.npy is provided

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for faster computation
    max_n_points = 500  # Maximum number of points to consider
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values[:n_points].copy()
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try increasing the number of points up to max_n_points
    for n_points in range(100, max_n_points + 1, 50):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate a random initial guess
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Optimization: minimize the maximum overlap integral
        # We use a convex optimization approach with a small epsilon
        # to approximate the max operation
        epsilon = 1e-3
        c = cp.Variable(n_points)
        objective = cp.Minimize(cp.Max(cp.sum(c * (1 - h[None, :] - h[None, :] + h[None, :] * h[None, :]))))
        
        # Constraints
        constraints = [
            cp.sum(c) == n_points / 2,
            0 <= c,
            c <= 1
        ]
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'OPTIMAL':
                h_opt = c.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
        except:
            pass
        
        # Check if we have a better solution
        if best_c5 < 0.380876:
            break
    
    # Final check with the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, len(best_h)
```
