```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Start with the initial n_points
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    # Check if initial solution is valid
    if np.sum(h_initial) != n_points / 2:
        h_initial = np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)])
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        if np.sum(h) != n_points / 2:
            return float('inf')
        # Compute the overlap integral
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    # Function to optimize h with constraints
    def optimize_h(h_initial, n_points, budget_s):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        obj = cp.Minimize(cp.Max(cp.multiply(cp.sum(h), dx), compute_c5(h)))
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
            if result is not None and result < float('inf'):
                return h.value
            else:
                return h_initial
        except:
            return h_initial
    
    # Optimize with initial guess
    h_opt = optimize_h(h_initial, n_points, budget_s)
    c5_bound = compute_c5(h_opt)
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 60
        dx = 2.0 / n_points
        h_initial = np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)])
        h_opt = optimize_h(h_initial, n_points, remaining_time)
        c5_bound = compute_c5(h_opt)
    
    # Final check
    if np.sum(h_opt) != n_points / 2:
        h_opt = np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)])
    
    return h_opt, c5_bound, n_points
```
