```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points for faster computation
    max_points = 500  # Maximum number of points to try
    max_iterations = 100  # Maximum number of iterations
    tolerance = 1e-4  # Tolerance for convergence
    
    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # Truncate to n_points
    h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for n_points in range(100, max_points + 1, 50):
        # Adjust h to fit new number of points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
        
        # Optimize h using gradient descent
        for _ in range(max_iterations):
            # Compute overlap integral
            c5 = evaluate_erdos_solution(h)
            
            # If we've reached the budget, break
            if time.time() - start_time > budget_s:
                break
            
            # Compute gradient of the overlap integral
            # For simplicity, use finite differences
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = h_plus / np.sum(h_plus) * (len(h_plus) / 2)
                c5_plus = evaluate_erdos_solution(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = h_minus / np.sum(h_minus) * (len(h_minus) / 2)
                c5_minus = evaluate_erdos_solution(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update h using gradient descent
            h -= 0.01 * grad
            
            # Ensure h stays within [0, 1] and integral is 1
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (len(h) / 2)
            
            # Check for convergence
            if np.abs(c5 - best_c5) < tolerance:
                break
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, len(best_h)
```
