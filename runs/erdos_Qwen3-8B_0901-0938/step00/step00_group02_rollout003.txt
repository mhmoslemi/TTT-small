import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values / (sum_h / (n_points / 2))
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # To handle circular convolution
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the overlap integral for a given h
    def objective(h):
        return evaluate_c5(h)
    
    # Function to compute the constraint sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization problem
    h = cp.Variable(n_points, bounds=(0, 1))
    objective_func = objective(h.value) if h.value is not None else np.inf
    problem = cp.Problem(cp.Minimize(objective_func), [constraint_sum(h) == 0])
    
    # Optimization loop
    best_h = initial_h_values.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    # Use a simple gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    decay = 0.999
    epsilon = 1e-8
    grad = np.zeros(n_points)
    velocity = np.zeros(n_points)
    
    for step in range(1000):
        h_values = best_h.copy()
        h_values = np.clip(h_values, 0, 1)
        c5 = evaluate_c5(h_values)
        
        # Compute gradient numerically
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_values.copy()
            h_plus[i] += 1e-6
            c5_plus = evaluate_c5(h_plus)
            h_minus = h_values.copy()
            h_minus[i] -= 1e-6
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-6)
        
        # Update velocity and weights
        velocity = momentum * velocity - learning_rate * grad
        h_values += velocity
        h_values = np.clip(h_values, 0, 1)
        
        # Check if we improved
        if evaluate_c5(h_values) < c5:
            best_h = h_values
            best_c5 = evaluate_c5(best_h)
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(final_h)
    
    return final_h, final_c5, n_points
