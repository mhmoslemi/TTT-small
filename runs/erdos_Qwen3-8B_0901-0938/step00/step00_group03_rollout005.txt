```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is provided

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    max_points = 500  # Maximum number of points to try
    best_c5 = float('inf')
    best_h = None
    
    # Try increasing number of points up to max_points
    for n_points in range(100, max_points + 1, 50):
        if time.time() - start_time > budget_s:
            break
        
        # Initialize h with a uniform distribution
        h = np.random.rand(n_points) * 2.0 - 1.0  # Random values in [-1, 1]
        h = np.clip(h, 0, 1)  # Clamp to [0, 1]
        h = h * (n_points / 2.0)  # Ensure integral is 1
        
        # Use a local optimizer to refine the solution
        def objective(x):
            h = x.reshape(n_points)
            integral = np.sum(h) * (2.0 / n_points)
            if np.abs(integral - 1.0) > 1e-6:
                return float('inf')
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(overlap)
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        # Check if the optimization was successful
        if result.success:
            h_opt = result.x
            c5 = evaluate_erdos_solution(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        
        # Try a more structured approach with a convex optimization
        # Define variables
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2.0, 0 <= h <= 1]
        objective = cp.Max(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * (2.0 / n_points)))
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver='ECOS')
        
        if prob.status == 'OPTIMAL':
            h_opt = h.value
            c5 = evaluate_erdos_solution(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
    
    # If no solution found, use the initial solution
    if best_h is None:
        best_h = initial_h_values
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, len(best_h))
```
