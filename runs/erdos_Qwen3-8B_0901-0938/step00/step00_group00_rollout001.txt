import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    max_iterations = 100
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = None
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / n_points
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i] = h[i:] if i < n_points else h[:n_points - i]
        h1_minus_h = 1.0 - h
        correlation = np.zeros(n_points)
        for i in range(n_points):
            correlation[i] = np.dot(h, h1_minus_h[i:])
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to compute the overlap integral with numpy correlate
    def compute_c5_np(h):
        h = np.array(h)
        dx = 2.0 / n_points
        h1_minus_h = 1.0 - h
        corr = signal.correlate(h, h1_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h using gradient descent
    def optimize_h(h_initial):
        h = h_initial.copy()
        for _ in range(max_iterations):
            # Compute the overlap integral
            c5 = compute_c5_np(h)
            if c5 < best_c5 - tolerance:
                best_c5 = c5
                best_h = h.copy()
                if best_c5 < 0.380876:
                    break
            # Compute the gradient (approximate using finite differences)
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = compute_c5_np(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                c5_minus = compute_c5_np(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            # Update h using gradient descent
            h -= 0.01 * grad
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            # Ensure the integral constraint is satisfied
            total = np.sum(h) * (2.0 / n_points)
            if total < 1 - 1e-5:
                h += (1 - total) / n_points
            elif total > 1 + 1e-5:
                h -= (total - 1) / n_points
        return h
    
    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.array(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    total = np.sum(h_initial) * (2.0 / n_points)
    if abs(total - 1) > 1e-5:
        h_initial += (1 - total) / n_points
    
    # Run the optimization
    h_optimized = optimize_h(h_initial)
    best_c5 = compute_c5_np(h_optimized)
    
    # Try increasing the number of points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0
        total = np.sum(h_initial) * (2.0 / n_points)
        h_initial += (1 - total) / n_points
        h_initial = np.clip(h_initial, 0, 1)
        h_optimized = optimize_h(h_initial)
        best_c5 = compute_c5_np(h_optimized)
    
    # Final check
    if best_c5 < 0.380876:
        print(f"Found better bound: {best_c5}")
    else:
        print(f"Current bound: {best_c5}")
    
    return (h_optimized.tolist(), best_c5, n_points)
