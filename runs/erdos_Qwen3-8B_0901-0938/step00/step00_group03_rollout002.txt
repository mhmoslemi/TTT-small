import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    max_iterations = 100
    learning_rate = 0.1
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = None
    
    # Initial guess
    h = np.random.rand(n_points)
    h = h * (2.0 / n_points)  # Normalize to ensure integral is 1
    
    # Optimization loop
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute overlaps
        overlaps = signal.correlate(h, 1 - h, mode="full")
        overlaps = overlaps * dx
        
        # Find maximum overlap
        max_overlap = np.max(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    gradient[i] += (1 - h[i + k]) * dx
                if i - k >= 0:
                    gradient[i] -= h[i - k] * dx
        
        # Update h
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Normalize to ensure integral is 1
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
