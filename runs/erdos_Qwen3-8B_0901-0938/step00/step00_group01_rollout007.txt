import numpy as np

import time

# Pre-imported functions

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]  # Use initial construction
    h = np.array(h, dtype=float)
    
    # Ensure the integral constraint is satisfied
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h, dtype=float)
        h = np.concatenate((h, h[:1]))  # Extend for full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to optimize h
    def optimize_h(h_initial, n_points, max_iter=100, tol=1e-6):
        h = h_initial.copy()
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute the gradients for each point
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_i = h[i]
                # Compute the contribution to C5 from h[i]
                # This is a simplified gradient approximation
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                c5_minus = compute_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update h using gradient descent
            step_size = 0.1
            h = h - step_size * grad
            
            # Ensure h stays in [0, 1]
            h = np.clip(h, 0, 1)
            
            # Ensure the integral constraint is satisfied
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-6:
                h = h / sum_h * (n_points / 2)
            
            # Check for convergence
            if current_c5 < 0.380876:
                break
        
        return h
    
    # Optimize h
    h_opt = optimize_h(h, n_points)
    best_c5 = compute_c5(h_opt)
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 100:
        n_points = 200

        h = initial_h_values.copy()[:n_points]
        h = np.array(h, dtype=float)

        if abs(sum_h - n_points / 2) > 1e-6:
            h = h / sum_h * (n_points / 2)
        
        h_opt = optimize_h(h, n_points)
        best_c5 = compute_c5(h_opt)
    
    # Return the best solution
    return h_opt, best_c5, n_points
