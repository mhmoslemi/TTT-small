import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Start with the initial n_points
    max_n_points = 100  # Max n_points to try
    max_iterations = 100  # Max number of iterations to try
    n_points_list = [45, 60, 80, 100]  # Try different n_points values
    
    for n in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial h values
        h = initial_h_values.copy()
        h = np.resize(h, (n,))
        h = h[:n]  # Truncate or pad to match n_points
        h = np.resize(h, (n,))
        h = h / np.sum(h) * (n / 2)  # Ensure the integral is 1
        
        # Define the optimization variables
        x = cp.Variable(n, integer=True)
        constraints = [
            cp.sum(x) == n // 2,  # Ensure the integral is 1
            0 <= x,
            x <= n
        ]
        
        # Define the objective function
        def objective(x_val):
            h_val = x_val / n * 2  # Scale to [0, 2]
            # Compute the overlap integral
            corr = signal.correlate(h_val, 1 - h_val, mode="full")
            c5 = np.max(corr) * (2.0 / n)
            return c5
        
        # Use a local optimization method
        result = optimize.minimize(objective, x.value, method='L-BFGS-B', bounds=[(0, n)] * n)
        h_opt = result.x / n * 2
        c5 = objective(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            print(f"n={n}, C5={c5}")
    
    # Evaluate the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C5: {final_c5}")
    
    return best_h, final_c5, len(best_h)
