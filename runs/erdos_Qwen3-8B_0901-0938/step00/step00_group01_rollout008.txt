import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    max_iter = 100  # Max iterations for optimization
    tolerance = 1e-5  # Tolerance for convergence
    best_c5 = float('inf')
    best_h = None
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = np.array(initial_h[:n_points])  # Ensure it's the same length as n_points
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        if np.abs(np.sum(h) * dx - 1) > 1e-8:
            return float('inf')
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Take the maximum value and multiply by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h using gradient descent
    def optimize_h(h, n_points, max_iter, tolerance):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        for _ in range(max_iter):
            # Compute the current C5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Compute the gradient (approximate)
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = np.clip(h_minus, 0, 1)
                c5_minus = compute_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            # Update h using gradient descent
            h -= 0.1 * grad
            h = np.clip(h, 0, 1)
            # Check for convergence
            if np.abs(np.sum(h) * dx - 1) < tolerance:
                break
        return h
    
    # Start optimization
    h = initial_h
    for _ in range(max_iter):
        h = optimize_h(h, n_points, max_iter, tolerance)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Check if time is up
        if time.time() - start_time > budget_s - 10:
            break
    
    # Try increasing the number of points if time allows
    if time.time() - start_time < budget_s - 60:
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.zeros(n_points)
        # Initialize with a better guess
        best_h = np.linspace(0, 1, n_points)
        best_h = np.clip(best_h, 0, 1)
        # Ensure the integral constraint is satisfied
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        # Optimize
        for _ in range(max_iter):
            h = optimize_h(best_h, n_points, max_iter, tolerance)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Check for convergence
            if np.abs(np.sum(h) * dx - 1) < tolerance:
                break
            # Check if time is up
            if time.time() - start_time > budget_s - 10:
                break
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
