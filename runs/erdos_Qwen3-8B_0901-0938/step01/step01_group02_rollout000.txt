```python
import numpy as np
import scipy.optimize as opt
import time
import copy
from scipy.optimize import Bounds
from scipy.optimize import minimize

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]  # Truncate to n_points
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure sum(h) == n_points / 2

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Ensure sum(h) == n_points / 2
        h = h.reshape(-1, 1)
        ones_minus_h = 1 - h
        # Compute cross-correlation
        corr = np.correlate(h.flatten(), ones_minus_h.flatten(), mode='full')
        # Compute the maximum value of the overlap integral
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Function to optimize h
    def optimize_h(h_initial):
        def objective(h):
            return evaluate_overlap(h)

        # Bounds for h
        bounds = Bounds(0, 1)

        # Optimization settings
        options = {
            'disp': False,
            'ftol': 1e-6,
            'maxiter': 1000
        }

        # Perform optimization
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, options=options)
        return result.x

    # Main loop to try different n_points and optimize
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s:
            break
        print(f"Trying with n_points = {n_points}")
        # Generate initial guess
        initial_h = initial_h_values.copy()
        initial_h = initial_h[:n_points]
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2)

        # Optimize
        optimized_h = optimize_h(initial_h)
        current_c5 = evaluate_overlap(optimized_h)

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h

        # Try a few random perturbations
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = optimized_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2)
            current_c5 = evaluate_overlap(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h

    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    print(f"Best C5 found: {final_c5}")
    return (best_h, final_c5, n_points)
```
