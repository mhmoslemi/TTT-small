```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = None
    
    # Initial guess
    h = np.array(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)
    
    # Evaluate initial solution
    c5 = evaluate_erdos_solution(h, dx)
    best_c5 = c5
    best_h = h.copy()
    
    # Optimization loop
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate the perturbed solution
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        
        # Accept the perturbation if it improves the solution
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            print(f"Iteration {iteration + 1}: C5 = {best_c5:.6f}")
        else:
            # Revert to the previous solution
            h = best_h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    print(f"Final C5: {final_c5:.6f}")
    
    return best_h, final_c5, n_points
```
