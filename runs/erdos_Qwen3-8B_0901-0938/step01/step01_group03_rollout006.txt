import numpy as np
import scipy.optimize as opt
import time
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    n_points = len(best_h)
    dx = 2.0 / n_points
    
    def objective_function(h_values):
        h = np.array(h_values)
        if np.sum(h) != n_points / 2:
            return float('inf')
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def optimize_h(h_initial):
        result = opt.minimize(
            objective_function,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        return result.x
    
    # Try a few optimization runs with different initializations
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_candidate = optimize_h(best_h)
        c5_candidate = objective_function(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    # Evaluate the best solution
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, len(best_h)
