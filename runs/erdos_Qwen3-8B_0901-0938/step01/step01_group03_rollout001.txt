import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    dx = 2.0 / n_points

    # Evaluate initial solution
    c5 = evaluate_erdos_solution(best_h, dx)
    best_c5 = c5
    print(f"Initial C5: {best_c5}")

    # Try different n_points values
    for n_points in [45, 60, 80, 100, 120, 150, 180, 200]:
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Normalize to satisfy ∫h(x)dx = 1
        h = np.clip(h, 0, 1)

        # Evaluate initial solution for this n_points
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Use CVXOPT for constrained optimization
        h_var = cp.Variable(n_points)
        constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
        objective = cp.Maximize(cp.max(cp.sum(h_var * (1 - cp.roll(h_var, 1)), axis=0) * dx))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_opt = np.round(h_var.value, 5)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

        # Use gradient descent with L-BFGS-B
        def objective_func(h):
            return evaluate_erdos_solution(h, dx)

        result = optimize.minimize(objective_func, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if result.success:
            h_opt = result.x
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        print(f"n_points={n_points}, Best C5: {best_c5}")

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    print(f"Final C5: {final_c5}")
    return best_h, final_c5, len(best_h)
