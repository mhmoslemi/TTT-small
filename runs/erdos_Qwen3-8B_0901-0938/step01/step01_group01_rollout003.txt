import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Use the provided initial construction
    initial_h = initial_h[:n_points]  # Ensure it's the correct length
    
    # Normalize the initial h to satisfy the integral constraint
    sum_h = np.sum(initial_h) * dx
    scaling_factor = (1.0 / sum_h) if sum_h != 0 else 1.0
    initial_h = initial_h * scaling_factor
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h))  # Extend to full length for correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Objective function for optimization: minimize the maximum overlap
    def objective(h):
        return compute_c5(h)
    
    # Constraints: sum(h) * dx = 1
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization: use a local optimizer with constraints
    def optimize_h(initial_h, n_points, max_iter=100, tol=1e-6):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        result = opt.minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=tol,
            options={'maxiter': max_iter, 'disp': False}
        )
        if result.success:
            return result.x
        else:
            return initial_h  # Return initial if optimization fails
    
    # Try different n_points to find the best solution within the budget
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    
    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 10:
            break
        # Generate a random initial guess
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h) * dx
        scaling_factor = (1.0 / sum_h) if sum_h != 0 else 1.0
        h = h * scaling_factor
        
        # Optimize
        optimized_h = optimize_h(h, n_points)
        current_c5 = compute_c5(optimized_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
            best_n_points = n_points
    
    # Final evaluation
    return (best_h, best_c5, best_n_points)
