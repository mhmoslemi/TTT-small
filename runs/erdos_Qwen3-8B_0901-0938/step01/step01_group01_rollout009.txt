import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import multiprocessing

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    n_points = len(best_h)
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral for a given h
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - 1]])  # Extend for full convolution
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to perturb h
    def perturb_h(h, perturbation_strength=0.1):
        h = h.copy()
        indices = np.random.choice(len(h), size=int(len(h) * 0.1), replace=False)
        h[indices] += np.random.uniform(-perturbation_strength, perturbation_strength, size=len(indices))
        h = np.clip(h, 0, 1)
        return h
    
    # Local search with perturbation
    for _ in range(100):
        if time.time() - start_time > budget_s:

    # Convert h to a variable
    h_var = cp.Variable(n_points)
    constraints = [cp.sum(h_var) == n_points / 2]

    # We approximate the maximum overlap by evaluating it at several shifts

    overlaps = []
    for k in shifts:
        if k == 0:
            continue
        h_shifted = np.roll(h_var, k)
        overlap = cp.multiply(h_var, 1 - h_shifted)
        overlaps.append(overlap)
    
    objective = cp.Minimize(cp.Max(overlaps))
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.SCS, max_iter=100)
