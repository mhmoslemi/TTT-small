import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Choose a manageable number of points for faster computation
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Start from the initial construction
    
    # Rescale to fit the current n_points
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.round(initial_h, 5)  # Remove small numerical errors
    
    # Ensure the initial_h satisfies the constraint sum(h) * dx = 1
    initial_sum = np.sum(initial_h) * dx
    if np.abs(initial_sum - 1) > 1e-6:
        initial_h = initial_h * (1 / initial_sum)
    
    # Define objective function
    def objective(h_values):
        h = h_values.astype(float)
        # Ensure h_values are in [0, 1]
        h = np.clip(h, 0, 1)
        # Ensure the constraint sum(h) * dx = 1
        h = h * (1 / (np.sum(h) * dx))
        # Compute the overlap integral
        h_shifted = np.zeros_like(h)
        for k in range(1, n_points - 1):
            h_shifted[k:] = h[:-k]
        overlap = np.sum(h * (1 - h_shifted)) * dx
        return overlap
    
    # Define constraint for sum(h) * dx = 1
    def constraint(h_values):
        return np.sum(h_values) * dx - 1
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (1 / (np.sum(h_initial) * dx))
    
    # Optimization settings
    maxiter = 100
    start_time = time.time()
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': maxiter}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    
    # Compute C₅ value for the best solution
    def compute_c5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        h_shifted = np.zeros_like(h)
        for k in range(1, n_points - 1):
            h_shifted[k:] = h[:-k]
        overlaps = np.sum(h * (1 - h_shifted), axis=1) * dx
        return np.max(overlaps)
    
    c5_bound = compute_c5(best_h)
    
    # Return the best solution
    return (best_h, c5_bound, n_points)
