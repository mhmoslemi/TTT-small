import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros((n_points, n_points))
    for i in range(n_points):
        h_shifted[i, :] = h
    corr = np.zeros(n_points)
    for k in range(n_points):
        corr[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # start with 45 points
    initial_h = initial_h_values()
    h = initial_h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Optimization strategy: use gradient descent with a custom loss function
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.zeros(n_points)
        for k in range(n_points):
            corr[k] = np.sum(h * (1 - np.roll(h, k))) * (2.0 / n_points)
        return np.max(corr)
    
    def objective_func_with_gradient(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.zeros(n_points)
        for k in range(n_points):
            corr[k] = np.sum(h * (1 - np.roll(h, k))) * (2.0 / n_points)
        c5_val = np.max(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Gradient from max corr[k]
            if corr[k] == c5_val:
                for i in range(n_points):
                    grad[i] += (1 - np.roll(h, k))[i] * (2.0 / n_points)
                    grad[i] -= h[i] * (2.0 / n_points) * np.roll(np.ones(n_points), k)[i]
        return c5_val, grad.flatten()
    
    # Use a simple gradient descent optimizer
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate objective
        c5_val, grad = objective_func_with_gradient(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        # Update h using gradient descent
        h -= 0.01 * grad
        # Ensure h stays in [0, 1] and sum to n_points / 2
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Further refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        c5_val, grad = objective_func_with_gradient(best_h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = best_h.copy()
        best_h -= 0.005 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Try a random perturbation
    perturbation = np.random.uniform(-0.05, 0.05, n_points)
    perturbed_h = np.clip(best_h + perturbation, 0, 1)
    perturbed_h = perturbed_h / np.sum(perturbed_h) * (n_points / 2)
    c5_perturbed = evaluate_erdos_solution(perturbed_h)
    if c5_perturbed < best_c5:
        best_c5 = c5_perturbed
        best_h = perturbed_h
    
    return best_h, best_c5, n_points
