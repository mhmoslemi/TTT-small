import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 100  # Start with 100 points for a balance between accuracy and speed
    
    # Initialize the function h with the provided initial values
    h = initial_h_values.copy()
    n_points_initial = len(h)
    if n_points_initial != n_points:
        h = np.zeros(n_points)
        # Distribute the initial values proportionally to maintain the integral constraint
        for i in range(n_points):
            h[i] = initial_h_values[i // n_points_initial] if i // n_points_initial < len(initial_h_values) else 0
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(h) * (2.0 / n_points)
    if abs(total_integral - 1.0) > 1e-6:
        # Adjust the values to satisfy the integral constraint
        scaling_factor = 1.0 / total_integral
        h = h * scaling_factor
    
    def evaluate_c5(h):
        # Compute the overlap integral
        dx = 2.0 / n_points
        # Compute the convolution
        convolved = np.correlate(h, 1 - h, mode="full")
        # The maximum overlap value
        c5 = np.max(convolved) * dx
        return c5
    
    def objective_function(h):
        # Objective is to minimize the maximum overlap
        return evaluate_c5(h)
    
    def constraints(h):
        # Constraint: integral of h must be 1
        integral = np.sum(h) * (2.0 / n_points)
        return [integral - 1.0]
    
    # Use scipy optimization with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective_function,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Extract the best solution found by the optimizer
    best_h = result.x
    
    # Evaluate the best solution
    best_c5 = evaluate_c5(best_h)
    
    # Try a few more iterations with finer grid if time allows
    elapsed_time = time.time() - time.time()
    if elapsed_time < budget_s - 5:
        n_points_new = n_points * 2
        new_h = np.zeros(n_points_new)
        # Resample the current solution to the new grid
        for i in range(n_points_new):
            new_h[i] = np.interp(i * 2.0 / n_points_new, np.linspace(0, 2, n_points), best_h)
        
        # Ensure the integral constraint is satisfied
        total_integral = np.sum(new_h) * (2.0 / n_points_new)
        if abs(total_integral - 1.0) > 1e-6:
            scaling_factor = 1.0 / total_integral
            new_h = new_h * scaling_factor
        
        # Evaluate the new solution
        new_c5 = evaluate_c5(new_h)
        
        # Compare and select the better solution
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    # Finally, return the best solution and the bound
    return (best_h, best_c5, len(best_h))
