import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-defined initial solution
initial_h_values = np.array([...])  # Placeholder; replace with actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    n_points = 100  # Adjust based on time budget
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Initial guess with random values
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    # Function to compute C₅
    def compute_c5(h_vals):
        h = h_vals
        shifted = np.zeros(2 * n_points - 1)
        for i in range(n_points):
            shifted[i:i + n_points] += 1 - h
        correlate = np.correlate(h, shifted, mode="full")
        c5 = np.max(correlate) * dx
        return c5
    
    # Function to compute the overlap integral for all k
    def compute_c5_all(h_vals):
        h = h_vals
        shifted = np.zeros(2 * n_points - 1)
        for i in range(n_points):
            shifted[i:i + n_points] += 1 - h
        correlate = np.correlate(h, shifted, mode="full")
        return correlate * dx
    
    # Optimization setup
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [cp.sum(h_var) == n_points / 2]
    c5 = cp.Maximize(cp.sum(cp.multiply(h_var, 1 - h_var)))
    
    # Solve using CVXPY with a solver
    prob = cp.Problem(c5, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=1000, time_limit=budget_s - (time.time() - start_time))
        if prob.status == 'optimal':
            h_opt = h_var.value
        else:
            h_opt = h.copy()
    except:
        h_opt = h.copy()
    
    # Refine using local optimization
    def objective_func(h_vals):
        c5_values = compute_c5_all(h_vals)
        return np.max(c5_values)
    
    result = optimize.minimize(
        objective_func,
        h_opt,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-4,
        options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Ensure the integral constraint is met
    if np.sum(best_h) * dx < 1 - 1e-6 or np.sum(best_h) * dx > 1 + 1e-6:
        best_h = initial_h_values.copy()
        best_c5 = compute_c5(best_h)
    
    # Final check to ensure time budget is not exceeded
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (initial_h_values, compute_c5(initial_h_values), len(initial_h_values))
    
    return (best_h, best_c5, n_points)
