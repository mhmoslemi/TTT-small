import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Number of points (choose a smaller number for faster optimization)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess: start with the provided initial configuration or random
    h = np.zeros(n_points)
    if initial_h_values.size > 0 and initial_h_values.shape[0] == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    
    # Ensure the integral constraint is satisfied
    h = h * (n_points / 2) / np.sum(h)
    
    # Define the optimization variable
    x = cp.Variable(n_points, lower=0, upper=1)
    
    # Objective: minimize the maximum overlap integral
    max_overlap = cp.Variable()
    constraints = [cp.sum(x) == n_points / 2]
    
    # Define the correlation operation
    def evaluate_correlation(x_vals):
        h_vals = x_vals
        corr = signal.correlate(h_vals, 1 - h_vals, mode="full")
        return corr * dx
    
    # We need to compute the max overlap for all shifts
    # For a given x_vals, evaluate the correlation and find the maximum
    # Then minimize that maximum (this is a non-convex problem)
    
    # Instead of exact optimization, we'll use a gradient-based method with a custom objective
    # Convert to a numpy array for optimization
    x_vals = x.value if x.value is not None else h
    
    # Define the objective function for optimization
    def objective(x_vals):
        corr = signal.correlate(x_vals, 1 - x_vals, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Optimization: minimize the maximum overlap
    # Use L-BFGS-B for a local minimum
    def optimize_h(h_initial):
        def objective_func(x):
            return objective(x)
        
        def gradient_func(x):
            # Compute the gradient numerically using finite differences
            eps = 1e-6
            grad = np.zeros_like(x)
            for i in range(len(x)):
                x_plus = x.copy()
                x_plus[i] += eps
                obj_plus = objective(x_plus)
                x_minus = x.copy()
                x_minus[i] -= eps
                obj_minus = objective(x_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * eps)
            return grad
        
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            jac=gradient_func,
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        return result.x
    
    # Optimization loop
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Run optimization
    for _ in range(5):  # Try multiple iterations with different starting points
        h = optimize_h(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        if time.time() - start_time > budget_s - 1:
            break
    
    print(f"Final C5: {best_c5}")
    return best_h, best_c5, n_points
