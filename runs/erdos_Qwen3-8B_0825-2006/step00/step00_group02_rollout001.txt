import numpy as np
import scipy.optimize as optimize
import time
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Global constants
BEST_C5 = 1.0
BEST_H_VALUES = None
BEST_N_POINTS = 0

# Helper function to compute the overlap integral
def compute_c5(h, dx):
    h_extended = np.zeros(len(h) * 2 - 1)
    h_extended[:len(h)] = h
    correlation = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(correlation)
    return max_correlation * dx

# Function to evaluate and update the best solution
def update_best_solution(h_values, c5_bound, n_points):
    global BEST_C5, BEST_H_VALUES, BEST_N_POINTS
    if c5_bound < BEST_C5:
        BEST_C5 = c5_bound
        BEST_H_VALUES = h_values
        BEST_N_POINTS = n_points

# Function to generate a random initial solution
def generate_random_h(n_points):
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    return h

# Function to refine a solution using optimization
def refine_solution(h_initial, n_points, seed=42):
    # Define variables
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    obj = cp.Minimize(cp.Max(cp.sum(cp.multiply(h, 1 - h[:n_points - k])) for k in range(1, n_points)))
    prob = cp.Problem(obj, constraints)
    
    # Solve the optimization problem
    try:
        result = prob.solve(solver=cp.SCS, max_iter=100, eps=1e-4, time_limit=30)
        if result is not None:
            return h.value
    except:
        pass
    return h_initial

# Main function to run the optimization
def run(seed=42, budget_s=60, **kwargs):
    global BEST_C5, BEST_H_VALUES, BEST_N_POINTS
    BEST_C5 = 1.0
    BEST_H_VALUES = None
    BEST_N_POINTS = 0
    start_time = time.time()
    
    # Initialize with the given initial construction
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    h_initial = h_initial / np.sum(h_initial) * (n_points_initial / 2)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = np.round(h_initial, 5)
    
    # Start with the initial solution
    h = copy.deepcopy(h_initial)
    n_points = n_points_initial
    dx = 2.0 / n_points
    c5_bound = compute_c5(h, dx)
    update_best_solution(h, c5_bound, n_points)
    
    # Try random initializations
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        n_points = min(200, n_points_initial)
        h = generate_random_h(n_points)
        dx = 2.0 / n_points
        c5_bound = compute_c5(h, dx)
        update_best_solution(h, c5_bound, n_points)
    
    # Refine the best solution found so far
    if BEST_H_VALUES is not None:
        h_refined = refine_solution(BEST_H_VALUES, BEST_N_POINTS)
        dx = 2.0 / BEST_N_POINTS
        c5_bound = compute_c5(h_refined, dx)
        update_best_solution(h_refined, c5_bound, BEST_N_POINTS)
    
    # If no solution found, use the initial one
    if BEST_H_VALUES is None:
        BEST_H_VALUES = h_initial
        BEST_N_POINTS = n_points_initial
        dx = 2.0 / BEST_N_POINTS
        c5_bound = compute_c5(BEST_H_VALUES, dx)
    
    # Ensure the solution is within the budget
    if time.time() - start_time > budget_s:
        BEST_H_VALUES = np.round(BEST_H_VALUES, 5)
        return (BEST_H_VALUES.tolist(), BEST_C5, BEST_N_POINTS)
    
    # Final check for best solution
    BEST_H_VALUES = np.round(BEST_H_VALUES, 5)
    return (BEST_H_VALUES.tolist(), BEST_C5, BEST_N_POINTS)
