```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    max_n_points = 1000
    n_points = 100  # start with 100 points, which is under 1k
    best_c5 = float('inf')
    best_h_values = None
    
    # Function to evaluate the C5 bound
    def evaluate(h_values):
        nonlocal best_c5
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h_values.copy()
        return c5
    
    # Start from initial_h_values
    if len(initial_h_values) <= max_n_points:
        h_initial = initial_h_values[:n_points].copy()
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if abs(sum_h - n_points / 2) > 1e-6:
            h_initial = h_initial * (n_points / 2) / sum_h
        h_initial = np.clip(h_initial, 0, 1)
        current_c5 = evaluate(h_initial)
    
    # Optimization loop
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s:
            break
        print(f"Trying n_points={n_points}")
        
        # Random initialization
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        current_c5 = evaluate(h)
        
        # Optimization with simulated annealing
        def objective(x):
            h = x.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate(h)
        
        result = optimize.minimize(objective, np.random.rand(n_points), method='L-BFGS-B', tol=1e-4, options={'maxiter': 100})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        current_c5 = evaluate(h_opt)
        
        # Local search
        for _ in range(10):
            h = h_opt.copy()
            idx = np.random.randint(n_points)
            h[idx] = np.random.rand()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate(h)
        
        if time.time() - start_time > budget_s:
            break
    
    return best_h_values, best_c5, len(best_h_values)
```
