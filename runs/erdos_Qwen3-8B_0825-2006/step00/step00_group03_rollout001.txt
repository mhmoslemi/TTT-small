```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Starting with 100 points, can be increased if needed
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()[:n_points]  # Clamp to n_points

    # Normalize initial h to satisfy the constraint sum(h) = n_points / 2
    h_initial = np.array(h_initial)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    def objective_function(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure the constraint is met
        # Compute the convolution and max overlap
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def optimize_h(h_values):
        h = cp.Variable(n_points)
        objective = cp.Minimize(objective_function(h))
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS)
        if prob.status != 'OPTIMAL':
            print("Warning: Optimization not optimal")
        return h.value

    # Initial optimization
    h_opt = optimize_h(h_initial)
    c5 = objective_function(h_opt)
    best_h = h_opt
    best_c5 = c5
    best_n_points = n_points

    # Try increasing the number of points if time allows
    for n_points in [150, 200, 250, 300]:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n_points
        # Get a new initial guess based on previous solution
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_n_points), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        def objective_function_new(h):
            h = h.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        def optimize_h_new(h_values):
            h = cp.Variable(n_points)
            objective = cp.Minimize(objective_function_new(h))
            constraints = [
                cp.sum(h) == n_points / 2,
                0 <= h,
                h <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS)
            if prob.status != 'OPTIMAL':
                print("Warning: Optimization not optimal")
            return h.value

        h_opt_new = optimize_h_new(h_initial)
        c5_new = objective_function_new(h_opt_new)
        if c5_new < best_c5:
            best_h = h_opt_new
            best_c5 = c5_new
            best_n_points = n_points

    # Final evaluation
    final_result = evaluate_erdos_solution(best_h)
    return best_h, final_result, best_n_points
```
