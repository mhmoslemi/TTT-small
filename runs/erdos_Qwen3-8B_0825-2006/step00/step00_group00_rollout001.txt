import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 45
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial guess
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Define the objective function for optimization
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Define the constraints
    constraints = [
        cp.sum(h_flat) == n_points / 2
    ]
    
    # Bounds on h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    h_flat = cp.Variable(n_points)
    prob = cp.Problem(cp.Minimize(objective(h_flat)), constraints)
    
    # Run the solver
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        if prob.status != 'OPTIMAL':
            print("Optimization failed.")
            return initial_h, evaluate_erdos_solution(initial_h), n_points
        
        h_opt = h_flat.value
        c5_bound = evaluate_erdos_solution(h_opt)
        return h_opt, c5_bound, n_points
    except:
        print("Optimization failed.")
        return initial_h, evaluate_erdos_solution(initial_h), n_points
