def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # fallback init
    try: parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),0,1000)
         if parent.ndim !=1 or parent.size==0 or not np.isfinite(parent).all():
              raise
    except Exception:
         parent = np.ones(256, dtype=np.float64)
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    
    n = parent.size if parent is not None else 256
    # Determine number of candidates to try.
    # We'll loop while time left.
    while time.monotonic() < deadline:
        # randomly decide which generator
        r = rng.random()
        if r < 0.33:
            cand = rng.random(n)  # uniform [0,1)
        elif r < 0.66:
            cand = rng.exponential(scale=1.0, size=n)
        else:
            # random permutation of parent values
            cand = parent.copy()
            rng.shuffle(cand)  # in-place shuffle
        # clip to [0,1000] just in case
        np.clip(cand, 0.0, 1000.0, out=cand)
        # Evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
        # Also try a small perturbation of best (like block scaling)
        # maybe after each accepted improvement, local search:
        # produce a candidate by scaling a random subsegment by exp(normal)
        # but limit time.
        if time.monotonic() >= deadline:
            break
        # optional: small local perturbation:
        # only a few times per iteration: produce perturbed version of best
        # but we already generate random candidate many times; could also incorporate local attempts.
        # Let's still add a Block perturbation as used earlier
        if rng.random() < 0.1: # occasionally
            # random block with multiplicative factor
            cand2 = best_seq.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand2[start:stop] *= factor
            # Optional symmetry blending
            if rng.random() < 0.35:
                cand2 = 0.5 * (cand2 + cand2[::-1])
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_seq = cand2
                best_score = score2
    # return best_seq as list of floats
    return [float(x) for x in best_seq]

def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 1.0)
    # initialize a from parent or random uniform
    if parent is defined:
         a = parent.astype(np.float64).copy()
    else:
         a = rng.random(256)  # fallback
    # ensure nonneg non-zero sum
    a = np.clip(a, 0.0, 1000.0)
    # Optionally, random shuffle for variability
    if rng.random() < 0.5:
         rng.shuffle(a)
    # sum_target
    sum_target = a.sum()
    if sum_target < 0.01:
         a = np.ones_like(a)
         sum_target = a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # Initialize step size
    step = 0.2
    # Perform gradient descent loop
    while time.monotonic() < deadline:
         # compute convolution
         b = np.convolve(a, a)
         idx = int(np.argmax(b))
         max_val = b[idx]
         # Compute gradient
         grad = np.zeros_like(a)
         # for i positions where idx-i in range
         # This can be vectorized:
         lo = max(0, idx - (len(a)-1))
         hi = min(len(a)-1, idx)
         # i range [lo, hi]
         # compute j = idx - i
         # grad[i] = 2 * a[j]
         i_range = np.arange(lo, hi+1)
         j_range = idx - i_range
         grad[i_range] = 2.0 * a[j_range]
         # take step
         a_new = a - step * grad
         # clip negative to zero
         a_new = np.clip(a_new, 0.0, 1000.0)
         # renormalize to original sum
         total = a_new.sum()
         if total < 1e-12:
             # avoid zero vector
             a_new = a.copy()
         else:
             a_new *= sum_target / total
         score = evaluate_sequence(a_new.tolist())
         # accept if better
         if np.isfinite(score) and score < best_score:
             best_seq = a_new.copy()
             best_score = score
         # simple acceptance rule: if score improved, move to a_new; else maybe keep current?
         # Use hill climbing: if improvement, update a = a_new; else maybe still accept with probability.
         if np.isfinite(score) and score < best_score + 1e-12: # improvement
             a = a_new
         else:
             # small random reset or step size decay
             # occasionally random perturbation to escape local minima
             if rng.random() < 0.05:
                 # random perturbation: multiply random block by factor >1
                 width = int(rng.integers(max(1, len(a)//200), max(2, len(a)//12)))

                 a_pert *= sum_target / a_pert.sum()
                 a = a_pert

         # check deadline
    # return best
    return [float(x) for x in best_seq]

def _geom_array(n, c):
    """geometric decreasing array a_i = exp(-c*i/n).  a_0 = 1."""
    eps = c / n
    r = math.exp(-eps)
    return r ** np.arange(n, dtype=np.float64)

    """Search for a low‑AC1 step function without dense matrices."""
    rng = np.random.default_rng(seed)
    # time guard – keep a few seconds for final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise a fallback sequence (the supplied parent if it exists)
    try:

            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    best_seq = parent.copy()

    # ------------------------------------------------------------------

    # and many permutations of the same value multiset.
    decay_consts = (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0)

                      if math.gcd(p, n) == 1]

        if time.monotonic() >= deadline:
            break
        base = _geom_array(n, c)                     # decreasing values ∈ (0,1]
