def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    # Determine n
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
    except Exception:
        n = 256
    # initial candidate: constant 1.0
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # Perform short random perturbations within time budget to try improve (unlikely)
    while time.monotonic() < deadline:
        # Randomly choose width and factor
        width = rng.integers(1, max(2, n//10))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))
        cand = best.copy()
        cand[start:start+width] *= factor
        # clip within [0, 1000]
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    return [float(x) for x in best]
