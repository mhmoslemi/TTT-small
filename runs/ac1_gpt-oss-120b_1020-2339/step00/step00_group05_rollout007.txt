def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    else:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    best = base.copy()
    best_score = evaluate(best)

    # candidate generation functions
    def candidate_from_exp(alpha):
        seq = np.exp(-alpha * np.arange(n))
        return apply_clamp(seq)

    def candidate_from_sym_exp(alpha):
        idx = np.arange(n)
        dist = np.minimum(idx, n-1-idx)
        seq = np.exp(-alpha * dist)
        return apply_clamp(seq)

    def candidate_from_sin():
        seq = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        return apply_clamp(seq)

    def candidate_linear():
        seq = np.linspace(1.0, 0.0, n)
        return apply_clamp(seq)

    def candidate_random():
        seq = rng.random(n) * 10.0  # random low amplitude to avoid overflow
        return apply_clamp(seq)

    # Pre-evaluate a set of parametric candidates
    alphas = np.logspace(-4, -2, 20)  # from 1e-4 to 1e-2
    for a in alphas:
        for kind in ('exp', 'sym'):
            if time.monotonic() >= deadline:
                break
            if kind == 'exp':
                seq = candidate_from_exp(a)
            else:
                seq = candidate_from_sym_exp(a)
            score = evaluate(seq)
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score

        if time.monotonic() >= deadline:
            break

    # evaluate other shapes
    for fn in (candidate_from_sin, candidate_linear):
        if time.monotonic() >= deadline:
            break
        seq = fn()
        score = evaluate(seq)
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score

    # random perturbations around best
    while time.monotonic() < deadline:
        # block perturbation similar to previous method
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand = best.copy()
        factor = np.exp(rng.normal(0.0, 0.12))
        cand[start:stop] *= factor
        if rng.random() < 0.35:

        cand = apply_clamp(cand)
        score = evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

        # occasionally attempt LP improvement
        if HAVE_CVXPY and time.monotonic() + 4.0 < deadline:
            # compute M from best
            M = np.max(np.convolve(best, best))
            # solve LP
            # try solve, with time limit internal maybe customizing via cp.settings?
            try:
                g = cp.Variable(n, nonneg=True)
                constraints = [g <= 1000,

                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
                if g.value is not None:
                    g_seq = np.maximum(0.0, np.minimum(g.value, 1000.0))

                    sum_g = g_seq.sum()
                    if sum_g > 0:

                        score = evaluate(cand)
                        if np.isfinite(score) and score < best_score:
                            best = cand
                            best_score = score
                # update best for next iteration
            except Exception:
                pass
            # break after one LP or attempt few times
            # maybe add cooldown
            # continue loop
        # loop continues

    return [float(x) for x in best]

def improve_by_local_moves(seq):

    curr_score = compute_score(current)

        k = np.argmax(conv)
        i_start = max(0, k - (n-1))

        j_range = k - i_range
        contrib = current[i_range] * current[j_range]
        idx = np.argmax(contrib)
        donor = i_range[idx] if current[i_range[idx]] >= current[j_range[idx]] else j_range[idx]

        recv = rng.integers(0, n)
        if recv == donor:
            continue
        # transfer fraction
        frac = rng.uniform(0.05, 0.2)
        delta = min(current[donor] * frac, 1000.0 - current[recv])
        if delta <= 0:
            continue
        new_current = current.copy()
        new_current[donor] -= delta
        new_current[recv] += delta
        # clip
        new_current = np.clip(new_current, 0.0, 1000.0)

            curr_score = new_score

                best_seq = new_current
                best_score = new_score
                no_improve = 0

            no_improve += 1
        # early exit if no improve for many steps
        if no_improve > 1000:
            break
    return best_seq, best_score

import numpy as np
import time

def _score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # starting point – use the provided parent if it looks sane
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)  # a harmless fallback

    n = parent.shape[0]
    best_seq = parent.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # helper to record a better candidate
    # ------------------------------------------------------------------
    def _maybe_update(seq):
        nonlocal best_seq, best_score
        sc = _score(seq)
        if sc < best_score:
            best_seq = seq
            best_score = sc

    # ------------------------------------------------------------------
    # explore a few parametric families (exponential, power‑law, …)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # exponential decays
    for a in np.logspace(-5, -2, num=15):
        seq = np.exp(-a * idx)
        _maybe_update(seq)

    # symmetric exponential (high at both ends, low in the middle)
    d = np.minimum(idx, n - 1 - idx)
    for a in np.logspace(-5, -2, num=15):
        seq = np.exp(-a * d)
        _maybe_update(seq)

    # linear decay from left to right
    _maybe_update(np.linspace(1.0, 0.0, n))

    # power‑law decays
    for p in np.linspace(0.1, 2.0, num=20):
        seq = (idx + 1.0) ** (-p)
        _maybe_update(seq)

    # pure sine (half‑period, non‑negative)
    _maybe_update(np.sin(np.pi * (idx + 0.5) / n))

    # raised cosine (non‑negative)
    _maybe_update((1.0 + np.cos(np.pi * idx / (n - 1))) * 0.5)

    # ------------------------------------------------------------------
    # hill‑climbing moves that directly attack the current maximal
    # convolution entry.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_score = best_score
    no_improve = 0
    iter_cnt = 0

    while time.monotonic() < deadline:
        # compute the convolution and the offending shift
        conv = np.convolve(current, current)
        k = int(np.argmax(conv))

        # indices that contribute to this shift
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range

        # contribution of each pair
        contrib = current[i_range] * current[j_range]
        if contrib.size == 0:
            break
        pair_idx = int(np.argmax(contrib))
        i_max = i_range[pair_idx]
