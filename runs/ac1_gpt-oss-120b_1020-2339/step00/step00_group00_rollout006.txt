```
def simplex_projection(v):
    # Project onto probability simplex (sum=1, v>=0)
    # Use algorithm O(n log n) for big n.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np
import time
import math

def _build_sidon_set(n, target, rng):
    """Greedy construction of a B2 (Sidon) set up to size ~target."""
    best = []
    # try a few random orders, keep the largest set found
    attempts = 0
    while attempts < 8:
        perm = rng.permutation(n)
        cur = []
        sums = set()
        for idx in perm:
            s_self = idx + idx
            if s_self in sums:
                continue
            ok = True
            for s in cur:
                if idx + s in sums:
                    ok = False
                    break
            if not ok:
                continue
            # add idx
            for s in cur:
                sums.add(idx + s)
            sums.add(s_self)
            cur.append(idx)
            if len(cur) >= target:
                break
        if len(cur) > len(best):
            best = cur
            if len(best) >= target:
                break
        attempts += 1
        # early stop if time is low
        if time.monotonic() > _deadline:
            break
    return best

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Generate a sequence with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # leave a small buffer for the final return
    global _deadline
    _deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # length of the problem (use provided parent if available)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 4806  # fallback length

    # Desired size of a Sidon set (roughly sqrt(n))
    target_sz = max(1, int(math.sqrt(n)))

    # Build the heavy index set
    heavy_idx = _build_sidon_set(n, target_sz, rng)
    m = len(heavy_idx)

    # If construction failed, fall back to a uniform vector
    if m == 0:
        a = np.ones(n, dtype=np.float64)
        a /= a.sum()
        return [float(x) for x in a]

    # Choose a small background weight ε.
    # Constraints:
    #   (n-m) * ε²  ≤ 2   (background self‑convolution)
    #   m * ε       ≤ 2   (heavy–background cross terms)
    eps1 = math.sqrt(2.0 / n)          # from background self‑conv.
    eps2 = 2.0 / m                     # from cross terms
    epsilon = min(eps1, eps2) * 0.9    # safety margin

    # Initialise the candidate
    a = np.full(n, epsilon, dtype=np.float64)
    a[heavy_idx] = 1.0

    # Evaluate once – keep the best seen
    best_vec = a.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # Simple hill‑climbing tweaks while time permits
    # Move a small amount of mass from a heavy position to a light one
    # (keeps total sum unchanged)
    while time.monotonic() < _deadline:
        # pick a heavy index and a distinct light index
        hi = rng.choice(heavy_idx)
        lo = rng.integers(0, n)
        if lo in heavy_idx:
            continue
        delta = best_vec[hi] * 0.05  # 5 % of the heavy mass
        if delta <= 0:
            continue
        trial = best_vec.copy()
        trial[hi] -= delta
        trial[lo] += delta
        # Ensure non‑negativity (should hold) and renormalise (sum unchanged)
        if trial[hi] < 0:
            continue
        score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score
        # occasional random reset to escape plateaus
