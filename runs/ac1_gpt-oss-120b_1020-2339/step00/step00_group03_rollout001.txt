import time
import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 bound evaluated by ``evaluate_sequence``.

    The algorithm starts from a uniform sequence (which attains the theoretical
    lower bound 2.0 for the given objective) and performs a few cheap random
    block perturbations in case a better‑scoring point is found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    # Leave a small safety margin for interpreter overhead.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # Determine the length to use (prefer the provided construction if any).
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:  # height_sequence_1 may be undefined
        base = None

    if (
        base is not None
        and base.ndim == 1
        and base.size > 0
        and np.isfinite(base).all()
    ):
        n = int(base.size)
        # Clip to the allowed range before using any statistics.
        base = np.clip(base, 0.0, 1000.0)
        total = base.sum()
        if total < 0.01:          # degenerate parent – fall back to a small sum
            total = 1.0
        uniform_val = total / n      # keep the same total mass
        best = np.full(n, uniform_val, dtype=np.float64)
    else:
        # No usable parent – choose a modest length and unit entries.
        n = 256
        best = np.full(n, 1.0, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Evaluate the starting candidate.
    # ----------------------------------------------------------------------
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # Simple random block perturbations (matrix‑free, O(n) memory).
    # ----------------------------------------------------------------------
    it = 0
    max_iters = 200            # safe hard cap; the time check is the real stop‑gap
    while time.monotonic() < deadline and it < max_iters:
        # Randomly choose a contiguous block and scale it.
        width = int(rng.integers(1, max(2, n // 20)))   # block size ≥1, ≤n/20
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # modest multiplicative noise

        cand = best.copy()
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Guard against an accidental near‑zero total (should never happen).
        if cand.sum() < 0.01:
            cand[0] += 0.01 - cand.sum()

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

        it += 1

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats).
    # ----------------------------------------------------------------------
    return [float(v) for v in best]
