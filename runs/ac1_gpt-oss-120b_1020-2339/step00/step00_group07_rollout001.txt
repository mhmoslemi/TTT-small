def project_onto_simplex(v, s=1):
    # ensure nonnegative sum s
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def simplex_projection(v):
    # Projects onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v)+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all non-positive, return uniform distribution
            return np.full_like(v, 1.0/len(v))
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    while time.monotonic() < deadline:
        # compute convolution
        b = np.convolve(a, a)
        maxb = b.max()
        m = int(b.argmax())
        # compute gradient direction (negative gradient)
        g = np.zeros_like(a)
        # Only indices i where 0 <= m-i < n
        lo = max(0, m - (n-1))
        hi = min(n-1, m)  # i up to min(n-1,m)
        # i in [lo, hi]
        # vectorized:
        i_range = np.arange(lo, hi+1)
        j_range = m - i_range
        g[i_range] = a[j_range]  # derivative/2; scaling not important
        # compute step size based on remaining time (decrease)
        remaining = deadline - time.monotonic()
        frac = remaining / float(budget_s)
        step = init_step * (frac ** 0.5)  # gradually decreasing
        step = max(step, min_step)

        # gradient descent candidate
        cand = a - step * g
        # ensure nonneg; project to simplex sum=1
        cand = np.maximum(cand, 0.0)
        cand = project_simplex(cand)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            a = cand  # adopt
            continue

        # else try random perturbation (like block)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        pert = a.copy()
        factor = np.exp(rng.normal(0.0, 0.12))
        pert[start:stop] *= factor
        if rng.random() < 0.35:
            pert = 0.5 * (pert + pert[::-1])
        pert = np.clip(pert, 0.0, 1000.0)
        # reproject to simplex sum=1
        pert = project_simplex(pert)
        score2 = float(evaluate_sequence(pert.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best = pert
            best_score = score2
            a = pert

        # occasionally (and as fallback) sample random dirichlet vector
        if rng.random() < 0.01:
            rand = rng.random(n)
            rand = rand / rand.sum()
            score3 = float(evaluate_sequence(rand.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best = rand
                best_score = score3
                a = rand

    # return best scaled back. maintain sum=1 (which evaluator rescales)
    result = [float(x) for x in best]
    return result
```

def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

import numpy as np
import time

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
