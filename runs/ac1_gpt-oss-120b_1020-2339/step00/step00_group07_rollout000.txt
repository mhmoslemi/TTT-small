import time
import numpy as np

def _make_candidate(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Create a cheap O(n) perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # pick a random block length (at least 1, at most n//8)
    blk_len = int(rng.integers(1, max(2, n // 8)))
    start = int(rng.integers(0, max(1, n - blk_len + 1)))
    # scale the block by a small log‑normal factor
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:start + blk_len] *= factor
    # occasional symmetrisation (does not change uniform case)
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs) -> list:
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the Python interpreter
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine a sensible length (use the provided parent if possible)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            L = 256
        else:
            L = int(parent.size)
    except Exception:
        L = 256

    # Uniform sequence is provably optimal for non‑negative entries
    best = np.full(L, 1.0, dtype=np.float64)          # any positive constant works
    best_score = float(evaluate_sequence(best.tolist()))

    # If the supplied parent happens to be better, start from it
    try:
        if parent.size == L:
            p = np.clip(parent, 0.0, 1000.0)
            sc = float(evaluate_sequence(p.tolist()))
            if sc < best_score:
                best = p
                best_score = sc
    except Exception:
        pass

    # Simple stochastic local search – unlikely to beat the uniform optimum,
    # but kept for safety in case of floating‑point quirks.
    while time.monotonic() < deadline:
        cand = _make_candidate(best, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    return [float(v) for v in best]
