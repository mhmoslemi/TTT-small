        g = np.zeros_like(a)

        a = a - lr * g

import time
import numpy as np

def _fast_score(arr: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative array."""
    s = arr.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(arr, arr)
    return 2.0 * arr.size * conv.max() / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs):
    """Search for a sequence with a low AC1 bound using cheap gradient steps.

    The routine works with the pre‑imported ``height_sequence_1`` when it is
    available; otherwise it falls back to a uniform vector of length 256.
    It respects the time budget and always returns a valid fallback candidate.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise problem size and a starter sequence
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:                     # no parent or it is unusable
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # start from a uniform vector (the known optimal shape for the bound)
    a = np.full(n, 1.0, dtype=np.float64)          # before clipping
    a = np.clip(a, 0.0, 1000.0)

    # keep the best sequence seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # also consider the supplied parent as a candidate
    parent = np.clip(parent, 0.0, 1000.0)
    sc_parent = float(evaluate_sequence(parent.tolist()))
    if sc_parent < best_score:
        best = parent.copy()
        best_score = sc_parent

    # ------------------------------------------------------------------
    # cheap descent loop – each iteration uses O(n) memory and O(n²) time
    # ------------------------------------------------------------------
    lr = 0.5                # initial step size for the sub‑gradient walk
    it = 0
    while time.monotonic() < deadline:
        # full convolution and the index where it attains its maximum
        conv = np.convolve(a, a)
        idx = int(np.argmax(conv))
        max_val = conv[idx]

        # compute the current bound (fast, no extra clipping needed)
        cur_score = 2.0 * n * max_val / (a.sum() * a.sum())
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # sub‑gradient of the max‑convolution entry
        j = idx - np.arange(n, dtype=np.int64)
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # gradient step, projection onto the non‑negative orthant,
        # and renormalisation to keep the sum fixed (scale does not matter)
        a = a - lr * g
        np.maximum(a, 0.0, out=a)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:                                   # degenerate case – reset to uniform
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # anneal the step size a little
        lr *= 0.9995
        it += 1

        # occasional random restart to avoid possible stagnation
        if it % 500 == 0 and rng.random() < 0.2:
            a = rng.dirichlet(np.ones(n), size=1)[0]

    # ------------------------------------------------------------------
    # final validation (authoritative scorer) and return
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    # the scorer is scale‑invariant; any small numerical deviation is fine
    return [float(x) for x in best]
