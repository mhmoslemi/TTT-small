import numpy as np
import time

def _uniform_sequence(length: int, value: float = 1.0) -> np.ndarray:
    """Return a length‑`length` array filled with `value` (clipped to allowed range)."""
    arr = np.full(length, float(value), dtype=np.float64)
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr


def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The theoretical minimum for the given objective is 2, attained by any
    uniform sequence.  This implementation returns a uniform vector of the
    appropriate length and attempts a tiny local search only if time permits.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine the problem size (length of the sequence).
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        # Fallback length when no parent is supplied.
        n = 256
        base = None

    # Start from a uniform sequence – this attains the known lower bound of 2.
    best = _uniform_sequence(n, 1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # If a parent exists and happens to be better, keep it.
    if base is not None:
        parent_score = float(evaluate_sequence(base.tolist()))
        if parent_score < best_score:
            best = base
            best_score = parent_score

    # Tiny random‑block perturbation loop – it can only improve or keep the
    # score, and stops early if the bound reaches the theoretical optimum.
    while time.monotonic() < deadline:
        # Propose a new candidate by scaling a random block of the current best.
        cand = best.copy()
        width = max(1, n // 120)                # O(n)‑size block
        start = rng.integers(0, n - width + 1)   # block start index
        scale = np.exp(rng.normal(0.0, 0.015))   # mild multiplicative change
        cand[start:start + width] *= scale
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # If we have reached the provable optimum we can stop.
            if best_score <= 2.0 + 1e-12:
                break

    # Return a plain Python list as required.
    return [float(x) for x in best]
