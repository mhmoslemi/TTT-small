def build_set(n, max_repr, rng):
    indices = rng.permutation(n)  # random order
    counts = np.zeros(2*n-1, dtype=np.int16)  # small ints
    selected = []
    for idx in indices:
        # Compute sums with already selected
        if selected:
            sums = idx + np.array(selected, dtype=np.int32)
            # check if any counts exceed limit after increment
            if np.any(counts[sums] >= max_repr):
                continue
        # check self sum
        if counts[2*idx] >= max_repr:
            continue
        # Passed: add idx
        if selected:
            counts[sums] += 1
        counts[2*idx] += 1
        selected.append(int(idx))
    return selected, counts

def _build_set(n, max_per_sum, rng):
    order = rng.permutation(n)
    counts = np.zeros(2*n-1, dtype=np.int16)
    selected = []
    for idx in order:
        # self check
        if counts[2*idx] >= max_per_sum:
            continue
        ok = True
        for s in selected:
            if counts[idx + s] >= max_per_sum:
                ok = False
                break

            continue
        # add

            counts[idx + s] += 1
        counts[2*idx] += 1
        selected.append(int(idx))
    return selected, counts

def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # initial fallback
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception

        parent = np.ones(256, dtype=np.float64)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # store best candidate found
    while time.monotonic() < deadline:
        # Choose max per sum (1..5)

        selected, counts = _build_set(n, max_per, rng)
        m = len(selected)
        if m < 2:
            continue
        B0 = int(counts.max())
        if B0 == 0:
            continue

import time
import numpy as np

# ------------------------------------------------------------
# Helper: construct a set of indices whose unordered sum
# multiplicities never exceed `max_per` (a sparse additive set).
# ------------------------------------------------------------
def _build_sparse_set(n: int, max_per: int, rng: np.random.Generator):
    """Greedy construction of an index set with bounded sum‐multiplicity.

    Returns
    -------
    selected : list[int]
        Chosen indices.
    counts : np.ndarray[int16]
        Unordered pair–sum frequencies (including i+i).
    """
    order = rng.permutation(n)
    counts = np.zeros(2 * n - 1, dtype=np.int16)   # sums range 0 … 2n‑2
    selected = []
    for idx in order:
        # self‑pair check
        if counts[2 * idx] >= max_per:
            continue
        ok = True
        for s in selected:
            if counts[idx + s] >= max_per:
                ok = False
                break
        if not ok:
            continue
        # incorporate the new index
        for s in selected:
            counts[idx + s] += 1
        counts[2 * idx] += 1
        selected.append(int(idx))
    return selected, counts
