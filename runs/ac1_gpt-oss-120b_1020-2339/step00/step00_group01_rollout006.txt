import numpy as np
import time

def _score(seq):
    """Helper to evaluate a sequence using the provided evaluator."""
    # evaluate_sequence is pre‑imported in the execution environment
    return float(evaluate_sequence(seq.tolist()))


def _gen_power_seq(p, n):
    """Generate a non‑negative step sequence a_i = (i+1)^p - i^p."""
    i = np.arange(n, dtype=np.float64)
    # (i+1)^p - i^p  (vectorised)
    a = (i + 1.0) ** p - i ** p
    # clip only to satisfy the evaluator limits (values are far below 1000)
    return np.clip(a, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a low‑score sequence using a simple power‑law family."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # initial candidate (fallback) – use the supplied parent if it exists
    # ----------------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
        best = parent.copy()
        best_score = _score(best)
    except Exception:
        # no valid parent – fall back to a uniform vector of modest length
        n = 256
        best = np.ones(n, dtype=np.float64)
        best_score = _score(best)

    # ----------------------------------------------------------------------
    # sweep a small set of exponents p in (0,1) – the family a_i = (i+1)^p-i^p
    # ----------------------------------------------------------------------
    # Candidate exponents; denser sampling around the region that gave good
    # results in theory (roughly 0.3–0.5).
    p_vals = np.array(
        [0.20, 0.25, 0.30, 0.33, 0.35, 0.37, 0.40, 0.45, 0.50, 0.55, 0.60],
        dtype=np.float64,
    )
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        cand = _gen_power_seq(p, n)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ----------------------------------------------------------------------
    # optional local refinement: small blockwise multiplicative perturbations
    # ----------------------------------------------------------------------
    def make_block_perturbation(base):
        """Return a cheap O(n) perturbation of ``base``."""
        cand = base.copy()
        width = int(rng.integers(max(1, n // 250), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.09)))
        cand[start:stop] *= factor
        # occasional symmetric averaging – harmless and cheap
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, 1000.0)

    # Run the simple hill‑climb while time permits
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    return [float(x) for x in best]
