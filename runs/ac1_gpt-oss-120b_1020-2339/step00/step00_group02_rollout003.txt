def _linear_candidate(length: int) -> np.ndarray:

    return np.linspace(1000.0, 0.0, length, dtype=np.float64)


def _pow_candidate(base: np.ndarray, p: float) -> np.ndarray:

    # base is already in [0,1]; keep the endpoint at 0.
    seq = np.power(base, p) * 1000.0
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq
