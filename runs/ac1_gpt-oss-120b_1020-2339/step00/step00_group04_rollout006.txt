def compute_score(arr):
    a = np.clip(arr, 0.0, 1000.0)
    n = a.shape[0]
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()

        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np
import time

# ------------------------------------------------------------
# Helper: compute the exact score used by the evaluator.
# ------------------------------------------------------------
def _score_vec(vec: np.ndarray) -> float:
    """Exact objective: 2*n*max(conv)/sum^2 with clamping."""
    a = np.clip(vec, 0.0, 1000.0)
    n = a.shape[0]
    b = np.convolve(a, a)          # length 2*n-1
    max_b = float(b.max())
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(max_b):
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Generate a candidate by a small random block perturbation.
# ------------------------------------------------------------
def _perturb_block(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    i0 = int(rng.integers(0, max(1, n - w + 1)))
    i1 = i0 + w
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[i0:i1] *= float(factor)
    if rng.random() < 0.35:                     # slight symmetrisation
        cand[:] = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Generate a candidate from a random Fourier series.
# ------------------------------------------------------------
def _fourier_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector by summing a few cosine modes."""
    modes = rng.integers(2, 6)          # 2‑5 active cosine modes
    coeffs = rng.uniform(-0.4, 0.4, size=modes)  # small random amplitudes
    idx = np.arange(n, dtype=np.float64)
    # start from a constant background (value 1) to keep positivity likely
    vals = np.ones(n, dtype=np.float64)
    for k, c in enumerate(coeffs, start=1):
        vals += c * np.cos(2.0 * np.pi * k * idx / n)
    vals = np.clip(vals, 0.0, 1000.0)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    # normalise – the absolute scale does not affect the score
    vals *= 1.0 / vals.sum()
    return vals

        convolution term.

    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.5)   # reserve a little slack

    # ------------------------------------------------------------------
    # Initial vector: use the provided parent if it looks valid,
    # otherwise fall back to a uniform step.
    # ------------------------------------------------------------------
    try:

    except NameError:

        n = cur.size
        # Normalise to a probability vector (the absolute scale does not matter)

    # Ensure the current vector respects the evaluator's clamping rules

    iteration = 0

    return [float(v) for v in out]
