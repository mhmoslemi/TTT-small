def max_convolution(seq):
    # Compute full convolution using FFT (real)
    n = seq.size
    size = 1 << (2 * n - 1).bit_length()  # next power of two > 2n-1
    # but we can just use n*2 maybe.
    # Using np.fft.rfft for real transforms.
    f = np.fft.rfft(seq, n*2)
    conv = np.fft.irfft(f * f, n*2)
    # The full convolution length is 2n-1. conv[:2*n-1] corresponds.
    return np.max(conv[:2*n-1])

import time
import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- determine length -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 256                       # fallback length if parent missing

    # ---- initial fallback (uniform) ---------------------------------------
    best_seq = np.ones(n, dtype=np.float64)          # uniform sequence → score = 2
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ---- try the supplied parent if it exists ------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(base, 0.0, 1000.0)
        sc = float(evaluate_sequence(parent.tolist()))
        if sc < best_score:
            best_seq = parent.copy()
            best_score = sc

    target = 1.50274349

    # ---- random subset search ---------------------------------------------
    # we use binary values (0/1); scaling cancels out in the score.
    min_m = max(30, int(0.03 * n))      # at least a few entries
    max_m = min(n, int(0.55 * n))       # not too dense (helps keep max_b low)

    # pre‑allocate a zero vector we can reuse (avoids repeated allocations of the same size)
    zero_vec = np.zeros(n, dtype=np.float64)

    while time.monotonic() < deadline:
        m = rng.integers(min_m, max_m + 1)               # number of non‑zero positions
        idx = rng.choice(n, size=m, replace=False)      # random distinct indices
        seq = zero_vec.copy()
        seq[idx] = 1.0                                   # binary amplitude

        # fast estimate using FFT (linear convolution)
        fft_len = n * 2
        f = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(f * f, fft_len)
        max_b = float(np.max(conv[: 2 * n - 1]))
        sum_a = float(m)                                 # because entries are 0/1
        if sum_a < 0.01:                                 # degenerate – shouldn't happen
            continue
        est_score = 2.0 * n * max_b / (sum_a * sum_a)

        # only call the authoritative evaluator if we have a chance to improve
        if est_score < best_score:
            real_score = float(evaluate_sequence(seq.tolist()))
            if real_score < best_score:
                best_seq = seq.copy()
                best_score = real_score
                if best_score < target:
                    break

    # return as plain Python floats (list)
    return [float(v) for v in best_seq]
