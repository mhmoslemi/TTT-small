import time
import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Heuristic search for a non‑negative step function that lowers the
    AC1 upper bound 2·n·max(conv)/sum².
    The method starts from a (nearly) uniform sequence, adds a tiny random
    perturbation to break symmetry and then repeatedly moves a small amount of
    mass from the mirror‑pair with the largest product to the pair with the
    smallest product.  Improvements are kept, and the best vector found before
    the time deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length n and initialise a vanilla sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    else:
        n = 256                     # fallback length
    # start from a uniform vector (sum = n) and add a tiny noise
    a = np.full(n, 1.0, dtype=np.float64)
    a += rng.normal(loc=0.0, scale=0.05, size=n)   # break symmetry
    a = np.clip(a, 0.0, None)
    a *= (n / a.sum())                            # keep sum = n

    # keep track of the best solution seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = a.copy()
    cur_score = best_score

    # time management
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 1. Compute products of mirrored pairs (i, n‑1‑i)
        # ------------------------------------------------------------------
        half = n // 2
        i_idx = np.arange(half, dtype=np.int64)
        j_idx = n - 1 - i_idx
        prod = cur[i_idx] * cur[j_idx]

        # include middle element when n is odd
        if n % 2:
            mid = half
            i_idx = np.append(i_idx, mid)
            j_idx = np.append(j_idx, mid)
            prod = np.append(prod, cur[mid] * cur[mid])

        # ------------------------------------------------------------------
        # 2. Find the pair with maximal and minimal product
        # ------------------------------------------------------------------
        max_pos = int(prod.argmax())
        min_pos = int(prod.argmin())

        # if all products are equal we perform a tiny random nudge
        if max_pos == min_pos:
            cur += rng.normal(scale=0.01, size=n)
            cur = np.clip(cur, 0.0, None)
            cur *= (n / cur.sum())
            cur_score = float(evaluate_sequence(cur.tolist()))
            continue

        i_hi, j_hi = int(i_idx[max_pos]), int(j_idx[max_pos])
        i_lo, j_lo = int(i_idx[min_pos]), int(j_idx[min_pos])

        # ------------------------------------------------------------------
        # 3. Transfer a small amount of mass from the high‑product pair to
        #    the low‑product pair (preserving the total sum)
        # ------------------------------------------------------------------
        # fraction of the smaller entry of the high‑product pair to move
        frac = rng.uniform(0.06, 0.18)
        delta = min(cur[i_hi], cur[j_hi]) * frac
        if delta <= 0.0:
            # shouldn't happen, but guard against stagnation
            cur += rng.normal(scale=0.01, size=n)
            cur = np.clip(cur, 0.0, None)
            cur *= (n / cur.sum())
            cur_score = float(evaluate_sequence(cur.tolist()))
            continue

        cand = cur.copy()
        cand[i_hi] -= delta
        cand[j_hi] -= delta
        cand[i_lo] += delta
        cand[j_lo] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)   # safety clip

        # ------------------------------------------------------------------
        # 4. Evaluate the candidate
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best = cand.copy()
                best_score = score
            if score < cur_score:

                cur_score = score
            else:
