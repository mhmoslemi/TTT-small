def compute_gradient(a, k):
    n = a.shape[0]
    i = np.arange(n)
    j = k - i
    valid = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    # diagonal:
    diag = valid & (i == j)
    grad[diag] = a[i[diag]]
    # off-diagonal:
    off = valid & (~diag)
    grad[off] = 2 * a[j[off]]
    return grad

def full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x using a power‑of‑two FFT (O(n log n))."""
    n = x.shape[0]
    L = 2 * n - 1
    nfft = 1 << ((L - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, nfft)
    conv = np.fft.irfft(fx * fx, nfft)
    return conv[:L]                              # exact linear convolution


def gradient_max_conv(x: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max_k (x*x)[k] w.r.t. x (projected onto non‑negative orthant)."""
    n = x.shape[0]
    i = np.arange(n)
    j = k - i

        grad[diag] = x[i[diag]]
    off = valid & (~diag)
    if np.any(off):
        grad[off] = 2.0 * x[j[off]]

    # keep the best candidate seen so far (exact evaluation)
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    # approximate best max for fast comparisons
    conv = full_convolution_fft(a)
    best_max = float(conv.max())

    # parameters for the two‑phase hill‑climb
    step = 0.5                     # initial gradient step size
    min_step = 1e-6
    max_width = max(2, n // 12)    # limit of random block width

    while time.monotonic() < deadline:
        # ----- 1) gradient step on the current worst convolution entry ---- #
        conv = full_convolution_fft(a)
        cur_max = float(conv.max())
        k = int(np.argmax(conv))

        grad = gradient_max_conv(a, k)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:                           # pathological, re‑randomise
            a_new = a + 1e-3 * rng.random(n)
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
        a_new = a_new / s
        conv_new = full_convolution_fft(a_new)
        new_max = float(conv_new.max())

        if new_max < cur_max:                  # accept improvement
            a = a_new
            step = min(step * 1.07, 5.0)        # modestly enlarge step
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
        else:                                   # reject – shrink step
            step = step * 0.52
            if step < min_step:
                # random jitter to escape potential stall
                a = best_a.copy()
                a += 0.001 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                step = 0.5

        # ----- 2) cheap random block perturbation (hill‑climb) ----------- #
        # aim to discover improvements that gradient may miss
        width = int(rng.integers(max(1, n // 200), max_width))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand = a.copy()
        cand[start:start + width] *= factor
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])   # optional symmetrisation
        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum()
        cand_max = float(full_convolution_fft(cand).max())
        if cand_max < cur_max:
            a = cand
            cur_max = cand_max
            if cand_max < best_max:
                best_max = cand_max
                best_a = cand.copy()

    # ----- final exact evaluation -------------------------------------- #
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score):
        # fallback to the last valid candidate
        final_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(final_score):
            best_a = a.copy()

    # enforce evaluator limits (non‑negative, ≤1000) and return list
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
