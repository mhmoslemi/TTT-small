        if parent_arr.ndim == 1 and parent_arr.size == n and np.isfinite(parent_arr).all():
            parent_norm = parent_arr / parent_arr.sum()
            maybe_update(parent_norm)
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # FFT‑based convolution (matrix‑free, O(n log n))
    # ----------------------------------------------------------------------
    max_len = 2 * n - 1
    fft_size = 1 << int(math.ceil(math.log2(max_len)))

    def conv_fft(x):
        fx = np.fft.rfft(x, fft_size)
        c = np.fft.irfft(fx * fx, fft_size)[:max_len]
        c[c < 0] = 0.0          # tiny negatives from rounding
        return c

    # ----------------------------------------------------------------------
    # Build Sidon sets by greedy insertion using random orders.
    # Keep the best (largest) one found within the time budget.
    # ----------------------------------------------------------------------
    attempt = 0
    while time.monotonic() < deadline and attempt < 300:
        attempt += 1
        order = rng.permutation(n)          # random order of indices
        used = np.zeros(max_len, dtype=bool)   # which i+j sums are already taken
        S = []                                 # selected indices

        for i in order:
            # self‑sum must be free
            ss = i + i
            if used[ss]:
                continue
            if S:
                sums = i + np.array(S, dtype=int)
                if used[sums].any():
                    continue
            # accept i
            S.append(int(i))
            used[ss] = True
            if len(S) > 1:
                prev = np.array(S[:-1], dtype=int)
                used[i + prev] = True

        m = len(S)
        if m < 2:
            continue

        seq = np.zeros(n, dtype=np.float64)
        seq[S] = 1.0 / m                # equal mass on the Sidon indices
        maybe_update(seq)

        # early exit if we already have a very good bound
        if best_score <= 1.02:
            break

    # ----------------------------------------------------------------------
    # Optional local refinement: projected gradient descent with occasional
    # random perturbations.  This works on the current best sequence.
    # ----------------------------------------------------------------------
    cur = best_seq.copy()
    while time.monotonic() < deadline:
        b = conv_fft(cur)
        max_b = b.max()
        k_star = int(b.argmax())

        idx = k_star - np.arange(n)
        rev = np.zeros_like(cur)
        mask = (idx >= 0) & (idx < n)
        rev[mask] = cur[idx[mask]]

        sum_a = cur.sum()          # should be 1 after normalisation
        grad = (4.0 * n / (sum_a * sum_a)) * (rev - max_b / sum_a)

        improved = False
        for eta in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = cur - eta * grad
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand = cand / s
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                cur = cand
                improved = True
                break
        if not improved:
            # small random block perturbation (fallback move)
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor
            if rng.random() < 0.3:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand = cand / s
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                cur = cand

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats)
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
```
