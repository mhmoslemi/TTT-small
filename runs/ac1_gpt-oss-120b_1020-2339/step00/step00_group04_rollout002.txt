import numpy as np, time

def _sidon_set(n, rng, deadline, max_trials=20):
    # compute theoretical max size
    max_possible = int(np.floor((-1 + np.sqrt(16 * n - 7)) / 2))
    best = []
    best_len = 0
    trials = 0
    while time.monotonic() < deadline and trials < max_trials:
        trials += 1
        order = rng.permutation(n)
        used = set()
        cur = []
        for i in order:
            # early exit if remaining positions insufficient to surpass best (optional)
            # Check for conflict
            conflict = False
            for j in cur:
                if (i + j) in used:
                    conflict = True
                    break
            if conflict or (2 * i) in used:
                continue
            # add i
            for j in cur:
                used.add(i + j)
            used.add(2 * i)
            cur.append(i)
        if len(cur) > best_len:
            best = cur
            best_len = len(cur)
            if best_len >= max_possible:
                break
    return best, max_possible
```

import numpy as np
import time
import math

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Construct a sequence of non‑negative heights that achieves a low
    AC1 upper bound.  The idea is to place a few entries at the maximal
    allowed value (1000) on a Sidon set of indices – i.e. a set where all
    i+j (including i+i) are distinct.  Then every convolution offset
    contains at most one product 1000·1000 (twice for i≠j), so the maximal
    autocorrelation value is 2·1000².  With enough such entries the score
    2·n·max_b / (sum_a)² drops well below the target 1.5027.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # length of the problem instance
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    # theoretical maximal size of a Sidon set in [0, n‑1]
    max_possible = int(math.floor((-1.0 + math.sqrt(16.0 * n - 7.0)) / 2.0))

    best_set = []
    best_len = 0

    # try many random orderings; keep the largest Sidon set found
    while time.monotonic() < deadline:
        order = rng.permutation(n)          # random permutation of 0..n‑1
        used = np.zeros(2 * n - 1, dtype=bool)  # used[i+j] flags
        cur = []
        # greedy insertion respecting the Sidon condition
        for i in order:
            if used[i + i]:
                continue
            conflict = False
            for j in cur:
                if used[i + j]:
                    conflict = True
                    break
            if conflict:
                continue
            # add i to the set
            for j in cur:
                used[i + j] = True
            used[i + i] = True
            cur.append(i)
        if len(cur) > best_len:
            best_len = len(cur)
            best_set = cur
            if best_len >= max_possible:       # cannot be improved further
                break

    # if for some reason no indices were selected, fall back to a simple parent
    if not best_set:
        try:
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
            if (parent.ndim == 1 and parent.size > 0
                    and np.isfinite(parent).all()):
                return [float(x) for x in parent]
        except Exception:
            pass
        return [1000.0] * n

    # build the final sequence: 1000 on the Sidon indices, 0 elsewhere
    seq = np.zeros(n, dtype=np.float64)
    seq[best_set] = 1000.0

    # final sanity check – the evaluator should never return inf here
    if not np.isfinite(evaluate_sequence(seq.tolist())):
        seq = np.full(n, 1000.0, dtype=np.float64)

    return [float(x) for x in seq]
