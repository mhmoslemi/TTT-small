import numpy as np
import time

def _make_block_perturbation(base, rng):
    """Multiplicative perturbation of a random contiguous block."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between 1% and ~10% of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal noise
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetric averaging
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Search for a non‑negative step sequence with a low AC1 upper bound.
    Uses a quick parametric scan (geometric / power families) followed by
    cheap block‑wise stochastic refinements.
    Returns the best sequence discovered as a list of floats.
    """
    rng = np.random.default_rng(seed)
    # leave a small margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Base sequence (fallback to uniform if the provided parent is unusable)
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Helper to evaluate and possibly replace the incumbent best.
    # ------------------------------------------------------------------
    def _maybe_update(arr):
        nonlocal best, best_score
        cand = np.clip(np.asarray(arr, dtype=np.float64), 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # 1) Coarse parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric a_i = r^i
    for r in np.linspace(0.99, 0.99995, 180):
        if time.monotonic() >= deadline:
            break
        a = r ** idx
        a /= np.max(a)                     # scale to keep max ≤ 1
        _maybe_update(a)

    # decreasing power law a_i = ((n-i)/n)^p
    for p in np.linspace(0.2, 5.0, 140):
        if time.monotonic() >= deadline:
            break
        a = ((n - idx) / n) ** p
        a /= np.max(a)
        _maybe_update(a)

    # combined geometric * power law
    for r in np.linspace(0.995, 0.9999, 30):
        if time.monotonic() >= deadline:
            break
        for p in np.linspace(0.5, 3.0, 25):
            if time.monotonic() >= deadline:
                break
            a = (r ** idx) * (((n - idx) / n) ** p)
            a /= np.max(a)
            _maybe_update(a)

    # ------------------------------------------------------------------
    # 2) Stochastic fine‑tuning using block perturbations.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # return primitive Python list as required
    return [float(x) for x in best]
