import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Construct a decreasing geometric sequence whose self‑convolution peak
    is provably small.  The length matches the provided ``height_sequence_1``
    when available, otherwise falls back to 256.
    """
    # ------------------------------------------------------------
    # Determine the length to use (avoid copying the parent values)
    # ------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = int(parent_arr.size)
    except Exception:
        n = 256  # safe default when the parent is absent

    # ------------------------------------------------------------
    # Build the geometric progression.
    # ------------------------------------------------------------
    if n == 1:
        seq = np.array([1.0])
    else:
        # Choose r so that the central convolution value equals the edge value:
        #   n * r^{n-1} = 1   =>   r = (1/n)^{1/(n-1)}
        # Using log/exp for numerical stability.
        r = np.exp(np.log(1.0 / n) / (n - 1))
        idx = np.arange(n, dtype=np.float64)
        seq = np.exp(idx * np.log(r))            # equivalent to r**idx
        # Enforce the evaluator's clipping range.
        seq = np.clip(seq, 0.0, 1000.0)

    # ------------------------------------------------------------
    # Return a plain Python list of floats.
    # ------------------------------------------------------------
    return [float(x) for x in seq]
