def project_to_simplex(v):
    # Ensure nonnegative and sum to 1.
    # Implementation: sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho = max i such that u[i] - cssv[i]/i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time, math
    
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = 6733  # Could use length of height_sequence_1 or set but we can use its length if defined
    # Determine length:
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = base.size
    except NameError:
        n = 6733
    best = None
    best_score = math.inf

    # Helper functions
    def max_conv_and_idx(a):
        size = 1 << ((2*n - 1).bit_length())
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:2*n-1]
        idx = int(conv.argmax())
        return float(conv[idx]), idx, conv

    def evaluate(a):
        # evaluate using fast version for speed: compute exact via evaluate_sequence at end
        M, _, _ = max_conv_and_idx(a)
        s = a.sum()
        return 2.0 * n * M / (s * s)

    # start with uniform as fallback
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = uniform.copy()
    best_score = evaluate(best)

    # also include original parent if exists
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.size == n:
            # normalize to sum 1
            if parent.sum() > 0:
                parent_norm = parent / parent.sum()
                score_parent = evaluate(parent_norm)
                if score_parent < best_score:
                    best = parent_norm.copy()
                    best_score = score_parent
    except Exception:
        pass

    # iterations
    while time.monotonic() < deadline:
        # random init
        # draw Dirichlet with concentration param maybe 0.5 to get varied distribution
        alpha = 0.5 + rng.random() * 3.0   # between 0.5 and 3.5
        a = rng.dirichlet(np.full(n, alpha))
        # local optimization
        # run inner steps limited by time
        inner_deadline = time.monotonic() + 0.4  # allocate small budget per restart
        while time.monotonic() < inner_deadline:
            M, kp, conv = max_conv_and_idx(a)
            # Compute gradient
            g = np.zeros_like(a)
            i_start = max(0, kp - (n - 1))
            i_end = min(n - 1, kp)
            if i_start <= i_end:
                I = np.arange(i_start, i_end + 1)
                J = kp - I
                g[I] = 2.0 * a[J]
            # step size
            # norm2 = np.dot(g, g)
            # if norm2 == 0: break
            # eta = 0.5 * M / (norm2 + 1e-12)
            eta = 0.5 * M / (np.sum(g) + 1e-12)  # simpler scaling
            a_new = a - eta * g
            # enforce nonnegativity
            np.maximum(a_new, 0, out=a_new)
            s = a_new.sum()
            if s <= 0:
                break
            a = a_new / s
            # evaluate quick score; if better than best, update best
            score = 2.0 * n * M / (s * s)  # approximated same as evaluate
            if score < best_score:
                best = a.copy()
                best_score = score
            # optional random jitter to escape local minima
            if rng.random() < 0.02:
                # add small random perturbation to a few entries
                idxs = rng.choice(n, size=int(0.01*n)+1, replace=False)
                perturb = rng.normal(0.0, 0.1, size=idxs.shape)
                a[idxs] = np.clip(a[idxs] + perturb, 0, None)
                a /= a.sum()
        # after inner loop, fallback to uniform maybe
    # Return best after scaling back to appropriate magnitude (e.g., multiply by 1)
    # Ensure values within [0,1000]
    best = np.clip(best, 0.0, 1000.0)
    # Scale to have sum >0.01, not zero; currently sum=1
    return best.tolist()

def propose_candidate(...):
    import numpy as np, time, math
    
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Determine n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size if base.ndim == 1 and base.size > 0 else 6733
    except Exception:
        n = 6733

    # fast convolution using fft
    size_fft = 1 << ((2*n - 1).bit_length())
    def max_conv(a):
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[:2*n-1]
        idx = int(conv.argmax())
        return conv[idx], idx, conv

    # function to compute score quickly
    def quick_score(a, M):
        s = a.sum()
        return 2.0 * n * M / (s*s)

    best = np.full(n, 1.0/n, dtype=np.float64)
    M, _, _ = max_conv(best)
    best_score = quick_score(best, M)

    # also incorporate parent if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            parent_norm = parent / max(parent.sum(), 1e-12)
            M2, _, _ = max_conv(parent_norm)
            sc = quick_score(parent_norm, M2)
            if sc < best_score:
                best = parent_norm.copy()
                best_score = sc
    except Exception:
        pass

    # main loop
    while time.monotonic() < deadline:
        # Random start Dirichlet
        alpha = 0.5 + rng.random() * 4.0   # 0.5 to 4.5
        a = rng.dirichlet(np.full(n, alpha))
        # ensure a not too close to zero
        # local refinement steps
        inner_deadline = time.monotonic() + 0.3  # 300ms per restart
        while time.monotonic() < inner_deadline:
            M, kp, _ = max_conv(a)
            # gradient w.r.t a for this kp
            g = np.zeros_like(a)
            i_start = max(0, kp - (n-1))
            i_end = min(n-1, kp)
            if i_start <= i_end:
                I = np.arange(i_start, i_end+1)
                J = kp - I
                g[I] = 2.0 * a[J]
            # choose step
            # compute step scaling: we want to reduce M by a fraction, approximate
            # Use line search: try a decreasing step
            step = 0.5 * M / (g.dot(g) + 1e-12)
            a_new = a - step * g
            # Clip, renorm
            np.maximum(a_new, 0, out=a_new)
            s = a_new.sum()
            if s <= 1e-12:
                break
            a = a_new / s
            # quick score
            score = 2.0 * n * M / (s*s)
            if score < best_score:
                best = a.copy()
                best_score = score
            # occasionally random jitter to escape
            if rng.random() < 0.01:
                idxs = rng.choice(n, size=max(1, int(0.005*n)), replace=False)
                a[idxs] = np.clip(a[idxs] + rng.normal(0, 0.02, size=idxs.shape), 0, None)
                a /= a.sum()
        # maybe break if best_score already low enough
        if best_score <= 1.50274349:
            break

    # final scaling: we can keep sum=1; scaling does not affect ratio, but ensure values <1000.
    best = np.clip(best, 0.0, 1000.0)
    # multiply by a factor to ensure sum not too tiny (though sum =1)
    return best.tolist()

                # keep numerical stability
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # --- occasional random jitter to escape flat regions -----------------
        if rng.random() < jitter_prob:
            # pick a small set of indices and shuffle a tiny amount of mass
            k = max(2, int(0.001 * n))
            idx = rng.choice(n, size=k, replace=False)
            mass = a[idx].copy()
            rng.shuffle(mass)
            a[idx] = mass
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # keep the best ever seen
        if cur_M < M:
            M = cur_M
            best_a = a.copy()

    # ------------------------------------------------------------------
    # Prepare final answer – enforce evaluator limits
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    # Normalisation is irrelevant for the score (scale‑invariant), but keep sum>0
    if best_a.sum() < 1e-12:
        best_a = uniform.copy()
    return best_a.tolist()
