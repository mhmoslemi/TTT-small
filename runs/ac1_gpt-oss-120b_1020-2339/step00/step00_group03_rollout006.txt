import time, math
import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Generate a non‑negative step‑function minimizing the AC1 upper bound.
    Uses a compact parametric family (exponential, linear, quadratic) and
    a short local search.  The result is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----------------------------------------------------------------------
    # Determine length n (use the provided parent if available)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = base.size
        else:
            n = 256
    else:
        n = 256

    # ----------------------------------------------------------------------
    # Helper to evaluate a candidate (authoritative evaluator)
    # ----------------------------------------------------------------------
    def eval_seq(arr):
        # evaluator expects a plain list
        return float(evaluate_sequence(arr.tolist()))

    # ----------------------------------------------------------------------
    # Basic candidates (uniform, linear, quadratic, exponential)
    # ----------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64) / n          # uniform, sum=1
    best_score = eval_seq(best_seq)

    # linear decreasing: a_i ∝ (n-i)
    lin = (np.arange(n, dtype=np.float64)[::-1] + 1)
    lin = lin / lin.sum()
    lin_score = eval_seq(lin)
    if lin_score < best_score:
        best_seq, best_score = lin, lin_score

    # quadratic decreasing: a_i ∝ (n-i)^2
    quad = (np.arange(n, dtype=np.float64)[::-1] + 1) ** 2
    quad = quad / quad.sum()
    quad_score = eval_seq(quad)
    if quad_score < best_score:
        best_seq, best_score = quad, quad_score

    # exponential family: a_i = C * r^i  (r = exp(-c/n)), sum normalized to 1
    def gen_exp(c):
        # r = exp(-c/n), r^n = exp(-c)
        r = math.exp(-c / n)
        r_n = math.exp(-c)               # exactly r**n
        C = (1.0 - r) / (1.0 - r_n)      # scaling so that sum = 1
        return C * (r ** np.arange(n, dtype=np.float64))

    # try several decay parameters
    for c in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        cand = gen_exp(c)
        sc = eval_seq(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
        # also the mirrored version (mass at the end instead of the start)
        cand_rev = cand[::-1]
        sc_rev = eval_seq(cand_rev)
        if sc_rev < best_score:
            best_seq, best_score = cand_rev, sc_rev

    # ----------------------------------------------------------------------
    # Light local search: random multiplicative block perturbations
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        # pick a random contiguous block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # apply a small log‑normal scaling to the block
        scale = math.exp(rng.normal(0.0, 0.06))
        cand[start:stop] *= scale
        # optional symmetric blending (helps shape the tail)
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        # enforce bounds required by the evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = eval_seq(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # ----------------------------------------------------------------------
    # Return the best found sequence as a plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
