def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # get parent if defined
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise Exception()
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)
    parent = np.clip(parent_arr, 0.0, 1000.0)
    best_arr = parent.copy()
    best_score = float(evaluate_sequence(best_arr.tolist()))
    n = len(parent)

    # constants
    V = 1000.0
    K_max = 2   # max unordered pair per sum

    # Function to construct candidate set
    def build_candidate():
        # random permutation
        order = np.arange(n)
        rng.shuffle(order)
        sum_counts = np.zeros(2*n-1, dtype=np.int16)
        selected = []
        for i in order:
            # check feasibility
            feasible = True
            # cross sums with existing
            for j in selected:
                s = i + j
                if sum_counts[s] >= K_max:   # adding would exceed
                    feasible = False
                    break
            if not feasible:
                continue
            # self sum
            if sum_counts[2*i] >= K_max:   # but should be 0
                continue
            # Accept
            selected.append(i)
            sum_counts[2*i] += 1
            for j in selected[:-1]:
                sum_counts[i + j] += 1
        # build array
        a = np.zeros(n, dtype=np.float64)
        if selected:
            a[selected] = V
        return a, len(selected)

    # Attempt many candidates
    while time.monotonic() < deadline:
        candidate_arr, m = build_candidate()
        # ensure nonzero sum
        if candidate_arr.sum() < 0.01:
            continue
        # evaluate

import numpy as np, time

def _build_sparse_sequence(n, K_max, rng):
    """Return a sequence of length n with values 1000 at selected indices.
    The selection is a greedy random construction respecting that each unordered
    pair of indices (i,j) with i<=j contributes at most K_max to the
    sum_counts array of i+j."""
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)
    # count of unordered pairs (including self) that map to each sum
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    selected = []
    add = selected.append
    for i in order:
        # fast reject if already full (optional)
        # Check all pair sums with current selected set
        ok = True
        # Self-sum check (always ok because count is 0 before insertion)
        if sum_counts[2 * i] >= K_max:
            continue
        for j in selected:
            s = i + j
            if sum_counts[s] >= K_max:
                ok = False
                break
        if not ok:
            continue
        # Accept i
        add(i)
        sum_counts[2 * i] += 1  # self pair
        for j in selected[:-1]:
            sum_counts[i + j] += 1
    seq = np.zeros(n, dtype=np.float64)
    if selected:
        seq[selected] = 1000.0
    return seq, len(selected)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a sparse step function with low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # obtain parent if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)
    parent = np.clip(base, 0.0, 1000.0)
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    n = len(parent)

    # parameters for the construction
    K_MAX = 3          # allow at most K_MAX unordered pairs per sum
    MIN_M_FOR_GOAL = 150  # heuristic threshold

    while time.monotonic() < deadline:
        # build a sparse candidate
        cand_seq, m = _build_sparse_sequence(n, K_MAX, rng)
        if m < 1:
            continue
        # quick theoretical bound: if m is too small, skip evaluation
        # theoretical_score ≈ 4 * n * K_MAX / m**2
        # if this bound already exceeds current best, we can skip
        # using a simple estimate to save time
        # (This bound is exact when all values are equal.) 
        if best_score is not None and (4.0 * n * K_MAX) / (m * m) >= best_score:
            # no chance to improve
            continue

            best_seq = cand_seq.copy()
            # stopping early if we beat the target
            if best_score < 1.5028:
                break

    # return as plain Python list
    return [float(v) for v in best_seq]

import numpy as np
import time

def _build_sparse_sequence(n: int, K_max: int, rng: np.random.Generator):
    """
    Greedy random construction of a set S ⊂ {0,…,n‑1} such that each unordered
    pair (i,j) with i≤j contributes at most K_max to the count of the sum i+j.
    The returned sequence has value 1000 at the selected indices and 0 elsewhere.
    """
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)

    # number of unordered pairs (including self‑pairs) that map to each possible sum
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    selected = []
    for i in order:
        # self‑pair check
        if sum_counts[2 * i] >= K_max:
            continue

        # check all cross‑pairs with already selected indices
        ok = True
        for j in selected:
            s = i + j
            if sum_counts[s] >= K_max:
                ok = False
                break
        if not ok:
            continue

        # accept i
        selected.append(i)
        sum_counts[2 * i] += 1                     # self‑pair (i,i)
        for j in selected[:-1]:                     # cross‑pairs (i,j)
            sum_counts[i + j] += 1

    seq = np.zeros(n, dtype=np.float64)
    if selected:
        seq[selected] = 1000.0
    return seq, len(selected)
