            sum_set.add(p + q)
        # optional early stop when size reaches sqrt(2*n) maybe

    return positions

import time
import numpy as np

def _generate_sidon_positions(n: int, rng: np.random.Generator) -> list:
    """
    Greedy construction of a Sidon set (all pairwise sums are distinct).
    Positions 0 and n-1 are avoided to keep endpoint squares small.
    """
    # random order improves chance of a larger set
    cand = list(range(1, n - 1))
    rng.shuffle(cand)

    pos = []               # accepted positions
    sums = set()           # seen sums a+b (including a+a)
    pos_set = set()        # for quick symmetry check

    for p in cand:
        # do not allow both a position and its mirror about the centre
        if (n - 1 - p) in pos_set:
            continue

        # check whether any sum would repeat
        conflict = False
        s_self = p + p
        if s_self in sums:
            conflict = True
        else:
            for q in pos:
                if p + q in sums:
                    conflict = True
                    break

        if conflict:
            continue

        # accept this position
        pos.append(p)
        pos_set.add(p)

        # record new sums (including the self‑sum)
        sums.add(s_self)
        for q in pos[:-1]:
            sums.add(p + q)

    return pos
