    except Exception:

    n = base_arr.size

    max_val = 1000.0  # use maximum allowed height

    # Linear decreasing shape
    if n == 1:
        candidate = np.array([max_val], dtype=np.float64)
    else:
        idx = np.arange(n, dtype=np.float64)
        candidate = max_val * (n - 1 - idx) / (n - 1)

    # clip bounds and ensure finite
    candidate = np.clip(candidate, 0.0, max_val)

    best_candidate = candidate

    parent = np.clip(base_arr, 0.0, max_val)
    parent_score = float(evaluate_sequence(parent.tolist()))
    if parent_score < best_score:
        best_candidate = parent
        best_score = parent_score

    # Simple local random perturbations for remaining time (optional)
    # Use O(n) memory per candidate
    while time.monotonic() < deadline:
        # propose a perturbation: multiply a random contiguous block by random factor near 1
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, n - width + 1))

        if np.isfinite(score) and score < best_score:

            best_score = score
