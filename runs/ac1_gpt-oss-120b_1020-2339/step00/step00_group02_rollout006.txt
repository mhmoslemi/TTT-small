    # maybe precompute plan to avoid reallocation? Not necessary.
    
    # Learning rate schedule
    lr = 0.5   # step for gradient after scaling factor (units of coefficient)
    # Actually we define step scaling factor after computing gradient.
    
    # optionally implement momentum, not needed.

    while time.monotonic() < deadline:

        # pad a to length L
        A = np.fft.rfft(a, n=L)
        c_full = np.fft.irfft(A * A, n=L)[:2*n-1]
        # get max and index
        m = c_full.max()
        k = int(np.argmax(c_full))
        # compute gradient g
        g = np.zeros_like(a)
        start = max(0, k - n + 1)
        end = min(k+1, n)
        if start < end:

            i = k - idx

def generate_sidon_set(n, target_m=None):
    sum_set = set()
    positions = []
    for i in range(n):
        # check if i can be added
        ok = True
        # check sum with itself
        if (i + i) in sum_set:

                    ok = False
                    break
        if ok:
            positions.append(i)

            for p in positions[:-1]: # previous ones
                sum_set.add(i + p)
            # optionally break if target reached

                break
    return positions

    # Determine length n

        if base.ndim != 1 or base.size == 0:

        n = base.size
    except Exception:
        n = 256  # fallback size
    # define function to compute objective f for given r
    # compute multiplicative overlap lengths L
    ks = np.arange(2*n - 1, dtype=np.float64)
    L = np.minimum(ks + 1, 2*n - 1 - ks)  # shape (2n-1,)

    best_score = math.inf
    best_seq = None

    # try several alphas; include previous parent as fallback
    alphas = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 6.0, 8.0]
    for alpha in alphas:

            break

            continue

        if r <= 0.0:

        idx = np.arange(n, dtype=np.float64)
        a = np.power(r, idx)
