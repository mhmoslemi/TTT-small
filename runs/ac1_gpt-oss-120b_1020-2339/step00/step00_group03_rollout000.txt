import time
import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Return a simple uniform (constant) non‑negative sequence.
    The constant value is chosen so that the total sum matches that of
    ``height_sequence_1`` when it is available, otherwise a safe default is used.
    This yields the theoretical lower bound 2.0 for the AC1 upper bound.
    """
    # Determine target length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        if n == 0:
            raise ValueError
    except Exception:
        # Fallback length if the parent sequence is unavailable
        n = 256
        base = None

    # Choose a constant value.
    # If we have a reference sequence, keep the total sum the same.
    if base is not None:
        const_val = float(np.mean(base))
    else:
        const_val = 1.0

    # Enforce non‑negativity and the evaluator's clipping range
    const_val = max(0.0, min(1000.0, const_val))

    seq = np.full(n, const_val, dtype=np.float64)

    # Ensure the sum is above the evaluator's minimum (0.01)
    if seq.sum() < 0.01:
        # Scale up minimally while staying within bounds
        scale = 0.02 / seq.sum()
        seq = np.clip(seq * scale, 0.0, 1000.0)

    # Convert to plain Python list as required by the evaluator
    return [float(x) for x in seq]
