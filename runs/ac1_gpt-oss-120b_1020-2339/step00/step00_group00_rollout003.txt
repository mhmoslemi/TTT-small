# fast evaluation of the AC1 bound (exact up to floating error)
def _fast_ac1(a: np.ndarray, size: int) -> float:
    n = a.size
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _gradient(a: np.ndarray, k: int) -> np.ndarray:
    """sub‑gradient of the convolution entry with index k w.r.t. a"""
    n = a.size
    g = np.zeros_like(a)
    if k < n:
        # indices 0..k participate
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        # the tail part (k >= n)
        start = k - (n - 1)
        length = 2 * n - 1 - k
        g[start:] = 2.0 * a[:length][::-1]
    return g
