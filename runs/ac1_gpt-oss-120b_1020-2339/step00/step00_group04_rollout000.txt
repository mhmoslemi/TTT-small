def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng= np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-0.5)
    # Determine length
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256
    # Uniform candidate
    base = np.full(n, 1.0, dtype=np.float64)  # any constant >0
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # simple random local search optional
    while time.monotonic() < deadline:
        # randomly perturb a small block
        width = rng.integers(1, max(2, n//20))
        start = rng.integers(0, max(0, n-width+1))
        candidate = best.copy()
        factor = np.exp(rng.normal(0.0, 0.1))
        candidate[start:start+width] *= factor
        # clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            # If we find better than 2 (unlikely), break
            if best_score < 1.8:
                break
    return [float(x) for x in best]

    The algorithm starts from a uniform sequence (which is optimal for the
    objective) and performs cheap random block perturbations while the time
    budget allows, keeping the best candidate.
    """
    rng = np.random.default_rng(seed)
    # reserve a small safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine the length to use.  If the parent construction ``height_sequence_1``

    # Uniform sequence is provably optimal for the objective:
    #   score = 2 * n * sum_i a_i^2 / (sum_i a_i)^2  >= 2,
    # with equality when all a_i are equal.
    # ------------------------------------------------------------------
    best = np.full(n, 1.0, dtype=np.float64)         # any positive constant works
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # A tiny random local‑search loop may discover an equally good
    # configuration (or, with rounding quirks, a marginally smaller score).
    # It respects the O(n) memory budget.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        width = int(rng.integers(1, max(2, n // 20)))

            best_score = score
            # early exit if we happen to beat the trivial bound (unlikely)
            if best_score < 1.99:
                break

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best]
