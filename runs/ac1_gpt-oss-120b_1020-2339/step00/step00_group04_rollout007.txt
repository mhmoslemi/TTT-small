    # compute convolution via FFT
    n = a.shape[0]
    size = 1 << (2*n - 1).bit_length()
    # maybe use next power of 2 >= 2n.
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[:2*n-1]  # relevant length
    return conv.max()

def compute_score(a):
    n = a.size
    # length for FFT: next power of two >= 2n-1
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)
```

import numpy as np
import time


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT (O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)                       # FFT of the zero‑padded vector
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]  # linear convolution (real)
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Exact score used by the authoritative evaluator (no clipping here)."""
    n = a.shape[0]
    max_b = _conv_max(a)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Heuristic local search that tries to lower the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # keep a safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # initialise a feasible starting sequence
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        cur = np.clip(base, 0.0, 1000.0)
    except Exception:                                   # height_sequence_1 missing / bad
        n = 256
        cur = np.ones(n, dtype=np.float64)

    if cur.sum() < 0.01:                               # avoid the trivial ∞ case
        cur[:] = 0.01

    n = cur.shape[0]
    best = cur.copy()
    cur_score = _score(cur)
    best_score = cur_score

    # ------------------------------------------------------------------
    # simple stochastic hill‑climbing:
    #   move a random fraction of mass from one index to another.
    # ------------------------------------------------------------------
    max_frac = 0.5            # max proportion of the source element that may be moved

    while time.monotonic() < deadline:
        i = rng.integers(n)                     # source index (will lose mass)
        j = rng.integers(n - 1)                 # target index (will gain mass)
        if j >= i:
            j += 1
        if cur[i] <= 1e-12:                     # nothing to give
            continue

        # choose a transferable amount
        delta = cur[i] * rng.uniform(0.0, max_frac)
        # respect the upper bound 1000 on the destination entry
        cap = 1000.0 - cur[j]
        if cap <= 0.0:
            continue
        delta = min(delta, cap)
        if delta <= 0.0:
            continue

        # propose the new vector
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _score(new)

        # accept only improvements (strict descent)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

    # return the best candidate found (as a plain list of floats)
    return [float(v) for v in best]
