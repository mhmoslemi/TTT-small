import numpy as np
import time
import math

def make_block_perturbation(base, rng):
    """Return a cheap O(n)-memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # a random contiguous block, width roughly between 0.5% and 10% of n
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Return a non‑negative step‑function improving the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # length of the problem and a cheap fallback (the supplied parent)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:                     # very unlikely, but safe
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # --------------------------------------------------------------
    # fast objective using FFT (exact up to floating error)
    # --------------------------------------------------------------
    def fast_score(a):
        s = a.sum()
        if s < 0.01:
            return math.inf
        L = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        m = conv.max()
        return 2.0 * n * m / (s * s)

    best_seq = parent.clip(0.0, 1000.0)
    best_score = fast_score(best_seq)

    # ------------------------------------------------------------------
    # Parameterised exponential families as starting points
    # ------------------------------------------------------------------
    L_full = 1 << ((2 * n - 1).bit_length())          # length for a*a
    L_grad = 1 << ((3 * n - 2).bit_length())          # length for a * mask_rev
    c_vals = [0.8, 1.0, 1.2, 1.4, 1.6, 2.0]

    for c in c_vals:
        if time.monotonic() >= deadline:
            break
        r = math.exp(-c / n)                          # decay factor
        a = r ** np.arange(n, dtype=np.float64)       # a_i = r^i
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum                                    # normalise sum to 1

        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()

        eta = 0.5                                      # initial step size
        no_improve = 0
        max_iter = 200

        for _ in range(max_iter):
            if time.monotonic() >= deadline:
                break

            # ----- compute convolution a*a and the mask of maximal lags -----
            fa = np.fft.rfft(a, L_full)
            conv = np.fft.irfft(fa * fa, n=L_full)[: 2 * n - 1]
            M = conv.max()
            mask = np.isclose(conv, M, atol=1e-12, rtol=0.0).astype(np.float64)

            # ----- gradient of the max‑convolution term --------------------
            rev_mask = mask[::-1]
            fa2 = np.fft.rfft(a, L_grad)
            fm2 = np.fft.rfft(rev_mask, L_grad)
            full = np.fft.irfft(fa2 * fm2, n=L_grad)[: 3 * n - 2]
            offset = 2 * n - 2
            grad_M = 2.0 * full[offset - np.arange(n)]

            # ----- full gradient of the objective -------------------------
            S = a.sum()                       # should be 1 after normalisation
            A = 2.0 * n
            grad_obj = A * (grad_M / (S * S) - 2.0 * M / (S * S * S))

            # ----- gradient step, projection and renormalisation ----------
            step = eta * grad_obj
            a_new = a - step
            np.clip(a_new, 0.0, None, out=a_new)
            s_new = a_new.sum()
            if s_new < 1e-12:                # avoid collapse – shrink step size
                eta *= 0.5
                continue
            a_new /= s_new

            new_score = fast_score(a_new)

            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 20:              # adapt step size if stagnating
                eta *= 0.5
                no_improve = 0

            a = a_new
            if best_score < 1.50274349:       # early exit if target reached
                break

    # --------------------------------------------------------------
    # Final local search (random block perturbations) if time permits
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best_seq, rng)
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, cand.copy()
            if best_score < 1.50274349:
                break

    # --------------------------------------------------------------
    # Return a safe, clipped list of floats
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
