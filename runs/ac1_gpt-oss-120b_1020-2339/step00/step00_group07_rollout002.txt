    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # init
    if 'height_sequence_1' in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    n = a.size

    while time.monotonic() < deadline:

        conv = np.convolve(a, a)
        max_val = conv.max()

        score = 2 * n * max_val

        if score < best_score:

            best_score = score
        # Choose operation step: gradient descent

        # compute gradient vector g

        factor = np.exp(-eta * g)

        a_sum = a.sum()

            a = np.full_like(a, 1.0 / n)

            if not ok:
                continue
            # accept i
            S.append(i)
            counts[i + i] += 1
            # update counts for cross pairs (i,j) and (j,i)
            for j in S[:-1]:
                counts[i + j] += 2
            if len(S) >= m_target:
                return S, C
        return S, C

    # ------------------------------------------------------------------
    # main search loop
    # ------------------------------------------------------------------
    TARGET = 1.50274349
    C_choices = [3, 4, 5]                     # try small max counts first
    while time.monotonic() < deadline:
        C = int(rng.choice(C_choices))
        # theoretical maximal size for a given C (pigeonhole bound)
        max_possible = int(math.isqrt(C * (2 * n - 1)))
        # size needed to beat the target bound with this C
        needed = int(math.ceil(math.sqrt(2.0 * n * C / TARGET)))
        # aim for a size comfortably between the two limits
        m_target = min(max_possible, max(needed, int(0.8 * max_possible)))
        if m_target < 2:
            continue

        # build a candidate support set
        S, used_C = greedy_set(C, m_target)
        m = len(S)
        if m < 2:
            continue

        # theoretical bound from C and m (guaranteed not to be exceeded)
        bound_est = 2.0 * n * used_C / (m * m)
        # skip if even this bound cannot improve the current best
        if bound_est >= best_score:
            continue

        # construct the actual step‑function (uniform weight on the support)
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / m

        # evaluate the exact score (authoritative)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            if best_score <= TARGET:
                break

    # ------------------------------------------------------------------
    # final safeguard – ensure a finite, non‑empty result
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        # fallback to the parent (already normalised)
        best = parent.copy()
    return [float(x) for x in best]
