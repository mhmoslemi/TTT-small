def hill_climb(a, c, deadline):

    sum_a = a.sum()  # constant
    cur_max = c.max()
    best_a = a.copy()
    best_max = cur_max
    while time.monotonic() < deadline:
        k = int(np.argmax(c))
        # Determine possible source indices participating in this max
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # Choose source index i randomly in that range
        i = rng.integers(i_start, i_end+1)
        j = k - i
        # Choose which of i,j to reduce: pick the larger value
        if a[i] >= a[j]:
            src = i
            partner = j
        else:
            src = j
            partner = i
        # Choose target index t outside the range i_start..i_end to move mass to
        # we can select random index not in [i_start,i_end]
        # To ensure not same as src, partner.
        t = rng.integers(0, n)

def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # init candidate
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    else:
        parent = None
    n = len(parent) if parent is not None else 256
    # start with uniform
    uniform = np.ones(n, dtype=np.float64)
    # compute scores
    best = uniform.copy()
    best_conv = np.convolve(best, best)
    best_max = best_conv.max()
    best_score = float(2 * n * best_max / (best.sum()**2))
    # optionally compare with parent if present and better
    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if parent_score < best_score:
            best = parent.copy()
            best_conv = np.convolve(best, best)
            best_max = best_conv.max()
            best_score = parent_score
    # now run hill-climbing
    a = best.copy()
    c = np.convolve(a, a)
    cur_max = c.max()
    cur_score = best_score
    best_a = a.copy()
    best_cur_max = cur_max
    # main loop
    while time.monotonic() < deadline:
        # try guided move
        k = int(np.argmax(c))
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # pick source index among i_start..i_end
        i = rng.integers(i_start, i_end + 1)
        j = k - i
        # choose src as the larger of the two
        if a[i] >= a[j]:
            src = i
            partner = j
        else:
            src = j
            partner = i
        # choose target index not in [i_start,i_end]
        t = rng.integers(0, n)
        # ensure t not src and not partner and not inside range
        attempts = 0
        while (i_start <= t <= i_end) or (t == src) or (t == partner):
            t = rng.integers(0, n)
            attempts += 1
            if attempts > 5:
                t = (t + n//2) % n
                break
        max_transfer = min(a[src], 1000.0 - a[t])
        if max_transfer <= 0:
            continue
        # delta fraction
        delta = max_transfer * rng.random() * 0.5  # up to 50%
        # perform move
        apply_delta(a, c, src, -delta)
        apply_delta(a, c, t, delta)
        new_max = c.max()
        if new_max < cur_max:
            cur_max = new_max
            cur_score = 2 * n * cur_max / (a.sum()**2)
            if cur_max < best_cur_max:
                best_cur_max = cur_max
                best_a = a.copy()
        else:
            # revert
            apply_delta(a, c, t, -delta)
            apply_delta(a, c, src, delta)
        # occasional random perturbation to avoid stagnation
        if rng.random() < 0.01:
            # random small perturbation on a random block
            block_start = rng.integers(0, n)
            block_len = rng.integers(1, min(100, n - block_start))
            factor = 1.0 + rng.normal(0.0, 0.05)
            # scale block
            old_vals = a[block_start:block_start+block_len].copy()
            new_vals = np.clip(old_vals * factor, 0.0, 1000.0)
            delta_vec = new_vals - old_vals
            # apply each delta sequentially (could be slower) or compute aggregated effects:
            for idx, d in enumerate(delta_vec):
                if abs(d) > 1e-12:
                    apply_delta(a, c, block_start+idx, d)
        # monitor time
    final = best_a.tolist()
    # ensure final is finite
    final = [float(x) for x in final]
    return final

import time
import numpy as np

def _apply_delta(x: np.ndarray, conv: np.ndarray, idx: int, delta: float) -> None:
    """Update convolution `conv` when `x[idx]` changes by `delta` (in‑place).

    The update uses the identity:
        (x+Δe_i)*(x+Δe_i) = x*x + 2Δ*(x conv e_i) + Δ²*(e_i conv e_i)
    where (x conv e_i) is x placed at positions i … i+len(x)-1.
    """
    if delta == 0.0:
        return
    old = x[idx]
    # contributions of the cross terms (using the old vector x)
    conv[idx: idx + x.size] += 2.0 * delta * x
    # the self‑product term
    conv[2 * idx] += delta * delta
    # finally update the entry itself
    x[idx] = old + delta


def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs):
    """Hill‑climbing search for a sequence with a low AC1 upper bound.

    The algorithm starts from a uniform vector (or from the provided
    `height_sequence_1` if it is better) and repeatedly transfers a small
    amount of mass between two positions while keeping the total sum fixed.
    Only moves that strictly lower the maximal convolution value are kept.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------ #
    # initialise candidate
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    else:
        parent = None

    n = parent.size if parent is not None else 256

    # uniform starting point (any positive constant works – scale is irrelevant)
    best = np.ones(n, dtype=np.float64)
    cur = best.copy()
    cur_conv = np.convolve(cur, cur)
    cur_max = cur_conv.max()
    best_max = cur_max
    best_seq = cur.copy()

    # possibly start from the supplied parent if it is already better
    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        uniform_score = float(evaluate_sequence(best.tolist()))
        if parent_score < uniform_score:
            cur = parent.copy()
            cur_conv = np.convolve(cur, cur)
            cur_max = cur_conv.max()
            best_seq = cur.copy()
            best_max = cur_max

    # ------------------------------------------------------------------ #
    # hill‑climbing loop
    while time.monotonic() < deadline:
        # pick a source index with probability proportional to its magnitude
        src = int(rng.choice(n, p=cur / cur.sum()))
        # pick a distinct target index (uniformly)
        tgt = rng.integers(0, n - 1)
        while tgt == src:
            tgt = rng.integers(0, n - 1)

        # maximal feasible transfer respecting bounds
        max_transfer = min(cur[src], 1000.0 - cur[tgt])
        if max_transfer <= 0.0:
            continue

        # try a modest random amount (up to 50 % of what can be moved)
        delta = max_transfer * rng.random() * 0.5

        # perform the move: src loses, tgt gains
        _apply_delta(cur, cur_conv, src, -delta)
        _apply_delta(cur, cur_conv, tgt, delta)

        new_max = cur_conv.max()
        if new_max < cur_max:                     # accept only improving moves
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_seq = cur.copy()
        else:                                      # revert the change
            _apply_delta(cur, cur_conv, tgt, -delta)
            _apply_delta(cur, cur_conv, src, delta)

        # occasional tiny random perturbation to escape plateaus
        if rng.random() < 0.005:
            w = rng.integers(0, n)
            length = rng.integers(1, min(32, n - w) + 1)
            factor = 1.0 + rng.normal(0.0, 0.03)
            orig = cur[w:w + length].copy()
            cur[w:w + length] = np.clip(orig * factor, 0.0, 1000.0)
            # update convolution for each changed entry
            for pos, delta_i in enumerate(cur[w:w + length] - orig):
                if delta_i != 0.0:
                    _apply_delta(cur, cur_conv, w + pos, delta_i)

            cur_max = cur_conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_seq = cur.copy()

    # ------------------------------------------------------------------ #
    # final sanity check – evaluate the best sequence exactly
    final = best_seq.tolist()
    # make sure we return plain Python floats
    return [float(v) for v in final]
