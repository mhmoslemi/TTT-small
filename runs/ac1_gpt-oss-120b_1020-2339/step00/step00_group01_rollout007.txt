def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., float(budget_s) - 2.)

    # Use height_sequence_1 if defined
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        if np.sum(parent_arr) == 0:
            raise ValueError
        a = parent_arr / np.sum(parent_arr)  # normalize to sum=1
    except Exception:
        n = 256
        a = np.full(n, 1.0/n, dtype=np.float64)

    n = a.shape[0]
    # pre compute FFT size
    Nfft = 1 << ((2*n - 1).bit_length())

    def conv_max(v):
        fa = np.fft.rfft(v, Nfft)
        conv = np.fft.irfft(fa * fa, Nfft)
        # keep only first 2*n-1 entries
        c = conv[:2*n-1]
        return float(c.max())

    def proj_simplex(v):
        # project onto simplex {x>=0, sum=1}
        # using sorting algorithm
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.nonzero(u - cssv / np.arange(1, u.size+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # initial evaluation
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    best_approx = 2 * n * conv_max(best_a)  # approx score (since sum=1)

    # subgradient descent loop
    t = 0
    step0 = 0.5   # initial step size
    while time.monotonic() < deadline:
        # compute convolution max and its index
        fa = np.fft.rfft(a, Nfft)
        conv = np.fft.irfft(fa * fa, Nfft)
        c = conv[:2*n-1]
        max_idx = int(np.argmax(c))
        max_val = float(c[max_idx])

        # compute gradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end+1)
            g[idx] = 2.0 * a[max_idx - idx]

        # step size schedule
        alpha = step0 / np.sqrt(t+1)

        # gradient step and projection
        a_new = a - alpha * g
        a = proj_simplex(a_new)

        # occasional random perturbation to avoid stagnation
        if rng.random() < 0.01:
            perturb = rng.normal(0.0, 0.05, size=n)
            a = proj_simplex(a + perturb)

        # approximate score (sum=1)
        approx_score = 2.0 * n * max_val  # since sum=1

        # if improved
        if approx_score < best_approx - 1e-9:
            # compute exact
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                best_approx = approx_score
        t += 1

    # after time, scale best_a to feasible values (already in 0..1)
    # ensure within bounds 0-1000 (multiplying by 1000 maybe)
    # The evaluator will clip anyway.
    result = (best_a * 1.0).tolist()
    # final sanity: ensure length non-zero and finite
    # convert to Python floats
    result = [float(x) for x in result]
    return result

import numpy as np
import time
import math

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x>=0, sum(x)=1}."""
    # Duchi et al. algorithm, O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, u.size + 1) > 0)[0]
    if rho.size == 0:          # all entries are zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a: np.ndarray, Nfft: int) -> float:
    """Return the maximum coefficient of the full convolution a*a."""
    fa = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(fa * fa, Nfft)
    # first 2*len(a)-1 entries are the true linear convolution
    c = conv[: 2 * a.size - 1]
    return float(c.max())
