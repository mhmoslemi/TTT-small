import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Generate a non‑negative step function whose AC1 upper bound is
    substantially below the trivial value 2.0.  The construction uses a
    discretised arcsine density (a_i ∝ 1/√(i·(n‑i))) which makes the
    self‑convolution almost flat, driving the score close to the theoretical
    lower bound of 1.
    """
    # random generator for any optional stochastic refinement
    rng = np.random.default_rng(seed)

    # time guard – leave a few seconds margin for final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # length of the problem instance (height_sequence_1 is supplied by the harness)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = int(base.size)
    except Exception:
        n = 256

    # ---------- core deterministic construction ----------
    # positions centred in the interval, offset by 0.5 to avoid zero denominator
    i = np.arange(n, dtype=np.float64) + 0.5
    # raw densities proportional to 1/√(i·(n‑i))
    raw = 1.0 / np.sqrt(i * (n - i))
    seq = raw / raw.sum()                     # normalise to sum == 1

    # ensure the sequence respects the evaluator's clipping bounds
    seq = np.clip(seq, 0.0, 1000.0)

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ---------- optional lightweight stochastic improvement ----------
    # simple mass‑transfer moves that preserve the total sum.
    # This loop runs only while time permits; it never harms the fallback.
    while time.monotonic() < deadline:
        # pick two distinct indices
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 == i2:
            continue

        # amount to move (a small fraction of the source entry)
        src_val = best_seq[i1]
        if src_val <= 0.0:
            continue
        delta = src_val * rng.random() * 0.03  # at most 3 % of the source

        cand = best_seq.copy()
        cand[i1] = src_val - delta
        cand[i2] = best_seq[i2] + delta
        # keep within allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # return a plain Python list of floats as required by the evaluator
    return [float(x) for x in best_seq]
