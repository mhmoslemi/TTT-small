    b = np.convolve(current, current)

    # compute gradient vector g = 2 * reversed version shifted by k

def compute_ac1(seq):
    s = np.sum(seq)
    if s < 1e-12:

    bmax = np.convolve(seq, seq).max()

def log_perturb(a, rng, sigma=0.02):
    cand = a.copy()
    log_cand = np.log(cand + 1e-15)
    log_cand += rng.normal(0.0, sigma, size=cand.shape)
    cand = np.exp(log_cand)
    cand = np.clip(cand, 0.0, None)
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand

def gradient_step(a, n):
    # compute current maxb and its index
    b = np.convolve(a, a)
    maxb = b.max()
    k = b.argmax()
    idx = np.arange(n)

    mask = (rev >= 0) & (rev < n)

import numpy as np
import time

def _compute_ac1(seq: np.ndarray) -> float:
    """Return the AC1 bound for a non‑negative sequence (sum may be any >0)."""
    s = np.sum(seq)
    if s <= 0.0:
        return float('inf')
    # full convolution, automatically zero‑pads out‑of‑range indices
    maxb = np.convolve(seq, seq).max()
    n = seq.shape[0]
    return 2.0 * n * maxb / (s * s)


def _init_shape(n: int) -> np.ndarray:
    """
    Initialise a symmetric 1/√(i)–type shape.
    a_i = ½(1/√(i+1) + 1/√(n-i)) and then normalise to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i))
    a /= a.sum()
    return a


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block of `base`."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.10)))  # modest log‑scale
    if rng.random() < 0.35:               # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return cand


def _log_perturbation(base: np.ndarray, rng: np.random.Generator, sigma: float = 0.02) -> np.ndarray:
    """Additive Gaussian noise in log‑space, then re‑normalise."""
    logcand = np.log(base + 1e-15)               # avoid log(0)
    logcand += rng.normal(0.0, sigma, size=base.shape)
    cand = np.exp(logcand)
    cand = np.clip(cand, 0.0, None)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _gradient_step(cur: np.ndarray, n: int):
    """
    One projected sub‑gradient step for the objective max(b)   (sum is kept 1).
    Returns a possibly improved candidate and its *raw* max‑convolution value.
    """
    b = np.convolve(cur, cur)
    maxb = b.max()
    k = int(b.argmax())

    idx = np.arange(n, dtype=np.int64)
    rev = k - idx
    mask = (rev >= 0) & (rev < n)
    g = np.zeros_like(cur)
    g[mask] = 2.0 * cur[rev[mask]]           # ∂max_b/∂a_i

    max_a = cur.max()
    max_g = g.max()
    if max_g == 0.0:
        return cur, maxb

    # try a few decreasing step factors
    for factor in (0.5, 0.25, 0.125, 0.0625):
        alpha = factor * max_a / max_g        # crude step size
        cand = cur - alpha * g
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s
        new_maxb = np.convolve(cand, cand).max()
        if new_maxb < maxb:
            return cand, new_maxb
    return cur, maxb


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Return a list of non‑negative coefficients (length inherited from
    ``height_sequence_1`` when present) that attempts to minimise the
    AC1 bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size) if parent.ndim == 1 and parent.size > 0 else 256
    else:
        n = 256

    # ------------------------------------------------------------------
    # Initialise with the analytical 1/√(i) shape (symmetrised)
    best = _init_shape(n)

    # Optionally use the provided parent (normalised) if it is better
    if 'height_sequence_1' in globals():
        try:

            if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
                parent_norm = parent / parent.sum()
                parent_score = _compute_ac1(parent_norm)
                best_score = _compute_ac1(best)
                if parent_score < best_score:
                    best = parent_norm
        except Exception:
            pass

    best_score = _compute_ac1(best)

    # ------------------------------------------------------------------
    # Time‑bounded optimisation loop
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    it = 0
    while time.monotonic() < deadline:
        # Randomly pick a mutation operator
        op = rng.integers(0, 3)

        if op == 0:                     # block multiplicative perturbation
            cand = _make_block_perturbation(best, rng)

        elif op == 1:                   # log‑space Gaussian perturbation
            cand = _log_perturbation(best, rng, sigma=0.02)

        else:                           # projected gradient step
            cand, new_maxb = _gradient_step(best, n)
            cand_score = 2.0 * n * new_maxb      # sum is 1 by construction
            if cand_score < best_score:
                best = cand
                best_score = cand_score
            it += 1
            continue                     # skip the generic evaluation below

        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s

        # Cheap evaluation using the same formula as the authoritative evaluator
        cand_score = _compute_ac1(cand)

        if cand_score < best_score:
            best = cand
            best_score = cand_score

        it += 1

    # Return a plain Python list of floats as required
    return [float(v) for v in best]
