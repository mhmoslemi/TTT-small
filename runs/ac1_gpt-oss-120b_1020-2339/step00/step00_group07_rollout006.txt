import time
import numpy as np
import scipy.sparse as sp

# optional linear programming backend
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False


def _conv_linear(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using FFT (fast for n~7k)."""
    n = a.size
    size = 2 * n - 1
    # next power‑of‑2 length for efficient FFT
    fft_len = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(A * A).real[:size]
    return conv


def _topk_indices(arr: np.ndarray, k: int) -> np.ndarray:
    """Return indices of the k largest values of arr (unsorted)."""
    if k >= arr.size:
        return np.arange(arr.size, dtype=int)
    # np.argpartition gives the k indices of the largest elements
    part = np.argpartition(-arr, k - 1)[:k]
    return part


def _build_constraint_matrix(a: np.ndarray, active: np.ndarray) -> sp.csr_matrix:
    """Build sparse matrix A where (A @ g)[r] = sum_i a_i * g_{k_r-i}."""
    n = a.size
    rows = []
    cols = []
    data = []
    for r, k in enumerate(active):
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        # j runs over the valid indices of g
        j = np.arange(start, end + 1, dtype=int)
        i = k - j
        rows.extend([r] * len(j))
        cols.extend(j.tolist())
        data.extend(a[i].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n), dtype=np.float64)
    return A


def _lp_step(a: np.ndarray, M: float, active: np.ndarray, max_iters: int = 500) -> np.ndarray:
    """
    Solve  max sum(g)  s.t. a * g <= M on the active indices.
    Returns a scaled g (same L1 norm as a) or None if LP failed.
    """
    if not _HAS_CVX:
        return None
    n = a.size
    A = _build_constraint_matrix(a, active)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M + 1e-12]  # tiny tolerance
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=max_iters, verbose=False)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.GLPK, warm_start=True, max_iters=max_iters, verbose=False)
        except Exception:
            return None

    if g.value is None:
        return None

    g_val = np.asarray(g.value, dtype=np.float64).reshape(-1)
    sum_a = a.sum()
    sum_g = g_val.sum()
    if sum_g <= 0:
        return None
    # scale to keep the total sum unchanged (scaling does not affect the score)
    g_scaled = g_val * (sum_a / sum_g)
    return g_scaled


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Simple O(n) perturbation used as a cheap fallback."""
    b = a.copy()
    n = b.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    b[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        b = 0.5 * (b + b[::-1])
    return np.clip(b, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Return a sequence of non‑negative heights with low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------ #
    # initialise from the supplied parent if possible, otherwise use  #
    # a uniform step function.                                            #
    # ------------------------------------------------------------------ #
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:  # pragma: no cover
        parent = None

    if parent is None or parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 6733
        a = np.full(n, 1.0 / n, dtype=np.float64)  # uniform, sum = 1
    else:
        a = np.clip(parent, 0.0, 1000.0)

    # ensure we work with a float array
    a = a.astype(np.float64, copy=False)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #
    # Main optimisation loop – a few LP steps punctuated by cheap       #
    # random perturbations.                                              #
    # ------------------------------------------------------------------ #
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # current convolution and its maximum
        conv = np.convolve(a, a)  # numpy uses an efficient method internally
        M = conv.max()

        # -------------------------------------------------------------- #
        # Build a small active set (indices with largest convolution)    #
        # -------------------------------------------------------------- #
        topk = 12
        active_idx = _topk_indices(conv, topk)

        # -------------------------------------------------------------- #
        # Try an LP step (if cvxpy is available).                        #
        # -------------------------------------------------------------- #
        g_scaled = None
        if _HAS_CVX:
            g_scaled = _lp_step(a, M, active_idx, max_iters=300)

        if g_scaled is not None:
            # convex combination – blending factor tuned for stability
            t = 0.5
            a = (1.0 - t) * a + t * g_scaled
            a = np.clip(a, 0.0, 1000.0)
        else:

            a = _random_block_perturb(a, rng)

        # -------------------------------------------------------------- #

        # -------------------------------------------------------------- #
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            # early exit if we already beat a generous target
            if best_score < 1.6:
                break

        # stop if no progress for many iterations
        if iteration >= 25:
            break

    return [float(x) for x in best]
