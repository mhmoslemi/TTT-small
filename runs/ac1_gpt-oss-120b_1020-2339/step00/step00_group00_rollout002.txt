def fast_score(a):
    # compute convolution via FFT (real). Use np.fft.rfft for speed.
    L = 1 << ((2 * len(a) - 1).bit_length())  # next pow2
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)  # length L, real
    # conv length L, actual conv needed first 2n-1 entries
    conv = conv[:2 * len(a) - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * len(a) * max_b / (sum_a ** 2)

    deadline = time.monotonic() + max(0,float(budget_s)-2)
    n = ...

    if height_sequence_1 exists:

        i = rng.integers(0, n)

        factor = np.exp(rng.normal(0.0, sigma))
        new_a = cur_a.copy()

        # clip

        if new_val < 0.0:
            new_val = 0.0
        new_a[i] = new_val
        # possibly also adjust other random index to keep sum stable?
        # not needed.

        new_score = fast_score(new_a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T,1e-12)):
            cur_a = new_a
            cur_score = new_score

import time
import numpy as np

# fast computation of the score (identical to evaluate_sequence but using FFT)
def _fast_score(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2*n*max(conv(arr,arr))/sum(arr)^2 (no clipping)."""
    # FFT based linear convolution
    fa = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * n - 1]               # keep the linear part only
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _greedy_sidon(n: int, rng: np.random.Generator, max_size: int = None):
    """Return a list of indices forming a Sidon set (all i+j distinct)."""
    # random order of candidates
    order = np.arange(n, dtype=np.int64)
    rng.shuffle(order)
    indices = []
    sums = set()
    for i in order:
        # quick reject if 2*i already taken
        if 2 * i in sums:
            continue
        collide = False
        for j in indices:
            if i + j in sums:
                collide = True
                break
        if not collide:
            # add i
            for j in indices:
                sums.add(i + j)      # i+j with previous elements
            sums.add(2 * i)           # i+i
            indices.append(int(i))
            if max_size is not None and len(indices) >= max_size:
                break
    return indices
