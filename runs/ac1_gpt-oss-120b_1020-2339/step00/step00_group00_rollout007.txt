def conv_max(a):
    n = a.size
    m = 1 << ((2*n-1).bit_length())   # next power of two >= 2n-1
    # compute FFT
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)  # length m
    # conv length m; actual conv values are first 2n-1 entries
    return np.max(conv[:2*n-1])

    deadline = time.monotonic() + budget_s - 2.0
    # Load parent if exist
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, 1000.0)
    else:
        base = np.ones(256, dtype=np.float64)
    # Normalize sum to 1
    if base.sum() == 0:
        base = np.ones_like(base)
    base = base / base.sum()
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    a = best.copy()
    n = a.size
    # Precompute shape for FFT
    m = 1 << ((2*n - 1).bit_length())
    # main loop
    while time.monotonic() < deadline:
        # propose perturbation
        # random choose operation type: swap, shift, block scaling
        op = rng.integers(0, 3)
        if op == 0:

            i = rng.integers(0, n)
            j = rng.integers(0, n)

            # amount to transfer
            delta = a[i] * rng.random() * 0.5  # up to half of a[i]
            if delta <= 0:

            a[i] -= delta
            a[j] += delta

            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))

            a[start:start+width] *= factor

import time
import numpy as np

def _conv_max(a, fft_len):
    """Return the maximal value of a*a (full convolution) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv.max()

    """
    Stochastic gradient + random perturbation search for a sequence that lowers
    the AC1 upper bound (the evaluator uses 2*n*max(conv)/(sum^2)).
    The algorithm works with O(n) memory and uses FFT‑based convolutions.
    """

    # time budget – leave a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------ #
    # initialise a – start from the supplied parent if available else uniform
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    else:
        a = np.ones(256, dtype=np.float64)

    if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
        a = np.ones(256, dtype=np.float64)

    # normalise to sum = 1 (scaling does not affect the objective)
    s = a.sum()
    if s <= 0.0:
        a = np.ones_like(a)
        s = a.size
    a = a / s

    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    # main optimisation loop
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # convolution of current point (already have its maximum)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        k = int(conv.argmax())                     # index of the maximal bin

        # build sub‑gradient of the max‑bin w.r.t. a
        g = np.zeros_like(a)
        if k < n:                                  # left half of the output
            g[: k + 1] = 2.0 * a[k :: -1]
        else:                                      # right half
            start = k - (n - 1)
            g[start:] = 2.0 * a[start:][::-1]

        # simple decreasing step size, never smaller than 1e-3
        lr = 0.1 * max(1.0 - iteration * 1e-5, 0.01)

        # propose a new point
        a_new = a - lr * g

        # occasional stochastic perturbation to avoid stagnation
        if rng.random() < 0.1:
            a_new += 0.02 * (rng.random(n) - 0.5) * a_new

        a_new = np.clip(a_new, 0.0, None)
        s_new = a_new.sum()
        if s_new <= 0.0:
            a_new = np.ones_like(a) / n
            s_new = 1.0
        a_new /= s_new

        # evaluate the proposal (FFT based, O(n log n))
        max_new = _conv_max(a_new, fft_len)
        score_new = 2.0 * n * max_new

        # accept if the proposal improves the current score
        if score_new < cur_score:
            a = a_new
            cur_score = score_new
        else:
            # tiny probability to accept uphill moves (helps escape local minima)
            if rng.random() < 0.001:
                a = a_new
                cur_score = score_new

        # keep the best ever encountered
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score < 1.50274349:          # target reached – stop early
                break

        # safety break if we have run too many iterations (unlikely)
        if iteration > 200000:
            break

    # ensure a non‑empty list of finite numbers
    return [float(x) for x in best]
