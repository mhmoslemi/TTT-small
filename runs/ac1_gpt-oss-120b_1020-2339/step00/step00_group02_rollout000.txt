import time
import numpy as np


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Return a non‑negative step‑function that (almost) minimizes the AC1 bound.

    The theoretical optimum for the given evaluator is a uniform vector,
    which yields a score of exactly 2.  We start with a uniform vector of
    length matching the provided parent (height_sequence_1 when available) and
    perform a very cheap random local search while time permits – any
    improvement over the uniform baseline is kept.
    """
    # ----- time management -------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----- determine length ------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback length if the parent is unavailable or invalid
        n = 256

    # ----- uniform starting candidate (optimal in theory) -----------------
    cand = np.full(n, 1.0, dtype=np.float64)          # any positive constant works
    cand = np.clip(cand, 0.0, 1000.0)

    best = cand
    best_score = float(evaluate_sequence(best.tolist()))

    # ----- tiny stochastic refinement (optional, respects time) ----------
    rng = np.random.default_rng(seed)
    while time.monotonic() < deadline:
        # perturb a modest block of entries – O(n) memory only
        width = max(1, n // 200)                     # block width
        start = rng.integers(0, n - width + 1)
        stop = start + width
        proposal = best.copy()
        # multiplicative Gaussian perturbation on the block
        proposal[start:stop] *= float(np.exp(rng.normal(0.0, 0.04)))
        # optional mild symmetrisation (keeps the sequence non‑negative)
        if rng.random() < 0.25:
            proposal = 0.5 * (proposal + proposal[::-1])
        # enforce the evaluator's hard bounds
        proposal = np.clip(proposal, 0.0, 1000.0)

        score = float(evaluate_sequence(proposal.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = proposal, score

    # ----- return as plain Python list ------------------------------------
    return [float(x) for x in best]
