import time

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # 1.  Obtain a reasonable starting vector.
    # ----------------------------------------------------------------------
    parent = None
    if "height_sequence_1" in globals():
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            parent = np.clip(arr, 0.0, 1000.0)

    if parent is None:
        # fallback – short constant line (any non‑empty vector works)
        parent = np.full(256, 1.0, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 2.  Simple grid search over power‑law families (a_i = (i+off)^-p).
    # ----------------------------------------------------------------------
    exponents = (0.45, 0.5, 0.55)
    offsets = (0.1, 0.5, 1.0, 2.0, 5.0, 10.0)

    for p in exponents:
        for off in offsets:
            seq = (np.arange(n, dtype=np.float64) + off) ** (-p)
            seq = np.clip(seq, 0.0, 1000.0)
            sc = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score = seq, sc

            # reversed order can change the convolution pattern
            rev = seq[::-1]
            sc = float(evaluate_sequence(rev.tolist()))
            if np.isfinite(sc) and sc < best_score:

            # symmetric average (just for completeness)
            sym = 0.5 * (seq + rev)
            sc = float(evaluate_sequence(sym.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score = sym, sc

    # ----------------------------------------------------------------------
    # 3.  Local refinement: move a small amount of mass away from the

    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
