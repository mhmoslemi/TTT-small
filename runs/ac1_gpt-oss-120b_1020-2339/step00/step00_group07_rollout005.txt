def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # keep 2 seconds for final
    # init a
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise Exception
        a = np.clip(init, 0.0, 1000.0)
    except Exception:
        a = np.ones(256, dtype=np.float64)
    # Normalize to sum 1
    a = a / a.sum()
    n = a.size
    # compute initial score
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()  # sum=1 => denominator 1
    best = a.copy()
    step = 0.1  # magnitude for gradient
    # iterate
    while time.monotonic() < deadline:
        if rng.random() < 0.3:
            # random block perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = a.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, None)
            cand = cand / cand.sum()
        else:
            # gradient descent on current max conv
            conv = np.convolve(a, a)
            idx = int(conv.argmax())
            # compute gradient
            g = np.zeros_like(a)
            # vectorized: for i where idx-i valid:
            # we can compute slice
            lo = max(0, idx - (n - 1))
            hi = min(idx, n - 1)
            # indices i from lo..hi inclusive produce valid j = idx - i
            i_range = np.arange(lo, hi+1, dtype=np.int64)
            j_range = idx - i_range
            g[i_range] = 2.0 * a[j_range]
            # update
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            total = cand.sum()
            if total == 0:
                cand = a.copy()
            else:
                cand = cand / total
        # evaluate
        conv_c = np.convolve(cand, cand)
        score = 2.0 * n * conv_c.max()  # sum=1
        # accept if better
        if score < best_score - 1e-12:
            best_score = score
            best = cand.copy()
            a = cand  # move hillclimbing
            # optionally increase step if improvement
            step = min(step * 1.05, 1.0)
        else:
            # maybe try smaller step or reject
            step = max(step * 0.95, 1e-4)
            # optionally keep current a unchanged
        # safety break if best_score < target 1.50274 etc.
        if best_score < 1.50274349:
            break
    # return best as list of floats
    return [float(x) for x in best]

import numpy as np, time

def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine runs a short stochastic‑gradient search that respects the
    scale‑invariance of the objective.  It works with the supplied
    ``height_sequence_1`` if present; otherwise it falls back to a uniform start.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------ #
    # initialise a feasible probability vector (sum = 1, non‑negative)
    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
            raise ValueError
        a0 = np.clip(a0, 0.0, 1000.0)
    except Exception:
        a0 = np.ones(256, dtype=np.float64)

    total = a0.sum()
    if total == 0.0:
        a0 = np.ones_like(a0)
        total = a0.sum()
    a = a0 / total                     # normalised to sum == 1
    n = a.size

    # ------------------------------------------------------------------ #
    # cheap evaluation (scale invariant, sum is 1)
    def score(vec: np.ndarray) -> float:
        conv = np.convolve(vec, vec)
        return 2.0 * n * conv.max()

    best = a.copy()
    best_score = score(best)

    # parameters for the search
    step = 0.12                # learning‑rate for gradient step
    alpha = 20.0               # softness of the soft‑max on the convolution
    min_step = 1e-5
    iter_no = 0

    while time.monotonic() < deadline:
        iter_no += 1

        # ---------------------------------------------------------- #
        # occasional random block perturbation (helps to escape)
        if rng.random() < 0.25:
            width = int(rng.integers(max(1, n // 220), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand = a.copy()
            cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0.0:
                cand = cand / s
        else:
            # -------------------------------------------------- #
            # gradient descent on a smooth approximation of max(conv)
            conv = np.convolve(a, a)                # length 2n‑1
            mx = conv.max()
            # soft‑max weights (numerically safe)
            w = np.exp(alpha * (conv - mx))
            w = w / w.sum()                         # length 2n‑1, positive, sum 1

            a_rev = a[::-1]

            cand = a - step * grad
