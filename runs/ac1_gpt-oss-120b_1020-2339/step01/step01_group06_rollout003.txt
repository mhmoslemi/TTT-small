    def shape_power(p, offset=0.0):
        idx = np.arange(n, dtype=np.float64)
        frac = (n-1 - idx)/(n-1)
        seq = (frac**p) + offset
        seq = np.clip(seq, 0.0, 1.0) * max_val
        return seq

    # initial candidates list
    candidates = []
    best_seq = uniform
    best_score = eval_seq(best_seq)
    # evaluate parent if exists
    try:

        parent_score = eval_seq(parent)
        if parent_score < best_score:
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # add power law p=2
    seq2 = shape_power(2.0)
    try:
        s2 = eval_seq(seq2)
        if s2 < best_score:
            best_seq, best_score = seq2, s2
    except Exception:
        pass

    # random search over p
    p = 2.0
    offset = 0.0

    deadline = time.monotonic() + max(0.0, float(budget_s)-0.5)
    while time.monotonic() < deadline:
        # Choose operation
        op = rng.integers(0, 5) # 0: modify p, 1: modify offset, 2: random swap mass move, 3: block scaling, 4: global scaling
        if op == 0:
            # adjust p (Gaussian)
            p_new = max(0.01, p * np.exp(rng.normal(0, 0.05))) # multiplicative noise
            seq = shape_power(p_new, offset)
            try:
                score = eval_seq(seq)
                if score < best_score:
                    best_seq, best_score = seq, score
                    p = p_new
            except Exception:
                pass
        elif op == 1:
            # adjust offset up or down
            offset_new = max(0.0, offset + rng.normal(0, 0.02))
            seq = shape_power(p, offset_new)
            try:
                score = eval_seq(seq)
                if score < best_score:
                    best_seq, best_score = seq, score
                    offset = offset_new
            except Exception:
                pass
        elif op == 2:

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

            a_i = best_seq[i]

            cand[i] -= delta
            cand[j] = min(max_val, cand[j] + delta)
            try:

                if score < best_score:

        elif op == 3:

        # seq expected numpy array
        # ensure nonnegative & clip
        seq = np.clip(seq, 0.0, max_val)
        sum_a = seq.sum()
        if sum_a < 0.01:
            return float('inf')
        b = np.convolve(seq, seq)
        max_b = b.max()
        return 2.0 * n * max_b / (sum_a * sum_a)

    # base candidates
    best_seq = np.full(n, max_val, dtype=np.float64)   # uniform
    best_score = fast_score(best_seq)

    if have_parent:
        parent_seq = np.clip(base, 0.0, max_val)
        parent_score = fast_score(parent_seq)
        if parent_score < best_score:
            best_seq = parent_seq.copy()
            best_score = parent_score

    # power law candidate p=2, alpha=0

        seq = np.clip(seq, 0.0, 1.0) * max_val

    if s < best_score:

            b = np.convolve(best_seq, best_seq)

            # reduce larger of the two
            if best_seq[i_star] >= best_seq[j_star]:

            cand[t] = min(max_val, cand[t] + reduction)
            score = fast_score(cand)
            if score < best_score:
                best_seq, best_score = cand, score

    stochastic mutations (mass transfer, block scaling and a targeted
    reduction of the largest convolution contribution).  All operations
    are O(n) or O(n log n) and respect the 110 s time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n from the provided parent (if any)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.shape[0])
        have_parent = base.ndim == 1 and base.size > 0 and np.isfinite(base).all()
    except Exception:
        n = 256
        have_parent = False

    if n == 1:
        # trivial case – any positive value works, keep it at the upper bound
        return [1000.0]

    max_val = 1000.0
    idx = np.arange(n, dtype=np.float64)
    # decreasing fraction from 1 down to 0
    frac = (n - 1 - idx) / (n - 1)

    # ------------------------------------------------------------------
    # Helper: build a candidate from parameters (p, alpha)
    #   a_i = ( (1‑i/(n‑1))**p * (1‑alpha) + alpha ) * max_val
    # ------------------------------------------------------------------
    def make_seq(p, alpha):
        seq = np.power(frac, p) * (1.0 - alpha) + alpha
        seq = np.clip(seq, 0.0, 1.0) * max_val
        return seq

    # ------------------------------------------------------------------
    # Fast scoring identical to the evaluator (avoids list conversion)
    # ------------------------------------------------------------------
    def fast_score(arr):
        a = np.clip(arr, 0.0, max_val)
        s = a.sum()
        if s < 0.01:
            return float("inf")
        b = np.convolve(a, a)
        return 2.0 * n * b.max() / (s * s)

    # ------------------------------------------------------------------
    # Initialise with a few deterministic candidates
    # ------------------------------------------------------------------
    best_seq = np.full(n, max_val, dtype=np.float64)      # uniform
    best_score = fast_score(best_seq)

    if have_parent:
        parent = np.clip(base, 0.0, max_val)
        parent_score = fast_score(parent)
        if parent_score < best_score:
            best_seq = parent.copy()
            best_score = parent_score

    # simple power‑law baseline (p = 2, no uniform component)
    seq = make_seq(2.0, 0.0)
    sc = fast_score(seq)
    if sc < best_score:
        best_seq, best_score = seq, sc

    # ------------------------------------------------------------------
    # Small grid over (p,alpha) to locate a promising region
    # ------------------------------------------------------------------
    for p_val in np.linspace(0.5, 5.0, 9):          # 0.5 … 5.0
        for alpha_val in (0.0, 0.1, 0.2, 0.4):
            cand = make_seq(float(p_val), float(alpha_val))
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                p, alpha = float(p_val), float(alpha_val)

    # initialise parameters for the continuous search
    if "p" not in locals():
        p, alpha = 2.0, 0.0

    # ------------------------------------------------------------------
    # Stochastic loop – respect the overall time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Choose a random modification type (probabilities tuned empirically)
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.30:                     # perturb exponent p (log‑normal step)
            p_new = max(0.01, p * np.exp(rng.normal(0.0, 0.07)))
            cand = make_seq(p_new, alpha)
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score, p = cand, sc, p_new

        elif r < 0.55:                   # perturb uniform mixing alpha
            alpha_new = np.clip(alpha + rng.normal(0.0, 0.04), 0.0, 1.0)
            cand = make_seq(p, alpha_new)
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score, alpha = cand, sc, alpha_new

        elif r < 0.70:                   # move a random small mass from i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if best_seq[i] <= 0.0:
                continue
            delta = best_seq[i] * rng.uniform(0.001, 0.05)
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] = min(max_val, cand[j] + delta)
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

        elif r < 0.85:                   # block scaling (smooth local change)
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.05))
            cand = best_seq.copy()
            cand[start:start + width] *= factor
            cand = np.clip(cand, 0.0, max_val)
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

        elif r < 0.95:                   # global mild scaling
            factor = 1.0 + rng.normal(0.0, 0.02)
            cand = np.clip(best_seq * factor, 0.0, max_val)
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

        else:                            # targeted reduction of the current max contribution
            b = np.convolve(best_seq, best_seq)
            k = int(np.argmax(b))                       # shift where the max occurs
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            i_range = np.arange(i0, i1 + 1)
            j_range = k - i_range
            contrib = best_seq[i_range] * best_seq[j_range]
            idx_max = int(np.argmax(contrib))
            i_star = i_range[idx_max]
            j_star = j_range[idx_max]
            # pick the larger participant of the pair
            target = i_star if best_seq[i_star] >= best_seq[j_star] else j_star
            # shrink it a bit and re‑inject the mass elsewhere
            factor = 1.0 - rng.uniform(0.01, 0.07)
            new_val = best_seq[target] * factor
            transfer = best_seq[target] - new_val
            dest = rng.integers(0, n)
            cand = best_seq.copy()
            cand[target] = new_val
            cand[dest] = min(max_val, cand[dest] + transfer)
            sc = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # Return the best sequence discovered, as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
