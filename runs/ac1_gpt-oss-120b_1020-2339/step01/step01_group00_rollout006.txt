def proj_simplex(v):
   """Project onto the probability simplex (non-negative, sum=1)."""
   # reference algorithm:
   n = len(v)
   u = np.sort(v)[::-1]   # descending
   cssv = np.cumsum(u) - 1
   ind = np.arange(n) + 1
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

import time

def _fft_len_for_n(n: int) -> int:
    """Return a power‑of‑two length suitable for a full convolution of size 2n‑1."""
    return 1 << ((2 * n - 1).bit_length())

def _max_convolution(a: np.ndarray, fft_len: int):
    """
    Compute the full (linear) convolution a*a using FFT,
    returning (max_value, index_of_max, full_convolution_array).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x≥0, sum(x)=1}.
    Implementation of the algorithm from
    https://doi.org/10.1007/978-3-540-71031-4_20
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform as a safe fallback
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise for numerical safety
    if w.sum() == 0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w.sum()
    return w
