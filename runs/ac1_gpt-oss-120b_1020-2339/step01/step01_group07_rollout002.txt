def conv_full(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)
    conv = conv[:2*n -1]  # however due to rounding errors may produce small negatives near zero.
    return conv

def perturb_max_pair(a, rng):
    n = len(a)
    conv = conv_full(a)  # compute
    max_idx = int(conv.argmax())
    # compute contributions
    # j index maps:
    # i from max(0, max_idx - (n-1)) to min(n-1, max_idx)
    start = max(0, max_idx - (n - 1))
    end = min(n - 1, max_idx)
    i_vals = np.arange(start, end + 1)
    j_vals = max_idx - i_vals
    products = a[i_vals] * a[j_vals]
    # find i_max with largest product
    idx = np.argmax(products)
    i = int(i_vals[idx])
    j = int(j_vals[idx])
    # delta selection
    if i == j:
        delta = a[i] * 0.1
        a[i] -= delta
        # distribute delta to random other index
        k = int(rng.integers(0, n))
        if k == i:
            k = (k + 1) % n
        a[k] += delta
    else:
        delta = min(a[i], a[j]) * 0.1
        a[i] -= delta
        a[j] -= delta
        total = 2 * delta
        # distribute to two random indices (could be same) but avoid i,j
        for _ in range(2):
            k = int(rng.integers(0, n))
            if k == i or k == j:
                k = (k + 1) % n
            a[k] += total / 2
    # ensure bounds
    np.clip(a, 0.0, 1000.0, out=a)

return a

import numpy as np
import time

def _conv_full(a):
    n = a.shape[0]
    m = 2*n - 1
    size = 1 << ((m).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)
    return conv[:m]

def _score(a):
    n = a.shape[0]
    conv = _conv_full(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _block_move(a, rng):
    n = a.shape[0]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))

    a[start:start+width] *= factor
    if rng.random() < 0.3:
        a[:] = (a + a[::-1]) * 0.5
    np.clip(a, 0.0, 1000.0, out=a)

import numpy as np
import time

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a power‑of‑2 FFT (real input)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[:m]

def _score(a: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative step‑function."""
    n = a.shape[0]
    conv = _conv_full(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _block_move(a: np.ndarray, rng: np.random.Generator) -> None:
    """Multiplicative perturbation of a random contiguous block."""
    n = a.shape[0]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    a[start:start + width] *= factor
    if rng.random() < 0.3:
        a[:] = (a + a[::-1]) * 0.5
    np.clip(a, 0.0, 1000.0, out=a)

    """
    Reduce the contribution to the currently maximal convolution entry.
    Moves a small fraction of the two entries that dominate that entry and
    redistributes the removed mass elsewhere.
    """
    n = a.shape[0]
    conv = _conv_full(a)
    k = int(conv.argmax())                 # index where convolution peaks
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_idx = np.arange(i_start, i_end + 1)
    j_idx = k - i_idx
    prod = a[i_idx] * a[j_idx]
    pos = int(prod.argmax())
    i = int(i_idx[pos])
    j = int(j_idx[pos])

    if i == j:                              # diagonal (a_i^2) dominates
        delta = a[i] * step_frac
        if delta <= 0:

    # ------------------------------------------------------------------
    # Initial candidates: uniform step and, if available, the parent.
    # ------------------------------------------------------------------
    cur = np.full(n, 1.0, dtype=np.float64)      # uniform start
    best = cur.copy()
    best_score = _score(best)

    # try to use the supplied parent sequence directly
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.shape == cur.shape:
            s = _score(parent)
            if s < best_score:
                best = parent.copy()
                best_score = s
        # start walk from the better of uniform/parent
        if best_score < _score(cur):
            cur = best.copy()
    except Exception:
        pass

    cur_score = _score(cur)

    # ------------------------------------------------------------------
    # Stochastic search mixing two simple orthogonal moves.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = cur.copy()
        # 30 % of the time we perform a targeted max‑pair move,
        # otherwise a cheap random block scaling.
        if rng.random() < 0.3:
            _maxpair_move(cand, rng, step_frac=0.05)
        else:
            _block_move(cand, rng)

        cand_score = _score(cand)

        # keep the best ever found
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # walk downhill (accept non‑worsening steps)
        if cand_score <= cur_score:
            cur = cand
            cur_score = cand_score

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
