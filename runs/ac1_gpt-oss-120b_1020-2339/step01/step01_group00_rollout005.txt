def hill_climb(a, time_deadline):
    best_local = a.copy()
    best_local_score = score(a)
    step_factor = 0.05  # initial step factor
    while time.monotonic() < time_deadline:
        # compute current conv and max index
        ft = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(ft * ft, fft_len)[:2*n-1]
        max_val = float(conv.max())
        kp = int(conv.argmax())
        # gradient: g_i = 2*a[kp-i] for i where 0<=kp-i<n else 0
        # Build g using vectorized approach:
        # we can compute g by shifting a:
        # effective indices i where i in [max(0,kp-(n-1)), min(n-1,kp)]
        lo = max(0, kp - (n-1))
        hi = min(n-1, kp)
        # idx = lo:hi inclusive
        idx = np.arange(lo, hi+1)
        partners = kp - idx
        # compute subgradient as a[partners] (not doubled to simplify)
        g_vals = a[partners]  # shape same size
        # Determine removal proportion: we want to reduce a at idx proportional to g_vals (larger product)
        # compute removal = step_factor * g_vals
        removal = step_factor * g_vals
        removal = np.minimum(removal, a[idx])
        # reduce a at idx
        a[idx] -= removal
        total_removed = removal.sum()
        if total_removed > 0:
            # distribute total_removed mass uniformly (or to tail)
            # uniform redistribution may raise max elsewhere; maybe better to add to ends:
            # For simplicity, uniform redistribution.
            a += total_removed / n
        # clip, renorm
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0:
            a = uniform.copy()
            s = a.sum()
        a /= s
        # evaluate
        cur_score = score(a)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local = a.copy()
            # possibly reduce step_factor for finer adjustments
            step_factor = max(step_factor * 0.9, 0.001)
        else:
            # if not improved, maybe increase step_factor slightly
            step_factor = min(step_factor * 1.05, 0.2)
    return best_local, best_local_score

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def max_conv_value(a: np.ndarray, fft_len: int) -> float:
    """Full convolution of a with itself via FFT, return its maximum."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def compute_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective used by the evaluator."""
    s = a.sum()
    if s < 1e-12:
        return math.inf
    m = max_conv_value(a, fft_len)
    return 2.0 * n * m / (s * s)


def build_power_shape(i_arr: np.ndarray, offset: float, p: float, beta: float) -> np.ndarray:
    """a_i = (i+offset)^{-p} * exp(-beta*i/n), normalised to sum = 1."""
    a = (i_arr + offset) ** (-p)
    if beta != 0.0:
        a = a * np.exp(-beta * i_arr / i_arr.size)
    total = a.sum()
    if total == 0.0:
        return a
    return a / total


def hill_climb(initial: np.ndarray, n: int, fft_len: int, deadline: float):
    """Simple projected‑gradient hill‑climb that only moves mass away from the
    currently maximal autocorrelation index. Returns the best vector seen."""
    a = initial.copy()
    best = a.copy()
    best_score = compute_score(a, n, fft_len)
    rng = np.random.default_rng(0)
    step_factor = 0.15

    while time.monotonic() < deadline:
        ft = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(ft * ft, fft_len)[: 2 * n - 1]
        kp = int(conv.argmax())

        lo = max(0, kp - (n - 1))
        hi = min(n - 1, kp)
        idx = np.arange(lo, hi + 1)
        partners = kp - idx

        # move a fraction of the partner mass away from the indices that
        # directly contribute to the maximal value
        removal = step_factor * a[partners]
        removal = np.minimum(removal, a[idx])
        a[idx] -= removal
        total_removed = removal.sum()
        if total_removed > 0.0:
            a += total_removed / n   # uniform redistribution

        # occasional random shuffle to escape plateaus
        if rng.random() < 0.02:
            k = max(2, int(0.001 * n))
            chosen = rng.choice(n, size=k, replace=False)
            mass = a[chosen].copy()
            rng.shuffle(mass)
            a[chosen] = mass

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        cur_score = compute_score(a, n, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            step_factor = max(step_factor * 0.9, 0.001)
        else:
            step_factor = min(step_factor * 1.05, 0.3)

    return best, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # ---- problem size -------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = int(base.size)
    except Exception:
        n = 6733                     # fallback length

    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ---- baseline candidates -------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = compute_score(best_seq, n, fft_len)

    # parent (if present)
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n:
            parent_norm = parent if parent.sum() == 0 else parent / parent.sum()
            sc = compute_score(parent_norm, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = parent_norm.copy()
    except Exception:
        pass

    # ---- parametric random search --------------------------------------
    i_arr = np.arange(n, dtype=np.float64)
    max_param_iters = 2000
    for _ in range(max_param_iters):
        if time.monotonic() - start > budget_s - 7.0:
            break

        shape_type = rng.integers(0, 3)          # 0: power, 1: power+exp, 2: pure exp
        offset = rng.uniform(1e-3, 20.0)
        p = rng.uniform(0.3, 1.2)
        beta = 0.0

        if shape_type == 0:                      # pure power‑law
            a = build_power_shape(i_arr, offset, p, beta)

        elif shape_type == 1:                    # power‑law with exponential tail
            beta = rng.uniform(0.0, 5.0)
            a = build_power_shape(i_arr, offset, p, beta)

        else:                                    # pure exponential decay
            lam = rng.uniform(5e-4, 2e-2)         # decay per index
            a = np.exp(-lam * i_arr)
            total = a.sum()
            if total == 0.0:
                continue
            a = a / total

        sc = compute_score(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # ---- hill‑climbing refinement --------------------------------------
    deadline = start + budget_s - 0.5
    if time.monotonic() < deadline:
        best_seq, best_score = hill_climb(best_seq, n, fft_len, deadline)

    # ---- final small mixing with the uniform vector (optional) ---------
    # a light convex combination can sometimes shave a tiny amount off
    for w in np.linspace(0.0, 0.5, 11):
        cand = (1.0 - w) * best_seq + w * uniform
        cand = cand / cand.sum()
        sc = compute_score(cand, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ---- enforce evaluator limits --------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 1e-12:
        best_seq = uniform.copy()
    return best_seq.tolist()
