def propose_candidate(seed=42, budget_s=110, **kwargs):

    # Determine length

    if n <= 0:

    # Ensure in range and sum positive
    init_seq = np.clip(init_seq, 0.0, 1000.0)
    s = init_seq.sum()
    if s < 1e-12:
        init_seq = np.ones(n, dtype=np.float64) / n
    else:
        init_seq = init_seq / s

    # Set convolution size
    size = 1 << ((2 * n - 2).bit_length())

    # fast evaluation
    def fast_ac1(a):

        conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
        maxv = conv.max()
        s = a.sum()
        if s == 0.0:
            return np.inf
        return 2.0 * n * maxv / (s * s)

    best_seq = init_seq.copy()
    best_score = fast_ac1(best_seq)

    # analytic families

def _fast_ac1(a: np.ndarray, size: int) -> float:
    """Fast (FFT‑based) evaluation of the AC1 bound for a non‑negative vector."""
    n = a.shape[0]
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    maxv = conv.max()
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * maxv / (s * s)

    """Sub‑gradient of the k‑th convolution entry (a*a)[k] w.r.t. a."""
