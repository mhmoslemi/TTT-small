def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # get parent if possible
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        n = 256
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
    # function to evaluate
    def eval_seq(seq):
        return float(evaluate_sequence(seq.tolist()))
    # generate families
    # 1: geometric decreasing
    for c in np.linspace(0.5, 3.0, 26):  # c = lambda * n
        if time.monotonic() >= deadline: break
        lam = c / n
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        sc = eval_seq(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    # 2: power decreasing
    p_vals = np.concatenate((
        np.linspace(0.05, 0.5, 10),
        np.linspace(0.6, 2.0, 8),
        np.linspace(2.5, 5.0, 5)
    ))
    idx = np.arange(n, dtype=np.float64)
    for p in p_vals:
        if time.monotonic() >= deadline: break
        t = n - idx  # n, n-1, ..., 1
        cand = np.power(t, p) - np.power(t-1, p)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = eval_seq(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    # 3: reversed power increasing (starting low increasing) maybe not needed
    # Random hill climbing
    while time.monotonic() < deadline:
        width = int(rng.integers(max(1, n // 400), max(2, n // 20)))
        start = int(rng.integers(0, n - width + 1))
        factor = math.exp(rng.normal(0.0, 0.1))
        cand = best.copy()
        cand[start:start+width] *= factor
        # occasional smoothing
        if rng.random() < 0.2:
            cand = (cand + np.roll(cand, 1)) / 2.0
        cand = np.clip(cand, 0.0, 1000.0)
        sc = eval_seq(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    # Return best as list of floats
    return [float(x) for x in best]

import numpy as np
import time
import math

def _score(seq):
    """Exact score using the provided evaluator."""
    # ``evaluate_sequence`` expects a list, but works with any sequence of floats
    return float(evaluate_sequence(seq.tolist()))


def _gen_geom(c, n):
    """Geometric (exponential) decreasing: a_i = exp(-c * i / n)."""
    lam = c / float(n)
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _gen_power_decay(p, n):
    """Power‑law decreasing: a_i = ((n - i) / n) ** p."""
    idx = np.arange(n, dtype=np.float64)
    return ((n - idx) / float(n)) ** p


def _gen_pow_inv(p, n):
    """Inv‑power decreasing: a_i = (i+1) ** (-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _gen_gaussian(ratio, n):
    """Gaussian centred on the middle with sigma = ratio * n."""
    sigma = max(1e-12, ratio * float(n))
    centre = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64) - centre
    return np.exp(-0.5 * (x / sigma) ** 2)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a low‑score non‑negative step sequence."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ----------------------------------------------------------------------

    except Exception:

        best_score = _score(best)

    # ----------------------------------------------------------------------

    # geometric family (parameter c ≈ 0.5 … 3)
    for c in np.linspace(0.5, 3.0, 25):

    # power‑law decreasing: ((n-i)/n)**p  with p ∈ [0.1, 5]
