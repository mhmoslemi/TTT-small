import time
import numpy as np


def compute_score(a: np.ndarray, n: int, fft_len: int):
    """Return (objective, max convolution value, index of max, full conv)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_c = conv.max()
    score = float(2.0 * n * max_c / (a.sum() ** 2))
    return score, float(max_c), int(conv.argmax()), conv


def simplex_projection(v: np.ndarray):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = u[0] - 1.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
