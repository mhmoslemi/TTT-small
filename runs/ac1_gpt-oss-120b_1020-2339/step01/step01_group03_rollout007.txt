    deadline = time.monotonic() + float(budget_s) - 0.5

    except: n = 256

    size = 1 << ((2 * n - 2).bit_length())

    best_seq = None
    best_score = float('inf')
    
    # Helper function to compute AC1 via FFT
    def fast_ac1(a):
        A = np.fft.rfft(a, size)
        conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
        return float(2 * n * conv.max() / (a.sum() ** 2))
    
    # initial candidates:
    # Uniform:
    a = np.full(n, 1.0 / n, dtype=np.float64)
    sc = fast_ac1(a)
    best_seq, best_score = a, sc

    for c in c_vals:

        a = (1 - r) * (r ** np.arange(n, dtype=np.float64))
        sc = fast_ac1(a)
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc

def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs):

        """Compute the AC1 bound using FFT (exact up to floating error)."""
        A = np.fft.rfft(a, size)
        conv = np.fft.irfft(A * A, size)[: 2 * n - 1]

    def _grad_max(a: np.ndarray, k: int) -> np.ndarray:
