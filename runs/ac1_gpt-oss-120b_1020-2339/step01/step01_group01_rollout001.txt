def generate_sidon(n, m, rng):
    positions = []
    sum_set = set()
    attempts = 0
    max_attempts = 10000 * m  # some large
    while len(positions) < m and attempts < max_attempts:
        cand = rng.integers(0, n)
        if cand in positions:
            attempts += 1
            continue
        new_sums = [cand + p for p in positions] + [2*cand]
        # check collisions
        if any(s in sum_set for s in new_sums):
            attempts += 1
            continue
        # add
        positions.append(cand)
        sum_set.update(new_sums)
        attempts = 0   # reset attempts after success
    # if failure, return whatever we have
    return positions
```

def _max_convolution(a: np.ndarray) -> float:
    """Return max of the full convolution a*a using FFT (linear time)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


def _generate_sidon_set(n: int, m: int, rng: np.random.Generator, deadline: float):
    """
    Random greedy construction of a Sidon (B2) set of size m inside {0,…,n‑1}.
    Returns a list of positions or None if it cannot be built before deadline.
    """
    # quick deterministic try (scan in order) – often succeeds for moderate m
    positions = []
    sums_set = set()
    for cand in range(n):
        if time.monotonic() >= deadline:
            return None
        coll = False
        for p in positions:
            if cand + p in sums_set:
                coll = True
                break
        if coll or (2 * cand) in sums_set:
            continue
        for p in positions:
            sums_set.add(cand + p)
        sums_set.add(2 * cand)
        positions.append(cand)
        if len(positions) >= m:
            return positions

    # random greedy retries
    max_outer = 15
    attempts_per = 5000 * m
    while max_outer > 0 and time.monotonic() < deadline:
        max_outer -= 1
        positions = []
        sums_set = set()
        attempts = 0
        while len(positions) < m and attempts < attempts_per:
            if time.monotonic() >= deadline:
                return None
            cand = int(rng.integers(0, n))
            if cand in positions:
                attempts += 1
                continue
            coll = False
            for p in positions:
                if cand + p in sums_set:
                    coll = True
                    break
            if coll or (2 * cand) in sums_set:
                attempts += 1
                continue
            for p in positions:
                sums_set.add(cand + p)
            sums_set.add(2 * cand)
            positions.append(cand)
            attempts = 0            # reset after a successful insertion
        if len(positions) == m:
            return positions
    return None
