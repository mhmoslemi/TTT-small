def fast_max_conv(a, size):
    # returns max of linear convolution via zero-padded FFT
    # a is 1d np.ndarray float64
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)

    # actual conv: conv[:2*len(a)-1]
    # so compute max

def _normalize_and_clip(a: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and keep the sum above the evaluator threshold."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:                     # avoid the forbidden tiny‑sum regime
        if s == 0.0:
            a = np.full_like(a, 0.01 / a.size)
        else:
            a = a * (0.01 / s)
    return a


def _fast_max_conv(a: np.ndarray, size: int) -> float:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns the maximal entry of the (2n‑1)‑length result.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # only the first 2*len(a)-1 entries are the true linear convolution
    effective_len = a.shape[0] * 2 - 1
    return float(conv[:effective_len].max())


def _fast_score(a: np.ndarray, n: int, size: int) -> float:
    """Cheap surrogate of the evaluator's score."""
    s = a.sum()
    if s == 0.0:
        return float("inf")
    M = _fast_max_conv(a, size)
    return 2.0 * n * M / (s * s)


def _gen_power_seq(p: float, n: int) -> np.ndarray:
    """a_i = (i+1)^p - i^p, a classic decreasing power‑law family."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** p - i ** p
    return a

    """a_i = (i+1)^(-alpha)  (pure power law, not a difference)."""
    i = np.arange(1, n + 1, dtype=np.float64)

        cand = _normalize_and_clip(_gen_power_seq(p, n))
        est = _fast_score(cand, n, fft_size)
        if est + 1e-8 < best_score:          # quick guard before expensive call
            off = float(evaluate_sequence(cand.tolist()))
            if off < best_score:
                best_seq, best_score = cand, off

    # Pure power‑law family
    for alpha in np.linspace(0.3, 1.6, 20):
        if time.monotonic() >= deadline:
            break
        cand = _normalize_and_clip(_gen_powlaw_seq(alpha, n))
        est = _fast_score(cand, n, fft_size)
        if est + 1e-8 < best_score:
            off = float(evaluate_sequence(cand.tolist()))
            if off < best_score:
                best_seq, best_score = cand, off

    # ----- randomised search loop -------------------------------------------
    while time.monotonic() < deadline:
        choice = rng.random()
        if choice < 0.35:                                 # Dirichlet draw
            a_par = 10 ** rng.uniform(-0.7, 0.9)           # alpha in ~[0.2,8]
            cand = _normalize_and_clip(_gen_dirichlet(a_par, n, rng))
        elif choice < 0.60:                               # block perturbation
            cand = _perturb_block(best_seq, rng)
            cand = _normalize_and_clip(cand)
        elif choice < 0.80:                               # fresh power‑diff
            p = rng.uniform(0.05, 0.85)
            cand = _normalize_and_clip(_gen_power_seq(p, n))
        elif choice < 0.95:                               # random entry perturb
            cand = _perturb_random_entries(best_seq, rng)
            cand = _normalize_and_clip(cand)
        else:                                              # pure power‑law
            alpha = rng.uniform(0.2, 2.0)
            cand = _normalize_and_clip(_gen_powlaw_seq(alpha, n))

        # cheap surrogate test
        est = _fast_score(cand, n, fft_size)
        if est + 1e-8 < best_score:                       # promising
            off = float(evaluate_sequence(cand.tolist()))
            if off < best_score:
                best_seq, best_score = cand, off

    # ----- final output ------------------------------------------------------
    return [float(x) for x in best_seq]
