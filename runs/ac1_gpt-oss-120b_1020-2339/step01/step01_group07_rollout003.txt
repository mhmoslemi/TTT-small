def greedy_sidon(k, n):
    indices = []
    sum_set = set()
    candidate = 0
    while len(indices) < k and candidate < n:
        conflict = False
        for q in indices:
            if (candidate + q) in sum_set:
                conflict = True
                break
        if not conflict and (2*candidate) not in sum_set:
            indices.append(candidate)
            for q in indices:
                sum_set.add(candidate + q)
        candidate += 1
    return indices

import numpy as np, time, math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Uses a fast FFT‑based hill‑climbing guided by the index where the
    convolution attains its maximum.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine length n from the optional parent sequence.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.shape[0]
    except Exception:
        n = 256                     # fallback length

    # ------------------------------------------------------------------
    # Helper: convolution max (via FFT) and its argument.
    # ------------------------------------------------------------------
    L = _next_pow2(2 * n - 1)

    def conv_max_idx(vec):
        freq = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(freq * freq, n=L)[: 2 * n - 1]
        idx = int(conv.argmax())
        return float(conv[idx]), idx

    # ------------------------------------------------------------------
    # Initial candidate: uniform + tiny noise, then normalise to sum=1.
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += rng.normal(scale=1e-4, size=n)          # break symmetry
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    cur_max, cur_idx = conv_max_idx(a)
    best_a = a.copy()
    best_score = 2.0 * n * cur_max

    # ------------------------------------------------------------------
    # Possibly improve with the provided parent sequence.
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.shape == (n,):
            s = float(evaluate_sequence(parent.tolist()))
            if s < best_score:
                best_a = parent.copy()
                best_score = s
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Hill‑climbing loop.
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        it += 1

        # indices that contribute to the current maximal convolution entry
        lo = max(0, cur_idx - (n - 1))
        hi = min(n - 1, cur_idx)

        # sub‑vector that actually participates
        sub = a[lo : hi + 1]
        if sub.size == 0:
            cur_max, cur_idx = conv_max_idx(a)
            continue

        # choose source (large) and destination (small) positions
        j_max_rel = int(np.argmax(sub))
        j_min_rel = int(np.argmin(sub))
        j_max = lo + j_max_rel
        j_min = lo + j_min_rel
        i_max = cur_idx - j_max
        i_min = cur_idx - j_min

        # safety fallback
        if not (0 <= i_max < n and 0 <= i_min < n) or i_max == i_min:
            i_max = int(rng.integers(0, n))
            i_min = int(rng.integers(0, n))
            if i_max == i_min:
                continue

        # move a substantial part of mass from i_max to i_min
        delta = 0.5 * a[i_max]                 # half of the source weight
        if delta <= 0.0:
            continue

        a[i_max] -= delta
        a[i_min] += delta

        new_max, new_idx = conv_max_idx(a)
        new_score = 2.0 * n * new_max

        # accept if we made progress; otherwise rollback
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if new_score < 2.0 * n * cur_max - 1e-12:
            cur_max, cur_idx = new_max, new_idx
        else:
            # revert the move
            a[i_max] += delta
            a[i_min] -= delta

        # occasional random perturbation to escape plateaus
        if it % 200 == 0 and rng.random() < 0.07:
            width = int(rng.integers(1, max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.04)))
            a[start : start + width] *= factor
            a = np.clip(a, 0.0, None)
            total = a.sum()
            if total > 0.0:
                a /= total
            cur_max, cur_idx = conv_max_idx(a)

        # if we have been stuck for a long while, re‑randomise a bit

            cur_max, cur_idx = conv_max_idx(a)

    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)

    # fallback to the best we have (could be the parent or uniform)
    return [float(x) for x in best_a]
