def proj_simplex(v):
    """Project onto simplex {x >= 0, sum x = 1}."""
    # uses algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def solve_lp_max_sum(f, M):
    n = len(f)
    # Build A_ub
    # Use coo format
    data = []
    row = []
    col = []
    for j in range(n):
        # column j: rows from j to j+n-1 inclusive (size n)
        rows = np.arange(j, j+n)
        row.extend(rows.tolist())
        col.extend([j]*n)
        data.extend(f.tolist())
    A = coo_matrix((data, (row, col)), shape=(2*n-1, n)).tocsr()
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)  # minimize -sum = maximize sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        raise ValueError

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _score(seq: np.ndarray) -> float:
    """AC1 upper bound for a non‑negative sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _proj_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _lp_expand(f, M, remaining):
    """
    Solve  max sum(g)  s.t. (f*g)[k] <= M  for all k,
    g >= 0.
    Returns scaled g (same sum as f) or None on failure/timeout.
    """
    if remaining < 8.0:        # leave head‑room for final steps
        return None
    n = f.shape[0]
    # build sparse matrix A where A[k, j] = f[k-j] (zero outside range)
    # column j contains f[0..n-1] placed at rows j..j+n-1
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel(order='F')
    cols = np.repeat(j, n)
    data = np.tile(f, n)
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # maximize sum(g)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=(0, None),
            method="highs",
            options={"presolve": True, "time_limit": remaining - 2.0},
        )
    except Exception:  # pragma: no cover
        return None
    if not res.success:
        return None
    g = res.x
    sum_f = np.sum(f)
    sum_g = np.sum(g)
    if sum_g <= 0:
        return None
    scale = sum_f / sum_g
    return g * scale
