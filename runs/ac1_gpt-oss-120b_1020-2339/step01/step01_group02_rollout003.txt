def proj_simplex(v):
    """Project onto simplex sum=1, v >=0."""
    # from Duchi et al.
    u = np.sort(v)[::-1] # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
