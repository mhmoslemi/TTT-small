def next_pow2(x):

def compute_conv(a, size):
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2 * a.size - 1]

def compute_score(a, size):
    conv = compute_conv(a, size)
    maxv = conv.max()
    s = a.sum()
    if s == 0:
        return np.inf
    return 2.0 * a.size * maxv / (s * s)

def add_grad_for_index(a, k, out, weight=1.0):
    n = a.size
    if k < n:
        out[:k + 1] += weight * 2.0 * a[k :: -1]
    else:
        start = k - (n - 1)
        length = 2 * n - 1 - k
        out[start:start+length] += weight * 2.0 * a[:length][::-1]

def gradient_multi(a, indices, weight=1.0):

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """smallest power of two >= x"""
    return 1 << ((x - 1).bit_length())

def _conv_max(a: np.ndarray, size: int) -> float:
    """maximal entry of a*a using FFT (a is real)"""
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * a.size - 1]
    return float(conv.max())

def _add_grad_for_index(a: np.ndarray, k: int, out: np.ndarray, weight: float = 1.0) -> None:
    """add sub‑gradient of the k‑th convolution entry (scaled by weight) to out."""
    n = a.size
    if k < n:
        out[: k + 1] += weight * 2.0 * a[k :: -1]
    else:
        start = k - (n - 1)
        length = 2 * n - 1 - k
        out[start:start + length] += weight * 2.0 * a[:length][::-1]

def _gradient_multi(a: np.ndarray, idx: np.ndarray) -> np.ndarray:
    """average sub‑gradient over several indices."""
    g = np.zeros_like(a)
    if len(idx) == 0:
        return g
    w = 1.0 / len(idx)
    for k in idx:
        _add_grad_for_index(a, int(k), g, weight=w)
    return g

def _candidate_score(a: np.ndarray, size: int) -> float:
    """AC1 score assuming a.sum() == 1 (i.e. 2*n*max(conv))."""
    maxv = _conv_max(a, size)
    return 2.0 * a.size * maxv

    """Return a non‑negative step‑function (as a Python list) whose AC1 bound is low."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Determine problem size and FFT length.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                      # pragma: no cover
        n = 256
        parent = None
    if n <= 0:
        n = 256
    size = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2.  Scan a few compact analytic families and keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def consider(a):
        nonlocal best_seq, best_score
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        sc = _candidate_score(a, size)
        if sc < best_score:
            best_seq = a.copy()
            best_score = sc

    # uniform
    consider(np.full(n, 1.0 / n))

    # parent (if supplied)
    if parent is not None:
        p = np.clip(parent, 0.0, 1000.0)
        consider(p)

    for c in np.linspace(0.2, 5.0, 40):

        consider(a)

    # power‑law families (decreasing and increasing)
    for p in np.linspace(-2.5, 2.5, 61):
        # decreasing
        a = ((np.arange(1, n + 1, dtype=np.float64) ** p)[::-1])
        consider(a)
        # increasing
        a = (np.arange(1, n + 1, dtype=np.float64) ** p)
        consider(a)

    # single bell (Gaussian)
    x = np.arange(n, dtype=np.float64)
    sigma = n / 5.0
    a = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
    consider(a)

    # ------------------------------------------------------------------
    # 3.  Refine the best candidate by a multi‑index sub‑gradient descent.
    # ------------------------------------------------------------------
    a = best_seq.copy()

        if cur_score < target:

            best_score = cur_score
            break
