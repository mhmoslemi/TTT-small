def ac1_bound(seq):
    s = seq.sum()
    if s < 0.01: return np.inf
    n = len(seq)
    # next power of 2 size: 2*n - 1 length, so conv_len = 2n-1
    L = 1
    while L < 2*n-1:
        L <<= 1
    # FFT
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_c = conv.max()
    return 2 * n * max_c / (s*s)

def ac1_bound(seq):
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    n = len(seq)
    # compute convolution via FFT
    L = 1
    while L < 2*n-1:
        L <<= 1
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_c = conv.max()
    return float(2 * n * max_c / (s * s))

def propose_candidate(seed=42, budget_s=110, **kwargs):

  import numpy as np, time

  rng = np.random.default_rng(seed)

  deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

  # find n
  try:
      base = np.asarray(height_sequence_1, dtype=np.float64)
      if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
          n = base.shape[0]
      else:
          raise ValueError
  except Exception:
      n = 256

  # fast bound function
  def ac1_bound(seq):
      s = seq.sum()
      if s < 0.01:
          return float('inf')
      n_local = seq.shape[0]
      # compute convolution via FFT
      L = 1
      while L < 2 * n_local - 1:
          L <<= 1
      fa = np.fft.rfft(seq, n=L)
      conv = np.fft.irfft(fa * fa, n=L)[:2 * n_local - 1]
      max_c = conv.max()
      return float(2 * n_local * max_c / (s * s))

  # baseline uniform
  best_seq = np.ones(n, dtype=np.float64)
  best_score = ac1_bound(best_seq)

  # check parent if exists
  try:
      parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
      if parent.shape == best_seq.shape:
          s_parent = ac1_bound(parent)
          if s_parent < best_score:
              best_seq = parent.copy()
              best_score = s_parent
  except Exception:
      pass

  # building binary sequences
  # function
  def build_binary(n_local, max_allowed, rng):

      # but keep simple.

import time
import numpy as np

def build_binary(n, max_allowed, rng):
    """Greedy construction of a 0‑1 vector whose convolution never exceeds max_allowed.
    Returns the vector, its ℓ₁‑norm (number of ones) and the actual maximal convolution entry."""
    sum_counts = [0] * (2 * n - 1)          # counts of ordered pair contributions for each shift
    positions = []                         # selected indices
    order = rng.permutation(n)             # random order each run
    for i in order:
        # diagonal contribution (i,i)
        if sum_counts[2 * i] + 1 > max_allowed:
            continue
        ok = True
        for j in positions:                # test all already chosen positions
            s = i + j
            if sum_counts[s] + 2 > max_allowed:
                ok = False
                break
        if not ok:
            continue
        # accept i
        sum_counts[2 * i] += 1
        for j in positions:
            sum_counts[i + j] += 2          # (i,j) and (j,i)
        positions.append(i)
    seq = np.zeros(n, dtype=np.float64)
    if positions:
        seq[positions] = 1.0
    max_c = max(sum_counts) if sum_counts else 0
    return seq, len(positions), max_c


def bound_score(n, m, max_c):
    """Exact AC1 bound for a 0‑1 vector of length n, ℓ₁‑norm m, max convolution entry max_c."""
    if m == 0:
        return float('inf')
    return 2.0 * n * max_c / (m * m)


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Determine length n (use the provided parent if possible)
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.shape[0])
    except Exception:
        n = 256                     # fallback length when no parent is present

    # ----------------------------------------------------------------------
    # Baseline: uniform vector (score = 2)
    # ----------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64)
    best_score = 2.0

    # ----------------------------------------------------------------------
    # Try the supplied parent sequence as a fallback candidate
    # ----------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.shape == best_seq.shape:
            # use the official evaluator once – it is cheap for a single call
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best_seq = parent.copy()
                best_score = parent_score
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Main stochastic search: build binary vectors with a bounded convolution peak.
    # The bound for the AC1 metric is 2·n·max_c / m², where m = #ones.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random upper bound for the convolution peak (higher allows more ones)
        max_allowed = int(rng.integers(5, 25))   # choose between 5 and 24 inclusive
        seq, m, max_c = build_binary(n, max_allowed, rng)
        if m == 0:
            continue
        cand_score = bound_score(n, m, max_c)
        if cand_score < best_score:
            best_seq = seq
            best_score = cand_score
            # early exit if we are already comfortably below the target
            if best_score < 1.45:
                break

    # ----------------------------------------------------------------------
    # Return the best-found vector as a plain Python list of floats.
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
