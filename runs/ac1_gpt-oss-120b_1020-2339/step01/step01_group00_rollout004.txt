def project_simplex(v):

def project_simplex(v):
    # Duchi et al.
    n = v.shape[0]
    # ensure not all zero
    if v.max() <= 0:
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time
import math

def _proj_simplex(v):
    """Projection of v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    if v.max() <= 0:
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    """
    Find a non‑negative step‑function (list of floats) that lowers the
    AC1 upper bound.  The routine builds a few analytical starts,
    picks the best, then refines it with a projected exponentiated‑gradient
    descent followed by a lightweight hill‑climbing phase.
    """
    rng = np.random.default_rng(seed)

        """Return (max value, index of max, full convolution)."""
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

    # ------------------------------------------------------------------

        if s < 1e-12:
            return np.full_like(v, 1.0 / n)
        return v / s

    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # parent if available
    if base is not None:
        parent = normalize(base)
        candidates.append(parent)

    # inverse‑sqrt family
    offsets = np.array([0.0, 0.5, 1.0, 2.0, 5.0, 10.0, 100.0, 1000.0])
    i = np.arange(n, dtype=np.float64)
    for off in offsets:

    # linear decreasing shape

    candidates.append(normalize(lin))

    # pick the best start according to the raw AC1 bound
    best_start = None
    best_score = math.inf
    for a in candidates:

        score = 2.0 * n * M  # sum(a)=1 after normalise
        if score < best_score:

            rng.shuffle(mass)
            a[ridx] = mass

    # ------------------------------------------------------------------

    if best_a.sum() < 1e-12:
        best_a = np.full(n, 1.0 / n)
    return best_a.tolist()
