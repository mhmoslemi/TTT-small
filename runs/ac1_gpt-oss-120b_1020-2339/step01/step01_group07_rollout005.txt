def propose_candidate(seed=42, budget_s=110, **kwargs):

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    a = rng.exponential(scale=1.0, size=n) + 1e-6
    a = a / a.sum()                     # normalise sum to 1
    best = a.copy()

    # ----------------------------------------------------------------------
    # Warm‑start from the provided parent if it is better
    # ----------------------------------------------------------------------

            if s < best_score:
                best = parent.copy()
                best_score = s
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Main stochastic‑gradient loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # 60 % of the time use a targeted reduction of the peak convolution
        if rng.random() < 0.6:
            conv = np.convolve(best, best)
            t = int(conv.argmax())                     # index of current maximum
            idx = np.arange(n)
            partner = t - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(best)
            contrib[mask] = best[mask] * best[partner[mask]]

            # indices with the largest contribution (at least 1 % of n)
            topk = max(1, n // 100)
            if topk < n:
                high = np.argpartition(contrib, -topk)[-topk:]
            else:
                high = np.arange(n)

            cand[high] *= factor
