import numpy as np

def _score(a: np.ndarray) -> float:
    """Return the AC1 upper‑bound score for a non‑negative step function."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max() / (s * s)

    # 6.  Final safety adjustment (should never trigger)
    # ----------------------------------------------------------------------
    if best.sum() < 0.01:
        best = np.clip(best, 0.0, 1000.0)
        if best.sum() < 0.01:
            best[0] = 0.01

    return [float(v) for v in best.tolist()]
