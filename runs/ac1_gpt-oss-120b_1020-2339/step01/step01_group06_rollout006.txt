import numpy as np, time, math

def propose_candidate(seed=42, budget_s=110, **kwargs):
    # Determine n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = int(base.shape[0])
            base_seq = np.clip(base, 0.0, 1000.0)
        else:
            n = 256
            base_seq = None
    except Exception:
        n = 256
        base_seq = None

    max_val = 1000.0
    idx = np.arange(n, dtype=np.float64)
    size_fft = 1 << ((2 * n - 1).bit_length())

    def fast_score(seq):
        # seq is numpy array float64, values within bounds
        # Compute convolution via FFT
        f = np.fft.rfft(seq, n=size_fft)
        conv = np.fft.irfft(f * f, n=size_fft)[:2 * n - 1]
        max_b = np.max(conv)
        sum_a = np.sum(seq)
        if sum_a < 0.01:
            return float('inf')
        return 2.0 * n * max_b / (sum_a * sum_a)

    best_seq = None
    best_score = float('inf')
    # Helper to update best
    def consider(seq):
        nonlocal best_seq, best_score
        # ensure within bounds
        # Already seq within [0, max_val] due to construction
        s = fast_score(seq)
        if s < best_score:
            best_score = s
            best_seq = seq.copy()

    # add uniform baseline
    uniform_seq = np.full(n, max_val, dtype=np.float64)
    consider(uniform_seq)

    if base_seq is not None:
        consider(base_seq)

    # Power law decreasing
    for p in np.linspace(0.5, 10.0, 40):
        seq = max_val * ((n - 1 - idx) / (n - 1)) ** p
        consider(seq)

    d = np.minimum(idx, n - 1 - idx) / half
    for p in np.linspace(0.5, 8.0, 30):

        consider(seq)

    # Exponential decreasing
    for a in np.linspace(0.0005, 0.015, 30):
        seq = max_val * np.exp(-a * idx)
        consider(seq)

    # Random perturbation loop
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    while time.monotonic() < deadline:
        # perturb best_seq globally: multiply by random factor near 1

import numpy as np
import time

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Search for a step‑function that minimizes the AC1 upper bound
    2*n*max(conv(a,a))/(sum(a)^2).  The search explores a few parametric
    families (power‑law, V‑shaped, exponential) and then performs cheap
    stochastic refinements.  All work is done with linear / FFT‑based
    convolution, i.e. O(n log n) memory and time.
    """
    # ------------------------------------------------------------------
    # Determine length and a fallback parent (if any)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = int(base.shape[0])
            parent = np.clip(base, 0.0, 1000.0)
        else:
            raise Exception
    except Exception:
        n = 256                      # modest fallback length
        parent = None

    max_val = 1000.0
    idx = np.arange(n, dtype=np.float64)

    # pre‑compute FFT size (next power of two covering full convolution)
    fft_len = 1 << ((2 * n - 1).bit_length())

    def fast_score(seq: np.ndarray) -> float:
        """Fast evaluation of the objective using an FFT‑based convolution."""
        f = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
        max_b = np.max(conv)
        sum_a = np.sum(seq)
        if sum_a < 0.01:
            return float("inf")
        return 2.0 * n * max_b / (sum_a * sum_a)

    # ------------------------------------------------------------------
    # Keep the best candidate seen so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def consider(seq: np.ndarray):
        nonlocal best_seq, best_score
        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # uniform all‑max (baseline, objective = 2)
    consider(np.full(n, max_val, dtype=np.float64))

    # parent, if provided
    if parent is not None:
        consider(parent)

    # ------------------------------------------------------------------
    # Parametric families
    # ------------------------------------------------------------------
    # 1) decreasing power‑law a_i = max_val * ((n-1-i)/(n-1))**p
    for p in np.linspace(0.5, 10.0, 45):
        seq = max_val * ((n - 1 - idx) / (n - 1)) ** p
        consider(seq)

    # 2) V‑shaped: high at both ends, low in the middle
    half = (n - 1) / 2.0
    d = np.minimum(idx, n - 1 - idx) / half                     # 0 at ends, 1 in centre
    for p in np.linspace(0.5, 8.0, 30):
        seq = max_val * (1.0 - d ** p)
        consider(seq)

    # 3) exponential decay a_i = max_val * exp(-alpha*i)
    for alpha in np.linspace(5e-4, 1.5e-2, 30):
        seq = max_val * np.exp(-alpha * idx)
        consider(seq)

    # ------------------------------------------------------------------
    # Light stochastic refinement (time‑bounded)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    while time.monotonic() < deadline:
        # tiny global scaling (does not change the objective but explores clipping)
        factor = 1.0 + rng.normal(0.0, 0.012)
        cand = np.clip(best_seq * factor, 0.0, max_val)
        sc = fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

        # occasional block perturbation
        if rng.random() < 0.3:
            block_len = int(rng.integers(max(1, n // 250), max(2, n // 50)))
            start = int(rng.integers(0, n - block_len + 1))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand = best_seq.copy()
            cand[start:start + block_len] *= factor
            cand = np.clip(cand, 0.0, max_val)
            sc = fast_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()

    # ------------------------------------------------------------------
    # Final validation with the authoritative evaluator and return list
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, max_val, dtype=np.float64)

    # The evaluator performs the same clipping, but we do it explicitly
    final = np.clip(best_seq, 0.0, max_val)

        _ = evaluate_sequence(final.tolist())
    except Exception:
        pass

    return [float(x) for x in final]
