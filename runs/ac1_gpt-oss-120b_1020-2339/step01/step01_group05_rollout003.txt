import time
import numpy as np

def _random_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random point on the simplex (Dirichlet(1))."""
    x = rng.exponential(1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return x / total


def _perturb(candidate: np.ndarray, rng: np.random.Generator,
             sigma: float) -> np.ndarray:
    """
    Produce a new candidate from ``candidate``.
    A random contiguous block is scaled, optionally symmetrised,
    and the whole vector is renormalised to keep the L1‑norm equal.
    """
    n = candidate.size
    blk_len = int(rng.integers(1, max(2, n // 8)))
    start = int(rng.integers(0, max(1, n - blk_len + 1)))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand = candidate.copy()
    cand[start:start + blk_len] *= factor

    # occasional averaging with the reversed sequence (helps anti‑correlation)
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])

    # renormalise to keep the total mass fixed (scaling does not affect the score)
    total = cand.sum()
    if total > 0.0:
        cand /= total
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs) -> list:
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- determine length -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = int(parent.size)
        else:
            n = 256
    except Exception:
        n = 256

    # ----- initialise best candidate ----------------------------------------
    # uniform distribution (scale does not matter, we keep L1‑norm = 1)
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # use the supplied parent if it is better (after normalising its sum)
    try:
        if parent.size == n:
            p = np.clip(parent, 0.0, 1000.0)
            s = p.sum()
            if s > 0.0:
                p = p / s
                sc = float(evaluate_sequence(p.tolist()))
                if sc < best_score:
                    best = p.copy()
                    best_score = sc
    except Exception:
        pass

    # ----- phase 1: random exploration --------------------------------------
    # spend a fraction of the budget on pure random points
    random_phase_time = 0.15 * (deadline - time.monotonic())
    while time.monotonic() < deadline - random_phase_time:
        cand = _random_candidate(rng, n)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ----- phase 2: local stochastic optimisation ---------------------------
    # start from the best found so far and perform many small perturbations
    current = best.copy()
    sigma = 0.5  # initial perturbation strength
    # schedule for sigma: exponential decay towards 0.02
    sigma_min = 0.02
    decay_rate = 0.9995

    while time.monotonic() < deadline:
        cand = _perturb(current, rng, sigma)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            current = cand
        else:
            # occasional move to a previously good point to escape plateaus
            if rng.random() < 0.01:
                current = best.copy()
        sigma = max(sigma_min, sigma * decay_rate)

    # return the best sequence discovered (as plain Python floats)
    return [float(v) for v in best]
