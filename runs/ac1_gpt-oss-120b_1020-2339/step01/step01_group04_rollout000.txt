def fast_score(a):
    # compute convolution via FFT
    n = len(a)
    # Next power of 2 for conv length
    m = 2*n -1
    size = 1 << (m-1).bit_length()
    FA = np.fft.rfft(a, size)
    conv = np.fft.irfft(FA*FA, size)[:m]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)
```

def move_mass(base):
    cand = base.copy()
    # choose donor index i where cand[i] > 0
    nonzero = np.where(cand > 1e-12)[0]
    if nonzero.size == 0:
        return cand
    i = rng.choice(nonzero)
    # choose receiver j (could be same but avoid)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    # fraction
    frac = rng.uniform(0.01, 0.2)
    delta = cand[i] * frac
    # cap delta if receiver near limit
    max_add = 1000.0 - cand[j]
    if delta > max_add:
        delta = max_add
    # ensure donor leaves at least 0
    if delta > cand[i]:
        delta = cand[i]
    cand[i] -= delta
    cand[j] += delta
    return cand
```

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _gen_power_seq(p, n):
    """Generate a decreasing power‑law step sequence."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** p - i ** p
    return np.clip(a, 0.0, 1000.0)


def _fast_score(a, n=None, _fft_size_cache={}):
    """FFT based estimate of the evaluator score (exact up to round‑off)."""
    if n is None:
        n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    # reuse size for given n to avoid recomputation
    if n not in _fft_size_cache:
        _fft_size_cache[n] = size
    size = _fft_size_cache[n]

    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    max_b = conv.max()
    s = a.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _move_mass(base, rng, target_sum):
    """Transfer a random fraction of mass from one index to another."""
    cand = base.copy()
    nz = np.where(cand > 1e-12)[0]
    if nz.size == 0:
        return cand
    i = rng.choice(nz)
    j = rng.integers(0, len(cand))
    while j == i:
        j = rng.integers(0, len(cand))
    frac = rng.uniform(0.01, 0.2)
    delta = cand[i] * frac
    # respect upper bound on receiver
    delta = min(delta, 1000.0 - cand[j])
    delta = min(delta, cand[i])
    cand[i] -= delta
    cand[j] += delta
    # renormalise sum exactly
    cur = cand.sum()
    if cur != 0.0:
        cand *= (target_sum / cur)
    return np.clip(cand, 0.0, 1000.0)


def _block_scale(base, rng, target_sum):
    """Scale a random contiguous block."""
    n = base.size
    cand = base.copy()
    # block size between n/250 and n/12
    lo = max(1, n // 250)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:start + width] *= factor
    cand = np.clip(cand, 0.0, 1000.0)
    # renormalise sum exactly
    cur = cand.sum()
    if cur != 0.0:
        cand *= (target_sum / cur)
    # occasional symmetrisation
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
