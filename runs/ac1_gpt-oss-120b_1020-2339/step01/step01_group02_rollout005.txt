import numpy as np, time
from scipy.optimize import linprog

def _conv_full(a):
    n = a.shape[0]
    L = 1 << (int(np.ceil(np.log2(2*n - 1))))
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv

def _score_arr(a):
    n = a.shape[0]
    conv = _conv_full(a)
    maxB = conv.max()
    sumA = a.sum()
    if sumA < 0.01:
        return np.inf
    return 2.0 * n * maxB / (sumA * sumA)

def _maybe_update(seq, best_seq_ref):
    # best_seq_ref is [best_seq, best_score] list
    best_seq, best_score = best_seq_ref
    sc = _score_arr(seq)
    if sc < best_score:
        best_seq_ref[0] = seq.copy()
        best_seq_ref[1] = sc
    return

def _transfer_move(a, rng):
    n = a.shape[0]
    conv = _conv_full(a)
    kmax = int(np.argmax(conv))
    i_start = max(0, kmax - (n - 1))
    i_end = min(kmax, n - 1)
    i_range = np.arange(i_start, i_end+1)
    j_range = kmax - i_range
    contrib = a[i_range] * a[j_range]
    if contrib.size == 0:
        return None
    pair_idx = int(np.argmax(contrib))
    i_max = i_range[pair_idx]
    j_max = j_range[pair_idx]
    donor = i_max if a[i_max] >= a[j_max] else j_max
    # receiver is the smallest entry (to spread mass)
    recv = int(np.argmin(a))
    if recv == donor:
        # pick second smallest
        sorted_idx = np.argsort(a)
        recv = sorted_idx[0] if sorted_idx[0] != donor else sorted_idx[1]
    # transfer fraction of donor
    frac = float(rng.uniform(0.05, 0.15))
    delta = a[donor] * frac
    delta = min(delta, 1000.0 - a[recv])
    if delta <= 0:
        return None
    cand = a.copy()
    cand[donor] -= delta
    cand[recv] += delta
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def _smooth_move(a, rng):
    n = a.shape[0]
    # random window size between 5 and 31 odd
    w = int(rng.integers(5, 33))
    if w % 2 == 0:
        w += 1
    # random sigma for Gaussian kernel
    sigma = rng.uniform(0.5, 2.0)
    half = w // 2
    xs = np.arange(-half, half+1)
    kernel = np.exp(-0.5 * (xs / sigma) ** 2)
    kernel = kernel / kernel.sum()
    # convolve using np.convolve (mode='same')
    smooth = np.convolve(a, kernel, mode='same')
    # mix a and smooth a bit
    t = rng.uniform(0.05, 0.2)
    cand = (1 - t) * a + t * smooth
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def _lp_step(a, rng, top_k=200, tmix=0.25):
    n = a.shape[0]
    conv = _conv_full(a)
    M = conv.max()
    total_shifts = conv.shape[0]
    if total_shifts <= top_k:
        ks = np.arange(total_shifts, dtype=np.int64)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    ks = np.sort(ks)   # optional
    m = ks.shape[0]
    # Build constraint matrix
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        A[row, j_vals] = a[i_vals]
    bounds = [(0.0, 1000.0)] * n
    c = -np.ones(n, dtype=np.float64)   # maximize sum(b)
    ub = np.full(m, M, dtype=np.float64)
    # Solve with HiGHS
    try:
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None
    b = res.x
    sum_a = a.sum()
    sum_b = b.sum()
    if sum_b == 0:
        return None
    # Scale b to keep total sum unchanged
    scale = sum_a / sum_b
    b_scaled = b * scale
    cand = (1 - tmix) * a + tmix * b_scaled
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

import numpy as np
import time
from scipy.optimize import linprog

# ---------- convolution and scoring utilities ----------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a via FFT (real)."""
    n = a.shape[0]
    L = 1 << (int(np.ceil(np.log2(2 * n - 1))))  # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _score_arr(a: np.ndarray) -> float:
    """Exact AC1 upper bound for a (non‑negative, clipped)."""
    n = a.shape[0]
    conv = _conv_full(a)
    maxB = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxB / (s * s)


# ---------- candidate updates ----------
def _maybe_update(seq: np.ndarray, best_ref: list):
    """Update best sequence if seq improves the score."""
    sc = _score_arr(seq)
    if sc < best_ref[1]:
        best_ref[0] = seq.copy()
        best_ref[1] = sc


def _transfer_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Move a fraction of mass from a large contributor to the smallest entry."""
    n = a.shape[0]
    conv = _conv_full(a)
    k = int(np.argmax(conv))

    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_rng = np.arange(i_start, i_end + 1)
    j_rng = k - i_rng
    contrib = a[i_rng] * a[j_rng]
    if contrib.size == 0:
        return None
    pair = int(np.argmax(contrib))
    i_max = i_rng[pair]
    j_max = j_rng[pair]

    donor = i_max if a[i_max] >= a[j_max] else j_max
    recv = int(np.argmin(a))
    if recv == donor:
        # pick a different tiny entry
        srt = np.argsort(a)
        recv = srt[0] if srt[0] != donor else srt[1]

    frac = float(rng.uniform(0.05, 0.15))
    delta = a[donor] * frac
    delta = min(delta, 1000.0 - a[recv])
    if delta <= 0.0:
        return None

    cand = a.copy()
    cand[donor] -= delta
    cand[recv] += delta
    return np.clip(cand, 0.0, 1000.0)


def _smooth_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Blend a with a short Gaussian blur."""
    n = a.shape[0]
    w = int(rng.integers(5, 33))
    if w % 2 == 0:
        w += 1
    sigma = rng.uniform(0.5, 2.0)
    half = w // 2
    xs = np.arange(-half, half + 1)
    ker = np.exp(-0.5 * (xs / sigma) ** 2)
    ker = ker / ker.sum()
    smooth = np.convolve(a, ker, mode='same')
    t = rng.uniform(0.05, 0.2)
    cand = (1 - t) * a + t * smooth
    return np.clip(cand, 0.0, 1000.0)


def _lp_step(a: np.ndarray, rng: np.random.Generator,
             top_k: int = 250, mix: float = 0.3) -> np.ndarray | None:
    """
    Linear‑programming refinement.
    Maximize sum(b) subject to (a * b)_k ≤ M for the top `top_k` shifts.
    """
    n = a.shape[0]
    conv = _conv_full(a)
    M = conv.max()
    total_shifts = conv.shape[0]

    if total_shifts <= top_k:
        ks = np.arange(total_shifts, dtype=np.int64)
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]
    ks = np.sort(ks)

    m = ks.size
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        A[row, j_vals] = a[i_vals]

    bounds = [(0.0, 1000.0)] * n
    c = -np.ones(n, dtype=np.float64)          # maximize sum(b)
    ub = np.full(m, M, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None

    b = res.x
    sum_a = a.sum()
    sum_b = b.sum()
    if sum_b == 0.0:
        return None
    scale = sum_a / sum_b
    b_scaled = b * scale
    cand = (1 - mix) * a + mix * b_scaled
    return np.clip(cand, 0.0, 1000.0)
