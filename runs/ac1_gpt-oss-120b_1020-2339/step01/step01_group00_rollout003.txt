def project_simplex(v):
    # v = np.asarray(v)

def propose_candidate(seed=42, budget_s=110, **kwargs):
    import numpy as np, time, math

    # get n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n = base.size
    except Exception:
        n = 6000

    fft_len = 1 << ((2*n - 1).bit_length())

    def conv_max(a):
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        return conv.max()

    def conv_full(a):
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        return conv

    def compute_bound(a):
        max_val = conv_max(a)
        return 2 * n * max_val   # sum a =1

    # Project onto simplex
    def proj_simplex(v):
        # v is 1D numpy array
        # elements can be any real
        # algorithm from Duchi et al.
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
        if rho.size == 0:
            # all zero? return uniform?
            return np.full_like(v, 1.0 / len(v))
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # Build initial shapes
    def build_inv_sqrt(off):
        i = np.arange(n, dtype=np.float64) + off
        a = 1.0 / np.sqrt(i + 1e-30)
        a /= a.sum()
        return a

    # candidate list
    offsets = np.concatenate((np.array([0.0,0.5,1.0,2.0,5.0,10.0]),
                               np.logspace(1,3,5)))  # maybe 5 values.
    best_a = None
    best_val = math.inf

    for off in offsets:

        val = compute_bound(a)
        if val < best_val:

            best_a = a

    # Try uniform as fallback
    uniform = np.full(n, 1.0 / n)

        best_val = val_unif
        best_a = uniform

    # make sure sum > 0.01

        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        start_candidates.append(a)

    # polynomial / triangular shape - using i*(n-1-i)

    a_tri = i * (n-1 - i)  # shape zero at ends

        a_sum = a.sum()

            start_candidates.append(a)

    for a in start_candidates:

        if val < best_val:

    # now run iterative refinement

            return np.full_like(v, 1.0 / v.shape[0])
        rho = rho[-1]

        w = np.maximum(v - theta, 0)
        return w

        i = np.arange(n, dtype=np.float64) + off

        # ensure no nan for i=0

        a /= a.sum()

import numpy as np
import time
import math

def proj_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(u.shape[0]) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a, fft_len):
    """Full convolution a*a using FFT (real)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv

def bound_of(a, n, fft_len):
    """Upper bound value 2*n*max(a*a) assuming sum(a)=1."""
    conv = full_convolution(a, fft_len)
    return 2.0 * n * conv.max()

def build_pow(off, p, n):
    """a_i = (i+off)^{-p}, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64) + off + 1e-12
    a = np.power(i, -p)
    a /= a.sum()
    return a

def build_exp(lam, n):
    """a_i = exp(-lam*i), normalised to sum 1."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
