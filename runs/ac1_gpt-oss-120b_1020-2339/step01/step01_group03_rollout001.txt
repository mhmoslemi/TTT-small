    else:

    best_seq = a.copy()

    size = 1 << ((2*n-2).bit_length())
    # define function to compute score
    def compute_score(vec):
        A = np.fft.rfft(vec, size)
        conv = np.fft.irfft(A * A, size)[:2*n-1]
        maxc = conv.max()
        return float(2 * n * maxc)  # denominator sum^2=1
    best_score = compute_score(a)
    best_seq = a.copy()
    # time limit
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5) # slight headroom

    # gradient descent parameters
    it = 0
    step0 = 0.8  # initial step
    beta = 0.9   # momentum
    v = np.zeros_like(a)
    eps_max = 0.001  # tolerance for near-maximum indices
    # also store current score
    cur_score = best_score
    while time.monotonic() < deadline:
        it += 1
        # compute conv
        A = np.fft.rfft(a, size)
        conv = np.fft.irfft(A * A, size)[: 2*n - 1]
        maxc = conv.max()

        thresh = maxc * (1 - eps_max)
        near_idx = np.nonzero(conv >= thresh)[0]
        # compute gradient sum over near indices
        g = np.zeros_like(a)
        for k in near_idx:
            start = max(0, k - (n - 1))
            end = min(n, k + 1)
            # i in [start, end)
            # g[i] += 2*a[k - i]
            i = np.arange(start, end)
            g[i] += 2.0 * a[k - i]

        # step size schedule
        step = step0 / np.sqrt(it)
        # momentum update
        v = beta * v - step * g
        a_new = a + v
        # projection onto nonnegatives
        np.maximum(a_new, 0, out=a_new)
        s = a_new.sum()
        if s <= 1e-12:
            # revert momentum and continue
            v = np.zeros_like(a)
            continue

        # keep best

def compute_ac1_score(arr: np.ndarray, n: int, size_fft: int) -> float:
    """Exact AC1 upper bound for a non‑negative sequence."""
    A = np.fft.rfft(arr, size_fft)
    conv = np.fft.irfft(A * A, size_fft)[: 2 * n - 1]
