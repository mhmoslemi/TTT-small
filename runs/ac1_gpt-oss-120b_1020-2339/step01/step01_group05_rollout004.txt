import time
import numpy as np

def _objective(arr: np.ndarray) -> float:
    """Exact same formula used by evaluate_sequence (without clipping)."""
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

    """
    Search for a non‑negative step‑function that minimizes the AC1 upper bound.
    Uses a fast local‑move hill‑climbing guided by the current convolution peak.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # ----- initialise ---------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        a = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback: modest length uniform >0
        L = 256
        a = np.full(L, 1.0, dtype=np.float64)

    n = a.size
    best = a.copy()
    best_score = _objective(best)

    # ----- main loop ----------------------------------------------------------
    # Parameters for the stochastic move
    max_frac_move = 0.20          # up to 20 % of the transferable mass
    small_sym_prob = 0.02        # occasional symmetrisation
    block_change_prob = 0.02     # occasional block scaling (like previous code)

    # Keep a copy of the current working vector (starts from best)
    cur = best.copy()

    while time.monotonic() < deadline:
        # current convolution and location of its maximum
        conv_cur = np.convolve(cur, cur)
        k_max = int(conv_cur.argmax())

        # overlapping interval for this shift
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)

        # sample an index i that contributes heavily to the peak
        idx_range = np.arange(lo, hi + 1)
        partner = k_max - idx_range
        contrib = cur[idx_range] * cur[partner]
        if contrib.sum() > 0.0:
            probs = contrib / contrib.sum()
            i = int(rng.choice(idx_range, p=probs))
        else:
            i = int(rng.integers(0, n))

        # pick a destination index j preferably outside the overlapping region
        if n - (hi - lo + 1) > 0:
            # pick until outside
            while True:

                if j < lo or j > hi:

        else:

            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1

        # decide how much to move

        if max_move <= 0.0:
