            best_score = sc

    # ----------------------------------------------------------------------
    # Light‑weight local search: random window shuffling and tiny scaling.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a random contiguous block
        start = int(rng.integers(0, n))
        width = int(rng.integers(1, max(2, n // 20)))   # modest block size
        stop = min(n, start + width)

        cand = best.copy()
        # permute the selected block
        cand[start:stop] = rng.permutation(cand[start:stop])

        # occasional global scaling (preserves shape, may escape plateaus)
        if rng.random() < 0.25:
            factor = float(np.exp(rng.normal(0.0, 0.04)))
            cand *= factor

        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
