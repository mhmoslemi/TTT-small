import numpy as np, time

def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Get parent if exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.shape[0]
    # Helper scoring function
    def score(seq):
        conv = np.convolve(seq, seq)
        return float(2.0 * n * np.max(conv) / (np.sum(seq) ** 2))

    best_seq = parent.copy()
    best_score = score(best_seq)

    # parametric families
    idx = np.arange(n, dtype=np.float64)

    # store candidates in a list of sequences to test
    def maybe_update(seq):
        nonlocal best_seq, best_score
        sc = score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc

    # Exponential families
    for a in np.logspace(-5, -2, num=12):
        maybe_update(np.exp(-a * idx))

    # Symmetric exponential
    d = np.minimum(idx, n-1-idx)
    for a in np.logspace(-5, -2, num=12):
        maybe_update(np.exp(-a * d))

    # Linear decay
    maybe_update(np.linspace(1.0, 0.0, n))

    # Power law
    for p in np.linspace(0.1, 3.0, num=15):
        maybe_update((idx+1.0) ** (-p))

    # Sine half-period (non-negative)
    maybe_update(np.sin(np.pi * (idx+0.5) / n))

    # Raised cosine
    maybe_update(0.5 * (1.0 + np.cos(np.pi * idx / (n-1))))

    # At this point best_seq may be good.
    current = best_seq.copy()
    cur_score = best_score

    # helper: compute contribution products for shift k
    # we will use inside loops.

    no_improve = 0
    iter_cnt = 0
    max_no_improve_restart = 2500
    # set parameters for targeted move
    while time.monotonic() < deadline:
        # compute convolution once per loop
        conv = np.convolve(current, current)
        max_idx = int(np.argmax(conv))
        # compute pair contributions for that shift
        i_start = max(0, max_idx - (n-1))
        i_end = min(max_idx, n-1)
        i_vals = np.arange(i_start, i_end+1)
        j_vals = max_idx - i_vals
        # contributions using elementwise product
        contrib = current[i_vals] * current[j_vals]
        # find index with max contribution
        pair_offset = int(np.argmax(contrib))
        i_max = i_vals[pair_offset]
        j_max = j_vals[pair_offset]

        # choose donor as larger of i_max or j_max
        donor = i_max if current[i_max] >= current[j_max] else j_max
        donor_val = current[donor]
        if donor_val <= 0.0:
            no_improve += 1
            if no_improve > max_no_improve_restart:
                current = best_seq.copy()
                cur_score = best_score
                no_improve = 0
            continue

        # sample a few candidate receivers
        n_candidates = 5
        candidate_receivers = rng.choice(n, size=n_candidates, replace=False)
        # ensure donor not in receivers
        candidate_receivers = [r for r in candidate_receivers if r != donor]
        # also add far index (e.g., n-1-donor) if distinct
        far = n-1-donor
        if far != donor and far not in candidate_receivers:
            candidate_receivers.append(far)

        # try also the index with smallest value (to spread)
        min_idx = int(np.argmin(current))
        if min_idx != donor and min_idx not in candidate_receivers:
            candidate_receivers.append(min_idx)

        best_local_seq = None
        best_local_score = cur_score

        for recv in candidate_receivers:
            # transfer fraction
            frac = float(rng.uniform(0.05, 0.18))
            delta = donor_val * frac
            # ensure receiver caps
            delta = min(delta, 1000.0 - current[recv])
            if delta <= 0.0:
                continue
            cand = current.copy()
            cand[donor] -= delta
            cand[recv] += delta
            # clip just in case
            # np.clip might allocate new array, but okay
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = score(cand)
            if cand_score < best_local_score:
                best_local_seq = cand
                best_local_score = cand_score

        if best_local_seq is not None and best_local_score < cur_score:
            current = best_local_seq
            cur_score = best_local_score
            no_improve = 0
            if cur_score < best_score:
                best_seq = current.copy()
                best_score = cur_score
        else:
            no_improve += 1
            if no_improve > max_no_improve_restart:
                current = best_seq.copy()
                cur_score = best_score
                no_improve = 0

        iter_cnt += 1
        # occasional wash: smooth or symmetric averaging
        if iter_cnt % 400 == 0 and time.monotonic() < deadline:
            # symmetric average with probability
            if rng.random() < 0.4:
                sym = 0.5 * (current + current[::-1])
                sym = np.clip(sym, 0.0, 1000.0)
                sym_score = score(sym)
                if sym_score < cur_score:
                    current = sym
                    cur_score = sym_score
                    if cur_score < best_score:
                        best_seq = sym.copy()
                        best_score = cur_score
            # random stretch: multiply a slice by random log-normal factor
            w = int(rng.integers(max(1, n // 250), max(2, n // 15)))
            s = int(rng.integers(0, max(1, n - w + 1)))
            pert = current.copy()
            factor = np.exp(rng.normal(0.0, 0.18, size=w))
            pert[s:s+w] *= factor
            pert = np.clip(pert, 0.0, 1000.0)
            pert_score = score(pert)
            if pert_score < cur_score:
                current = pert
                cur_score = pert_score
                if cur_score < best_score:
                    best_seq = pert.copy()
                    best_score = cur_score

    # final fallback: ensure list of floats
    return [float(x) for x in best_seq]

def generate_sidon_seq(n, rng, epsilon=0.02):
    used = np.zeros(2*n-1, dtype=np.bool_)
    sel = []
    order = rng.permutation(n)
    for i in order:
        # check self sum
        if used[2*i]:
            continue
        conflict = False
        for j in sel:
            if used[i + j]:
                conflict = True
                break
        if conflict:
            continue
        # add
        for j in sel:
            used[i + j] = True
        used[2*i] = True
        sel.append(i)
    # construct sequence
    seq = np.full(n, epsilon, dtype=np.float64)
    for idx in sel:
        seq[idx] = 1.0
    return seq, len(sel)

import numpy as np
import time

def _ac1_score(seq: np.ndarray, n: int) -> float:
    """Return the AC1 upper bound for a non‑negative sequence."""
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv) / (np.sum(seq) ** 2))


def _sidon_candidate(n: int, rng: np.random.Generator, eps: float) -> np.ndarray:
    """
    Build a Sidon‑like 0/1 pattern and add a small constant background `eps`.
    The set is constructed greedily, guaranteeing that every unordered sum i+j
    occurs at most once.  The returned array has values 1 on the Sidon positions
    and `eps` elsewhere.
    """
    used = np.zeros(2 * n - 1, dtype=bool)          # which sums are already used
    selected = []                                   # Sidon indices
    order = rng.permutation(n)                     # random order of attempts
    for i in order:
        if used[2 * i]:
            continue
        bad = False
        for j in selected:
            if used[i + j]:
                bad = True
                break
        if bad:
            continue
        # accept i
        for j in selected:
            used[i + j] = True
        used[2 * i] = True
        selected.append(i)

    seq = np.full(n, eps, dtype=np.float64)
    if selected:
        seq[selected] = 1.0
    return seq
