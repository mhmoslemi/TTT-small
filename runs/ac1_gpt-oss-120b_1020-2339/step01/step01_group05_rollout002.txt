import time
import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    # get parent length if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
        if n <= 0:
            raise Exception
    except Exception:
        n = 256
    # generate linear decreasing sequence with sum 1
    # a_i = 2*(n-i) / (n*(n+1))
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, ..., 1
    a = 2.0 * idx / (n * (n + 1))
    # ensure nonneg and <=1000
    np.clip(a, 0.0, 1000.0, out=a)

    # optional improvement steps (simple gradient)
    # compute initial score
    best = a
    best_score = float(evaluate_sequence(best.tolist()))
    # time tracking
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    rng = np.random.default_rng(seed)
    step_frac = 0.05  # start with 5% move
    while time.monotonic() < deadline:
        # compute convolution and max index
        conv = np.convolve(best, best)
        k = int(np.argmax(conv))
        n_len = best.size
        lo = max(0, k - (n_len - 1))
        hi = min(n_len - 1, k)
        # partner values for indices lo..hi
        partners = best[k - np.arange(lo, hi + 1)]
        # identify index with max partner and min partner
        rel_i_max = int(np.argmax(partners))
        rel_i_min = int(np.argmin(partners))
        i_max = lo + rel_i_max
        i_min = lo + rel_i_min
        if i_max == i_min:
            # random move as fallback
            i_max = rng.integers(0, n_len)
            i_min = rng.integers(0, n_len)
            if i_max == i_min:
                continue
        # step size relative to source value
        step = step_frac * best[i_max]
        if step <= 0:
            continue
        cand = best.copy()
        cand[i_max] -= step
        cand[i_min] += step
        # ensure nonnegative, also maintain sum (should)
        if cand[i_max] < 0:
            continue
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
        else:
            # occasionally reduce step fraction or random perturb
            step_frac *= 0.99  # decay
            if step_frac < 1e-4:
                step_frac = 0.05
    return [float(v) for v in best]

def compute_score(arr):
    total = np.sum(arr)
    if total < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    maxb = np.max(conv)
    n = arr.shape[0]
    return float(2.0 * n * maxb / (total * total))

import time
import numpy as np

def _linear_candidate(length: int) -> np.ndarray:
    """
    Returns the decreasing linear sequence a_i = 2*(length-i)/(length*(length+1)).
    Its sum equals 1 and all entries lie in [0,1].
    """
    # values n, n-1, ..., 1
    vals = np.arange(length, 0, -1, dtype=np.float64)
    a = 2.0 * vals / (length * (length + 1))
    return a

def _score(arr: np.ndarray) -> float:
    """Fast computation of the AC1 upper‑bound score."""
    total = np.sum(arr)
    if total < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    return float(2.0 * arr.size * np.max(conv) / (total * total))

def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs) -> list:
    """
    Produce a non‑negative step‑function (list of floats) with a low AC1 upper bound.
    The core construction is a linearly decreasing sequence; a short
    gradient‑descent refinement is performed if time allows.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length: use provided parent if possible, else a modest default
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Base candidate: decreasing linear profile (sum == 1)
    # ------------------------------------------------------------------
    best = _linear_candidate(n)
    best_score = _score(best)

    # ------------------------------------------------------------------
    # If the supplied parent is better, start from it instead
    # ------------------------------------------------------------------
    try:
        if parent.size == n:
            parent = np.clip(parent, 0.0, 1000.0)
            sc = _score(parent)
            if sc < best_score:
                best = parent.copy()
                best_score = sc
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Simple greedy refinement: move a small amount from an index that
    # contributes most to the current maximal convolution entry to one that
    # contributes least.  This respects the sum constraint and keeps the
    # vector non‑negative.  The loop runs only while time permits.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    step_frac = 0.05          # initial fraction of the source entry to move
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        k = int(np.argmax(conv))                # index of the current maximum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)

        # partner values a_{k-i} for all i that actually appear in the sum
        idx_range = np.arange(lo, hi + 1)
        partners = best[k - idx_range]

        # indices i with largest and smallest partner values
        rel_i_max = int(np.argmax(partners))
        rel_i_min = int(np.argmin(partners))
        i_max = lo + rel_i_max
        i_min = lo + rel_i_min

        if i_max == i_min:
            # fallback: pick two distinct random positions
            i_max = rng.integers(0, n)
            i_min = rng.integers(0, n - 1)
            if i_min >= i_max:
                i_min += 1
            if i_max == i_min:
                continue

        # size of the transfer; never remove more than is present
        move = step_frac * best[i_max]
        if move <= 0.0:
            step_frac *= 0.9
            continue

        cand = best.copy()
        cand[i_max] -= move
        cand[i_min] += move

        # maintain feasibility (non‑negative, within allowed range)
        if cand[i_max] < 0.0:
            continue
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            # keep step size for successful moves
        else:
            # gradually shrink the step fraction if progress stalls
            step_frac *= 0.99
            if step_frac < 1e-4:
                step_frac = 0.05

    return [float(v) for v in best]
