import numpy as np, time

def _score(seq):
    return float(evaluate_sequence(seq))

def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # Initialize with parent or uniform
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception
        seq = parent.copy()
    except Exception:
        n = 256
        seq = np.ones(n, dtype=np.float64)
    n = seq.shape[0]

    # evaluate initial
    best = seq.copy()
    best_score = _score(best.tolist())
    # try power law candidates (telescoping sum)
    # generate array of indices once
    idx = np.arange(n, dtype=np.float64)
    # exponent sweep
    for p in np.linspace(0.2, 0.9, 15):
        cand = (idx + 1.0) ** p - idx ** p
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc

    # main search loop
    while time.monotonic() < deadline:
        # decide move type
        if rng.random() < 0.7:
            # targeted transfer move
            # compute convolution and locate max
            b = np.convolve(best, best)
            k = int(np.argmax(b))
            # compute partner array
            i = np.arange(n)
            j = k - i
            mask = (j >= 0) & (j < n)
            partner = np.empty_like(best)
            partner.fill(0.0)
            partner[mask] = best[j[mask]]
            prod = best * partner
            total = prod.sum()
            if total == 0.0:
                # fallback to random block perturbation
                width = int(rng.integers(1, max(2, n // 20)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                factor = float(np.exp(rng.normal(0.0, 0.08)))
                cand = best.copy()
                cand[start:start+width] *= factor
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand.tolist())
                if sc < best_score:
                    best = cand
                    best_score = sc
                continue
            # sample a high‑contribution index
            probs = prod / total
            i_high = rng.choice(n, p=probs)
            # choose an index with small partner (prefer zero)
            low_mask = partner == partner.min()
            low_candidates = np.where(low_mask)[0]
            if low_candidates.size == 0:
                i_low = rng.integers(0, n)
            else:
                i_low = rng.choice(low_candidates)
            delta = best[i_high] * 0.015  # 1.5% transfer
            if delta <= 0:
                continue
            cand = best.copy()
            cand[i_high] -= delta
            cand[i_low] += delta
            # enforce bounds
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand.tolist())
            if sc < best_score:
                best = cand
                best_score = sc

def _score(seq):
    """Wrapper around the provided evaluator."""
    # seq is a Python list of floats (converted from a NumPy array)

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Return a sequence of non‑negative heights (list of floats) with a small
    AC1 upper‑bound score. The method starts from any available parent
    (height_sequence_1) or a simple uniform vector, tries a few power‑law
    families, then performs a fast hill‑climb that moves mass away from the
    indices that dominate the maximal convolution value.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the harness timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ----------------------------------------------------------------------
    # 1. initialise – use the supplied parent if it exists, else uniform
    # ----------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0,
                         1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        seq = parent.copy()
    except Exception:
        # fallback – modest length uniform vector (sum will be ≥0.01)
        n = 256
        seq = np.ones(n, dtype=np.float64)
    n = seq.shape[0]

    # ----------------------------------------------------------------------
    # 2. evaluate initial candidate and keep it as the current best
    # ----------------------------------------------------------------------
    best = seq.copy()
    best_score = _score(best.tolist())

    # ----------------------------------------------------------------------
    # 3. quick sweep of the telescoping power‑law family
    #    a_i = (i+1)^p - i^p  with p in [0.2, 0.9]
    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(0.20, 0.90, 15, dtype=np.float64):
        cand = (idx + 1.0) ** p - idx ** p
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand.tolist())
        if sc < best_score:
            best = cand
            best_score = sc

    # ----------------------------------------------------------------------
    # 4. main optimisation loop – until time runs out
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 4a. targeted mass‑transfer move (≈ gradient descent on the
        #     maximal convolution entry)
        # --------------------------------------------------------------
        if rng.random() < 0.7:
            # convolution and location of the current worst lag
            conv = np.convolve(best, best)
            k = int(np.argmax(conv))

            # partner[i] = best[k‑i] if that index exists, else 0
            i = np.arange(n)
            j = k - i
            mask = (j >= 0) & (j < n)
            partner = np.zeros_like(best)
            partner[mask] = best[j[mask]]

            # contribution of each entry to the peak value
            prod = best * partner
            tot = prod.sum()
            if tot == 0.0:
                # fall back to a cheap random perturbation
                width = int(rng.integers(max(1, n // 250), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                factor = float(np.exp(rng.normal(0.0, 0.09)))
                cand = best.copy()
                cand[start:start + width] *= factor
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand.tolist())
                if sc < best_score:
                    best, best_score = cand, sc
                continue

            # high‑impact index (probability ∝ contribution)
            probs = prod / tot
            i_high = rng.choice(n, p=probs)

            # pick a low‑impact index (prefer zero partner)
            low_mask = partner == partner.min()
            low_choices = np.where(low_mask)[0]
            i_low = (
                rng.choice(low_choices)
                if low_choices.size
                else rng.integers(0, n)
            )

            # move a small fraction of mass from i_high to i_low
            delta = best[i_high] * 0.015  # 1.5 % of the high value
            if delta > 0.0:
                cand = best.copy()
                cand[i_high] -= delta
                cand[i_low] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)

    # 5. return the best sequence found (as plain Python floats)
    # ----------------------------------------------------------------------
