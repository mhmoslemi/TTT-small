def propose_candidate(seed=42, budget_s=110, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-1.0)  # keep margin
    try:
        n = len(height_sequence_1)
        a0 = np.maximum(height_sequence_1.astype(np.float64), 0.0)
    except Exception:
        n = 256
        a0 = np.ones(n, dtype=np.float64)
    # normalise sum
    sum0 = a0.sum()
    if sum0 < 1e-12:
        a0 = np.full(n, 1.0/n)
        sum0 = 1.0
    a0 = a0 / sum0

    # function for evaluating objective given a (normed)
    size = 1 << ((2*n - 2).bit_length())
    def compute_conv_max(a):
        A = np.fft.rfft(a, size)
        conv = np.fft.irfft(A * A, size)[:2*n-1]
        k = int(conv.argmax())
        return conv[k], k, conv

    # initial candidates
    best_a = a0.copy()
    best_val, _, _ = compute_conv_max(best_a)
    best_score = 2*n*best_val

    # geometric candidate
    c_opt = 1.256  # approximate solution from previous code
    r = math.exp(-c_opt / n)
    a_geo = (1.0 - r) * (r ** np.arange(n, dtype=np.float64))
    a_geo = a_geo / a_geo.sum()
    val_geo, _, _ = compute_conv_max(a_geo)
    if 2*n*val_geo < best_score:
        best_score = 2*n*val_geo
        best_a = a_geo.copy()

    # power-law decreasing candidate
    for p in (-0.5, -1.0, -1.5, -2.0):
        a_pow = (np.arange(1, n+1, dtype=np.float64) ** p)
        a_pow = a_pow / a_pow.sum()
        val_pow, _, _ = compute_conv_max(a_pow)
        if 2*n*val_pow < best_score:
            best_score = 2*n*val_pow
            best_a = a_pow.copy()

    # random Dirichlet candidates
    for _ in range(5):
        # sample Dirichlet with concentration 0.5 for sparse shape
        a_rand = rng.gamma(shape=0.5, scale=1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        val_rand, _, _ = compute_conv_max(a_rand)
        if 2*n*val_rand < best_score:
            best_score = 2*n*val_rand
            best_a = a_rand.copy()
    # Now perform gradient descent from best_a
    a = best_a.copy()
    size = 1 << ((2*n - 2).bit_length())
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr = 0.1
    it = 0
    # Keep best overall
    overall_best_a = a.copy()
    overall_best_score = best_score

    while time.monotonic() < deadline:
        it += 1
        Afft = np.fft.rfft(a, size)
        conv = np.fft.irfft(Afft * Afft, size)[:2*n-1]
        max_val = float(conv.max())
        k_max = int(conv.argmax())
        # compute subgradient for max entry
        # gradient vector length n
        # use _gradient implementation
        # we can inline for speed
        n_local = n
        g = np.zeros_like(a)
        if k_max < n_local:
            # slice assign
            rev = a[k_max::-1]
            g[:k_max+1] = 2.0 * rev
        else:
            start = k_max - (n_local - 1)
            length = 2 * n_local - 1 - k_max
            rev = a[:length][::-1]
            g[start:] = 2.0 * rev
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection: non-negativity
        np.maximum(a_new, 0.0, out=a_new)
        sum_new = a_new.sum()
        if sum_new == 0.0:
            # reinitialize some random direction to avoid degenerate. skip update.
            # optionally perturb small epsilon to each coordinate.
            a_new = a  # keep old
            sum_new = a_new.sum()
        else:
            a_new /= sum_new
        # compute new objective
        Afft_new = np.fft.rfft(a_new, size)
        conv_new = np.fft.irfft(Afft_new * Afft_new, size)[:2*n-1]
        max_val_new = float(conv_new.max())
        score_new = 2 * n_local * max_val_new
        # keep best ever
        if score_new < overall_best_score:
            overall_best_score = score_new
            overall_best_a = a_new.copy()
            # if we already beat target, break early
            if overall_best_score < 1.50274349:
                break
        # accept new point if improvement; else maybe do not accept but still keep momentum.
        if score_new <= 2 * n_local * max_val:  # compare to previous objective
            a = a_new
        # else keep old a but continue with momentum (Keeps m, v unchanged)
        # optional: also periodically reset momentum if stalled

    # final
    final_a = overall_best_a
    # Clip and return as list
    np.clip(final_a, 0.0, 1000.0, out=final_a)
    return final_a.tolist()

import time
import math
import numpy as np

def _conv_max(a: np.ndarray, size: int):
    """Return (max convolution value, index of max, full convolution) for a."""
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * a.size - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    if a0.sum() < 1e-12:

    a0 = a0 / a0.sum()                     # enforce sum = 1
    size = 1 << ((2 * n - 2).bit_length()) # FFT length (next power of two)
