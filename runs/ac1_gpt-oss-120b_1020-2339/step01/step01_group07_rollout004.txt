import numpy as np, time, math

def propose_candidate(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # init a
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            raise ValueError
    except Exception:
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)
    n = int(a.shape[0])
    # apply clipping and minimal sum check
    np.clip(a, 0.0, 1000.0, out=a)
    # ensure sum not too low
    if a.sum() < 0.01:
        a[:] = 1.0
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also compare with uniform as baseline (maybe better)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uscore = float(evaluate_sequence(uniform.tolist()))
    if uscore < best_score:
        best = uniform.copy()
        best_score = uscore
    # main loop
    while time.monotonic() < deadline:
        if rng.random() < 0.7:
            # targeted reduction move
            # compute convolution
            b = np.convolve(a, a)
            k_max = int(np.argmax(b))
            # indices range where i + j = k_max, i and j in [0, n-1]
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            # compute contributions a[i] * a[k_max - i]
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]
            # choose index with largest contribution
            idx = int(np.argmax(contrib))
            i0 = i_range[idx]
            j0 = j_range[idx]
            # decide which to shrink (the larger one)
            if a[i0] >= a[j0]:
                src = i0
                dst_candidate = j0
            else:
                src = j0
                dst_candidate = i0
            # shrink src by factor <1
            # sample factor: exp(-abs(N(0, sigma))) ensures <1
            sigma = 0.07
            factor = math.exp(-abs(rng.normal(0.0, sigma)))
            # ensure not too small to avoid zero

            old_val = a[src]
            new_val = old_val * factor
            delta = old_val - new_val
            # pick destination index to receive delta (random not src)
            dst = int(rng.integers(0, n))
            while dst == src:
                dst = int(rng.integers(0, n))
            cand = a.copy()
            cand[src] = new_val
