    # init seq

        parent = np.asarray(height_sequence_1, dtype=np.float64)

        parent = np.where(np.isfinite(parent), parent, 0.0)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        # fallback size
        n = 256
        parent = np.ones(n, dtype=np.float64)

    n = parent.shape[0]
    # normalize sum to 1
    seq = parent.copy()
    # Ensure non-negative
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total < 1e-12:
        seq = np.ones(n) / n
    else:
        seq = seq / total

    # function to compute score
    fft_len = 1 << ((2*n-1).bit_length())
    def compute_score(s):
        # s is 1D array sum to 1 (maybe not exactly)
        # compute conv via FFT
        F = np.fft.rfft(s, fft_len)
        conv = np.fft.irfft(F * F, fft_len)[:2*n-1]
        maxb = conv.max()
        sum_a = s.sum()
        if sum_a < 0.01:
            return float('inf')
        return float(2.0 * n * maxb / (sum_a*sum_a))

    best_seq = seq.copy()
    best_score = compute_score(best_seq)

    # initial parametric families to improve baseline
    idx = np.arange(n, dtype=np.float64)
    # exponential families
    for a in np.logspace(-5, -2, 15):
        cand = np.exp(-a * idx)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # symmetric exponential decays
    d = np.minimum(idx, n-1-idx)
    for a in np.logspace(-5, -2, 15):
        cand = np.exp(-a * d)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:

    # linear decay

    cand = cand / cand.sum()
    sc= compute_score(cand)
    if sc<best_score:
        best_seq,best_score=cand,sc

        cand = (idx+1.0)**(-p)
        cand = cand / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_seq,best_score = cand,sc

import numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Return a non‑negative sequence (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – try to use the provided parent, otherwise fall back
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, 1000.0)
        base = np.where(np.isfinite(base), base, 0.0)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
    except Exception:                     # no parent available
        base = np.ones(256, dtype=np.float64)

    n = base.shape[0]

    # work with a probability vector (sum = 1); scaling does not affect the
    # objective because the evaluator is homogeneous.
    base = np.maximum(base, 0.0)
    if base.sum() < 1e-12:
        base = np.ones(n, dtype=np.float64) / n
    else:
        base = base / base.sum()

    # ------------------------------------------------------------------
    # 2️⃣  Evaluation helper (FFT‑based, O(n log n))
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2n‑1

    def _score(v: np.ndarray) -> float:
        """Exact score used by the official evaluator (but faster)."""
        F = np.fft.rfft(v, fft_len)
        conv = np.fft.irfft(F * F, fft_len)[: 2 * n - 1]
        maxb = float(conv.max())
        s = float(v.sum())
        if s < 0.01:                 # never happens for our normalised vectors
            return float('inf')
        return 2.0 * n * maxb / (s * s)

    # ------------------------------------------------------------------
    # 3️⃣  Scan a few analytic families – they sometimes give a very good start
    # ------------------------------------------------------------------
    best_seq = base.copy()
    best_score = _score(best_seq)

    idx = np.arange(n, dtype=np.float64)

    # exponential decay
    for a in np.logspace(-5, -2, 15):
        cand = np.exp(-a * idx)
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # symmetric exponential (high at both ends)
    d = np.minimum(idx, n - 1 - idx)
    for a in np.logspace(-5, -2, 15):
        cand = np.exp(-a * d)
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # linear decay left→right
    cand = np.linspace(1.0, 0.0, n)
    cand /= cand.sum()
    sc = _score(cand)
    if sc < best_score:
        best_seq, best_score = cand, sc

    # power‑law
    for p in np.linspace(0.1, 2.0, 20):

        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # arcsine‑type (large weight at the ends)
    eps = 1e-12
    cand = 1.0 / np.sqrt(
        np.clip(idx + 0.5, eps, None) * np.clip(n - 0.5 - idx, eps, None)

    cand /= cand.sum()
    sc = _score(cand)
    if sc < best_score:

    # half‑wave sine (always non‑negative)
    cand = np.sin(np.pi * (idx + 0.5) / n)
    cand = np.clip(cand, 0.0, None)
    cand /= cand.sum()

    if sc < best_score:

    if sc < best_score:

    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score

    while time.monotonic() < deadline:

        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = cur[i_range] * cur[j_range]

        # pick one of the two indices of the strongest pair
        pair = int(np.argmax(contrib))
        i_max = i_range[pair]

        donor = i_max if cur[i_max] >= cur[j_max] else j_max

        if cur[donor] < 1e-12:

        # pick a receiver distinct from donor (prefer a low‑mass index)
        recv = int(rng.integers(0, n))
        while recv == donor:
            recv = int(rng.integers(0, n))

        # move a modest fraction of the donor's mass
        frac = float(rng.uniform(0.02, 0.15))

        cand[recv] += delta

        cand_score = _score(cand)

        # accept if better, otherwise occasional uphill moves (simulated annealing)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:

            if T > 0 and rng.random() < math.exp(-(cand_score - cur_score) / T):

                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 2500:
                    cur, cur_score = best_seq.copy(), best_score
                    no_improve = 0

        it += 1

        # every few hundred iterations apply a stronger perturbation
        if it % 800 == 0:

            w = int(rng.integers(max(2, n // 250), max(4, n // 60)))
            s = int(rng.integers(0, n - w + 1))

            pert[s : s + w] *= factor
            pert = np.clip(pert, 0.0, None)
            pert /= pert.sum()
            p_score = _score(pert)
            if p_score < best_score:
                best_seq, best_score = pert, p_score
                cur, cur_score = pert, p_score

            # occasional symmetrisation (helps because convolution is symmetric)
            if rng.random() < 0.18:

                sym /= sym.sum()

                    best_seq, best_score = sym, s_score
                    cur, cur_score = sym, s_score

        if time.monotonic() + 0.05 > deadline:

    # ------------------------------------------------------------------
    # 5️⃣  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # safety
    return [float(v) for v in best_seq]
