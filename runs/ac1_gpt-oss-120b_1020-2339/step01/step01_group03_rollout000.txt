    g = np.zeros_like(a)
    if k < n:

        # That length equals n - start.

    return g

def fast_ac1_score(a, size, n):
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return np.inf
    return 2.0 * n * max_val / (s * s)

def gradient_at_k(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        g[:k+1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        length = 2 * n - 1 - k
        g[start:] = 2.0 * a[:length][::-1]
    return g

def _fast_ac1_score(a: np.ndarray, size: int, n: int) -> float:
    """Fast evaluation of the AC1 bound (exact up to floating error)."""
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    return float(2.0 * n * max_val / (s * s)) if s > 0 else np.inf

        if s_new == 0.0:                     # degenerate – keep the old point
            a_new = a.copy()
            s_new = a_new.sum()
        else:
            a_new /= s_new

        # evaluate the new candidate (fast FFT)
        Anew = np.fft.rfft(a_new, size)
        conv_new = np.fft.irfft(Anew * Anew, size)[: 2 * n - 1]
        max_val_new = conv_new.max()
        score_new = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        # keep the best ever seen
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            last_improve = it
            if best_score < 1.50274349:      # target reached – stop early
                break

        # greedy acceptance for the current iterate
        if score_new <= score_curr:
            a = a_new
            score_curr = score_new
        else:
            # occasional random perturbation if we have stalled for a while
            if it - last_improve > 2000:
                # small Gaussian noise, stay non‑negative, renorm
                noise = rng.normal(scale=0.01, size=n)
                a = a + noise
                np.maximum(a, 0.0, out=a)
                s_a = a.sum()
                if s_a == 0.0:
                    a = np.ones(n, dtype=np.float64) / n
                else:
                    a /= s_a
                # recompute current score after perturbation
                A = np.fft.rfft(a, size)
                conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
                score_curr = 2.0 * n * conv.max() / (a.sum() ** 2)
                last_improve = it

    # ---- 3. Final safeguard ----------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(n, dtype=np.float64) / n

    return best_seq.tolist()
