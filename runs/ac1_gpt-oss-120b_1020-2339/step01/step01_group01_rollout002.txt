            init = np.clip(init, 0.0, 1000.0)
        else:
            init = None
    else:
        init = None

    if init is None:
        n = 256
        best = np.full(n, 1.0, dtype=np.float64)
    else:
        best = init.copy()
        n = best.size

    # Normalise to unit L1 mass – the bound is scale‑invariant.
    total = best.sum()
    if total == 0.0:
        best[:] = 1.0
        total = best.sum()
    best = best / total

    L = 1 << ((2 * n - 1).bit_length())
    best_score = compute_score(best, L)

    # ------------------------------------------------------------------
    # 2. Systematic search over power‑law shapes.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    p_vals = np.linspace(0.2, 1.5, 66)                     # many exponents
    offsets = np.array([0.0, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0, 10.0],
                       dtype=np.float64)

    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        for off in offsets:
            a = (idx + 1.0 + off) ** (-p)                 # avoid division by zero
            a = np.clip(a, 0.0, 1000.0)
            a = a / a.sum()
            sc = compute_score(a, L)
            if sc < best_score:
                best, best_score = a, sc

            rev = a[::-1]
            sc = compute_score(rev, L)
            if sc < best_score:
                best, best_score = rev, sc

            sym = 0.5 * (a + rev)
            sc = compute_score(sym, L)
            if sc < best_score:
                best, best_score = sym, sc

    # ------------------------------------------------------------------
    # 3. Gradient‑descent style refinement on the currently active max.
    # ------------------------------------------------------------------
    step = 0.1
    while time.monotonic() < deadline:
        conv = full_convolution(best, L)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # gradient of the bound (up to a positive constant factor)
        grad = rev_contrib(best, k_max) - max_val * best

        cand = best - step * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            step *= 0.5
            continue
        cand = cand / s

        cand_score = compute_score(cand, L)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            step = min(step * 1.05, 1.0)
            continue

        # No improvement – shrink step, occasionally shake the solution.
        step *= 0.5
        if step < 1e-6:
            pert = best * (1.0 + 0.05 * rng.standard_normal(n))
            pert = np.clip(pert, 0.0, None)
            pert = pert / pert.sum()
            pert_score = compute_score(pert, L)
            if pert_score < best_score:
                best, best_score = pert, pert_score
            step = 0.1

    # ------------------------------------------------------------------
    # 4. Final safety adjustment.
    # ------------------------------------------------------------------
    if best.sum() < 0.01:
        best = np.clip(best, 0.0, 1000.0)
        if best.sum() < 0.01:
            best[0] = 0.01

    return [float(v) for v in best]
