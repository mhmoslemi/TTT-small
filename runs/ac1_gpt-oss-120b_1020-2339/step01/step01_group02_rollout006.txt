def simplex_projection(v):
    """Project onto the simplex {x>=0, sum=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _score_vec(seq: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative sequence (no clipping)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – put all mass on the largest entry
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        seq = (idx + 1.0) ** (-p)
        seq = seq / seq.sum()
        _maybe_update(seq)

    # half‑period sine (non‑negative)
    seq = np.sin(np.pi * (idx + 0.5) / n)
    seq = np.maximum(seq, 0.0)
    seq = seq / seq.sum()
    _maybe_update(seq)

    # raised cosine (non‑negative)
    seq = (1.0 + np.cos(np.pi * idx / (n - 1))) * 0.5
    seq = seq / seq.sum()
    _maybe_update(seq)

    # ------------------------------------------------------------------
    # start a focused local search from the current best
    # ------------------------------------------------------------------
    p = best_seq.copy()
    cur_score = best_score

    iter_cnt = 0
    no_improve = 0
    base_frac = 0.15  # typical transfer fraction

    while time.monotonic() < deadline:
        # convolution and the most offending shift
        conv = np.convolve(p, p)
        k = int(conv.argmax())

        # indices that actually appear in this shift
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx

        # contributions of each ordered pair to conv[k]
        contrib = p[i_idx] * p[j_idx]
        pair_pos = int(contrib.argmax())
        i_max = i_idx[pair_pos]
        j_max = j_idx[pair_pos]

        donor = i_max if p[i_max] >= p[j_max] else j_max

        # choose a receiver that does *not* contribute to this shift
        # (i.e. outside the active index interval)
        if i_start > 0:
            low = np.arange(0, i_start, dtype=int)
        else:
            low = np.empty(0, dtype=int)
        if i_end < n - 1:
            high = np.arange(i_end + 1, n, dtype=int)
        else:
            high = np.empty(0, dtype=int)

        if low.size + high.size > 0:
            # blend the two blocks without constructing a huge array
            if low.size == 0:
                recv = int(rng.choice(high))
            elif high.size == 0:
                recv = int(rng.choice(low))
            else:
                if rng.random() < low.size / (low.size + high.size):
                    recv = int(rng.choice(low))
                else:
                    recv = int(rng.choice(high))
        else:
            # fallback – any index different from donor
            recv = int(rng.integers(0, n))
            while recv == donor:
                recv = int(rng.integers(0, n))

        # amount to move (a modest fraction of the donor)
        if p[donor] <= 0.0:
            no_improve += 1
            if no_improve > 2000:
                p = best_seq.copy()
                cur_score = best_score
                no_improve = 0
            continue

        frac = base_frac * (0.5 + rng.random())          # 0.075 … 0.225
        delta = min(p[donor] * frac, 1.0 - p[recv])      # keep total sum = 1
        if delta <= 0.0:
            continue

        cand = p.copy()
        cand[donor] -= delta
        cand[recv] += delta

        cand_score = _score_vec(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()

        if cand_score < cur_score:
            p = cand
            cur_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2500:
                p = best_seq.copy()
                cur_score = best_score
                no_improve = 0

        iter_cnt += 1

        # occasional broader perturbation to shake out plateaus
        if iter_cnt % 500 == 0 and time.monotonic() < deadline:
            w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            s = int(rng.integers(0, max(1, n - w + 1)))
            e = s + w
            pert = best_seq.copy()
            pert[s:e] *= np.exp(rng.normal(0.0, 0.12, size=w))
            if rng.random() < 0.35:
                pert = 0.5 * (pert + pert[::-1])
            pert = np.clip(pert, 0.0, None)
            pert = pert / pert.sum()
            pert_score = _score_vec(pert)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.copy()
            if pert_score < cur_score:
                p = pert
                cur_score = pert_score

    # ------------------------------------------------------------------
    # Return the best sequence seen (values already lie in [0,1000])
    # ------------------------------------------------------------------
    out = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in out.tolist()]
