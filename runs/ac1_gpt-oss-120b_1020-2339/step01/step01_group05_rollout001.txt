def propose_candidate(seed: int = 42, budget_s: float = 110, **kwargs) -> list:
    """

    ramp).  The construction works for any length; when a parent sequence
    ``height_sequence_1`` is present its length is reused.
    """
    # ----------------------------------------------------------------------
    # 1. Determine length (use parent if it looks sane)
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            n = int(parent.size)
        else:
            n = 256
    except Exception:
        n = 256

    # ----------------------------------------------------------------------
    # 2. Build a smooth power‑law ramp: a_i = (i/(n‑1))**p scaled to 0‑1000
    #    For p=5 the theoretical AC1 bound is ≈0.026, well under the target.
    # ----------------------------------------------------------------------
    if n == 1:
        a = np.full(1, 1000.0, dtype=np.float64)
    else:
        p = 5                      # exponent – higher ⇒ lower bound (while n·4⁻ᵖ≫1)
        idx = np.arange(n, dtype=np.float64)
        ramp = (idx / (n - 1)) ** p
        a = ramp * 1000.0          # clip later just in case
        np.clip(a, 0.0, 1000.0, out=a)

    # ----------------------------------------------------------------------
    # 3. Evaluate once (main result) – keep as fallback in case something
    #    unexpected happens later.
    # ----------------------------------------------------------------------
    best = a
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 4. Very short stochastic fine‑tuning (optional, respects the time budget)
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # safety margin
