    best_seq = None
    best_score = float('inf')

    if sum_a < 0.01:
        return float("inf")
    L = 2 * n - 1
    # linear convolution via FFT (real‑to‑real)
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    max_b = float(conv.max())
    return 2.0 * n * max_b / (sum_a * sum_a)


# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=110, **kwargs):
    """
    Generate a non‑negative step‑function that (hopefully) improves
    the AC1 bound.  The routine respects the time budget, explores a
    few analytic families of shapes, and then performs a light‑weight
    stochastic refinement.
    """
    # --------------------------------------------------------
    # Determine problem size (use the provided parent if possible)
    # --------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        ok_parent = (
            base.ndim == 1
            and base.size > 0
            and np.isfinite(base).all()
        )
    except Exception:
        ok_parent = False

    if ok_parent:
        n = int(base.shape[0])
    else:
        n = 256                     # fall‑back length

    # --------------------------------------------------------
    # Pre‑compute reusable index helpers
    # --------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    frac = (n - 1 - idx) / (n - 1)          # runs from 1 down to 0

    # --------------------------------------------------------
    # Bookkeeping of the best candidate seen so far
    # --------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    if ok_parent:
        parent = np.clip(base, 0.0, 1000.0)
        try:
            sc = compute_ac1_score(parent)
            best_seq = parent.copy()
            best_score = sc
        except Exception:
            pass

    # --------------------------------------------------------
    # Helper to evaluate a candidate and keep the best one
    # --------------------------------------------------------
    def maybe_update(seq):
        nonlocal best_seq, best_score
        cand = np.clip(seq, 0.0, 1000.0)
        sc = compute_ac1_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # --------------------------------------------------------
    # Time control
    # --------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 3.0)   # leave a small head‑room

    # --------------------------------------------------------
    # 1) Systematic exploration of simple analytic families
    # --------------------------------------------------------

    # power‑law decay a_i = 1000 * ((n‑1‑i)/(n‑1))**p
    for p in np.linspace(0.2, 12.0, 60):
        if time.monotonic() >= deadline:
            break
        maybe_update(1000.0 * np.power(frac, p))

    # exponential decay a_i = 1000 * exp(-λ i/(n‑1))
    for lam in np.linspace(0.0, 12.0, 60):
        if time.monotonic() >= deadline:
            break
        maybe_update(1000.0 * np.exp(-lam * idx / (n - 1)))

    # square‑root of a power‑law (extra curvature)
    for p in np.linspace(0.2, 3.0, 30):
        if time.monotonic() >= deadline:
            break
        maybe_update(1000.0 * np.sqrt(np.power(frac, p)))

    # convex combos of power‑law and exponential
    for w in (0.2, 0.5, 0.8):
        for p in (1.0, 2.0, 4.0, 6.0):
            for lam in (0.5, 1.0, 2.0, 4.0):
                if time.monotonic() >= deadline:
                    break
                combo = w * np.power(frac, p) + (1.0 - w) * np.exp(-lam * idx / (n - 1))
                maybe_update(1000.0 * combo)

    # --------------------------------------------------------
    # 2) Stochastic refinement (block scaling, global jitter,
    #    gentle sinusoidal modulation)
    # --------------------------------------------------------
    rng = np.random.default_rng(seed)

    # Use the best shape found so far as a moving base
    base_shape = best_seq.copy() if best_seq is not None else 1000.0 * np.power(frac, 2.0)

    while time.monotonic() < deadline:
        cand = base_shape.copy()

        # Global multiplicative jitter
        if rng.random() < 0.6:
            cand *= 1.0 + rng.normal(0.0, 0.02)

        # Small sinusoidal wiggle (keeps non‑negativity after clipping)
        if rng.random() < 0.4:
            amp = rng.uniform(0.0, 0.1)               # relative amplitude
            freq = rng.uniform(1.0, 5.0)              # few full cycles across the array
            phase = rng.uniform(0.0, 2.0 * np.pi)
            wiggle = 1.0 + amp * np.sin(2.0 * np.pi * freq * idx / (n - 1) + phase)
            cand *= wiggle

        # Random contiguous block scaling
        if rng.random() < 0.5:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            block_factor = 1.0 + rng.normal(0.0, 0.15)
            cand[start:start + width] *= block_factor

        cand = np.clip(cand, 0.0, 1000.0)

        sc = compute_ac1_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            base_shape = best_seq       # steer future mutations toward the new best

    # --------------------------------------------------------
    # Fallback (should never trigger)
    # --------------------------------------------------------
    if best_seq is None:
        best_seq = 1000.0 * np.power(frac, 2.0)

    # Return a plain Python list of floats
    return [float(v) for v in best_seq]
