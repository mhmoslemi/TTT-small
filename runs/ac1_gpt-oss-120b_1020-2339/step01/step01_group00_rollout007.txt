Pseudo-code for simplex projection:

```
def proj_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum=1}."""
    # using algorithm from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending sort
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    # find rho = last index where condition true
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def proj_simplex(v):
    # ensure nonnegative sum to 1
    # v shape (n,)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]

    w[w < 0] = 0

import numpy as np
import time

# --------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# --------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive: return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # renormalise (numerical safety)
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / n)
    w /= s
    return w
