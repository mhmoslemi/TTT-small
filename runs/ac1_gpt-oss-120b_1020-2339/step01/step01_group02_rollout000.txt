def compute_score(seq):
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

import time
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound for a non‑negative sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.shape[0]

    best_seq = parent.copy()
    best_score = _score(best_seq)

    # ---- deterministic candidates ------------------------------------
    # sorted descending values (largest → smallest)
    sorted_desc = np.sort(parent)[::-1]

    # 1) alternating extremes (largest, smallest, 2nd largest, 2nd smallest,…)
    seq_alt = np.empty_like(parent)
    left, right, pos = 0, n - 1, 0
    while left <= right:
        seq_alt[pos] = sorted_desc[left]
        left += 1
        pos += 1
        if pos < n:
            seq_alt[pos] = sorted_desc[right]
            right -= 1
            pos += 1
    sc = _score(seq_alt)
    if sc < best_score:
        best_seq, best_score = seq_alt.copy(), sc

    # 2) zig‑zag ends (fill from left, then right, etc.)
    seq_zig = np.empty_like(parent)
    l, r, idx = 0, n - 1, 0
    while l <= r:
        seq_zig[l] = sorted_desc[idx]
        idx += 1
        l += 1
        if l <= r:
            seq_zig[r] = sorted_desc[idx]
            idx += 1
            r -= 1
    sc = _score(seq_zig)
    if sc < best_score:
        best_seq, best_score = seq_zig.copy(), sc

    # 3) stride (coprime step) permutation – spreads large values uniformly
    step = 37  # 37 is coprime to most practical lengths (e.g. 4498)
    pos = (np.arange(n) * step) % n
    seq_str = np.empty_like(parent)
    seq_str[pos] = sorted_desc
    sc = _score(seq_str)
    if sc < best_score:
        best_seq, best_score = seq_str.copy(), sc

    # 4) a few random shuffles of the sorted list
    for _ in range(5):
        cand = sorted_desc.copy()
        rng.shuffle(cand)
        sc = _score(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc

    # ---- local improvement loop ------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        # convolution and worst shift
        conv = np.convolve(cur, cur)
        k = int(np.argmax(conv))

        # indices that form this shift
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = cur[i_range] * cur[j_range]
        if contrib.size == 0:
            break
        pair = int(np.argmax(contrib))
        i_max = i_range[pair]
        j_max = j_range[pair]

        # donor: the larger entry of the pair
        donor = i_max if cur[i_max] >= cur[j_max] else j_max
        if cur[donor] <= 0.0:
            # pick any positive entry
            pos = np.where(cur > 0)[0]
            if pos.size == 0:
                break
            donor = int(rng.choice(pos))

        # receiver: the current smallest entry (helps spread mass)
        recv = int(np.argmin(cur))
        if recv == donor:
            # second smallest if needed
            args = np.argsort(cur)
            recv = int(args[1] if args[1] != donor else args[2] % n)

        # fraction to move – decays over time
        elapsed = time.monotonic() - start
        frac = 0.15 * (1.0 - elapsed / (budget_s - 0.5)) + 0.005
        delta = min(cur[donor] * frac, 1000.0 - cur[recv])
        if delta <= 0.0:
            no_improve += 1
            if no_improve > 2000:
                cur, cur_score = best_seq.copy(), best_score
                no_improve = 0
            continue

        cand = cur.copy()
        cand[donor] -= delta
        cand[recv] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
        else:

            if rng.random() < 0.001 * (1.0 - elapsed / (budget_s - 0.5)):
                cur, cur_score = cand, cand_score
            no_improve += 1
            if no_improve > 2500:
                cur, cur_score = best_seq.copy(), best_score
                no_improve = 0

        if it % 100 == 0:

            j_swap = int(rng.integers(0, n))
            if i_swap != j_swap:
                swap_cand = cur.copy()
                swap_cand[i_swap], swap_cand[j_swap] = swap_cand[j_swap], swap_cand[i_swap]
                swap_score = _score(swap_cand)
                if swap_score < cur_score:
                    cur, cur_score = swap_cand, swap_score
                    if swap_score < best_score:
                        best_seq, best_score = swap_cand.copy(), swap_score

        # occasional broader perturbation
        if it % 250 == 0:

            if pert_score < best_score:
                best_seq, best_score = pert.copy(), pert_score
            if pert_score < cur_score:
                cur, cur_score = pert, pert_score

        it += 1

    return [float(v) for v in best_seq]
