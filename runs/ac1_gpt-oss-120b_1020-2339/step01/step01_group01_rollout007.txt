def propose_candidate(seed=42,budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 1.5)

    # get n
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr,0,1000)
        n = parent_arr.size
    else:
        n = 256
        parent_arr = np.full(n, 1.0)

    best = parent_arr.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # if parent not good (maybe >?), we still evaluate.

    # grid search: power law
    exponents = np.linspace(0.2, 1.2, 11)  # inclusive
    offsets = np.geomspace(0.01, 10.0, 8)  # geometric spacing
    for p in exponents:
        for off in offsets:
            seq = (np.arange(n, dtype=np.float64) + off) ** (-p)
            seq = np.clip(seq, 0.0, 1000.0)
            sc = float(evaluate_sequence(seq.tolist()))
            if sc < best_score:
                best = seq
                best_score = sc
            rev = seq[::-1]
            sc = float(evaluate_sequence(rev.tolist()))
            if sc < best_score:
                best = rev
                best_score = sc

    # random exponent candidates
    for _ in range(30):
        p = rng.uniform(0.1, 2.0)
        off = 10**rng.uniform(-3, 1)  # 0.001 to 10
        seq = (np.arange(n, dtype=np.float64) + off) ** (-p)
        seq = np.clip(seq,0,1000)
        if rng.random() < 0.5:
            seq = seq[::-1]
        sc = float(evaluate_sequence(seq.tolist()))
        if sc < best_score:
            best = seq
            best_score = sc

    # random uniform or gamma
    for _ in range(20):
        # gamma shape 0.5 (skewed)
        seq = rng.gamma(shape=0.5, scale=1.0, size=n)
        # normalize maybe? scaling doesn't affect ratio.
        seq = np.clip(seq,0,1000)
        sc = float(evaluate_sequence(seq.tolist()))
        if sc < best_score:
            best = seq
            best_score = sc

    # main hill climbing loop
    while time.monotonic() < deadline:
        move_type = rng.integers(0, 100)
        cand = None
        if move_type < 70:  # mass transfer
            # compute convolution for best just once? We need max index for targeted move
            b = np.convolve(best, best)
            k_max = int(np.argmax(b))
            i0 = max(0, k_max - (n - 1))
            i1 = min(n - 1, k_max)
            i_idx = np.arange(i0, i1 + 1, dtype=np.int64)
            j_idx = k_max - i_idx
            # contributions
            prod = best[i_idx] * best[j_idx]
            pos = int(np.argmax(prod))
            src = int(i_idx[pos])
            if rng.random() < 0.4:
                src = int(j_idx[pos])
            dst = int(rng.integers(0, n))
            while dst == src:
                dst = int(rng.integers(0, n))
            # fraction delta
            frac = rng.uniform(0.05, 0.3)  # up to 30%
            delta = best[src] * frac
            # ensure nonneg
            delta = min(delta, best[src])
            delta = min(delta, 1000.0 - best[dst])
            if delta <= 0:
                continue
            cand = best.copy()
            cand[src] -= delta
            cand[dst] += delta
        elif move_type < 90:  # swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            while j == i:
                j = int(rng.integers(0, n))
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:  # random segment scaling
            width = int(rng.integers(1, max(2, n // 200)))  # small segment
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.2))
            cand = best.copy()
            cand[start:start+width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)

        if cand is None:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            # optional: early continue to next iteration

    # final safety: ensure sum>0.01
    if np.sum(best) < 0.01:
        best[0] = max(best[0], 0.01)

    return [float(v) for v in best]

import time, numpy as np

def propose_candidate(seed=42, budget_s=110, **kwargs):
    """Search for a non‑negative step function with a lower AC1 bound."""
    rng = np.random.default_rng(seed)
    # give a short safe margin before the sandbox kill
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # 1.  Base vector (parent if supplied, else modest constant)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            parent = None
    else:
        parent = None

    if parent is None:
        parent = np.full(256, 1.0, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 2.  Exhaustive grid over power‑law families (both decreasing & rev)
    # ------------------------------------------------------------------
    exps = np.linspace(0.45, 1.20, 16)          # many exponents, covering the sweet spot
    offs = np.geomspace(1e-3, 10.0, 10)        # from tiny to moderate offsets

    idx = np.arange(n, dtype=np.float64)

    for p in exps:
        for off in offs:
            # decreasing shape a_i = (i+off)^(-p)
            seq = (idx + off) ** (-p)
            seq = np.clip(seq, 0.0, 1000.0)
            sc = float(evaluate_sequence(seq.tolist()))
            if sc < best_score:
                best, best_score = seq, sc

            # increasing shape (reverse of the decreasing one)
            rev = seq[::-1]
            sc = float(evaluate_sequence(rev.tolist()))
            if sc < best_score:
                best, best_score = rev, sc

            # a small symmetric blend – occasionally useful
            avg = 0.5 * (seq + rev)
            sc = float(evaluate_sequence(avg.tolist()))
            if sc < best_score:
                best, best_score = avg, sc

    # ------------------------------------------------------------------
    # 3.  Random power‑law samples (to explore beyond the grid)
    # ------------------------------------------------------------------
    for _ in range(30):
        p = rng.uniform(0.3, 1.4)
        off = 10 ** rng.uniform(-3, 1)          # 1e-3 … 10
        seq = (idx + off) ** (-p)
        seq = np.clip(seq, 0.0, 1000.0)
        if rng.random() < 0.5:
            seq = seq[::-1]                     # try the increasing variant
        sc = float(evaluate_sequence(seq.tolist()))
        if sc < best_score:
            best, best_score = seq, sc

    # ------------------------------------------------------------------
    # 4.  Random Gamma‑distributed vectors (wide variety)
    # ------------------------------------------------------------------
    for _ in range(20):
        seq = rng.gamma(shape=0.6, scale=2.0, size=n)  # skewed positive values
        seq = np.clip(seq, 0.0, 1000.0)
        sc = float(evaluate_sequence(seq.tolist()))
        if sc < best_score:
            best, best_score = seq, sc

    # ------------------------------------------------------------------
    # 5.  Local hill‑climbing moves guided by the current convolution peak
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # compute the full convolution of the current best
        b = np.convolve(best, best)
        k_max = int(np.argmax(b))

        # indices i,j that contribute to this maximal entry
        i0 = max(0, k_max - (n - 1))
        i1 = min(n - 1, k_max)
        i_idx = np.arange(i0, i1 + 1, dtype=np.int64)
        j_idx = k_max - i_idx
        prod = best[i_idx] * best[j_idx]

        # pick the pair with the largest product – the biggest offender
        pos = int(np.argmax(prod))
        src = int(i_idx[pos]) if rng.random() < 0.5 else int(j_idx[pos])

        if best[src] <= 0.0:
            continue

        dst = int(rng.integers(0, n))
        while dst == src:
            dst = int(rng.integers(0, n))

        # move a modest random fraction of mass from src to dst
        delta = best[src] * rng.uniform(0.03, 0.20)
        delta = min(delta, best[src])
        delta = min(delta, 1000.0 - best[dst])
        if delta <= 0.0:

        cand[dst] += delta

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            continue

        # occasionally try a random swap – may break a stalemate
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)

            cand2[i], cand2[j] = cand2[j], cand2[i]

    # 6.  Safety – guarantee a non‑empty, feasible solution
    # ------------------------------------------------------------------
    if np.sum(best) < 0.01:
        best = np.clip(best, 0.0, 1000.0)
        if np.sum(best) < 0.01:
            best[0] = 0.01

    return [float(v) for v in best]
