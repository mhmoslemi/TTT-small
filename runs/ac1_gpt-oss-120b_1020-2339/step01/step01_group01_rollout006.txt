                continue
            delta = allow * rng.uniform(0.02, 0.14)
            if delta <= 0.0:
                continue

            # incremental update of the convolution
            b_new = b.copy()
            # remove contributions of src
            b_new[src:src + n] -= 2.0 * delta * a
            b_new[2 * src] += delta * delta
            # add contributions of dst
            b_new[dst:dst + n] += 2.0 * delta * a
            b_new[2 * dst] += delta * delta
            if src != dst:
                b_new[src + dst] -= 2.0 * delta * delta

            new_max = b_new.max()
            new_score = 2.0 * n * new_max / (sum_a * sum_a)
            if new_score < local_best_score - 1e-12:
                # apply this move
                local_best_score = new_score
                best_a = a.copy()
                best_a[src] -= delta
                best_a[dst] += delta
                best_b = b_new
                improved = True

        if improved:
            a = best_a
            b = best_b
            best_score = local_best_score
            continue

        # ------------------------------------------------------------------
        # 4.  Occasional random stretch (escape local minima).
        # ------------------------------------------------------------------
        if rng.random() < 0.005:
            width = int(rng.integers(1, max(2, n // 120)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            alt = a.copy()
            alt[start:stop] *= np.exp(rng.normal(0.0, 0.12, size=width))
            alt = np.clip(alt, 0.0, 1000.0)
            if alt.sum() >= 0.01:
                sc = float(evaluate_sequence(alt.tolist()))
                if sc < best_score:
                    a = alt
                    b = np.convolve(a, a)
                    best_score = sc
                    continue

        # ------------------------------------------------------------------
        # 5.  Periodic full recomputation to avoid drift (cheap, occasional).
        # ------------------------------------------------------------------
        if iters % 2000 == 0:
            b = np.convolve(a, a)
            max_b = b.max()
            best_score = 2.0 * n * max_b / (sum_a * sum_a)

    # ------------------------------------------------------------------
    # 6.  Final safety – ensure sum is non‑trivial.
    # ------------------------------------------------------------------
    if a.sum() < 0.01:
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            a[0] = 0.01

    return [float(v) for v in a]
