def compute_ac1(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    n = seq.shape[0]
    # compute convolution via FFT
    m = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    # conv length = m, only first 2n-1 entries are valid
    max_b = np.max(conv[:2*n-1])
    return 2.0 * n * max_b / (sum_a * sum_a)

import math

def _ac1_score(seq: np.ndarray, n: int) -> float:
    """Fast AC1 score using FFT (does not clip the sequence)."""
    seq = np.clip(seq, 0.0, MAX_VAL)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    max_b = np.max(conv[: 2 * n - 1])
    return 2.0 * n * max_b / (sum_a * sum_a)


def _power_seq(p: float, n: int) -> np.ndarray:
    """a_i = MAX_VAL * ((n-1-i)/(n-1))**p  (monotone decreasing)."""
    if n == 1:
        return np.full(1, MAX_VAL, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    frac = (n - 1 - idx) / (n - 1)
    seq = MAX_VAL * np.power(frac, p)
    return np.clip(seq, 0.0, MAX_VAL)
