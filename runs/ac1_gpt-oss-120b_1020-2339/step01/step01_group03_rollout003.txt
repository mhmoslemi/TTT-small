import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _full_conv(a: np.ndarray, size: int) -> np.ndarray:
    """Full (linear) self‑convolution via FFT, length 2*len(a)-1."""
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    return conv[: 2 * a.size - 1]


def _grad_at(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the self‑convolution entry with index k w.r.t. a."""
    n = a.size
    g = np.zeros_like(a)
    if k < n:
        # indices 0..k participate
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        # tail part (k >= n)
        start = k - (n - 1)
        length = 2 * n - 1 - k
        g[start:] = 2.0 * a[:length][::-1]
    return g


def _score_from_conv(conv: np.ndarray, sum_a: float, n: int) -> float:
    """Exact AC1 score from a convolution array."""
    return float(2.0 * n * conv.max() / (sum_a ** 2))
