import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    sum_needed = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_half:
            h_initial = h_initial[:n_half]
        h_start = h_initial.copy()
    else:
        h_start = np.zeros(n_half)

    def compute_overlap(h):
        if np.sum(h) == 0:
            return np.inf
        scale = sum_needed / np.sum(h)
        h_scaled = np.clip(h * scale, 0, 1)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    best_c5 = np.inf
    best_h = np.zeros(n_half)

    def objective(x):
        nonlocal best_c5, best_h
        c = compute_overlap(x)
        if c < best_c5:
            best_c5 = c
            best_h = x.copy()
        return c

    start_time = time.time()
    try:
        result = differential_evolution(objective, bounds, 
                                       strategy='best1bin', 
                                       popsize=30, 
                                       tol=1e-6, 
                                       mutation=(0.5, 0.9),
                                       recombination=0.9,
                                       maxiter=int(200),
                                       seed=seed + 100)
        best_c5 = result.fun
        best_h = result.x
    except:
        pass

    if best_c5 > np.inf:
        try:
            result = minimize(objective, best_h, 
                              method='L-BFGS-B', 
                              bounds=bounds, 
                              tol=1e-6,
                              options={'maxiter': 50})
            best_c5 = result.fun
            best_h = result.x
        except:
            pass

    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points * 2
