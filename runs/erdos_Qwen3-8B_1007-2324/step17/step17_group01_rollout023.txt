            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def compute_gradient(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (len(h_full) - 1)
            grad = np.zeros_like(h)
            h_len = len(h_full)
            for i in range(len(h)):
                i_plus = (i + k + h_len) % h_len
                i_minus = (i - k + h_len) % h_len
                grad[i] = (1 - h_full[i_plus]) * dx - h_full[i_minus] * dx
            return grad
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            jac=compute_gradient,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_points)}],
            options={
                'ftol': 1e-12,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    return best_h, best_c5, n_points
