import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    # Use the parent solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    
    # Perturb initial solution to explore nearby regions
    if start_points:
        h_perturb = start_points[0] + np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (integral_target / np.sum(h_perturb))
        start_points.append(h_perturb)
    
    # Binary initial guess
    binary_initial = np.random.rand(n_points)
    binary_initial = np.round(binary_initial)
    binary_initial = binary_initial * (integral_target / np.sum(binary_initial))
    start_points.append(binary_initial)
    
    # Random initial guess with modified distribution
    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    start_points.append(random_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        # Normalize starting point
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            h_full = h
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        # Define optimization bounds and constraint
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use COBYLA and SLSQP with adjusted settings
        for method in ['COBYLA', 'SLSQP']:
            if method == 'COBYLA':
                result = scipy.optimize.minimize(
                    compute_overlap,
                    h_start,
                    method=method,
                    bounds=bounds,
                    constraints=cons,
                    options={
                        'maxiter': 100,
                        'disp': False,
                        'rhobeg': 0.2,
                        'rhoend': 0.01
                    }
                )
            else:
                result = scipy.optimize.minimize(
                    compute_overlap,

                    method=method,
                    bounds=bounds,
                    constraints=cons,
                    options={
                        'ftol': 1e-10,
                        'maxiter': 150,
                        'disp': False,
                        'eps': 1e-8
                    }
                )

                current_c5 = compute_overlap(result.x)
