import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    initial_guesses = []

    initial_guesses.append(binary_initial)

    # Random initial guess
    random_initial = np.random.rand(n_half)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    initial_guesses.append(random_initial)

    structured_initial = np.zeros(n_half)

    structured_initial[segments:segments*2] = 0.0
    structured_initial[segments*2:segments*3] = 1.0

    initial_guesses.append(structured_initial)

    def generate_gauss(n_half):

        h = np.zeros(n_half)

        h[1:pos] = 0.5

        h = h * (integral_target / np.sum(h))

    initial_guesses.append(generate_gauss(n_half))

    def compute_overlap(h_values):
