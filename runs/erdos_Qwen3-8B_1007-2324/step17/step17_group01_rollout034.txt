    result_slsqp = opt.minimize(
        objective_slsqp, best_h_de,
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    # Optional resolution doubling if time allows
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        n_points *= 2
        dx = 2.0 / n_points
        target_sum_double = n_points / 2
        h_double = np.concatenate([best_h, best_h])
        h_double = h_double * (target_sum_double / np.sum(h_double) if np.sum(h_double) > 1e-6 else 1.0)
        h_double = np.clip(h_double, 0.0, 1.0)
        h_double = h_double * (target_sum_double / np.sum(h_double))
        
        def objective_de_double(x):
            h = x
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        bounds_double = [(0.0, 1.0) for _ in range(n_points)]
        result_de_double = opt.differential_evolution(
            objective_de_double, bounds_double, strategy='best1bin',
            popsize=4, maxiter=30, tol=1e-6, mutation=(0.5, 1), recombination=0.9, seed=seed
        )
        best_h_double = result_de_double.x
        best_c5_double = result_de_double.fun
        
        def objective_slsqp_double(x):
            h = x
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result_slsqp_double = opt.minimize(
            objective_slsqp_double, best_h_double,
            method='SLSQP', bounds=bounds_double,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_double}],
            options={'ftol': 1e-10, 'maxiter': 150, 'disp': False}
        )
        best_h_double = result_slsqp_double.x
        best_c5_double = result_slsqp_double.fun
        
        if best_c5_double < best_c5:
            best_h = best_h_double
            best_c5 = best_c5_double
    
    # Final clipping and sum correction
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = final_h * (target_sum / np.sum(final_h) if np.sum(final_h) > 1e-6 else 1.0)
    final_h = np.clip(final_h, 0.0, 1.0)
    
    return final_h, best_c5, n_points
