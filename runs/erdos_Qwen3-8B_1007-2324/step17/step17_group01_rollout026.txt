            h_start = h_start * scale
            h_start = np.clip(h_start, 0.0, 1.0)
        
        # Define objective function with penalty
        def obj_func(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            penalty = 1e6 * abs(np.sum(h) - integral_target)
            return max_overlap + penalty
        
        # Run differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        try:
            de_result = scipy.optimize.differential_evolution(
                obj_func, bounds, strategy='best1bin', maxiter=100, popsize=20,
                tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed
            )
            h_de = de_result.x
            h_de = np.clip(h_de, 0.0, 1.0)
            
            # Run SLSQP refinement
            def compute_overlap(x):
                return np.max(np.correlate(x, 1 - x, mode='full')) * dx
            
            def constraint(x):
                return np.sum(x) - integral_target
            
            slsqp_result = scipy.optimize.minimize(
                compute_overlap, h_de, method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones(n_points)}],
                options={'ftol': 1e-12, 'maxiter': 50, 'disp': False, 'eps': 1e-8}
            )
            
            if slsqp_result.success:
                current_c5 = slsqp_result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = slsqp_result.x
        except:
            pass
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = np.array(best_h, dtype=np.float64)
    return best_h, best_c5, n_points
