import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    penalty_factor = 1e6

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        integral = np.sum(h)
        penalty = penalty_factor * (integral - integral_target) ** 2
        return compute_overlap(h) + penalty

    initial_guesses = []
    if 'initial_h_values' in globals():
        parent = initial_h_values[:n_points]
        parent = (parent - np.min(parent)) / (np.max(parent) - np.min(parent) + 1e-10)
        initial_guesses.append(parent)
    
    random_guess = np.random.rand(n_points) * integral_target / np.mean(np.random.rand(n_points))
    initial_guesses.append(random_guess)

    def generate_gaussian(n):
        x = np.linspace(0, 2, n)
        peaks = np.linspace(0.5, 1.5, 5)
        h = np.zeros(n)
        for peak in peaks:
            h += np.exp(-((x - peak)/0.25)**2)
        h = h / np.sum(h) * integral_target
        return h
    initial_guesses.append(generate_gaussian(n_points))

    best_h = None
    best_c5 = np.inf
    max_evals = 10000

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        guess_sum = np.sum(guess)
        guess = guess * (integral_target / guess_sum) if guess_sum > 1e-6 else guess
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=400,
            popsize=30,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.5,
            seed=seed,
            disp=False,
            polish=True,
            vectorize=True
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    if best_h is None:
        best_h = initial_guesses[0]
        best_c5 = compute_overlap(best_h)

    return best_h, best_c5, n_points
