        h_initial = np.array(initial_h_values)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        if len(h_initial) < n_points:
            h_initial = np.concatenate([h_initial, np.zeros(n_points - len(h_initial))])
        elif len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        h_initial = h_initial * (integral_target / np.sum(h_initial))
        start_points.append(h_initial)

    # Binary initial guess
    binary_initial = np.random.rand(n_points)
    binary_initial = binary_initial * (integral_target / np.sum(binary_initial))
    start_points.append(binary_initial)

    # Structured initial guess with high/low segments
    def make_segmented(n):
        seg = np.zeros(n)
        for i in range(n // 4):
            seg[i] = 1.0
        for i in range(n // 4, 2 * n // 4):
            seg[i] = 0.0
        for i in range(2 * n // 4, 3 * n // 4):
            seg[i] = 1.0
        for i in range(3 * n // 4, n):
            seg[i] = 0.0
        return seg

    start_points.append(make_segmented(n_points))

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
