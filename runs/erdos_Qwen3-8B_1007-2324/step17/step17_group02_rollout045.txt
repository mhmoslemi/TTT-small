import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    start_points = []
    
    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        # Extract first half and mirror
        h_half_initial = h_initial[:n_half]
        start_points.append(h_half_initial)
    
    # Generate symmetric initial guess from cosine function
    x_values = np.linspace(0, 1, n_half)
    h_initial = 0.5 + 0.5 * np.cos( (np.pi / 2) * x_values )
    sum_initial = np.sum(h_initial)
    h_initial_scaled = h_initial * (15.0 / sum_initial)
    start_points.append(h_initial_scaled)
    
    # Add some random perturbations to initial guesses
    for _ in range(2):
        # Random start with sum constraint
        h_rand = np.random.rand(n_half)
        h_rand_sum = np.sum(h_rand)
        h_rand_scaled = h_rand * (15.0 / h_rand_sum)
        start_points.append(h_rand_scaled)
    
    best_c5 = np.inf
    best_h_half = np.zeros(n_half)
    
    # Function to compute the overlap integral
    def compute_overlap(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # Optimization settings
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}]
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    # Run optimization for each starting point
    for h_start in start_points:
        # Ensure the starting point is within bounds
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = h_start * (integral_target / 2) / np.sum(h_start) if np.sum(h_start) > 1e-6 else h_start * (integral_target / 2) / np.sum(h_start)
        
        # Define the optimization problem
        result = scipy.optimize.minimize(
            compute_overlap, 
            h_start, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={
                'ftol': 1e-9, 
                'maxiter': 100, 
                'disp': False, 
                'eps': 1e-8
            }
        )
        
        # Check for success
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_half = result.x
    
    # Construct the full h_values
    best_h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    return best_h_full, best_c5, n_points
