import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = integral_target

        current_sum = np.sum(h_start)
        if current_sum < 1e-6:

        else:
            h_start = h_start * (integral_target / current_sum)

            corr = np.correlate(h_full, 1 - h_full, mode='full')

    final_c5 = compute_overlap(best_h)
    run_time = time.time() - start_time
    if run_time > budget_s:
        print("Budget exceeded, returning best found")
    return (best_h, final_c5, n_points)
