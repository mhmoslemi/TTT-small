    # Use the parent solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        start_points.append(h_initial[:n_points])

    # Generate three harmonic initial guesses
    x = np.linspace(0, 2, n_points)
    start_points.append(0.5 + 0.3 * np.cos(np.pi * x))
    start_points.append(0.5 + 0.5 * np.cos(np.pi * x))
    start_points.append(0.5 + 0.3 * np.cos(2 * np.pi * x))
    start_points = [np.clip(h, 0, 1) for h in start_points]

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        # Normalize to satisfy integral constraint
        if np.sum(h_start) < 1e-6:
            h_scaled = h_start * (integral_target / np.sum(h_start))
        else:
            h_scaled = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Use L-BFGS-B with tighter tolerances
        result = scipy.optimize.minimize(
            compute_overlap,
            h_scaled,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={
                'ftol': 1e-8,
                'gtol': 1e-8,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
