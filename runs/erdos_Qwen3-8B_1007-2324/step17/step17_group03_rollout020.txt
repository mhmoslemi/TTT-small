import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
        elif len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        start_points = [h_initial]
    else:
        start_points = []

    # Add perturbed versions of initial guess
    for _ in range(3):
        h_perturb = h_initial.copy()
        noise = np.random.normal(0, 0.15, n_points)
        h_perturb += noise
        h_perturb = np.clip(h_perturb, 0, 1)
        # Rescale to maintain integral
        h_perturb = h_perturb * (integral_target / np.sum(h_perturb))
        start_points.append(h_perturb)

    # Add random binary pattern
    binary_start = np.zeros(n_points)
    n_ones = int(np.round(0.3 * n_points))
    idxs = np.random.choice(n_points, n_ones, replace=False)
    binary_start[idxs] = 1
    start_points.append(binary_start)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    start_time = time.time()

    for h_start in start_points:
        # Ensure valid starting point
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))

        def objective(x):
            h_full = np.concatenate([x, x[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - integral_target / 2

        # Bounds for first half of h
        bounds = [(0.0, 1.0) for _ in range(n_points // 2)]
        res = optimize.minimize(
            objective,
            h_start[:n_points // 2],
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-9,
                'maxiter': 150,
                'disp': False,
                'eps': 1e-8
            },
            tol=1e-6
        )

        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = np.concatenate([res.x, res.x[::-1]])

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 5:
            break

    return best_h, best_c5, n_points
