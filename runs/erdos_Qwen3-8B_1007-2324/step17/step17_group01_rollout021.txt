        h_de,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons_slsqp,
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    )
    
    current_time = time.time()
    if current_time - start_time > budget_s:
        h_best = result_slsqp.x
        c5_bound = objective_slsqp(h_best)
        return h_best, c5_bound, n_points
    
    h_best = result_slsqp.x
    c5_bound = objective_slsqp(h_best)
    
    return h_best, c5_bound, n_points
