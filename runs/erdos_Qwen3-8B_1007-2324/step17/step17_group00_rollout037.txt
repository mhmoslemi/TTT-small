        x_normalized = x * (n_half / sum_x)
        h = np.concatenate([x_normalized, x_normalized[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (n_half / np.sum(h_start))
        else:
            h_start = h_start * (n_half / np.sum(h_start))
        
        def obj(x):
            return objective(x)
        
        start_time = time.time()
        result = differential_evolution(obj, bounds, strategy='best1bin', 
                                        popsize=20, mutation=(0.5, 0.9), 
                                        maxiter=100, tol=1e-6, seed=seed + 1000)
        time_used = time.time() - start_time
        time_remaining -= time_used
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if time_remaining > 50:
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_half}]
        result_local = minimize(objective, best_h, method='SLSQP', 
                                bounds=bounds, constraints=constraints, 
                                options={'ftol': 1e-9, 'maxiter': 50})
        if result_local.success:
            current_c5 = result_local.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_local.x
    
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * (n_half / sum_best)
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = objective(best_h)
    return best_h_full, final_c5, n_points
