import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        # Enforce symmetry
        if n_points % 2 == 0 and n_half == n_points // 2:
            start_points.append(h_initial[:n_half])
        else:
            start_points.append(h_initial)

    # Binary initial guess
    binary_initial = np.zeros(n_half)
    num_ones = n_half // 2
    indices = np.random.choice(n_half, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)

    best_c5 = np.inf
    best_h = np.zeros(n_half)

    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - integral_target / 2

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones_like}],
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
