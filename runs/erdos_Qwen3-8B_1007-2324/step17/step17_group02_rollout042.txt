    random_initial *= integral_target
    start_points.append(random_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={
                'ftol': 1e-9,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final adjustment to ensure integral constraint and bounds
    best_h = best_h * (integral_target / np.sum(best_h))
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_overlap(best_h)
    
    return best_h, final_c5, n_points
