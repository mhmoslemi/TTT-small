def run(seed=42, budget_s=100, **kwargs):

    start_points = []
    
    if 'initial_h_values' in globals():

        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    
    # Structured perturbations around initial solution
    if 'initial_h_values' in globals():
        h_perturbed = np.array(initial_h_values)
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        start_points.append(h_perturbed)
    
    # Binary initial guess
    binary_initial = np.zeros(n_half)
    num_ones = n_half // 2
    indices = np.random.choice(n_half, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    start_points.append(binary_full[:n_points])
    
    # Random initial guess
    random_initial = np.random.rand(n_half)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    start_points.append(random_full[:n_points])
    
    best_c5 = np.inf
    best_h = np.zeros(n_half)
    
    for h_start in start_points:
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def constraint(x):
            return np.sum(x) - integral_target / 2
        
        # Repair function
        def repair(x):
            target_sum = integral_target
            total = np.sum(x)
            if abs(total) < 1e-10:
                return np.clip(x, 0.0, 1.0)
            scale = target_sum / total
            x_corrected = x * scale
            x_corrected = np.clip(x_corrected, 0.0, 1.0)
            while not np.isclose(np.sum(x_corrected), target_sum, atol=1e-5):
                d = (target_sum - np.sum(x_corrected)) / len(x_corrected)
                x_corrected += d * 0.5
                x_corrected = np.clip(x_corrected, 0.0, 1.0)
            return x_corrected
        
        # Apply repair to the starting point
        h_start_repaired = repair(h_start)
        
        result = scipy.optimize.minimize(
            compute_overlap,
