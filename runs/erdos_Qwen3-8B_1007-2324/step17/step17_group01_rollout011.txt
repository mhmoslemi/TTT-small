import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_candidates = [h_initial]
    else:
        h_candidates = []
    
    # Binary initial guess
    binary_initial = np.zeros(n_points // 2)
    num_ones = n_points // 4
    indices = np.random.choice(n_points // 2, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    h_candidates.append(binary_full[:n_points])
    
    # Random initial guess
    random_initial = np.random.rand(n_points // 2)
    random_initial = random_initial * (target_sum / np.sum(random_initial))
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    h_candidates.append(random_full[:n_points])
    
    for h_start in h_candidates:
        h_start = h_start * (target_sum / np.sum(h_start))
        h_start = np.clip(h_start, 0.0, 1.0)
        
        def objective(x):
            h = x
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def constraint(x):
            return np.sum(x) - target_sum
        
        # DE optimization
        def objective_with_penalty(x):
            h = x
            penalty = 1e6 * (np.sum(h) - target_sum)**2
            return objective(h) + penalty
        
        result_de = scipy.optimize.differential_evolution(
            objective_with_penalty,
            [(0.0, 1.0) for _ in range(n_points)],
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=60,
            seed=seed
        )
        
        best_DE = result_de.x
        
        # SLSQP refinement
        result_slsqp = scipy.optimize.minimize(
            objective,
            best_DE,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-9,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        current_c5 = result_slsqp.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x.copy()
    
    return best_h, best_c5, n_points
