import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

        h_initial = h_initial * (integral_target / np.sum(h_initial))

        h_perturb = np.clip(h_initial + perturb, 0, 1)
        h_perturb = h_perturb * (integral_target / np.sum(h_perturb))
        start_points.append(h_perturb)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    for h_start in start_points:
        def objective(x):
            h = np.array(x)
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            max_corr = np.max(corr)
