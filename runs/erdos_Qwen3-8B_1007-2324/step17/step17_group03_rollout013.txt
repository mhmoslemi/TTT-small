import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    
    # Binary initial guess
    binary_initial = np.zeros(n_points // 2)
    num_ones = (n_points // 2) // 2
    indices = np.random.choice(n_points // 2, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    start_points.append(binary_full[:n_points])
    
    # Random initial guess
    random_initial = np.random.rand(n_points // 2)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    start_points.append(random_full[:n_points])
    
    # Sinusoidal initial guess
    x_points = np.linspace(0, 2, n_points)
    sine_initial = 0.5 + 0.5 * np.sin(np.pi * x_points)
    sine_initial = sine_initial[:n_points // 2]
    sine_full = np.concatenate([sine_initial, sine_initial[::-1]])
    start_points.append(sine_full[:n_points])
    
    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)
    
    for h_start in start_points:
        if np.sum(h_start) < 1e-6:

        else:
            h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start[:n_points // 2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[

                'disp': False,

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
