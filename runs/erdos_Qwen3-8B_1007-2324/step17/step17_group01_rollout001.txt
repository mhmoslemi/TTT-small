import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    # Handle initial_h_values if available
    h_initial = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
        if np.sum(h_initial) != integral_target:
            h_initial = h_initial * (integral_target / np.sum(h_initial))
    
    # Function to compute C5 value
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # Function to apply constraint scaling
    def scale_h(h):
        total = np.sum(h)
        if abs(total - integral_target) > 1e-8:
            h = h * (integral_target / total)
        return h
    
    # Define penalty for DE optimization
    def de_objective(x):
        h = scale_h(x)
        return compute_c5(h) + 1e6 * np.abs(np.sum(h) * dx - 1)
    
    # Define SLSQP objective and constraint
    def slsqp_objective(h):
        return compute_c5(h)
    
    def slsqp_constraint(h):
        return np.sum(h) * dx - 1
    
    # Generate initial guesses
    start_points = []
    if 'initial_h_values' in globals():
        start_points.append(h_initial)
    # Phase-shifted binary pattern
    binary = np.zeros(n_points)
    half = n_points // 2
    ones = np.random.choice(half, min(half // 2, 10), replace=False)
    binary[ones] = 1
    binary = np.concatenate([binary, binary[::-1]])
    start_points.append(binary[:n_points])
    # Random initial guess
    rand = np.random.rand(n_points)
    rand = rand * (integral_target / np.sum(rand))
    start_points.append(rand)
    
    # Run DE optimization for each starting point
    for h_start in start_points:
        h_start = scale_h(h_start)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = scipy.optimize.differential_evolution(
            de_objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed + len(start_points),
            disp=False
        )
        if result.success:
            h_candidate = result.x
            h_candidate = scale_h(h_candidate)
            # Refine with SLSQP
            cons = [{'type': 'eq', 'fun': slsqp_constraint}]
            res = scipy.optimize.minimize(
                slsqp_objective,
                h_candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
            )
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
