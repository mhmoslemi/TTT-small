        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        c5 = compute_c5(x)
        penalty = 1e10 * (np.sum(x) - integral_target)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    result_de = None
    try:
        result_de = scipy.optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            max_evals=40000,
            popsize=15,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            polish=False
        )
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.6:
        result_de = None

    best_h = np.zeros(n_half)
    best_c5 = np.inf
    if result_de is not None:
        candidate = result_de.x
        sum_candidate = np.sum(candidate)
        if abs(sum_candidate - integral_target) < 1e-5:
            current_c5 = compute_c5(candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate

    if 'initial_h_values' in globals() and result_de is None:
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_half]
        h_initial = h_initial * (integral_target / np.sum(h_initial))
        best_h = h_initial
        best_c5 = compute_c5(best_h)

    if best_h.size > 0:
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
        bounds_local = [(0.0, 1.0) for _ in range(n_half)]
        try:
            result_local = scipy.optimize.minimize(
                compute_c5,
                best_h,
                method='L-BFGS-B',
                bounds=bounds_local,
                constraints=constraints,
                options={
                    'ftol': 1e-12,
                    'maxiter': 100
                }
            )
            if result_local.success:
                best_h = result_local.x
                best_c5 = compute_c5(best_h)
        except:
            pass

    sum_best = np.sum(best_h)
    if not (abs(sum_best - integral_target) < 1e-5):
        best_h = np.zeros(n_half)
        best_c5 = np.inf

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
