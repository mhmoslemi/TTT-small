    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, maxiter=50, tol=1e-6)
    best_h_de = result_de.x
    best_c5_de = objective(best_h_de)
    
    def compute_overlap(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scaled_h = h * (target_sum / current_sum)
        scaled_h = np.clip(scaled_h, 0.0, 1.0)
        h_full = np.concatenate([scaled_h, scaled_h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    result_slsqp = scipy.optimize.minimize(
        compute_overlap,
        best_h_de,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones(n_points)}],
        options={
            'ftol': 1e-12,
            'maxiter': 200,
            'disp': False,
            'eps': 1e-8
        }
    )
    
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
    else:
        best_c5 = best_c5_de
        best_h = best_h_de
    
    final_sum = np.sum(best_h)
    if np.abs(final_sum - target_sum) > 1e-6:
        best_h = best_h * (target_sum / final_sum)
        best_h = np.clip(best_h, 0.0, 1.0)
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
