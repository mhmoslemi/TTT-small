import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    start_points = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) == n_points:
            start_points.append(h_initial)
    
    # Generate additional initial points
    num_extra = 2
    for _ in range(num_extra):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        start_points.append(h_rand)
    
    best_h = np.zeros(n_points)
    best_c5 = np.inf
    start_time = time.time()
    
    for h_start in start_points:
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = h_start * (target_sum / np.sum(h_start))
        
        # DE optimization with constraint penalty
        def objective_de(x):
            h = x.copy()
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            penalty = abs(np.sum(h) - target_sum) * 1e5
            return c5 + penalty
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_de = scipy.optimize.differential_evolution(
            objective_de, bounds, strategy='best1bin',
            maxiter=50, popsize=30, tol=1e-6, mutation=(0.5, 1),
            recombination=0.8, seed=seed, polish=True, disp=False
        )
        de_sol = result_de.x
        
        # SLSQP refinement
        def objective_slsqp(x):
            h = x.copy()
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def constraint(x):
            return np.sum(x) - target_sum
        
        result_slsqp = scipy.optimize.minimize(
            objective_slsqp, de_sol,
            method='SLSQP', bounds=[(0.0, 1.0)]*n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'ftol': 1e-12, 'maxiter': 100, 'disp': False}
        )
        slsqp_sol = result_slsqp.x
        
        current_c5 = objective_slsqp(slsqp_sol)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = slsqp_sol.copy()
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, n_points
