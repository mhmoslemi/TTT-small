        h_sym = np.concatenate([h_initial[:n_points//2], h_initial[n_points//2-1::-1]])
        start_points.append(h_sym[:n_points])
        # Add perturbed version
        perturbed = h_sym + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        start_points.append(perturbed[:n_points])
    else:
        # Fallback to symmetric uniform distribution
        x_initial = np.full(n_points // 2, 15.0 / (n_points // 2))
        h_sym = np.concatenate([x_initial, x_initial[::-1]])
        start_points.append(h_sym[:n_points])
    
    best_c5 = np.inf
    best_x = np.zeros(n_points // 2)
    
    for h_start_full in start_points:
        x_start = h_start_full[:n_points // 2]
        
        def compute_overlap(x):
            h = np.concatenate([x, x[::-1]])
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            x_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - 15.0, 'jac': np.ones_like(x)}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 150,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_x = result.x
    
    h_values = np.clip(np.concatenate([best_x, best_x[::-1]]), 0.0, 1.0)
    assert abs(np.sum(h_values) * dx - 1.0) < 1e-6, "Integral constraint violated"
    return h_values, best_c5, n_points
