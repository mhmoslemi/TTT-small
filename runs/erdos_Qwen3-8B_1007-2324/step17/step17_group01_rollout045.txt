        h_initial = h_initial * (target_sum / np.sum(h_initial))
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # Define constraints for SLSQP
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    # Try multiple starting points
    best_h = h_initial
    best_c5 = compute_overlap(best_h)
    
    # Binary pattern
    binary_h = np.zeros(n_points)
    num_ones = int(np.ceil(target_sum))
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_h[indices] = 1
    binary_h = binary_h * (target_sum / np.sum(binary_h))
    res = scipy.optimize.minimize(compute_overlap, binary_h, method='SLSQP',
                                  bounds=bounds_slsqp, constraints=cons,
                                  options={'ftol': 1e-12, 'maxiter': 300})
    if res.success:
        current_c5 = res.fun
        if current_c5 < best_c5:
            best_h = res.x
            best_c5 = current_c5
    
    # Random pattern
    random_h = np.random.rand(n_points)
    random_h = random_h * (target_sum / np.sum(random_h))
    res = scipy.optimize.minimize(compute_overlap, random_h, method='SLSQP',
                                  bounds=bounds_slsqp, constraints=cons,
                                  options={'ftol': 1e-12, 'maxiter': 300})
    if res.success:
        current_c5 = res.fun
        if current_c5 < best_c5:
            best_h = res.x
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
