import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values[:n_points])
        start_points.append(h_initial)
        for _ in range(3):
            perturb = h_initial.copy()
            noise = np.random.normal(0, 0.02, n_points)
            perturb += noise
            perturb = np.clip(perturb, 0, 1)
            perturb = perturb * (integral_target / np.sum(perturb))
            start_points.append(perturb)
    
    binary_initial = np.zeros(n_points // 2)
    num_ones = n_points // 4
    indices = np.random.choice(n_points // 2, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    start_points.append(binary_full[:n_points])
    
    random_initial = np.random.rand(n_points // 2) * integral_target
    random_initial /= np.sum(random_initial)
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    start_points.append(random_full[:n_points])
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like(x)}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={
                'ftol': 1e-9,
                'maxiter': 250,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, n_points
