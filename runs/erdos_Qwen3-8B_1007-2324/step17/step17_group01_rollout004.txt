import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    
    # Perturb parent
    if 'initial_h_values' in globals():
        for _ in range(3):
            perturbed = h_initial.copy()
            noise = np.random.normal(0, 0.05, n_points)
            perturbed += noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed * (integral_target / np.sum(perturbed))
            start_points.append(perturbed)
    
    # Binary initial guess
    binary_initial = np.zeros(n_half)
    num_ones = n_half // 2
    indices = np.random.choice(n_half, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    start_points.append(binary_full[:n_points])
    
    # Random initial guess
    random_initial = np.random.rand(n_half)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    start_points.append(random_full[:n_points])
    
    # Sine-like initial guess
    x = np.linspace(0, 2, n_points)
    y = 0.5 * (1 + np.sin(np.pi * x))
    y = y * (integral_target / np.sum(y))
    start_points.append(y[:n_points])
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def constraint(x):
            return np.sum(x) - integral_target
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start[:n_half],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones_like}],
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            h_candidate = result.x.copy()

            c5 = compute_overlap(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
    
    return np.concatenate([best_h[:n_half], best_h[:n_half][::-1]]), best_c5, n_points
