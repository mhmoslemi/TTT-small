    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        # Scale initial guess to meet integral constraint
        sum_initial = np.sum(h_initial)
        if sum_initial == 0:
            initial_guess = np.random.rand(n_points)
            initial_guess = initial_guess * (integral_target / np.sum(initial_guess))
        else:
            initial_guess = h_initial * (integral_target / sum_initial)
    else:
        # Generate a random initial guess
        initial_guess = np.random.rand(n_points)
        initial_guess = initial_guess * (integral_target / np.sum(initial_guess))
    
    # Define the objective function
    def obj(h_candidate):
        h = h_candidate.copy()
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf  # Avoid division by zero
        h_scaled = h * (integral_target / sum_h)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # Bounds for each h[i] (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Run differential evolution
    result = opt.differential_evolution(
        obj,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-9,
        maxiter=100,
        polish=True,
        x0=initial_guess,
        options={'ftol': 1e-9, 'maxiter': 100, 'mutation': (0.5, 1.0), 'recombination': 0.8}
    )
    
    # Compute the final best solution
    best_h_unscaled = result.x
    sum_unscaled = np.sum(best_h_unscaled)
    best_h = best_h_unscaled * (integral_target / sum_unscaled)
    
    # Compute C5 value
    h_full = np.concatenate([best_h, best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx
    
    # Ensure constraints are satisfied
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Bounds violation"
    assert np.abs(np.sum(best_h) - integral_target) < 1e-3, "Integral constraint failure"
    
    return best_h, c5_bound, n_points
