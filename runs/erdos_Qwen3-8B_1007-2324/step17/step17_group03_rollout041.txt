    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        # Ensure h_start is a 60-element array and scale to target integral
        h_start = np.clip(h_start, 0, 1)
        sum_h = np.sum(h_start)
        if sum_h < 1e-6:
            h_start = h_start * (integral_target / sum_h)
        else:
            h_start = h_start * (integral_target / sum_h)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_points)}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
