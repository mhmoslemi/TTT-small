import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(h):
        c5 = compute_c5(h)
        sum_h = np.sum(h)
        penalty = 1e5 * np.abs(sum_h - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=100,
        seed=seed,
        polish=True,
        disp=False
    )

    best_h = result.x
    current_sum = np.sum(best_h)
    if abs(current_sum - target_sum) > 1e-3:
        scale = target_sum / current_sum
        best_h = np.clip(best_h * scale, 0, 1)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
