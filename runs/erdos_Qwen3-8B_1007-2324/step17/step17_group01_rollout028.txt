    start_points.append(random_full[:n_points])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - integral_target

        def compute_overlap_with_gradient(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            max_corr = np.max(corr)
            max_index = np.argmax(corr)
            shift = max_index
            grad = np.zeros(n_points)
            for m in range(n_points):
                m_plus = m + shift
                if m_plus < n_points:
                    term1 = (1 - h[m_plus]) * dx
                else:
                    term1 = 1.0 * dx
                m_minus = m - shift
                if m_minus >= 0:
                    term2 = -h[m_minus] * dx
                else:
                    term2 = 0.0
                grad[m] = term1 + term2
            return max_corr * dx, grad

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones(n_points)}],
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            },
            jac=lambda x: compute_overlap_with_gradient(x)[1]
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()

    return best_h, best_c5, n_points
