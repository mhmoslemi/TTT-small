    # Random initial guess with modified distribution
    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    start_points.append(random_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        # Normalize starting point
        if np.sum(h_start) < 1e-6:

        def compute_overlap(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Define optimization bounds and constraint
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like(x)}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use COBYLA with adjusted settings
        result = scipy.optimize.minimize(
