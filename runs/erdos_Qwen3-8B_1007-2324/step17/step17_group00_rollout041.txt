    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_values_initial = h_initial[:n_half]
        if np.sum(h_values_initial) != 0:
            h_values_initial = (h_values_initial / np.sum(h_values_initial)) * integral_target
        else:
            h_values_initial = np.full(n_half, integral_target / n_half)
        start_points.append(h_values_initial)
        start_points.append(h_values_initial[::-1])

    for _ in range(4):
        rand_vec = np.random.rand(n_half)
        rand_vec = (rand_vec / np.sum(rand_vec)) * integral_target
        start_points.append(rand_vec)

    def objective(x):
        h_values = x
        h_full = np.concatenate([h_values, h_values[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        penalty = 1e10 * (np.sum(x) - integral_target)**2
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    de_result = None
    try:
        de_params = {
            'strategy': 'best1bin',
            'popsize': 15,
            'mutation': (0.5, 1),
            'recombination': 0.8,
            'tol': 1e-6,
            'maxiter': 100,
            'disp': False
        }
        de_result = scipy.optimize.differential_evolution(objective, bounds, **de_params)
    except:
        pass

    elapsed = time.time() - start_time

    if de_result is None or elapsed > budget_s * 0.7:
        best_x = np.zeros(n_half)
        best_c5 = np.inf
        for x in start_points:
            if np.sum(x) != 0:
                x = (x / np.sum(x)) * integral_target
            h_values = x
            h_full = np.concatenate([h_values, h_values[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_x = x
        best_h = best_x
        final_c5 = best_c5
    else:
        best_x = de_result.x
        best_c5_val = de_result.fun
        def local_objective(x):
            h_values = x
            h_full = np.concatenate([h_values, h_values[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
        local_result = scipy.optimize.minimize(local_objective, best_x, method='L-BFGS-B', bounds=[(0, 1)] * n_half, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-12})
        best_x = local_result.x
        final_c5 = local_result.fun

    if np.sum(best_x) != 0:
        best_x = (best_x / np.sum(best_x)) * integral_target
    best_h_full = np.concatenate([best_x, best_x[::-1]])
    return best_h_full, final_c5, n_points
