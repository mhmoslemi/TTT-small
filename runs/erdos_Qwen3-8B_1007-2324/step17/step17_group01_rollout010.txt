import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = np.array([])

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h[:n_points]
        initial_h = np.clip(initial_h * (target_sum / np.sum(initial_h)), 0, 1)
    
    def repair(h):
        s = np.sum(h)
        if np.abs(s - target_sum) < 1e-10:
            return h
        scale = target_sum / s
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective(h):
        h_r = repair(h)
        corr = np.correlate(np.concatenate([h_r, h_r[::-1]]), 1 - np.concatenate([h_r, h_r[::-1]]), mode='full')
        return np.max(corr) * dx

    def compute_overlap(h):
        h_r = repair(h)
        corr = np.correlate(np.concatenate([h_r, h_r[::-1]]), 1 - np.concatenate([h_r, h_r[::-1]]), mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    if 'initial_h_values' in globals():
        result = scipy.optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=20,
            tol=1e-12,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True,
            disp=False
        )
    else:
        result = scipy.optimize.minimize(
            compute_overlap,
            np.random.rand(n_points) * (target_sum / np.random.rand(n_points).sum()),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones(n_points)}],
            options={'ftol': 1e-12, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
        )

    best_h = repair(result.x)
    best_c5 = compute_overlap(best_h)
    best_h_full = np.concatenate([best_h, best_h[::-1]])
