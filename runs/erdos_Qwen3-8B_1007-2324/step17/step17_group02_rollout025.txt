import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target = n_points / 2
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        start_points.append(h_initial[:n_points])

    # Sine-based initial guess
    x = np.linspace(0, 2, n_points)
    sine_h = 0.5 + 0.5 * np.sin(np.pi * x)
    start_points.append(sine_h)

    # Binary initial guess
    binary = np.zeros(n_points)
    num_ones = n_points // 2
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary[indices] = 1
    start_points.append(binary)

    # Random initial guess
    random = np.random.rand(n_points)
    random = random * (target / np.sum(random))
    random = np.clip(random, 0, 1)
    start_points.append(random)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': np.ones(n_points)})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for h_start in start_points:
        h_start = h_start * (target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start * (target / np.sum(h_start))
        h_start = np.clip(h_start, 0.0, 1.0)
        result = scipy.optimize.minimize(compute_c5, h_start, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-12, 'maxiter': 200})
        if result.success:
            current_c5 = compute_c5(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return (best_h, best_c5, n_points)
