    h_harmonic = 0.5 + 0.5 * np.cos(np.pi * x)
    h_harmonic = np.clip(h_harmonic, 0, 1)
    current_sum = np.sum(h_harmonic)
    scale = integral_target / current_sum
    h_harmonic = h_harmonic * scale
    h_harmonic = np.clip(h_harmonic, 0, 1)
    start_points.append(h_harmonic)
    
    random_start = np.random.rand(n_points)
    random_start = random_start * (integral_target / np.sum(random_start))
    start_points.append(random_start)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def objective(x):
            return compute_overlap(x)
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like(x)}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={
                'ftol': 1e-9,
                'gtol': 1e-9,
                'maxiter': 100,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
