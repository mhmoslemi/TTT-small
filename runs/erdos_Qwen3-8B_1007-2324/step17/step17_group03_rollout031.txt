        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        return c5 + penalty

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        initial_sum = np.sum(initial_h)
        scale = integral_target / initial_sum
        adjusted_initial = np.clip(initial_h * scale, 0.0, 1.0)
    else:
        adjusted_initial = np.random.rand(n_points)
        adjusted_initial = np.clip(adjusted_initial * (integral_target / np.sum(adjusted_initial)), 0.0, 1.0)

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_c5 = objective(best_h)

    adjusted_h = best_h.copy()
    current_sum = np.sum(adjusted_h)
    target_sum = integral_target
    if abs(current_sum - target_sum) > 1e-6:
        scale = target_sum / current_sum
        adjusted_h = np.clip(adjusted_h * scale, 0.0, 1.0)
    h_full = np.concatenate([adjusted_h, adjusted_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx

    return adjusted_h, final_c5, n_points
