        except:
            result_de = scipy.optimize.minimize(objective_de, h_start, method='L-BFGS-B', bounds=bounds)
        
        h_de = result_de.x
        sum_de = np.sum(h_de)
        if sum_de > 0:
            h_de = h_de * (sum_half_target / sum_de)
        
        def compute_overlap_slsqp(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_half_target, 'jac': np.ones_like(x)}]
        result_slsqp = scipy.optimize.minimize(
            compute_overlap_slsqp, h_de, method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_half)], constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False, 'eps': 1e-8})
        
        current_c5 = compute_overlap_slsqp(result_slsqp.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
