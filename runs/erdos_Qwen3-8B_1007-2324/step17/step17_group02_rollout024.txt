import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    integral_target = n_points / 2
    dx = 2.0 / n_points

    start_points = []

    # Use the parent solution if available
    if 'initial_h_values' in globals():

        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)

    # Binary initial guess
    binary_initial = np.zeros(n_points)
    num_ones = int(np.round(integral_target))
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_initial[indices] = 1.0
    binary_initial /= np.sum(binary_initial)
    binary_initial *= integral_target
    start_points.append(binary_initial)

    # Random initial guess
    random_initial = np.random.rand(n_points)
    random_initial /= np.sum(random_initial)
    random_initial *= integral_target
    start_points.append(random_initial)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for h_start in start_points:

        def compute_overlap(h):

            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = scipy.optimize.minimize(

            method='SLSQP',
            bounds=bounds,
