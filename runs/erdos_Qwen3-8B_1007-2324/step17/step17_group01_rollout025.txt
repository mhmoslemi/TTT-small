        # Binary initial guess
        binary_initial = np.zeros(n_points // 2)
        num_ones = n_points // 4
        indices = np.random.choice(n_points // 2, num_ones, replace=False)
        binary_initial[indices] = 1
        binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
        start_points.append(binary_full[:n_points])
        
        # Random initial guess
        random_initial = np.random.rand(n_points)
        random_initial = random_initial * (integral_target / np.sum(random_initial))
        start_points.append(random_initial)
        
        for h_start in start_points:
            def compute_overlap(h):
                h_full = h
                corr = np.correlate(h_full, 1 - h_full, mode='full')
                return np.max(corr) * dx
            
            def constraint(x):
                return np.sum(x) - integral_target
            
            result = scipy.optimize.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones(n_points)}],
                options={
                    'ftol': 1e-12,
                    'maxiter': 200,
                    'disp': False,
                    'eps': 1e-8
                }
            )
            
            if result.success:
                current_c5 = compute_overlap(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        
        # Early exit if time is up
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
