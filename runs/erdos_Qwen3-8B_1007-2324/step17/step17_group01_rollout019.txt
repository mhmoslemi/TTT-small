import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = np.zeros(60)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_start = initial_h_values
    else:
        h_start = np.zeros(n_points)
        num_ones = n_points // 4
        idx = np.random.choice(n_points, num_ones, replace=False)
        h_start[idx] = 1.0

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    def optimize_h(h_init):
        result = scipy.optimize.minimize(
            compute_overlap,
            h_init,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones_like}],
            options={
                'ftol': 1e-8,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            return result.x, compute_overlap(result.x)
        return np.zeros(n_points), np.inf

    best_h, best_c5 = optimize_h(h_start)
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        new_h = np.zeros(n_points)
        num_ones = n_points // 4
        shift = n_points // 2
        idx = np.random.choice(n_points, num_ones, replace=False)
        new_h[idx] = 1.0
        new_h[shift:shift+num_ones] = 1.0
        new_h = new_h * (target_sum / np.sum(new_h))
        h_candidate, c5_candidate = optimize_h(new_h)
        if c5_candidate < best_c5:
            best_h, best_c5 = h_candidate, c5_candidate
        new_h = np.zeros(n_points)
        num_ones = np.random.randint(1, n_points // 2)
        idx = np.random.choice(n_points, num_ones, replace=False)
        new_h[idx] = np.random.rand(num_ones)

        h_candidate, c5_candidate = optimize_h(new_h)
        if c5_candidate < best_c5:
            best_h, best_c5 = h_candidate, c5_candidate

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
