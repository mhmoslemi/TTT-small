def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial_full = np.array(initial_h_values)

            h_initial_full = h_initial_full[:n_points]

        start_points.append(variables_initial)
    
    # Binary initial guess
    binary_initial = np.zeros(n_half)
    num_ones = n_half // 2
    indices = np.random.choice(n_half, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)
    
    # Random initial guess
    random_initial = np.random.rand(n_half)
    random_initial = random_initial * (target_sum / np.sum(random_initial))
    start_points.append(random_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_half)
    start_time = time.time()
    
    for variables in start_points:
        current_sum = np.sum(variables)
        if current_sum < 1e-12:
            variables = variables * (target_sum / 1e-12)
        else:
            variables = variables * (target_sum / current_sum)
        
        def objective(vars):
            h_full = np.concatenate([vars, vars[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def constraint(vars):
            return np.sum(vars) - target_sum
        
        result = scipy.optimize.minimize(
            objective,
            variables,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones_like(variables)}],
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = result.fun
