import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h_initial = None
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]

        c5 = np.max(corr) * dx
        penalty = 1e4 * np.abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Check for initial_h_values and use as starting point
    if h_initial is not None:
        start_points = [h_initial]
    else:
        start_points = [np.random.rand(n_points)]

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for _ in range(3):
        result_de = scipy.optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=20,
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.7,
            tol=1e-10,
            polish=True,
            disp=False,
            vectorized=True
        )
        h_de = result_de.x

        def compute_c5(h):

            corr = np.correlate(h_full, 1 - h_full, mode='full')
