import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    # Use the parent solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    
    # Generate structured initial guesses
    # Sine wave with normalization
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.5 * np.cos(np.pi * x)
    h_sine = h_sine / np.sum(h_sine) * integral_target
    start_points.append(h_sine)
    
    # Symmetric binary pattern
    binary = np.zeros(n_points)
    binary[:n_points//2] = 1.0
    start_points.append(binary)
    
    # Random with normalization
    rand = np.random.rand(n_points)
    rand = rand * (integral_target / np.sum(rand))
    start_points.append(np.clip(rand, 0, 1))
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_c5(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1.0 - h_full, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_points)}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = scipy.optimize.minimize(
            compute_c5,
            h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 150}
        )
        
        if result.success:
            current_c5 = compute_c5(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
