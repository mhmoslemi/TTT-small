        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr_index = np.argmax(corr)
        k = max_corr_index
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            term1 = 0.0
            if i + k < len(h_full):
                term1 = (1 - h_full[i + k]) * dx
            term2 = 0.0
            if i - k >= 0:
                term2 = -h_full[i - k] * dx
            gradient[i] = term1 + term2
        return gradient

    constraint = lambda x: np.sum(x) - integral_target
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.minimize(
        compute_overlap,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        jac=compute_gradient,
        options={'ftol': 1e-12, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
    )

    if result.success:
        best_c5 = result.fun
        best_h = result.x.copy()
    else:
        best_c5 = np.inf
        best_h = np.zeros(n_points)

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
