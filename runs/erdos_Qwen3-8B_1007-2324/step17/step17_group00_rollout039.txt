            h_half = np.full(n_half, 1.0 / n_half)
        else:
            h_half = h_half * (target_sum / sum_h)
        initial_points.append(h_half)
        h_reversed = np.flip(h_half)
        h_reversed = h_reversed * (target_sum / np.sum(h_reversed))
        initial_points.append(h_reversed)

    for _ in range(4):
        h_rand = np.random.rand(n_half)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        initial_points.append(h_rand)

    def compute_c5(x):
        h_values = x
        h_full = np.concatenate([h_values, h_values[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective_de(x):
        c5 = compute_c5(x)
        penalty = 1e10 * (np.sum(x) - target_sum) ** 2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    result_de = differential_evolution(
        objective_de,
        bounds,
        maxiter=100,
        popsize=15,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.7,
        tol=1e-10,
        disp=False,
        polya=True,
        seed=seed
    )
    elapsed_time = time.time() - start_time
    time_left = budget_s - elapsed_time

    best_x = result_de.x
    if time_left > 40:
        def objective_lbfgs(x):
            return compute_c5(x)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result_lbfgs = scipy.optimize.minimize(
            objective_lbfgs,
            best_x,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=cons,
            options={
                'ftol': 1e-12,
                'max_iter': 1000,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result_lbfgs.success:
            best_x = result_lbfgs.x

    final_c5 = compute_c5(best_x)
    return np.concatenate([best_x, best_x[::-1]]), final_c5, n_points
