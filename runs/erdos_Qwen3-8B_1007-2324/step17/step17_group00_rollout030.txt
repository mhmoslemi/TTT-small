import numpy as np

import scipy.optimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    def compute_overlap(x):
        h = x.copy()
        h_full = np.zeros(n_points)
        h_full[:n_half] = h
        h_full[n_half:] = h[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        corr = corr[n_points-1:]  # Only consider k in [0, 2]
        max_corr = np.max(corr) * dx
        return max_corr

    def objective(x):
        total = np.sum(x)
        weight_overlap = 0.9
        weight_integral = 0.1
        penalty = 1e6 * (total - integral_target)**2
        overlap_val = compute_overlap(x)
        penalized = weight_overlap * overlap_val + weight_integral * penalty
        return penalized

    initial_parents = []
    if 'initial_h_values' in globals():
        parent = initial_h_values
        parent = parent[:n_points]
        parent = (parent - np.min(parent)) / (np.max(parent) + 1e-10)
        initial_parents.append(parent)

    initial_parents += [
        np.random.rand(n_points) * 0.5 + 0.25,
        np.sin(np.linspace(0, 10, n_points) * np.pi) + 0.5,
        np.random.rand(n_points) * 0.5 + 0.5
    ]

    best_sol = None
    best_score = float('inf')

    for attempt in range(3):
        bounds = [(0,1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin',

        if result.success:

            score = compute_overlap(candidate)
            if score < best_score:

    best_h = best_sol if best_sol is not None else np.random.rand(n_points)
    best_h = (best_h - np.min(best_h)) / (np.max(best_h) + 1e-10)
    final_h = best_h[:n_half] if n_points % 2 == 0 else best_h[:n_half + 1]
