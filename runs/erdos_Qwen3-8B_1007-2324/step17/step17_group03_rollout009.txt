import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        elif len(h_initial) < n_points:
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
        start_points.append(h_initial)
    
    # Binary initial guess
    binary_initial = np.zeros(n_points)
    num_ones = int(integral_target)
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)
    
    # Random initial guess
    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    start_points.append(random_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-10,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
