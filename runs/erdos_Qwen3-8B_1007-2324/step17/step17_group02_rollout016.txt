def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        start_points.append(np.array(initial_h_values))

        angles = np.linspace(0, 2 * np.pi, n_points)
        sine = 0.5 + 0.5 * np.sin(freq * angles)
        sine *= integral_target / np.sum(sine)
        return sine
    
    start_points.append(generate_sine_start())
    
    def generate_square_start():
        square = np.zeros(n_points)
        for i in range(n_points // 4):
            square[i::(n_points // 4)] = 1
        square *= integral_target / np.sum(square)
        return square
    
    start_points.append(generate_square_start())
    
    def generate_random_start():
        return np.random.rand(n_points) * (integral_target / np.sum(np.random.rand(n_points)))
    
    start_points.append(generate_random_start())
    
    best_c5 = float('inf')
