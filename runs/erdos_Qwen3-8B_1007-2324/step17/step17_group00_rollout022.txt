import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)

    # Binary initial guess with spaced 1s
    binary_initial = np.zeros(n_half)
    num_ones = n_half // 4
    indices = np.random.choice(n_half, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    start_points.append(binary_full[:n_points])

    # Random initial guess
    random_initial = np.random.rand(n_half)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    start_points.append(random_full[:n_points])

    # Structured initial guess with alternating regions
    structured_initial = np.zeros(n_half)
    segments = n_half // 8
    for i in range(4):
        structured_initial[i*segments:(i+1)*segments] = 1.0 if i % 2 == 0 else 0.0
    structured_full = np.concatenate([structured_initial, structured_initial[::-1]])
    start_points.append(structured_full[:n_points])

    # Spike initial guess with random positions
    spike_initial = np.zeros(n_half)
    spike_count = n_half // 5
    spike_indices = np.random.choice(n_half, spike_count, replace=False)
    spike_initial[spike_indices] = 1.0
    spike_full = np.concatenate([spike_initial, spike_initial[::-1]])
    start_points.append(spike_full[:n_points])

    best_c5 = np.inf
    best_h = np.zeros(n_half)

    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start[:n_half],
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-9,
                'maxiter': 500,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    return best_h_full, final_c5, n_points
