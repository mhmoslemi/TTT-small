import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum of h_values for [0,1]

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial[:n_half])

    # Generate random feasible points
    for _ in range(4):
        rand = np.random.rand(n_half)
        rand = rand * (integral_target / np.sum(rand))
        start_points.append(rand)

    best_c5 = np.inf
    best_h = np.zeros(n_half)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        max_over = compute_overlap(x)
        penalty = 1e10 * (np.sum(x) - integral_target)**2
        return max_over + penalty

    start_time = time.time()
    result_de = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        max_evals=40000,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    elapsed = time.time() - start_time

    if elapsed > budget_s * 0.6:
        best_h = result_de.x

    final_c5 = compute_overlap(best_h)
