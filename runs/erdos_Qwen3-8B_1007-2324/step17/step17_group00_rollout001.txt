import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_half  # sum(h_values) * dx = 1 => sum(h_values) = 15

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_half:
            h_initial = h_initial[:n_half]
        start_points.append(h_initial)
        start_points.append(np.flip(h_initial))

    # Generate random feasible initial points
    for _ in range(4):
        r = np.random.rand(n_half)
        r = r * (integral_target / np.sum(r))
        start_points.append(r)

    best_c5 = np.inf
    best_h = np.zeros(n_half)

    # Time budget allocation
    t0 = time.time()
    remaining_time = budget_s * 0.666

    # Differential Evolution global optimization
    def objective(x):
        x = np.clip(x, 0, 1)
        h = x
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e10 * (np.sum(x) - integral_target)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    try:
        result = scipy.optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-6,
            maxiter=1000,
            seed=seed,
            polish=False,
            disp=False,
            args=(),
            constraints=(),
            vectorized=True,
            workers=-1,
            x0=start_points[0]
        )
        best_h = np.clip(result.x, 0, 1)
        best_c5 = objective(best_h)
    except:
        pass

    # Local polishing with L-BFGS-B
    if time.time() - t0 < budget_s * 0.666:
        try:
            result = scipy.optimize.minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0) for _ in range(n_half)],
                tol=1e-6,
                options={'maxiter': 1000, 'ftol': 1e-12, 'eps': 1e-8}

            best_h = np.clip(result.x, 0, 1)
            best_c5 = objective(best_h)
        except:
            pass

    # Final evaluation
    h_full = np.concatenate([best_h, best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx
    return best_h, final_c5, n_points
