    best_h = np.zeros(n_half)
    
    start_time = time.time()
    result_de = differential_evolution(
        compute_overlap,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        seed=seed + 1000
    )
    time_used = time.time() - start_time
    remaining_time = budget_s - time_used
    
    best_h = result_de.x
    best_c5 = compute_overlap(best_h)
    
    if remaining_time > 0:
        try:
            result_slqp = minimize(
                compute_overlap,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
                options={'ftol': 1e-8, 'maxiter': 50}
            )
            if result_slqp.success:
                best_c5 = result_slqp.fun
                best_h = result_slqp.x
        except:
            pass
    
    for h_start in initial_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))
        current_c5 = compute_overlap(h_start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_start
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
