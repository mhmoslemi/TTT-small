def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    n_half = n_points // 2

        random = random * (integral_target / np.sum(random))
        start_points.append(random)

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e12 * (np.sum(x) - integral_target) ** 2
        return c5 + penalty
