import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h_clip)
        penalty = 1e4 * abs(sum_h - integral_target)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    try:
        while time.time() - start_time < budget_s - 2:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=20,
                tol=1e-6,
                maxiter=100,
                mutation=(0.5, 1.0),
                recombination=0.8,
                seed=seed
            )
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    except:
        pass

    return best_h, best_c5, n_points
