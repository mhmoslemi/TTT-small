import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = initial_h * (integral_target / total)
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        penalty = 1e5 * (sum_h - integral_target)**2
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Differential evolution for global search
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed
    )
    best_h = result_de.x
    best_c5 = objective(best_h)

    # SLSQP refinement
    def objective_slsqp(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    # Final refinement with perturbations
    shifts = [1, 2, 5, 10, 140, -1, -2, -5, -10, -140]
    for shift in shifts:
        if time.time() - start_time > budget_s - 1:

        if sum_start > 0:

                method='SLSQP',

                constraints=constraints,
                options={'ftol': 1e-9, 'maxiter': 50}
            )
