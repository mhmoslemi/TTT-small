import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 320  # Larger grid for better resolution
    dx = 2.0 / n_points
    integral_target = n_points // 2  # Ensure exact integer division

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair_integral(h):
        current_sum = np.sum(h)
        needed = integral_target - current_sum
        if abs(needed) < 1e-5:
            return h.copy()
        weights = np.abs(h - h.mean()) + 1e-10
        cum_weights = weights / weights.sum()
        indices = np.random.choice(n_points, size=20, p=cum_weights, replace=False)
        h = h.copy()
        for idx in indices:
            if abs(needed) < 1e-5:
                break
            max_val = min(1.0, h[idx] + 0.1)
            min_val = max(0.0, h[idx] - 0.1)
            if max_val == min_val:
                h[idx] += needed / len(indices)
                break
            else:
                space_size = max_val - min_val
                dl = min(needed, space_size)
                h[idx] = h[idx] + dl / space_size
                needed -= dl
        return np.clip(h, 0, 1)

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = repair_integral(initial_h + perturb)
        population.append(new_h)

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    best_idx = 0

    for gen in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Tournament selection
        tournament = np.random.choice(len(population), size=10)
        selected = population[np.argmax([evaluate(p) for p in tournament])]
        # Crossover
        child = np.copy(selected)
        if np.random.rand() < 0.9:
            idx1, idx2 = np.random.choice(len(population), size=2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            crossover = np.random.rand(n_points) < 0.5
            child = np.where(crossover, parent1, parent2)
        # Mutation
        mutate = np.random.rand(n_points) < 0.1
        child[mutate] += np.random.normal(0, 0.1, np.sum(mutate))
        child = repair_integral(child)
        # Evaluate and select
        child_c5 = evaluate(child)
        if child_c5 < best_c5:
            best_h = child.copy()
            best_c5 = child_c5
            best_idx = len(population)
        population.append(child)
        if len(population) > population_size:
            population.pop(np.argmin([evaluate(p) for p in population]))

    # Local search refinement
    def local_search(h):
        result = minimize(evaluate, h, method='Nelder-Mead', bounds=[(0,1)]*n_points,
                          options={'xatol': 1e-5, 'maxiter': 50})
        return result.x

    best_h = local_search(best_h)
    best_c5 = evaluate(best_h)
    return best_h, best_c5, n_points
