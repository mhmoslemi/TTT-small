            noise = np.random.normal(0, 0.02, 140)
            perturbed_h = best_h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            sum_perturbed = np.sum(perturbed_h)
            if sum_perturbed > 0:
                perturbed_h *= (70.0 / sum_perturbed)
            result = minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 30, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()

    full_h = np.concatenate([best_h, best_h[::-1]])
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return full_h, best_c5, n_points
    return full_h, best_c5, n_points
