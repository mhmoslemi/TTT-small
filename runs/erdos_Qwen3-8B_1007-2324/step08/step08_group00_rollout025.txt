import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_half = initial_h[:n_half]
        initial_h_full = np.concatenate([h_half, h_half[::-1]])
    else:
        x = np.linspace(0, 2, n_points)
        distance = np.abs(x - 1)
        h = 1 - distance
        h = np.clip(h, 0, 1)
        h_half = h[:n_half]
        initial_h_full = np.concatenate([h_half, h_half[::-1]])

    h_opt_initial = initial_h_full[:n_half]

    def objective(h_opt):
        h_array = np.concatenate([h_opt, h_opt[::-1]])
        h_clip = np.clip(h_array, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{'type': 'eq', 'fun': lambda h_opt: np.sum(h_opt) - integral_target}]

    best_h_opt = h_opt_initial.copy()
    best_c5 = objective(best_h_opt)

    trials = 50
    max_iter = 100

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_opt_start = best_h_opt.copy()
        noise = np.random.normal(0, 0.03, n_half)
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        sum_start = np.sum(h_opt_start)
        if sum_start > 0:
            h_opt_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_opt_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h_opt = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points

    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
