import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair(h):
        h = h.copy().astype(float)
        s = np.sum(h)
        if abs(s - integral_target) < 1e-5:
            return h
        delta = (integral_target - s) / n_points
        h += delta
        return np.clip(h, 0, 1)

    def perturb(h, temp=0.3):
        h = h.copy()
        idxs = np.random.choice(n_points, size=20, replace=False)
        noise = np.random.normal(0, 0.2, n_points)
        h[idxs] += noise[idxs] * (0.5 - (h[idxs] - 0.5) / 0.5)
        return repair(h)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h
    best_c5 = objective(initial_h)

    starts = [initial_h]
    for _ in range(2):
        starts.append(perturb(initial_h, temp=0.1 + _ * 0.1))

    for start in starts:
        if time.time() - start_time > budget_s - 2:
            break
        try:
            result = minimize(
                objective,
                start,
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    return best_h, best_c5, n_points
