import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e5 * abs(sum_h - integral_target)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use initial guess to seed the population
    initial_pop = [initial_h] + [np.random.uniform(0, 1, n_points) for _ in range(10)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        seed=seed
    )

    best_h = result.x
    best_h_clip = np.clip(best_h, 0, 1)
    corr = np.correlate(best_h_clip, 1 - best_h_clip, mode='full')

        best_h[0] += needed
        best_h = np.clip(best_h, 0, 1)
