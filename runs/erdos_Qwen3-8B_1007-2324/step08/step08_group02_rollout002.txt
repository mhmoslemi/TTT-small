import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target = n_points / 2
    field = np.linspace(0, 2, n_points)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points) * 0.5
        initial_h = initial_h * (target / np.sum(initial_h))

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        deficit = target - np.sum(h_clip)
        scale = 1.0
        if np.sum(h_clip) > 0:
            scale = 1.0 + deficit / np.sum(h_clip)
        h_scaled = h_clip * scale
        h_final = np.clip(h_scaled, 0, 1)
        if np.sum(h_final) > 0:
            h_final = h_final * (target / np.sum(h_final))
        return h_final

    def objective(h):
        h_feas = repair(h)
