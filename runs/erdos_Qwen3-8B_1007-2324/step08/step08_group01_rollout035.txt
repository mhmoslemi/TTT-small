import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_scaled = h.copy()
        sum_clipped = np.sum(h_scaled)
        if sum_clipped == 0:
            return 1e6
        scale_factor = integral_target / sum_clipped
        h_scaled *= scale_factor
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Custom differential evolution setup
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        vectorized=True,
        x0=initial_h
    )

    best_h = result.x
    best_c5 = result.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
