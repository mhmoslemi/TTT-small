import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points) * 0.5
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (target / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0, 1)

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if total == 0:
            return h_clip
        scale = target / total
        return h_clip * scale

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target}]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    num_starts = 20
    max_iter = 100
    for _ in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
        pert = np.random.normal(0, 0.05, n_points)
        h_start = initial_h + pert
        h_start = np.clip(h_start, 0, 1)
        h_start = repair(h_start)

        try:
            result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
