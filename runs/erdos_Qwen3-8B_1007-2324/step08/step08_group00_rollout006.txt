from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

        distance = np.abs(x - 1)
        h = 1 - distance
        h = np.clip(h, 0, 1)
        h_half = h[:n_half]
        initial_h = np.concatenate([h_half, h_half[::-1]])

    h_opt_initial = initial_h[:n_half]
    h_opt_initial = np.clip(h_opt_initial, 0, 1)
    sum_opt = np.sum(h_opt_initial)
    h_opt_initial *= (integral_target / sum_opt) if sum_opt > 0 else 1.0

    def objective(h_opt):
        h_array = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

        try:
            result = minimize(
