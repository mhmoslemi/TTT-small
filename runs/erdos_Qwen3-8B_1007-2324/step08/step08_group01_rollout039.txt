import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200  # Reduced from 280 for better exploration
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(best_h)

    # Adaptive exploration with local refinement
    num_restarts = 50
    for _ in range(num_restarts):
        if time.time() - start_time > budget_s - 10:
            break
        noise = np.random.normal(0, 0.05, n_points)
        perturb_h = best_h + noise
        perturb_h = np.clip(perturb_h, 0, 1)
        perturb_sum = np.sum(perturb_h)
        if perturb_sum > 0:
            perturb_h *= (target_sum / perturb_sum)
        # Optimize from perturbed solution
        result = minimize(
            evaluate_erdos,
            perturb_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
