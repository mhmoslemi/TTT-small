import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = initial_h * (integral_target / total)
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    max_iter = 300
    trials = 150
    shift_range = 10

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)

        h_start = np.roll(initial_h, shift)
        noise = np.random.normal(0, 0.02, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    if time.time() - start_time < budget_s - 10:
