from scipy.optimize import minimize
import time

    integral = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)

    def calc_c5(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = calc_c5(best_h)

    def optimize_perturbations(h0):
        for _ in range(5):
            noise = np.random.normal(0, 0.03, n_points)
            new_h = h0 + noise
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h *= integral / sum_new
            res = minimize(calc_c5, new_h, method='SLSQP',
                           bounds=[(0,1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral}],
