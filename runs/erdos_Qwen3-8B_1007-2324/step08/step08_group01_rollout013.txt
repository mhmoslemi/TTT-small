import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    shifts = [1.0, -1.0, 0.5, -0.5, 2.0, -2.0, 0.25, -0.25, dx, -dx, dx*2, -dx*2, 0.0]
    num_shifts = len(shifts)
    shift_positions = [round(s / dx) for s in shifts]
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        for shift_idx in range(num_shifts):
            shift = shifts[shift_idx]
            shift_pos = shift_positions[shift_idx]
            shifted_h = np.roll(initial_h, shift_pos)
            noise = np.random.normal(0, 0.05, n_points)
            shifted_h += noise
            shifted_h = np.clip(shifted_h, 0, 1)
            sum_shifted = np.sum(shifted_h)
            if sum_shifted > 0:
                shifted_h *= (integral_target / sum_shifted)
            try:
                result = minimize(
                    evaluate_erdos,
                    shifted_h,
                    method='SLSQP',
                    bounds=[(0.0, 1.0) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],
                    options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass
    
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
