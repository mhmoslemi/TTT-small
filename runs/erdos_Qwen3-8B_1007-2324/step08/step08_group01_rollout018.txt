import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 128

    integral_target = n_points / 2

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(best_h)
    current_c5 = best_c5

    max_iter = int(budget_s * 0.6)

        np.random.seed(seed + _)  # Ensure reproducibility
        noise = np.random.uniform(-0.05, 0.05, n_points)

        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed > 0:

        else:
            perturbed_h = np.full(n_points, integral_target / n_points)
        # Local optimization

                perturbed_h,
