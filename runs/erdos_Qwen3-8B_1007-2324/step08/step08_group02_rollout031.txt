import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    initial_h = np.clip(initial_h, 0, 1)
    initial_sum = np.sum(initial_h)
    if initial_sum > 0:
        initial_h = initial_h * (integral_target / initial_sum)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return float('inf')
        h_scaled = h_clip * (integral_target / sum_clip)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    result = None
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            x0=initial_h
        )
    except:
        pass

    best_h = initial_h
    best_c5 = objective(best_h)
    if result is not None:
        if result.success:
            best_h = result.x
            best_c5 = result.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
