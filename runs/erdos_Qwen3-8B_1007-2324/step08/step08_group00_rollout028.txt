import numpy as np
from scipy.optimize import dual_annealing, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points // 2  # sum(h) = 140 for n_points=280

    # Initialize with provided or generated symmetric function
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        distance = np.abs(x - 1)
        h = 1 - distance
        h = np.clip(h, 0, 1)
        initial_h = h[:n_half]

    # Normalize initial guess to meet integral constraint
    initial_sum = np.sum(initial_h)
    scale_factor = integral_target / (initial_sum if initial_sum != 0 else 1)
    initial_h = np.clip(initial_h * scale_factor, 0, 1)

    # Objective with penalty for integral constraint
    def objective_func(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5_val = np.max(corr) * dx
        penalty = 1e5 * np.abs(np.sum(h_vec) - integral_target)
        return c5_val + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    best_score = float('inf')
    best_vec = np.zeros(n_half)

    # Conduct multi-start optimization within time budget
    n_trials = min(15, int((budget_s - 5) / 6))  # 6 seconds per trial
    for i in range(n_trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + i)
        # Generate starting point with random noise
        x0 = initial_h * (1 + 0.05 * np.random.randn(n_half))
        x0 = np.clip(x0, 0, 1)
        # Regulate sum to meet integral_target
        sum0 = np.sum(x0)
        if abs(sum0 - integral_target) > 20:
            continue
        x0 *= integral_target / (sum0 + 1e-7)
        x0 = np.clip(x0, 0, 1)

        # Global search with dual_annealing
        res_g = dual_annealing(objective_func, bounds=bounds, x0=x0, max_evals=3000, seed=seed + i)
        if res_g.success:
            # Local refinement with L-BFGS-B
            res_l = minimize(objective_func, res_g.x, method='L-BFGS-B', bounds=bounds, options={'maxiter': 500})

            current_score = objective_func(x0)
            current_vec = x0

        if current_score < best_score:
            best_score = current_score
            best_vec = current_vec

    h_values = np.concatenate([best_vec, best_vec[::-1]])
