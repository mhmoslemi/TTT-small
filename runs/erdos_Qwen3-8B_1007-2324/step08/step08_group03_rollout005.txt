import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair_integral(h):
        current_sum = np.sum(h)
        needed = integral_target - current_sum
        if abs(needed) < 1e-5:
            return h.copy()
        h = h.copy()
        for i in range(n_points):
            if abs(needed) < 1e-5:
                break
            delta = min(1.0 - h[i], max(0.0, h[i] - 0.0), abs(needed))
            h[i] += delta
            needed -= delta
        return np.clip(h, 0, 1)

    def perturb_solution(base_h, temp=0.3):
        h = base_h.copy()
        idxs = np.random.choice(n_points, size=40, replace=False)
        noise = np.random.normal(0, 0.15 * np.std(h), size=n_points)
        h[idxs] += noise[idxs] * (0.5 - (h[idxs] - 0.5) / 0.5)
        return repair_integral(h)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    starts = [initial_h]
    for _ in range(4):
        starts.append(perturb_solution(initial_h, temp=0.3 + 0.05 * _))
    starts.append(perturb_solution(initial_h, temp=0.2))

    for start in starts:
        if time.time() - start_time > budget_s - 2:
            break
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    return best_h, best_c5, n_points
