import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    current_h = initial_h.copy()
    current_c5 = evaluate_erdos(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Parameters for SA
    T = 50.0
    cooling_rate = 0.995
    min_T = 0.1
    perturb_scale_low = 0.03
    perturb_scale_high = 0.15
    n_steps_per_decay = 50

    while T > min_T and time.time() - start_time < budget_s - 10:
        for _ in range(n_steps_per_decay):
            perturb = np.random.uniform(-perturb_scale_high * T, perturb_scale_high * T, n_points)
            new_h = current_h + perturb
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h *= (integral_target / sum_new)
            else:
                new_h = np.full(n_points, integral_target / n_points)
            new_c5 = evaluate_erdos(new_h)
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                current_h = new_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        T *= cooling_rate

    # Local refinement with SLSQP
    if time.time() - start_time < budget_s - 10:
        result = minimize(
            evaluate_erdos,
            best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
