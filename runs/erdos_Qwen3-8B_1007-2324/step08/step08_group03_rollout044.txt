from scipy.optimize import differential_evolution

    start_time = time.time()
    n_points = 128  # Reduced for faster execution

        h = np.full(n_points, 0.5)
    
    # Repair initial to meet sum constraint
    def repair(h_vec):
        s = np.sum(h_vec)
        if abs(s - target_sum) < 1e-5:
            return h_vec
        factor = target_sum / s
        h = h_vec * factor
        h = np.clip(h, 0, 1)
        # Final adjustment to hit target_sum
        diff = target_sum - np.sum(h)
        while abs(diff) > 1e-5:
            idx = np.argmax(h)
            if diff > 0:
                inc = min(1 - h[idx], diff)
                h[idx] += inc
                diff -= inc

                h[idx] -= dec
                diff += dec
