import numpy as np
from scipy.optimize import dual_annealing, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    best_score = None
    best_vec = None

    if 'initial_h_values' in globals():
        h_opt_initial = np.array(initial_h_values)[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        distance = np.abs(x - 1)
        h = 1 - distance
        h = np.clip(h, 0, 1)
        h_opt_initial = h[:n_half]

    def objective_func(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5_val = np.max(corr) * dx
        penalty = 1e6 * np.abs(np.sum(h_vec) - integral_target)
        return c5_val + penalty

    bounds = [(0.0, 1.0)] * n_half

    n_trials = min(30, int((budget_s - 5) / 10))
    for i in range(n_trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + i)
        x0 = h_opt_initial * (1 + 0.05 * np.random.randn(n_half))
        x0 = np.clip(x0, 0, 1)
        sum0 = np.sum(x0)
        if abs(sum0 - integral_target) > 15:
            continue
        x0 *= min(max(sum0 / integral_target, 0.95), 1.05)
        x0 = np.clip(x0, 0, 1)
        x0 *= integral_target / (np.sum(x0) + 1e-7)

        res_g = dual_annealing(objective_func, bounds=bounds, x0=x0, max_evals=10000, gamma=0.995, seed=seed + i)
        if res_g.success and res_g.fun < 1e7:
            res_l = minimize(objective_func, res_g.x, method='L-BFGS-B', bounds=bounds, options={'maxiter': 5000})
            cand_score, cand_vec = res_l.fun, res_l.x
        else:

        if best_score is None or cand_score < best_score:
            best_score, best_vec = cand_score, cand_vec

    h_values = np.concatenate([best_vec, best_vec[::-1]])
