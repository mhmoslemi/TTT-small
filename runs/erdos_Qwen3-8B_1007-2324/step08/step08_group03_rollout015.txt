import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2  # 140

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair_integral(h):
        current_sum = np.sum(h)
        needed = integral_target - current_sum
        if abs(needed) < 1e-5:
            return h.copy()
        h = h.copy()
        idxs = np.random.choice(n_points, size=10, replace=False)
        for idx in idxs:
            if abs(needed) < 1e-5:
                break
            max_val = min(1.0, h[idx] + 0.1)
            min_val = max(0.0, h[idx] - 0.1)
            if max_val == min_val:
                scale = 1.0 / len(idxs)
                h[idx] += needed * scale
                break
            else:
                space_size = max_val - min_val
                dl = min(needed, space_size)
                h[idx] = h[idx] + dl / space_size
                needed -= dl
        return np.clip(h, 0, 1)

        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')

    result = differential_evolution(
