   ```python
           n_half = n_points // 2
           h_initial_full = init_from_parent(np.array(initial_h_values), n_half)
       else:
           n_half = n_points // 2
           h_initial_full = np.ones(n_points) * (integral_target / n_points)
       
       def build_symmetric_h(h_vec):
           return np.concatenate([h_vec[:n_half], h_vec[:n_half][::-1]])
       
       def objective(h_vec):
           h_full = build_symmetric_h(h_vec[:])
           # Ensure valid range and integral
           h_full = np.clip(h_full, 0, 1)
           corr = np.correlate(h_full, 1 - h_full, mode='full')
           max_corr_val = np.max(corr)
           return max_corr_val * dx, h_full
       
       def grad_objective(h_vec):
           # Exact analytical gradient
           h_full = build_symmetric_h(h_vec[:])
           correlations = np.correlate(h_full, 1 - h_full, mode='full')
           max_idx = np.argmax(correlations)
           # Discrete derivative calculation
           grad = np.zeros_like(h_vec)
           # For non-symmetric targets (just one sample for simplicity)
           # Partial derivative w.r.t each component
           for j in range(len(h_vec)):
               pert = np.zeros(len(h_vec))
               pert[j] = 1e-4
               h_pert = build_symmetric_h(h_vec + pert)
               corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
               forward = np.max(corr_pert)
               h_pert = build_symmetric_h(h_vec - pert)
               corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
               backward = np.max(corr_pert)
               grad[j] = (forward - backward) / (2e-4)
           return grad * dx
       
       bounds = [(0,1)] * n_half
       
       def constraint(x):
           sum_x = np.sum(build_symmetric_h(x))
           return sum_x - integral_target
       
       # Global phase (random exploration)
       best_val = float('inf')
       best_vec = None
       for i in range(10):  # 10 global trials
           if time.time() - start_time > budget_s:
               break
           # Random walk
           noise = np.random.randn(len(bounds)) * 0.01
           candidate = np.clip(np.array(h_initial_full) + noise, 0, 1)
           # Ensure integral constraint
           candidate_vec = candidate[:n_half]
           sum_val = constraint(candidate_vec.argsort()[:5])  # select positions to adjust
           scale = integral_target / sum_val if abs(sum_val) > 1e-5 else 1
           if abs(scale-1) > 0.01:
               candidate_vec *= scale
               candidate_vec = np.clip(candidate_vec, 0, 0.1 if scale>1 else 0.9, 1)
           cost = objective(candidate_vec)[0]
           if cost < best_val:
               best_val = cost
               best_vec = candidate_vec.copy()
       
       # Local refinement
       if best_vec is not None:
           res = minimize(
               objective,
               best_vec,
               method='L-BFGS-B',
               bounds=bounds,
               constraints={'type': 'eq', 'fun': constraint},
               jac=grad_objective,
               options={'maxiter': int(2e3), 'disp': False, 'eps': 1e-2}
           )
           if res.success and res.fun < best_val:
               best_val = res.fun
               best_vec = res.x.copy()
       
       h_values = build_symmetric_h(best_vec)
       return h_values, best_val, n_points
   ```
