   For a shift \(k\), the directional derivative of the \(k\)-th autocorrelation term is:
   \[
       \frac{\partial}{\partial h[j]} \left( \sum_i h[i] \cdot (1 - h[i+k]) \Delta x \right) = (1 - h[j+k]) \Delta x.
   \]

   **Surrogate Approach**: Penalize deviations from \(\sum h[i] = 140\) using:
   \[
       \text{Total objective} = \max_k \text{Correlation}[k] \cdot \Delta x + \lambda \left| \sum h[i] - 140 \right|
   \]  
   where \(\lambda = 10^6\) ensures constraint satisfaction.

   ```python
   from scipy.optimize import dual_annealing, minimize
   import time

   def run(seed=42, budget_s=100, **kwargs):
       start_time = time.time()
       n_points = 280
       n_half = n_points // 2
       dx = 2.0 / n_points  # ~0.007142857
       integral_target = 0.5 / dx # 70.0

       # Load/construct initial guess
       if 'initial_h_values' in globals():
           h_opt_initial = np.array(initial_h_values)[:n_half] * (n_half / 70.0) 
           # Adjust length to n_half and normalize to satisfy integral constraint
       else:
           x = np.linspace(0, 2, n_points)
           distance = np.abs(x - 1)
           h = 1 - distance
           h = np.clip(h, 0, 1)
           h_opt_initial = h[:n_half]

       def objective_func(h_vec):
           # Build full function and compute correlation
           h_full = np.concatenate([h_vec, h_vec[::-1]])
           corr = np.correlate(h_full, 1-h_full, mode='full')
           c5_val = np.max(corr) * dx
           penalty = 1e6 * np.abs(np.sum(h_vec) - integral_target)
           return c5_val + penalty

       bounds = [(0.0, 1.0)] * n_half
       ```

   ```python
       best_vec = None

       # Multi-start with dual_annealing
       n_trials = min(30, int((budget_s - 5) / 10))  # 10 seconds per trial
       for i in range(n_trials):
           if time.time() - start_time > budget_s - 1: break
           np.random.seed(seed + i)
           
           # Sample feasible starting points near the target
           x0 = h_opt_initial * (1 + 0.05 * np.random.randn(n_half))
           x0 = np.clip(x0, 0, 1)
           # Regulate sum: 
           sum0 = np.sum(x0)
           if abs(sum0 - integral_target) > 15: continue
           x0 *= min(max(sum0 / integral_target, 0.95), 1.05)
           x0 = np.clip(x0, 0, 1)
           x0 *= integral_target / (np.sum(x0) + 1e-7)
           
           # Global optimization with dual_annealing
           res_g = dual_annealing(objective_func, bounds=bounds, args=(),
                                   max_evals=100000, 
                                   x0=x0,
                                   gamma=0.995,
                                   seed=seed + i)
           
           if res_g.success and res_g.fun < 1e7:
               # Local refinement with gradient descent
               method='L-BFGS-B'
               res_l = minimize(objective_func, res_g.x, method=method, 
                                 bounds=bounds, options={'maxiter': 5000})
               
               cand_score = res_l.fun
               cand_vec = res_l.x
           else:
               cand_score, cand_vec = objective_func(x0), x0

           # Update best candidate
           if best_score is None or cand_score < best_score:
               best_score = cand_score
               best_vec = cand_vec
       ```
