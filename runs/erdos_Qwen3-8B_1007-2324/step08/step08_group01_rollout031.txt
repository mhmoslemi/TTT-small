import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    guesses = [initial_h.copy()]
    np.random.seed(seed)
    
    for _ in range(4):
        perturbation = np.random.normal(0, 0.1, n_points)
        guess = initial_h + perturbation
        guess = np.clip(guess, 0, 1)
        sum_guess = np.sum(guess)
        if sum_guess > 0:
            guess *= integral_target / sum_guess
        guesses.append(guess)
    
    binary_pattern = np.zeros(n_points)
    binary_pattern[:int(n_points / 2)] = 1
    np.random.shuffle(binary_pattern)
    guesses.append(binary_pattern)
    
    sine_pattern = 0.5 * (1 + np.sin(np.linspace(0, 2*np.pi, n_points)))
    sine_pattern = sine_pattern / np.sum(sine_pattern) * integral_target
    guesses.append(sine_pattern)

    for guess in guesses:
        h_initial = guess.copy()
        result = minimize(
            evaluate_erdos,
            h_initial,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s - 10:
            break

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
