    if 'initial_h_values' in globals():

    def objective_func(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5_val = np.max(corr) * dx
        penalty = 1e6 * np.abs(np.sum(h_vec) - integral_target)
        return c5_val + penalty

    bounds = [(0.0, 1.0)] * n_half
    best_score = float('inf')
    best_vec = h_opt_initial.copy()

    n_trials = min(30, int((budget_s - 5) / 10))
    for i in range(n_trials):
        if time.time() - start_time > budget_s - 1:
