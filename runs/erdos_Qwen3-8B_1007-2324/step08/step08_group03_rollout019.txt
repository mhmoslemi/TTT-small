def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()

        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    needed = integral_target - sum_initial
    indices = np.random.choice(n_points, size=30, replace=False)
    for idx in indices:
        if abs(needed) < 1e-5:
            break
        current_val = initial_h[idx]
        max_add = min(1.0 - current_val, needed)
        max_sub = max(0.0, current_val)
        if max_add > 0:
            delta = max_add
        elif max_sub > 0:
            delta = -max_sub
        else:
            delta = needed / len(indices)
        initial_h[idx] += delta
        needed -= delta
    initial_h = np.clip(initial_h, 0, 1)

    def repair(h):
        current_sum = np.sum(h)
        needed = integral_target - current_sum
        if abs(needed) < 1e-5:

        for idx in indices:
            if abs(needed) < 1e-5:
                break
            current_val = h[idx]
            max_add = min(1.0 - current_val, needed)
            max_sub = max(0.0, current_val)
            if max_add > 0:
                delta = max_add
            elif max_sub > 0:
                delta = -max_sub
            else:
                delta = needed / len(indices)

    def objective(h):
        h_repaired = repair(h.copy())
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    population = [initial_h.copy() for _ in range(50)]
    best_h = initial_h.copy()
