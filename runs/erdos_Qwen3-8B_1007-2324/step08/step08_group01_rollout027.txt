from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 128  # Smaller n_points for faster evaluation
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (integral_target / np.sum(initial_h) if np.sum(initial_h) > 0 else 1.0)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    # Use more diverse perturbation strategies
    shifts = [1.0, -1.0, 0.5, -0.5, 0.25, -0.25, 0.0]
    for shift in shifts:
        shift_positions = round(shift / dx)
        shifted_h = np.roll(initial_h, shift_positions)
        noise = np.random.normal(0, 0.03, n_points)
        shifted_h += noise
        shifted_h = np.clip(shifted_h, 0, 1)
        sum_shifted = np.sum(shifted_h)
        if sum_shifted > 0:
            shifted_h *= (integral_target / sum_shifted)
        try:
            result = minimize(
                evaluate_erdos,
                shifted_h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],
                options={'maxiter': 300, 'ftol': 1e-10, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        if time.time() - start_time > budget_s - 10:
            break
