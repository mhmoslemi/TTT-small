import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fft import rfft, irfft
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (target / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        h_fft = rfft(h_clip)
        one_minus_h_fft = rfft(1 - h_clip)
        cross_fft = h_fft * np.conj(one_minus_h_fft)
        cross_corr = irfft(cross_fft)
        max_corr = np.max(cross_corr)
        c5 = max_corr * dx
        penalty = 1e5 * (sum_h - target)**2 / target**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Differential Evolution
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, maxiter=50, tol=1e-5,
        mutation=(0.5, 1), recombination=0.8, seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    # SLSQP Refinement
    def objective_slsqp(h):
        h_clip = np.clip(h, 0, 1)
        h_fft = rfft(h_clip)
        one_minus_h_fft = rfft(1 - h_clip)
        cross_fft = h_fft * np.conj(one_minus_h_fft)
        cross_corr = irfft(cross_fft)
        return np.max(cross_corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target}]
    result_slsqp = minimize(
        objective_slsqp, best_h, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
