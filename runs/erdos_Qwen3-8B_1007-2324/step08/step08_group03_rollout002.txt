            return h

                h[i] = min(1.0, h[i] + diff)
                diff = 0
            elif h[i] > 0:
                h[i] = max(0.0, h[i] + diff)
                diff = 0
        return np.clip(h, 0, 1)
    
    def eval_overlap(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def crossover(a, b):

        b = b.copy()
