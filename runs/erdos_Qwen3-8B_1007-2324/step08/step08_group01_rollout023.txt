import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initial guess
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate variations
    variations = [initial_h.copy()]

    # Add noise variation
    noise = np.random.normal(0, 0.02, n_points)
    h_noise = initial_h + noise
    h_noise = np.clip(h_noise, 0, 1)
    sum_h = np.sum(h_noise)
    if sum_h > 0:
        h_noise *= (integral_target / sum_h)
    variations.append(h_noise)

    # Add blocky variation
    block_h = np.zeros(n_points)
    for i in range(n_points//2):
        block_h[i*2 : (i+1)*2] = 1
    variations.append(block_h)

    # Add random variation
    h_random = np.random.rand(n_points)
    h_random = h_random * (integral_target / np.sum(h_random))
    h_random = np.clip(h_random, 0, 1)
    variations.append(h_random)

    for h_start in variations:
        try:
            result = minimize(
                evaluate_erdos,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        if time.time() - start_time > budget_s - 10:
            break

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
