import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    start_time = time.time()

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        vars_initial = initial_h[:140]
    else:
        vars_initial = np.random.rand(140) * 0.2 + 0.4

    def project_vars(vars):
        h = np.concatenate([vars, np.flip(vars)])
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        scale = integral_target / sum_clip if sum_clip > 0 else 1.0
        h_scaled = h_clip * scale
        return h_scaled[:140]

    def objective(vars):
        h = np.concatenate([vars, np.flip(vars)])
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    bounds = [(0.0, 1.0) for _ in range(140)]
    best_h_vars = vars_initial.copy()
    best_c5 = objective(best_h_vars)
    best_h = project_vars(best_h_vars)

    shifts = [1, 2, 5, 10, 140, -1, -2, -5, -10, -140]
    for shift in shifts:
        if time.time() - start_time > budget_s - 1:
            break
        shifted_vars = np.roll(best_h_vars, shift)
        noise = np.random.normal(0, 0.01, 140)
        perturbed_vars = np.clip(shifted_vars + noise, 0, 1)
        try:
            result = minimize(
                objective,
                perturbed_vars,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 120, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_h_vars = project_vars(result.x)
                current_c5 = objective(current_h_vars)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h_vars = current_h_vars
        except:
            pass

    elapsed = time.time() - start_time
    best_h = project_vars(best_h_vars)
    return best_h, best_c5, n_points
