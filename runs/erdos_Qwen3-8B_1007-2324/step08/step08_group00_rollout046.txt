import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # to satisfy ∫h dx = 1
    
    def init_from_parent(h_parent, n_half):
        h_full = np.concatenate([h_parent[:n_half], h_parent[:n_half][::-1]])
        sum_val = np.sum(h_full)
        scale = integral_target / sum_val if sum_val > 0 else 1
        return h_full * scale
    
    if 'initial_h_values' in globals():
        n_half = n_points // 2
        h_initial_full = init_from_parent(np.array(initial_h_values), n_half)
    else:
        n_half = n_points // 2
        h_initial_full = np.ones(n_points) * (integral_target / n_points)
    
    def build_symmetric_h(h_vec):
        return np.concatenate([h_vec[:n_half], h_vec[:n_half][::-1]])
    
    def objective(h_vec):
        h_full = build_symmetric_h(h_vec)
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr_val = np.max(corr)
        return max_corr_val * dx, h_full
    
    def grad_objective(h_vec):
        h_full = build_symmetric_h(h_vec)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - n_points + 1
        grad = np.zeros_like(h_vec)
        dx_val = dx
        for i in range(len(h_vec)):
            x_i = i * dx_val
            x_shifted = x_i + k
            if 0 <= x_shifted <= 2:
                j = int(round(x_shifted / dx_val))
                if 0 <= j < n_points:
                    grad[i] = (1 - h_full[j]) * dx_val
                else:
                    grad[i] = 1.0 * dx_val
            else:
                grad[i] = 1.0 * dx_val
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraint = {'type': 'eq', 'fun': lambda h_vec: np.sum(build_symmetric_h(h_vec)) - integral_target}
    
    best_vec = np.array(h_initial_full[:n_half])
    best_val = objective(best_vec)[0]
    
    trials = 6
    for t in range(trials):
        if time.time() - start_time > budget_s:
            break
        np.random.seed(seed + t)
        h_opt_start = best_vec.copy()
        noise = np.random.normal(0, 0.02, n_half)
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        sum_start = np.sum(build_symmetric_h(h_opt_start))
        scale = integral_target / sum_start if abs(sum_start) > 1e-5 else 1
        h_opt_start *= scale
        h_opt_start = np.clip(h_opt_start, 0, 1)
        cost = objective(h_opt_start)[0]
        if cost < best_val:
            best_val = cost
            best_vec = h_opt_start
    
    result = minimize(
        objective,
        best_vec,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        jac=grad_objective,
        options={'maxiter': 600, 'disp': False, 'eps': 1e-2}
    )
    
    if result.success and result.fun < best_val:
        best_val = result.fun
        best_vec = result.x
    
    h_values = build_symmetric_h(best_vec)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_values, best_val, n_points)
    return (h_values, best_val, n_points)
