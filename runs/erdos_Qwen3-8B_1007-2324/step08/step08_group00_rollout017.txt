from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2

    # Initialize symmetric h with parent or basic function
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_full = np.concatenate([h_initial[:n_points//2], h_initial[:n_points//2][::-1]])
    else:
        h_full = np.ones(n_points) * (integral_target / n_points)
        h_full = np.concatenate([h_full[:n_points//2], h_full[:n_points//2][::-1]])

    def objective(x):
        # Build symmetric h from first half
        h = np.concatenate([x, x[::-1]])
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        # Ensure integral constraint
        return np.sum(np.concatenate([x, x[::-1]])) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]

    best_h = h_full[:n_points//2].copy()
    best_c5 = objective(best_h)

    # Global exploration phase
    for trial in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + trial)
        noise = np.random.normal(0, 0.02, n_points // 2)
        candidate = np.clip(best_h + noise, 0, 1)
        # Scale to meet integral constraint
        sum_candidate = np.sum(np.concatenate([candidate, candidate[::-1]]))
        if sum_candidate > 0:
            candidate *= (integral_target / sum_candidate)
        # Check if this is better
        current_c5 = objective(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate

    # Local refinement phase
    try:
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    except:
        pass

    # Return best solution
    h_values = np.concatenate([best_h, best_h[::-1]])
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
    return h_values, best_c5, n_points
