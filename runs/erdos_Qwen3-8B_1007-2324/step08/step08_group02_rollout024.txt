import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target_full = n_points / 2
    integral_target_half = integral_target_full / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        first_half = initial_h[:140]
        if np.sum(first_half) > 0:
            scale = integral_target_half / np.sum(first_half)
        else:
            scale = 1.0
        x_initial = np.clip(first_half * scale, 0, 1)
    else:
        x_initial = np.random.rand(140) * 0.4 + 0.3
        x_initial = np.clip(x_initial * (integral_target_half / np.sum(x_initial)), 0, 1)
    
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        h_clip = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(140)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target_half}]
    
    best_x = x_initial
    best_c5 = objective(best_x)
    start_time = time.time()
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 10:
            break
        np.random.seed(seed + 1000 + _)
        perturb = np.random.normal(0, 0.05, 140)
        x_perturbed = np.clip(x_initial + perturb, 0, 1)
        x_perturbed = np.clip(x_perturbed * (integral_target_half / np.sum(x_perturbed)), 0, 1)
        try:
            res = minimize(objective, x_perturbed, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8})
            if res.success and res.fun < best_c5:
                best_x = res.x
                best_c5 = res.fun
        except:
            pass
    
    best_h = np.concatenate([best_x, best_x[::-1]])
    return best_h, best_c5, n_points
