import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    def repair_individual(h):
        h_clip = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clip)
        if sum_clipped == 0:
            return h_clip
        scale = integral_target / sum_clipped
        h_scaled = h_clip * scale
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    def objective(h):
        h_feas = repair_individual(h)
        corr = np.correlate(h_feas, 1 - h_feas, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    shift_steps = [1, 2, 5, 10, 14, -1, -2, -5, -10, -14]
    num_trials = 30
    max_iter = 100
    trials_remaining = num_trials

    while trials_remaining > 0 and time.time() - start_time < budget_s - 5:

        shift = np.random.choice(shift_steps)
        noise = np.random.normal(0, 0.02, n_points)
        h_start = np.roll(initial_h, shift) + noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        trials_remaining -= 1

    if time.time() - start_time < budget_s - 5:
        for t in range(10):

            noise = np.random.normal(0, 0.015, n_points)
            h_start += noise
            h_start = np.clip(h_start, 0, 1)
