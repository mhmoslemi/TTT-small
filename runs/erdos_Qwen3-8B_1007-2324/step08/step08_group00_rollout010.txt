from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    # Initialize with parent if available

        h_half = initial_h[:n_half]
    else:

        h_half = np.random.rand(n_half) * 0.5 + 0.5
        h_half = h_half / np.sum(h_half) * integral_target

    # Objective function
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        c5_val = max_corr * dx
        penalty = 0 if np.abs(np.sum(h_full) - n_points) < 1e-3 else 1e6 * np.abs(np.sum(h_full) - n_points)
        return c5_val + penalty

    # Genetic operators
    def crossover(a, b):
        return np.array([ai if np.random.rand() < 0.5 else bi for ai, bi in zip(a, b)])

    def mutate(x, mu, sigma):
        perturb = np.random.normal(0, sigma, len(x))
        x_new = np.clip(x + perturb, 0, 1)
        return x_new * (integral_target / np.sum(x_new) if np.sum(x_new) > 1e-6 else 1)

    # Evolutionary search
    pop_size = 50
    pop = [h_half + np.random.normal(0, 0.03, n_half) for _ in range(pop_size)]
    pop = [np.clip(p, 0, 1) * (integral_target / np.sum(np.clip(p, 0, 1))) for p in pop]

    best_c5 = float('inf')
    best_h = h_half.copy()

    for gen in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        # Tournament selection
        selected = np.random.choice(pop, size=5, replace=False)
        selected = [np.clip(p, 0, 1) * (integral_target / np.sum(np.clip(p, 0, 1))) for p in selected]
        fit = [objective(p) for p in selected]
        winner = selected[np.argmin(fit)]
        
        # Mutation
        if np.random.rand() < 0.2:
            winner = mutate(winner, mu=0.03, sigma=0.05)
        else:
            winner = mutate(winner, mu=0.01, sigma=0.02)
        
        # Crossover
        if np.random.rand() < 0.3:
            other = np.random.choice(pop)
            winner = crossover(winner, other)
        
        # Update population
        pop = [p for p in pop if np.random.rand() > 0.1]
        pop.append(winner)
        if len(pop) > pop_size:
            pop.pop(np.argmax([objective(p) for p in pop]))
        
        # Local refinement
        x0 = winner.copy()
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(np.clip(x, 0, 1)) - integral_target}]
        res = minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_half, constraints=cons, tol=1e-8)
        if res.success:
            candidate = res.x
            c5 = objective(candidate)
            if c5 < best_c5:
                best_c5 = c5

    # Final refinement
    x0 = best_h
