import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 140

    # Initial setup
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_erdos(current_h)
    time_used = 0

    # SA parameters
    T_initial = 50.0
    cooling_rate = 0.995
    min_T = 0.1
    perturb_scale_high = 0.15

    T = T_initial
    while T > min_T and time_used < budget_s - 10:
        # Generate perturbation
        noise = np.random.normal(0, perturb_scale_high * T, n_points)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_clipped = np.sum(new_h)
        if sum_clipped == 0:
            new_h = current_h.copy()
        else:
            scale = integral_target / sum_clipped
            new_h *= scale
        # Evaluate
        new_c5 = evaluate_erdos(new_h)
        # Acceptance criterion
        if new_c5 < best_c5:
            current_h = new_h.copy()
            best_c5 = new_c5
        else:
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                current_h = new_h.copy()
                best_c5 = new_c5
        # Cool down
        T *= cooling_rate
        time_used = time.time() - start_time

    # Local refinement with SLSQP if time allows
    if time_used < budget_s - 10:
        def objective(h):
            return evaluate_erdos(h)
        constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target})
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8})
        if result.success:
            best_h = result.x
            best_c5 = result.fun

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
