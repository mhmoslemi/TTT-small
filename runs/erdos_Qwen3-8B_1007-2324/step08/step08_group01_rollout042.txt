import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair_individual(individual):
        clipped = np.clip(individual, 0, 1)
        current_sum = np.sum(clipped)
        if current_sum == 0:
            return np.zeros_like(clipped) + 1.0
        scale = target_sum / current_sum
        return clipped * scale

    def evaluate_erdos(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    pop_size = 100
    pop = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        pop[i] = repair_individual(pop[i])

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(best_h)

    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        new_pop = []
        for _ in range(pop_size):
            a, b = np.random.choice(pop, 2, replace=False)

            child = np.concatenate([a[:point], b[point:]])
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = repair_individual(child)
            new_pop.append(child)
        pop = new_pop
        fitness = [evaluate_erdos(h) for h in pop]
        idx = np.argmin(fitness)
        best_h = pop[idx]
        best_c5 = fitness[idx]

    final_h = repair_individual(best_h)
    final_c5 = evaluate_erdos(final_h)
    elapsed = time.time() - start_time
