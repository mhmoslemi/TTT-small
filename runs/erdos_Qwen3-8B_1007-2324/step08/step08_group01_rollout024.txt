import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(best_h)

    T_initial = 50.0
    cooling_rate = 0.995
    min_T = 0.1
    perturb_scale = 0.15
    current_h = initial_h.copy()
    current_c5 = evaluate_erdos(current_h)
    T = T_initial

    while T > min_T and time.time() - start_time < budget_s - 10:
        for _ in range(100):
            perturb = np.random.normal(0, perturb_scale * T / T_initial, n_points)
            new_h = current_h + perturb
            clipped = np.clip(new_h, 0, 1)
            sum_clipped = np.sum(clipped)
            if sum_clipped == 0:
                scaled = np.zeros_like(clipped)
            else:
                scaled = np.clip(clipped * (integral_target / sum_clipped), 0, 1)
            new_c5 = evaluate_erdos(scaled)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = scaled.copy()
                current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        T *= cooling_rate

    if time.time() - start_time < budget_s - 10:
        result = minimize(
            evaluate_erdos,
            best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],
