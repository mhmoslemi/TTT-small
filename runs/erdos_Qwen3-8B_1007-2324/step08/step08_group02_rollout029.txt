import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points)

    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = initial_h * (integral_target / total)
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    max_trials = 50
    for trial in range(max_trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + trial)
        pert = np.random.normal(0, 0.03, n_points)
        h_start = initial_h + pert
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (integral_target / np.sum(h_start)) if np.sum(h_start) > 0 else h_start

        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
