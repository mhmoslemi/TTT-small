import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    # Simulated Annealing
    temp = 1.0
    cooling_rate = 0.997
    steps = int(10000)  # Adjusted for faster exploration
    current_h = initial_h.copy()

    for _ in range(steps):
        if time.time() - start_time > budget_s - 10:
            break
        noise = np.random.normal(0, 0.05, n_points)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h *= (integral_target / sum_new)
        else:
            new_h = np.random.rand(n_points)
            new_h *= (integral_target / np.sum(new_h))
        new_c5 = evaluate_erdos(new_h)
        delta = new_c5 - evaluate_erdos(current_h)
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h
        temp *= cooling_rate

    # Local Refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    try:
        result = minimize(
            evaluate_erdos,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
    except:
        pass

    if elapsed > budget_s:

    return best_h, best_c5, n_points
