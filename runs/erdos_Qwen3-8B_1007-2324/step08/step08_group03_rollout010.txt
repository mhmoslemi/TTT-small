import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0.0, 1.0)

    def repair(h):
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) < 1e-8:
            return h
        indices = np.random.choice(n_points, size=30, replace=False)
        for idx in indices:
            val = h[idx]
            max_change = min(1.0 - val, val)
            if diff > 0:
                h[idx] = min(val + max_change, 1.0)
            else:
                h[idx] = max(val - max_change, 0.0)
            diff -= max_change if diff > 0 else -max_change
            if abs(diff) < 1e-8:
                break
        return np.clip(h, 0.0, 1.0)

    def evaluate(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h_clipped, 1.0 - h_clipped, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def local_objective(h):
        return evaluate(h)

    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]

    for _ in range(10):
        if time.time() - start_time > budget_s - 2:
            break
        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = repair(perturbed)
        population.append(perturbed)

    population = np.array(population)
    best_fitness = float('inf')
    best_h = initial_h.copy()

    for gen in range(50):
        if time.time() - start_time > budget_s - 2:
            break
        fitness = np.array([evaluate(ind) for ind in population])
        elite_idx = np.argsort(fitness)[::-1]
        elites = population[elite_idx[:5]]

    best_h = result.x
    best_c5 = evaluate(best_h)
