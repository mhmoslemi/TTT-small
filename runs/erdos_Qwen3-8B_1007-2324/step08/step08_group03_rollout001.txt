from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair_integral(h):
        diff = np.sum(h) - integral_target
        if abs(diff) < 1e-5:
            return h.copy()
        h = h.copy()
        for _ in range(100):
            if abs(diff) < 1e-5:
                break
            idx = np.random.randint(n_points)
            max_val = 1.0 if h[idx] < 1 else h[idx]
            min_val = 0.0 if h[idx] > 0 else h[idx]
            possible = max_val - min_val
            if possible < 1e-5:
                break
            delta = min(possible, abs(diff))
            h[idx] += delta * np.sign(diff)
            diff -= delta
        return np.clip(h, 0, 1)

    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    evals = 0

    def perturb(h, temp=0.3):
        h = h.copy()
        idxs = np.random.choice(n_points, size=30, replace=False)
        noise = np.random.normal(0, 0.15 * np.std(h), size=n_points)
        h[idxs] += noise[idxs] * (0.5 - (h[idxs] - 0.5) / 0.5)
        return repair_integral(h)

    if 'initial_h_values' in globals():
        starts = [initial_h] + [perturb(initial_h, temp=0.3 + i*0.05) for i in range(5)]
    else:
        starts = [np.full(n_points, 0.5)] + [perturb(np.full(n_points, 0.5), temp=0.3 + i*0.05) for i in range(5)]

    for start in starts:
        if time.time() - start_time > budget_s - 2:
            break
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
