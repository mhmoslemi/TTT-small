import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    # Generate diverse starting points
    trials = []
    # Perturbed initial
    noise = np.random.normal(0, 0.05, n_points)
    h_perturb = initial_h + noise
    h_perturb = np.clip(h_perturb, 0, 1)
    sum_perturb = np.sum(h_perturb)
    if sum_perturb > 0:
        h_perturb *= (integral_target / sum_perturb)
    trials.append(h_perturb)

    # Random
    h_rand = np.random.uniform(0, 1, n_points)
    sum_rand = np.sum(h_rand)
    if sum_rand > 0:
        h_rand *= (integral_target / sum_rand)
    trials.append(h_rand)

    # Shifted parents
    shifts = [1, -1, 1/dx, -1/dx, dx, -dx, 0.5, -0.5]
    for shift in shifts:

        shifted_h = np.roll(initial_h, shift_idx)
        noise = np.random.normal(0, 0.03, n_points)
        shifted_h += noise
        shifted_h = np.clip(shifted_h, 0, 1)
        sum_shifted = np.sum(shifted_h)
        if sum_shifted > 0:
            shifted_h *= (integral_target / sum_shifted)
        trials.append(shifted_h)

    # Optimize each trial
    for h_start in trials:
        if time.time() - start_time > budget_s - 10:

        result = minimize(

            h_start,
