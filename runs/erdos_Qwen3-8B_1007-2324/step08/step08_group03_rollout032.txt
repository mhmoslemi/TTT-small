import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def obj(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    starts = [initial_h]
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        start = initial_h + noise
        start = np.clip(start, 0, 1)
        starts.append(start)

    best_h = initial_h.copy()
    best_c5 = obj(initial_h)

    for start in starts:
        if time.time() - start_time > budget_s - 2:
            break
        try:
            res = minimize(obj, start, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-8)
            if res.success:
                if res.fun < best_c5:
                    best_c5 = res.fun
                    best_h = res.x
        except:
            pass

    return best_h, best_c5, n_points
