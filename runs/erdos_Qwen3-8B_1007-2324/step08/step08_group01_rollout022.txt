import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

    # Generate multiple random perturbations
    num_perturbations = 10
    np.random.seed(seed)
    for _ in range(num_perturbations):
        # Create a perturbed version of initial_h
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h = initial_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Rescale to meet the integral constraint
        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed > 0:
            perturbed_h *= (integral_target / sum_perturbed)

        # Run optimization
        result = minimize(
            evaluate_erdos,
            perturbed_h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

        return np.max(corr) * dx

    num_perturbations = 10
    perturbations = []

        noise = np.random.normal(0, 0.05, n_points)

        perturbed = np.clip(perturbed, 0, 1)

        perturbations.append(perturbed)
