import time

def run(seed=42, budget_s=100, **kwargs):

    def evaluate_erdos(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos(initial_h)

        noise = np.random.normal(0, 0.1, n_points)
        perturbed = initial_h + noise

        if sum_perturbed > 0:

        perturbations.append(perturbed)

    # Add random start
