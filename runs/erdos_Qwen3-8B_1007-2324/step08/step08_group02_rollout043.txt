            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_x = result.x
                best_h_values = np.concatenate([best_x, best_x[::-1]])

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_values, best_c5, n_points

    return best_h_values, best_c5, n_points
