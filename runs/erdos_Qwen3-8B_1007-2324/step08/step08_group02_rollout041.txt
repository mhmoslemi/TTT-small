import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_vars = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_full = np.zeros(n_points)
        h_full[:n_vars] = initial_h[:n_vars]
        h_full[n_vars:] = initial_h[:n_vars][::-1]
        current_sum = np.sum(h_full)
        scale = integral_target / current_sum
        h_full = np.clip(h_full * scale, 0, 1)
        initial_vars = h_full[:n_vars]
    else:
        initial_vars = np.full(n_vars, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n_vars)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 70.0}]

    def objective(x):
        h_full = np.zeros(n_points)
        h_full[:n_vars] = x
        h_full[n_vars:] = np.flip(x)
        h_clip = np.clip(h_full, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    result = minimize(objective, initial_vars, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-9, 'disp': False})
    best_h = result.x
    best_c5 = result.fun

    if time.time() - start_time < budget_s - 5:

            if time.time() - start_time > budget_s - 1:
                break

            perturbed = np.clip(perturbed, 0, 1)
            sum_perturbed = np.sum(perturbed)
            if sum_perturbed > 0:
                perturbed *= 70.0 / sum_perturbed
