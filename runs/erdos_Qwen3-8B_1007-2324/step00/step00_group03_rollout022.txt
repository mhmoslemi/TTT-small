def c5(h):
    n = len(h)
    dx = 2 / n
    c = np.correlate(h, 1-h, mode='full')
    return np.max(c) * dx

def grad(h, n):
    c = np.correlate(h, 1-h, mode='full')
    m = np.argmax(c)
    d = m - (n-1)
    g = np.zeros(n)
    dx = 2 / n
    for j in range(n):
        if j + d >= n: t1 = 1
        else: t1 = 1 - h[j+d]
        if j - d < 0: t2 = 0
        else: t2 = h[j-d]
        g[j] = dx * (t1 - t2)
    return g

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    H = initial_h_values()
    best = H.copy()
    best_c = c5(best)
    for n in [70, 140, 280]:
        if n == 70:
            h = H
        else:
            h = np.repeat(np.repeat(H, 2), 2) if n == 280 else np.repeat(H, 2)
            h = h * (n/2) / np.sum(h)
        t0 = time.time()
        while time.time() - t0 < budget_s:
            g = grad(h, n)
            h = np.clip(h - 0.01*g, 0, 1)
            h = h * (n/2) / np.sum(h)
            curr = c5(h)
            if curr < best_c:
                best_c, best = curr, h.copy()
            if np.all(np.abs(g) < 1e-6): break
    return best, best_c, len(best)
