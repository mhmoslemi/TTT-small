def repair(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    sum_clipped = np.sum(h_clipped)
    if sum_clipped == 0:
        return h_clipped
    scale_factor = (n_points / 2) / sum_clipped
    scaled_h = h_clipped * scale_factor
    scaled_h_clipped = np.clip(scaled_h, 0, 1)
    sum_scaled = np.sum(scaled_h_clipped)
    for _ in range(2):
        if sum_scaled == 0:
            break
        scale_factor = (n_points / 2) / sum_scaled
        scaled_h_clipped = scaled_h_clipped * scale_factor
        scaled_h_clipped = np.clip(scaled_h_clipped, 0, 1)
        sum_scaled = np.sum(scaled_h_clipped)
    return scaled_h_clipped

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
