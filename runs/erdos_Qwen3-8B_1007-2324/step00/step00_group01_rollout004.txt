import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2  # 23.5

    # Use initial_h_values if available
    initial_h = initial_h_values

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    # Number of restarts
    num_restarts = 10
    for _ in range(num_restarts):
        # Perturb initial_h with small noise
        perturbation = np.random.laplace(0, 0.015, size=n_points)
        h_initial = initial_h + perturbation
        h_initial = np.clip(h_initial, 0, 1)
        current_sum = np.sum(h_initial)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        h_initial = h_initial * scale

        # Define objective function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        # Constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraint_sum = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])

        # Run optimization
        result = minimize(
            fun=objective,
            x0=h_initial,
            method='COBYLA',
            bounds=bounds,
            constraints=constraint_sum,
            options={'maxiter': 1000}
        )

        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            current_sum = np.sum(h_opt)
            scale = target_sum / current_sum if current_sum != 0 else 1.0
            h_opt = h_opt * scale

            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(corr) * dx

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                print(f"New best C5: {best_c5}")

        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    if best_h is None:
        best_h = initial_h
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
