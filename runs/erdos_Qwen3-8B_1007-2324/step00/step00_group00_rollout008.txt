def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)

        h = np.ones(n_points) * (target_sum / n_points)

        h = np.clip(h, 0, 1)

        return h

    def objective(h_flat):
        h = h_flat.copy()
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
