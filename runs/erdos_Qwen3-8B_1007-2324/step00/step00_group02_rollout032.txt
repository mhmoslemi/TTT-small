import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    initial_h = initial_h_values
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-8:
        scale = (n_points / 2) / sum_h
        initial_h = initial_h * scale
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        s = np.sum(h)
        if s == 0:
            return float('inf')
        scaling_factor = (n_points / 2) / s
        h_scaled = h * scaling_factor
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    h_values = result.x
    sum_h = np.sum(h_values)
    if abs(sum_h - n_points / 2) > 1e-8:
        scale = (n_points / 2) / sum_h
        h_values = h_values * scale
    c5_bound = objective(h_values)
