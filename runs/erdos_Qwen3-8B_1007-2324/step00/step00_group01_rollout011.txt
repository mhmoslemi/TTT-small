import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    if not hasattr(initial_h_values, '__len__'):
        n_points = 47
        h_values = np.repeat(1.0 / n_points, n_points)
        initial_h_values = h_values
    initial_h = initial_h_values
    best_h = initial_h.copy()

    n_points_best = len(initial_h)
    t0 = time.time()
