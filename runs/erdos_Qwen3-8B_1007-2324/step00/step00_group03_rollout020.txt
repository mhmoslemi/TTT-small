def compute_cost(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def calculate_gradient(h, n, d, dx):
    grad = np.zeros(n)
    for j in range(n):
        if j + d < n:
            term1 = 1 - h[j + d]
        else:
            term1 = 1.0
        if j - d >= 0:
            term2 = h[j - d]
        else:
            term2 = 0.0
        grad[j] = (term1 - term2) * dx
    return grad
