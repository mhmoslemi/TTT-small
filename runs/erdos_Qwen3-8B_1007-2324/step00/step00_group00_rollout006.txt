import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize with provided or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(n_points) * (target_sum / n_points)
    
    # Project to feasible set
    def project_to_simplex(h_flat):
        h_proj = np.clip(h_flat, 0, 1)
        total_proj = np.sum(h_proj)
        if abs(total_proj - target_sum) > 1e-6:
            scale = target_sum / total_proj
            h_proj *= scale
        return h_proj
    
    # Objective function
    def compute_objective(h_flat):
        h_1_minus = 1.0 - h_flat
        corr = np.correlate(h_flat, h_1_minus, mode='full')
        return corr.max() * dx
    
    # Initial projection
    h = project_to_simplex(h)
    best_h = h.copy()
    best_c5 = compute_objective(best_h)
    start_time = time.time()
    
    # Optimization loop
    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbations
        idx = np.random.randint(0, n_points)
        perturb = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()
        new_h[idx] += perturb
        new_h = project_to_simplex(new_h)
        
        current_c5 = compute_objective(new_h)
        if current_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = current_c5
            h = best_h
    
    return best_h, best_c5, n_points
