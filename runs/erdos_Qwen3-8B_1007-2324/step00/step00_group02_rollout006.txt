import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    np.random.seed(seed)
    
    def calculate_C5(h):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        total = np.sum(h)
        sc = 0.5 / (total / n_points) if abs(total) > 1e-7 else 0.5
        return h * sc
    
    best_h = repair(initial_h.copy())
    best_c5 = calculate_C5(best_h)
    
    time_start = time.time()
    
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        h = best_h.copy()
        idxs = np.random.choice(n_points, 5, replace=False)
        for i in idxs:
            h[i] += np.random.uniform(-0.1, 0.1)
        h = repair(h)
        c5 = calculate_C5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, n_points
