    dx = 2.0 / n_points

    def objective(h):

        current_sum = np.sum(h_clipped)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        h_scaled = np.clip(h_clipped * scale, 0, 1)
        for _ in range(3):
            current_sum = np.sum(h_scaled)
            if abs(current_sum - target_sum) < 1e-6:
                break
            scale = target_sum / current_sum
            h_scaled = np.clip(h_scaled * scale, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0., 1.) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   maxiter=400, popsize=15, tol=1e-6,
                                   mutation=(0.5, 1), recombination=0.9,
                                   seed=seed, disp=False)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:

            scale = target_sum / current_sum

        c5_best = np.max(np.correlate(h_scaled, 1 - h_scaled, mode='full')) * dx
        return (h_scaled, c5_best, n_points)
