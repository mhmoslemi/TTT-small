            x_initial = np.linspace(0, 2, n_initial)
            f = interp1d(x_initial, initial_h, kind='linear')
            x_new = np.linspace(0, 2, n)
            h0 = f(x_new)
            h0 = np.clip(h0, 0, 1)
        dx = 2.0 / n

        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}

        with np.errstate(all='ignore'):
            res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})

        if res.success:
            h_opt = np.clip(res.x, 0, 1)
            c5 = obj(h_opt)
        else:
            h_opt = h0
            c5 = obj(h_opt)

        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

        if np.random.SeedSequence(time_start).spawn(1)[0] > 1000:
            break

    return (best_h, best_c5, len(best_h))
