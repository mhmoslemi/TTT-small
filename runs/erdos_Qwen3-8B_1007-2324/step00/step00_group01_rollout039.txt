        if n == 47:
            initial_guess = initial_h_values
        else:
            initial_guess = np.full(n, 0.5)
        
        dx_val = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_val
        
        result = minimize(
            fun=objective,
            x0=initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100}
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n
    
    return (best_h, best_c5, best_n)
