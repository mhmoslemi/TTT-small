import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 45
        h = np.random.rand(n)
        h = np.sort(h)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = (h * 100).astype(np.float64)
        total = np.sum(h)
        scale = (n / 2) / total if total != 0 else 1.0
        h *= scale
    n_points = len(h)
    dx = 2.0 / n_points
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    for _ in range(1000):
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        total = np.sum(h_perturbed)

        h_perturbed *= scale
        c5 = compute_c5(h_perturbed)
        if c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5
        if time.time() - start_time >= budget_s:
