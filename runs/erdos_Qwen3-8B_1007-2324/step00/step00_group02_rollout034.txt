import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    # Use the provided initial_h_values if available
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    # Define bounds [0, 1] for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function to minimize: max overlap integral
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return (2.0 / n_points) * np.max(corr)
    
    # Penalty function for constraint violation (sum(h) != n_points / 2)
    def objective_penalty(h):
        c5 = objective(h)
        sum_h = np.sum(h)
        # Use a large penalty to enforce constraint
        penalty = 1e4 * abs(sum_h - n_points / 2)
        return c5 + penalty
    
    # Run differential evolution with constraints
    result = differential_evolution(
        objective_penalty, bounds,
        popsize=15, maxiter=50, tol=1e-6,
        seed=seed, dispatch='parallel'
    )
    
    # Extract the optimized h and compute final C₅
    h_opt = result.x
    c5 = objective(h_opt)
    
    # Return as required
    return h_opt.tolist(), c5, n_points
