import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = len(h)
        target_sum = n_points / 2
        # Normalize initial sum
        h = h / np.sum(h) * target_sum
    else:
        n_points = 47
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum

    def objective(x):
        h_vec = x
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        dx = 2.0 / n_points
        max_overlap = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(h_vec) - target_sum)
        return max_overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=200,
        seed=seed
    )
    return (result.x, result.fun, n_points)
