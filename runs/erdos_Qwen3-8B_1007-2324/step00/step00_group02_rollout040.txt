def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    def repair_h(h_vals):
        h_clipped = np.clip(h_vals, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        target_sum = n_points / 2.0
        if current_sum == 0:
            return np.zeros_like(h_clipped)
        scale_factor = target_sum / current_sum
        return h_clipped * scale_factor

    h = repair_h(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    iterations = 0
    max_iter = int(1000 * (budget_s / 10))  # Adjusted for time budget

    while time.time() - start_time < budget_s:

        # Generate perturbation
        noise = np.random.normal(0, 0.1, n_points)
        new_h = best_h + noise
        new_h = repair_h(new_h)
