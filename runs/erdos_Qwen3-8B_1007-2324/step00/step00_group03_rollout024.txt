                else:
                    term1 = 1
                if (j - d) >= 0:
                    term2 = current_h[j - d]
                else:
                    term2 = 0
                grad[j] = dx_val * (term1 - term2)
            eta = 0.01
            current_h = np.clip(current_h - eta * grad, 0, 1)
            sum_current = np.sum(current_h)
            if sum_current != 0:
                current_h = current_h * (n / 2) / sum_current
            new_c5 = compute_cost(current_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
            total_time = time.time() - start_time
    return best_h, best_c5, len(best_h)
