                          0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5,
                          0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5,
                          0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5,
                          0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5, 0.4, 0.3, 0.5, 0.5])
    h = default_h.copy()
    h = h * (n_points / 2.0) / np.sum(h)

    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    def normalize_h(h_val):
        scale = (n_points / 2.0) / np.sum(h_val)
        return h_val * scale

    def grad_descent_step(h_val):
        eps = 1e-4
        grad = np.zeros_like(h_val)
        for i in range(len(h_val)):
            h_perturb = h_val.copy()
            h_perturb[i] += eps
            h_norm = normalize_h(h_perturb)
            grad[i] = (compute_c5(h_norm) - compute_c5(h_val)) / eps
        h_new = h_val - 0.1 * grad
        return normalize_h(h_new)

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        h = grad_descent_step(h)
    
    final_c5 = compute_c5(h)
    return h, final_c5, n_points
