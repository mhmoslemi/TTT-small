import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def obj(h):
        h_arr = h.reshape(-1)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0)] * n_points
    initial_guess = np.array(initial_h_values)
    
    result = minimize(obj, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-6})
    
    h_vals = result.x
    c5 = obj(h_vals)
    return (h_vals, c5, n_points)
