    def optimize_h(h_initial, n):
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]
        
        def obj(x):
            return compute_c5(x, n)
        
        result = minimize(obj, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8})
        return result.x if result.success else h_initial, float('inf')
    
    for n in resolutions:
        x_initial = np.linspace(0, 2, n_initial)
        x_new = np.linspace(0, 2, n)
        interp_func = interp1d(x_initial, initial_h, kind='linear', fill_value='extrapolate')
        h_initial = interp_func(x_new)
        
        h_opt, c5 = optimize_h(h_initial, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n
    
    return (best_h, best_c5, best_n)
