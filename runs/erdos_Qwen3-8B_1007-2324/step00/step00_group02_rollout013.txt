import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in kwargs:
        h_initial = np.array(kwargs['initial_h_values'])
    else:
        n = 45
        h_initial = np.zeros(n)
        h_initial[::2] = 1.0
    n_points = len(h_initial)
    
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if total == 0:
            return np.zeros(n_points)
        scale = (0.5 * n_points) / total
        return h * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - 0.5 * n_points
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    h = repair(h_initial)
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
    best_h = repair(result.x)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points

def repair_full(h, n_points):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if total == 0:
        return np.zeros(n_points)
    scale = (0.5 * n_points) / total
    return h * scale
