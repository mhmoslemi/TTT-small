    best_h = parent_h.copy()
    best_c5 = compute_cost(parent_h)
    remaining_time = budget_s
    
    for i, n in enumerate(n_list):
        budget = remaining_time * (1 / len(n_list))
        h = upsample(parent_h, n)
        h = optimize_h(h, n, budget)
        current_c5 = compute_cost(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        remaining_time -= budget
    
    return best_h, best_c5, len(best_h)
