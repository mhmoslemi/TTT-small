import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 45
    dx = 2 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(n_points) * (target_sum / n_points)

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def project_to_simplex(h):
        h_proj = np.clip(h, 0, 1)
        total = np.sum(h_proj)
        if np.abs(total - target_sum) > 1e-8:
            scale = target_sum / total
            h_proj *= scale
        return h_proj

    def objective(h_flat):
        return compute_c5(h_flat)

    def update_perturbation(h):
        h = h.copy()
        idx = np.random.choice(n_points)
        h[idx] += np.random.uniform(-0.2, 0.2)
        h = project_to_simplex(h)
        return h

    best_h = h
    best_c5 = compute_c5(best_h)
    start = time.time()

    for _ in range(2000):
        if time.time() - start > budget_s - 2:
            break
        h_candidate = update_perturbation(best_h)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    return best_h, best_c5, n_points
