import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_initial = len(initial_h)
    target_n = 94
    best_h = initial_h.copy()
    best_c5 = float('inf')
    best_n = n_initial

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_candidate(parent, target_n):
        N = len(parent)
        fs = fft(parent)
        N_new = target_n
        fs_padded = np.zeros(N_new, dtype=complex)
        fs_padded[:N] = fs
        h_new = np.real(ifft(fs_padded))
        h_new = np.clip(h_new, 0, 1)
        target_sum = target_n / 2
        scale = (target_sum - np.sum(h_new)) / (target_n * 1.0)
        h_new = np.clip(h_new + scale, 0, 1)
        return h_new

    def obj_func(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - len(x)/2

    start = time.time()
    while time.time() - start < budget_s - 1:
        if len(best_h) == n_initial:
            h_candidate = generate_candidate(initial_h, target_n)
        else:
            h_candidate = generate_candidate(best_h, target_n)
        bounds = [(0, 1)] * len(h_candidate)
