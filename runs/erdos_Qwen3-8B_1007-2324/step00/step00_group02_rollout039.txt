            if np.sum(new_h) > 0:
                scale = (n_points / 2.0) / np.sum(new_h)
                new_h = new_h * scale
            new_h = np.clip(new_h, 0, 1)
        
        c5 = compute_c5(new_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
