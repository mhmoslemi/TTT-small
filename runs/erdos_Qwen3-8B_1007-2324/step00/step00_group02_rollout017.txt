import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    
    # Use provided initial_h_values if available
    initial_h = kwargs.get('initial_h_values', np.repeat(0.5, n_points))
    initial_sum = np.sum(initial_h)
    if abs(initial_sum - n_points / 2) > 1e-6:
        diff = n_points / 2 - initial_sum
        initial_h[0] += diff
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx
    
    # Constraints
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization with SLSQP
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x
    best_c5 = result.fun
    
    return (best_h, best_c5, n_points)
