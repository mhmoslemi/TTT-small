import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.ones(n_points) * target_sum / n_points

    def compute_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project_to_simplex(h):
        h_proj = np.clip(h, 0, 1)
        total_proj = np.sum(h_proj)
        if np.abs(total_proj - target_sum) > 1e-6:
            scale = target_sum / total_proj
            h_proj *= scale
        return h_proj

    def perturb(h):
        h_new = h.copy()
        idx = np.random.randint(n_points)
        h_new[idx] += np.random.uniform(-0.1, 0.1)
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new)
        if total != 0:
            h_new *= target_sum / total
        return h_new

    h = project_to_simplex(initial_h)
    best_val = compute_objective(h)
    best_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s:
        h = perturb(h)
        val = compute_objective(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    best_h = project_to_simplex(best_h)
    final_val = compute_objective(best_h)
    return best_h, final_val, n_points
