import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    h = h * (n_points / 2) / total if total != 0 else h

    def obj(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x) * dx
        x = x * (n_points / 2) / total if total != 0 else x
        corr = correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) * dx - 1

    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = minimize(obj, h, bounds=bounds, constraints={'type': 'eq', 'fun': cons}, method='L-BFGS-B', tol=1e-8)
        h = result.x
        if result.success:
            break
    c5 = obj(h)
    return (h, c5, n_points)
