        dx = 2.0 / n
        sum_h = np.sum(h_initial)
        if abs(sum_h - n / 2) > 1e-6:
            scale = (n / 2) / sum_h
            h_initial = h_initial * scale

        def objective(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        bounds = [(0.0, 1.0) for _ in range(n)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}

        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        if result.success:
            h_opt = result.x
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n

    return (best_h, best_c5, best_n)
