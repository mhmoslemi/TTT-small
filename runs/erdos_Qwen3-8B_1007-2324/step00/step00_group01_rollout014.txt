import numpy as np
from scipy.optimize import differential_evolution
import time

def repair(h, target_sum):
    h_clamped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clamped)
    if np.isclose(current_sum, target_sum):
        return h_clamped
    scale_factor = target_sum / current_sum
    h_scaled = h_clamped * scale_factor
    h_final = np.clip(h_scaled, 0.0, 1.0)
    return h_final

    max_corr = np.max(correlation)
