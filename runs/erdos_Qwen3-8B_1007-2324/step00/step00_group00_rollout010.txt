import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Use initial_h_values if available
    try:
        initial_h = initial_h_values
    except NameError:
        initial_h = np.ones(n_points) * (target_sum / n_points)
    initial_h /= np.sum(initial_h)
    initial_h *= target_sum

    def objective(h_flat):
        h = h_flat.copy()
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def penalty_func(h_flat):
        h = h_flat.copy()
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        sum_h = np.sum(h)
        if np.abs(sum_h - target_sum) > 1e-6:
            return np.inf
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use differential evolution with adjusted parameters
    result = differential_evolution(
        penalty_func,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=1000
    )

    best_h = result.x
    final_c5 = objective(best_h)
    return best_h, final_c5, n_points
