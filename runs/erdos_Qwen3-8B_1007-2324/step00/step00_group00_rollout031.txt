    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(x):
        sum_first_44 = np.sum(x)
        penalty = 0.0
        if sum_first_44 < 21.5 or sum_first_44 > 22.5:
            penalty = 1e6 * (abs(sum_first_44 - 22.5) + abs(sum_first_44 - 21.5))
        
        h_array = np.zeros(n_points)
        h_array[:44] = x
        h_array[44] = target_sum - sum_first_44
        h_array[44] = np.clip(h_array[44], 0, 1)
        
        h_shifted = 1.0 - h_array
        corr = correlate(h_array, h_shifted, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(44)]
    x0 = h[:44]
    
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    if result.success:
        best_h = np.zeros(n_points)
        best_h[:44] = result.x
        sum_first_44 = np.sum(result.x)
        best_h[44] = target_sum - sum_first_44
        best_h[44] = np.clip(best_h[44], 0, 1)
        c5_bound = objective(result.x)
    else:
        best_h = h.copy()
        c5_bound = evaluate_erdos_solution(best_h)
    
    return (best_h, c5_bound, n_points)
