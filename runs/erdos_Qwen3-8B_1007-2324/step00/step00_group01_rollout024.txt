def run(seed=42, budget_s=100, **kwargs):

    for n in n_points_options:

            dx_val = 2.0 / n
            def objective(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx_val
            bounds = [(0.0, 1.0)] * n
            constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}
            result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
