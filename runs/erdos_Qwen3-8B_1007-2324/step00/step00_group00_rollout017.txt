import numpy as np
from numpy.random import RandomState

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    target_sum = n_points / 2
    h = initial_h_values.copy()

    def evaluate(h):

        h[44] = np.clip(h[44], 0.0, 1.0)
        return evaluate_erdos_solution(h)

    best_h = h.copy()
    best_c5 = evaluate(best_h)
    max_iter = 1000

    for _ in range(max_iter):
