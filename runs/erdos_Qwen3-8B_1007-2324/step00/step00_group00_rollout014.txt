import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)

        x = np.linspace(0, 2, n_points)
        h_initial = 0.5 * (1 - np.cos(2 * np.pi * x / n_points))

        sum_h = np.sum(h_initial)
        scale = target_sum / sum_h
        h_initial = h_initial * scale
    
    h_current = h_initial.copy()
    best_h = h_current.copy()
    best_c5 = float('inf')
    
    time_start = time.time()
    
    def project_candidate(h_44):
        h_clipped = np.clip(h_44, 0, 1)
        sum_44 = np.sum(h_clipped)
        last = target_sum - sum_44
        if 0 <= last <= 1:
            return np.concatenate([h_clipped, [last]])
        return None
    
    def evaluate_c5(h):

        corr = np.correlate(h, one_minus_h, mode='full')

        if h_candidate is not None:
