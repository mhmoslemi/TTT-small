def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

            return float('inf')
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-5:
            return float('inf')
        scale = (n_points / 2) / total
        h = h * scale
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Perturbation
