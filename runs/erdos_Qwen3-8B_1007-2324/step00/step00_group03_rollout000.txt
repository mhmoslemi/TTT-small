import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    np.random.seed(seed)
    
    # Start with initial_h_values (70 points)
    h = np.array(initial_h_values)
    n_points = 140
    h = np.repeat(h, 2)  # Upsample to 140 points
    target_sum = n_points / 2
    h = h.astype(float)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iter_limit = int(100000 * (budget_s / 100))  # Adjust based on time
    
    for _ in range(iter_limit):
        # Perturb random elements
        indices = np.random.choice(len(h), size=30, replace=False)
        delta = np.random.uniform(-0.01, 0.01, len(h))
        h_perturbed = h + delta
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Adjust sum constraint
        current_sum = np.sum(h_perturbed)
        if abs(current_sum - target_sum) > 1e-4:
            scale = target_sum / current_sum
            h_perturbed *= scale
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
