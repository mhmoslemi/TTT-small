        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum < target_sum:
            mask = (h > 0) & (h < 1)
            if np.any(mask):
                h[mask] = (h[mask] * target_sum) / current_sum
        else:
            mask = (h > 0) & (h < 1)
            if np.any(mask):
                h[mask] = (h[mask] * target_sum) / current_sum
        
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return (best_h, best_c5, n_points)
