import numpy as np
import scipy.optimize as opt
import scipy.fft as fft
import time

initial_h_values = np.array([...])  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = np.zeros(0)
    best_n = 0
    
    def score(h):
        nonlocal best_c5
        if len(h) == 0:
            return np.inf
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)
        return c5
    
    def initial_candidate(parent_h, new_n):
        if len(parent_h) != 47:
            return np.zeros(new_n)
        freq = fft.fft(parent_h)
        idx = np.abs(freq) > 0.9 * np.max(np.abs(freq))
        freq[idx] *= np.random.uniform(0.8, 1.2)
        freq[~idx] = 0
        new_h = fft.ifft(freq).real
        new_h = np.clip(new_h, 0, 1)
        new_h = np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, len(new_h)), new_h)
        new_h = np.round(new_h * new_n).astype(float) / new_n
        new_sum = np.sum(new_h)
        if np.abs(new_sum - new_n / 2) > 1e-3:
            new_h = np.clip(new_h + (new_n / 2 - new_sum) / new_n, 0, 1)
        return new_h
    
    def local_optimize(h, target_n):
        def obj(x):
            h_x = x.reshape(-1)
            corr = np.correlate(h_x, 1 - h_x, mode='full')
            return np.max(corr) * (2.0 / len(h_x))
        bounds = [(0, 1) for _ in range(target_n)]
        result = opt.minimize(
            obj,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x
    
    for n in [74, 94]:
        if time.time() - start_time > budget_s - 1:

        score(h)
