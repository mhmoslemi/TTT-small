import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in kwargs:
        initial_h = kwargs['initial_h_values']
    else:
        n_points = 45
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 1.0
        initial_h[n_points//2:] = 0.0
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return (np.max(conv) * dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1)] * n_points
    x0 = initial_h.copy()
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    options = {'maxiter': 300, 'ftol': 1e-6, 'disp': False}

    start_time = time.time()
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options=options)

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        return h * scale

    best_h = repair(result.x)
    c5_bound = objective(best_h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, c5_bound, n_points
    return best_h, c5_bound, n_points
