        scale_factor = (n_points / 2) / total if total != 0 else 1.0
        h_repaired = h_new * scale_factor
        h_repaired = np.clip(h_repaired, 0.0, 1.0)
        total_repaired = np.sum(h_repaired)
        if abs(total_repaired - n_points / 2) > 1e-5:
            scale_factor = (n_points / 2) / total_repaired
            h_repaired = h_repaired * scale_factor
            h_repaired = np.clip(h_repaired, 0.0, 1.0)
        current_c5 = _evaluate_c5(h_repaired, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_repaired.copy()
        if time.time() - start_time > budget_s:
            break
    
    return best_h.tolist(), best_c5, n_points

def _evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
