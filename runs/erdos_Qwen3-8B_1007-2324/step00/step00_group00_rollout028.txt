    dx = 2.0 / n_points

    def objective(x):
        x = np.array(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons_eq = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    cons_ineq_low = [{'type': 'ineq', 'fun': lambda x: x}]
    cons_ineq_high = [{'type': 'ineq', 'fun': lambda x: 1 - x}]
    cons = cons_eq + cons_ineq_low + cons_ineq_high

    start_time = time.time()
    result = opt.minimize(
        objective, h_initial, method='SLSQP', constraints=cons,
        tol=1e-8, options={'maxiter': 200, 'ftol': 1e-9}
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_initial, 0.38085, n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
