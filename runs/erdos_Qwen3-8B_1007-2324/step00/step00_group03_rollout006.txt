    corr = np.correlate(h, 1 - h, mode='full')
    return np.argmax(corr)

def calculate_gradient_vec(h, d, dx):
    n = len(h)
    term1 = np.zeros(n)
    term1[d:] = 1.0 - h[d:]
    term1[:d] = 1.0
    term2 = np.zeros(n)
    term2[d:] = h[:n - d]
    return dx * (term1 - term2)

def clip_and_project(h, n):
    h_clip = np.clip(h, 0, 1)
    current_sum = np.sum(h_clip)
    if current_sum == 0:
        return np.zeros(n)
    scale = (n / 2) / current_sum
    return h_clip * scale

def run(seed=42, budget_s=100):
    h = initial_h_values()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_cost(h)
    learning_rate = 0.01
    start_time = time.time()
    prev_best_c5 = best_c5
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        d = find_max_shift(h)
        grad = calculate_gradient_vec(h, d, 2.0 / n)
        h_new = h - learning_rate * grad
        h = clip_and_project(h_new, n)
        current_c5 = compute_cost(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if abs(current_c5 - prev_best_c5) < 1e-6:
            break
        prev_best_c5 = current_c5
    return best_h, best_c5, n
