import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use initial_h_values if available, else uniform distribution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_initial = np.array(initial_h_values)

        h_initial = np.ones(n_points) * target_sum / n_points
    
    def compute_objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        len_o = len(h)
        # Compute cross-correlation with 1-h
        corr = np.zeros(2 * len_o - 1)
        for lag in range(-(len_o-1), len_o):
            shifted = np.roll(1 - h, lag)
            corr[lag + len_o - 1] = np.dot(h, shifted)
        c5 = np.max(corr) * dx
        return c5
    
    def project_to_simplex(h):
        h_proj = np.clip(h, 0, 1)
        total = np.sum(h_proj)
        if abs(total - target_sum) > 1e-6:
            scale = target_sum / total
            h_proj *= scale
        return h_proj
    
    def perturb_solution(h):
        h = h.copy()

            # Global change: random single point

            h[idx] = new_val
        else:
            # Local refinement: adjust two points
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            val1, val2 = h[idx1], h[idx2]
            ratio = (target_sum - np.sum(h)) / n_points
            if random.random() < 0.5:
                # Increase
                h[idx1] = min(1.0, val1 + ratio)
                h[idx2] = min(1.0, val2 + ratio)
            else:
                # Decrease
                h[idx1] = max(0.0, val1 - ratio)
                h[idx2] = max(0.0, val2 - ratio)
        return project_to_simplex(h)
    
    best_h = project_to_simplex(h_initial)
