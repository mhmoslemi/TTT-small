import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in kwargs:
        initial_h = kwargs['initial_h_values']
        n_points = len(initial_h)
    else:
        n_points = 45
        initial_h = np.full(n_points, 0.5)
    
    def repair_full(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            scale = 0
        else:
            scale = (n_points / 2) / total
        h = h * scale
        return h
    
    initial_h = repair_full(initial_h)
    
    def objective(h):
        cross = np.correlate(h, 1 - h, mode='full')
        return np.max(cross) * (2.0 / n_points)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    best_h = repair_full(result.x)
    cross = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cross) * (2.0 / n_points)
    
    return (best_h, c5_bound, n_points)
