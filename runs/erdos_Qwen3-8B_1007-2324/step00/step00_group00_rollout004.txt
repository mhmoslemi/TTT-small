import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h with parent input if available or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(n_points, 1.0 / n_points * target_sum)
    
    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 1.0 / n_points * target_sum)
        scale = target_sum / total
        return h * scale
    
    h = project(h)
    
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Small perturbation
        idx = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new = h.copy()
        h_new[idx] += delta
        h_new = project(h_new)
        current_c5 = compute_c5(h_new)
        if current_c5 < best_c5:
            best_h, best_c5 = h_new, current_c5
            h = h_new
        
        # Large perturbation with 10% chance
        if np.random.random() < 0.1:
            num_reset = np.random.randint(1, 6)
            indices = np.random.choice(n_points, num_reset, replace=False)
            h_reset = h.copy()
            for idx in indices:
                h_reset[idx] = np.random.uniform(0, 1)

            current_c5 = compute_c5(h_reset)
            if current_c5 < best_c5:
