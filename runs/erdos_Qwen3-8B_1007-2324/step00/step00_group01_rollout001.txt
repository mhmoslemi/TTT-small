import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', np.random.rand(47))
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for _ in range(10):
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = initial_h_values + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_clipped = np.sum(h_perturbed)
        h_perturbed *= target_sum / sum_clipped
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            sum_opt = np.sum(h_opt)
            h_opt *= target_sum / sum_opt
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        if time.time() - start_time > budget_s * 0.8:
            break

    c5_initial = objective(initial_h_values)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = initial_h_values

    return (best_h, best_c5, n_points)
