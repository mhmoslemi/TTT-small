    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.01, 0.01)
        new_val = np.clip(h[i] + delta, 0.0, 1.0)
        change = new_val - h[i]
        j = np.random.choice(n_points, size=1, replace=False)[0]
        while j == i:
            j = np.random.choice(n_points, size=1, replace=False)[0]
        new_j = np.clip(h[j] - change, 0.0, 1.0)
        new_h = h.copy()
        new_h[i] = new_val
        new_h[j] = new_j
        corr = correlate(new_h, 1 - new_h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
        h = new_h
    
    return (best_h, best_c5, n_points)
