def compute_cost(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, d, dx):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        if j + d < n:
            term1 = 1 - h[j + d]
        else:
            term1 = 1
        if j - d >= 0:
            term2 = h[j - d]
        else:
            term2 = 0
        grad[j] = dx * (term1 - term2)
    return grad

def upsample(prev_h, new_n):
    old_n = len(prev_h)
    scale = old_n / new_n
    repeats = new_n // old_n
    upsampled = np.repeat(prev_h, repeats)
    upsampled *= scale
    return upsampled
