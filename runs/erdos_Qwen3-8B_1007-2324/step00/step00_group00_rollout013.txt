import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Get initial vector
    initial_vector = initial_h_values  # Assume it's available
    if initial_vector is None:
        initial_vector = np.full(n_points, 0.5)

    def project_to_simplex(h):
        h_proj = np.clip(h, 0.0, 1.0)
        total_proj = np.sum(h_proj)
        if total_proj == 0:
            return np.zeros_like(h)
        scale = target_sum / total_proj
        return h_proj * scale

    h0 = project_to_simplex(initial_vector)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    start_time = time.time()
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h0, objective(h0), n_points

    final_h = result.x
    final_c5 = objective(final_h)
    return final_h, final_c5, n_points
