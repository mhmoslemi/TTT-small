        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0.0, 1.0)
            current_sum = np.sum(current_h)
            if abs(current_sum - target_sum) > 1e-6:
                current_h = current_h * target_sum / current_sum
            current_c5 = compute_c5(current_h, n)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n = n
    
    return (best_h, best_c5, best_n)
