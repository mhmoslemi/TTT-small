    def optimize(initial_guess):
        def obj(h):

            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0., 1.)] * len(initial_guess)
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
        best_c5 = float('inf')

        for trial in range(10):

        x = np.linspace(0, 2, n_points)
        dist = np.abs(x - 1)
        h_tri = 1 - dist
        h_tri = np.clip(h_tri, 0, 1)
        h_opt_initial_1 = h_tri[:n_half]
        h_opt_initials.append(h_opt_initial_1)

    x_radial = np.linspace(0, 2, n_points)
    h_radial = (1.0 + np.cos(np.pi * x_radial)) / 2.0

            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0., 1.)] * len(initial_guess)

        best_c5 = float('inf')

            h_start = initial_guess + perturb
            h_clipped = np.clip(h_start, 0, 1)
            h_scaled = h_clipped * (integral_target / np.sum(h_clipped) if np.sum(h_clipped) != 0 else 1)
            res = minimize(obj, h_scaled, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
