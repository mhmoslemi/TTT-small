    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum(h_upper)

    if 'initial_h_values' in globals():
        initial_upper = np.array(initial_h_values)[:n_half]
        sum_upper = np.sum(initial_upper)
        scale = integral_target / sum_upper if sum_upper != 0 else 1.0
        initial_upper = initial_upper * scale
    else:
        np.random.seed(seed)
        initial_upper = np.random.rand(n_half) * 0.5 + 0.25
        sum_upper = np.sum(initial_upper)
        scale = integral_target / sum_upper if sum_upper != 0 else 1.0
        initial_upper = initial_upper * scale

    def objective(h_upper):
        h_full = np.concatenate([h_upper, h_upper[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint(h_upper):
        return np.sum(h_upper) - integral_target

    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    best_upper = initial_upper.copy()
    best_c5 = objective(best_upper)

    result = minimize(objective, best_upper, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8})
    if result.success:
        best_upper = result.x
        best_c5 = objective(best_upper)

    h_full = np.concatenate([best_upper, best_upper[::-1]])
    h_full = np.clip(h_full, 0, 1)
    sum_h = np.sum(h_full) * dx
    if np.abs(sum_h - 1.0) > 1e-6:
        scale = 1.0 / sum_h
        h_full *= scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_upper, best_upper[::-1]]), best_c5, n_points

    return h_full, best_c5, n_points
