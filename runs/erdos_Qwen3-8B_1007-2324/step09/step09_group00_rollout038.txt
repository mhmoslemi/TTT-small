    best_c5 = compute_c5(best_h_opt)
    current_h_opt = best_h_opt.copy()
    
    T0 = 0.1
    cooling_factor = 0.995
    max_iter = 10000
    iter_count = 0
    
    while (time.time() - start_time) < budget_s:
        iter_count += 1
        T = T0 * (cooling_factor ** iter_count)
        # Perturb current_h_opt
        noise = np.random.normal(0, 0.05, n_half)
        new_h_opt = current_h_opt + noise
        new_h_opt = np.clip(new_h_opt, 0.0, 1.0)
        sum_new = np.sum(new_h_opt)
        if sum_new != 0:
            new_h_opt = new_h_opt * (integral_target / sum_new)
        else:
            new_h_opt = np.zeros_like(new_h_opt)
        # Compute c5 for new_h_opt
        c5_candidate = compute_c5(new_h_opt)
        # Acceptance
        delta = c5_candidate - best_c5
        if delta < 0:
            current_h_opt = new_h_opt
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h_opt = current_h_opt.copy()
        else:
            p = np.exp(-delta / T)
            if np.random.rand() < p:
                current_h_opt = new_h_opt
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h_opt = current_h_opt.copy()
    
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    h_full = np.clip(h_full, 0.0, 1.0)
    sum_full = np.sum(h_full) * dx
    if abs(sum_full - 1) > 1e-6:
        h_full = h_full / sum_full * 1
    return (h_full, best_c5, n_points)
