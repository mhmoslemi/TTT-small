        noise = np.random.normal(0, 0.008, len(h_opt_start))
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        if np.sum(h_opt_start) > 0:
            h_opt_start *= (integral_target / np.sum(h_opt_start))
        try:
            result = minimize(
                objective,
                h_opt_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-8, 'disp': False}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
