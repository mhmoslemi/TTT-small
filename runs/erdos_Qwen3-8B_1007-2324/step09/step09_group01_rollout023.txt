from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()

    integral_target = n_points / 2  # sum(h) * dx = 1
    
    def init_from_parent(h_parent, n_half):
        h_full = np.concatenate([h_parent[:n_half], h_parent[:n_half][::-1]])
        sum_val = np.sum(h_full)
        scale = integral_target / sum_val if sum_val > 0 else 1
        return h_full * scale
    
    h_initial = np.array([])
    if 'initial_h_values' in globals():
        n_half = n_points // 2
        h_initial = init_from_parent(np.array(initial_h_values), n_half)
    else:
        n_half = n_points // 2
        h_half = np.zeros(n_half)
        start_idx = int(35)  # For n_points=280, 35th index
        end_idx = start_idx + 70
        h_half[start_idx:end_idx] = 1.0
        h_initial = np.concatenate([h_half, h_half[::-1]])
    
    def build_h(params):
        h_vec = np.zeros(n_half)
        h_vec[:] = params
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        return h_full
    
    def objective(params):
        h_full = build_h(params)
        h_full = np.clip(h_full, 0, 1)

        integral = np.sum(h_full) * dx
        return integral - integral_target

    result = differential_evolution(
