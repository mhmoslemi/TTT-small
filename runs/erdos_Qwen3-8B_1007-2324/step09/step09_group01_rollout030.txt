        return True

    n_breaks = 3  # 3 breaks = 4 segments
    initial_params = np.random.rand(n_breaks + n_breaks) * 2
    initial_params[n_breaks:] = np.random.rand(n_breaks)
    initial_params[n_breaks:] = np.clip(initial_params[n_breaks:], 0, 1)
    initial_params[:n_breaks] = np.sort(np.random.rand(n_breaks) * 2)
    initial_params[:n_breaks] = np.clip(initial_params[:n_breaks], 0, 2)

    if not is_valid(initial_params):
        initial_params[:n_breaks] = np.linspace(0, 2, n_breaks + 1)[1:-1]

    bounds = [(0.0, 2.0) for _ in range(n_breaks)]
    bounds += [(0.0, 1.0) for _ in range(n_breaks)]

    result = differential_evolution(
        objective,
        bounds,
        constraints=[('eq', constraint)],
        disp=False,
        popsize=15,
        maxiter=40,
        tol=1e-5,
        mutation=(0.8, 1.0),
        recombination='blend',
        polish=True,
        callback=lambda x: None
    )

    best_params = result.x
    best_h = build_h(best_params[:n_breaks], best_params[n_breaks:])

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), result.fun, n_points

    return np.concatenate([best_h, best_h[::-1]]), result.fun, n_points
