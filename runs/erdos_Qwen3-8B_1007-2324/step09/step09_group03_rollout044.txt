import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    n_half = n_points // 2
    dx_full = 2.0 / n_points
    integral_target = n_points / 4  # sum(h) * dx_full = 0.5 => sum(h) = 0.5 / dx_full = n_points / 4

    # Generate initial guesses
    def generate_initial():
        x = np.linspace(0, 1, n_half)
        # Triangular
        h_tri = 1.0 - x
        sum_tri = np.sum(h_tri)
        h_tri_scaled = h_tri * integral_target / sum_tri
        # Radial
        h_rad = (np.cos(np.pi * x) + 1) / 2
        sum_rad = np.sum(h_rad)
        h_rad_scaled = h_rad * integral_target / sum_rad
        return h_tri_scaled, h_rad_scaled

    h_tri_init, h_rad_init = generate_initial()

    def obj(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx_full

    bounds = [(0., 1.)] * n_half
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = h_tri_init.copy()

    # Optimize triangular
    np.random.seed(seed)
    h_start_tri = h_tri_init.copy()
    res_tri = minimize(obj, h_start_tri, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
    if res_tri.success and res_tri.fun < best_c5:
        best_c5 = res_tri.fun
        best_h = res_tri.x

    # Optimize radial
    np.random.seed(seed + 1)
    h_start_rad = h_rad_init.copy()
    res_rad = minimize(obj, h_start_rad, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
    if res_rad.success and res_rad.fun < best_c5:
        best_c5 = res_rad.fun
        best_h = res_rad.x

    # Further refinement
    trials = 5
    for i in range(trials):
        if time.time() - start > budget_s - 1:
