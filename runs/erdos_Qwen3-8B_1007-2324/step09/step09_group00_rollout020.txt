import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4.0

    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_half].copy()
    else:
        h_opt = np.random.rand(n_half)
        sum_h = np.sum(h_opt)
        h_opt = h_opt * (integral_target / sum_h)

    def objective(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = objective(h_opt)
    best_h_opt = h_opt.copy()

    T = 0.5
    cooling_rate = 0.995
    max_iter = 1000

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
        noise = np.random.normal(0, 0.02, n_half)
        h_new = np.clip(h_opt + noise, 0, 1)
        sum_h_new = np.sum(h_new)
        if sum_h_new > 0:
            h_new = h_new * (integral_target / sum_h_new)
        new_c5 = objective(h_new)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h_opt = h_new.copy()
        else:
            delta = best_c5 - new_c5
            if np.random.rand() < np.exp(-delta / T):
                best_c5 = new_c5
                best_h_opt = h_new.copy()
        T *= cooling_rate
        h_opt = best_h_opt

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points

    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
