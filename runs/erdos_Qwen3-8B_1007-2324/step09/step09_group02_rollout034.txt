import numpy as np
from scipy.optimize import dual_annealing, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4  # sum of the first half

    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values[:n_half]
        sum_initial = np.sum(initial_guess)
        if sum_initial != 0:
            initial_guess = initial_guess * (target_sum / sum_initial)
    else:
        np.random.seed(seed)
        initial_guess = np.random.rand(n_half)
        initial_guess = initial_guess * (target_sum / np.sum(initial_guess))

    # Define the objective function
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Define the constraint for the sum
    def constraint_sum(x):
        return np.sum(x) - target_sum

    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    # Use dual_annealing with the constraints
    result = dual_annealing(
        objective,
        bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        x0=initial_guess,
        maxfun=1000,
        disp=False
    )

    # Extract the best solution
    best_x = result.x
    best_c5 = objective(best_x)

    # Perform local refinement with SLSQP
    def objective_local(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result_local = minimize(
        objective_local,
        best_x,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200}
    )
    best_x = result_local.x
    best_c5 = objective_local(best_x)

    # Ensure the solution is symmetric and meets the constraints
    h_full = np.concatenate([best_x, best_x[::-1]])
    sum_h = np.sum(h_full) * dx
    if not np.isclose(sum_h, 1.0, atol=1e-3):
        scale = 1.0 / sum_h
        h_full = h_full * scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_x, best_x[::-1]]), best_c5, n_points

    return h_full, best_c5, n_points
