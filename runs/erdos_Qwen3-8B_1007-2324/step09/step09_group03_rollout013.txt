import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    n_base = 140
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_base]
    else:
        x = np.linspace(0, 2, n_base)
        dist = np.abs(x - 1)
        h = 1 - dist
        h = np.clip(h, 0, 1)
        h_opt_initial = h[:n_base // 2]

    # Radial initial guess
    x_radial = np.linspace(0, 2, n_base)
    k = 20.0
    radial = np.exp(-k * (x_radial - 1)**2)
    radial = np.clip(radial, 0, 1)
    h_radial_initial = radial[:n_base // 2]

    def generate_candidate(base_h):
        h_full = np.concatenate([base_h, base_h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        scale = integral_target / np.sum(h)
        return h * scale

    def optimize_candidate(base_h, name='triangular'):
        bounds = [(0., 1.)] * len(base_h)
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
        h_start = base_h.copy()
        np.random.seed(seed + 100 + hash(name))
        perturb = np.random.normal(0, 0.01, len(base_h))
        h_start = repair(h_start + perturb)
        res = minimize(generate_candidate, h_start, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 200})
        return res.x, res.fun

    best_c5 = float('inf')
    best_h = h_opt_initial.copy()

    trials = 3
    for i in range(trials):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + i)
        h1, c1 = optimize_candidate(h_opt_initial, 'triangular')
        h2, c2 = optimize_candidate(h_radial_initial, 'radial')
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
