        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    temperature = T0
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.02, len(h_opt))
        h_candidate = h_opt + noise
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate == 0:
            continue
        scaling = integral_target / sum_candidate
        h_candidate = h_candidate * scaling
        new_c5 = objective(h_candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h_opt = h_candidate.copy()
        else:
            delta = best_c5 - new_c5
            prob = np.exp(-delta / temperature)
            if np.random.rand() < prob:
                best_c5 = new_c5
                best_h_opt = h_candidate.copy()
        temperature *= cooling_rate

    best_h_opt = np.clip(best_h_opt, 0, 1)
    sum_best = np.sum(best_h_opt)
    if sum_best != 0:
        best_h_opt *= (integral_target / sum_best)
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
