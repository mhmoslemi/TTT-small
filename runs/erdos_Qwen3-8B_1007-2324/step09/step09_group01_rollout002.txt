from scipy.optimize import minimize
import time

        h_half[starts[1]:] = 0.5
        h_initial_full = np.concatenate([h_half, h_half[::-1]])
    
    def build_symmetric_h(h_vec):
        return np.concatenate([h_vec, h_vec[::-1]])
    
    def objective(h_vec):
        h_full = np.clip(build_symmetric_h(h_vec), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(build_symmetric_h(x)) - integral_target
    
    # Initial guess from parent
