import numpy as np
from scipy.optimize import dual_annealing
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    sum_half_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_half = initial_h[:n_points//2]
    else:
        np.random.seed(seed)
        h_half = np.random.rand(n_points//2)
        sum_half = np.sum(h_half)
        h_half = h_half * sum_half_target / sum_half if sum_half != 0 else h_half

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - sum_half_target

    bounds = [(0.0, 1.0) for _ in range(len(h_half))]

    result = dual_annealing(objective, bounds, constraints=[{'type': 'eq', 'fun': constraint}],
                            x0=h_half, maxfun=1000, disp=False, tol=1e-6)

    best_h_half = result.x
    best_c5 = objective(best_h_half)
    h_full = np.clip(np.concatenate([best_h_half, best_h_half[::-1]]), 0, 1)
    sum_full = np.sum(h_full) * dx
    if not np.isclose(sum_full, 1.0):
        h_full *= 1.0 / sum_full

    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
