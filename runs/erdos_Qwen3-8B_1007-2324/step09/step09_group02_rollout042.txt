import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    n_half = n_points // 2
    integral_target = n_points / 4  # 70 for n_points=280

    if 'initial_h_values' in globals():
        initial_guess = np.array(initial_h_values[:n_half])
    else:
        np.random.seed(seed)
        initial_guess = np.random.rand(n_half) * 0.5 + 0.25  # Random initial guess
        # Adjust to meet integral constraint
        if np.sum(initial_guess) != 0:
            initial_guess = initial_guess * (integral_target / np.sum(initial_guess))
        else:
            initial_guess = np.full(n_half, 1.0 / n_half)

    def objective(x):
        h_half = x
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def grad_objective(x):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            obj_plus = objective(x_plus)
            x_minus = x.copy()
            x_minus[i] -= eps
            obj_minus = objective(x_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], jac=grad_objective, tol=1e-6, options={'maxiter': 300})

    best_x = result.x
    best_c5 = objective(best_x)
    best_h_full = np.concatenate([best_x, best_x[::-1]])
    best_h_full = np.clip(best_h_full, 0, 1)
    sum_full = np.sum(best_h_full) * dx
    if not np.isclose(sum_full, 1.0, atol=1e-3):
        scale = 1.0 / sum_full
        best_h_full *= scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([initial_guess, initial_guess[::-1]]), objective(initial_guess), n_points
    return best_h_full, best_c5, n_points
