from numpy.fft import fft, ifft

def compute_cross_correlation(h):
    n = len(h)
    # Compute the cross-correlation using FFT
    # This is equivalent to the convolution of h with reversed h
    h_fft = fft(h, n * 2)
    reversed_h = h[::-1]

    cross_corr_fft = h_fft * reversed_h_fft
