        h_opt_start = best_h_opt.copy()
        noise = np.random.normal(0, 0.01, n_half)
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        sum_start = np.sum(h_opt_start)
        if sum_start > 0:
            h_opt_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_opt_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h_opt = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
    
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
