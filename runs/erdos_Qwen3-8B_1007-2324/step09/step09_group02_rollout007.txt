    final_h_values = np.concatenate([final_solution, final_solution[::-1]])
    final_h_values = np.clip(final_h_values, 0, 1)
    sum_h = np.sum(final_h_values) * dx
    if abs(sum_h - 1) > 1e-3:
        scale = 1.0 / sum_h
        final_h_values *= scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best, best[::-1]]), best_c5, n_points

    return final_h_values, best_c5, n_points
