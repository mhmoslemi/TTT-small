import numpy as np
from scipy.optimize import dual_annealing
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = 140
    dx = 2.0 / n_points
    integral_target = 140  # sum(x) * dx = 1 => sum(x) = 140

    # Initialize the initial guess
    if 'initial_h_values' in globals():
        initial_x = np.array(initial_h_values)[:n_half]
    else:
        np.random.seed(seed)
        initial_x = np.random.rand(n_half)
        sum_initial = np.sum(initial_x)
        if sum_initial != 0:
            initial_x = initial_x * integral_target / sum_initial

    # Objective function
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Constraint function
    def constraint(x):
        return np.sum(x) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    cons = [{'type': 'eq', 'fun': constraint}]

    # Run dual annealing optimization
    result = dual_annealing(objective, bounds, constraints=cons, x0=initial_x, maxfun=500)

    # Extract optimal solution
    optimal_x = result.x
    h_full = np.concatenate([optimal_x, optimal_x[::-1]])

    # Scale to ensure integral constraint
    sum_h = np.sum(h_full) * dx
    scale = 1.0 / sum_h if abs(sum_h) > 1e-10 else 1.0
    h_full *= scale

    # Recompute C5 after scaling
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx

    # Check time budget
    if time.time() - start_time > budget_s - 1:
        return np.concatenate([initial_x, initial_x[::-1]]), c5_bound, n_points

    return h_full, c5_bound, n_points
