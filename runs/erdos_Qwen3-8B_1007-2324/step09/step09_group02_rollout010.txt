import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = 140

    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h_full = np.array(initial_h_values)
        h_half = h_full[:n_half]
    else:
        np.random.seed(seed)
        h_half = np.random.rand(n_half) * 0.5 + 0.25
        sum_current = np.sum(h_half)
        h_half = h_half * (integral_target / sum_current) if sum_current != 0 else h_half

    # Ensure symmetry and integral constraint
    h_full = np.concatenate([h_half, h_half[::-1]])
    sum_current = np.sum(h_full)
    if sum_current != 0:
        h_full = h_full * (integral_target / sum_current)

    # Local optimization to minimize the maximum overlap
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(np.concatenate([x, x[::-1]])) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    cons = [{'type': 'eq', 'fun': constraint}]
    result = minimize(objective, h_half, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 300})
    best_solution = result.x
    best_c5 = objective(best_solution)

    # Final refinement
    result = minimize(objective, best_solution, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})

    # Ensure solution meets integral constraint and bounds
    h_full = np.clip(np.concatenate([best_solution, best_solution[::-1]]), 0, 1)
    sum_current = np.sum(h_full)
    if sum_current != 0:

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_solution, best_solution[::-1]]), best_c5, n_points
