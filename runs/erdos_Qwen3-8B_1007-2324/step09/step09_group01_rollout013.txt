import numpy as np
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        # Create symmetric two-interval step function
        n_half = n_points // 2
        h_half = np.zeros(n_half)
        start_idx = int(35)  # For n_points=280, 35th index
        end_idx = start_idx + 70
        h_half[start_idx:end_idx] = 1.0
        h_initial = np.concatenate([h_half, h_half[::-1]])
    
    def objective(h_vec):
        h_full = h_vec.copy()
        h_full = np.clip(h_full, 0, 1)
        sum_val = np.sum(h_full)
        scale = (integral_target) / sum_val if sum_val > 0 else 1
        h_full = h_full * scale
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # Basinhopping with L-BFGS-B as local minimizer
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = basinhopping(
        objective,
        h_initial,
        niter=50,
        T=1.0,
        stepsize=0.5,
        minimizer_kwargs={
            'method': 'L-BFGS-B',
            'bounds': bounds,
            'ftol': 1e-8,
            'disp': False
        }
    )
    
    best_h_opt = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
