import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    S = n_points / 2  # sum(h) must be S

    # Initialize h_half
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_half = initial_h[:n_half]
    else:
        np.random.seed(seed)
        h_half = np.random.rand(n_half) * 0.5 + 0.25
        sum_half = np.sum(h_half)
        if sum_half != 0:
            h_half = h_half * (S / sum_half)

    # Objective function with penalty
    def obj(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        penalty = 0 if np.abs(np.sum(h_full) - S) < 1e-2 else 1e6 * np.abs(np.sum(h_full) - S)
        return c5 + penalty

    # Bounds for h_half
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    # Differential evolution
    res = differential_evolution(obj, bounds, strategy='best1bin', popsize=30, tol=1e-4, mutation=(0.5, 0.9), recombination=0.8, seed=seed, polish=True, disp=False)

    best_h = res.x
    best_c5 = obj(best_h)

    # Local refinement with SLSQP
    def refine_obj(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def refine_constr(x):
        return np.sum(np.concatenate([x, x[::-1]])) - S

    cons = [{'type': 'eq', 'fun': refine_constr}]
    res_refine = minimize(refine_obj, best_h, method='SLSQP', bounds=[(0., 1.)]*n_half, constraints=cons, options={'maxiter': 200, 'ftol': 1e-7})
    best_h = res_refine.x
    best_c5 = refine_obj(best_h)

    # Final sanity check and normalization
    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0., 1.)
    sum_full = np.sum(h_full)
    if np.abs(sum_full - S) > 1e-2:
        scale = S / sum_full
        h_full = h_full * scale
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
    return h_full, best_c5, n_points
