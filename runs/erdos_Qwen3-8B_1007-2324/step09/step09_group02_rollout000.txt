        sum_half = np.sum(h_half)
        h_half = h_half * (140.0 / sum_half) if sum_half != 0 else h_half

    # Objective to minimize max correlation
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Integral constraint
    def constraint(x):
        return np.sum(np.concatenate([x, x[::-1]])) - 140.0

    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(len(h_half))]

    # Initial optimization with SLSQP
    result = minimize(objective, h_half, method='SLSQP',
                     constraints=[{'type': 'eq', 'fun': constraint}],
                     bounds=bounds, options={'maxiter': 200, 'ftol': 1e-9})

    # Refine further with local search
    if result.success:

                return np.max(corr) * dx
