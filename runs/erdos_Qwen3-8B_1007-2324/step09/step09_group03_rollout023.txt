def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum h_vec = 70.0

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_half]
        # Generate step function initial guess
        step_initial = np.zeros(n_half)
        step_initial[70:] = 1.0
        initial_guesses = [h_opt_initial, step_initial]
    else:
        # Generate triangular and step initial guesses
        x = np.linspace(0, 1, n_half)
        dist = np.abs(x - 0.5)
        h_tri = 1 - dist
        h_tri = np.clip(h_tri, 0, 1)
        step_initial = np.zeros(n_half)
        step_initial[70:] = 1.0
        initial_guesses = [h_tri, step_initial]

    def obj(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.)] * n_half
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = np.zeros(n_half)

    trials_per_guess = 10
    for initial_guess in initial_guesses:
        for i in range(trials_per_guess):
            if time.time() - start > budget_s - 1:
                break
            np.random.seed(seed + i)
            perturb = np.random.normal(0, 0.02, n_half)
            h_start = np.clip(initial_guess + perturb, 0, 1)
            h_start = (h_start * integral_target) / (np.sum(h_start) + 1e-12)
            res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
