from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_half_target = n_points / 4

    sum_final = np.sum(final_h_values) * dx
