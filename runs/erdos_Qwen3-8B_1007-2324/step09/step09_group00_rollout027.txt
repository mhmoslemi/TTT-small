import numpy as np
import time

def evaluate_c5(h_opt, n_points, dx):
    h_full = np.concatenate([h_opt, h_opt[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140 if 'initial_h_values' in globals() else 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4

    if 'initial_h_values' in globals():

    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt_initial = h[:n_half]

    best_h_opt = h_opt_initial.copy()
    best_c5 = evaluate_c5(best_h_opt, n_points, dx)

    T0 = 0.15
    cooling_rate = 0.995
    max_iter = 1000
    noise_scale = 0.03
    iterations = int((budget_s - 1) * 100)

    for i in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = best_h_opt.copy()
        noise = np.random.normal(0, noise_scale, n_half)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new == 0:
            continue
        scale = target_sum / sum_new
        h_new = h_new * scale

        current_c5 = evaluate_c5(h_new, n_points, dx)
        delta = current_c5 - best_c5
        if np.random.rand() < np.exp(-delta / (T0 * (cooling_rate ** i))):
            best_h_opt = h_new
            best_c5 = current_c5

    best_h_opt = np.clip(best_h_opt, 0, 1)
    sum_best = np.sum(best_h_opt)

    final_c5 = evaluate_c5(best_h_opt, n_points, dx)
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return h_full, final_c5, n_points
