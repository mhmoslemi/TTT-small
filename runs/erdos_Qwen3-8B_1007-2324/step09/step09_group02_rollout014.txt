import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1 => sum(h) = 140

    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_half = np.array(initial_h_values)[:n_half]
    else:
        np.random.seed(seed)
        h_half = np.random.rand(n_half) * 0.5 + 0.25
        sum_half = np.sum(h_half)
        h_half = h_half * (integral_target / sum_half) if sum_half != 0 else h_half

    # Define full vector for objective and constraints
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    # Constraint: integral constraint
    def constraint(x):
        return np.sum(np.concatenate([x, x[::-1]])) - integral_target

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    # Optimize using SLSQP with tight settings
    result = minimize(
        objective,
        h_half,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    best_h = result.x if result.success else h_half

    # Ensure bounds and integral
    h_full = np.concatenate([best_h, best_h[::-1]])
    h_full = np.clip(h_full, 0, 1)
    sum_full = np.sum(h_full) * dx
    if abs(sum_full - 1) > 1e-5:
        scale = 1.0 / sum_full
        h_full *= scale

    # Final evaluation
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    c5 = np.max(corr)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), c5, n_points

    return h_full, c5, n_points
