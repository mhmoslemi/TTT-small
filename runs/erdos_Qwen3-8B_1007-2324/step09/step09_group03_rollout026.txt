import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # Corrected from previous error

    # Use initial_h_values if provided, else generate a simple one
    if 'initial_h_values' in globals():

    else:
        x = np.linspace(0, 2, n_points)
        dist = np.abs(x - 1)
        h = 1 - dist
        h = np.clip(h, 0, 1)
        h_opt_initial = h[:n_half]

    def obj(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.)] * n_half
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = h_opt_initial.copy()

    trials = 30  # Increased number of trials for more exploration
    for i in range(trials):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + i)
        # Use different perturbation strategies for diversity
        if i % 2 == 0:
            perturb = np.random.normal(0, 0.05, n_half)  # Larger perturbations for exploration

            perturb = np.random.uniform(-0.1, 0.1, n_half)  # Uniform perturbation

        h_start = (h_start * integral_target) / (np.sum(h_start) + 1e-12)
