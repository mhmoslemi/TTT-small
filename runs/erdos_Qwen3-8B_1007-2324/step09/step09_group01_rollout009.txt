import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    def build_symmetric_h(h_vec):
        return np.concatenate([h_vec[:n_half], h_vec[:n_half][::-1]])
    
    if 'initial_h_values' in globals():
        h_initial_full = np.array(initial_h_values)
        h_initial = h_initial_full[:n_half]
    else:
        n_half = n_points // 2
        h_initial = np.zeros(n_half)
        start_idx = 35
        end_idx = start_idx + 70
        h_initial[start_idx:end_idx] = 1.0
    
    n_half = n_points // 2
    
    def objective(h_vec):
        h_full = build_symmetric_h(h_vec)
        sum_val = np.sum(h_full)
        scale = integral_target / sum_val if sum_val > 0 else 1
        h_full_scaled = h_full * scale
        h_full_scaled = np.clip(h_full_scaled, 0, 1)
        corr = np.correlate(h_full_scaled, 1 - h_full_scaled, mode='full')
        max_corr_val = np.max(corr)
        return max_corr_val * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    best_h_opt = h_initial.copy()
    best_c5 = objective(best_h_opt)
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=200,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
