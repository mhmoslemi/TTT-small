    def evaluate(individual):
        h_full = np.concatenate([individual, individual[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def tournament_selection(population, k=5):
        selected = np.random.choice(len(population), size=k, replace=False)
        best_idx = selected[0]
        best_val = evaluate(population[best_idx])
        for i in selected[1:]:
            current_val = evaluate(population[i])
            if current_val < best_val:
                best_val = current_val
                best_idx = i
        return population[best_idx]

    def crossover(a, b):
        child = np.copy(a)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = b[i]
        return child
