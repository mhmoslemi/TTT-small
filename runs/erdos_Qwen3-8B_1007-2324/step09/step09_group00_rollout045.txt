from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 256

    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.concatenate([initial_h[:n_half], initial_h[:n_half][::-1]])

        np.random.seed(seed)
        h_opt = np.random.rand(n_half)
        h_opt /= np.sum(h_opt)
        h_opt *= integral_target
        initial_h = h_opt
    
    def objective(h_opt):
        h_array = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{'type': 'eq', 'fun': lambda h_opt: np.sum(h_opt) - integral_target}]
    
    best_h_opt = initial_h.copy()
    best_c5 = objective(best_h_opt)
    
    trials = 30
    max_iter = 200
    
    for t in range(trials):

        try:
            result = minimize(
