            h_full_pert = build_symmetric_h(h_pert)
            corr_pert = np.correlate(h_full_pert, 1 - h_full_pert, mode='full')
            backward = np.max(corr_pert)
            grad[j] = (forward - backward) / (2 * eps)
        return grad * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    def constraint(x):
        return np.sum(build_symmetric_h(x)) - integral_target
    
    best_h_opt = h_initial_full[:n_half].copy()
    best_c5 = objective(best_h_opt)
    
    trials = 30
    max_iter = 200
    
    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_opt_start = best_h_opt.copy()
        # Add noise to global search
        noise = np.random.normal(0, 0.03, n_half)
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        # Ensure integral constraint
        sum_val = np.sum(h_opt_start)
        scale = integral_target / sum_val if sum_val > 0 else 1
        h_opt_start = h_opt_start * scale
        h_opt_start = np.clip(h_opt_start, 0, 1)
        cost = objective(h_opt_start)
        if cost < best_c5:
            best_c5 = cost
            best_h_opt = h_opt_start.copy()
    
    try:
        result = minimize(
            objective,
            best_h_opt,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            jac=grad_objective,
            options={'maxiter': max_iter, 'disp': False, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            best_c5 = result.fun
            best_h_opt = result.x.copy()
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
    
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
