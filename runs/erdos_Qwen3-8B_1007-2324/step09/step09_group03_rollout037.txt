import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum h_vec = 70.0

    # Generate initial guess based on available data or radial shape
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_half]
    else:
        # Radial shape: symmetric sine wave
        x = np.linspace(0, 1, n_half)
        h_radial = 0.5 + 0.5 * np.sin(np.pi * x)
        sum_radial = np.sum(h_radial)
        scaling_factor = (0.5 * n_half) / sum_radial
        h_opt_initial = h_radial * scaling_factor

    def obj(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.)] * n_half
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = h_opt_initial.copy()

    trials = 20
    for i in range(trials):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + i)
        # Perturb initial guess
        h_start = np.clip(h_opt_initial + np.random.normal(0, 0.02, n_half), 0, 1)
        h_start = (h_start * integral_target) / (np.sum(h_start) + 1e-12)
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 250})

            best_c5 = res.fun
            best_h = res.x

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
