        h_start_full = (h_start_full * integral_target) / (np.sum(h_start_full) + 1e-12)
        res_full = minimize(obj_full, h_start_full, method='SLSQP', bounds=bounds_full, constraints=cons_full, options={'maxiter': 100})
        if res_full.success and res_full.fun < best_c5:
            best_c5 = res_full.fun
            best_h = np.concatenate([res_full.x, res_full.x[::-1]])

    h_values = np.clip(best_h, 0, 1)
    assert np.isclose(np.sum(h_values)*dx, 1.0, atol=1e-6), "Integral check failed"
    return h_values, best_c5, n_points
