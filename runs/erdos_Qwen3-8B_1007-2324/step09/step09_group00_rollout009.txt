from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():

    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        initial_h = np.concatenate([h[:n_half], h[:n_half][::-1]])
    
    def objective(h_opt):
        h_array = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{'type': 'eq', 'fun': lambda h_opt: np.sum(h_opt) - integral_target}]
    
    best_h_opt = initial_h[:n_half]
    best_c5 = objective(best_h_opt)
    
    trials = 50
    max_iter = 200
    
    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_opt_start = best_h_opt.copy()
        noise = np.random.normal(0, 0.03, n_half)
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        sum_start = np.sum(h_opt_start)
        if sum_start > 0:
            h_opt_start *= (integral_target / sum_start)
        try:
            result = minimize(

                method='SLSQP',
                bounds=bounds,
                constraints=constraints,

            )
            if result.success:
                current_c5 = result.fun
