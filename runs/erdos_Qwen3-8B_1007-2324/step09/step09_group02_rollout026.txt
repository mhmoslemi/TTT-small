import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = integral_target

    # Initialize from provided initial values
    if 'initial_h_values' in globals():
        initial_half = np.array(initial_h_values)
    else:
        np.random.seed(seed)
        initial_half = np.random.rand(n_points // 2) * 0.5 + 0.25
        sum_half = np.sum(initial_half)
        if sum_half != 0:
            initial_half = initial_half * (integral_target / sum_half)

    # Define objective: minimize max correlation
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Define integral constraint
    def constraint_sum(x):
        return np.sum(np.concatenate([x, x[::-1]])) - integral_target

    # Bounds for each variable
    bounds = [(0.0, 1.0) for _ in range(len(initial_half))]

    # Constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Optimizer settings
    opts = {
        'ftol': 1e-8,
        'maxiter': 300,
        'disp': False
    }

    # Run optimization
    result = minimize(objective, initial_half, method='SLSQP', bounds=bounds, constraints=cons, options=opts)

    # Handle optimization success
    final_half = result.x if result.success else initial_half
    final_h_values = np.concatenate([final_half, final_half[::-1]])
    final_h_values = np.clip(final_h_values, 0, 1)

    # Enforce integral constraint
    sum_final = np.sum(final_h_values) * dx
    if np.abs(sum_final - 1) > 1e-4:
        scale = 1.0 / sum_final
        final_h_values *= scale

    # Evaluate final C5
    corr = np.correlate(final_h_values, 1 - final_h_values, mode='full')
    c5_bound = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([final_half, final_half[::-1]]), c5_bound, n_points
