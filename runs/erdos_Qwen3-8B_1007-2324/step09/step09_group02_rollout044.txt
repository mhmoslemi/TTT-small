    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - integral_half_target})
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = minimize(objective, h_half, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000})

    if result.success:
        best_h_half = result.x
    else:
        best_h_half = h_half

    h_full = np.concatenate([best_h_half, best_h_half[::-1]])
    sum_h_full = np.sum(h_full) * dx
    if np.abs(sum_h_full - 1) > 1e-6:
        scale = 1.0 / sum_h_full
        h_full *= scale
    h_full = np.clip(h_full, 0.0, 1.0)
    sum_h_full = np.sum(h_full) * dx
    if np.abs(sum_h_full - 1) > 1e-6:
        scale = 1.0 / sum_h_full
        h_full *= scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([h_half, h_half[::-1]]), result.fun, n_points
    return h_full, result.fun, n_points
