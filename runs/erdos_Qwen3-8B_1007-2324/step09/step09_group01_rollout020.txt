from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():

        h_initial_full = h_initial_full * scale
        h_initial_full = np.clip(h_initial_full, 0, 1)
        initial_h = h_initial_full[:n_half]
    else:
        initial_h = np.zeros(n_points // 2)
        initial_h[35:105] = 1.0
        h_initial_full = np.concatenate([initial_h, initial_h[::-1]])
        h_initial_full = np.clip(h_initial_full, 0, 1)

        scale = integral_target / sum_val if sum_val > 0 else 1
        h_initial_full = h_initial_full * scale
        h_initial_full = np.clip(h_initial_full, 0, 1)
        initial_h = h_initial_full[:n_points // 2]
    
    bounds = [(0., 1.) for _ in range(n_points // 2)]
    
    def compute_c5(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')

    best_h = initial_h.copy()

        h_opt += noise
        h_opt = np.clip(h_opt, 0, 1)

        scale = integral_target / sum_val if sum_val > 0 else 1
        h_opt = h_opt * scale
        h_opt = np.clip(h_opt, 0, 1)
        c5 = compute_c5(h_opt)

            best_h = h_opt.copy()
    
    def objective(x):
        h_vec = x
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    try:
        result = differential_evolution(
            objective,
            bounds,

            maxiter=100,
            popsize=15,

        best_h = result.x

    except:

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
