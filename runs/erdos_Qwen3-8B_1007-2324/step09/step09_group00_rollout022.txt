def run(seed=42, budget_s=100, **kwargs):

    def compute_c5(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(h_opt):
        return compute_c5(h_opt)

    # Basin Hopping parameters
    maxiter = 100
    T = 1.0
    step_size = 0.1
    # Use constrained optimization

        h_sum = np.sum(h)
