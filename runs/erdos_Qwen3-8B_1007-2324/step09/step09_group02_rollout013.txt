def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2

    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_half = initial_h[:n_half]

        h_half = np.random.rand(n_half) * 0.5 + 0.25  # Random symmetric initial guess
        sum_half = np.sum(h_half)
        h_half = h_half * (integral_target / sum_half) if sum_half != 0 else h_half

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = minimize(compute_c5, h_half, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})

    best_solution = result.x
    final_h_values = np.clip(np.concatenate([best_solution, best_solution[::-1]]), 0, 1)
    sum_final = np.sum(final_h_values) * dx
    if np.abs(sum_final - 1) > 1e-3:
        final_h_values *= 1.0 / sum_final

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_solution, best_solution[::-1]]), result.fun, n_points
