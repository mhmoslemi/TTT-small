import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    def map_to_h(x, n_points):
        return np.interp(np.linspace(0, 2, n_points), [0, 2], x)
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_resampled = map_to_h(h_initial, n_points)
    else:
        h_resampled = np.zeros(n_points)
        center = n_points // 2
        start = center - 35
        end = center + 35
        h_resampled[start:end] = 1.0
    
    def build_h(params, n_segments):
        break_points = np.sort(params[:n_segments-1])
        heights = params[n_segments-1:]
        boundaries = np.concatenate([[0], break_points, [2]])
        h = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(boundaries[i] * n_points / 2)
            end_idx = int(boundaries[i+1] * n_points / 2)
            if start_idx < end_idx:
                h[start_idx:end_idx] = heights[i]
        h = np.clip(h, 0, 1)
        return h
    
    def objective(params, n_segments):
        h = build_h(params, n_segments)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint(params, n_segments):
        h = build_h(params, n_segments)
        integral = np.sum(h) * dx
        return integral - integral_target
    
    bounds = []
    for _ in range(3): bounds.append((0, 2))
    for _ in range(4): bounds.append((0, 1))
    
    best_h = h_resampled
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    for n_segments in [4, 6]:
        if time.time() - start > budget_s - 5: break
        np.random.seed(seed)
        guess = np.random.uniform(0, 2, 3) + np.random.uniform(0, 1, 4)
        guess = np.clip(guess, 0, 2)
        result = differential_evolution(
            lambda x: objective(x, n_segments),
            bounds,
            constraints=[lambda x: constraint(x, n_segments)],
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        h = build_h(result.x, n_segments)
        h = np.clip(h, 0, 1)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    elapsed = time.time() - start
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
