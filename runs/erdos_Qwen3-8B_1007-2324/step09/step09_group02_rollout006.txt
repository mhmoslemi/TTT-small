        return c5 + penalty

    def tournament_selection(population, k=5):

        best_idx = selected[0]
        best_val = evaluate(population[best_idx])
        for i in selected[1:]:
            current_val = evaluate(population[i])
            if current_val < best_val:
                best_val = current_val
                best_idx = i
        return population[best_idx]

    def crossover(a, b):
        child = np.copy(a)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = b[i]
        return child

    def mutate(individual, mutation_rate=0.1):

                individual[i] += np.random.normal(0, 0.05)

        sum_current = np.sum(individual)
        if sum_current != 0:
            individual *= 140.0 / sum_current
        return individual

        def objective(x):
