import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = 140

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_half = np.array(initial_h_values)[:n_half]
    else:
        np.random.seed(seed)
        h_half = np.random.rand(n_half) * 0.5 + 0.25
        sum_half = np.sum(h_half)
        h_half = h_half * (integral_target / sum_half) if sum_half != 0 else h_half

    # Objective function
    def obj(x):
        h = np.concatenate([x, x[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint function
    def con(x):
        return np.sum(x) - integral_target

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{'type': 'eq', 'fun': con}]

    # Initial optimization
    res = minimize(obj, h_half, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
    final_h = np.concatenate([res.x, res.x[::-1]])
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx

    # Fallback to initial guess if optimization failed
    if not res.success:
        final_h = np.concatenate([h_half, h_half[::-1]])
        final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx

    # Final checks
    final_h = np.clip(final_h, 0, 1)
    sum_final = np.sum(final_h) * dx
    if abs(sum_final - 1) > 1e-3:
        scale = 1.0 / sum_final
        final_h *= scale

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([res.x, res.x[::-1]]), final_c5, n_points

    return final_h, final_c5, n_points
