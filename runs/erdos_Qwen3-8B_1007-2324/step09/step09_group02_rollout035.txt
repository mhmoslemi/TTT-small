        child = np.copy(a)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = b[i]
        return child

    def mutate(individual, mutation_rate=0.15):
        for i in range(len(individual)):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        sum_current = np.sum(individual)
        if sum_current != 0:
            individual *= 70.0 / sum_current
        return individual

    def refine(individual):
        def objective(x):
            h_full = np.concatenate([x, x[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        def constraint(x):
            return np.sum(np.concatenate([x, x[::-1]])) - 140
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(len(individual))]
        result = minimize(objective, individual, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
        return result.x if result.success else individual

    population_size = 100
    population = [h_half.copy() for _ in range(population_size)]
    best_solution = h_half.copy()
    best_c5 = evaluate(best_solution)

    for generation in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        parent1 = tournament_selection(population)
        parent2 = tournament_selection(population)
        child = crossover(parent1, parent2)
        mutated = mutate(child)
        current_c5 = evaluate(mutated)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = mutated.copy()
        population.append(mutated)
        population.pop(np.argmax([evaluate(ind) for ind in population]))

    final_solution = refine(best_solution)
    final_c5 = evaluate(final_solution)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_solution = final_solution

    final_h_values = np.concatenate([final_solution, final_solution[::-1]])
    final_h_values = np.clip(final_h_values, 0, 1)
    sum_final = np.sum(final_h_values) * dx
    if np.abs(sum_final - 1) > 1e-3:
        scale = 1.0 / sum_final
        final_h_values *= scale

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_solution, best_solution[::-1]]), best_c5, n_points

    return final_h_values, best_c5, n_points
