import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_half_target = 70.0  # sum of first half must be 70

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_half_initial = np.array(initial_h_values)[:n_half]
        sum_h = np.sum(h_half_initial)
        if abs(sum_h - integral_half_target) > 1e-3:
            h_half_initial = h_half_initial * (integral_half_target / sum_h)
    else:
        np.random.seed(seed)
        h_half_initial = np.random.rand(n_half) * 0.5 + 0.25
        sum_h = np.sum(h_half_initial)
        h_half_initial = h_half_initial * (integral_half_target / sum_h)

    def objective(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_half_target}]
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = minimize(objective, h_half_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
    h_half_opt = result.x
    h_full_opt = np.concatenate([h_half_opt, h_half_opt[::-1]])
    h_full_opt = np.clip(h_full_opt, 0.0, 1.0)

    sum_full = np.sum(h_full_opt) * dx
    if abs(sum_full - 1.0) > 1e-3:
        scale = 1.0 / sum_full
        h_full_opt = h_full_opt * scale

    corr = np.correlate(h_full_opt, 1 - h_full_opt, mode='full')
    c5_bound = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (np.concatenate([h_half_initial, h_half_initial[::-1]]), np.inf, n_points)
    return (h_full_opt, c5_bound, n_points)
