import numpy as np
from scipy.optimize import dual_annealing, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    required_sum = 70.0

    if 'initial_h_values' in globals():
        x0 = np.array(initial_h_values[:n_half])
    else:
        np.random.seed(seed)
        x0 = np.random.rand(n_half)
        x0 = x0 * (required_sum / np.sum(x0)) if np.sum(x0) != 0 else x0

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        penalty = 1e4 * (np.sum(x) - required_sum)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    result = dual_annealing(objective, bounds, x0=x0, maxfun=1000, seed=seed)

    best_x = result.x
    best_x_refine = best_x.copy()

    def objective_refine(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum}]
    result_refine = minimize(objective_refine, best_x, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})

    best_x_refine = result_refine.x
    h_full = np.concatenate([best_x_refine, best_x_refine[::-1]])
    h_full = np.clip(h_full, 0, 1)
    sum_full = np.sum(h_full) * dx
    if not np.isclose(sum_full, 1, atol=1e-3):
        scale = 1.0 / sum_full
        h_full *= scale

    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (np.concatenate([best_x, best_x[::-1]]), best_c5, n_points)

    return h_full, final_c5, n_points
