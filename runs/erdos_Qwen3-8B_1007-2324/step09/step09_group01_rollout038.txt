        cost = objective(h_opt_start)
        if cost < best_c5:
            best_c5 = cost
            best_h_opt = h_opt_start.copy()
    
    try:
        result = minimize(
            objective,
            best_h_opt,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            jac=grad_objective,
            options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            best_c5 = result.fun
            best_h_opt = result.x.copy()
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.array(best_h_opt), best_c5, n_points
    
    return np.array(best_h_opt), best_c5, n_points
