    integral_target = n_points / 2  # sum(h) * dx = 1

    # Load initial_h_values if available
    initial_guess = None
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        h_initial_half = h_initial[:n_half]
        initial_guess = h_initial_half.copy()
    else:
        # Generate symmetric two-interval initial guess
        h_half = np.zeros(n_half)
        start_idx = int(0.35 * n_half)
        end_idx = start_idx + int(0.7 * n_half)
        h_half[start_idx:end_idx] = 1.0
        sum_h_initial_half = np.sum(h_half)
        sum_full = 2 * sum_h_initial_half
        scale = integral_target / sum_full
        h_initial_half_scaled = h_half * scale
        initial_guess = h_initial_half_scaled.copy()

    # Define objective function
    def objective(x):
        h_half = x
        h_full = np.concatenate([h_half, h_half[::-1]])
        sum_h = np.sum(h_full)
        # Scale to meet integral constraint
        scale = integral_target / sum_h if sum_h != 0 else 1.0
        h_scaled = h_full * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    corr = np.correlate(h_full_best, 1 - h_full_best, mode='full')
