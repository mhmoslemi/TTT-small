import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum(h) * dx = 1 => sum(h) = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        x0 = np.array(initial_h_values[:n_half])
    else:
        np.random.seed(seed)
        x0 = np.random.rand(n_half) * 0.5 + 0.25
        sum_x = np.sum(x0)
        if sum_x > 1e-12:
            x0 = x0 * (integral_target / sum_x)
        else:
            x0 = np.full(n_half, 0.5)

    # Objective function
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Constraints: sum(x) = integral_target
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
    bounds = [(0.0, 1.0) for _ in range(n_half)]

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
    best_solution = result.x

    # Final clipping and scaling
    final_h_values = np.concatenate([best_solution, best_solution[::-1]])
    final_h_values = np.clip(final_h_values, 0, 1)
    sum_final = np.sum(final_h_values) * dx
    scale = 1.0 / sum_final
    final_h_values *= scale

    # Compute C5
    corr = np.correlate(final_h_values, 1 - final_h_values, mode='full')
    c5_bound = np.max(corr) * dx

    # Respect time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([best_solution, best_solution[::-1]]), c5_bound, n_points
