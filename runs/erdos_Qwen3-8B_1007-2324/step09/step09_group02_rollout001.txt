import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_half = np.array(initial_h_values)[:n_points//2]
    else:
        h_half = np.full(n_points//2, 0.5)

    def objective(h):
        h_full = np.concatenate([h, h[::-1]])
        return evaluate_erdos_solution(h_full)

    def constraint(h):
        return np.sum(h) - integral_target

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(len(h_half))]

    result = minimize(objective, h_half, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})

    best_h = result.x
    best_c5 = objective(best_h)

    final_h = np.concatenate([best_h, best_h[::-1]])
    final_h = np.clip(final_h, 0, 1)
    sum_final = np.sum(final_h) * dx
    if np.abs(sum_final - 1) > 1e-5:
        final_h *= 1.0 / sum_final

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([h_half, h_half[::-1]]), objective(h_half), n_points

    return final_h, best_c5, n_points
