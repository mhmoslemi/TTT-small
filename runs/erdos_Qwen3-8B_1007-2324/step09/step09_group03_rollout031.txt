import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = n_points/2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        dist = np.abs(x - 1)
        h = 1 - dist
        h = np.clip(h, 0, 1)
        h_initial = h

    bounds = [(0., 1.)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h = h_initial.copy()

    trials = 30
    for i in range(trials):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + i)
        perturb = np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_initial + perturb, 0, 1)
        h_start = (h_start * integral_target) / (np.sum(h_start) + 1e-12)
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 150})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    h_values = best_h
    return h_values, best_c5, n_points
