import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    integral_target = n_points / 4  # sum(h) == 70

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_points // 2]
    else:
        h_opt_initial = np.zeros(n_points // 2)

    def obj(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * 2.0 / n_points

    bounds = [(0., 1.)] * (n_points // 2)
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = np.zeros(n_points // 2)

    # Multi-phase optimization
    trials = 12
    for i in range(trials):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + i)
        
        # Base perturbation with Gaussian
        perturb = np.random.normal(0, 0.03, n_points // 2)
        h_start = np.clip(h_opt_initial + perturb, 0, 1)
        
        # Repair: scale to meet constraint
        h_start = (h_start * integral_target) / (np.sum(h_start) + 1e-12)
        
        # Use L-BFGS-B for faster convergence
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 150, 'ftol': 1e-6, 'gtol': 1e-5})
        
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
