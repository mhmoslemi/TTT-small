import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 256
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():

        if len(initial_h) == n_points:
            initial_h = initial_h[:n_half]
        else:

        h = np.clip(1 - np.abs(x - 1), 0, 1)
        initial_h = np.concatenate([h[:n_half], h[:n_half][::-1]])

    def objective(h_opt):
        h_array = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    best_h_opt = initial_h

    trials = 30
    max_iter = 300

        noise = np.random.normal(0, 0.01, n_half)
        h_opt_start += noise
        h_opt_start = np.clip(h_opt_start, 0, 1)
        sum_start = np.sum(h_opt_start)
