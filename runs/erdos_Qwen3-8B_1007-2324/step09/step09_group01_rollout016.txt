import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        n_half = n_points // 2
        h_initial = np.array(initial_h_values)
        params_initial = h_initial[:n_half]
    else:
        n_half = n_points // 2
        h_half = np.zeros(n_half)
        start_idx = 35
        end_idx = start_idx + 70
        h_half[start_idx:end_idx] = 1.0
        params_initial = h_half.copy()

    def objective(params):
        h = np.concatenate([params, params[::-1]])
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = integral_target / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False,
        polish=True
    )

    best_params = result.x
    best_c5 = result.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_values = np.concatenate([best_params, best_params[::-1]])
        h_values = np.clip(h_values, 0, 1)
        sum_h = np.sum(h_values)
        scale = integral_target / sum_h
        h_values = h_values * scale
        h_values = np.clip(h_values, 0, 1)
        return h_values, best_c5, n_points

    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    scale = integral_target / sum_h
    h_values = h_values * scale
    h_values = np.clip(h_values, 0, 1)
