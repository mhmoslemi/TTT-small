import numpy as np
from scipy.optimize import minimize
import time
from scipy.ndimage import gaussian_filter1d

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 280
    n_coarse = 140
    dx = 2.0 / n_points
    integral_full = n_points / 2  # sum(h) == 140 for full grid
    integral_coarse = n_points / 4  # sum(h) == 70 for coarse grid

    # Initialize with parent or create default
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_coarse_initial = h_initial[:n_coarse//2]
    else:
        x = np.linspace(0, 2, n_coarse)
        h_triangle = 1 - np.abs(x - 1)
        h_triangle = np.clip(h_triangle, 0, 1)
        h_coarse_initial = h_triangle[:n_coarse//2]

    # Apply Gaussian smoothing to initial guess
    sigma = 0.05
    h_smoothed = gaussian_filter1d(h_coarse_initial, sigma)

    def obj_coarse(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_coarse = [(0, 1)] * (n_coarse // 2)
    constraints_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_coarse}]

    best_c5_coarse = float('inf')
    best_h_coarse = h_smoothed.copy()

    # Coarse grid multi-trial optimization
    for i in range(10):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + i)
        perturb = np.random.normal(0, 0.01, n_coarse // 2)
        h_start = np.clip(h_smoothed + perturb, 0, 1)
        h_start = (h_start * integral_coarse) / (np.sum(h_start) + 1e-12)

            best_h_coarse = res.x

    h_full_initial = np.concatenate([best_h_coarse, best_h_coarse[::-1]])

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds_full = [(0, 1)] * n_points
