                            s, e = sorted_breaks[-1], 2.0
                        else:
                            s, e = sorted_breaks[j-1], sorted_breaks[j]
                        if s <= x < e:
                            h_vals[i] = heights[j] * (integral_target / np.sum((e - s) * heights))
                            break
                best_h = h_vals
        except:
            pass
    
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
