    h_opt_280_start = np.concatenate([best_h_140, best_h_140[::-1]])

    best_c5_280 = float('inf')
    best_h_280 = h_opt_280_start.copy()

    trials_280 = 5
    for i in range(trials_280):
        if time.time() - start > budget_s - 1:
            break
        np.random.seed(seed + trials_140 + i)
        perturb = np.random.normal(0, 0.01, n_points_280)
        h_start = np.clip(h_opt_280_start + perturb, 0, 1)
        h_start = (h_start * integral_target_280) / (np.sum(h_start) + 1e-12)
        res = minimize(
            lambda h: compute_c5(h, n_points_280, dx_280),
            h_start,
            method='SLSQP',
            bounds=[(0., 1.)]*n_points_280,
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_280}],
            options={'maxiter': 200}
        )
        if res.success and res.fun < best_c5_280:
            best_c5_280 = res.fun
            best_h_280 = res.x

    return best_h_280, best_c5_280, n_points_280
