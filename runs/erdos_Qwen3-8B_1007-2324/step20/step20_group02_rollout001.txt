import scipy.optimize

    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points // 2  # 150

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        base = h_initial[:n_points // 2]
        start_points.append(base * integral_target / np.sum(base))
        start_points.append(np.roll(base * integral_target / np.sum(base), 20))
        start_points.append(np.clip(base * integral_target / np.sum(base) * 1.1, 0, 1))
        start_points.append(np.roll(base * integral_target / np.sum(base), 40))
        start_points.append(np.clip(base * integral_target / np.sum(base) * 0.9, 0, 1))

    # Fallback initializations
    random_initial = np.random.rand(n_points // 2)
    random_initial = random_initial * integral_target / np.sum(random_initial)
    start_points.append(random_initial)

    binary_initial = np.zeros(n_points // 2)
    num_ones = (n_points // 2) // 2
    indices = np.random.choice(n_points // 2, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial * integral_target / np.sum(binary_initial))

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],

                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}

        if result.success:
