import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - integral_target / 2

    initial_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_initial_half = h_initial[:half_length]
        scale = (integral_target / 2) / np.sum(h_initial_half)
        initial_points.append(h_initial_half * scale)

        perturbed = h_initial_half * scale
        perturbed += np.random.normal(0, 0.01, half_length)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral_target / 2) / np.sum(perturbed)
        initial_points.append(perturbed)

        rotated = h_initial_half * scale
        shift = np.random.randint(1, half_length)
        rotated = np.roll(rotated, shift)
        rotated = rotated * (integral_target / 2) / np.sum(rotated)
        initial_points.append(rotated)

    binary = np.zeros(half_length)
    num_ones = half_length // 2
    indices = np.random.choice(half_length, num_ones, replace=False)
    binary[indices] = 1
    initial_points.append(binary)

    random = np.random.rand(half_length)
    random = random * (integral_target / 2) / np.sum(random)
    initial_points.append(random)

    for _ in range(5):
        noise = np.random.normal(0, 0.02, half_length)
        perturbed = np.clip(perturbed + noise, 0, 1)
        perturbed = perturbed * (integral_target / 2) / np.sum(perturbed)
        initial_points.append(perturbed)

    best_c5 = float('inf')
    best_h = np.zeros(half_length)

    start_time = time.time()
    for h_start in initial_points:
        h_start = h_start * (integral_target / 2) / np.sum(h_start)
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_length)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'ftol': 1e-10, 'maxiter': 300, 'disp': False}
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s - 10:
            break

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
