import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points // 2  # 150

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points // 2:
            h_initial = h_initial[:n_points // 2]
        h_initial = h_initial * integral_target / np.sum(h_initial)
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 50))
        start_points.append(np.clip(h_initial * 1.1, 0, 1))
        start_points.append(np.roll(h_initial, 100))
        start_points.append(np.clip(h_initial * 0.9, 0, 1))
    else:
        # Random initial guess
        random_initial = np.random.rand(n_points // 2)
        random_initial = random_initial * integral_target / np.sum(random_initial)
        start_points.append(random_initial)
        start_points.append(np.zeros(n_points // 2))
        start_points.append(np.ones(n_points // 2))
        start_points.append(np.random.rand(n_points // 2) * 0.5)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * integral_target / np.sum(h_start)

        def objective(x):
            return compute_overlap(x)

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
        bounds = [(0.0, 1.0) for _ in range(n_points // 2)]

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-10, 'maxiter': 150, 'disp': False, 'eps': 1e-8}
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
