import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    sum_h_target = n_points // 2  # 30

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        kernel = 1 - h_full
        corr = np.correlate(h_full, kernel, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent_flat = initial_h_values[:n_points//2]
        parent_scaled = parent_flat * (sum_h_target / np.sum(parent_flat))
        start_points.append(parent_scaled)
        perturbed = parent_scaled + np.random.normal(0, 0.02, parent_scaled.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (sum_h_target / np.sum(perturbed))
        start_points.append(perturbed)
        rotated = np.roll(parent_scaled, np.random.randint(1, n_points//2))
        rotated = rotated * (sum_h_target / np.sum(rotated))
        start_points.append(rotated)

    binary_initial = np.zeros(n_points//2)
    binary_initial[np.random.choice(n_points//2, size=n_points//4, replace=False)] = 1
    start_points.append(binary_initial)

    random_initial = np.random.rand(n_points//2)
    random_initial = random_initial * (sum_h_target / np.sum(random_initial))
    start_points.append(random_initial)

    for _ in range(2):
        noise = np.random.normal(0, 0.03, n_points//2)
        perturbed = np.clip(parent_scaled + noise, 0, 1)
        perturbed = perturbed * (sum_h_target / np.sum(perturbed))
        start_points.append(perturbed)

    best_c5 = np.inf
    best_h = np.zeros(n_points//2)

    start_time = time.time()

    for h_start in start_points:
        h_start_sum = np.sum(h_start)
        if h_start_sum == 0:
            continue
        h_start = h_start * (sum_h_target / h_start_sum)
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0, 1)] * len(h_start),
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h_target, 'jac': np.ones_like(x)}],
            options={'ftol': 1e-9, 'maxiter': 400, 'disp': False, 'eps': 1e-8}
        )
        if result.success and time.time() - start_time < budget_s:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
