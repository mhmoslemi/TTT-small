import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    if 'initial_h_values' in globals():

        start_points.append(np.clip(h_initial[:n_points//2] * 1.1, 0, 1))
        start_points.append(np.clip(h_initial[:n_points//2] * 0.9, 0, 1))
        start_points.append(np.random.rand(n_points//2) * (integral_target / 2) / np.sum(np.random.rand(n_points//2)))
    else:
        h_initial = np.random.rand(n_points//2)
        h_initial = h_initial * (integral_target / 2) / np.sum(h_initial)
        start_points.append(h_initial)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        h_start = h_start * (integral_target / 2) / np.sum(h_start)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',

                'maxiter': 300,
                'disp': False,
                'eps': 1e-8

        if result.success:
