import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 15))
        start_points.append(np.clip(h_initial * 1.2, 0, 1))
        start_points.append(np.clip(h_initial * 0.8, 0, 1))
        start_points.append(np.roll(h_initial, 30))
    
    binary_initial = np.zeros(n_points)
    num_ones = n_points // 4
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)
    
    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    start_points.append(random_initial)
    
    perturbed_initial = np.array(initial_h_values[:n_points]) if 'initial_h_values' in globals() else np.zeros(n_points)
    perturbed_initial += np.random.normal(0, 0.05, size=n_points)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (integral_target / np.sum(perturbed_initial))
    start_points.append(perturbed_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start * integral_target
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
            ],
            options={

                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
