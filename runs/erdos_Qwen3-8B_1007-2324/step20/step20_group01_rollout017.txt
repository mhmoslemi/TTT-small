import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    sum_h_req = n_points / 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        kernel = 1 - h_full
        corr = np.correlate(h_full, kernel, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        base_h = np.array(initial_h_values[:30])
        base_h = base_h * (sum_h_req / np.sum(base_h))
        start_points.append(base_h)
        perturbed = base_h + np.random.normal(0, 0.05, 30)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (sum_h_req / np.sum(perturbed))
        start_points.append(perturbed)
        rotated = np.roll(base_h, np.random.randint(1, 30))
        rotated = rotated * (sum_h_req / np.sum(rotated))
        start_points.append(rotated)
        binary = np.zeros(30)
        binary[:15] = 1
        start_points.append(binary)
        random = np.random.rand(30)
        random = random * (sum_h_req / np.sum(random))
        start_points.append(random)

    for _ in range(2):
        noise = np.random.normal(0, 0.02, 30)
        perturbed = np.clip(np.array(initial_h_values[:30]) + noise, 0, 1)
        perturbed = perturbed * (sum_h_req / np.sum(perturbed))
        start_points.append(perturbed)

    best_c5 = float('inf')
    best_h = np.zeros(30)

    for h_start in start_points:
        try:
            result = scipy.optimize.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * 30,
                constraints=[
                    {'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h_req}
                ],
                options={
                    'ftol': 1e-10,
                    'maxiter': 200,
                    'disp': False,
                    'eps': 1e-8
                }
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
