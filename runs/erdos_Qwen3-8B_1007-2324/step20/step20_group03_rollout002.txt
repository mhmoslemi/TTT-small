import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n1 = n_points // 2
    best_c5 = np.inf
    best_h = np.zeros(n1)

    # Handle initial_h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        h_initial = h_initial[:n1]
        initial_guess = h_initial * integral_target / np.sum(h_initial)
    else:
        initial_guess = np.random.rand(n1)
        initial_guess = initial_guess * integral_target / np.sum(initial_guess)

    def evaluate(x):
        x = np.clip(x, 0, 1)
        if np.abs(np.sum(x) - integral_target) > 1e-5:
            return 1e10
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n1)]

    result = scipy.optimize.differential_evolution(
        func=evaluate,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )

    best_h = result.x
    best_c5 = evaluate(best_h)
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
