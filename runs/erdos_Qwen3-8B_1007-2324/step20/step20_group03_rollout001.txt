import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair(x):
        x_clip = np.clip(x, 0, 1)
        s = np.sum(x_clip)
        if s < 1e-8:
            return np.full_like(x_clip, 0.5 * target_sum / len(x_clip))
        return x_clip * (target_sum / s)
    
    def obj(x):
        x_rep = repair(x)
        x_full = np.concatenate([x_rep, x_rep[::-1]])
        corr = np.correlate(x_full, 1 - x_full, mode='full')
        return np.max(corr) * dx
    
    initial = None
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values)
        if len(initial) > n_half:
            initial = initial[:n_half]
        elif len(initial) < n_half:
            initial = np.concatenate([initial, np.zeros(n_half - len(initial))])
        initial = repair(initial)
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    if initial is not None:
        pop = [initial]
    else:
        pop = []
    
    for _ in range(30):
        x = np.random.rand(n_half)
        pop.append(repair(x))
    
    best = min(pop, key=obj)
    
    for _ in range(150):
        a, b, c = np.random.choice(n_half, 3, replace=False)
        mutant = pop[np.random.randint(len(pop))].copy()
        mutant[(a + b + c) % n_half] += np.random.uniform(-0.1, 0.1)
        mutant = repair(mutant)
        if obj(mutant) < obj(best):
            best = mutant
    
    def bfgs(x):
        def f(x):
            x_rep = repair(x)
            x_full = np.concatenate([x_rep, x_rep[::-1]])
            corr = np.correlate(x_full, 1 - x_full, mode='full')
            return np.max(corr) * dx
        res = scipy.optimize.minimize(f, x, method='BFGS', bounds=bounds, tol=1e-7, options={'maxiter': 100})
        return res.x
    
    best = bfgs(best)
    
    final = np.concatenate([best, best[::-1]])
    return final, obj(best), n_points
