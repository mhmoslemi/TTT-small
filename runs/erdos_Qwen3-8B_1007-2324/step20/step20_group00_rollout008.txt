import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:half_length]
        scale_factor = (integral_target / 2) / np.sum(h_initial_half)
        scaled_initial = h_initial_half * scale_factor
        start_points.append(scaled_initial)

        for shift in range(1, half_length):
            rotated = np.roll(scaled_initial, shift)
            start_points.append(rotated)

        reversed_initial = scaled_initial[::-1].copy()
        start_points.append(reversed_initial)

        for _ in range(5):
            dirichlet = np.random.dirichlet(np.ones(half_length)) * integral_target / 2
            start_points.append(dirichlet)

        for _ in range(3):
            perturbed = scaled_initial + np.random.normal(0, 0.05, half_length)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (integral_target / 2) / np.sum(perturbed)
            start_points.append(perturbed)

        low_pass = np.convolve(scaled_initial, np.ones(2)/2, mode='same')
        low_pass = np.clip(low_pass, 0, 1)
        low_pass = low_pass * (integral_target / 2) / np.sum(low_pass)
        start_points.append(low_pass)

    best_c5 = np.inf
    best_h = np.zeros(half_length)

    for h_start in start_points:
        h_start = h_start * (integral_target / 2) / np.sum(h_start)
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_length)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}],
            options={'ftol': 1e-10, 'maxiter': 300, 'disp': False}

        if result.success:
