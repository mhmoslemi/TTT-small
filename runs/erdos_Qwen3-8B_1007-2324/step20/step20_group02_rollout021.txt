import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points // 2

    start_points = []

    if 'initial_h_values' in globals():

        h_initial = h_initial[:n_points // 2]
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 30))
        start_points.append(np.clip(h_initial * 1.2, 0, 1))

        start_points.append(np.clip(h_initial * 0.8, 0, 1))
        start_points.append(np.random.rand(n_points // 2))
        for i in range(len(start_points)):
            start_points[i] = start_points[i] * integral_target / np.sum(start_points[i])
    else:
        binary = np.zeros(n_points // 2)
        binary[:50] = 1.0

        random_start = np.random.rand(n_points // 2)
