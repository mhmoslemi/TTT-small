        noise = np.random.normal(0, 0.02, half_length)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale = integral_target / sum_perturbed
        start_points.append(perturbed * scale)

    best_c5 = np.inf
    best_h = np.zeros(half_length)

    for h_start in start_points:
        # Ensure constraint is satisfied
        sum_h = np.sum(h_start)
        if abs(sum_h - integral_target) > 1e-6:
            scale = integral_target / sum_h
            h_start = h_start * scale

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_length)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(half_length)}
            ],
            options={
                'ftol': 1e-10,
                'maxiter': 500,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
