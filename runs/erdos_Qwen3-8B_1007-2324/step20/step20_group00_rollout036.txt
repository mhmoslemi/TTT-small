        bounds=[(0.0, 1.0) for _ in range(L)],
        constraints=cons,
        options={
            'ftol': 1e-12,
            'maxiter': 600,
            'disp': False,
            'eps': 1e-8
        }
    )

    best_h = result_slsqp.x
    final_c5 = compute_overlap(best_h)
    h_values = np.concatenate([best_h, best_h[::-1]])

    return h_values, final_c5, n_points
