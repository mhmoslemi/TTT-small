            def repair(x):
                # x is a numpy array of 30 (or can be a population row, but we do individually)
                current_sum = np.sum(x)
                desired_sum = 15.0
                if abs(current_sum - desired_sum) < 1e-5:
                    return x
                # We can scale the vector to the constraint plane, but ensure [0,1] bounds are maintained.
                if abs(current_sum) < 1e-5:
                    # If current_sum is nearly zero, we cannot scale arbitrarily.
                    return x * (desired_sum/1e-5)  # but then may go out of bounds.
                x = x * (desired_sum / current_sum)
                # If any element becomes >1, we set it to 1 and then renormalize the sum.
                x[x>1] = 1
                # If any element becomes <0, set to 0.
                x[x<0] = 0
                # Recompute new sum and desired_sum
                new_sum = np.sum(x)
                if abs(new_sum - desired_sum) > 1e-5:
                    # Distribute the remaining sum adjustment by increasing the values proportionally.
                    gap = desired_sum - new_sum
                    x += x * gap / new_sum
                return x
