import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    half_length = n_points // 2
    initial_population = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:half_length]
        # Perturbed version
        perturbed = h_initial_half.copy()
        noise = np.random.normal(0, 0.05, half_length)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_population.append(perturbed)
        # Rotated
        shift = np.random.randint(1, half_length)
        rotated = np.roll(perturbed, shift)
        initial_population.append(rotated)
        # Scaled
        scaled = perturbed.copy()
        scaled *= 0.95 + np.random.rand() * 0.1
        scaled = np.clip(scaled, 0, 1)
        scaled = scaled * (target_sum / np.sum(scaled))
        initial_population.append(scaled)

    # Binary initial
    binary = np.zeros(half_length)
    num_ones = half_length // 2
    indices = np.random.choice(half_length, num_ones, replace=False)
    binary[indices] = 1
    initial_population.append(binary)
    # Random uniform
    random = np.random.rand(half_length)
    random = random * (target_sum / np.sum(random))
    initial_population.append(random)

    # More initial guesses
    for _ in range(5):
        noise = np.random.normal(0, 0.05, half_length)
        perturbed = np.clip(h_initial_half + noise, 0, 1)
        perturbed = perturbed * (target_sum / np.sum(perturbed))
        initial_population.append(perturbed)

    def compute_overlap(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scale_factor = target_sum / current_sum
        h_scaled = h * scale_factor
        if np.any(h_scaled < 0) or np.any(h_scaled > 1):
            return np.inf
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = np.zeros(half_length)
    best_c5 = np.inf

    # Evaluate initial population
    for h_start in initial_population:
        current_c5 = compute_overlap(h_start)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_start

    # Run differential evolution
    result = scipy.optimize.differential_evolution(
        compute_overlap,
        bounds=[(0.0, 1.0) for _ in range(half_length)],
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        tol=1e-6,
        disp=False,
        polish=False,
        vectorized=True
    )

    if result.success:
        best_h = result.x
        best_c5 = compute_overlap(best_h)

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
