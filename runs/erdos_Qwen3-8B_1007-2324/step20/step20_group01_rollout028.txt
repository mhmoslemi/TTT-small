import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x_sum = np.sum(x)
        if x_sum == 0:
            return np.inf
        scaled_x = x * (integral_target / x_sum)
        h_full = np.concatenate([scaled_x, scaled_x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:half_length]
        scale_factor = (integral_target / 2) / np.sum(h_initial_half)
        h_initial_scaled = h_initial_half * scale_factor
        start_points.append(h_initial_scaled)
        for _ in range(2):
            noise = np.random.normal(0, 0.05, half_length)
            perturbed = h_initial_scaled + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            perturbed_scale = (integral_target / 2) / perturbed_sum
            perturbed = perturbed * perturbed_scale
            start_points.append(perturbed)

    if not start_points:
        random_h = np.random.rand(half_length)
        random_h_sum = np.sum(random_h)
        random_h_scaled = random_h * (integral_target / 2) / random_h_sum
        start_points.append(random_h_scaled)

    result = scipy.optimize.differential_evolution(
        objective,
        bounds=[[0.0, 1.0] for _ in range(half_length)],
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        vectorized=True,
        seed=seed,
        polish=False,
        disp=False
    )

    best_h = result.x
    best_h_scaled = np.clip(best_h * (integral_target / np.sum(best_h)), 0, 1)
    best_c5 = compute_overlap(best_h_scaled)

    return np.concatenate([best_h_scaled, best_h_scaled[::-1]]), best_c5, n_points
