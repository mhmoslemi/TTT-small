import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points // 2

    start_points = []

    if 'initial_h_values' in globals():

        h_initial = h_initial[:n_points // 2]
        h_scaled = h_initial * (integral_target / np.sum(h_initial))

        start_points.append(h_scaled)

        start_points.append(np.clip(h_scaled * 1.2, 0, 1))
        start_points.append(np.clip(h_scaled * 0.8, 0, 1))
        binary_initial = np.zeros(n_points // 2)
        num_ones = (n_points // 2) // 2
        np.random.shuffle(np.arange(n_points // 2))
        binary_initial[:num_ones] = 1
        start_points.append(binary_initial)
    else:
        random_initial = np.random.rand(n_points // 2)
        random_initial = random_initial * (integral_target / np.sum(random_initial))
        start_points.append(random_initial)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)

        def compute_overlap(h):

            corr = np.correlate(h_full, 1 - h_full, mode='full')

        result = scipy.optimize.minimize(

        if result.success:

            if current_c5 < best_c5:
