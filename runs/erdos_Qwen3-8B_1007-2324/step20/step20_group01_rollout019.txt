import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 25

    def compute_overlap(h):
        h_full = h
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - integral_target

    def constraint_jac(h):
        return np.ones_like(h)

    start_points = []

    if 'initial_h_values' in locals() and initial_h_values is not None:
        parent = np.array(initial_h_values)[:n_points]
        if len(parent) < n_points:
            parent = np.pad(parent, (0, n_points - len(parent)), 'constant')
        start_points.append(parent * integral_target / np.sum(parent))

        perturbed = parent.copy()
        perturbed += np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * integral_target / np.sum(perturbed)
        start_points.append(perturbed)

        rotated = parent.copy()
        shift = np.random.randint(1, n_points//2)
        rotated = np.roll(rotated, shift)
        rotated = rotated * integral_target / np.sum(rotated)
        start_points.append(rotated)

    binary = np.zeros(n_points)
    num_ones = n_points // 2
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary[indices] = 1
    start_points.append(binary * integral_target / np.sum(binary))

    random_start = np.random.rand(n_points)
    random_start = random_start * integral_target / np.sum(random_start)
    start_points.append(random_start)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()
    for i, h_start in enumerate(start_points):
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            res = opt.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint, 'jac': constraint_jac}],
                options={'ftol': 1e-10, 'maxiter': 500, 'disp': False}
            )
            if res.success:
                current_c5 = compute_overlap(res.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x
        except:
            pass

    return (best_h, best_c5, n_points)
