        reversed_initial = reversed_initial * (integral_target / 2) / np.sum(reversed_initial)
        start_points.append(reversed_initial)

        # Binary and random starting points
        binary_initial = np.zeros(half_length)

        random_initial = np.random.rand(half_length)

def compute_overlap(h_half):
    h_full = np.concatenate([h_half, h_half[::-1]])
    corr = np.correlate(h_full.astype(float), 1 - h_full.astype(float), mode='full')
