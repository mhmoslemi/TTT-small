def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    half_length = n_points // 2
    integral_target = n_points // 4  # 15
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full.astype(float), 
                           1 - h_full.astype(float), 
                           mode='full')
        return np.max(corr) * dx
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_initial_half = h_initial[:half_length]
        sum_initial = np.sum(h_initial_half)
        scale_factor = integral_target / sum_initial if sum_initial > 0 else 1.0
        h_initial_scaled = h_initial_half * scale_factor
        start_points.append(h_initial_scaled)
        
        perturbed = h_initial_scaled.copy()
        perturbed += np.random.normal(0, 0.05, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale_perturbed = integral_target / sum_perturbed if sum_perturbed > 0 else 1.0
        perturbed = perturbed * scale_perturbed
        start_points.append(perturbed)
        
        rotated = h_initial_scaled.copy()
        shift = np.random.randint(1, half_length)
        rotated = np.roll(rotated, shift)
        sum_rotated = np.sum(rotated)
        scale_rotated = integral_target / sum_rotated if sum_rotated > 0 else 1.0
        rotated = rotated * scale_rotated
        start_points.append(rotated)
        
        reversed_initial = h_initial_scaled[::-1].copy()
        sum_reversed = np.sum(reversed_initial)
        scale_reversed = integral_target / sum_reversed if sum_reversed > 0 else 1.0
        reversed_initial = reversed_initial * scale_reversed
        start_points.append(reversed_initial)
        
        shifted_i = np.roll(h_initial_scaled, np.random.randint(1, half_length))
        sum_shifted = np.sum(shifted_i)
        scale_shifted = integral_target / sum_shifted if sum_shifted > 0 else 1.0
        shifted_i = shifted_i * scale_shifted
        start_points.append(shifted_i)
        
        binary_initial = np.zeros(half_length)
        num_ones = half_length // 2 + 1
        indices = np.random.choice(half_length, num_ones, replace=False)
        binary_initial[indices] = 1
        sum_binary = np.sum(binary_initial)
        scale_binary = integral_target / sum_binary if sum_binary > 0 else 1.0
        binary_initial = binary_initial * scale_binary
        start_points.append(binary_initial)
        
        random_initial = np.random.rand(half_length)
        sum_random = np.sum(random_initial)
        scale_random = integral_target / sum_random if sum_random > 0 else 1.0
        random_initial = random_initial * scale_random
        start_points.append(random_initial)
        
        random_initial2 = np.random.rand(half_length)
        sum_random2 = np.sum(random_initial2)
        scale_random2 = integral_target / sum_random2 if sum_random2 > 0 else 1.0
        random_initial2 = random_initial2 * scale_random2
        start_points.append(random_initial2)
        
        t = np.linspace(0, half_length, half_length)
        sine_wave = np.sin(2 * np.pi * t / half_length)
        sine_wave = sine_wave - np.min(sine_wave)
        sine_wave = sine_wave / np.sum(sine_wave)
        sine_wave = sine_wave * integral_target
        start_points.append(sine_wave)
    
    best_c5 = np.inf
    best_h = np.zeros(half_length)
    
    for h_start in start_points:
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_length)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(half_length)}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 1000,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
