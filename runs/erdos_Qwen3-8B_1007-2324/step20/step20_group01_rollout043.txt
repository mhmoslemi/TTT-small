                h,
                method='L-BFGS-B',
                bounds=[(0., 1.) for _ in range(half_length)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
                options={'ftol': 1e-10, 'maxiter': 100, 'disp': False}
            )
            if result.success:
                evaluate(result.x)

    if not start_points:
        initial_guess = np.random.rand(half_length)
        s = np.sum(initial_guess)
        if s != 0:
            initial_guess = initial_guess * (integral_target / s)
        evaluate(initial_guess)

    result = scipy.optimize.differential_evolution(
        compute_overlap,
        [(0., 1.) for _ in range(half_length)],
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        polish=True,
        seed=seed
    )
    evaluate(result.x)

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
