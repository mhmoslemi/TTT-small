import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    int_tol = 1e-5

    def scale_h(h, target):
        current = np.sum(h)
        if current < 1e-10:
            return h
        scale = target / current
        return np.clip(h * scale, 0, 1)

    def objective(h):
        h_scaled = scale_h(h, target_sum)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def check_constraints(h):
        total = np.sum(h)
        return np.abs(total - target_sum) < int_tol

    # Initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
        h_initial = scale_h(h_initial, target_sum)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = scale_h(h_initial, target_sum)

    # Differential Evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = scipy.optimize.differential_evolution(
        objective,
