import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30
    half_length = n_points // 2  # 30

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        parent = np.array(kwargs['initial_h_values'])[:n_points]
        h_initial_half = parent[:half_length]
        scale_factor = integral_target / np.sum(h_initial_half)
        start_points.append(h_initial_half * scale_factor)
        for _ in range(3):
            noise = np.random.normal(0, 0.02, half_length)
            perturbed = np.clip(h_initial_half * scale_factor + noise, 0, 1)
            perturbed = perturbed * integral_target / np.sum(perturbed)
            start_points.append(perturbed)

    for _ in range(4):
        h_rand = np.random.rand(half_length)
        h_scaled = h_rand * integral_target / np.sum(h_rand)
        start_points.append(h_scaled)

    def constraint(x):
        return np.sum(x) - integral_target

    def obj(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(half_length)]

    best_h = np.zeros(half_length)
    best_c5 = float('inf')

    start_time = time.time()
    for h_start in start_points:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = opt.minimize(
            obj,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-8, 'maxiter': 300, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
