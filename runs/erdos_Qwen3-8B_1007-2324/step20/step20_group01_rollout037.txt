            seed=seed,
            disp=False
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h_scaled = best_h * (integral_target) / sum_best
        best_h_scaled = np.clip(best_h_scaled, 0.0, 1.0)
    else:
        best_h_scaled = best_h

    best_h_full = np.concatenate([best_h_scaled, best_h_scaled[::-1]])
    return best_h_full, best_c5, n_points
