import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) < n_points:
            initial_h = np.tile(initial_h, n_points // len(initial_h) + 1)[:n_points]
        start_points.append(initial_h)
        start_points.append(np.roll(initial_h, 20))
        start_points.append(np.clip(initial_h * 1.1, 0, 1))
        start_points.append(np.clip(initial_h * 0.9, 0, 1))
        start_points.append(np.zeros(n_points))
        num_ones = int(integral_target)
        indices = np.random.choice(n_points, num_ones, replace=False)
        start_points[-1][indices] = 1

    else:
        random_initial = np.random.rand(n_points)
        random_initial = random_initial * (integral_target / np.sum(random_initial))
        start_points.append(random_initial)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)
        else:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
