import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # adjust if needed

    def objective(x):

    bounds = [(0.0, 1.0) for _ in range(L)]

    # Run DE optimization
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=40,
        mutation=(0.5, 1),  # mutation factor and bounds
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False
    )

    # Extract best solution from DE
    best_h = result.x
    best_c5 = objective(best_h)

    # Refine with SLSQP
    def compute_overlap(x):
        h = x
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')

        bounds=bounds_slsqp,
        constraints=[

        ],
        options={

            'maxiter': 100,
            'disp': False,
            'eps': 1e-8

    # Update best_h with SLSQP result

    # Final check
    # Ensure sum is approximately integral_target
    sum_h = np.sum(best_h)
