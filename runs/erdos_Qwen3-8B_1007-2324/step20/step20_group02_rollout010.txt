import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = h_initial[:n_points]
        start_points = [h_initial]
        start_points.append(np.roll(h_initial, 10))
        start_points.append(np.roll(h_initial, 20))
        start_points.append(np.roll(h_initial, 30))
        start_points.append(np.roll(h_initial, 40))
        start_points.append(np.roll(h_initial, 50))
        start_points.append(np.clip(h_initial * 1.2, 0, 1))
        start_points.append(np.clip(h_initial * 0.8, 0, 1))
        start_points.append(np.random.permutation(h_initial))
        start_points.append(np.flip(h_initial))
        start_points.append(np.roll(h_initial, 15))
        start_points.append(np.roll(h_initial, 25))
        start_points.append(np.roll(h_initial, 35))
        start_points.append(np.roll(h_initial, 45))
        start_points.append(np.roll(h_initial, 55))
        start_points.append(np.random.rand(n_points) * 0.5)
        start_points.append(np.random.rand(n_points) * 0.6)
        start_points.append(np.random.rand(n_points) * 0.7)
        start_points.append(np.random.rand(n_points) * 0.8)
        start_points.append(np.random.rand(n_points) * 0.9)
    else:
        start_points = [np.random.rand(n_points)]
        start_points.append(np.random.rand(n_points) * 0.5)
        start_points.append(np.random.rand(n_points) * 0.6)
        start_points.append(np.random.rand(n_points) * 0.7)
        start_points.append(np.random.rand(n_points) * 0.8)
        start_points.append(np.random.rand(n_points) * 0.9)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (integral_target / np.sum(h_start))

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-10,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
