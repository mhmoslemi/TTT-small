def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    integral = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points//2]
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 10))
        start_points.append(np.clip(h_initial * 1.2, 0, 1))
        start_points.append(np.roll(h_initial, 20))
        start_points.append(np.clip(h_initial * 0.8, 0, 1))

    binary = np.zeros(n_points//2)
    n_ones = (n_points//2) // 2
    idxs = np.random.choice(n_points//2, n_ones, replace=False)
    binary[idxs] = 1
    start_points.append(binary)

    rand = np.random.rand(n_points//2)
    rand = rand * (integral / 2) / np.sum(rand)
    start_points.append(rand)

    if 'initial_h_values' in globals():
        perturbed = np.array(initial_h_values[:n_points//2])
        perturbed += np.random.normal(0, 0.05, size=perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral / 2) / np.sum(perturbed)
        start_points.append(perturbed)

    best_c5 = float('inf')
    best_h = np.zeros(n_points//2)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        h_start = h_start * (integral / 2) / np.sum(h_start)
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',

            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral / 2, 'jac': np.ones_like(x)}],

        if result.success:
