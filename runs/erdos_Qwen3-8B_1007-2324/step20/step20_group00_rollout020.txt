import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    half_length = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points // 4
    
    def compute_overlap(x):
        if np.sum(x) == 0:
            return np.inf
        x_norm = x * (target_sum / np.sum(x))
        h_full = np.concatenate([x_norm, x_norm[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    initial_points = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_half = h_initial[:half_length]
        initial_points.append(h_half)
        shifts = np.random.randint(1, half_length, size=3)
        for shift in shifts:
            shifted = np.roll(h_half, shift)
            initial_points.append(shifted)
        initial_points.append(h_half[::-1])
        for _ in range(3):
            noise = np.random.normal(0, 0.05, half_length)
            noisy = np.clip(h_half + noise, 0, 1)
            noisy = noisy * target_sum / np.sum(noisy)
            initial_points.append(noisy)
        for _ in range(2):

            mask = np.random.choice(half_length, half_length//2 + 1, replace=False)
            binary[mask] = 1
            binary = binary * target_sum / np.sum(binary)
            initial_points.append(binary)
    else:
        initial_points.append(np.random.rand(half_length))
        initial_points.append(np.random.rand(half_length))
        initial_points.append(np.random.rand(half_length))
    
    processed_points = []
    for pt in initial_points:
        pt_clipped = np.clip(pt, 0, 1)
        sum_clipped = np.sum(pt_clipped)
        if sum_clipped == 0:
            pt_clipped = np.random.rand(half_length)

            pt_clipped = pt_clipped * target_sum / sum_clipped

    for guess in processed_points:
        result = scipy.optimize.differential_evolution(
            compute_overlap,
            bounds=[(0.0, 1.0) for _ in range(half_length)],

    return h_values, best_c5, n_points
