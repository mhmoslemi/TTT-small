import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60

    dx = 2.0 / n_points

    def compute_overlap(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full.astype(float), 1 - h_full.astype(float), mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - integral_target

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:

        h_initial_half = h_initial[:half_length]
        scale_factor = integral_target / (2 * np.sum(h_initial_half))
        h_initial_scaled = h_initial_half * scale_factor
        start_points.append(h_initial_scaled)

        h_start = h_initial_scaled.copy()
        h_start += np.random.normal(0, 0.05, h_start.shape)
        h_start = np.clip(h_start, 0, 1)
