import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != half_length:

        h_initial = h_initial * (integral_target / 2) / np.sum(h_initial)
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 5))
        start_points.append(h_initial[::-1])
        start_points.append(np.clip(h_initial + np.random.normal(0, 0.04, h_initial.shape), 0, 1))
        start_points.append(np.clip(h_initial + np.random.normal(0, 0.07, h_initial.shape), 0, 1))
        start_points.append(np.random.permutation(h_initial))
        start_points.append(np.zeros_like(h_initial))
        indices = np.random.choice(half_length, half_length // 2 + 1, replace=False)
        start_points.append(np.zeros_like(h_initial))

    best_c5 = np.inf

    for h_start in start_points:
        h_start = h_start * (integral_target / 2) / np.sum(h_start)
        result = scipy.optimize.minimize(
