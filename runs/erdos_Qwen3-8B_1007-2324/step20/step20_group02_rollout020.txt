import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points // 2]
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 10))
        start_points.append(h_initial[::-1])
        start_points.append(np.clip(h_initial * 1.1, 0, 1))
        start_points.append(np.clip(h_initial * 0.9, 0, 1))
        start_points.append(np.random.rand(n_points // 2) * 0.5)
        start_points.append(np.zeros(n_points // 2))
        start_points.append(np.clip(h_initial + np.random.normal(0, 0.05, h_initial.shape), 0, 1))
        start_points.append(np.array([0.5]*n_points // 2))

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)
        else:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)

        def objective(x):
            return compute_overlap(x)

        def constraint(x):
            return np.sum(x) - integral_target / 2

        res = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-9,
                'maxiter': 150,
                'disp': False,
                'eps': 1e-8
            }
        )

        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
