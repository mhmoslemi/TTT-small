import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30
    half_length = n_points // 2  # 30

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in kwargs:
        h_initial = np.array(kwargs['initial_h_values'])
        h_initial = h_initial[:n_points]
        start_points.append(h_initial)
        h_scaled = h_initial * (integral_target / np.sum(h_initial)) if np.sum(h_initial) > 1e-7 else h_initial
        start_points.append(h_scaled)
        start_points.append(np.roll(h_scaled, 1))
        start_points.append(np.roll(h_scaled, -1))
        start_points.append(np.clip(h_scaled + np.random.normal(0, 0.05, n_points), 0, 1))
        start_points.append(np.clip(h_scaled + np.random.normal(0, 0.02, n_points), 0, 1))

    for _ in range(3):
        start = np.random.rand(n_points)
        start = start * integral_target / np.sum(start) if np.sum(start) > 1e-7 else start
        start_points.append(start)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_points)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()
    for h_start in start_points:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = opt.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-8,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-6
            }
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return (np.clip(best_h, 0, 1), best_c5, n_points)
