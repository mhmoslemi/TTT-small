import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - integral_target / 2

    def objective(h):
        return compute_overlap(h)

    def initialize_points():
        points = []
        if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
            parent = np.array(kwargs['initial_h_values'][:n_points])
            parent_half = parent[:half_length]
            scale = (integral_target / 2) / np.sum(parent_half)
            points.append(parent_half * scale)
        for _ in range(5):
            h = np.random.rand(half_length)
            h = h * (integral_target / 2) / np.sum(h)
            points.append(h)
        for _ in range(4):
            if points:
                base = np.random.choice(points, 1)[0]
                noise = np.random.normal(0, 0.02, half_length)
                h = np.clip(base + noise, 0, 1)
                h = h * (integral_target / 2) / np.sum(h)
                points.append(h)
        return points

    start_points = initialize_points()
    best_c5 = float('inf')
    best_solution = np.zeros(half_length)

    bounds = [(0.0, 1.0) for _ in range(half_length)]
    constraints = [{'type': 'eq', 'fun': constraint, 'jac': np.ones(half_length)}]

    start_time = time.time()
    for h in start_points:
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            res = opt.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
            )
            if res.success:
                current_c5 = compute_overlap(res.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = res.x
        except:
            pass

    h_full = np.concatenate([best_solution, best_solution[::-1]])
    return h_full, best_c5, n_points
