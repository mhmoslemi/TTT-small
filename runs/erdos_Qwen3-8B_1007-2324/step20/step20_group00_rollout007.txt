import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_integral = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_initial_half = h_initial[:half_length]
        scale_factor = (target_integral / 2) / np.sum(h_initial_half)
        h_initial_scaled = h_initial_half * scale_factor
        start_points.append(h_initial_scaled)

        perturbed = h_initial_scaled.copy()
        perturbed += np.random.normal(0, 0.02, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (target_integral / 2) / np.sum(perturbed)
        start_points.append(perturbed)

        rotated = h_initial_scaled.copy()
        shift = np.random.randint(1, half_length)
        rotated = np.roll(rotated, shift)
        rotated = rotated * (target_integral / 2) / np.sum(rotated)
        start_points.append(rotated)

        reversed_initial = h_initial_scaled[::-1].copy()
        reversed_initial = reversed_initial * (target_integral / 2) / np.sum(reversed_initial)
        start_points.append(reversed_initial)

        shifted_i = np.roll(h_initial_scaled, np.random.randint(1, half_length))
        shifted_i = shifted_i * (target_integral / 2) / np.sum(shifted_i)
        start_points.append(shifted_i)

        binary_initial = np.zeros(half_length)
        num_ones = half_length // 2 + 1
        indices = np.random.choice(half_length, num_ones, replace=False)
        binary_initial[indices] = 1
        start_points.append(binary_initial)

        random_initial = np.random.rand(half_length)
        random_initial = random_initial * (target_integral / 2) / np.sum(random_initial)
        start_points.append(random_initial)

        random_initial2 = np.random.rand(half_length)
        random_initial2 = random_initial2 * (target_integral / 2) / np.sum(random_initial2)
        start_points.append(random_initial2)

        perturbed_large = h_initial_scaled.copy()
        perturbed_large += np.random.normal(0, 0.05, perturbed_large.shape)
        perturbed_large = np.clip(perturbed_large, 0, 1)
        perturbed_large = perturbed_large * (target_integral / 2) / np.sum(perturbed_large)
        start_points.append(perturbed_large)

    def objective(h):
        c5 = compute_overlap(h)
        penalty = 1e9 * (np.sum(h) - target_integral) ** 2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(half_length)]
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=40,
        tol=1e-6,
        mutation=(0.8, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )

    best_h = result.x

    def slsqp_refine(h):
        result = scipy.optimize.minimize(
            compute_overlap,
            h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_length)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_integral, 'jac': np.ones_like(x)}],
            options={'ftol': 1e-12, 'maxiter': 600, 'disp': False, 'eps': 1e-8}
        )
        return result.x if result.success else h

    best_h = slsqp_refine(best_h)

    best_c5 = compute_overlap(best_h)
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
