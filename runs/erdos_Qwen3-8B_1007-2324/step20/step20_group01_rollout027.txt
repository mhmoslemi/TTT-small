import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h = np.zeros(half_length)

    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        parent_h = parent_h[:n_points]
        parent_half = parent_h[:half_length]
        total_parent = np.sum(parent_half)
        scale_factor = (integral_target / 2.0) / max(1e-10, total_parent)
        parent_scaled = parent_half * scale_factor
        start_points = [parent_scaled]
        for _ in range(4):
            perturb = np.random.normal(0, 0.01, half_length)
            new_h = parent_scaled + perturb
            new_h = np.clip(new_h, 0, 1)
            total_new = np.sum(new_h)
            if total_new > 1e-7:
                new_h = new_h * (integral_target / 2.0) / total_new
            start_points.append(new_h)
        for _ in range(3):
            random_h = np.random.rand(half_length)
            total_rand = np.sum(random_h)
            if total_rand > 1e-7:
                random_h = random_h * (integral_target / 2.0) / total_rand
            start_points.append(random_h)
    else:
        start_points = []
        for _ in range(6):
            random_h = np.random.rand(half_length)
            total_rand = np.sum(random_h)
            if total_rand > 1e-7:
                random_h = random_h * (integral_target / 2.0) / total_rand
            start_points.append(random_h)

    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - 15, 'jac': lambda x: np.ones_like(x)})
    bounds = [(0.0, 1.0) for _ in range(half_length)]

    options = {'disp': False, 'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}

    start_time = time.time()
    for h_start in start_points:
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            result = opt.minimize(compute_overlap, h_start, method='SLSQP', bounds=bounds, constraints=constraints, options=options)
            if result.success:
                current_c5 = compute_overlap(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
