        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:half_length]
        start_points.append(h_initial_half.copy())
        perturbed = h_initial_half.copy() + np.random.normal(0, 0.01, half_length)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        start_points.append(perturbed * 15.0 / perturbed_sum)
        rotated = np.roll(h_initial_half, np.random.randint(1, half_length))
        rotated_sum = np.sum(rotated)
        start_points.append(rotated * 15.0 / rotated_sum)
        binary = np.zeros(half_length)
        binary[np.random.choice(half_length, half_length//2, replace=False)] = 1
        start_points.append(binary)
        random = np.random.rand(half_length)
        random_sum = np.sum(random)
        start_points.append(random * 15.0 / random_sum)

    for _ in range(3):
        noise = np.random.normal(0, 0.02, half_length)
        perturbed = h_initial_half.copy() + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        start_points.append(perturbed * 15.0 / perturbed_sum)

    result = scipy.optimize.differential_evolution(
        compute_overlap,
        bounds=[(0.0, 1.0) for _ in range(half_length)],
        popsize=15,
        maxiter=100,
        tol=1e-5,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_c5 = np.inf
    best_h = np.zeros(half_length)
    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    else:
        for point in start_points:
            current_c5 = compute_overlap(point)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = point

    best_h_sum = np.sum(best_h)
    if best_h_sum != 0:
        best_h_scaled = best_h * 15.0 / best_h_sum
    else:
        best_h_scaled = np.zeros(half_length)
    h_values = np.concatenate([best_h_scaled, best_h_scaled[::-1]])

    return h_values, best_c5, n_points
