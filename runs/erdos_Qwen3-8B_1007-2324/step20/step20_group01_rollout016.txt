        kernel = 1 - h_full
        corr = np.correlate(h_full, kernel, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals():

        perturbed = h_initial_scaled.copy()
        perturbed += np.random.normal(0, 0.01, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)

        start_points.append(perturbed)

        rotated = h_initial_scaled.copy()
