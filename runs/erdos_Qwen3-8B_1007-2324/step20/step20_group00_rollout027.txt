import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    half_length = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points // 4  # 15

    def compute_overlap(x):
        scaled_x = x * (integral_target / np.sum(x))
        h_full = np.concatenate([scaled_x, scaled_x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    initial_points = []

    # Use provided initial points if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:half_length]
        scaled_initial = h_initial_half * (integral_target / np.sum(h_initial_half))
        initial_points.append(scaled_initial)
        # Perturbations
        perturbed = scaled_initial.copy()
        perturbed += np.random.normal(0, 0.05, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral_target / np.sum(perturbed))
        initial_points.append(perturbed)
        # Reversed
        reversed_initial = scaled_initial[::-1].copy()
        reversed_initial = reversed_initial * (integral_target / np.sum(reversed_initial))
        initial_points.append(reversed_initial)
        # Shifted
        shifted = np.roll(scaled_initial, np.random.randint(1, half_length))
        shifted = shifted * (integral_target / np.sum(shifted))
        initial_points.append(shifted)
        # Binary
        binary = np.zeros(half_length)
        num_ones = half_length // 2 + 1
        indices = np.random.choice(half_length, num_ones, replace=False)
        binary[indices] = 1
        binary = binary * (integral_target / np.sum(binary))
        initial_points.append(binary)
        # Random
        random = np.random.rand(half_length)
        random = random * (integral_target / np.sum(random))
        initial_points.append(random)
    else:
        # Fallback to random and wavelet initializations
        random = np.random.rand(half_length)
        random = random * (integral_target / np.sum(random))
        initial_points.append(random)
        # Wavelet freqs
        wavelet_freqs = [1, 3, 5, 7]
        for f in wavelet_freqs:
            t = np.linspace(0, half_length, half_length)
            wavelet = np.sin(2 * np.pi * f * t / half_length)

            wavelet /= np.sum(wavelet)

            initial_points.append(wavelet)

    best_c5 = np.inf

    return np.concatenate([best_h_scaled, best_h_scaled[::-1]]), best_c5, n_points
