def run(seed=42, budget_s=100, **kwargs):

    n_points = 60

    dx = 2.0 / n_points
    sum_h = 15.0

        kernel = 1.0 - h_full
        max_corr = 0.0

            corr = np.dot(shifted, kernel) * dx
            if corr > max_corr:

        rotated = np.roll(rotated, shift)
        rotated = rotated * (sum_h / np.sum(rotated))
