import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_length = n_points // 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - integral_target / 2

    def initialize_points(parent=None):
        points = []
        if parent is not None:

            scale = (integral_target / 2) / total_parent
            base = parent_half * scale
            points.append(base)
            points.append(np.roll(base, 1))
            points.append(np.roll(base, 10))
            points.append(np.roll(base, 20))
            points.append(np.roll(base, 30))
        for _ in range(4):
            h = np.random.rand(half_length)
            total = np.sum(h)
            if total == 0:
                h += 1e-10
            h = h * (integral_target / 2) / total
            points.append(h)
        return points

    start_points = initialize_points(kwargs.get('initial_h_values'))

    best_c5 = float('inf')
    best_h = np.zeros(half_length)

    for h_start in start_points:
        try:
            result = opt.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(half_length)],
                constraints=[{'type': 'eq', 'fun': constraint, 'jac': np.ones(half_length)}],
                options={'ftol': 1e-10, 'maxiter': 500, 'disp': False}

            if result.success:
                current_c5 = compute_overlap(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    best_h_full = np.concatenate([best_h, best_h[::-1]])
