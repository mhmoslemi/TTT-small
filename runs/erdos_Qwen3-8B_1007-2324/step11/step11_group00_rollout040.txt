            child2 = p2.copy()
            if np.random.rand() < 0.8:
                idx = np.random.randint(1, n_points - 1)
                child1[:idx], child2[:idx] = p2[:idx], p1[:idx]
            if np.random.rand() < mut_rate:
                child1 += np.random.normal(0, 0.015, n_points)
                child1 = repair_h(child1, target_sum)
            if np.random.rand() < mut_rate:
                child2 += np.random.normal(0, 0.015, n_points)
                child2 = repair_h(child2, target_sum)
            new_pop.extend([child1, child2])
        population = new_pop[:pop_size]

    best_h = repair_h(best_h, target_sum)
    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, n_points
