import numpy as np
import time
from scipy.optimize import minimize
from numpy.random import rand, normal

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.rand(n_half)
    
    def repair(h_vec, integral_target):
        n = len(h_vec)
        def objective(x):
            return np.sum((x - h_vec)**2)
        bounds = [(0, 1) for _ in range(n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
        result = minimize(objective, h_vec, method='SLSQP', bounds=bounds, constraints=cons)
        return result.x

    h_opt = repair(h_opt, integral_target)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)

    population = [h_opt.copy()]
    for _ in range(50):
        if _ < 20:
            h = np.random.rand(n_half)
        else:
            h = np.random.normal(0.5, 0.05, n_half)
        h = np.clip(repair(h, integral_target), 0, 1)
        population.append(h)

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        idxs = np.argsort(fitness)
        parents = [population[i] for i in idxs[:20]]
        
        new_population = []
        for i in range(20):
            p1 = parents[i % len(parents)]
            p2 = parents[(i+1) % len(parents)]
            child = p1 * np.random.rand() + p2 * (1 - np.random.rand())
            child = np.clip(repair(child, integral_target), 0, 1)
            new_population.append(child)
        
        for i in range(len(new_population)):
            child = new_population[i]
            noise = 0.01 * normal(size=n_half)
            child += noise
            child = np.clip(repair(child, integral_target), 0, 1)
            new_population[i] = child
        
        population = new_population + population[:20]
        fitness = [compute_overlap(ind) for ind in population]
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs[:50]]
        current_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()

    def objective(x):
        h_opt = x.reshape((n_half,))
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = minimize(objective, best_h, bounds=[(0,1)]*n_half, method='SLSQP')
    if result.success:
        best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
