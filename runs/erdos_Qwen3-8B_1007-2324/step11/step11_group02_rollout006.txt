import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)
        h_opt = repair(h_opt, integral_target)

    def repair(h_vec, integral):
        n = len(h_vec)
        def constraint_func(x):
            return np.sum(x) - integral
        bounds = [(0, 1)] * n
        res = minimize(lambda x: 0, x0=h_vec, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_func}])
        return res.x

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [h_opt.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points//2)
        h = repair(h, integral_target)
        population.append(h)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        sorted_idx = np.argsort(fitness)
        parents = [population[i] for i in sorted_idx[:10]]
        new_pop = []
        for i in range(len(parents)):
            p1 = parents[i % len(parents)]
            p2 = parents[(i+1) % len(parents)]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            child = repair(child, integral_target)
            new_pop.append(child)
        population = new_pop + [population[0]] * (population_size - len(new_pop))
        current_c5 = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_c5)
        if current_c5[best_idx] < best_c5:
            best_c5 = current_c5[best_idx]
            best_h = population[best_idx].copy()

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
