    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    initial_guesses = []
    if 'initial_h_values' in globals():
        initial_guesses.append(np.array(initial_h_values))
    x = np.linspace(0, 2, n_points)
    initial_guesses.append((1 + np.sin(np.pi * x)) / 2)
    initial_guesses.append(np.zeros_like(x))
    initial_guesses.append(1 - np.abs(x - 1))
    initial_guesses.append(1 - np.clip(np.abs(x - 1) - 0.4, 0, 1))

    best_h = None
    best_c5 = float('inf')

    for guess in initial_guesses:
        h_opt = np.clip(guess[:n_half], 0, 1)
        current_sum = np.sum(h_opt)
        if current_sum == 0:
            continue
        h_opt = repair_h(h_opt, integral_target)

        best_h_candidate = h_opt.copy()
        best_c5_candidate = compute_overlap(best_h_candidate)
        current_temp = 0.5
        cooling_rate = 0.998
        iter_count = 0
        max_iters = 5000

        while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
            iter_count += 1
            noise_std = 0.025 * current_temp
            noise = np.random.normal(0, noise_std, n_half)
            candidate = np.clip(h_opt + noise, 0, 1)
            candidate = repair_h(candidate, integral_target)
            candidate_c5 = compute_overlap(candidate)
            if candidate_c5 < best_c5_candidate or np.random.rand() < np.exp((best_c5_candidate - candidate_c5) / current_temp):
                best_h_candidate = candidate
                best_c5_candidate = candidate_c5
            current_temp *= cooling_rate

        if best_c5_candidate < best_c5:
            best_h = best_h_candidate.copy()
            best_c5 = best_c5_candidate

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
