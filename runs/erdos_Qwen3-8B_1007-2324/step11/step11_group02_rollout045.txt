            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def constraint_sum(x):
            return np.sum(x) - integral_target

        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_half)]

        result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
