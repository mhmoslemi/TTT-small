import numpy as np
import cma
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2

    # Initial solution
    if 'initial_h_values' in globals():
        initial_mean = np.array(initial_h_values) if len(initial_h_values) == n_points else np.random.rand(n_points) * 0.5 + 0.5
    else:
        initial_mean = np.random.rand(n_points) * 0.5 + 0.5

    def repair(h):
        center = n_points // 2
        indices = np.arange(n_points)
        weights = 1.0 / (np.abs(indices - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)

    def objective(x):
        h = repair(x)
        h_double = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_double, 1 - h_double, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    start_time = time.time()
    es = cma.CMAEvolutionStrategy(initial_mean, 0.1, {'popsize': 16, 'maxiter': 1000})
    best_h = initial_mean.copy()
    best_c5 = objective(initial_mean)

    while time.time() - start_time < budget_s:
        solutions = es.ask()
        obj_values = [objective(sol) for sol in solutions]
        es.tell(solutions, obj_values)
        if es.result[0][1] < best_c5:
            best_c5 = es.result[0][1]
            best_h = es.result[0][0]

    return best_h, best_c5, n_points
