import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)

    def repair(h, target):
        n = len(h)
        current_sum = np.sum(h)
        if np.isclose(current_sum, target):
            return np.clip(h, 0, 1)
        scale = target / current_sum if current_sum != 0 else 1.0
        h_scaled = h * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        if np.isclose(np.sum(h_clipped), target):
            return h_clipped
        diff = target - np.sum(h_clipped)
        h_clipped += diff / n
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    h_opt = repair(h_opt, integral_target)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)

    population_size = 50
    population = [h_opt.copy() for _ in range(population_size)]
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        top_indices = np.argsort(fitness)[:int(len(population)*0.2)]
        parents = [population[i] for i in top_indices]

        new_population = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = (p1 + p2) * 0.5 + np.random.normal(0, 0.03, p1.shape)
            child = repair(child, integral_target)
            new_population.append(child)

        population = new_population
        for ind in population:
            if compute_overlap(ind) < best_c5:
                best_c5 = compute_overlap(ind)
                best_h = ind.copy()

    def objective(x):
        h = x.reshape((n_points//2,))
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points//2)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
    res = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
    if res.success:
        best_h = res.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
