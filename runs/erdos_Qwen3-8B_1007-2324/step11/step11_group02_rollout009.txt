    def repair(h, target_sum):
        curr_sum = np.sum(h)
        if np.isclose(curr_sum, target_sum):
            return h
        scale = target_sum / curr_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        # Adjust final sum
        delta = target_sum - np.sum(h)
        if delta != 0:
            idx = np.argsort(h)
            h[idx[-1]] += delta
        return h

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def evaluate(h):
        h = repair(h, target_sum)
        return compute_overlap(h)

    population = [h_opt.copy()]
    for _ in range(49):
        h = np.random.rand(n_points//2)
        population.append(h)

    best_h = h_opt.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    elapsed = 0

    for _ in range(500):
        if elapsed > budget_s - 1:
            break
        fitness = [evaluate(ind) for ind in population]
