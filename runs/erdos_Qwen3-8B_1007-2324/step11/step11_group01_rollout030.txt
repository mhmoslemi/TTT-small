        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def fitness_function(x):
        h_candidate = np.array(x, dtype=np.float64)
        h_repaired = repair(h_candidate, integral_target)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    mean = h_initial_repaired.copy()
    sigma = 0.2
    pop_size = 20
    options = {
        'ftarget': None,
        'tolfx': 1e-6,
        'tolx': 1e-6,
        'popsize': pop_size,
        'seed': seed,
    }
    
    start_time = time.time()
    es = cma.CMAEvolutionStrategy(mean, sigma, options)
    best_solution = mean.copy()
    best_c5 = fitness_function(mean)
    
    while True:
        if time.time() - start_time > budget_s:
            break
        es.optimize()
        current_best = es.result().x
        current_c5 = fitness_function(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = current_best.copy()
    
    h_values = repair(best_solution, integral_target)
    final_c5 = fitness_function(h_values)
    return h_values, final_c5, n_points
