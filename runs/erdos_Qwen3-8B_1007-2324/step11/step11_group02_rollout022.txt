import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points // 2]
    else:
        h_opt = np.random.rand(n_points // 2)

    def repair(h, target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        return np.clip(h * scaling, 0, 1)

    h_opt = repair(h_opt, integral_target)

    def compute_C5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_C5(x)

    bounds = [(0, 1) for _ in range(n_points // 2)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target})

    start_time = time.time()
    result = minimize(objective, h_opt, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = compute_C5(best_h)
    elapsed = time.time() - start_time

    # If time permits, refine further
    if elapsed < budget_s - 1:
        result = minimize(objective, best_h, method='Nelder-Mead', bounds=bounds, tol=1e-6)
        best_h = result.x
        best_c5 = compute_C5(best_h)

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
