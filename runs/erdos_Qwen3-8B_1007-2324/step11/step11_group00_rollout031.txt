import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    col = 12

    def repair_h(h, target):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            return np.zeros_like(h)
        scale = target / s
        return np.clip(h_clipped * scale, 0, 1)

    def compute_c5(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def mutate(h, std=0.03):
        noise = np.random.normal(0, std, h.shape)
        return repair_h(h + noise, integral_target)

    def evolve():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = h[:n_half]
        else:
            h = np.random.rand(n_half)
        h = repair_h(h, integral_target)
        best = h.copy()
        best_c5 = compute_c5(h)
        for _ in range(5000):
            if time.time() - start_time > budget_s - 1:
                break
            h1 = mutate(best)
            h2 = mutate(best)
            h3 = mutate(best)
            vals = [compute_c5(h1), compute_c5(h2), compute_c5(h3)]
            h = [h1, h2, h3][np.argmin(vals)]

                best_c5 = compute_c5(h)
        return best, best_c5

    h_opt, c5 = evolve()
