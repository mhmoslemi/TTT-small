import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h_opt[:n_half]

    def repair(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = target_sum / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def evaluate(h_opt):
        return compute_overlap(h_opt)

    start_time = time.time()
    T = 1.0
    cooling_rate = 0.998
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = evaluate(current_h_opt)

    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.05, n_half)
        perturbed = current_h_opt + noise
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed, integral_target)
        candidate_c5 = evaluate(repaired)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((candidate_c5 - best_c5) / T):
            current_h_opt = repaired
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h_opt = current_h_opt.copy()
        T *= cooling_rate

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points

    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
