    def repair(h, target):

            return np.zeros_like(h)

        h = np.clip(h * scale, 0, 1)
        clamped = h < 1.0
        diff = (s - np.sum(h)) / np.sum(clamped)
        h[clamped] = np.clip(h[clamped] + diff, 0, 1)
        return h

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population = [h_opt.copy()]
    for _ in range(49):
        h = np.random.rand(n_points//2)

    start_time = time.time()

    for _ in range(1000):
