import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum(h) * dx = 1 => sum(h) = 0.5 * n_points

        h = np.clip(1 - 0.25 * np.abs(x - 1), 0, 1)
        h_opt = h

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, target):
        if np.all(h == 0):
            return np.zeros_like(h)
        scale = target / np.sum(h)
        return np.clip(h * scale, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(np.concatenate([best_h, best_h[::-1]]))
    temp = 0.9
