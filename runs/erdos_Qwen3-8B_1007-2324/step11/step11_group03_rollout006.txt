import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4.0
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 1, n_half)
        h_opt = np.clip(1 - np.abs(x - 0.5), 0, 1)
    
    def repair(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * target / s, 0, 1)
    
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    h_opt = repair(h_opt, integral_target)
    best_h = h_opt.copy()
    best_c5 = compute_c5(h_opt)
    
    start = time.time()
    T = 0.6
    cooling = 0.999
    for _ in range(int(1e5)):
        if time.time() - start > budget_s - 1:

        noise = np.random.laplace(0, 0.03, n_half)
        new_h = h_opt + noise
        new_h = repair(new_h, integral_target)
        c5 = compute_c5(new_h)
