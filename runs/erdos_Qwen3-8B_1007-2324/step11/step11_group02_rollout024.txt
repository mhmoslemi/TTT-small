        population = new_pop + [np.random.rand(n_half) for _ in range(population_size // 2)]
        population = [repair(h, integral_target) for h in population]

    best_h = min(population, key=compute_overlap)
    best_c5 = compute_overlap(best_h)
    if time.time() - start_time < budget_s - 1:
        best_h = local_optimize(best_h)

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
