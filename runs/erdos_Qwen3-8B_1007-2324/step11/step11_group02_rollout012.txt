def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

        h_opt = h_initial[:n_points//2]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        def obj(x):
            return np.sum((x - h)**2)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target})

        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons)
        return result.x

    h_opt = repair(h_opt)

        h_full = np.concatenate([x, x[::-1]])

        return np.max(corr) * dx
