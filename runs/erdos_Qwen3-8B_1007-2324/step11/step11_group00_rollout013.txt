import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        # Reshape to n_points
        h_opt = np.resize(h_initial, n_points)
    else:
        h_opt = np.random.rand(n_points)
        h_opt = repair_h(h_opt, integral_target)

    def repair_h(h, target):
        h = np.clip(h, 0, 1)
        current = np.sum(h)
        if current == 0:
            return np.zeros_like(h)
        scale = target / current
        return h * scale

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_c5(best_h)
    pop_size = 20
    generations = 100

    for gen in range(generations):
        pop = [best_h]
        for _ in range(pop_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            h = repair_h(h_opt + noise, integral_target)
            pop.append(h)
        fitness = np.array([compute_c5(h) for h in pop])
        idx = np.argsort(fitness)
        parents = np.array(pop)[idx[:pop_size // 2]]
        new_pop = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            pt = np.random.randint(1, n_points)
            c1 = np.concatenate([p1[:pt], p2[pt:]])
            c2 = np.concatenate([p2[:pt], p1[pt:]])
            new_pop.append(repair_h(c1, integral_target))
            new_pop.append(repair_h(c2, integral_target))
        for i in range(len(new_pop)):
            noise = np.random.normal(0, 0.01, n_points)
            new_pop[i] = repair_h(new_pop[i] + noise, integral_target)
        pop = new_pop
        for h in pop:
            c = compute_c5(h)
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c
        if time.time() - start_time > budget_s - 1:
            break

    h_full = np.concatenate([best_h, best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    best_c5 = np.max(corr) * dx
    return h_full, best_c5, n_points
