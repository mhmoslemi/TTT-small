def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2].copy()
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points//2].copy()
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        h2 = 1 - h_full
        corr = np.correlate(h_full, h2, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    T = 0.5
    cooling_rate = 0.995
    current_h = repair(h_opt)
    best_h = current_h.copy()
    best_c5 = compute_overlap(current_h)
    iterations = 0
    max_iter = 100000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        iterations += 1
        noise = np.random.standard_cauchy(len(current_h)) * T * 0.1
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed)
        c5 = compute_overlap(repaired)
        if c5 < best_c5 or np.random.rand() < np.exp((c5 - best_c5) / T):
            current_h = repaired
            best_c5 = c5
            best_h = current_h.copy()
        T *= cooling_rate
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
