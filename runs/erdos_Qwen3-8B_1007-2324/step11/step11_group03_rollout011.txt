import numpy as np

from scipy.optimize import cma

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        current_h = h_initial[:n_half]

        current_h = np.random.rand(n_half)

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        return np.clip(h * scale, 0, 1)

    def objective(x):
        h_opt = repair(x)
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start = time.time()
    result = cma.fmin(
        objective, current_h, 0.1,

    )

    best_c5 = objective(best_x)
    elapsed = time.time() - start
