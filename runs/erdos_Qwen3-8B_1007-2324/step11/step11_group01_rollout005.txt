import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        pop = [initial_h[:n_points//2] for _ in range(50)]
    else:
        pop = [np.random.uniform(0, 1, n_points//2) for _ in range(50)]
        for i in range(len(pop)):
            pop[i] = repair(pop[i], integral_target)

    def repair(h, target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_idx = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_idx] = sorted_vals
        curr_sum = np.sum(new_h)
        if curr_sum == 0:
            return np.zeros_like(h)
        scaling = target / curr_sum
        new_h *= scaling
        return np.clip(new_h, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = pop[0].copy()
    best_c5 = compute_overlap(best_h)
    start = time.time()

    for _ in range(10000):
        if time.time() - start > budget_s - 1:
            break
        p1, p2 = np.random.choice(pop, 2, replace=False)
        child = np.zeros_like(p1)
        for i in range(len(p1)):
            u = np.random.rand()
            if u < 0.5:
                beta = (2*u) ** 10
            else:
                beta = 1 / (2*(1-u)) ** 10
            child[i] = p1[i] + (p2[i] - p1[i]) * beta
        child = np.clip(child, 0, 1)
        child = repair(child, integral_target)
        c5 = compute_overlap(child)
        if c5 < best_c5:
            best_h, best_c5 = child, c5

    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
