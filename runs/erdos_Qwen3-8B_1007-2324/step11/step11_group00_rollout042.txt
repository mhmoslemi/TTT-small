        h_initial = np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (integral_target / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0, 1)

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = integral_target / current_sum
        return h * scale

    def compute_c5_fft(h):
        h_full = np.concatenate([h, h[::-1]])
        h_full_fft = fft.fft(h_full)
        one_minus_h = 1 - h_full
        one_minus_h_fft = fft.fft(one_minus_h)
        corr_fft = fft.ifft(h_full_fft * np.conj(one_minus_h_fft))
        corr = np.real(corr_fft)
        return np.max(corr) * dx

    h_initial_repaired = repair(h_initial)
    best_c5 = compute_c5_fft(h_initial_repaired)
    best_h = h_initial_repaired

    current_temp = 1.0
    cooling_rate = 0.995
    max_iter = 10000
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        noise_std = 0.025 * current_temp
        noise = np.random.normal(0, noise_std, n_points)
        candidate = h_initial_repaired + noise
        candidate = repair(candidate)
        candidate_c5 = compute_c5_fft(candidate)
        if candidate_c5 < best_c5:
            best_h = candidate
            best_c5 = candidate_c5
        else:
            delta = candidate_c5 - best_c5
            prob = np.exp(-delta / current_temp)
            if np.random.rand() < prob:
                best_h = candidate
                best_c5 = candidate_c5
        current_temp *= cooling_rate
        if iter_count % 100 == 0:
            if np.all(np.abs(candidate_c5 - best_c5) < 1e-6):
                break

    best_c5 = compute_c5_fft(best_h)
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
