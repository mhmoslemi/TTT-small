import numpy as np
import time
from numpy.random import rand, normal
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # smaller for faster evaluation
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum(h) == n_points / 2

    # Use provided initial solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)

    def repair(h, target_sum):
        current = np.sum(h)
        if current == 0:
            return np.zeros_like(h)
        scaling = target_sum / current
        new_h = h * scaling
        new_h = np.clip(new_h, 0, 1)
        return new_h

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 30
    population = [h_opt.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points//2)
        h = repair(h, integral_target)
        population.append(h)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    start_time = time.time()

    # Evolution loop
    for _ in range(500):  # 500 generations
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:20]]

        new_population = []
        for i in range(20):
            p1 = parents[i % len(parents)]
            p2 = parents[(i+1) % len(parents)]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            new_population.append(child)

        for i in range(len(new_population)):
            child = new_population[i]
            noise = 0.1 * normal(size=n_points//2)
            child += noise
            child = np.clip(child, 0, 1)
            child = repair(child, integral_target)
            new_population[i] = child

        # Combine new and old
        new_population.extend(population)
        fitness_all = [compute_overlap(ind) for ind in new_population]
        sorted_indices = np.argsort(fitness_all)
        population = [new_population[i] for i in sorted_indices[:population_size]]

        # Track best
        current_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()

    # Local optimization
    final_h = best_h.copy()
    def objective(x):
        h_opt = x.reshape((n_points//2,))
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = minimize(objective, final_h, bounds=[(0,1)]*(n_points//2), method='L-BFGS-B')
    if result.success:
        best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
