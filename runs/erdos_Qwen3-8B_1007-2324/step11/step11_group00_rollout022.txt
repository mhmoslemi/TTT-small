import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h):
        current_sum = np.sum(h)
        scaling = integral_target / current_sum if current_sum != 0 else 1.0
        return np.clip(h * scaling, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(np.concatenate([best_h, best_h[::-1]]))
    current_temp = 0.5
    cooling_rate = 0.999
    iter_count = 0
    max_iters = 5000

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        noise_std = 0.025 * current_temp
        noise = np.random.normal(0, noise_std, n_points)
        candidate = np.clip(h_opt + noise, 0, 1)
        candidate = repair_h(candidate)
        candidate_c5 = compute_overlap(np.concatenate([candidate, candidate[::-1]]))
        if candidate_c5 < best_c5:
            best_h = candidate
            best_c5 = candidate_c5
        current_temp *= cooling_rate

    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
