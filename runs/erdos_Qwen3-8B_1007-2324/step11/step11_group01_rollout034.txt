        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(x):
        h = x.copy()
        repaired_h = repair(h, integral_target)
        return compute_overlap(repaired_h)
    
    # Set up initial guess
    if 'initial_h_values' in globals():
        initial_guess = h_initial[:]
    else:
        initial_guess = np.random.rand(n_points)
    
    # Configure CMA-ES
    sigma = 0.5
    options = {
        'popsize': 20,
        'maxiter': 150,
        'ftarget': None,
        'bounds': [(0.0, 1.0) for _ in range(n_points)],
        'disp': False
    }
    
    start_time = time.time()
    with cma.CMAEvolutionStrategy(initial_guess, sigma, options=options) as es:
        while not es.stop():
            if time.time() - start_time > budget_s - 1:
                break
            es.tell(es.ask(), [objective(ind) for ind in es.ask()])
            es.retrain()
    
    # Extract best solution
    best_h = es.best_x
    best_c5 = compute_overlap(best_h)
    
    # Repair the final solution
    best_h_repaired = repair(best_h, integral_target)
    best_h_repaired = np.clip(best_h_repaired, 0, 1)
    
    # Return the h_values, which is the n_points samples
    return np.array(best_h_repaired), best_c5, n_points
