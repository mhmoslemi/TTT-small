import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_half]
    else:
        h_opt = np.random.rand(n_half)

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        h = h * scale
        return np.clip(h, 0, 1)

    h_opt = repair(h_opt)

    def obj(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}
    bounds = [(0, 1) for _ in range(n_half)]

    def optimize_with_repair(h):
        res = minimize(obj, h, bounds=bounds, constraints=cons, method='L-BFGS-B')
        return repair(res.x)

    start = time.time()
    try:
        for _ in range(200):
            if time.time() - start > budget_s - 1:
                break
            h_candidate = optimize_with_repair(h_opt)
            h_opt = h_candidate
    except:
        pass

    h_opt = optimize_with_repair(h_opt)
    h_full = np.concatenate([h_opt, h_opt[::-1]])
    final_c5 = obj(h_opt)
    return h_full, final_c5, n_points
