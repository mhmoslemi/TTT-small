def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():

        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.tile([1,0,1,0], n_points//4)[:n_points//2]
    
    def repair(h, integral_target, mode=1):
        n = len(h)
        target_sum = integral_target
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        h_scaled = h * scaling
        h_clamped = np.clip(h_scaled, 0, 1)
        if mode == 1:
            return h_clamped
        elif mode == 2:
            while True:
                valley = np.min(h_clamped)
                if valley < 0:
                    h_clamped = np.clip(h_clamped, 0, 1)
                    continue
                hill = np.max(h_clamped)
                if hill > 1:
                    h_clamped = np.clip(h_clamped, 0, 1)
                    continue
                current_sum = np.sum(h_clamped)
                if abs(current_sum - target_sum) < 1e-6:
                    return h_clamped
                scaling = target_sum / current_sum
                h_clamped = h_clamped * scaling
                h_clamped = np.clip(h_clamped, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    T = 0.5
    cooling_rate = 0.998
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = compute_overlap(current_h_opt)
    
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break

        perturbed = current_h_opt + noise

        if candidate_c5 < best_c5:
            current_h_opt = repaired

        T *= cooling_rate
