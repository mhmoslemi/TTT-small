import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)
    
    def repair(h):
        def cons(x):
            return np.sum(x) - integral_target
        bounds = Bounds(0, 1)
        res = minimize(lambda x: np.sum((x - h)**2), h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons})
        return res.x
    
    h_opt = repair(h_opt)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        h1 = repair(np.random.rand(n_points//2))
        h2 = repair(np.random.rand(n_points//2))
        h_cross = (h1 + np.random.rand() * h2) / 2
        h_cross = repair(h_cross)
        c5 = compute_overlap(h_cross)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cross
    
    def objective(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*(n_points//2))
    if result.success:
        best_h = result.x
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
