import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)
        h_opt = repair(h_opt, integral_target)

    def repair(h, integral_target):
        diff = integral_target - np.sum(h)
        if np.abs(diff) < 1e-6:
            return h
        h[-1] += diff
        return np.clip(h, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_opt = x
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def local_optimize(h):
        bounds = [(0, 1)] * len(h)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target})
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            return result.x
        return h

    def evaluate(h):
        return compute_overlap(h)

    best_h = h_opt.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    population = [h_opt.copy() for _ in range(20)]
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [evaluate(ind) for ind in population]
        top_indices = np.argsort(fitness)[:10]
        new_population = []
        for i in range(10):
            p1, p2 = population[top_indices[i % 10]], population[top_indices[(i+1) % 10]]
            child = (p1 + np.random.rand(len(p1)) * p2) / 2
            child = repair(child, integral_target)
            new_population.append(child)
        for h in new_population:
            if np.random.rand() < 0.3:
                h += np.random.normal(0, 0.05, len(h))
                h = repair(h, integral_target)
        population = new_population + [h for h in population if evaluate(h) < best_c5]

    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
