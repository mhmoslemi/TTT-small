import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_half]
    
    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    h_opt = repair(h_opt, integral_target)
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x.reshape(-1)
        x = repair(x, integral_target)
        return compute_overlap(x)
    
    bounds = [(0, 1) for _ in range(n_half)]
    
    # Optimizer parameters
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-4, options={'maxiter': 10})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
        else:
            break
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
