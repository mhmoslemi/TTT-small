import numpy as np
import time
from cma import CMAES

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from cma import CMAES
    
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_half]
    
    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    best_h_opt = h_opt.copy()
    best_c5 = compute_overlap(best_h_opt)
    
    # CMA-ES setup
    def objective(x):
        x = np.clip(x, 0, 1)
        repaired = repair(x, integral_target)
        c5 = compute_overlap(repaired)
        return c5  # Minimize
    
    # Initial mean and covariance
    mean = h_opt
    sigma = 0.1
    bounds = np.array([[0.0, 1.0]] * n_half)
    
    # Run CMA-ES
    es = CMAES(mean=np.copy(mean), sigma=sigma, bounds=bounds, maxiter=2000, popsize=50, seed=seed)
    
    for i in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
        es.tell(es.ask(), [objective(es.ask()[0])])
        es.next()
        current_mean = es.mean
        current_c5 = objective(current_mean)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_opt = current_mean.copy()
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
    
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
