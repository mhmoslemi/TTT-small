import numpy as np
import time
import warnings

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4
    h_opt = np.zeros(n_points // 2)
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points // 2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points // 2]
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.min(corr) * dx
    
    def perturb(h, noise_std=0.015):
        noise = np.random.normal(0, noise_std, len(h))
        perturbed = h + noise
        return np.clip(perturbed, 0, 1)
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    time_start = time.time()
    max_iter = int(budget_s * 0.9)  # Leave buffer for evaluation
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
        h_pert = perturb(best_h)
        h_repair = repair(h_pert)
        c5 = compute_overlap(h_repair)
        
        if c5 < best_c5 or np.random.rand() < np.exp((c5 - best_c5) / (20 * (1 - _/max_iter))):
            best_h = h_repair
            best_c5 = c5
    
    h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    return h_full, final_c5, n_points
