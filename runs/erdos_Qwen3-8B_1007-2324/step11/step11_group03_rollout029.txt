def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from scipy.signal import correlate

    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points//2]

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    X = []
    y = []
    def sample_candidate(h):
        noise = np.random.normal(0, 0.02, len(h))
        candidate = h + noise
        candidate = np.clip(candidate, 0, 1)
        repaired = repair(candidate, integral_target)
        return repaired

    for _ in range(10):
        candidate = sample_candidate(h_opt)
        X.append(candidate)
        y.append(compute_c5(candidate))

    X = np.array(X)
    y = np.array(y)
    kernel = Matern(nu=1.5)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)
    gp.fit(X, y)

    best_h = h_opt.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        mean, std = gp.predict(X, return_std=True)
        next_idx = np.random.choice(len(X), p=1 / (1 + np.exp(-mean)))
        best_h = X[next_idx].copy()
        best_h += np.random.normal(0, 0.03, len(best_h))
        best_h = np.clip(best_h, 0, 1)
        best_h = repair(best_h, integral_target)
        curr_c5 = compute_c5(best_h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
        X = np.vstack([X, best_h])
        y = np.append(y, curr_c5)
        gp.fit(X, y)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points

    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
