import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize

    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_half]

    def repair(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target / s
        return np.clip(h * scale, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(h_vec):
        h_repaired = repair(h_vec, integral_target)
        return compute_overlap(h_repaired)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    best_time = time.time()

    T = 2.5
    cooling = 0.996
    noise = 0.06
    max_iter = 10000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        noise_vec = np.random.normal(0, noise, n_half)
        perturbed = h_opt + noise_vec
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed, integral_target)
        candidate_c5 = compute_overlap(repaired)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((candidate_c5 - best_c5) / T):
            h_opt = repaired
            best_c5 = candidate_c5
            best_h = h_opt.copy()
        T *= cooling
        noise *= 0.98

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points

    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
