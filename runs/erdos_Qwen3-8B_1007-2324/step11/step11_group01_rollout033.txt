import numpy as np

def run(seed=42, budget_s=100, **kwargs):

        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    n_points = len(h_initial)
    integral_target = n_points / 2
    
    def repair(h, integral_target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def objective(x):
        h = repair(x, integral_target)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    
    try:
        sigma = 0.5
        popsize = 40
        es = cma.CMAEvolutionStrategy(h_initial, sigma, {'popsize': popsize})
        for gen in range(1000):
            if time.time() - start_time > budget_s - 1:
                break

            current_mean = es.mean
            current_c5 = objective(current_mean)
            if current_c5 < best_c5:
                best_h = current_mean.copy()
                best_c5 = current_c5

        pass
    
    best_h = repair(best_h, integral_target)
    final_c5 = objective(best_h)
    return np.concatenate([best_h, best_h[::-1]]), final_c5, n_points
