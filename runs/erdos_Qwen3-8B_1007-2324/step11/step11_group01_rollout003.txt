def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.tile([1,1,0,0], n_points//4)[:n_points//2]
    
    def repair(h, integral_target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def compute_overlap(h_opt):
        h_double = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_double, 1 - h_double, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    T = 0.2
    cooling_rate = 0.999
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = compute_overlap(current_h_opt)
    
    for _ in range(20000):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.1, n_points//2)
        perturbed = current_h_opt + noise
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed, integral_target)
        candidate_c5 = compute_overlap(repaired)
        
        if candidate_c5 < best_c5:
            current_h_opt = repaired
            best_h_opt = current_h_opt.copy()
            best_c5 = candidate_c5
        elif np.random.rand() < np.exp((candidate_c5 - best_c5) / T):
            current_h_opt = repaired
        
        T *= cooling_rate
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
