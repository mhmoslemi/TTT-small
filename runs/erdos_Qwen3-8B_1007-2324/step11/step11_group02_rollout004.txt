import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)
        h_opt = repair(h_opt, integral_target)
    
    def repair(h, integral):
        n = len(h)
        w = 1.0 / (np.abs(np.arange(n) - n//2) + 1e-6)
        idx = np.argsort(-w)
        h_sorted = np.sort(h)
        new_h = np.zeros_like(h)
        new_h[idx] = h_sorted[::-1]
        sum_h = np.sum(new_h)
        if sum_h == 0:
            return np.zeros_like(h)
        scaling = integral / sum_h
        new_h *= scaling
        return np.clip(new_h, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1-h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - integral_target
    
    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    population_size = 50
    population = [h_opt.copy() for _ in range(population_size)]
    fitness = [compute_overlap(p) for p in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        top_idx = np.argsort(fitness)[:20]
        for i in top_idx:
            h = population[i].copy()
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                new_h = result.x
                new_h = repair(new_h, integral_target)
                fitness[i] = compute_overlap(new_h)
                population[i] = new_h
                if fitness[i] < best_c5:
                    best_c5 = fitness[i]
                    best_h = new_h.copy()
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
