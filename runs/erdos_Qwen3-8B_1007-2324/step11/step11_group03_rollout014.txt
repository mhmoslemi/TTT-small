import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 140
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4.0
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_half]
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()
    start_time = time.time()
    
    T = 1.0
    cooling_rate = 0.998
    for _ in range(20000):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.015, n_half)
        perturbed = h_opt + noise
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed)
        candidate_c5 = compute_overlap(repaired)
        
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((candidate_c5 - best_c5) / T):
            best_c5 = candidate_c5
            best_h = repaired.copy()
        
        T *= cooling_rate
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
