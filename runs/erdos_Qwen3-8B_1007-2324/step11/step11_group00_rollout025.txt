def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Initial h_values if provided
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points]
    else:
        # Random initial configuration
        h_opt = np.random.rand(n_points)
        h_opt = h_opt / np.sum(h_opt) * integral_target
        h_opt = np.clip(h_opt, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h, target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def mutate(h, mut_rate=0.1):
        noise = np.random.normal(0, 0.1, h.shape)
        h += noise * mut_rate
        h = repair_h(h, integral_target)
        return np.clip(h, 0, 1)
    
    def crossover(parent1, parent2):
        return (parent1 + parent2) / 2
    
    # Genetic algorithm
    population = [h_opt]
    pop_size = 10
    gen = 0
    best_h = h_opt
    best_c5 = compute_overlap(h_opt)
    max_generations = 500
    
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        gen += 1
        new_population = []
        
        # Evaluate and select top performers
        scores = [compute_overlap(h) for h in population]
        sorted_indices = np.argsort(scores)
        top_parents = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Generate new solutions
        for _ in range(pop_size):
            parent1 = np.random.choice(top_parents)

            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)

        current_c5 = compute_overlap(current_best)

            best_c5 = current_c5
