    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        def objective(x):
            h_opt = x.reshape((n_points//2, ))
            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = minimize(objective, best_h, bounds=[(0,1)]*(n_points//2), method='L-BFGS-B')
        if result.success:
            best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
