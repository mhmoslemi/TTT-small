import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)[:n_points]
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
    h_initial = h_initial * integral_target / np.sum(h_initial)

    def compute_c5(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scaling = integral_target / sum_clipped
        return h_clipped * scaling

    pop_size = 20
    population = [h_initial + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    population = [repair(h) for h in population]
    
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        t1 = np.random.choice(population, 5)
        t2 = np.random.choice(population, 5)
        p1 = min(t1, key=compute_c5)
        p2 = min(t2, key=compute_c5)
        cross = np.random.randint(1, n_points)
        child = np.copy(p1)
        child[cross:] = p2[cross:]
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.03, n_points)
        else:
            noise = np.random.normal(0, 0.01, n_points)
        child += noise
        child = repair(child)
        worst = np.argmax([compute_c5(h) for h in population])
        if compute_c5(child) < compute_c5(population[worst]):

        current = min(population, key=compute_c5)
        if compute_c5(current) < best_c5:

            best_c5 = compute_c5(current)

    final_c5 = compute_c5(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
