def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():

    def repair(candidate):
        current_sum = np.sum(candidate)
        if current_sum == 0:
            return np.zeros_like(candidate)
        scale_factor = integral_target / current_sum

    current_candidate = repair(current_candidate)

        return max_corr * dx
    
    best_candidate = current_candidate.copy()
