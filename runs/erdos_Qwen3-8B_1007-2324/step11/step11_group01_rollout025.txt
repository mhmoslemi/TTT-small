import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def repair(h, integral):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral / current_sum
        new_h *= scaling
        return np.clip(new_h, 0, 1)

    def compute_c5(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def tournament_selection(population, size=3):
        candidates = np.random.choice(len(population), size, replace=False)
        return min([population[i] for i in candidates], key=compute_c5)

    def sbx_crossover(p1, p2, eta=10):
        u = np.random.rand(len(p1))
        o1, o2 = np.copy(p1), np.copy(p2)
        for i in range(len(p1)):
            if u[i] <= 0.5:
                beta = np.random.rand()
                if beta <= 0.5:
                    beta = (2 * beta) ** (1 / (eta + 1)) - 1
                else:
                    beta = (2 * (1 - beta)) ** (1 / (eta + 1)) - 1
                o1[i] = p1[i] + beta * (p2[i] - p1[i])
                o2[i] = p2[i] - beta * (p2[i] - p1[i])
        return np.clip(o1, 0, 1), np.clip(o2, 0, 1)

    def polynomial_mutation(ind, eta=20):
        for i in range(len(ind)):
            if np.random.rand() < 0.1:
                u = np.random.rand()
                if u <= 0.5:
                    beta = (2 * u) ** (1 / (eta + 1))
                    ind[i] = np.clip(ind[i] + beta * (1 - ind[i]), 0, 1)
                else:
                    beta = (2 * (1 - u)) ** (1 / (eta + 1))
                    ind[i] = np.clip(ind[i] - beta * ind[i], 0, 1)
        return ind

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points // 2]

    h_opt = repair(h_opt, integral_target)

    best_h = population[0].copy()

    start_time = time.time()

        o1 = polynomial_mutation(o1)
        o2 = polynomial_mutation(o2)
