    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (integral_target / s), 0, 1)
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 0.6
