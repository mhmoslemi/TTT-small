import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.random.rand(n_points)
    
    def repair(h, target):
        current = np.sum(h)
        if np.isclose(current, target):
            return h
        scale = target / current
        h_scaled = h * scale
        h_clip = np.clip(h_scaled, 0, 1)
        current_clip = np.sum(h_clip)
        if np.isclose(current_clip, target):
            return h_clip
        scale2 = target / current_clip
        return h_clip * scale2

    h_initial = repair(h_initial, integral_target)

    def objective(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def compute_overlap(x):
        return objective(x)

    bounds = [(0, 1)] * n_points
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}

    best_h = h_initial
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Local optimization
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-10})
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        # Perturb and optimize
        perturb = np.random.rand(n_points) * 0.05 - 0.025
        h_perturbed = np.clip(best_h + perturb, 0, 1)
        h_perturbed = repair(h_perturbed, integral_target)

        if result_perturb.success and result_perturb.fun < best_c5:

            best_h = result_perturb.x

    final_h = best_h
