                child1 = np.concatenate([p1[:idx], p2[idx:]])
                child2 = np.concatenate([p2[:idx], p1[idx:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(p1)
                offspring.append(p2)

        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                offspring[i] += noise
            offspring[i] = repair(offspring[i])

        population = offspring

    best_solution = repair(best_solution)
    best_c5 = compute_overlap(best_solution)
    elapsed = time.time() - start_time
    return best_solution, best_c5, n_points
