import numpy as np
from scipy.optimize import cma

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    from scipy.optimize import cma
    import time

    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        initial_mean = h_initial[:n_half]
    else:
        initial_mean = np.random.rand(n_half)

    def repair(h, it):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        sc = it / s
        return np.clip(h * sc, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return dx * np.max(corr)

    def objective(x):
        h_opt = repair(x, integral_target)
        return compute_overlap(h_opt)

    options = {
        'bounds': [(0.0, 1.0) for _ in range(n_half)],
        'ftol': 1e-10,
        'gtol': 1e-10,
        'seed': seed
    }

    start_time = time.time()
    res = cma.fmin(objective, initial_mean, 0.1, options=options, disp=False)
    elapsed = time.time() - start_time

    best_x = res[0]
    c5 = compute_overlap(best_x)
    h_values = np.concatenate([best_x, best_x[::-1]])

    return h_values, c5, n_points
