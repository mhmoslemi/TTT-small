    def repair(g):
        current_sum = np.sum(g)
        if current_sum == 0:

    h_opt = repair(h_opt)

        corr = np.correlate(g_full, 1 - g_full, mode='full')
        return np.max(corr) * dx

    def mutate(g, temp):
        return np.clip(g + np.random.normal(0, 0.03 * temp, n_half), 0, 1)

    h_opt = repair(h_opt)
