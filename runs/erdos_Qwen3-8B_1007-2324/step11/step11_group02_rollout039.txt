def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) must be this value

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = repair(h_initial, integral_target)
    else:
        h_opt = np.random.rand(n_points)
        h_opt = repair(h_opt, integral_target)

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h = h * scaling_factor
        new_h = np.clip(new_h, 0, 1)
        return new_h

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [h_opt.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair(h, integral_target)
        population.append(h)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:20]]
        
        new_population = []
        for i in range(20):
            p1 = parents[i % len(parents)]
            p2 = parents[(i+1) % len(parents)]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            new_population.append(child)
        
        for i in range(len(new_population)):
            child = new_population[i]
            noise = 0.05 * normal(size=n_points)
            child += noise
            child = repair(child, integral_target)
            new_population[i] = child
        
        new_population.extend(population)
        fitness_all = [compute_overlap(ind) for ind in new_population]
        sorted_indices = np.argsort(fitness_all)
        population = [new_population[i] for i in sorted_indices[:population_size]]
        
        current_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        def objective(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        result = minimize(objective, best_h, method='Nelder-Mead', bounds=[(0,1)]*n_points)
        if result.success:
            best_h = result.x

    return best_h, best_c5, n_points
