import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    dx = 2.0 / n_points
    integral_target = n_points / 4

    else:
        h_parent = np.random.rand(n_points)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * integral_target / np.sum(h_parent)
    
    def repair(h):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_idx = np.argsort(weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_idx] = sorted_vals
        scale = integral_target / np.sum(new_h)
        new_h *= scale
        return np.clip(new_h, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    population = [h_parent.copy()]
    for _ in range(49):
        population.append(np.random.rand(n_points))
        population[-1] = population[-1] * integral_target / np.sum(population[-1])
    
    best_h = h_parent.copy()
    best_c5 = compute_overlap(best_h)
    
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        idx1, idx2 = np.random.choice(len(population), 2, replace=False)
        parent1, parent2 = population[idx1], population[idx2]
        
        # SBX crossover
        beta = np.random.rand(n_points)
        beta = np.where(beta < 0.5, np.random.rand(n_points) ** (2 / (eta - 1)), 
                        (1 - 2 * np.random.rand(n_points)) ** (2 / (eta + 1)))
        child = 0.5 * (parent1 + parent2) * (1 - beta) + beta * (parent1 - parent2)
        child = np.clip(child, 0, 1)
        
        # Polynomial mutation
        mut_rate = 0.1
        mut = np.random.rand(n_points) < mut_rate
