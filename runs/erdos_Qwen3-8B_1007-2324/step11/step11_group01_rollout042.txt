    def repair(x):

        sorted_indices = np.argsort(distances)
        sorted_vals = np.sort(x)[::-1]

        new_x *= scaling
        return np.clip(new_x, 0, 1)

        corr = np.correlate(h_full, 1 - h_full, mode='full')

        return max_corr * dx
