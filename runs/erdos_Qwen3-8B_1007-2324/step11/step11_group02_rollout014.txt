import numpy as np
import time
from numpy.random import rand, uniform
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.rand(n_points//2)
        h_opt = repair(h_opt, integral_target)

    def repair(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        h = h * scaling
        return h

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [h_opt.copy() for _ in range(population_size)]
    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        top_idx = np.argsort(fitness)[:10]
        parents = [population[i] for i in top_idx]

        new_population = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + np.random.rand(*p1.shape) * (p2 - p1)) / 2
            child = repair(child, integral_target)
            new_population.append(child)

        population = new_population
        current_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()

    if time.time() - start_time < budget_s - 1:
        def objective(x):
            h = x.reshape((n_points//2,))
            return compute_overlap(h)

        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*(n_points//2))
        if res.success:
            best_h = res.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
