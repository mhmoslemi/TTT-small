        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)

        return np.max(corr) * dx

    def repair_h(h, integral_target):
        current_sum = np.sum(h)

            return np.zeros_like(h)
        scaling = integral_target / current_sum

            if compute_overlap(population[i]) < compute_overlap(population[best]):
                best = i
        return population[best]
