import numpy as np
import time
import cma

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        initial_mean = h_initial[:n_points//2]
    else:
        initial_mean = np.random.rand(n_points//2)
    
    def repair(h_vec):
        n = len(h_vec)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h_vec)[::-1]
        new_h = np.zeros_like(h_vec)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h_vec)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def evaluate(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    es = cma.CMAEvolutionStrategy(initial_mean, 0.5, {'bounds': (0.0, 1.0)})
    best_solution = initial_mean.copy()
    best_c5 = evaluate(best_solution)

        if es.generation > 100:
            break
