    optimizer = cma.CMAEvolutionStrategy(mean, sigma, {'popsize': pop_size})
    start_time = time.time()
    
    best_h = mean.copy()
    best_c5 = objective(mean.copy())
    
    while time.time() - start_time < budget_s - 1:
        optimizer.optimize(objective)
        current_best = optimizer.best
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
