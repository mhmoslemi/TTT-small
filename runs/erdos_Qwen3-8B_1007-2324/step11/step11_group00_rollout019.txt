import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points]
    else:
        h_opt = np.random.rand(n_points)

    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = integral_target / sum_h
        h = np.clip(h * scale, 0, 1)
        return h

    h_opt = repair(h_opt)

    def compute_c5(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    population_size = 50
    population = [repair(h_opt + np.random.normal(0, 0.03, n_points)) for _ in range(population_size)]
    fitness = [compute_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]

    while time.time() - start_time < budget_s - 1:
        new_population = []
        for _ in range(population_size // 2):
            p1 = random.choice(population)
            p2 = random.choice(population)
            child1 = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
            child2 = np.concatenate([p2[:n_points//2], p1[n_points//2:]])

        population = new_population
        fitness = [compute_c5(ind) for ind in population]
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_h = population[idx]
            best_c5 = fitness[idx]
    return np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1), best_c5, n_points
