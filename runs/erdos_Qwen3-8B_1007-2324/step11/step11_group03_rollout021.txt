import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half].copy()
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_half].copy()

    def repair(h, integral_target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        factor = integral_target / s
        return np.clip(h * factor, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_opt = repair(x, integral_target)
        return compute_overlap(h_opt)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    result = scipy.optimize.minimize(objective, h_opt, bounds=bounds, method='L-BFGS-B', tol=1e-10)
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return h_full, best_c5, n_points
