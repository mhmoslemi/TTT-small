def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        # Generate symmetric random initial configuration
        h_initial = np.random.uniform(0, 1, size=n_half)
        h_initial_sum = np.sum(h_initial)
        scaling = (n_points / 4) / h_initial_sum
        h_initial = h_initial * scaling
        h_initial = np.clip(h_initial, 0, 1)
        h_opt = h_initial
    
    def repair(h, integral_target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    T = 0.2
    cooling_rate = 0.995
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = compute_overlap(current_h_opt)
    
    for _ in range(20000):
        if time.time() - start_time > budget_s - 1:
            break
        # Symmetric perturbation
        i, j = np.random.choice(range(n_half), 2, replace=False)
        # Swap values and mirror them
        perturbed = current_h_opt.copy()

        repaired = repair(perturbed, integral_target)

            if candidate_c5 < best_c5:

                best_h_opt = current_h_opt.copy()
        T *= cooling_rate
