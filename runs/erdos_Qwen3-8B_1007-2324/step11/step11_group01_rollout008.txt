import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points//2]
        population = [h_initial]
    else:
        population = []
        for _ in range(50):
            h_rand = np.random.rand(n_points//2)
            h_repaired = repair(h_rand, integral_target)
            population.append(h_repaired)
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    def repair(h, integral_target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    best_h = population[0]
    best_c5 = compute_overlap(best_h)
    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        p1, p2 = np.random.choice(population, 2, replace=False)
        alpha = np.random.rand()
        off = alpha * p1 + (1 - alpha) * p2
        mut_mask = np.random.rand(n_points//2) < 0.1
        off[mut_mask] += np.random.normal(0, 0.01, sum(mut_mask))

        off_repaired = repair(off, integral_target)
