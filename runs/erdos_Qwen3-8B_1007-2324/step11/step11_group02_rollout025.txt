import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt = repair_h(h_opt, integral_target)

    def repair_h(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target / s
        return np.clip(h * scale, 0, 1)

    def objective(x):
        h = x.reshape((n_half,))
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_half

    def optimize_local():
        result = minimize(objective, h_opt, method='L-BFGS-B', bounds=bounds, tol=1e-5)
        if result.success:
            return result.x
        return h_opt

    h_opt = optimize_local()
    final_h = np.concatenate([h_opt, h_opt[::-1]])
    return final_h, objective(h_opt), n_points
