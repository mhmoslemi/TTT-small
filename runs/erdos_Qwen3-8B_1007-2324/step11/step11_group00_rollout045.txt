import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    integral_target = n_points / 2
    np.random.seed(seed)
    random.seed(seed)
    
    def repair_h(h, target):
        clipped = np.clip(h, 0, 1)
        current_sum = np.sum(clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        repaired = np.clip(clipped * scaling, 0, 1)
        return repaired
    
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

        perturbed = h_base + np.random.normal(0, 0.05, n_points)
