import numpy as np
import time
import cma

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import cma
    
    # Set up initial parameters and data
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Check if initial_h_values is provided
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
        integral_target = n_points / 2
    else:
        # Default random initialization
        initial_h = np.random.rand(n_points)
        integral_target = n_points / 2
    
    # Repair function to satisfy integral constraint and bounds
    def repair(h, target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    # Repair initial solution
    initial_repaired = repair(initial_h, integral_target)
    
    # Define objective function
    def objective(x):
        h_candidate = repair(x, integral_target)
        h_full = np.concatenate([h_candidate, h_candidate[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    # Set up CMA-ES
    np.random.seed(seed)
    es = cma.CMAEvolutionStrategy(initial_repaired, 0.1, {'popsize': 40})
    
    best_h = initial_repaired.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    while True:
        if time.time() - start_time > budget_s - 1:
            break
        es.optimize()
        x = es.result.x
        current_c5 = objective(x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()
    
    # Prepare result
    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
