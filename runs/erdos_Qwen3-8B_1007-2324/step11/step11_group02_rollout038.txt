from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h = h_initial[:n_points]
    else:
        h = np.random.rand(n_points)

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target_sum / s
        return np.clip(h * scale, 0, 1)

    h = repair(h)

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_with_bounds(x):

        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = target_sum / s
        x = x * scale
        return x

    def objective_with_bounds(x):
        x = repair_with_bounds(x)
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    best_h = h

    start_time = time.time()
