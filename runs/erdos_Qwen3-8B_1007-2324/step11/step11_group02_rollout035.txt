        for i in range(len(new_population)):
            child = new_population[i]
            noise = 0.05 * normal(size=n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = repair(child, integral_target)
            new_population[i] = child
        
        new_population.extend(population)
        fitness_all = [compute_overlap(ind) for ind in new_population]
        sorted_indices = np.argsort(fitness_all)
        population = [new_population[i] for i in sorted_indices[:population_size]]
        
        current_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        def objective(x):
            h_opt = x
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            return np.max(corr) * dx
        
        result = minimize(objective, best_h, bounds=[(0,1)]*n_points, method='L-BFGS-B')
        if result.success:
            best_h = result.x

    return best_h, best_c5, n_points
