import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair_h(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_base = initial_h[:n_points]
    else:
        x = np.linspace(0, 2, n_points)
        h_base = np.clip(1 - np.abs(x - 1), 0, 1)

    population = [h_base]
    for _ in range(4):
        mut = np.random.normal(0, 0.1, n_points)
        candidate = np.clip(h_base + mut, 0, 1)
        candidate = repair_h(candidate, integral_target)
        population.append(candidate)

    best_h = population[0]
    best_c5 = compute_overlap(best_h)
    for h in population:
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    max_generations = 100
    pop_size = 5
    for gen in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
        parents = []
        for _ in range(2):
            idxs = np.random.choice(len(population), 3, replace=False)
            best = min(population[idxs], key=compute_overlap)
            parents.append(best)
        child = (parents[0] + parents[1]) / 2
        noise = np.random.normal(0, 0.05, n_points)
        child += noise
        child = np.clip(child, 0, 1)
        child = repair_h(child, integral_target)
        child_c5 = compute_overlap(child)
        all_scores = [compute_overlap(h) for h in population]
        worst_idx = np.argmax(all_scores)
        population[worst_idx] = child
        if child_c5 < best_c5:
            best_h = child
            best_c5 = child_c5

    elapsed = time.time() - start_time
