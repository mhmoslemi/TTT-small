        return max_corr * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        population = [h_initial]
    else:
        population = []

    population_size = 50
    while len(population) < population_size:
        h_rand = np.random.rand(n_points)
        h_repaired = repair(h_rand)
        population.append(h_repaired)

    fitness = [compute_overlap(h) for h in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_c5 = fitness[best_index]

    start_time = time.time()
    max_iterations = 10000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        def tournament_selection(pop, fit):
            indices = np.random.choice(len(pop), 3, replace=False)
            best_idx = indices[np.argmin(fit[indices])]
            return pop[best_idx]

        parent1 = tournament_selection(population, fitness)
        parent2 = tournament_selection(population, fitness)

        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(n_points):
            beta = np.random.rand()
            if beta <= 0.5:
                beta = 1 + (2 * beta) * 10
                delta = abs(parent1[i] - parent2[i])
                child1[i] = parent1[i] + beta * delta
                child2[i] = parent2[i] - beta * delta
            else:
                beta = 1 + (2 * (1 - beta)) * 10
                delta = abs(parent1[i] - parent2[i])
                child1[i] = parent1[i] + beta * delta
                child2[i] = parent2[i] - beta * delta
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)

        mutation_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                child1[i] += np.random.normal(0, 0.01)
                child2[i] += np.random.normal(0, 0.01)
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)

        child1 = repair(child1)
        child2 = repair(child2)

        c5_c1 = compute_overlap(child1)
        c5_c2 = compute_overlap(child2)

        worst_index = np.argmax(fitness)
        if c5_c1 < fitness[worst_index]:
            population[worst_index] = child1
            fitness[worst_index] = c5_c1
        elif c5_c2 < fitness[worst_index]:
            population[worst_index] = child2
            fitness[worst_index] = c5_c2

        if min(fitness) < best_c5:
            best_c5 = min(fitness)
            best_h = population[np.argmin(fitness)]

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
