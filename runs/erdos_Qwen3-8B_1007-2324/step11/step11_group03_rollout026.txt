from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4.0

    if 'initial_h_values' in globals():

        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def repair(h_opt):
        current_sum = np.sum(h_opt)
        if current_sum == 0:
            return np.zeros_like(h_opt)
        scaling_factor = integral_target / current_sum
        return np.clip(h_opt * scaling_factor, 0, 1)

        return compute_overlap(h_repaired)
