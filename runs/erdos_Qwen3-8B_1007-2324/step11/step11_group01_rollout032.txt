    center_index = n_half // 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half)
    
    def repair(h):
        weights = 1.0 / (np.abs(np.arange(n_half) - center_index) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h_opt):
        repaired_h = repair(h_opt)
        return compute_overlap(repaired_h)
    
    bounds = [(0, 1) for _ in range(n_half)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=16,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed = time.time() - start_time
    
    best_h_opt = result.x
    best_c5 = result.fun
    
    best_h_opt_repaired = repair(best_h_opt)
    h_values = np.concatenate([best_h_opt_repaired, best_h_opt_repaired[::-1]])
    
    return h_values, best_c5, n_points
