    best_c5 = objective(best_x)
    
    # Refine with L-BFGS-B
    def objective_refine(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - integral_target_half
    
    result_l_bfgs = scipy.optimize.minimize(
        objective_refine,
        best_x,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 200, 'ftol': 1e-6, 'disp': False})
    
    if result_l_bfgs.success:
        best_x = result_l_bfgs.x.copy()
        best_c5 = objective_refine(best_x)
    
    # Final check and build h_values
    h_values = np.concatenate([best_x, best_x[::-1]])
    sum_h = np.sum(h_values) * dx
    if abs(sum_h - 1) > 1e-6:
        h_values = h_values * (1 / sum_h)
    return h_values, best_c5, n_points
