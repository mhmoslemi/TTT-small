        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for start in start_points:
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]

        def objective(x):
            h_half = x
            h_full = np.concatenate([h_half, h_half[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })

        result = scipy.optimize.minimize(
            objective,
            start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'disp': False, 'ftol': 1e-9, 'eps': 1e-8}
        )

        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            current_c5 = compute_overlap(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.concatenate([h_opt, h_opt[::-1]])

    return best_h, best_c5, n_points
