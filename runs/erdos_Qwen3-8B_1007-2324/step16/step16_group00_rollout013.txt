import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2
    
    best_c5 = np.inf
    best_h = np.zeros(n_half)
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_start = h_initial[:n_half]
    else:
        h_start = np.random.rand(n_half) * (integral_target / np.sum(np.random.rand(n_half)))
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint(x):
        return np.sum(x) - integral_target / 2
    
    result = scipy.optimize.minimize(
        objective,
        h_start,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
    )
    
    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
