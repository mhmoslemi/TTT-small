        parents = [pop[i] for i in np.argsort(fit)[:10]]
        off = []
        for _ in range(len(pop) - 10):
            a, b = np.random.choice(parents, 2, replace=False)
            c = crossover(a, b)
            if np.random.rand() < 0.1:
                c = mutate(c)
            off.append(c)
        pop = parents + off

    best_idx = np.argmin([compute_overlap(ind) for ind in pop])
    best_h = pop[best_idx]

    def obj(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_len)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 15}]
    res = scipy.optimize.minimize(obj, best_h, method='SLSQP',
                                  bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h = res.x if res.success else best_h
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    best_h = best_h * 15 / s if s else np.zeros(half_len)
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, compute_overlap(best_h), n_points
