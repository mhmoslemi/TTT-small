```python
def generate_base_pattern_variation(n_points, method='periodic', phase=0):
    """Produce structured trademarks shown to improve overlap bounds"""
    if method == 'periodic':
        t = np.linspace(0, 2 * np.pi * phase, n_points)
        pattern = 0.5 + 0.2 * np.sin(t) + 0.3 * np.cos(t * 2)
        pattern = pattern / np.max(np.abs(pattern)) * 0.5 + 0.25  # Normalize
        return pattern
    elif method == 'symmetric':
        t = np.linspace(0, 2, n_points) * np.pi / 180  # In radians
        return 0.5 * (1 + np.cos(t))
```
