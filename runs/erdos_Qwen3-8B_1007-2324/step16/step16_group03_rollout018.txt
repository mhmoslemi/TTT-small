import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60

    n_opt = n_points // 2
    integral_target = n_opt

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    time_limit = budget_s * 0.8
    start_time = time.time()

    def objective(x):
        h = x
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        penalty = 1e4 * np.abs(np.sum(x) - integral_target)
        return overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_opt)]

    # Use differential evolution for optimization
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,

        tol=1e-4
