import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    integral_target = n_points / 2

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    best_c5 = compute_overlap(best_h)

    def objective(x):
        return compute_overlap(x)

    def constraint(x):
        return np.sum(x) - integral_target

    cons = ({'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)})

    time_limit = budget_s * 0.9
    start_time = time.time()

    for _ in range(3):
        if time.time() - start_time > time_limit:

        result = scipy.optimize.minimize(
            objective,
            x0=h_initial,
            method='SLSQP',
