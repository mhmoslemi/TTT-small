import scipy.optimize

        h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target / 2,
        'jac': lambda x: np.ones_like(x)
    })

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)

    result = scipy.optimize.minimize(
        compute_overlap,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 300, 'disp': False, 'ftol': 1e-9, 'eps': 1e-8}
    )

    if result.success:
