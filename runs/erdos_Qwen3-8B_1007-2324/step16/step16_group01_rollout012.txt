import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target_half = 15

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        # Process 60-point initial vector to 30-point half
        h_opt = h_initial[:n_points//2].copy()
    else:
        h_opt = np.random.rand(n_points//2)

    # Normalize initial to meet integral constraint
    if np.sum(h_opt) == 0:
        h_opt = np.random.rand(n_points//2)
    h_opt = h_opt * integral_target_half / np.sum(h_opt)

    def objective(x):
        h_half = x
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        penalty = 1e5 * abs(np.sum(x) - integral_target_half)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]

    # Differential evolution with coarse global search
    result_de = scipy.optimize.differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=50, maxiter=100, tol=1e-6,
        mutation=(0.5, 1), recombination=0.7,
        seed=seed, polish=False, disp=False
    )

    best_x_de = result_de.x
    best_c5_de = objective(best_x_de)

    # L-BFGS-B refinement with exact constraint
    def objective_refined(x):
        h_half = x
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - integral_target_half

    bounds_lbgfs = [(0.0, 1.0) for _ in range(n_points//2)]

    result_lbgfs = scipy.optimize.minimize(
        objective_refined, best_x_de, method='L-BFGS-B',
        bounds=bounds_lbgfs, constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 200, 'disp': False}
    )

    best_x = result_lbgfs.x
    best_c5 = objective_refined(best_x)

    # Build full vector with exact integral
    h_full = np.concatenate([best_x, best_x[::-1]])

        best_x = np.clip(best_x, 0, 1)
        h_full = np.concatenate([best_x, best_x[::-1]])

    return h_full, best_c5, n_points
