import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_opt = n_points // 2

    if 'initial_h_values' in globals():
        v_initial = np.array(initial_h_values)
        v_initial = v_initial[:n_opt]
    else:
        v_initial = np.random.rand(n_opt)
        while np.abs(np.sum(v_initial) - integral_target) > 1e-6:
            v_initial = np.random.rand(n_opt)

    def compute_overlap(h):
        h_fft = np.fft.fft(h)
        g = 1 - h
        g_fft = np.fft.fft(g)
        conv_fft = h_fft * g_fft
        conv = np.fft.ifft(conv_fft).real
        return np.max(conv) * dx

    def objective(x):
        return compute_overlap(x)

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    best_h = v_initial.copy()
    best_c5 = compute_overlap(best_h)

    time_remaining = budget_s * 0.9
    start_time = time.time()

    result = scipy.optimize.minimize(
        objective,
        x0=v_initial,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_opt)],
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    for _ in range(20):
        if time.time() - start_time > time_remaining:
            break
        perturb = np.random.normal(0, 0.05, n_opt)
        candidate = best_h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if np.abs(candidate_sum - integral_target) > 1e-6:
            adjustment = (integral_target - candidate_sum) / n_opt
            candidate += adjustment * np.random.choice([-1, 1], size=n_opt)
            candidate = np.clip(candidate, 0, 1)
        current_c5 = compute_overlap(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
