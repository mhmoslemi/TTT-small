import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 4  # 1 / dx

    # Initialize with provided starting values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_pop = [h_initial[:n_points//2].copy()]
    else:
        t = np.linspace(0, 2 * np.pi, n_points)
        h_pop = [0.5 * (1 + np.cos(t))]

    def repair(x):
        y = np.clip(x, 0.0, 1.0)
        total = np.sum(y) * dx
        if np.abs(total - 1.0) > 1e-6:
            scale = 1.0 / total if total > 1e-6 else 1.0
            y = y * scale
            y = np.clip(y, 0.0, 1.0)
        return y

    def compute_overlap(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Generate initial population
    for _ in range(4):
        t = np.linspace(0, 2 * np.pi, n_points)
        h = 0.5 * (0.7 * np.sin(t) + 0.3 * np.cos(2 * t))
        h = np.clip(h, 0.0, 1.0)
        h = repair(h)
        h_pop.append(h)
    
    best_h = h_pop[0]
    best_c5 = compute_overlap(best_h)
    time_start = time.time()

    # Genetic algorithm loop
    for gen in range(30):
        if time.time() - time_start > budget_s:
            break
        # Evaluate population
        fitness = [compute_overlap(h) for h in h_pop]
        # Select top 20
        top_idx = np.argsort(fitness)[:20]
        top_candidates = [h_pop[i] for i in top_idx]
        if gen % 5 == 0:
            # Add random restarts
            for _ in range(5):
                rand_h = np.random.rand(n_points//2)
                rand_h = repair(rand_h)
                top_candidates.append(rand_h)
        
        # Crossover and mutation
        new_pop = []
        for i in range(len(top_candidates)):
            for j in range(i+1, len(top_candidates)):
                parent1 = top_candidates[i]
                parent2 = top_candidates[j]
                child = (parent1 + parent2) / 2
                noise = np.random.normal(0, 0.1, size=len(child))
                child += noise
                child = repair(child)
                new_pop.append(child)
        
        # Replace old population
        h_pop = top_candidates + new_pop
        if len(h_pop) > 50:
            h_pop = h_pop[:50]
        
        # Track best
        for h in h_pop:
            c5 = compute_overlap(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Local refinement
    def objective(x):
        return compute_overlap(x)
    
    result = scipy.optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points//2)],
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target},
        options={'ftol': 1e-12, 'maxiter': 50, 'disp': False}
    )
    if result.success:
        best_h = result.x
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    
    return final_h, final_c5, n_points
