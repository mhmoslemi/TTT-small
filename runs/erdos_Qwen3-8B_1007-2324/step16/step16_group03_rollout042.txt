            while np.abs(np.sum(h_full) - integral_target) > 1e-6:
                h_full = np.random.uniform(0, 1, n_points).clip(0, 1)
    else:
        h_full = np.random.uniform(0, 1, n_points).clip(0, 1)
        while np.abs(np.sum(h_full) - integral_target) > 1e-6:
            h_full = np.random.uniform(0, 1, n_points).clip(0, 1)
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_overlap(h_full)
    best_h = h_full.copy()
    
    time_limit = budget_s * 0.8
    start_time = time.time()
    
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')

    cons = ({
