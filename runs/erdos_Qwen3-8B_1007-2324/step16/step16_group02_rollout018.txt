import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h_candidate)
        scale = integral_target / sum_clipped
        h_scaled = h_clipped * scale
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    def objective(h_candidate):
        h_repaired = repair(h_candidate)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_points//2]

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )

    best_h = result.x
    full_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = objective(best_h)
    return full_h, final_c5, n_points
