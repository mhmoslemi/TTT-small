import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        parent = initial_h_values[:n_points]
        parent_sum = np.sum(parent)
        if np.abs(parent_sum - integral_target) > 1e-6:
            parent = parent * (integral_target / parent_sum)
    else:
        parent = np.random.rand(n_points)
        parent = parent * (integral_target / np.sum(parent))

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if np.isclose(sum_clipped, 0):
            return np.inf
        scaling = integral_target / sum_clipped
        h_repaired = np.clip(h_clipped * scaling, 0, 1)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-12,
        mutation=(0.8, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False,
        polish=True,
        vectorized=True
    )

    best_h = result.x
    final_c5 = objective(best_h)

    return best_h, final_c5, n_points
