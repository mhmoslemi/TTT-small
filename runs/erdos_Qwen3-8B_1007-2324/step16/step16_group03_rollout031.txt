import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.random.uniform(0, 1, n_points).clip(0, 1)
        while np.abs(np.sum(h_initial) - integral_target) > 1e-6:
            h_initial = np.random.uniform(0, 1, n_points).clip(0, 1)

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = np.max(corr) * dx
        return overlap

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h_initial.copy()

    start_time = time.time()
    result = scipy.optimize.minimize(objective, x0=x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-8})
    elapsed = time.time() - start_time

    if result.success and elapsed < budget_s:
        best_c5 = result.fun
        best_h = result.x
    else:
        best_c5 = compute_overlap(x0)
        best_h = x0

    return best_h, best_c5, n_points
