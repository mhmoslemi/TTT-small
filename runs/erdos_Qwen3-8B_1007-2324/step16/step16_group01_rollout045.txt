        best_ind = best_ind / np.sum(best_ind) * sum_half
        best_ind = np.clip(best_ind, 0, 1)
    
    h_full = np.concatenate([best_ind, best_ind[::-1]])
    c5 = compute_fitness(best_ind)
    return h_full, c5, n_points
