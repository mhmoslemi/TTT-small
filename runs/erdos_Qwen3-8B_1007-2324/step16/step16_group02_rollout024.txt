import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_initial_population(parent=None):
        popsize = 15
        if parent is not None:
            t_samples = np.random.standard_t(df=5, size=(popsize-1, n_points))
            t_samples *= 0.1
            initial_pop = np.vstack([np.array(parent), t_samples])
        else:
            t_samples = np.random.standard_t(df=5, size=(popsize, n_points))
            t_samples *= 0.1
            initial_pop = t_samples
        return np.clip(initial_pop, 0, 1)

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        initial_pop = generate_initial_population(parent)
    else:
        initial_pop = generate_initial_population(None)

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            scaling = 0
        else:
            scaling = target_sum / sum_clipped
        x_scaled = x_clipped * scaling
        x_repaired = np.clip(x_scaled, 0, 1)
        h_full = np.concatenate([x_repaired, x_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = scipy.optimize.differential_evolution(
        fun=objective, bounds=bounds,
        strategy='best1bin', popsize=15,
        tol=1e-10, mutation=(0.8, 1.0),
        crossover=0.9, maxiter=100,
        recombination='binomial', polish=True,
        disp=False, seed=seed
    )

    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
