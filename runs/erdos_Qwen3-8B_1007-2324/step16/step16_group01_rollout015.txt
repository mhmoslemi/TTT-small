import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60

    integral_target_half = 15

    bounds_de = [(0.0, 1.0) for _ in range(n_points//2)]
    result_de = scipy.optimize.differential_evolution(
        objective_de,
        bounds_de,
        strategy='best1bin',
        popsize=50,
        maxiter=120,
        mutation=(0.5, 1.0),
        recombination=0.7,
        tol=1e-6,

        return np.max(corr) * dx
