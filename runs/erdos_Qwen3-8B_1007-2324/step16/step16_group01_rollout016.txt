def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = 15  # Sum of the first half

    # Get initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        # Take the first half
        h_initial_half = h_initial[:n_points//2]
        # Scale to target_sum
        h_initial_half = h_initial_half / np.sum(h_initial_half) * target_sum
        h_initial_half = np.clip(h_initial_half, 0, 1)
    else:
        # Generate random initial guess
        h_initial_half = np.random.rand(n_points//2)
        h_initial_half = h_initial_half / np.sum(h_initial_half) * target_sum
        h_initial_half = np.clip(h_initial_half, 0, 1)

    # Define objective function
    def compute_overlap(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Constraints: sum(h_half) = target_sum
    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum,
        'jac': lambda x: np.ones_like(x)
    })

    # Function to run optimization with a starting point
    def optimize_starting_point(h_start):
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'disp': False, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            h_opt = result.x.copy()
            h_opt = np.clip(h_opt, 0, 1)
            # Check if sum is correct
            sum_opt = np.sum(h_opt)
            if abs(sum_opt - target_sum) > 1e-6:
                h_opt = h_opt * target_sum / sum_opt
                h_opt = np.clip(h_opt, 0, 1)
            return h_opt, compute_overlap(h_opt)
        else:
            return h_start, compute_overlap(h_start)

    # Try multiple starting points
    # Start with the initial guess
    best_h, best_c5 = optimize_starting_point(h_initial_half)
    
    # Try a few perturbed starting points
    for _ in range(3):
        perturbed = h_initial_half + np.random.normal(0, 0.1, size=n_points//2)

            best_c5 = c_perturbed
