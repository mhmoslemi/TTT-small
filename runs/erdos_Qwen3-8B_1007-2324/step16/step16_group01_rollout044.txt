                perturb = np.random.normal(0, 0.05, size=n_points//2)
                child += perturb
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child > 0:
                child = child * (integral_target / sum_child)
            new_population.append(child)
        population = new_population

    fitness = [compute_overlap(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx]

    def objective(x):
        h_opt = x.copy()
        h_opt = np.clip(h_opt, 0, 1)
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    result = scipy.optimize.minimize(
        objective,
        best_h_half,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 50, 'disp': False, 'ftol': 1e-8, 'eps': 1e-8}
    )

    optimized_h_half = result.x.copy()
    optimized_h_half = np.clip(optimized_h_half, 0, 1)
    sum_opt = np.sum(optimized_h_half)
    if sum_opt > 0:
        optimized_h_half = optimized_h_half * (integral_target / sum_opt)
    final_h_full = np.concatenate([optimized_h_half, optimized_h_half[::-1]])
    corr = np.correlate(final_h_full, 1 - final_h_full, mode='full')
    best_c5 = np.max(corr) * dx
    return final_h_full, best_c5, n_points
