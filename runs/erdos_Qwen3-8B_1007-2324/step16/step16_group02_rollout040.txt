import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(ind):
        h = np.clip(ind, 0, 1)
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            scale = 1.0 / total
            h = np.clip(h * scale, 0, 1)
        return h

    def compute_c5(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        return compute_c5(h)

    if 'initial_h_values' in kwargs and initial_h_values is not None and len(initial_h_values) == n_points:
        h_init = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h_init = np.clip(1 - np.abs(x - 1), 0, 1)

    h_init = repair(h_init)
    pop_size = 100
    population = [h_init.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        population[i] = repair(population[i])

    start_time = time.time()
    best_h = h_init.copy()
    best_c5 = fitness(best_h)

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            if np.random.rand() < 0.9:
                alpha = np.random.rand()
                c1 = alpha * p1 + (1 - alpha) * p2
                c2 = alpha * p2 + (1 - alpha) * p1
            else:
                split = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
            if np.random.rand() < 0.3:
                c1 += np.random.normal(0, 0.1, n_points)
                c2 += np.random.normal(0, 0.1, n_points)
            new_pop.extend([repair(c1), repair(c2)])
        population = new_pop

        f_vals = [fitness(p) for p in population]
        if np.min(f_vals) < best_c5:
            best_idx = np.argmin(f_vals)
            best_c5 = f_vals[best_idx]
            best_h = population[best_idx].copy()

    return best_h, best_c5, n_points
