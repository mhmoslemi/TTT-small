        if result.success:
            current_h = result.x
            h_full = np.concatenate([current_h, current_h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_first_half = current_h
    
    best_h_full = np.concatenate([best_h_first_half, best_h_first_half[::-1]])
    return best_h_full, best_c5, n_points
