import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(individual):
        h = np.clip(individual, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-6:
            h = np.full_like(h, 0.5)
        else:
            scale = integral_target / total
            h = np.clip(h * scale, 0, 1)
        return h

    def compute_c5(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
        h_initial = h_initial[:n_points//2]

    h_initial_repaired = repair(h_initial)
    best_h = h_initial_repaired.copy()
    best_c5 = compute_c5(best_h)

    pop_size = 100
    population = [repair(np.random.rand(n_points)) for _ in range(pop_size - 1)]
    population.append(repair(h_initial))

    mutation_prob = 0.3
    crossover_prob = 0.9
    n_gen = int(budget_s * 0.8)

    for gen in range(n_gen):
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        new_population = []
        for _ in range(pop_size):
            p1, p2 = np.random.randint(pop_size, size=2)
            parent1, parent2 = population[p1], population[p2]

            child = np.zeros_like(parent1)
            if np.random.rand() < crossover_prob:
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = parent1.copy()

            for i in range(n_points):
                if np.random.rand() < mutation_prob:
                    child[i] += np.random.normal(0, 0.1)
            child = repair(child)
            new_population.append(child)

        population = new_population

    return best_h, best_c5, n_points
