    initial_guess = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        initial_guess.append(h_initial)
    
    # Additional initialization
    binary_initial = np.zeros(n_points)
    num_ones = int(target_sum)
    binary_initial[:num_ones] = 1
    initial_guess.append(binary_initial)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for h_start in initial_guess:
        h_scaled = h_start * (target_sum / np.sum(h_start))
        result = scipy.optimize.minimize(
            compute_overlap, 
            h_scaled, 
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}],
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 500, 'disp': False}
        )
        
        if result.success:
            curr_c5 = compute_overlap(result.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = result.x
    
    return best_h, best_c5, n_points

def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    max_corr = 0.0
    for k in range(-n + 1, n):
        if k == 0:
            continue
        x_start = max(0, -k)
        x_end = min(n - 1, n - k - 1)
        if x_start > x_end:
            continue
        total = 0.0
        for i in range(x_start, x_end + 1):
            j = int(round((i * dx + k * dx) / dx))
            if 0 <= j < n:
                total += h[i] * (1 - h[j])
        max_corr = max(max_corr, total * dx)
    return max_corr
