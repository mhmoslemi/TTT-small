import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2

    # Initialize with provided starting values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
    else:
        # Generate a starting pattern
        t = np.linspace(0, 2*np.pi, n_points)
        pattern1 = 0.5 * (np.sin(t) ** 2)
        pattern2 = 0.5 * (1 + np.cos(t)) / 2
        alpha = np.random.rand()
        h_initial = alpha * pattern1 + (1 - alpha) * pattern2
        h_initial = np.clip(h_initial, 0.0, 1.0)
        # Scale to meet integral target
        current_sum = np.sum(h_initial)
        h_initial = h_initial * (integral_target / current_sum)
    
    # Repair function to enforce constraints
    def repair(x):
        x = np.clip(x, 0.0, 1.0)
        current_sum = np.sum(x)
        if np.isclose(current_sum, integral_target, atol=1e-6):
            return x
        # Scale to correct sum
        scale = integral_target / current_sum
        x = x * scale
        x = np.clip(x, 0.0, 1.0)
        # If still not correct, add remaining sum to last element
        remaining = integral_target - np.sum(x)
        if remaining > 1e-6:
            x[-1] += remaining
            x = np.clip(x, 0.0, 1.0)
        return x
    
    # Compute overlap
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # Objective function for differential evolution
    def objective(x):
        return compute_overlap(repair(x))
    
    # Bounds for differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    start_time = time.time()
    result_de = opt.differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        maxiter=50, 
        popsize=20, 
        tol=1e-6, 
        mutation=(0.5, 1),

        seed=seed
    )
    elapsed = time.time() - start_time

    )

    final_c5 = compute_overlap(final_h)
