import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        base = np.array(initial_h_values[:n_points//2])
    else:
        base = np.random.rand(n_points//2)
        base *= integral_target / np.sum(base)

    def repair(x):
        x_clipped = np.clip(x, 0, 1)
        total = np.sum(x_clipped)
        if total < 1e-6:
            return np.random.rand(len(x)) * integral_target / np.sum(np.random.rand(len(x)))
        scale = integral_target / total
        x_scaled = x_clipped * scale
        x_final = np.clip(x_scaled, 0, 1)
        return x_final

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x_repaired = repair(x)
        return compute_overlap(x_repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=10,
        tol=1e-12,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, n_points
