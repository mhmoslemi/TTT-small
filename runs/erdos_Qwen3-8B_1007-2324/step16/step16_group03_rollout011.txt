import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_opt = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_opt / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        v_initial = h_initial[:n_opt]
    else:
        v_initial = np.random.uniform(0, 1, n_opt)

    def objective(v):
        h = np.concatenate([v, v[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(v) - integral_target) ** 3
        return max_overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_opt)]
    start_time = time.time()

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=10,
        mutation=(0.8, 1.0),
        recombination=0.9,
        tol=1e-8,
        maxiter=200,
        mutation_size=0.5,
        vectors_evaluated=1000
    )

    if result.success and (time.time() - start_time) < budget_s:
        best_v = result.x
    else:
        best_v = v_initial

    h_best = np.concatenate([best_v, best_v[::-1]])
    corr_best = np.correlate(h_best, 1 - h_best, mode='full')
    c5_best = np.max(corr_best) * dx

    return h_best, c5_best, n_points
