import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
    
    def repair(individual):
        h = np.clip(individual, 0, 1)
        total = np.sum(h)
        if abs(total - integral_target) > 1e-4:
            h = h * (integral_target / total)
        return h
    
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    pop_size = 100
    individuals = [repair(np.random.rand(n_points)) for _ in range(pop_size)]
    best_h = repair(h_initial)
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for _ in range(int((budget_s * 0.8) / 0.01)):
        fitness = np.array([compute_c5(ind) for ind in individuals])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = individuals[best_idx].copy()
        
        if time.time() - start_time > budget_s - 0.1:
            break
        
        new_population = []
        for _ in range(pop_size // 2):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            parent1, parent2 = individuals[idx1], individuals[idx2]
            
            if np.random.rand() < 0.9:
                alpha = np.random.rand(n_points)
                child = parent1 * alpha + parent2 * (1 - alpha)
            else:
                child = np.random.rand(n_points)
            
            child = repair(child)
            new_population.append(child)
        
        individuals = new_population + [best_h] * (pop_size - len(new_population))
    
    return best_h, best_c5, n_points
