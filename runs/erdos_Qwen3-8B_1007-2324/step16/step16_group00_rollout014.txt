import numpy as np
import time
import scipy.optimize
import sys

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    integral_target = n_points / 2
    dx = 2.0 / n_points

    initial_h_values = kwargs.get('initial_h_values', None)
    start_points = []

    if initial_h_values is not None and len(initial_h_values) == n_points:
        start_points.append(np.array(initial_h_values))
    
    # Perturb initial guess with small noise
    if initial_h_values is not None and len(initial_h_values) == n_points:
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(initial_h_values + perturbation, 0, 1)
        start_points.append(perturbed)
    
    # Add other starting guesses if not already in start_points
    if len(start_points) < 3:
        # Binary initial guess
        binary_initial = np.zeros(n_points // 2)
        num_ones = (n_points // 2) // 2
        indices = np.random.choice(n_points // 2, num_ones, replace=False)
        binary_initial[indices] = 1
        binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
        start_points.append(binary_full[:n_points])
        
        # Random initial guess
        random_initial = np.random.rand(n_points)
        random_initial = random_initial * (integral_target / np.sum(random_initial))
        start_points.append(random_initial)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)

    start_time = time.time()
    for h_start in start_points:
        # Scale to meet integral constraint
        h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        # Optimization with trust-constr
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='trust-constr',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
            ],
            options={
                'maxiter': 200,
                'gtol': 1e-8,
                'ftol': 1e-8,
                'disp': False
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final evaluation
    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, n_points
