    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    
    # Binary initial guess
    binary_initial = np.zeros(n_half)
    num_ones = n_half // 2
    indices = np.random.choice(n_half, num_ones, replace=False)
    binary_initial[indices] = 1
    binary_full = np.concatenate([binary_initial, binary_initial[::-1]])
    start_points.append(binary_full[:n_points])
    
    # Random initial guess
    random_initial = np.random.rand(n_half)
    random_initial = random_initial * (n_points / 4 / np.sum(random_initial))
    random_full = np.concatenate([random_initial, random_initial[::-1]])
    start_points.append(random_full[:n_points])
    
    best_c5 = np.inf
    best_h_full = np.zeros(n_points)
    
    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start[:n_half],
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_h = result.x
            current_h_full = np.concatenate([current_h, current_h[::-1]])
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_full = current_h_full
    
    return best_h_full, best_c5, n_points
