import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    n_opt = n_points // 2
    integral_target = n_opt / 2  # sum(v) * dx * 2 = 1

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        step = int(np.ceil(len(h_initial) / n_points))
        v_initial = h_initial[::step][:n_opt]
    else:
        v_initial = np.random.uniform(0, 1, n_opt)
        while np.abs(np.sum(v_initial) - integral_target) > 1e-6:
            v_initial = np.random.uniform(0, 1, n_opt)

    def objective(v):
        h = np.concatenate([v, v[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        score = np.max(corr) * dx
        penalty = 1e5 * abs(np.sum(v) - integral_target) ** 3
        return score + penalty

    bounds = [(0.0, 1.0) for _ in range(n_opt)]

    start_time = time.time()
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=10,
        mutation=(0.8, 1.0),
        recombination=0.9,
        tol=1e-8,
        maxiter=500,
        callback=lambda x: None,
        disp=False,
        polish=True,
        seed=seed
    )

    best_v = result.x
    best_score = objective(best_v)

    # Fallback to initial guess if optimization fails
    if time.time() - start_time > budget_s * 0.9 or result.success:
        best_v = v_initial
        best_score = objective(best_v)

    h_best = np.concatenate([best_v, best_v[::-1]])
    return h_best, best_score, n_points
