import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    # Configuration
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) = n_points / 2 ensures ∫h(x)dx = 1

    # Initialize population
    def repair(individual):
        h = np.clip(individual, 0, 1)
        total = np.sum(h)
        if abs(total - target_sum) > 1e-6:
            scale = target_sum / total
            h = np.clip(h * scale, 0, 1)
        return h

    # Compute C5
    def compute_c5(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Tournament selection
    def tournament_selection(population, k=3):
        candidates = np.random.choice(len(population), k)
        return population[np.argmin([compute_c5(population[i]) for i in candidates])]

    # Initialize population
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values)
    else:
        x = np.linspace(-1, 1, n_points)
        h_initial = np.clip(1 - np.abs(x), 0, 1)
        h_initial = h_initial[:n_points//2]
    population = [repair(np.random.rand(n_points)) for _ in range(100)]
    population = [repair(h) for h in population]
    population[0] = repair(h_initial)

    best_h = population[0]
    best_c5 = compute_c5(best_h)
    time_start = time.time()

    # Genetic algorithm
    while time.time() - time_start < budget_s:
        new_population = []
        for _ in range(len(population) // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            # Crossover
            child1 = np.copy(parent1)
            child2 = np.copy(parent2)
            if np.random.rand() < 0.9:
                split = np.random.randint(1, n_points - 1)
                child1[:split] = parent2[:split]
                child2[:split] = parent1[:split]
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.3:
                    child1[i] += np.random.normal(0, 0.05)
                    child2[i] += np.random.normal(0, 0.05)
            child1 = repair(child1)
            child2 = repair(child2)
            new_population.extend([child1, child2])
        # Elitism
        population = new_population[:len(population)]
        current_best = min(population, key=compute_c5)
        if compute_c5(current_best) < best_c5:
            best_c5 = compute_c5(current_best)
            best_h = current_best

    return best_h, best_c5, n_points
