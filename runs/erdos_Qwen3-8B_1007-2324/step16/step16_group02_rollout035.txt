        h_opt = np.clip(h_opt, 0, 1)

    # Function to evaluate maximum overlap integral
    def compute_overlap(h_val):
        h_full = np.concatenate([h_val, h_val[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Objective for optimization
    def objective(x):
        return compute_overlap(x)

    # Constraints: 0 ≤ h[i] ≤ 1 and ∑h[i] = integral_target
    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': lambda x: np.ones_like(x)}

    best_c5 = np.inf
    best_h = h_opt.copy()

    # Run multi-strategy optimization within time budget
    start_time = time.time()
    for _ in range(10):  # Balance exploration with time
        # Strategy 1: Perturb current best
        perturb = np.random.normal(0, 0.05, len(h_opt))
        h_perturbed = np.clip(h_opt + perturb, 0.0, 1.0)
        h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))

        result = scipy.optimize.minimize(
            objective, h_perturbed, method='SLSQP',
            bounds=bounds, constraints=constraint,
            options={'ftol': 1e-12, 'maxiter': 80, 'disp': False, 'iprint': -1}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

        # Strategy 2: Mirror initial guess and tweak
        h_alt = np.concatenate([h_opt, h_opt[::-1]])[::-1]
        h_alt = np.clip(h_alt, 0.0, 1.0)
        h_alt = h_alt[:len(h_opt)]
        perturb = np.random.normal(0, 0.05, len(h_alt))
        h_alt = np.clip(h_alt + perturb, 0.0, 1.0)
        h_alt = h_alt * (integral_target / np.sum(h_alt))

        result = scipy.optimize.minimize(
            objective, h_alt, method='SLSQP',
            bounds=bounds, constraints=constraint,
            options={'ftol': 1e-12, 'maxiter': 80, 'disp': False, 'iprint': -1}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

        # Strategy 3: Random restart
        h_rand = np.random.rand(len(h_opt))
        h_rand = h_rand * (integral_target / np.sum(h_rand))

        result = scipy.optimize.minimize(
            objective, h_rand, method='SLSQP',
            bounds=bounds, constraints=constraint,
            options={'ftol': 1e-12, 'maxiter': 80, 'disp': False, 'iprint': -1}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

        # Early exit if budget is exceeded
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final output: optimized h, C₅, and resolution
    return best_h, best_c5, n_points
