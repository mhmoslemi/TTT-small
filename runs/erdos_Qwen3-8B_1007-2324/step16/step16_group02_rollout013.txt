import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize population
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values)
    else:
        x = np.linspace(-1, 1, n_points)
        h_initial = np.clip(1 - np.abs(x), 0, 1)
        h_initial = h_initial[:n_points//2]

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-8:
            return np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        scale = target_sum / total
        return h * scale

    h_initial = repair(h_initial)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    pop_size = 100
    mutation_prob = 0.3
    crossover_prob = 0.8
    max_iter = int(budget_s * 0.8)

    population = [h_initial.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        population.append(repair(h_rand))

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    for _ in range(max_iter):
        def fitness(ind):
            return compute_c5(ind)
        fitnesses = np.array([fitness(ind) for ind in population])
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx]

        if time.time() - start_time > budget_s:
            break

        new_pop = []
        for _ in range(pop_size):
            a, b = np.random.choice(pop_size, 2, replace=False)
            parent = population[a] if fitnesses[a] < fitnesses[b] else population[b]
            child = parent.copy()
            if np.random.rand() < crossover_prob:
                other = population[np.random.randint(pop_size)]
                child = (parent + np.random.rand(n_points) * other) / 2
            if np.random.rand() < mutation_prob:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            child = repair(child)
            new_pop.append(child)
        population = new_pop

    return best_h, compute_c5(best_h), n_points
