def run(seed=42, budget_s=100, **kwargs):

    n_points = 60

    target_sum = n_points / 2
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values)
    else:

        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - target_sum) > 1e-5:
            scale = target_sum / total if total != 0 else 1.0
            h = np.minimum(np.maximum(h * scale, 0), 1)
        return h
    
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def fitness(h):
        return compute_c5(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return repair(h)
    
    def crossover(a, b):
        return np.where(np.random.rand(len(a)) < 0.5, a, b)
    
    h_pop = [repair(np.random.rand(n_points)) for _ in range(100)]
    h_pop = [repair(h) for h in h_pop]
    
    start_time = time.time()

        new_pop = []

        h_pop = new_pop
