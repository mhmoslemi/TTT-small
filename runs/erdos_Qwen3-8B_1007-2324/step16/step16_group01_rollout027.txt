import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    half_len = n_points // 2

    def compute_overlap(x):
        h_full = np.concatenate([x, x[::-1]])
        one_minus_h = 1 - h_full
        corr = np.correlate(h_full, one_minus_h, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = initial_h_values[:half_len]
        h_initial_sum = np.sum(h_initial)
        if h_initial_sum > 0:
            h_initial = h_initial * (integral_target / 2) / h_initial_sum
    else:
        h_initial = np.random.rand(half_len)
        h_initial = h_initial / np.sum(h_initial) * (integral_target / 2)
        h_initial = np.clip(h_initial, 0, 1)

    bounds = [(0.0, 1.0) for _ in range(half_len)]
    integral_half = integral_target / 2

    def objective(x):
        return compute_overlap(x) + 1e6 * np.abs(np.sum(x) - integral_half)

    result_de = scipy.optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=50,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed
    )
    best_x = result_de.x

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_half}]
    result_refine = scipy.optimize.minimize(
        fun=compute_overlap,
        x0=best_x,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )

    best_x = result_refine.x
    best_x = np.clip(best_x, 0, 1)
    h_full = np.concatenate([best_x, best_x[::-1]])
    sum_h = np.sum(h_full) * dx
    if abs(sum_h - 1) > 1e-6:
        scale = 1.0 / sum_h
        h_full = h_full * scale

    return h_full, compute_overlap(best_x), n_points
