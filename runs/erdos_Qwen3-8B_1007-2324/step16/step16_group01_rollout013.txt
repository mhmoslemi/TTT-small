import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    half_length = n_points // 2
    dx = 2.0 / n_points
    target_sum = 15  # integral constraint

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:half_length].copy()
        h_opt = h_initial_half * target_sum / np.sum(h_initial_half)
    else:
        h_opt = np.random.rand(half_length)
        h_opt = h_opt * target_sum / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Stochastic refinement phase
    for _ in range(50):
        perturb = np.random.normal(0, 0.05, half_length)
        h_candidate = h_opt + perturb
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        if compute_overlap(h_candidate) < compute_overlap(h_opt):
            h_opt = h_candidate.copy()

    # Deterministic optimization
    bounds = [(0.0, 1.0) for _ in range(half_length)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result = scipy.optimize.minimize(
        compute_overlap,
        h_opt,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-9}
    )

    if result.success:
        h_opt = result.x.copy()
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * target_sum / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)
    else:
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * target_sum / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)

    h_full = np.concatenate([h_opt, h_opt[::-1]])
    return h_full, compute_overlap(h_opt), n_points
