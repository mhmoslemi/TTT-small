import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    parent = kwargs.get('initial_h_values', None)
    if parent is not None and len(parent) == n_points:
        h_initial = np.array(parent)
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
        h_initial = h_initial[:n_points//2]

    def repair(individual):
        h = np.clip(individual, 0, 1)
        total = np.sum(h)
        if abs(total - integral_target) > 1e-5:
            scale = integral_target / total
            h = np.minimum(np.maximum(h * scale, 0), 1)
        return h

    def compute_c5(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def evaluate(individual):
        return compute_c5(individual)

    def tournament_selection(population):
        idxs = np.random.choice(len(population), 3)
        return population[idxs[np.argmin([evaluate(p) for p in population[idxs]])]]

    h_initial = repair(h_initial)
    pop_size = 100
    pop = [repair(np.random.rand(n_points)) for _ in range(pop_size)]
    best_h = h_initial
    best_c5 = evaluate(best_h)
    gen = 0
    max_gen = int(budget_s * 0.9)

    while gen < max_gen and time.time() < time.time() + budget_s:
        new_pop = []
        for _ in range(pop_size // 2):
            a = tournament_selection(pop)
            b = tournament_selection(pop)
            if np.random.rand() < 0.9:
