import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_opt = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        step = int(np.ceil(len(h_initial) / n_points))
        h_opt = h_initial[::step][:n_opt]
    else:
        h_opt = np.random.uniform(0, 1, n_opt)
        while np.abs(np.sum(h_opt) - n_opt / 2) > 1e-5:
            h_opt = np.random.uniform(0, 1, n_opt)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_opt / 2,
             'jac': lambda x: np.ones_like(x)})

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    time_left = budget_s * 0.9

    for _ in range(20):
        if time_left < 0.1:
            break
        if np.random.rand() < 0.8:
            perturb = np.random.uniform(-0.1, 0.1, n_opt)
            h_candidate = np.clip(h_opt + perturb, 0, 1)
        else:
            h_candidate = np.random.uniform(0, 1, n_opt)
            while np.abs(np.sum(h_candidate) - n_opt / 2) > 1e-5:
                h_candidate = np.random.uniform(0, 1, n_opt)

        result = scipy.optimize.minimize(
            objective, h_candidate, method='SLSQP', bounds=[(0,1)]*n_opt,
            constraints=cons, options={'maxiter': 150, 'ftol': 1e-8, 'disp': False})
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
            time_left -= time.time() - start_time

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
