        max_corr = np.max(corr) * dx
        penalty = 1e5 * (sum_h - target_sum) ** 2
        return max_corr + penalty

    start_points = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
    else:
        binary_initial = np.zeros(n_points)
        num_ones = n_points // 3
        indices = np.random.choice(n_points, num_ones, replace=False)
        binary_initial[indices] = 1
        start_points.append(binary_initial)
        random_initial = np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points)))
        start_points.append(random_initial)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for h_start in start_points:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = scipy.optimize.differential_evolution(
            compute_overlap,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            disp=False
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        initial_c5 = compute_overlap(h_start)
        if initial_c5 < best_c5:
            best_c5 = initial_c5
            best_h = h_start.copy()

    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = np.clip(best_h * scale, 0, 1)

    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, n_points
