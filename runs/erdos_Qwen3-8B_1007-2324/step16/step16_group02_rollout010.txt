import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize with provided starting values if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values[:n_points])
        # Generate initial population from parent and random samples
        pop_size = 15
        pop = []
        for _ in range(pop_size):
            perturb = np.random.normal(0, 0.1, n_points)
            candidate = h_parent + perturb
            candidate = np.clip(candidate, 0.0, 1.0)
            sum_candidate = np.sum(candidate)
            if sum_candidate > 1e-6:
                candidate *= integral_target / sum_candidate
            else:
                candidate = np.random.rand(n_points)
                candidate *= integral_target / np.sum(candidate)
            pop.append(candidate)
        # Add some random samples to diversify
        for _ in range(pop_size // 2):
            candidate = np.random.rand(n_points)
            candidate *= integral_target / np.sum(candidate)
            pop.append(candidate)
    else:
        # Random initial population
        pop = []
        for _ in range(15):
            candidate = np.random.rand(n_points)
            candidate *= integral_target / np.sum(candidate)
            pop.append(candidate)
    pop = np.array(pop)

    # Objective function with repair
    def objective(h):
        h_repaired = np.clip(h, 0.0, 1.0)
        sum_repaired = np.sum(h_repaired)
        if sum_repaired < 1e-6:
            return float('inf')
        scale = integral_target / sum_repaired
        h_scaled = h_repaired * scale
        h_final = np.clip(h_scaled, 0.0, 1.0)
        h_full = np.concatenate([h_final, h_final[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Run differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(

        strategy='best1bin',
        mutation=(0.8, 0.8),
        recombination=0.9,
        popsize=15,
        tol=1e-12,
        maxiter=100,
        disp=False,
        seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
