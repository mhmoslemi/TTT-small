import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = 15  # sum of h_half

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_half = initial_h[:n_points//2]
        sum_initial = np.sum(h_half)
        if sum_initial > 0:
            h_half = h_half * integral_target / sum_initial
        h_half = np.clip(h_half, 0, 1)
    else:
        h_half = np.random.rand(n_points//2)
        h_half = h_half / np.sum(h_half) * integral_target
        h_half = np.clip(h_half, 0, 1)
    population = [h_half.copy() for _ in range(20)]

    def compute_overlap(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def evaluate(ind):
        return compute_overlap(ind)

    # Genetic algorithm parameters
    generations = 30
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.7

    for gen in range(generations):
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]

        new_population = []
        for _ in range(len(population)):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[idx1], population[idx2]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, len(p1))
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, child.shape)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 0:
                    scale = integral_target / sum_child
                    child *= scale
                child = np.clip(child, 0, 1)
            new_population.append(child)
        population = new_population

    # Select best from GA
    fitness = [evaluate(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]

    # Local optimization
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target,
             'jac': lambda x: np.ones_like(x)})
    bounds = [(0.0, 1.0) for _ in range(len(best_h))]
    result = scipy.optimize.minimize(
        objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-9}

    if result.success:

            h_opt = np.clip(h_opt, 0, 1)
        current_c5 = objective(h_opt)
        if current_c5 < best_c5:

            best_h = h_opt

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
