import numpy as np

import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    n_opt = n_points // 2
    integral_target = n_opt / 2  # sum(h_opt) * dx = 1 => sum(h_opt) = n_opt / 2 * dx * n_points = n_opt / 2
    dx_half = dx * n_points  # dx * n_points is 2, but for h_opt sum is n_opt / 2

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

        return compute_overlap(x)

    best_c5 = np.inf
    best_h = np.zeros(n_points)
