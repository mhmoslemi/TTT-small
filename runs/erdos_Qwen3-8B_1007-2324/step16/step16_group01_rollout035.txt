        h_opt = np.clip(h_opt, 0, 1)
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        sum_full = np.sum(h_full) * dx
        if abs(sum_full - 1) > 1e-6:
            h_opt = h_opt * (integral_target / 2) / np.sum(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
        current_c5 = compute_overlap(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
