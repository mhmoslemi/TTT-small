import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 4
    n_half = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        step = int(np.ceil(len(h_initial) / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)

    def objective(v):
        h = np.concatenate([v, v[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        score = np.max(corr) * dx
        penalty = 1e5 * abs(np.sum(v) - integral_target)**3
        return score + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=10,
        tol=1e-6,
        mutation=(0.8, 1.2),
        recombination=0.9,
        maxiter=100,
        seed=seed,
        disp=False
    )

    best_v = result.x
    best_score = objective(best_v)
    best_h = np.concatenate([best_v, best_v[::-1]])

    return best_h, best_score, n_points
