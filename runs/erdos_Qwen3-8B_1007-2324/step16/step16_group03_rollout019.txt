import scipy.optimize
import time

    integral_target = n_points / 4
    n_half = n_points // 2

            h_opt = np.random.choice([0, 1], size=n_half, p=[0.3, 0.7])

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    time_limit = budget_s * 0.8
    start_time = time.time()

    def objective(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        return overlap

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    result = scipy.optimize.minimize(
