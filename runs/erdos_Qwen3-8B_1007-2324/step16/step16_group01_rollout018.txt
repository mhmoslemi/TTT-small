import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = 15
    n_half = n_points // 2
    
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_full = np.concatenate([h_initial[:n_half], h_initial[:n_half][::-1]])
        h_full = h_full / np.sum(h_full) * (target_sum * 2)
        h_full = np.clip(h_full, 0, 1)
        h_opt = h_full[:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt = h_opt / np.sum(h_opt) * target_sum
        h_opt = np.clip(h_opt, 0, 1)
    
    def optimize(start):
        result = scipy.optimize.minimize(
            objective,
            start,
            method='SLSQP',
            bounds=[(0,1)]*n_half,
            constraints={'type':'eq','fun':constraint},
            options={'maxiter':200,'ftol':1e-9,'eps':1e-8}
        )
        return result.x
    
    best_h = optimize(h_opt)
    best_c5 = objective(best_h)
    
    if 'initial_h_values' in globals():
        h_perturb = h_opt + np.random.normal(0, 0.05, n_half)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        perturb_result = optimize(h_perturb)
        if objective(perturb_result) < best_c5:
            best_h = perturb_result
            best_c5 = objective(best_h)
        
        x = np.linspace(0, 1, n_half)
        h_sine = 0.5 * (1 + np.sin(np.pi * x))
        h_sine = np.clip(h_sine, 0, 1)
        h_sine = h_sine * (target_sum / np.sum(h_sine))
        sine_result = optimize(h_sine)
        if objective(sine_result) < best_c5:
            best_h = sine_result
            best_c5 = objective(best_h)
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
