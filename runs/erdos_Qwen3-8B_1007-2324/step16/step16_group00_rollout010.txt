import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    integral_target = n_points / 2
    time_limit = budget_s

    def initial_guess():
        if 'initial_h_values' in globals():
            h_initial = np.array(initial_h_values)
            h_initial = h_initial[:n_points]
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (integral_target / np.sum(h_initial))
        return h_initial

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Provided initial guess
    start_points = [initial_guess()]

    # Add structured binary patterns
    for _ in range(2):
        binary_pattern = np.zeros(n_points)
        num_ones = n_points // 4
        indices = np.random.choice(n_points, num_ones, replace=False)
        binary_pattern[indices] = 1
        start_points.append(binary_pattern)

    # Add a symmetric triangular profile
    def triangular_profile():
        t = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h = np.minimum(1, (t - 1) / 1.0)
        h = np.minimum(1, 1 - (t - 1) / 1.0)
        return h * (integral_target / np.sum(h))
    start_points.append(triangular_profile())

    best_c5 = np.inf
    best_h = np.zeros(n_points)
    total_time = 0

    for h_start in start_points:
        start_time = time.time()
        # Enforce integral constraint
        h_start = h_start * (integral_target / np.sum(h_start))
        # Ensure domain bounds
        h_start = np.clip(h_start, 0.0, 1.0)

        def objective(x):
            h = x.copy()
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - integral_target

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = scipy.optimize.minimize(
            objective, h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}

        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            total_time += time.time() - start_time
            if total_time > time_limit:
                break

    # Final check and output

    final_c5 = compute_overlap(best_h)
