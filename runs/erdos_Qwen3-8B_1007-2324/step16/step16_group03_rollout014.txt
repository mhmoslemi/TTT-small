import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_opt = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_opt]
    else:
        h_opt = np.random.uniform(0, 1, n_opt).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_opt).clip(0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    time_left = budget_s * 0.9
    start_time = time.time()

    def objective(x):
        return compute_overlap(x)

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    def perturb(x):
        noise = np.random.normal(0, 0.1, len(x))
        return np.clip(x + noise, 0, 1)

    for _ in range(10):
        if time.time() - start_time > time_left:
            break
        x0 = perturb(h_opt)
        result = scipy.optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=[(0., 1.)]*len(x0),
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
