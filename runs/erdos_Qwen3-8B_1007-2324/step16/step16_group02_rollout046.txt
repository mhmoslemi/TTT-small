                for i in range(len(x_scaled)):
                    if x_scaled[i] < 1.0:
                        add = excess * (1.0 - x_scaled[i]) / len(x_scaled)
                        x_scaled[i] += add
                        excess -= add
                        if np.abs(excess) < 1e-6:
                            break
        return x_scaled

    def objective(x):
        x_repaired = repair(x)
        h_full = np.concatenate([x_repaired, x_repaired[::-1]])
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    if 'initial_h_values' in globals():
        initial_guess = repair(np.array(initial_h_values))
    else:
        initial_guess = np.random.rand(n_points)
        initial_guess = repair(initial_guess)

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        disp=False,
        polish=True
    )

    best_x = result.x
    best_c5 = result.fun

    result_local = minimize(
        objective,
        best_x,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(repair(x)) - integral_target}]
    )

    if result_local.success:
        best_x = result_local.x
        best_c5 = result_local.fun

    best_x_repaired = repair(best_x)
    final_c5 = objective(best_x_repaired)
    return best_x_repaired, final_c5, n_points
