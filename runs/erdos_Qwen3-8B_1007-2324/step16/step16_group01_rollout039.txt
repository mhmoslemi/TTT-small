            x_normed = np.zeros_like(x)
        else:
            x_normed = x * (15.0 / np.sum(x))
        h_full = np.concatenate([x_normed, x_normed[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds_de = [(0.0, 1.0) for _ in range(n_points//2)]
    
    result_de = scipy.optimize.differential_evolution(
        objective_de,
        bounds_de,
        popsize=50,
        maxiter=100,
        mutation=(0.5, 1.0),
        recombination=0.7,
        strategy='best1bin',
        tol=1e-6,
        disp=False
    )
    
    best_x_de = result_de.x
    
    def objective_local(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - integral_target_half
    
    bounds_lbg = [(0.0, 1.0) for _ in range(n_points//2)]
    
    result_lbg = scipy.optimize.minimize(
        objective_local,
        best_x_de,
        method='L-BFGS-B',
        bounds=bounds_lbg,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 500, 'disp': False}
    )
    
    best_x = result_lbg.x
    
    h_full = np.concatenate([best_x, best_x[::-1]])
    sum_h = np.sum(h_full) * dx
    if abs(sum_h - 1) > 1e-6:
        best_x = best_x * integral_target_half / np.sum(best_x)
        h_full = np.concatenate([best_x, best_x[::-1]])
    
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx
    
    return h_full, c5_bound, n_points
