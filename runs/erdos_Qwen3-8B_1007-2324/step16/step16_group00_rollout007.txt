import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def project_to_feasible(x):
        x = np.clip(x, 0, 1)
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-6:
            return x
        if diff > 0:
            idx = np.argmin(x)
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = np.argmax(x)
            x[idx] = max(0.0, x[idx] + diff)
        return x

    def compute_overlap(h):
        h = project_to_feasible(h)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        start_points = [h_initial]
    else:
        start_points = []

    half = n_points // 2
    binary = np.zeros(half)
    idxs = np.random.choice(half, half // 2, replace=False)
    binary[idxs] = 1
    start_points.append(np.concatenate([binary, binary[::-1]]))

    random = np.random.rand(half)
    random = random * (target_sum / np.sum(random))
    start_points.append(np.concatenate([random, random[::-1]]))

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for h_start in start_points:
        h_start = np.clip(h_start, 0, 1)
        h_start = project_to_feasible(h_start)
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start[:n_points//2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points//2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum / 2}],
            options={'ftol': 1e-10, 'maxiter': 150, 'disp': False}
        )
        if result.success:
            h_candidate = project_to_feasible(result.x)
            h_full = np.concatenate([h_candidate, h_candidate[::-1]])
            c5 = compute_overlap(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
