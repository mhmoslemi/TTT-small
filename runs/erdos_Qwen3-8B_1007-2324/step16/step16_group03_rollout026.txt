import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 4
    n_half = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        step = int(np.ceil(len(h_initial) / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    time_limit = budget_s * 0.9
    start_time = time.time()

    def objective(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        return overlap

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    result = scipy.optimize.minimize(
        objective,
        x0=h_opt,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Additional refinement with CG
    result = scipy.optimize.minimize(
        objective,
        x0=best_h,
        method='CG',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=cons,
        options={'maxiter': 100, 'gtol': 1e-8}
    )

    if result.success and compute_overlap(result.x) < best_c5:
        best_c5 = compute_overlap(result.x)
        best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
