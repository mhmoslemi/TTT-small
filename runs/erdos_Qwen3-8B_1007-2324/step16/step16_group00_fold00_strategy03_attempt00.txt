import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, n_initial=60, n_max=240):
    np.random.seed(seed)
    global initial_h_values
    if hasattr(initial_h_values, 'shape') and len(initial_h_values) > 1:
        h_parent = initial_h_values
    else:
        h_parent = np.ones(n_initial) * 0.5  # Ensure fallback
    
    n_start = len(h_parent)
    dx = 2.0 / n_initial
    integral_target = n_initial  # Since SLSQP used mode="full" from n_parent=60

    def run_level(n):
        def project_feasible(x):
            return np.clip(x, 0, 1) * (integral_target/np.sum(x)) if np.sum(x>0)>0 else x.copy()

        def objective(x):
            projection = project_feasible(x)
            h_full = np.concatenate([projection, 1-projection[::-1]])
            corr = np.correlate(h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0,1)]*n
        res_de = optimize.differential_evolution(objective, bounds, seed=seed, max_evals=200)
        best_cand = res_de.x
        best_val = objective(best_cand)

        return best_cand, best_val

    results = {}
    levels = [120, 240]
    start_time = time.time()
    for level in levels:
        if (time.time() - start_time) > budget_s * 0.9: break
        h_cur, c5_val = run_level(level)
        results[level] = (h_cur, c5_val)
    
    best_solution = min([(k, v) for k, (h, v) in results.items()], key=lambda x: x[1])[1
    return best_solution, best_solution, len(best_solution)
