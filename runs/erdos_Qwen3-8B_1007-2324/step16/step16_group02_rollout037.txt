    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(len(h_opt))}
    
    best_c5 = np.inf
    best_h = h_opt.copy()
    
    for _ in range(5):
        perturb = np.random.normal(0, 0.1, len(h_opt))
        h_perturbed = np.clip(h_opt + perturb, 0.0, 1.0)
        if np.sum(h_perturbed) > 1e-6:
            h_perturbed *= integral_target / np.sum(h_perturbed)
        else:
            h_perturbed = np.random.rand(len(h_opt))
            h_perturbed *= integral_target / np.sum(h_perturbed)
        
        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'ftol': 1e-10, 'maxiter': 100, 'disp': False, 'eps': 1e-8, 'iprint': -1}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    h_alt = np.concatenate([h_opt, h_opt[::-1]])[::-1]
    h_alt = np.clip(h_alt, 0.0, 1.0)
    h_alt = h_alt[:len(h_opt)]
    perturb = np.random.normal(0, 0.1, len(h_alt))
    h_alt = np.clip(h_alt + perturb, 0.0, 1.0)
    if np.sum(h_alt) > 1e-6:
        h_alt *= integral_target / np.sum(h_alt)
    else:
        h_alt = np.random.rand(len(h_alt))
        h_alt *= integral_target / np.sum(h_alt)
    
    result = scipy.optimize.minimize(
        objective,
        h_alt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'ftol': 1e-10, 'maxiter': 100, 'disp': False, 'eps': 1e-8, 'iprint': -1}
    )
    if result.success:
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    h_rand = np.random.rand(len(h_opt))
    h_rand *= integral_target / np.sum(h_rand)
    result = scipy.optimize.minimize(
        objective,
        h_rand,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'ftol': 1e-10, 'maxiter': 100, 'disp': False, 'eps': 1e-8, 'iprint': -1}
    )
    if result.success:
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
