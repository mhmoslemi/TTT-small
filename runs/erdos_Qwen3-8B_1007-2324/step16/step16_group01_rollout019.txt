        step = int(np.ceil(len(h_initial) / n_points))

        if np.sum(h_opt) > 0:
            h_opt /= np.sum(h_opt)

        h_opt *= integral_target / 2
        h_opt = np.clip(h_opt, 0, 1)
    
    def compute_overlap(h_half):
        h_full = np.concatenate([h_half, h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    constraints = ({
        'type': 'eq',

    })

    result = scipy.optimize.minimize(
