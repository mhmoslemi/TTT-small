            constraints=constraint,
            options={
                'ftol': 1e-12,
                'maxiter': 50,
                'disp': False,
                'eps': 1e-8,
                'iprint': -1
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    return final_h, final_c5, n_points
