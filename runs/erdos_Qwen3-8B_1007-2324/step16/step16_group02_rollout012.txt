import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    global initial_h_values

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - target_sum) > 1e-5:
            scale = target_sum / total
            h = np.minimum(np.maximum(h * scale, 0), 1)
        return h

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def evaluate(h):
        try:
            return compute_c5(h)
        except:
            return 1e10

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_pop = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h_pop = np.clip(1 - np.abs(x - 1), 0, 1)
        h_pop = h_pop[:n_points//2]

    h_pop = repair(h_pop)

    pop_size = 100
    best_h = h_pop.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(200):
        if time.time() > budget_s - 0.1:
            break
        parents = np.random.choice(pop_size, size=pop_size, replace=True)
        new_pop = []
        for i in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            h1, h2 = h_pop[parents[idx1]], h_pop[parents[idx2]]
            child = np.copy(h1)
            if np.random.rand() < 0.7:
                split = np.random.randint(1, n_points)
                child[:split] = h2[:split]
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            child = repair(child)
            new_pop.append(child)
        h_pop = np.array(new_pop)
        for h in h_pop:
            c = evaluate(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    return best_h, best_c5, n_points
