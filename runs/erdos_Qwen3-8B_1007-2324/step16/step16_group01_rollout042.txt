    scale = integral_target_half / sm if sm else 1
    h_half = h_half * scale
    h_half = np.clip(h_half, 0, 1)
    
    def obj(x):
        x = x.copy()
        sm = np.sum(x)
        scale = integral_target_half / sm if sm else 1
        x = x * scale
        x = np.clip(x, 0, 1)
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target_half, 'jac': lambda x: np.ones_like(x)})
    bnds = [(0.0, 1.0) for _ in range(len(h_half))]
    res = scipy.optimize.minimize(obj, h_half, method='SLSQP', bounds=bnds, constraints=cons, options={'maxiter': 200})
    if res.success:
        h_opt = res.x.copy()
        h_opt = np.clip(h_opt, 0, 1)
        sm = np.sum(h_opt)
        scale = integral_target_half / sm if sm else 1
        h_opt = h_opt * scale
        h_opt = np.clip(h_opt, 0, 1)
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        final_c5 = np.max(corr) * dx
        best_h = h_opt
        best_c5 = final_c5
    
    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
