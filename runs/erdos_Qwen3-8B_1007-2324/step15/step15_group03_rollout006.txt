import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution
    import time

    n_points = 60
    dx = 2.0 / n_points
    target_sum = 30.0

    # Initialize with provided or random starting point
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values).astype(float)
    else:
        h_initial = np.random.rand(n_points)

    # Repair function to ensure constraints
    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        s0 = np.sum(h_clipped)
        if s0 < 1e-7:
            return np.full_like(h, target_sum / n_points)
        elif abs(s0 - target_sum) < 1e-7:
            return h_clipped
        else:
            scaling = target_sum / s0
            return h_clipped * scaling

    # Repair initial guess
    h_initial_repaired = repair(h_initial)

    # Objective function to minimize max overlap
    def objective(x):
        x_clean = repair(x)
        h_full = np.concatenate([x_clean, x_clean[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial_repaired
    best_c5 = objective(best_h)
    start_time = time.time()

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 0.9),
            recombination=0.7,
            tol=1e-5,
            maxiter=200,
            disp=False,
            polish=True,
            vectorized=True
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            return best_h, best_c5, n_points
        if result.success:
            current_h = repair(np.array(result.x))
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    except:
        pass

    return best_h, best_c5, n_points
