        initial_full = np.concatenate([initial_full[:30], initial_full[:30][::-1]])
        initial_full = initial_full[:60]
    best_h = initial_full.copy()
    best_c5 = compute_c5(best_h, 60)
    start_time = time.time()
    for _ in range(int(budget_s * 1000)):
        indices_1 = np.where(best_h == 1)[0]
        indices_0 = np.where(best_h == 0)[0]
        if len(indices_1) > 0 and len(indices_0) > 0:
            i = np.random.choice(indices_1)
            j = np.random.choice(indices_0)
            mutant = best_h.copy()
            mutant[i], mutant[j] = mutant[j], mutant[i]
            mutant_c5 = compute_c5(mutant, 60)
            if mutant_c5 < best_c5:
                best_h = mutant
                best_c5 = mutant_c5
        else:
            if np.sum(best_h) == 60:
                best_h[0] = 0
            else:
                best_h[0] = 1
            mutant_c5 = compute_c5(best_h, 60)
            if mutant_c5 < best_c5:
                best_c5 = mutant_c5
        if time.time() - start_time > budget_s - 1:
            break
    return best_h, best_c5, 60

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
