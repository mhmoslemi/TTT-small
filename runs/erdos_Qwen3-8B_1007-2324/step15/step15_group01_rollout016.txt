import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = None

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) > n_points:
            initial_h = initial_h[:n_points]
        initial_h = initial_h[:n_half]
    else:
        initial_h = np.random.rand(n_half)

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 1e-6:
            return h * (target_sum / s)
        else:
            return np.random.rand(len(h)) * (target_sum / len(h))

    def compute_c5(h_vec):
        h = normalize_h(h_vec)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    best_h = initial_h.copy()
    best_c5 = float('inf')

    for _ in range(3):
        if np.random.rand() > 0.5:
            init = initial_h.copy()
            perturb = np.random.normal(0, 0.1, n_half)
            init = np.clip(init + perturb, 0, 1)
            init = normalize_h(init)
        else:
            init = np.random.rand(n_half)
            init = normalize_h(init)

        result = scipy.optimize.differential_evolution(
            compute_c5,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            seed=seed
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    full_h = np.concatenate([best_h, best_h[::-1]])
    return full_h, best_c5, n_points
