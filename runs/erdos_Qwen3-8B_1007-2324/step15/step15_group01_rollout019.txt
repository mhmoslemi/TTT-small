import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    # Generate initial guesses
    h_initial = np.array([])
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_initial = h_initial[::step][:n_half]
    else:
        h_initial = np.clip(1 - np.abs(np.linspace(0, 2, n_half) - 1), 0, 1)

    def generate_binary_start(n_half):
        return np.random.permutation(np.concatenate([np.ones(n_half // 2), np.zeros(n_half // 2)]))

    def generate_random_start(n_half):
        return np.random.rand(n_half) * (integral_target / np.sum(np.random.rand(n_half)))

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def optimize_start(start):
        x0 = np.clip(start, 0.0, 1.0)
        if np.sum(x0) > 1e-6:
            x0 = x0 * (integral_target / np.sum(x0))
        else:
            x0 = generate_random_start(n_half)
        result = scipy.optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)
            },
            options={
                'ftol': 1e-12,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        return result.x if result.success else None

    best_h = h_initial.copy()
    best_c5 = np.inf

    # Try initial guess
    h_opt = optimize_start(h_initial)
    if h_opt is not None:
