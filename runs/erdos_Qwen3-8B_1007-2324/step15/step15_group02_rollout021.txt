import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    integral_target = n_points / 4
    n_half = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half)

    # Enforce integral constraint
    current_sum = np.sum(h_opt)
    if abs(current_sum - integral_target) > 1e-6:
        h_opt *= integral_target / current_sum
    h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    def objective(x):
        h = x
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]

    # First optimization with SLSQP
    result = scipy.optimize.minimize(
        objective,
        x0=h_opt,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    if result.success:

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Second optimization with L-BFGS-B
    result = scipy.optimize.minimize(
        objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
