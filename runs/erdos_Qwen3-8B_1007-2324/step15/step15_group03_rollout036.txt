        corr = np.correlate(repaired, 1.0 - repaired, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = repair(initial_guess)
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx
    
    start_time = time.time()
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-6,
            maxiter=100,
            disp=False,
            polish=True,
            vectorized=True
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            return best_h, best_c5, n_points
        if result.success:
            current_h = repair(result.x)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    except:
        pass
    
    return best_h, best_c5, n_points
