import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize guess
    h_opt = np.zeros(n_points // 2)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        step = int(np.ceil(len(h_initial) / n_points))
        h_opt = h_initial[::step][:n_points // 2]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_points // 2]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    x0 = h_opt.copy()

    best_c5 = np.inf
    best_h = x0.copy()

    # Multi-start optimization
    for _ in range(5):  # More perturbations for diversity
        if np.random.rand() < 0.5:
            # Binary-like start: piecewise constant
            m = int(np.random.randint(2, 10))
            h_perturbed = np.zeros_like(h_opt)
            for i in range(m):
                start = np.random.randint(0, len(h_opt) - 1)
                end = np.random.randint(start + 1, len(h_opt))
                h_perturbed[start:end] = 1.0
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        else:
            # Perturb existing guess
            h_perturbed = x0 + np.random.normal(0, 0.1, size=len(x0))
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            if np.sum(h_perturbed) > 1e-6:
                h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
            else:
                h_perturbed = np.random.rand(len(x0)) * (integral_target / len(x0))

        # Optimization with SLSQP
        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)

            options={
                'ftol': 1e-9,

                'disp': False,

            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Final evaluation
    final_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
