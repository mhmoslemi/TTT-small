import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        if original_len != n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, original_len), h_initial)
        h_opt = h_initial.copy()
    else:
        h_opt = np.random.rand(n_points)
        h_opt /= np.sum(h_opt)
        h_opt *= integral_target
        h_opt = np.clip(h_opt, 0, 1)
    
    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        valid_shifts = corr[n_points - 1: 2 * n_points - 1]
        return np.max(valid_shifts) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })
    
    def exponential_smoothing(x, alpha=0.5):
        return np.convolve(x, [alpha, 1 - alpha], mode='same')
    
    h_opt = exponential_smoothing(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    
    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        valid_shifts = corr[n_points - 1: 2 * n_points - 1]
        return np.max(valid_shifts) * dx
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    
    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)
    
    result = scipy.optimize.minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'disp': False, 'ftol': 1e-9}

        current_c5 = compute_overlap(h_full)
        if current_c5 < best_c5:

            best_h = h
    
    h_values = np.concatenate([best_h, best_h[::-1]])
