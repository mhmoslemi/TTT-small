import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    n_half = n_points // 2
    target_sum = n_points / 4
    dx = 2.0 / n_points

    h_initial = np.zeros(n_half)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_half]
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (target_sum / np.sum(h_initial))

    def adjust_feasibility(x):
        s = np.sum(x)
        if abs(s - target_sum) < 1e-5:
            return x
        scale = target_sum / max(1e-10, s)
        x_scaled = x * scale
        x_clipped = np.clip(x_scaled, 0, 1)
        s_clipped = np.sum(x_clipped)
        if abs(s_clipped - target_sum) < 1e-5:
            return x_clipped
        scale2 = target_sum / s_clipped
        return x_clipped * scale2

    def objective(x):

        h_full = np.concatenate([x_feasible, x_feasible[::-1]])

        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in range(n_half)]
    result = scipy.optimize.differential_evolution(
