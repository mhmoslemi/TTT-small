        'jac': lambda x: np.ones_like(x)
    }]
    
    best_h = h_initial.copy()
    best_c5 = compute_overlap(best_h)
    best_time = 0.0
    start_time = time.time()

    # Try multiple optimizations with different initial guesses
    for _ in range(3):
        # Perturb the initial guess
        perturbation = np.random.normal(0, 0.1, size=n_points)
        h_perturb = h_initial + perturbation
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * integral_target / np.sum(h_perturb)
        h_perturb = np.clip(h_perturb, 0, 1)

        # Optimize using SLSQP
        result = scipy.optimize.minimize(
            compute_overlap,
            h_perturb,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if result.success:
            h_candidate = result.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            # Ensure integral constraint is satisfied
            h_candidate = h_candidate * integral_target / np.sum(h_candidate)
            h_candidate = np.clip(h_candidate, 0, 1)
            current_c5 = compute_overlap(h_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate
            # Check time
            elapsed = time.time() - start_time
            if elapsed > budget_s - 10:
                break

    # Additional optimization with a random initial guess
    h_random = np.random.rand(n_points)
    h_random = h_random * integral_target / np.sum(h_random)
    h_random = np.clip(h_random, 0, 1)
    result = scipy.optimize.minimize(
        compute_overlap,

        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

    )
    if result.success:
        h_candidate = result.x.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        # Ensure integral constraint is satisfied
        h_candidate = h_candidate * integral_target / np.sum(h_candidate)

        current_c5 = compute_overlap(h_candidate)
