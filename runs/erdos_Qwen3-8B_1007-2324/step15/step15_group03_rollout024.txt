        initial_guess = np.where(initial_h > 0.5, 1.0, 0.0)
        current_sum = np.sum(initial_guess)
        if current_sum != 30:
            idx = np.where(initial_guess == 1)[0]
            if current_sum < 30:
                to_add = 30 - current_sum
                add_indices = idx[:to_add] if len(idx) >= to_add else idx
                initial_guess[add_indices] = 1
            else:
                to_remove = current_sum - 30
                remove_indices = idx[:to_remove]
                initial_guess[remove_indices] = 0
    else:
        initial_guess = np.array([0.5] * 30)

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points // 2

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == target_sum:
            return h
        if current_sum < target_sum:
            need = target_sum - current_sum
            indices = np.where(h == 0)[0]
            h[indices[:need]] = 1
        else:
            need = current_sum - target_sum
            indices = np.where(h == 1)[0]
            h[indices[:need]] = 0
        return h

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(len(initial_guess))]

    best_h = np.concatenate([initial_guess, initial_guess[::-1]])
    best_c5 = initial_c5

    start_time = time.time()
    try:

            bounds,

            popsize=15,
