    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    start_time = time.time()
    time_limit = budget_s * 0.9

    # First optimization with SLSQP
    def obj(x):
        h_opt = x
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    result = scipy.optimize.minimize(
        obj,
        x0=h_opt,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    if time.time() - start_time > time_limit:
        final_h = np.concatenate([best_h, best_h[::-1]])
        return final_h, best_c5, n_points

    # Second optimization with L-BFGS-B
    def obj2(x):
        h_opt = x
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = scipy.optimize.minimize(
        obj2,
        x0=best_h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8}
    )

    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
