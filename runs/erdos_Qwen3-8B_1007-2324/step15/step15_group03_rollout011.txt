from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    # Use the provided initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == 60:
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.random.rand(60)
        # Initialize with a flat profile and sparse spikes for diversity
        initial_h[:30] = np.random.uniform(0.5, 1.0, 30)
        initial_h[30:] = np.random.uniform(0.0, 0.5, 30)

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2  # 30.0

    def repair(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clipped)
        if abs(total - target_sum) < 1e-7:
            return h_clipped
        scaling = target_sum / total
        return h_clipped * scaling

    def objective(x):
        x_clean = repair(x)
        h_full = x_clean
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Start with the initial guess
    initial_guess = initial_h.copy()
    initial_corr = np.correlate(initial_guess, 1 - initial_guess, mode='full')
    best_c5 = np.max(initial_corr) * dx
    best_h = initial_guess.copy()

    start_time = time.time()
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,  # Increased population size for better exploration
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-5,
            maxiter=200,
            disp=False,
            polish=True,
            vectorized=True
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            return best_h, best_c5, n_points
        if result.success:
