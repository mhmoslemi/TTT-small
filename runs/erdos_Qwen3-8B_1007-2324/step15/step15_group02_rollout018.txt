import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_half = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        # Reshape to n_points//2 as per previous code
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    result = scipy.optimize.minimize(
        objective,
        x0=h_opt,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-9}
    )

    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = h_opt
        best_c5 = objective(h_opt)

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
