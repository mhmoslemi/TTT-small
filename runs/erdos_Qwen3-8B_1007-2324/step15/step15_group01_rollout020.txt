import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def adjust(x, target):
        s = np.sum(x)
        if np.isclose(s, target, atol=1e-5):
            return x
        scale = target / max(1e-10, s)
        x = x * scale
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if np.isclose(s, target, atol=1e-5):
            return x
        scale = target / max(1e-10, s)
        return x * scale

    def compute_overlap(x):
        h = np.concatenate([x, x[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(adjust(x, integral_target))

    if 'initial_h_values' in locals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_half:
            step = int(np.ceil(len(h_initial) / n_half))
            h_opt = h_initial[::step][:n_half]
        else:
            h_opt = h_initial
        h_opt = adjust(h_opt, integral_target)
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]
        h_opt = adjust(h_opt, integral_target)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    x0 = h_opt.copy()

    best_c5 = np.inf
    best_h = x0.copy()

    options = {'ftol': 1e-12, 'maxiter': 100, 'disp': False, 'eps': 1e-8}

    for _ in range(2):
        if np.random.rand() < 0.5:
            perturb = np.random.normal(0, 0.05, size=n_half)
            h_pert = np.clip(x0 + perturb, 0.0, 1.0)
            h_pert = adjust(h_pert, integral_target)
        else:
            h_pert = np.random.rand(n_half)
            h_pert = adjust(h_pert, integral_target)

        result = scipy.optimize.minimize(
            objective,
            h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target},
            options=options
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    return final_h, final_c5, n_points
