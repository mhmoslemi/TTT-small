import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # Corrected to ensure sum(h) * dx = 1
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:

        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt /= np.sum(h_opt)
        h_opt *= integral_target
        h_opt = np.clip(h_opt, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    
    start_time = time.time()
    
    # Use SLSQP with hard constraints
    result = scipy.optimize.minimize(
        compute_overlap,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    if result.success:
        h = result.x.copy()
        h = np.clip(h, 0, 1)
        h_full = np.concatenate([h, h[::-1]])
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1) > 1e-6:
            scale = 1.0 / (h_sum * dx)
            h = h * scale
            h = np.clip(h, 0, 1)
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    final_h = np.concatenate([best_h, best_h[::-1]])
