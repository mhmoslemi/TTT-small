import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Efficient for time budget
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # 0.5 * dx * n_points = 1

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt /= np.sum(h_opt)
        h_opt *= integral_target
        h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()

    result = scipy.optimize.minimize(
        compute_overlap,
        h_opt,
        method='COBYLA',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'disp': False, 'ftol': 1e-10}
    )

    best_h = result.x.copy()
    best_h = np.clip(best_h, 0, 1)
    final_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    
    # Early exit if time is exhausted
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        return final_h, final_c5, n_points

    # Use spectral initialization
    h_freq = np.fft.fft(best_h)
    h_freq *= np.maximum(1 - 0.5 * np.abs(h_freq), 0)
    h_initial = np.real(np.fft.ifft(h_freq))[:n_half]
    h_initial = np.clip(h_initial, 0, 1)
    h_initial *= integral_target / np.sum(h_initial)

    result = scipy.optimize.minimize(
        compute_overlap,
        h_initial,
        method='COBYLA',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'disp': False, 'ftol': 1e-10}
    )
