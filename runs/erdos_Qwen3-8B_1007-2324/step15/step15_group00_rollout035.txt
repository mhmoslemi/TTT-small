import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [40, 80, 120]
    best_h = None
    best_c5 = 1.0
    total_time = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        integral_target = n_points / 2
        n_half = n_points // 2
        h_opt = np.zeros(n_half)
        
        if 'initial_h_values' in globals():
            h_initial = np.array(initial_h_values)
            original_len = len(h_initial)
            step = int(np.ceil(original_len / n_points))
            h_opt = h_initial[::step][:n_half]
        else:
            h_opt = np.random.rand(n_half)
            h_opt = h_opt * (integral_target / np.sum(h_opt)) if np.sum(h_opt) > 0 else np.zeros_like(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_half)]
        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })
        
        weights = np.random.beta(5, 8, size=20)
        start_time = time.time()
        
        for w in weights:
            result = scipy.optimize.minimize(
                lambda x: w * compute_overlap(x) + (1 - w) * np.var(x),
                h_opt,
                method='COBYLA',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'disp': False}
            )
            
            if result.success:
                h_candidate = result.x.copy()
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (integral_target / np.sum(h_candidate)) if np.sum(h_candidate) > 0 else np.zeros_like(h_candidate)
                current_c5 = compute_overlap(h_candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_candidate

                    break
        
        total_time += time.time() - start_time
