        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        return w * overlap + (1 - w) * np.var(h)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    for _ in range(2):
        w = np.random.beta(2, 32)
        result = scipy.optimize.minimize(
            lambda x: objective(x, w),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.success:
            h = result.x.copy()
            h = np.clip(h, 0, 1)
            h_full = np.concatenate([h, h[::-1]])
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h

        if time.time() - start_time > budget_s * 0.8:
            break

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
