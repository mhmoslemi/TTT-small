    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h_binary = (h > 0.5).astype(int)
    else:
        h_binary = np.random.randint(0, 2, n_points)
    # Repair to meet target_ones
    def repair(h):
        current = np.sum(h)
        if current == target_ones:
            return h
        diff = current - target_ones
        if diff > 0:
            ones = np.where(h == 1)[0]
            for _ in range(diff):
                idx = np.random.choice(ones)
                h[idx] = 0
        else:
            zeros = np.where(h == 0)[0]
            for _ in range(-diff):
                idx = np.random.choice(zeros)
                h[idx] = 1
        return h

    h_binary = repair(h_binary)

    # Define evaluation function
    def evaluate(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Start optimization
    pop_size = 20
    mutation_rate = 0.1
    crossover_rate = 0.8
    max_gen = 200

    population = [h_binary.copy() for _ in range(pop_size)]
    best_h = h_binary.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate and select parents
        fitness = np.array([evaluate(ind) for ind in population])
        parents = np.argsort(fitness)[:pop_size//2]
        # Crossover and mutate
        new_pop = []
        for i in range(0, pop_size, 2):
            a, b = population[parents[i]], population[parents[i+1]]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points)
                child1 = np.concatenate([a[:split], b[split:]])
                child2 = np.concatenate([b[:split], a[split:]])
            else:
                child1, child2 = a, b
            # Mutate
            for child in [child1, child2]:
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] ^= 1
            # Repair
            child1 = repair(child1)
            child2 = repair(child2)
            new_pop.extend([child1, child2])
        population = new_pop
        # Update best
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
