def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = (initial_h > 0.5).astype(int)

        initial_h = np.random.randint(0, 2, n_points)

        diff = target_sum - current_sum
        if diff != 0:
            flip_indices = np.random.choice(n_points, abs(diff), replace=False)
            initial_h[flip_indices] ^= 1

    def repair(h):
        h = h.copy()
        current_sum = np.sum(h)
        target = target_sum
        if current_sum == target:
            return h
        diff = target - current_sum
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if zeros.size > 0:
                flip_indices = np.random.choice(zeros, diff, replace=False)
                h[flip_indices] = 1
        else:
            ones = np.where(h == 1)[0]
            if ones.size > 0:
                flip_indices = np.random.choice(ones, -diff, replace=False)
                h[flip_indices] = 0
        return h

    def objective(x):
        h = np.round(x).astype(int)
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guess = initial_h.astype(float)
    initial_guess = repair(initial_guess)

    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_guess.copy()

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.9,

            maxiter=200,
            disp=False,
            polish=True,
            vectorized=True
        )
        if result.success:
            current_h = np.round(result.x).astype(int)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h

    except:
        pass

    elapsed = time.time() - start_time
