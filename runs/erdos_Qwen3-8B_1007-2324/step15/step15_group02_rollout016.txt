        'jac': lambda x: np.ones_like(x)
    })
    
    # Tuned L-BFGS-B optimization
    result = scipy.optimize.minimize(
        obj,
        x0=h_opt,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    if result.success:
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
