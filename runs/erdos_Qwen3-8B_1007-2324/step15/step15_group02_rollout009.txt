import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    dx = 2.0 / n_points
    integral_target = n_points / 2
    h_opt = None

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_initial = len(h_initial)
        step = int(np.ceil(n_initial / n_points))
        h_opt = h_initial[::step][:n_points//2]
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points//2).clip(0, 1)
    else:
        h_opt = np.random.uniform(0, 1, n_points//2).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points//2).clip(0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return np.max(np.correlate(np.concatenate([x, x[::-1]]), 1 - np.concatenate([x, x[::-1]]), mode='full')) * dx

    constraints = {
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }

    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]

    time_limit = budget_s * 0.9
    start_time = time.time()
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    for _ in range(5):
        result = scipy.optimize.minimize(
            objective,
            x0=h_opt,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > time_limit:
            break

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
