import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        x = np.linspace(0, 2, n_points)
        h_base = np.clip(1 - np.abs(x - 1), 0, 1)

    # Generate variations
    variations = [h_base]
    # Perturb
    perturb = np.random.normal(0, 0.05, len(h_base))
    h_perturbed = np.clip(h_base + perturb, 0, 1)
    h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
    variations.append(h_perturbed)
    # Reverse
    h_reversed = np.clip(h_base[::-1], 0, 1)
    h_reversed = h_reversed * (integral_target / np.sum(h_reversed))
    variations.append(h_reversed)
    # Random
    h_random = np.random.rand(len(h_base)) * (integral_target / np.sum(h_random))
    variations.append(h_random)

    best_c5 = np.inf
    best_h = h_base.copy()
    start_time = time.time()

    for h_start in variations:
        h_start = np.clip(h_start, 0, 1)
        if np.sum(h_start) < 1e-6:
            h_start = np.random.rand(len(h_start)) * (integral_target / np.sum(h_start))
        else:
            h_start = h_start * (integral_target / np.sum(h_start))

        def objective(x):
            h_full = np.concatenate([x, x[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}
        result = scipy.optimize.minimize(
            objective, h_start, method='SLSQP', bounds=[(0,1)]*len(h_start),
            constraints=constraints, options={'ftol': 1e-10, 'maxiter': 300}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        if time.time() - start_time > budget_s:
            break

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
