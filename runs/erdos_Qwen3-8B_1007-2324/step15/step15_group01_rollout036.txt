    bounds = [(0.0, 1.0) for _ in range(n_half)]
    x0 = initial_h
    
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    best_x = result.x
    best_h_full = np.concatenate([best_x, best_x[::-1]])
    best_c5 = result.fun
    
    return best_h_full, best_c5, n_points
