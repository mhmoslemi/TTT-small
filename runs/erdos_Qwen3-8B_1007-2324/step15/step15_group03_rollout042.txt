import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 60
        positions = np.arange(n_points)
        values = np.cos(2 * np.pi * positions / n_points)
        h = np.where(values > 0, 1.0, 0.0)
        if np.sum(h) != n_points // 2:
            h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            np.random.shuffle(h)
    n_points = 60
    dx = 2.0 / n_points

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    max_iter = 100000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        indices_ones = np.where(h == 1)[0]
        indices_zeros = np.where(h == 0)[0]
        if not indices_ones.size or not indices_zeros.size:
            break
        i = np.random.choice(indices_ones)
        j = np.random.choice(indices_zeros)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
