        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        variance = np.var(h)
        return overlap + 1e-3 * variance

    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]

    result = scipy.optimize.minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    if result.success:
        best_h = result.x
    else:
        best_h = h_opt.copy()

    # Final verification and normalization
    h_full = np.concatenate([best_h, best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx

    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1) > 1e-6:
        best_h = best_h * (integral_target / np.sum(best_h))
        best_h = np.clip(best_h, 0, 1)

    return np.concatenate([best_h, best_h[::-1]]), c5_bound, n_points
