    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points//2]
    else:
        h_initial = np.random.uniform(0, 1, n_points//2).clip(0, 1)
        while np.abs(np.sum(h_initial) - integral_target) > 1e-6:
            h_initial = np.random.uniform(0, 1, n_points//2).clip(0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_initial)
    best_h = h_initial.copy()

    def obj(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        return -overlap  # Minimize overlap

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    time_limit = budget_s * 0.8
    start_time = time.time()

    for _ in range(10):
        result = scipy.optimize.minimize(
            obj,
            x0=best_h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(len(best_h))],
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > time_limit:
            break

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
