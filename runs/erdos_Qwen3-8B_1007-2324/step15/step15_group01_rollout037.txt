import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    # Generate initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    x0 = h_opt.copy()

    best_c5 = np.inf
    best_h_opt = x0.copy()

    # Run multiple optimizations
    for _ in range(5):  # More iterations
        # Perturb initial guess
        perturb = np.random.normal(0, 0.05, size=n_half)
        h_perturbed = np.clip(x0 + perturb, 0.0, 1.0)
        # Enforce integral constraint
        if np.sum(h_perturbed) > 1e-6:
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        else:
            h_perturbed = np.random.rand(n_half)
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))

        # Optimization
        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)
            },
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_opt = result.x

    # Final evaluation
    final_h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    final_c5 = compute_overlap(best_h_opt)
    return final_h_full, final_c5, n_points

def run(seed=42, budget_s=100, **kwargs):

    # Generate initial guess
    if 'initial_h_values' in globals():

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    best_c5 = np.inf
