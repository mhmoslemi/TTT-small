import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points][:n_half]
    else:

        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    x0 = h_opt.copy()

    best_c5 = np.inf
    best_h = x0.copy()

    for _ in range(5):
        perturb = np.random.normal(0, 0.1, size=n_half)
        h_perturbed = np.clip(x0 + perturb, 0.0, 1.0)
        if np.sum(h_perturbed) > 1e-6:
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        else:
            h_perturbed = np.random.rand(n_half)
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))

        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like},
            options={'ftol': 1e-12, 'maxiter': 50, 'disp': False, 'eps': 1e-8}
        )
        if result.success:
            curr_c5 = result.fun
            if curr_c5 < best_c5:
