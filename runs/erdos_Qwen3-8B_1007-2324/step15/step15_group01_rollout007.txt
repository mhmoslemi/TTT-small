import numpy as np
import time
from sklearn.cmaes import CMAEvolutionStrategy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def adjust_for_feasibility(x, target):
        integral = np.sum(x)
        if abs(integral - target) < 1e-5:
            return x
        scale = target / max(1e-10, integral)
        x_scl = x * scale
        x_clipped = np.clip(x_scl, 0, 1)
        integral_clipped = np.sum(x_clipped)
        if abs(integral_clipped - integral) < 0.05 * target:
            return x_clipped
        return x_clipped * (target / min(1e-10, integral_clipped))

    def generate_initial(n_half, target, parent=None):
        if parent is not None and len(parent) == n_half:
            candidates = [np.clip(parent, 0, 1)]
            for _ in range(4):
                perturb = np.random.normal(0, 0.01, n_half)
                cand = parent + perturb
                cand = adjust_for_feasibility(cand, target)
                candidates.append(cand)
        else:
            base = np.random.dirichlet(np.ones(n_half)*10, size=5)
            candidates = [adjust_for_feasibility(c, target) for c in base]
        return candidates

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x_vals = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x_vals - 1), 0, 1)[:n_half]

    initial_pop = generate_initial(n_half, integral_target, h_opt)
    initial_pop = np.array(initial_pop)
    initial_flat = initial_pop.flatten()
    sigma = 0.5
    es = CMAEvolutionStrategy(initial_flat, sigma, {'bounds': [(0.0, 1.0)] * n_half})

    best_x = initial_pop[0]
    best_c5 = objective(best_x)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        population = es.ask()
        for i in range(len(population)):
            x = population[i]
            x = adjust_for_feasibility(x, integral_target)
            population[i] = x
        obj_vals = [objective(x) for x in population]
        best_idx = np.argmin(obj_vals)
        best_x = population[best_idx]
        best_c5 = obj_vals[best_idx]
        es.tell(population, obj_vals)
        es.rewind()

    h_values = np.concatenate([best_x, best_x[::-1]])
    return h_values, best_c5, n_points
