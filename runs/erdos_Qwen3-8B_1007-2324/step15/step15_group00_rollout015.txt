import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt /= np.sum(h_opt)
        h_opt *= integral_target
        h_opt = np.clip(h_opt, 0, 1)
    
    def smooth_h(h, window=3):
        h_smooth = np.zeros_like(h)
        for i in range(len(h)):
            start = max(0, i - window // 2)
            end = min(len(h), i + window // 2 + 1)
            h_smooth[i] = np.mean(h[start:end])
        return h_smooth
    
    h_opt_smoothed = smooth_h(h_opt)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = scipy.optimize.LinearConstraint(np.ones(n_half), [integral_target], [integral_target])
    
    best_h = h_opt.copy()
