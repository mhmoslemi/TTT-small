import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 60:
        h_initial = initial_h_values.astype(float)
    else:
        h_initial = np.random.rand(60)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = (30.0 / np.sum(h_initial)) * h_initial

    n_points = 60
    dx = 2.0 / n_points
    target_sum = 30.0

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            return np.linspace(0, 1, n_points)
        return (target_sum / s) * h_clipped

    def objective(x):
        try:
            x_clean = repair(x)
            corr = np.correlate(x_clean, 1 - x_clean, mode='full')
            return np.max(corr) * dx
        except:
            return float('inf')

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 0.9),
        recombination=0.9,
        tol=1e-5,
        maxiter=300,
        disp=False,
        polish=True,
        vectorized=True
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h_initial, objective(h_initial), n_points

    best_h = result.x
    best_c5 = result.fun
    return repair(best_h), best_c5, n_points
