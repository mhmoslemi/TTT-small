import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_half = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half].copy()
        current_sum = np.sum(h_opt)
        scaling_factor = integral_target / current_sum
        h_opt = h_opt * scaling_factor
    else:
        h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = scipy.optimize.minimize(
        objective,
        x0=h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-10}
    )

    best_h = result.x
    best_c5 = compute_overlap(best_h)

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
