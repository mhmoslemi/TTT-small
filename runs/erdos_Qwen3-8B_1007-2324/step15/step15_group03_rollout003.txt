import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    else:
        n_points = 60
        h_initial = np.random.rand(n_points)

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        s0 = np.sum(h_clipped)
        if abs(s0 - target_sum) < 1e-7:
            return h_clipped
        scaling = target_sum / s0
        return h_clipped * scaling

    def objective(x):
        h_opt = repair(x)
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=300,
        polish=True,
        vectorized=True
    )
    elapsed = time.time() - start_time

    if elapsed > budget_s - 1:
        best_h = repair(h_initial)
        best_c5 = objective(h_initial)
    else:
        if result.success:
            best_h = repair(result.x)
            best_c5 = result.fun
        else:
            best_h = repair(h_initial)
            best_c5 = objective(h_initial)

    return best_h, best_c5, n_points
