        h_opt = np.array(initial_h_values)
        if len(h_opt) != n_points:
            h_opt = h_opt[:n_points]
    else:
        h_opt = np.random.uniform(0, 1, n_points)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': lambda x: np.ones_like(x)})

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    start_time = time.time()
    result = scipy.optimize.minimize(
        compute_overlap,
        x0=h_opt,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-10}
    )

    if result.success:
        best_c5 = compute_overlap(result.x)
        best_h = result.x

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        result = scipy.optimize.minimize(
            compute_overlap,
            x0=best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            best_c5 = compute_overlap(result.x)
            best_h = result.x

    return best_h, best_c5, n_points
