import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    if 'initial_h_values' in globals():

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = np.inf
    best_h = h_opt.copy()

        h_perturbed = np.clip(h_opt + perturb, 0.0, 1.0)
        h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed)) if np.sum(h_perturbed) > 1e-6 else np.random.rand(n_points) * (integral_target / n_points)

            method='SLSQP',

            constraints=[

        )

        if result.success:
