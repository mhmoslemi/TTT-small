import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    n_half = n_points // 2
    
    h_opt = np.empty(n_half)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_half).clip(0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })
    
    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
    )
    
    if result.success:
        best_h = result.x
    else:
        best_h = h_opt
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, compute_overlap(best_h), n_points
