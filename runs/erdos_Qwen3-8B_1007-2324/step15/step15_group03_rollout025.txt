import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    n_points = 60
    dx = 2.0 / n_points
    target_sum = 30

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        binary_h = (h_initial >= 0.5).astype(int)
        current_sum = np.sum(binary_h)
        if current_sum != target_sum:
            ones = np.where(binary_h == 1)[0]
            zeros = np.where(binary_h == 0)[0]
            if current_sum > target_sum:
                diff = current_sum - target_sum
                idxs = np.random.choice(ones, size=diff, replace=False)
                binary_h[idxs] = 0
            else:
                diff = target_sum - current_sum
                idxs = np.random.choice(zeros, size=diff, replace=False)
                binary_h[idxs] = 1
        current_h = binary_h
    else:
        current_h = np.random.choice([0, 1], size=n_points)
        current_sum = np.sum(current_h)
        if current_sum != target_sum:
            if current_sum < target_sum:
                diff = target_sum - current_sum
                current_h[np.random.choice(np.where(current_h == 0)[0], size=diff)] = 1
            else:
                diff = current_sum - target_sum
                current_h[np.random.choice(np.where(current_h == 1)[0], size=diff)] = 0
        current_h = np.array(current_h, dtype=int)

    best_h = current_h.copy()
    best_c5 = compute_c5(current_h)
    T = 1.0
    cooling_rate = 0.995
    iterations = 0
