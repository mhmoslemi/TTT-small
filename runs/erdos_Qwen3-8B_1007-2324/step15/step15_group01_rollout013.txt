import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2  # Corrected to ensure ∫h(x)dx = 1

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        # Resample to match n_points//2 length using linear interpolation
        h_opt = np.interp(np.linspace(0, original_len-1, n_points//2), np.arange(original_len), h_initial)
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_points//2]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    x0 = h_opt.copy()

    best_c5 = np.inf
    best_h_opt = x0.copy()
    elapsed = 0

    start_time = time.time()
    for _ in range(3):  # Multiple restarts for diversity
        perturb = np.random.normal(0, 0.1, size=n_points//2)
        h_perturbed = np.clip(x0 + perturb, 0.0, 1.0)
        integral_current = np.sum(h_perturbed)
        if abs(integral_current - integral_target) > 1e-5:
            scale = integral_target / max(1e-10, integral_current)
            h_perturbed = h_perturbed * scale
            h_perturbed = np.clip(h_perturbed, 0, 1)
        # Final check
        if abs(np.sum(h_perturbed) - integral_target) > 1e-5:
            h_perturbed = np.random.rand(n_points//2)
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)
            },
            options={'ftol': 1e-10, 'maxiter': 150, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_opt = result.x

    # Final evaluation
    final_h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    final_c5 = compute_overlap(best_h_opt)
    return final_h_full, final_c5, n_points
