            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
            h_opt = result.x.copy()
        if time.time() - start_time > time_limit:
            break

    h_full = np.concatenate([best_h, best_h[::-1]])
    integral = np.sum(h_full) * dx
    if abs(integral - 1.0) > 1e-6:
        scaling = 1.0 / integral
        h_full = h_full * scaling
        h_full = np.clip(h_full, 0, 1)
    return h_full, best_c5, n_points
