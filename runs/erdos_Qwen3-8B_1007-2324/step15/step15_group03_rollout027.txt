    # Determine initial h_values
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_initial = np.where(initial_h > 0.5, 1, 0).astype(int)
        count_ones = np.sum(h_initial)
        if count_ones < 30:
            zeros = np.where(h_initial == 0)[0]
            add_indices = np.random.choice(zeros, size=30 - count_ones, replace=False)
            h_initial[add_indices] = 1
        elif count_ones > 30:
            ones = np.where(h_initial == 1)[0]
            remove_indices = np.random.choice(ones, size=count_ones - 30, replace=False)
            h_initial[remove_indices] = 0
        h_initial = h_initial.astype(bool).astype(int)  # Ensure 0/1
    else:
        h_initial = np.zeros(60, dtype=int)
        h_initial[:30] = 1
        np.random.shuffle(h_initial)
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    temperature = 1.0
    cooling_rate = 0.99
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Generate a new candidate via mutation
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue  # no swap possible
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        new_h = best_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c5 = compute_c5(new_h)
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / temperature)
            if np.random.random() < prob:
                best_h = new_h
                best_c5 = new_c5
        
        temperature *= cooling_rate
    
    return best_h, best_c5, 60

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / 60
    return max_corr * dx
