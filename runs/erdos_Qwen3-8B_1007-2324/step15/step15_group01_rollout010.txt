import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        interp_x = np.linspace(0, 2, len(initial_h_values))
        h_initial = np.interp(np.linspace(0, 2, n_half), interp_x, initial_h_values)
    else:

    h_opt = h_initial[:n_half]

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def adjust_for_feasibility(x, target):

        if abs(integral - target) < 1e-6:
