        result = minimize(objective, initial_positions, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_positions = result.x

    if best_positions is None:
        best_positions = np.sort(np.random.rand(10)) * 2.0
    h_solution = np.zeros(n_points)
    for i in range(len(best_positions)):
        t_i = best_positions[i]
        left = t_i - 0.5
        right = t_i + 0.5
        left_idx = int(np.round(left / dx))
        right_idx = int(np.round(right / dx))
        left_idx = np.clip(left_idx, 0, n_points - 1)
        right_idx = np.clip(right_idx, 0, n_points - 1)
        h_solution[left_idx:right_idx] += 1.0 / len(best_positions)
    h_solution = np.clip(h_solution, 0, 1)
    integral = np.sum(h_solution) * dx
    h_solution = h_solution * (1 / integral) if abs(integral - 1) > 1e-6 else h_solution
    final_h = np.concatenate([h_solution, h_solution[::-1]])
    return final_h, best_c5, n_points
