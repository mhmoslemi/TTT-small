import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        h_opt = np.pad(h_initial[:n_points], (0, n_points - original_len), 'constant') if original_len < n_points else h_initial[:n_points]
    else:
        h_opt = np.random.uniform(0, 1, n_points).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points).clip(0, 1)

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_overlap(np.concatenate([h_opt, h_opt[::-1]]))

    time_limit = budget_s * 0.8

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        return compute_overlap(h_full)

        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run SLSQP
    result = scipy.optimize.minimize(
