import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = n_points/2 to satisfy ∫h(x)dx=1

    # Initialize h_opt with initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_opt = h_initial.copy()
    else:
        h_opt = np.random.uniform(0, 1, n_points)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    # Optimize with multiple weightings
    num_weights = 15
    weights = np.random.beta(2, 32, num_weights).clip(0.01, 0.99)

    time_limit = budget_s * 0.8
    start_time = time.time()

    for w in weights:
        def obj(x):
            h = x.copy()
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            overlap = np.max(corr) * dx
            sq = np.sum(h ** 2)
            return w * overlap + (1 - w) * (-sq)

        cons = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })

        result = scipy.optimize.minimize(
            obj,
            x0=h_opt,
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-10}
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        if time.time() - start_time > time_limit:
            break

    final_h = best_h
    return final_h, best_c5, n_points
