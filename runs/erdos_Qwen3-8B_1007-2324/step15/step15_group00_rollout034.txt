            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            overlap = np.max(corr) * dx
            variance = np.var(h)
            return overlap + 10 * variance
        
        bounds = [(0.0, 1.0) for _ in range(N)]
        def constraint_sum(x):
            return np.sum(x) - integral_target
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = scipy.optimize.minimize(
            objective,
            x0,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'disp': False, 'ftol': 1e-10}
        )
        
        if result.success:
            h_current = result.x.copy()
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (integral_target / np.sum(h_current))
            h_current = np.clip(h_current, 0, 1)
            h_full = np.concatenate([h_current, h_current[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, len(final_h)
