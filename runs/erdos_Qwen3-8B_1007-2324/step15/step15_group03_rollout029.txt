            ones = np.where(initial_h == 1)[0]
            zeros = np.where(initial_h == 0)[0]
            if len(ones) > 30:
                idx1, idx0 = ones[0], zeros[0]
                initial_h[idx1], initial_h[idx0] = 0, 1
            elif len(ones) < 30:
                idx1, idx0 = zeros[0], ones[0]
                initial_h[idx1], initial_h[idx0] = 1, 0
    else:
        np.random.seed(seed)
        initial_h = np.zeros(60, dtype=int)
        initial_h[:30] = 1

    current_h = initial_h.copy()
    best_h = current_h.copy()
    current_c5 = compute_c5(current_h)
    best_c5 = current_c5
    temperature = 1.0
    start_time = time.time()

    cooling_rate = 0.99
    while True:
        if time.time() - start_time > budget_s:
            break
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if not ones.size or not zeros.size:
            break
        idx1 = np.random.choice(ones)
        idx0 = np.random.choice(zeros)
        new_h = current_h.copy()
        new_h[idx1], new_h[idx0] = 0, 1
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        temperature *= cooling_rate

    return best_h, best_c5, 60
