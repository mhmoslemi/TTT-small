        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / (n_points // 2)))
        h_opt = h_initial[::step][:n_points // 2]
    else:
        h_opt = np.random.uniform(0, 1, n_points // 2).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points // 2).clip(0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    num_weights = 50
    weights = np.random.beta(2, 32, num_weights).clip(0.01, 0.99)

    time_limit = budget_s * 0.8
    start_time = time.time()

    for w in weights:
        def obj(x):
            h = x.copy()
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            overlap = np.max(corr) * dx
            sq = np.sum(h ** 2)
            return w * overlap + (1 - w) * (-sq)
        cons = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })
        result = scipy.optimize.minimize(
            obj,
            x0=h_opt,
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(len(h_opt))],
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-10}
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > time_limit:
            break

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
