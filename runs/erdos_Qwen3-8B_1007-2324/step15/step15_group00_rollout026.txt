import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        h_opt = h_initial.copy()
    else:
        h_opt = np.random.rand(n_points)
        h_opt = h_opt / np.sum(h_opt) * integral_target
        h_opt = np.clip(h_opt, 0.0, 1.0)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')

        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        overlap = compute_overlap(h)
        return overlap + 0.1 * np.var(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = scipy.optimize.minimize(

        h_opt,

        bounds=bounds,

        options={'maxiter': 100, 'ftol': 1e-8}
    )

    if result.success:
        h = result.x.copy()

        c5 = compute_overlap(h)
    else:
        h = h_opt.copy()
        c5 = compute_overlap(h)
