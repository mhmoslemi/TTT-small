import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def create_binary_initial():
        ones = np.ones(n_points // 2)
        zeros = np.zeros(n_points // 2)
        h = np.concatenate([ones, zeros])
        np.random.shuffle(h)
        return h

    def create_random_initial():
        h = np.random.rand(n_points)
        h = h * (integral_target / np.sum(h))
        return h

    initial_guesses = []
    if 'initial_h_values' in globals():
        initial_guesses.append(np.array(initial_h_values))
    initial_guesses.append(create_binary_initial())
    initial_guesses.append(create_random_initial())

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for init in initial_guesses:
        init = np.clip(init, 0, 1)
        init = init * (integral_target / np.sum(init)) if np.sum(init) > 1e-8 else np.random.rand(n_points) * (integral_target / np.sum(np.random.rand(n_points)))
        result = scipy.optimize.minimize(
            compute_overlap,
            init,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)
            },
            options={'ftol': 1e-12, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x

    h_full = best_h
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx
    return h_full, final_c5, n_points
