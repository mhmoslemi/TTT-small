    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]

    best_c5 = float('inf')
    best_h = h_opt.copy()
    weights = np.linspace(0.5, 2.0, 5)

    for w in weights:
        result = scipy.optimize.minimize(
            lambda x: objective(x, w),
            h_opt,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            current_c5 = compute_c5(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
