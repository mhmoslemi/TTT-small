import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Increased resolution for better approximation
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt /= np.sum(h_opt)
        h_opt *= integral_target
        h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    start_time = time.time()

    # Use SLSQP for better constraint handling
    def objective(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        return overlap

    for _ in range(5):
        result = scipy.optimize.minimize(
            objective,
            h_opt,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-6}
        )
        if result.success:
            h = result.x.copy()
            h = np.clip(h, 0, 1)
            h_full = np.concatenate([h, h[::-1]])
            h_sum = np.sum(h) * dx
            if abs(h_sum - 1) > 1e-6:
                h = h * (integral_target / np.sum(h))
                h = np.clip(h, 0, 1)
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        h_opt = best_h  # Use best solution as next initial guess

        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

    final_h = np.concatenate([best_h, best_h[::-1]])
