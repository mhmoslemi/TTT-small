import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_points]
    else:
        h_opt = np.random.uniform(0, 1, n_points).clip(0, 1)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points).clip(0, 1)

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        overlap = compute_overlap(h)
        penalty = np.sum(h**2) * 1e-3
        return overlap + penalty

    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target, 'jac': lambda h: np.ones_like(h)})

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.minimize(
        objective,
        x0=h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-9}
    )

    best_h = result.x if result.success else h_opt
    best_c5 = compute_overlap(best_h)

    return best_h, best_c5, n_points
