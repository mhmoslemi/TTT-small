import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_initial = len(h_initial)
        if n_initial > n_points:
            h_opt = h_initial[:n_points//2]
        else:
            h_opt = h_initial
            step = int(np.ceil(n_initial / (n_points//2)))
            h_opt = h_initial[::step][:n_points//2]
    else:
        h_opt = np.random.uniform(0, 1, n_points//2)
        while np.abs(np.sum(h_opt) - integral_target) > 1e-6:
            h_opt = np.random.uniform(0, 1, n_points//2)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return compute_overlap(x)

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]

    result = scipy.optimize.minimize(
        obj,
        x0=h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-10}
    )

    if result.success:
        best_c5 = compute_overlap(result.x)
        best_h = result.x
    else:
        best_c5 = compute_overlap(h_opt)
        best_h = h_opt

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
