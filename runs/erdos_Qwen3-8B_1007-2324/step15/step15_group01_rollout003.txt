import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    n_half = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        original_len = len(initial_h_values)
        original_x = np.linspace(0, 2, original_len)
        new_x = np.linspace(0, 2, n_points)
        h_initial = np.interp(new_x, original_x, initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h_opt[:n_half]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    best_c5 = np.inf
    best_h_opt = h_opt.copy()

    start_time = time.time()
    for _ in range(5):  # More trials for better exploration
        # Generate asymmetric perturbation
        perturb = np.random.normal(0, 0.1, size=n_half)
        h_perturbed = np.clip(h_opt + perturb, 0.0, 1.0)
        if np.sum(h_perturbed) > 1e-6:
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        else:
            h_perturbed = np.random.rand(n_half)
            h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))

        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)
            },
            options={
                'ftol': 1e-10,
                'maxiter': 150,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h_opt = result.x

        if time.time() - start_time > budget_s - 5:
            break

    final_h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    final_c5 = compute_overlap(best_h_opt)
    return final_h_full, final_c5, n_points
