import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        x_initial = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_new = np.interp(x_new, x_initial, initial_h_values)
        h_opt = h_new[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = 1 - np.abs(x - 1)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt[:n_half]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    x0 = h_opt.copy()

    best_c5 = np.inf
    best_h_opt = x0.copy()
    elapsed = 0

    for _ in range(5):
        if elapsed >= budget_s:
            break
        perturb = np.random.normal(0, 0.1, size=n_half)
        h_perturbed = np.clip(x0 + perturb, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        scale_factor = integral_target / current_sum if current_sum > 1e-6 else 1.0
        h_perturbed = h_perturbed * scale_factor

        start = time.time()
        result = scipy.optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={
                'type': 'eq',
                'fun': lambda x: np.sum(x) - integral_target,
                'jac': lambda x: np.ones_like(x)
            },
            options={
                'ftol': 1e-12,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        elapsed += time.time() - start
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_opt = result.x

    final_h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    final_c5 = compute_overlap(best_h_opt)
    return final_h_full, final_c5, n_points
