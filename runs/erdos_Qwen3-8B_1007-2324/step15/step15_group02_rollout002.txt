import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_initial = h_initial[::step][:n_points]
    else:
        h_initial = np.random.uniform(0, 1, n_points).clip(0, 1)
        while np.abs(np.sum(h_initial) - integral_target) > 1e-6:
            h_initial = np.random.uniform(0, 1, n_points).clip(0, 1)

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        overlap = compute_overlap(x)
        return overlap

    def constraint(x):
        return np.sum(x) - integral_target

    cons = ({'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = compute_overlap(best_h)
    remaining_time = budget_s * 0.75

    for _ in range(30):
        result = scipy.optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        h_initial = best_h
        if time.time() - start_time > budget_s - remaining_time:
            break

    return best_h, best_c5, n_points
