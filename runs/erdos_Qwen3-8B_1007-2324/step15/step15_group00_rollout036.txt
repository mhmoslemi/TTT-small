        h_initial = np.array(initial_h_values)
        step = int(np.ceil(len(h_initial) / n_points))
        h_initial = h_initial[::step]
        h_initial = h_initial[:n_points]
        h_initial = h_initial * (integral_target / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0, 1)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (integral_target / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint(h):
        return np.sum(h) - integral_target

    cons = [{'type': 'eq', 'fun': constraint}]

    result = scipy.optimize.minimize(
        compute_overlap,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-8}
    )

    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    return h_opt, compute_overlap(h_opt), n_points
