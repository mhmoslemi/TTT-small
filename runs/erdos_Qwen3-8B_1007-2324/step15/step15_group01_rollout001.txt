    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    best_c5 = np.inf
    best_h_opt = h_opt.copy()

    def adjust_for_feasibility(x, target):

        if abs(integral_current - target) < 1e-5:
            return x
        scale = target / max(1e-10, integral_current)
        x_scl = x * scale

        if abs(integral_clipped - integral_current) < 0.05 * target:

            x_final = x_clipped * (target / min(1e-10, integral_clipped))

    def generate_initial_guess():

        return candidate

    start_time = time.time()
    for _ in range(5):  # Try multiple initializations
        h_perturbed = generate_initial_guess()
        result = scipy.optimize.minimize(
