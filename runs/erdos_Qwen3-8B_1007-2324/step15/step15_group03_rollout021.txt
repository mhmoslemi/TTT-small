from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values).astype(float)
    else:
        h_initial = np.random.rand(60)
        h_initial = repair(h_initial, 60, 30)
