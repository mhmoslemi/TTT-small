import scipy.optimize
import time

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / 60)
    
    def objective(x):
        h = x.copy()
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / 60)
    
    bounds = [(0.0, 1.0) for _ in range(60)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 30,
        'jac': lambda x: np.ones_like(x)
    })
    
    best_h = h_initial.copy()
    best_c5 = compute_overlap(best_h)
    
    start_time = time.time()

            method='SLSQP',
