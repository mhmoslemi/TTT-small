                else:
                    indices = np.where(h == 0)[0]
                    flip_indices = indices[:abs(diff)]
                    h[flip_indices] = 1
            h = h.astype(float)
    else:
        h = np.zeros(60, dtype=int)
        for m in range(60):
            angle = 2 * np.pi * 3 * m**2 / 60
            if np.cos(angle) > 0:
                h[m] = 1
        count = np.sum(h)
        if count != 30:
            diff = count - 30
            if diff > 0:
                indices = np.where(h == 1)[0]
                flip_indices = indices[:diff]
                h[flip_indices] = 0
            else:
                indices = np.where(h == 0)[0]
                flip_indices = indices[:abs(diff)]
                h[flip_indices] = 1
        h = h.astype(float)
    n_points = 60
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    max_steps = int(budget_s * 1000)
    steps = 0
    while steps < max_steps:
        i, j = np.random.choice(n_points, 2, replace=False)
        if h[i] != h[j]:
            new_h = h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_c5 = evaluate(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        steps += 1
        if time.time() - start_time > budget_s - 1:
            break
    return best_h, best_c5, n_points
