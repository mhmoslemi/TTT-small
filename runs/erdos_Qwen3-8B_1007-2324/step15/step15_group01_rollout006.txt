import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1) * 2, 0, 1)[:n_half]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def adjust_for_feasibility(x, target):
        s = np.sum(x)
        if np.isclose(s, target, atol=1e-6):
            return x
        scale = target / max(1e-10, s)
        x_scaled = np.clip(x * scale, 0, 1)
        s_clipped = np.sum(x_scaled)
        if np.isclose(s_clipped, target, atol=1e-6):
            return x_scaled
        factor = target / max(1e-10, s_clipped)
        return np.clip(x_scaled * factor, 0, 1)

    best_c5 = np.inf
    best_h = h_opt.copy()

    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        perturb = np.random.normal(0, 0.1, n_half)
        h_perturbed = np.clip(h_opt + perturb, 0, 1)
        h_perturbed = adjust_for_feasibility(h_perturbed, integral_target)
        res = scipy.optimize.minimize(
            compute_overlap, h_perturbed,
            method='SLSQP',
            bounds=[(0., 1.)]*n_half,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_half)},
            options={'ftol': 1e-9, 'maxiter': 100, 'disp': False}
        )
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x

    final_h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    return final_h_full, final_c5, n_points
