            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                child[idx] ^= 1
            child = repair(child)
            new_population.append(child)
        population = new_population
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        generations += 1

    for _ in range(5):
        idx = np.random.randint(n_points)
        new_h = best_h.copy()
        new_h[idx] ^= 1
        new_h = repair(new_h)
        current_c5 = evaluate(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h

    return best_h, best_c5, n_points
