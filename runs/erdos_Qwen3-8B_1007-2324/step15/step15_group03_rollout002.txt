import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    import numpy as np
    import time

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        s0 = np.sum(h_clipped)
        if s0 == 0 or abs(s0 - target_sum) < 1e-7:
            h_clean = np.arange(n_points) / (n_points - 1)
            return np.clip(h_clean, 0, 1)
        return (target_sum / s0) * h_clipped

    def objective(x):
        try:
            if len(x) != n_points:
                return float('inf')
            x_clean = repair(x)
            corr = np.correlate(x_clean, 1 - x_clean, mode='full')
            c5_val = np.max(corr) * dx
            return c5_val
        except:
            return float('inf')

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = repair(h_initial)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h_initial
    best_c5 = objective(h_initial)

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.8, 1.2),
            recombination=0.9,
            tol=1e-6,
            maxiter=250,
            disp=False,
            polish=True,
            vectorized=True
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            return best_h, best_c5, len(best_h)
        if result.success:
            current_h = repair(np.array(result.x))
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    except:
        pass

    return best_h, best_c5, len(best_h)
