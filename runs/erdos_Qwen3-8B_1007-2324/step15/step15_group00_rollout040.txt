            h_opt,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'disp': False, 'ftol': 1e-10}
        )
        if result.success:
            h = result.x.copy()
            h = np.clip(h, 0, 1)
            h_full = np.zeros(2 * n_points)
            h_full[:n_half] = h
            h_true = h_full[:n_half]
            h_sum = np.sum(h) * dx
            if abs(h_sum - 1) > 1e-6:
                h = h * (integral_target / np.sum(h))
                h = np.clip(h, 0, 1)
            current_c5 = compute_overlap(h[:n_half])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h[:n_half]
    
    for _ in range(3):
        optimize()
        time_remaining -= 5
        if time_remaining <= 0:
            break
    
    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
