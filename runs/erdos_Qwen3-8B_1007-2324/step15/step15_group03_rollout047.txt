import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points // 2

        initial_h = initial_h.astype(float)

    if current_sum != target_sum:
        diff = target_sum - current_sum
        indices = np.where(initial_h == 0)[0]

            if not indices.size:
                break
            idx = np.random.choice(indices)

        indices = np.where(initial_h == 1)[0]

            idx = np.random.choice(indices)
            initial_h[idx] = 0

        h_full = np.concatenate([h, h])
