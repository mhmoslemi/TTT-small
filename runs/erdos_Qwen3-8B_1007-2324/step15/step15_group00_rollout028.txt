            h_new = np.clip(h_new, 0.0, 1.0)
            h_full = np.concatenate([h_new, h_new[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * (2.0 / size)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h_start = h_new.copy()
        
        elapsed_time += time.time() - start
        if elapsed_time >= budget_s:
            break

    return best_h, best_c5, len(best_h)
