        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

        n_half = n_points // 2
        dx = 2.0 / n_points
        integral_target = n_points / 2

        if initial_h is not None:
            new_x = np.linspace(0, 2, n_points)
            initial_x = np.linspace(0, 2, original_len)
            h_initial = np.interp(new_x, initial_x, initial_h)
        else:
            x = np.linspace(0, 2, n_points)
            h_initial = np.clip(1 - np.abs(x - 1), 0, 1)

        sum_initial = np.sum(h_initial)
        scaling_factor = integral_target / sum_initial
        h_initial = np.clip(h_initial * scaling_factor, 0, 1)
        x0 = h_initial[:n_half]

        def compute_overlap(h_opt):
            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_half)]
