        x = np.linspace(0, 2, n_points)
        h_initial = np.zeros(n_points)
        h_initial[:140] = 1 - np.abs(x[:140] - 0.5)
        h_initial[140:] = 1 - np.abs(x[140:] - 1.5)
        h_initial = np.clip(h_initial, 0, 1)
        h_opt = h_initial[:140]

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = (n_points / 2) / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)
