import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = 64
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target_half = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 1, n_half)
        h_opt = 1 - np.abs(x - 0.5)

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target_half / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    h_opt = repair(h_opt)

    def objective(h):
        h_repaired = repair(h)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    result = minimize(objective, x0=h_opt, method='Nelder-Mead', options={'xatol': 1e-4, 'fatol': 1e-4, 'maxiter': 200})
    elapsed = time.time() - start_time

    if elapsed > budget_s - 1:
        h_repaired = repair(h_opt)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
        return np.concatenate([h_repaired, h_repaired[::-1]]), c5, n_points

    best_h = result.x
    h_repaired_best = repair(best_h)
    h_full_best = np.concatenate([h_repaired_best, h_repaired_best[::-1]])
    c5_best = np.max(np.correlate(h_full_best, 1 - h_full_best, mode='full')) * dx

    return np.concatenate([h_repaired_best, h_repaired_best[::-1]]), c5_best, n_points
