import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    n_points = 64
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 1, n_half)
        h_opt = 1 - np.abs(x - 0.5)
        current_sum = np.sum(h_opt)
        scaling_factor = integral_target / current_sum
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    start_time = time.time()
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1',
            mutation=0.7,
            recombination=0.9,
            maxiter=100,
            popsize=20,
            tol=1e-4,
            disp=False,
            polish=True,
            vectorized=True
        )
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            repaired = repair(h_opt)
            h_full = np.concatenate([repaired, repaired[::-1]])
            c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx

import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 1, n_half)
        h_opt = 1 - np.abs(x - 0.5)
        current_sum = np.sum(h_opt)
        scaling_factor = integral_target / current_sum
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    start_time = time.time()
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1',
            mutation=0.8,
            recombination=0.9,
            maxiter=100,
            popsize=20,
            tol=1e-4,
            disp=False,
            polish=True,
            vectorized=True
        )
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            repaired = repair(h_opt)
            h_full = np.concatenate([repaired, repaired[::-1]])
            c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
            return np.concatenate([repaired, repaired[::-1]]), c5, n_points
