import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        desired_sum = n_points / 2.0
        h_opt = h_initial
    else:

        n_points = 280

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)

        if s == 0:
            return compute_overlap(np.zeros_like(h))

        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - desired_sum,
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='COBYLA',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 1000,
            'disp': False,
            'rhobeg': 1.0,
            'eps': 1e-8

    )
