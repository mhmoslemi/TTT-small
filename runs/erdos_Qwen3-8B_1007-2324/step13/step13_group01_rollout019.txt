import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        h_opt = h_initial[:n_points//2]
        integral_target = n_points / 4.0

        n_points = 280
        h_opt = np.random.rand(n_points//2)
        integral_target = n_points / 4.0
        dx = 2.0 / n_points
