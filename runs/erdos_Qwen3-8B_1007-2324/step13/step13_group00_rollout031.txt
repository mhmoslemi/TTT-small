        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=150,
        tol=1e-7,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([h_opt, h_opt[::-1]]), compute_overlap(h_opt), n_points
    
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
