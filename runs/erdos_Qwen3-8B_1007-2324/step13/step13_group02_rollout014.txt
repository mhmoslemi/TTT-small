import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        n_half = 140
        dx = 2.0 / n_points

    else:
        x = np.linspace(0, 1, 140)
        h_opt = 1 - np.abs(x - 0.5)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (70.0 / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = 70.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(140)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 70.0,
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 3000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
