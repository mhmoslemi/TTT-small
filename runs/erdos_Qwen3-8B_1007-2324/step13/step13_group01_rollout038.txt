import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        h_opt = h_initial[:n_points//2]
    else:
        n_points = 280
        h_opt = np.random.rand(n_points//2)
    
    integral_target = n_points / 4
    dx = 2.0 / n_points

    def repair(x):
        s = np.sum(x)
        if s <= 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    h_opt = repair(h_opt)

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]

    result = scipy.optimize.basinhopping(
        objective,
        h_opt,
        niter=100,
        T=1.0,
        stepsize=0.5,
        minimizer_kwargs={
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': constraints,
            'options': {
                'ftol': 1e-10,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        },
        niter_success=5,
        seed=seed
    )

    best_h_opt = result.x
    best_c5 = result.fun
    full_h = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return full_h, best_c5, n_points
