import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            raise ValueError("Initial values length mismatch")
        h_opt = h_initial
    else:
        h_opt = np.zeros(n_points)
        for i in range(n_points):
            if i < 70:
                h_opt[i] = i / 70.0
            elif i < 140:
                h_opt[i] = 1.0 - (i - 70) / 70.0
            elif i < 210:
                h_opt[i] = (i - 140) / 70.0
            else:
                h_opt[i] = 1.0 - (i - 210) / 70.0

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = 140.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 140.0,
        'jac': lambda x: np.ones_like(x)
    }]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 1000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h_opt, compute_overlap(h_opt), n_points

    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    return best_h_opt, best_c5, n_points
