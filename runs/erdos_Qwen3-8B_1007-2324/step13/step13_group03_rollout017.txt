import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    integral_target = n_points // 2  # sum(h) = 32

    # Initialize h_opt based on parent or generate new pattern
    if 'initial_h_values' in globals():
        parent_initial = initial_h_values[:n_points//2]
        h_opt = parent_initial.astype(float)

        h_opt = 1 - np.abs(x - 0.5)

    current_sum = np.sum(h_opt)

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]

    start_time = time.time()

    result = minimize(
        objective,
        h_opt,

        bounds=bounds,
