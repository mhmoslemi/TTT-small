import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        nx = n_points
        dx = 2.0 / nx
        h_opt = h_initial[:nx]
    else:
        x = np.linspace(0, 1, 140)
        h_opt = 1 - np.abs(x - 0.5)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (70.0 / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt[:140]
        nx = 140
        dx = 2.0 / nx

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = (nx * 0.5) / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(nx), lb=nx * 0.5, ub=nx * 0.5)

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={
            'ftol': 1e-9,
            'maxiter': 3000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), 2 * nx

    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return h_full, best_c5, 2 * nx
