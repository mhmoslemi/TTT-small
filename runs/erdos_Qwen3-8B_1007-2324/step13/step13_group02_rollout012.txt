import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        h_opt = h_initial[:140]
    else:
        n_points = 280
        x = np.linspace(0, 2, n_points)
        # Generate multi-peak initial guess
        num_peaks = 5
        positions = np.linspace(0.3, 1.7, num_peaks)
        widths = 0.2
        h = np.zeros(n_points)
        for pos in positions:
            h += 1 - np.abs(x - pos)/widths
        total = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / total) * (n_points / 2.0)
        h = np.clip(h, 0, 1)
        h_opt = h[:140]
    
    dx = 2.0 / n_points
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = (n_points / 2.0) / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(140)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2})
    
    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-8,
            'maxiter': 500,
            'disp': False,
            'eps': 1e-8
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), n_points
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
