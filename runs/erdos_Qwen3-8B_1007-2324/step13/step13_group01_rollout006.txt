import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        h_opt = h_initial[:n_points//2]
    else:
        n_points = 280
        h_opt = np.random.rand(n_points//2)
    integral_target = n_points / 2
    dx = 2.0 / n_points

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    h_opt = repair(h_opt)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_c5(h_opt)

    start_time = time.time()

    temp = 1.0
    cooling_rate = 0.995
    steps = 0
    max_steps = int(budget_s * 100)

    while steps < max_steps:
        if time.time() - start_time > budget_s:
            break
        noise = np.random.randn(len(h_opt)) * 0.1
        h_candidate = h_opt + noise
        h_candidate = repair(h_candidate)
        current_c5 = best_c5
        new_c5 = compute_c5(h_candidate)
        if new_c5 < current_c5:
            h_opt = h_candidate
            best_c5 = new_c5
        else:
            delta = new_c5 - current_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                h_opt = h_candidate
                best_c5 = new_c5
        temp *= cooling_rate
        steps += 1

    full_h = np.concatenate([h_opt, h_opt[::-1]])
    return full_h, best_c5, n_points
