        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def objective(h_flat):
        h = h_flat.copy()
        repaired = repair(h)
        corr = np.correlate(repaired, 1 - repaired, mode='full')
        return np.max(corr) * dx
    
    h_initial_flat = h_full_initial
    
    start_time = time.time()
    result = minimize(objective, h_initial_flat, method='Nelder-Mead', tol=1e-4, options={'maxiter': 500})
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = repair(h_initial_flat)
        c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
        return h_full, c5, n_points
    
    best_h_flat = result.x
    h_full = repair(best_h_flat)
    c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
    
    return h_full, c5, n_points
