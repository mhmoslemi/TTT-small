from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        dx = 2.0 / n_points
        h_opt = h_initial[:140]
    else:
        n_points = 140
        dx = 2.0 / n_points
        grid = np.linspace(0, 1, n_points)
        h = np.zeros_like(grid)
        np.random.seed(seed)
        centers = np.random.uniform(0.25, 0.75, 5)
        for c in centers:
            base_length = 0.15
            h += np.clip(1 - np.abs(grid - c) / (base_length / 2), 0, 1)
        total = np.sum(h)
        scaling = 70.0 / (total + 1e-10)
        h = h * scaling
        h_opt = h
    h_opt = np.clip(h_opt, 0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = 70.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(140)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 70.0,
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 2000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), 280

    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return h_full, best_c5, 280
