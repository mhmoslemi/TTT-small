        h_opt = h_initial[:n_half]
    else:
        n_points = 280
        n_half = n_points // 2
        h_opt = np.random.rand(n_half)
    
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    def repair(x):
        current_sum = np.sum(x)
        if abs(current_sum) < 1e-5:
            return np.zeros_like(x)
        scale = integral_target / current_sum
        return np.clip(x * scale, 0.0, 1.0)
    
    def objective(x):
        x_repaired = repair(x)
        h_full = np.concatenate([x_repaired, x_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(repair(x)) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]
    
    result = opt.minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_half)],
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h_opt = result.x
    best_c5 = result.fun
    full_h = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return full_h, best_c5, n_points
