import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)
    else:
        grid = np.linspace(0, 2, n_points)
        base_length = 0.5
        k = 5
        centers = np.random.uniform(0.25, 1.75, k)
        h = np.zeros_like(grid)
        for c in centers:
            h += 1 - np.abs(grid - c) / (base_length / 2)
        h = np.clip(h, 0, 1)
        scale_factor = 140.0 / np.sum(h)
        h = h * scale_factor
        h = np.clip(h, 0, 1)
        h_opt = h.copy()
    
    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = 140.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - 140.0, 'jac': lambda x: np.ones_like(x)})
    
    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-10, 'maxiter': 3000, 'disp': False, 'eps': 1e-8}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h_opt, compute_overlap(h_opt), n_points
    
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    return best_h_opt, best_c5, n_points
