import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        n_half = 140
        dx = 2.0 / n_points
        h_opt = h_initial[:n_half]
    else:
        # Binary initial guess: 1s every other point
        h_opt = np.zeros(n_points)

        scale = (n_points / 2) / sum_h
        h_opt = np.clip(h_opt * scale, 0, 1)

    def compute_overlap(h_opt):
        h_full = h_opt
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = (n_points / 2) / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - n_points / 2,
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={

            'disp': False,

    best_h_opt = result.x
