        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]

    x0 = h_initial.copy()

    start_time = time.time()
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-12,
            'maxiter': 200,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_opt = x0
        c5 = compute_overlap(h_opt)
        return h_opt, c5, n_points

    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    return best_h_opt, best_c5, n_points
