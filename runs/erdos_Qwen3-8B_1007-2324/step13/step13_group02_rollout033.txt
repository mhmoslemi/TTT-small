            end = (i + 1) * interval_width
            mask = (x >= start) & (x <= end)
            x_in_interval = x[mask] - start
            x_normalized = x_in_interval / interval_width
            triangle = np.where(x_normalized < 0.5, x_normalized, 1 - x_normalized)
            h[mask] = triangle
        h_opt = h

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):

        current_sum = np.sum(h)
        if current_sum == 0:

        repaired = np.clip(h * scaling, 0, 1)

    })

        method='SLSQP',
