import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64
    dx = 2.0 / n_points
    integral_target = n_points // 2  # sum(h) = 32

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) == n_points:
            h_opt = h_initial[:n_points//2]
        else:
            x = np.linspace(0, 1, n_points//2)
            h_opt = 1 - np.abs(x - 0.5)
            current_sum = np.sum(h_opt)
            scaling_factor = (integral_target / 2.0) / current_sum
            h_opt = np.clip(h_opt * scaling_factor, 0, 1)
    else:
        x = np.linspace(0, 1, n_points//2)
        h_opt = 1 - np.abs(x - 0.5)
        current_sum = np.sum(h_opt)
        scaling_factor = (integral_target / 2.0) / current_sum
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)

    def repair(h):
        current_sum = np.sum(h)

            return np.zeros_like(h)

    start_time = time.time()
