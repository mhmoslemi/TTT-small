import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        h_opt = h_initial[:n_points]
    else:
        n_points = 280
        h_opt = np.zeros(n_points)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum < 1e-8:
            return compute_overlap(np.zeros_like(h))
        scaling = 140.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 140.0,
        'jac': lambda x: np.ones_like(x)
    }]

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-8,
            'maxiter': 1500,
            'disp': False,
            'eps': 1e-8,
            'iprint': -1
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([h_opt, h_opt[::-1]]), compute_overlap(h_opt), n_points

    best_c5 = compute_overlap(best_h)
