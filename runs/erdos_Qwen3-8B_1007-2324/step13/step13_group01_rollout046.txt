        strategy='best1bin',

import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 4
    dx = 2.0 / n_points

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    def objective(x):
        x_repaired = repair(x)
        h_full = np.concatenate([x_repaired, x_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]

    start_time = time.time()
    result = scipy.optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=20,
        maxiter=50,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        strategy='best1bin',
        seed=seed
    )
    end_time = time.time()

    # Early exit if time is exhausted
    if end_time - start_time > budget_s:
        return np.concatenate([h_opt, h_opt[::-1]]), objective(h_opt), n_points

    best_h_opt = result.x
    best_c5 = result.fun
