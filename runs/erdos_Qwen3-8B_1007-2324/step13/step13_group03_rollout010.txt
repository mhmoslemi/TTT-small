import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = 64
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_half = np.interp(np.linspace(0, 1, n_points // 2), np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        h_half = np.ones(n_points // 2) * 0.5

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]
    
    start_time = time.time()
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1',
            mutation=0.9,
            recombination=0.9,
            maxiter=150,
            popsize=20,
            tol=1e-4,
            disp=False,
            polish=True,
            vectorized=True
        )
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            repaired = repair(h_half)
            h_full = np.concatenate([repaired, repaired[::-1]])
            c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
            return np.concatenate([repaired, repaired[::-1]]), c5, n_points
