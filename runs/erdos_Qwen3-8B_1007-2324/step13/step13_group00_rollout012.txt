import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    # Determine n_points based on the initial_h_values length
    if 'initial_h_values' in globals():
        initial_len = len(initial_h_values)
        n_points = initial_len
    else:
        n_points = 60  # default if no initial values are provided
    
    # Compute dx and integral target
    dx = 2.0 / n_points
    integral_target = n_points / 2  # since sum(h) * dx = 1 → sum(h) = n_points / 2
    
    # Initialize h_opt based on initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        else:
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
        h_opt = h_initial[:n_points//2]  # take first half for symmetric approach
    else:
        # Default initialization: triangular shape
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h_opt[:n_points//2]
    
    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    start_time = time.time()
    
    # Use differential evolution first to explore search space
    result_de = differential_evolution(objective, bounds, 
                                      strategy='best1bin', 
                                      maxiter=50, popsize=15, tol=1e-6, 
                                      mutation=(0.5, 1), recombination=0.8, 
                                      disp=False, polish=True)
    
    # Apply SLSQP for refinement
    x0 = result_de.x
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })
    result_slsqp = minimize(objective, x0, method='SLSQP', 
                            bounds=bounds, constraints=constraints,
                            options={'ftol': 1e-10, 'maxiter': 100, 'disp': False})
    
    best_h_opt = result_slsqp.x
    best_c5 = compute_overlap(best_h_opt)
    best_h = np.concatenate([best_h_opt, best_h_opt[::-1]])
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([x0, x0[::-1]]), compute_overlap(x0), n_points
    
    return best_h, best_c5, n_points
