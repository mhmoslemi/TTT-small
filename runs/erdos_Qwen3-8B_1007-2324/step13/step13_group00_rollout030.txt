        h_opt = h_initial.copy()
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = x.copy()
        repaired_h = repair(h)
        return compute_overlap(repaired_h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tolerance=1e-5,
        maxiter=200,
        polish=True,
        workers=1,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_initial_repaired = repair(h_opt)
        best_c5 = compute_overlap(h_initial_repaired)
        return h_initial_repaired, best_c5, n_points
    
    best_h = result.x
    best_h_repaired = repair(best_h)
    best_c5 = compute_overlap(best_h_repaired)
    
    return best_h_repaired, best_c5, n_points
