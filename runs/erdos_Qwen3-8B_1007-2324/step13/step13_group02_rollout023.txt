from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 140
        h_opt = h_initial[:n_points]
    else:
        x = np.linspace(0, 1, 140)
        h_opt = 1 - np.abs(x - 0.5)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (70.0 / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0, 1)

    dx = 1.0 / 140.0

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = 70.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(140)]
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-8,
        mutation=(0.2, 0.8),
        recombination=0.5,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), 280
