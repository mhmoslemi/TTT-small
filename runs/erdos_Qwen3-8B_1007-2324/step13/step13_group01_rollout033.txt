        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-10,
        maxiter=100,
        seed=seed
    )

    best_h_opt = result.x
    best_c5 = result.fun
    full_h = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return full_h, best_c5, n_points
