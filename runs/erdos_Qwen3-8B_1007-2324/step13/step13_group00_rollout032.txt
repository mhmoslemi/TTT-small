        initial_h = np.array(initial_h_values)
        m = len(initial_h)
        n_points = 80
        x_initial = np.linspace(0, 2, m)
        x_new = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_new, x_initial, initial_h)
    else:
        n_points = 80
        x = np.linspace(0, 2, n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_overlap(h_opt):
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        return np.max(corr) * (2.0 / n_points)

    integral_target = n_points / 2
    dx = 2.0 / n_points

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]

    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 500,
            'disp': False,
            'eps': 1e-8
        }
    )
    elapsed = time.time() - start_time

    best_h = result.x
    best_c5 = compute_overlap(best_h)
    return best_h, best_c5, n_points
