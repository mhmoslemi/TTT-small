import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        x_initial = h_initial[:n_points//2]
    else:
        n_points = 280
        x_initial = np.random.rand(n_points//2)

    target_sum = n_points / 4
    dx = 2.0 / n_points

    def objective(x):

        corr = np.correlate(h_full, 1 - h_full, mode='full')

        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(len(x_initial))]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-10,
        maxiter=100,
        seed=seed,
    )

    best_x = result.x
    s = np.sum(best_x)
