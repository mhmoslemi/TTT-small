import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        n_half = n_points // 2
        h_opt = h_initial[:n_half]
    else:
        n_points = 280
        n_half = n_points // 2
        h_opt = np.random.rand(n_half)
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    def objective(x):
        h_repaired = repair(x)
        full_h = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(full_h, 1 - full_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    # Run differential evolution with limited iterations to fit budget
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    elapsed = time.time() - start_time
    # Early stop if budget exhausted
    if elapsed > budget_s - 1:
        result = scipy.optimize.OptimizeResult(x=h_opt, fun=objective(h_opt), success=True)
    best_h = result.x
    best_c5 = result.fun
    full_h = np.concatenate([best_h, best_h[::-1]])
    return full_h, best_c5, n_points
