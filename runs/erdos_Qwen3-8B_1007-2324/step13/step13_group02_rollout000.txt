import numpy as np
import time
from scipy.optimize import minimize, linprog

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = 280
        h_opt = h_initial
    else:
        x = np.linspace(0, 1, 140)
        h_opt = 1 - np.abs(x - 0.5)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (70.0 / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0, 1)
        n_points = 140

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = (n_points / 2) / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - (n_points / 2),
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-8,
            'maxiter': 3000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), 2 * n_points

    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return h_full, best_c5, 2 * n_points
