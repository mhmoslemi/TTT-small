            maxiter=500,
            popsize=20,
            tol=1e-4,
            disp=False,
            polish=True,
            vectorized=True
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            repaired = repair(h_opt)
            h_full = np.concatenate([repaired, repaired[::-1]])
            c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
            return np.concatenate([repaired, repaired[::-1]]), c5, n_points

        best_h_opt = result.x
        repaired_best = repair(best_h_opt)
        h_full_best = np.concatenate([repaired_best, repaired_best[::-1]])
        c5_best = np.max(np.correlate(h_full_best, 1 - h_full_best, mode='full')) * dx
        return np.concatenate([repaired_best, repaired_best[::-1]]), c5_best, n_points

    except Exception as e:

        return np.concatenate([repaired, repaired[::-1]]), c5, n_points
