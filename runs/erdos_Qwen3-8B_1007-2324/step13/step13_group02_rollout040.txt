import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    dx = 2.0 / n_points
    sum_required = n_points / 2

    else:

        k = 5
        centers = np.random.uniform(0.25, 1.75, size=k)
        h = np.zeros(n_points)
        for c in centers:
            dist = np.abs(grid - c)
            t = 1 - dist / 0.25
            t = np.clip(t, 0, 1)
            h += t
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 0:
            h *= sum_required / total
        h = np.clip(h, 0, 1)
        h_opt = h

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = sum_required / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

        method='SLSQP',
