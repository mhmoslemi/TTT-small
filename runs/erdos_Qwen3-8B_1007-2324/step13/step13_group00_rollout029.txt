            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    if best_h is None:
        if h_initial is not None:
            h = h_initial
        else:
            x = np.linspace(0, 2, 60)
            h = 1 - np.abs(x - 1)
            h = np.clip(h, 0, 1)
        return np.concatenate([h, h[::-1]]), compute_overlap(h), 60
    else:
        h_full = np.concatenate([best_h, best_h[::-1]])
        return h_full, best_c5, len(best_h)
