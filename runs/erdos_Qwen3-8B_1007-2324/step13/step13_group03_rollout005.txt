def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = 64
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():

        h_opt = h_initial[:n_points]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()

            objective,
            bounds,
