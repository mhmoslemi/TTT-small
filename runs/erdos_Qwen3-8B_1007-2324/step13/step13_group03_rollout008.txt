import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = 256
    integral_target = n_points / 4
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points // 2]
    else:
        x = np.linspace(0, 1, n_points // 2)
        h_opt = 1 - np.abs(x - 0.5)
        current_sum = np.sum(h_opt)
        scaling_factor = integral_target / current_sum
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]
    
    start_time = time.time()
    
    try:

            strategy='best1bin',
            mutation=(0.6, 1.0),

            polish=True,
            vectorized=True
