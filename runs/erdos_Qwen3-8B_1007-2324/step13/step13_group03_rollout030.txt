    integral_target = n_points // 2  # 32

    def generate_initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            x = i * dx
            if x <= 0.5:
                h[i] = 2 * x
            else:
                h[i] = 2 * (1 - x)
        for i in range(n_points // 2, n_points):
            h[i] = h[n_points - 1 - (i - n_points // 2)]
        return h

    h_initial = generate_initial_guess()

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def objective(x):
        h = repair(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1',
            mutation=0.7,
            recombination=0.9,
            maxiter=100,
            popsize=20,
            tol=1e-4,
            disp=False,
            polish=True,
            vectorized=True
        )
    except:
        result = None

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = repair(h_initial)
        c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
        return np.concatenate([h_full, h_full[::-1]]), c5, n_points

    best_h = result.x
    best_h = repair(best_h)
    c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return best_h, c5, n_points
