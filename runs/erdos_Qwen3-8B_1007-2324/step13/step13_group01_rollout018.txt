import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        h_opt = h_initial[:n_points//2]
    else:
        n_points = 280
        h_opt = np.random.rand(n_points//2)
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    h_opt = repair(h_opt)

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    current_h = h_opt.copy()
    best_h = h_opt.copy()
    best_c5 = objective(best_h)
    T = 1.0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        perturbation = np.random.randn(len(current_h)) * T * 0.1
        new_h = current_h + perturbation
        new_h = repair(new_h)
        current_c5 = objective(current_h)
        new_c5 = objective(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h.copy()
        if objective(current_h) < best_c5:
            best_h = current_h.copy()
            best_c5 = objective(best_h)
        T *= 0.995

    full_h = np.concatenate([best_h, best_h[::-1]])
    return full_h, best_c5, n_points
