def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    else:
        n_points = 140
        x = np.linspace(0, 1, n_points)

        h_initial = np.clip(h_initial, 0, 1)
        total = np.sum(h_initial)
        h_initial = h_initial * (70.0 / total) if total != 0 else np.zeros_like(h_initial)
        n_points = 280
        h_initial = np.concatenate([h_initial, h_initial[::-1]])
    dx = 2.0 / n_points

    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clip = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clip)
        if total == 0:
            return np.zeros_like(h)
        scaling = 1.0 / total
        return h_clip * scaling

    # PSO parameters
    n_particles = 50
    max_iter = 100
    inertia = 0.729
    phi1 = 1.494
    phi2 = 1.494

    particles = np.random.uniform(0, 1, (n_particles, n_points))
    velocities = np.random.uniform(-1, 1, (n_particles, n_points))

    personal_best = particles.copy()
    personal_best_vals = np.array([compute_overlap(p) for p in personal_best])
    global_best = personal_best[np.argmin(personal_best_vals)]
    global_best_value = np.min(personal_best_vals)

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + phi1 * r1 * (personal_best[i] - particles[i]) + phi2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = repair(particles[i])
