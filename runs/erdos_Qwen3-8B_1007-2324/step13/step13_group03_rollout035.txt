    integral_target = n_points / 2
    
    h_initial = np.array([])
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1.0
        current_sum = np.sum(h_initial)
        scaling_factor = integral_target / current_sum
        h_initial = np.clip(h_initial * scaling_factor, 0, 1)
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def objective(x):
        h = x.copy()
        repaired = repair(h)
        corr = np.correlate(repaired, 1 - repaired, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            mutation=(0.5, 1.0),
            recombination=0.9,
            maxiter=100,
            popsize=30,
            tol=1e-4,
            disp=False,
            polish=True,
            vectorized=True
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            repaired = repair(h_initial)
            c5 = np.max(np.correlate(repaired, 1 - repaired, mode='full')) * dx
            return repaired, c5, n_points
        
        best_h = result.x
        repaired_best = repair(best_h)
        c5_best = np.max(np.correlate(repaired_best, 1 - repaired_best, mode='full')) * dx
        
        return repaired_best, c5_best, n_points
    
    except:
        repaired = repair(h_initial)
        c5 = np.max(np.correlate(repaired, 1 - repaired, mode='full')) * dx
        return repaired, c5, n_points
