        # Generate initial guess for this grid size
        if initial_h_available and n_points == grid_sizes[0]:
            # Interpolate initial_h to current grid size
            x_parent = np.linspace(0, 2, len(initial_h))

        else:
            # Use symmetric triangular wave
            x = np.linspace(0, 2, n_points)

        scaling_factor = integral_target / sum_h if sum_h != 0 else 1.0
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)
