        dx = 2.0 / n_points

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = n_points / 2.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

    start_time = time.time()
    result = minimize(

        h_opt,
        method='SLSQP',
