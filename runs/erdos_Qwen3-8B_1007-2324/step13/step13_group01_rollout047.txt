import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
        h_opt = h_initial[:n_points//2].copy()
    else:
        n_points = 280
        h_opt = np.random.rand(n_points//2)
    integral_target = n_points / 4
    dx = 2.0 / n_points

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    h_opt = repair(h_opt)

    def objective(x):
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    temp = 10.0
    cooling_rate = 0.995
    max_iter = 2000
    best_c5 = objective(h_opt)
    best_h = h_opt.copy()

    for _ in range(max_iter):
        perturbation = np.random.normal(0, 0.05, h_opt.shape)
        x_new = h_opt + perturbation

        c5_new = objective(x_new)
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            if c5_new < best_c5:
                best_c5 = c5_new
            h_opt = x_new.copy()
        temp *= cooling_rate
        if temp < 1e-5:

    full_h = np.concatenate([h_opt, h_opt[::-1]])
