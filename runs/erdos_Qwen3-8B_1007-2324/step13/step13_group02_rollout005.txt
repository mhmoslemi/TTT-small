from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.zeros_like(x)
        for i in range(5):
            center = 0.4 * i + 0.2
            start = center - 0.2
            end = center + 0.2
            mask = (x >= start) & (x <= end)
            for idx in np.where(mask)[0]:
                dx_val = x[idx] - center
                h[idx] = 1 - np.abs(dx_val) / 0.2
        total_area = np.sum(h) * dx
        scale = 140.0 / (np.sum(h) * dx)
        h = np.clip(h * scale, 0, 1)
        h_initial = h
    h_opt = h_initial.copy()

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = 140.0 / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 140.0,
        'jac': lambda x: np.ones_like(x)
    })

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 1000,
            'disp': False,
            'eps': 1e-8
        }
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:

        return h_full, compute_overlap(h_opt), 280

    best_h_opt = result.x
