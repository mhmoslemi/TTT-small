            return h_full, c5, n_points
        else:
            best_h = result.x
            repaired_best = repair(best_h)
            h_full = repaired_best
            c5_best = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
            return h_full, c5_best, n_points
        
    except Exception as e:
        repaired = repair(h_initial)
        h_full = repaired
        c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
        return h_full, c5, n_points
