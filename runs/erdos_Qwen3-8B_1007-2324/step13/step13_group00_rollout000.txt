        x = np.linspace(0, 2, n_points_min)
        h_base = np.clip(1 - np.abs(x - 1), 0, 1)
    
    def repair(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target / s
        return np.clip(h * scale, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    for n in grid_sizes:
