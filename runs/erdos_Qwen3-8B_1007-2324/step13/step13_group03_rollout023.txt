import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 64
    n_half = n_points // 2
    dx = 2.0 / n_points
    sum_opt_target = n_points / 4  # sum of h_opt is integral_target / 2

    # Start with the parent solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
        # Scale to match target sum
        current_sum = np.sum(h_opt)
        scaling_factor = sum_opt_target / current_sum
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)
    else:
        # Symmetric triangular starting guess
        x = np.linspace(0, 1, n_half)
        h_opt = 1 - np.abs(x - 0.5)
        current_sum = np.sum(h_opt)
        scaling_factor = sum_opt_target / current_sum
        h_opt = np.clip(h_opt * scaling_factor, 0, 1)

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:

        scaling_factor = sum_opt_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='Nelder-Mead',
        bounds=bounds,
        options={'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 200}

    best_h_opt = result.x
    repaired_best = repair(best_h_opt)
    h_full_best = np.concatenate([repaired_best, repaired_best[::-1]])
    c5_best = np.max(np.correlate(h_full_best, 1 - h_full_best, mode='full')) * dx
