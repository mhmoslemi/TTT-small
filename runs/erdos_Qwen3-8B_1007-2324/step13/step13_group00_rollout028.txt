        if initial_h is not None and len(initial_h) >= n_points:
            x = np.linspace(0, 2, n_points)
            x_ref = np.linspace(0, 2, len(initial_h))
            h_initial = np.interp(x, x_ref, initial_h)
        else:
            x = np.linspace(0, 2, n_points)
            h_initial = np.clip(1 - np.abs(x - 1) / 1.0, 0, 1)
        
        h_opt = h_initial[:n_points//2]
        integral_target = n_points / 2
        dx = 2.0 / n_points
        
        def objective(x):
            h = x.copy()
            repaired = repair(h, integral_target)
            return compute_overlap(repaired)
        
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })
        
        result = minimize(
            objective,
            h_opt,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'disp': False, 'ftol': 1e-8}
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.concatenate([result.x, result.x[::-1]])
    
    return best_h, best_c5, len(best_h) if best_h is not None else 60
