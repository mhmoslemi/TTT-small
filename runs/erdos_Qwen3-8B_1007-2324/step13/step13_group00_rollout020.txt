import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Increased resolution for better control

    integral_target = n_points / 2  # sum(h) = integral_target

    # Initialize with a non-symmetric pattern if available, or generate new
    if 'initial_h_values' in globals():

                h_initial[i] = initial[idx]
