import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    grid_sizes = [40, 60, 80]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        parent_len = len(parent_h)
    else:
        parent_h = None
        parent_len = 0
    
    for n_points in grid_sizes:
        dx = 2.0 / n_points
        integral_target = n_points / 2
        
        if parent_h is not None and parent_len >= n_points:
            step = int(np.ceil(parent_len / n_points))
            h_initial = parent_h[::step][:n_points]
        else:
            x = np.linspace(0, 2, n_points)
            h_initial = 1 - np.abs(x - 1)
        
        def repair(h):
            s = np.sum(h)
            if s == 0:
                return np.zeros_like(h)
            scale = integral_target / s
            return np.clip(h * scale, 0, 1)
        
        h_initial = repair(h_initial)
        
        def objective(x):
            h = repair(x)
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0., 1.) for _ in range(n_points)]
        x0 = h_initial.copy()
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target})
        
        start_time = time.time()
        try:
            result = scipy.optimize.minimize(
                objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                options={'ftol': 1e-9, 'maxiter': 300, 'disp': False, 'eps': 1e-8}
            )
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:

                current_c5 = objective(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                    best_n = n_points
        except:
            pass
    
    return best_h, best_c5, best_n
