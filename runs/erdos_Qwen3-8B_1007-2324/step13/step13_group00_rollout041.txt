    def repair(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = target_sum / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_overlap(h, n_points):
        dx = 2.0 / n_points
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Grid refinement strategy
    n_points_candidates = [40, 60, 80]
    best_c5 = float('inf')
