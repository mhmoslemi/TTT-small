import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 280
    dx = 2.0 / n_points
    sum_h = 70.0

    def generate_initial():
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        centers = np.random.uniform(0.25, 1.75, 5)
        for c in centers:
            dist = np.abs(grid - c)
            tent = np.clip(1 - dist / 0.25, 0, 1)
            h += tent
        h = np.clip(h, 0, 1)
        h = h * (sum_h / np.sum(h)) if np.sum(h) > 1e-10 else h
        return h

    if h_initial is not None:
        h_opt = h_initial[:n_points]
    else:
        h_opt = generate_initial()

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        if np.sum(h) == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = sum_h / np.sum(h)
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - sum_h,
        'jac': lambda x: np.ones_like(x)
    }]

    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 1000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), n_points

    best_h = result.x
    best_c5 = compute_overlap(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
