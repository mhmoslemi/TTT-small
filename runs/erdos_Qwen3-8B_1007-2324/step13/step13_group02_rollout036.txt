    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - 70.0, 'jac': lambda x: np.ones_like(x)})
    
    start_time = time.time()
    result = minimize(
        objective,
        h_opt,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 3000,
            'disp': False,
            'eps': 1e-8
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        return h_full, compute_overlap(h_opt), n_points
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
