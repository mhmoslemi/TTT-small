        base_n_points = n_points_list[0]
        base_x = np.linspace(0, 2, base_n_points)
        initial_x = np.linspace(0, 2, len(initial_guess))
        h_initial = np.interp(base_x, initial_x, initial_guess)
        h_initial = np.clip(h_initial, 0, 1)
        h_opt_initial = h_initial[:base_n_points // 2]
    else:
        base_n_points = n_points_list[0]
        x = np.linspace(0, 2, base_n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt_initial = h_initial[:base_n_points // 2]

    dx = 2.0 / base_n_points
    integral_target = base_n_points / 2.0

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for n_points in n_points_list:
        if best_h is None:
            h_opt = h_opt_initial.copy()
        else:
            prev_n_half = len(best_h) // 2
            prev_x = np.linspace(0, 2, 2 * prev_n_half)
            new_x = np.linspace(0, 2, 2 * n_points)
            h_new = np.interp(new_x, prev_x, best_h)
            h_new = np.clip(h_new, 0, 1)
            h_opt = h_new[:n_points // 2]
        integral_target = n_points / 2.0

        def objective(x):
            h = x.copy()
            repaired = repair(h, integral_target)
            return compute_overlap(repaired)

        bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })

        result = scipy.optimize.minimize(
            objective,
            h_opt,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={
                'disp': False,
                'maxiter': 200,
                'ftol': 1e-8,
                'rhobeg': 1e-3,
                'eps': 1e-8
            }
        )

        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.concatenate([h_opt, h_opt[::-1]])

    if best_h is None:
        h_values = np.concatenate([h_opt_initial, h_opt_initial[::-1]])
        c5_bound = compute_overlap(h_opt_initial)
    else:
        h_values = best_h
        c5_bound = best_c5

    return h_values, c5_bound, len(h_values)
