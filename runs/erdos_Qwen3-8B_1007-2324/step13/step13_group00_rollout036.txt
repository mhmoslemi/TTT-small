        integral_target = n_points / 4.0

        if initial_h is not None:
            x_initial = np.linspace(0, 2, len(initial_h))
            x_new = np.linspace(0, 2, n_points)
            h_initial_new = np.interp(x_new, x_initial, initial_h)
            h_opt_initial = h_initial_new[:n_points // 2]
        else:
            x = np.linspace(0, 2, n_points)
            h_full = 1 - np.abs(x - 1)
            h_full = np.clip(h_full, 0, 1)
            h_opt_initial = h_full[:n_points // 2]

        def repair(h, target):
            current_sum = np.sum(h)
            if current_sum == 0:
                return np.zeros_like(h)
            scaling = target / current_sum
            return np.clip(h * scaling, 0, 1)

        h_opt_initial = repair(h_opt_initial, integral_target)

        def compute_overlap(h_opt):
            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def objective(x):
            h_opt = x.copy()
            repaired = repair(h_opt, integral_target)
            return compute_overlap(repaired)

        bounds = [(0.0, 1.0) for _ in range(n_points // 2)]

        result = minimize(
            objective,
            x0,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,

        )

        if result.success:
