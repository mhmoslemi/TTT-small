def run(seed=42, budget_s=100, **kwargs):

        if sum_pop > 0:

    best_h = pop[0]

    generations = 50
    mutation_prob = 0.3
    crossover_prob = 0.7

    for _ in range(generations):
        parents = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(pop, size=3, replace=False)
            winner = tournament[np.argmin([evaluate_erdos_solution(h) for h in tournament])]
            parents.append(winner)

        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            if np.random.rand() < crossover_prob:
                child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
            else:
                child1, child2 = p1, p2
            offspring.append(child1)
            offspring.append(child2)

        for h in offspring:
            for j in range(n_points):
                if np.random.rand() < mutation_prob:
                    h[j] += np.random.normal(0, 0.05)
                    h[j] = np.clip(h[j], 0, 1)
                sum_h = np.sum(h)
                if sum_h != 0:
                    h *= integral_target / sum_h

        new_pop = []
        for h in offspring:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
