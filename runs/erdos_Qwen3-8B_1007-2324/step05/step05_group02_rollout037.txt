            segment = h_fine_initial[start_idx:end_idx]
            h_coarse_initial.append(np.mean(segment))
        h_coarse_initial = np.array(h_coarse_initial)
        h_coarse_initial = np.clip(h_coarse_initial, 0, 1)
        sum_val = np.sum(h_coarse_initial)
        if sum_val > 0:
            h_coarse_initial *= (integral_target_coarse / sum_val)
    else:
        h_coarse_initial = np.full(n_coarse, 0.5)

    # Coarse grid optimization
    def objective_coarse(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_coarse

    bounds_coarse = [(0.0, 1.0) for _ in range(n_coarse)]
    constraints_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_coarse}]
    
    result_coarse = differential_evolution(
        objective_coarse,
        bounds_coarse,
        constraints=constraints_coarse,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        tol=1e-5,
        maxiter=50,
        polish=True
    )
    best_coarse = result_coarse.x
    best_c5_coarse = result_coarse.fun

    # Upsample coarse to fine grid
    def upscale(h_coarse):
        h_fine = []
        for i in range(n_coarse):
            h_fine.extend([h_coarse[i]] * 3)
            if i < n_coarse - 1:
                h_fine.append(0.5 * h_coarse[i] + 0.5 * h_coarse[i+1])
            else:
                h_fine.append(h_coarse[i])
        h_fine = np.array(h_fine)
        sum_val = np.sum(h_fine)
        if sum_val > 0:
            h_fine *= (integral_target_fine / sum_val)
        return h_fine

    h_fine = upscale(best_coarse)
    
    # Fine grid optimization
    def objective_fine(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_fine

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
    constraints_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_fine}]
    
    result_fine = minimize(
        objective_fine,
        h_fine,
        method='SLSQP',
        bounds=bounds_fine,
        constraints=constraints_fine,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    best_fine = result_fine.x
    best_c5_fine = result_fine.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_fine, best_c5_fine, n_fine
    
    return best_fine, best_c5_fine, n_fine
