        h = np.clip(h, 0, 1)
        return h

    # Symmetric widths: w1 = w3, w2 = 2 - 2*w1
    for w1 in np.linspace(0.4, 0.6, 50):
        w2 = 2.0 - 2 * w1
        if w2 > 0:
            weights = [w1, w2, w1]
            h = generate_h(weights)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Asymmetric widths with coarser grid
    base_vals = np.linspace(0.4, 0.8, 50)
    for b1 in base_vals:
        for b2 in base_vals:
            b3 = 2.0 - b1 - b2
            if b3 > 0:
                weights = [b1, b2, b3]
                h = generate_h(weights)
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    elapsed = time.time() - start
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
