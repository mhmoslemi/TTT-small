import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    # Ensure integral constraint
    def adjust_integral(h):
        total = np.sum(h)
        if np.isclose(total, integral_target):
            return h
        scale = integral_target / total
        return h * scale

    initial_h = adjust_integral(initial_h)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr) * dx
        entropy = -np.sum(np.log(1 + np.abs(h - 0.5)))
        return max_corr - 0.001 * entropy

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    # Generate diverse initial points
    np.random.seed(seed)
    starts = [initial_h]
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + noise
        perturbed = np.clip(adjust_integral(perturbed), 0, 1)
        starts.append(perturbed)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
