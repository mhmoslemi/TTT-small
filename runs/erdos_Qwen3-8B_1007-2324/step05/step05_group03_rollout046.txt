        for j in range(n_points):
            j_plus_k = (j + lag) % n_points
            j_minus_k = (j - lag) % n_points
            grad[j] = dx * (1 - h[j_plus_k] - h[j_minus_k])
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        norm = np.sum(h)
        h = h * (integral_target / norm) if norm > 0 else h
    
    # Final evaluation
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    elapsed = time.time() - start_time
    return h.tolist(), c5, n_points
