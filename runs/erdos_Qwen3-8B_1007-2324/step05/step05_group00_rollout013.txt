import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_perturbations(h0, seed):
        np.random.seed(seed)
        shifts = np.random.uniform(-0.4, 0.4, size=n_points)
        h_shifted = np.roll(h0, np.random.randint(0, n_points))
        h_perturbed = h_shifted + np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed > 0:
            h_perturbed *= (integral_target / sum_perturbed)
        return h_perturbed

    starts = [initial_h]
    for i in range(10):
        starts.append(generate_perturbations(initial_h, seed + i + 100))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='COBYLA',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
