from scipy.optimize import minimize
import time

    def repair(h):
        h_cpy = np.clip(h, 0, 1)
        s = np.sum(h_cpy)
        if abs(s - integral_target) > 1e-4:
            h_cpy *= integral_target / s
        return h_cpy
    
    def objective(h):
        h_clip = repair(h)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    num_starts = 15

        start = repair(start)
