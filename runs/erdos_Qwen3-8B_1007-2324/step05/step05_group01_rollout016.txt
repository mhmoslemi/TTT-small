import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    
    def generate_h(base_widths):
        h = np.zeros(n_points)
        start = 0.0
        for b in base_widths:
            end = start + b
            for idx in range(n_points):
                x = idx * dx
                if x < start or x >= end:
                    continue
                rel = x - start
                val = 2 * (rel if rel < b/2 else b - rel) / b
                h[idx] += val
            start = end
        h = np.clip(h, 0, 1)
        return h
    
    # Three-tent grid search
    w1_vals = np.linspace(0.3, 0.9, 20)
    w2_vals = np.linspace(0.3, 0.9, 20)
    for w1 in w1_vals:
        for w2 in w2_vals:
            w3 = 2.0 - w1 - w2
            if w3 <= 0:
                continue
            base_widths = [w1, w2, w3]
            h = generate_h(base_widths)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Two-tent grid search
    w_vals = np.linspace(0.3, 0.9, 20)
    for w in w_vals:
        w2 = 2.0 - w
        base_widths = [w, w2]
        h = generate_h(base_widths)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
