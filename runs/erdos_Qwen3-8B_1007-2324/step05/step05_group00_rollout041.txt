import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:

        h = (h < 0.5).astype(float)

        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = h.copy()

    for _ in range(40):
