import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    integral_target = n_points / 2
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def generate_start(h):
        np.random.seed(seed)
        shift = np.random.randint(0, n_points)
        h_shifted = np.roll(h, shift)
        noise = np.random.normal(0, 0.025, n_points)
        h_noisy = h_shifted + noise
        h_noisy = np.clip(h_noisy, 0, 1)
        sum_h = np.sum(h_noisy)
        h_noisy *= (integral_target / sum_h) if sum_h > 0 else 0.5
        return h_noisy
    
    starts = [initial_h]
    for i in range(20):
        starts.append(generate_start(initial_h))
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
