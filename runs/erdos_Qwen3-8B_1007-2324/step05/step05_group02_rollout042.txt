        return np.max(corr) * dx_coarse

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_coarse_sum}]

    best_coarse = coarse_h.copy()
    best_c5 = objective_coarse(best_coarse)

    # Coarse optimization
    res = minimize(objective_coarse, best_coarse, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-9})
    if res.success and res.fun < best_c5:
        best_c5 = res.fun
        best_coarse = res.x

    # Transfer to fine grid
    def expand_coarse_to_fine(coarse):
        fine = np.zeros(n_fine)
        current_idx = 0
        for i in range(n_coarse):
            if i % 2 == 0:
                count = 3
            else:
                count = 4
            fine[current_idx:current_idx + count] = coarse[i]
            current_idx += count
        return fine

    fine_h = expand_coarse_to_fine(best_coarse)

    def objective_fine(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_fine

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
    constraints_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_fine_sum}]

    res_fine = minimize(objective_fine, fine_h, method='SLSQP', bounds=bounds_fine, constraints=constraints_fine, options={'maxiter': 100, 'ftol': 1e-9})
    if res_fine.success and res_fine.fun < best_c5:
        best_c5 = res_fine.fun
        best_h = res_fine.x
    else:
        best_h = fine_h

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_fine
    return best_h, best_c5, n_fine
