import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 140
    dx = 2.0 / n_points
    integral_target = n_points / 2.0
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
        initial_h = initial_h * integral_target / np.sum(initial_h)
    
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - integral_target
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def optimize_once(h0):
        result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            return result.x, objective(result.x)
        return h0, np.inf
    
    best_h, best_c5 = initial_h.copy(), objective(initial_h)
    start_time = time.time()
    
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * integral_target / np.sum(perturbed)
        h, c5 = optimize_once(perturbed)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
