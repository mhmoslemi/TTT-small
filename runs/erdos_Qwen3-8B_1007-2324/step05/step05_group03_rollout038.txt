    population = np.zeros((pop_size, n_points))

        temp = np.random.rand(n_points)
        temp = np.clip(temp, 0, 1)
        temp_sum = np.sum(temp)
        if temp_sum != 0:
            temp *= integral_target / temp_sum
        population[i, :] = temp
    
    def evaluate_fitness(individual):
        h_clip = np.clip(individual, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr)
        sum_dev = abs(np.sum(h_clip) - integral_target)
        penalty = sum_dev * 10000
        return max_corr * dx + penalty
