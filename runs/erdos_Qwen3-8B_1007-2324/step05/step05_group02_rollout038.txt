    if 'initial_h_values' in globals():
        coarse_initial = resample(initial_h_values, n_coarse)
        coarse_initial = np.clip(coarse_initial, 0, 1)
    else:
        coarse_initial = np.full(n_coarse, 0.5)
        coarse_initial = np.clip(coarse_initial, 0, 1)

    def obj(h, dx, target):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds_coarse = [(0.0, 1.0) for _ in range(n_coarse)]
    cons_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_coarse}]
    sol_coarse = minimize(obj, coarse_initial, args=(dx_coarse, target_coarse), method='SLSQP', bounds=bounds_coarse, constraints=cons_coarse, options={'maxiter': 150, 'ftol': 1e-9, 'disp': False})
    best_coarse = sol_coarse.x if sol_coarse.success else coarse_initial
    c5_coarse = obj(best_coarse, dx_coarse, target_coarse)

    fine_initial = resample(best_coarse, n_fine)
    fine_initial = np.clip(fine_initial, 0, 1)
    sum_fine = np.sum(fine_initial)
    if sum_fine > 0:
        fine_initial *= target_fine / sum_fine

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
    cons_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_fine}]
    sol_fine = minimize(obj, fine_initial, args=(dx_fine, target_fine), method='SLSQP', bounds=bounds_fine, constraints=cons_fine, options={'maxiter': 150, 'ftol': 1e-9, 'disp': False})
    best_fine = sol_fine.x if sol_fine.success else fine_initial
    c5_fine = obj(best_fine, dx_fine, target_fine)

    final_h = best_fine if c5_fine < c5_coarse else best_coarse
    final_c5 = c5_fine if c5_fine < c5_coarse else c5_coarse
    final_n = n_fine if c5_fine < c5_coarse else n_coarse

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_fine if c5_fine < c5_coarse else best_coarse, c5_fine if c5_fine < c5_coarse else c5_coarse, n_fine if c5_fine < c5_coarse else n_coarse)
    
    return (final_h, final_c5, final_n)
