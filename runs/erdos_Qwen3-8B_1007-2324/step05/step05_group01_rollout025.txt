        w2 = np.random.uniform(0.4, 0.8)

        if 0.4 <= w3 <= 0.8:
            h = generate_h(w1, w2, w3)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return (best_h, best_c5, n_points)
