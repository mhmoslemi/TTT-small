from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    def generate_watershed(h0, seed=None):
        if seed is not None:
            np.random.seed(seed)
        noise = np.random.normal(0, 0.02, n_points)
        shifted = np.roll(h0 + noise, np.random.randint(-40, 40))
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    starts = [initial_h]
    for i in range(20):
        starts.append(generate_watershed(initial_h, seed + i))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )

                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    if elapsed > budget_s:
