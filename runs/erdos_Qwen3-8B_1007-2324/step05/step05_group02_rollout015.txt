from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    np.random.seed(seed)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    best_time = 0
    
    # Generate diverse starting points
    starts = []
    starts.append(initial_h.copy())
    np.random.seed(seed + 1)
    starts.append(initial_h + np.random.normal(0, 0.05, n_points))
    starts.append(initial_h + np.random.normal(0, 0.08, n_points))
    starts.append(initial_h + np.random.normal(0, 0.1, n_points))
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',

                constraints=constraints,

            )
            if result.success and result.fun < best_c5:

                best_h = result.x.copy()

            break
    
    elapsed = time.time() - start_time

        return best_h, best_c5, n_points
