import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()

    integral_target = n_points / 2

        if total == 0:

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h
    best_c5 = objective(initial_h)

        starts.append(h_start)

        result = minimize(
