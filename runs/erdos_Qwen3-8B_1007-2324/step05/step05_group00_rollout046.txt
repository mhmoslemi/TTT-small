def run(seed=42, budget_s=100, **kwargs):

                h[i] = 1.0
        return h

        h = h.copy()

        end = start + np.random.randint(5, 10)
        h[start:end+1] = 1.0 - h[start:end+1]
        current_sum = np.sum(h)
        delta = current_sum - integral_target
        if delta != 0:
            if delta > 0:
                idx = np.where(h == 1)[0][0]
            else:
                idx = np.where(h == 0)[0][0]
            h[idx] = 1.0 - h[idx]
        return h

    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
