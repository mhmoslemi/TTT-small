import numpy as np

import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Load initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    
    # Function to generate piecewise linear h from base_widths
    def generate_h(m, base_widths):
        h = np.zeros(n_points)
        start = 0.0
        for i in range(m):
            b = base_widths[i]
            end = start + b
            for idx in range(n_points):
                x = idx * dx
                if x < start or x >= end:
                    continue
                rel = x - start
                if rel < b / 2:
                    val = 2 * rel / b
                else:
                    val = 2 * (b - rel) / b
                h[idx] += val
            start = end
        h = np.clip(h, 0, 1)
        return h
    
    # Define objective function to minimize max overlap
    def obj_func(base_widths):
        m = len(base_widths)
        if m == 0:
            return 1000.0
        h = generate_h(m, base_widths)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Optimize for m=2, 3, 4
    for m in [2, 3, 4]:
        if m == 2:
            base_list = np.linspace(0.4, 1.6, 31)
            for b in base_list:
                base_widths = [b, 2 - b]
                current_c5 = obj_func(base_widths)
                if current_c5 < best_c5:

                    best_h = generate_h(m, base_widths)

        elif m == 4:
