        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_fine

    constraints_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_fine}]

    result_fine = minimize(objective_fine, fine_h_initial, method='SLSQP',
                           bounds=bounds_fine, constraints=constraints_fine,
                           options={'maxiter': 100, 'ftol': 1e-9, 'disp': False})

    if 'initial_h_values' in globals():
        initial_h_fine = np.array(initial_h_values)
        sum_initial = np.sum(initial_h_fine)
        initial_h_fine *= integral_target_fine / sum_initial
        result_initial = minimize(objective_fine, initial_h_fine, method='SLSQP',
                                  bounds=bounds_fine, constraints=constraints_fine,
                                  options={'maxiter': 100, 'ftol': 1e-9, 'disp': False})
        if result_initial.fun < result_fine.fun:
            result_fine = result_initial

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return result_fine.x, result_fine.fun, n_fine

    return result_fine.x, result_fine.fun, n_fine
