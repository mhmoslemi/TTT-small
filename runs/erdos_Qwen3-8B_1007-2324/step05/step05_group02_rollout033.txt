import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_fine = 280
    n_coarse = 160
    integral = n_fine / 2
    dx = 2.0 / n_fine

    if 'initial_h_values' in globals():
        parent_h = initial_h_values
    else:
        parent_h = np.full(n_fine, 0.5)

    def interpolate_and_scale(h_source, n):
        x_source = np.linspace(0, 2, len(h_source))
        x_target = np.linspace(0, 2, n)
        h_interpolated = np.interp(x_target, x_source, h_source)
        total = np.sum(h_interpolated)
        if total == 0:
            return np.zeros(n)
        scaling = (n / 2) / total
        return h_interpolated * scaling

    def objective(h, dx):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_coarse)]
    cons_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_coarse / 2}]

    current_h = interpolate_and_scale(parent_h, n_coarse)
    res_coarse = minimize(
        lambda h: objective(h, 2 / n_coarse),
        current_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons_coarse,
        options={'maxiter': 200, 'ftol': 1e-9}
    )

    best_h = parent_h.copy()
    best_c5 = objective(best_h, dx)

    if res_coarse.success:
        current_h = res_coarse.x
        c5 = objective(current_h, 2 / n_coarse)
        if c5 < best_c5:
            best_c5 = c5
            best_h = interpolate_and_scale(current_h, n_fine)

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
    cons_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral}]

    res_fine = minimize(
        lambda h: objective(h, dx),
        best_h,
        method='SLSQP',
        bounds=bounds_fine,
        constraints=cons_fine,
        options={'maxiter': 200, 'ftol': 1e-9}
    )

    if res_fine.success:
        best_h = res_fine.x
        best_c5 = res_fine.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_fine

    return best_h, best_c5, n_fine
