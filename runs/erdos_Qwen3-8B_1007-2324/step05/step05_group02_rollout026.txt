from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=100, **kwargs):

    n_fine = 280
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    integral_target = n_coarse / 2

    if 'initial_h_values' in globals():

        initial_h_fine = np.clip(initial_h_fine * (n_fine / 2) / np.sum(initial_h_fine), 0, 1)

    coarse_x = np.linspace(0, 2, n_coarse)

    def objective_coarse(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')

        penalty = 1e4 * abs(np.sum(h) - integral_target)
        return max_corr + penalty

    result_coarse = differential_evolution(

    fine_h_initial *= scale_factor
    fine_h_initial = np.clip(fine_h_initial, 0, 1)

    def objective_fine(h):

        return np.max(corr) * dx_fine
