    # Optimize for m=3 with L-BFGS-B
    result = minimize(
        lambda x: objective(x, 3),
        x0=[0.6, 0.6],
        bounds=[(0.4, 0.8), (0.4, 0.8)],
        method='L-BFGS-B',
        options={'maxiter': 100}
    )
    if result.success:
        h = generate_h_fast([result.x[0], result.x[1], 2 - result.x[0] - result.x[1]])
        c5 = result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Optimize for m=2 with L-BFGS-B
    result = minimize(
        lambda x: objective(x, 2),
        x0=[1.0],
        bounds=[(0.4, 0.8)],
        method='L-BFGS-B',
        options={'maxiter': 100}
    )
    if result.success:
        h = generate_h_fast([result.x[0], 2 - result.x[0]])
        c5 = result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5, n_points
