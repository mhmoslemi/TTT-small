    best_c5 = float('inf')
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_sum = np.sum(initial_h)
        if initial_sum > 0:
            initial_h *= (integral_target / initial_sum)
        # Adjust to fit smaller n_points
        if initial_h.shape[0] != n_points:
            initial_h = np.resize(initial_h, n_points)
            initial_h = np.clip(initial_h, 0, 1)
            initial_sum = np.sum(initial_h)
            if initial_sum > 0:
                initial_h *= (integral_target / initial_sum)
        population = [initial_h.copy() for _ in range(population_size)]
    else:
        # Generate random initial population
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h *= (integral_target / h_sum)
            population.append(h)
    
    # Evaluate initial population
    population_fitness = []
    for h in population:
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        c5 = np.max(corr) * dx
        sum_diff = np.abs(np.sum(h_clip) - integral_target)
        population_fitness.append(c5 + sum_diff * 1000)
    
    # Track best solution
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx]
    best_c5 = population_fitness[best_idx]
    
    # Main loop
    while generations < 100 and time.time() - start_time < budget_s:
        # Selection: tournament selection
        selected_parents = []
        for _ in range(population_size):
            candidates = np.random.choice(population, tournament_size)
            candidate_fitness = [population_fitness[i] for i in range(len(population)) if population[i] in candidates]
            best_candidate = candidates[np.argmin(candidate_fitness)]
            selected_parents.append(best_candidate)
        
        # Crossover
        children = []
        for i in range(0, population_size, 2):
            if i + 1 >= population_size:
                break
            parent1 = selected_parents[i]
            parent2 = selected_parents[i+1]
            child1 = np.copy(parent1)
            child2 = np.copy(parent2)
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    child1[j] = parent2[j]
                    child2[j] = parent1[j]
            children.append(child1)
            children.append(child2)
        
        # Mutate children
        mutated_children = []
        for child in children:
            mutated = np.copy(child)
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    perturbation = np.random.normal(0, 0.05, 1)
                    mutated[j] += perturbation[0]
                    mutated[j] = np.clip(mutated[j], 0, 1)
            mutated = repair(mutated, integral_target)
            mutated_children.append(mutated)
        
        # Evaluate children
        children_fitness = []
        for h in mutated_children:
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            c5 = np.max(corr) * dx
            sum_diff = np.abs(np.sum(h_clip) - integral_target)
            children_fitness.append(c5 + sum_diff * 1000)
        
        # Combine population and children
        new_population = population + mutated_children
        
        # Evaluate all in new_population
        all_fitness = []
        for h in new_population:
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            c5 = np.max(corr) * dx
            sum_diff = np.abs(np.sum(h_clip) - integral_target)
            all_fitness.append(c5 + sum_diff * 1000)
        
        # Select top survivors
        survivor_indices = np.argsort(all_fitness)[:population_size]
        population = [new_population[i] for i in survivor_indices]
        population_fitness = all_fitness[survivor_indices]
        
        # Update best solution
        current_best_idx = np.argmin(population_fitness)
        current_c5 = population_fitness[current_best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[current_best_idx]
        
        generations += 1
    
    # Final evaluation
    h_clip = np.clip(best_h, 0, 1)
    corr = np.correlate(h_clip, 1 - h_clip, mode='full')
    best_c5 = np.max(corr) * dx
    return best_h, best_c5, n_points

def repair(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-6:
        return h
    scale = target_sum / current_sum
    h_scaled = h * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    current_sum_scaled = np.sum(h_scaled)
    if abs(current_sum_scaled - target_sum) < 1e-6:
        return h_scaled
    diff = target_sum - current_sum_scaled
    idx = np.argmin(h_scaled)
    h_scaled[idx] += diff
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
