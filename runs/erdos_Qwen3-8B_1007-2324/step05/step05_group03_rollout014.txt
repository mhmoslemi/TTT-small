import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (integral_target / np.sum(initial_h))
        initial_h = np.clip(initial_h, 0, 1)
    
    pop_size = 500
    population = [initial_h.copy() for _ in range(pop_size)]
    
    def evaluate_fitness(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    tournament_size = 3
    crossover_prob = 0.7
    mutation_rate = 0.3
    generations = 50
    
    best_h = initial_h.copy()
    best_c5 = evaluate_fitness(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = [evaluate_fitness(ind) for ind in population]
        
        parents = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(population, size=tournament_size, replace=False)
            t_fitness = [evaluate_fitness(ind) for ind in tournament]
            winner = tournament[np.argmin(t_fitness)]
            parents.append(winner.copy())
        
        children = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1, p2 = parents[i], parents[i+1]
                if np.random.rand() < crossover_prob:
                    child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                    child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)
                else:
                    child1, child2 = p1.copy(), p2.copy()
                children.extend([child1, child2])
            else:
                children.append(parents[i].copy())
        
        for child in children:
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    delta = np.random.normal(0, 0.05)
                    child[i] += delta
                    child[i] = np.clip(child[i], 0, 1)
        
        for child in children:
            s = np.sum(child)
            if s > 0:
                child *= (integral_target / s)
        
        population += children
        combined = list(zip(population, fitness + [evaluate_fitness(c) for c in children]))
        combined.sort(key=lambda x: x[1])
        population = [ind for ind, _ in combined[:pop_size]]
        
        current_fitness = evaluate_fitness(population[0])
        if current_fitness < best_c5:
            best_c5 = current_fitness
            best_h = population[0].copy()
    
    final_fitness = evaluate_fitness(best_h)
    s = np.sum(best_h)
    if s > 0:
        best_h *= (integral_target / s)
    
    elapsed = time.time() - start_time
    return best_h, final_fitness, n_points
