def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_coarse = 80
    n_fine = 280
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    target_coarse = n_coarse / 2  # 40
    target_fine = n_fine / 2  # 140

    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        # Downsample initial_h_values (280 points) to coarse grid (80 points)
        coarse_initial = []
        step_size = 3.5
        for i in range(n_coarse):
            start = int(i * step_size)
            end = int((i+1) * step_size)
            # take average of initial_h_values[start:end]
            avg = np.mean(initial_h_values[start:end])
            coarse_initial.append(avg)
        coarse_initial = np.array(coarse_initial)
    else:
        # Use uniform initial guess for coarse grid
        coarse_initial = np.full(n_coarse, target_coarse / n_coarse)

    # Optimize on coarse grid
    # Define objective for coarse grid
    def objective_coarse(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_coarse

    bounds = [(0.0, 1.0) for _ in range(n_coarse)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_coarse}]
    result_coarse = minimize(
        objective_coarse,
        coarse_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    best_coarse = result_coarse.x
    best_c5_coarse = result_coarse.fun

    # Upsample best_coarse to fine grid
    fine_initial = []
    for v in best_coarse:
        fine_initial.extend([v] * 3)
    fine_initial = np.array(fine_initial)
    fine_initial *= (3.5 / 3)  # scale to get correct sum
    fine_initial = np.clip(fine_initial, 0, 1)

    # Optimize on fine grid
    def objective_fine(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_fine

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
    constraints_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_fine}]
    result_fine = minimize(

    dx_fine = 2.0 / n_fine
    target_coarse = n_coarse / 2
    target_fine = n_fine / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        coarse_initial = []
        step_size = 3.5
        for i in range(n_coarse):
            start = int(i * step_size)
            end = int((i + 1) * step_size)
            slice_end = min(end, len(initial_h))
            avg = np.mean(initial_h[start:slice_end])
            coarse_initial.append(avg)
        coarse_initial = np.array(coarse_initial)
    else:
        coarse_initial = np.full(n_coarse, target_coarse / n_coarse)

    def objective_coarse(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_coarse

    bounds_coarse = [(0.0, 1.0) for _ in range(n_coarse)]
    constraints_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_coarse}]
    result_coarse = minimize(
        objective_coarse,
        coarse_initial,
        method='SLSQP',
        bounds=bounds_coarse,
        constraints=constraints_coarse,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    best_coarse = result_coarse.x
    best_c5_coarse = result_coarse.fun

    def objective_fine(h):
        h_clip = np.clip(h, 0, 1)

        return np.max(corr) * dx_fine

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
