from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_variants(h0, seed):
        np.random.seed(seed)
        variants = [h0.copy()]
        for _ in range(10):
            noise = np.random.normal(0, 0.05, n_points)
            shifted = np.roll(h0 + noise, np.random.randint(-40, 40))
            shifted = np.clip(shifted, 0, 1)
            sum_shifted = np.sum(shifted)
            if sum_shifted > 0:
                shifted *= (integral_target / sum_shifted)
            variants.append(shifted)
        return variants

    starts = generate_variants(initial_h, seed)
    starts = [np.clip(s, 0, 1) for s in starts]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for i, start in enumerate(starts):
        if time.time() - start_time > budget_s:
            break
        try:
            result = minimize(

                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
