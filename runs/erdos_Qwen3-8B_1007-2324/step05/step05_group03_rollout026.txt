import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    for _ in range(num_starts):
        perturb = np.random.normal(0, 0.02, n_points)

        sum_perturbed = np.sum(start_perturbed)
        if sum_perturbed > 0:
            start_perturbed *= integral_target / sum_perturbed
        starts.append(start_perturbed)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
