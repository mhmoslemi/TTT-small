import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_watershed(h0, seed=None):
        if seed is not None:
            np.random.seed(seed)
        noise = np.random.normal(0, 0.02, n_points)
        shift = np.random.randint(-40, 40)
        shifted = np.roll(h0 + noise, shift)
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    starts = [initial_h]
    for i in range(15):
        starts.append(generate_watershed(initial_h, seed + i))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
