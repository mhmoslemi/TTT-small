import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    num_starts = 15
    starts = []

    np.random.seed(seed)
    noise = np.random.normal(0, 0.1, n_points)
    base = initial_h + noise
    base = np.clip(base, 0, 1)
    base_sum = np.sum(base)
    if base_sum > 0:
        base *= (integral_target / base_sum)
    starts.append(base)

    for i in range(num_starts - 1):
        np.random.seed(seed + i)
        perturb = np.random.normal(0, 0.05, n_points)
        start_perturbed = initial_h + perturb
        start_perturbed = np.clip(start_perturbed, 0, 1)
        sum_perturbed = np.sum(start_perturbed)
        if sum_perturbed > 0:
            start_perturbed *= (integral_target / sum_perturbed)
        starts.append(start_perturbed)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = None
    best_c5 = float('inf')
    max_iter = 1000
    iter_count = 0

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        iter_count += 1
        if iter_count >= max_iter:
            break

    if best_h is None:
        best_h = initial_h
        best_c5 = objective(initial_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
