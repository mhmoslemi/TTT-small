import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    def optimize(start):
        res = minimize(
            objective,
            start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
        )
        return res.x, res.fun
    
    best_h, best_c5 = initial_h.copy(), objective(initial_h)
    elapsed = time.time() - start_time
    
    # Single optimized run
    if elapsed < budget_s:
        h_opt, c5_opt = optimize(initial_h)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Perturbation-based refinement
    for _ in range(3):
        if elapsed >= budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        start_pert = initial_h + perturb
        start_pert = np.clip(start_pert, 0, 1)
        sum_pert = np.sum(start_pert)
        if sum_pert > 0:
            start_pert *= (integral_target / sum_pert)
        h_opt, c5_opt = optimize(start_pert)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        elapsed = time.time() - start_time
    
    return best_h, best_c5, n_points
