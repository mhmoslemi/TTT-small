import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_variations(h0):
        variations = []
        variations.append(h0)
        variations.append(np.roll(h0, 50))
        variations.append(np.roll(h0, -50))
        variations.append(1 - h0)
        variations.append(np.flip(h0))
        variations.append(h0 + np.random.normal(0, 0.01, n_points))
        variations.append(h0 + np.random.normal(0, 0.02, n_points))
        variations.append((h0 * np.random.rand(n_points)).clip(0, 1))
        variations.append(h0 * 0.9 + 0.05)
        variations.append(h0 * 1.1 - 0.05)
        for i in range(len(variations)):
            var = variations[i]
            sum_var = np.sum(var)
            if sum_var > 0:
                variations[i] = var * (integral_target / sum_var)
        return variations

    starts = generate_variations(initial_h)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='COBYLA',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
