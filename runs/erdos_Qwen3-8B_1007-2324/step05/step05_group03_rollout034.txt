        initial_h = np.full(n_points, 0.5)
    
    def repair(individual):
        s = np.sum(individual)
        if s == 0:
            return np.zeros_like(individual)
        return individual * (integral_target / s)
    
    population = np.zeros((500, n_points))
    population[0] = repair(initial_h)
    for i in range(1, 500):
        ind = np.random.uniform(0, 1, n_points)
        population[i] = repair(ind)
    
    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    for _ in range(100):
        fitness = np.array([evaluate(ind) for ind in population])
        parents = np.array([population[np.random.choice(500, 3, replace=False)][np.argmin(fitness[np.random.choice(500, 3, replace=False)])] for _ in range(500)])
        
        offspring = np.zeros_like(population)
        for i in range(0, 500, 2):
            if np.random.rand() < 0.7:
                p1, p2 = parents[i], parents[i+1]
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        offspring[i, j] = p1[j]
                        offspring[i+1, j] = p2[j]
                    else:
                        offspring[i, j] = p2[j]
                        offspring[i+1, j] = p1[j]
            else:
                offspring[i] = parents[i]
                offspring[i+1] = parents[i+1]
        
        for i in range(500):
            for j in range(n_points):
                if np.random.rand() < 0.3:
                    mutation = np.random.normal(0, 0.1)
                    offspring[i, j] += mutation
                    offspring[i, j] = np.clip(offspring[i, j], 0, 1)
            offspring[i] = repair(offspring[i])
        
        combined = np.vstack((population, offspring))
        combined_fitness = np.array([evaluate(ind) for ind in combined])
        indices = np.argsort(combined_fitness)[:500]
        population = combined[indices]
        
        current_best = population[0]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    final_h = repair(best_h)
    final_c5 = evaluate(final_h)
    return final_h, final_c5, n_points
