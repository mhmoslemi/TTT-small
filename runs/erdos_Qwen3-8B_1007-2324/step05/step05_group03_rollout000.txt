import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    pop_size = 500
    tournament_size = 3
    crossover_prob = 0.7
    mutation_prob = 0.3
    
    def evaluate_fitness(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    def repair_sum(h):
        total = np.sum(h)
        if abs(total - integral_target) < 1e-5:
            return h
        scale = integral_target / total
        return np.clip(h * scale, 0, 1)
    
    def mutate(h, eta=0.5):
        h_mut = h.copy()
        for i in range(len(h_mut)):
            if np.random.rand() < mutation_prob:
                delta = np.random.uniform(-1, 1) * (1 - 2 * np.random.rand()) ** eta
                h_mut[i] += delta
                h_mut[i] = np.clip(h_mut[i], 0, 1)
        return repair_sum(h_mut)
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if np.random.rand() < crossover_prob:
                child[i] = parent2[i]
        return repair_sum(child)
    
    def tournament_selection(population):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best = min(population[indices], key=evaluate_fitness)
        return best
    
    def evolve(population):
        new_pop = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            new_pop.append(mutate(child1))
            new_pop.append(mutate(child2))
        return new_pop
    
    population = [initial_h.copy() for _ in range(pop_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate_fitness(initial_h)
    
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        population = evolve(population)
        for h in population:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:

    return best_h, best_c5, n_points
