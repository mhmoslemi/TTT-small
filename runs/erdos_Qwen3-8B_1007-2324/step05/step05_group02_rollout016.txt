from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()

    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    np.random.seed(seed)
    num_starts = 50
    starts = []
    
    # Base perturbation: noise + scaling + projection
    noise = np.random.normal(0, 0.15, n_points)
    parent_with_noise = initial_h + noise
    parent_with_noise = np.clip(parent_with_noise, 0, 1)
    sum_val = np.sum(parent_with_noise)
    if sum_val > 0:
        parent_with_noise *= (integral_target / sum_val)
    starts.append(parent_with_noise)
    
    # Add diversity via structured perturbations
    for i in range(num_starts - 1):
        np.random.seed(seed + i)
        perturb = np.random.normal(0, 0.05, n_points)
        start_perturbed = initial_h + perturb
        start_perturbed = np.clip(start_perturbed, 0, 1)
        sum_val = np.sum(start_perturbed)
        if sum_val > 0:
            start_perturbed *= (integral_target / sum_val)
        starts.append(start_perturbed)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    best_time = 0
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x.copy()
                best_time = time.time() - start_time
        except:
            pass
        
        if best_time >= budget_s:
            break
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
