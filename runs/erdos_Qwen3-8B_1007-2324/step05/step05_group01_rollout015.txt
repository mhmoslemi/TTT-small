import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_h(w1, w2):
        w3 = 2.0 - w1 - w2
        if w3 <= 0:
            return None
        h = np.zeros(n_points)
        start = 0.0
        for i in range(3):
            width = [w1, w2, w3][i]
            end = start + width
            for idx in range(n_points):
                x = idx * dx
                if x < start or x >= end:
                    continue
                rel = x - start
                if rel < width / 2:
                    val = 2 * rel / width
                else:
                    val = 2 * (width - rel) / width
                h[idx] += val
            start = end
        return h

    def objective(params):
        w1, w2 = params
        h = generate_h(w1, w2)
        if h is None:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0.4, 0.8), (0.4, 0.8)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=100, popsize=15, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, seed=seed
    )
    best_params = result.x
    best_h = generate_h(best_params[0], best_params[1])
    best_c5 = objective(best_params)

    if best_h is None:
        best_h = initial_h
        best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    return best_h, best_c5, n_points
