            w1_opt, w2_opt, w3_opt = result.x
            base_widths_opt = [w1_opt, w2_opt, w3_opt]
            h = generate_h(3, base_widths_opt)
            corr = np.correlate(h, 1 - h, mode='full')
            c5_opt = np.max(corr) * dx
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h.copy()
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
