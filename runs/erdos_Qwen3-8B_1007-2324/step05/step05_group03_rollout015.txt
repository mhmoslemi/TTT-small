import numpy as np
import time
from numpy.random import rand, normal, choice

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    pop_size = 500
    max_generations = 100
    mutation_prob = 0.3
    crossover_prob = 0.7
    tournament_size = 3

    def evaluate_fitness(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h * (integral_target / np.sum(initial_h))
        population = [initial_h + normal(0, 0.05, n_points) for _ in range(pop_size)]
        for i in range(pop_size):
            population[i] = population[i] * (integral_target / np.sum(population[i]))
    else:
        population = [rand(n_points) * (integral_target / np.sum(rand(n_points))) for _ in range(pop_size)]

    best_h = population[0]
    best_fitness = evaluate_fitness(best_h)

    for _ in range(max_generations):
        fitness = [evaluate_fitness(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_fitness:
            best_fitness = fitness[best_idx]
            best_h = population[best_idx]
        if time.time() - start_time > budget_s:
            break

        new_population = []
        for _ in range(pop_size):
            tournament = choice(range(pop_size), size=tournament_size, replace=False)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            winner = population[winner_idx]

            if np.random.rand() < crossover_prob:
                other_idx = choice(range(pop_size), size=1, replace=False)[0]
                other = population[other_idx]
                child = winner * 0.5 + other * 0.5
            else:
                child = winner

            if np.random.rand() < mutation_prob:
                for i in range(n_points):
                    child[i] += normal(0, 0.02)
                    child[i] = np.clip(child[i], 0, 1)
            sum_child = np.sum(child)
            child = child * (integral_target / sum_child) if sum_child != 0 else child
            new_population.append(child)

        population = new_population

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    best_h = best_h * (integral_target / sum_best) if sum_best != 0 else best_h
    elapsed = time.time() - start_time
    return best_h, best_fitness, n_points
