import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr) * dx
        sum_h = np.sum(h_clip)
        penalty = 1e4 * (sum_h - integral_target)**2
        return max_corr + penalty

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)

    starts = [initial_h]
    for i in range(15):
        noise = np.random.normal(0, 0.03, n_points)
        shifted = np.roll(initial_h + noise, np.random.randint(-40, 40))
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        starts.append(shifted)

    for start in starts:
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=15,
                tol=1e-8,
                mutation=(0.5, 1),
                recombination=0.8,
                seed=seed,
                maxiter=100
            )
            current_h = result.x
            current_c5 = evaluate(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
