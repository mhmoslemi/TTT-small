import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        scale = integral_target / np.sum(initial_h)
        initial_h = initial_h * scale
    else:
        initial_h = np.full(n_points, 0.5)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr)
        penalty = 1e3 * abs(np.sum(h) - integral_target)
        return max_corr * dx + penalty
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=30,
        tol=1e-6,
        mutation=(0.4, 0.9),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False,
        vectorized=True
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
