from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    num_starts = 8
    starts = []
    
    np.random.seed(seed)
    noise = np.random.normal(0, 0.05, n_points)
    parent_with_noise = initial_h + noise
    parent_with_noise = np.clip(parent_with_noise, 0, 1)
    sum_parent = np.sum(parent_with_noise)
    if sum_parent > 0:
        parent_with_noise *= (integral_target / sum_parent)
    starts.append(parent_with_noise)
    
    for i in range(num_starts - 1):
        np.random.seed(seed + i)
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        start_perturbed = initial_h + perturb
        start_perturbed = np.clip(start_perturbed, 0, 1)
        sum_perturbed = np.sum(start_perturbed)
        if sum_perturbed > 0:
            start_perturbed *= (integral_target / sum_perturbed)
        starts.append(start_perturbed)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        h_fft = np.fft.fft(h_clip)
        one_minus_h = 1 - h_clip
        one_minus_h_fft = np.fft.fft(one_minus_h)
        corr_fft = np.fft.ifft(h_fft * np.conj(one_minus_h_fft))
        corr = np.real(corr_fft) * dx
        return np.max(corr)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = None
    best_c5 = float('inf')
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
