        return max_corr * dx_fine - 0.0001 * std_corr
    
    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]

    # Coarse grid optimization
    result_coarse = differential_evolution(

    )

    # Interpolate to fine grid

        x_coarse = np.linspace(0, 2, n_coarse)

        h_fine = np.clip(h_fine, 0, 1)
        scale = integral_target / np.sum(h_fine)
