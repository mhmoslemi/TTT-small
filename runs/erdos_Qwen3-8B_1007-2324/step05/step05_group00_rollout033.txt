import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1

    def evaluate_c5(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    def objective(h):
        return evaluate_c5(h)

    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)

    def generate_candidates(base_h):
        candidates = [base_h.copy()]
        for _ in range(5):
            h = base_h.copy()
            idx = np.random.choice(n_points, 20, replace=False)
            h[idx] = 1 - h[idx]
            candidates.append(h)
        return candidates

    def perturb_and_optimize(base_h):
        candidates = generate_candidates(base_h)
        for h in candidates:
            h_float = h.astype(float)
            result = minimize(
                objective,
                h_float,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    def refine_candidate(h):
        result = minimize(
            objective,
            h.astype(float),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 400, 'ftol': 1e-9, 'eps': 1e-8}
        )
        if result.success:
            return result.x
        return h

    perturb_and_optimize(h_initial)
    best_h = refine_candidate(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:

    return best_h, best_c5, n_points
