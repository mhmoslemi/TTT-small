            start *= (integral_fine / sum_val)
        try:
            result = minimize(
                objective_fine,
                start,
                method='SLSQP',
                bounds=bounds_fine,
                constraints=constraints_fine,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            if result.success and result.fun < best_c5_fine:
                best_c5_fine = result.fun
                best_fine = result.x.copy()
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_fine, best_c5_fine, n_fine
    
    return best_fine, best_c5_fine, n_fine
