import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    population_size = 40
    population = []
    if 'initial_h_values' in globals():

        initial_h = initial_h[:n_half]

    for _ in range(population_size - 1):

        total = np.sum(h_clipped)

            return np.copy(h)
