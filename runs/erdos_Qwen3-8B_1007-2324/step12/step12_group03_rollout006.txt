import time

def run(seed=42, budget_s=100, **kwargs):

    if 'initial_h_values' in globals():

    pop_size = 30
    population = np.zeros((pop_size, n_half))
    population[0] = parent[:n_half]
    for i in range(1, pop_size):
        population[i] = np.clip(parent[:n_half] + np.random.normal(0, 0.05, n_half), 0, 1)
    
    def repair_constrain(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (integral_target / s), 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    # GA parameters
    gen = 0
    max_gen = 20
    crossover_rate = 0.7
    mutation_rate = 0.1
    tournament_size = 5
    
    while time.time() - start_time < budget_s - 10 and gen < max_gen:
        gen += 1
        indices = np.random.choice(pop_size, tournament_size*2, replace=False)
        candidates = population[indices]
        fits = np.array([compute_overlap(c) for c in candidates])
        selected = candidates[np.argsort(fits)][:2]
        
        if np.random.rand() < crossover_rate:
            p1, p2 = selected

            child2 = np.concatenate([p2[:split], p1[split:]])

            child1, child2 = selected
