        for (c5, h) in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    final_h = best_h
    final_h = repair(final_h)
    final_c5 = compute_overlap(final_h)

    full_h = np.concatenate([final_h, final_h[::-1]])
    return full_h, final_c5, n_points
