    h_opt = h_initial_repaired[:n_points//2]

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    start_time = time.time()

    # Perform multiple perturbations
    for _ in range(10):
        # Perturb the current best solution
        perturb = np.random.normal(0, 0.1, size=h_opt.shape)
        perturbed = np.clip(h_opt + perturb, 0.0, 1.0)
        perturbed_repaired = repair(perturbed, integral_target)
        
        # SLSQP optimization
        bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })
        result = scipy.optimize.minimize(
            objective,
            perturbed_repaired,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-10,
                'maxiter': 150,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
