import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200  # Reduced for faster convergence with DE
    half_length = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:half_length]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:half_length]

    def repair_h(h):
        current_sum = np.sum(h)
        scaling = integral_target / current_sum if current_sum != 0 else 0
        return np.clip(h * scaling, 0, 1)

    def objective(h):
        h_repaired = repair_h(h)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_length)]

    # Differential evolution with tighter tolerance and fewer iterations for speed
    result = differential_evolution(
