import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]
    
    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(h_opt):
        return compute_overlap(h_opt)
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    start_time = time.time()
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1',
        mutation=(0.7, 0.3),
        crossover=0.9,
        popsize=20,
        maxiter=50,
        tol=1e-8,
        workers=1
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        best_h_opt = h_opt
        best_c5 = compute_overlap(h_opt)
    else:
        best_h_opt = result.x
        best_c5 = result.fun
    
    final_h = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return final_h, best_c5, n_points
