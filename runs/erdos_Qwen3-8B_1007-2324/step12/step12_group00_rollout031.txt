import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    global initial_h_values
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():

        h_opt = h_initial[:n_half]

        if current_sum == 0:

        scaling_factor = integral_target / current_sum
