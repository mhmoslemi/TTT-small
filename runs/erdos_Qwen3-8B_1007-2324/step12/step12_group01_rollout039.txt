                parent1 = parents[i % population_size]
                parent2 = parents[(i + 1) % population_size]
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = np.clip(child, 0, 1)
            else:
                child = parents[i % population_size].copy()

            if np.random.rand() < mutation_rate:
                mu = 0.02
                sigma = 0.03
                noise = np.random.normal(0, sigma, n_points//2)
                child += noise
                child = np.clip(child, 0, 1)

            child_repaired = repair_parent(child, dx)
            new_population.append(child_repaired)

        population = new_population

        current_fitness = [compute_c5(h, dx) for h in population]
        best_idx = np.argmin(current_fitness)
        current_c5 = current_fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx]

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points

def repair_parent(h_candidate, dx):
    current_sum = np.sum(h_candidate) * dx
    if abs(current_sum) < 1e-6:
        return np.zeros_like(h_candidate)
    scaling = 1.0 / current_sum
    h_repaired = np.clip(h_candidate * scaling, 0, 1)
    return h_repaired

def compute_c5(h_vec, dx):
    h_repaired = repair_parent(h_vec, dx)
    h_full = np.concatenate([h_repaired, h_repaired[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
