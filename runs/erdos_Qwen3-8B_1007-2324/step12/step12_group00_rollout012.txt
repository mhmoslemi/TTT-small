import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]
    
    def repair(h, target):
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scale = target / np.sum(h)
        scaled = h * scale
        clipped = np.clip(scaled, 0, 1)
        masked = np.where(clipped < 1, clipped, 1)
        residual = target - np.sum(masked)
        if residual != 0:
            idx = np.where(clipped >= 1)[0]
            if idx.size > 0:
                masked[idx] = np.minimum(1, masked[idx] + residual / idx.size)
        return masked
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = x.copy()
        repaired = repair(h, integral_target)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    elapsed = 0
    start_time = time.time()
    
    def optimize_start(h):
        result = scipy.optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                                        options={'ftol': 1e-10, 'maxiter': 200})
        return result.x
    
    best_h = optimize_start(best_h)
    best_c5 = compute_overlap(best_h)
    elapsed = time.time() - start_time
