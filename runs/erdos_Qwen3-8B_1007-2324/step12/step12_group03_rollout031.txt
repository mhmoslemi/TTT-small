import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    np.random.seed(seed)
    pop_size = 40
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points//2]
    else:
        x = np.linspace(0, 1, n_points//2)
        h_opt = x

    population = [np.copy(h_opt) for _ in range(pop_size)]
    for i in range(pop_size):
        noise = np.random.normal(0, 0.05, n_points//2)
        population[i] += noise
        population[i] = np.clip(population[i], 0, 1)
        current_sum = np.sum(population[i])
        population[i] *= target_sum / current_sum

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        return h_clipped * (target_sum / current_sum)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)

    F = 0.8
    max_generations = 1000
    generation = 0

    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            r1, r2, r3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = r1 + F * (r2 - r3)
            trial = np.where(np.random.rand(n_points//2) < 0.5, mutant, population[i])
            trial = repair(trial)
            trial_c5 = compute_overlap(trial)
            if trial_c5 < compute_overlap(population[i]):
                population[i] = trial
                if trial_c5 < best_c5:
                    best_h = trial.copy()
                    best_c5 = trial_c5

    full_h = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return full_h, best_c5, n_points
