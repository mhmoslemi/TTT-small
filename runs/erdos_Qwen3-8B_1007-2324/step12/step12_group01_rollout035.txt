        current_global = np.min(personal_best_c5)
        if current_global < global_best_c5:
            global_best_c5 = current_global
            global_best = personal_best[np.argmin(personal_best_c5)]
        if np.random.rand() < 0.1:
            for i in range(population_size):
                if np.random.rand() < 0.05:
                    particles[i] = np.random.rand(n_points//2)
                    particles[i] = repair(particles[i])
                    c5 = compute_C5(particles[i])
                    if c5 < personal_best_c5[i]:
                        personal_best[i] = particles[i].copy()
                        personal_best_c5[i] = c5

    best_h = repair(global_best)
    best_c5 = compute_C5(best_h)
    return (best_h, best_c5, n_points)
