        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt_initial = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]

    population_size = 30
    population = [h_opt_initial + np.random.randn(n_half) * 0.05 for _ in range(population_size)]
    population = [np.clip(np.concatenate([h, h[::-1]]), 0, 1) for h in population]
    population = [h[:n_half] for h in population]
    population = [np.clip(h, 0, 1) for h in population]
    population = [h * (integral_target / np.sum(h)) if np.sum(h) > 0 else np.zeros_like(h) for h in population]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    fitness = [compute_overlap(h) for h in population]

    def repair_constrain(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (target / s), 0, 1)

    def sa(h, temp_init, cooling_rate, iters):
        h_current = h.copy()
        c5_current = compute_overlap(h_current)
        temp = temp_init
        for _ in range(iters):
            noise = np.random.normal(0, 0.04 * temp, n_half)
            h_candidate = np.clip(h_current + noise, 0, 1)
            h_candidate = repair_constrain(h_candidate, integral_target)
            c5_candidate = compute_overlap(h_candidate)
            if c5_candidate < c5_current or np.random.rand() < np.exp((c5_current - c5_candidate) / temp):
                h_current = h_candidate
                c5_current = c5_candidate
            temp *= cooling_rate
        return h_current, c5_current

    generations = 20
    crossover_rate = 0.7
    mutation_rate = 0.1
    tournament_size = 5

    for _ in range(generations):
        selected = np.random.choice(len(population), tournament_size, replace=False)
        selected_fitness = [fitness[i] for i in selected]
        best_idx = selected[np.argmin(selected_fitness)]
        parent1 = population[best_idx]
        selected2 = np.random.choice(len(population), tournament_size, replace=False)
        selected_fitness2 = [fitness[i] for i in selected2]
        best_idx2 = selected2[np.argmin(selected_fitness2)]
        parent2 = population[best_idx2]

        if np.random.rand() < crossover_rate:
            split = np.random.randint(1, n_half)
            child1 = np.concatenate([parent1[:split], parent2[split:]])
            child2 = np.concatenate([parent2[:split], parent1[split:]])
        else:
            child1, child2 = parent1.copy(), parent2.copy()

        if np.random.rand() < mutation_rate:
            child1 += np.random.randn(n_half) * 0.05
            child1 = np.clip(child1, 0, 1)
        if np.random.rand() < mutation_rate:
            child2 += np.random.randn(n_half) * 0.05
            child2 = np.clip(child2, 0, 1)

        child1 = repair_constrain(child1, integral_target)
        child2 = repair_constrain(child2, integral_target)

        new_fitness1 = compute_overlap(child1)
        new_fitness2 = compute_overlap(child2)

        worst_idx = np.argmax(fitness)
        population[worst_idx] = child1
        fitness[worst_idx] = new_fitness1

        worst_idx2 = np.argmax(fitness)
        population[worst_idx2] = child2
        fitness[worst_idx2] = new_fitness2

    best_idx = np.argmin(fitness)
    best_h_opt = population[best_idx]
    best_c5 = fitness[best_idx]

    sa_h1, sa_c51 = sa(best_h_opt, 1.0, 0.997, 1000)
    sa_h2, sa_c52 = sa(best_h_opt, 1.0, 0.995, 1000)
    best_h = sa_h1 if sa_c51 < sa_c52 else sa_h2
    best_c5 = min(best_c5, sa_c51, sa_c52)

    h_full = np.concatenate([best_h, best_h[::-1]])
    h_full = np.clip(h_full, 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
