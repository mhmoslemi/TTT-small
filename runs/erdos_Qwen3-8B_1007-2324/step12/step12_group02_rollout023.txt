import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt_initial = initial_h[:n_half]
    else:
        h_opt_initial = np.full(n_half, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def objective(candidate):
        current_sum = np.sum(candidate)
        if current_sum == 0:
            return float('inf')
        scaling = (n_points / 4) / current_sum
        scaled = candidate * scaling
        h_full = np.concatenate([scaled, scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = differential_evolution(
        objective,
        bounds,
        x0=h_opt_initial,
        popsize=30,
        mutation=0.5,
        recombination=0.9,
        strategy='best1bin',
        maxiter=300,
        tol=1e-6,
        seed=seed
    )

    best_h_opt = result.x
    current_sum = np.sum(best_h_opt)
    if current_sum == 0:
        scaled_h_opt = np.zeros_like(best_h_opt)
    else:
        scaling = (n_points / 4) / current_sum
        scaled_h_opt = best_h_opt * scaling
    h_full = np.concatenate([scaled_h_opt, scaled_h_opt[::-1]])

    elapsed = time.time() - start_time
    return h_full, result.fun, n_points
