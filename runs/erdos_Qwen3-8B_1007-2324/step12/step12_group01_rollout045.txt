def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(h):
        current_sum = np.sum(h)
        if current_sum <= 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_initial = initial_h[:n_points//2]
        h_initial_repaired = repair(h_initial)
    else:
        h_initial_repaired = np.random.rand(n_points//2) * 0.5
        h_initial_repaired = repair(h_initial_repaired)

    pop_size = 50
    particles = []
    velocities = []
    for _ in range(pop_size):
        pos = h_initial_repaired + np.random.normal(0, 0.1, n_points//2)
        pos = np.clip(pos, 0, 1)
        pos = repair(pos)
        vel = np.random.uniform(-0.1, 0.1, n_points//2)
        particles.append(pos)
        velocities.append(vel)

    def compute_C5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    personal_best_positions = [p.copy() for p in particles]
    personal_best_fitness = [compute_C5(p) for p in particles]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = particles[global_best_idx].copy()
    best_c5 = personal_best_fitness[global_best_idx]

    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 1000

    iter_count = 0

        iter_count += 1
        time_left = budget_s - (time.time() - start_time)

            vel = velocities[i] * omega + c1 * r1 * (personal_best_positions[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            velocities[i] = vel
