import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points//2]

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        s = np.sum(h)
        if s <= 0:
            return np.zeros_like(h)
        scaling = integral_target / s
        return np.clip(h * scaling, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 1.0
    cooling_rate = 0.998
    iter_count = 0
    max_iters = 15000

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        noise_std = 0.04 * current_temp
        noise = np.random.normal(0, noise_std, n_points//2)
        candidate = np.clip(h_opt + noise, 0, 1)
        candidate = repair(candidate)
        candidate_c5 = compute_overlap(candidate)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / current_temp):
            best_h = candidate
            best_c5 = candidate_c5
        current_temp *= cooling_rate

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
