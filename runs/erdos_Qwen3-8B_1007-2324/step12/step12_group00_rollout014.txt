import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (integral_target / s), 0, 1)

    def objective(x):
        h = repair(x)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in range(n_half)]

    start_time = time.time()
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        mutation=(0.7, 1),
        recombination=0.9,
        popsize=20,
        maxiter=100,
        seed=seed,
        tol=1e-6,
        disp=False
    )
    elapsed = time.time() - start_time

    best_x = result.x
    best_h = repair(best_x)
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, objective(best_x), n_points
