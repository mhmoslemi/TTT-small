def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h_opt = initial_h[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = np.concatenate([h[:n_points//2], h[:n_points//2][::-1]])[:n_points//2]

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 1.0
    cooling_rate = 0.995
