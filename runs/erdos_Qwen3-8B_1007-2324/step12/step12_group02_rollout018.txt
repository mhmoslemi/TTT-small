import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # ensures sum(h_full) * dx == 1

    # Initialize with provided or default starting point
    if 'initial_h_values' in globals():
        h_opt_initial = np.array(initial_h_values)[:n_half]

        h_opt_initial = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    # Repair function to maintain integral constraint and bounds

        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)

    # Compute maximum overlap for a given h_opt
    def objective(h_opt):

        h_full = np.concatenate([h_opt_repaired, h_opt_repaired[::-1]])
