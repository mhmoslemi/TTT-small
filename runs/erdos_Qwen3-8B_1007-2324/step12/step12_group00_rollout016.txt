import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) = integral_target

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_points//2]

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        repaired = repair(h)
        return compute_c5(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]
    constraint_eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    # Generate perturbations
    np.random.seed(seed)
    candidates = [h_opt.copy()]
    for _ in range(5):
        noise = np.random.normal(0, 0.1, size=h_opt.shape)
        perturbed = h_opt + noise
        repaired_perturbed = repair(perturbed)
        candidates.append(repaired_perturbed)

    for candidate in candidates:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        result = scipy.optimize.minimize(
            objective,
            candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=[constraint_eq],
            options={
                'ftol': 1e-10,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(repair(current_h))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                elapsed = time.time() - start_time
                if elapsed > budget_s - 5:
                    break

    h_values = np.concatenate([best_h, best_h[::-1]])
    return h_values, best_c5, n_points
