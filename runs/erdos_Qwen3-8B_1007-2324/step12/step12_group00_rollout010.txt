import numpy as np

from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():

        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s

    def objective(x):
        h_opt = repair(x)
        return compute_overlap(h_opt)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    result = differential_evolution(
        objective,

        x0=h_opt_initial
