import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points//2]

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum <= 0:
            return np.clip(h * 1e1, 0, 1)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    def compute_C5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_pop = repair(h_opt + np.random.normal(0, 0.05, n_points//2), integral_target)
    velocities = np.random.uniform(-0.1, 0.1, (50, n_points//2))
    personal_best = h_pop.copy()
    global_best = h_pop.copy()
    best_c5 = compute_C5(global_best)
    omega, c_p, c_g = 0.7, 1.5, 1.5
    iter_count = 0
    max_iter = 10000
    time_start = time.time()

    while time.time() - time_start < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        for i in range(50):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c_p * r1 * (personal_best[i] - h_pop[i]) + c_g * r2 * (global_best - h_pop[i])
            h_pop[i] = np.clip(h_pop[i] + velocities[i], 0, 1)
            h_pop[i] = repair(h_pop[i], integral_target)
            current_c5 = compute_C5(h_pop[i])
            if current_c5 < compute_C5(personal_best[i]):
                personal_best[i] = h_pop[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                global_best = h_pop[i].copy()
        if iter_count % 100 == 0:
            h_pop[np.random.choice(50, 10)] = repair(h_opt + np.random.normal(0, 0.05, n_points//2), integral_target)

    h_full = np.clip(np.concatenate([global_best, global_best[::-1]]), 0, 1)
    return h_full, best_c5, n_points
