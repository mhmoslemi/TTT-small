import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 180
    n_half = n_points // 2
    dx = 2.0 / n_points

        parent = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        parent = np.clip(1 - np.abs(x - 1), 0, 1)
        parent = np.concatenate([parent[:n_half], parent[:n_half][::-1]])[:n_half]

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        return np.clip(h * scale, 0, 1)

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population = [repair(parent + np.random.normal(0, 0.05, n_half)) for _ in range(30)]
    fitness = [compute_overlap(ind) for ind in population]

            c1 = np.concatenate([p1[:split], p2[split:]])
            c2 = np.concatenate([p2[:split], p1[split:]])

        c1 = repair(c1 + np.random.normal(0, 0.1, n_half))

    temp = 1.0
