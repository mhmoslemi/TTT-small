import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # sum(h) = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scaling = integral_target / sum_h
        return h * scaling

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_initial = initial_h[:n_half]
    else:
        np.random.seed(seed)
        h_initial = np.random.uniform(0, 1, n_half)
        h_initial = repair(h_initial)

    population_size = 40
    population = [h_initial.copy() for _ in range(population_size)]
    for i in range(population_size):
        noise = np.random.normal(0, 0.04, n_half)
        population[i] += noise
        population[i] = repair(population[i])

    F = 0.8
    pc = 0.9
    max_generations = 1000
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    generation = 0

    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        fitness = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        new_population = []

    elapsed = time.time() - start_time
    return best_h, final_c5, n_points
