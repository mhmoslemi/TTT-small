import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize
    
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]
    
    def repair(h, target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    h_opt = repair(h_opt, integral_target)
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    
    perturbation_scale = 0.05
    max_trials = 200
    
    start_time = time.time()
    
    for _ in range(max_trials):
        perturbed = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=best_h.shape)
        perturbed = repair(perturbed, integral_target)
        
        bounds = [(0.0, 1.0) for _ in range(len(perturbed))]
        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })
        
        result = scipy.optimize.minimize(
            lambda x: compute_overlap(x),
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-8,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                if time.time() - start_time > budget_s - 1:
                    break
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
