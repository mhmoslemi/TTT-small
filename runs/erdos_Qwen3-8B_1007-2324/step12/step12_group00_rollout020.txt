import numpy as np

import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 160
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) >= n_points:
            h_opt = h_initial[:n_points]
        else:
            h_opt = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        h_opt = np.random.uniform(0, 1, n_points)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })
    
    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-10,
            'maxiter': 800,
            'disp': False,
            'eps': 1e-8
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([h_opt, h_opt[::-1]]), compute_overlap(h_opt), n_points
    
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
