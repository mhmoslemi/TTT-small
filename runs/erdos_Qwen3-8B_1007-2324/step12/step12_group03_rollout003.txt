import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4
    max_generations = 50
    pop_size = 50
    elite_frac = 0.2
    tournament_size = 5
    mut_rate = 0.1
    cross_rate = 0.7

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_initial = initial_h[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = np.clip(1 - np.abs(x - 1), 0, 1)
        h_initial = np.concatenate([h_initial[:n_points//2], h_initial[:n_points//2][::-1]])[:n_points//2]

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h)
        scale = integral_target / total
        return np.clip(h * scale, 0, 1)

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        return compute_c5(h)

    population = [np.copy(h_initial) + np.random.normal(0, 0.05, n_points//2) for _ in range(pop_size-1)]
    population.append(h_initial)
    population = [repair(p) for p in population]
    best_h = population[0]
    best_c5 = fitness(best_h)

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        fitnesses = [fitness(ind) for ind in population]
        elite_idx = np.argsort(fitnesses)[:int(elite_frac * pop_size)]
        elites = [population[i] for i in elite_idx]
        new_population = elites
        while len(new_population) < pop_size:
            a, b = np.random.choice(population, size=2, replace=False)
            if np.random.rand() < cross_rate:
                split = np.random.randint(1, len(a)-1)
                child = np.concatenate([a[:split], b[split:]])
            else:
                child = np.copy(a)
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.05, len(child))
                child += noise
            child = repair(child)
            new_population.append(child)
        population = new_population
        curr_best = min(population, key=fitness)
        curr_c5 = fitness(curr_best)
        if curr_c5 < best_c5:
            best_h = curr_best
            best_c5 = curr_c5

    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
