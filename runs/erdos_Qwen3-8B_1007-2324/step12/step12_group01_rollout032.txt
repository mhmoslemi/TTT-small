        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points//2]
    
    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    
    def generate_candidates(h_opt, noise_std):
        candidates = []
        for _ in range(5):
            noise = np.random.normal(0, noise_std, h_opt.shape)
            candidate = np.clip(h_opt + noise, 0, 1)
            candidate = repair_h(candidate, integral_target)
            candidates.append(candidate)
        return candidates
    
    def evaluate_candidates(candidates):
        results = []
        for cand in candidates:
            c5 = compute_overlap(cand)
            results.append((c5, cand))
        return results
    
    best_c5 = compute_overlap(best_h)
    time_start = time.time()

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
