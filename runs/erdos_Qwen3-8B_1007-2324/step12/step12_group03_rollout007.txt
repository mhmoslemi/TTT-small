import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return h.copy()
        scale_factor = integral_target / total
        h_scaled = np.clip(h * scale_factor, 0, 1)
        return h_scaled

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    pop_size = 40
    pop = [repair(np.random.rand(n_half)) for _ in range(pop_size - 1)]
    pop.append(best_h)
    pop = np.array(pop)

    def fitness(h):
        return compute_overlap(h)

    def tournament_selection():
        indices = np.random.choice(len(pop), 5, replace=False)
        return pop[indices[np.argmin([fitness(p) for p in pop[indices]])]]

    start_time = time.time()
    max_iter = 50
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        new_pop = []
        for _ in range(pop_size // 2):
            p1 = tournament_selection()
            p2 = tournament_selection()
            child = np.zeros_like(p1)
            split = np.random.randint(1, n_half - 1)
            child[:split] = p1[:split]
            child[split:] = p2[split:]
            child = repair(child)
            new_pop.append(child)
        pop = np.array(new_pop)
        pop[0] = best_h
        for h in pop:
            if fitness(h) < best_c5:
                best_h = h.copy()
                best_c5 = fitness(h)
    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
