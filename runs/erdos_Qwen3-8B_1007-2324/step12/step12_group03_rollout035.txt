    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h_clipped)
        scaling = integral_target / total
        return h_clipped * scaling

    # Compute overlap
    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = initial_population
    best_h = population[0].copy()
    best_c5 = compute_overlap(best_h)

    # DE parameters
    F = 0.8
    CR = 0.9
    max_generations = 1000
    generation = 0
    start_time = time.time()

    while generation < max_generations and time.time() - start_time < budget_s:
        generation += 1
        for i in range(len(population)):
            # Select three random indices
            while True:
                r1, r2, r3 = np.random.choice(len(population), 3, replace=False)
                if r1 != i and r2 != i and r3 != i:
                    break
            # Mutation: r3 + F * (r1 - r2)
            mutant = population[r3] + F * (population[r1] - population[r2])
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_half):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]

            # Compute C5

            # Update if better

                    best_h = trial.copy()

    # Final best h
    final_h = np.concatenate([best_h, best_h[::-1]])
