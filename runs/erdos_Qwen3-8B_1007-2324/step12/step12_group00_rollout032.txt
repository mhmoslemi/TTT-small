import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:140]
    else:
        x = np.linspace(0, 1, 140)
        h_opt_initial = np.clip(1 - np.abs(x - 0.5), 0, 1)
    
    n_points = 280
    dx = 2.0 / n_points
    integral_target = 140.0  # sum(h_opt) * dx = 1
    
    def repair(h_opt):
        current_sum = np.sum(h_opt)
        if current_sum == 0:
            return np.zeros_like(h_opt)
        scaling = integral_target / current_sum
        scaled = h_opt * scaling
        scaled = np.clip(scaled, 0, 1)
        return scaled
    
    def compute_overlap(h_opt):
        repaired = repair(h_opt)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(140)]
    
    start_time = time.time()
    
    result = differential_evolution(
        compute_overlap,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-10,
        maxiter=300,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        updating='nop',
        disp=False
    )
    
    elapsed = time.time() - start_time
    
    if elapsed > budget_s - 1:
        best_h_opt = h_opt_initial
    else:
        best_h_opt = result.x
    
    best_h_opt_repaired = repair(best_h_opt)
    best_h_full = np.concatenate([best_h_opt_repaired, best_h_opt_repaired[::-1]])
    best_c5 = compute_overlap(best_h_opt_repaired)
    
    return (best_h_full, best_c5, n_points)
