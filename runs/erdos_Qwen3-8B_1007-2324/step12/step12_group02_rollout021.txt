    else:
        h_opt = np.zeros(n_points)

        h_opt = (h_opt * integral_target / np.sum(h_opt)).clip(0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h):

        h_sum = np.sum(h)
        if h_sum == 0:

        scaling = integral_target / h_sum
