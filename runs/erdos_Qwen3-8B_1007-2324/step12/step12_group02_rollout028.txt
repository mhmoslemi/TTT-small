import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h_half = initial_h[:n_half]
    else:
        initial_h_half = np.full(n_half, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def objective(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        total = np.sum(x_clipped)
        scaling = integral_target / total if total != 0 else 1.0
        scaled = x_clipped * scaling
        h_full = np.concatenate([scaled, scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = differential_evolution(
        objective,
        bounds,
        popsize=30,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=300,
        x0=initial_h_half,
        tol=1e-5
    )

    best_x = result.x
    best_c5 = objective(best_x)
    h_full = np.concatenate([best_x, best_x[::-1]])
    h_full = np.clip(h_full, 0.0, 1.0)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
