        population.append(candidate)

    for i in range(len(population)):
        candidate = population[i]
        candidate = repair_h(candidate, integral_target)
        population[i] = candidate

    def compute_overlap(h):

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = population[0].copy()
