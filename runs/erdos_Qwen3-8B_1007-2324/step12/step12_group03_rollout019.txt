import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_half  # sum(candidate) = 70 for n_points=280

    def repair_h(candidate, target):
        current_sum = np.sum(candidate)
        if current_sum == 0:
            return np.zeros_like(candidate)
        scaling = target / current_sum
        return np.clip(candidate * scaling, 0, 1)

    def compute_overlap(candidate):
        h_full = np.concatenate([candidate, candidate[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    initial_population_size = 50
    if 'initial_h_values' in globals():
        initial_candidate = np.array(initial_h_values[:n_half])
        initial_candidate = repair_h(initial_candidate, integral_target)
        population = [initial_candidate]
        for _ in range(initial_population_size - 1):
            candidate = np.random.uniform(0, 1, n_half)
            candidate = repair_h(candidate, integral_target)
            population.append(candidate)
    else:
        population = [np.random.uniform(0, 1, n_half) for _ in range(initial_population_size)]
        for i in range(initial_population_size):
            population[i] = repair_h(population[i], integral_target)

    max_generations = 50
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        new_population = []
        for _ in range(len(population)):
            tournament = np.random.choice(population, 5, replace=False)
            best = min(tournament, key=compute_overlap)
            new_population.append(best)
        new_population = np.array(new_population)
        for i in range(0, len(new_population), 2):
            if i + 1 < len(new_population):
                p1, p2 = new_population[i], new_population[i + 1]
                split = np.random.randint(1, n_half)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
                if np.random.rand() < 0.1:
                    noise = np.random.normal(0, 0.05, n_half)
                    c1 += noise
                    c1 = repair_h(c1, integral_target)
                if np.random.rand() < 0.1:
                    noise = np.random.normal(0, 0.05, n_half)
                    c2 += noise
                    c2 = repair_h(c2, integral_target)
                new_population[i], new_population[i + 1] = c1, c2
        population = new_population

    best_idx = np.argmin([compute_overlap(c) for c in population])
    best_h = population[best_idx]
    best_c5 = compute_overlap(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
