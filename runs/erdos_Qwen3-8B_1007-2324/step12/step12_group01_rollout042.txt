import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair_h(h):
        current_sum = np.sum(h)
        if abs(current_sum) < 1e-10:
            return np.ones_like(h) * 0.5
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    def compute_c5(h_vec):
        h_opt = h_vec.copy()
        integral_val = np.sum(h_opt) * dx
        ideal_integral = n_points / 2
        scaling = ideal_integral / integral_val
        h_fixed = np.clip(h_opt * scaling, 0, 1)
        h_full = np.concatenate([h_fixed, h_fixed[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        best_overlap = np.max(corr)
        return best_overlap * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h_opt = initial_h[:n_points//2]
        population = [np.clip(initial_h_opt + np.random.normal(0, 0.05, n_points//2), 0, 1) for _ in range(50)]
        population = [repair_h(p) for p in population]
    else:
        population = [np.clip(np.random.rand(n_points//2), 0, 1) for _ in range(50)]
        population = [repair_h(p) for p in population]

    best_solution = population[0]
    best_c5 = compute_c5(best_solution)
    generations = 0
    max_generations = 200

    while time.time() - start_time < budget_s - 1 and generations < max_generations:
        generations += 1
        new_population = []
        for _ in range(len(population)):
            tournament = random.sample(population, min(5, len(population)))
            winner = min(tournament, key=compute_c5)
            if random.random() < 0.7:
                parent2 = random.choice(population)
                alpha = np.random.rand()
                child = alpha * winner + (1 - alpha) * parent2
                child = np.clip(child, 0, 1)
                child = repair_h(child)
                new_population.append(child)
            else:
                mutant = winner.copy()
                noise = np.random.normal(0, 0.03, n_points//2)
                mutant += noise
                mutant = np.clip(mutant, 0, 1)
                mutant = repair_h(mutant)
                new_population.append(mutant)
        population = new_population
        for ind in population:
            current_c5 = compute_c5(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = ind.copy()

    best_c5 = compute_c5(best_solution)
    elapsed = time.time() - start_time
    return best_solution, best_c5, n_points
