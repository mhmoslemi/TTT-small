                c1 = repair(c1)
                c2 = repair(c2)
            else:
                c1, c2 = p1.copy(), p2.copy()
            if random.random() < mutation_rate:
                noise = np.random.normal(0, mutation_std, n_half)
                c1 += noise
                c1 = repair(c1)
                c2 += noise
                c2 = repair(c2)
            new_population.append(c1)
            new_population.append(c2)
        candidates = [ (compute_overlap(indiv), indiv) for indiv in new_population ]
        candidates.sort()
        population = [ indiv for (f, indiv) in candidates[:50] ]
        fitness = [ compute_overlap(indiv) for indiv in population ]
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()

    best_h = repair(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
