        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        u = u_initial
    else:
        u = result.x
    
    # Final repair
    sum_u = np.sum(u)
    scale = target_sum / sum_u
    u = np.clip(u * scale, 0, 1)
    best_c5 = objective(u)
    
    h_values = np.concatenate([u, u[::-1]])
    return h_values, best_c5, n_points
