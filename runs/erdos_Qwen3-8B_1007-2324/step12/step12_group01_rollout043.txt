            new_pop.append(child)
        pop = new_pop
        curr_best = pop[np.argmin(fitness)]
        curr_c5 = eval_c5(curr_best)
        if curr_c5 < best_c5:
            best_h = curr_best
            best_c5 = curr_c5
        if time.time() - start_time > budget_s - 1:
            break

    final_h = repair(best_h)
    final_h_full = np.concatenate([final_h, final_h[::-1]])
    return final_h_full, best_c5, n_points
