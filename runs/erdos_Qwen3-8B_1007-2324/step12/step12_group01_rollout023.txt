        x = np.linspace(0, 2, n_points)

        h_opt = h[:n_points//2]

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    def objective(h):
        return compute_overlap(h)

    def constraint(h):
        return np.sum(h) - integral_target

    cons = [{'type': 'eq', 'fun': constraint}]

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 1.0

    iter_count = 0
    max_iters = 15000
    patience = 0
    max_patience = 1000

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:

        noise_std = 0.04 * current_temp
