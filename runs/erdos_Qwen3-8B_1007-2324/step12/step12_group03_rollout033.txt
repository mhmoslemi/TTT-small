        initial_h = np.array(initial_h_values)
        parent_candidate = initial_h[:n_points]
    else:
        parent_candidate = np.random.rand(n_points)

    # Create population: parent + 49 random vectors
    population = [parent_candidate]
    for _ in range(49):
        candidate = np.random.rand(n_points)
        candidate = repair_h(candidate, integral_target)
        population.append(candidate)

    def compute_overlap(h_candidate):
        h_full = np.clip(np.concatenate([h_candidate, h_candidate[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)

    max_generations = 500
    best_h = parent_candidate.copy()
    best_c5 = compute_overlap(best_h)

    for _ in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = [compute_overlap(ind) for ind in population]
        # Tournament selection
        def tournament_selection():
            indices = np.random.choice(len(population), size=5, replace=False)
            return population[np.argmin(fitness[indices])]
        parent1 = tournament_selection()
        parent2 = tournament_selection()
        # Crossover
        if np.random.rand() < 0.7:
            split = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:split], parent2[split:]])
        else:
            child = parent1
        # Mutation
        noise = np.random.normal(0, 0.05, n_points)
        mutant = child + noise
        mutant = repair_h(mutant, integral_target)
        # Replace worst individual
        worst_idx = np.argmax(fitness)
        population[worst_idx] = mutant
        # Update best
        current_c5 = compute_overlap(mutant)
        if current_c5 < best_c5:
            best_h = mutant.copy()
            best_c5 = current_c5

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
