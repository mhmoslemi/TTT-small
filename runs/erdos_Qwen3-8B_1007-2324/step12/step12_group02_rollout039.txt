import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)
        h_opt = h_opt[:n_points]
    else:
        h_opt = np.zeros(n_points)
        h_opt[:int(n_points / 2)] = 1.0
        np.random.shuffle(h_opt)
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h, target):
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scaling = target / np.sum(h)
        return np.clip(h * scaling, 0, 1)
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 1.0
    cooling_rate = 0.995
    iter_count = 0
    max_iters = 10000

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        old_c5 = compute_overlap(best_h)
        noise_std = 0.05 * current_temp
        noise = np.random.normal(0, noise_std, n_points)
        candidate = np.clip(best_h + noise, 0, 1)
        candidate = repair_h(candidate, integral_target)
        candidate_c5 = compute_overlap(candidate)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / current_temp):
            best_h = candidate
            best_c5 = candidate_c5
        current_temp *= cooling_rate

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
