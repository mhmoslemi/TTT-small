        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        y = 1 - h_full
        len_cc = 2 * len(h_full) - 1
        x_fft = fft(h_full, len_cc)
        y_fft = fft(y, len_cc)
        cc_fft = x_fft * np.conj(y_fft)
        cc = ifft(cc_fft).real
        return np.max(cc) * dx
    
    population_size = 40
    population = []
    for _ in range(population_size):
        perturbation = np.random.normal(0, 0.05, n_half)
        candidate = h_opt + perturbation
        candidate = repair_h(candidate, integral_target)
        population.append(candidate)
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap_fft(best_h)
    
    F = 0.8
    CR = 0.9
    generations = 0
    max_generations = 1000
    
    while time.time() - start_time < budget_s - 1 and generations < max_generations:
        generations += 1
        new_population = []
        for i in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)
            r1, r2, r3 = population[indices[0]], population[indices[1]], population[indices[2]]
            mutant = r1 + F * (r2 - r3)
            trial = np.copy(population[i])
            for j in range(n_half):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = repair_h(trial, integral_target)
            trial_c5 = compute_overlap_fft(trial)
            target_c5 = compute_overlap_fft(population[i])
            if trial_c5 < target_c5:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        for candidate in population:
            current_c5 = compute_overlap_fft(candidate)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5
    
    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
