import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]
    
    def repair(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target / s
        return np.clip(h * scale, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        perturb = np.random.uniform(-0.1, 0.1, size=n_half)
        h_perturb = np.clip(h_opt + perturb, 0, 1)
        h_repaired = repair(h_perturb, integral_target)
        bounds = [(0.0, 1.0) for _ in range(n_half)]
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(x),
            h_repaired,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        if res.success:
            h_candidate = res.x
            c5 = compute_overlap(h_candidate)
            if c5 < best_c5:
                best_h = h_candidate
                best_c5 = c5
    
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
