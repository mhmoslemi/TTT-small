import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2
    h_opt = np.array(initial_h_values[:n_points//2] if 'initial_h_values' in globals() else [1 - np.abs(i - n_points//2) / (n_points//2) for i in range(n_points//2)])
    
    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * integral_target / s, 0, 1)
    
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    pop_size = 50
    particles = np.array([repair(np.random.rand(n_points//2)) for _ in range(pop_size)])
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points//2))
    pbest = particles.copy()
    pbest_c5 = np.array([compute_c5(p) for p in particles])
    gbest = particles[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)
    best_c5 = gbest_c5
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = repair(particles[i])
            c5 = compute_c5(particles[i])
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
        if np.min(pbest_c5) < best_c5:
            best_c5 = np.min(pbest_c5)
            gbest = pbest[np.argmin(pbest_c5)]
        if (time.time() - start_time) > budget_s - 5:
            break
    return np.concatenate([gbest, gbest[::-1]]), best_c5, n_points
