import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values[:n_points])
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        s = np.sum(h)
        if s <= 0:
            return np.clip(h * 1e5, 0, 1)
        scaling = integral_target / s
        return np.clip(h * scaling, 0, 1)
    
    h_opt = repair(h_opt)
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()
    
    temp = 1.0
    cooling_rate = 0.995
    max_iter = int(budget_s * 1000)
    iterations = 0
    
    while time.time() - time.time() < budget_s and iterations < max_iter:
        iterations += 1
        noise_std = 0.05 * temp
        noise = np.random.normal(0, noise_std, n_points)
        candidate = np.clip(h_opt + noise, 0, 1)
        candidate = repair(candidate)
        candidate_c5 = compute_overlap(candidate)
        
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / temp):
            best_c5 = candidate_c5
            best_h = candidate
    
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
