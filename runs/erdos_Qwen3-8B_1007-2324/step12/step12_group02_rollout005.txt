import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]

    def repair_h(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (integral_target / s), 0, 1)

    def objective(h_opt):
        h_repaired = repair_h(h_opt)
        h_full = np.clip(np.concatenate([h_repaired, h_repaired[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_half)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=500,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        polish=True,
        disp=False,
        seed=seed

    best_h_opt = result.x
    best_c5 = objective(best_h_opt)

    h_full = np.clip(np.concatenate([best_h_opt, best_h_opt[::-1]]), 0, 1)
    elapsed = time.time() - start_time
