        corr = np.correlate(h_full, complement, mode='full')
        return np.max(corr) * dx

    fitness = [compute_overlap(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break

        new_population = []
        for _ in range(pop_size):
            tournament = np.random.choice(population, tournament_size, replace=False)
            t_fitness = [compute_overlap(ind) for ind in tournament]
            winner = tournament[np.argmin(t_fitness)]
            new_population.append(winner)

        for i in range(pop_size):
            if np.random.rand() < crossover_prob:
                j = np.random.randint(pop_size)
                parent1 = new_population[i]
                parent2 = new_population[j]
                point = np.random.randint(1, n_half - 1)
                child = np.concatenate([parent1[:point], parent2[point:]])
            else:
                child = new_population[i]
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_half)
                child += noise
            child = repair_h(child)
            new_population[i] = child

        new_fitness = [compute_overlap(ind) for ind in new_population]
        best_new = new_population[np.argmin(new_fitness)]
        if compute_overlap(best_new) < best_c5:
            best_h = best_new
            best_c5 = compute_overlap(best_new)
        population = new_population

    final_h = best_h
    final_h_full = np.clip(np.concatenate([final_h, final_h[::-1]]), 0, 1)
    final_c5 = compute_overlap(final_h)
    return final_h_full, final_c5, n_points
