        h_opt_initial = initial_h[:n_points//2]
        pop_size = 40
        population = [h_opt_initial + np.random.normal(0, 0.05, n_points//2) for _ in range(pop_size)]
    else:
        pop_size = 40
        population = [np.random.uniform(0, 1, n_points//2) for _ in range(pop_size)]
    
    def repair_h(h, target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    population = [repair_h(ind, target_for_opt) for ind in population]
    best_h = population[0]
    best_c5 = compute_overlap(best_h)
    
    generations = 0
    while time.time() - start_time < budget_s - 1 and generations < 200:
        generations += 1
        np.random.shuffle(population)
        for i in range(len(population)):
            j, k, l = np.random.choice(len(population), 3, replace=False)
            mutant = population[j] + 0.8 * (population[k] - population[l])
            mutant = np.clip(mutant, 0, 1)
            mask = np.random.rand(len(mutant)) < 0.5
            trial = np.where(mask, mutant, population[i])
            trial = repair_h(trial, target_for_opt)
            trial_c5 = compute_overlap(trial)
            if trial_c5 < best_c5 or np.random.rand() < 0.05:
                population[i] = trial
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial
    
    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
