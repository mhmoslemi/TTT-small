def run(seed=42, budget_s=100, **kwargs):

    import time

    dx = 2.0 / n_points
    integral_target = n_points / 2  # 140.0
    
    # Load initial_h_values if available
    if 'initial_h_values' in globals():

        # Fallback to symmetric triangular initial guess
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_points//2]
    
    def repair(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)
    
    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })
    
    start_time = time.time()
    result = scipy.optimize.minimize(
        objective,
        h_opt,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

    )
    best_h = result.x

    elapsed = time.time() - start_time
