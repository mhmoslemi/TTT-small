import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):

        if abs(current_sum) < 1e-6:

        scaling = target_sum / current_sum
        h_repaired = np.clip(h * scaling, 0, 1)
        return h_repaired

    def compute_c5(h_vec):
        h_opt = h_vec
        current_sum = np.sum(h_opt)
        if abs(current_sum) < 1e-6:
            scaling = target_sum / 0.5
        else:
            scaling = target_sum / current_sum
        h_fixed = np.clip(h_opt * scaling, 0, 1)
        h_full = np.concatenate([h_fixed, h_fixed[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Repair initial_h
        initial_h_repaired = repair(initial_h, target_sum)

        population = [initial_h_repaired + np.random.normal(0, 0.02, len(initial_h_repaired)) for _ in range(75)]
        population = [repair(h, target_sum) for h in population]
