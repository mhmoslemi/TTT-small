import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # Corrected target for integral constraint

    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, (1 - h_full), mode='full')
        return np.max(corr) * dx

    def repair_h(h, target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        return np.clip(h * scaling, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 1.5
    cooling_rate = 0.98
    iter_count = 0
    max_iters = 8000  # Adjusted for time budget
    mutation_probs = [0.5, 0.3, 0.2]  # Gaussian, Uniform, Horseshoe

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        # Select mutation type
        mut_idx = np.random.choice(3, p=mutation_probs)
        if mut_idx == 0:  # Gaussian
            noise = np.random.normal(0, 0.15 * current_temp, n_half)
            candidate = np.clip(h_opt + noise, 0, 1)
        elif mut_idx == 1:  # Uniform
            noise = np.random.uniform(-0.2, 0.2, n_half)
            candidate = np.clip(h_opt + noise, 0, 1)
        else:  # Horseshoe
            start = np.random.randint(0, n_half - 1)
            end = np.random.randint(start + 1, n_half)
            candidate = h_opt.copy()
            candidate[start:end] = 1 - candidate[start:end]
            candidate = np.clip(candidate, 0, 1)
        # Repair and evaluate
        candidate = repair_h(candidate, integral_target)
        candidate_c5 = compute_overlap(candidate)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / current_temp):
            best_h = candidate
            best_c5 = candidate_c5
        current_temp *= cooling_rate

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
