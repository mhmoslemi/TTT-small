import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        u_initial = h_initial[:n_points//2]
    else:
        u_initial = np.random.uniform(0, 1, n_points//2)
        sum_u = np.sum(u_initial)
        scaling = integral_target / max(sum_u, 1e-10)
        u_initial = np.clip(u_initial * scaling, 0, 1)
    
    def objective(u):
        h = np.concatenate([u, u[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        
        sum_u = np.sum(u)
        if abs(sum_u - integral_target) > 1e-8:
            scaling = integral_target / max(sum_u, 1e-10)
            u_scaled = np.clip(u * scaling, 0, 1)
            h_scaled = np.concatenate([u_scaled, u_scaled[::-1]])
            corr_scaled = np.correlate(h_scaled, 1 - h_scaled, mode='full')
            max_corr = np.max(corr_scaled) * dx
        
        return max_corr
    
    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    start_time = time.time()
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-8,
