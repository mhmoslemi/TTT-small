            child2 = mutate(child2, mut_std)
            # Repair
            child1 = repair_h(child1, integral_target)
            child2 = repair_h(child2, integral_target)
            new_population.append(child1)
            new_population.append(child2)
        # Replace old population with new_population and keep the best
        combined_population = population + new_population
        # Sort by fitness
        fitness_new = [compute_overlap(h) for h in combined_population]
        sorted_indices = np.argsort(fitness_new)
