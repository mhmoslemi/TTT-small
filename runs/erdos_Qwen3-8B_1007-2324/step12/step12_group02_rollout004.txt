import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # target sum for half interval

    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_half]
    else:
        h_opt = np.full(n_half, 0.5)

    def objective(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            h_scaled = np.zeros_like(h)
        else:
            scale = integral_target / sum_h
            h_scaled = np.clip(h * scale, 0, 1)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_half)]

    try:
        result = differential_evolution(
            objective, bounds,
            popsize=30, tol=1e-5, maxiter=300,
            callback=lambda x: None,
            recombination='blend',
            mutation=(0.5, 1.0)
        )
    except:
        result = None

    best_h = result.x if result else h_opt
    best_c5 = objective(best_h)

    elapsed = time.time() - start_time
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
