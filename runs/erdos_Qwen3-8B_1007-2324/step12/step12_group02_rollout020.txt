import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt_initial = initial_h[:n_half]
    else:
        h_opt_initial = np.full(n_half, 0.5)

    def objective(x):
        h_opt = np.clip(x, 0, 1)
        sum_h = np.sum(h_opt)
        if sum_h == 0:
            return np.inf
        scale = integral_target / sum_h
        h_opt = h_opt * scale
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_half)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=30, mutation=(0.5, 1), crossover=0.9,
        maxiter=150, tol=1e-6, seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)

        h_full *= scale

    elapsed = time.time() - start_time
