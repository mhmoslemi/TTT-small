import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt_initial = initial_h[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt_initial = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]

    def repair_h(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target / s
        return np.clip(h * scale, 0, 1)

    def objective(h_opt):
        h_repaired = repair_h(h_opt, integral_target)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0., 1.) for _ in range(n_half)]
    result = opt.differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, maxiter=100,
        tol=1e-8, seed=seed, polish=True, recombination=0.7, mutation=(0.7, 1.0),
        disp=False
    )
    best_h = np.concatenate([result.x, result.x[::-1]])
    best_c5 = objective(result.x)
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
