import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_half]
        population = [h_opt.copy() + np.random.normal(0, 0.05, n_half) for _ in range(40)]
    else:
        population = [np.random.rand(n_half) for _ in range(40)]
        for i in range(len(population)):
            p = population[i]
            scale = integral_target / np.sum(p)
            population[i] = np.clip(p * scale, 0, 1)

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        return np.clip(h * scale, 0, 1)

    def fitness(h):
        return compute_overlap(h)

    best_h = population[0]
    best_c5 = fitness(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(len(population)):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + 0.8 * (b - c)
            trial = np.where(np.random.rand(n_half) < 0.9, mutant, population[i])
            trial = repair(trial)
            trial_c5 = fitness(trial)
            if trial_c5 < fitness(population[i]):
                population[i] = trial
                if trial_c5 < best_c5:
                    best_h = trial
                    best_c5 = trial_c5

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    return h_full, best_c5, n_points
