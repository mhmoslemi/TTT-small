    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        scaled = h * scaling_factor
        clipped = np.clip(scaled, 0, 1)
        new_sum = np.sum(clipped)
        if np.isclose(new_sum, integral_target):
            return clipped
        remaining = integral_target - new_sum
        overshoot = np.where(scaled > 1)[0]
        if len(overshoot) == 0:
            return clipped
        for idx in overshoot:
            clipped[idx] = 1.0
        overshoot_val = 1.0 - scaled[overshoot]
        if remaining > 0:
            scale = remaining / np.sum(overshoot_val)
            clipped[overshoot] = 1.0 - (1.0 - scaled[overshoot]) * scale
        return clipped
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)

    x0 = h_opt.copy()
    constraints = ({

        'jac': lambda x: np.ones_like(x)
    })
    
    start_time = time.time()
    result = scipy.optimize.minimize(
