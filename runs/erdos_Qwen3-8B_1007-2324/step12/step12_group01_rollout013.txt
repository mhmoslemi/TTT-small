def run(seed=42, budget_s=100, **kwargs):

        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points//2]
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        s = np.sum(h)
        if s <= 0:
            return np.clip(h * 1e1, 0, 1)
        return np.clip(h * (integral_target / s), 0, 1)
    
    # Initial evaluation
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    
    # Optimization parameters
    max_iter = 10000
    temp = 1.0
    cooling_rate = 0.998
    noise_std = 0.05

        temp *= cooling_rate
    
    # Final repair
