import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair_h(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        return np.clip(h * scale, 0, 1)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_half]
    else:
        h_opt = np.random.rand(n_half)
    h_opt = repair_h(h_opt)

    population_size = 20
    population = [h_opt.copy() for _ in range(population_size)]

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    max_generations = 30
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = np.array([compute_overlap(ind) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        new_pop = []
        for _ in range(population_size // 2):
            t = np.random.choice(range(population_size), 5, replace=False)
            parent1 = population[t[np.argmin(fitness[t])]]
            if np.random.rand() < 0.7:
                c = np.random.randint(1, n_half)
                child1 = np.concatenate([parent1[:c], population[t[0]][c:]])
                child2 = np.concatenate([population[t[0]][:c], parent1[c:]])
                noise = np.random.normal(0, 0.05, n_half)
                child1 += noise
                child2 += noise
                child1 = repair_h(child1)
                child2 = repair_h(child2)
                new_pop.extend([child1, child2])
            else:
                child1 = np.random.normal(parent1, 0.05, n_half)
                child2 = np.random.normal(parent1, 0.05, n_half)
                child1 = repair_h(child1)
                child2 = repair_h(child2)
                new_pop.extend([child1, child2])
