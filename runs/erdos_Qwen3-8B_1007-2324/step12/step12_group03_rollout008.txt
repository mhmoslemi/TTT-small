    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        complement = 1 - h_full
        corr = np.correlate(h_full, complement, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return h.copy()
        scale_factor = integral_target / total
        return np.clip(h * scale_factor, 0, 1)

    pop_size = 40
    population = [h_opt.copy()]

        candidate = np.clip(h_opt + (noise - 0.5), 0, 1)
        candidate = repair(candidate)
        population.append(candidate)

    def fitness(h):
        return compute_overlap(h)

    best_h = h_opt.copy()
    best_c5 = fitness(best_h)
