def run(seed=42, budget_s=100, **kwargs):

    h_opt_initial = None

        initial_h = np.array(initial_h_values)
        h_opt_initial = initial_h[:n_points//2]
    else:
        # Generate a default initial h
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt_initial = h[:n_points//2]

    # Repair initial h_opt
    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    h_opt_initial = repair(h_opt_initial, integral_target)

    # Initialize population
    num_particles = 50
    particles = []
    velocities = []
    for _ in range(num_particles):
        noise = np.random.normal(0, 0.1, len(h_opt_initial))
        candidate = h_opt_initial + noise
        candidate = np.clip(candidate, 0, 1)
        candidate = repair(candidate, integral_target)
        particles.append(candidate)
        velocities.append(np.random.uniform(-0.1, 0.1, len(candidate)))

    # Compute initial fitness
    def compute_C5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initial personal bests and global best
    pbest = particles.copy()

    gbest = pbest[np.argmin(pbest_values)]

    start_time = time.time()
    omega = 0.7
    c_p = 1.5
    c_g = 1.5
    max_iter = 10000
    iter_count = 0
    improve_count = 0

import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt_initial = initial_h[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt_initial = h[:n_points//2]

    def repair(h, target):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * target / s, 0, 1)

    h_opt_initial = repair(h_opt_initial, integral_target)

    num_particles = 50
    particles = []
    velocities = []
    for _ in range(num_particles):
        noise = np.random.normal(0, 0.1, len(h_opt_initial))
        candidate = h_opt_initial + noise
        candidate = np.clip(candidate, 0, 1)
        candidate = repair(candidate, integral_target)
        particles.append(candidate)
        velocities.append(np.random.uniform(-0.1, 0.1, len(candidate)))

    def compute_C5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    pbest = particles.copy()
    pbest_values = [compute_C5(p) for p in pbest]
    gbest = pbest[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)

    start_time = time.time()
    omega = 0.7
    c_p = 1.5
    c_g = 1.5
    max_iter = 10000
    iter_count = 0

    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        improve_count = 0
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()

            new_x = particles[i] + v
            new_x = np.clip(new_x, 0, 1)
            new_x = repair(new_x, integral_target)
            new_c5 = compute_C5(new_x)
            if new_c5 < pbest_values[i]:
                pbest[i] = new_x
                pbest_values[i] = new_c5
                improve_count += 1
            if new_c5 < gbest_value:
                gbest = new_x
                gbest_value = new_c5
            particles[i] = new_x
        if improve_count == 0:

        if iter_count % 100 == 0:

    gbest = repair(gbest, integral_target)
    return gbest, gbest_value, n_points
