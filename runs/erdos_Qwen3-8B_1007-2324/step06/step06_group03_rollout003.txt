import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-5:
            return np.ones(n_points) * 0.5
        return h_clip * (integral_target / total)

    def generate_random_feasible():
        h = np.random.rand(n_points)
        h = repair(h * (integral_target / np.sum(h)))
        return h

    if 'initial_h_values' in globals():
        initial_solution = repair(np.array(initial_h_values))
    else:
        initial_solution = generate_random_feasible()

    pop_size = 50
    pop = np.array([initial_solution] + [generate_random_feasible() for _ in range(pop_size - 1)])
    best_fitness = evaluate(initial_solution)
    best_solution = initial_solution.copy()

    for gen in range(40):
        if time.time() - start_time > budget_s:
            break
        new_pop = np.zeros_like(pop)
        for idx in range(pop_size):

            candidates = pop[candidates_idx]
            winner_idx = np.argmin([evaluate(candidates[i]) for i in range(3)])
            parent_t = candidates[winner_idx]
