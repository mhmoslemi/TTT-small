    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    def initial_h_func():
        if 'initial_h_values' in globals():
            return np.array(initial_h_values)
        return np.full(n_points, 0.5)
    
    initial_h = initial_h_func()
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    trials = 30
    max_iter = 500
    
    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()
        shift = np.random.uniform(-n_points//4, n_points//4)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.015, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
