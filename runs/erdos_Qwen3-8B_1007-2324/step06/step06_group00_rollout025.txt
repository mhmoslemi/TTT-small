    elapsed = time.time() - start_time
    if elapsed > budget_s or result is None:
        best_h = initial_pop[0]
        best_c5 = objective(best_h)
    else:
        best_h = result.x
        best_c5 = result.fun

    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = repair(best_h)
    if np.sum(best_h) != integral_target:
        best_h = repair(best_h)

    return best_h, best_c5, n_points
