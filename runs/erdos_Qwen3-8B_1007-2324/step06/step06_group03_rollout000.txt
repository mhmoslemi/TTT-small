import time

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2

        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.full(n_points, 0.5)

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-5:
            return np.ones(n_points) * 0.5
        return h_clip * (integral_target / total)

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def generate_feasible():
        h = np.random.rand(n_points)
        return repair(h * (integral_target / np.sum(h)))

    h_start = repair(h_initial)
    pop_size = 50
    pop = np.array([h_start] + [generate_feasible() for _ in range(pop_size - 1)])
    best_h = h_start.copy()
    best_c5 = evaluate(h_start)

    param_mu = 0.05
    param_sigma = 0.15
    n_gen = 40

    for gen in range(n_gen):

            break
        new_pop = []

            parent = np.random.choice(pop)
            if np.random.random() < 0.8:
                noise = np.random.normal(0, param_sigma, n_points)
                child = 0.5 * parent + 0.5 * noise
