def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    integral_target = n_points / 2
    dx = 2.0 / n_points

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clip)
        if abs(sum_clipped - integral_target) < 1e-5:
            return h_clip
        scaling_factor = integral_target / sum_clipped
        h_scaled = h_clip * scaling_factor
        h_scaled_clipped = np.clip(h_scaled, 0, 1)
        return h_scaled_clipped

    def generate_random_feasible():
        h = np.random.rand(n_points)
        h_scaled = h * (integral_target / np.sum(h))
        return repair(h_scaled)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = generate_random_feasible()

    pop_size = 50
    population = [initial_h] + [generate_random_feasible() for _ in range(pop_size - 1)]

    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)

    for gen in range(40):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(pop_size):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = (parent1 + parent2) / 2
            noise = np.random.normal(0, 0.02, n_points)
            child = child + noise
            child = np.clip(child, 0, 1)
            child = repair(child)
            new_population.append(child)
        fitness = [evaluate(ind) for ind in new_population]
        for i in range(pop_size):
            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = new_population[i]
        population = new_population

    best_h = repair(best_h)
    return best_h, best_c5, n_points
