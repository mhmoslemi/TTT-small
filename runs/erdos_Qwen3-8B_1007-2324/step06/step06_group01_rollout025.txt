from scipy.optimize import minimize

    def repair_integral(h):
        total = np.sum(h)
        if abs(total - integral_target) < 1e-6:
            return np.clip(h, 0, 1)
        scale = integral_target / total
        h_scaled = h * scale
        return np.clip(h_scaled, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = h_initial.copy()
    best_c5 = objective(h_initial)
    max_iter = 200
    trials = 20
    seed_offset = seed

        try:
