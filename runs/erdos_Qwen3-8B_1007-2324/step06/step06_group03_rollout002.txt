from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    else:
        initial_h = np.full(n_points, 0.5)
    
    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total_clip = np.sum(h_clip)
        if total_clip == 0:
            return np.full_like(h, 1.0 / n_points)
        scale = integral_target / total_clip
        return np.clip(h_clip * scale, 0, 1)
    
    def objective(h):
        h_clipped = repair(h)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    starts = [initial_h]

        shifted = np.clip(shifted, 0, 1)

        starts.append(shifted)
