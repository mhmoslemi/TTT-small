import time

    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Generate a structured initial guess with evenly spaced 1s and 0s
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1.0
        initial_h = np.clip(initial_h, 0, 1)
        sum_initial = np.sum(initial_h)
        if sum_initial > 0:
            initial_h *= (integral_target / sum_initial)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    best_time = 0

    starts = []
    # Add initial guess
    starts.append(initial_h)
    # Generate more structured starting points with varied patterns
    for shift in range(0, 10):
        shifted = np.roll(initial_h, shift)
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)

        starts.append(shifted)
    
    # Add randomly perturbed versions

        if sum_noisy > 0:
            noisy *= (integral_target / sum_noisy)
        starts.append(noisy)

    # Optimize from all starting points
    for i, start in enumerate(starts):
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
