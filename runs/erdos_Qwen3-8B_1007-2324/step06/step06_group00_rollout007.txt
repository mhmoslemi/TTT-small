from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(float)
        h *= integral_target / np.sum(h)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = h.copy()
    best_c5 = objective(best_h)

    for _ in range(50):
        h_perturb = best_h.copy()
        idx = np.random.choice(n_points, size=10, replace=False)
        h_perturb[idx] = 1 - h_perturb[idx]
        result = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
