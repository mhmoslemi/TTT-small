    def generate_periodic_variations(base_h, seed=None, num_variations=5):

        variations = []

            shifted = np.clip(shifted, 0, 1)
            sum_shifted = np.sum(shifted)
            if sum_shifted > 0:
                shifted *= (integral_target / sum_shifted)
            variations.append(shifted)
        return variations

    starts = [initial_h]
    for i in range(5):
        starts.append(generate_periodic_variations(initial_h, seed=seed + i))
    starts.append(generate_periodic_variations(initial_h, seed=seed + 100))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
