import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_watershed(h0, seed=None):
        if seed is not None:
            np.random.seed(seed)
        noise = np.random.normal(0, 0.015, n_points)
        shift = np.random.randint(-30, 30)
        shifted = np.roll(h0 + noise, shift)
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    def random_binary_guess():
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[indices] = 1
        return h

    starts = [initial_h]
    for i in range(20):
        starts.append(generate_watershed(initial_h, seed + i))
    starts.append(random_binary_guess())

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    best_time = 0

    for i, start in enumerate(starts):
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed >= budget_s - best_time:
            break

    return best_h, best_c5, n_points
