import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200  # Reduced for faster computation
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_periodic(h0, shift, seed=None):
        if seed is not None:
            np.random.seed(seed)
        period = int(10 * n_points / 2)
        h = np.tile(h0, period // n_points + 1)[:n_points]
        h = np.roll(h, shift)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= (integral_target / sum_h)
        return h

    starts = [initial_h]
    for i in range(14):
        starts.append(generate_periodic(initial_h, i, seed + i))
    starts.append(generate_periodic(initial_h, 50, seed + 100))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    best_time = 0

    for i, start in enumerate(starts):
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1800, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed >= budget_s - best_time:
            break

    return best_h, best_c5, n_points
