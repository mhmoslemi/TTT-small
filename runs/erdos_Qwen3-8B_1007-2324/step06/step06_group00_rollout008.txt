import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    target_sum = n_points / 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values).copy()
        initial = np.clip(initial, 0, 1)
        initial = initial * (target_sum / np.sum(initial))

        initial = np.random.rand(n_points)

        initial = initial * (target_sum / np.sum(initial))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        current_sum = np.sum(h_clip)
        if not np.isclose(current_sum, target_sum):
            scale = target_sum / current_sum
            h_clip = h_clip * scale
            h_clip = np.clip(h_clip, 0, 1)
        corr = np.correlate(h_clip, 1.0 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    try:
        result = differential_evolution(
            objective, bounds, strategy='best1', maxiter=500, popsize=20,
            tol=1e-6, mutation=(0.8, 1.0), recombination=0.7, args=(n_points,)
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            h_values = initial
            c5_bound = objective(initial)
        else:
            h_values = result.x
            c5_bound = result.fun

        h_values = initial
        c5_bound = objective(initial)

    if not np.isclose(current_sum, target_sum):
