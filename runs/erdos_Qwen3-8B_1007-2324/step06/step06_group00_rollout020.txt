import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.zeros(n_points)
        ones_needed = integral_target
        indices = np.random.choice(n_points, size=int(ones_needed), replace=False)
        h[indices] = 1.0

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 200, 'ftol': 1e-9})
    best_h = result.x
    best_c5 = result.fun

    for _ in range(20):
        h_perturb = best_h.copy()
        idx = np.random.randint(0, n_points)
        h_perturb[idx] = 1 - h_perturb[idx]
        result_perturb = minimize(objective, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        if result_perturb.success and result_perturb.fun < best_c5:
            best_c5 = result_perturb.fun
            best_h = result_perturb.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
