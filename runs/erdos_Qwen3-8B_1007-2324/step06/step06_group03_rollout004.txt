import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    integral_target = n_points / 2
    dx = 2.0 / n_points

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-5:
            return np.full(n_points, 0.5)
        return h_clip * (integral_target / total)

    if 'initial_h_values' in globals():
        initial = repair(np.array(initial_h_values))
    else:
        initial = np.random.rand(n_points)
        initial = repair(initial)

    pop = [initial] + [np.random.rand(n_points) for _ in range(49)]
    pop = [repair(p) for p in pop]

    best_fitness = evaluate(initial)
    best_sol = initial.copy()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(len(pop)):
            a, b = np.random.choice(len(pop), 2, replace=False)
            parent = pop[np.argmin([evaluate(p) for p in (pop[a], pop[b])])]
            child = parent + np.random.normal(0, 0.05, n_points)
            child = np.clip(child, 0, 1)
            child = repair(child)
            new_pop.append(child)
        pop = new_pop
        for p in pop:
            f = evaluate(p)
            if f < best_fitness:
                best_fitness = f
                best_sol = p.copy()

    best_sol = repair(best_sol)
    return best_sol, best_fitness, n_points
