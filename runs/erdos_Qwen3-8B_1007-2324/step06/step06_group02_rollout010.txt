from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    h_initial = h0.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial *= integral_target / np.sum(h_initial)

    def perturb(h, seed=None):
        if seed is not None:
            np.random.seed(seed)
        h_pert = h.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert *= integral_target / np.sum(h_pert)
        return h_pert

    starts = [h_initial]
    starts.append(perturb(h_initial, seed + 1))
    starts.append(perturb(h_initial, seed + 2))
    starts.append(perturb(h_initial, seed + 3))
    starts.append(perturb(h_initial, seed + 4))
    starts.append(perturb(h_initial, seed + 5))

    best_h = h_initial
    best_c5 = objective(h_initial)

    for i, start in enumerate(starts):
        try:
            result = minimize(
