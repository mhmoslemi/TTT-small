import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Normalize and apply
        h = ((h > 0.5) * 1).astype(float)
        h = h * (integral_target / np.sum(h))
    else:
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(float)
        h = h * (integral_target / np.sum(h))

    def objective(params):
        T, a = params
        x = np.linspace(0, 2, n_points)
        r = np.mod(x, T)
        alpha = a * T
        beta = alpha + T / 2
        h = ((r >= alpha) & (r < beta)).astype(float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.5, 5.0), (0.0, 0.5)]
    initial_guess = [1.0, 0.5]

    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 300, 'ftol': 1e-9})

    best_T, best_a = result.x
    x = np.linspace(0, 2, n_points)
    r = np.mod(x, best_T)
    alpha = best_a * best_T
    beta = alpha + best_T / 2
    h_values = ((r >= alpha) & (r < beta)).astype(float)
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    best_c5 = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_values, best_c5, n_points

    return h_values, best_c5, n_points
