    # Build the h_values
    mod = np.mod(x_values, best_T)
    mask = (mod >= best_alpha) & (mod < best_alpha + best_T / 2)
    h_values = mask.astype(np.float64)
    
    # Compute the best C5
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_values, best_c5, n_points
    
    return h_values, best_c5, n_points
