import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 512
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.permutation([1.0] * (n_points // 2) + [0.0] * (n_points // 2))

    def objective(h_vec):
        h_clip = np.clip(h_vec, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]

    best_c5 = float('inf')
    best_h = h.copy()

    candidates = [h]
    periodic_h = np.zeros(n_points)
    periodic_h[:n_points // 2] = 1.0
    candidates.append(periodic_h)

    for _ in range(3):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral_target / np.sum(perturbed))
        candidates.append(perturbed)

    for init in candidates:
        try:
            result = minimize(
                objective, init, method='SLSQP', bounds=bounds,
                constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
