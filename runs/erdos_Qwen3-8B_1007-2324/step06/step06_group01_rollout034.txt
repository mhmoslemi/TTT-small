import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = initial_h * (140.0 / np.sum(initial_h))
    initial_h = np.clip(initial_h, 0, 1)

    def compute_c5(h):
        h_clip = h.clip(0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - 140.0}]

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    trials = 20
    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()
        shift = np.random.randint(-n_points//4, n_points//4)
        h_start = np.roll(h_start, shift)
        noise = np.random.normal(0, 0.04, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (140.0 / sum_start)
        try:
            result = minimize(
                compute_c5,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 400, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:

                    best_h = result.x.copy()
