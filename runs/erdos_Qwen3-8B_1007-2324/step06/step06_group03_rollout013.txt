import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

        scale = integral_target / total
        return np.clip(h_clip * scale, 0, 1)

    def objective(h):
        h_repaired = repair(h)

        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
