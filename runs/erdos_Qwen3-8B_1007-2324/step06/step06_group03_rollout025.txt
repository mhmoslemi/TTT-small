import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total) < 1e-12:
            return np.full_like(h_clip, integral_target / n_points)
        scale = integral_target / total
        return np.clip(h_clip * scale, 0, 1)

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Generate diverse initial variants
    variants = [initial_h]
    np.random.seed(seed)
    for _ in range(10):
        noise = np.random.normal(0, 0.05, n_points)
        shifted = np.roll(initial_h + noise, np.random.randint(-40, 40))
        shifted = np.clip(shifted, 0, 1)
        variants.append(shifted)

    best_h = initial_h
    best_c5 = objective(initial_h)

    for variant in variants:
        try:
            # Run differential evolution with initial guess
            result = differential_evolution(
                fun=objective,
                bounds=bounds,
                strategy='best1bin',
                popsize=20,
                mutation=(0.5, 1),
                recombination=0.8,
                tol=1e-6,
                maxiter=int(budget_s * 0.6),
                initial_guess=variant,
                disp=False,
                vectorized=True
            )
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
