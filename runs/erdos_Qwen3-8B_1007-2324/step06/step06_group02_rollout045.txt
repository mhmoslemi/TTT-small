    total = np.sum(h_tri)
    h_tri = h_tri * (integral_target / total) if total > 0 else np.zeros(n_points)

    def objective(h):
        h_clamped = np.clip(h, 0, 1)
        total = np.sum(h_clamped)
        if total == 0:
            return float('inf')
        h_scaled = h_clamped * (integral_target / total)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Set initial population based on provided or generated structures
    population = np.array([initial_h])
    for _ in range(5):
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = h_tri + noise
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total > 0:
            perturbed *= (integral_target / total)
        population = np.vstack((population, perturbed))

    # Run differential evolution with early stopping
    result = differential_evolution(
        objective, bounds,
        population_size=10, strategy='best1bin',
        maxiter=200, tolerance=1e-6, disp=False,
        polish=True, mutation=(0.5, 1), recombination=0.9
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1e-6:
        result = differential_evolution(
            objective, bounds,
            population_size=5, strategy='best1bin',
            maxiter=100, tolerance=1e-6, disp=False,
            polish=True, mutation=(0.5, 1), recombination=0.9
        )

    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
