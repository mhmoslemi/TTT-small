import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-5:
            return h_clip
        return h_clip * (integral_target / total)

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def generate_random_feasible():
        h = np.random.rand(n_points)
        h = repair(h)
        return h

    population = [repair(initial_h)]
    for _ in range(49):
        population.append(generate_random_feasible())

    best_h = population[0]
    best_c5 = evaluate(best_h)

    for gen in range(100):
