import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 140

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        c5 = np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    initial_x = initial_h

    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,

        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h) - 1e6 * np.abs(np.sum(best_h) - integral_target)

    elapsed = time.time() - start_time
