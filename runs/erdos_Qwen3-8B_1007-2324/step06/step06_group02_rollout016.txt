import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    candidates = [initial_h]
    
    # Generate diverse initial candidates
    def make_pattern(period, offset, scale):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            t = i / n_points * 2.0
            val = 0.5 * (1 + np.cos(2 * np.pi * (t - offset) / period))
            pattern[i] = val * scale
        return pattern
    
    candidates.append(make_pattern(1.0, 0.5, 1.0))
    candidates.append(make_pattern(0.5, 0.25, 1.0))
    candidates.append(make_pattern(1.5, 0.75, 1.0))
    candidates.append(1.0 * (np.random.rand(n_points) < 0.5))
    
    for h_initial in candidates:
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        if abs(sum_h - target_sum) > 1e-6:
            h_initial = h_initial * (target_sum / sum_h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        def objective(h):
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            return np.max(corr) * dx
        
        try:
            result = differential_evolution(
                objective,
                bounds,
                seed=seed,
                strategy='best1bin',
                popsize=15,
                maxiter=100,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                args=(dx,)
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
