        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    # Generate periodic variations
    def generate_periodic(h_base, period_length):

            return h_base.copy()
        repeats = n_points // period_length
        base_segment = h_base[:period_length]
        periodic_h = np.tile(base_segment, repeats)
        remaining = n_points - repeats * period_length
        if remaining > 0:
            periodic_h = np.concatenate((periodic_h, base_segment[:remaining]))
        scale_factor = integral_target / np.sum(periodic_h)
        return periodic_h * scale_factor
