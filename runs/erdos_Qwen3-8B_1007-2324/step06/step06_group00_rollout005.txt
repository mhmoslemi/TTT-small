import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values).astype(float)
        initial_pop = [np.clip(initial_h + 0.1 * np.random.randn(n_points), 0, 1) for _ in range(20)]
    else:
        initial_pop = [np.clip(np.random.randn(n_points), 0, 1) for _ in range(20)]

    def repair(h):
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-5:
            return h
        scale = integral_target / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        h_repaired = repair(h_clip)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=200,
            popsize=20,
            tol=1e-6,
            mutation=(0.8, 1.0),
            recombination=0.7,
            options={'mutation': (0.8, 1.0), 'recombination': 0.7}
        )
    except:
        best_h = initial_pop[0]
        best_c5 = objective(best_h)
        return best_h, best_c5, n_points

    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time

    best_h = repair(best_h)
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
