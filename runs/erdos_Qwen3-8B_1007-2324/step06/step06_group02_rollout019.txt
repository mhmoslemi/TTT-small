import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def compute_c5(h):
        h_clip = np.clip(h, 0, 1)
        max_overlap = 0.0
        for k in range(-n_points + 1, n_points):
            x1 = np.arange(0, n_points)
            x2 = (x1 + k) % n_points
            overlap = np.sum(h_clip[x1] * (1 - h_clip[x2])) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    best_time = 0

    starts = [initial_h]
    for i in range(15):
        noise = np.random.normal(0, 0.02, n_points)
        shifted = np.roll(initial_h + noise, i)
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        starts.append(shifted)

    for i, start in enumerate(starts):
        try:
            result = minimize(
                compute_c5,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1200, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:

                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed >= budget_s - best_time:
            break

    return best_h, best_c5, n_points
