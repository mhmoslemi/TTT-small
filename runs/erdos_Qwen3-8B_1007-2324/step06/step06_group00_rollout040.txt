import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 140

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.random.rand(n_points)
        h = (h < 0.5).astype(float)
        h *= integral_target / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    # Fibonacci-like grid search for T values
    T_values = [0.5, 1.0, 2.0]
    x_array = np.linspace(0, 2, n_points)

    # Perform exhaustive search over T and a
    for T in T_values:
        N = 2.0 / T
        w = 1.0 / N
        a_values = np.linspace(0, T - w, 100)
        for a in a_values:
            fractional = np.mod(x_array, T)
            condition = (fractional >= a) & (fractional < a + w)
            h_candidate = condition.astype(float)

            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate.copy()

    # Optimizer refinement

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
