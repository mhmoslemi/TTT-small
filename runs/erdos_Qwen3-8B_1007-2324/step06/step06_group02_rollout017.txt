from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_sinusoidal():
        x = np.linspace(0, 2, n_points)
        k_values = [1, 2, 3, 4, 5, 6]
        res = []
        for k in k_values:
            h = 0.5 + 0.5 * np.sin(2 * np.pi * k * x)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (integral_target / sum_h)
            h *= scale
            res.append(h)
        return res

    def generate_shuffled(start):
        np.random.seed(seed + 100)
        noise = np.random.normal(0, 0.015, n_points)
        shifted = np.roll(start + noise, np.random.randint(-30, 30))
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    starts = [initial_h]
    for i in range(5):
        starts.append(generate_shuffled(initial_h))
    starts += generate_sinusoidal()

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    best_time = 0

    for i, start in enumerate(starts):
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1200, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            elapsed = time.time() - start_time
            if elapsed >= budget_s - best_time:
                break
        except:
            pass

    return best_h, best_c5, n_points
