import numpy as np
from scipy.optimize import dual_annealing
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    max_trials = 10

    for trial in range(max_trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + trial)
        # Generate initial guess via shifting and noise
        shift = np.random.uniform(-40, 40)
        shifted = np.roll(initial_h, int(shift))
        noise = np.random.normal(0, 0.02, n_points)
        h_start = shifted + noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        # Run dual annealing
        result = dual_annealing(
            objective,
            bounds,
            constraints=constraints,
            options={
                'initial_temp': 0.15,
                'final_temp': 1e-6,
                'maxiter': 5000,
                'maxfun': 5000,
                'tol': 1e-8
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return np.clip(best_h, 0, 1), best_c5, n_points
