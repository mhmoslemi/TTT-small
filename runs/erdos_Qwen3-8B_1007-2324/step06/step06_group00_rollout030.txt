        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constrained_objective(h):
        h_clip = np.clip(h, 0.0001, 0.9999)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(h) - integral_target)
        return max_corr + penalty

    result = differential_evolution(
        constrained_objective, bounds,
        strategy='best1bin', maxiter=50, popsize=20,
        tol=1e-5, mutation=(0.5, 1), recombination=0.8,
        seed=seed
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
