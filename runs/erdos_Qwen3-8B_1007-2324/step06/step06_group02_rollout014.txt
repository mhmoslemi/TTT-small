import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 120  # Reduced for efficiency
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

        initial_h = np.full(n_points, 0.5)

    def generate_variants(h0):

        variants = []

            shift = np.random.randint(-15, 15)

            total = np.sum(shifted)
            if total > 0:

            variants.append(shifted)
        return variants

    starts = generate_variants(initial_h)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')

                start,
