from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

    else:
        initial_h = np.full(n_points, 0.5)

    def generate_watershed(h0, seed=None):
        if seed is not None:
            np.random.seed(seed)
        noise = np.random.normal(0, 0.01, n_points)
        shift = np.random.randint(-30, 30)
        shifted = np.roll(h0 + noise, shift)
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    def generate_periodic(n_points, period=5):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % period < period // 2:
                h[i] = 1
        total = np.sum(h)
        if total > 0:
            h *= (integral_target / total)
        return h

    starts = [initial_h]
    for i in range(10):
        starts.append(generate_watershed(initial_h, seed + i))
    starts.append(generate_periodic(n_points, period=7))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    best_time = 0

    for i, start in enumerate(starts):
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
