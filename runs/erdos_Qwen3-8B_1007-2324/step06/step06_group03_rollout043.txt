import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.full_like(h, 0.5)
        scale = integral_target / sum_h
        h_scaled = h * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-8,
        maxiter=100,
        seed=seed,
        polish=False,
        vectorized=True
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        c5 = objective(initial_h)
        return initial_h, c5, n_points

    return result.x, result.fun, n_points
