import numpy as np
from scipy.optimize import dual_annealing

def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')

        h_shifted = np.roll(initial_h, shift)

        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        else:
            h_start = np.full(n_points, integral_target / n_points)

        try:
            result = dual_annealing(
                objective,
                bounds,
                constraints=constraints,
                options={
                    'initial_temp': 0.5,
                    'final_temp': 1e-6,
                    'maxiter': 1000,
                    'maxfun': 1000
                }
            )
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
