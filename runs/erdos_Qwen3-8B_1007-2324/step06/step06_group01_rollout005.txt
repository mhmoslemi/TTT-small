import numpy as np
from scipy.optimize import dual_annealing
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    # Generate diverse initial guesses
    initial_guesses = []
    for shift in range(-20, 21):
        shifted = np.roll(initial_h, shift)
        noise = np.random.normal(0, 0.02, n_points)
        shifted += noise
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        initial_guesses.append(shifted)

    # Run optimization
    for guess in initial_guesses:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            result = dual_annealing(
                objective,
                bounds=bounds,
                constraints=constraints,
                options={
                    'initial_temp': 1.0,
                    'final_temp': 1e-5,
                    'maxiter': 1000,
                    'maxfun': 1000,
                    'disp': False,
                    'tol': 1e-6,
                    'seed': seed
                },
                x0=guess
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass

    # Enforce constraints
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - integral_target) > 1e-6:
        best_h *= (integral_target / sum_best)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
