        h[(r >= interval_low) & (r < interval_high)] = 1
        return h

    def objective(params):
        T, a = params
        if T <= 0:
            return float('inf')
        h = build_h(T, a)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.25, 5.0), (0.0, 0.5)]
    initial_guess = [1.0, 0.25]

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Estimate T and a from initial_h_values
        idx = np.where(np.diff(initial_h) > 0)[0]
        if len(idx) > 0:
            period = np.mean(np.diff(idx))
            initial_guess[0] = period
            initial_guess[1] = 0.25

    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200, 'ftol': 1e-9})

    best_T, best_a = result.x
    h = build_h(best_T, best_a)
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h, best_c5, n_points

    return h, best_c5, n_points
