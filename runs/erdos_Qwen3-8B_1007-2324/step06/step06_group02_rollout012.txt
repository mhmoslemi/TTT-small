from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    starts = [initial_h]

        shifted = np.clip(shifted, 0, 1)
        shifted_sum = np.sum(shifted)
        if shifted_sum > 0:

        starts.append(shifted)
    
    # Optimize each start
    for i, start in enumerate(starts):
