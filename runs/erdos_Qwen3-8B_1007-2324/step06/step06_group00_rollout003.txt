import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_pop = [np.clip(initial_h_values.astype(float) + 0.1 * np.random.randn(n_points), 0, 1) for _ in range(20)]
    else:
        initial_pop = [np.clip(np.random.randn(n_points), 0, 1) for _ in range(20)]

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-6:
            return h_clip
        scale = integral_target / total
        h_scaled = h_clip * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    def objective(h):
        h_clip = repair(h)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    try:
        result = differential_evolution(
            objective, bounds, 
            strategy='best1bin', 
            maxiter=500, 
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            polish=True
        )
    except:
        result = None

    best_h = result.x if result and result.success else initial_pop[0]
    best_c5 = objective(best_h) if result and result.success else objective(initial_pop[0])

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
