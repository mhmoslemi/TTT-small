        penalty = 1e6 * abs(np.sum(h_clip) - integral_target)
        return c5_val + penalty

    def evaluate_correlation(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    # Multi-start differential evolution
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   maxiter=75, popsize=20, tol=1e-6,
                                   mutation=(0.5, 1), recombination=0.8,
                                   disp=False)

    best_h = result.x
    best_c5 = evaluate_correlation(best_h)

    return best_h, best_c5, n_points
