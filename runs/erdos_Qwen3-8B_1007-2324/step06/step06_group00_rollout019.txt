import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points / 2
    start_time = time.time()

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        T_start = np.random.uniform(0.25, 5.0)
        a_start = np.random.uniform(0.0, 1.0)
    else:
        T_start = np.random.uniform(0.25, 5.0)
        a_start = np.random.uniform(0.0, 1.0)

    def objective(params):
        T, a = params
        alpha = T * a
        x = np.linspace(0, 2, n_points, endpoint=False)
        r = x % T
        interval_start = alpha
        interval_end = alpha + T / 2
        mask = (r >= interval_start) & (r < interval_end)
        h = mask.astype(float)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.25, 5.0), (0.0, 1.0)]
    initial_guess = [T_start, a_start]
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200, 'ftol': 1e-9})

    if result.success:
        best_T, best_a = result.x
        best_alpha = best_T * best_a
        x = np.linspace(0, 2, n_points, endpoint=False)
        r = x % best_T
        interval_start = best_alpha
        interval_end = best_alpha + best_T / 2
        mask = (r >= interval_start) & (r < interval_end)
        best_h = mask.astype(float)
        c5_bound = result.fun
    else:
        # Fallback to initial guess
        best_T, best_a = initial_guess
        best_alpha = best_T * best_a
        x = np.linspace(0, 2, n_points, endpoint=False)
        r = x % best_T
        interval_start = best_alpha
        interval_end = best_alpha + best_T / 2
        mask = (r >= interval_start) & (r < interval_end)
        best_h = mask.astype(float)
        c5_bound = objective(initial_guess)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
