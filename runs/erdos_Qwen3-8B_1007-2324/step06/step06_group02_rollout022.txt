            start *= target_sum / np.sum(start)
        else:
            start = np.full_like(start, 0.5)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed
        )

        if result.success:
            current_c5 = result.fun - 1e3 * np.abs(np.sum(result.x) - target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()

        if time.time() - start_time >= budget_s:
            break

    # Final clipping and sum adjustment
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) != 0:
        best_h *= target_sum / np.sum(best_h)
    else:
        best_h = np.full_like(best_h, 0.5)
    return best_h, best_c5, n_points
