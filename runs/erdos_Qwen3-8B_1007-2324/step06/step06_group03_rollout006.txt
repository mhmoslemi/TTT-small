import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-5:
            return np.ones(n_points) * 0.5
        return h_clip * (integral_target / total)

    def generate_random():
        h = np.random.rand(n_points)
        return repair(h * (integral_target / np.sum(h)))
