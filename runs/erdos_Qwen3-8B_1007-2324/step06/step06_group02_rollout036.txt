import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def generate_periodic_candidate(T):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            pos_in_period = x - int(x / T) * T
            if pos_in_period < T / 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return h

    candidates = []
    if 'initial_h_values' in globals():
        candidates.append(initial_h)
    T_values = [1.0, 0.5, 2.0, 0.25, 1.5, 0.75]
    for T in T_values:
        candidates.append(generate_periodic_candidate(T))

    best_h = initial_h.copy()
    best_c5 = float('inf')

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    for candidate in candidates:
        sum_candidate = np.sum(candidate)
        scale = integral_target / sum_candidate if sum_candidate != 0 else 1.0
        scaled = np.clip(candidate * scale, 0, 1)
        if np.sum(scaled) < integral_target - 1e-5:
            delta = (integral_target - np.sum(scaled)) / n_points
            scaled += delta
        scaled = np.clip(scaled, 0, 1)
        initial_guess = scaled

        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
