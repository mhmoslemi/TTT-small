def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def create_even_intervals(n_points, k):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        interval_length = 1.0 / k
        for i in range(k):
            start = i * interval_length
            end = start + interval_length
            start_idx = int(start / dx)
            end_idx = int(end / dx)
            start_idx = max(0, min(start_idx, n_points - 1))
            end_idx = max(0, min(end_idx, n_points - 1))
            h[start_idx:end_idx] = 1.0
        return h

    starts = [initial_h]
    for k in [4, 5, 6, 7, 8]:
        starts.append(create_even_intervals(n_points, k))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    best_time = 0

    for i, start in enumerate(starts):
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1300, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed >= budget_s - best_time:
            break

    return best_h, best_c5, n_points
