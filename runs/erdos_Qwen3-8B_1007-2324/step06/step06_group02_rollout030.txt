                h_grid_opt = np.zeros(n_points)
                for i in range(m):
                    start, end = kinks[i], kinks[i + 1]
                    start_idx = int(start / dx)
                    end_idx = int(end / dx)
                    h_grid_opt[start_idx:end_idx + 1] = h_vals_opt[i]
                best_h = h_grid_opt

    return best_h, best_c5, n_points
