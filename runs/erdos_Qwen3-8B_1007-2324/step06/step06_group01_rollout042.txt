import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    num_particles = 20
    max_iter = 100
    w, c1, c2 = 0.7, 1.5, 1.5

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    particles = []
    velocities = []
    pbest = []
    pvals = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        particles.append(initial_h)
        velocities.append(np.zeros(n_points))
        pbest.append(initial_h.copy())
        pvals.append(objective(initial_h))
        for _ in range(num_particles - 1):
            new_vec = initial_h + np.random.normal(0, 0.05, n_points)
            new_vec = np.clip(new_vec, 0, 1)
            sum_new = np.sum(new_vec)
            if sum_new > 0:
                new_vec *= (integral_target / sum_new)
            particles.append(new_vec)
            velocities.append(np.zeros(n_points))
            pbest.append(new_vec.copy())
            pvals.append(objective(new_vec))
    else:
        for _ in range(num_particles):
            vec = np.random.rand(n_points)
            vec *= (integral_target / np.sum(vec))
            particles.append(vec)
            velocities.append(np.zeros(n_points))
            pbest.append(vec.copy())
            pvals.append(objective(vec))

    gbest = pbest[np.argmin(pvals)]
    gval = np.min(pvals)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            if sum_new > 0:
                new_pos *= (integral_target / sum_new)
            else:
                new_pos = particles[i].copy()
            new_val = objective(new_pos)
            if new_val < pvals[i]:
                pbest[i] = new_pos.copy()
                pvals[i] = new_val
        current_gval = np.min(pvals)
        if current_gval < gval:
            gval = current_gval
            gbest = pbest[np.argmin(pvals)]

    return gbest, gval, n_points
