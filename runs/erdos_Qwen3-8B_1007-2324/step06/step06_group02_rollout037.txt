        kink_indices = np.where(diffs > 1e-8)[0] + 1
        kink_positions = np.sort(kink_indices * dx)
        spacing = np.mean(np.diff(kink_positions)) if len(kink_positions) > 1 else 0.5
        T = spacing
    else:
        T = 0.5

    best_c5 = float('inf')
    best_h = initial_h.copy()

    x_values = np.linspace(0, 2, n_points)

    for m in range(2, 26):
        kinks = np.linspace(0, 2, m + 1)
        segment_lengths = np.diff(kinks)
        integral_weights = segment_lengths * dx

        def obj_func(h_vals):
            h_full = np.zeros(n_points)
            for i in range(m):
                start = np.searchsorted(kinks, kinks[i], side='right')
                end = np.searchsorted(kinks, kinks[i+1], side='left')
                h_full[start:end] = h_vals[i]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def constraint_func(h_vals):
            return np.dot(h_vals, integral_weights) - 1.0

        if 'initial_h_values' in globals():
            h_vals_init = np.zeros(m)
            for i in range(m):
                start = np.searchsorted(kinks, kinks[i], side='right')
                end = np.searchsorted(kinks, kinks[i+1], side='left')
                h_vals_init[i] = np.mean(initial_h[start:end])
        else:
            h_vals_init = np.full(m, 0.5)

        bounds = [(0.0, 1.0) for _ in range(m)]
        constraints = [{'type': 'eq', 'fun': constraint_func}]
        try:
            result = minimize(obj_func, h_vals_init, method='L-BFGS-B', bounds=bounds, constraints=constraints)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    h_full = np.zeros(n_points)
                    for i in range(m):
                        start = np.searchsorted(kinks, kinks[i], side='right')
                        end = np.searchsorted(kinks, kinks[i+1], side='left')
                        h_full[start:end] = result.x[i]
                    best_h = h_full
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break

    return best_h, best_c5, n_points
