import numpy as np
from scipy.optimize import dual_annealing
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    trials = 10
    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        shift = np.random.randint(-n_points // 2, n_points // 2)
        h_start = np.roll(initial_h, shift)
        h_start += np.random.normal(0, 0.03, n_points)
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
        try:
            result = dual_annealing(
                objective,
                bounds=bounds,
                constraints=constraints,
                options={'initial_temp': 0.1, 'final_temp': 1e-7, 'maxiter': 1000, 'maxfun': 1000, 'disp': False},
                x0=h_start
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
