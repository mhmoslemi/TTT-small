import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def evaluate(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total = np.sum(h_clip)
        if abs(total - integral_target) < 1e-8:
            return np.full_like(h_clip, 0.5)
        return h_clip * (integral_target / total)

    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values)
    else:
        initial = np.random.rand(n_points)
    initial = repair(initial)

    pop_size = 50
    pop = [initial + np.random.normal(0, 0.02, n_points) for _ in range(pop_size)]
    pop = [repair(sol) for sol in pop]

    best = initial
    best_c5 = evaluate(best)

    gen_limit = 50
    for gen in range(gen_limit):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size):
            t = np.random.choice(pop, 3, replace=False)
            parent = t[np.argmin([evaluate(c) for c in t])]
            noise = np.random.normal(0, 0.05, n_points)
            child = parent + noise
            new_pop.append(repair(child))
        pop = new_pop
        for c in pop:
            c5 = evaluate(c)
            if c5 < best_c5:
                best_c5 = c5
                best = c.copy()
    best = repair(best)
    return best, best_c5, n_points
