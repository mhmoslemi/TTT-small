            if merged and merged[-1][1] + 1 >= s:
                merged[-1] = (merged[-1][0], max(merged[-1][1], e))
            else:
                merged.append((s, e))
        intervals = merged
    elif len(intervals) < m:
        # Split one interval into multiple
        s, e = intervals[0]
        split = (s, s + (e - s) // m)
        intervals = [(s, split[1])] + [(split[1]+1, e)] + intervals[1:]
    
    # Prepare initial guess for optimization
    x0 = []
    for s, e in intervals:
        x0.extend([s, e])
    
    # Objective function
    def objective(x):
        intervals = []
        for i in range(m):
            s = int(round(x[2*i]))
            e = int(round(x[2*i + 1]))
            s = max(0, min(s, n_points - 1))
            e = max(s, min(e, n_points - 1))
            intervals.append((s, e))
        # Check overlaps
        sorted_intervals = sorted(intervals, key=lambda x: x[0])
        for i in range(1, m):
            if sorted_intervals[i-1][1] + 1 >= sorted_intervals[i][0]:
                return float('inf')
        # Compute h and correlation
        h = np.zeros(n_points)
        for s, e in intervals:
            h[s:e+1] = 1
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Constraints
    bounds = [(0, n_points-1) for _ in range(2*m)]
    cons = []
    for i in range(m):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*i + 1]})
    for i in range(m):
        for j in range(i+1, m):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: x[2*j] - x[2*i + 1]})
    cons.append({'type': 'eq', 'fun': lambda x: sum(x[2*i + 1] - x[2*i] + 1 for i in range(m)) - n_points/2})
    
    # Optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_x = result.x
    
    # Generate final h
    h = np.zeros(n_points)
    for i in range(m):
        s = int(round(best_x[2*i]))
        e = int(round(best_x[2*i + 1]))
        s = max(0, min(s, n_points - 1))
        e = max(s, min(e, n_points - 1))
        h[s:e+1] = 1
    
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    return (h, best_c5, n_points)
