import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = np.zeros(140)
    best_n = 140
    initial_guesses = []
    n_values = [140, 280]
    
    # Add initial_h_values and upsampled version
    initial_guesses.append(initial_h_values)
    initial_guesses.append(np.repeat(initial_h_values, 2))
    
    # Generate random guesses
    for n in n_values:
        for _ in range(3):
            half = n // 2
            h = np.zeros(n, dtype=int)
            h[:half] = 1
            np.random.shuffle(h)
            initial_guesses.append(h.astype(float))
    
    start_time = time.time()
    
    for guess in initial_guesses:
        n = len(guess)
        if n > 280:
            continue
        # Ensure sum is correct
        if np.isclose(np.sum(guess), n / 2):
            pass
        else:
            guess /= np.sum(guess) * (n / 2)
        # Objective function
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        bounds = [(0.0, 1.0) for _ in range(n)]
