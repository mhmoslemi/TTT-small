    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        best_h = initial_h
        best_c5 = float('inf')
        dx = 2.0 / n_points
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
