import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = h_initial.sum()
    scale = target_sum / current_sum
    h_initial = h_initial * scale
    
    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h_clipped * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    cons = [sum_constraint]
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    num_restarts = 16  # Increased restarts for better exploration
    
    for _ in range(num_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
        perturbation = np.random.normal(0, 0.15, n_points)  # Larger perturbation for broader search
        h_perturbed = h_initial + perturbation
        h_perturbed = adjust_h(h_perturbed)
        
        result = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 600})
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    best_h = adjust_h(best_h)
    return best_h.tolist(), best_c5, n_points
