import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with provided initial_h_values or random
    if 'initial_h_values' in locals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / h.sum())

    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        total = h_clipped.sum()
        if total == 0:
            return np.zeros_like(h)
        scale = target_sum / total
        return h_clipped * scale

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    num_restarts = 10
    max_iter = 300

    for _ in range(num_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
        # Generate diverse perturbations
        pert = np.random.normal(0, 0.15, n_points)
        h_perturbed = h + pert
        h_perturbed = adjust_h(h_perturbed)

        result = minimize(
            objective,
            h_perturbed,
            method='trust-constr',
            bounds=bounds,
            constraints=[sum_constraint],
            options={'maxiter': max_iter, 'xscale': 'reciprocal'}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = adjust_h(best_h)
    return best_h.tolist(), best_c5, n_points
