        cons = {'type': 'eq', 'fun': lambda x: constraint(x, s)}
        result = minimize(lambda x: compute_c5(x, s), endpoints, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_endpoints = np.sort(result.x)
                intervals = [(best_endpoints[2*i], best_endpoints[2*i+1]) for i in range(s)]
                n_points = 1000
                dx = 2.0 / n_points
                x = np.linspace(0, 2, n_points)
                h = np.zeros(n_points)
                for a, b in intervals:
                    h[(x >= a) & (x < b)] = 1.0
                best_h = h
                best_n = n_points
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, best_n)
