from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    h_initial = initial_h_values.copy()

    scale = target_sum / current_sum if current_sum != 0 else 1
    h_initial = h_initial * scale
    
    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        if abs(s - target_sum) < 1e-5:
            return h_clipped
        if s == 0:
            return np.zeros_like(h)
        factor = target_sum / s
        v_scaled = factor * h_clipped
        v_scaled = np.clip(v_scaled, 0, 1)
        return v_scaled
    
    def objective(h):
        corr = np.correlate(adjust_h(h), 1 - adjust_h(h), mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    cons = [sum_constraint]
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    num_restarts = 12

        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = adjust_h(h_perturbed)
