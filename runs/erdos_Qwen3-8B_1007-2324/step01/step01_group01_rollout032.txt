import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 560
    n = n_points
    dx = 2.0 / n
    
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.tile(initial_h_values, 4)
    else:
        initial_h = np.full(n, 0.5)
    
    np.random.seed(seed)
    initial_points = [initial_h]
    for _ in range(2):
        h = np.random.rand(n)
        sum_h = np.sum(h)
        h = h * (n / 2) / sum_h
        h = np.clip(h, 0, 1)
        initial_points.append(h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_func(h):
        return np.sum(h) - n / 2
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    best_c5 = float('inf')
    best_h = np.full(n, 0.5)
    
    for init in initial_points:
        sum_h = np.sum(init)
        scale = (n / 2) / sum_h if sum_h != 0 else 1.0
        init = init * scale
        init = np.clip(init, 0, 1)
        try:
            result = minimize(objective, init, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 50, 'ftol': 1e-6})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        if time.time() - start_time > budget_s - 10:
            break
    
    sum_h = np.sum(best_h)
    scale = (n / 2) / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    best_h = np.clip(best_h, 0, 1)
    
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
