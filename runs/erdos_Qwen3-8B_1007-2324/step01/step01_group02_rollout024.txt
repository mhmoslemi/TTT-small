        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_initial)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    best_h = np.array(initial_h_values)
    best_c5 = compute_c5(best_h)
    best_n = n_initial

    start_time = time.time()
    np.random.seed(seed)

    s_values = [5, 6, 7]

            positions_sorted = np.sort(positions)
            intervals = []

                end = positions_sorted[2 * i + 1]
                intervals.append((start, end))

            if total_length == 0:
                continue
            h_values = np.zeros(n_initial)
            x_grid = np.linspace(0, 2, n_initial)
            for start, end in intervals:
                mask = (x_grid >= start) & (x_grid <= end)
                h_values[mask] = 1.0
            h_values = h_values * (1 / (np.sum(h_values) * 2.0 / n_initial))
            current_c5 = compute_c5(h_values)
