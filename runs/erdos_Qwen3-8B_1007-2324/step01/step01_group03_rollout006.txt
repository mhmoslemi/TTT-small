import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Initialize from provided values or uniform
    h_initial = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (target_sum / h_initial.sum())

    best_h = h_initial
    best_c5 = objective(best_h)
    start_time = time.time()

    # Restart loop with adaptive perturbation
    num_restarts = 5
    for _ in range(num_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Perturb current best
        perturbation = np.random.normal(0, 0.025, n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (target_sum / h_perturbed.sum())

        # Setup constraints
        cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # COBYLA initial exploration
        res = minimize(objective, h_perturbed, method='COBYLA', bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'disp': False})
        if res.success:
            h_refined = res.x
            # Trust-constr refinement
            res_refine = minimize(objective, h_refined, method='trust-constr', bounds=bounds,
                                 constraints=cons, options={'maxiter': 100, 'disp': False})
            if res_refine.success:
                current_c5 = objective(res_refine.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res_refine.x

    # Final enforce constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / best_h.sum())
    return best_h.tolist(), best_c5, n_points
