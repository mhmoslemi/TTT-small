import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n = 280  # Balance resolution and time
    dx = 2.0 / n
    
    if 'initial_h_values' in globals():
        initial_h = np.repeat(initial_h_values, 2)
        # Ensure integral constraint
        if np.abs(np.sum(initial_h) - n / 2) > 1e-5:

            initial_h *= scale
    else:

        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def constraint_func(h):
        return np.sum(np.clip(h, 0, 1)) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    # Use more robust optimization with bounds and constraints
    result = minimize(
        objective, 
        initial_h, 
        method='L-BFGS-B', 
        bounds=bounds,
