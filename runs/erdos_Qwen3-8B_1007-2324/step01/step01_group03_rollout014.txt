from scipy.optimize import minimize
import time

    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h_clipped * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    num_restarts = 10

        h_perturbed = h_initial + perturbation
        h_perturbed = adjust_h(h_perturbed)
