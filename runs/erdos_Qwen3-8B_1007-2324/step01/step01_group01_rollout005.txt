import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_integral = n_points / 2.0

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.full(n_points, 0.5)

    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_integral / current_sum
        return h_clipped * scale

    def objective(h):
        h_proj = project_h(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, mutation=(0.8, 1), recombination=0.9,
        tol=1e-8, maxiter=50, seed=seed, vectorized=True
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = h_initial
        c5_bound = objective(h_values)
        return h_values, c5_bound, n_points

    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
