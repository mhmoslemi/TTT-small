import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h_values
    if 'initial_h_values' in globals():
        initial_h = np.tile(initial_h_values, 2)

    initial_points = [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(num_starts)]
    initial_points = [np.clip(p, 0, 1) for p in initial_points]

    best_h = None
    best_c5 = float('inf')

    for h_start in initial_points:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def constraint_func(h):
            return np.sum(h) - target_sum

        constraints = [{'type': 'eq', 'fun': constraint_func}]

        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(h_start)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_start

    if best_h is None:
        best_h = initial_h
        best_c5 = objective(best_h)
