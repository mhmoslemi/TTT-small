import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    np.random.seed(seed)

    for _ in range(15):  # More variations
        h = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h)
        h *= scale
        result = minimize(lambda x: compute_c5(x), h, method='SLSQP',
                          bounds=bounds, constraints=cons, tol=1e-7)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

        if time.time() - start_time > budget_s - 1:
            break

    return (best_h, best_c5, n_points)
