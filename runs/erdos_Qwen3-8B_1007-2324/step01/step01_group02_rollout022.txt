from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 1000  # n_points
    
    s_values = [1, 2, 3, 4, 5]
    num_trials_per_s = 5  # number of candidates to try per s
    
    for s in s_values:
        # Generate multiple candidates for this s
        for _ in range(num_trials_per_s):
            # Initial points for this s
            num_points = 2 * s + 1
            initial_points = np.random.uniform(0, 2, num_points)

            def objective(points_flat):
                points = points_flat.copy()
                points.sort()

                intervals = []
                for i in range(s):
                    start = points[2*i]
                    end = points[2*i + 1]
                    intervals.append( (start, end) )
                # Compute total length S
                S = sum(end - start for start, end in intervals)
                if S == 0:
                    return np.inf  # avoid division by zero
                # Build h_values
                n_points = 1000
                dx = 2.0 / n_points
                h_values = np.zeros(n_points)
                for i in range(n_points):
                    x = i * dx
                    for start, end in intervals:
                        if start <= x < end:
                            h_values[i] = 1.0 / S
                            break
                corr = np.correlate(h_values, 1 - h_values, mode='full')
                c5 = np.max(corr) * dx
                return c5
            
            # Set bounds for each point
            bounds = [(0.0, 2.0) for _ in range(num_points)]
            
            # Initial guess
            initial_guess = initial_points.copy()
            
            # Perform optimization
            result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    # Generate h_values for the best solution
                    points = result.x.copy()
                    points.sort()
                    intervals = []
                    for i in range(s):
                        start = points[2*i]
                        end = points[2*i + 1]
                        intervals.append( (start, end) )

                            if start <= x < end:
                                h_values[i] = 1.0 / S
                                break
                    best_h = h_values
                    best_n = n_points
            # Check time budget
            if time.time() - start_time > budget_s - 1:
                break
    return (best_h, best_c5, best_n)
