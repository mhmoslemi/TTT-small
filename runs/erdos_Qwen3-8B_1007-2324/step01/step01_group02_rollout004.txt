from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_initial = len(initial_h)
    target_sum = n_initial / 2

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        return compute_c5(h_flat)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_initial)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)
    best_n = n_initial

    np.random.seed(seed)
    start_time = time.time()

    for _ in range(10):  # 10 independent trials
        # Generate random initial guess
        h_flat = np.random.rand(n_initial)
        h_flat = h_flat * target_sum / np.sum(h_flat)

        if result.success:
