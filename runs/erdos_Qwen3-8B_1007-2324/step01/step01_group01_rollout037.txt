import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use provided initial solution or uniform 0.5
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values).copy()
    else:
        initial_h = np.full(n_points, 0.5)

    def project(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = target_sum / s
        h_clamped = np.clip(h, 0.0, 1.0)
        return h_clamped * scale

    def objective(h):
        h_proj = project(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.8, 1),
            recombination=0.9,
            seed=seed,
            disp=False,
            maxiter=100
        )
    except:
        result = None

    elapsed = time.time() - start_time
    if elapsed > budget_s or result is None:
        h_values = initial_h
        h_proj = project(h_values)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        c5_bound = np.max(corr) * dx
    else:
        h_values = result.x
        h_proj = project(h_values)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        c5_bound = np.max(corr) * dx

    return h_values, c5_bound, n_points
