import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 560  # Use 560 for higher precision
    n = n_points
    dx = 2.0 / n
    target_sum = n / 2

    # Check for initial_h_values
    if 'initial_h_values' in globals():
        parent = initial_h_values
        # Scale up parent to n_points (assuming parent is 140 points)
        initial_h = np.repeat(parent, int(n // len(parent)))
    else:
        initial_h = np.full(n, 0.5)
    
    starts = [initial_h.copy()]
    np.random.seed(seed)
    # Add a couple of random starts
    for _ in range(2):
        h_rand = np.random.rand(n)
        h_rand = np.clip(h_rand, 0, 1)
        scale = target_sum / np.sum(h_rand)
        h_rand *= scale
        starts.append(h_rand)
    
    best_c5 = float('inf')
    best_h = None
    elapsed = 0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_func(h):
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    for start in starts:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        result = minimize(
            objective,
            start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    elapsed = time.time() - start_time
    if best_h is None:
        best_h = initial_h
    
    return best_h, best_c5, n_points
