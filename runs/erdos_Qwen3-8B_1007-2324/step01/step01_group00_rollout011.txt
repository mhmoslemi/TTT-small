import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    h = np.repeat(initial_h_values, 2)
    # Ensure total mass and bounds
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

    best_h = result.x
    best_c5 = result.fun
