import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 560
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        base_pattern = initial_h_values
        h_initial = np.tile(base_pattern, 4)
    else:
        h_initial = np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    result = minimize(
        objective, 
        h_initial, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_initial, objective(h_initial), n_points
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
