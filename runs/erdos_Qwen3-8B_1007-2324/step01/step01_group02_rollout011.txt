import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.reshape(-1)
        penalty = 1e3 * abs(np.sum(h) - target_sum)
        return compute_c5(h) + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    np.random.seed(seed)

    for _ in range(5):
        perturbed = initial_h.copy()
        noise = np.random.normal(0, 0.05, len(perturbed))
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        scale = target_sum / np.sum(perturbed) if np.sum(perturbed) > 0 else 1.0
        perturbed *= scale
        result = differential_evolution(
            objective, bounds, popsize=15, maxiter=50, tol=1e-6, seed=seed, strategy='best1bin'
        )

            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h

        if time.time() - start_time > budget_s - 10:

    return (best_h, best_c5, n_points)
