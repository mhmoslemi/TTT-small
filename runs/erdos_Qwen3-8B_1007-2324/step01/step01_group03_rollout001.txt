from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with provided initial values or uniform
    if 'initial_h_values' in globals():

    else:

    current_sum = h_initial.sum()
    scale = target_sum / current_sum
    h_initial = h_initial * scale
    
    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h_clipped * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    cons = [sum_constraint]
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    
    # Phase 1: COBYLA for diversity
    for _ in range(4):
        if time.time() - start_time > budget_s * 0.6:
            break

        h_perturbed = h_initial + 0.02 * np.random.normal(size=n_points)
        h_perturbed = adjust_h(h_perturbed)

        result = minimize(objective, h_perturbed, method='COBYLA', bounds=bounds, constraints=cons, options={'maxiter': 200})

            if current_c5 < best_c5:
