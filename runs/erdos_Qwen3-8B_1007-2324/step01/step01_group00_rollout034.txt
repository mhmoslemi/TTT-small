        sorted_intervals = sorted(intervals, key=lambda x: x[0])
        for i in range(1, m):
            if sorted_intervals[i-1][1] >= sorted_intervals[i][0] - 1:
                return -1
        return 0

    def sum_cons(x):
        intervals = []
        for i in range(m):
            s = int(x[2*i])
            e = int(x[2*i + 1])
            intervals.append((s, e))
        total_len = sum(e - s + 1 for s, e in intervals)
        return total_len - n_points / 2

    bounds = [(0, n_points - 1) for _ in range(2 * m)]
    constraints = (
        {'type': 'ineq', 'fun': non_overlapping},
        {'type': 'eq', 'fun': sum_cons}
    )

    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    best_x = result.x
    best_intervals = []
    for i in range(m):
        s = int(best_x[2*i])
        e = int(best_x[2*i + 1])
        best_intervals.append((s, e))
    grid = np.zeros(n_points)
    for s, e in best_intervals:
        if e >= s and s >= 0 and e < n_points:
            grid[s:e+1] = 1
    best_c5 = result.fun
    return (grid, best_c5, n_points)
