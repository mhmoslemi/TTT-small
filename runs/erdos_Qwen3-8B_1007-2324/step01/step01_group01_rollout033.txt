import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_integral = n_points / 2.0
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    np.random.seed(seed)
    
    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scale = target_integral / sum_clipped
        return h_clipped * scale
    
    def objective(h):
        h_proj = project(h)
        corr = np.correlate(h_proj, 1.0 - h_proj, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=20,
            tol=1e-8,
            mutation=(0.8, 1.0),
            recombination=0.9,
            disp=False,
            polish=True
        )
    except:
        result = None
    
    elapsed = time.time() - start_time
    if elapsed > budget_s or result is None:
        h_values = initial_h
        final_h = project(h_values)
        corr = np.correlate(final_h, 1.0 - final_h, mode='full')
        c5_bound = np.max(corr) * dx
    else:
        h_values = result.x
        final_h = project(h_values)
        corr = np.correlate(final_h, 1.0 - final_h, mode='full')
        c5_bound = np.max(corr) * dx
    
    return (final_h, c5_bound, n_points)
