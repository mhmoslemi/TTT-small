import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)

    def project(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scale = target_sum / sum_clipped
        return h_clipped * scale

    def objective(h):
        h_proj = project(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        popsize=20,
        maxiter=100,
        tol=1e-8,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        initial_h_proj = project(initial_h)
        corr = np.correlate(initial_h_proj, 1 - initial_h_proj, mode='full')
        c5 = np.max(corr) * dx
        return initial_h, c5, n_points

    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
