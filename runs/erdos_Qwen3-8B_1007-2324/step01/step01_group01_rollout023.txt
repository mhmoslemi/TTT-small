import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 560  # Higher resolution for better optimization
    dx = 2.0 / n_points
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        if len(initial_h_values) == 140:
            initial_h = np.repeat(initial_h_values, 4)
        else:
            initial_h = np.copy(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    integral_target = n_points / 2.0
    
    def objective(h):
        cell = np.clip(h, 0, 1)
        corr = np.correlate(cell, 1 - cell, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - integral_target
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.copy(initial_h), objective(initial_h), n_points
    
    h_values = np.copy(result.x)
    c5_bound = result.fun
    return h_values, c5_bound, n_points
