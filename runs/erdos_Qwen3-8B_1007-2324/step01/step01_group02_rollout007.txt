from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_initial = len(initial_h)
    target_sum = n_initial / 2

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_initial)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)
    best_n = n_initial

    start_time = time.time()

    # Generate multiple variations from initial_h
    for _ in range(10):

        perturbed = initial_h.copy()
        perturbed += np.random.normal(0, 0.05, len(perturbed))
        perturbed = np.clip(perturbed, 0, 1)
        scale = target_sum / np.sum(perturbed)
        perturbed *= scale
        h_flat = perturbed

        # Optimization with tighter tolerance
        result = minimize(objective, h_flat, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:

            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                best_n = len(current_h)
