import numpy as np

def run(seed=42, budget_s=100, **kwargs):

    n_points = 47
    dx = 2.0 / n_points

        v_clip = np.clip(v, 0, 1)
        s = v_clip.sum()

            return v_clip

            s2 = v_scaled.sum()
            if s2 > 0:
                v_scaled = (target_sum / s2) * v_scaled
        return v_scaled

    def fitness(v):
        v_fixed = repair(v)
        corr = np.correlate(v_fixed, 1 - v_fixed, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():

        h_initial = repair(h_initial)
        population = [np.clip(h_initial + np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(50)]

        population = [np.clip(np.random.uniform(0, 1, n_points), 0, 1) for _ in range(50)]

    best_fitness = float('inf')
    best_h = population[0]
    start_time = time.time()

    for _ in range(80):
