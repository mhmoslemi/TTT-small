import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    n = 280  # Higher resolution for better exploration
    
    # Generate diverse starting points
    start_points = []
    # Original upsampled initial
    h = np.repeat(initial_h_values, 2)
    start_points.append(h)
    # Random binary
    idx = np.random.choice(n, n//2, replace=False)
    h_rand = np.zeros(n)
    h_rand[idx] = 1
    start_points.append(h_rand)
    # Smoothed variant
    def smooth_h(h, sigma=1.0):
        smoothed = gaussian_filter(h, sigma)
        smoothed = np.clip(smoothed, 0, 1)
        total = np.sum(smoothed)
        scale = (n / 2) / total
        return smoothed * scale
    start_points.append(smooth_h(h))
    # Interleaved
    h_inter = np.empty(n)
    h_inter[::2] = 1
    h_inter[1::2] = 0
    start_points.append(h_inter)
    
    # Evaluate each starting point
    for h0 in start_points:
        # Normalize sum
        if np.sum(h0) != n/2:
            h0 = h0 * (n/2) / np.sum(h0)
        # Objective function
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            max_corr = np.max(corr)
            return max_corr * (2.0 / n)
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
        bounds = [(0.0, 1.0) for _ in range(n)]
        # Run optimization
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n
    
    return (best_h, best_c5, best_n)
