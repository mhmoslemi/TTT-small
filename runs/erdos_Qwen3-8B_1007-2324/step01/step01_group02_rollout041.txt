def compute_c5_from_intervals(x_vars, l_vars, n_points=1000):
    s = int(len(x_vars) / 2)
    intervals = []
    for i in range(s):
        start = x_vars[2*i]
        length = l_vars[2*i]
        end = start + length
        intervals.append( (start, end) )
    # Now, generate h_values
    x_grid = np.linspace(0, 2, n_points)
    h_values = np.zeros(n_points)
    for i in range(s):
        start, end = intervals[i]
        mask = (x_grid >= start) & (x_grid <= end)
        h_values[mask] = 1.0
    # Compute correlation
    dx = 2.0 / n_points
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    return np.max(corr) * dx
