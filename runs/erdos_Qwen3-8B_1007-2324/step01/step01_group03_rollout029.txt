import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Higher resolution for better approximation
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
    h_initial = np.clip(h_initial, 0, 1)
    current_sum = h_initial.sum()
    scale = target_sum / current_sum
    h_initial = h_initial * scale

    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h_clipped * scale

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = Bounds(0.0, 1.0)
    sum_constraint = {'type': 'eq', 'fun': lambda h: h.sum() - target_sum}

    best_c5 = float('inf')
    best_h = h_initial.copy()

    start_time = time.time()
    num_restarts = 15  # Increased number of restarts

    for _ in range(num_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = adjust_h(h_perturbed)

        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=[sum_constraint], options={'maxiter': 200})
        if result.success:
            current_h = result.x

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = adjust_h(best_h)
    return best_h.tolist(), best_c5, n_points
