def compute_c5_from_intervals(intervals, n_points=1000):
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for start, end in intervals:
        # Find indices where x is between start and end
        # Using np.searchsorted to find the indices
        left = np.searchsorted(x, start)
        right = np.searchsorted(x, end)
        h[left:right] = 1.0
    # Compute 1 - h
    one_minus_h = 1.0 - h
    # Compute full correlation
    corr = np.correlate(h, one_minus_h, mode='full')
    # Find the maximum correlation value
    max_corr = np.max(corr)
    return max_corr * dx
