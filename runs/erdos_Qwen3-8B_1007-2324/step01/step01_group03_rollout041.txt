            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    best_h = np.clip(best_h, 0, 1)
    current_sum = best_h.sum()
    if current_sum != 0:
        best_h = best_h * (target_sum / current_sum)
    
    return best_h.tolist(), best_c5, n_points
