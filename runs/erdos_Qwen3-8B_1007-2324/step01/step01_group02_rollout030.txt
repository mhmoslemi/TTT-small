from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    def objective(vars, s, n_points):
        starts = vars[:s]
        lengths = vars[s:]
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(s):
            start = starts[i]
            end = start + lengths[i]
            start_idx = max(0, int(start / dx))
            end_idx = min(n_points, int(end / dx) + 1)
            if start_idx < end_idx:
                h[start_idx:end_idx] = 1.0
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(vars, s):
        return np.sum(vars[s:]) - 1.0
    
    for s in s_values:
        n_guesses = 5

            starts = starts[idx]

        for guess in initial_guesses:

            bounds = [(0.0, 2.0) for _ in range(2*s)]
            result = minimize(lambda x: objective(x, s, 1000), guess, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: constraint(x, s)})
