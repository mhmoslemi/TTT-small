import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_initial = len(initial_h)
    target_sum = n_initial / 2
    dx = 2.0 / n_initial

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.copy()
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) > 1e-6:
            return np.inf
        return compute_c5(h)

    bounds = [(0.0, 1.0) for _ in range(n_initial)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    np.random.seed(seed)

    def perturb(h, scale=0.15):
        h = h.copy()
        noise = np.random.randn(len(h)) * scale
        h += noise
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h)
        return h * scale

    for _ in range(5):
        h = perturb(best_h)
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        if time.time() - start_time > budget_s - 1:
            break

    return (best_h, best_c5, n_initial)
