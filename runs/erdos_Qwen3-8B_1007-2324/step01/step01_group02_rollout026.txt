    for k in [5, 6, 7]:
        if time.time() - start_time > budget_s - 1:
            break
        endpoints = np.random.uniform(0, 2, 2 * k)
        def obj(endpoints):
            sorted_endpoints = np.sort(endpoints)
            intervals = []
            for i in range(k):
                st = sorted_endpoints[2*i]
                ed = sorted_endpoints[2*i + 1]
                intervals.append((st, ed))
            total_len = sum(ed - st for st, ed in intervals)
            if total_len == 0:
                return float('inf')
            x = np.linspace(0, 2, 1000)
            h = np.zeros(1000)
            for st, ed in intervals:
                mask = (x >= st) & (x <= ed)
                h[mask] = 1
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / 1000)
        bounds = [(0.0, 2.0) for _ in range(2*k)]
        result = minimize(obj, endpoints, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if result.success:
            sorted_endpts = np.sort(result.x)
            intervals = []
            for i in range(k):
                st = sorted_endpts[2*i]
                ed = sorted_endpts[2*i + 1]
                intervals.append((st, ed))
            total_len = sum(ed - st for st, ed in intervals)
            if total_len == 0:
                continue
            x = np.linspace(0, 2, 1000)
            h = np.zeros(1000)
            for st, ed in intervals:
                mask = (x >= st) & (x <= ed)
                h[mask] = 1
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / 1000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = 1000

    return (best_h, best_c5, best_n)
