import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_sum = 70.0  # since dx * sum(h) = 1 → sum(h) = 1/dx = 70

    # Initialize with the initial_h_values if available, else uniform
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    # Project the initial guess
    initial_h = project(initial_h)

    def project(h):
        sum_h = np.sum(h)
        if np.isclose(sum_h, 0):
            return h
        scale = target_sum / sum_h
        projected = h * scale
        projected = np.clip(projected, 0.0, 1.0)
        return projected

    def objective(h):
        h_proj = project(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with early termination if time runs out
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='rand',
        popsize=20,
        tol=1e-8,
        mutation=(0.8, 1.0),
        recombination=0.9,
        maxiter=100,
        workers=1,
        polish=True,
        disp=False,
        x0=initial_h
    )

    elapsed = time.time() - start_time

        h_values = initial_h
        c5_bound = objective(h_values)
