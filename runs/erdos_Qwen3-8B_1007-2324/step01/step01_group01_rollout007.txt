from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280  # Increased resolution for better convergence
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        if len(h_initial) != n_points:
            h_initial = np.repeat(h_initial, 2)[:n_points]  # Scale to match new n
    else:
        h_initial = np.full(n_points, 0.5)

    def objective(h):
        # Clip to [0,1] and ensure sum is integral_target
        h_clip = np.clip(h, 0, 1)
        # Safe projection to meet integral constraint
        sum_clip = np.sum(h_clip)
        if abs(sum_clip - integral_target) > 1e-5:
            scale = integral_target / sum_clip
            h_clip = h_clip * scale
        # Compute correlation
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(np.clip(h, 0, 1)) - integral_target}]

    # Fallback if optimization stalls
    fallback_h = np.full(n_points, 0.5)
    best_c5 = objective(fallback_h)
    best_h = fallback_h.copy()

    # Run optimization
    result = minimize(
        objective, 
        h_initial, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    # Update best solution
    h_values = result.x
    c5_bound = result.fun
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_values
