import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    t_start = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.repeat(initial_h_values, 2)
    else:
        initial_h = np.full(n_points, 0.5)

    np.random.seed(seed)
    starts = []
    starts.append(initial_h)
    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = initial_h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        s = np.sum(perturbed)
        perturbed *= target_sum / s
        starts.append(perturbed)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def con(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': con}]

    best_h, best_c = initial_h, obj(initial_h)

    for start in starts:
        res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
        if res.success and res.fun < best_c:
            best_c = res.fun
            best_h = res.x
        if time.time() - t_start > budget_s:
            break

    return best_h, best_c, n_points
