from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    h_initial = initial_h_values
    n_points = len(h_initial)

        else:
            i += 1
    m = len(intervals)

    initial_guess = np.array([float(s) for (s, e) in intervals for _ in range(2)], dtype=np.float64)

        h = np.zeros(n_points, dtype=np.float64)
        for i in range(m):
            s = int(np.round(x[2*i]))
            e = int(np.round(x[2*i + 1]))
            if s <= e:
                h[s:e+1] = 1.0
        return h
    
    def objective(x):
        h = generate_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

            e = x[2*i + 1]
            total += e - s + 1.0
        return total - (n_points / 2)

                e_i = x[2*i + 1]
                s_j = x[2*j]

        return 0.0

            e = x[2*i + 1]
            if s > e:

        return 0.0
