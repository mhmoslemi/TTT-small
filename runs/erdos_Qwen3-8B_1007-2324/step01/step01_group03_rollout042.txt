import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 94  # Use double-resolution for better optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from provided values and adapt if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        # Fallback to uniform distribution
        h_initial = np.zeros(n_points)
        h_initial[:n_points//2] = 1.0

    h_initial = np.clip(h_initial, 0, 1)
    current_sum = h_initial.sum()
    scale = target_sum / current_sum
    h_initial = h_initial * scale

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = LinearConstraint(np.ones(n_points), target_sum, target_sum)
    constraints = [sum_constraint]

    best_c5 = float('inf')
    best_h = h_initial.copy()

    start_time = time.time()
    num_restarts = 10  # More restarts to escape local optima

    for _ in range(num_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
        # Perturb with adaptive noise
        perturbation = np.random.normal(0, 0.03, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = h_perturbed.sum()
        scale = target_sum / current_sum
        h_perturbed = h_perturbed * scale

        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    # Final enforcement of constraints
    best_h = np.clip(best_h, 0, 1)
    current_sum = best_h.sum()
    scale = target_sum / current_sum
    best_h = best_h * scale

    return best_h.tolist(), best_c5, n_points
