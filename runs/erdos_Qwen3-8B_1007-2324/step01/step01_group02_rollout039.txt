import time

def run(seed=42, budget_s=100, **kwargs):

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_initial)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)
    best_n = n_initial
    start_time = time.time()

    variations = []
    for _ in range(10):

        noise = np.random.normal(0, 0.05, len(perturbed))
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        scale = target_sum / np.sum(perturbed)
        perturbed *= scale
        variations.append(perturbed)

    for variation in variations:
        h_flat = variation
        result = minimize(objective, h_flat, method='Nelder-Mead', bounds=bounds, constraints=cons, tol=1e-6)
