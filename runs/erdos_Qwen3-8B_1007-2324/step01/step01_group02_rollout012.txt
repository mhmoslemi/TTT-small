import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n = len(initial_h)
    target_sum = n / 2

    def objective(h_flat):
        h = np.array(h_flat)
        if np.sum(h) == 0:
            return float('inf')
        scale = target_sum / np.sum(h)
        h_scaled = h * scale
        dx = 2.0 / len(h_scaled)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n)]

    np.random.seed(seed)
    start_time = time.time()

    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        maxiter=200, popsize=15, mutation=(0.5, 1),
        recombination=0.9, tol=1e-6, seed=seed,
        polish=True, vectorized=True
    )

    best_h = result.x
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time

    return (best_h, best_c5, n)
