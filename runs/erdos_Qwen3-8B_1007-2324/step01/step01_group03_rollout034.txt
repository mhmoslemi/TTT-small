        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        n = len(h)
        lag = (max_idx - (n - 1))
        grad = np.zeros(n)
        for i in range(n):
            idx = i + lag
            if 0 <= idx < n:
                grad[i] = (1 - h[idx]) * dx
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    cons = [sum_constraint]
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    num_restarts = 15
    
    for _ in range(num_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = adjust_h(h_perturbed)
        
        result = minimize(objective, h_perturbed, method='trust-constr', bounds=bounds, constraints=cons, jac=grad_obj, options={'maxiter': 300})
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    best_h = adjust_h(best_h)
    return best_h.tolist(), best_c5, n_points
