        for i in range(m):
            start = x[2*i]
            end = x[2*i + 1]
            intervals.append((start, end))
        grid = np.zeros(n_points)
        for s, e in intervals:
            if s <= e:
                grid[s:e+1] = 1
        corr = np.correlate(grid, 1 - grid, mode='full')
        return np.max(corr) * (2.0 / n_points)
    cons = []
    for i in range(m):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - x[2*i]})
    for i in range(m-1):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - x[2*(i+1)] + 1})
    def total_length(x):
        return np.sum([x[2*i+1] - x[2*i] + 1 for i in range(m)]) - n_points / 2
    cons.append({'type': 'eq', 'fun': total_length})
    result = minimize(objective, variables, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    optimized_intervals = []
    for i in range(m):
        s = result.x[2*i]
        e = result.x[2*i + 1]
        optimized_intervals.append((s, e))
    h_values = np.zeros(n_points)
    for s, e in optimized_intervals:
        h_values[s:e+1] = 1
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * (2.0 / n_points)
    return (h_values, c5_bound, n_points)
