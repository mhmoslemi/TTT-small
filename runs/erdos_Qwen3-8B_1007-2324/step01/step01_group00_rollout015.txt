import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 140
    start_points = []
    
    h_initial = initial_h_values.copy()
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

    start_points.append(h_smoothed)
