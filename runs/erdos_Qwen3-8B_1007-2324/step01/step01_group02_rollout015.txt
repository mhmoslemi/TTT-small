import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_flat):
        h = h_flat.reshape(-1)
        c5 = compute_c5(h)
        penalty = 1000 * np.abs(np.sum(h) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), 
        recombination=0.8, seed=seed
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        return (initial_h, compute_c5(initial_h), n_points)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
