import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    target_integral = n_points / 2
    
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.full(n_points, 0.5)
    
    def obj(x):
        x_proj = np.clip(x, 0, 1)
        s = np.sum(x_proj)
        if s == 0:
            return float('inf')
        scale = target_integral / s
        x_scaled = x_proj * scale
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0., 1.) for _ in range(n_points)]
    result = differential_evolution(obj, bounds, strategy='best1bin', 
                                    maxiter=200, popsize=15, tol=1e-6, 
                                    mutation=(0.7, 1.5), recombination=0.8, 
                                    seed=seed, disp=False)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_best = h
        c5 = obj(h)
    else:
        h_best = result.x
        c5 = result.fun
    
    return h_best, c5, n_points
