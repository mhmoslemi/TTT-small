from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    def clip_and_scale(h):

        if s == 0:
            return np.full(n_points, 1.0)
        return np.clip(h, 0, 1) * (integral_target / s)

    initial_h = clip_and_scale(initial_h)
    np.random.seed(seed)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint(h):
        return np.sum(h) - integral_target

    def feasible_h(h):
        return clip_and_scale(h)

    starts.append(feasible_h(initial_h * 1.1))

    starts.append(feasible_h(np.roll(initial_h, 1)))

    best_c5 = float('inf')
    best_h = initial_h.copy()

    for start in starts:
        try:
