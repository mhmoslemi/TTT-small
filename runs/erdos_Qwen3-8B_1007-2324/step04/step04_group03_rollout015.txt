import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_guesses = [np.array(initial_h_values)]
    else:
        initial_guesses = []

    # Centered hump: 140 values at 1.0
    centered = np.zeros(n_points)
    start = (n_points - 140) // 2
    end = start + 140
    centered[start:end] = 1.0
    initial_guesses.append(centered)

    # Dual hump: two humps at edges
    dual = np.zeros(n_points)
    dual[0:70] = 1.0
    dual[210:280] = 1.0
    initial_guesses.append(dual)

    # Add perturbed versions of patterns
    for guess in initial_guesses[:2]:
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = guess + noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed *= (integral_target / sum_perturbed)
        initial_guesses.append(perturbed)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    constraints_full = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        sum_guess = np.sum(guess)
        if sum_guess > 0:
            guess *= (integral_target / sum_guess)
        result = minimize(objective, guess, method='SLSQP', bounds=bounds_full, constraints=constraints_full,
                          options={'maxiter': 200, 'ftol': 1e-8, 'disp': False})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5

        if elapsed > budget_s:
