            x_new = np.linspace(0, 2, n_points)
            initial_h = np.interp(x_new, x_initial, initial_h)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint}]

    num_starts = 5
    starts = [initial_h.copy() for _ in range(num_starts)]
    for i in range(num_starts):
        noise = np.random.normal(0, 0.05, n_points)
        starts[i] += noise
        starts[i] = np.clip(starts[i], 0, 1)
        if np.sum(starts[i]) > 0:
            starts[i] *= (target_sum / np.sum(starts[i]))

    best_h, best_c5 = initial_h, objective(initial_h)
    for start in starts:
        result = minimize(objective, start, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-8, 'disp': False})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5, best_h = current_c5, result.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
