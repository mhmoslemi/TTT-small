    base_perturb = np.clip(base_perturb, -0.2, 0.2)
    base = initial_h + base_perturb
    base = np.clip(base, 0, 1)
    sum_base = np.sum(base)
    if sum_base > 0:
        base *= (integral_target / sum_base)
    starts.append(base)
    
    for i in range(10):
        noise = np.random.normal(0, 0.05, n_points)
        noise = np.clip(noise, -0.15, 0.15)
        start = initial_h + noise
        start = np.clip(start, 0, 1)
        sum_start = np.sum(start)
        if sum_start > 0:
            start *= (integral_target / sum_start)
        starts.append(start)
    
    def project(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return np.full(n_points, 0.5)
        return h_clip * (integral_target / sum_clip)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    for start in starts:
        try:
            result = minimize(
