import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)

    def objective(h):
        n = len(h)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(np.abs(corr)) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    starts = []
    starts.append(initial_h)
    np.random.seed(seed)
    noise = np.random.normal(0, 0.1, n_points)
    p_start = initial_h + noise
    p_start = np.clip(p_start, 0, 1)
    p_sum = np.sum(p_start)
    if p_sum > 0:
        p_start *= (integral_target / p_sum)
    starts.append(p_start)
    b_start = np.zeros(n_points)
    b_start[::2] = 1
    b_start = np.clip(b_start, 0, 1)
    b_sum = np.sum(b_start)
    if b_sum > 0:
        b_start *= (integral_target / b_sum)
    starts.append(b_start)
    for i in range(4):
        np.random.seed(seed + i)
        r_start = np.random.uniform(0, 1, n_points)
        r_start = np.clip(r_start, 0, 1)
        r_sum = np.sum(r_start)
        if r_sum > 0:
            r_start *= (integral_target / r_sum)
        starts.append(r_start)

    for start in starts:
        try:

                objective,
                start,
                method='L-BFGS-B',
