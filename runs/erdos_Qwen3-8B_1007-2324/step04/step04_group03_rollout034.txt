from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    # Generate centered hump
    centered = np.zeros(n_points)
    mid = n_points // 2
    start_idx = mid - 70
    end_idx = mid + 70
    centered[start_idx:end_idx] = 1.0
    initial_guesses.append(centered)

    # Generate dual hump
    dual = np.zeros(n_points)
    dual[0:70] = 1.0
    dual[140:210] = 1.0
    initial_guesses.append(dual)

    # Perturb parent solution if available
    if 'initial_h_values' in globals():
        initial_parent = np.array(initial_h_values)
        perturbed = initial_parent + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed *= (target_sum / sum_perturbed)
        initial_guesses.append(perturbed)

    # Objective function with sum constraint penalty
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e4 * abs(sum_h - target_sum)
        return c5 + penalty

    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with initial guesses
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=30, maxiter=100, tol=1e-8, 
                                   seed=seed, args=())

    best_h = result.x
    best_c5 = objective(best_h)

    # Adjust to meet sum constraint
    if np.abs(np.sum(best_h) - target_sum) > 1e-3:
        best_h = best_h * (target_sum / np.sum(best_h))
        best_h = np.clip(best_h, 0, 1)
        best_c5 = objective(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
