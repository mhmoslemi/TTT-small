import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    best_c5 = float('inf')
    best_h = np.zeros(n_points, dtype=np.int64)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.int64)
    else:
        h = np.zeros(n_points, dtype=np.int64)
        h[:140] = 1
        np.random.shuffle(h)

    def evaluate(h):
        nonlocal best_c5
        h_float = h.astype(np.float64)
        corr = np.correlate(h_float, 1 - h_float, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h[:]
        return c5

    evaluate(h)
    iterations = 0
    while time.time() - start_time < budget_s:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i, j = np.random.choice(zeros), np.random.choice(ones)
        h[i], h[j] = 1, 0
        iterations += 1
        evaluate(h)

    best_h = np.clip(best_h, 0, 1).astype(float)
    return best_h, best_c5, n_points
