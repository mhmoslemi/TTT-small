import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        n = len(h)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_clip
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_clip
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(np.abs(corr)) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    best_c5 = result.fun
    best_h = result.x

    # Add block pattern start
    b_start = np.zeros(n_points)
    b_start[::2] = 1
    b_start = np.clip(b_start, 0, 1)
    b_sum = np.sum(b_start)
    if b_sum > 0:
        b_start *= (integral_target / b_sum)
    result_block = minimize(
        objective,
        b_start,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 600, 'ftol': 1e-9, 'disp': False}
    )

        best_h = result_block.x
