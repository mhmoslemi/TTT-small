import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def compute_c5(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - integral_target

    # Generate diverse initial solutions
    initial_solutions = []

    # Single hump
    m = n_points // 2
    start = (n_points - m) // 2
    h_single = np.zeros(n_points)
    h_single[start:start + m] = 1.0
    initial_solutions.append(h_single)

    # Dual hump
    m = (n_points // 2) // 2
    start1 = (n_points - m) // 2 - m
    end1 = start1 + m
    start2 = (n_points - m) // 2 + m
    end2 = start2 + m
    h_dual = np.zeros(n_points)
    h_dual[start1:end1] = 1.0
    h_dual[start2:end2] = 1.0
    initial_solutions.append(h_dual)

    # Perturbed parent if available
    if 'initial_h_values' in globals():
        h_perturb = np.array(initial_h_values)
        noise = np.random.normal(0, 0.05, n_points)
        h_perturb += noise
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        if sum_h > 0:
            h_perturb *= (integral_target / sum_h)
        initial_solutions.append(h_perturb)

    # Random-like initial solution
    h_rand = np.random.rand(n_points)
    h_rand = np.clip(h_rand, 0, 1)
    sum_rand = np.sum(h_rand)
    if sum_rand > 0:
        h_rand *= (integral_target / sum_rand)
    initial_solutions.append(h_rand)

    best_h = None
