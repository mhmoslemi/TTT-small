import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 560
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = np.full(n_points, 0.5)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    if np.sum(initial_h) > 0:
        initial_h *= (integral_target / np.sum(initial_h))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        h_sum = np.sum(h_clip)
        if h_sum > 0:
            h_rescale = h_clip * (integral_target / h_sum)
        else:
            h_rescale = np.zeros_like(h_clip)
        n = len(h_rescale)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_rescale
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_rescale
        corr = np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(o_pad))).real
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, tol=1e-5, mutation=(0.5, 1.0),
        recombination=0.9, maxiter=100,
        x0=initial_h, workers=1, vectorized=True
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return result.x, result.fun, n_points
    return result.x, result.fun, n_points
