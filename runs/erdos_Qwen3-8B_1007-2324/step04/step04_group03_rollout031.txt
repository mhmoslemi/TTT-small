        h_perturb += noise
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        sum_perturb = np.sum(h_perturb)
        if sum_perturb > 0:
            h_perturb *= (target_sum / sum_perturb)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        initial_candidates.append(h_perturb)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')

    for start in initial_candidates:
        try:
            result = differential_evolution(
                lambda h: _objective(h, x, dx, target_sum),
                bounds,
                strategy='best1bin',
                maxiter=50,
                popsize=20,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                polish=True,
                options={'mutation_scale': 0.5, 'init': start}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Fallback to initial_h if no improvement found
    if best_h is None:
        best_h = initial_h
        best_c5 = _objective(best_h, x, dx, target_sum)

    # Ensure sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = np.clip(best_h * (target_sum / sum_h), 0.0, 1.0)
    # Final C5 evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points

def _objective(h, x, dx, target_sum):
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-5:
        factor = target_sum / sum_h
        h = np.clip(h * factor, 0.0, 1.0)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
