import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.round(np.clip(initial_h_values, 0, 1)).astype(int)
    else:
        initial_h = np.zeros(n_points, dtype=int)
        initial_h[:integral_target] = 1
        np.random.shuffle(initial_h)

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)

    if np.sum(best_h) != integral_target:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            best_h[ones[0]] = 0
            best_h[zeros[0]] = 1
        best_c5 = evaluate_erdos_solution(best_h, dx)

    while time.time() - start_time < budget_s - 0.5:
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if len(zeros) > 0 and len(ones) > 0:
            i, j = np.random.choice(zeros), np.random.choice(ones)
            new_h = best_h.copy()
            new_h[i] ^= 1
            new_h[j] ^= 1
            current_c5 = evaluate_erdos_solution(new_h, dx)
            if current_c5 < best_c5:
                best_h = new_h
                best_c5 = current_c5

    return best_h, best_c5, n_points
