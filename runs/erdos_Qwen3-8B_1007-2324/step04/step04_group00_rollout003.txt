from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    try:
        initial_h = np.array(initial_h_values)

    def objective(h):

        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 10000 * (np.sum(h) - integral_target) ** 2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    def generate_start(h0, seed=None):

            np.random.seed(seed)
        noise = np.random.normal(0, 0.05, n_points)
        shifted = np.roll(h0 + noise, np.random.randint(-20, 20))
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            scaled = (integral_target / sum_shifted) * shifted
        else:
            scaled = np.full(n_points, 0.5)
        return scaled

    starts = [initial_h]
    for i in range(20):
        starts.append(generate_start(initial_h, seed + i))

    for start in starts:

            result = minimize(
