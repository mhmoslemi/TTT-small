        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if np.abs(current_sum - target_sum) > 1e-8:
            scale = target_sum / current_sum
            h_scaled = h * scale
            h_final = np.clip(h_scaled, 0, 1)
        else:
            h_final = h

        n = n_points
        len_corr = 2 * n - 1
        pad_size = len_corr - n
        h_padded = np.zeros(len_corr, dtype=np.complex128)
        h_padded[:n] = h_final
        g = 1 - h_final
        g_padded = np.zeros(len_corr, dtype=np.complex128)
        g_padded[:n] = g

        H = fftpack.fft(h_padded)
        G = fftpack.fft(g_padded)
        product = H * np.conj(G)
        corr_fft = fftpack.ifft(product)
        corr = np.real(corr_fft)

        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1exp',
        mutation=0.9,
        crossover=0.7,
        popsize=50,
        maxiter=50,
        tol=1e-8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_c5 = result.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
