def project_constrained(h):
    """Project vector h onto feasible set while enforcing constraints"""
    n = len(h)
    # First, enforce sum constraint
    s = np.sum(h)
    if abs(s - integral_target) > 1e-8:
        scale = integral_target / s if abs(s) > 1e-8 else n / 2  # Arbitrary fallback
        h = h * scale
    
    # Enforce value bounds
    h = np.clip(h, 0, 1)
    
    # Handle cases where clipping fails to meet sum constraint
    s_clipped = np.sum(h)
    if abs(s_clipped - integral_target) > 1e-6:
        need_to_add = (integral_target - s_clipped) / n
        if need_to_add > 1e-6:  # Avoid tiny adjustments causing large displacement
            add_to_idx = np.random.randint(n)
            h[add_to_idx] = min(h[add_to_idx] + need_to_add, 1.0)
        # Small probability for inverse operation
        elif need_to_add < -1e-6:
            remove_from_idx = np.random.randint(n)
            h[remove_from_idx] = max(h[remove_from_idx] + need_to_add, 0.0)
    
    return h
