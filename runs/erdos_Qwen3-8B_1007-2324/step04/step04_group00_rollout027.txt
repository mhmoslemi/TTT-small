        i = np.random.randint(n_points)
        h_candidate[i] ^= 1
        # Repair sum
        current_sum = np.sum(h_candidate)
        if current_sum != integral_target:
            if current_sum > integral_target:
                # Flip a 1 to 0
                j = np.random.choice(np.where(h_candidate == 1)[0])
                h_candidate[j] = 0
            else:
                # Flip a 0 to 1
                j = np.random.choice(np.where(h_candidate == 0)[0])
                h_candidate[j] = 1
        # Compute C5
        current_c5 = evaluate_erdos_solution(h_candidate, dx)
        if current_c5 < best_c5:
            best_h = h_candidate
            best_c5 = current_c5

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
