import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values)
    else:
        initial = np.full(n_points, 0.5)

    def repair(h):
        h = np.array(h)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scale = target_sum / np.sum(h)
        h = h * scale
        return np.clip(h, 0, 1)

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=50, tol=1e-8, mutation=(0.8, 1.2), recombination=0.7,
        maxiter=100, seed=seed
    )

    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
