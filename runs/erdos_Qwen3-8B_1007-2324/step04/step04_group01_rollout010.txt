import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 240  # Reduced for faster optimization
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    initial_h_values = np.array([0.5] * n_points)  # Default if no parent
    
    # Generate diverse initial starts
    num_starts = 30
    starts = []
    
    np.random.seed(seed)
    # First start: large random perturbation
    perturb = np.random.uniform(-0.3, 0.3, n_points)
    parent_with_noise = initial_h_values + perturb
    parent_with_noise = np.clip(parent_with_noise, 0, 1)
    sum_perturbed = np.sum(parent_with_noise)
    if sum_perturbed > 0:
        parent_with_noise *= (integral_target / sum_perturbed)
    starts.append(parent_with_noise)
    
    for i in range(num_starts - 1):
        np.random.seed(seed + i)
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        start_perturbed = initial_h_values + perturb
        start_perturbed = np.clip(start_perturbed, 0, 1)
        sum_perturbed = np.sum(start_perturbed)
        if sum_perturbed > 0:
            start_perturbed *= (integral_target / sum_perturbed)
        starts.append(start_perturbed)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
