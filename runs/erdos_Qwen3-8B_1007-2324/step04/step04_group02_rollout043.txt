        max_overlap = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(h) - target_sum)
        return max_overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=300,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
