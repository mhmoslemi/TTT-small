import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def constraint_full(h):
        return np.sum(h) - integral_target

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    constraints_full = [{'type': 'eq', 'fun': constraint_full}]
    
    starts = [initial_h]
    np.random.seed(seed)
    np.random.shuffle(initial_h)
    starts.append(initial_h)
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        noisy_start = initial_h + noise
        noisy_start = np.clip(noisy_start, 0, 1)
        sum_noisy = np.sum(noisy_start)
        if sum_noisy > 0:
            noisy_start *= (integral_target / sum_noisy)
        starts.append(noisy_start)
    
    best_h = None
    best_c5 = float('inf')
    for start in starts:
        try:
            result = minimize(objective, start, method='L-BFGS-B', bounds=bounds_full, 
                              constraints=constraints_full, options={'maxiter': 500, 'ftol': 1e-8, 'disp': False})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
