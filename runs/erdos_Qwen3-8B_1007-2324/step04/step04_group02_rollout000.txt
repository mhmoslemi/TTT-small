import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def clip_and_scale(h):
        s = np.sum(h)
        if s == 0:
            return np.full(n_points, 1.0)
        return np.clip(h, 0, 1) * (integral_target / s)

    def objective(h):
        h_clip = clip_and_scale(h)
        h_pad = np.zeros(2 * n_points - 1)
        h_pad[:n_points] = h_clip
        o_pad = np.zeros(2 * n_points - 1)
        o_pad[:n_points] = 1 - h_clip
        corr = np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(o_pad))).real
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    np.random.seed(seed)
    starts = [clip_and_scale(initial_h)]
    starts.append(clip_and_scale(initial_h * (1 + np.random.normal(0, 0.05, n_points))))
    starts.append(clip_and_scale(np.random.uniform(0, 1, n_points)))

    for start in starts:
        try:
            res = minimize(objective, start, method='L-BFGS-B', bounds=bounds, 
                           constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target},

        except:
            pass
