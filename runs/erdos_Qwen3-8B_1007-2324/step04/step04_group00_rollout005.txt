import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    # Generate diverse starts
    def generate_starts(h0, seeds):
        np.random.seed(seed)
        starts = [h0.copy()]
        for s in seeds:
            np.random.seed(s)
            noise = np.random.normal(0, 0.05, n_points)
            shifted = np.roll(h0 + noise, np.random.randint(-20, 20))
            shifted = np.clip(shifted, 0, 1)
            sum_shifted = np.sum(shifted)
            if sum_shifted > 0:
                shifted *= (integral_target / sum_shifted)
            starts.append(shifted)
        return starts

    seeds = [seed + i for i in range(15)]
    starts = generate_starts(initial_h, seeds)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
            )

                    best_h = result.x
        except:
            pass
