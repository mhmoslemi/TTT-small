import numpy as np
import cma
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    def project(h):
        h_clip = np.clip(h, 0.0, 1.0)
        total_sum = np.sum(h_clip)
        if total_sum < 1e-10:
            h_proj = 0.5 * np.ones_like(h_clip)
        else:
            scale = integral_target / total_sum
            h_proj = np.clip(h_clip * scale, 0.0, 1.0)
        return h_proj
    
    def fitness(x):
        h_proj = project(x)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    if 'initial_h_values' in globals():
        initial_mean = initial_h
        sigma = 0.4
    else:
        initial_mean = np.random.uniform(0.0, 1.0, n_points)
        sigma = 0.4
    
    es = cma.CMAEvolutionStrategy(initial_mean, sigma, {'bounds': bounds, 'popsize': 20})
    
    best_h = initial_mean
    best_c5 = fitness(best_h)
    
    while time.time() - start_time < budget_s:
        es.optimize(fitness)
        best_x = es.best_x
        best_fit = es.best_f
        if best_fit < best_c5:
            best_c5 = best_fit
            best_h = best_x
    
    best_h_proj = project(best_h)
    corr = np.correlate(best_h_proj, 1 - best_h_proj, mode='full')
    best_c5 = np.max(corr) * dx
