import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (target_sum / np.sum(initial_h))

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h_scaled = h * (target_sum / sum_h)
        n = len(h_scaled)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_scaled
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_scaled
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(np.abs(corr)) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-6,
        maxiter=1000,
        seed=seed,
        workers=1
    )

    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
