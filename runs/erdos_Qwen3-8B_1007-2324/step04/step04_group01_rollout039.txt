    integral_target = n_points / 2

        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return np.zeros_like(h_basis)
        scale = (integral_target) / sum_clip
        return h_clip * scale

    for i in range(4):
        t = np.linspace(0, 2, n_points)
        wave = 0.5 + 0.5 * np.sin(2 * np.pi * (t * i) / 4)

        candidates.append(projected)

        h_clip = np.clip(h, 0.0, 1.0)

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
