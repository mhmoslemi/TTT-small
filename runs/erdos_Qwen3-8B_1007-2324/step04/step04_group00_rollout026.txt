    a_values = np.linspace(0.5, 1.0, 100)
    best_c5 = float('inf')
    best_a = 0.5

    for a in a_values:
        h = build_h(a)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_a = a

    def objective(a):
        h = build_h(a)
        return evaluate_c5(h)

    result = minimize(
        objective,
        x0=[best_a],
        method='L-BFGS-B',
        bounds=[(0.5, 1.0)],
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    if result.success:
        best_a = result.x[0]
        best_h = build_h(best_a)
        best_c5 = result.fun
    else:
        best_h = build_h(best_a)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
