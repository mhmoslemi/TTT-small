from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair_h(h):

        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-6:
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        adjustment = target_sum - current_sum
        h[-1] += adjustment
        h = np.clip(h, 0, 1)
        return h

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        n = len(h)
        h_fft = np.fft.fft(h)
        g = 1.0 - h
        g_fft = np.fft.fft(g)
        g_reversed = g[::-1]
        g_reversed_fft = np.fft.fft(g_reversed)
        corr_fft = h_fft * g_reversed_fft
        corr = np.fft.ifft(corr_fft).real
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h):
        h_repaired = repair_h(h)
        return compute_c5(h_repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    starts = []
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        noisy = initial_h + noise
        noisy = np.clip(noisy, 0, 1)
        sum_noisy = np.sum(noisy)
        noisy = noisy * (target_sum / sum_noisy) if sum_noisy > 0 else noisy
        starts.append(noisy)

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=50,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=True,
        disp=False,
        vectorize=True
    )

    elapsed = time.time() - start_time
    h_best = result.x if elapsed <= budget_s else initial_h
    h_best = repair_h(h_best)
    final_c5 = compute_c5(h_best)
