import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate initial guesses
    initial_guesses = []

    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        # Perturb parent's initial_h_values
        parent_h = np.array(initial_h_values)
        # Add some noise
        noise = np.random.normal(0, 0.05, n_points)
        noisy_h = parent_h + noise
        noisy_h = np.clip(noisy_h, 0, 1)
        # Normalize sum
        sum_noise = np.sum(noisy_h)
        if sum_noise > 0:
            noisy_h *= target_sum / sum_noise
        initial_guesses.append(noisy_h)
    else:
        # Default initial guess: uniform
        initial_guesses.append(np.full(n_points, 0.5))

    # Generate centered hump
    centered_h = np.zeros(n_points)
    mid = n_points // 2
    length = int(target_sum)
    start_idx = mid - length // 2
    end_idx = mid + length // 2
    if end_idx > n_points:
        end_idx = n_points
    centered_h[start_idx:end_idx] = 1.0
    sum_centered = np.sum(centered_h)
    if sum_centered > 0:
        centered_h *= target_sum / sum_centered
    initial_guesses.append(centered_h)

    # Generate dual hump
    dual_h = np.zeros(n_points)
    # First hump
    length = int(target_sum / 2)
    start1 = length
    end1 = start1 + length
    dual_h[start1:end1] = 1.0
    # Second hump
    start2 = n_points - length - 1
    end2 = start2 + length
    dual_h[start2:end2] = 1.0
    sum_dual = np.sum(dual_h)
    if sum_dual > 0:
        dual_h *= target_sum / sum_dual
    initial_guesses.append(dual_h)

    # Generate another random initial guess
    random_h = np.random.rand(n_points)
    sum_random = np.sum(random_h)
    if sum_random > 0:
        random_h *= target_sum / sum_random
    initial_guesses.append(random_h)

    # Now, use differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h = np.clip(h, 0, 1)
        S = np.sum(h)
        if S == 0:
            return np.inf
        scale = target_sum / S
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Run differential evolution
    result = differential_evolution(objective, bounds, popsize=30, strategy='best1bin', 
                                   maxiter=100, tol=1e-8, mutation=(0.5, 1), recombination=0.9, 
                                   polish=True, init='random', x0=initial_guesses, 
                                   seed=seed, callback=None, disp=False)

    # Check elapsed time and return the best solution
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Fallback to the parent's initial solution if not found
        if 'initial_h_values' in globals():
            h_values = np.array(initial_h_values)
        else:
            h_values = np.full(n_points, 0.5)
    else:
        if result.success:
            h_values = result.x
        else:
            # Fallback
            if 'initial_h_values' in globals():
                h_values = np.array(initial_h_values)
            else:
                h_values = np.full(n_points, 0.5)
    
    # Compute c5_bound
    h_values = np.clip(h_values, 0, 1)
    S = np.sum(h_values)
    scale = target_sum / S
    h_scaled = h_values * scale
    corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
    c5_bound = np.max(corr) * dx

    return h_values, c5_bound, n_points

import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    initial_guesses = []

    # Generate initial guesses
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        noise = np.random.normal(0, 0.05, n_points)
        noisy_h = parent_h + noise
        noisy_h = np.clip(noisy_h, 0, 1)
        sum_noise = np.sum(noisy_h)
        if sum_noise > 0:
            noisy_h *= target_sum / sum_noise
        initial_guesses.append(noisy_h)
    else:
        initial_guesses.append(np.full(n_points, 0.5))

    # Centered hump
    centered_h = np.zeros(n_points)
    length = int(target_sum)
    start = (n_points - length) // 2
    end = start + length
    centered_h[start:end] = 1.0
    sum_centered = np.sum(centered_h)
    if sum_centered > 0:
        centered_h *= target_sum / sum_centered
    initial_guesses.append(centered_h)

    # Dual hump
    dual_h = np.zeros(n_points)
    length = int(target_sum / 2)
    start1 = length
    end1 = start1 + length
    dual_h[start1:end1] = 1.0
    start2 = n_points - length - 1
    end2 = start2 + length
    dual_h[start2:end2] = 1.0
    sum_dual = np.sum(dual_h)
    if sum_dual > 0:
        dual_h *= target_sum / sum_dual
    initial_guesses.append(dual_h)

    # Random initial guess
    random_h = np.random.rand(n_points)
    sum_random = np.sum(random_h)
    if sum_random > 0:
        random_h *= target_sum / sum_random
    initial_guesses.append(random_h)

    # Objective function
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        S = np.sum(h_clip)
        if S == 0:
            return np.inf  # to avoid division by zero
        scale = target_sum / S
        h_scaled = h_clip * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        popsize=30,
        strategy='best1bin',
        maxiter=100,  # can be adjusted for time
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.9,
        polish=True,
        x0=initial_guesses,
        seed=seed,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        # Fallback
        if 'initial_h_values' in globals():
            h_values = np.array(initial_h_values)
        else:
            h_values = np.full(n_points, 0.5)
    else:
        if result.success:
            h_values = result.x
        else:
            h_values = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    # Compute final c5_bound
    h_values = np.clip(h_values, 0, 1)
    S = np.sum(h_values)
    if S == 0:
        c5_bound = np.inf
    else:
        scale = target_sum / S
        h_scaled = h_values * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5_bound = np.max(corr) * dx

    return h_values, c5_bound, n_points
