import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair(h):
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) < 1e-6:
            return np.clip(h, 0, 1)
        scale = target_sum / sum_h
        h_scaled = h * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    starts = [initial_h.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        noisy = initial_h + noise
        noisy = np.clip(noisy, 0, 1)
        sum_noisy = np.sum(noisy)
        if sum_noisy > 1e-6:
            noisy *= (target_sum / sum_noisy)
        starts.append(noisy)

    best_h = None
    best_c5 = float('inf')

    for start in starts:
        try:
            result = minimize(
                fun=objective,
                x0=start,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
