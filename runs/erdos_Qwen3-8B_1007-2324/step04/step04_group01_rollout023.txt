import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    initial_h = np.array([])  # Placeholder, actual value loaded by the harness
    if len(initial_h) == 0:
        initial_h = np.full(n_points, 0.5)
    
    num_starts = 30  # Increased number of starts for better exploration
    starts = []
    
    np.random.seed(seed)
    for i in range(num_starts):
        if i == 0:
            # Primary start: Gaussian noise perturbation
            perturb = np.random.normal(0, 0.15, n_points)
        else:
            # Diverse perturbations: random swap, noise, and sinusoidal modulation
            if i % 3 == 1:
                perturb = np.random.normal(0, 0.05, n_points)
            elif i % 3 == 2:
                perturb = np.random.choice([-1, 1], size=n_points) * 0.15
            else:
                perturb = np.sin(2 * np.pi * np.linspace(0, 1, n_points)) * 0.1
        start_perturbed = initial_h + perturb
        start_perturbed = np.clip(start_perturbed, 0, 1)
        sum_perturbed = np.sum(start_perturbed)
        if sum_perturbed > 0:
            start_perturbed *= (integral_target / sum_perturbed)
        starts.append(start_perturbed)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    best_iter = 0
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_iter = len(starts)
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
