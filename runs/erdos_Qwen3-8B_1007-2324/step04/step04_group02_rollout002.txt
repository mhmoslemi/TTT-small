    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)

    def projective_clip(h):
        h_clip = np.clip(h, 0, 1)
        s = np.sum(h_clip)
        if s == 0:
            return np.full_like(h_clip, 1.0)
        return h_clip * (integral_target / s)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        n = len(h_clip)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_clip
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_clip
        corr = np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(o_pad))).real
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    starts = [initial_h]
    np.random.seed(seed)

        start = initial_h + perturbation
        start = np.clip(start, 0, 1)
        s = np.sum(start)
        if s > 0:
            start *= (integral_target / s)
        starts.append(start)

    best_c5 = float('inf')
    best_h = initial_h.copy()

    for start in starts:
