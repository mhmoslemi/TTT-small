def run(seed=42, budget_s=100, **kwargs):

    dual[start1:end1] = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_solutions = []

    # Generate centered hump
    centered = np.zeros(n_points)
    spread = int(0.4 * n_points)
    start = n_points // 2 - spread // 2
    end = start + spread
    if end < n_points:
        centered[start:end] = 1.0
    total = np.sum(centered)
    if total > 0:
        centered = centered * (integral_target / total)
    initial_solutions.append(centered)

    # Generate dual hump
    dual = np.zeros(n_points)
    spread = int(0.2 * n_points)
    start1 = n_points // 2 - spread - spread // 2
    end1 = start1 + spread
    start2 = n_points // 2 - spread // 2
    end2 = start2 + spread
    if end1 < n_points:
        dual[start1:end1] = 1.0
    if end2 < n_points:
        dual[start2:end2] = 1.0
    total = np.sum(dual)
    if total > 0:
        dual = dual * (integral_target / total)
    initial_solutions.append(dual)

    # Use parent solution if available
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        perturbed = parent + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed *= (integral_target / sum_perturbed)
        initial_solutions.append(perturbed)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')

        return np.sum(h) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_points)]
