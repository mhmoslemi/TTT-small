import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def projective_clip(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s == 0:
            return np.full_like(h_clipped, 1)
        return h_clipped * (integral_target / s)

    initial_h = projective_clip(initial_h)

    starts = [initial_h]

    np.random.seed(seed)
    for _ in range(4):
        noise = np.random.normal(0, 0.1, n_points)
        p_start = initial_h + noise
        p_start = projective_clip(p_start)
        starts.append(p_start)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        n = len(h)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_clip
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_clip
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(np.abs(corr)) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = initial_h.copy()

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                projected_h = projective_clip(result.x)
                current_c5 = objective(projected_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = projected_h.copy()
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
