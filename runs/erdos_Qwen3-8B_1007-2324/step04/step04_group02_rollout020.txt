import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def project(h):
        h_clip = np.clip(h, 0, 1)
        if np.sum(h_clip) == 0:
            return np.zeros_like(h_clip)
        return h_clip * (target / np.sum(h_clip))

    def objective(h):
        h_proj = project(h)
        n = len(h_proj)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_proj
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_proj
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda h: np.sum(project(h)) - target
    }]

    best_h = initial_h.copy()
    best_c5 = float('inf')

    np.random.seed(seed)
    starts = [initial_h]
    for _ in range(4):
        noise = np.random.normal(0, 0.1, n_points)
        h_start = initial_h + noise
        h_start = project(h_start)
        starts.append(h_start)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
