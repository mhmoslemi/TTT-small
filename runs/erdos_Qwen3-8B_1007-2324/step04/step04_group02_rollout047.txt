        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(np.abs(corr)) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
