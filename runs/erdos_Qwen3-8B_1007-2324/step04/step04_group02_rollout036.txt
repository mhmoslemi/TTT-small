    p_start = np.clip(p_start, 0, 1)
    p_sum = np.sum(p_start)
    if p_sum > 0:

    starts.append(p_start)

    b_start[::4] = 1
    b_start = np.clip(b_start, 0, 1)

    for i in range(5):
        np.random.seed(seed + i)
        r_start = np.random.uniform(0, 1, n_points)
        r_sum = np.sum(r_start)
        if r_sum > 0:
            r_start *= (integral_target / r_sum)
        starts.append(r_start)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        h_pad = np.zeros(2 * n_points - 1)
        h_pad[:n_points] = h_clip
        o_pad = np.zeros(2 * n_points - 1)

        corr = np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(o_pad))).real

    bounds = [(0.0, 1.0) for _ in range(n_points)]
