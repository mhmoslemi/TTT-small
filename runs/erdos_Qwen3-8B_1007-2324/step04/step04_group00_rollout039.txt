import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    x_values = np.linspace(0, 2, n_points)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    # Try regular pulse configurations for M=2,3,4
    for M in [2, 3, 4]:
        b = 2.0 / M
        h_vals = np.zeros(n_points)
        for i in range(M):
            s = i * b
            e = (i + 1) * b
            for idx in range(n_points):
                x = x_values[idx]
                if s <= x <= e:
                    if x <= s + b/2:
                        val = 2 * (x - s) / b
                    else:
                        val = 2 * (s + b - x) / b
                    h_vals[idx] = val
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals
