    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        noisy_start = initial_h + noise
        noisy_start = np.clip(noisy_start, 0, 1)
        sum_noisy = np.sum(noisy_start)
        if sum_noisy > 0:
            noisy_start *= (integral_target / sum_noisy)
        starts.append(noisy_start)

    best_h = None
    best_c5 = float('inf')
    for start in starts:
        try:
            def objective(h):
                h_clip = np.clip(h, 0, 1)
                corr = np.correlate(h_clip, 1 - h_clip, mode='full')
                return np.max(corr) * dx

            def constraint(h):
                return np.sum(h) - integral_target

            bounds = [(0.0, 1.0) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint}]
            result = minimize(
                objective, start, method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_h = result.x
                current_sum = np.sum(current_h)
                if abs(current_sum - integral_target) > 1e-4:
                    scale = integral_target / current_sum
                    current_h = current_h * scale
                    current_h = np.clip(current_h, 0, 1)
                h_clip = np.clip(current_h, 0, 1)
                corr = np.correlate(h_clip, 1 - h_clip, mode='full')
                current_c5 = np.max(corr) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
