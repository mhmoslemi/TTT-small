import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_h = None
    best_c5 = float('inf')

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - target_sum

    def evaluate_solution(h):
        return np.max(np.correlate(np.clip(h, 0, 1), 1 - np.clip(h, 0, 1), mode='full')) * dx

    # Generate initial guesses
    initial_guesses = []
    # Single hump
    single_h = np.zeros(n_points)
    start_single = n_points // 2 - 70
    single_h[start_single:start_single+140] = 1.0
    initial_guesses.append(single_h)
    # Dual hump
    dual_h = np.zeros(n_points)
    dual_h[0:70] = 1.0
    dual_h[210:280] = 1.0
    initial_guesses.append(dual_h)
    # Perturb parent if available
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        for _ in range(3):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(parent_h + noise, 0, 1)
            sum_perturbed = np.sum(perturbed)
            if sum_perturbed > 0:
                perturbed *= (target_sum / sum_perturbed)
            initial_guesses.append(perturbed)

    for guess in initial_guesses:
        guess = np.clip(guess, 0, 1)
        sum_guess = np.sum(guess)
        if sum_guess > 0:
            guess = guess * (target_sum / sum_guess)
        try:
            # Differential evolution with penalty
            def penalty_func(h):
                return objective(h) + 1e6 * (np.sum(h) - target_sum)**2
            result = scipy.optimize.differential_evolution(
                penalty_func, bounds=[(0,1)]*n_points, popsize=20, maxiter=100,
                tol=1e-4, seed=seed, strategy='best1bin', mutation=(0.5, 1), recombination=0.8
            )
            if result.success:
                current_h = result.x
                current_c5 = evaluate_solution(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:
            pass
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Fallback to parent solution if needed
    if best_h is None and 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_c5 = evaluate_solution(best_h)
    elif best_h is None:
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_solution(best_h)

    return best_h, best_c5, n_points
