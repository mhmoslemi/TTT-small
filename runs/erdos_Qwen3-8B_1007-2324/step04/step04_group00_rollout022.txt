    initial_h = initial_h * (integral_target / np.sum(initial_h))

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        c5 = np.max(corr) * dx
        # Penalize deviation from integral constraint
        integral = np.sum(h_clip) * dx

        return c5 + penalty

    best_c5 = float('inf')
    best_h = initial_h.copy()
