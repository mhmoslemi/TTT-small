import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 128
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        penalty = 1e4 * abs(np.sum(h) - integral_target)
        return np.max(corr) * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Differential Evolution
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    # SLSQP Refinement
    def constraint_full(h):
        return np.sum(h) - integral_target

    constraints = [{'type': 'eq', 'fun': constraint_full}]
    result_slsqp = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    # Final scaling
    scale = integral_target / np.sum(best_h)
    best_h = np.clip(best_h * scale, 0, 1)

    return best_h, best_c5, n_points
