    for i in range(population_size):
        h_candidate = initial_population[i]
        h_clipped = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h != 0:
            h_scaled = h_clipped * (target_sum / sum_h)
        else:
            h_scaled = np.zeros_like(h_clipped)
        initial_population[i] = h_scaled
    
    # Run differential evolution
    result = differential_evolution(objective, bounds, popsize=population_size, strategy='best1bin', 
                                   mutation=(0.5, 1), recombination=0.7, tol=1e-8, 
                                   maxiter=50, seed=seed, x0=initial_population)
    
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
