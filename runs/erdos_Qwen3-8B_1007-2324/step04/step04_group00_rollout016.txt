import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        indices = np.random.choice(n_points, integral_target, replace=False)
        initial_h = np.zeros(n_points, dtype=np.float64)
        initial_h[indices] = 1.0

    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)

    def find_zero_one(h):
        zero_mask = (h == 0.0)
        one_mask = (h == 1.0)
        if np.any(zero_mask) and np.any(one_mask):
            i = np.random.choice(np.where(zero_mask)[0])
            j = np.random.choice(np.where(one_mask)[0])
            return i, j
        return None, None

    while time.time() - start_time < budget_s:
        i, j = find_zero_one(h)
        if i is None or j is None:
            break
        old_i = h[i]
        old_j = h[j]

            h[i] = old_i
            h[j] = old_j

    if elapsed > budget_s:

    return best_h, best_c5, n_points
