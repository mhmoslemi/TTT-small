def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280

    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

    perturbations = []

        noise = np.random.normal(0, 0.15, n_points)
        start = initial_h + noise
        start = np.clip(start, 0, 1)
        sum_start = np.sum(start)
        if sum_start > 0:
            start *= (integral_target / sum_start)
        perturbations.append(start)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr) * dx
        sum_h = np.sum(h_clip)
        penalty = 1e4 * abs(sum_h - integral_target)
        return max_corr + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h
    best_c5 = objective(best_h)
    
    for start in perturbations:

                bounds,

            if result.success:
