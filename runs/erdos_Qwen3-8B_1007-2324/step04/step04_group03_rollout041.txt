import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    target_integral = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Create a structured initial guess
        m = 5
        l = 1.0 / m
        spacing = l / 2
        h = np.zeros(n_points)
        pos = 0
        while pos + l < 2:
            start_idx = int(pos / dx)
            end_idx = int((pos + l) / dx)
            h[start_idx:end_idx] = 1
            pos += l + spacing

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return 0
        h_scaled = h_clip * (target_integral / sum_h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False,
    )

    best_h = result.x
    elapsed = time.time() - start_time

    if elapsed > budget_s:
        return initial_h, 0.0, n_points

    best_h_clipped = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h_clipped)
    if sum_h == 0:
        best_c5 = 0.0
    else:
        best_h_scaled = best_h_clipped * (target_integral / sum_h)
        corr = np.correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
        best_c5 = np.max(corr) * dx

    return best_h, best_c5, n_points
