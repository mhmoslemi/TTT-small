        result = differential_evolution(objective, bounds, popsize=30, maxiter=100, tol=1e-8, seed=seed)
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
