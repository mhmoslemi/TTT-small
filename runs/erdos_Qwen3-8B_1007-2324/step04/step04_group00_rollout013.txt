import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(np.float64)
        while h.sum() != integral_target:
            if h.sum() < integral_target:
                idx = np.random.choice(np.where(h == 0)[0])
                h[idx] = 1
            else:
                idx = np.random.choice(np.where(h == 1)[0])
                h[idx] = 0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    max_iter = int(budget_s / (1.5 * n_points))

        elapsed = time.time() - start_time

            break
        h = best_h.copy()
        i = np.random.randint(n_points)
        h[i] = 1 - h[i]
        sum_h = h.sum()
        if sum_h != integral_target:
            if sum_h < integral_target:
                j = np.random.choice(np.where(h == 0)[0])
                h[j] = 1
            else:
                j = np.random.choice(np.where(h == 1)[0])
                h[j] = 0
        current_c5 = evaluate_erdos_solution(h, dx)
