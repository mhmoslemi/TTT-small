from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def build_h(h, dx, n_points):
        h = np.clip(h, 0, 1)
        return np.array(h)

    def compute_overlap(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_overlap(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = compute_overlap(initial_h)

    # Generate diverse starts with more varied perturbations
    def generate_start(h, seed):
        np.random.seed(seed)
        noise = np.random.normal(0, 0.05, n_points)
        shifted = np.roll(h + noise, np.random.randint(-20, 20))
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    starts = [initial_h]
    for i in range(15):  # Increase number of starts
        starts.append(generate_start(initial_h, seed + i))

    for start in starts:
        try:
            result = minimize(

                start,
                method='SLSQP',

                constraints=constraints,
