        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def constraint_full(h):
        return np.sum(h) - integral_target

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    constraints_full = [{'type': 'eq', 'fun': constraint_full}]

    np.random.seed(seed)
    starts = [initial_h]
    for _ in range(5):
        noisy_start = initial_h + np.random.normal(0, 0.04, n_points)
        noisy_start = np.clip(noisy_start, 0, 1)
        sum_noisy = np.sum(noisy_start)
        if sum_noisy > 0:
            noisy_start *= (integral_target / sum_noisy)
        starts.append(noisy_start)

    best_h = None
    best_c5 = float('inf')
    for start in starts:
        try:
            result = minimize(objective, start, method='SLSQP', bounds=bounds_full, 
                              constraints=constraints_full, options={'maxiter': 300, 'ftol': 1e-8, 'disp': False})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
