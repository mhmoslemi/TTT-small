import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

                idx = np.random.randint(n_points)
                initial_h[idx] = 1

                idx = np.random.randint(n_points)
                initial_h[idx] = 0
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        h = best_h.copy()
        h[i] ^= 1
        h[j] ^= 1
        current_sum = np.sum(h)
        if current_sum != target_sum:
            if current_sum > target_sum:
                ones = np.where(h == 1)[0]
                if len(ones) > 0:
                    idx = np.random.choice(ones)
                    h[idx] = 0
            else:
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    idx = np.random.choice(zeros)
                    h[idx] = 1
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
