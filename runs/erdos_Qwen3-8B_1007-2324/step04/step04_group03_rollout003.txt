import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-5:
            return h
        scale = integral_target / current_sum
        scaled_h = h * scale
        scaled_h = np.clip(scaled_h, 0, 1)
        return scaled_h

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')

    # Generate diverse initial guesses
    starts = []
    # Perturb initial guess
    for _ in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        start = np.clip(initial_h + noise, 0, 1)
        start_sum = np.sum(start)
        if start_sum > 0:
            start *= (integral_target / start_sum)
        starts.append(start)

    # Add Golomb-inspired pattern
    def generate_golomb_like():
        pattern = np.zeros(n_points)
        step = n_points // 4
        for i in range(0, n_points, step):
            pattern[i] = 1.0
        pattern = pattern / np.sum(pattern) * integral_target
        return pattern
    starts.append(generate_golomb_like())

    for start in starts:
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                popsize=5,
                tol=1e-8,
                maxiter=100,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return (best_h, best_c5, n_points)
