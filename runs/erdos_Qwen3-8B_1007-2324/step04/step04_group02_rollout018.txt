import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = np.array([])
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def clip_and_scale(h):
        s = np.sum(h)
        scale = integral_target / max(1e-10, s)
        h_scaled = h * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    def objective(h):
        h_clip = clip_and_scale(h)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    starts = [clip_and_scale(initial_h)]
    np.random.seed(seed)
    starts.append(clip_and_scale(initial_h + np.random.normal(0, 0.1, n_points)))
    starts.append(clip_and_scale(np.zeros(n_points)))
    starts.append(clip_and_scale(np.random.uniform(0, 1, n_points)))
    starts.append(clip_and_scale(np.kron(np.random.choice([0, 1], size=n_points//4), np.ones(4))))
    starts.append(clip_and_scale(np.kron(np.random.choice([0, 1], size=n_points//8), np.ones(8))))

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
