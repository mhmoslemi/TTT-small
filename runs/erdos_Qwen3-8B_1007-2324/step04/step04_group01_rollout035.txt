import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    def project(h_vec):
        sum_h = np.sum(h_vec)
        if sum_h == 0:
            return np.zeros_like(h_vec)
        scaling = integral_target / sum_h
        return np.clip(h_vec * scaling, 0, 1)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    h = np.clip(h, 0, 1)
    h = project(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    learning_rate = 0.05
    noise_scale = 0.1
    min_noise_scale = 0.01
    plateau_steps = 0
    max_plateau = 5
    max_steps = 1000
    
    for _ in range(max_steps):
        if time.time() - start_time > budget_s:
            break
        current_c5 = compute_c5(h)
        eps = 1e-4
        h_plus = np.clip(h + eps, 0, 1)
        h_plus = project(h_plus)
        h_minus = np.clip(h - eps, 0, 1)
        h_minus = project(h_minus)
        obj_plus = compute_c5(h_plus)
        obj_minus = compute_c5(h_minus)
        grad = (obj_plus - obj_minus) / (2 * eps)
        
        noise = np.random.normal(0, noise_scale, n_points)
        h_new = h + learning_rate * grad + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = project(h_new)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            best_c5 = new_c5
            plateau_steps = 0
        else:
            plateau_steps += 1
        
        if plateau_steps >= max_plateau:
            noise_scale = max(min_noise_scale, noise_scale * 0.9)
            plateau_steps = 0
    
    h = project(h)
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return h, best_c5, n_points
