import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return float('inf')
        h_norm = h_clip * (integral_target / sum_h)
        n = len(h_norm)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_norm
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_norm
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(corr) * dx

    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=15, tol=1e-8, maxiter=500,
                                   seed=seed, vectorized=True)

    best_h = result.x
    best_c5 = result.fun

    if elapsed > budget_s:
        return best_h, best_c5, n_points
