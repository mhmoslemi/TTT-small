def run(seed=42, budget_s=100, **kwargs):

    if 'initial_h_values' in globals():

    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)

    starts = []
    starts.append(initial_h)

    # Perturbed start
    np.random.seed(seed)
    noise = np.random.normal(0, 0.1, n_points)
    p_start = initial_h + noise
    p_start = np.clip(p_start, 0, 1)
    p_sum = np.sum(p_start)
    if p_sum > 0:
        p_start *= (integral_target / p_sum)
    starts.append(p_start)

    # Block pattern start
    b_start = np.zeros(n_points)
    b_start[::2] = 1
    b_start = np.clip(b_start, 0, 1)
    b_sum = np.sum(b_start)
    if b_sum > 0:
        b_start *= (integral_target / b_sum)
    starts.append(b_start)

    # Random starts
    for i in range(4):
        np.random.seed(seed + i)
        r_start = np.random.uniform(0, 1, n_points)
        r_sum = np.sum(r_start)
        if r_sum > 0:
            r_start *= (integral_target / r_sum)
        starts.append(r_start)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return float('inf')
        scale = integral_target / sum_clip
        h_scaled = h_clip * scale
        n = len(h_scaled)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_scaled
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_scaled
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}

                current_c5 = result.fun
