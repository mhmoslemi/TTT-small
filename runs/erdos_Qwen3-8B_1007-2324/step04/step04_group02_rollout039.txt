import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        n = len(h)
        h_pad = np.zeros(2 * n - 1)
        h_pad[:n] = h_clip
        o_pad = np.zeros(2 * n - 1)
        o_pad[:n] = 1 - h_clip
        fft_h = np.fft.fft(h_pad)
        fft_o = np.fft.fft(o_pad)
        corr = np.fft.ifft(fft_h * np.conj(fft_o)).real
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    starts = [initial_h]
    np.random.seed(seed)
    starts.append(initial_h + np.random.normal(0, 0.1, n_points))
    starts.append(np.zeros(n_points))
    starts.append(np.full(n_points, 0.5))
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    for start in starts:
        try:
            result = minimize(objective, start, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
