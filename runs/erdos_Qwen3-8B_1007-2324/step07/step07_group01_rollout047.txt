import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.full(n_points, 0.5)

    # Normalize and clip to [0,1]
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h *= (integral_target / sum_h)

    best_c5 = float('inf')
    best_h = h.copy()

    # Objective function
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    # Diversity operations
    def apply_shift(h):
        shift = np.random.randint(-n_points//2, n_points//2)
        return np.roll(h, shift)
    def apply_scale(h):
        factor = np.random.choice([0.9, 1.1])
        return h * factor
    def apply_overflip(h):
        start = np.random.randint(0, n_points - 2)
        end = np.random.randint(start + 2, n_points)
        return np.concatenate([h[:start], 1 - h[start:end], h[end:]])

    trials = 0
    max_trials = 100
    while trials < max_trials and time.time() - start_time < budget_s - 1:
        trials += 1
        # Apply diversity operation
        if np.random.rand() < 0.6:
            h = apply_shift(h)
        elif np.random.rand() < 0.3:
            h = apply_scale(h)
        else:
            h = apply_overflip(h)
        # Clip and normalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:

        try:

                    best_c5 = current_c5
                    best_h = result.x.copy()
                    if best_c5 <= 0.380876:
                        break
        except:
            pass

    return best_h, best_c5, n_points
