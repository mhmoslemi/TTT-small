import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = np.array(initial_h_values).copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (integral_target / np.sum(initial_h))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    def generate_candidates(base):
        candidates = []
        candidates.append(base)
        candidates.append(1.0 - base)
        candidates.append(np.roll(base, 50))
        candidates.append(np.roll(base, 100))
        candidates.append(np.roll(base, 150))
        return [np.clip(c, 0, 1) * (integral_target / np.sum(np.clip(c, 0, 1))) for c in candidates]

    candidates = generate_candidates(initial_h)

    for h_start in candidates:
        h_start = h_start.copy()
        if np.sum(h_start) > 0:
            h_start *= (integral_target / np.sum(h_start))
        try:
            result = minimize(objective, h_start, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    trials = 15
    for t in range(trials):

        h_start = initial_h.copy()

        noise = np.random.normal(0, 0.02, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
