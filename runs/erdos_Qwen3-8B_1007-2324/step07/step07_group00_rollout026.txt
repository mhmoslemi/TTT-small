import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Create a piecewise function: 1 in first half, 0 in second
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 1.0
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h)
        initial_h = initial_h * (integral_target / sum_h)
        initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return 1e6  # Penalize zero sum
        scale = integral_target / sum_h
        h_scaled = h_clip * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use differential evolution with best1 strategy
    result = differential_evolution(objective, bounds, strategy='best1',
                                   maxiter=200, popsize=20, tol=1e-7,
                                   mutation=(0.5, 1), recombination=0.8,
                                   seed=seed, disp=False)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    h_best = result.x
    c5_best = result.fun
    return h_best, c5_best, n_points
