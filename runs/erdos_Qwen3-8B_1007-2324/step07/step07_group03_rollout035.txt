        result = minimize(obj_560, h_start, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5_560:
                best_c5_560 = current_c5
                best_h_560 = result.x

    final_h = best_h_560 if best_c5_560 < best_c5_280 else best_h_280
    final_c5 = best_c5_560 if best_c5_560 < best_c5_280 else best_c5_280
    n_points_final = n_points_560 if best_c5_560 < best_c5_280 else n_points_280

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_560, best_c5_560, n_points_560

    return final_h, final_c5, n_points_final
