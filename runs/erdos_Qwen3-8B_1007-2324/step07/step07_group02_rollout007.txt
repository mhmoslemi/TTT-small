import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def repair_integral(h):
        s = np.sum(h)
        if abs(s - integral_target) < 1e-6:
            return h
        needed = integral_target - s
        weights = np.abs(h - h.mean()) + 1e-10
        cum_weights = weights / weights.sum()
        indices = np.random.choice(n_points, size=10, p=cum_weights)
        for i in indices:
            if abs(needed) < 1e-6:
                break
            max_val = min(1.0, h[i] + 0.1)
            min_val = max(0.0, h[i] - 0.1)
            if max_val == min_val:
                h[i] += needed / len(indices)
                needed = 0
                break
            else:
                adjust = min(needed, max_val - h[i]) if max_val > h[i] else min(needed, h[i] - min_val)
                h[i] += adjust
                needed -= adjust
        return np.clip(h, 0, 1)

    def perturb(h, seed):
        np.random.seed(seed)
        h = h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        return repair_integral(h)

    starts = [initial_h]
    for i in range(5):
        starts.append(perturb(initial_h, seed + i))
        starts.append(perturb(initial_h, seed + i + 10))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    for start in starts:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            res = minimize(objective, start, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        except:
            pass

    return best_h, best_c5, n_points
