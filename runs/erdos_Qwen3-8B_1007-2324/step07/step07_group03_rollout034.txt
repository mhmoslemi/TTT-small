    constraints_base = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_base}]
    
    for t in range(trials_base):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_start = initial_h_base.copy()
        shift = np.random.uniform(-n_points_base/2, n_points_base/2)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.03, n_points_base)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target_base / sum_start)
        try:
            result = minimize(
                objective_base,
                h_start,
                method='SLSQP',
                bounds=bounds_base,
                constraints=constraints_base,
                options={'maxiter': max_iter_base, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    # Refined resolution optimization
    if 'initial_h_values' in globals():
        initial_h_refined = np.tile(initial_h_base, 2)
    else:
        initial_h_refined = np.full(n_points_refined, 0.5)
    
    trials_refined = 5
    max_iter_refined = 200
    
    def objective_refined(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_refined
    
    bounds_refined = [(0.0, 1.0) for _ in range(n_points_refined)]
    constraints_refined = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_refined}]
    
    for t in range(trials_refined):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t + trials_base)
        h_start = initial_h_refined.copy()
        shift = np.random.uniform(-n_points_refined/2, n_points_refined/2)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.01, n_points_refined)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target_refined / sum_start)
        try:
            result = minimize(
                objective_refined,
                h_start,
                method='SLSQP',
                bounds=bounds_refined,
                constraints=constraints_refined,
                options={'maxiter': max_iter_refined, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    if best_h is None:
        best_h = initial_h_base.copy()
        best_c5 = objective_base(initial_h_base)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
