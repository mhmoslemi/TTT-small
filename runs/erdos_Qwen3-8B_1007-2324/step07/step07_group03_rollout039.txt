        if i == 0:
            prev_h = h
        else:
            prev_h = best_h
        m = len(prev_h)
        assert n_points == 2 * m
        current_h = np.repeat(prev_h, 2)

        # Add random noise and shift
        noise_scale = 0.03 / (2 ** i)
        np.random.seed(seed + i)
        h_guess = current_h + np.random.normal(0, noise_scale, n_points)
        h_guess = np.clip(h_guess, 0, 1)

        # Nelder-Mead phase
        def objective(h):
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            return np.max(corr) * (2.0 / n_points)

        result_nelder = minimize(
            objective,
            h_guess,
            method='Nelder-Mead',
            options={'xatol': 1e-4, 'maxiter': 100, 'disp': False}
        )

        # SLSQP phase
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        result_slsqp = minimize(
            objective,
            result_nelder.x,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6, 'disp': False}
        )

        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x.copy()

        # Time check again
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Fallback if no solution found
    if best_h is None:
        if 'initial_h_values' in globals():
            return initial_h_values, float('inf'), 280
        else:
            return np.full(280, 0.5), float('inf'), 280

    # Verify integral constraint
    sum_h = np.sum(best_h)
    assert np.isclose(sum_h, len(best_h) / 2), "Integral not preserved"
    return (best_h, best_c5, len(best_h))
