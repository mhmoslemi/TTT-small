    dx_interpolated = 2.0 / n_points_interpolated
    integral_target_interpolated = n_points_interpolated / 2

    original_x = np.linspace(0, 2, n_points)
    new_x = np.linspace(0, 2, n_points_interpolated)
    interpolated_h = np.interp(new_x, original_x, best_h_280)
    interpolated_h_scaled = interpolated_h * 0.5

    def objective_560(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_interpolated

    constraints_560 = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_interpolated}]
    best_h_560 = interpolated_h_scaled.copy()
    best_c5_560 = objective_560(best_h_560)
    max_iter_560 = 50
    trials_560 = 5

    for t in range(trials_560):
        if time.time() - start_time > budget_s - 20:
            break
        np.random.seed(seed + 10 + t)
        h_start = best_h_560.copy()
        shift = np.random.uniform(-30, 30)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.005, n_points_interpolated)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target_interpolated / sum_start)
        try:
            result = minimize(objective_560, h_start, method='SLSQP', bounds=[(0.0, 1.0)]*n_points_interpolated,
                              constraints=constraints_560, options={'maxiter': max_iter_560})
            if result.success and result.fun < best_c5_560:
                best_c5_560 = result.fun
                best_h_560 = result.x
        except:
            pass

    best_c5 = min(best_c5_280, best_c5_560)
    best_h = best_h_560 if best_c5 == best_c5_560 else best_h_280
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points_interpolated
    return best_h, best_c5, n_points_interpolated
