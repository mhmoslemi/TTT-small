import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 280

    integral_target = n_points // 2  # 140

            return h.copy()

            if abs(needed) < 1e-8:
                break
            max_val = min(1.0, h_adjusted[idx] + 0.1)
            min_val = max(0.0, h_adjusted[idx] - 0.1)
            if max_val == min_val:
                h_adjusted[idx] += needed / len(adjustment_indices)
                needed = 0.0
                break

                space_size = max_val - min_val
                dl = min(needed, space_size)
                h_adjusted[idx] = h_adjusted[idx] + dl / space_size
