            h = np.clip(h, 0.0, 1.0)
        return h.copy()

    # Initialize h
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    current_sum = np.sum(initial_h)
    if abs(current_sum - integral_target) > 1e-5:
        initial_h = repair_integral(initial_h)

    # Define objective function
    def objective(h):
        h_repaired = repair_integral(h)
        h_clip = np.clip(h_repaired, 0.0, 1.0)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    # Bounds for DE
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        seed=seed
    )

    # Extract best solution
    best_h = result.x
    best_c5 = objective(best_h)

    # Ensure time constraint
    current_time = time.time() - start_time
    if current_time > budget_s - 1:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
