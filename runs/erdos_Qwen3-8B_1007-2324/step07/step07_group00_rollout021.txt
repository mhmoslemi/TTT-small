import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (integral_target / np.sum(initial_h))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    trials = 20
    max_iter = 500

    generating_shifts = [1, 2, 3, 4, -1, -2, -3, -4, 5, -5, 6, -6, 7, -7, 8, -8]

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()
        shift = np.random.choice(generating_shifts)
        h_start = np.roll(h_start, shift)
        noise = np.random.normal(0, 0.02, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        h_start *= (integral_target / sum_start) if sum_start > 0 else 1.0
        try:
            result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
