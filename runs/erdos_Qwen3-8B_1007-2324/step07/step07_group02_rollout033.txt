```python
import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair_to_feasible(h):
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-5:
            return h
        delta = integral_target - current_sum
        indices = np.random.choice(n_points, size=10, replace=False)
        for i in indices:
            if delta == 0:
                break
            max_adj = 1 - h[i]
            min_adj = 0 - h[i]
            if delta > 0:
                amount = min(max_adj, delta)
            else:
                amount = max(min_adj, delta)
            h[i] += amount
            delta -= amount
        h = np.clip(h, 0, 1)
        return h

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = repair_to_feasible(h)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = repair_to_feasible(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=50,
        maxiter=1500,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        tol=1e-6
    )

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return result.x, result.fun, n_points
    return result.x, result.fun, n_points
```
