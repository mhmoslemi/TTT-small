from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    def repair_integral(h):
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-6:
            return h.copy()
        needed = integral_target - current_sum
        weights = np.abs(h - h.min()) + 1e-10
        cum_weights = weights / weights.sum()
        indices = np.random.choice(n_points, size=10, p=cum_weights, replace=False)
        h_adj = h.copy()
        for idx in indices:
            if abs(needed) < 1e-6:
                break
            max_val = min(1.0, h_adj[idx] + 0.1)
            min_val = max(0.0, h_adj[idx] - 0.1)
            if max_val == min_val:
                h_adj[idx] += needed / len(indices)
                needed = 0.0
                break
            else:
                space = max_val - min_val
                dl = min(needed, space)
                h_adj[idx] = h_adj[idx] + dl / space
                needed -= dl
        return np.clip(h_adj, 0, 1)
