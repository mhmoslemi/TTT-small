    best_h_fine = h_fine_initial.copy()
    best_c5_fine = objective(best_h_fine)
    bounds_fine = [(0.0, 1.0) for _ in range(n_points_fine)]
    constraints_fine = [{'type': 'eq', 'fun': constraint}]

    trials_fine = 5
    for t in range(trials_fine):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + trials_coarse + t)
        h_start = h_fine_initial.copy()
        shift = np.random.uniform(-50, 50)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.01, n_points_fine)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (n_points_fine / 2) / sum_start
        try:
            result = minimize(objective, h_start, method='SLSQP', bounds=bounds_fine, constraints=constraints_fine,
                              options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_fine:
                    best_c5_fine = current_c5
                    best_h_fine = result.x
        except:
            pass

    if best_c5_coarse < best_c5_fine:
        return best_h_coarse, best_c5_coarse, n_points_initial
    else:
        return best_h_fine, best_c5_fine, n_points_fine
