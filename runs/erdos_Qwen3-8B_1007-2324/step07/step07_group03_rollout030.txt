        if sum_start_slsqp > 0:
            h_start_slsqp *= (integral_target / sum_start_slsqp)

        result_slsqp = minimize(objective, h_start_slsqp, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8})

        current_c5 = result_slsqp.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_slsqp.x
            best_n = n_points
            current_h = result_slsqp.x

    return best_h, best_c5, best_n
