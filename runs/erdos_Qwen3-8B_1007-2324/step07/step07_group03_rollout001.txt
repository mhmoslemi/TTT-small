import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    def optimize(n, initial_h, dx, integral_target, budget):
        np.random.seed(seed)
        h_start = initial_h.copy()
        h_start = np.roll(h_start, np.random.randint(-n//4, n//4))
        noise = np.random.normal(0, 0.015, n)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        h_start *= (integral_target / sum_start) if sum_start > 0 else 1
        result = minimize(objective, h_start, method='Nelder-Mead', bounds=bounds, constraints=constraints, options={'xatol': 1e-8, 'maxiter': 300})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        return best_c5, best_h

    for _ in range(2):
        if time.time() - start_time > budget_s - 1:
            break
        h_refined = np.tile(initial_h, 2) * 0.5
        n_refined = len(h_refined)
        dx_refined = 2.0 / n_refined
        integral_target_refined = n_refined / 2
        if n_refined == n_points:
            continue
        optimized_c5, optimized_h = optimize(n_refined, h_refined, dx_refined, integral_target_refined, budget_s - (time.time() - start_time))
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            best_h = optimized_h
        if n_refined > n_points:
            break

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
