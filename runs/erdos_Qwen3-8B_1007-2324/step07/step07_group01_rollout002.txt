import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize with parent pattern or uniform distribution
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (integral_target / np.sum(initial_h))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Generate diverse pattern candidates
    def generate_patterns():
        def sin_pattern():
            return 0.5 * np.sin(np.linspace(0, np.pi, n_points)) + 0.5
        def cos_pattern():
            return 0.5 * np.cos(np.linspace(0, 2*np.pi, n_points)) + 0.5
        def rect_pattern():
            return np.where(np.linspace(0, 2, n_points) < 1, 1, 0) * 0.5 + 0.5
        def sq_wave():
            return 0.5 + 0.5 * np.sign(np.sin(np.linspace(0, 2*np.pi, n_points)))
        def inv_parent():
            return 1 - initial_h
        def stretch_pattern():
            return np.concatenate((np.linspace(0, 1, 10), np.zeros(n_points - 10)))
        
        return [
            sin_pattern(), cos_pattern(), rect_pattern(), sq_wave(), inv_parent(),
            stretch_pattern(), np.random.rand(n_points), np.full(n_points, 0.5)
        ]

    patterns = generate_patterns()
    patterns = [np.clip(p, 0, 1) * (integral_target / np.sum(np.clip(p, 0, 1))) for p in patterns]

    for pattern in patterns:
        if time.time() - start_time > budget_s - 5:
            break
        try:
            result = minimize(objective, pattern, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
