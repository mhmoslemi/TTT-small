import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Process 280 points first
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Function to optimize
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    # For 280 points
    trials = 15
    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()
        shift = np.random.uniform(-n_points//2, n_points//2)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.03, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    # Now process 560 points using the best from 280
    if best_h is not None:
        n_points = 560
        dx = 2.0 / n_points
        integral_target = n_points / 2
        h_560 = np.tile(best_h, 2) * 0.5
        # Now perform optimization on h_560
        trials = 15
        for t in range(trials):
            if time.time() - start_time > budget_s - 1:
                break
            np.random.seed(seed + t + trials)
            h_start = h_560.copy()
            shift = np.random.uniform(-n_points//2, n_points//2)
            h_start = np.roll(h_start, int(shift))
            noise = np.random.normal(0, 0.01, n_points)
            h_start += noise
            h_start = np.clip(h_start, 0, 1)
            sum_start = np.sum(h_start)
            if sum_start > 0:
                h_start *= (integral_target / sum_start)
            try:
                result = minimize(
                    objective,
                    h_start,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass

    # Also check the fallback to 280 if 560 failed
    if best_h is None:
        best_h = initial_h

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
