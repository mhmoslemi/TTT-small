import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 256
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        initial_h = 1 - np.abs(x - 1)
        sum_initial = np.sum(initial_h)
        factor = integral_target / sum_initial
        initial_h = np.clip(initial_h * factor, 0, 1)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        factor = integral_target / sum_h if sum_h > 0 else 1.0
        projected_h = np.clip(h_clip * factor, 0, 1)
        corr = np.correlate(projected_h, 1 - projected_h, mode='full')
        return np.max(corr) * dx

    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            dispatch_every=1,
            polish=True,
            callback=None,
        )
    except:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            dispatch_every=1,
            polish=True,
            callback=None,
        )

    best_h = np.clip(result.x, 0, 1)
    best_c5 = result.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
