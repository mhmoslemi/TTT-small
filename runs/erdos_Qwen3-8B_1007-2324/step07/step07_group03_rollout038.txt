        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_560

    cons_560 = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_560}]
    bounds_560 = [(0.0, 1.0) for _ in range(n_points_560)]

    best_h_560 = h_560.copy()
    best_c5_560 = objective_560(best_h_560)

    if time.time() - start_time < budget_s - 10:
        np.random.seed(seed + trials)
        noise = np.random.normal(0, 0.01, n_points_560)
        h_start = best_h_560 + noise
        h_start = np.clip(h_start, 0, 1)
        if np.sum(h_start) > 0:
            h_start *= (integral_target_560 / np.sum(h_start))
        try:
            result = minimize(
                objective_560,
                h_start,
                method='SLSQP',
                bounds=bounds_560,
                constraints=cons_560,
                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_560:
                    best_c5_560 = current_c5
                    best_h_560 = result.x
        except:
            pass

    final_c5 = best_c5_560 if best_c5_560 < best_c5_280 else best_c5_280
    final_h = best_h_560 if final_c5 == best_c5_560 else best_h_280
    final_n = n_points_560 if final_c5 == best_c5_560 else n_points_280

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return final_h, final_c5, final_n
    return final_h, final_c5, final_n
