            scaled = h_clip * (integral_target / sum_h)
        corr = np.correlate(scaled, 1 - scaled, mode='full')
        return np.max(corr) * dx

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        seed=seed
    )

    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
