import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Construct initial guess: symmetric triangular wave with random perturbation
        x = np.linspace(0, 2, n_points)
        t = (x - 1) / 1.0  # Periodic over [0, 2]
        # Create symmetric triangle: rise for t < 0.5, fall for t > 1.5
        initial_h = np.clip(1 - np.abs(2 * (np.abs(t) - 0.5)), 0, 1)
        initial_h = np.random.permutation(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = initial_h * (integral_target / total)
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    trials = 50  # Increased trial count
    max_iter = 300  # Slightly reduced iterations per trial

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        # Generate perturbation: base guess with random shifts and amplitude jitter
        h_start = initial_h.copy()
        shift = np.random.uniform(-n_points // 3, n_points // 3)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.03, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target / sum_start)
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    if elapsed > budget_s:
