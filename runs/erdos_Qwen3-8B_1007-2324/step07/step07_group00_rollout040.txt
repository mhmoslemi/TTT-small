        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=15, maxiter=100, tol=1e-6,
                                   mutation=(0.5, 1), recombination=0.8,
                                   seed=seed, disp=False)

    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
