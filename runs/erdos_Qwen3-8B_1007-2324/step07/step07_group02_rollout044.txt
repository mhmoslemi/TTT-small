    # Repair initial_h to meet integral constraint
    initial_h = np.clip(initial_h, 0, 1)
    current_sum = np.sum(initial_h)
    if current_sum != integral_target:
        scale = integral_target / current_sum
        initial_h = initial_h * scale
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        current_sum = np.sum(h_clip)
        if current_sum == 0:
            return np.inf
        scale = integral_target / current_sum
        h_repaired = h_clip * scale
        n = len(h_repaired)
        h_fft = np.fft.fft(h_repaired, 2 * n - 1)
        g_fft = np.fft.fft(1.0 - h_repaired, 2 * n - 1)
        corr_fft = h_fft * np.conj(g_fft)
        corr = np.fft.ifft(corr_fft).real
        max_corr = np.max(corr) * dx
        return max_corr

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False
    )

    best_h = result.x
    best_c5 = objective(best_h)

    # Ensure final repair
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    scale = integral_target / current_sum
    best_h = best_h * scale
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
