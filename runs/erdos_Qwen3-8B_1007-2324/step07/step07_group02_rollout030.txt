import time

    dx = 2.0 / n_points
    integral_target = n_points / 2
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Ensure initial_h is valid
    initial_h = np.clip(initial_h, 0, 1)
    current_sum = np.sum(initial_h)
    if current_sum != 0:
        initial_h *= (integral_target / current_sum)
    else:
        initial_h = np.full(n_points, 1.0 / n_points)  # Fallback
    
    # Generate multiple start points
    starts = [initial_h]
    for i in range(3):
        # Add noise and scale
        noise = np.random.normal(0, 0.05, n_points)
        perturb = initial_h + noise
        perturb = np.clip(perturb, 0, 1)
        sum_perturb = np.sum(perturb)
        scale = integral_target / sum_perturb
        starts.append(perturb * scale)
    
    # Objective function
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    # Constraints
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
