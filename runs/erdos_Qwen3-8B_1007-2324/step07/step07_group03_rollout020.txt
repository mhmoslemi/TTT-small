import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx_280 = 2.0 / n_points
    integral_target_280 = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective_280(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_280

    bounds_280 = [(0.0, 1.0) for _ in range(n_points)]
    constraints_280 = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_280}]

    best_h_280 = initial_h.copy()
    best_c5_280 = objective_280(initial_h)
    max_iter_280 = 500
    trials_280 = 15

    for t in range(trials_280):
        if time.time() - start_time > budget_s - 5:
            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()
        shift = np.random.uniform(-30, 30)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.03, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target_280 / sum_start)
        try:
            result = minimize(
                objective_280,
                h_start,
                method='SLSQP',
                bounds=bounds_280,
                constraints=constraints_280,
                options={'maxiter': max_iter_280, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_280:
                    best_c5_280 = current_c5
                    best_h_280 = result.x
        except:
            pass

    # Interpolate to 560 points
    n_points_560 = 560
    dx_560 = 2.0 / n_points_560
    integral_target_560 = n_points_560 / 2

    x_old = np.linspace(0, 2, n_points)
    x_new = np.linspace(0, 2, n_points_560)
    h_560_interpolated = np.interp(x_new, x_old, best_h_280)

    def objective_560(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_560

    bounds_560 = [(0.0, 1.0) for _ in range(n_points_560)]
    constraints_560 = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target_560}]

    best_h_560 = h_560_interpolated.copy()
    best_c5_560 = objective_560(h_560_interpolated)
    max_iter_560 = 500
    trials_560 = 10

    for t in range(trials_560):
        if time.time() - start_time > budget_s - 5:
            break
        np.random.seed(seed + trials_280 + t)
        h_start = best_h_560.copy()
        shift = np.random.uniform(-20, 20)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.01, n_points_560)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start *= (integral_target_560 / sum_start)
        try:
            result = minimize(
                objective_560,
                h_start,
                method='SLSQP',
                bounds=bounds_560,
                constraints=constraints_560,
                options={'maxiter': max_iter_560, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun

        except:
            pass

    if elapsed > budget_s:

    return best_h_560, best_c5_560, n_points_560
