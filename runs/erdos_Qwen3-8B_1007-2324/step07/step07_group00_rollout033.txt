import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    initial_sym = np.zeros(n_points)
    initial_sym[:half] = initial_h[:half]
    initial_sym[half:] = initial_h[:half][::-1]
    total_sym = np.sum(initial_sym) * dx
    scale = integral_target / total_sym
    initial_sym = initial_sym * scale
    initial_sym = np.clip(initial_sym, 0, 1)
    h_opt = initial_sym[:half]

    def build_full_h(h_opt):
        full_h = np.zeros(n_points)
        full_h[:half] = h_opt
        full_h[half:] = h_opt[::-1]
        return full_h

    def objective(h_opt):
        full_h = build_full_h(h_opt)
        corr = np.correlate(full_h, 1 - full_h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half)]
    constraints = [{'type': 'eq', 'fun': lambda h_opt: np.sum(h_opt) - (n_points / 4)}]

    best_h_opt = h_opt.copy()
    best_c5 = objective(best_h_opt)
    trials = 20
    max_iter = 500

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        perturbation = np.random.normal(0, 0.02, half)
        h_start = h_opt + perturbation

        sum_start = np.sum(h_start)
        if sum_start > 0:
            h_start = h_start * (n_points / 4) / sum_start
        h_start = np.clip(h_start, 0, 1)
        try:
            result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h_opt = result.x
        except:
            pass

    best_full_h = build_full_h(best_h_opt)
    elapsed = time.time() - start_time
    if elapsed > budget_s:

    return best_full_h, best_c5, n_points
