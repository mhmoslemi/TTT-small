from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    # Enforce symmetry
    symmetric_h = np.zeros(n_points)
    for i in range(n_points):
        symmetric_h[i] = initial_h[i]
        symmetric_h[n_points - 1 - i] = 1 - symmetric_h[i]
    initial_h = symmetric_h.copy()
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = initial_h * (integral_target / total)
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    trials = 20
    max_iter = 500

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()
        shift = np.random.uniform(-n_points/2, n_points/2)
        h_start = np.roll(h_start, int(shift))
        noise = np.random.normal(0, 0.01, n_points//2)
        h_start[:n_points//2] += noise
        h_start[:n_points//2] = np.clip(h_start[:n_points//2], 0, 1)
        h_start[n_points//2:] = 1 - h_start[:n_points//2]

        try:

                objective,
                h_start,
                method='COBYLA',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'disp': False}
            )
            if result.success:
