            # Nelder-Mead
            res_n = minimize(lambda x: compute_c5(x, 2.0/n), h0, method='Nelder-Mead',
                             bounds=bounds, options={'xatol': 1e-4, 'maxiter': 100})
            # SLSQP with integral constraint
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]
            res_s = minimize(lambda x: compute_c5(x, 2.0/n), res_n.x, method='SLSQP',
                             bounds=bounds, constraints=cons, options={'ftol': 1e-6, 'maxiter': 500})
            if res_s.success:
                if res_s.fun < best_c5:
                    best_c5 = res_s.fun
                    best_h = res_s.x
                    n_best = n
        optimize(n1, h_start)

    # Level 2: 280 points
    n2 = 280
    h_start = interpolate(best_h, n2)
    if time.time() - start_time < budget_s - 30:
        optimize(n2, h_start)

    # Level 3: 560 points
    n3 = 560
    h_start = interpolate(best_h, n3)
    if time.time() - start_time < budget_s - 40:
        optimize(n3, h_start)

    # Final check against initial
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 280/2}]
        res = minimize(lambda x: compute_c5(x, 2.0/280), h_start, method='SLSQP',
                      bounds=[(0,1)]*280, constraints=cons, options={'ftol':1e-6, 'maxiter':500})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
            n_best = 280

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_best
    return best_h, best_c5, n_best
