import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 256  # Optimal resolution for balance between speed and precision
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize with provided or default symmetric pattern
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * (integral_target / sum_initial)
    else:
        initial_h = np.full(n_points, 0.5) * (integral_target / n_points)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.inf
        h_proj = h_clip * (integral_target / sum_h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with optimized parameters
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        polish=True,
        disp=False,
        vectorized=True
    )

    # Handle case where optimization fails
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = initial_h.copy()
        best_c5 = objective(initial_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
