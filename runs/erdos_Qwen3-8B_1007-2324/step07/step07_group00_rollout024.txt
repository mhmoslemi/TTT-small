from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 256
    dx = 2.0 / n_points
    integral_target = n_points / 2
    max_iter = 200

        objective,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
