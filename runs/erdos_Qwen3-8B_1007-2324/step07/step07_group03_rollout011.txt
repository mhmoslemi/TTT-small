from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    resolutions = [280, 560]
    best_h = np.zeros(280)
    best_c5 = float('inf')
    n_points = 280

    else:
        initial_h = np.full(280, 0.5)

    for n in resolutions:
        dx = 2.0 / n
        integral_target = n / 2.0

        if n == 280:
            current_h = np.array(initial_h) if 'initial_h_values' in globals() else np.full(n, 0.5)
        else:
            if 'initial_h_values' in globals():
                current_h = np.tile(initial_h, 2) * 0.5
            else:
                current_h = np.full(n, 0.5)

        current_h = np.clip(current_h, 0, 1)
        if np.sum(current_h) > 0:
            current_h *= (integral_target / np.sum(current_h))

        trials = 15
        for t in range(trials):
            if time.time() - start_time > budget_s - 1:
                break
            np.random.seed(seed + t)
            shift = np.random.uniform(-n//4, n//4)
            perturbed = np.roll(current_h, int(shift))

            perturbed += noise

            if np.sum(perturbed) > 0:
                perturbed *= (integral_target / np.sum(perturbed))

            def objective(h):
                hc = np.clip(h, 0, 1)
                corr = np.correlate(hc, 1 - hc, mode='full')
                return np.max(corr) * dx

            bounds = [(0.0, 1.0) for _ in range(n)]
