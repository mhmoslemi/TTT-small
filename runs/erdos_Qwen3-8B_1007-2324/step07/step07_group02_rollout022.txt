import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    def perturb(h):
        np.random.seed(seed)
        noise = np.random.normal(0, 0.05, n_points)
        shift = np.random.randint(-n_points//8, n_points//8)
        h_pert = np.roll(h + noise, shift)
        h_pert = np.clip(h_pert, 0, 1)
        total = np.sum(h_pert)
        if total > 0:
            h_pert *= (integral_target / total)
        return h_pert

    starts = [initial_h]
    for i in range(4):
        starts.append(perturb(initial_h))
    starts.append(perturb(perturb(initial_h)))

    for start in starts:
        if time.time() - start_time > budget_s - 2:
            break

                objective,
                start,
                method='SLSQP',
                bounds=bounds,

            )

                    best_h = result.x
        except:
