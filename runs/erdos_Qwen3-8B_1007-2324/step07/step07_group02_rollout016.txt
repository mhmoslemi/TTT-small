import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def repair_integral(h):
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-5:
            return h.copy()
        needed = integral_target - current_sum
        weights = np.abs(h - 0.5)
        indices = np.random.choice(n_points, size=10, replace=False, p=weights/weights.sum())
        for idx in indices:
            if abs(needed) < 1e-5:
                break
            max_add = 1.0 - h[idx]
            max_sub = h[idx] - 0.0
            if needed > 0:
                delta = min(max_add, needed)
            else:
                delta = max(max_sub, needed)
            h[idx] += delta
            needed -= delta
            h[idx] = np.clip(h[idx], 0, 1)
        return h

    starts = [initial_h.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        shifted = np.roll(initial_h + noise, np.random.randint(-n_points//8, n_points//8))
        starts.append(repair_integral(shifted))

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    for start in starts:
        if time.time() - start_time > budget_s - 2:
            break
        try:
            res = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}],
                options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x
        except:
            pass

    best_h = repair_integral(best_h)
    corr_full = np.correlate(best_h.clip(0, 1), (1 - best_h).clip(0, 1), mode="full")
    c5_val = np.max(corr_full) * dx
    return best_h, c5_val, n_points
