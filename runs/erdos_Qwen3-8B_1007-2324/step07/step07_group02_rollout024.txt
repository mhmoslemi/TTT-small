from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

    else:
        initial_h = np.full(n_points, 0.5)

    def generate_variant(h, seed):
        np.random.seed(seed)
        noise = np.random.normal(0, 0.015, n_points)
        shift = np.random.randint(-n_points//8, n_points//8)
        shifted = np.roll(h + noise, shift)
        shifted = np.clip(shifted, 0, 1)
        sum_shifted = np.sum(shifted)
        if sum_shifted > 0:
            shifted *= (integral_target / sum_shifted)
        return shifted

    starts = [initial_h]
    for i in range(8):
        starts.append(generate_variant(initial_h, seed + i))
    starts.append(generate_variant(initial_h, seed + 40))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for idx, start in enumerate(starts):
        if time.time() - start_time > budget_s - 2:
            break
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy='best1bin',
                maxiter=100,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),

                polish=True,
                constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
