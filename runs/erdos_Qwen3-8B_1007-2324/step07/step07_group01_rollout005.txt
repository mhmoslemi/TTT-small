import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    h *= integral_target / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    def objective(h_vec):
        h_clip = np.clip(h_vec, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def apply_geometric(h):
        scale = 1.1 if np.random.rand() < 0.5 else 0.9
        new_h = h * scale
        new_h = np.clip(new_h, 0, 1)
        new_h *= integral_target / np.sum(new_h)
        return new_h

    def apply_shift(h):
        shift = np.random.randint(-n_points // 2, n_points // 2)
        new_h = np.roll(h, shift)
        new_h = np.clip(new_h, 0, 1)
        new_h *= integral_target / np.sum(new_h)
        return new_h

    def apply_invert(h):
        start = np.random.randint(0, n_points - 1)
        end = np.random.randint(start + 1, n_points)

        new_h[start:end] = 1 - new_h[start:end]
        new_h = np.clip(new_h, 0, 1)
        new_h *= integral_target / np.sum(new_h)
        return new_h

    while time.time() - start_time < budget_s - 1:

            new_h = apply_invert(best_h)
