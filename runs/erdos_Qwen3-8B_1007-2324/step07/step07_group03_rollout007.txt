from scipy.optimize import minimize
import time

            break
        dx = 2.0 / n_points
        integral_target = n_points / 2

        if n_points == 280:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values).copy()
            else:
                h = np.full(n_points, 0.5)

            h = np.tile(np.array(initial_h_values), 2) * 0.5

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= (integral_target / sum_h)

        def objective(x):
            x_clip = np.clip(x, 0, 1)
            corr = np.correlate(x_clip, 1 - x_clip, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]

                break

            h_start = h.copy()

            h_start = np.roll(h_start, int(shift))
            noise = np.random.normal(0, 0.015 if n_points == 280 else 0.005, n_points)
            h_start += noise
            h_start = np.clip(h_start, 0, 1)
            sum_start = np.sum(h_start)
            if sum_start > 0:
                h_start *= (integral_target / sum_start)

            try:
                result = minimize(objective, h_start, method='Nelder-Mead', bounds=bounds, constraints=constraints, options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 300})
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
