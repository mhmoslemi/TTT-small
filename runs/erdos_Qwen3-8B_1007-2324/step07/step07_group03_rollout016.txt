from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 300  # Increased sample count for better resolution

    if 'initial_h_values' in globals():

    else:
        initial_h = np.full(n_points, 0.5)

    def objective(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    max_iter = 1000
    trials = 50

    for t in range(trials):

            break
        np.random.seed(seed + t)
        h_start = initial_h.copy()

        noise = np.random.normal(0, 0.015, n_points)  # Reduced noise magnitude
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
        sum_start = np.sum(h_start)
        if sum_start > 0:

        try:
            result = minimize(
                objective,
                h_start,
                method='Nelder-Mead',  # Switched to Nelder-Mead for better exploration
                bounds=bounds,
                constraints=constraints,
