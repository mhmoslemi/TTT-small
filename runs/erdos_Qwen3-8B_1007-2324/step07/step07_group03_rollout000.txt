import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    best_c5 = 1.0
    best_h = np.array([])
    n_points_list = [280, 560]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        integral_target = n_points / 2
        if 'initial_h_values' in globals():
            initial_h = np.array(initial_h_values)
        else:
            initial_h = np.full(n_points, 0.5)
        
        def objective(h):
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
        
        trials = 25
        for t in range(trials):
            if time.time() - start_time > budget_s - 1:
                break
            h_start = initial_h.copy()
            shift = np.random.uniform(-n_points//4, n_points//4)
            h_start = np.roll(h_start, int(shift))
            noise = np.random.normal(0, 0.01 if n_points == 560 else 0.03, n_points)
            h_start += noise
            h_start = np.clip(h_start, 0, 1)
            sum_start = np.sum(h_start)
            if sum_start > 0:
                h_start *= (integral_target / sum_start)
            
            try:
                result = minimize(
                    objective,
                    h_start,
                    method='Nelder-Mead',
                    bounds=bounds,
                    constraints=constraints,
                    options={'xatol': 1e-8, 'maxiter': 300}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass
            
            try:
                result = minimize(
                    objective,
                    h_start,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass
    
    elapsed = time.time() - start_time

        return best_h, best_c5, len(best_h)
