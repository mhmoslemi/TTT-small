    best_h = h_start.copy()
    best_c5 = objective(h_start[:n_points // 2])

    trials = 20
    max_iter = 300

    for t in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + t)
        perturbation = np.random.normal(0, 0.05, n_points // 2)
        h_perturb = h_start[:n_points // 2] + perturbation
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_first = np.sum(h_perturb)
        if sum_first > 0:
            scale = (n_points / 4) / sum_first
            h_perturb = h_perturb * scale
            h_perturb = np.clip(h_perturb, 0, 1)
        try:
            result = minimize(
                objective,
                x0=h_perturb,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-9, 'eps': 1e-8, 'disp': False}
            )
            if result.success:
                current_x = result.x
                current_h = np.zeros(n_points)
                for i in range(n_points // 2):
                    current_h[i] = current_x[i]
                    current_h[n_points - 1 - i] = current_x[i]
                current_h = np.clip(current_h, 0, 1)
                current_c5 = objective(current_x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:
            pass

    for i in range(n_points // 2):
        if best_h[i] != best_h[n_points - 1 - i]:
            best_h[n_points - 1 - i] = best_h[i]
    return best_h, best_c5, n_points
