import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize initial solution
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (integral_target / sum_h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    num_trials = 50
    for trial in range(num_trials):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(seed + trial)
        
        # Apply diversification
        if np.random.rand() < 0.4:
            shift = np.random.randint(-n_points//2, n_points//2)
            h_perturbed = np.roll(h, shift)
        elif np.random.rand() < 0.3:
            scale = np.random.uniform(0.9, 1.1)
            h_perturbed = h * scale
            h_perturbed = np.clip(h_perturbed, 0, 1)
            sum_p = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (integral_target / sum_p) if sum_p > 0 else np.zeros(n_points)
        else:
            start = np.random.randint(0, n_points - 10)
            end = start + np.random.randint(10, 20)
            h_perturbed = h.copy()
            h_perturbed[start:end] = 1 - h_perturbed[start:end]
            sum_p = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (integral_target / sum_p) if sum_p > 0 else np.zeros(n_points)
        
        # Objective function
        def objective(x):
            h_clip = np.clip(x, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            return np.max(corr) * dx
        
        # Constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
        
        # Optimization
        result = minimize(objective, h_perturbed, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()

    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
