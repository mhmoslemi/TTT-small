import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (integral_target / np.sum(initial_h))

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_c5 = float('inf')
    best_h = initial_h.copy()

    def generate_candidates(h):
        candidates = []
        candidates.append(h.copy())
        candidates.append(1.0 - h.copy())
        shifted = np.roll(h, int(np.random.uniform(-40, 40)))
        candidates.append(shifted)
        modulated = h * (1.0 + np.sin(np.linspace(0, 2 * np.pi, n_points)))
        candidates.append(np.clip(modulated, 0, 1))
        return candidates

    if 'initial_h_values' in globals():
        candidates = generate_candidates(initial_h)
    else:
        candidates = [initial_h]

    for c in candidates:
        scaled = c * (integral_target / np.sum(c))
        try:
            result = minimize(objective, scaled, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return best_h, best_c5, n_points
