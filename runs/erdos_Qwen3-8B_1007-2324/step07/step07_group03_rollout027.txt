import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points_primary = 280
    integral_primary = n_points_primary / 2
    dx_primary = 2.0 / n_points_primary

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points_primary, 0.5)

    def obj_primary(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx_primary

    bounds_primary = [(0.0, 1.0) for _ in range(n_points_primary)]
    cons_primary = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_primary}]

    best_h = initial_h.copy()
    best_c5 = obj_primary(initial_h)

            break

        h_start = initial_h.copy()

        sum_h = np.sum(h_start)
