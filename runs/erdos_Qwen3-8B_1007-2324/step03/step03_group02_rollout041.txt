from scipy.optimize import minimize

        starts.append(structured_start)

            h_rand = np.random.rand(n_points)
            h_start = np.clip(h_rand, 0, 1)
            starts.append(h_start)
    else:
        np.random.seed(seed)
        for _ in range(num_starts):
            h_rand = np.random.rand(n_points)
            h_start = np.clip(h_rand, 0, 1)
            starts.append(h_start)
    
    best_h = None
    best_c5 = float('inf')
    
    for h_start in starts:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            return np.max(corr) * dx

            objective,
            h_start,
            method='L-BFGS-B',

        )
        
        h_opt = np.clip(result.x, 0, 1)
        sum_opt = np.sum(h_opt)

            h_opt = h_opt * (integral_target / sum_opt)

            h_opt = np.zeros(n_points)

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
