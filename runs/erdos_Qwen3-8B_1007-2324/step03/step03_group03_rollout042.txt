def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()

    best_c5 = float('inf')

    # Initialize candidates
    candidates = []
    if 'initial_h_values' in globals():

        def generate_periodic(n):
            base = np.array([1.0, 1.0, 0.0, 0.0])

            target = n / 2
            diff = target - sum_pattern
            idx = np.where(pattern == 0)[0]
            if diff > 0 and len(idx) > 0:

                pattern[idx[-adjust:]] = 1.0
