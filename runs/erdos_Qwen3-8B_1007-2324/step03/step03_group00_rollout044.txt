import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)  # default if no parent available

    # Use parent if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            # Re-construct from available parent
            initial_h = np.repeat(initial_h[:70], 2)
        else:
            initial_h = np.copy(initial_h)
    else:
        initial_h = initial_h_values

    # Normalize to satisfy integral constraint
    initial_sum = np.sum(initial_h)
    if abs(initial_sum - n_points / 2) > 1e-6:
        scale = (n_points / 2) / initial_sum
        initial_h = initial_h * scale
        initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_func}]

    # Initial guess
    x0 = initial_h

    # Optimization
    result = minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-9, 
            'gtol': 1e-9, 
            'maxiter': 300,
            'disp': False
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    h_values_sum = np.sum(h_values)

    c5_bound = result.fun
