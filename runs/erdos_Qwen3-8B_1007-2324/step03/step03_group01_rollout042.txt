        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = None
    best_c5 = float('inf')
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 3000, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    if best_h is None:
        best_h = initial_h
        best_c5 = objective(initial_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
