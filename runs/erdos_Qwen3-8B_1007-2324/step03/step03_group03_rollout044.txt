        binary[::3] = 1.0
        binary = binary * (len(h_initial)/2) / np.sum(binary)
        candidates.append(binary)
    
    # Optimization loop
    for i, h in enumerate(candidates):
        n = len(h)
        if n > 420:
            continue
        # Clip and normalize to simplex
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h = h * (n/2) / total
        # Objective function
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        # Constraints
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        # Optimize
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                n_points = n
    
    # Final check for time
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, len(best_h))
