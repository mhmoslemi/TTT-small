import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_half = 70
    n_full = 140
    dx = 2.0 / n_full

    if 'initial_h_values' in globals():
        initial_h = initial_h_values[:n_half]
    else:
        np.random.seed(seed)
        initial_h = np.random.uniform(0, 1, n_half)

    def repair(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        target = n_half
        if abs(sum_x - target) > 1e-7:
            delta = (target - sum_x) / len(x)
            x += delta
            x = np.clip(x, 0, 1)
        return x

    initial_h = repair(initial_h)
    population = [initial_h.copy() for _ in range(50)]

    def objective(x):
        h = np.zeros(n_full)
        for i in range(n_half):
            h[i] = x[i]
            h[n_full-1-i] = x[i]
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    F, CR = 0.8, 0.9
    best_solution = initial_h.copy()
    best_value = objective(best_solution)

    while time.time() - start_time < budget_s:
        new_population = []
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            trial = np.copy(a)
            for j in range(len(mutant)):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = repair(trial)
            trial_val = objective(trial)
            if trial_val < objective(population[i]):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        for candidate in population:
            cand_val = objective(candidate)
            if cand_val < best_value:

    return h_values, best_value, n_full
