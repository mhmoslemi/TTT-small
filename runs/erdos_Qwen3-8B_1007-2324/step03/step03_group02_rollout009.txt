import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)

    def repair(h):
        h_safe = np.clip(h, 0, 1)
        sum_safe = np.sum(h_safe)
        if sum_safe == 0:
            h_safe = np.random.rand(n_points) * 0.1
        else:
            h_safe = h_safe * target_sum / sum_safe
        return np.clip(h_safe, 0, 1)

    def obj_func(h):

        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
