from scipy.optimize import minimize

import time

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)

    else:
        base_h = np.random.rand(70)
        base_h = base_h / np.sum(base_h) * (70 / 2)
    
    def project_to_simplex(x, n):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        factor = (n / 2) / s if s > 0 else 0
        return x * factor
    
    def generate_candidates(base, n):
        candidates = []
        # Upsample
        if len(base) < n:
            upsampled = np.repeat(base, (n + len(base) - 1) // len(base))
            upsampled = upsampled[:n]
            candidates.append(project_to_simplex(upsampled, n))
        # Periodic pattern
        p = 8
        num_reps = (n + p - 1) // p
        periodic = np.tile([1] * p + [0] * p, num_reps)[:n]
        candidates.append(project_to_simplex(periodic, n))
        # Random perturbation
        perturbed = base.copy()
        perturbed += np.random.normal(0, 0.1, len(base))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = project_to_simplex(perturbed, n)
        candidates.append(perturbed)
        # Interleaved
        interleaved = np.zeros(n)
        interleaved[::2] = 1.0
        interleaved = project_to_simplex(interleaved, n)
        candidates.append(interleaved)
        return candidates
    
    required_resolutions = [100, 200, 400]
    for n in required_resolutions:
        if time.time() - start_time > budget_s:
            break
        candidates = generate_candidates(base_h, n)
        for cand in candidates:
            if np.isnan(np.sum(cand)):
                continue
            def objective(x):
                corr = correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n)
            bounds = [(0.0, 1.0) for _ in range(n)]
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
            result = minimize(objective, cand, method='trust-constr', bounds=bounds, constraints=cons, options={'max_iter': 50})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:

                    best_h = result.x
