import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 100  # Reduced for faster optimization
    n = n_points
    dx = 2.0 / n
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        if len(initial_h) != n:
            # Interpolate or repeat for matching length
            if len(initial_h) * 2 == n:
                initial_h = np.repeat(initial_h, 2)
            else:

        initial_h = initial_h * scale
        initial_h = np.clip(initial_h, 0, 1)

    if n % 2 == 0:
        initial_h = np.concatenate([initial_h[:n//2], initial_h[:n//2][::-1]])
    else:
        initial_h = np.concatenate([initial_h[:n//2], [initial_h[n//2]], initial_h[:n//2][::-1]])
    
    # Final clip and normalization
    initial_h = np.clip(initial_h, 0, 1)
    initial_h_sum = np.sum(initial_h)
    if abs(initial_h_sum - n / 2) > 1e-6:
        scale = (n / 2) / initial_h_sum
        initial_h = initial_h * scale
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def constraint_func(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    # Run optimization with increased precision
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}

        h_values = h_values * scale
        h_values = np.clip(h_values, 0, 1)
    
    c5_bound = result.fun
    return h_values, c5_bound, n_points
