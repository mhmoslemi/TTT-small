import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)

    candidates = []

            base[:k] = 1.0

            scale = (n / 2) / np.sum(pattern)
            pattern *= scale
            pattern = np.clip(pattern, 0, 1)
            candidates.append(pattern)
        # Random candidate
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand / np.sum(h_rand) * (len(h_initial)/2)
        candidates.append(h_rand)
    
    for h in candidates:
        n = len(h)
        if n > 1000:

        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)

            if current_c5 < best_c5:

                best_h = result.x
                n_points = n
