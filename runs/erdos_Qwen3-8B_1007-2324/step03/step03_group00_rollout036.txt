    initial_h = np.clip(initial_h, 0, 1)
    initial_h_sum = np.sum(initial_h)
    if not np.isclose(initial_h_sum, n / 2):
        scale = (n / 2) / initial_h_sum
        initial_h = initial_h * scale
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def constraint_func(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points
    
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values_sum = np.sum(h_values)
    if not np.isclose(h_values_sum, n / 2):
        scale = (n / 2) / h_values_sum
        h_values = h_values * scale
        h_values = np.clip(h_values, 0, 1)
    
    c5_bound = result.fun
    return h_values, c5_bound, n_points
