import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280

    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():

    else:

    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h *= (integral_target / sum_initial)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Generate several starting points
    starts = []
    np.random.seed(seed)
    
    # Start 1: baseline
    starts.append(initial_h.copy())
    
    # Start 2: perturbed initial with noise
    noise = np.random.normal(0, 0.05, n_points)
    perturbed = initial_h + noise
    perturbed = np.clip(perturbed, 0, 1)
    sum_perturbed = np.sum(perturbed)
    if sum_perturbed > 0:

    starts.append(perturbed)

    block_start = np.zeros(n_points)
    block_start[:n_points//2] = 1.0
    starts.append(block_start)
