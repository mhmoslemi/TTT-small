import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    num_starts = 6
    starts = []
    
    np.random.seed(seed)
    noise = np.random.normal(0, 0.05, n_points)
    start = initial_h + noise
    start = np.clip(start, 0, 1)
    sum_start = np.sum(start)
    if sum_start > 0:
        start *= (integral_target / sum_start)
    starts.append(start)
    
    np.random.seed(seed + 1)
    random_start = np.random.uniform(0, 1, n_points)
    sum_rand = np.sum(random_start)
    if sum_rand > 0:
        random_start *= (integral_target / sum_rand)
    starts.append(random_start)
    
    perturb = np.random.normal(0, 0.02, n_points)
    start = initial_h + perturb
    start = np.clip(start, 0, 1)
    sum_start = np.sum(start)
    if sum_start > 0:
        start *= (integral_target / sum_start)
    starts.append(start)
    
    starts.append(initial_h[::-1])
    starts.append(np.zeros(n_points) + 0.5)
    starts.append(np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]))
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h
    best_c5 = objective(best_h)
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
