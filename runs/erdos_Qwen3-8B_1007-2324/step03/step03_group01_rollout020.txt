import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    # Normalize initial_h to meet integral constraint
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * (integral_target / sum_initial)
    else:
        initial_h[:] = 0.5

    starts = [initial_h]

    # Add noise-perturbed start
    np.random.seed(seed)
    perturbed = initial_h + np.random.normal(0, 0.1, n_points)
    perturbed = np.clip(perturbed, 0, 1)
    sum_perturbed = np.sum(perturbed)
    if sum_perturbed > 0:
        perturbed = perturbed * (integral_target / sum_perturbed)
    starts.append(perturbed)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        n = len(h_clip)
        h_padded = np.zeros(2 * n - 1)
        h_padded[:n] = h_clip
        th_padded = np.zeros(2 * n - 1)
        th_padded[:n] = 1.0 - h_clip
        fft_h = np.fft.fft(h_padded)
        fft_th = np.fft.fft(th_padded)
        corr = np.fft.ifft(fft_h * np.conj(fft_th))
        max_corr = np.max(np.abs(corr))
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1500, 'ftol': 1e-8, 'disp': False}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
