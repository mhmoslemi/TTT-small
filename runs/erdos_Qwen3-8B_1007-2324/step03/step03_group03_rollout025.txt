from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    best_c5 = float('inf')

            break
        candidates = []

            upsampled = np.repeat(base_pattern, 2)

        else:
            downsampled = base_pattern[:n]
            candidates.append(downsampled)
