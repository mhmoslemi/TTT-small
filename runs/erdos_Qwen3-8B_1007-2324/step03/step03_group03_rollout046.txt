            max_corr = np.max(corr)
            return max_corr * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        
        result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'max_iter': 100})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
