        sum_x = np.sum(x)
        if abs(sum_x - target_sum) > 1e-6:
            delta = target_sum - sum_x
            x += delta / len(x)
            x = np.clip(x, 0.0, 1.0)
        h = np.zeros(full_n_points)
        for i in range(n_points):
            h[i] = x[i]
            h[full_n_points - 1 - i] = x[i]
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) if corr.size > 0 else 0
        return -max_corr * dx

    try:
        result = differential_evolution(
            objective,
            bounds,
            popsize=50,
            strategy='best1bin',
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-10,
            seed=seed,
            x0=initial_population,
            maxiter=200
        )
    except:
        result = None

    elapsed = time.time() - start_time
    if elapsed > budget_s or result is None:
        h_full = np.zeros(full_n_points)
        for i in range(n_points):
            h_full[i] = initial_population[i]
            h_full[full_n_points - 1 - i] = initial_population[i]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        c5_bound = max_corr * dx
        return h_full, c5_bound, full_n_points

    best_x = result.x
    h_full = np.zeros(full_n_points)
    for i in range(n_points):
        h_full[i] = best_x[i]
        h_full[full_n_points - 1 - i] = best_x[i]
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    max_corr = np.max(corr)
    c5_bound = max_corr * dx
    return h_full, c5_bound, full_n_points
