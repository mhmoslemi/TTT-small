import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        return h * (integral_target / np.sum(h))

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        np.random.seed(seed)
        num_starts = max(1, 2 * (budget_s // 10))
        starts = [initial_h.copy() + np.random.normal(0, 0.05, n_points) for _ in range(num_starts)]
        starts = [np.clip(s, 0, 1) for s in starts]
        for _ in range(num_starts - len(starts)):
            starts.append(np.random.rand(n_points))
    else:
        np.random.seed(seed)
        num_starts = max(1, 2 * (budget_s // 10))
        starts = [np.random.rand(n_points) for _ in range(num_starts)]

    best_h = None
    best_c5 = float('inf')
    total_evals = 0

    for h_start in starts:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

        def objective(h):
            nonlocal total_evals
            total_evals += 1
            if total_evals > 500:
                return best_c5
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            return np.max(corr) * dx

        result = minimize(
            objective,
            h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )

        h_opt = repair(result.x)
        h_clip = np.clip(h_opt, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        if 'initial_h_values' in globals():
            h_initial = np.array(initial_h_values)
            h_clip = np.clip(h_initial, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            c5 = np.max(corr) * dx
            return h_initial, c5, n_points
        else:
            h = np.full(n_points, 0.5)
            h_clip = np.clip(h, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            c5 = np.max(corr) * dx
            return h, c5, n_points

    return best_h, best_c5, n_points
