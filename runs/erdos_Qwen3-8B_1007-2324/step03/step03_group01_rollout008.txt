import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_coarse = 104
    n_fine = 280
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    integral_coarse = n_coarse / 2
    integral_fine = n_fine / 2

    def objective_coarse(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx_coarse

    def objective_fine(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx_fine

    if 'initial_h_values' in globals():
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, n_coarse)
        h_initial_coarse = np.interp(x_coarse, x_fine, initial_h_values)
    else:
        h_initial_coarse = np.full(n_coarse, 0.5)

    bounds_coarse = [(0.0, 1.0) for _ in range(n_coarse)]
    constraints_coarse = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_coarse}]
    result_coarse = minimize(objective_coarse, h_initial_coarse, method='SLSQP', bounds=bounds_coarse,
                             constraints=constraints_coarse, options={'maxiter': 100, 'ftol': 1e-9, 'disp': False})

    if not result_coarse.success:
        h_fine_initial = np.full(n_fine, 0.5)
    else:
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine_initial = np.interp(x_fine, x_coarse, result_coarse.x)
