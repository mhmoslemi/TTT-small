import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    starts = []
    
    # Base start: parent solution
    starts.append(initial_h.copy())
    
    # Perturbation 1: Gaussian noise with adaptive amplitude
    np.random.seed(seed)
    noise = np.random.normal(0, 0.05, n_points)
    perturbed = initial_h + noise
    perturbed = np.clip(perturbed, 0, 1)
    sum_perturbed = np.sum(perturbed)
    if sum_perturbed > 1e-6:
        perturbed *= (integral_target / sum_perturbed)
    starts.append(perturbed)
    
    # Perturbation 2: Reversed baseline
    reversed_h = initial_h[::-1].copy()
    reversed_h = np.clip(reversed_h, 0, 1)
    sum_reversed = np.sum(reversed_h)
    if sum_reversed > 1e-6:
        reversed_h *= (integral_target / sum_reversed)
    starts.append(reversed_h)
    
    # Perturbation 3: Sinusoidal modulation
    phase = np.linspace(0, np.pi, n_points)
    sin_h = 0.5 + 0.5 * np.sin(phase)
    sin_h = np.clip(sin_h, 0, 1)
    sum_sin = np.sum(sin_h)
    if sum_sin > 1e-6:
        sin_h *= (integral_target / sum_sin)
    starts.append(sin_h)
    
    # Perturbation 4: Alternating banding
    band_width = 10
    band_h = np.zeros(n_points)
    for i in range(n_points):
        if (i // band_width) % 2 == 0:
            band_h[i] = 1.0
        else:
            band_h[i] = 0.0
    band_h = np.clip(band_h, 0, 1)
    sum_band = np.sum(band_h)
    if sum_band > 1e-6:
        band_h *= (integral_target / sum_band)
    starts.append(band_h)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')

        try:

                objective,
                start,
                method='L-BFGS-B',
