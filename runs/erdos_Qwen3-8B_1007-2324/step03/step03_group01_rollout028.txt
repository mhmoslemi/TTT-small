        parent_with_noise *= (integral_target / sum_parent)
    starts.append(parent_with_noise)
    
    # Second start: reversed parent
    reversed_h = initial_h[::-1]
    reversed_h = np.clip(reversed_h, 0, 1)
    sum_rev = np.sum(reversed_h)
    if sum_rev > 0:

    starts.append(reversed_h)
