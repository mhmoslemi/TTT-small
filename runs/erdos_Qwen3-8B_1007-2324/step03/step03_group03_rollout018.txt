import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_used = 0
    
    def project_simplex(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if np.isclose(s, len(x)/2):
            return x
        scale = (len(x)/2) / s if s != 0 else 0
        return x * scale
    
    def objective(x):
        corr = correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / len(x))
    
    if 'initial_h_values' in globals():
        base = initial_h_values
    else:
        base = np.zeros(70)
        base[::5] = 1.0
        base = np.repeat(base, 2)
    
    for n in [100, 200, 400]:
        if time.time() - start_time > budget_s:
            break
        candidates = []
        candidates.append(np.tile(base[:n//2], 2)[:n])
        candidates.append(np.repeat(base[:n//2], 2)[:n])
        candidates.append(project_simplex(np.random.rand(n)))
        candidates.append(project_simplex(base[:n]))
        
        for cand in candidates:
            if np.isclose(np.sum(cand), n/2):
                x0 = cand
            else:
                x0 = project_simplex(cand)
            
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
            bounds = [(0.0, 1.0) for _ in range(n)]
            res = minimize(objective, x0, method='trust-constr', bounds=bounds, constraints=cons, options={'max_iter': 50})
            
            if res.success:
                curr_c5 = res.fun
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = res.x
                    n_used = n
    
    return (best_h, best_c5, n_used)
