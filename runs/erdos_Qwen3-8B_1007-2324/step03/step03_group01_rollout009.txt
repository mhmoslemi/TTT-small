import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    num_starts = 15
    starts = []
    
    # Use initial_h as base and add noise variations
    np.random.seed(seed)
    base_h = np.clip(initial_h, 0, 1)
    sum_base = np.sum(base_h)
    base_h = base_h * (integral_target / sum_base) if sum_base > 0 else base_h
    
    # Simple variations
    for _ in range(num_starts):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = base_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed *= (integral_target / sum_perturbed)
        starts.append(perturbed)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
