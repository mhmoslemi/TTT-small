import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points

        initial_h = np.array(initial_h_values)

        for h in candidates:
            h_clipped = np.clip(h, 0, 1)

            c5 = np.max(corr) * dx
