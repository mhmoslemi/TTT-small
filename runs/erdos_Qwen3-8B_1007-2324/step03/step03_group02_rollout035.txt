import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Initialize starting points
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points)

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return np.zeros_like(h_clip)
        scale = (n_points / 2) / sum_clip
        return h_clip * scale

    def objective_func(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    # Generate multiple starting points
    num_starts = 10
    starts = [initial_h]
    np.random.seed(seed)
    for _ in range(num_starts - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturb = initial_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        starts.append(h_perturb)

    best_h = None
    best_c5 = float('inf')

    for h_start in starts:
        # Differential evolution for global search
        result = differential_evolution(
            objective_func,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            strategy='best1bin',

            maxiter=50,
            tol=1e-5,
