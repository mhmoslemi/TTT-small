import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    # Initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates.append(h_initial)
        candidates.append(np.repeat(h_initial, 2))  # Upsampled
    
    def generate_period_pattern(period, n):
        base = np.array([1]*(period//2) + [0]*(period - period//2), dtype=np.float64)
        num_repeats = n // period
        remainder = n % period
        pattern = np.tile(base, num_repeats)
        if remainder > 0:
            pattern = np.concatenate([pattern, base[:remainder]])
        return pattern
    
    # Add period-4 and period-5 patterns
    for n in [len(initial_h_values), 140, 280, 420]:
        if n < 70:
            continue
        try:
            period4 = generate_period_pattern(4, n)
            candidates.append(period4)
        except:
            pass
        try:
            period5 = generate_period_pattern(5, n)
            candidates.append(period5)
        except:
            pass
    
    # Random candidate
    h_rand = np.random.rand(len(initial_h_values))
    h_rand /= np.sum(h_rand) * (len(initial_h_values)/2)
    candidates.append(h_rand)
    
    # Optimization loop
    for i, h in enumerate(candidates):
        n = len(h)
        if n > 1000:
            continue
        # Project to simplex
        if not np.isclose(np.sum(h), n/2):
            h = h / np.sum(h) * (n/2)
        h = np.clip(h, 0, 1)
        # Objective function
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        # Constraints
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        # Optimize
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'max_iter': 200})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                n_points = n
    
    # Final check for time
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, len(best_h))
