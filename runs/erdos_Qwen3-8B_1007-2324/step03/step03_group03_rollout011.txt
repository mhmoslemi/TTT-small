def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    candidates = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates.append(h_initial)
        candidates.append(np.repeat(h_initial, 2))
        def generate_periodic_pattern(n_points, period=5, ones_per_period=2):
            base = [1.0]*ones_per_period + [0.0]*(period - ones_per_period)
            base = np.array(base)
            full_pattern = np.tile(base, (n_points + period - 1) // period)
            full_pattern = full_pattern[:n_points]
            total = np.sum(full_pattern)
            full_pattern = (full_pattern / total) * (n_points / 2)
            return full_pattern
        for n in [140, 280, 420]:
            candidates.append(generate_periodic_pattern(n, period=5, ones_per_period=2))
            candidates.append(generate_periodic_pattern(n, period=4, ones_per_period=1))
            candidates.append(generate_periodic_pattern(n, period=6, ones_per_period=3))
        candidates.append(np.random.rand(len(h_initial)))
    
    for h in candidates:
        n = len(h)
        if n > 420:
            continue
        if not np.isclose(np.sum(h), n/2):
            h = h / np.sum(h) * (n/2)
        h = np.clip(h, 0, 1)
        h = h * (n/2) / np.sum(h)
        
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'max_iter': 200})
        if result.success:
            current_h = np.clip(result.x, 0, 1)
            current_h = current_h * (n/2) / np.sum(current_h)
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                n_points = n
