import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    # Normalize initial_h
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    if sum_initial > 0:
        initial_h = initial_h * (integral_target / sum_initial)
    else:
        initial_h = np.full(n_points, 140 / n_points)

    num_starts = 20
    starts = []

    # Parent with noise
    np.random.seed(seed)
    noise = np.random.normal(0, 0.05, n_points)
    parent_with_noise = initial_h + noise
    parent_with_noise = np.clip(parent_with_noise, 0, 1)
    sum_p = np.sum(parent_with_noise)
    if sum_p > 0:
        parent_with_noise = parent_with_noise * (integral_target / sum_p)
    starts.append(parent_with_noise)

    # Random starts
    for i in range(num_starts - 1):
        np.random.seed(seed + i)
        random_start = np.random.uniform(0, 1, n_points)
        sum_r = np.sum(random_start)
        if sum_r > 0:
            random_start = random_start * (integral_target / sum_r)
        else:
            random_start = np.full(n_points, 140 / n_points)
        starts.append(random_start)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return float('inf')
        h_scaled = h_clip * (integral_target / sum_h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, n_points
