from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

        clipped = np.clip(x, 0, 1)

        scale = target_sum / current_sum
        return clipped * scale
    
    def objective(x):
        corr = correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
