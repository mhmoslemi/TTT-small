import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    candidates = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates.append(h_initial)
        # Upsample
        x_vals = np.linspace(0, 2, len(h_initial))
        h_upsampled = np.interp(np.linspace(0, 2, n_points), x_vals, h_initial)
        candidates.append(h_upsampled)
        # Sine wave
        x = np.linspace(0, 2, n_points)
        h_sine = (1 + np.sin(2 * np.pi * 0.5 * x)) / 2
        candidates.append(h_sine)
        # Random
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * target_sum
        candidates.append(h_rand)
    
    for h in candidates:
        if len(h) != n_points:
            continue
        # Normalize to simplex
        h = h / np.sum(h) * target_sum
        # Objective function
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            scaled_corr = corr * dx
