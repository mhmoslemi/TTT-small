import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    # Initialize candidates from initial_h_values if available
    candidates = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates.append(h_initial)
        # Upsample
        candidates.append(np.repeat(h_initial, 2))
        # Smoothed
        smoothed = np.convolve(h_initial, np.ones(5)/5, mode='same')
        smoothed = np.clip(smoothed, 0, 1)
        candidates.append(smoothed)
        # Random perturbation
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand / np.sum(h_rand) * (len(h_initial)/2)
        candidates.append(h_rand)
        # Random binary pattern
        h_bin = np.random.randint(0, 2, size=len(h_initial))
        h_bin = h_bin / np.sum(h_bin) * (len(h_initial)/2)
        candidates.append(h_bin)
        # Alternating pattern
        h_alt = np.zeros_like(h_initial)
        h_alt[::2] = 1.0
        h_alt[::4] = 0.5
        candidates.append(h_alt)
    else:
        n_base = 70
        h_base = np.zeros(n_base)
        h_base[:14] = 1.0
        h_base[14:28] = 0.5
        h_base[28:] = 1.0
        candidates.append(h_base)
    
    # Process candidates
    for idx, h in enumerate(candidates):
        n = len(h)
        if n > 400:
            continue
        # Project to simplex
        if not np.isclose(np.sum(h), n / 2):
            h = h / np.sum(h) * (n / 2)
        # Objective function
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        # Optimization
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        result = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-4, options={'max_iter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                n_points = n
        # Time check
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, len(best_h))
