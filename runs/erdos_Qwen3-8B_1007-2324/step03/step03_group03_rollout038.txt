            h = np.repeat(h0, int(n // len(h0)))[:n]
            if not np.isclose(np.sum(h), n/2):
                h = h / np.sum(h) * (n/2)
            h = np.clip(h, 0, 1)
            
            def objective(x):
                corr = correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n)
            
            bounds = [(0.0, 1.0) for _ in range(n)]
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 150})
            if res.success and res.fun < best_c5:
                best_c5, best_h = res.fun, res.x
                if time.time() - start_time > budget_s:
                    break
    
    # Fallback to random if parent not available
    if best_h is None:
        n = 140
        h = np.random.rand(n)
        h = h / np.sum(h) * (n/2)
        h = np.clip(h, 0, 1)
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 150})
        best_c5, best_h = res.fun, res.x
    
    return (best_h, best_c5, len(best_h))
