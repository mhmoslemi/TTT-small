import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Generate diverse starting points
    starts = []
    np.random.seed(seed)
    # Perturb initial with noise
    noise = np.random.normal(0, 0.05, n_points)
    parent_with_noise = initial_h + noise
    parent_with_noise = np.clip(parent_with_noise, 0, 1)
    sum_parent = np.sum(parent_with_noise)
    if sum_parent > 0:
        parent_with_noise *= (integral_target / sum_parent)
    starts.append(parent_with_noise)
    
    # Generate additional starts with segment shifts
    for i in range(4):
        np.random.seed(seed + i)
        perturb = np.zeros_like(initial_h)
        idx = np.random.permutation(n_points)[::4]
        perturb[idx] = np.random.uniform(-0.2, 0.2, len(idx))
        start = initial_h + perturb
        start = np.clip(start, 0, 1)
        sum_start = np.sum(start)
        if sum_start > 0:
            start *= (integral_target / sum_start)
        starts.append(start)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = None
    best_c5 = float('inf')
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    if best_h is None:
        best_h = initial_h
        best_c5 = objective(initial_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
