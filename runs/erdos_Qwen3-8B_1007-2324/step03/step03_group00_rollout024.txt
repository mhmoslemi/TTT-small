import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_full = 140
    n_half = 70
    dx = 2.0 / n_full
    repair_factor = 35.0  # target sum of half vector
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    # Initialize population based on initial_h_values if available
    if 'initial_h_values' in globals() and len(initial_h_values) == n_full:
        population = [np.clip(initial_h_values[:n_half], 0, 1)]
    else:
        population = [np.random.rand(n_half)]
    
    # Repair function
    def repair(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if abs(total - repair_factor) > 1e-6:
            delta = (repair_factor - total) / len(x)
            x += delta
            x = np.clip(x, 0, 1)
        return x
    
    # Objective function
    def objective(x):
        x = repair(x)
        h_full = np.zeros(n_full)
        for i in range(n_half):
            h_full[i] = x[i]
            h_full[n_full - 1 - i] = x[i]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return -np.max(corr) * dx  # Minimize negative to find minimum
    
    # DE parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_gen = 1000
    gen = 0
    best = population[0]
    best_obj = objective(best)
    
    # Initialize population
    while len(population) < pop_size:
        candidate = np.random.rand(n_half)
        population.append(repair(candidate))
    
    while gen < max_gen and time.time() - start_time < budget_s:
        gen += 1
        new_pop = []
        for i in range(pop_size):
            # Mutation
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_half):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = repair(trial)
            # Evaluate
            trial_obj = objective(trial)
            if trial_obj < objective(population[i]):
                new_pop.append(trial)
            else:
                new_pop.append(population[i])
        population = new_pop
        # Update best
        for candidate in population:
            if objective(candidate) < best_obj:
                best = candidate
                best_obj = objective(candidate)
    
    # Build final h_full
    best = repair(best)
    h_full = np.zeros(n_full)
    for i in range(n_half):
        h_full[i] = best[i]
        h_full[n_full - 1 - i] = best[i]
    
    return h_full, -best_obj, n_full
