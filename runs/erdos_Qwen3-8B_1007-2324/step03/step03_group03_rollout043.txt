import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    n_points = 400
    dx = 2.0 / n_points

    # Generate initial candidates
    candidates = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        # Reshape to 400 points if needed
        if len(h_initial) != n_points:
            h_initial = np.kron(h_initial, np.ones((n_points // len(h_initial),)))
        candidates.append(h_initial)
    
    # Sine wave with optimized frequency
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + np.sin(2 * np.pi * 0.8 * x))
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * (n_points / 2) / np.sum(h_sine)
    candidates.append(h_sine)

    # Gaussian-modulated sine wave
    h_gaussian = 0.5 * (1 + np.sin(2 * np.pi * 0.8 * x)) * np.exp(-0.5 * (x - 1)**2)
    h_gaussian = np.clip(h_gaussian, 0, 1)
    h_gaussian = h_gaussian * (n_points / 2) / np.sum(h_gaussian)
    candidates.append(h_gaussian)

    # Random candidate
    h_random = np.random.rand(n_points)
    h_random = h_random / np.sum(h_random) * (n_points / 2)
    candidates.append(h_random)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

    for h in candidates:
        if not np.isclose(np.sum(h), n_points / 2, atol=1e-6):
            h = h / np.sum(h) * (n_points / 2)
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'max_iter': 500, 'tol': 1e-6})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
