import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    n_half = 70
    dx = 2.0 / n_points

    def repair(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if abs(total - 35) > 1e-7:
            deficit = 35 - total
            x += deficit / len(x)
            x = np.clip(x, 0, 1)
        return x

    if 'initial_h_values' in globals():
        initial = initial_h_values[:n_half]
        sum_initial = np.sum(initial)
        if abs(sum_initial - 35) > 1e-7:
            scale = 35.0 / sum_initial
            initial = initial * scale
            initial = np.clip(initial, 0, 1)
            sum_initial = np.sum(initial)
            if abs(sum_initial - 35) > 1e-7:
                deficit = 35 - sum_initial
                initial += deficit / len(initial)
                initial = np.clip(initial, 0, 1)
    else:
        initial = np.random.uniform(0, 1, n_half)

    initial = repair(initial)

    def objective(x):
        x_repaired = repair(x)
        h = np.zeros(n_points)
        for i in range(n_half):
            h[i] = x_repaired[i]
            h[n_points - 1 - i] = x_repaired[i]
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return -c5

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=50, maxiter=500, tol=1e-10,
                                   mutation=(0.8, 1.0), recombination=0.9,
                                   seed=seed, polish=True, disp=False)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = np.zeros(n_points)
        for i in range(n_half):
            h_values[i] = initial[i]
            h_values[n_points - 1 - i] = initial[i]
        c5_bound = -objective(initial)
        return h_values, c5_bound, n_points

    best_x = result.x
    best_x_repaired = repair(best_x)
    h_values = np.zeros(n_points)
    for i in range(n_half):
        h_values[i] = best_x_repaired[i]
        h_values[n_points - 1 - i] = best_x_repaired[i]
    c5_bound = -result.fun

    sum_h = np.sum(h_values) * dx
    if abs(sum_h - 1.0) > 1e-6:
        scale = 1.0 / sum_h
        h_values = h_values * scale

    return h_values, c5_bound, n_points
