import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    required_sum = n_points / 2

    candidates = []

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        # Upsample with linear interpolation
        x = np.linspace(0, 2, len(h_initial))
        x_new = np.linspace(0, 2, n_points)
        h_upsampled = np.interp(x_new, x, h_initial)
        candidates.append(h_upsampled)
        candidates.append(h_initial)
    else:
        # Periodic pattern
        period = 5
        pattern = np.zeros(n_points)
        for i in range(n_points // period):
            pattern[i*period:(i+1)*period] = 1.0 / period
        pattern = pattern * (required_sum / np.sum(pattern))
        candidates.append(pattern)
        # Sine wave
        x = np.linspace(0, 2, n_points)
        f = 0.8
        sine = (np.sin(2 * np.pi * f * x) + 1) / 2
        sine = sine * (required_sum / np.sum(sine))
        candidates.append(sine)
        # Gaussian-modulated sine
        gaussian_sine = (np.sin(2 * np.pi * f * x) * np.exp(-0.1 * (x - 1)**2) + 1) / 2
        gaussian_sine = gaussian_sine * (required_sum / np.sum(gaussian_sine))
        candidates.append(gaussian_sine)
        # Random
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * required_sum
        candidates.append(h_rand)

        sum_clipped = np.sum(h_clipped)
        if not np.isclose(sum_clipped, required_sum):
