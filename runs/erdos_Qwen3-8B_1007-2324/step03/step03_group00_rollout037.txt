        initial_first_half = initial_first_half * scale
        initial_first_half = np.clip(initial_first_half, 0, 1)

    # Objective function to minimize max overlap
    def objective(x):
        full = np.concatenate([x, x[::-1]])
        corr = np.correlate(full, 1 - full, mode='full')
        return np.max(corr) * dx

    # Optimization bounds and constraint
    bounds = [(0.0, 1.0) for _ in range(70)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]

    # Run optimization
    result = minimize(
        objective,
        initial_first_half,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    # Respect time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = np.concatenate([initial_first_half, initial_first_half[::-1]])
        c5_bound = objective(initial_first_half)
        return h_values, c5_bound, n_points

    # Final adjustments
    best_first_half = result.x
    best_first_half = np.clip(best_first_half, 0, 1)
    sum_best = np.sum(best_first_half)
    if np.abs(sum_best - 35) > 1e-6:
        scale = 35.0 / sum_best
        best_first_half = best_first_half * scale
        best_first_half = np.clip(best_first_half, 0, 1)
    h_values = np.concatenate([best_first_half, best_first_half[::-1]])
    c5_bound = result.fun

    return h_values, c5_bound, n_points
