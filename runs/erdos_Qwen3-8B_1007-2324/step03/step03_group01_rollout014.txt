import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    num_starts = 15
    starts = []
    
    # First start: initial_h
    starts.append(initial_h)
    
    # Wavelet-modified starts
    for _ in range(3):
        wavelet = np.random.normal(0, 0.05, n_points)
        wavelet = np.clip(wavelet, -0.1, 0.1)
        modified = initial_h + wavelet
        modified = np.clip(modified, 0, 1)
        sum_modified = np.sum(modified)
        if sum_modified > 0:
            modified *= (140 / sum_modified)
        starts.append(modified)
    
    # Random starts
    for _ in range(num_starts - 4):
        np.random.seed(seed + _)
        random_start = np.random.uniform(0, 1, n_points)
        sum_random = np.sum(random_start)
        if sum_random > 0:
            random_start *= (140 / sum_random)
        starts.append(random_start)
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - 140}]
    
    best_h = None
    best_c5 = float('inf')
    
    for start in starts:
        try:
            result = minimize(
                objective,
                start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
