import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    np.random.seed(seed)
    candidates = []

        for _ in range(5):
            candidates.append(np.random.rand(n_points))

        h = np.clip(indiv, 0, 1)
        sum_h = np.sum(h)
        if sum_h < 1e-6:
            return np.random.rand(n_points) * 0.1
        h = h * (target_sum / sum_h)
        return h
    
    best_h = None
    best_c5 = float('inf')
    
    for c in candidates:

            h_clip = np.clip(h, 0, 1)

        h_opt = np.clip(result.x, 0, 1)
