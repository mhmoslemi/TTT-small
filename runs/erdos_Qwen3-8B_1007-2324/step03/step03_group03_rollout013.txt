import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 0

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates = [h_initial]
        # Upsample to 280 points
        h_upsamp = np.repeat(h_initial, 2)
        candidates.append(h_upsamp)
        # Add random perturbation
        h_perturb = h_initial + 0.1 * np.random.rand(len(h_initial)) - 0.05
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (len(h_initial)/2) / np.sum(h_perturb)
        candidates.append(h_perturb)
    else:
        candidates = [np.random.rand(140) * 140 / 2]
    
    # Try different n_points
    n_values = [140, 280, 420]
    
    for n in n_values:
        for h in candidates:
            if len(h) != n:
                h = np.resize(h, n)
            # Ensure sum is n/2
            h = h * (n/2) / np.sum(h)
            h = np.clip(h, 0, 1)
            # Objective
            def objective(x):
                corr = correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n)
            # Constraints
            bounds = [(0.0, 1.0) for _ in range(n)]
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
            # Optimize
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                              options={'max_iter': 50, 'ftol': 1e-6, 'eps': 1e-6})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    n_points = n
                    if time.time() - start_time > budget_s - 1:
                        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
