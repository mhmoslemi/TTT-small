import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (integral_target) / sum_h
        return h * scale

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        h_initial = np.zeros(n_points)
        num_pulses = int(np.sqrt(n_points))
        positions = np.linspace(0, 2, num_pulses, endpoint=False)
        for pos in positions:
            idx = int(pos * n_points / 2)
            h_initial[idx] = 1
        np.random.seed(seed)
        h_initial += np.random.normal(0, 0.1, h_initial.shape)
        h_initial = repair(h_initial)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_estimate(h):
        h_low = repair(np.clip(h - 1e-5, 0, 1))
        h_high = repair(np.clip(h + 1e-5, 0, 1))
        return (objective(h_high) - objective(h_low)) / 2e-5

    # Differential evolution
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    h_de = result_de.x

    # L-BFGS-B refinement
    result_lbgfgs = minimize(
        objective,
        h_de,
        method='L-BFGS-B',
        bounds=bounds,
        jac=grad_estimate,
        options={'maxiter': 200, 'ftol': 1e-9}
    )

    h_opt = repair(result_lbgfgs.x)

    corr_final = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(corr_final) * dx

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_initial = repair(h_initial)
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        c5 = np.max(corr) * dx
        return h_initial, c5, n_points

    return h_opt, c5, n_points
