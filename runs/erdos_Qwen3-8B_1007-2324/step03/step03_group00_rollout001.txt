from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_half = 70
    dx = 2.0 / n_points

        h = np.zeros(n_points)
        for i in range(n_half):
            h[i] = x[i]
            h[n_points - 1 - i] = x[i]
        return h

    if 'initial_h_values' in globals():

        h_half = initial[:n_half]
        h_half = np.clip(h_half, 0, 1)
        h_sum = np.sum(h_half)
        if abs(h_sum - n_half) > 1e-6:
            scale = n_half / h_sum
            h_half = h_half * scale
            h_half = np.clip(h_half, 0, 1)
        initial = h_half
    else:
        initial = np.full(n_half, 0.5)

    def objective(x):

        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        return -max_val * dx

    def repair(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        target = n_half
        if abs(total - target) > 1e-7:
            deficit = target - total
            x += deficit / len(x)
            x = np.clip(x, 0, 1)
        return x

    def de_step(pop, F=0.8, CR=0.9):

        for i in range(len(pop)):

            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = repair(mutant)
            trial = np.random.rand(len(mutant)) < CR
            child = np.where(trial, mutant, pop[i])
            new_pop.append(child)
        return new_pop

    pop = [np.random.rand(n_half) for _ in range(50)]
