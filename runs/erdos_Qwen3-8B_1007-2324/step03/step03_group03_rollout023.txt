import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 0

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / len(x))

    if 'initial_h_values' in globals():
        base = initial_h_values
        for n in [100, 200, 400]:
            # Generate periodic candidate
            base_pattern = np.array([1.0, 1.0, 0.0, 0.0])
            num_repeats = n // len(base_pattern) + 1
            candidate = np.tile(base_pattern, num_repeats)[:n]
            candidate = np.clip(candidate, 0, 1)
            total = np.sum(candidate)
            if total == 0:
                continue
            candidate = candidate * (n / 2) / total
            # Optimize
            result = minimize(objective, candidate, method='trust-constr',
                              bounds=[(0,1)]*n, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    n_points = n
    # Handle remaining budget for fallback
    if time.time() - start_time < budget_s:
        # Fallback to base pattern
        base_pattern = np.array([1.0, 1.0, 0.0, 0.0])
        num_repeats = len(best_h) // len(base_pattern) + 1
        fallback = np.tile(base_pattern, num_repeats)[:len(best_h)]
        fallback = np.clip(fallback, 0, 1)
        total = np.sum(fallback)
        if total > 0:
            fallback = fallback * (len(best_h) / 2) / total
        if best_h is None:
            best_h = fallback
        elif np.max(np.abs(fallback - best_h)) < 1e-6:
            best_c5 = np.max(np.correlate(fallback, 1 - fallback, mode='full')) * (2.0 / len(fallback))
    return (best_h, best_c5, n_points)
