import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return float('inf')
        h_scaled = h_clip * (integral_target / sum_clip)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    np.random.seed(seed)
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            mutation=(0.5, 1),
            crossover=0.9,
            popsize=10,
            tol=1e-6,
            maxiter=50,
            seed=seed,
            polish=True,
            disp=False
        )
        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = initial_h
        best_c5 = objective(initial_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_h = initial_h
        best_c5 = objective(initial_h)
    
    return best_h, best_c5, n_points
