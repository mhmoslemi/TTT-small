def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():

    initial_h = np.clip(initial_h, 0, 1)

        h_clip = np.clip(h, 0, 1)
        N = len(h_clip)
        h_padded = np.zeros(2 * N - 1)
        h_padded[:N] = h_clip
        one_minus_h = 1 - h_clip

        one_minus_h_padded[:N] = one_minus_h
        fft_h = np.fft.fft(h_padded)
        fft_1mh = np.fft.fft(one_minus_h_padded)
        corr = np.fft.ifft(fft_h * np.conj(fft_1mh))
        max_corr = np.max(np.abs(corr.real))
        return max_corr * dx
