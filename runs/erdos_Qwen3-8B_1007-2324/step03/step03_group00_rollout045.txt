import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    m = 70  # segments in [0,1]
    n = 2 * m  # total sample points
    delta = 1.0 / m  # segment length in [0,1]
    delta_x = 2.0 / n  # sample spacing
    n_points = n

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        # Extract initial_y from first m+1 points (assuming symmetry)
        initial_y = initial_h[:m+1]
        # Ensure sum constraint is satisfied
        sum_val = sum(initial_y[i] + initial_y[i+1] for i in range(m))
        if not np.isclose(sum_val, m):
            scale = m / sum_val
            initial_y = initial_y * scale
    else:
        initial_y = np.full(m + 1, 0.5)

    def generate_h_values(y):
        h = []
        for x in np.linspace(0, 2, n_points):
            if x <= 1:
                i = int(x / delta)
                i = min(i, m - 1)
                if i < m:
                    t = (x - i * delta) / delta
                    y_val = y[i] + t * (y[i + 1] - y[i])
                else:
                    y_val = y[m]
            else:
                x_prime = 2 - x
                i = int(x_prime / delta)
                i = min(i, m - 1)
                if i < m:
                    t = (x_prime - i * delta) / delta
                    y_val = y[i] + t * (y[i + 1] - y[i])
                else:
                    y_val = y[m]
            h.append(y_val)
        return np.array(h)

    def objective(y):
        h = generate_h_values(y)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * delta_x

        return sum(y[i] + y[i + 1] for i in range(m)) - m

    bounds = [(0.0, 1.0) for _ in range(m + 1)]

    result = minimize(
