import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_candidate = h_initial.copy()
        def objective(h):
            h_clip = np.clip(h, 0, 1)
            corr = correlate(h_clip, 1 - h_clip, mode='full', method='fft')
            return np.max(corr) * dx
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, h_candidate, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
        h_opt = np.clip(result.x, 0, 1)
        sum_opt = np.sum(h_opt)
        if sum_opt > 0:
            h_opt = h_opt * (integral_target / sum_opt)
        else:
            h_opt = np.zeros(n_points)
        current_c5 = objective(h_opt)
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5
    else:
        h_candidate = np.random.rand(n_points)
    
    if time.time() - start_time > budget_s - 20:
        if 'initial_h_values' in globals():
            h_clip = np.clip(initial_h_values, 0, 1)
            corr = correlate(h_clip, 1 - h_clip, mode='full', method='fft')
            c5 = np.max(corr) * dx
            return initial_h_values, c5, n_points
        else:
            h = np.full(n_points, 0.5)
            h_clip = np.clip(h, 0, 1)
            corr = correlate(h_clip, 1 - h_clip, mode='full', method='fft')
