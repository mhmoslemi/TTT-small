def run(seed=42, budget_s=100, **kwargs):

        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros(n_points)
        scale = (integral_target) / sum_clipped
        return h_clipped * scale

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points)

    variations = [initial_h]
    for _ in range(4):
        noise = np.random.normal(0, 0.15, n_points)
        h = initial_h + noise
        h = np.clip(h, 0, 1)
        h = repair(h)
        variations.append(h)

    for _ in range(4):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = repair(h)
        variations.append(h)

    best_h = None
    best_c5 = float('inf')

    for h_start in variations:

            return np.max(corr) * dx

        try:
