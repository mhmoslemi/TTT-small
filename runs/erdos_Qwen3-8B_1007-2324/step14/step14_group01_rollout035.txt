        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        h_repaired = repair(h)
        return compute_overlap(h_repaired)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin', 
        maxiter=100, 
        popsize=20, 
        mutation=(0.5, 1.0), 
        recombination=0.9, 
        tol=1e-6,
        polish=True,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return initial_h, compute_overlap(initial_h), n_points
    
    best_h = result.x
    best_h = repair(best_h)
    best_c5 = compute_overlap(best_h)
    return best_h, best_c5, n_points
