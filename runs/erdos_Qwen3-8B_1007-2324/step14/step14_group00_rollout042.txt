def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 4
    
    if 'initial_h_values' in globals():

    def compute_c5(h_half):
        h_full = np.concatenate([h_half, 1 - h_half[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair_positions(h):
        h_clamped = np.clip(h, 0, 1)
        sum_clamped = np.sum(h_clamped)
        if np.isclose(sum_clamped, target_sum):
            return h_clamped
        if sum_clamped == 0:
            return np.full_like(h_clamped, target_sum / n_half)
        scaling_factor = target_sum / sum_clamped
        h_scaled = h_clamped * scaling_factor
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled
    
    n_particles = 20
    w, c1, c2 = 0.9, 2.0, 2.0
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_half))
    positions = np.array([h_initial.copy() for _ in range(n_particles)])
    pbest_positions = positions.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest_position = np.array([0.0]*n_half)
    gbest_fitness = np.inf
    
    for i in range(n_particles):
        fitness = compute_c5(positions[i])
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_position = positions[i].copy()
            gbest_fitness = fitness
    
    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_half), np.random.rand(n_half)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest_positions[i] - positions[i]) + c2 * r2 * (gbest_position - positions[i])
