import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    eps = 1e-8

    # Initialize h
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        x_initial = h_initial[:n_half]
    else:
        x_initial = np.clip(1 - np.abs(np.linspace(0, 2, n_points) - 1), 0, 1)[:n_half]

    def build_h(x):
        h = np.zeros(n_points)
        h[:n_half] = x
        h[n_half:] = 1.0 - x[::-1]
        return h

    def compute_c5(x):
        h = build_h(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # PSO setup
    n_particles = 30
    max_iter = 500
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    pos = np.random.uniform(0, 1, (n_particles, n_half))
    pos[0, :] = x_initial
    vel = np.random.uniform(-0.1, 0.1, (n_particles, n_half))
    pbest_pos = pos.copy()
    pbest_val = np.inf * np.ones(n_particles)
    gbest_pos = pos[0, :]
    gbest_val = np.inf
    time_start = time.time()

    # PSO loop
    for _ in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
        for i in range(n_particles):
            curr_val = compute_c5(pos[i])
            if curr_val < pbest_val[i]:
                pbest_val[i] = curr_val
                pbest_pos[i, :] = pos[i, :]
            r1, r2 = np.random.rand(2, n_half)
            vel[i, :] = inertia * vel[i, :] + cognitive * r1 * (pbest_pos[i, :] - pos[i, :]) + social * r2 * (gbest_pos - pos[i, :])

    best_h = build_h(gbest_pos)
    return best_h, gbest_val, n_points
