        h = np.clip(h, 0, 1)
        s2 = np.sum(h)
        scale2 = integral_target / s2
        return h * scale2

    def objective(x):
        h = repair(x)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    result = differential_evolution(
        objective, bounds, popsize=15, mutation=(0.5, 1.0),
        recombination=0.8, tol=1e-6, maxiter=100, seed=seed
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (initial_h, np.max(np.correlate(np.concatenate([initial_h, initial_h[::-1]]), 1 - np.concatenate([initial_h, initial_h[::-1]]), mode='full')) * dx, n_points)

    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
