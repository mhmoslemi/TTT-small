    best_h = h.copy()
    best_c5 = float('inf')

    def compute_overlap(h_binary):
        corr = np.correlate(h_binary, 1 - h_binary, mode='full')

    def repair(child):

    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    gen_budget = int(budget_s * 0.8)  # Reserve 20% for final evaluation

    start_time = time.time()

        fitnesses = [compute_overlap(ind) for ind in population]
