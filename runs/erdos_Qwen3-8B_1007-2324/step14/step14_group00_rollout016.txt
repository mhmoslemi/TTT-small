import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    max_iter = 500

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.uniform(0, 1, size=n_half)

    num_particles = 30
    particles = np.random.uniform(0, 1, (num_particles, n_half))
    velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_half))
    pbest_positions = particles.copy()
    pbest_fitness = np.array([compute_overlap(p) for p in pbest_positions])
    gbest_position = pbest_positions[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    best_h = gbest_position.copy()

    start_time = time.time()
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1: break
        for i in range(num_particles):
            r1, r2 = np.random.rand(n_half), np.random.rand(n_half)
            velocities[i] = 0.5 * velocities[i] + 1.5 * r1 * (pbest_positions[i] - particles[i]) + 1.5 * r2 * (gbest_position - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            curr = compute_overlap(particles[i])

            if curr < gbest_fitness:
                gbest_fitness = curr
                gbest_position = particles[i].copy()
                best_h = gbest_position.copy()

    h_full = np.concatenate([best_h, 1 - best_h[::-1]])
    return h_full, gbest_fitness, n_points
