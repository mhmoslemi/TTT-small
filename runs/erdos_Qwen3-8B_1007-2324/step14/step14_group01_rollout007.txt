import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    # Load initial guess if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Default: triangular function clipped to [0, 1]
        x = np.linspace(0, 2, 60)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
        initial_h = h
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Corrected from previous error
    
    def repair(x):
        current_sum = np.sum(x)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        return np.clip(x * scale, 0, 1)
    
    def objective(x):
        h = repair(x)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Differential Evolution with adaptive strategy
    result_de = differential_evolution(
        objective, bounds, 
        strategy='best1bin', 
        popsize=20, 
        maxiter=100, 
        tol=1e-6,
        seed=seed,
        mutation=(0.5, 1.0)
    )
    
    # Local refinement with L-BFGS-B
    result_local = minimize(
        objective, result_de.x,
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    # Select better solution
    best_result = result_local if result_local.fun < result_de.fun else result_de
    
    # Final repair and evaluation
    best_h = repair(best_result.x)
    final_c5 = objective(best_h)
    
    return best_h, final_c5, len(best_h)
