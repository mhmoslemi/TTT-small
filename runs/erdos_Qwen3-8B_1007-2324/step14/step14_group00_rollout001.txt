import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points

    # Initial guess based on parent's symmetric construction
    h_opt = np.zeros(n_half)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half].copy()
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = 1 - np.abs(x - 1)
        h_opt = h_opt[:n_half].copy()
        h_opt = np.clip(h_opt, 0, 1)

    # Objective function with symmetry
    def objective(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Bounds and constraints
    bounds = [(0.0, 1.0)] * n_half

    # Run differential evolution with adaptive time allocation
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    elapsed = time.time() - start_time

    # Early exit if time is critical
    if elapsed > budget_s - 1:
        best_h_opt = h_opt
    else:
        best_h_opt = result.x

    best_c5 = objective(best_h_opt)
    h_values = np.concatenate([best_h_opt, 1 - best_h_opt[::-1]])
    return h_values, best_c5, n_points
