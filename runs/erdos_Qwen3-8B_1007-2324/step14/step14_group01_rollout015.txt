```python
import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.zeros(n_points)

    def repair(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)

    if 'initial_h_values' in globals():
        h_initial = repair(initial_h)
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = 1 - np.abs(x - 1)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = repair(h_initial)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        repaired = repair(h)
        h_full = np.concatenate([repaired, repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-6,
        maxiter=300,
        mutation=(0.5, 'best1bin'),
        crossover=(0.9, 'bin'),
        seed=seed,
        vectorized=True
    )

    best_h = repair(result.x)
    h_full = np.concatenate([best_h, best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx
    return best_h, c5_bound, n_points
```
