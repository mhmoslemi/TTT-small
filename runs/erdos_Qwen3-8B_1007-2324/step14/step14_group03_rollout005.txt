def run(seed=42, budget_s=100, **kwargs):

    if 'initial_h_values' in globals():

        k = 4
        base_length = 0.4
        centers = np.sort(np.random.uniform(0.25, 1.75, k))
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for c in centers:
            tent = 1 - np.abs(grid - c) / (base_length / 2)
            h += np.clip(tent, 0, 1)
        current_sum = np.sum(h)
        scale = required_sum / current_sum
        h = np.clip(h * scale, 0, 1)

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = required_sum / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - required_sum,
        'jac': lambda x: np.ones_like(x)
    }]

    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={

            'maxiter': 3000,
            'disp': False,
            'eps': 1e-8
        }
    )

    elapsed = time.time() - start_time
