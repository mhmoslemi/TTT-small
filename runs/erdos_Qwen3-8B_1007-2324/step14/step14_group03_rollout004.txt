import numpy as np
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280

    required_sum = n_points / 2.0

        h[indices] = 1

    population = [np.zeros(n_points, dtype=int) for _ in range(pop_size)]
    for i in range(pop_size):
        indices = np.random.choice(n_points, size=int(required_sum), replace=False)
        population[i][indices] = 1

    def compute_overlap(h_binary):
        corr = np.correlate(h_binary, 1 - h_binary, mode='full')
        return np.max(corr) * dx

        target_sum = int(required_sum)
        if current_sum > target_sum:
            excess = current_sum - target_sum
            indices_to_flip = np.where(ind == 1)[0]
            flip_indices = np.random.choice(indices_to_flip, size=excess, replace=False)
            ind[flip_indices] = 0

            needed = target_sum - current_sum
            indices_to_flip = np.where(ind == 0)[0]
            flip_indices = np.random.choice(indices_to_flip, size=needed, replace=False)
            ind[flip_indices] = 1
        return ind

    def fitness(ind):

    fitness_values = [fitness(ind) for ind in population]
