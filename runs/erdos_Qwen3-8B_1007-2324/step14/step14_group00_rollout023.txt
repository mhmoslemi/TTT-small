import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 60
    dx = 2.0 / n_points
    n_vars = n_points // 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_vars]
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = 1.0 - np.abs(x - 1.0)
        h_initial = h_initial[:n_vars]

    num_particles = 50
    particles = np.clip(np.random.normal(h_initial, 0.05, (num_particles, n_vars)), 0, 1)
    velocities = np.random.uniform(-0.05, 0.05, (num_particles, n_vars))
    pbest = particles.copy()

    gbest = particles[0].copy()
    gbest_score = np.inf

    max_iter = 1000
    w, c1, c2 = 0.8, 1.5, 1.5
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(num_particles):
            pos = particles[i]
            h_full = np.zeros(n_points)
            h_full[:n_vars] = pos
            h_full[n_vars:] = 1.0 - pos[::-1]
            corr = np.correlate(h_full, 1.0 - h_full, mode='full')
            c5 = np.max(corr) * dx
            if c5 < pbest_scores[i]:
                pbest_scores[i] = c5
                pbest[i] = pos.copy()
            if c5 < gbest_score:
                gbest_score = c5
                gbest = pos.copy()
        for i in range(num_particles):
            r1, r2 = np.random.rand(2, n_vars)

    best_h[n_vars:] = 1.0 - gbest[::-1]
    return (best_h, gbest_score, n_points)
