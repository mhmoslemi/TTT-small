from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.zeros(60)

    def repair(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return np.clip(h * scaling, 0, 1)

        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')

        return np.max(corr) * dx

    def objective(x):
        h = repair(x, 30)
        return compute_overlap(h)

    bounds = [(0.0, 1.0) for _ in range(60)]
    start_time = time.time()

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.9,
