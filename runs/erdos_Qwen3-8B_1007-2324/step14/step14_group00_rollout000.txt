import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize
    
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt_initial = 1 - np.abs(x - 1)
        h_opt_initial = h_opt_initial[:n_half]
    
    def objective(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    
    start_time = time.time()
    result = scipy.optimize.differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        polish=True,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        best_h_opt = h_opt_initial
    else:
        best_h_opt = result.x
    
    best_c5 = objective(best_h_opt)
    h_values = np.concatenate([best_h_opt, 1 - best_h_opt[::-1]])
    
    return h_values, best_c5, n_points
