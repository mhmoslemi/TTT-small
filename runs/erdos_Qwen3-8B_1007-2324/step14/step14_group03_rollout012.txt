import numpy as np

    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(child):
        current_sum = np.sum(child)
        diff = current_sum - required_sum
        if diff != 0:
            indices = np.where(child == 1)[0]
            zeros = np.where(child == 0)[0]
            if diff > 0:
                if len(indices) > diff:
                    flip_indices = indices[:diff]
                    child[flip_indices] = 0
                else:
                    child[indices] = 0
                    flip_indices = zeros[:diff - len(indices)]
                    child[flip_indices] = 1
            else:
                if len(zeros) > -diff:
                    flip_indices = zeros[:abs(diff)]
                    child[flip_indices] = 1
                else:
                    child[zeros] = 1
                    flip_indices = indices[:abs(diff) - len(zeros)]
                    child[flip_indices] = 0
        return child

    def generate_initial_population(pop_size):
        pop = []
        for _ in range(pop_size):

            pop.append(arr)
        return np.array(pop)

    if 'initial_h_values' in globals():

        h_bin_initial = (h_initial > 0.5).astype(int)
