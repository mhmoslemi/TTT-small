def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize
    from scipy.optimize import minimize

    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) * dx = 1 => sum(h) = 1/dx = n_points / 2

    # Check initial_h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        if original_len >= n_half:
            step = int(np.ceil(original_len / n_points))
            h_opt = h_initial[::step][:n_half]
        else:
            h_opt = h_initial
    else:
        h_opt = np.zeros(n_half)
        for i in range(n_half):
            h_opt[i] = i * dx  # linear from 0 to 1

    # Objective function
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Define objective for optimization
    def objective(x):
        return compute_overlap(x)

        'jac': lambda x: np.ones_like(x)

        noise = np.random.normal(0, amplitude, len(h))
        return np.clip(h + noise, 0.0, 1.0)

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize
    from scipy.optimize import minimize

    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        h_opt = np.zeros(n_half)
        for i in range(n_half):
            h_opt[i] = i * dx

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return compute_overlap(x)

    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    })

    def perturb(h, amplitude):
        noise = np.random.normal(0, amplitude, len(h))
        return np.clip(h + noise, 0.0, 1.0)

    start_time = time.time()
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt.copy()

    max_iter = 20

        if time.time() - start_time > budget_s - 1:
            break
        perturbed = perturb(h_opt, amp)
        result = minimize(
            objective,
            perturbed,
            method='SLSQP',
            bounds=[(0,1)] * len(perturbed),
            constraints=constraints,
            options={

                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
