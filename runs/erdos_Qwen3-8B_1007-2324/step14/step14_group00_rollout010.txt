        h_opt_initial *= (target_sum / np.sum(h_opt_initial))

    def repair_position(pos):
        clamped = np.clip(pos, 0, 1)
        current_sum = np.sum(clamped)
        delta = target_sum - current_sum
        if delta != 0:
            scale = (target_sum + 1e-10) / (current_sum + 1e-10)
            adjusted = clamped * scale
            adjusted = np.clip(adjusted, 0, 1)
            pos[:] = adjusted

    num_particles = 20
    particles = np.random.rand(num_particles, 30)
    velocities = np.random.rand(num_particles, 30) * 0.1

    pbest_pos = particles.copy()
    pbest_obj = np.full(num_particles, np.inf)
    gbest_pos = particles[0].copy()
    gbest_obj = np.inf

    max_iter = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5

    start_time = time.time()

    for _ in range(max_iter):
        for i in range(num_particles):
            repair_position(particles[i])
            h_full = np.concatenate([particles[i], 1.0 - particles[i][::-1]])
            corr = np.correlate(h_full, 1.0 - h_full, mode='full')

            if c5 < pbest_obj[i]:

                pbest_pos[i] = particles[i].copy()

                gbest_obj = c5
                gbest_pos = particles[i].copy()

        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest_pos[i] - particles[i]) + social * r2 * (gbest_pos - particles[i])

        elapsed = time.time() - start_time
