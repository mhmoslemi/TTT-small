import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        n_points = len(parent_h) * 2
    else:
        n_points = 560
        parent_h = None
    
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    def objective(x):
        x = repair(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = None
    best_c5 = float('inf')

    def generate_start(parent=None):

            start = np.zeros(n_points)
            start[:len(parent)] = parent
        else:
            start = np.random.rand(n_points)
        return repair(start)

    if 'initial_h_values' in globals():
        starters = [generate_start(parent_h), 
                    generate_start(parent_h + 0.05 * np.random.randn(n_points)),
                    generate_start(parent_h - 0.05 * np.random.randn(n_points)),
                    generate_start()]
    else:
        starters = [generate_start()]
