        return np.clip(x * scale, 0, 1)
    
    h_starts = [h_opt]
    noise = np.random.normal(0, 0.02, size=h_opt.shape)
    h_starts.append(np.clip(h_opt + noise, 0, 1))
    h_starts.append(np.random.rand(280))
    h_starts.append(np.random.rand(280))
    
    best_c5 = float('inf')
    best_h = None
    
    for start in h_starts:
        dx = 2.0 / n_points
        def objective(x):
            full_h = np.concatenate([x, x[::-1]])
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            return np.max(corr) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 280.0}]
        bounds = [(0.0, 1.0) for _ in range(280)]
        
        result = opt.minimize(
            objective,
            start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if best_h is not None:
        full_h = np.concatenate([best_h, best_h[::-1]])
        return full_h, best_c5, n_points
    else:
        if 'initial_h_values' in globals():
            full_h = np.concatenate([np.array(initial_h_values), np.array(initial_h_values)[::-1]])
            return full_h, float('inf'), 280
        else:
            return np.random.rand(560), float('inf'), 560
