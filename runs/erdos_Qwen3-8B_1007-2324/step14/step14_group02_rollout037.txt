import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    n_points = 560
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
        elif len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)
    
    def repair(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        s_clipped = np.sum(x_clipped)
        if s_clipped == 0:
            return np.zeros_like(x)
        scale = (n_points / 2) / s_clipped
        x_repaired = x_clipped * scale
        return x_repaired
    
    h_initial_repaired = repair(h_initial)
    
    def objective(x):
        x_repaired = repair(x)
        corr = np.correlate(x_repaired, 1 - x_repaired, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.dual_annealing(objective, bounds, 
                                options={'maxfun': int(100 / 0.5), 'initial_temp': 50.0, 'seed': seed})
    
    best_h = repair(result.x)
    best_c5 = result.fun
    
    return best_h.tolist(), best_c5, n_points
