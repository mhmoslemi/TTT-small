    if runtime > budget_s - 1:  # Ensuring budget is met
        print(f"Budget exceeded for resolution {n_points}: {runtime+1:.2f} seconds")
        return initial_h, initial_c5, n_points  # Fallback if overridden

    return best_candidate_repaired, best_c5, n_points
