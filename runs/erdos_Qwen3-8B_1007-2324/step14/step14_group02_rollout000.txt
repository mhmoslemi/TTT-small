import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 560
    integral_target = n_points / 2.0
    dx = 2.0 / n_points

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        scale = integral_target / s
        return np.clip(x * scale, 0, 1)

    starting_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_full = np.concatenate([h_initial, h_initial[::-1]])
        starting_points.append(h_full)
        h_perturbed = h_full + np.random.normal(0, 0.02, len(h_full))
        h_perturbed = repair(h_perturbed)
        starting_points.append(h_perturbed)
    else:
        for _ in range(4):
            h_half = np.random.rand(n_points // 2)
            h_full = np.concatenate([h_half, h_half[::-1]])
            starting_points.append(h_full)

    best_h = None
    best_c5 = float('inf')

    for h in starting_points:
        h_repair = repair(h)
        def objective(x):
            full_h = np.concatenate([x, x[::-1]])
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
        bounds = [(0.0, 1.0) for _ in range(len(h_repair))]
        try:
            result = opt.minimize(objective, h_repair, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'ftol': 1e-10, 'maxiter': 150, 'disp': False})
            if result.success:
                c5 = result.fun
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.concatenate([h_repair, h_repair[::-1]])
        except:
            pass

    return best_h, best_c5, n_points
