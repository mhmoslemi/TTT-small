import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    n_points = 560
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = np.concatenate([h_initial, h_initial[:n_points - len(h_initial)]])
        elif len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points)
    
    def repair(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        s = np.sum(x_clipped)
        if s == 0:
            return np.zeros_like(x)
        scale = (n_points / 2) / s
        return x_clipped * scale
    
    def objective(x):
        h = repair(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = opt.dual_annealing(objective, bounds, 
                                maxfun=150, 
                                x0=h_initial, 
                                disp=False)
    
    best_h = repair(result.x)
    return best_h, result.fun, n_points
