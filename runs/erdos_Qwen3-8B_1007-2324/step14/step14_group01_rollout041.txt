def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

        scaling2 = target / sum_clipped
        return np.clip(clipped * scaling2, 0, 1)

    h_initial = repair(h_initial, integral_target)

        return np.max(corr) * dx

    def objective(x):
        h_opt = x.copy()
        repaired = repair(h_opt, integral_target)
        return compute_overlap(repaired)
