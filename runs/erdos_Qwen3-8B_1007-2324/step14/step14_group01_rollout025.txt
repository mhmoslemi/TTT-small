import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    def repair(h):
        current_sum = np.sum(h)
        scaling = integral_target / current_sum if current_sum != 0 else 1.0
        return np.clip(h * scaling, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    initial_guess = initial_h.copy()

    start_time = time.time()
    result = differential_evolution(
        lambda x: compute_overlap(repair(np.array(x))),
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=100,
        tol=1e-8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_h = repair(best_h)
    best_c5 = compute_overlap(best_h)
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return initial_guess, compute_overlap(repair(initial_guess)), n_points
    return best_h, best_c5, n_points
