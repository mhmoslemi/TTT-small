        h = np.zeros(n_points)
        for c in centers:
            tent = 1.0 - np.abs(grid - c) / (base_length / 2)
            h += np.clip(tent, 0, 1)
        current_sum = np.sum(h)
        scale = required_sum / current_sum
        h = np.clip(h * scale, 0, 1)

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = required_sum / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - required_sum,
        'jac': lambda x: np.ones_like(x)
    }]

    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-8,
            'maxiter': 5000,
            'disp': False,
            'eps': 1e-9
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h, compute_overlap(h), n_points

    best_h = result.x
    best_c5 = compute_overlap(best_h)
    return best_h, best_c5, n_points
