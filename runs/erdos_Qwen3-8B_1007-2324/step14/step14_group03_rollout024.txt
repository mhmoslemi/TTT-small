import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Smaller grid for faster optimization
    dx = 2.0 / n_points
    required_sum = n_points / 2.0
    
    # Initialize with provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)

            h = np.clip(h, 0, 1)
    else:
        # Construct a base solution with uniform placement
        h = np.zeros(n_points)
        base_length = 0.5
        centers = np.linspace(0.25, 1.75, 5)
        for c in centers:
            h += np.clip(1 - np.abs(np.arange(n_points) * dx - c) / (base_length / 2), 0, 1)
        h = np.clip(h * required_sum / np.sum(h), 0, 1)
    
    def compute_c5(h_full):

        corr = correlate(h_full, 1 - h_full, mode='full')

        current_sum = np.sum(h)
        scaling = required_sum / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_c5(repaired)
    
    # Optimization bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
