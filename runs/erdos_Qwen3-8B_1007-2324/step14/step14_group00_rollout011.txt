import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 60
    dx = 2.0 / n_points
    n_half = n_points // 2
    integral_target = 0.5  # because sum of first half * dx = 0.5

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 1, n_half)
        h_opt = 1 - np.abs(x - 0.5)
        h_opt = np.clip(h_opt, 0, 1)
        sum_h_opt = np.sum(h_opt) * dx
        h_opt *= (integral_target / sum_h_opt) if sum_h_opt > 0 else 1

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def objective(x):
        h_opt = x.copy()
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h_opt)
        penalty = 1e5 * np.abs(sum_h - 15)

    result = scipy.optimize.differential_evolution(

        mutation=(0.5, 1.0),
        recombination=0.8,
        strategy='best1bin',
