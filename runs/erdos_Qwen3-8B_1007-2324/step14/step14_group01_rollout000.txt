import time
from scipy.optimize import differential_evolution, minimize, basinhopping
from pyswarm import pso

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 4.0  # 15.0

    if 'initial_h_values' in globals():

    else:
        x = np.linspace(0, 2, n_points)

        initial_h = np.clip(initial_h, 0, 1)

    def repair(x):
        current_sum = np.sum(x)
        if abs(current_sum - integral_target) < 1e-5:
            scale = 1.0
        else:
            scale = integral_target / current_sum
        x = np.clip(x * scale, 0, 1)
        for _ in range(3):
            missing = integral_target - np.sum(x)
            if abs(missing) < 0.1:
                break
            factor = 1 + (15 * missing) / np.sum(x)
            x = np.clip(x * min(factor, 2), 0, 1)
        return x

    def objective(x):
        try:
            h = repair(x)
            np.clip(h, 0, 1, out=h)
            if np.sum(h) < 0 or np.sum(h) > 2 * integral_target:
                return np.inf
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        except:
            return np.inf

    def optimize_de():
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       mutation=(0.5, 1.0), recombination=0.8,

        return result.x

                         swarmsize=50, 
                         debug=False, 
                         seed=seed)
        return xopt

    def optimize_basinhopping():
        result = basinhopping(objective, initial_h[:n_points],
