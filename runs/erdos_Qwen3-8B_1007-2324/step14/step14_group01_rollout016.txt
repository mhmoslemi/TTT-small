import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
        initial_h = h

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    def objective(h):
        h_repaired = repair(h)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = initial_h if 'initial_h_values' in globals() else np.random.uniform(0, 1, n_points)

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False,
        x0=x0
    )

    best_h = repair(result.x)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
