import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution

    n_points = 60
    n_half = 30
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        initial_guess = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        initial_guess = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    def objective(x):
        h_opt = x.copy()
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.8, 1.0),
        recombination=0.9,
        seed=seed,
        maxiter=100,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        best_h = initial_guess
    else:
        best_h = result.x

    h_full = np.concatenate([best_h, 1 - best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5 = np.max(corr) * dx
    return np.concatenate([best_h, 1 - best_h[::-1]]), c5, n_points
