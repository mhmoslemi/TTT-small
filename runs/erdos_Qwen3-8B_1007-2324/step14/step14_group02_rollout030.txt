import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    n_points = 560
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        x_clipped = np.clip(x, 0.0, 1.0)
        s_clipped = np.sum(x_clipped)
        scale = target_sum / s_clipped
        x_repaired = x_clipped * scale
        x_repaired = np.clip(x_repaired, 0.0, 1.0)
        return x_repaired

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values).copy()  # Use parent's values
        h_initial = np.repeat(h_initial, 2)  # Duplicate to match n_points=560
        h_initial = repair(h_initial)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = repair(h_initial)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.dual_annealing(objective, bounds, maxfun=int(budget_s / 0.2), disp=False)
    best_h = repair(result.x)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return best_h, best_c5, n_points
