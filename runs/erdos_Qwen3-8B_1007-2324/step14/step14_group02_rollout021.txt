    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = opt.dual_annealing(objective, bounds, maxfun=int(budget_s * 0.5), disp=False)
    best_h = result.x
    best_c5 = result.fun
    
    final_h = repair(best_h)
    final_c5 = objective(final_h)
    
    return final_h, final_c5, n_points
