import numpy as np
import time
from scipy.optimize import differential_evolution, basinhopping, minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    import numpy as np
    import time

    # Initialize starting point
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, 60)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
        initial_h = h

    n_points = len(initial_h)
    dx = 2.0 / n_points
    integral_target = n_points / 4.0

    def repair(x):
        current_sum = np.sum(x)
        if np.isclose(current_sum, integral_target, atol=1e-5):
            return x
        scale = integral_target / current_sum
        x_scaled = np.clip(x * scale, 0, 1)
        current_sum_scaled = np.sum(x_scaled)
        if abs(current_sum_scaled - integral_target) > 1e-5:
            scale = integral_target / current_sum_scaled
            x_scaled = np.clip(x_scaled * scale, 0, 1)
        return x_scaled

    def objective(x):
        try:
            h = repair(x)
            np.clip(h, 0, 1, out=h)
            if np.sum(h) < 0 or np.sum(h) > 2 * integral_target:
                return np.inf
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        except:
            return np.inf

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    # Run differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    res_de = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1.0), recombination=0.8, 
                                   tol=1e-5, maxiter=50, seed=seed)
    best_h = repair(res_de.x)
    best_c5 = objective(best_h)

    # Run basin hopping
    res_bh = basinhopping(objective, initial_h, niter=50, T=1.0, 
                          stepsize=0.1, minimizer_kwargs={"method": "L-BFGS-B", "bounds": bounds})
    best_h = repair(res_bh.x)
    best_c5 = min(best_c5, objective(best_h))

    # Run L-BFGS-B local optimization
    res_lbgfs = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, 
                         options={'maxiter': 20})
    best_h = repair(res_lbgfs.x)
    best_c5 = min(best_c5, objective(best_h))

    # Run particle swarm optimization
    try:
        from pyswarm import pso
        lb = [0.0] * n_points
        ub = [1.0] * n_points
        x_pso, f_pso = pso(objective, lb, ub, nwarmup=50, swarmsize=20, 
                           minstep=1e-8, maxiter=50, debug=False)
        best_h = repair(x_pso)
        best_c5 = min(best_c5, objective(best_h))
    except ImportError:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
