import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.repeat(0.5, 60)
    x_initial = initial_h[:30]
    sum_initial = np.sum(x_initial)
    if sum_initial == 0:
        x_initial = np.zeros_like(x_initial)
    else:
        scaling = 15.0 / sum_initial
        x_initial = np.clip(x_initial * scaling, 0, 1)
    bounds = [(0.0, 1.0) for _ in range(30)]
    
    def repair(x, target_sum):
        if np.sum(x) == 0:
            return np.zeros_like(x)
        scaling = target_sum / np.sum(x)
        return np.clip(x * scaling, 0, 1)
    
    def objective(x):
        x_repaired = repair(x, 15.0)
        h = np.concatenate([x_repaired, x_repaired[::-1]])
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (1.0 / 30.0)
    
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=300,
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.5,
        seed=seed,
        disp=False
    )
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([x_initial, x_initial[::-1]]), objective(x_initial), 60
    best_x = result.x
    best_h = np.concatenate([best_x, best_x[::-1]])
    best_c5 = objective(best_x)
    return best_h, best_c5, 60
