import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 60
    dx = 2.0 / n_points
    integral_target = 30.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        initial_h = 1 - np.abs(x - 1)
        initial_h = np.clip(initial_h, 0, 1)

    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        return np.clip(h * scale, 0, 1)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_overlap(repair(h))

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        mutation=(0.5, 1.0),
        crossover=0.9,
        maxiter=150,
        tol=1e-8,
        seed=seed,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        best_h = repair(initial_h)
        return best_h, compute_overlap(best_h), n_points

    best_h = result.x
    best_h_repaired = repair(best_h)
    final_c5 = compute_overlap(best_h_repaired)
    return best_h_repaired, final_c5, n_points
