import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution

    def repair(h):
        s = np.sum(h)
        if s == 0: return np.zeros_like(h)
        scale = 30.0 / s
        return np.clip(h * scale, 0, 1)

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, 60)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
        h_initial = h
        h_initial = repair(h_initial)

    def objective(h_vec):
        h = repair(h_vec)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / 60)

    bounds = [(0.0, 1.0) for _ in range(60)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin', 
        popsize=15, mutation=(0.5, 1), recombination=0.9, 
        tol=1e-8, seed=seed, maxiter=300, 
        disp=False, polish=True
    )
    
    best_h = repair(result.x)
    h_full = np.concatenate([best_h, best_h[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * (2.0 / 60)
    
    return best_h, final_c5, 60
