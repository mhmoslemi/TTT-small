    else:
        h_initial = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, size=required_sum, replace=False)
        h_initial[indices] = 1

    def compute_overlap(h):
        if h.ndim == 1:
            h = h.reshape(-1, 1)
        h1 = np.concatenate([h, np.zeros((1, n_points))], axis=0)
        h2 = np.concatenate([np.zeros((1, n_points)), h], axis=0)
        shift = np.zeros((2 * n_points, n_points))
        for i in range(n_points):
            shift[i, i:] = 1
        overlap = np.dot(h1, h2.T) * dx
        return np.max(overlap)

    def repair(child):
        count = np.sum(child)
        if count > required_sum:
            idxs = np.where(child == 1)[0]
            to_flip = idxs[:count - required_sum]
            child[to_flip] = 0
        elif count < required_sum:
            idxs = np.where(child == 0)[0]
            to_flip = idxs[:required_sum - count]
            child[to_flip] = 1
        return child

    def fitness(h):
        return compute_overlap(h)

    population = np.zeros((100, n_points), dtype=int)
    for i in range(population.shape[0]):
        indices = np.random.choice(n_points, size=required_sum, replace=False)
        population[i, indices] = 1

    start_time = time.time()
    for gen in range(100):
        fitness_values = np.array([fitness(ind) for ind in population])
        parents = []
        for _ in range(population.shape[0]):
            idxs = np.random.choice(population.shape[0], 2, replace=False)
            if fitness_values[idxs[0]] < fitness_values[idxs[1]]:
                parents.append(population[idxs[0]])
            else:
                parents.append(population[idxs[1]])
        new_population = []
        for i in range(0, population.shape[0], 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < population.shape[0] else parents[i]
            if np.random.rand() < 0.7:
                idx = np.random.randint(1, n_points-1)
                c1 = np.concatenate([p1[:idx], p2[idx:]])
                c2 = np.concatenate([p2[:idx], p1[idx:]])
            else:
                c1, c2 = p1.copy(), p2.copy()
            for c in [c1, c2]:
                for j in range(n_points):
                    if np.random.rand() < 0.05:
                        c[j] = 1 - c[j]
            c1 = repair(c1)
            c2 = repair(c2)
            new_population.extend([c1, c2])
        population = np.array(new_population, dtype=int)
        if time.time() - start_time > budget_s - 1:
            break

    best_h = population[np.argmin(fitness(population))]
    return best_h.astype(float), compute_overlap(best_h), n_points
