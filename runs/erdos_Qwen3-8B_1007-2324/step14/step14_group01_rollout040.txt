import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import minimize

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum of h is integral_target

    start_time = time.time()

    # Determine initial guess
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_initial = initial_h[:30]
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = 1 - np.abs(x[:30] - 1)
        h_initial = np.clip(h_initial, 0, 1)

    def repair(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return np.zeros_like(h)
        scaling = integral_target / sum_clip
        scaled = h_clip * scaling
        scaled = np.clip(scaled, 0, 1)
        sum_scaled = np.sum(scaled)
        if np.abs(sum_scaled - integral_target) > 1e-5:
            return scaled
        return scaled

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = 1.0
    best_h = np.zeros(n_points)

    max_restarts = 10
    for restart in range(max_restarts):
        h_opt = h_initial.copy()
        if restart > 0:
            noise = np.random.normal(0, 0.1, h_opt.shape)
            h_opt = np.clip(h_opt + noise, 0, 1)
        h_opt_repaired = repair(h_opt)

        if np.sum(h_opt_repaired) < 1e-5:
            continue

        def objective(x):
            h = x.copy()
            h_repaired = repair(h)
            return compute_overlap(h_repaired)

        bounds = [(0.0, 1.0) for _ in range(len(h_opt_repaired))]

        result = minimize(
            objective,
            h_opt_repaired,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'disp': False, 'eps': 1e-8}

        if result.success:

            current_c5 = compute_overlap(current_h)

            current_h = h_opt_repaired

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.concatenate([current_h, current_h[::-1]])

        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

    return best_h, best_c5, n_points
