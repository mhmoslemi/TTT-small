            }
        )

        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            h_opt = current_h.copy()

        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

    best_h_clamped = np.clip(best_h, 0.0, 1.0)
    full_h = np.concatenate([best_h_clamped, best_h_clamped[::-1]])

    return full_h, best_c5, n_points
