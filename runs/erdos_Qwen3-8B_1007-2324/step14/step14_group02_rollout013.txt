import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    initial_h = kwargs.get('initial_h_values', None)
    n_points = 560
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def interpolate_initial(initial_h):
        if initial_h is None:
            return np.random.rand(n_points)
        x = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        return np.interp(x_new, x, initial_h)

    h_initial = interpolate_initial(initial_h)

    def repair(x):
        x_clip = np.clip(x, 0.0, 1.0)
        current_sum = np.sum(x_clip)
        if current_sum == 0:
            return np.zeros_like(x)
        scale = target_sum / current_sum
        scaled = x_clip * scale
        scaled = np.clip(scaled, 0.0, 1.0)
        return scaled

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    h_initial = repair(h_initial)

    result = opt.dual_annealing(objective, bounds=[(0, 1)] * n_points, x0=h_initial, maxiter=100, max_evals=500, disp=False)
    best_h = result.x
    best_c5 = result.fun
    best_h = repair(best_h)

    return best_h, best_c5, n_points
