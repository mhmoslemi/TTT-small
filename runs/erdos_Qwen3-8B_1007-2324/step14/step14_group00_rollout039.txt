    import time
    
    n_points = 60
    n_vars = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # 60/4 = 15
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_vars]
    else:
        np.random.seed(seed)
        h_opt = np.random.uniform(0, 1, n_vars)
        h_opt = h_opt * (integral_target / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (integral_target / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0, 1)
    
    def compute_c5(h):
        h_full = np.concatenate([h, 1 - h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    n_particles = 20
    n_iter = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    particles = np.random.uniform(0, 1, (n_particles, n_vars))
    for i in range(n_particles):
        particles[i, :] = particles[i, :] * (integral_target / np.sum(particles[i, :]))
        particles[i, :] = np.clip(particles[i, :], 0, 1)
        particles[i, :] = particles[i, :] * (integral_target / np.sum(particles[i, :]))
        particles[i, :] = np.clip(particles[i, :], 0, 1)
    
    velocities = np.random.uniform(-1, 1, (n_particles, n_vars))
    personal_best = particles.copy()
    global_best = particles[0, :].copy()
    best_c5 = compute_c5(global_best)
    
    start_time = time.time()
    for _ in range(n_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        
        for i in range(n_particles):
            r1 = np.random.rand(n_vars)
            r2 = np.random.rand(n_vars)
            velocities[i, :] = w * velocities[i, :] + c1 * r1 * (personal_best[i, :] - particles[i, :]) + c2 * r2 * (global_best - particles[i, :])
            particles[i, :] += velocities[i, :]
        
        for i in range(n_particles):
            particles[i, :] = np.clip(particles[i, :], 0, 1)
            current_sum = np.sum(particles[i, :])
            if current_sum != 0:
                scale = integral_target / current_sum
                particles[i, :] = np.clip(particles[i, :] * scale, 0, 1)
        
        for i in range(n_particles):
            current_c5 = compute_c5(particles[i, :])
            if current_c5 < compute_c5(personal_best[i, :]):
                personal_best[i, :] = particles[i, :].copy()
                if current_c5 < best_c5:
                    global_best = personal_best[i, :].copy()
                    best_c5 = current_c5
    
    final_h = np.concatenate([global_best, 1 - global_best[::-1]])
    return final_h, best_c5, n_points
