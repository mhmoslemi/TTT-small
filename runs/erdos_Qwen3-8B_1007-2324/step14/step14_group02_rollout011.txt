import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    n_points = 560
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.random.rand(n_points)
    else:
        h = np.random.rand(n_points)
    
    def repair(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        s = np.sum(x_clipped)
        if np.isclose(s, 0.0, atol=1e-8):
            return np.zeros_like(x)
        scale = (n_points / 2.0) / s
        x_repaired = x_clipped * scale
        return np.clip(x_repaired, 0.0, 1.0)
    
    h = repair(h)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = opt.dual_annealing(objective, bounds, 
                                maxiter=100, 
                                disp=False,
                                maxfun=int(budget_s * 10))
    
    best_h = repair(result.x)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
