import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for _ in range(5):
            c = np.random.uniform(0.25, 1.75)
            t = np.clip(1 - np.abs(np.linspace(0, 2, n_points) - c) / 0.5, 0, 1)
            h += t
        h *= (required_sum / np.sum(h))
        h = np.clip(h, 0, 1)

    def compute_c5(h_full):
        corr = correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_c5(np.zeros_like(h))
        scaling = required_sum / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_c5(repaired)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - required_sum,
        'jac': lambda x: np.ones_like(x)
    }]

    start_time = time.time()
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'ftol': 1e-9,
            'maxiter': 1000,
            'disp': False,
            'eps': 1e-8,
            'iprint': -1
        }
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h, compute_c5(h), n_points

    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
