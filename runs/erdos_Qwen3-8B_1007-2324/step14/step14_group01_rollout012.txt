from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100):

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Default initialization: triangular function
        x = np.linspace(0, 2, 60)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
        initial_h = h

    integral_target = 15.0

        current_sum = np.sum(x)
        if current_sum == 0:
            return np.zeros_like(x)
        scale = integral_target / current_sum

        clipped = np.clip(scaled, 0, 1)
        new_sum = np.sum(clipped)
        if new_sum == 0:

        final_scale = integral_target / new_sum

    n_points = len(initial_h)

        corr = np.correlate(h, 1 - h, mode='full')

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
