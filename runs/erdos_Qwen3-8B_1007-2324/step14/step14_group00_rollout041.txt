    if 'initial_h_values' in globals():

        h_opt_initial = np.random.rand(n_half)
        h_opt_initial = h_opt_initial / np.sum(h_opt_initial) * integral_target
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]
    
    start_time = time.time()
    best_c5 = float('inf')

        scale = integral_target / start_sum
        start = start * scale
        start_points.append(start)
    
    for start in start_points:
        if time.time() - start_time > budget_s - 1:
            break
        result = scipy.optimize.minimize(
