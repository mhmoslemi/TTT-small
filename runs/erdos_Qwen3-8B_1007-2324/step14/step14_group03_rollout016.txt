        np.random.shuffle(bits)
        population.append(bits)

    best_h = h.copy()
    best_c5 = compute_overlap(h)

    start_time = time.time()
    for gen in range(generations):
        fitness = [compute_overlap(ind) for ind in population]
        best_current = population[np.argmin(fitness)]
        if compute_overlap(best_current) < best_c5:
            best_c5 = compute_overlap(best_current)
            best_h = best_current.copy()
        if time.time() - start_time > budget_s - 1:
            break
        new_population = []
        for _ in range(pop_size):
            parent1 = np.random.choice(population, size=2, replace=False)
            parent1 = parent1[np.argmin([compute_overlap(p) for p in parent1])]
            parent2 = np.random.choice(population, size=2, replace=False)
            parent2 = parent2[np.argmin([compute_overlap(p) for p in parent2])]
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            if np.random.rand() < mutation_prob:
                num_mutations = np.random.randint(1, 5)
                indices = np.random.choice(n_points, size=num_mutations, replace=False)
                child[indices] ^= 1
            current_sum = np.sum(child)
            diff = current_sum - required_sum
            if abs(diff) > 1e-6:
                if diff > 0:
                    ones_indices = np.where(child == 1)[0]
                    if len(ones_indices) > diff:
                        child[ones_indices[:diff]] = 0
                    else:
                        child[ones_indices] = 0
                        needed = diff - len(ones_indices)
                        zeros_indices = np.where(child == 0)[0]
                        child[zeros_indices[:needed]] = 1
                else:
                    zeros_indices = np.where(child == 0)[0]
                    needed = -diff
                    if len(zeros_indices) > needed:
                        child[zeros_indices[:needed]] = 1
                    else:
                        child[zeros_indices] = 1
                        needed = -diff - len(zeros_indices)
                        ones_indices = np.where(child == 1)[0]
                        child[ones_indices[:needed]] = 0
            new_population.append(child)
        population = new_population

    return best_h, best_c5, n_points
