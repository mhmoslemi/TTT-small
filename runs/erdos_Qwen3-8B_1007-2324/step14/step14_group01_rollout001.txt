    def repair(x):
        current_sum = np.sum(x)
        if abs(current_sum - integral_target) < 1e-5:
            scale = 1.0
        else:
            scale = integral_target / current_sum
        corrected = np.clip(x * scale, 0, 1)
        for _ in range(3):
            missing = integral_target - np.sum(corrected)
            if abs(missing) < 0.1:
                break
            factor = 1 + (15 * missing) / np.sum(corrected)
            corrected = np.clip(x * min(factor, 2), 0, 1)
        return corrected

    def objective(x):

            np.clip(h, 0, 1, out=h)
            if np.sum(h) < 0 or np.sum(h) > 2 * integral_target:
                return np.inf
            corr = np.correlate(h, 1 - h, mode='full')
