import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        orig_len = len(h_initial)
        h_opt = h_initial[::max(1, orig_len // n_half)][:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_half)]

    def objective(x):
        return compute_overlap(x)

    result = scipy.optimize.minimize(
        objective, h_opt, method='SLSQP',
        bounds=bounds, options={'ftol': 1e-12, 'maxiter': 100}
    )
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)

    start_time = time.time()
    for _ in range(5):
        perturbation = np.random.normal(0, 0.03, n_half)
        h_perturbed = np.clip(best_h_opt + perturbation, 0, 1)
        result = scipy.optimize.minimize(
            objective, h_perturbed, method='SLSQP',
            bounds=bounds, options={'ftol': 1e-12, 'maxiter': 100}
        )
        if result.success:
            curr_c5 = compute_overlap(result.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h_opt = result.x
        if time.time() - start_time > budget_s - 1:
            break

    return np.concatenate([best_h_opt, 1 - best_h_opt[::-1]]), best_c5, n_points
