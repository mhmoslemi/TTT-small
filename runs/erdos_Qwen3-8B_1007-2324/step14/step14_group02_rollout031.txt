            options={'maxfun': 300, 'disp': False},
            x0=start
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    best_h = repair(best_h)
    return best_h, best_c5, n_points
