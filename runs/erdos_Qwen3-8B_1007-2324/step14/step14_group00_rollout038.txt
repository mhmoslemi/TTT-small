import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4  # 15

    # Initialize h_opt
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_half))
        h_opt = h_initial[::step][:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    best_c5 = float('inf')
    best_h = h_opt.copy()

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    n_weights = 50  # Adjust based on time constraints
    sampled_weights = [np.random.beta(2, 32) for _ in range(n_weights)]

    for w in sampled_weights:
        if time.time() - start_time > budget_s - 1:
            break
        def objective(x):
            h_clamped = np.clip(x, 0.0, 1.0)
            h_full = np.concatenate([h_clamped, h_clamped[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            overlap = np.max(corr) * dx
            variance = np.sum((h_clamped - 0.5)**2) * dx
            return w * overlap + (1 - w) * variance
        # Constraints for integral
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_half)}]
        bounds = [(0.0, 1.0) for _ in range(n_half)]
        result = scipy.optimize.minimize(objective, h_opt, method='COBYLA', bounds=bounds, constraints=constraints)
        if result.success:
            current_h = result.x
            current_overlap = compute_overlap(current_h)
            if current_overlap < best_c5:
                best_c5 = current_overlap
                best_h = current_h.copy()

    full_h = np.concatenate([best_h, best_h[::-1]])
    return full_h, best_c5, n_points
