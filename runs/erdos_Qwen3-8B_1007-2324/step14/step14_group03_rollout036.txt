    dx = 2.0 / n_points

    def compute_overlap(h_binary):
        corr = np.correlate(h_binary, 1 - h_binary, mode='full')
        return np.max(corr) * dx

    # Initial population
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h_binary = (h > 0.5).astype(int)
        current_sum = np.sum(h_binary)
        diff = required_sum - current_sum
        if diff > 0:
            zeros = np.where(h_binary == 0)[0]
            h_binary[zeros[:diff]] = 1
        elif diff < 0:
            ones = np.where(h_binary == 1)[0]
            h_binary[ones[:abs(diff)]] = 0
        pop = [h_binary]
    else:
        while True:
            h_binary = np.random.rand(n_points) > 0.5
            if np.sum(h_binary) == required_sum:
                break
        pop = [h_binary]

    pop_size = len(pop)
    generations = 100
    mutation_rate = 0.05
    best_h = pop[0]
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    for _ in range(generations):
        new_pop = []
        for _ in range(pop_size):
            # Selection
            indices = np.random.choice(len(pop), 3, replace=False)
            parent = pop[np.argmin([compute_overlap(pop[i]) for i in indices])]
            # Crossover
            split = np.random.randint(1, n_points)
            child = np.zeros(n_points, dtype=int)
            child[:split] = parent[:split]
            child[split:] = np.random.randint(0, 2, n_points - split)
            # Repair
            cnt = np.sum(child)
            if cnt > required_sum:
                ones = np.where(child == 1)[0]
                flip = np.random.choice(ones, cnt - required_sum, replace=False)
                child[flip] = 0
            elif cnt < required_sum:
                zeros = np.where(child == 0)[0]
                flip = np.random.choice(zeros, required_sum - cnt, replace=False)
                child[flip] = 1
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] ^= 1
            # Repair after mutation
            cnt = np.sum(child)
            if cnt > required_sum:
                ones = np.where(child == 1)[0]
                flip = np.random.choice(ones, cnt - required_sum, replace=False)
                child[flip] = 0
            elif cnt < required_sum:
                zeros = np.where(child == 0)[0]
                flip = np.random.choice(zeros, required_sum - cnt, replace=False)
                child[flip] = 1
            new_pop.append(child)
        # Evaluate and combine
        new_fit = [compute_overlap(ind) for ind in new_pop]
        combined = np.vstack((pop, new_pop))
        fit = [compute_overlap(ind) for ind in combined]
        indices = np.argsort(fit)[:pop_size]
        pop = combined[indices]
        current_best = np.min(fit[indices])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = pop[np.argmin(fit[indices])]
        if time.time() - start_time > budget_s - 1:
            break
    return best_h, best_c5, n_points
