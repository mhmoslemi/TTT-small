import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def optimize(h_opt):
        def objective(x):
            return compute_overlap(x)
        bounds = [(0.0, 1.0) for _ in range(n_half)]
        constraints = [{
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        }]
        x0 = h_opt.copy()
        start_time = time.time()
        result = scipy.optimize.minimize(
            objective, x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-12,
                'maxiter': 1000,
                'disp': False,
                'eps': 1e-8
            }
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            return None
        if result.success:
            return result.x
        return None

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt = h_initial[::step][:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    best_c5 = float('inf')
    best_h_opt = h_opt

    for _ in range(5):
        noise = np.random.normal(0, 0.1, n_half)
        perturbed = np.clip(h_opt + noise, 0, 1)
        optimized = optimize(perturbed)
        if optimized is not None:
            current_c5 = compute_overlap(optimized)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_opt = optimized

    best_h_full = np.concatenate([best_h_opt, 1 - best_h_opt[::-1]])
    return best_h_full, best_c5, n_points
