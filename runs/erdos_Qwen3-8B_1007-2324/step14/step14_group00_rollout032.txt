        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                if time.time() - start_time > budget_s - 1:
                    break
    
    h_full = np.concatenate([best_h, 1 - best_h[::-1]])
    return h_full, best_c5, n_points
