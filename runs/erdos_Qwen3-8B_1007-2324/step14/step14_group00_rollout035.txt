        h_initial = np.array(initial_h_values)
        original_len = len(h_initial)
        step = int(np.ceil(original_len / n_points))
        h_opt_initial = h_initial[::step][:n_half]
    else:
        h_opt_initial = np.random.rand(n_half)
        h_opt_initial = h_opt_initial * (15.0 / np.sum(h_opt_initial))

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    constraints = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - integral_target,
        'jac': lambda x: np.ones_like(x)
    }]

    # Sample weights from Beta distribution
    weights = np.random.beta(2, 32, size=20)
    best_c5 = float('inf')
    best_h_opt = h_opt_initial.copy()

    start_time = time.time()
    for w in weights:
        def objective(x):
            h_opt = x.copy()
            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            reg = np.sum(h_opt ** 2)
            return w * c5 + (1 - w) * reg

        result = scipy.optimize.minimize(
            objective,
            h_opt_initial,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-12,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            h_opt = result.x
            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h_opt = h_opt

        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break

    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    return h_full, best_c5, n_points
