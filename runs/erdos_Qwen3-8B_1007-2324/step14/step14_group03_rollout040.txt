def run(seed=42, budget_s=100, **kwargs):

    n_points = 280
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)

        current_sum = np.sum(h_binary)
        diff = required_sum - current_sum
        zeros = np.where(h_binary == 0)[0]
        ones = np.where(h_binary == 1)[0]
        if diff > 0:
            h_binary[zeros[:diff]] = 1
        elif diff < 0:
            h_binary[ones[:abs(diff)]] = 0
        h = h_binary
    else:
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h = np.zeros(n_points, dtype=float)
        h[indices] = 1.0

    def compute_overlap(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = x.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return compute_overlap(np.zeros_like(h))
        scaling = required_sum / current_sum
        repaired = np.clip(h * scaling, 0, 1)
        return compute_overlap(repaired)

        'fun': lambda x: np.sum(x) - required_sum,
        'jac': lambda x: np.ones_like(x)
    }]

    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
