import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    initial_h_values = kwargs.get('initial_h_values')
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h_binary = (h > 0.5).astype(int)
        cnt = np.sum(h_binary)
        if cnt != required_sum:
            diff = int(required_sum - cnt)
            if diff > 0:
                z = np.where(h_binary == 0)[0]
                h_binary[z[:diff]] = 1
            else:
                o = np.where(h_binary == 1)[0]
                h_binary[o[:abs(diff)]] = 0
        pop = [h_binary]
    else:
        pop = np.random.randint(0, 2, (50, n_points))
        for i in range(50):
            while np.sum(pop[i]) != required_sum:
                idx = np.random.randint(n_points)
                pop[i][idx] ^= 1

    def compute_overlap(h_bin):
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        return np.max(corr) * dx

    best_h = pop[0].copy()
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    for _ in range(100):
        fit = np.array([compute_overlap(ind) for ind in pop])
        curr_best = pop[np.argmin(fit)]
        if fit[np.argmin(fit)] < best_c5:
            best_c5 = fit[np.argmin(fit)]
            best_h = curr_best.copy()

        parents = []
        for _ in range(50):
            t = np.random.choice(50, 3, replace=False)
            winner = t[np.argmin(fit[t])]
            parents.append(pop[winner].copy())

        children = []
        for i in range(0, 50, 2):
            p1, p2 = parents[i], parents[i+1]
            pt = np.random.randint(1, n_points-1)
            c1 = np.concatenate([p1[:pt], p2[pt:]])
            c2 = np.concatenate([p2[:pt], p1[pt:]])
            children.extend([c1, c2])

        for c in children:
            if np.random.rand() < 0.05:
                idx = np.random.randint(n_points)
                c[idx] ^= 1

        for c in children:
            cnt = np.sum(c)
            if cnt != required_sum:
                if cnt < required_sum:
                    z = np.where(c == 0)[0]
                    c[z[:int(required_sum - cnt)]] = 1
                else:
                    o = np.where(c == 1)[0]
                    c[o[:int(cnt - required_sum)]] = 0

        pop = np.array(children)
        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
