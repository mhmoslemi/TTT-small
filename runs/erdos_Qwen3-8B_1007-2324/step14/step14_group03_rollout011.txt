        else:
            offspring = [parents[0].copy(), parents[1].copy()]

        # Mutation
        for i in range(len(offspring)):
            child = offspring[i].copy()
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] ^= 1
            # Repair sum
            current_sum = np.sum(child)
            if current_sum != required_sum:
                diff = current_sum - required_sum
                for _ in range(abs(diff)):
                    if diff > 0:
                        # Flip a 1 to 0
                        idx = np.random.choice(np.where(child == 1)[0])
                        child[idx] = 0
                    else:
                        # Flip a 0 to 1
                        idx = np.random.choice(np.where(child == 0)[0])
                        child[idx] = 1
            offspring[i] = child

        # Replace the population with new individuals
        population = offspring + [ind for ind in population if evaluate_individual(ind) < best_c5]

        # Update best solution
        current_fitness = [evaluate_individual(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx]

    return best_h.astype(float), best_c5, n_points
