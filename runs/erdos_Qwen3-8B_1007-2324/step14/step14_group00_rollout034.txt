    if initial_h_values is not None:
        h_initial = np.array(initial_h_values)
        h_opt_initial = h_initial[:n_half]
    else:
        h_opt_initial = np.random.uniform(0, 1, size=n_half)

    n_particles = 30
    n_iterations = 100
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.1

    particles = []
    velocities = []
    p_best_positions = []
    p_best_values = []
    g_best_position = None
    g_best_value = float('inf')

    for _ in range(n_particles):
        h_opt = np.random.uniform(0, 1, n_half)
        particles.append(h_opt)
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_half))
        p_best_positions.append(h_opt.copy())
        p_best_values.append(compute_overlap(h_opt))

    for i in range(n_particles):
        if p_best_values[i] < g_best_value:
            g_best_value = p_best_values[i]
            g_best_position = p_best_positions[i].copy()

    start_time = time.time()
    for iteration in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_half), np.random.rand(n_half)
            velocity = velocities[i] + w * (p_best_positions[i] - particles[i]) * r1 + c1 * (p_best_positions[i] - particles[i]) * r2 + c2 * (g_best_position - particles[i]) * r2
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            velocities[i] = velocity
            particles[i] += velocity
            particles[i] = np.clip(particles[i], 0, 1)
            current_value = compute_overlap(particles[i])
            if current_value < p_best_values[i]:
                p_best_positions[i] = particles[i].copy()
                p_best_values[i] = current_value

        for i in range(n_particles):
            if p_best_values[i] < g_best_value:
                g_best_value = p_best_values[i]
                g_best_position = p_best_positions[i].copy()

    h_opt_best = g_best_position
    h_full = np.concatenate([h_opt_best, 1 - h_opt_best[::-1]])
    c5_bound = compute_overlap(g_best_position)
    return h_full, c5_bound, n_points
