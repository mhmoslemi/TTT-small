            corr = np.correlate(h_full, 1 - h_full, mode='full')
            overlap = np.max(corr) * dx
            # Variance of h, since mean is 1 under constraint
            variance = np.sum((h - 1)**2)
            return w * overlap + (1 - w) * variance

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}]
        bounds = [(0.0, 1.0) for _ in range(n_half)]

        result = scipy.optimize.minimize(
            objective,
            h_opt,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'disp': False, 'ftol': 1e-8}
        )

        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

        # Early termination if time runs out
        if time.time() - start_time > budget_s - 10:
            break

    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
