    gbest = particles[0].copy()
    gbest_error = np.inf

    for i in range(num_particles):
        err = compute_overlap(particles[i])
        pbest_error[i] = err
        if err < gbest_error:
            gbest = particles[i].copy()
            gbest_error = err

    start_time = time.time()
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        for p in range(num_particles):
            r1 = np.random.rand(n_vars)
            r2 = np.random.rand(n_vars)
            vel[p] = w * vel[p] + c1 * r1 * (pbest[p] - particles[p]) + c2 * r2 * (gbest - particles[p])
            particles[p] += vel[p]
            particles[p] = np.clip(particles[p], 0, 1)
            err = compute_overlap(particles[p])
            if err < pbest_error[p]:
                pbest[p] = particles[p].copy()
                pbest_error[p] = err
                if err < gbest_error:
                    gbest = particles[p].copy()
                    gbest_error = err

    initial_c5 = compute_overlap(h_opt)
    if initial_c5 < gbest_error:
        best_h_full = np.concatenate([h_opt, 1 - h_opt[::-1]])
        best_c5 = initial_c5
    else:
        best_h_full = np.concatenate([gbest, 1 - gbest[::-1]])
        best_c5 = gbest_error

    return best_h_full, best_c5, n_points
