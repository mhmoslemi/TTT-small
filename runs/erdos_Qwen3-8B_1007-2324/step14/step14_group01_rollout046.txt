    for _ in range(num_restarts):

        noise = np.random.uniform(-0.1, 0.1, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = repair(perturbed, integral_target)
        result = minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        if result.success:
            current_h = result.x
        else:
            current_h = perturbed
        current_h_repaired = repair(current_h, integral_target)
        current_c5 = compute_overlap(current_h_repaired)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h_repaired

    return best_h, best_c5, n_points
