import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair(h, target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target / current_sum
        return np.clip(h * scaling, 0, 1)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h[:n_points]
        initial_h = repair(initial_h, target_sum)
    else:
        x = np.linspace(0, 2, n_points)
        initial_h = 1 - np.abs(x - 1)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = repair(initial_h, target_sum)

    def objective(h):
        h = repair(h, target_sum)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1.0),
        crossover=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False,
        seed=seed
    )

    best_h = result.x
    best_c5 = objective(best_h)
    best_h = repair(best_h, target_sum)
    best_c5 = objective(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return initial_h, objective(initial_h), n_points

    return best_h, best_c5, n_points
