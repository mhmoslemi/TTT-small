import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize

    n_points = 60
    n_half = 30
    dx = 2.0 / n_points
    integral_target = 15.0

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial_half = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_initial_half = np.clip(1 - np.abs(x - 1), 0, 1)[:n_half]

    def objective(x):
        h_full = np.concatenate([x, 1 - x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e6 * abs(np.sum(x) - integral_target)
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_half)]
    x0 = h_initial_half.copy()

    start_time = time.time()
    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        x0=x0,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        maxiter=500,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return np.concatenate([x0, 1 - x0[::-1]]), objective(x0) - 1e6 * abs(np.sum(x0) - 15), n_points

    best_h_opt = result.x
    best_c5 = objective(best_h_opt) - 1e6 * abs(np.sum(best_h_opt) - 15)
    return np.concatenate([best_h_opt, 1 - best_h_opt[::-1]]), best_c5, n_points
