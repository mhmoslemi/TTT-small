                indices = np.where(h_binary == 1)[0]
                flip_indices = np.random.choice(indices, current_sum - target_sum, replace=False)
                h_binary[flip_indices] = 0
            else:
                indices = np.where(h_binary == 0)[0]
                flip_indices = np.random.choice(indices, target_sum - current_sum, replace=False)
                h_binary[flip_indices] = 1
        h = h_binary
    else:
        h = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, target_sum, replace=False)
        h[indices] = 1

    def compute_overlap(h_bin):
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    num_generations = 100
    population = [np.zeros(n_points, dtype=int) for _ in range(pop_size)]
    for i in range(pop_size):
        indices = np.random.choice(n_points, target_sum, replace=False)
        population[i][indices] = 1
    fitness = [compute_overlap(pop) for pop in population]

    start_time = time.time()
    for gen in range(num_generations):
        def select_parent():
            candidates = np.random.choice(range(pop_size), size=3, replace=False)
            return population[np.argmin([fitness[i] for i in candidates])]
        p1, p2 = select_parent(), select_parent()
        crossover = np.random.randint(1, n_points - 1)
        child = np.copy(p1)
        child[crossover:] = p2[crossover:]

        child_sum = np.sum(child)
        if child_sum != target_sum:
            if child_sum > target_sum:
                ones = np.where(child == 1)[0]
                flip = np.random.choice(ones, child_sum - target_sum, replace=False)
                child[flip] = 0
            else:
                zeros = np.where(child == 0)[0]
                flip = np.random.choice(zeros, target_sum - child_sum, replace=False)
                child[flip] = 1

        if np.random.rand() < 0.05:
            idx = np.random.randint(n_points)
            child[idx] = 1 - child[idx]
            child_sum = np.sum(child)
            if child_sum != target_sum:
                if child_sum > target_sum:
                    ones = np.where(child == 1)[0]
                    flip = np.random.choice(ones, child_sum - target_sum, replace=False)
                    child[flip] = 0
                else:
                    zeros = np.where(child == 0)[0]
                    flip = np.random.choice(zeros, target_sum - child_sum, replace=False)
                    child[flip] = 1

        worst_idx = np.argmax(fitness)
        child_fit = compute_overlap(child)
        if child_fit < fitness[worst_idx]:
            population[worst_idx] = child
            fitness[worst_idx] = child_fit
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

    best_idx = np.argmin(fitness)
    return population[best_idx], fitness[best_idx], n_points
