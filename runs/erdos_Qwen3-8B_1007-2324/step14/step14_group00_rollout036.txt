            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-12,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    final_h = np.concatenate([best_h, 1.0 - best_h[::-1]])
    return final_h, best_c5, n_points
