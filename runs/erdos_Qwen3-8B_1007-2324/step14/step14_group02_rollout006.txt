import numpy as np
import scipy.optimize as opt

def repair(x, n_points):
    s = np.sum(x)
    if s == 0:
        return np.zeros_like(x)
    x_clipped = np.clip(x, 0.0, 1.0)
    scale = (n_points / 2) / np.sum(x_clipped)
    x_repaired = x_clipped * scale
    return x_repaired

def objective(x, n_points):
    h = repair(x, n_points)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 560
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = np.concatenate([h_initial, h_initial])
        elif len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        h_initial = repair(h_initial, n_points)
    else:
        h_initial = np.random.rand(n_points)
        h_initial = repair(h_initial, n_points)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.dual_annealing(objective, bounds, args=(n_points,), maxfun=1000)
    best_h = result.x
    best_h = repair(best_h, n_points)
    best_c5 = objective(best_h, n_points)
    return best_h, best_c5, n_points
