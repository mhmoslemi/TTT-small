import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import time

    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    # Repair initial guess
    current_sum = np.sum(initial_h)
    if current_sum != 0:
        scaling = target_sum / current_sum
        initial_h = np.clip(initial_h * scaling, 0, 1)
    else:
        initial_h = np.full(n_points, 0.5)

    # Define repair function
    def repair(x):
        current_sum = np.sum(x)
        if current_sum == 0:
            return np.zeros_like(x)
        scaling = target_sum / current_sum
        return np.clip(x * scaling, 0, 1)

    # Compute overlap
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Objective function
    def objective(x):
        h_candidate = repair(x)
        return compute_overlap(h_candidate)

    # Bounds and DE settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    strategy = 'best1bin'
    popsize = 20
    tol = 1e-6
    mutation = (0.5, 1)
    recombination = 0.8
    maxiter = 100

    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        strategy=strategy,
        popsize=popsize,
        tol=tol,
        mutation=mutation,
        recombination=recombination,
        seed=seed,
        polish=True,
        disp=False,
        maxiter=maxiter,
        vectorized=True
    )

    best_h = repair(result.x)
    best_c5 = compute_overlap(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        best_h = initial_h
        best_c5 = compute_overlap(best_h)

    return best_h, best_c5, n_points
