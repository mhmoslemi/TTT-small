    else:
        n_points = 560
        h_opt = np.random.rand(n_points)

    def repair(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        s = np.sum(x_clipped)
        if s == 0:
            return np.zeros_like(x)
        scale = (n_points / 2) / s
        return x_clipped * scale

    h_opt = repair(h_opt)
    dx = 2.0 / n_points

    def objective(x):
        x_repaired = repair(x)
        corr = np.correlate(x_repaired, 1 - x_repaired, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.dual_annealing(objective, bounds=bounds, options={
        'maxiter': 100,
        'tmax': 100,
        'tmin': 0.1,
        'maxfun': max(10, int(budget_s * 5)),
    })

    best_h = result.x
    best_h = repair(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return best_h, c5_bound, n_points
