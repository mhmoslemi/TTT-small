def run(seed=42, budget_s=100, **kwargs):

    n_points = 60

    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if initial_h is not None:
        def normalize(h):
            s = np.sum(h)
            if s < 1e-6:
                return h * 0.0
            return h * (integral_target / s)
        start_points.append(normalize(initial_h))
        for _ in range(3):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def binary_initial():
        res = np.random.rand(n_points)
        res[np.random.choice(n_points, size=n_points//2, replace=False)] = 1.0
        return normalize(res)

    def random_initial():
        res = np.random.rand(n_points)
        return normalize(res)

    def structured_initial():
        res = np.zeros(n_points)
        seg_size = n_points // 4
        res[:seg_size] = 1.0
        res[seg_size:seg_size*2] = 0.0
        res[seg_size*2:seg_size*3] = 1.0
        res[seg_size*3:] = 0.0
        return normalize(res)

    start_points.extend([binary_initial(), random_initial(), structured_initial(), structured_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
