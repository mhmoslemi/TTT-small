import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    def build_h(c, w):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(4):
            start = c[i] - w[i] / 2
            end = c[i] + w[i] / 2
            h = np.maximum(h, (x >= start) * (x <= end))
        return h
    
    def compute_overlap(c, w):
        h = build_h(c, w)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

            def objective(x):
                c = x[::2]

            result = scipy.optimize.minimize(
