        start_points.append(first_half)
    
    # Generate random initial guesses
    def create_initial():
        base = np.random.rand(n_points // 2)
        base = base / np.sum(base) * 15  # ensure sum is 15
        return base
    
    start_points.append(create_initial())
    start_points.append(np.random.rand(n_points//2) * 15)
    start_points.append(np.zeros(n_points//2) + 15/(n_points//2))
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = h_start * 15 / np.sum(h_start)
        h_start = np.clip(h_start, 0.0, 1.0)
        
        def compute_overlap(first_half):
            full_h = np.concatenate([first_half, first_half[::-1]])
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            corr = corr * dx
            return np.max(corr)
        
        def objective(x):
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - 15
        
        result = opt.minimize(
            objective,
            h_start,
            bounds=[(0.0, 1.0) for _ in range(len(h_start))],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.concatenate([result.x, result.x[::-1]])
    
    return best_h, best_c5, n_points
