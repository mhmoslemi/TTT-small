    binary[ones] = 1.0
    binary = binary * (integral_target / np.sum(binary))
    start_points.append(binary)

    random = np.random.rand(n_points)
    random = random * (integral_target / np.sum(random))
    start_points.append(random)

    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(2 * np.pi * 0.5 * x)
    sine = np.clip(sine, 0.0, 1.0)
    sine = sine * (integral_target / np.sum(sine))
    start_points.append(sine)

    block = np.zeros(n_points)
    block[:n_points//3] = 1.0
    block[n_points//3:2*n_points//3] = 0.5
    block[2*n_points//3:] = 0.25
    block = block * (integral_target / np.sum(block))
    start_points.append(block)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def project_to_feasible(h):
        sum_h = np.sum(h) * dx
        if sum_h == 0:
            return h
        scale_factor = integral_target / sum_h
        h_scaled = h * scale_factor
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        return h_scaled

    def compute_overlap_with_penalty(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped) * dx
        penalty = 1e6 * np.abs(sum_h - 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx + penalty

    def compute_overlap(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        result_de = scipy.optimize.differential_evolution(
            fun=compute_overlap_with_penalty,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            disp=False
        )
        if result_de.success:
            current_h = result_de.x
            current_h_projected = project_to_feasible(current_h)
            current_c5 = compute_overlap(current_h_projected)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result_slsqp = scipy.optimize.minimize(
                compute_overlap,
                current_h_projected,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={
                    'ftol': 1e-12,
                    'maxiter': 300,
                    'disp': False,
                    'eps': 1e-8
                }
            )
            if result_slsqp.success:
                current_c5 = compute_overlap(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        else:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result_slsqp = scipy.optimize.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={
                    'ftol': 1e-12,
                    'maxiter': 300,
                    'disp': False,
                    'eps': 1e-8
                }
            )
            if result_slsqp.success:
                current_c5 = compute_overlap(result_slsqp.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

    return best_h, best_c5, n_points
