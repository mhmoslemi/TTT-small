import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30
    half_length = n_points // 2  # 30

    h_initial = None
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_initial = h_initial[:half_length]

    def objective(h_vec):
        sum_h = np.sum(h_vec)
        if np.abs(sum_h) < 1e-6:
            h_norm = np.ones_like(h_vec)
        else:
            scaling = integral_target / sum_h
            h_norm = h_vec * scaling
        h_full = np.concatenate([h_norm, h_norm[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_length)]
    if h_initial is not None:
        initial_guess = h_initial
    else:
        initial_guess = np.random.rand(half_length)

    result = scipy.optimize.differential_evolution(

        mutation=(0.4, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        polish=True,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_c5 = objective(best_h)
    best_h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
