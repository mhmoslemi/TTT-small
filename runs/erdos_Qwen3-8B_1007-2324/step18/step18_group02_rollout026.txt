import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s < 1e-6:
            return h * 0.0
        return h * (integral_target / s)

    start_points = []

    if initial_h is not None:
        start_points.append(normalize(initial_h))
        for _ in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def structured_initial():
        res = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                res[i] = 1.0
        return normalize(res)

    def low_freq_initial():
        res = np.ones(n_points) * 0.5
        res[::2] += 0.5
        res[1::2] -= 0.5
        return normalize(res)

    def random_initial():
        half = n_points // 2
        res = np.random.rand(half) * (integral_target / np.sum(np.random.rand(half)))
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([structured_initial(), low_freq_initial(), random_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
            options={'ftol': 1e-10, 'maxiter': 100, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    return best_h, best_c5, n_points
