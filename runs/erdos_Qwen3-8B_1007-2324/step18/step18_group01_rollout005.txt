        start_points = []

    # Add structured initializations
    def create_symmetric_guess():
        base = np.random.rand(n_points // 2)

        return np.concatenate([base, base[::-1]])[:n_points]
    
    start_points.append(create_symmetric_guess())
    start_points.append(np.ones(n_points) * 0.5)
    start_points.append(np.zeros(n_points) + 0.5)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        # Ensure symmetry and normalize
        h_start = np.concatenate([h_start[:n_points//2], h_start[:n_points//2][::-1]])
        h_start = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def objective(x):
            h = np.concatenate([x, x[::-1]])
            return compute_overlap(h)

        def constraint(x):
            return np.sum(x) - integral_target / 2

        result = scipy.optimize.minimize(
            objective,
            h_start[:n_points//2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points//2)],
