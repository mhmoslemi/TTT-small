        h_initial = np.array(initial_h_values)
        if len(h_initial) != n:
            h_initial = h_initial[:n]
        start_points.append(h_initial[:n-1])
    
    start_points.append(np.random.rand(n-1))
    start_points.append(np.full(n-1, 0.5))
    
    best_c5 = np.inf
    best_h = np.zeros(n)
    
    for x_start in start_points:
        x_start = np.clip(x_start, 0.0, 1.0)
        if x_start.shape[0] != n-1:
            x_start = x_start[:n-1]
        
        def obj(x):
            total = np.sum(x)
            y = integral_target - total
            h = np.concatenate([x, [y]])
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        cons = [
            {'type': 'ineq', 'fun': lambda x: integral_target - np.sum(x)},
            {'type': 'ineq', 'fun': lambda x: np.sum(x) + 1 - integral_target}
        ]
        bounds = [(0.0, 1.0) for _ in range(n-1)]
        
        options = {
            'ftol': 1e-8,
            'maxiter': 50,
            'disp': False,
            'eps': 1e-8
        }
        
        try:
            result = scipy.optimize.minimize(
                obj,
                x_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options=options
            )
            if result.success:
                current_h = np.concatenate([result.x, [integral_target - np.sum(result.x)]])
                if np.all(current_h >= 0) and np.all(current_h <= 1):
                    current_c5 = obj(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h
        except:
            pass
    
    if best_c5 == np.inf:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)[:n]
        else:
            best_h = np.full(n, 0.5)
        best_c5 = obj(np.full(n-1, 0.5))
    
    return best_h, best_c5, n
