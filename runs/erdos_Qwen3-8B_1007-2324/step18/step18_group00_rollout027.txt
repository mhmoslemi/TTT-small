import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points // 2  # 30
    half_length = n_points // 2  # 30

    population = []

    # Initialize population with initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        initial_h = np.clip(initial_h, 0, 1)
        if np.sum(initial_h) > 0:
            initial_h = initial_h * (integral_target / np.sum(initial_h))
        population.append(initial_h[:half_length])
    else:
        population.append(np.random.rand(half_length))

    # Generate additional initial parents
    for _ in range(3):
        perturbed = population[0].copy()
        perturbed += np.random.normal(0, 0.05, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) > 0:
            perturbed = perturbed * (integral_target / np.sum(perturbed))
        population.append(perturbed)

    # Add rotated version
    if len(population) > 0:
        rotated = np.roll(population[0], np.random.randint(1, half_length))
        rotated = np.clip(rotated, 0, 1)
        if np.sum(rotated) > 0:
            rotated = rotated * (integral_target / np.sum(rotated))
        population.append(rotated)

    start_time = time.time()
    best_c5 = np.inf
    best_h = population[0].copy()

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')

            fitnesses.append(current_c5)

        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        current_best = population[min_fitness_idx]
        current_best_c5 = fitnesses[min_fitness_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()

        # Select top half as parents
