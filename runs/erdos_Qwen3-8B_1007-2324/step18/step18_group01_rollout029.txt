import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Load initial guess if available
    initial_h = None

    start_points.append(create_initial())

    start_points.append(np.zeros(n_points) + 0.5)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_overlap(h):

            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def objective(x):
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - integral_target / 2
        
        result = scipy.optimize.minimize(

            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
