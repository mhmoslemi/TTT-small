import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2
    
    # Load initial solution if available

        if 'initial_h_values' in globals():

    def build_h(c, w):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(4):
            left = c[i] - w[i]/2
            right = c[i] + w[i]/2
            mask = (x >= left) & (x <= right)
            count = np.sum(mask)
            if count > 0:
                h[mask] = (w[i] / (dx * count))
        return np.clip(h, 0.0, 1.0)
    
    def enforce_bounds(c, w):
        for i in range(4):
            if w[i] < 0.1:
                w[i] = 0.1
            if c[i] - w[i]/2 < 0:
                c[i] = w[i]/2
            if c[i] + w[i]/2 > 2:
                c[i] = 2 - w[i]/2
        return c, w
    
    def compute_overlap(c, w):

            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(c_w):
        c, w = c_w[:4], c_w[4:]
