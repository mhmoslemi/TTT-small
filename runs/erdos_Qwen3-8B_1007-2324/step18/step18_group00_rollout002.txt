import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.zeros(n_points)
        h_initial[:15] = 1.0
        h_initial[30:45] = 1.0
        h_initial = h_initial * (integral_target / np.sum(h_initial))

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    # Transformations of initial_h_values
    starts = [h_initial]
    starts.append(np.roll(h_initial, 10))
    starts.append(np.clip(h_initial + np.random.normal(0, 0.05, n_points), 0, 1))
    starts.append(np.clip(h_initial * np.random.uniform(0.8, 1.2, n_points), 0, 1))
    starts.append(np.concatenate([h_initial[:30], np.zeros(30)]))
    starts.append(np.concatenate([np.zeros(30), h_initial[:30]]))
    starts.append(np.concatenate([h_initial[:15], h_initial[15:30], np.zeros(15)]))
    starts.append(np.zeros(n_points))
    starts.append(np.random.rand(n_points) * integral_target / np.sum(np.random.rand(n_points)))

    for h_start in starts:
        h_start = h_start[:n_points]
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (integral_target / np.sum(h_start))

        def objective(x):
            return compute_overlap(x)

        constraints = ({
            'type': 'eq',
            'fun': lambda x: np.sum(x) - integral_target,
            'jac': lambda x: np.ones_like(x)
        })

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=constraints,
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
