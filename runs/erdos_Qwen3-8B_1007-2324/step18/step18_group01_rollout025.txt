import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    S = n_points / 2
    best_c5 = np.inf
    best_x = np.random.rand(n_points - 1)
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) == n_points:
            start_x = h_initial[:n_points - 1]
            start_x = np.clip(start_x, 0, 1)
            start_x = start_x * (S / np.sum(start_x)) if np.sum(start_x) > 0 else start_x
            best_x = start_x
    
    def compute_obj(x):
        total = np.sum(x)
        y = S - total
        h = np.concatenate([x, [y]])
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def constraint1(x):
        return np.sum(x) - (S - 1)
    
    def constraint2(x):
        return S - np.sum(x)
    
    bounds = [(0.0, 1.0) for _ in range(n_points - 1)]
    constraints = [{'type': 'ineq', 'fun': constraint1}, {'type': 'ineq', 'fun': constraint2}]
    
    for _ in range(5):
        x_start = best_x * 0.9 + np.random.rand(n_points - 1) * 0.1
        x_start = np.clip(x_start, 0, 1)
        x_start = x_start * (S / np.sum(x_start)) if np.sum(x_start) > 0 else x_start
        
        result = scipy.optimize.minimize(
            compute_obj,
            x_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 50, 'disp': False}
        )
        if result.success:
            current_c5 = compute_obj(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_x = result.x
    
    total = np.sum(best_x)
    y = S - total
    h = np.concatenate([best_x, [y]])
    h = np.clip(h, 0, 1)
    
    return h, best_c5, n_points
