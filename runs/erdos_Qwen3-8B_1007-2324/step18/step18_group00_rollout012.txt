import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial[:n_points // 2])

    def shift_parent(parent, shift_amount):
        return np.roll(parent, shift_amount)

    def invert_segment(parent, window):
        return np.concatenate([parent[:window], parent[window:][::-1]])

    def blend_parent(parent, noise_scale=0.05):
        noise = np.random.normal(0, noise_scale, len(parent))
        return parent * 0.7 + noise * 0.3

    if 'initial_h_values' in globals():
        parent = h_initial[:n_points // 2]
        start_points.append(shift_parent(parent, np.random.randint(1, n_points // 2)))
        start_points.append(invert_segment(parent, np.random.randint(1, n_points // 2)))
        start_points.append(blend_parent(parent))

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = np.zeros(n_points // 2)
    best_c5 = np.inf

    for h_start in start_points:
        h_start = h_start.copy()
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)
        else:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)

        def objective(h):
            h_clipped = np.clip(h, 0, 1)
            h_full = np.concatenate([h_clipped, h_clipped[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like(x)}],
            options={'ftol': 1e-10, 'maxiter': 300, 'disp': False, 'eps': 1e-8}
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
