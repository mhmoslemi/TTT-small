import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Load initial guess
    initial_h = np.array([])
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        start_points = [initial_h]
    else:
        start_points = []

    # Add periodic and symmetric initial guesses
    def create_periodic():
        base = np.random.rand(n_points // 2)
        base /= np.sum(base)
        base *= integral_target / 2
        return np.concatenate([base, base[::-1]])[:n_points]
    
    def create_symmetric():
        base = np.linspace(0, 1, n_points // 2)
        base = np.sort(base)
        base /= np.sum(base)
        base *= integral_target / 2
        return np.concatenate([base, base[::-1]])[:n_points]
    
    start_points.append(create_periodic())
    start_points.append(create_symmetric())
    start_points.append(np.random.rand(n_points) * 2 - 1)
    start_points.append(np.zeros(n_points) + 0.5)

    best_c5 = np.inf
    best_h = np.zeros(n_points)
    start_time = time.time()

    for h_start in start_points:
        h = h_start.copy()
        h = h * (integral_target / np.sum(h))
        h = np.clip(h, 0, 1)

        def compute_overlap(h_):
            h_full = np.concatenate([h_, h_[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def objective(x):
            return compute_overlap(x)

        def constraint(x):
            return np.sum(x) - integral_target

        # Use SLSQP with tighter tolerances
        result = scipy.optimize.minimize(
            objective,
            h[:n_points // 2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-10,
                'maxiter': 50,
                'disp': False,
                'eps': 1e-8
            },
            jac=lambda x: np.zeros_like(x)  # Gradient approximated via finite differences
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        if time.time() - start_time > budget_s - 5:
            break

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)

    return best_h_full, final_c5, n_points
