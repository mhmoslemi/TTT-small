import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    pop_size = 10
    max_gen = 500
    best_c5 = np.inf
    best_h = np.zeros(n_points)

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        population = [h_initial]
    else:
        population = []

    # Generate initial population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (integral_target / np.sum(h))
        population.append(h)
    population = [np.clip(p, 0, 1) for p in population]

    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = []
        for h in population:
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        # Select top half
        idx = np.argsort(fitness)
        selected = [population[i] for i in idx[:pop_size//2]]
        # Generate new population
        new_pop = []
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = (p1 + p2) / 2
            mutate = np.random.rand(n_points) < 0.1
            child[mutate] += np.random.normal(0, 0.05, size=np.sum(mutate))
            child = np.clip(child, 0, 1)
            child = child * (integral_target / np.sum(child))
            new_pop.append(child)
        population = new_pop
        # Update best
        for h in population:
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5, n_points
