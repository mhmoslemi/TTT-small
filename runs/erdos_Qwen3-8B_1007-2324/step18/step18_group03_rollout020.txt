import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Use parent solution if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_start = h_initial
    else:
        h_start = np.random.rand(n_points)
        h_start = h_start * (integral_target / np.sum(h_start))
    
    # Normalize starting point
    h_start = h_start * (integral_target / np.sum(h_start))
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = scipy.optimize.minimize(
        compute_overlap,
        h_start,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={
            'ftol': 1e-10,
            'maxiter': 300,
            'disp': False,
            'eps': 1e-8
        }
    )
    
    # Perturb initial solution and optimize
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturbation = 0.05 * np.random.randn(n_points)
        h_perturbed = np.clip(h_start + perturbation, 0, 1)
        h_perturbed = h_perturbed * (integral_target / np.sum(h_perturbed))
        
        result_perturb = scipy.optimize.minimize(
            compute_overlap,
            h_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={
                'ftol': 1e-10,
                'maxiter': 150,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result_perturb.success and compute_overlap(result_perturb.x) < compute_overlap(result.x):
            result = result_perturb
    
    if time.time() - start_time > budget_s:
        best_h = h_start
        best_c5 = compute_overlap(h_start)
    else:
        best_h = result.x
        best_c5 = compute_overlap(best_h)
    
    return best_h, best_c5, n_points
