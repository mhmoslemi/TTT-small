import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():

            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if initial_h is not None:
        def normalize(h):
            s = np.sum(h)
            if s < 1e-6:
                return h * 0.0
            return h * (integral_target / s)
        start_points.append(normalize(initial_h))
        for i in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def structured_initial():
        half = n_points // 2
        seg = half // 4
        res = np.zeros(half)
        res[:seg] = 1.0
        res[seg:seg*2] = 0.5
        res[seg*2:seg*3] = 1.0
        res[seg*3:] = 0.5
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def ternary_initial():
        half = n_points // 2
        seg = half // 3
        res = np.zeros(half)

        res[seg:seg*2] = 0.5
        res[seg*2:] = 1.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([structured_initial(), ternary_initial()])

    def random_initial():
        half = n_points // 2
        res = np.random.rand(half) * (integral_target / np.sum(np.random.rand(half)))
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.append(random_initial())

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
