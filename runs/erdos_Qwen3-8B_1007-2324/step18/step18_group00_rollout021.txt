    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    start_points = []

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        start_points.append(h_initial)

    # Generate more varied starting points from parent
    if 'initial_h_values' in globals():
        # Add noise to parent
        noise = np.random.normal(0, 0.05, len(h_initial))
        noisy_start = h_initial + noise
        noisy_start = np.clip(noisy_start, 0, 1)
        noisy_start = noisy_start * (integral_target / 2) / np.sum(noisy_start)
        start_points.append(noisy_start)

        # Shift parent
        shift = np.random.randint(1, len(h_initial))
        shifted_start = np.roll(h_initial, shift)
        shifted_start = shifted_start * (integral_target / 2) / np.sum(shifted_start)
        start_points.append(shifted_start)

        # Windowed inversion
        window_size = np.random.randint(1, len(h_initial) // 2)
        window_start = np.random.randint(0, len(h_initial) - window_size)
        windowed_start = h_initial.copy()
        windowed_start[window_start:window_start+window_size] = windowed_start[window_start:window_start+window_size][::-1]
        windowed_start = windowed_start * (integral_target / 2) / np.sum(windowed_start)
        start_points.append(windowed_start)

    # Binary sparse
    binary_initial = np.zeros(len(h_initial))
    num_ones = len(h_initial) // 2
    indices = np.random.choice(len(h_initial), num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)

    # Uniform random
    random_initial = np.random.rand(len(h_initial))
    random_initial = random_initial * (integral_target / 2) / np.sum(random_initial)
    start_points.append(random_initial)

    # Rotated
    rotated_start = np.roll(h_initial, np.random.randint(1, len(h_initial)))
    rotated_start = rotated_start * (integral_target / 2) / np.sum(rotated_start)
    start_points.append(rotated_start)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        # Normalize to meet the integral constraint
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)
        else:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        # Optimize
        result = scipy.optimize.minimize(

            method='SLSQP',
