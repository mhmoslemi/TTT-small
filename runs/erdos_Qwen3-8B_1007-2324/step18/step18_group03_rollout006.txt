import numpy as np

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():

            h_initial = h_initial[:n_points]
        start_points.append(h_initial)

    def generate_initial():
        def binary():
            b = np.zeros(n_points)
            ones = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
            b[ones] = 1.0
            b = b * (integral_target / np.sum(b))
            return b

        def random():
            r = np.random.rand(n_points)
            r = r * (integral_target / np.sum(r))
            return r

        def sine():
            x = np.linspace(0, 2, n_points)
            s = 0.5 + 0.5 * np.sin(2 * np.pi * 0.4 * x)
            s = np.clip(s, 0.0, 1.0)
            s = s * (integral_target / np.sum(s))
            return s

        def block():
            b = np.zeros(n_points)
            b[:n_points//3] = 1.0
            b[n_points//3:2*n_points//3] = 0.5
            b[2*n_points//3:] = 0.25
            b = b * (integral_target / np.sum(b))
            return b

        def tri():
            x = np.linspace(0, 2, n_points)
            t = np.clip(1 - np.abs(x - 1) / 1, 0, 1)
            t = t * (integral_target / np.sum(t))
            return t

        def combo():
            c = np.zeros(n_points)
            intervals = np.random.choice(n_points, size=4, replace=False)
            for i in intervals:
                c[i] = 1.0
            c = c * (integral_target / np.sum(c))
            return c

        yield binary()
        yield random()
        yield sine()
        yield block()
        yield tri()
        yield combo()

    for p in generate_initial():
        start_points.append(p)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for h_start in start_points:

        def compute_overlap(h):
            h = np.clip(h, 0, 1)
            h = h * (integral_target / np.sum(h))
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
