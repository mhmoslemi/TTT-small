            lambda x: compute_overlap(x),
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
            options={'ftol': 1e-12, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = res.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    return best_h, best_c5, n_points
