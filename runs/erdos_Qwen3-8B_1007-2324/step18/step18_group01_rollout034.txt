        integral = np.sum(h) * dx
        h = h * (1 / integral) * (n_points / 2)
        return h
    
    start_points.append(create_symmetric_cosine())
    start_points.append(create_checkerboard())
    start_points.append(np.random.rand(n_points) * 2 - 1)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def objective(x):
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - integral_target / 2
        
        result = scipy.optimize.minimize(
            objective,
            h_start[:n_points // 2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-10,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    final_c5 = compute_overlap(best_h)
    return best_h_full, final_c5, n_points
