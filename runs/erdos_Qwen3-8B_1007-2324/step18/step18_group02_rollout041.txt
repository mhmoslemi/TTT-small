        res = res * integral_target / np.sum(res)
        return res

    def random_initial():
        res = np.random.rand(n_points)
        res = res * integral_target / np.sum(res)
        return res

    def structured_initial():
        seg = n_points // 4
        res = np.zeros(n_points)
        res[:seg] = 1.0
        res[seg:seg*2] = 0.0
        res[seg*2:seg*3] = 1.0
        res[seg*3:] = 0.0
        res = res * integral_target / np.sum(res)
        return res

    start_points.extend([binary_initial(), random_initial(), structured_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
            options={'ftol': 1e-9, 'maxiter': 100, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    return best_h, best_c5, n_points
