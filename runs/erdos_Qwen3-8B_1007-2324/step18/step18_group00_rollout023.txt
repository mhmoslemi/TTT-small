import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points

    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]

    if 'initial_h_values' in globals():

        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        h_initial = initial_h[:n_points // 2]
        result = scipy.optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=50,
            maxiter=100,

            objective,
            bounds,
            strategy='best1bin',

            maxiter=100,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            tol=1e-8,

        )

    h_best = result.x
