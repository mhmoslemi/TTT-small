import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if initial_h is not None:
        def normalize(h):
            s = np.sum(h)
            if s < 1e-6:
                return h * 0.0
            return h * (integral_target / s)
        start_points.append(normalize(initial_h))
        for _ in range(2):
            noise = np.random.normal(0, 0.01, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def binary_initial():
        h = np.zeros(n_points//2)
        num_ones = (n_points//2)//2
        idxs = np.random.choice(n_points//2, num_ones, replace=False)
        h[idxs] = 1.0
        return normalize(np.concatenate([h, h[::-1]]))

    def random_initial():
        h = np.random.rand(n_points//2) * (integral_target / np.sum(np.random.rand(n_points//2)))
        return normalize(np.concatenate([h, h[::-1]]))

    def structured_initial():
        h = np.zeros(n_points//2)
        seg = (n_points//2) // 4
        h[:seg] = 1.0
        h[seg:seg*2] = 0.0
        h[seg*2:seg*3] = 1.0
        h[seg*3:] = 0.0
        return normalize(np.concatenate([h, h[::-1]]))

    def periodic_initial():
        h = np.zeros(n_points//2)
        period = (n_points//2) // 2
        h[:period] = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(period) / period)
        return normalize(np.concatenate([h, h[::-1]]))

    start_points.extend([binary_initial(), random_initial(), structured_initial(), periodic_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points//2)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points//2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points//2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2}],
            options={'ftol': 1e-10, 'maxiter': 300, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
