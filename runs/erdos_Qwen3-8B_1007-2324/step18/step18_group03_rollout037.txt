import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points
    target = n_points / 2

    def create_pulse_function(params):
        h_vals = np.zeros(n_points)
        x_grid = np.linspace(0, 2, n_points)
        for i in range(4):
            c = params[3*i]
            w = params[3*i + 1]
            a = params[3*i + 2]
            w = max(0.01, min(w, 2.0))
            left = c - w / 2
            right = c + w / 2
            left_clipped = max(0.0, left)
            right_clipped = min(2.0, right)
            mask = (x_grid >= left_clipped) & (x_grid <= right_clipped)
            h_vals[mask] += a
        h_vals = np.clip(h_vals, 0.0, 1.0)
        integral = np.sum(h_vals * dx)
        scale = target / integral if integral != 0 else 1.0
        h_vals *= scale
        return h_vals

    def compute_overlap(params):
        h = create_pulse_function(params)
        correl = np.correlate(h, 1 - h, mode='full')
        return np.max(correl) * dx

    initial_params_list = []
    for _ in range(5):
        params = np.random.rand(12)
        updated_params = np.copy(params)
        for i in range(12):
            if i % 3 == 0:
                updated_params[i] = np.clip(updated_params[i], 0.0, 2.0)
            elif i % 3 == 1:
                updated_params[i] = np.clip(updated_params[i], 0.01, 2.0)
            else:
                updated_params[i] = np.clip(updated_params[i], 0.0, 1.0)
        initial_params_list.append(updated_params)

    best_c5 = np.inf

    for params_initial in initial_params_list:
        constraints = []
        bounds = []
        for i in range(12):

                bounds.append((0.01, 2.0))

                bounds.append((0.0, 1.0))

        result = scipy.optimize.minimize(
