import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = np.clip(h_initial[:n_points], 0, 1)
        start_points.append(h_initial)
    else:
        start_points.append(np.random.rand(n_points))
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - integral_target

    best_c5 = np.inf
    best_h = np.zeros(n_points)
    best_h_full = np.zeros(2 * n_points)

    for h_start in start_points:
        if np.sum(h_start) == 0:
            h_start = np.random.rand(n_points) / 2
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'ftol': 1e-12, 'maxiter': 100, 'disp': False}
        )
        
        if result.success:
            current_val = compute_overlap(result.x)
            if current_val < best_c5:
                best_c5 = current_val
                best_h = result.x
                best_h_full = np.concatenate([best_h, best_h[::-1]])

    return best_h_full, best_c5, n_points
