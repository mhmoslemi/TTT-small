def construct_diffused_parent(parent_h, diffusivity=0.1):
    """Simulate a diffusion process to explore function space"""
    n = len(parent_h)
    dtype = parent_h.dtype
    # Convert to continuous signal for diffusion processing
    parent_signal = np.interp(np.linspace(0,2,n), np.linspace(0,2,n), parent_h)
    
    # Step 1: Translate parent to origin
    translated = parent_signal - np.mean(parent_signal)
    
    # Step 2: Add Gaussian diffusion noise
    noise = np.random.normal(size=n) * diffusivity * np.std(translated)
    noisy_signal = translated + noise
    
    # Step 3: Transform and stabilize
    shifted_signal = (noisy_signal - np.min(noisy_signal)) / (np.max(noisy_signal)-np.min(noisy_signal))
