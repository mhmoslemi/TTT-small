        def run(seed=42, budget_s=100, **kwargs):
            import time
            import numpy as np
            from scipy.optimize import differential_evolution, minimize

            start_time = time.time()
            n_points_val = 60
            dx = 2.0 / n_points_val
            integral_target = n_points_val // 2
            m = n_points_val // 2

            # Define compute_overlap for both functions
            def compute_overlap(h):
                h_val = np.clip(h, 0, 1)
                h_full = np.concatenate([h_val, h_val[::-1]])
                corr = np.correlate(h_full, 1-h_full, mode='full')
                return np.max(corr) * dx

            # With penalty for DE
            penalty_factor = 50   # Tuning

            def objective_de(h):
                h_val = np.clip(h, 0, 1)
                h_full = np.concatenate([h_val, h_val[::-1]])
                corr = np.correlate(h_full, 1-h_full, mode='full')
                max_corr_val = np.max(corr)
                c5_val = max_corr_val * dx
                constraint_error = np.abs(np.sum(h) - integral_target)
                return c5_val + penalty_factor * constraint_error**2

            bounds = [(0.0, 1.0) for _ in range(m)]
            res_de = differential_evolution(objective_de, bounds, strategy='best1', maxiter=5, popsize=10, tol=1e-6, seed=seed, mutation=(0.5, 1.0), recombination='blend')

            def objective_slsqp(h):
                # Pure objective, no penalty
                return compute_overlap(h)

            # Constraints for SLSQP: sum(h)=integral_target

            # We also bound the h in [0,1] for SLSQP
