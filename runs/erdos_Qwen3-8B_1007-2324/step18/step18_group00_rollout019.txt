import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 60
    half_length = n_points // 2
    integral_target = n_points / 2  # 30
    dx = 2.0 / n_points  # 2/60

    # Initialize population
    population_size = 20
    population = []

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(initial_parent) > half_length:
            initial_parent = initial_parent[:half_length]
        initial_parent = (initial_parent * integral_target) / np.sum(initial_parent)
        population.append(initial_parent)
    else:
        # Create random initial parents
        for _ in range(population_size):
            h = np.random.rand(half_length)
            h = h * (integral_target / np.sum(h))
            population.append(h)

    best_c5 = np.inf
    best_h = np.zeros(half_length)

    # Main loop
    max_generations = 200
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate all individuals
        current_fitness = []
        for h in population:
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            current_c5 = np.max(corr) * dx
            current_fitness.append(current_c5)
        
        # Select top parents
        sorted_indices = np.argsort(current_fitness)
        top_parents = [population[i] for i in sorted_indices[:int(population_size / 2)]]

        # Generate new candidates
        new_population = []
        for _ in range(population_size):
            if np.random.rand() < 0.7:
                # Crossover
                p1, p2 = np.random.choice(top_parents, 2, replace=False)
                child = (p1 + p2) / 2

                parent = np.random.choice(top_parents)
                child = parent.copy()

                indices = np.random.choice(half_length, size=5, replace=False)
                noise = np.random.normal(0, 0.05, size=half_length)

                best_h = h.copy()

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
