import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    if 'initial_h_values' in globals():

        if len(h_initial) != n_points:

    best_h = np.zeros(n_points)
