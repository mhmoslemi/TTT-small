            h_full = np.concatenate([h_start, h_start[::-1]])
        else:
            optimal_len = n_points // 2
            h_full = np.concatenate([h_start, [h_start[-1]], h_start[::-1]])
        
        h_full = h_full * integral_target / np.sum(h_full)
        h_full = np.clip(h_full, 0, 1)
        h_full = h_full * integral_target / np.sum(h_full)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def objective(x):
            x = x * integral_target / np.sum(x)
            x = np.clip(x, 0, 1)
            x = x * integral_target / np.sum(x)
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - integral_target / 2
        
        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(len(h_start))],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-12,
                'maxiter': 100,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if n_points % 2 == 0:
        best_h_full = np.concatenate([best_h, best_h[::-1]])
    else:
        best_h_full = np.concatenate([best_h, [best_h[-1]], best_h[::-1]])
    
    final_c5 = compute_overlap(best_h)
    
    return best_h_full, final_c5, n_points
