            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            compute_overlap,
            h_normalized,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_half)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 0.01
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
