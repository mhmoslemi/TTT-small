import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():

            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s < 1e-6:
            return h * 0.0
        return h * (integral_target / s)

    start_points = []

    if initial_h is not None:
        start_points.append(normalize(initial_h))
        for _ in range(3):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def binary_initial():
        half = n_points // 2
        res = np.zeros(half)
        num_ones = half // 2
        res[np.random.choice(half, num_ones, replace=False)] = 1
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def random_initial():
        half = n_points // 2
        res = np.random.rand(half) * (integral_target / np.sum(np.random.rand(half)))
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def structured_initial():
        half = n_points // 2
        seg = half // 4
        res = np.zeros(half)
        res[:seg] = 1.0
        res[seg:seg*2] = 0.0
        res[seg*2:seg*3] = 1.0
        res[seg*3:] = 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def zigzag_initial():
        half = n_points // 2
        res = np.zeros(half)
        for i in range(half):
            res[i] = 1.0 if (i % 3 == 0) else 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([binary_initial(), random_initial(), structured_initial(), zigzag_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points//2],

            bounds=[(0.0, 1.0) for _ in range(n_points//2)],
