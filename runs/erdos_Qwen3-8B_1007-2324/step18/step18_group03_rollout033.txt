            yield sine
            block = np.zeros(n_points)
            block[:n_points//4] = 1.0
            block[n_points//4:2*n_points//4] = 0.5
            block[2*n_points//4:3*n_points//4] = 0.25
            block[3*n_points//4:] = 0.125
            yield block * (integral_target / np.sum(block))
            interleaved = np.array([1.0 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0.0 for i in range(n_points)])
            yield interleaved * (integral_target / np.sum(interleaved))

        for p in generate_initial():
            start_points.append(p)

        for h_start in start_points:
            h_start = h_start * (integral_target / np.sum(h_start))

            def compute_overlap(h):
                h_clipped = np.clip(h, 0.0, 1.0)
                h_norm = h_clipped * (integral_target / np.sum(h_clipped))
                corr = np.correlate(h_norm, 1 - h_norm, mode='full')
                return np.max(corr) * dx

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = scipy.optimize.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-12, 'maxiter': 500, 'disp': False, 'eps': 1e-8}
            )

            if result.success:
                current_c5 = compute_overlap(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x

    return best_h, best_c5, len(best_h)
