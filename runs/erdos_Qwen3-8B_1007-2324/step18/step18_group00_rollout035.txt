```python
import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target = n_points / 2  # 30

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    h_initial = initial_h[:n_points//2] if initial_h is not None else np.random.rand(n_points//2)

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        penalty = 1e3 * abs(sum_h - target)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlap = np.max(corr) * dx
        return overlap + penalty

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        disp=False,
        polish=True,
        vectorized=True
    )

    h_best = result.x
    h_best = np.clip(h_best, 0, 1)
    sum_h = np.sum(h_best)
    if abs(sum_h - target) > 1e-6:
        h_best = h_best * (target / sum_h)

    h_full = np.concatenate([h_best, h_best[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5 = np.max(corr) * dx

    return h_full, c5, n_points
```
