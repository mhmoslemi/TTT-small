    h_vals_final = np.clip(h_vals_final, 0, 1)
    total = np.sum(h_vals_final * dx)
    scale = target_sum / total if total != 0 else 1.0
    h_vals_final *= scale
    
    corr = np.correlate(h_vals_final, 1 - h_vals_final, mode='full')
    c5_bound = np.max(corr) * dx
    
    return h_vals_final, c5_bound, n_points
