import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if initial_h is not None:
        def normalize(h):
            s = np.sum(h)
            return h * (integral_target / s) if s > 1e-6 else h * 0.0
        def perturb(h, std=0.05):
            return normalize(h + np.random.normal(0, std, n_points))
        start_points.append(normalize(initial_h))
        for _ in range(4):
            start_points.append(perturb(initial_h))
        start_points.append(normalize(np.random.rand(n_points)))

    def structured_initial():
        half = n_points // 2
        seg = half // 4
        res = np.zeros(half)
        for i in range(1, 5):
            res[seg*(i-1):seg*i] = 1.0 if i % 2 else 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def checker_initial():
        half = n_points // 2
        res = np.zeros(half)
        for i in range(half // 4):
            res[i*2] = 1.0
            res[i*2 + 1] = 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([structured_initial(), checker_initial()])
    if initial_h is not None:
        start_points.append(normalize(np.random.rand(n_points)))

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
            options={'ftol': 1e-12, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    return best_h, best_c5, n_points
