        else:

        for _ in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            s_perturbed = np.sum(perturbed)
            if s_perturbed < 1e-6:
                start_points.append(np.zeros(n_points))
            else:
                start_points.append(perturbed * (integral_target / s_perturbed))

    def random_initial():
        h = np.random.rand(n_points)
        s = np.sum(h)
        if s < 1e-6:
            return np.zeros(n_points)
        return h * (integral_target / s)

    def binary_initial():
        h = np.zeros(n_points)
        num_ones = int(n_points / 2)
        indices = np.random.choice(n_points, num_ones, replace=False)
        h[indices] = 1.0
        s = np.sum(h)
        if s < 1e-6:
            return np.zeros(n_points)
        return h * (integral_target / s)

    def structured_initial():
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        s = np.sum(h)
        if s < 1e-6:
            return np.zeros(n_points)
        return h * (integral_target / s)

    start_points.extend([binary_initial(), random_initial(), structured_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(x),
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
            options={'ftol': 1e-12, 'maxiter': 150, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    return best_h, best_c5, n_points
