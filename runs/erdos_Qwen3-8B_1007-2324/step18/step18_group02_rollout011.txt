import numpy as np

def run(seed=42, budget_s=100, **kwargs):

            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if initial_h is not None:
        def normalize(h):
            s = np.sum(h)
            if s < 1e-6:
                return h * 0.0
            return h * (integral_target / s)
        start_points.append(normalize(initial_h))
        for _ in range(3):
            noise = np.random.normal(0, 0.03, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def binary_initial():
        half = n_points // 2
        res = np.zeros(half)
        num_ones = half // 2 + np.random.randint(2)
        res[np.random.choice(half, num_ones, replace=False)] = 1
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def random_initial():
        half = n_points // 2
        res = np.random.rand(half) * (integral_target / np.sum(np.random.rand(half)))
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def structured_initial():
        half = n_points // 2
        seg = half // 3
        res = np.zeros(half)
        res[:seg] = 1.0
        res[seg:seg*2] = 0.5
        res[seg*2:] = 1.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def balanced_initial():
        half = n_points // 2
        res = np.zeros(half)
        res[::2] = 1.0
        res[1::2] = 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([binary_initial(), random_initial(), structured_initial(), balanced_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points // 2],

            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2}],
