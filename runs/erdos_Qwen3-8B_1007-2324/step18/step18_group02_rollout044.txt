                return h * 0.0
            return h * (integral_target / s)
        start_points.append(normalize(initial_h))
        for _ in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def binary_initial():
        half = n_points // 2
        res = np.zeros(half)
        num_ones = half // 2
        res[np.random.choice(half, num_ones, replace=False)] = 1
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def random_initial():
        half = n_points // 2
        res = np.random.rand(half) * (integral_target / np.sum(np.random.rand(half)))
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def structured_initial_1():
        half = n_points // 2
        seg = half // 4
        res = np.zeros(half)
        res[:seg] = 1.0
        res[seg:seg*2] = 0.0
        res[seg*2:seg*3] = 1.0
        res[seg*3:] = 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def structured_initial_2():
        half = n_points // 2
        seg = half // 4
        res = np.zeros(half)
        res[:seg] = 1.0
        res[seg*2:seg*3] = 1.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([binary_initial(), random_initial(), structured_initial_1(), structured_initial_2()])

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points // 2],
            method='COBYLA',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2}],
            options={'maxiter': 200, 'disp': False, 'ftol': 1e-12}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
