def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    h = np.zeros(n_points)

        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[-n_points:]

        x = x * (integral_target / np.sum(x))
