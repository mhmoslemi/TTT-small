import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 40
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Load initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = np.pad(h_initial[:n_points], (0, n_points - len(h_initial[:n_points])))
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (integral_target / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0, 1)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    def generate_h(theta, dim):
        theta = np.clip(theta, 0.0, 1.0)
        if dim == 0:
            return np.ones(n_points) * (integral_target / n_points)
        act = 1.0 / (1.0 + np.exp(-10.0 * (theta - 0.5)))
        h = np.zeros(n_points)
        h[:dim] = act[:dim]
        h[1:n_points-1] = act[1:dim]
        h[-dim:] = act[-dim:]
        h = np.clip(h, 0, 1)
        h = h * (integral_target / np.sum(h))
        return h
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(theta):
        h = generate_h(theta, 15)
        return compute_overlap(h)
    
    def constraint(theta):
        h = generate_h(theta, 15)
        return np.sum(h) - integral_target
    
    def perturb(theta, noise_scale):
        return theta + np.random.normal(0, noise_scale, theta.shape)
    
    theta_initial = np.random.rand(15)
    theta_initial = theta_initial * (integral_target / 15)
    theta_initial = np.clip(theta_initial, 0, 1)
    
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        theta = perturb(theta_initial, 0.05)
        result = scipy.optimize.minimize(
            objective,
            theta,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(15)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-8,
                'maxiter': 50,
                'disp': False
            }
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = generate_h(result.x, 15)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (integral_target / np.sum(best_h))
    final_c5 = compute_overlap(best_h)
    
    return best_h, final_c5, n_points
