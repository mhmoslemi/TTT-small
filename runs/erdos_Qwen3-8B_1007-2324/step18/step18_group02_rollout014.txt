import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def normalize(h, target):
        s = np.sum(h)
        if s < 1e-6:
            return h * 0.0
        return h * (target / s)

    start_points = []

    if initial_h is not None:
        start_points.append(normalize(initial_h.copy(), integral_target))
        for _ in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed, integral_target))

    def binary_init():
        half = n_points // 2
        res = np.zeros(half)
        num_ones = int(np.ceil(half * 0.3))
        res[np.random.choice(half, num_ones, replace=False)] = 1
        return normalize(np.concatenate([res, res[::-1]])[:n_points], integral_target)

    def random_init():
        half = n_points // 2
        res = np.random.rand(half) * (integral_target / np.sum(np.random.rand(half)))
        return normalize(np.concatenate([res, res[::-1]])[:n_points], integral_target)

    def spaced_init():
        half = n_points // 2
        res = np.zeros(half)
        for i in range(0, half, 5):
            res[i:i+2] = 1.0
        return normalize(res, integral_target)

    start_points.extend([binary_init(), random_init(), spaced_init()])

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points // 2],
            method='SLSQP',
            bounds=[(0., 1.) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2}],

        if res.success:
