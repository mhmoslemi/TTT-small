import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    # Load initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points = [h_initial]
    else:
        start_points = []

    # Add structured initializations
    def create_initial():
        base = np.random.rand(n_points // 2)
        base /= np.sum(base)
        base *= integral_target / 2
        return np.concatenate([base, base[::-1]])[:n_points]

    start_points.append(create_initial())

    # Symmetric step function with 3 intervals
    def step_initial():
        intervals = 3
        vals = np.zeros(n_points // 2)
        vals[0:intervals] = 1.0
        vals[intervals:2*intervals] = 0.5
        vals[2*intervals:] = 0.0
        vals = vals * (integral_target / np.sum(vals))
        return np.concatenate([vals, vals[::-1]])[:n_points]
    start_points.append(step_initial())

    # Uniform guess
    def uniform_initial():
        return np.full(n_points, 0.5) * (integral_target / np.sum(np.full(n_points, 0.5)))

    start_points.append(uniform_initial())

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def objective(x):
            h = np.zeros(n_points)
            h[:len(x)] = x
            h = np.concatenate([h, h[::-1]])
            return compute_overlap(h)

        def constraint(x):
            return np.sum(x) - integral_target / 2

        result = scipy.optimize.minimize(
            objective,
            h_start[:n_points // 2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': constraint}],
