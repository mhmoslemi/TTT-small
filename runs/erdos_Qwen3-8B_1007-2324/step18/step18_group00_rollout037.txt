import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    # Use initial_h_values if available

        if len(initial_h) != n_points:

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_overlap = np.max(corr) * dx
        # Penalty for integral constraint
        penalty = 1000 * abs(np.sum(h_clipped) - target_sum)

    # Bounds
    bounds = [(0.0, 1.0) for _ in range(n_points // 2)]

    popsize = min(100, int(budget_s * 10))  # Adjusted for time

        objective,
        bounds,
        strategy='best1bin',
        popsize=popsize,
        maxiter=maxiter,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        disp=False,
        vectorized=True
    )

    # Extract best solution
    best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
