import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    start_points = []

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial[:n_points // 2])

    # Perturbed parent variations
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)[:n_points//2]
        perturbed = h_initial + np.random.normal(0, 0.04, h_initial.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral_target / 2) / np.sum(perturbed)
        start_points.append(perturbed)

        rotated = np.roll(h_initial, np.random.randint(1, len(h_initial)))
        rotated = rotated * (integral_target / 2) / np.sum(rotated)
        start_points.append(rotated)

        split_half = len(h_initial) // 2
        split = np.concatenate([h_initial[:split_half], 1 - h_initial[split_half:]])
        split = split * (integral_target / 2) / np.sum(split)
        start_points.append(split)

    # Binary and random initializations
    binary_initial = np.zeros(n_points // 2)
    num_ones = (n_points // 2) // 2
    indices = np.random.choice(n_points // 2, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)

    random_initial = np.random.rand(n_points // 2)
    random_initial = random_initial * (integral_target / 2) / np.sum(random_initial)
    start_points.append(random_initial)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        # Normalize to meet the integral constraint
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)
        else:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        # Optimize
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
