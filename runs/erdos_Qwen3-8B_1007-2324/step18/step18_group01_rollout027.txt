import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target_total = n_points / 2
    integral_target_half = integral_target_total / 2
    
    initial_guesses = []
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        sym_h_first_half = h_initial[:n_points//2]
        sym_h_second_half = [1 - x for x in reversed(sym_h_first_half)]
        sym_h = np.concatenate([sym_h_first_half, sym_h_second_half])
        initial_guesses.append(sym_h)
    
    def create_random_symmetric_guess():
        n_half = n_points // 2
        first_half = np.random.rand(n_half)
        first_half /= np.sum(first_half)
        first_half *= integral_target_half
        second_half = [1 - x for x in reversed(first_half)]
        return np.concatenate([first_half, second_half])
    
    initial_guesses.append(create_random_symmetric_guess())
    initial_guesses.append(np.full(n_points, 0.5))
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_guess in initial_guesses:
        h_first = h_guess[:n_points//2]
        h_first = np.clip(h_first, 0, 1)
        
        def compute_overlap(x):

            corr = np.correlate(full_h, 1 - full_h, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n_points//2)]
        
        result = scipy.optimize.minimize(
            objective,
            h_first,
            method='SLSQP',
            bounds=bounds,
            options={'ftol': 1e-8, 'maxiter': 100, 'disp': False}
        )
