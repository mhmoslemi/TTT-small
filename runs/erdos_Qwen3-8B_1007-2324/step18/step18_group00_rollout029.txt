def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    def normalize(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) == 0:
            return np.random.rand(len(h)) * (integral_target / len(h))
        scale = (integral_target) / np.sum(h)
        return h * scale

    def fitness(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():

        initial_h = normalize(initial_h)
    else:
        initial_h = np.random.rand(n_points//2)
        initial_h = normalize(initial_h)

    bounds = [(0.0, 1.0) for _ in range(n_points//2)]
    result = differential_evolution(
        fitness,

        strategy='best1bin',

        popsize=12,
