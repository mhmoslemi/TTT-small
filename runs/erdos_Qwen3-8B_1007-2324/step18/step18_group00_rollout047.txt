    # Generate smoothed version (moving average)

        window_size = 5
        smoothed = np.convolve(h, np.ones(window_size)/window_size, mode='same')
        return smoothed

        h_smooth = apply_smoothing(h_initial)
        start_points.append(h_smooth)
