import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Increased for better resolution
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Load initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points = [h_initial]
    else:
        start_points = []
    
    # Generate structured initial guesses
    def create_initial():
        # Symmetric initial guess
        half = n_points // 2
        base = np.zeros(half)
        base = np.concatenate([base, np.ones(half)])
        return base[:n_points]
    
    start_points.append(create_initial())
    start_points.append(np.random.rand(n_points) * 2 - 1)
    start_points.append(np.zeros(n_points) + 0.5)
    start_points.append(np.random.rand(n_points))
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def objective(x):
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - integral_target / 2
        
        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-12,
                'maxiter': 50,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Binary rounding step
    def to_binary(h):
        b = np.round(h)
        diff = integral_target - np.sum(b)
        if diff != 0:
            idx = np.where(b == 0)[0]
            if diff > 0:
                b[idx[:diff]] = 1
            else:
                b[idx[-diff:]] = 0
        return b
    
    binary_h = to_binary(best_h)
    binary_c5 = compute_overlap(binary_h)
    
    if binary_c5 < best_c5:
        best_c5 = binary_c5
        best_h = binary_h
    
    return best_h, best_c5, n_points
