import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    # Load initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) > n_points:
            initial_h = initial_h[:n_points]
        else:
            initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), 'constant')
    else:
        initial_h = np.zeros(n_points)

    # Generate diverse initial population
    start_points = [initial_h]
    start_points.append(np.random.rand(n_points) * 0.5)
    start_points.append(np.zeros(n_points))
    start_points.append(np.ones(n_points) * 0.5)

    best_c5 = np.inf
    best_h = initial_h.copy()

    start_time = time.time()
    total_evals = 0

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Optimize each start point
    for h_start in start_points:
        # Normalize to meet integral constraint
        h_start = h_start * (integral_target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start * 0
        h_start = (h_start * integral_target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start

        # Add small perturbation to avoid flat regions
        h_start += np.random.normal(0, 0.05, size=n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (integral_target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start

        # Optimization
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
            options={
                'ftol': 1e-12,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        total_evals += 1
        if time.time() - start_time > budget_s:
            break

    # Apply final optimization to best candidate
    result = scipy.optimize.minimize(
        compute_overlap,
        best_h,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target}],
