import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def normalize(h):
        s = np.sum(h)
        if s < 1e-6:
            return np.zeros_like(h)
        return h * (integral_target / s)

    def compute_overlap(h):
        h_norm = normalize(h)
        h_full = np.concatenate([h_norm, h_norm[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def structured_initial1():
        half = n_points // 2
        seg = half // 5
        res = np.zeros(half)
        res[::seg*2] = 1.0
        res[seg*2::seg*2] = 0.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def structured_initial2():
        half = n_points // 2
        res = np.zeros(half)
        res[::(half//4)] = 1.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def patterned_initial():
        half = n_points // 2
        res = np.linspace(0, 1, half)
        res = np.tile(res, 2)[:n_points]
        return normalize(res)

    start_points = [structured_initial1(), structured_initial2(), patterned_initial()]

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    start_time = time.time()
    for h_start in start_points:
        result = scipy.optimize.differential_evolution(
            fun=compute_overlap,
            bounds=bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            disp=False
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
