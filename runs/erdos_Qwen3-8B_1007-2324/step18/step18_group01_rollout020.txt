import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2.0

        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    start_points = []
    
    if 'initial_h_values' in globals():

        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)

    start_points.append(np.random.rand(n_points))
    start_points.append(np.full(n_points, 1.0 / n_points))
    if 'initial_h_values' in globals():
        perturb = np.random.normal(0, 0.05, n_points)
        start_points.append((initial_h_values[:n_points] + perturb) * 0.5)
        start_points.append(np.flip(initial_h_values[:n_points]))
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        def objective(x):
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - integral_target
        
        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-8,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
