import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 4  # Adjusted to correct integral constraint
    
    # Load initial guess
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points = [h_initial]
    else:
        start_points = []
    
    # Generate symmetric initial guess
    def create_symmetric():
        base = np.random.rand(n_points // 2)
        base /= np.sum(base)
        base *= integral_target
        sym = np.concatenate([base, base[::-1]])
        return sym[:n_points]  # Ensure length is n_points
    
    start_points.append(create_symmetric())
    
    # Add a few more starting points
    start_points.append(np.random.rand(n_points // 2) * 2 - 1)
    start_points.append(np.zeros(n_points // 2) + 0.5)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        # Ensure the starting point is for the first half
        h_start = h_start[:n_points//2]
        h_start = h_start * (integral_target / np.sum(h_start))
        h_start = np.clip(h_start, 0, 1)
        
        # Define the objective and constraint
        def compute_overlap(h_half):
            h_full = np.concatenate([h_half, h_half[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        def objective(x):
            return compute_overlap(x)
        
        def constraint(x):
            return np.sum(x) - integral_target
        
        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points//2)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={
                'ftol': 1e-12,
                'maxiter': 250,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.concatenate([result.x, result.x[::-1]])
    
    # Final evaluation
    final_c5 = compute_overlap(best_h[:n_points//2])
    
    return best_h[:n_points], final_c5, n_points
