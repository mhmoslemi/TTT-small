import numpy as np

import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    start_points = []
    
    # Use parent solution if available
    if 'initial_h_values' in globals():

        start_points.append(h_initial)
    
    # Generate diverse initial points
    def generate_initial():
        # Binary pattern with varied distribution
        binary = np.zeros(n_points)
        ones = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
        binary[ones] = 1.0
        binary = binary * (integral_target / np.sum(binary))
        yield binary
        
        # Random pattern
        random = np.random.rand(n_points)
        random = random * (integral_target / np.sum(random))
        yield random
        
        # Sine-based pattern
        x = np.linspace(0, 2, n_points)
        sine = 0.5 + 0.5 * np.sin(2 * np.pi * 0.5 * x)
        sine = np.clip(sine, 0.0, 1.0)
        sine = sine * (integral_target / np.sum(sine))
        yield sine
        
        # Triangular pattern
        x = np.linspace(0, 2, n_points)
        triangular = 0.5 + 0.5 * np.abs(x - 1.0)
        triangular = np.clip(triangular, 0.0, 1.0)
        triangular = triangular * (integral_target / np.sum(triangular))
        yield triangular
        
        # Block pattern
        block = np.zeros(n_points)
        block[:n_points//3] = 1.0
        block[n_points//3:2*n_points//3] = 0.5
        block[2*n_points//3:] = 0.25
        block = block * (integral_target / np.sum(block))
        yield block
    
    for p in generate_initial():
        start_points.append(p)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    for h_start in start_points:
        # Normalize starting point
        h_start = h_start * (integral_target / np.sum(h_start))
        
        # Define compute_overlap
        def compute_overlap(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # SLSQP optimization
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
