        perturbed = perturbed * (integral_target / 2) / np.sum(perturbed)
        start_points.append(perturbed)

        rotated = h_initial_scaled.copy()
        shift = np.random.randint(1, half_length)
        rotated = np.roll(rotated, shift)
        rotated = rotated * (integral_target / 2) / np.sum(rotated)
        start_points.append(rotated)

    binary_initial = np.zeros(half_length)
    num_ones = half_length // 2
    indices = np.random.choice(half_length, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)

    random_initial = np.random.rand(half_length)
    random_initial = random_initial * (integral_target / 2) / np.sum(random_initial)
    start_points.append(random_initial)

    best_c5 = np.inf
    best_h = np.zeros(half_length)

    for h_start in start_points:
        h_start = h_start * (integral_target / 2) / np.sum(h_start)
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_length)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
