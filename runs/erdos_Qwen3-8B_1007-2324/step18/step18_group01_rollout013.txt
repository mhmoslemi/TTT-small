import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n = 60
    integral_target = n / 2

        if len(initial_h) != n:

        y = integral_target - sum_x

        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

        return integral_target - np.sum(x)
    
    def sum_constraint2(x):
        return np.sum(x) - (integral_target - 1)
    
    constraints = [
        {'type': 'ineq', 'fun': sum_constraint1},
        {'type': 'ineq', 'fun': sum_constraint2}
    ]

    if initial_h is not None:
