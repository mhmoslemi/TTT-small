import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def create_h_vals(params):
        x_grid = np.linspace(0, 2, n_points)
        h_vals = np.zeros(n_points)
        for i in range(4):
            c = params[3*i]
            w = params[3*i+1]

            start = max(0.0, c - w/2)
            end = min(2.0, c + w/2)
            mask = (x_grid >= start) & (x_grid <= end)
            h_vals[mask] += a
        h_vals = np.clip(h_vals, 0.0, 1.0)
        integral = np.sum(h_vals * dx)
        scale = target_sum / integral if integral != 0 else 0
        return h_vals * scale

    def objective(params):
        h = create_h_vals(params)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = []

        start_points.append(h_initial)

    for _ in range(3):
        params = []
        for _ in range(4):
            params.append(np.random.uniform(0, 2))
            params.append(np.random.uniform(0.2, 1.8))

        start_points.append(params)

    best_c5 = float('inf')
