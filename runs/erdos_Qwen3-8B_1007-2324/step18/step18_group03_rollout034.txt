    initial_params.append([0.25, 0.5, 1.0, 0.75, 0.5, 1.0, 1.25, 0.5, 1.0, 1.75, 0.5, 1.0])
    
    for params_start in initial_params:
        try:
            result = scipy.optimize.minimize(
                compute_overlap,
                params_start,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success:
                h_vals = np.zeros(n_points)
                x_grid = np.linspace(0, 2, n_points)
                for i in range(4):
                    c, w, a = result.x[3*i], result.x[3*i+1], result.x[3*i+2]
                    left, right = c - w/2, c + w/2
                    valid_idx = np.logical_and(x_grid >= left, x_grid <= right)
                    h_vals[valid_idx] += a
                h_vals = np.clip(h_vals, 0, 1)
                integral = np.sum(h_vals) * dx
                scale = target_sum / integral if integral != 0 else 1.0
                h_scaled = h_vals * scale
                corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
                current_c5 = np.max(corr) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_scaled
        except:
            pass
    
    return best_h, best_c5, n_points
