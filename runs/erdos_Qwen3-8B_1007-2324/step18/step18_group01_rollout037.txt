        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    initial_guesses = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        x_initial = h_initial[:59]
        initial_guesses.append(x_initial)
        x_uniform = np.full(59, 0.5)
        initial_guesses.append(x_uniform)
    else:
        x_uniform = np.full(59, 0.5)
        initial_guesses.append(x_uniform)

    x_random = np.random.rand(59)
    sum_x = np.sum(x_random)
    scale = 29.5 / (sum_x if sum_x != 0 else 1.0)
    x_random = np.clip(x_random * scale, 0, 1)
    initial_guesses.append(x_random)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for x0 in initial_guesses:
        def objective(x):
            sum_x = np.sum(x)
            y = integral_target - sum_x
            h = np.concatenate([x, [y]])
            return compute_overlap(h)

        def constraint_sum_min(x):
            return np.sum(x) - 29

        def constraint_sum_max(x):
            return 30 - np.sum(x)

        constraints = [
            {'type': 'ineq', 'fun': constraint_sum_min},
            {'type': 'ineq', 'fun': constraint_sum_max}
        ]
        bounds = [(0.0, 1.0)] * 59

        result = scipy.optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-12,
                'maxiter': 50,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_h = np.concatenate([result.x, [integral_target - np.sum(result.x)]])
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    return best_h, best_c5, n_points
