        h_opt = result.x

    h_clipped = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_clipped)
    if sum_h < 1e-6:
        h_clipped = h_clipped * (target_sum / np.sum(h_clipped))
    elif sum_h > target_sum + 1e-6:
        h_clipped = h_clipped * (target_sum / sum_h)

    h_values = np.concatenate([h_clipped, h_clipped[::-1]])
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5 = np.max(corr) * dx
    return h_values, c5, n_points
