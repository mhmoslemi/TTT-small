            c = np.mean(x_grid[np.abs(h_vals - i * 0.5) < 0.1])
            w = np.max(x_grid[h_vals > 0]) - np.min(x_grid[h_vals > 0])
            a = np.max(h_vals)
            params[idx:idx+3] = [c, w, a]
        initial_params = params
    
    def objective(params):
        h_unclipped = create_pulse_function(params)
        h_clipped = np.clip(h_unclipped, 0, 1)
        S = np.sum(h_clipped * dx)
        if S == 0:
            return float('inf')
        h_scaled = h_clipped * (target_sum / S)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = []
    for i in range(12):
        if i % 3 == 0:
            bounds.append((0.0, 2.0))
        elif i % 3 == 1:
            bounds.append((0.001, 2.0))
        else:
            bounds.append((0.0, 100.0))
    
    result = scipy.optimize.minimize(
        objective,
        initial_params,
        method='SLSQP',
        bounds=bounds,
        options={'ftol': 1e-10, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
    )
    
    best_params = result.x
    h_unclipped = create_pulse_function(best_params)
    h_clipped = np.clip(h_unclipped, 0, 1)
    S = np.sum(h_clipped * dx)
    if S != 0:
        h_scaled = h_clipped * (target_sum / S)
    else:
        h_scaled = np.zeros(n_points)
    best_c5 = objective(best_params)
    return h_scaled, best_c5, n_points
