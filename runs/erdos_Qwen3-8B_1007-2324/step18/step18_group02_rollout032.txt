import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if initial_h is not None:
        def normalize(h):
            s = np.sum(h)
            if s < 1e-6:
                return h * 0.0
            return h * (integral_target / s)
        start_points.append(normalize(initial_h))
        for _ in range(2):
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = initial_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            start_points.append(normalize(perturbed))

    def structured_initial():
        half = n_points // 2
        seg = half // 4
        res = np.zeros(half)
        res[:seg] = 1.0
        res[seg:seg*2] = 0.5
        res[seg*2:seg*3] = 1.0
        res[seg*3:] = 0.5
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    def sparsify_initial():
        half = n_points // 2
        res = np.zeros(half)
        num_ones = max(1, half // 5)
        res[np.random.choice(half, num_ones, replace=False)] = 1.0
        return normalize(np.concatenate([res, res[::-1]])[:n_points])

    start_points.extend([structured_initial(), sparsify_initial()])

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        res = scipy.optimize.minimize(
            lambda x: compute_overlap(np.clip(x, 0, 1)),
            h_start[:n_points // 2],
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2}],
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False, 'eps': 1e-8}
        )
        if res.success:
            c5 = compute_overlap(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
