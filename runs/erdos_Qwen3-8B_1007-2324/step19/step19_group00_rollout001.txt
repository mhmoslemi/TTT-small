import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) == integral_target:
            return h
        scale = integral_target / np.sum(h)
        return h * scale

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop):
        candidates = np.random.choice(len(pop), 3, replace=False)
        return [pop[i] for i in candidates]

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        if len(parent) > n_points:
            parent = parent[:n_points]
        parent = repair(parent[:n_points//2] * (integral_target / 2) / np.sum(parent[:n_points//2]))
        pop = [parent]
        for _ in range(49):
            h = np.random.rand(n_points//2) * (integral_target / 2)
            pop.append(repair(h))
    else:
        pop = [np.random.rand(n_points//2) * (integral_target / 2) for _ in range(50)]
        pop = [repair(h) for h in pop]

    best_h = pop[0]
    best_c5 = compute_overlap(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(50):
            parents = tournament_selection(pop)
            p1, p2 = parents
            child1 = np.where(np.random.rand(n_points//2) < 0.5, p1, p2)
            child2 = np.where(np.random.rand(n_points//2) < 0.5, p1, p2)
            if np.random.rand() < 0.3:
                child1 += np.random.normal(0, 0.02, n_points//2)
                child2 += np.random.normal(0, 0.02, n_points//2)
            child1 = repair(child1)
            child2 = repair(child2)
            new_pop.append(child1)
            new_pop.append(child2)
        pop = new_pop[:50]
        pop.sort(key=compute_overlap)
        current_c5 = compute_overlap(pop[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[0]

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
