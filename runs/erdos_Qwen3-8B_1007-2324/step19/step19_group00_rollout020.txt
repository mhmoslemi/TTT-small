import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scaling = 30.0 / sum_clipped
        return h_clipped * scaling

    population_size = 50
    population = []

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        population.append(initial_h)
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        population.append(repair(h))

    best_h = min(population, key=compute_overlap)
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break

        tournament = np.random.choice(population, size=3, replace=False)
        parent1 = min(tournament, key=compute_overlap)
        parent2 = min(tournament, key=compute_overlap)

        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child1[i] = parent1[i]
                child2[i] = parent2[i]
            else:
                child1[i] = parent2[i]
                child2[i] = parent1[i]

        if np.random.rand() < 0.3:
            noise = np.random.normal(0, 0.05, n_points)
            child1 = np.clip(child1 + noise, 0, 1)
            child1 = repair(child1)
        if np.random.rand() < 0.3:
            noise = np.random.normal(0, 0.05, n_points)
            child2 = np.clip(child2 + noise, 0, 1)
            child2 = repair(child2)

        candidates = population + [child1, child2]
        population = np.random.choice(candidates, size=population_size, replace=False)

        for h in population:
            c5 = compute_overlap(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5

    return best_h, best_c5, n_points
