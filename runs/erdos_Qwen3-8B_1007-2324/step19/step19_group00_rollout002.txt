import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    integral_target = n_points / 2
    dx = 2.0 / n_points
    max_evals = int(budget_s * 1000)  # ~1000 evals per second

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - integral_target) < 1e-8:
            return h
        scale = integral_target / s
        h = h * scale
        delta = integral_target - np.sum(h)
        if abs(delta) < 1e-8:
            return h
        h[-1] += delta
        return np.clip(h, 0, 1)

    def tournament_selection(pop, scores, k=4):
        elites = np.argsort(scores)[:k]
        return pop[np.random.choice(elites)]

    def crossover(a, b):
        mask = np.random.rand(len(a)) < 0.5
        return np.where(mask, a, b)

    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.1:
                h[i] = 1 - h[i] + np.random.normal(0, 0.02)
                h[i] = np.clip(h[i], 0, 1)
        return h

    def init_population(n_pop):

            h = np.random.rand(n_points)
            h = h * (integral_target / np.sum(h))
            pop.append(h)
        return [repair(h) for h in pop]

    pop_size = 50
    pop = init_population(pop_size)
    best_h = pop[0]
    best_c5 = compute_overlap(best_h)
    evals = 0

    start_time = time.time()
    while evals < max_evals and time.time() - start_time < budget_s:
        scores = [compute_overlap(h) for h in pop]
        for _ in range(pop_size // 2):
            a = tournament_selection(pop, scores)
            b = tournament_selection(pop, scores)
            child = crossover(a, b)
            child = mutate(child)

            best_c5 = current_best
            best_h = pop[scores.index(current_best)]
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
