def run(seed=42, budget_s=100, **kwargs):

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scaling = integral_target / sum_clipped
        repaired = h_clipped * scaling
        return repaired

    # Binary initial
    binary_initial = np.zeros(n_points)
    num_ones = n_points // 2
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_initial[indices] = 1

    population.append(scaled_binary)

    while len(population) < 50:

    for i in range(len(population)):
