        h_initial = np.random.rand(n_points)
        h_initial *= integral_target / np.sum(h_initial)

    def compute_overlap(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            h_normalized = np.full(n_points, integral_target / n_points)
        else:
            h_normalized = h * (integral_target / sum_h)
        h_full = np.concatenate([h_normalized, h_normalized[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.differential_evolution(
        compute_overlap,
        bounds,
        strategy='best1bin',
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=100,
    )

    best_h = result.x
    best_c5 = compute_overlap(best_h)
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
