import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    n_points = 60

    else:
        start_points = []

    # Generate more diverse initial guesses
    def generate_initials(base):
        initials = []
        # Smoothed version
        smoothed = np.convolve(base, np.ones(3), mode='same') / 3
        initials.append(np.clip(smoothed, 0, 1))
        # Uniform random
        random = np.random.rand(n_points // 2) * (integral_target / 2) / np.sum(np.random.rand(n_points // 2))
        initials.append(random)
        # Rotated
        rotated = np.roll(base, np.random.randint(1, n_points // 2))
        initials.append(rotated * (integral_target / 2) / np.sum(rotated))
        # Mirrored
        mirrored = np.flip(base)
        initials.append(mirrored * (integral_target / 2) / np.sum(mirrored))
        # Random perturbation
        perturbed = base + np.random.normal(0, 0.1, n_points // 2)
        perturbed = np.clip(perturbed, 0, 1)
        initials.append(perturbed * (integral_target / 2) / np.sum(perturbed))
        return initials

    if 'initial_h_values' in globals():
        for init in generate_initials(h_initial[:n_points//2]):
            start_points.append(init)

    best_c5 = np.inf
    best_h = np.zeros(n_points//2)
    elapsed = 0.0

    # Precompute for faster correlation
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        h_start_normalized = h_start.copy()
        if np.sum(h_start_normalized) < 1e-6:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)
        else:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)

        def objective(x):
            return compute_overlap(x)

        def constraint(x):
            return np.sum(x) - integral_target / 2

        constraints = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points//2)]

        result = scipy.optimize.minimize(

            h_start_normalized,
