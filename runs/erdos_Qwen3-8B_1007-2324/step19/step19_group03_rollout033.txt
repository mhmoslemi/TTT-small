        for _ in range(population_size - 1):
            binary = generate_binary(half_n, target_sum)
            population.append(binary)
    else:
        for _ in range(population_size):
            binary = generate_binary(half_n, target_sum)
            population.append(binary)

    best_h = None
    best_c5 = np.inf
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate_c5(ind) for ind in population]
        new_pop = []
        for _ in range(population_size):
            t = np.random.choice(population, size=5, replace=False)
            best_t = t[np.argmin(fitness[t])]
            new_pop.append(best_t)
        for i in range(len(new_pop)):
            ind = new_pop[i].copy()
            if np.random.random() < 0.1:
                idx = np.random.randint(half_n)
                ind[idx] ^= 1
                ind = repair_binary(ind, target_sum)
            new_pop[i] = ind
        population = new_pop
        for ind in population:
            c5 = evaluate_c5(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
