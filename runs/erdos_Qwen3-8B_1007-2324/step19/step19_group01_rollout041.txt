import scipy.optimize

    start_points.append(np.roll(h_initial, 10))
    start_points.append(np.roll(h_initial, 20))
    start_points.append(np.clip(h_initial * 1.1, 0, 1))
    start_points.append(np.clip(h_initial * 0.9, 0, 1))
    start_points.append(np.random.rand(n_points//2) * integral_target / np.sum(np.random.rand(n_points//2)))

    best_c5 = np.inf
    best_h = np.zeros(n_points//2)

    for h_start in start_points:

        h_start = np.clip(h_start, 0, 1)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        # Use L-BFGS-B for faster convergence
        result = scipy.optimize.minimize(

            h_start,
            method='L-BFGS-B',

                'disp': False
            }

        if result.success:
