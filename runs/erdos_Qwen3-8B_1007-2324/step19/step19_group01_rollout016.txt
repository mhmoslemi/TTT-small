import numpy as np
from skopt import BayesianOptimization
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    initial_points = []
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_initial = np.clip(h_initial, 0, 1)
        if np.sum(h_initial) > 1e-6:
            h_initial = h_initial * (target_sum / np.sum(h_initial))
        else:
            h_initial = np.zeros(n_points)
        initial_points.append(h_initial)
        initial_points.append(np.roll(h_initial, 10))
        initial_points.append(np.clip(h_initial * 1.2, 0, 1))
    else:
        binary = np.zeros(n_points)
        idx = np.random.choice(n_points, n_points // 2, replace=False)
        binary[idx] = 1
        initial_points.append(binary)
        rand = np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points)))
        initial_points.append(rand)

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        penalty = 1e6 * abs(sum_h - target_sum)
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    optimizer = BayesianOptimization(f=objective, pbounds=bounds, random_state=seed)
    for p in initial_points:
        optimizer.probe(params=p, obj=objective(p))

    start_time = time.time()
    while time.time() - start_time < budget_s:
        optimizer.suggest()
        optimizer.maximize(n_iter=1)

    best_params = optimizer.probest()
    best_h_clipped = np.clip(best_params["params"], 0, 1)
    sum_h = np.sum(best_h_clipped)
    scale = target_sum / sum_h if sum_h > 1e-6 else 1
    best_h_scaled = best_h_clipped * scale
    h_full = np.concatenate([best_h_scaled, best_h_scaled[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    best_c5 = np.max(corr) * dx

    return best_h_scaled, best_c5, n_points
