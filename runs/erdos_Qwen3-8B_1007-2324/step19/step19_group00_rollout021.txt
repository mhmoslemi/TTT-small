import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:

        scale_factor = (integral_target / 2) / np.sum(h_initial_half)
        h_initial_scaled = h_initial_half * scale_factor
        start_points.append(h_initial_scaled)

        perturbed = h_initial_scaled.copy()
        perturbed += np.random.normal(0, 0.01, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral_target / 2) / np.sum(perturbed)
        start_points.append(perturbed)

        rotated = h_initial_scaled.copy()
        shift = np.random.randint(1, half_length)
        rotated = np.roll(rotated, shift)
        rotated = rotated * (integral_target / 2) / np.sum(rotated)
        start_points.append(rotated)

    binary_initial = np.zeros(half_length)

    binary_initial[indices] = 1
