import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    half_points = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_half_initial = h_initial[:half_points]

        h_normalized = h * (integral_target / np.sum(h))

        max_corr = np.max(corr)
        c5 = max_corr * dx
