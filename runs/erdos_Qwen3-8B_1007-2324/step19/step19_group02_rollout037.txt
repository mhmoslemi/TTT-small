import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, len(h_initial)), 
                                 h_initial)
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points) * 0.5

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def _projection(x):
        s = integral_target / max(1, np.sum(x))
        return np.clip(x * s, 0, 1)

    def _scale(x):
        s = integral_target / max(1, np.sum(x))
        return x * s

    start_points = []

    # Uniform random
    random_initial = np.random.rand(n_points)
    random_initial = _scale(random_initial)
    start_points.append(random_initial)

    # Sine-like

        start_points.append(rotated_initial)

        randomized = _scale(randomized)

    best_c5 = np.inf

    for h_start in start_points:
        if np.sum(h_start) < 1e-6:
