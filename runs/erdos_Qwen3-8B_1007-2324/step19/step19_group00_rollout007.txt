import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(solution):
        s = np.sum(solution)
        if abs(s - integral_target) < 1e-8:
            return solution
        scale = integral_target / s
        scaled = solution * scale
        scaled = np.clip(scaled, 0, 1)
        s_new = np.sum(scaled)
        if abs(s_new - integral_target) > 1e-8:
            delta = integral_target - s_new
            indices = np.argsort(scaled)
            for i in indices:
                if scaled[i] > 0:
                    scaled[i] += delta
                    delta = 0
                    break
        return scaled

    def create_individual():
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values)
            if len(parent) > n_points:
                parent = parent[:n_points]
            h = parent.copy()
            np.random.shuffle(h)
            h = h[:n_points//2]
            h = h * (integral_target / 2) / np.sum(h)
            h = repair(h)
            return h
        return np.random.rand(n_points//2) * (integral_target / 2) / np.sum(np.random.rand(n_points//2))

    pop_size = 50
    population = [create_individual() for _ in range(pop_size)]
    best_h = population[0]
    best_c5 = compute_overlap(best_h)

    start_time = time.time()
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        tournament = np.random.choice(population, size=3)
        parents = [tournament[np.argmin([compute_overlap(ind) for ind in tournament])]]
        if np.random.rand() < 0.7:
            # Uniform crossover
            parent1, parent2 = np.random.choice(parents + [np.random.choice(population)], size=2)
            child1 = np.where(np.random.rand(n_points//2) < 0.5, parent1, parent2)
            child2 = np.where(np.random.rand(n_points//2) < 0.5, parent2, parent1)
        else:
            child1 = np.random.rand(n_points//2) * (integral_target / 2) / np.sum(np.random.rand(n_points//2))
            child2 = np.random.rand(n_points//2) * (integral_target / 2) / np.sum(np.random.rand(n_points//2))
        # Mutation
        if np.random.rand() < 0.3:
            child1 += np.random.normal(0, 0.05, n_points//2)
            child1 = np.clip(child1, 0, 1)
            child1 = repair(child1)
        if np.random.rand() < 0.3:
            child2 += np.random.normal(0, 0.05, n_points//2)
            child2 = np.clip(child2, 0, 1)
            child2 = repair(child2)
        # Replace worst
        population.append(child1)
        population.append(child2)
        population.pop(np.argmax([compute_overlap(ind) for ind in population]))
        # Update best
        current_c5 = compute_overlap(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
