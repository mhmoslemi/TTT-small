import scipy.optimize

    # Regular binary sparse start

    # Perturbed parent
    if 'initial_h_values' in globals():

        perturbed_initial += np.random.normal(0, 0.01, size=perturbed_initial.shape)
        perturbed_initial = np.clip(perturbed_initial, 0, 1)
        perturbed_initial = perturbed_initial * (integral_target / 2) / np.sum(perturbed_initial)
        start_points.append(perturbed_initial)

    # Uniform random start

    for h_start in start_points:

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])

        # Optimize with trust-constr
        result = scipy.optimize.minimize(
