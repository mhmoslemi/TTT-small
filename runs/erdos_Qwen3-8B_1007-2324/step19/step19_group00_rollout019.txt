import numpy as np
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def generate_random_solution():
        h = np.random.rand(n_points)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = integral_target / total
        return np.clip(h * scale, 0, 1)

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = integral_target / total
        return h * scale

    if 'initial_h_values' in globals():
        base_pop = [np.array(initial_h_values[:n_points])] * 5
        population = []
        for h_base in base_pop:
            population.append(h_base)
            population.append(generate_random_solution())
            h_mut = h_base.copy()
            h_mut += np.random.normal(0, 0.02, h_base.shape)
            h_mut = repair(h_mut)
            population.append(h_mut)
            h_mut = 1 - h_base.copy()
            h_mut = repair(h_mut)
            population.append(h_mut)
        population = [repair(x) for x in population[:50]]
    else:
        population = [generate_random_solution() for _ in range(50)]

    best_h = population[0]
    best_c5 = compute_overlap(best_h)
    start_time = time.time()

    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        sel1 = np.random.choice(population, 3, replace=False)
        sel2 = np.random.choice(population, 3, replace=False)
        parent1 = min(sel1, key=compute_overlap)
        parent2 = min(sel2, key=compute_overlap)
        # Crossover
        child = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        # Mutation
        if np.random.rand() < 0.1:
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] += np.random.normal(0, 0.02)
                else:
                    child[i] = 1 - child[i]
        child = repair(child)
        # Update best
        child_c5 = compute_overlap(child)
        if child_c5 < best_c5:
            best_c5 = child_c5
            best_h = child
        # Replace population
        population.append(child)
        if len(population) > 50:
            pop_scores = np.array([compute_overlap(x) for x in population])
            worst_idx = np.argmax(pop_scores)
            population.pop(worst_idx)
        # Keep best_h in population
        population.append(best_h)
        population = population[:50]

    return best_h, best_c5, n_points
