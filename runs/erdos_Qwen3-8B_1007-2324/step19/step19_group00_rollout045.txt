import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    population_size = 50
    population = []
    if 'initial_h_values' in globals():
        initial_parent = np.array(initial_h_values)
        if len(initial_parent) != n_points:
            initial_parent = initial_parent[:n_points]
        population.append(initial_parent)
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (integral_target / np.sum(h))
        population.append(h)
    
    best_h = None
    best_c5 = np.inf
    start_time = time.time()
    
    for _ in range(5000):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_overlap(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        parents = []
        for _ in range(2):
            tournament = np.random.choice(len(population), 3, replace=False)
            best_tournament = tournament[np.argmin(fitness[tournament])]
            parents.append(population[best_tournament])
        
        child1 = np.zeros(n_points)
        child2 = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child1[i] = parents[0][i]
                child2[i] = parents[1][i]
            else:
                child1[i] = parents[1][i]
                child2[i] = parents[0][i]
        
        def repair(h):
            h = h + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h *= integral_target / sum_h
            return h
        
        mutated1 = repair(child1)
        mutated2 = repair(child2)
        
        population.append(mutated1)
        population.append(mutated2)
        
        new_fitness = [compute_overlap(ind) for ind in population]
        population = sorted(zip(new_fitness, population), key=lambda x: x[0])[:population_size]
        population = [x[1] for x in population]
    
    return best_h, best_c5, n_points
