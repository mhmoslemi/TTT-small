import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30

    start_points = []

    def generate_initials(base):
        initials = []
        # Smoothed
        smoothed = np.convolve(base, np.ones(3), mode='same') / 3
        initials.append(np.clip(smoothed, 0, 1))
        # Random
        random = np.random.rand(n_points // 2)
        norm = random.sum() / (integral_target / 2)
        initials.append((random / norm) * (integral_target / 2))
        # Rotated
        rotated = np.roll(base, np.random.randint(1, n_points // 2))
        norm = rotated.sum() / (integral_target / 2)
        initials.append((rotated / norm) * (integral_target / 2))
        return initials

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) > n_points:
            h_initial = h_initial[:n_points]
        start_points.extend(generate_initials(h_initial[:n_points // 2]))

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        h_start = h_start.copy()
        total = np.sum(h_start)
        if abs(total) < 1e-6:
            h_start *= (integral_target / 2) / 0.1  # fallback
        else:
            h_start *= (integral_target / 2) / total

        def objective(h_vec):
            h = h_vec.copy()
            total = np.sum(h)
            if abs(total - integral_target / 2) > 1e-6:
                return np.inf
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='L-BFGS-B',
