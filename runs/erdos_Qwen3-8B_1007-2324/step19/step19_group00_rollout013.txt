import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    pop_size = 50
    start_time = time.time()
    population = []

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)[:n_points]
        parent = parent * (integral_target / np.sum(parent))
        population.append(parent)

    while len(population) < pop_size:
        h = np.random.rand(n_points) * (integral_target / np.sum(np.random.rand(n_points)))
        population.append(h)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-8:
            return h
        scale = integral_target / current_sum
        h = np.clip(h * scale, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - integral_target) < 1e-8:
            return h
        h = h * (integral_target / current_sum)
        h = np.clip(h, 0, 1)
        return h

    best_h = np.zeros(n_points)
    best_c5 = np.inf

    while time.time() - start_time < budget_s:
        fitness = np.array([compute_overlap(h) for h in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        new_population = []
        for _ in range(pop_size):
            tournament = np.random.choice(population, 3, replace=False)
            parent1 = min(tournament, key=lambda x: compute_overlap(x))
            parent2 = np.random.choice(population, 1, replace=False)[0]
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            child = repair(child)
            new_population.append(child)
        population = new_population

    return best_h, best_c5, n_points
