        strategy='best1bin',
        tol=1e-6,
        maxiter=100,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_c5 = objective(best_h)
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
