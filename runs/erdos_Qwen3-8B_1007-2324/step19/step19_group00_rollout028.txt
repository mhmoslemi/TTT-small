import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    integral_target = n_points // 2  # 30
    dx = 2.0 / n_points

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(vec):
        s = np.sum(vec)
        if np.isclose(s, integral_target):
            return vec
        scale = integral_target / s
        vec = np.clip(vec * scale, 0, 1)
        return vec

    def generate_individual():
        vec = np.random.rand(n_points)
        vec *= integral_target / np.sum(vec)
        return vec

    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_scaled = initial_h[:n_points] * (integral_target / np.sum(initial_h[:n_points//2]))
        population.append(h_scaled)
        for _ in range(49):
            population.append(generate_individual())
    else:
        for _ in range(50):
            population.append(generate_individual())

    best_h = population[0]
    best_c5 = compute_overlap(best_h)
    start_time = time.time()

    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        t = np.random.choice(population, size=3, replace=False)
        p1, p2 = sorted(t, key=lambda x: compute_overlap(x))[:2]

        child1 = np.where(np.random.rand(n_points) < 0.5, p1, p2)
        child2 = np.where(np.random.rand(n_points) < 0.5, p2, p1)

        def mutate(vec):
            mutated = vec + np.random.normal(0, 0.05, vec.shape)
            mutated = np.clip(mutated, 0, 1)
            return mutated

        child1 = mutate(child1)
        child2 = mutate(child2)
        child1 = repair(child1)
        child2 = repair(child2)

        population.append(child1)
        population.append(child2)
        population.sort(key=lambda x: compute_overlap(x))
        population = population[:50]

        current_c5 = compute_overlap(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]

    return best_h, best_c5, n_points
