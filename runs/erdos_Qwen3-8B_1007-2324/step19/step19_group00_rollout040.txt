            h = repair(h)
        return h

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        population = [initial_h]
        for _ in range(49):
            population.append(generate_initial())
    else:
        population = [generate_initial() for _ in range(50)]

    best_h = population[0]
    best_c5 = compute_overlap(best_h)
    time_start = time.time()

    for _ in range(10000):  # max 10k generations
        if time.time() - time_start > budget_s:
            break

        # Evaluate population
        fitnesses = [compute_overlap(h) for h in population]
        # Tournament selection
        parent1 = tournament_selection(population)
        parent2 = tournament_selection(population)
        # Crossover
        child = crossover(parent1, parent2)
        # Mutate
        child = mutate(child)
        # Repair
        child = repair(child)
        # Replace worst
        idx = np.argmax(fitnesses)
        if compute_overlap(child) < compute_overlap(population[idx]):
            population[idx] = child
        # Update best
        for h in population:
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
