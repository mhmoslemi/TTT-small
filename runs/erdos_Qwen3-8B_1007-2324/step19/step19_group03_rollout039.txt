    population_size = 50
    population = [initial_parent.copy()]
    for shift in range(1, 5):
        rotated = np.roll(initial_parent, shift)
        population.append(rotated)
    for _ in range(population_size - 2 - 4):
        candidate = np.random.choice([0, 1], size=n_points//2)
        current_sum = np.sum(candidate)
        needed = 15 - current_sum
        if needed != 0:
            if needed > 0:
                zeros = np.where(candidate == 0)[0]
                indices = np.random.choice(zeros, size=needed, replace=False)
                candidate[indices] = 1
            else:
                ones = np.where(candidate == 1)[0]
                indices = np.random.choice(ones, size=-needed, replace=False)
                candidate[indices] = 0
        population.append(candidate)

    best_c5 = np.inf
    best_h = np.zeros(n_points//2)

    generations = 200
    tournament_size = 5
    mutation_rate = 0.1

    for gen in range(generations):
        fitness = []
        for candidate in population:
            h_full = np.concatenate([candidate, candidate[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(-c5)

        best_idx = np.argmin(fitness)
        best_c5_current = -fitness[best_idx]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_idx].copy()

        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size)
            best_tournament = tournament[np.argmin(fitness[tournament])]
            parents.append(population[best_tournament])

        offspring = []
        for i in range(population_size // 2):
            parent1 = parents[i]
            parent2 = parents[population_size//2 - 1 - i]
            idx = np.random.randint(n_points//2)
            child = np.concatenate([parent1[:idx], parent2[idx:]])
            current_sum = np.sum(child)
            needed = 15 - current_sum
            if needed != 0:
                if needed > 0:
                    zeros = np.where(child == 0)[0]
                    indices = np.random.choice(zeros, size=needed, replace=False)
                    child[indices] = 1
                else:
                    ones = np.where(child == 1)[0]
                    indices = np.random.choice(ones, size=-needed, replace=False)
                    child[indices] = 0
            if np.random.random() < mutation_rate:
                flip_index = np.random.randint(n_points//2)
                child[flip_index] = 1 - child[flip_index]
                current_sum = np.sum(child)
                needed = 15 - current_sum
                if needed != 0:
                    if needed > 0:
                        zeros = np.where(child == 0)[0]
                        indices = np.random.choice(zeros, size=needed, replace=False)
                        child[indices] = 1
                    else:
                        ones = np.where(child == 1)[0]
                        indices = np.random.choice(ones, size=-needed, replace=False)
                        child[indices] = 0
            offspring.append(child)

        population = offspring

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
