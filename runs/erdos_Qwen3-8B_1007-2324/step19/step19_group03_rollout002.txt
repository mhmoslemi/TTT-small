import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

        start_h = h_initial[:n_points // 2]
    else:
        start_h = np.random.rand(n_points // 2) * (integral_target / 2) / np.sum(np.random.rand(n_points // 2))

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    def objective(h):
        return compute_overlap(h)

    def constraint(h):
        return np.sum(h) - integral_target / 2

    cons = {'type': 'eq', 'fun': constraint, 'jac': np.ones_like}

    for _ in range(3):  # Multiple trials for robustness
        h_start = start_h.copy()
        if np.sum(h_start) < 1e-6:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)
        else:
            h_start = h_start * (integral_target / 2) / np.sum(h_start)

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
