import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    start_points = []

    start_points.append(binary)

    random_init = np.random.rand(n_points//2)

        perturbed = perturbed * (target_sum / 2) / np.sum(perturbed)
        start_points.append(perturbed)

    def compute_overlap(h):
