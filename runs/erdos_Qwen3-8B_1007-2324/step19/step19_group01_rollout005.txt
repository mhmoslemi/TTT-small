        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

        h_initial = h_initial[:n_points // 2]
        start_points.append(h_initial)

        start_points.append(np.clip(h_initial * 0.9, 0, 1))

    for _ in range(5):

    best_c5 = np.inf
