import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Increase points for better resolution
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 40

    start_points = []

    # Use initial_h_values if available

        if len(h_initial) != n_points:

        start_points.append(h_initial)

    def generate_initials(base):
        # Generate diverse initial points: smoothed, inverted, rotated, sparse, dense
        initials = []
        # Smoothed version
        smoothed = np.convolve(base, np.ones(3), mode='same') / 3
        initials.append(np.clip(smoothed, 0, 1))
        # Inverted version
        inverted = 1.0 - base
        initials.append(inverted * (integral_target / 2) / np.sum(inverted))
        # Rotated
        rotated = np.roll(base, np.random.randint(1, n_points // 2))
        initials.append(rotated * (integral_target / 2) / np.sum(rotated))
        # Random sparse
        sparse = np.zeros(n_points // 2)
        cnt = np.random.randint(1, 5)
        idx = np.random.choice(n_points // 2, cnt, replace=False)
        sparse[idx] = 1.0
        initials.append(sparse * (integral_target / 2) / np.sum(sparse))
        # Random dense
        dense = np.random.rand(n_points // 2)
        initials.append(dense * (integral_target / 2) / np.sum(dense))
        return initials

    if 'initial_h_values' in globals():
        for init in generate_initials(h_initial):
            start_points.append(init)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)
    start_time = time.time()

    for h_start in start_points:
        h_start_normalized = h_start.copy()
        # Normalize if sum is near zero
        if np.sum(h_start_normalized) < 1e-6:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start_normalized,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}

                'maxiter': 400,
                'disp': False,
                'eps': 1e-8
            }
