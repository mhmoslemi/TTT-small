        elapsed = time.time() - start_time
        remaining = max(0, budget_s - elapsed)
        if remaining < 0.1:
            break
        maxiter = int(remaining * 2)
        result = scipy.optimize.differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=maxiter,
            popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed
        )
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
