import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):

    def generate_initials(base):
        initials = []

        smoothed = np.convolve(base, np.ones(3), mode='same') / 3
        initials.append(np.clip(smoothed, 0, 1))
        # Uniform random with normalization
        random = np.random.rand(n_points // 2)
        norm = np.sum(random) / (integral_target / 2)
        initials.append(random / norm)
        # Rotated
        rotated = np.roll(base, np.random.randint(1, n_points // 2))
        norm = np.sum(rotated) / (integral_target / 2)
        initials.append(rotated / norm)
        # Alternating pattern
        alternating = np.tile([1, 0, 1, 0], n_points // 4)[:n_points // 2]
        norm = np.sum(alternating) / (integral_target / 2)
        initials.append(alternating / norm)
        return initials

    # Add generated initial guesses
    if 'initial_h_values' in globals():

            start_points.append(init)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        h_start_normalized = h_start.copy()
        if np.sum(h_start_normalized) < 1e-6:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)
        else:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
