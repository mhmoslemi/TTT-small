        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) < n_points:
            initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), 'constant')
        elif len(initial_h) > n_points:
            initial_h = initial_h[:n_points]
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
        initial_h = np.clip(initial_h, 0, 1)

    def objective(params):
        x = np.array([params[f"x{i}"] for i in range(n_points)])
        x_clipped = np.clip(x, 0, 1)
        sum_x = np.sum(x_clipped)
        if sum_x == 0:
            return np.inf
        x_norm = x_clipped * (target_sum / sum_x)
        return compute_overlap(x_norm)

    pbounds = {f"x{i}": (0.0, 1.0) for i in range(n_points)}
    bo = BayesianOptimization(f=objective, pbounds=pbounds, random_state=seed)
    bo.minimize(n_initial_points=5, n_iter=15)

    best_candidate = bo.x
    def refine(h_candidate):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = {
            'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum,
            'jac': np.ones(n_points)
        }
        result = scipy.optimize.minimize(
            compute_overlap,
            h_candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )
        return result.x if result.success else h_candidate

    refined_h = refine(best_candidate)
    best_c5 = compute_overlap(refined_h)
    return refined_h, best_c5, n_points
