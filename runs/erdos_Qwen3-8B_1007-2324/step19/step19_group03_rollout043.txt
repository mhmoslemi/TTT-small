def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():

        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial[:n_points//2])

    def generate_initials(base):

        for kernel in [3, 5]:
            smoothed = np.convolve(base, np.ones(kernel), mode='same') / kernel
            initials.append(np.clip(smoothed, 0, 1))
        base_rand = np.random.rand(n_points//2)
        base_rand = base_rand * (integral_target / np.sum(base_rand))
        initials.append(base_rand)
        for shift in range(1, n_points//2):
            rotated = np.roll(base, shift)
            rotated = rotated * (integral_target / np.sum(rotated))
            initials.append(rotated)
        return initials

        for init in generate_initials(h_initial[:n_points//2]):
