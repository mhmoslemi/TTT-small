def run(seed=42, budget_s=100, **kwargs):

    n_points = 60
    dx = 2.0 / n_points

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    start_points = []

    if 'initial_h_values' in globals():

        perturbed = h_initial_scaled.copy()
        perturbed += nr.normal(0, 0.05, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (integral_target / np.sum(perturbed))
        start_points.append(perturbed)
        rotated = h_initial_scaled.copy()
        shift = nr.randint(1, n_points//2)
        rotated = np.roll(rotated, shift)
        rotated = rotated * (integral_target / np.sum(rotated))
        start_points.append(rotated)
        binary = np.zeros(n_points)

        indices = nr.choice(n_points, num_ones, replace=False)

        binary = binary * (integral_target / np.sum(binary))

        random = nr.rand(n_points) * (integral_target / np.sum(random))

    while len(start_points) < 50:
