import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
    else:
        h_initial = np.random.rand(n_points) * 0.5 + 0.5
        h_initial *= integral_target / np.sum(h_initial)

    def objective_de(h_candidate):
        sum_h = np.sum(h_candidate)
        scale = integral_target / sum_h if sum_h else 1.0
        scaled = h_candidate * scale
        clipped = np.clip(scaled, 0, 1)
        return compute_overlap(clipped)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = scipy.optimize.differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        maxiter=60,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True
    )
    
    best_h_de = result_de.x
    best_c5_de = compute_overlap(best_h_de)
    
    def objective_slsqp(h):
        return compute_overlap(h)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}]
    result_slsqp = scipy.optimize.minimize(
        objective_slsqp,
        best_h_de,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
    )
    
    best_h = result_slsqp.x
    best_c5 = compute_overlap(best_h)
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
