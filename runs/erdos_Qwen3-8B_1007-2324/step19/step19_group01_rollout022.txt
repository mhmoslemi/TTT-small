import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        h_initial = h_initial * (integral_target / np.sum(h_initial)) if np.sum(h_initial) > 1e-6 else np.zeros(n_points)
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 10))
        start_points.append(np.clip(h_initial * 1.1, 0, 1))
        start_points.append(np.clip(h_initial * 0.9, 0, 1))
    else:
        h_start = np.random.rand(n_points) * 0.5
        h_start = h_start * integral_target / np.sum(h_start)
        start_points.append(h_start)
        start_points.append(np.roll(h_start, 10))
        start_points.append(np.clip(h_start * 1.1, 0, 1))
        start_points.append(np.clip(h_start * 0.9, 0, 1))

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    for h_start in start_points:
        h_start = h_start * integral_target / np.sum(h_start) if np.sum(h_start) > 1e-6 else np.zeros(n_points)
        h_start = np.clip(h_start, 0, 1)

        def objective(x):
            x = np.clip(x, 0, 1)
            total = np.sum(x)
            penalty = 0 if abs(total - integral_target) < 1e-6 else (total - integral_target)**2 * 1e6
            return compute_overlap(x) + penalty

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}],
            options={
                'ftol': 1e-12,
                'maxiter': 400,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
