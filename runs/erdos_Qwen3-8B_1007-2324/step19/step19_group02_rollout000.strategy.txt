def _projection(x):  # Enforce L1~integral constraints
    # Lipschitz invariant projection with adaptive scaling
    s = (n_points/2) / max(1, np.sum(x))
    x_out = np.clip(x * s, 0, 1)
    return x_out

def _evaluate_objective(x):
    # Auto-diff compatible
    x_tensor = tf.convert_to_tensor(x, dtype=tf.float32)
    x_tensor_full = tf.concat([x_tensor, tf.reverse(x_tensor, [-1])], 0)
    corr_val = tf.reduce_max(tf.nn.conv1d(x_tensor_full, 
                                          tf.concat([1 - x_tensor_full, x_tensor_full[:-1]], 0),
                                          stride=1, padding='VALID')
    return tf.reduce_max(corr_val) * dx
