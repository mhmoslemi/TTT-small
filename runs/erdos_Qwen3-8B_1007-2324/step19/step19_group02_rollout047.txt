import scipy.optimize as opt

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 40

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 1))
        start_points.append(np.roll(h_initial, -1))
        start_points.append(h_initial * 0.8 + 0.1)
        start_points.append(h_initial * 1.2 - 0.1)
        start_points.append(np.sin(np.linspace(0, 2 * np.pi, n_points) * 0.5) * 0.5 + 0.5)
        start_points.append(np.random.rand(n_points) * 0.1 + 0.45)
        start_points.append(np.random.rand(n_points) * (integral_target / n_points) / np.mean(start_points[-1]))

    else:
        start_points.append(np.random.rand(n_points) * (integral_target / n_points) / np.mean(start_points[0]))

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        h_start = np.clip(h_start, 0, 1)
        h_scaled = h_start * (integral_target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start * (integral_target / n_points)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = opt.minimize(
            compute_overlap,
            h_scaled,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_points)}],

        if result.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
