import numpy as np
from skopt import BayesianOptimization
from skopt.space import Real

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2
    start_point = np.array(initial_h_values[:n_points//2]) if 'initial_h_values' in globals() else np.random.rand(n_points//2)
    start_point = start_point * (integral_target / np.sum(start_point))

    def objective(params):
        x = np.array([params[f'x{i}'] for i in range(n_points//2)])
        x_clipped = np.clip(x, 0.0, 1.0)
        if np.sum(x_clipped) < 1e-10:
            return np.inf
        x_normalized = x_clipped * (integral_target / np.sum(x_clipped))
        h_full = np.concatenate([x_normalized, x_normalized[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    param_space = [Real(0, 1, name=f'x{i}') for i in range(n_points//2)]
    optimizer = BayesianOptimization(
        f=objective,
        pbounds={f'x{i}': (0.0, 1.0) for i in range(n_points//2)},
        random_state=seed
    )
    if 'initial_h_values' in globals():
        initial_x = {f'x{i}': start_point[i] for i in range(n_points//2)}
        optimizer.add_outcomes({initial_x: objective(initial_x)})
    
    optimizer.maximize(n_iter=50)
    best_params = optimizer.x
    best_h = np.concatenate([best_params, best_params[::-1]])
    best_c5 = objective(best_params)
    return best_h, best_c5, n_points
