import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60

    integral_target = n_points / 2  # 30

    start_points = []

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

    def generate_initials(base):
        initials = []
        # Smoothed version
        smoothed = np.convolve(base, np.ones(3), mode='same') / 3
        initials.append(np.clip(smoothed, 0, 1))
        # Uniform random
        random = np.random.rand(n_points // 2) * (integral_target / 2) / np.sum(np.random.rand(n_points // 2))
        initials.append(random)
        # Rotated
        rotated = np.roll(base, np.random.randint(1, n_points // 2))
        initials.append(rotated * (integral_target / 2) / np.sum(rotated))
        # Mirrored
        mirrored = np.flip(base)
        initials.append(mirrored * (integral_target / 2) / np.sum(mirrored))
        # Scaled
        scaled = base * 0.8 + 0.1
        initials.append(np.clip(scaled, 0, 1))
        return initials

    if 'initial_h_values' in globals():
        for init in generate_initials(h_initial[:n_points // 2]):
            start_points.append(init)

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    for h_start in start_points:
        h_start_normalized = h_start.copy()
        if np.sum(h_start_normalized) < 1e-6:

        else:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
