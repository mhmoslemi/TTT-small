        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    pbounds = {f'x{i}': (0.0, 1.0) for i in range(dim)}
    bo = BayesianOptimization(objective, pbounds, random_state=seed)
    bo.minimize(n_initial_points=5, acquisition='ei')

    best_c5 = np.inf
    best_h = np.zeros(dim)

    for _ in range(5):
        candidate = bo.ask()
        x_candidate = np.array(candidate)
        def compute_overlap(x):
            h = x
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 0.5}]
        result = scipy.optimize.minimize(
            compute_overlap, x_candidate,
            method='SLSQP', bounds=[(0.0, 1.0)]*dim,
            constraints=constraints,
            options={'ftol': 1e-12, 'maxiter': 300, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        bo.tell(candidate, best_c5)

    h_full_best = np.concatenate([best_h, best_h[::-1]])
    return h_full_best, best_c5, n_points
