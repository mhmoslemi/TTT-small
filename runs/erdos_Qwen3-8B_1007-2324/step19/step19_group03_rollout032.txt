            candidate = np.random.choice([0,1], size=n_points, p=[0.5,0.5])
            while np.sum(candidate) != target_sum:
                candidate = np.random.choice([0,1], size=n_points, p=[0.5,0.5])
            initial_population.append(candidate)
        for _ in range(2):
            rotated = np.roll(initial_binary, np.random.randint(1, n_points//2))
            initial_population.append(rotated)
    else:
        for _ in range(5):
            candidate = np.random.choice([0,1], size=n_points, p=[0.5,0.5])
            while np.sum(candidate) != target_sum:
                candidate = np.random.choice([0,1], size=n_points, p=[0.5,0.5])
            initial_population.append(candidate)

    population = initial_population

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = None
    best_c5 = np.inf
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        new_population = []
        for _ in range(len(population)):
            tournament = np.random.choice(len(population), size=5, replace=False)
            best_t = tournament[np.argmin([fitness[i] for i in tournament])]
            parent1 = population[best_t]
            tournament = np.random.choice(len(population), size=5, replace=False)
            best_t = tournament[np.argmin([fitness[i] for i in tournament])]
            parent2 = population[best_t]
            point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:point], parent2[point:]])
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                child[idx] = 1 - child[idx]
            current_sum = np.sum(child)
            diff = target_sum - current_sum
            if diff != 0:
                if diff > 0:
                    zeros = np.where(child == 0)[0]
                    flip = zeros[:diff]
                    child[flip] = 1
                else:
                    ones = np.where(child == 1)[0]
                    flip = ones[:abs(diff)]
                    child[flip] = 0
            new_population.append(child)
        population = new_population

    return best_h, best_c5, n_points
