import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points // 2  # 30

    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) < n_points:
            h_initial = h_initial[:n_points]
        h_initial = h_initial[:n_points // 2]

        # Generate transformed initial guesses
        start_points = [h_initial]
        start_points.append(np.roll(h_initial, 10))
        start_points.append(np.roll(h_initial, 20))
        start_points.append(np.roll(h_initial, 30))
        start_points.append(np.flip(h_initial))
        start_points.append(h_initial * 0.95 + 0.025)
    else:
        # Fallback to random initial guess
        start_points = [np.random.rand(n_points // 2) * (integral_target / 2) / np.sum(np.random.rand(n_points // 2))]

    # Timer for budget
    start_time = time.time()

    for h_start in start_points:
        # Scale to meet integral constraint
        h_start = h_start * (integral_target / np.sum(h_start)) if np.sum(h_start) > 1e-6 else h_start * 0
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}],
            options={
                'ftol': 1e-8,
                'maxiter': 200,
                'disp': False,
                'eps': 1e-6
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        # Check time budget
        if time.time() - start_time > budget_s - 5:
            break

    best_h_full = np.concatenate([best_h, best_h[::-1]])
