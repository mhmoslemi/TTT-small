        sine_initial = sine_initial * (integral_target / np.sum(sine_initial))
        start_points.append(sine_initial)

    # Additional variations based on initial_h_values
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        # Rotated variant
        rotated_initial = np.roll(h_initial, np.random.randint(1, n_points // 2))
        start_points.append(rotated_initial)
        # Perturbed variant

        perturbed_initial = np.clip(perturbed_initial, 0, 1)
        perturbed_initial = perturbed_initial * (integral_target / np.sum(perturbed_initial))
        start_points.append(perturbed_initial)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        # Clip and normalize to meet the integral constraint
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        # Optimization with L-BFGS-B for better performance
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[
