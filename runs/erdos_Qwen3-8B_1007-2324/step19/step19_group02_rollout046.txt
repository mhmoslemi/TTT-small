    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial[:n_points]
        h_start = h_initial
    else:
        h_start = np.random.rand(n_points // 2)
        h_start = h_start * (integral_target / 2.0) / np.sum(h_start)

    # Define objective and constraint
    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Optimization
    result = scipy.optimize.minimize(
        compute_c5,
        h_start,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(len(h_start))],
        constraints=[
            {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2.0,
             'jac': np.ones(len(h_start))}]
    )

    best_h = h_start
    best_c5 = compute_c5(best_h)
    if result.success:
        best_h, best_c5 = result.x, compute_c5(result.x)

    # Perturb and optimize again
    perturbed = best_h + np.random.normal(0, 0.05, best_h.shape)
    perturbed = np.clip(perturbed, 0, 1)
    perturbed = perturbed * (integral_target / 2.0) / np.sum(perturbed)
    perturbed_c5 = compute_c5(perturbed)
    if perturbed_c5 < best_c5:
        best_h, best_c5 = perturbed, perturbed_c5

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
