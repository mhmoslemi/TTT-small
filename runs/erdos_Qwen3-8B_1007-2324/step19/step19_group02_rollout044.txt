        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        # Optimize
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points//2)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-12,
                'maxiter': 300,
                'disp': False,
                'eps': 1e-8
            }
        )

        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
