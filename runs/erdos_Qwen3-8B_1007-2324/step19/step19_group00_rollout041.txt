    best_h = population[0]
    best_c5 = compute_overlap(best_h)

    for _ in range(max_evals):
        def select_parent():
            idxs = np.random.choice(len(population), tournament_size, replace=False)
            return population[np.argmin([compute_overlap(population[i]) for i in idxs])]

        parent1 = select_parent()
        parent2 = select_parent()

        child1, child2 = np.zeros(n_points), np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < crossover_rate:
                child1[i], child2[i] = parent1[i], parent2[i]
            else:
                child1[i], child2[i] = parent2[i], parent1[i]

        child1 = mutate(child1)
        child2 = mutate(child2)

        c1 = compute_overlap(child1)
        c2 = compute_overlap(child2)

        if c1 < best_c5:
            best_c5 = c1
            best_h = child1.copy()
        if c2 < best_c5:
            best_c5 = c2
            best_h = child2.copy()

        population = np.vstack([population, child1, child2])
        fitness = [compute_overlap(ind) for ind in population]
        sorted_idx = np.argsort(fitness)
        population = population[sorted_idx[:pop_size]]

    return best_h, best_c5, n_points
