        delta = (30.0 - total_updated) * 0.5

def evaluate_fitness(values):
    if not isinstance(values, np.ndarray):
        values = np.array(values)
    try:
        corr = np.correlate(values, 1-values, mode='full')
        max_corr = np.max(corr)
        return max_corr, np.isfinite(max_corr)
    except Exception as e:
        return float('inf'), False

def tournament_selection(population, fitnesses):
    tournament_size = min(4, len(population))
    candidates = np.random.choice(len(population), tournament_size, replace=False)
    best_index = candidates[0]
    best_fitness = fitnesses[candidates].min()
    return best_index, best_fitness

def crossover(parent1, parent2):
    alpha = 0.9  # Crossover probability
    mask = np.random.random(len(parent1)) < alpha
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2
