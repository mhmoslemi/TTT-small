            child1 = parent1.copy()
            child2 = parent2.copy()
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child1[j] = parent2[j]
                if np.random.rand() < 0.5:
                    child2[j] = parent1[j]
            offspring.append(child1)
            offspring.append(child2)

        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] = 1.0 - h[j]
                    h[j] += np.random.normal(0, 0.01)
                    h[j] = np.clip(h[j], 0, 1)
            offspring[i] = repair(h)

        offspring_fitness = [compute_overlap(h) for h in offspring]
        for i in range(pop_size):
            if offspring_fitness[i] < pop_fitness[i]:
                population[i] = offspring[i]
                pop_fitness[i] = offspring_fitness[i]

        current_best_idx = np.argmin(pop_fitness)
        if pop_fitness[current_best_idx] < best_c5:
            best_c5 = pop_fitness[current_best_idx]
            best_h = population[current_best_idx]

    return best_h, best_c5, n_points
