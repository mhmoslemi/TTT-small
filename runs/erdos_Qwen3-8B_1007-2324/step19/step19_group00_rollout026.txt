import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-6:
            h[0] = 1.0
            return h
        scale = integral_target / total
        h = h * scale
        return h

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def crossover(parent1, parent2):
        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(len(parent1)):
            if np.random.rand() < 0.5:
                child1[i] = parent1[i]
                child2[i] = parent2[i]
            else:
                child1[i] = parent2[i]
                child2[i] = parent1[i]
        return child1, child2

    def mutate(h):
        noise = np.random.normal(0, 0.02, len(h))
        h += noise
        h = np.clip(h, 0, 1)
        h = repair(h)
        return h

    population_size = 50
    population = []

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = initial_h[:n_points]
        population.append(initial_h)

    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    population.append(random_initial)

    binary = np.zeros(n_points)
    num_ones = n_points // 2
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary[indices] = 1
    binary = binary * (integral_target / np.sum(binary))
    population.append(binary)

    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (integral_target / np.sum(h))
        population.append(h)

    for i in range(len(population)):
        population[i] = repair(population[i])

    best_h = population[0]
    best_c5 = compute_overlap(population[0])

    tournament_size = 4
    crossover_prob = 0.9
    mutation_prob = 0.1

    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = [compute_overlap(ind) for ind in population]
        new_population = []
        for _ in range(population_size):
            tournament = np.random.choice(population, tournament_size, replace=False)
            winner = min(tournament, key=lambda x: compute_overlap(x))
            new_population.append(winner)

        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_prob:
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                child1, child2 = crossover(parent1, parent2)
                new_population[i] = child1
                new_population[i+1] = child2

        for i in range(population_size):
            if np.random.rand() < mutation_prob:
                new_population[i] = mutate(new_population[i])

        for i in range(population_size):
            new_population[i] = repair(new_population[i])

        for ind in new_population:
            current_c5 = compute_overlap(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind

        population = new_population

    return best_h, best_c5, n_points
