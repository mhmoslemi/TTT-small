import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_point = h_initial
    else:
        start_point = np.random.rand(n_points)
        start_point = start_point * (integral_target) / np.sum(start_point)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_overlap(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = scipy.optimize.minimize(
        objective,
        start_point,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
        ],
        options={
            'ftol': 1e-10,
            'maxiter': 500,
            'disp': False,
            'eps': 1e-8
        }
    )

    if not result.success:
        best_h = np.random.rand(n_points)
        best_h = best_h * (integral_target) / np.sum(best_h)
        best_c5 = compute_overlap(best_h)
    else:
        best_h = result.x
        best_c5 = compute_overlap(best_h)

    return best_h, best_c5, n_points
