import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = 30.0
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    timer = time.time()
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if abs(total) < 1e-6:
            return np.random.rand(n_points)
        scale = integral_target / total
        return h * scale
    
    def generate_random():
        h = np.random.rand(n_points)
        total = np.sum(h)
        h = h * (integral_target / max(total, 1e-6))
        return repair(h)
    
    def gaussian_perturb(h):
        return repair(h + np.random.normal(0, 0.01, n_points))
    
    def oppositional_perturb(h):
        return repair(1.0 - h)
    
    if 'initial_h_values' in globals():
        base_pop = [initial_h_values[:n_points]] * 5
        pop = []
        for h_base in base_pop:
            h = np.clip(h_base, 0.0, 1.0)
            total = np.sum(h)
            if abs(total) < 1e-6:
                h = np.random.rand(n_points)
            scale = integral_target / total
            h = h * scale
            pop.append(h)
            pop.append(generate_random())
            pop.append(gaussian_perturb(h_base))
            pop.append(oppositional_perturb(h_base))
        pop = [repair(x) for x in pop[:50]]
    else:
        pop = [generate_random() for _ in range(50)]
    
    population = pop[:]
    
    for gen in range(200):
        if time.time() - timer > budget_s:
            break
        fitness = [compute_overlap(h) for h in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
        new_pop = [population[best_idx]]
        while len(new_pop) < len(population):
            candidates = np.random.choice(len(population), 3, replace=False)
            winner = candidates[np.argmin(fitness[candidates])]
            p1 = population[winner]
            candidates = np.random.choice(len(population), 3, replace=False)
            winner = candidates[np.argmin(fitness[candidates])]
            p2 = population[winner]
            child = np.random.choice([p1, p2], size=n_points)
            if np.random.rand() < 0.1:
                child += np.random.normal(0, 0.01, n_points)
                if np.random.rand() < 0.5:
                    child = 1.0 - child
            new_pop.append(repair(child))
        population = new_pop
    
    return best_h, best_c5, n_points
