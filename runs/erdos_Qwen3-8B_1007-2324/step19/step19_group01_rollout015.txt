import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    integral_target = n_points / 2

    start_points = []

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = h_initial[:n_points]
        start_points.append(h_initial)
        start_points.append(np.roll(h_initial, 20))
        start_points.append(np.roll(h_initial, 40))
        start_points.append(np.clip(h_initial * 1.1, 0, 1))
        start_points.append(np.clip(h_initial * 0.9, 0, 1))
        start_points.append(np.random.permutation(h_initial))

    random_initial = np.random.rand(n_points)
    random_initial = random_initial * (integral_target / np.sum(random_initial))
    start_points.append(random_initial)

    binary_initial = np.zeros(n_points)
    num_ones = n_points // 5
    indices = np.random.choice(n_points, num_ones, replace=False)
    binary_initial[indices] = 1
    start_points.append(binary_initial)

    perturbed_initial = np.array(initial_h_values) if 'initial_h_values' in globals() else np.zeros(n_points)
    perturbed_initial += np.random.normal(0, 0.05, size=perturbed_initial.shape)
    perturbed_initial = np.clip(perturbed_initial, 0, 1)
    perturbed_initial = perturbed_initial * (integral_target / np.sum(perturbed_initial))
    start_points.append(perturbed_initial)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        h_start = h_start * (integral_target / np.sum(h_start))

        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        def objective(x):
            x = np.clip(x, 0, 1)
            x = x * (integral_target / np.sum(x))
            return compute_overlap(x)

        result = scipy.optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],

        if result.success:
