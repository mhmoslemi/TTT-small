import numpy as np
import scipy.optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        h_initial = np.random.rand(n_points) * integral_target / np.mean(h_initial)

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode="full")
        return np.max(corr) * dx

    def scale_h(h):
        if np.sum(h) < 1e-6:
            s = integral_target / 2.0
        else:
            s = (integral_target / 2.0) / np.sum(h)
        return np.clip(h * s, 0, 1)

    start_points = []
    if 'initial_h_values' in globals():
        start_points.append(scale_h(h_initial))
        perturb = np.random.normal(0, 0.05, size=n_points)
        start_points.append(scale_h(h_initial + perturb))
        shift = np.random.randint(1, n_points // 2)
        start_points.append(scale_h(np.roll(h_initial, shift)))
        start_points.append(scale_h(np.roll(h_initial, -shift)))

    start_points.append(np.random.rand(n_points) * integral_target / np.mean(h_initial))
    start_points.append(np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5)
    start_points.append(np.sin(np.linspace(0, 2 * np.pi, n_points)) * 0.5 + 0.5)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for h_start in start_points:
        try:
            result = scipy.optimize.minimize(
                compute_overlap,
                h_start,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=[
                    {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones(n_points)}
                ],
                options={
                    'ftol': 1e-12,
                    'maxiter': 300,
                    'disp': False,
                    'eps': 1e-8
                }
            )
            if result.success:
                current_c5 = compute_overlap(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    return best_h, best_c5, n_points
