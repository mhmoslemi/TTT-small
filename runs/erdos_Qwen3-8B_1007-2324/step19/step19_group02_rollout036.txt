import numpy as np

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def project(h):
        s = integral_target / (np.sum(h) + 1e-12)
        return np.clip(h * s, 0, 1)

    def evaluate(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x = np.linspace(0, 2, len(h_init))
            h_init = np.interp(np.linspace(0, 2, n_points), x, h_init)
        population = [h_init]
    else:
        h_binary = np.zeros(n_points)
        idx = np.random.choice(n_points, int(integral_target), replace=False)
        h_binary[idx] = 1
        population = [h_binary]

    for _ in range(10):
        h = np.random.rand(n_points)
        h = project(h)
        population.append(h)

    best_h = population[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        fitnesses = [evaluate(h) for h in population]
        top = np.argsort(fitnesses)[:len(population)//2]
        selected = [population[i] for i in top]
        new_pop = []
        for _ in range(len(population)):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = (p1 + p2) * 0.5 + np.random.normal(0, 0.05, n_points)
            child = project(np.clip(child, 0, 1))
            new_pop.append(child)
        population = new_pop
        for h in population:
            c = evaluate(h)
            if c < best_c5:
                best_c5 = c
                best_h = h

    best_h = project(best_h)
    return best_h, best_c5, n_points
