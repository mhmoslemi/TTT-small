        if 'initial_h_values' in globals():
            perturbation = np.random.normal(0, 0.05, size=n_points // 2)
            perturbed = h_initial + perturbation
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (integral_target / np.sum(perturbed)) if np.sum(perturbed) > 0 else perturbed
            initial_points.append(perturbed)
        else:
            initial_points.append(np.random.rand(n_points // 2) * (integral_target / np.sum(np.random.rand(n_points // 2))))
    initial_points.append(h_initial)
    for point in initial_points:
        obj_val = objective(point)
        optimizer.tell(point, obj_val)

    # Run Bayesian optimization within the budget
    start_time = time.time()
    while time.time() - start_time < budget_s:
        next_point = optimizer.ask()
        h = np.array(next_point)
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            obj_val = np.inf
        else:
            h_scaled = h_clipped * (integral_target / total)
            h_full = np.concatenate([h_scaled, h_scaled[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            obj_val = np.max(corr) * dx
        optimizer.tell(next_point, obj_val)

    # Get the best solution
    best_point = optimizer.get_best()[0]
    best_c5 = objective(best_point)

    # Local optimization using SLSQP
    def objective_local(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.inf
        h_scaled = h_clipped * (integral_target / total)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = scipy.optimize.minimize(
        objective_local,
        best_point,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
        constraints=[
            {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target, 'jac': np.ones_like}
        ],
        options={
            'ftol': 1e-12,
            'maxiter': 300,
            'disp': False,
            'eps': 1e-8
        }
    )

    if result.success:
        final_h = result.x
        final_c5 = result.fun
    else:
        final_h = best_point
        final_c5 = best_c5

    h_full = np.concatenate([final_h, final_h[::-1]])
    return h_full, final_c5, n_points
