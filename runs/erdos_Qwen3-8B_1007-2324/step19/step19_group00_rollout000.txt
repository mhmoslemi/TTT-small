import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    integral_target = n_points // 2
    dx = 2.0 / n_points
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.full_like(h, 1.0 / len(h))
        scale = integral_target / h_sum
        h = h * scale
        h_sum = np.sum(h)
        delta = integral_target - h_sum
        if abs(delta) < 1e-10:
            return h
        h = h * (1 - delta / h_sum)
        return h
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def tournament_selection(pop, fitness, size=4):
        indices = np.random.choice(len(pop), size, replace=False)
        best_idx = np.argmin([fitness[i] for i in indices])
        return pop[indices[best_idx]]
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        mask = np.random.rand(len(parent1)) < 0.5
        child[mask] = parent2[mask]
        return repair(child)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return repair(h)
    
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        if len(init_h) != n_points:
            init_h = init_h[:n_points]
        pop = [repair(init_h) for _ in range(50)]
    else:
        pop = [np.random.rand(n_points) for _ in range(50)]
        for i in range(len(pop)):
            pop[i] = repair(pop[i])
    
    fitness = np.array([compute_overlap(h) for h in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx]
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    gen = 0
    while time.time() - start_time < budget_s:
        gen += 1
        new_pop = []
        for _ in range(len(pop) // 2):
            p1 = tournament_selection(pop, fitness)
            p2 = tournament_selection(pop, fitness)
            child = crossover(p1, p2)
            new_pop.append(child)
            new_pop.append(mutate(child))
        pop = new_pop
        fitness = np.array([compute_overlap(h) for h in pop])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx]
    
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
