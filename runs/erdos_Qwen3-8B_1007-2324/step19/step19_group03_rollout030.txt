import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    integral_target = n_points / 2  # 30
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        start_points = [np.ones(n_points // 2)]
    
    # Generate more diverse initial guesses
    def generate_initials(base):
        initials = []
        # Smoothed version
        smoothed = np.convolve(base, np.ones(3), mode='same') / 3
        initials.append(np.clip(smoothed, 0, 1))
        # Uniform random initialization
        random = np.random.rand(n_points // 2)
        random_sum = random.sum()
        if random_sum > 0:
            random = random * (integral_target / 2) / random_sum
        else:
            random = np.full(n_points // 2, integral_target / 2 / n_points // 2)
        initials.append(random)
        # Rotated initialization
        rotated = np.roll(base, np.random.randint(1, n_points // 2))
        rotated_sum = rotated.sum()
        if rotated_sum > 0:
            rotated = rotated * (integral_target / 2) / rotated_sum
        else:
            rotated = np.full(n_points // 2, integral_target / 2 / n_points // 2)
        initials.append(rotated)
        return initials
    
    if 'initial_h_values' in globals():
        for init in generate_initials(h_initial[:n_points // 2]):
            start_points.append(init)
    
    best_c5 = np.inf
    best_h = np.zeros(n_points // 2)
    start_time = time.time()
    
    for h_start in start_points:
        h_start_normalized = h_start.copy()
        if np.sum(h_start_normalized) < 1e-6:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)
        else:
            h_start_normalized = h_start_normalized * (integral_target / 2) / np.sum(h_start_normalized)
        
        def compute_overlap(h):
            h_full = np.concatenate([h, h[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = scipy.optimize.minimize(
            compute_overlap,
            h_start_normalized,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points // 2)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - integral_target / 2, 'jac': np.ones_like}
            ],
            options={
                'ftol': 1e-9,
                'maxiter': 250,
                'disp': False,
                'eps': 1e-8
            }
        )
        
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            if time.time() - start_time > budget_s - 1:
                break
    
    best_h_full = np.concatenate([best_h, best_h[::-1]])
    return best_h_full, best_c5, n_points
