def run(seed=42, budget_s=100, **kwargs):

    n_points = 200

    def objective(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    T0 = 0.5
    cooling_rate = 0.999
    best_h = h_opt.copy()
    best_c5 = objective(best_h)
    current_h = h_opt.copy()
    current_c5 = best_c5
    T = T0

    while time.time() - start_time < budget_s - 1:
        noise = np.random.normal(0, 0.05, n_half)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * (integral_target / sum_new)
        new_c5 = objective(new_h)
        delta = new_c5 - current_c5
        if delta < 0:
            if np.random.rand() < np.exp(-delta / T):
                current_h, current_c5 = new_h, new_c5
        else:
            current_h, current_c5 = new_h, new_c5
        if current_c5 < best_c5:

        T *= cooling_rate
