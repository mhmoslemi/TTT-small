import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt = np.clip(h_opt, 0, 1)
        sum_h = np.sum(h_opt)
        h_opt = h_opt * (integral_target / sum_h)

    def objective(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    T0 = 0.15
    cooling_rate = 0.99
    best_h = h_opt.copy()
    best_c5 = objective(best_h)
    current_h = h_opt.copy()
    current_c5 = best_c5
    T = T0

    while time.time() - start_time < budget_s - 1:
        noise = np.random.normal(0, 0.05, n_half)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * (integral_target / sum_new)
        new_c5 = objective(new_h)
        delta = new_c5 - current_c5
        if delta < 0:
            if np.random.rand() < np.exp(-delta / T):
                current_h, current_c5 = new_h, new_c5
        else:
            current_h, current_c5 = new_h, new_c5
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
        T *= cooling_rate

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * (integral_target / sum_best)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
