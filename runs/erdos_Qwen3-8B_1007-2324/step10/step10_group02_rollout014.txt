import time

def run(seed=42, budget_s=100, **kwargs):

        h_opt = h_opt * (target_sum / np.sum(h_opt))
    
    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s) < 1e-10:
            return np.ones_like(h) * target_sum / len(h)
        factor = (target_sum) / s
        return h * factor

    def compute_c5(h):
        h_full = np.concatenate([h, h[::-1]])
        h_full = repair(h_full)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
