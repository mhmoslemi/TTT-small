import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    import scipy.optimize as optimize

    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt = h_opt / np.sum(h_opt) * integral_target

    def repair(h, integral_target):
        n = len(h)
        weights = np.abs(np.arange(n) - n // 2)
        sorted_indices = np.argsort(weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(x, h_opt, integral_target):
        h = x.copy()
        h = repair(h, integral_target)
        return compute_overlap(h)

    def constraint(x):
        return np.sum(x) - integral_target

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_half)]

    start_time = time.time()
    result = optimize.minimize(objective, h_opt.copy(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
