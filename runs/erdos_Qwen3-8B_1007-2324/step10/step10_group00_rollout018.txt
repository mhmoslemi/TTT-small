import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2

    def obj(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h *= integral_target / s
        return h

    h = repair(h_opt)
    best_h = h.copy()
    best_c5 = obj(h)

    T = 0.3
    cooling = 0.995
    iter_count = 0
    max_iter = int(budget_s * 100)
