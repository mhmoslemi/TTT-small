import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    np.random.seed(seed)
    
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2  # sum(h) == n_points / 2
    
    # Initialize with parent or sine wave
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_points//2]
    else:
        x = np.linspace(0, 2, n_points)
        h_opt = 0.5 * (1 + np.sin(np.pi * x))
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt[:n_points//2]
    
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def repair(perturbed):
        s = np.sum(perturbed)
        if s == 0:
            return np.zeros_like(perturbed)
        scale = integral_target / s
        return np.clip(perturbed * scale, 0, 1)
    
    # Optimization
    start_time = time.time()
    T = 1.0
    cooling_rate = 0.998
    current_h = h_opt.copy()
    best_h = current_h.copy()
    best_c5 = compute_overlap(current_h)
    
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.005, len(current_h))
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed)
        c5 = compute_overlap(repaired)
        if c5 < best_c5 or np.random.rand() < np.exp((c5 - best_c5)/T):
            best_h = repaired.copy()
            best_c5 = c5
        T *= cooling_rate
    
    elapsed = time.time() - start_time
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
