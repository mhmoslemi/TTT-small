import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, integral_target):
        n = len(h)
        sorted_idx = np.argsort(np.abs(2 * np.arange(n) - (n - 1)))
        weights = np.abs(2 * np.arange(n) - (n - 1)) + 1
        weights_norm = weights / np.sum(weights)
        sorted_vals = np.sort(h)
        h_corrected = np.zeros_like(h)
        for i in range(n):
            pos = n - 1 - np.abs(n - 1 - sorted_idx[i])
            h_corrected[i] = sorted_vals[min(max(int(pos), 0), n-1)]
        h_clipped = np.clip(h_corrected, 0, 1)
        if np.sum(h_clipped) > 0:
            h_scaled = h_clipped * (integral_target / np.sum(h_clipped))
        else:
            h_scaled = np.zeros_like(h_clipped)
        return np.clip(h_scaled, 0, 1)

    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:140]
    else:
        x = np.linspace(0, 1, 140)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h.copy()

    population = [h_opt + np.random.normal(0, 0.05, 140) for _ in range(50)]
    population = [np.clip(repair_h(h, integral_target), 0, 1) for h in population]
    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)

    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        if compute_overlap(current_best) < best_c5:
            best_h = current_best.copy()
            best_c5 = compute_overlap(best_h)
        new_population = []
        for i in range(0, 50, 2):

        population = new_population
