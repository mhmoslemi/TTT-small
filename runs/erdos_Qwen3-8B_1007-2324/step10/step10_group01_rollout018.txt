def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    n_half = 140
    dx = 2.0 / n_points
    integral_target = n_points // 4  # 70

    if 'initial_h_values' in globals():

        sorted_vals = np.sort(h)
