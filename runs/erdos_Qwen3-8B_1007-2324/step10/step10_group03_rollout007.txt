def run(seed=42, budget_s=100, **kwargs):

        h_opt = h[:n_points//2]

    def repair(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = integral_target / current_sum
        scaled = np.clip(h * scale, 0, 1)
        return scaled

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def objective(h_flat):
        h = h_flat[:n_points//2]
        return compute_overlap(h)

    def constraint(h_flat):
        return np.sum(h_flat[:n_points//2]) - integral_target

    def bounds_func(x):
        return [(0, 1) for _ in range(n_points//2)]

    def local_refine(h_opt):
        x0 = h_opt.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_func, constraints={'type': 'eq', 'fun': constraint})
        return res.x

    start_time = time.time()
    T = 0.1
    cooling_rate = 0.999
    current_h = h_opt.copy()
    best_h = current_h.copy()
    best_c5 = compute_overlap(current_h)

    for _ in range(5000):

            break
