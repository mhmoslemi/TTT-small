import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    def initial_h():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            return h[:n_half]
        else:
            x = np.linspace(0, 1, n_half)
            return np.clip(1 - np.abs(x - 0.5), 0, 1)

    def repair_h(h, integral_target):
        n = len(h)
        sorted_idx = np.argsort(np.abs(2 * np.arange(n) - (n - 1)))
        weights = np.abs(2 * np.arange(n) - (n - 1)) + 1
        weights_norm = weights / np.sum(weights)
        sorted_vals = np.sort(h)
        h_corrected = np.zeros_like(h)
        for i in range(n):
            pos = n - 1 - np.abs(n - 1 - sorted_idx[i])
            h_corrected[i] = sorted_vals[min(max(pos, 0), n - 1)]
        h_clipped = np.clip(h_corrected, 0, 1)
        scale = integral_target / (np.sum(h_clipped) if np.sum(h_clipped) > 0 else 1)
        return np.clip(scale * h_clipped, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def crossover(parent1, parent2):
        child1, child2 = parent1.copy(), parent2.copy()
        idx = np.random.randint(3, size=len(parent1))
        for i in range(len(parent1)):
            if idx[i] > 0:
                child1[i], child2[i] = parent2[i], parent1[i]
            else:
                child1[i] = 0.5 * parent1[i] + 0.5 * parent2[i]
                child2[i] = 0.5 * parent2[i] + 0.5 * parent1[i]
        return np.clip(child1, 0, 1), np.clip(child2, 0, 1)

    def mutate(h, temp):
        child = h.copy()
        noise = np.random.normal(0, temp, len(h))
        child += noise
        child = np.clip(child, 0, 1)
        child = repair_h(child, integral_target)
        return child

    h_opt = initial_h()
    h_opt = repair_h(h_opt, integral_target)

    best_c5 = compute_overlap(best_h)

    for gen in range(gens):
