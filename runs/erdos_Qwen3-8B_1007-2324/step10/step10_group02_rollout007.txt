import numpy as np

def run(seed=42, budget_s=100, **kwargs):

    n_lead = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points // 2

    def repair_type1(values, target):
        vals = np.clip(values, 0, 1)
        total = np.sum(vals)
        if total == 0:
            return np.full_like(vals, target / len(vals))
        return vals * (target / total)

    def repair_type2(values, target):
        vals = np.clip(values, 0, 1)
        for _ in range(100):

            if abs(total - target) < 1e-5:

            vals += np.random.rand(len(vals)) * 0.01 * (target - total) / len(vals)
            vals = np.clip(vals, 0, 1)
        return vals

    def compute_c5(values):
        h_full = np.concatenate([values, values[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
