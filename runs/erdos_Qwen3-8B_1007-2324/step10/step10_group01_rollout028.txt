import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points // 2

    # Initialize population
    if 'initial_h_values' in globals():

        h_opt = h

    # Repair function
    def repair_h(h, target):
        n = len(h)
        sorted_idx = np.argsort(np.abs(2 * np.arange(n) - (n - 1)))

        weights_norm = weights / np.sum(weights)

        h_corrected = np.zeros_like(h)
        for i in range(n):
            pos = n - 1 - np.abs(n - 1 - sorted_idx[i])
            h_corrected[i] = sorted_vals[min(max(pos, 0), n - 1)]
        h_clipped = np.clip(h_corrected, 0, 1)
        if np.sum(h_clipped) > 0:
            h_scaled = h_clipped * (target / np.sum(h_clipped))
        else:
            h_scaled = np.zeros_like(h_clipped)
        return np.clip(h_scaled, 0, 1)

    def compute_overlap(h_opt):

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population_size = 50

    for _ in range(population_size - 1):

        candidate = np.clip(h_opt + mutation, 0, 1)

        population.append(candidate)

    best_h = h_opt.copy()
