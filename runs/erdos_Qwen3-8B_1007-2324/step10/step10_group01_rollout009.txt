import numpy as np
from scipy.optimize import minimize
import time

def repair_h(h, integral_target=70, n=140):
    sorted_idx = np.argsort(np.abs(2 * np.arange(n) - (n-1)))

    sorted_vals = np.sort(h)
    h_corrected = np.zeros_like(h)
    for i in range(n):
        pos = n - 1 - np.abs(n - 1 - sorted_idx[i])
        h_corrected[sorted_idx[i]] = sorted_vals[min(max(int(pos), 0), n-1)]
    h_clipped = np.clip(h_corrected, 0, 1)
    if np.sum(h_clipped) == 0:
        return np.zeros_like(h_clipped)
    scale = integral_target / np.sum(h_clipped)
    return np.clip(scale * h_clipped, 0, 1)

def compute_overlap(h_flat):
    h_full = np.concatenate([h_flat, h_flat[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * (2.0/280)
