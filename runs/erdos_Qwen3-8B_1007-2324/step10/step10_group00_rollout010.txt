import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 2 * dx  # 1.0
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.zeros(n_points)

    # Initialize symmetric start
    if 'initial_h_values' not in globals():
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        initial_h = np.concatenate([h[:n_half], h[:n_half][::-1]])
    h_opt = initial_h[:n_half].copy()

    def objective(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = integral_target / s
        return h * scale

    h_opt = repair(h_opt)
    best_h = h_opt.copy()
    best_c5 = objective(h_opt)
    current_h = h_opt.copy()
    current_c5 = best_c5
    T = 0.1
    cooling_rate = 0.995
    time_start = time.time()
