import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start = time.time()
    n_points = 160
    n_half = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2 * dx
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        curr = h[:n_half].copy()
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        curr = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]
    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (target_sum / s)
    def obj(h):
        full = np.concatenate([h, h[::-1]])
        corr = np.correlate(full, 1 - full, mode='full')
        return np.max(corr) * dx
    curr = repair(curr)
    best = curr.copy()
    best_c = obj(curr)
    T = 1.0
    cool = 0.998
    while time.time() - start < budget_s:
        noise = np.random.normal(0, 0.01, n_half)
        new = curr + noise
        new = repair(new)
        new_c = obj(new)
        if new_c < best_c or np.random.rand() < np.exp((new_c - best_c)/T):

            if new_c < best_c:
                best, best_c = new, new_c
        T *= cool
    best = repair(best)
    full_h = np.concatenate([best, best[::-1]])
