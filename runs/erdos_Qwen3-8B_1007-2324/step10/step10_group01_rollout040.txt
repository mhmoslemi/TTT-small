    best_fitness = np.inf
    best_h = population[0]
    for h in population:
        fitness = compute_overlap(h)
        if fitness < best_fitness:
            best_fitness = fitness
            best_h = h.copy()

    max_generations = 100
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        fitnesses = [compute_overlap(h) for h in population]
        sorted_indices = np.argsort(fitnesses)
        selected_indices = sorted_indices[:int(population_size * 0.2)]
        parents = [population[i] for i in selected_indices]
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = (parent1 + parent2) / 2
            noise = np.random.normal(0, 0.01, n_half)
            child += noise
            child = np.clip(child, 0, 1)
            child = repair_h(child, integral_target)
            new_population.append(child)
        population = new_population
        for h in population:
            fitness = compute_overlap(h)
            if fitness < best_fitness:
                best_fitness = fitness
                best_h = h.copy()

    h_full = np.concatenate([best_h, best_h[::-1]])
    elapsed = time.time() - start_time
    return h_full, best_fitness, n_points
