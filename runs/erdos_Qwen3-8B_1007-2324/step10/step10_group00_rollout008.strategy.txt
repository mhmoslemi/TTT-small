Its primary limitations are the likelihood of converging to local optima rather than the global minimum, and the computational expense derived from the high-dimensional space of 140-dimensional vectors. def create_numba_objective(n_half, dx):
    """Numba-optimized objective function"""
    import numpy as np
    from numba import njit
    @njit(cache=True)
    def obj(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        outlier = 1 - h_full
        corr = np.correlate(h_full, outlier, mode="full")
        return np.max(corr) * dx  # Already scaled
    return obj

def create_numba_objective_any(n_points, dx):
    """Alternative objective if input representation differs"""
    import numpy as np

        outlier = 1 - h_full
        corr = np.correlate(h_full, outlier, mode="full")
        return np.max(corr) * dx

def repair_solution(h, n_half, desired_sum=70.0):
    """Clip to [0,1] and adjust sum"""
