import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.uniform(0, 1, n_half)
        h_opt = repair(h_opt, integral_target)

    def repair(h, integral_target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    population_size = 50
    population = [h_opt] + [np.random.uniform(0, 1, n_half) for _ in range(population_size - 1)]
    for i in range(population_size):
        population[i] = repair(population[i], integral_target)

    start_time = time.time()
    for gen in range(300):
        if time.time() - start_time > budget_s - 1:
            break
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population, size=5)
            best = tournament[np.argmin([compute_overlap(ind) for ind in tournament])]
            parents.append(best)
        offspring = []
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i + 1]
            alpha = np.random.uniform(0, 1)
            child1 = alpha * p1 + (1 - alpha) * p2
            child2 = (1 - alpha) * p1 + alpha * p2
            for child in [child1, child2]:
                noise = np.random.normal(0, 0.01, len(child))
                child += noise
                child = np.clip(child, 0, 1)
                offspring.append(repair(child, integral_target))
        new_pop = population + offspring
        new_pop.sort(key=lambda x: compute_overlap(x))
        population = new_pop[:population_size]

    best = population[0]
    bounds = [(0, 1) for _ in range(n_half)]
    def objective(x):
        x = x.reshape((n_half,))
        h_full = np.concatenate([x, x[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    result = minimize(objective, best, method='L-BFGS-B', bounds=bounds, tol=1e-6)
    best = repair(result.x, integral_target)
    h_full = np.concatenate([best, best[::-1]])
    c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
    return np.concatenate([best, best[::-1]]), c5, n_points
