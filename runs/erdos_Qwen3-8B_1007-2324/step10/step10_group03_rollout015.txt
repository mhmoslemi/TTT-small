def run(seed=42, budget_s=100, **kwargs):

        h_opt[n_points//2:] = 0.0
    
    # Define repair function
    def repair(h, integral_target, seed):
        np.random.seed(seed)
        weights = np.abs(np.sin(np.pi * np.arange(len(h)) / (n_points//2)))
        ordered = np.argsort(weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[ordered] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)
    
    # Compute correlation
    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    # Main optimization
    start_time = time.time()
    T = 0.5
    cooling_rate = 0.998
    current_h = h_opt.copy()
    best_h = current_h.copy()
    best_c5 = compute_overlap(current_h)
    
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb

        perturbed = current_h + noise
