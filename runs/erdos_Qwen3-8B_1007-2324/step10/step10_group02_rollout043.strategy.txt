                candidate_refined, candidate_fitness = run_local_optimization(c, budget_s - spent)
                candidates.append(candidate_refined)
                best_c5 = min(best_c5, candidate_fitness)
            else:
                break
        
        # Prune and prepare final result
        best_solutions = prune_candidates(candidates, max_keep=5)
        final_candidate = best_solutions[0]
        break
    
    # Prepare outputs
    dx = 2.0 / n_points
    dx_times_c5 = best_c5 * (2/n_points)
    h_values = repair(final_candidate)
    c5_bound = dx_times_c5
    n_points = 280
    
    # Small compound validation
    assert np.all(h_values >= 0) and np.all(h_values <= 1)
    assert np.isclose(np.sum(h_values), 140, rtol=1e-4)
    try:
        compute_c5(h_values)
    except:
        pass
    
    # Time tracking cleanup
    total_time = time.time() - start_time
    if total_time > budget_s:
        print(f"WARNING: RAN {total_time:.2f} SECONDS (> {budget_s}s LIMIT)")
        print("Returning last found candidate.")
    
    return h_values, c5_bound, n_points
