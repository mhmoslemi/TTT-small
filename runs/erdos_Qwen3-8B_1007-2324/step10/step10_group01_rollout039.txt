    x0 = initial_h_val.copy() if 'initial_h_values' in globals() else np.random.rand(n_half)

    result = minimize(objective, x0, method='COBYLA', bounds=bounds, options={'disp': False, 'maxiter': 2000})

    best_h = repair_h(result.x, integral_target, n_half)
    best_c5 = compute_overlap(best_h)

    elapsed = time.time() - start_time
    return best_h, best_c5, n_half
