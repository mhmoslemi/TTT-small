        return np.max(corr) * dx

    def objective(h_vec):
        return compute_c5(h_vec)

    def constraint(h_vec):
        return np.sum(h_vec) - target_sum

    def repair(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if abs(total) < 1e-7:
            return np.ones_like(h_clipped) * (target_sum / len(h_clipped))
        factor = target_sum / total
        return h_clipped * factor

    def optimize():
        h = h_opt.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        T = 1.0
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            noise = np.random.normal(0, 0.03 * T, n_lead)
            new_h = h + noise
            new_h = repair(new_h)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
                h = new_h
                best_c5 = new_c5
                best_h = h.copy()
            T *= 0.998
        return best_h, best_c5

    h_opt, best_c5 = optimize()
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = repair(h_opt)
    elapsed = time.time() - start_time
    return np.concatenate([h_opt, h_opt[::-1]]), best_c5, n_points
