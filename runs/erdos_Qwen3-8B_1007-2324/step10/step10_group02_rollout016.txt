import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280

    population_size = 50

        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    
    # Differential evolution parameters
    F = 0.8
    CR = 0.9
    
    while time.time() - start_time < budget_s - 1:

        for _ in range(population_size):
