    def objective(h_opt):
        h_opt = repair(h_opt)
        return compute_overlap(h_opt)
    
    start_time = time.time()
    if 'initial_h_values' in globals():
        x0 = h_initial[:n_half]
    else:
        x0 = np.random.rand(n_half)
    
    result = differential_evolution(objective, bounds, x0=x0, strategy='best1bin', 
                                   maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), 
                                   recombination=0.9, disp=False)
    
    best_h_opt = result.x
    best_c5 = compute_overlap(best_h_opt)
    
    def objective_local(x):
        return compute_overlap(x)
    result_local = minimize(objective_local, best_h_opt, method='L-BFGS-B', bounds=bounds)
    best_h_opt = result_local.x
    best_c5 = compute_overlap(best_h_opt)
    
    h_full = np.concatenate([best_h_opt, best_h_opt[::-1]])
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx
    
    return np.concatenate([best_h_opt, best_h_opt[::-1]]), final_c5, n_points
