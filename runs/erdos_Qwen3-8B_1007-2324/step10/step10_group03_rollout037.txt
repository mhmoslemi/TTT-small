def run(seed=42, budget_s=100, **kwargs):

        h_opt = h[:n_points // 2]
    
    def repair(h, integral_target):
        n = len(h)
        target_sum = integral_target
        current_sum = np.sum(h)
        if np.isclose(current_sum, target_sum):
            return h
        delta = (target_sum - current_sum) / n
        h += delta
        h = np.clip(h, 0, 1)
        return h
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    T = 0.2
    cooling_rate = 0.998
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
