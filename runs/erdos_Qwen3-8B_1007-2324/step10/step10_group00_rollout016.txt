    def objective(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])

        return np.max(corr) * dx

    def repair(h_vec):

        s = np.sum(h_vec)
        if s <= 1e-10:
            return np.full_like(h_vec, desired_sum / len(h_vec))
        return h_vec * (desired_sum / s)

    best_h = h_opt.copy()
    best_c5 = objective(best_h)
    current_h = h_opt.copy()
