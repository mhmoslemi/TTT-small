import time

def run(seed=42, budget_s=100, **kwargs):

        h_opt = h[:n_points//2]

    def repair(h, target):
        diff = np.sum(h) - target
        if abs(diff) < 1e-6:
            return h
        idx = np.argsort(np.abs(h - 0.5))
        h[idx[:int(abs(diff))]] += np.sign(diff) * 0.05
        h = np.clip(h, 0, 1)
        while np.abs(np.sum(h) - target) > 1e-6:
            idx = np.argsort(np.abs(h - 0.5))
            h[idx[0]] += np.sign(np.sum(h) - target) * 0.05
            h = np.clip(h, 0, 1)
        return h

    def compute_overlap(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    T = 1.0
    cooling = 0.998
