    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values[:n_points])
    else:
        h_opt = np.random.rand(n_points)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
    
    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(h_opt)
    best_h_opt = h_opt.copy()
    
    T_start = 0.5
    cooling_factor = 0.995
    max_iter = 600
    
    current_temperature = T_start
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.05 * current_temperature, n_points)
        new_h = h_opt + noise
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        new_h = new_h * (target_sum / new_h_sum)
        current_c5 = compute_c5(h_opt)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temperature):
            h_opt = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_opt = h_opt.copy()
        current_temperature *= cooling_factor
    
    best_h_opt = np.clip(best_h_opt, 0, 1)
    best_h_opt = best_h_opt * (target_sum / np.sum(best_h_opt))
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_opt, best_c5, n_points
    return best_h_opt, best_c5, n_points
