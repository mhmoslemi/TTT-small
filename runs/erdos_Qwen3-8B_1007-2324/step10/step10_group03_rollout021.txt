def run(seed=42, budget_s=100, **kwargs):

    import time

    def repair(h, integral_target):
        current_sum = np.sum(h)
        scaling_factor = integral_target / np.clip(current_sum, 1e-12, np.inf)
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    T = 0.1
    cooling_rate = 0.995
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = compute_overlap(current_h_opt)
    
    for _ in range(4000):
        if time.time() - start_time > budget_s - 1:
            break
