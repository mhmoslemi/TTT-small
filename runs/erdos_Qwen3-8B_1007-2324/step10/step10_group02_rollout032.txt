        s = np.sum(h_vec)
        if abs(s - target_sum) < 1e-5:
            return h_vec
        factor = target_sum / s
        return np.clip(h_vec * factor, 0, 1)

    best_c5 = compute_c5(h_opt)
    best_h = h_opt.copy()
    
    T_start = 0.5
