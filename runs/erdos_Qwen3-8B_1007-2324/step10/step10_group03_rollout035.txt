def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time

    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points // 2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_points // 2]

    def repair(h, integral_target):
        current_sum = np.sum(h) * dx
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    start_time = time.time()
    T = 0.1
    cooling_rate = 0.997
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = compute_overlap(current_h_opt)

    for _ in range(7000):
        if time.time() - start_time > budget_s - 1.0:
            break
        noise = np.random.normal(0, 0.005, n_points // 2)
        perturbed = current_h_opt + noise
        perturbed = np.clip(perturbed, 0, 1)
        repaired = repair(perturbed, integral_target)
        candidate_c5 = compute_overlap(repaired)

        if candidate_c5 < best_c5 or np.random.rand() < np.exp((candidate_c5 - best_c5) / T):

    elapsed = time.time() - start_time
