import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    n_lead = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4
    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values[:n_lead])
    else:
        h_opt = np.random.rand(n_lead)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
    
    def compute_c5(h_opt):
        h_clip = np.clip(h_opt, 0, 1)
        sum_clip = np.sum(h_clip)
        h_scaled = h_clip * (target_sum / sum_clip)
        h_full = np.concatenate([h_scaled, h_scaled[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(h_opt)
    best_h_opt = h_opt.copy()
    T_start = 0.5
    cooling_factor = 0.995
    max_iter = 800
    
    current_temperature = T_start
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        noise = np.random.normal(0, 0.01 * current_temperature, n_lead)
        new_h = h_opt + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            new_h = np.ones(n_lead) * (target_sum / n_lead)
        else:
            new_h = new_h * (target_sum / sum_new)
        current_c5 = compute_c5(h_opt)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temperature):
            h_opt = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_opt = h_opt.copy()
        current_temperature *= cooling_factor
    
    best_h_opt = np.clip(best_h_opt, 0, 1)
    best_h_opt = best_h_opt * (target_sum / np.sum(best_h_opt))
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h_opt, best_h_opt[::-1]]), best_c5, n_points
