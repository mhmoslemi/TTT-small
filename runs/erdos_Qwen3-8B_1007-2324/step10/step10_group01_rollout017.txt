import time

def run(seed=42, budget_s=100, **kwargs):

    def compute_overlap(h):
        h_full = np.zeros(n_points)
        h_full[:len(h)] = h
        h_full[len(h):] = h[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr * dx)
        return max_corr

    def repair_h(h, target):
        excess = np.sum(h) - target
        if abs(excess) < 1e-6:
            return h
        diff = np.zeros_like(h)
        if excess > 0:
            diff = np.clip(h, 0, 1) - 1
        else:
            diff = np.clip(h, 0, 1)
        scale_factor = target / (np.sum(h) + 1e-12)
        return np.clip(h + diff * (scale_factor - 1), 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 0.2
    cooling_rate = 0.997
    iter_count = 0
