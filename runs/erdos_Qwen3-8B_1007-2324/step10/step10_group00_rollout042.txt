        h_opt = np.concatenate([h[:n_points // 2], h[:n_points // 2][::-1]])[:n_points // 2]
    
    def repair_solution(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        current_sum = np.sum(h_vec)
        if abs(current_sum) < 1e-10:
            return np.ones_like(h_vec) * (desired_sum / len(h_vec))
        return h_vec * (desired_sum / current_sum)
    
    def objective(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def get_max_k(h_vec):
        h_full = np.concatenate([h_vec, h_vec[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.argmax(corr)
    
    best_h = h_opt.copy()
    best_c5 = objective(best_h)
    current_h = h_opt.copy()
    current_c5 = best_c5
    T = 0.15
    cooling_rate = 0.99
    noise_scale = 0.05
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < 10000:
        iterations += 1
        noise = np.random.normal(0, noise_scale, len(current_h))
        new_h = current_h + noise
        new_h = repair_solution(new_h)
        new_c5 = objective(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h, current_c5 = new_h, new_c5
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
        T *= cooling_rate
        noise_scale *= 0.995
    
    best_h = repair_solution(best_h)
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
    return np.concatenate([best_h, best_h[::-1]]), best_c5, n_points
