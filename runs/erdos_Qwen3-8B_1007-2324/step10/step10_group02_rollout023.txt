            h_opt = h_opt * (target_sum / total)
        else:
            h_opt = np.zeros_like(h_opt) + 0.5

    def repair(candidate):
        clipped = np.clip(candidate, 0, 1)
        total = np.sum(clipped)
        if total > 0:
            return clipped * (target_sum / total)
        return np.zeros_like(candidate)

    def compute_c5(h_opt):

        corr = np.correlate(h_full, 1 - h_full, mode='full')

    best_c5 = compute_c5(h_opt)
    best_h = h_opt.copy()

    temp_start = 1.0
