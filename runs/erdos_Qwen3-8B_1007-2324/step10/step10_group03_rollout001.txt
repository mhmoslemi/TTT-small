def run(seed=42, budget_s=100, **kwargs):

        h /= np.sum(h) * dx
    
    def repair(h, target):
        if np.allclose(np.sum(h) * dx, target, atol=1e-5):
            return h
        delta = (target - np.sum(h) * dx) / (n_points * dx)
        h += delta
        h = np.clip(h, 0, 1)
        return h
    
    def evaluate(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    best_score = best_c5
    
    start_time = time.time()
    
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:

        new_h = np.clip(h + pert, 0, 1)
        new_h = repair(new_h, integral_target)
        current_c5 = evaluate(new_h)
