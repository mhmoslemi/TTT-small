    def objective(h_vec):

        return np.max(corr) * dx

    def repair(h_vec):

        s = np.sum(h_vec)
        if abs(s) < 1e-10:

        return h_vec * (desired_sum / s)

    h_curr = repair(h_opt.copy())
    best_h = h_curr.copy()
