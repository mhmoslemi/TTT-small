import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution
    
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_points//2]
    else:
        h_opt = np.random.uniform(0, 1, n_points//2)
        h_opt = repair(h_opt, integral_target)
    
    def repair(h, target):
        n = len(h)
        weights = 1.0 / (np.abs(np.arange(n) - n//2) + 1e-6)
        sorted_idx = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_idx] = sorted_vals
        scale = target / np.sum(new_h)
        return np.clip(new_h * scale, 0, 1)
    
    def objective(h):
        h_repaired = repair(h, integral_target)
        h_full = np.concatenate([h_repaired, h_repaired[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(len(h_opt))]
    
    start = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=15, maxiter=100, mutation=(0.5, 1),
                                   recombination=0.9, tol=1e-6, seed=seed)
    elapsed = time.time() - start
    
    best_h = repair(result.x, integral_target)
    best_c5 = objective(best_h)
    
    h_full = np.concatenate([best_h, best_h[::-1]])
    return h_full, best_c5, n_points
