import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    import time
    
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    integral_target = n_points / 4
    
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = h[:n_half]
    
    def repair(h, integral_target):
        n = len(h)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / sum_h
        return np.clip(h * scaling_factor, 0, 1)
    
    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    T = 0.5
    cooling_rate = 0.998
    current_h_opt = h_opt.copy()
    best_h_opt = current_h_opt.copy()
    best_c5 = compute_overlap(current_h_opt)
    
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break

        perturbed = current_h_opt.copy()
