import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    integral_target = n_points // 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h_opt = initial_h[:n_points // 2]
    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = np.concatenate([h[:n_points//2], h[:n_points//2][::-1]])[:n_points//2]

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, integral_target):
        n = len(h)
        h_clipped = np.clip(h, 0, 1)
        distances = np.abs(2 * np.arange(n) - (n - 1))
        sorted_idx = np.argsort(distances)
        weights = distances + 1
        weights_norm = weights / np.sum(weights)
        sorted_vals = np.sort(h_clipped)
        h_corrected = np.zeros_like(h)
        for i in range(n):
            pos = n - 1 - np.abs(n - 1 - sorted_idx[i])
            h_corrected[i] = sorted_vals[min(max(int(pos), 0), n-1)]
        h_clipped2 = np.clip(h_corrected, 0, 1)
        sum_h = np.sum(h_clipped2)
        if sum_h == 0:
            return np.zeros_like(h_corrected)
        scale = integral_target / sum_h
        return np.clip(h_corrected * scale, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 0.1
    cooling_rate = 0.994
    iter_count = 0
    max_iters = 3000

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        noise = np.random.normal(0, 0.02, len(best_h))
        candidate = np.clip(best_h + noise, 0, 1)
        candidate = repair_h(candidate, integral_target)
        candidate_c5 = compute_overlap(candidate)
        if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / current_temp):
            best_h = candidate.copy()
            best_c5 = candidate_c5
        current_temp *= cooling_rate

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
    elapsed = time.time() - start_time
    return h_full, best_c5, n_points
