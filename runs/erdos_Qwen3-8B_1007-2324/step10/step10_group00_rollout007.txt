import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    n_half = n_points // 2
    dx = 2.0 / n_points
    desired_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)[:n_half]
    else:
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sinc((x - 1) * 2))
        h_start = h_start[:n_half]
    
    def repair(h):
        h_clip = np.clip(h, 0, 1)
        if np.abs(np.sum(h_clip) - desired_sum) < 1e-5:
            return h_clip
        scale = desired_sum / np.sum(h_clip)
        return h_clip * scale
    
    def objective(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_h = repair(h_start)
    best_c5 = objective(best_h)
    
    def finite_diff_grad(h):
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = repair(h_minus)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    max_iter = 500
    step_size = 0.1
    noise_scale = 0.1
    early_stop = 0
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        grad = finite_diff_grad(best_h)
        h_new = best_h - step_size * grad
        h_new = repair(h_new)
        h_new += np.random.normal(0, noise_scale, n_half)
        h_new = repair(h_new)
        new_c5 = objective(h_new)
        if new_c5 < best_c5:
            best_h = h_new

            early_stop = 0
        else:

        if early_stop > 100:
            break
    
    best_h = repair(best_h)
    final_h = np.concatenate([best_h, best_h[::-1]])
    final_c5 = objective(best_h)
