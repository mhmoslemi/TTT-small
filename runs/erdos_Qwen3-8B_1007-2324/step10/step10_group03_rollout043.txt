    dx = 2.0 / n_points
    integral_target = n_points / 4

    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_opt = h_initial[:n_half]
    else:
        h_opt = np.random.rand(n_half)
        h_opt = repair(h_opt, integral_target)

    def repair(h, integral_target):
        n = len(h)
        center = n // 2
        weights = 1.0 / (np.abs(np.arange(n) - center) + 1e-6)
        sorted_indices = np.argsort(-weights)
        sorted_vals = np.sort(h)[::-1]
        new_h = np.zeros_like(h)
        new_h[sorted_indices] = sorted_vals
        current_sum = np.sum(new_h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling_factor = integral_target / current_sum
        new_h *= scaling_factor
        return np.clip(new_h, 0, 1)

    def compute_overlap(h_opt):
        h_full = np.concatenate([h_opt, h_opt[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = [h_opt.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_half)
        h = repair(h, integral_target)
        population.append(h)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(h_opt)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        fitness = [compute_overlap(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:20]]
        
        new_population = []
        for i in range(20):
            p1 = parents[i % len(parents)]
            p2 = parents[(i+1) % len(parents)]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            new_population.append(child)
        
        for i in range(len(new_population)):
            child = new_population[i]
            noise = 0.05 * normal(size=n_half)
            child += noise
            child = np.clip(child, 0, 1)
            child = repair(child, integral_target)
            new_population[i] = child
        
        new_population.extend(population)
        fitness_all = [compute_overlap(ind) for ind in new_population]
        sorted_indices = np.argsort(fitness_all)
        population = [new_population[i] for i in sorted_indices[:population_size]]
        
        current_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx].copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        def objective(x):
            h_opt = x.reshape((n_half, ))
            h_full = np.concatenate([h_opt, h_opt[::-1]])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        result = minimize(objective, best_h, bounds=[(0,1)]*n_half, method='Nelder-Mead')
        if result.success:
            best_h = result.x

    final_h = np.concatenate([best_h, best_h[::-1]])
    return final_h, best_c5, n_points
