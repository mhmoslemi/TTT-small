import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 140  # Halved for faster optimization
    dx = 2.0 / n_points
    integral_target = n_points / 4  # Ensures ∫h = 1

    if 'initial_h_values' in globals():
        h_opt = np.array(initial_h_values)[:n_points//2]
    else:
        # Construct a piecewise linear initial guess with 7 intervals
        h_opt = np.zeros(n_points//2)
        for i in range(1, 7):
            h_opt[20*i-10] = 1.0  # Place 1s spaced evenly
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (integral_target / np.sum(h_opt))

    def objective(h):
        h_full = np.concatenate([h, h[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h = h_opt.copy()
    best_c5 = objective(best_h)
    current_h = h_opt.copy()
    current_c5 = best_c5
    T = 0.5  # Higher initial temperature for broader exploration
    cooling_rate = 0.998
    noise_std = 0.05  # Larger initial noise for exploration

    while time.time() - start_time < budget_s - 1:
        noise = np.random.normal(0, noise_std, n_points//2)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h = new_h * (integral_target / sum_new)
        new_c5 = objective(new_h)
        delta = new_c5 - current_c5
        if delta < 0:
            if np.random.rand() < np.exp(-delta / T):
                current_h, current_c5 = new_h, new_c5
        else:
            current_h, current_c5 = new_h, new_c5
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
        T *= cooling_rate
        noise_std *= 0.98  # Reduce noise magnitude over time

    best_h = np.clip(best_h, 0, 1)
