import time

    def repair_h(h, target):
        n = len(h)
        sorted_idx = np.argsort(np.abs(2 * np.arange(n) - (n - 1)))
        weights = np.abs(2 * np.arange(n) - (n - 1)) + 1
        weights_norm = weights / np.sum(weights)
        sorted_vals = np.sort(h)
        h_corrected = np.zeros_like(h)
        for i in range(n):
            pos = n - 1 - np.abs(n - 1 - sorted_idx[i])
            h_corrected[i] = sorted_vals[min(max(int(pos), 0), n-1)]
        h_clipped = np.clip(h_corrected, 0, 1)
        scale = target / (np.sum(h_clipped) if np.sum(h_clipped) > 0 else 1)
        return np.clip(h_clipped * scale, 0, 1)

    def compute_overlap(h):
        h_full = np.clip(np.concatenate([h, h[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def crossover(parent1, parent2):
        child1 = parent1.copy()
        child2 = parent2.copy()
        idx = np.random.randint(3, size=len(parent1))
        for i in range(len(parent1)):
            blend = 0.5 * parent1[i] + 0.5 * parent2[i]
            if idx[i] > 0:
                child1[i] = parent2[i]
                child2[i] = parent1[i]
            else:
                child1[i] = blend
                child2[i] = blend
        return np.clip(child1, 0, 1), np.clip(child2, 0, 1)

        return child

    def tournament_selection(pop, k=5):
