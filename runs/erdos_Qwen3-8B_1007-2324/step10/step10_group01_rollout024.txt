import time

def run(seed=42, budget_s=100, **kwargs):

    else:
        x = np.linspace(0, 2, n_points)
        h = np.clip(1 - np.abs(x - 1), 0, 1)
        h_opt = np.concatenate([h[:n_half], h[:n_half][::-1]])[:n_half]

    def compute_overlap(h_opt):
        h_full = np.clip(np.concatenate([h_opt, h_opt[::-1]]), 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_h(h, integral_target):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = integral_target / current_sum
        return np.clip(h * scaling, 0, 1)

    best_h = h_opt.copy()
    best_c5 = compute_overlap(best_h)
    current_temp = 0.5
    cooling_rate = 0.995
    iter_count = 0
    max_iters = 5000

    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:

        noise_std = 0.025 * current_temp
        noise = np.random.normal(0, noise_std, n_half)

            best_h = candidate

        current_temp *= cooling_rate

    h_full = np.clip(np.concatenate([best_h, best_h[::-1]]), 0, 1)
