import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    target_sum = n_points // 2  # 140
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h_part = np.array(initial_h_values[:n_points//2], dtype=float)
    else:
        h_part = np.random.rand(n_points//2)
        h_part = h_part * (target_sum / np.sum(h_part))
    
    def repair(h_part):
        s = np.sum(h_part)
        h_part = h_part * (target_sum / s)
        h_part = np.clip(h_part, 0, 1)
        s_clipped = np.sum(h_part)
        h_part = h_part * (target_sum / s_clipped)
        return h_part

    def compute_c5(h_part):
        h_full = np.concatenate([h_part, h_part[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_h_part = h_part.copy()
    best_c5 = compute_c5(best_h_part)
    
    # PSO parameters
    n_particles = 10
    particles = [np.random.rand(n_points//2) * (target_sum / np.sum(np.random.rand(n_points//2))) for _ in range(n_particles)]
    velocities = [np.random.randn(n_points//2) * 0.01 for _ in range(n_particles)]
    personal_best = particles[:]
    personal_best_c5 = [compute_c5(p) for p in particles]
    global_best = min(personal_best, key=compute_c5)
    global_best_c5 = compute_c5(global_best)
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] += 0.5 * r1 * (personal_best[i] - particles[i]) + 0.5 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            # Clip and repair
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = repair(particles[i])
            current_c5 = compute_c5(particles[i])
            # Update personal best
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = current_c5
            # Update global best
            if current_c5 < global_best_c5:
                global_best = particles[i].copy()
                global_best_c5 = current_c5
    
    best_h_part = repair(best_h_part)
    best_h_values = np.concatenate([best_h_part, best_h_part[::-1]])
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_values, global_best_c5, n_points
    return best_h_values, global_best_c5, n_points
