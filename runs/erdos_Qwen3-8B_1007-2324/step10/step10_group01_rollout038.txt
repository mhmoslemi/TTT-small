        h_full = np.concatenate([h_flat, h_flat[::-1]])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        return compute_overlap(repair_h(h))
    
    bounds = [(0.0, 1.0) for _ in range(n_half)]
    start_time = time.time()
    
    x0 = initial_h_val if 'initial_h_values' in globals() else np.random.rand(n_half)
    
    result = minimize(objective, x0, method='COBYLA', bounds=bounds, 
                      options={'maxiter': 1000, 'disp': False})
    
    best_h = repair_h(result.x)
    best_c5 = compute_overlap(best_h)
    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
