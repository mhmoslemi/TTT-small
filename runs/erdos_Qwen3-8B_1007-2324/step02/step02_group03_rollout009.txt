import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points_base = 100
    n_points_medium = 140
    dx_base = 2.0 / n_points_base
    dx_medium = 2.0 / n_points_medium

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points_base, 0.5)

    def repair(h, n):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target = n / 2
        if current_sum == 0:
            return np.zeros_like(h)
        h = h * (target / current_sum)
        h = np.clip(h, 0, 1)
        final_sum = np.sum(h)
        if final_sum == 0:
            return np.zeros_like(h)
        h = h * (target / final_sum)
        return h

    def objective(h, dx):
        h_repaired = repair(h, len(h))
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx

    def interpolate_grid(h_coarse, n_old, n_new):
        h_new = np.zeros(n_new)
        step_coarse = 2.0 / n_old
        step_new = 2.0 / n_new
        for i in range(n_new):
            x = i * step_new
            idx = int(x / step_coarse)
            if idx < n_old:
                h_new[i] = h_coarse[idx]
            else:
                h_new[i] = h_coarse[-1]
        return h_new

    bounds_base = [(0.0, 1.0) for _ in range(n_points_base)]
    bounds_medium = [(0.0, 1.0) for _ in range(n_points_medium)]

    if 'initial_h_values' in globals():
        best_h = initial_h
        best_c5 = objective(initial_h, dx_base)
    else:
        best_h = np.full(n_points_base, 0.5)
        best_c5 = objective(best_h, dx_base)

    # DE phase on base grid
    result_de = differential_evolution(
        fun=lambda h: objective(h, dx_base),
        bounds=bounds_base,
        strategy='best1bin',
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-8,
        maxiter=300,
        disp=False
    )
    h_de = result_de.x
    c5_de = result_de.fun
    if c5_de < best_c5:
        best_h = h_de
        best_c5 = c5_de

    # Interpolate to medium grid
    h_interpolated = interpolate_grid(best_h, n_points_base, n_points_medium)
    h_interpolated = repair(h_interpolated, n_points_medium)

    # SLSQP phase on medium grid
    def objective_medium(h):
        return objective(h, dx_medium)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_medium / 2}]
    result_slsqp = minimize(
        objective_medium,
        h_interpolated,
        method='SLSQP',
        bounds=bounds_medium,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    h_final = result_slsqp.x
    c5_final = result_slsqp.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h.tolist(), best_c5, n_points_base)

    return (h_final.tolist(), c5_final, n_points_medium)
