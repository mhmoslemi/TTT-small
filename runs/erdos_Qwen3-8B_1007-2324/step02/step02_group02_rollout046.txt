        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (len(h_initial)/2) / np.sum(h_perturb)
        candidates.append(h_perturb)
        h_inter = np.zeros_like(h_initial)
        h_inter[::2] = 1.0
        h_inter[1::2] = 0.0
        candidates.append(h_inter)
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand * (len(h_initial)/2) / np.sum(h_rand)
        candidates.append(h_rand)
    
    for h in candidates:
        n = len(h)
        if n > 1000:
            continue
        
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n)
        
        def grad(x):
            corr = correlate(x, 1 - x, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)
            grad_val = np.zeros_like(x)
            for i in range(n):
                if 0 <= i + k < n:
                    grad_val[i] += 1.0 - x[i + k]
                if 0 <= i - k < n:
                    grad_val[i] -= x[i - k]
            return grad_val * (2.0 / n)
        
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}
        bounds = [(0.0, 1.0) for _ in range(n)]
        
        result = minimize(objective, h, method='SLSQP', jac=grad, bounds=bounds, constraints=cons, options={'maxiter': 100})
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            n_points = n
    
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, len(best_h))
