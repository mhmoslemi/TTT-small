    else:
        initial_h = np.full(n_medium, 0.5)

    def repair(h, n_points):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if np.isclose(current_sum, 0):
            return np.zeros(n_points)
        scale = (n_points / 2) / current_sum
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        final_sum = np.sum(h_scaled)
        if np.isclose(final_sum, 0):
            return np.zeros(n_points)
        scale_final = (n_points / 2) / final_sum
        return h_scaled * scale_final

    def interpolate_medium_to_base(h_medium, n_medium, n_base):
        x_medium = np.linspace(0, 2, n_medium)
        x_base = np.linspace(0, 2, n_base)
        h_base = np.interp(x_base, x_medium, h_medium)
        return h_base

    def interpolate_base_to_medium(h_base, n_base, n_medium):
        x_base = np.linspace(0, 2, n_base)
        x_medium = np.linspace(0, 2, n_medium)
        h_medium = np.interp(x_medium, x_base, h_base)
        return h_medium

    def objective_de(h):
        h_repaired = repair(h, n_base)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * (2.0 / n_base)
        return c5

    def objective_slsqp(h):
        h_repaired = repair(h, n_medium)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * (2.0 / n_medium)
        return c5

    if 'initial_h_values' in globals():
        initial_h_base = interpolate_medium_to_base(initial_h, n_medium, n_base)
    else:
        initial_h_base = np.full(n_base, 0.5)

    bounds_base = [(0.0, 1.0) for _ in range(n_base)]

    result_base = differential_evolution(objective_de, bounds_base, popsize=50, tol=1e-6, maxiter=200, mutation=(0.5, 1), recombination=0.8)
    best_h_base = result_base.x

    best_h_medium = interpolate_base_to_medium(best_h_base, n_base, n_medium)

    def constraint_slsqp(h):
        return np.sum(h) - n_medium / 2

    cons = [{'type': 'eq', 'fun': constraint_slsqp}]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_medium)]

    result_slsqp = minimize(objective_slsqp, best_h_medium, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})

    elapsed = time.time() - start_time
    h_values = result_slsqp.x
    c5_bound = result_slsqp.fun

    if elapsed > budget_s:
        h_values = best_h_medium
        c5_bound = objective_slsqp(h_values)
        return h_values.tolist(), c5_bound, n_medium

    return h_values.tolist(), c5_bound, n_medium
