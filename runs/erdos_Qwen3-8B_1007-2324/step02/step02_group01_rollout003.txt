def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    # Initialize population
    def initial_population():
        if 'initial_h_values' in globals():
            base = initial_h_values.copy()
        else:
            base = np.full(n_points, 0.5)
        pop = []

            vec = base + 0.1 * np.random.randn(n_points)
            vec = np.clip(vec, 0, 1)
            vec = vec * (integral_target / np.sum(vec)) if np.sum(vec) > 0 else vec
            pop.append(vec)
        return np.array(pop)

    population = initial_population()

    def clip_and_scale(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return np.zeros_like(x)
        scale = integral_target / total
        return x * scale

    def evaluate(individual):
        h = individual.astype(float)
        h_clip = clip_and_scale(h)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def mutation(parent):
        mutant = parent + 0.1 * np.random.randn(n_points)
        mutant = clip_and_scale(mutant)
        return mutant

    def crossover(parent, mutant):
        mask = np.random.rand(n_points) < 0.9
        return np.where(mask, mutant, parent)

    best_solution = population[0]
    best_score = evaluate(best_solution)
    start_time = time.time()
