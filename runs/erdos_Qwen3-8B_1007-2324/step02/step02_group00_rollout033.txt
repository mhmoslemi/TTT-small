import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def constraint_func(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    # Improved optimization with tolerance and iterations
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,

    )

    if elapsed > budget_s:

    h_values = np.clip(h_values, 0, 1)
    h_values_sum = np.sum(h_values)
