import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    starts = []

    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        noise = np.random.normal(0, 0.25, n_points)
        h_perturbed = h_initial + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed > 1e-8:
            h_perturbed = h_perturbed * (integral_target / sum_perturbed)
        starts.append(h_perturbed)

    num_starts = 12
    for _ in range(num_starts - 1):
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        if sum_h > 1e-8:
            h = h * (integral_target / sum_h)
        h = np.clip(h, 0, 1)
        starts.append(h)

    best_h = None
    best_c5 = float('inf')

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_func(h):
        return np.sum(h) - integral_target

    constraints = [{'type': 'eq', 'fun': constraint_func}]

    for idx, start in enumerate(starts):
        result = minimize(
            objective,
            start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1200, 'ftol': 1e-9, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    elapsed = time.time() - start_time
    return best_h, best_c5, n_points
