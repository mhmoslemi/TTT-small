from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points // 2

    def initial_population():
        if 'initial_h_values' in globals():

            pop *= (integral_target / np.sum(pop))

    def fitness(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode="full")
        return -np.max(corr) * dx

        return np.sum(h) - integral_target

    bounds = [(0.0, 1.0) for _ in range(n_points)]
