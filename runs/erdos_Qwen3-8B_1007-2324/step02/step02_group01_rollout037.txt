```python
    if best_h is None:
        best_h = initial_h
        best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points
    
    return best_h, best_c5, n_points
```
