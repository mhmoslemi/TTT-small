    n_base = 100
    n_medium = 140
    dx_base = 2.0 / n_base
    dx_medium = 2.0 / n_medium
    
    if 'initial_h_values' in globals():
        h_initial_medium = np.array(initial_h_values)
    else:
        h_initial_medium = np.random.uniform(0, 1, n_medium)
    
    # Phase 1: DE on base grid
    def repair_phase1(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        target_sum = n_base / 2
        if current_sum == 0:
            return h
        scale_factor = target_sum / current_sum
        h_scaled = h * scale_factor
        h_final = np.clip(h_scaled, 0.0, 1.0)
        final_sum = np.sum(h_final)
        adjust_factor = target_sum / final_sum
        return h_final * adjust_factor
    
    def objective_phase1(h):
        h_repaired = repair_phase1(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx_base
    
    # Interpolate initial guess to base grid
    x_initial = np.linspace(0, 2, n_medium)
    x_base = np.linspace(0, 2, n_base)
    h_initial_base = np.interp(x_base, x_initial, h_initial_medium)
    
    result_phase1 = differential_evolution(
        objective_phase1,
        bounds=[(0., 1.)] * n_base,
        strategy='best1bin',
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        maxiter=100,
        tol=1e-6,
        seed=seed
    )
    
    h_best_base = result_phase1.x
    
    # Interpolate to medium grid
    x_base = np.linspace(0, 2, n_base)
    x_medium = np.linspace(0, 2, n_medium)
    h_medium = np.interp(x_medium, x_base, h_best_base)
    
    # Phase 2: SLSQP optimization on medium grid
    def repair_phase2(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        target_sum = n_medium / 2
        if current_sum == 0:
            return h
        scale_factor = target_sum / current_sum
        h_scaled = h * scale_factor
        h_final = np.clip(h_scaled, 0.0, 1.0)
        final_sum = np.sum(h_final)
        adjust_factor = target_sum / final_sum
        return h_final * adjust_factor
    
    def objective_phase2(h):
        h_repaired = repair_phase2(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx_medium
    
    def constraint_phase2(h):
        return np.sum(h) - n_medium / 2
    
    cons_phase2 = [{'type': 'eq', 'fun': constraint_phase2}]
    bounds_phase2 = [(0., 1.)] * n_medium
    
    result_phase2 = minimize(
        objective_phase2,
        h_medium,
        method='SLSQP',
        bounds=bounds_phase2,
        constraints=cons_phase2,
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        best_c5 = min(result_phase1.fun, result_phase2.fun)
        return (h_medium if best_c5 == result_phase2.fun else h_best_base, 
                best_c5, n_medium)
    
    return result_phase2.x, result_phase2.fun, n_medium
