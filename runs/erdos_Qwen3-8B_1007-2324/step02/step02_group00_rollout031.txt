import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values).copy()
        # Ensure length matches n_points
        if len(initial_h) < n_points:
            initial_h = np.concatenate([initial_h, initial_h[-2:]])
        elif len(initial_h) > n_points:
            initial_h = initial_h[:n_points]
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Clip and adjust sum
    initial_h = np.clip(initial_h, 0, 1)
    initial_h_sum = np.sum(initial_h)
    if abs(initial_h_sum - n_points / 2) > 1e-5:
        scale = (n_points / 2) / initial_h_sum
        initial_h = initial_h * scale
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    result = minimize(
        objective, 
        initial_h, 
        method='L-BFGS-B', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points
    
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values_sum = np.sum(h_values)
    if abs(h_values_sum - n_points / 2) > 1e-5:
        scale = (n_points / 2) / h_values_sum
        h_values = h_values * scale
        h_values = np.clip(h_values, 0, 1)
    
    c5_bound = result.fun
    return h_values, c5_bound, n_points
