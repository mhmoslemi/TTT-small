import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - integral_target
    
    # Start points
    starts = []
    if 'initial_h_values' in globals():
        initial_h = np.clip(np.array(initial_h_values), 0, 1)
        current_sum = np.sum(initial_h)
        gap = integral_target - current_sum
        if abs(gap) > 1e-6:
            indices = np.random.choice(n_points, 5, replace=False)
            initial_h[indices] += gap / n_points
            initial_h = np.clip(initial_h, 0, 1)
        starts.append(initial_h)
    
    for _ in range(15):
        start = np.zeros(n_points)
        indices = np.random.choice(n_points, integral_target, replace=False)
        start[indices] = 1.0
        starts.append(start)
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for start in starts:
        result = minimize(
            objective,
            start,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    if best_h.size == 0:
        best_h = np.zeros(n_points)
        indices = np.random.choice(n_points, integral_target, replace=False)
        best_h[indices] = 1.0
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
