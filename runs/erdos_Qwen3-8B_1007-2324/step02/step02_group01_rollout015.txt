import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    def obj(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    def cons_func(h):
        return np.sum(h) - integral_target

    starts = []
    # Start 1: parent with noise
    h = initial_h.copy()
    h += np.random.normal(0, 0.1, n_points)
    h = np.clip(h, 0, 1)
    h = h * (integral_target / np.sum(h))
    h = np.clip(h, 0, 1)
    starts.append(h)
    # Start 2: random
    h = np.random.uniform(0, 1, n_points)
    h = h * (integral_target / np.sum(h))
    h = np.clip(h, 0, 1)
    starts.append(h)
    # Start 3: alternating
    h = np.zeros(n_points)
    h[::2] = 0.2
    h[1::2] = 0.8
    h = h * (integral_target / np.sum(h))
    h = np.clip(h, 0, 1)
    starts.append(h)
    # Start 4: cosine
    h = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_points) / n_points)
    h = h * (integral_target / np.sum(h))
    h = np.clip(h, 0, 1)
    starts.append(h)
    # Start 5: triangular
    h = np.kaiser(n_points, 10)
    h = h / h.max()
    h = h * (integral_target / np.sum(h))
    h = np.clip(h, 0, 1)
    starts.append(h)

    best_h = initial_h
    best_c5 = obj(initial_h)

    for start in starts:
        cons = [{'type': 'eq', 'fun': cons_func}]
        res = minimize(obj, start, bounds=[(0, 1)]*n_points, constraints=cons, method='SLSQP', options={'maxiter': 500})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points

    best_h_clip = np.clip(best_h, 0, 1)
    corr = np.correlate(best_h_clip, 1 - best_h_clip, mode='full')
    final_c5 = np.max(corr) * dx
    return best_h, final_c5, n_points
