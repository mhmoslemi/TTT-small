import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points

    def objective(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e7 * np.abs(np.sum(h) - n_points / 2)
        return max_corr + penalty

    bounds = [(0.0, 1.0)] * n_points

    def sum_constraint(x):
        return np.sum(x) - n_points / 2

    initial_h = kwargs.get("initial_h_values", None)
    if initial_h is not None:
        initial_h = np.array(initial_h[:n_points])
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        x0 = initial_h + np.random.normal(0, 0.05, n_points)
        x0 = np.clip(x0, 0, 1)
    else:
        x0 = np.random.rand(n_points)
        x0 = x0 * (n_points / 2) / np.sum(x0)

    res_de = differential_evolution(
        objective, bounds, max_evals=5000,
        strategy='best1bin', mutation=0.5,
        seed=seed, popsize=15, recombination=0.8
    )

    res_slsqp = minimize(
        lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
        res_de.x, method='SLSQP',
        bounds=bounds, constraints={'type': 'eq', 'fun': sum_constraint},
        options={'maxiter': 50, 'ftol': 1e-8}
    )

    final_h = np.clip(res_slsqp.x, 0, 1)
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx

    if time.time() - start_time > budget_s - 1:
        return (final_h.tolist(), final_c5, n_points)

    return (final_h.tolist(), final_c5, n_points)
