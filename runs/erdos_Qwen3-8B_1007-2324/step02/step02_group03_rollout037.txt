    h_slsqp = result_slsqp.x
    c5_slsqp = result_slsqp.fun

    best_c5 = min(c5_de, c5_slsqp)
    best_h = h_de if c5_de < c5_slsqp else h_slsqp
    best_n_points = n_points if c5_de < c5_slsqp else n_points_medium

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h.tolist(), objective(initial_h), n_points

    return best_h.tolist(), best_c5, best_n_points
