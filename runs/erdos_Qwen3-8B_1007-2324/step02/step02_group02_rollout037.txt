import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points_val = 140
    dx = 2.0 / n_points_val
    initial_h_values = kwargs.get('initial_h_values', None)
    
    def de_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = np.sum(x_clipped)
        if total == 0:
            return np.inf
        scale_factor = (n_points_val / 2) / total
        scaled_x = x_clipped * scale_factor
        corr = np.correlate(scaled_x, 1 - scaled_x, mode='full')
        return dx * np.max(corr)
    
    bounds = [(0.0, 1.0)] * n_points_val
    initial_point = None
    if initial_h_values is not None:
        x0 = np.clip(initial_h_values, 0, 1)
        x0_sum = np.sum(x0)
        scale_factor = (n_points_val / 2) / x0_sum
        initial_point = x0 * scale_factor
    
    res_de = None
    if time.time() - start_time < budget_s * 0.7:
        try:
            res_de = differential_evolution(de_objective, bounds, max_evals=5000, strategy='best1',
                                           mutation=0.5, seed=seed, x0=initial_point)
        except:
            res_de = differential_evolution(de_objective, bounds, max_evals=5000, strategy='best1',
                                           mutation=0.5, seed=seed)
    
    def slsqp_objective(x):
        x_clipped = np.clip(x, 0, 1)
        total = np.sum(x_clipped)
        scale_factor = (n_points_val / 2) / total
        scaled_x = x_clipped * scale_factor
        corr = np.correlate(scaled_x, 1 - scaled_x, mode='full')
        return dx * np.max(corr)
    
    def sum_constraint(x):
        x_clipped = np.clip(x, 0, 1)
        return np.sum(x_clipped) - (n_points_val / 2)
    
    res_slsqp = minimize(slsqp_objective, res_de.x if res_de else np.random.rand(n_points_val),
                         method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': sum_constraint},
                         options={'maxiter': 50})
    
    best_candidate = np.clip(res_slsqp.x, 0, 1)
    corr = np.correlate(best_candidate, 1 - best_candidate, mode='full')
    c5_value = dx * np.max(corr)
    return (best_candidate.tolist(), c5_value, n_points_val)
