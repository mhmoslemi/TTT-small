import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    # Initialize h using provided initial values or default
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)
    
    # Ensure correct length
    if len(initial_h) != n_points:
        initial_h = initial_h[:n_points]
    
    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - integral_target
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    start_time = time.time()
    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
    )
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
