        interleaved[1::2] = 0.0
        candidates.append(interleaved)

        n = 100
        h_rand = np.random.rand(n)

        candidates.append(h_rand)

    # Optimization loop
    for i, h in enumerate(candidates):
        n = len(h)
        if n > 1000:
            continue
        dx = 2.0 / n
        sum_h = np.sum(h)
        if abs(sum_h - n / 2) > 1e-5:
            h = h * (n / 2) / sum_h
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})

        def gradient(x):

            max_val = np.max(corr)

                if 0 <= i + lag / dx < n:
