from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 100  # Reduced grid size for faster convergence
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Default pattern: alternating 0.5 with a slight perturbation
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 0.5 + 0.1 * np.sin(2 * np.pi * i / n_points)
        initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_func}]

    result = minimize(
        objective, 
        initial_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
