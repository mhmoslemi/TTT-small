import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    required_sum = n_points // 2  # 70
    
    if 'initial_h_values' in globals():
        # Downsample parent solution (n=280) to n=140 by averaging pairs
        h_down = np.zeros(n_points)
        for i in range(n_points):
            idx = i * 2
            if idx < len(initial_h_values):
                h_down[i] = (initial_h_values[idx] + initial_h_values[idx+1]) / 2
            else:
                h_down[i] = initial_h_values[-1]
        x0 = h_down
    else:
        # Uniform initial guess
        x0 = np.full(n_points, required_sum / n_points)
    
    def project_solution(x):
        x_clip = np.clip(x, 0, 1)
        s = np.sum(x_clip)
        if abs(s - required_sum) < 1e-5:
            return x_clip
        if s == 0:
            return np.full_like(x_clip, required_sum / n_points)
        x_scaled = x_clip * (required_sum / s)
        x_final = np.clip(x_scaled, 0, 1)
        s_final = np.sum(x_final)
        if abs(s_final - required_sum) > 1e-5:
            scale_factor = required_sum / s_final
            x_final = np.clip(x_final * scale_factor, 0, 1)
        return x_final
    
    def objective(x):
        x_proj = project_solution(x)
        corr = np.correlate(x_proj, 1 - x_proj, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    popsize = min(40, int(budget_s))
    maxiter = int(budget_s * 2)
    
    result = differential_evolution(
        objective, bounds, 
        popsize=popsize, 
        maxiter=maxiter, 
        mutation=0.8, 
        recombination=0.7, 
        seed=seed, 
        tol=1e-6
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = project_solution(x0)
        c5_bound = objective(x0)
        return h_values, c5_bound, n_points
    
    best_h = result.x
    best_h = project_solution(best_h)
    c5_bound = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx
    
    return best_h, c5_bound, n_points
