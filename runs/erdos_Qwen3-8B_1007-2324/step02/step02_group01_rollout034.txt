from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.full(n_points, 0.5)

        return np.sum(h) - integral_target
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    start_time = time.time()
    result = minimize(
        objective,
