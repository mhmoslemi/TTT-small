import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    initial_h = np.full(n_points, 0.5)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)

    best_c5 = float('inf')
    best_h = initial_h.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]

    num_starts = 12
    for start_idx in range(num_starts):
        if start_idx == 0:
            h_start = initial_h.copy()
            noise = np.random.normal(0, 0.1, n_points)
            h_start += noise
            h_start = np.clip(h_start, 0, 1)
            sum_h = np.sum(h_start)
            if sum_h != 0:
                scale = integral_target / sum_h
                h_start *= scale
            else:
                h_start = np.full(n_points, integral_target / n_points)
        else:
            h_start = np.random.uniform(0, 1, n_points)
            sum_h = np.sum(h_start)
            if sum_h != 0:
                scale = integral_target / sum_h
                h_start *= scale
            else:
                h_start = np.full(n_points, integral_target / n_points)

        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        scale = integral_target / sum_h
        best_h *= scale
    else:
        best_h = np.full(n_points, integral_target / n_points)

    return best_h, best_c5, n_points
