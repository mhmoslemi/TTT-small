import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    n = n_points
    dx = 2.0 / n

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n, 0.5)

    def objective(h):
        h_fft = np.fft.fft(h, n=2*n - 1)
        g_fft = np.fft.fft(1 - h, n=2*n - 1)
        cross_corr_fft = h_fft * np.conj(g_fft)
        cross_corr = np.fft.ifft(cross_corr_fft).real
        max_corr = np.max(cross_corr)
        penalty = 1e5 * abs(np.sum(h) - n / 2)
        return max_corr * dx + penalty

    bounds = [(0.0, 1.0) for _ in range(n)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=100,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
