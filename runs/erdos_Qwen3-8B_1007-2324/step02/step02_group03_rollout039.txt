import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points_coarse = 100
    dx_coarse = 2.0 / n_points_coarse
    n_points_medium = 140
    dx_medium = 2.0 / n_points_medium

    if 'initial_h_values' in globals():
        x_medium = np.linspace(0, 2, len(initial_h_values))
        x_coarse = np.linspace(0, 2, n_points_coarse)
        initial_h = np.interp(x_coarse, x_medium, initial_h_values)
    else:
        initial_h = np.full(n_points_coarse, 0.5)

    def repair(h, n_points, dx):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum == 0:
            scale_factor = 1.0
        else:
            scale_factor = target_sum / current_sum
        h_scaled = h * scale_factor
        h_scaled = np.clip(h_scaled, 0, 1)
        final_sum = np.sum(h_scaled)
        adjust_factor = target_sum / final_sum if final_sum != 0 else 1.0
        return h_scaled * adjust_factor

    def objective_de(h):
        h_repaired = repair(h, n_points_coarse, dx_coarse)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx_coarse

    bounds_de = [(0.0, 1.0) for _ in range(n_points_coarse)]

    result_de = None
    try:
        result_de = differential_evolution(
            objective_de, bounds_de, strategy='best1bin',
            popsize=50, mutation=(0.5, 1), recombination=0.9,
            maxiter=100, tol=1e-6, disp=False, seed=seed
        )
    except:
        pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = initial_h
        c5_bound = objective_de(initial_h)
        return (h_values, c5_bound, n_points_coarse)

    def interpolate_grid(coarse, n_coarse, n_new):
        x_coarse = np.linspace(0, 2, n_coarse)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_coarse, coarse)
        return h_new

    h_medium = interpolate_grid(result_de.x, n_points_coarse, n_points_medium)

    def objective_slqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_medium

    def constraint_func(h):
        return np.sum(h) - n_points_medium / 2

    bounds_slqp = [(0.0, 1.0) for _ in range(n_points_medium)]
    constraints_slqp = [{'type': 'eq', 'fun': constraint_func}]

    result_slqp = None
    try:
        result_slqp = minimize(
            objective_slqp, h_medium, method='SLSQP',
            bounds=bounds_slqp, constraints=constraints_slqp,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
    except:
        result_slqp = minimize(objective_slqp, h_medium, method='L-BFGS-B',
                              bounds=bounds_slqp, options={'maxiter': 50})

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_medium, objective_slqp(h_medium), n_points_medium)

    h_values = result_slqp.x
    c5_bound = result_slqp.fun
    return (h_values, c5_bound, n_points_medium)
