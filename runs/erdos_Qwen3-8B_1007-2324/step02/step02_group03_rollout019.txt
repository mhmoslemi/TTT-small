import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    def repair(h, n):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros(n)
        scale = (n / 2) / s
        return h * scale

    def compute_c5(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def interpolate(prev_h, n):
        x = np.linspace(0, 2, len(prev_h))
        x_new = np.linspace(0, 2, n)
        return np.interp(x_new, x, prev_h)

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(140, 0.5)

    phases = [(50, 60), (100, 30), (200, 10)]

    for n_points, time_limit in phases:
        if time.time() - start_time > budget_s:
            break
        if n_points == 50:
            current_h = interpolate(initial_h, n_points)
        elif n_points == 100:
            current_h = interpolate(best_h, n_points) if best_h is not None else interpolate(initial_h, n_points)
        else:
            current_h = interpolate(best_h, n_points) if best_h is not None else interpolate(initial_h, n_points)
        current_h_repaired = repair(current_h, n_points)
        if n_points == 50 or n_points == 100:
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = differential_evolution(
                lambda x: compute_c5(repair(x, n_points), n_points),
                bounds,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.9,
                tol=1e-6,
                maxiter=100,
                polish=True,
                vectorized=True,
                disp=False
            )
            optimized_h = result.x
        else:
            def obj(x):
                return compute_c5(repair(x, n_points), n_points)
            def cons(x):
                return np.sum(repair(x, n_points)) - n_points / 2
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            res = minimize(obj, current_h_repaired, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'maxiter': 100, 'ftol': 1e-8, 'disp': False})
            optimized_h = res.x
        c5 = compute_c5(repair(optimized_h, n_points), n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = optimized_h
        if time.time() - start_time > budget_s:
            break

    best_h = repair(best_h, len(best_h)) if best_h is not None else initial_h
    c5_final = compute_c5(best_h, len(best_h))
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, compute_c5(initial_h, len(initial_h)), len(initial_h)
    return best_h, c5_final, len(best_h)
