from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

        max_corr = np.max(corr) * dx
        penalty = 1e6 * np.abs(np.sum(h) - n_points/2)
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:

    h_values = result.x
    # Ensure sum constraint is satisfied
    sum_h = np.sum(h_values)

    c5_bound = np.max(corr) * dx
