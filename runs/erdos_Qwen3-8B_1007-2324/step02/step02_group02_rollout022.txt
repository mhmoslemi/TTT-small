from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    candidates = []

    # Initialize from parent
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates.append(h_initial)
        # Upsample
        if len(h_initial) * 2 <= 1000:
            candidates.append(np.repeat(h_initial, 2))
        # Smooth
        smoothed = np.convolve(h_initial, np.ones(5)/5, mode='same')
        smoothed = np.clip(smoothed, 0, 1)
        candidates.append(smoothed)
        # Interleave
        interleaved = np.zeros_like(h_initial)
        interleaved[::2] = 1.0
        interleaved[1::2] = 0.0
        candidates.append(interleaved)
        # Random
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand / np.sum(h_rand) * (len(h_initial)/2)
        candidates.append(h_rand)

    for h_candidate in candidates:
        n_points = len(h_candidate)
        if n_points > 1000:
            continue
        def objective(x):
            h = np.clip(x, 0, 1)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            c5 = max_corr * dx
            penalty = 1000 * abs(np.sum(h) - n_points/2)
            return c5 + penalty
        bounds = [(0.0, 1.0)] * n_points
        try:
            result = differential_evolution(objective, bounds, strategy='best1bin',
                                           popsize=20, maxiter=100, tol=1e-6,
                                           mutation=(0.5, 1), recombination=0.9, seed=seed)
            if result.success:
                current_h = result.x
                h = np.clip(current_h, 0, 1)
                corr = np.correlate(h, 1 - h, mode='full')

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:

    if best_h is None and 'initial_h_values' in globals():
