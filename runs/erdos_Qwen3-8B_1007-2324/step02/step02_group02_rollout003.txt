from scipy.optimize import differential_evolution
import time

        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        penalty = 1000 * abs(sum_h - len(h)/2) / (len(h)/2) if len(h) > 0 else 0
        return compute_c5(h) + penalty
