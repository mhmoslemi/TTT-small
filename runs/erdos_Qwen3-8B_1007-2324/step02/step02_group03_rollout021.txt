            result = differential_evolution(
                obj,
                [(0.0, 1.0)] * n,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                tol=1e-8,
                maxiter=100,
                mutation_strength=0.8
            )
        else:
            bounds = [(0.0, 1.0) for _ in range(n)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}
            result = minimize(
                obj,
                current_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        current_h = result.x
        current_h_repaired = repair(current_h, n)
        current_c5 = obj(current_h_repaired)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h_repaired
            best_n = n
        previous_h = current_h_repaired

    return best_h, best_c5, best_n
