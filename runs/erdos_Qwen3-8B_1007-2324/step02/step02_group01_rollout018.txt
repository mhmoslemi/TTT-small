        x_scaled = x_clip * scale
        x_scaled = np.clip(x_scaled, 0, 1)

    def objective(x):
        h_clip = repair(x)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx
