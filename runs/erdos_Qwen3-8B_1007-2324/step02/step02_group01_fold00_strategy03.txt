def edge_repair(x):
    h_clip = np.clip(x, 0, 1)
    while not (np.all(h_clip >= 0) and np.all(h_clip <= 1)):
        h_clip = clipping_repair(x)  # Clip out-of-bounds values

def integral_repair(x, integral_target):
    total = np.sum(x)
    if abs(total - integral_target) < 1e-5:
        return x
    # Adjust proportionally (random binomial move)
    directions = np.random.choice([-1, 1], size=x.shape, p=[0.5, 0.5])
    correction = (integral_target - total) / max(1, total)
    x_adjusted = x + correction * directions
    x_adjusted = clipping_repair(x_adjusted)
    x_adjusted = integral_repair(x_adjusted, integral_target)  # Recursive call
    return x_adjusted
