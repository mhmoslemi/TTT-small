import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    np.random.seed(seed)
    
    num_starts = 12
    starts = []
    
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        noise = np.random.normal(0, 0.1, n_points)
        starts.append(h_initial + noise)
        for _ in range(num_starts - 1):
            starts.append(np.random.uniform(0, 1, n_points))
    else:
        for _ in range(num_starts):
            starts.append(np.random.uniform(0, 1, n_points))
    
    def project(h):
        h_clip = np.clip(h, 0, 1)
        s = np.sum(h_clip)
        if s == 0:
            return np.full(n_points, 0.5)
        return h_clip * (integral_target / s)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - integral_target}]
    
    best_h = None
    best_c5 = float('inf')
    
    for h_start in starts:
        h_proj = project(h_start)
        result = minimize(
            lambda h: np.max(np.correlate(np.clip(h, 0, 1), 1 - np.clip(h, 0, 1), mode='full')) * dx,
            h_proj,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s:

        if result.success and result.fun < best_c5:

            best_h = result.x

        best_c5 = np.max(np.correlate(np.clip(best_h, 0, 1), 1 - np.clip(best_h, 0, 1), mode='full')) * dx
    
    return best_h, best_c5, n_points
