import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    # Initial setup
    n_points_val = 140
    dx = 2.0 / n_points_val
    initial_h_list = kwargs.get("initial_h_values", None)

    # Normalize initial_h_values if provided
    if initial_h_list is not None:
        initial_h = np.array(initial_h_list)
        if len(initial_h) < n_points_val:
            initial_h = np.interp(np.linspace(0, 2, n_points_val), np.linspace(0, 2, len(initial_h)), initial_h)
        elif len(initial_h) > n_points_val:
            initial_h = initial_h[:n_points_val]
        initial_sum = np.sum(initial_h)
        scale_factor = (n_points_val / 2.0) / initial_sum
        initial_h = initial_h * scale_factor
        initial_h = np.clip(initial_h, 0, 1)
    else:
        initial_h = np.random.rand(n_points_val)
        initial_h = initial_h / np.sum(initial_h) * (n_points_val / 2.0)
        initial_h = np.clip(initial_h, 0, 1)

    # Objective for DE with penalty
    def objective(x):
        h_x = np.clip(x, 0, 1)
        corr = np.correlate(h_x, 1 - h_x, mode='full')
        max_corr = np.max(corr) * dx
        penalty = 1e7 * abs(np.sum(h_x) - n_points_val / 2.0)
        return max_corr + penalty

    bounds = [(0.0, 1.0)] * n_points_val

    # DE with fallback to L-BFGS-B if needed
    res_de = differential_evolution(objective, bounds, strategy='best1', mutation=0.5, popsize=15, maxiter=100, seed=seed)
    res_slsqp = minimize(
        lambda x: np.max(np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')) * dx,

        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_val / 2.0},
        options={'maxiter': 20}
    )

    if not res_slsqp.success:
        best_candidate = res_de.x
    else:
        best_candidate = res_slsqp.x

    best_candidate = np.clip(best_candidate, 0, 1)
    integral_val = np.sum(best_candidate) * dx
    max_corr_val = np.max(np.correlate(best_candidate, 1 - best_candidate, mode='full')) * dx
