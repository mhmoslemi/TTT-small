import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    initial_h_values = kwargs.get("initial_h_values")
    n_points_val = 140
    dx = 2.0 / n_points_val

    def objective(x):
        h = np.clip(x, 0, 1)
        total = np.sum(h)
        penalty = 1e5 * abs(total - n_points_val / 2)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx + penalty

    bounds = [(0.0, 1.0)] * n_points_val

    if initial_h_values is not None:
        base_x = np.array(initial_h_values)
        base_x = base_x[:n_points_val] / (np.sum(base_x[:n_points_val]) / (n_points_val / 2)) * (n_points_val / 2)
        x0 = (base_x + np.random.normal(size=n_points_val) * 0.1).clip(0, 1)
    else:
        x0 = np.random.rand(n_points_val)

    try:
        res_de = differential_evolution(objective, bounds, strategy='best1', mutation=0.5, max_evals=3000, popsize=15, seed=seed, polish=False)
    except:
        res_de = minimize(objective, x0, method='L-BFGS-B', bounds=bounds)

    def slsqp_objective(x):
        h = np.clip(x, 0, 1)
        return np.max(correlate(h, 1 - h, mode='full')) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_val / 2})
    res_slsqp = minimize(slsqp_objective, res_de.x, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-9})

    best_candidate = np.clip(res_slsqp.x, 0, 1)
    max_corr_val = np.max(correlate(best_candidate, 1 - best_candidate, mode='full')) * dx

    end_time = time.time()
    if end_time - start_time > budget_s - 5:
        return (best_candidate.tolist(), max_corr_val, n_points_val)

    return (best_candidate.tolist(), max_corr_val, n_points_val)
