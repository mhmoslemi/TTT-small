        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale = (n_points / 2) / np.sum(h_perturbed)
        h_perturbed *= scale
        candidates.append(h_perturbed)
        h_smoothed = np.convolve(h_initial, np.ones(5)/5, mode='same')
        h_smoothed = np.clip(h_smoothed, 0, 1)
        scale = (n_points / 2) / np.sum(h_smoothed)
        h_smoothed *= scale
        candidates.append(h_smoothed)
        h_inter = np.zeros_like(h_initial)
        h_inter[::2] = 1.0
        h_inter[1::2] = 0.0
        scale = (n_points / 2) / np.sum(h_inter)
        h_inter *= scale
        candidates.append(h_inter)
        h_rand = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h_rand)
        h_rand *= scale
        candidates.append(h_rand)
    else:
        n_points = 140
        h_default = np.ones(n_points) * 0.5
        scale = (n_points / 2) / np.sum(h_default)
        h_default *= scale
        candidates = [h_default]
    
    for h in candidates:
        n = len(h)
        if n > 1000:
            continue
        dx = 2.0 / n
        
        def objective(x):
            corr = correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'max_iter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                n_points = n
    
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, best_c5, len(best_h))
