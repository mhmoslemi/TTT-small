from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.repeat(initial_h_values, 2)
    else:
        h = np.full(n_points, 0.5)
    
    def repair(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if abs(total - integral_target) < 1e-5:
            return x
        delta = (integral_target - total) / n_points
        x += delta * np.random.choice([-1, 1], size=n_points)
        x = np.clip(x, 0, 1)
        return x
    
    def objective(x):
        x = repair(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def perturb(x, temp):
        x = x.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                x[i] = np.random.rand()
            else:
                x[i] += np.random.normal(0, 0.05 * temp)
                x[i] = np.clip(x[i], 0, 1)
        return repair(x)
    
    h = repair(h)
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
