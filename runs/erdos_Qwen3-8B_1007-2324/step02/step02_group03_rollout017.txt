import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_base = 100
    n_medium = 140
    dx_base = 2.0 / n_base
    dx_medium = 2.0 / n_medium

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.full(n_base, 0.5)

    def repair(h, n, dx):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return h
        scale = (n / 2) / s
        h = np.clip(h * scale, 0, 1)
        s = np.sum(h)
        adjust = (n / 2) / s
        return h * adjust

    def obj_de(x):
        h = repair(x, n_base, dx_base)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_base

    bounds_de = [(0.0, 1.0) for _ in range(n_base)]
    result_de = differential_evolution(obj_de, bounds_de, strategy='best1bin', popsize=50, tol=1e-6, maxiter=100, disp=False)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_init, obj_de(h_init), n_base

    h_coarse = result_de.x
    old_x = np.linspace(0, 2, n_base)
    new_x = np.linspace(0, 2, n_medium)
    h_medium = np.interp(new_x, old_x, h_coarse)
    h_medium = repair(h_medium, n_medium, dx_medium)

    def obj_med(h):
        h = repair(h, n_medium, dx_medium)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_medium

    bounds_med = [(0., 1.) for _ in range(n_medium)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_medium / 2}]

    result_med = minimize(obj_med, h_medium, method='SLSQP', bounds=bounds_med, constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_final = h_medium
    else:
        h_final = result_med.x

    h_final = repair(h_final, n_medium, dx_medium)
    c5 = obj_med(h_final)
    return h_final, c5, n_medium
