import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():

        perturb = np.random.normal(0, 0.1, n_points)
        initial_pop += perturb
        initial_pop = np.clip(initial_pop, 0, 1)
        sum_pop = np.sum(initial_pop)
        if sum_pop == 0:
            initial_pop = np.full_like(initial_pop, 0.5)
        else:
            initial_pop *= (integral_target / sum_pop)
        population = [initial_pop]
        for _ in range(49):
            rand_pop = np.random.rand(n_points)
            sum_rand = np.sum(rand_pop)
            if sum_rand == 0:
                rand_pop = np.full_like(rand_pop, 0.5)
            else:
                rand_pop *= (integral_target / sum_rand)
            population.append(rand_pop)
    else:
        population = [np.random.rand(n_points) * (integral_target / np.sum(np.random.rand(n_points))) for _ in range(50)]

    best_h = population[0]
    best_c5 = float('inf')
    start_time = time.time()

    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            trial = population[i].copy()
            mask = np.random.rand(n_points) < 0.9
            trial[mask] = mutant[mask]
            trial = np.clip(trial, 0, 1)
            sum_trial = np.sum(trial)
            if sum_trial == 0:

            if c5 < best_c5:

                best_h = trial
            new_population.append(trial)
        population = new_population

    return best_h, best_c5, n_points
