            trial = np.copy(population[i])
            u = np.random.rand(n_points)
            trial = np.where(u < cr, mutant, trial)
            trial_clip = np.clip(trial, 0, 1)
            sum_clip = np.sum(trial_clip)
            if not np.isclose(sum_clip, integral_target, atol=1e-6):
                delta = integral_target - sum_clip
                trial_clip[-1] += delta
                trial_clip = np.clip(trial_clip, 0, 1)
            new_pop.append(trial_clip)
        population = np.array(new_pop, dtype=np.float64)
        for ind in population:
            h_clip = np.clip(ind, 0, 1)
            corr = np.correlate(h_clip, 1 - h_clip, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points
