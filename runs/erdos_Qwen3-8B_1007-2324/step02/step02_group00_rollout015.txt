import numpy as np
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 160  # Balance resolution and speed
    dx = 2.0 / n_points
    n = n_points
    best_c5 = float('inf')
    best_h = np.zeros(n)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros(n)
        scale = (n / 2) / s
        return np.clip(h * scale, 0, 1)

    # Initial population
    if 'initial_h_values' in globals():
        initial_h = np.repeat(initial_h_values, 2)[:n]
        initial_h = normalize(initial_h)
    else:
        initial_h = np.random.rand(n)
        initial_h = normalize(initial_h)

    population = [initial_h + np.random.normal(0, 0.05, n) for _ in range(20)]
    population = [normalize(p) for p in population]

    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        # Keep top 20% as elites
        elite_size = int(len(population) * 0.2)
        elites = sorted_pop[:elite_size]
        # Generate new population
        new_pop = elites
        while len(new_pop) < len(population):
            a, b = np.random.choice(len(sorted_pop), 2, replace=False)
            parent1, parent2 = sorted_pop[a], sorted_pop[b]
            # Crossover
            split = np.random.randint(1, n-1)
            child = np.concatenate([parent1[:split], parent2[split:]])
            # Mutate
            mutate = np.random.rand(n) < 0.1
            child[mutate] += np.random.uniform(-0.1, 0.1)
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_pop.append(child)
        population = new_pop
        # Track best
        current_best = sorted_pop[0]
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best

    return best_h, best_c5, n_points
