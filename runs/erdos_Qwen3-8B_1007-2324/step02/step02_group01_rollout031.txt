import random

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.uniform(0, 1, n_points)

            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)
        population.append(h)

    def compute_c5(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        max_corr = np.max(corr)

            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + 0.8 * (b - c)
            mutant = np.clip(mutant, 0, 1)
            current_sum = np.sum(mutant)
            if current_sum > 0:

            mutant = np.clip(mutant, 0, 1)

                if np.random.rand() < 0.9:
                    trial[j] = mutant[j]
            trial_c5 = compute_c5(trial)
            parent_c5 = compute_c5(population[i])
            if trial_c5 < parent_c5:

        current_best = min(population, key=compute_c5)
        current_c5 = compute_c5(current_best)
