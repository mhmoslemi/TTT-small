        x_coarse = np.linspace(0, 2, n_coarse)
        current_solution = np.interp(x_coarse, x_parent, initial_h)
    else:
        current_solution = np.full(n_coarse, 0.5)

    current_solution = repair(current_solution, n_coarse)

    for (grid_size, phase_time) in phases:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        x_prev = np.linspace(0, 2, current_grid_size)
        x_current = np.linspace(0, 2, grid_size)
        current_solution = np.interp(x_current, x_prev, current_solution)
        current_solution = repair(current_solution, grid_size)

        if grid_size == n_coarse:
            result = differential_evolution(
                objective,
                bounds=[(0.0, 1.0)] * grid_size,
                x0=current_solution,
                strategy='best1bin',
                mutation=0.7,
                recombination=0.5,
                popsize=15,
                tol=1e-6,
                maxiter=100,
                disp=False
            )
            current_solution = result.x
        elif grid_size == n_medium:
            result = differential_evolution(
                objective,
                bounds=[(0.0, 1.0)] * grid_size,
                x0=current_solution,
                strategy='best1bin',
                mutation=0.7,
                recombination=0.5,
                popsize=15,
                tol=1e-6,
                maxiter=100,
                disp=False
            )
            current_solution = result.x
        elif grid_size == n_fine:
            result = minimize(
                objective,
                current_solution,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * grid_size,
                constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - grid_size / 2}]
            )
            if result.success:
                current_solution = result.x
            else:
                current_solution = np.copy(current_solution)
        current_solution = repair(current_solution, grid_size)
        if time.time() - start_time > elapsed + phase_time:
            break

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return current_solution, objective(current_solution), n_fine
    return current_solution, objective(current_solution), n_fine
