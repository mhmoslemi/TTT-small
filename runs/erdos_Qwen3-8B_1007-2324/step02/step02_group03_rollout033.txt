import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points_base = 100
    n_points_medium = 140
    dx_base = 2.0 / n_points_base
    dx_medium = 2.0 / n_points_medium

    # Initialize with given initial_h_values or random starting point
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        # Interpolate to base grid
        x_base = np.linspace(0, 2, n_points_base)
        x_initial = np.linspace(0, 2, len(initial_h_values))
        h_initial_base = np.interp(x_base, x_initial, initial_h_values)
    else:
        h_initial_base = np.full(n_points_base, 0.5)

    def repair(h, n_points):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum == 0:
            scale_factor = 1.0
        else:
            scale_factor = target_sum / current_sum
        h_sclipped = h * scale_factor
        h_final = np.clip(h_sclipped, 0, 1)
        final_sum = np.sum(h_final)
        adjust_factor = target_sum / final_sum if final_sum != 0 else 1.0
        return h_final * adjust_factor

    def objective_DE(candidate):
        h_repaired = repair(candidate, n_points_base)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx_base

    bounds_DE = [(0.0, 1.0) for _ in range(n_points_base)]
    result_phase1 = differential_evolution(objective_DE, bounds_DE, strategy='best1bin', popsize=20, mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=150)

    best_h_phase1 = result_phase1.x

    h_repaired_medium = repair(h_medium, n_points_medium)

    def objective_medium(h):
        corr = np.correlate(h, 1 - h, mode='full')
