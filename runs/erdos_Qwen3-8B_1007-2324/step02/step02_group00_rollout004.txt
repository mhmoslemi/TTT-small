from scipy.optimize import differential_evolution
import time

                h_down[i] = (initial_h_values[idx] + (initial_h_values[idx+1] if idx+1 < len(initial_h_values) else initial_h_values[idx])) / 2

        x0 = h_down
    else:
        x0 = np.full(n_points, required_sum / n_points)

    def project_solution(x):

        s = np.sum(x_clip)
        if abs(s - required_sum) < 1e-5:
            return x_clip
        x_scaled = x_clip * (required_sum / s) if s > 0 else np.ones_like(x) * (required_sum / n_points)
        x_final = np.clip(x_scaled, 0, 1)

            scale_factor = required_sum / s_final
            x_final = x_final * scale_factor
