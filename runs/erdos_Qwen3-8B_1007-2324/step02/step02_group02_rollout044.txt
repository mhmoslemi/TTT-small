import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates = [h_initial]
        for _ in range(3):
            perturb = np.random.normal(0, 0.01, len(h_initial))
            perturbed = h_initial + perturb
            perturbed = np.clip(perturbed, 0, 1)
            sum_perturbed = np.sum(perturbed)
            if sum_perturbed != 0:
                perturbed = perturbed * (len(h_initial)/2) / sum_perturbed
            candidates.append(perturbed)
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand / np.sum(h_rand) * (len(h_initial)/2)
        candidates.append(h_rand)

    for h in candidates:
        n = len(h)
        if n > 1000:
            continue
        dx = 2.0 / n

        def objective(x):
            current_h = x
            corr = correlate(current_h, 1 - current_h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        def gradient(x):
            current_h = x
            corr = correlate(current_h, 1 - current_h, mode='full')
            m = np.argmax(corr)
            lag_index = m - (n - 1)
            grad = np.zeros(n)
            for i in range(n):
                i_plus_k = i + lag_index
                term1 = 1.0 - current_h[i_plus_k] if 0 <= i_plus_k < n else 0.0

                grad[i] = term1 - term2
            return grad

        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, jac=gradient, tol=1e-6, options={'max_iter': 100})
