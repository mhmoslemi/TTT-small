from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points_base = 100
    n_points_medium = 140
    dx_base = 2.0 / n_points_base
    dx_medium = 2.0 / n_points_medium

        initial_h = np.full(n_points_base, 0.5)

    def repair(h, dx):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = len(h) * dx
        if np.abs(current_sum - target_sum) < 1e-8:
            return h
        scale = target_sum / current_sum
        h = np.clip(h * scale, 0, 1)
        current_sum = np.sum(h)
        if np.abs(current_sum - target_sum) < 1e-8:
            return h
        scale = target_sum / current_sum
        return np.clip(h * scale, 0, 1)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_base

    bounds_base = [(0.0, 1.0) for _ in range(n_points_base)]
    result_de = differential_evolution(
        objective,
        bounds_base,
        strategy='best1bin',
        popsize=30,

        polish=True,
