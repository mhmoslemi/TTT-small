    np.random.seed(seed)

    if 'initial_h_values' in globals():

        candidates.append(interleaved)
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand / np.sum(h_rand) * (len(h_initial)/2)
        candidates.append(h_rand)
    else:
        n_default = 140
        h_rand = np.random.rand(n_default)
        h_rand = h_rand / np.sum(h_rand) * (n_default / 2)
        candidates.append(h_rand)
    
    for h in candidates:
        n_points = len(h)

        dx = 2.0 / n_points

            h_vec = np.clip(x, 0.0, 1.0)

            score = max_corr * dx
            sum_h = np.sum(h_vec)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
