import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 280
    dx = 2.0 / n_points
    integral_target = n_points / 2

    if 'initial_h_values' in globals():
        parent = initial_h_values
        h = parent + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale = 140.0 / h_sum if h_sum != 0 else 1.0
        h = h * scale
        starts = [h]
    else:
        starts = [np.random.uniform(0, 1, n_points) for _ in range(12)]
        for i in range(len(starts)):
            h = starts[i]
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            scale = 140.0 / h_sum if h_sum != 0 else 1.0
            starts[i] = h * scale

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint_func = lambda h: np.sum(h) - integral_target

    best_h = None
    best_c5 = float('inf')

    for start in starts:
        res = minimize(
            objective,
            start,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return starts[0], objective(starts[0]), n_points

    return best_h, best_c5, n_points
