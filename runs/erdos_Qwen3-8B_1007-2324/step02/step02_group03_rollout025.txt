    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective_DE(initial_h), n_stage1
    
    best_h_stage1 = result_de.x
    best_h_stage2 = np.repeat(best_h_stage1, 3)[:n_stage2]  # Interpolate roughly to 140 points
    sum_h = np.sum(best_h_stage2)
    scale = (target_integral) / sum_h
    best_h_stage2 *= scale
    
    def objective_SLSQP(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_integral}]
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_stage2)]
    
    result_slsqp = minimize(objective_SLSQP, best_h_stage2, method='SLSQP', bounds=bounds_slsqp, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-8})
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h_stage2, objective_SLSQP(best_h_stage2), n_stage2
    
    return result_slsqp.x, result_slsqp.fun, n_stage2
