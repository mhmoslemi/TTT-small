    initial_guess = initial_h[:-1] * integral_target / np.sum(initial_h[:-1])

    def objective(h):
        sum_first = np.sum(h)
        last = np.clip(integral_target - sum_first, 0, 1)
        full_h = np.concatenate([h, [last]])
        h_clip = np.clip(full_h, 0, 1)
        corr = np.correlate(h_clip, 1 - h_clip, mode='full')
        return np.max(corr) * dx

    result = differential_evolution(
        objective,
        bounds,
        popsize=50,
        maxiter=50,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        dispatch='parallel'
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = initial_h.copy()
        c5_bound = objective(initial_guess)
        return h_values, c5_bound, n_points

    h_values = np.concatenate([result.x, [np.clip(integral_target - np.sum(result.x), 0, 1)]])
    c5_bound = result.fun
    return h_values, c5_bound, n_points
