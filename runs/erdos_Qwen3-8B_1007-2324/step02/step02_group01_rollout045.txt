def adjust_sum(h, target_sum=140):
    h_clip = np.clip(h, 0, 1)
    current_sum = np.sum(h_clip)
    delta = target_sum - current_sum
    if delta == 0:
        return h_clip
    # Calculate how much we can increase each element
    max_possible_increase = 1 - h_clip
    # Find the indices where we can increase
    indices = np.where(max_possible_increase > 0)[0]
    if len(indices) == 0:
        return h_clip  # Cannot adjust, sum is already at max
    # Calculate total possible increase
    total_possible = np.sum(max_possible_increase)
    if delta <= total_possible:
        # Distribute the delta proportionally
        max_possible = max_possible_increase[indices]
        weights = max_possible / total_possible
        increase = delta * weights
        h_clip[indices] += increase
    else:
        # Cannot fully adjust, distribute as much as possible
        h_clip[indices] = np.minimum(h_clip[indices] + delta / len(indices), 1)
    return h_clip

    def adjust_sum(h, target=integral_target):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target - current_sum
        if delta == 0:
            return h
        max_possible = 1 - h
        usable = max_possible > 1e-6
        if np.any(usable):
            max_add = np.sum(max_possible[usable])
            if abs(delta) <= max_add:
                weights = max_possible[usable] / max_add
                h[usable] += delta * weights
            else:
                h[usable] = np.clip(h[usable] + delta / np.sum(usable), 0, 1)
        return h

    def objective(h):
        h_adj = adjust_sum(h)
        corr = np.correlate(h_adj, 1 - h_adj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    pop_size = 50
    population = []
    for _ in range(pop_size):

            h += np.random.normal(0, 0.05, h.shape)
            population.append(adjust_sum(h))
        else:
            h = np.random.rand(n_points)
            population.append(adjust_sum(h))

    best_h = population[0]
    best_c5 = objective(best_h)

    generation = 0

    while time.time() - start_time < budget_s and generation < max_gen:
