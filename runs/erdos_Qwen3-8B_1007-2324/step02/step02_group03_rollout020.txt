import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)
    
    def objective_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e3 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution
    result_de = differential_evolution(
        objective_de, bounds, 
        strategy='best1bin', 
        popsize=15, 
        mutation=0.5, 
        recombination=0.8, 
        tol=1e-8, 
        maxiter=100, 
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective_de(initial_h), n_points
    
    h_de = result_de.x
    
    # Refine with SLSQP
    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    result_slsqp = minimize(
        objective_slsqp, 
        h_de, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_de, objective_de(h_de), n_points
    
    h_values = result_slsqp.x
    c5_bound = result_slsqp.fun
    return h_values, c5_bound, n_points
