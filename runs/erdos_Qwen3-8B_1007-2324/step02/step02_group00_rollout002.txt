import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    required_sum = n_points // 2  # 70

    def project_solution(x):
        x_clip = np.clip(x, 0, 1)
        s = np.sum(x_clip)
        if abs(s - required_sum) < 1e-5:
            return x_clip
        x_scaled = x_clip * (required_sum / s) if s > 0 else np.ones_like(x) * (required_sum / n_points)
        x_final = np.clip(x_scaled, 0, 1)
        s_final = np.sum(x_final)
        if abs(s_final - required_sum) > 1e-5:
            scale_factor = required_sum / s_final
            x_final = x_final * scale_factor
            x_final = np.clip(x_final, 0, 1)
            s_final = np.sum(x_final)
            if abs(s_final - required_sum) > 1e-5:
                scale_factor = required_sum / s_final
                x_final = x_final * scale_factor
        return x_final

    if 'initial_h_values' in globals():
        h_down = np.zeros(n_points)
        for i in range(n_points):
            idx = 2 * i
            if idx + 1 < len(initial_h_values):
                h_down[i] = (initial_h_values[idx] + initial_h_values[idx+1]) / 2
            else:
                h_down[i] = initial_h_values[idx]
        x0 = h_down
    else:
        x0 = np.full(n_points, required_sum / n_points)

    def objective(x):
        x_proj = project_solution(x)
        corr = np.correlate(x_proj, 1 - x_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    pop_size = min(40, int(budget_s) // 2)
    max_iter = int(budget_s * 2)
    result = differential_evolution(
        objective, bounds, popsize=pop_size, maxiter=max_iter,
        mutation=0.8, recombination=0.7, seed=seed, tol=1e-6,
        vectorize=True
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = project_solution(x0)
        c5_bound = objective(x0)
        return h_values, c5_bound, n_points

    best_h = result.x
    best_h = project_solution(best_h)
