from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    np.random.seed(seed)
    
    # Initialize candidates
    candidates = []
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        candidates.append(h_initial)
        # Upsample
        candidates.append(np.repeat(h_initial, 2))
        # Smoothed
        smoothed = np.convolve(h_initial, np.ones(5)/5, mode='same')
        smoothed = np.clip(smoothed, 0, 1)
        candidates.append(smoothed)
        # Random perturbed
        h_perturb = h_initial + np.random.normal(0, 0.05, len(h_initial))
        h_perturb = np.clip(h_perturb, 0, 1)
        candidates.append(h_perturb)
        # Random seed
        h_rand = np.random.rand(len(h_initial))
        h_rand = h_rand / np.sum(h_rand) * (len(h_initial)/2)
        candidates.append(h_rand)
    else:
        # Default grid: 140 points
        n_default = 140
        h_default = np.zeros(n_default)
        h_default[:n_default//2] = 1.0
        candidates.append(h_default)
    
    # Optimize each candidate
    for h in candidates:
        n = len(h)
        if n > 1000:
            continue
        bounds = [(0.0, 1.0) for _ in range(n)]
        
        def objective(x):
            h_vec = np.clip(x, 0, 1)
            dx = 2.0 / n
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')

            sum_h = np.sum(h_vec)

        # Run DE

            objective, bounds, strategy='best1bin',
