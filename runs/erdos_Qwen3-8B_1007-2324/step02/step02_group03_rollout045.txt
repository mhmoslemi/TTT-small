def resample_down(parent, m):

    dx_parent = 2.0 / n
    dx_new = 2.0 / m
    new_h = []
    for i in range(m):
        x_new = i * dx_new
        # find j such that parent[j] is closest to x_new
        j = int(x_new / dx_parent)
        # ensure j is within bounds
        j = min(max(j, 0), n-1)
        new_h.append(parent[j])
    new_h = np.array(new_h)
    # clip and scale
    new_h = np.clip(new_h, 0, 1)
    sum_h = np.sum(new_h)
    if sum_h == 0:
        return np.zeros(m)
    scale = (m / 2) / sum_h
    return new_h * scale

            return np.max(corr) * (2.0 / 100)
        
        bounds_medium = [(0.0, 1.0) for _ in range(100)]
        result_medium = differential_evolution(
            objective_medium,
            bounds_medium,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False
        )
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            best_h = h_medium
            best_c5 = objective_medium(h_medium)
        else:
            best_h = result_medium.x
            best_c5 = result_medium.fun
            h_fine = resample_up(best_h, 200)
            
            def objective_fine(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * (2.0 / 200)
            
            bounds_fine = [(0.0, 1.0) for _ in range(200)]
            result_fine = minimize(
                objective_fine,

                method='SLSQP',
                bounds=bounds_fine,
                constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - 200/2}],

            )
            elapsed = time.time() - start_time
            if elapsed > budget_s:

                best_c5 = objective_fine(h_fine)
