        initial_h = np.full(n, 0.5)
        scale = (n / 2) / initial_h_sum
        initial_h = initial_h * scale
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        h_scaled = h.copy()
        h_scaled_sum = np.sum(h_scaled)
        if h_scaled_sum == 0:
            return float('inf')
        scale = (n / 2) / h_scaled_sum
        h_scaled = h_scaled * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_values = initial_h
        c5_bound = objective(h_values)
        return h_values, c5_bound, n_points
    
    h_values = result.x
    h_scaled = h_values.copy()
    h_scaled_sum = np.sum(h_scaled)
    if h_scaled_sum == 0:
        h_scaled = np.zeros_like(h_scaled)
    else:
        scale = (n / 2) / h_scaled_sum
        h_scaled = h_scaled * scale
        h_scaled = np.clip(h_scaled, 0, 1)
    c5_bound = result.fun
    return h_scaled, c5_bound, n_points
