import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 160
    n = n_points
    dx = 2.0 / n

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1000 * abs(np.sum(h) - n / 2)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n)]

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=20,
        tol=1e-8,
        seed=seed
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    h_values = result.x
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    c5_bound = compute_c5(h_values)
    return h_values, c5_bound, n_points
