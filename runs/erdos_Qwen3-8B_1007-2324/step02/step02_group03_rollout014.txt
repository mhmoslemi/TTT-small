import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.full(n_points, 0.5)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_func}]

    def optimize_h():
        # Initial optimization with differential evolution to escape local optima
        result_de = differential_evolution(
            objective,
            bounds,
            constraints=[{'type': 'eq', 'fun': constraint_func}],
            maxiter=50,
            popsize=20,
            strategy='best1bin',
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.8,
            disp=False
        )
        de_sol = result_de.x
        
        # Refine with L-BFGS-B method for faster convergence
        result_lbgfs = minimize(
            objective,
            de_sol,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        return result_lbgfs

    result = optimize_h()
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h), n_points

    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
