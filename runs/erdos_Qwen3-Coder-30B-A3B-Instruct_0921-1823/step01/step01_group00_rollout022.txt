import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

    swarm = np.random.rand(swarm_size, n_points) * 0.5 + 0.25  # Random in [0.25, 0.75]

    # Initialize personal best positions
    pbest = swarm.copy()
    pbest_scores = np.array([evaluate_erdos_solution(p, n_points) for p in pbest])
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_scores)
    gbest = pbest[global_best_idx].copy()
    gbest_score = pbest_scores[global_best_idx]

    velocity = np.zeros_like(swarm)

            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])

            current_score = evaluate_erdos_solution(swarm[i], n_points)

            if current_score < pbest_scores[i]:

                # Update global best
