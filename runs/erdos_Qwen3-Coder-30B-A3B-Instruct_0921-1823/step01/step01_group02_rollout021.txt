import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    best_h = h0.copy()
    dx = 2.0 / n_points
    
    def compute_overlap(h_vec):
        # Correlation with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap, corr
    
    def objective_with_penalty(h_vec, penalties):
        max_overlap, corr = compute_overlap(h_vec)
        return max_overlap
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    # Main optimization loop
    max_iter = 5000
    patience = 50
    best_C = float('inf')
    last_improvement = 0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Gradient-based optimization with adaptive approach
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        current_time = time.time()
        
        # Current best values
        current_C, current_corr = compute_overlap(best_h)
        if current_C < best_C:
            best_C = current_C
            last_improvement = i
            
        if i - last_improvement > patience:
            # Perturb and restart if no improvement
            noise = np.random.normal(0, 0.01, n_points)
            best_h = np.clip(best_h + noise, 0, 1)
            # Normalize
            best_h = best_h * (n_points / 2.0) / np.sum(best_h)
            
        # Optimize with current approach
        try:
            res = minimize(
                lambda h: compute_overlap(h)[0],
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8},
                callback=lambda x: None
            )
            if res.success:
                best_h = res.x.copy()
                # Normalize to satisfy constraint
                best_h = best_h * (n_points / 2.0) / np.sum(best_h)
            else:
                # Fallback to just the constraint if optimization fails
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points / 2.0) / np.sum(best_h)
        except Exception as e:
            # Fallback to simple projection
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    
    # Final QCQP refinement
    try:
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2.0)
        
        # Overlap constraints for all shifts
        for k in range(1, n_points):
            expr = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    expr += h[i] * (1 - h[j])
            constraints.append(expr <= C * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            best_h = h.value.copy()
    except:
        pass
    
    # Final compute C5
    final_C, _ = compute_overlap(best_h)
    
    return best_h, final_C, n_points
