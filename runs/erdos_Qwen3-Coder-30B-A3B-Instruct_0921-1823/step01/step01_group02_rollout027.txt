import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        dx = 2.0 / n_points
        h0 = np.ones(n_points) * 0.5
        
    # Precompute indices for efficient correlation computation
    max_shift = n_points - 1
    dx = 2.0 / n_points
    
    def compute_overlap(h_vec):
        # Compute all correlations: correlate h with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take the maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def compute_gradient(h_vec, shift_idx):
        # Analytical gradient for a specific shift
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = i + shift_idx
            if j < n_points:
                grad[i] -= (1 - h_vec[j])
            j = i - shift_idx
            if j >= 0:
                grad[i] -= h_vec[j]
        return grad * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    de = 0.001  # small perturbation
    eval_count = 0
    
    # Start with the initial solution
    current_h = h0.copy()
    current_C = compute_overlap(current_h)
    
    # Iterative refinement using gradient-based methods
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Determine which shifts to consider (top few max overlaps)
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        # Get indices of the max values (shifts)
        max_indices = np.argsort(corr)[-10:]  # Check top 10 shifts
        max_indices = max_indices[::-1]  # Sort descending
        
        improved = False
        
        for shift_idx in max_indices:
            if shift_idx == 0:
                continue  # Skip zero shift as it's not a valid overlap
                
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Objective function for specific shift
            def obj_func(h_vec):
                # Compute only this specific overlap
                expr = 0
                for i in range(n_points):
                    j = i + shift_idx
                    if j < n_points:
                        expr += h_vec[i] * (1 - h_vec[j])
                return expr * dx
            
            # Gradient for this shift
            def grad_func(h_vec):
                return compute_gradient(h_vec, shift_idx)
            
            # Local optimization
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                # Use SLSQP with gradient info
                res = minimize(obj_func, current_h.copy(), method='SLSQP', 
                             bounds=bounds, constraints=cons, 
                             jac=grad_func, options={'maxiter': 100})
                
                if res.success:
                    new_h = res.x.copy()
                    new_C = compute_overlap(new_h)
                    eval_count += 1
                    if new_C < current_C:
                        current_h = new_h
                        current_C = new_C
                        improved = True
                        break # Move to next iteration to re-evaluate all shifts
            except:
                pass
                
        if not improved and iteration > 20:  # fallback perturbation
            # Perturb slightly in random directions
            perturbation = np.random.normal(0, 0.001, n_points)
            new_h = current_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            # Project onto constraint
            curr_sum = np.sum(new_h)
            scale_factor = (n_points / 2.0) / curr_sum if curr_sum != 0 else 1
            new_h = new_h * scale_factor
            new_C = compute_overlap(new_h)
            if new_C < current_C:
                current_h = new_h
                current_C = new_C
            
        if time.time() - start_time > budget_s - 0.1:
            break

    return current_h.copy(), current_C, n_points
