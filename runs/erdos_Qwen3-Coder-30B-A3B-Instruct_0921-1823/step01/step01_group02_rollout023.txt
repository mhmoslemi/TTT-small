from scipy.optimize import minimize
import time

        # Normalized by dx
        dx = 2.0 / n_points
        # Compute all correlations: correlate h with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        C = np.max(corr) * dx
        return C

    def gradient(h_vec):
        # Numerical gradient computation
        eps = 1e-8
        grad = np.zeros_like(h_vec)
        base_obj = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

    # Optimize with scipy
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial optimization with L-BFGS-B
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6})
        if res.success:
            temp_h = res.x.copy()
            temp_C = objective(temp_h)
            if temp_C < best_C:
                best_h = temp_h
                best_C = temp_C
    except:
        pass
        
    # Iterative local refinement with focused optimization
    iterations = 0
    patience = 0
    max_patience = 10
    last_improvement = 0
    elapsed = time.time() - start_time

        iterations += 1
        elapsed = time.time() - start_time

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - n_points + 1

            dx = 2.0 / n_points
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            max_val = np.max(corr)
            # Add penalty for violating current critical constraint
            penalty = 0
            if max_shift_idx < len(corr) and corr[max_shift_idx] > max_val * 0.999:
                penalty = 100 * (corr[max_shift_idx] - max_val * 0.999)**2
            return max_val * dx + penalty

                new_C = objective(new_h)

                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except:

        if patience > max_patience:

            perturb_factor = 0.01

            # Project back to feasible region
            perturbed_h = np.clip(perturbed_h, 0, 1)
            # Correct sum constraint
            current_sum = np.sum(perturbed_h)
            if current_sum > 0:

            patience = 0

        h = cp.Variable(n_points)
        C = cp.Variable()

        constraints.append(cp.sum(h) == n_points / 2.0)

        dx = 2.0 / n_points
        for k in range(1, n_points):
            expr = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    expr += h[i] * (1 - h[j])
            constraints.append(expr <= C * dx)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            final_h = h.value.copy()
            final_C = float(C.value)
            if final_C < best_C:
                best_h = final_h
                best_C = final_C
    except:
        pass
        
    elapsed = time.time() - start_time
    return best_h.copy(), best_C, n_points
