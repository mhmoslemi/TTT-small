import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():

    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_ineq(h):
        return np.minimum(h, 1.0) - h  # h <= 1
          
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq},
            {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}]
    
    # Optimization loop
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts
    max_restarts = 50
    restart_count = 0
    
    # Enhanced multi-start approach
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Try different perturbation strategies
        if restart_count < 10:
            # Small random perturbations
            perturbation_magnitude = 0.03
        elif restart_count < 30:
            # Medium perturbations
            perturbation_magnitude = 0.07
        else:
            # Larger perturbations to escape local minima
            perturbation_magnitude = 0.15
            
        # Perturb with noise
        noise = np.random.normal(0, perturbation_magnitude, n_points)
        h_perturbed = np.clip(h_current + noise, 0, 1)
        
        # Re-normalize after perturbation
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Try multiple optimizers
        optimizers = ['COBYLA', 'SLSQP']
        
        for opt in optimizers:
            try:
                res = minimize(
                    objective,
                    h_perturbed,
                    method=opt,
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 300, 'catol': 1e-6}
                )
                
                if res.success:
                    h_new = res.x
                    c5_new = obj(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        break  # Break after finding an improvement
            except:
                continue
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
        
    # Final computation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
