import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    # Optimization loop with multiple approaches and restarts
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts
    max_restarts = 50
    restart_count = 0
    
    # Different optimization methods to try
    methods = ['SLSQP', 'COBYLA']
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Try different starting points
        if restart_count == 0:
            # Start with the initial solution and add noise
            h_perturbed = h_current + np.random.normal(0, 0.05, n_points)
        elif restart_count < 10:
            # Introduce larger perturbation
            h_perturbed = h_current + np.random.normal(0, 0.1, n_points)
        else:
            # For later restarts, use smaller noise with random shifts
            h_perturbed = np.random.uniform(0, 1, n_points)
            
        # Ensure it satisfies the integral constraint
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Try different optimization methods
        method = methods[(restart_count % len(methods))]
        
        try:
            res = minimize(
                objective,
                h_perturbed,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'disp': False}
            )
            
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
        
    # Final computation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
