import numpy as np
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    # Initialize with slightly perturbed version of current best
    for i in range(n_particles):
        h_perturbed = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
        # Re-normalize
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        particles.append(h_perturbed.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(h_perturbed.copy())
        personal_best_vals.append(evaluate_erdos_solution(h_perturbed, n_points))
    
    # Global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (personal_best[i] - particles[i]) +
                            social * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            fitness = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if fitness < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = fitness
                
                # Check global best
                if fitness < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = fitness
                    stagnation_count = 0  # Reset stagnation counter
                else:
                    stagnation_count += 1
                    
        # Early stopping if no improvement for a while
        if stagnation_count > 100:
            break
            
        # Adaptive inertia weight
        if iteration > 100 and np.random.rand() < 0.1:
            inertia = max(0.3, inertia - 0.01)
            
    # Final refinement
    final_h = global_best.copy()
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)
