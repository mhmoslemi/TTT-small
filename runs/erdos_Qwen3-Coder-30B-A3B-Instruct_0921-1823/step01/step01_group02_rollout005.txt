from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    def objective(h_vec):
        dx = 2.0 / n_points
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only the overlapping parts contribute
        max_overlap = np.max(corr)
        return max_overlap * dx

    # Constraint: integral of h equals 1
    def constraint_sum(h_vec):
        dx = 2.0 / n_points
        return np.sum(h_vec) * dx - 1.0

    # Conservation of a vector with domain [0,1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization with L-BFGS-B
    cons = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 100000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    best_h = h0.copy()
    best_C = objective(best_h)
    
    # Multiple restarts with perturbations
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break

        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        try:
            res = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, 
                          constraints=cons, options=options)
            
            if res.success and res.fun < best_C:
                best_h = res.x.copy()
                best_C = res.fun
                
        except:
            pass
    
    # Refinement via CVXPY
    try:
        import cvxpy as cp
        
        # Vectorized form
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2.0)

            for i in range(n_points):
                j = i + k
                if j < n_points:
                    expr += h[i] * (1 - h[j])
            constraints.append(expr <= C * dx)
        
        objective_c = cp.Minimize(C)
        prob = cp.Problem(objective_c, constraints)
        
        # Solve using a fast solver
        prob.solve(solver=cp.ECOS, verbose=False)

            if c_final < best_C:
