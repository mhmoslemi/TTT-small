import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better resolution
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():

    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper function for optimization
    def objective(h):
        return evaluate_erdos_solution(h, n_points)
        
    # Constraint functions
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_ineq(h):
        return np.minimum(h, 1.0) - h  # h <= 1
      
    # Simple optimization using SLSQP with restarts
    best_h = h_current.copy()
    best_c5 = objective(best_h)
    
    # Optimization parameters
    max_restarts = 50
    iter_per_restart = 400
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Generate perturbed starting point
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_current + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Re-normalize to maintain constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            # Optimize with SLSQP
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[
                    {'type': 'eq', 'fun': constraint_eq},
                    {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}
                ],
                options={'maxiter': iter_per_restart, 'ftol': 1e-8}
            )
            
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
    
    # Final refinement
    try:
        res = minimize(
            objective,
            best_h.copy(),
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': constraint_eq},
                {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}
            ],
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        
        if res.success:
            h_new = res.x
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
