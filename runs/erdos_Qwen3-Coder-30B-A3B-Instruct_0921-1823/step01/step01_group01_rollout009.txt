import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    return np.max(conv) * dx

def project_to_simplex(h, target_sum=1.0):
    """Project h to the feasible set with integral = target_sum."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if np.isclose(current_sum, target_sum):
        return h
    scale = target_sum / current_sum
    return h * scale
