        return np.max(corr) * dx

        dx = 2.0 / n_points
        # Compute all correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        
        # Add penalty for the shifts that currently show highest overlap
        penalty = 0
        if penalty_shifts is not None:
            for k in penalty_shifts:
                if k < len(corr):
                    if corr[k] > max_corr * 0.95:  # Apply penalty if close to max
                        penalty += 1000 * (corr[k] - max_corr * 0.95)**2
        
        return max_corr * dx + penalty
    
    def compute_constraints(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    # Start with gradient-based optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': compute_constraints}
    
    iter_count = 0
    max_iter_no_improvement = 50
    no_improvement_count = 0
    best_iter_C = 1e6

        if time.time() - start_time > budget_s * 0.95:
            break

        max_shifts = min(n_points, 20)  # Limit to first 20 shifts only

            top_shifts = np.argsort(corr)[-max_shifts:][::-1]

            res = minimize(
                lambda x: objective_with_penalty(x, penalty_shifts),

            except:
                pass
        except:
            pass
    
    # Final verification
    final_C = compute_C(best_h)
