import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increased discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    # Optimization loop with multiple strategies and restarts
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts with different approaches
    max_restarts = 50
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Try different random perturbations
        if restart_count < 20:
            # Small perturbations
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.01, n_points), 0, 1)
        else:
            # Medium perturbations
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.03, n_points), 0, 1)
            
        # Re-normalize to satisfy integral constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Try different optimization methods
        methods = ['SLSQP', 'COBYLA']
        method_idx = restart_count % len(methods)
        method = methods[method_idx]
        
        try:
            res = minimize(
                objective,
                h_perturbed,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500}
            )
            
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
    
    # Final fine-tuning step
    try:
        res = minimize(
            objective,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        if res.success:
            h_final = res.x
            c5_final = obj(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Final computation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
