import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    # Optimization loop
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Multi-start approach with multiple strategies
    max_restarts = 50
    restart_count = 0
    
    strategies = [
        lambda h: np.clip(h + np.random.normal(0, 0.03, n_points), 0, 1),
        lambda h: np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1),
        lambda h: np.clip(h * (1 + np.random.normal(0, 0.02, n_points)), 0, 1),
    ]
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Choose a random strategy
        strategy = np.random.choice(strategies)
        h_perturbed = strategy(h_current)
        
        # Re-normalize
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            # Try both SLSQP and COBYLA
            methods = ['SLSQP', 'COBYLA']
            for method in methods:
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                res = minimize(
                    objective,
                    h_perturbed,
                    method=method,
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 300, 'disp': False}
                )
                
                if res.success:
                    h_new = res.x
                    c5_new = obj(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        
        except:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
        
        # Occasionally refine with local search
        if restart_count % 5 == 0 and restart_count > 0:
            # Apply small perturbations to the best solution
            h_refined = best_h + np.random.normal(0, 0.01, n_points)
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            try:
                res = minimize(
                    objective,
                    h_refined,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 100, 'disp': False}
                )
                
                if res.success:
                    h_new = res.x
                    c5_new = obj(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        
            except:
                pass
    
    # Final computation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
