import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase resolution 
    dx = 2.0 / n_points
    
    # Initialize with the pre-existing construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Initial construction - a simple symmetric approximation
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    inertia_w = 0.7
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.1
    
    # Initialize particles and velocities
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    personal_best_positions = np.zeros((n_particles, n_points))
    personal_best_scores = np.full(n_particles, np.inf)
    
    # Initialize particles around the current solution
    for i in range(n_particles):
        noise = np.random.normal(0, 0.05, n_points)
        particles[i] = np.clip(h_current + noise, 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        personal_best_positions[i] = particles[i].copy()
        personal_best_scores[i] = evaluate_erdos_solution(particles[i], n_points)
    
    global_best_position = personal_best_positions[np.argmin(personal_best_scores)].copy()
    global_best_score = np.min(personal_best_scores)
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = inertia_w * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            score = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best if necessary
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
        
        # Adapt parameters based on convergence
        if iteration > 100 and iteration % 50 == 0:
            if abs(global_best_score - np.min(personal_best_scores)) < 1e-6:
                inertia_w = min(0.9, inertia_w + 0.05)
                if inertia_w > 0.9:
                    c1, c2 = 1.0, 2.0
                else:
                    c1, c2 = 1.5, 1.5
    
    # Final refinement
    final_h = global_best_position.copy()
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    # Ensure final validity
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)
