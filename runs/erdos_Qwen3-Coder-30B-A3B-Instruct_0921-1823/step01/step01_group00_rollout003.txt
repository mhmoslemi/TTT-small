import numpy as np
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

    positions = np.random.uniform(0, 1, (n_particles, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Initialize personal best
    pbest_positions = positions.copy()
    pbest_scores = np.full(n_particles, np.inf)
    
    # Initialize global best
    gbest_idx = 0
    gbest_score = np.inf
    
    # Constraint adjustment: normalize sum to 1
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Initial positions and personal bests
    for i in range(n_particles):
        positions[i] = normalize(positions[i])
        score = evaluate_erdos_solution(positions[i], n_points)
        pbest_scores[i] = score
        if score < gbest_score:
            gbest_score = score
            gbest_idx = i
    
    # PSO iterations
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest_positions[i] - positions[i]) +
                social * r2 * (positions[gbest_idx] - positions[i])
            )
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply boundary constraints
            positions[i] = np.clip(positions[i], 0, 1)
            
            # Apply sum constraint
            positions[i] = normalize(positions[i])
            
            # Evaluate
            score = evaluate_erdos_solution(positions[i], n_points)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = positions[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest_idx = i
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        # Early termination if stagnated
        if stagnation_count >= max_stagnation:
            break
            
    # Final optimization with L-BFGS if improvement possible
    h_final = pbest_positions[gbest_idx]
    best_c5 = gbest_score
    
    # Fine-tune if needed
    if time.time() - start_time < budget_s * 0.9:

            return evaluate_erdos_solution(h, n_points)

    final_c5 = evaluate_erdos_solution(h_final, n_points)
    
    return (h_final.tolist(), final_c5, n_points)
