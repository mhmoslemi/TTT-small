import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    return np.max(conv) * dx

def project_to_simplex(h, target_sum=1.0):
    """Project h to the feasible set with integral = target_sum."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if np.isclose(current_sum, target_sum):
        return h
    scale = target_sum / current_sum
    return h * scale

def gradient_step(h, learning_rate=0.01):

    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(conv)
    max_k = max_idx - (n - 1)  # This is the shift k that maximizes overlap

    # Compute gradient: derivative of integral h(x)(1-h(x+k)) w.r.t. h(x)
    # Since h(x) contributes to the integral for all x where x+k is valid,
    # The gradient wrt h(x) is (1 - h(x+k)) 
    grad = np.zeros(n)
    for i in range(n):
        j = i + max_k  # j = x + k
        if 0 <= j < n:
            # Gradient contribution from h[i] to the overlap at shift k
            grad[i] += (1 - h[j])
    
    # apply gradient step
    h_new = h - learning_rate * grad
    
    # clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    # project to maintain integral constraint
    h_new = project_to_simplex(h_new)
    
    return h_new
