from scipy.optimize import minimize

        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        C = np.max(corr) * dx
        return C

        return np.sum(h_vec) - n_points / 2.0

    def compute_gradient(h_vec, k):
        # Returns gradient of overlap w.r.t. h for shift k
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = i + k
            if j < n_points:
                grad[i] += (1 - h_vec[j])
            # Reverse: also consider h[j] term in the derivative
            i_rev = j
            k_rev = n_points - k
            if i_rev < n_points:
                grad[i_rev] -= h_vec[i]
        return grad

    # Cycling strategy: optimize each shift in turn
    max_iter = 5000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s:

        current_C = compute_C(h_current)

        bounds = [(0, 1) for _ in range(n_points)]
