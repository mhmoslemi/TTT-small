import numpy as np
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increased from 45 for better resolution
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    w = 0.7  # inertia weight
    c1 = c2 = 1.5  # cognitive and social coefficients
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    # Initialize swarm around the current solution
    for i in range(n_particles):
        if i == 0:
            # First particle is the current solution
            h = h_current.copy()
        else:
            # Perturb the current solution
            h = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h, n_points)
        personal_best.append(h.copy())
        personal_best_scores.append(score)
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iteration
    iter_count = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Re-normalize
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Check global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    last_improvement = iter_count
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        iter_count += 1
        
        # Early stopping if no improvement for a while
        if stagnation_count > 100:
            break
            
        # Adaptive inertia
        if iter_count > 100:
            w = 0.7 + 0.3 * (1 - iter_count / max_iter)
    
    # Final evaluation
    final_score = evaluate_erdos_solution(global_best, n_points)
    
    return (global_best.tolist(), final_score, n_points)
