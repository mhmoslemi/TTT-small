import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization
    dx = 2.0 / n_points
    
    # Use initial construction if it exists

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_ineq(h):
        return np.minimum(h, 1.0) - h  # h <= 1
          
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq},
            {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}]
    
    # Optimization loop with progressive refinement
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts with better initializations
    max_restarts = 50
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Try different perturbation strategies
        if restart_count < 10:
            # Random perturbation with noise
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.03, n_points), 0, 1)
        elif restart_count < 30:
            # Structured perturbation
            h_perturbed = h_current.copy()
            # Slightly modify some segments
            segments = 5
            segment_size = n_points // segments
            for i in range(segments):
                start_idx = i * segment_size
                end_idx = start_idx + segment_size if i < segments - 1 else n_points
                if np.random.rand() > 0.5:
                    # Add small random change
                    h_perturbed[start_idx:end_idx] += np.random.normal(0, 0.01, end_idx - start_idx)
                    h_perturbed[start_idx:end_idx] = np.clip(h_perturbed[start_idx:end_idx], 0, 1)
        else:
            # More aggressive perturbation
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
        
        # Re-normalize
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except Exception as e:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
    
    # Final cleanup to ensure proper constraints
    h_final = best_h.copy()
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final computation
    final_c5 = evaluate_erdos_solution(h_final, n_points)
    
    return (h_final.tolist(), final_c5, n_points)
