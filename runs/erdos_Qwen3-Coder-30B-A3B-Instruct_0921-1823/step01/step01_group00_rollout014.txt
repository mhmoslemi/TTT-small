import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

    np.random.seed(seed)
    start_time = time.time()
    
    # parameters

    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Initialize with a slight perturbation of the current solution
    for i in range(n_particles):
        particle = h_current + np.random.normal(0, 0.02, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize global best
    g_best = h_current.copy()
    g_best_score = evaluate_erdos_solution(g_best, n_points)
    
    # PSO loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive_component = cognitive * r1 * (particle - particle)
            social_component = social * r2 * (g_best - particle)
            velocity = inertia * velocity + cognitive_component + social_component
            
            # Update position
            new_particle = particle + velocity
            new_particle = np.clip(new_particle, 0, 1)
            new_particle = new_particle * (n_points / 2) / np.sum(new_particle)
            
            # Evaluate new particle
            score = evaluate_erdos_solution(new_particle, n_points)
            
            # Update particle's best
            particles[i] = new_particle
            velocities[i] = velocity
            
            # Update global best if improved
            if score < g_best_score:
                g_best = new_particle.copy()
                g_best_score = score
    
    # Final refinement with local optimization
    h_final = g_best.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
        
    def constraint_ineq(h):
        return np.minimum(h, 1.0) - h
        
    cons = [{'type': 'eq', 'fun': constraint_eq},
            {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}]
    
    # Use SLSQP for final optimization
    try:
        res = minimize(
            obj,
            h_final,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200}
        )

            h_final = res.x
            g_best_score = obj(h_final)
    except:
        pass

    h_final = np.clip(h_final, 0, 1)

    final_c5 = evaluate_erdos_solution(h_final, n_points)
    
    return (h_final.tolist(), final_c5, n_points)
