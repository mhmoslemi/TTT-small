import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    return np.max(conv) * dx

def project_to_simplex(h, target_sum=1.0):
    """Project h to the feasible set with integral = target_sum."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if np.isclose(current_sum, target_sum):
        return h
    scale = target_sum / current_sum
    return h * scale

def compute_gradient(h, k_max):
    """Compute gradient with respect to maximum overlap shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    
    # For convolution h * (1-h), the gradient with respect to h[i] 
    # includes terms from each position j where h[j] contributes to the overlap at shift k_max
    # Specifically, at shift k_max, we're looking at h[i] * (1-h[i+k_max]) 
    # So the gradient for h[i] = 1 - h[i+k_max] where relevant
    for i in range(n):
        j = i + k_max
        if 0 <= j < n:
            grad[i] = -(1 - h[j])
    return grad
