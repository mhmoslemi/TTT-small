import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        # Create a structured pattern: alternating high/low segments
        segment_size = max(1, n_points // 9)  # Ensure at least 1
        for i in range(n_points):
            if (i // segment_size) % 2 == 0:
                h0[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
            else:
                h0[i] = 0.5 - 0.3 * np.sin(2 * np.pi * i / n_points)
        
        # Normalize to integral constraint
        h0 = h0 * (n_points / 2.0) / np.sum(h0)
        
    # Objective function
    def objective(h_vec):
        dx = 2.0 / n_points
        # Correlation of h with (1-h) for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return the maximum overlap (normalized)
        return np.max(corr) * dx
    
    # Constraint: integral equals 1
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    # Bound constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial optimization
    best_h = h0.copy()
    best_C = objective(best_h)
    
    # PSO-inspired random search
    for _ in range(1000):
        # Random perturbation
        h_new = best_h + np.random.normal(0, 0.05, n_points)
        h_new = np.clip(h_new, 0, 1)
        # Normalize
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        C_new = objective(h_new)
        if C_new < best_C:
            best_h = h_new.copy()
            best_C = C_new
        
        if time.time() - start_time > budget_s * 0.8:
            break
    
    # Local refinement with L-BFGS-B
    options = {'maxiter': 1000, 'disp': False}
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                       constraints=cons, options=options, callback=lambda x: None)
        if res.success and res.fun < best_C:
            best_h = res.x.copy()
            best_C = res.fun
    except:
        pass
    
    # More refined local optimization
    if time.time() - start_time < budget_s * 0.9:

            pass
