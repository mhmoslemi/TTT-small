from scipy.optimize import minimize
import time

    # Gradient-based local optimization with critical shift focusing
    def gradient_step(h, k):
        """Compute gradient of overlap for shift k, and update h"""
        dx = 2.0 / n_points
        # Compute overlap for shift k
        overlap = np.sum(h[:-k] * (1 - h[k:]))
        # Gradient of overlap w.r.t h[i]
        grad = np.zeros_like(h)
        for i in range(len(h)-k):
            grad[i] += (1 - h[i+k])  # d/dh[i] of h[i]*(1-h[i+k])
            grad[i+k] -= h[i]        # d/dh[i+k] of h[i]*(1-h[i+k])
        return grad * dx
    
    def compute_all_overlaps(h):
        """Compute all overlaps for different shifts"""
        dx = 2.0 / n_points
        overlaps = []
        for k in range(1, n_points):
            # Compute overlap for shift k
            if k < len(h):
                overlap = np.sum(h[:-k] * (1 - h[k:]))
                overlaps.append((overlap * dx, k))
        return overlaps
    
    # Simulated Annealing inspired approach: random perturbations
    def perturb_solution(h, temperature):
        proposed = h.copy()
        idx = np.random.randint(0, n_points)
        # Small random change
        delta = np.random.normal(0, 0.01 * temperature)
        proposed[idx] = np.clip(proposed[idx] + delta, 0, 1)
        # Re-normalize
        proposed *= (n_points/2.0) / np.sum(proposed)
        return proposed
    
    # Iterative refinement using simulated annealing approach
    current_h = best_h.copy()
    current_C = best_C
    temperature = 1.0
    iter_count = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Reduce temperature
        temperature = max(0.001, temperature * 0.999)

        new_h = perturb_solution(current_h, temperature)

            current_C = new_C

                best_C = new_C

            prob = np.exp(-(new_C - current_C) / temperature)

                current_h = new_h
                current_C = new_C

    bounds = [(0, 1) for _ in range(n_points)]

        C = evaluate_C5(h_vec, n_points)
        return C
        
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

            refined_C = evaluate_C5(refined_h, n_points)
            if refined_C < best_C:
                best_h = refined_h
                best_C = refined_C
    except:
        pass
    
    return best_h.copy(), best_C, n_points

def evaluate_C5(h, n_points):
    """Evaluate C5 for given h"""
    dx = 2.0 / n_points
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is at the center of the full correlation
    C = np.max(corr) * dx
    return C
