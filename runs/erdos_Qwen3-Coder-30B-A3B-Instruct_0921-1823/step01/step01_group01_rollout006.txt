import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    return np.max(conv) * dx

def project_to_simplex(h, target_sum=1.0):
    """Project h to the feasible set with integral = target_sum."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if np.isclose(current_sum, target_sum):
        return h
    scale = target_sum / current_sum
    return h * scale

        max_k = max_idx - (len(h) - 1)
        
        # Construct shifted version of h for the max_k shift
        shifted_h = np.roll(h, max_k)
        
        # Compute gradient for each position based on the contribution to overlap
        grad = np.zeros(len(h))
        for i in range(len(h)):
            # For each h[i], it contributes to overlap when shifted by max_k,
            # which means when x + max_k = i, so x = i - max_k
            if 0 <= i - max_k < len(h):
                grad[i] = shifted_h[i - max_k]
                
        # Descend along the negative gradient
        h_new = h - lr * grad

        h = h_new
        
        # Early stopping
        if abs(c5_new - evaluate_c5(h)) < tolerance:
            break
            
        # Adapt learning rate (meta-learning)
        if iteration > 1 and abs(c5_new - evaluate_c5(h)) < 1e-5:
            lr *= 0.9
            
    return best_h, best_c5
