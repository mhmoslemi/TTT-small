import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        dx = 2.0 / n_points
        h0 = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Evaluation function
    def evaluate(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Optimization using gradient-based method
    def optimize_step(h_start):
        def objective(h):
            return evaluate(h)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Use L-BFGS-B for better convergence
        res = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-8}, callback=lambda x: None)
        
        if res.success:
            return res.x.copy(), res.fun
        return h_start.copy(), evaluate(h_start)
    
    # Iterative refinement applying optimizations to critical shifts
    best_h = h0.copy()
    best_C = evaluate(best_h)
    
    # Track time
    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed
    
    iter_count = 0
    while elapsed < budget_s * 0.95:  # Leave some buffer for finalization
        # Refine using optimization
        h_new, C_new = optimize_step(best_h)
        
        if C_new < best_C:
            best_h = h_new
            best_C = C_new
        
        elapsed = time.time() - start_time
        iter_count += 1
        
        # Occasionally add small random perturbations to avoid local minima
        if iter_count % 10 == 0 and elapsed < budget_s * 0.9:
            # Add small random perturbation to explore
            perturbation = np.random.normal(0, 0.001, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Normalize
            h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
            
            _, C_perturbed = optimize_step(h_perturbed)
            if C_perturbed < best_C:
                best_h = h_perturbed
                best_C = C_perturbed
        
        # Early exit if improvement is minimal
        if abs(best_C - evaluate(best_h)) < 1e-6:
            break
            
        elapsed = time.time() - start_time
    
    # Final correction with mixture of methods
    h_final, C_final = optimize_step(best_h)
    if C_final < best_C:
        best_h = h_final
        best_C = C_final
    
    # Ensure final constraint satisfaction
    final_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2.0) / final_sum
    
    return best_h.copy(), best_C, n_points
