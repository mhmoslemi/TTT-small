import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Create a structured initial construction
        n_points = 45
        dx = 2.0 / n_points
        h0 = np.zeros(n_points)
        # Create a pattern with alternating high/low segments
        # First, let's create a sawtooth pattern to reduce overlaps
        for i in range(n_points):
            x = i * dx
            h0[i] = 0.5 + 0.3 * np.sin(np.pi * x)
        h0 = np.clip(h0, 0, 1)
        # Normalize to get correct integral
        h0 = h0 * (n_points / 2.0) / np.sum(h0)

    def compute_C5(h_vec):
        dx = 2.0 / n_points
        # Correlate h with (1 - h) for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Try multiple local optimization runs with different perturbations
    best_h = h0.copy()
    best_C = compute_C5(h0)
    
    # Do several optimization attempts with different starting points
    for attempt in range(20):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Slightly perturb the current best solution
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize again
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Optimize this perturbed version using L-BFGS-B
        try:
            res = minimize(
                compute_C5,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200}
            )
            if res.success:
                C = res.fun
                if C < best_C:
                    best_h = res.x.copy()
                    best_C = C
        except:
            pass

    # Final verification
    final_C = compute_C5(best_h)
    
    return best_h.copy(), final_C, n_points
