        dx = 2.0 / n_points

        corr = np.correlate(h, 1 - h, mode='full')

        return C

    def compute_gradient(h, shift):
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        # For shift k, the derivative of h[i] * (1 - h[i+k]) w.r.t. h[i] is (1 - h[i+k])
        # and w.r.t. h[i+k] is -h[i]
        # We accumulate these for the overlap constraint corresponding to shift k
        for i in range(len(h)):
            j = i + shift
            if 0 <= j < len(h):
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        return grad

    max_iter = 10000
    iter_count = 0
    time_limit = budget_s * 0.95  # Leave some time for cleanup

    for it in range(max_iter):
        elapsed = time.time() - start_time

            break

        corr = np.correlate(h_best, 1 - h_best, mode='full')

        shift = max_idx - (n_points - 1)

        def func(h):
            dx = 2.0 / n_points
            # Compute only the particular constraint term for this shift

                    total += h[i] * (1 - h[i + shift])
