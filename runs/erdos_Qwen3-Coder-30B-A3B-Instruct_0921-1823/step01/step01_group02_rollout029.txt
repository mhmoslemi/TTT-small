def adjust_to_integral(h_vec, dx):
    # Adjust to make sure sum(h) * dx = 1
    current_sum = np.sum(h_vec)
    target_sum = 1.0 / dx
    scale = target_sum / current_sum if current_sum != 0 else 1.0
    h_new = h_vec * scale
    # Clamp to [0,1]
    h_new = np.clip(h_new, 0, 1)
    return h_new
