import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Objective function
    def objective(h_vec):
        # Compute all correlations: correlate h with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        C = np.max(corr) * dx
        return C

    # Constraint function: integral = 1
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for minimization
    cons = {'type': 'eq', 'fun': constraint_sum}

    # First optimization with SLSQP
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
        if res.success:
            best_h = res.x.copy()
            best_C = res.fun
        else:
            best_h = h0.copy()
            best_C = objective(h0)
    except:
        best_h = h0.copy()
        best_C = objective(h0)
    
    # Additional local refinement with different method
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success and res.fun < best_C:
            best_h = res.x.copy()
            best_C = res.fun
    except:
        pass
    
    # Further refinement by iterative optimization on critical shifts
    # Track time
    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed
    
    if remaining_time < 10:
        return best_h.copy(), best_C, n_points
    
    # Add small perturbations to try escaping local minima
    perturbation_magnitude = 0.02
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        pert_h = best_h + np.random.normal(0, perturbation_magnitude, n_points)
        pert_h = np.clip(pert_h, 0, 1)
        # Ensure integral constraint
        scale = (n_points/2.0) / np.sum(pert_h)
        pert_h = pert_h * scale
        
        try:
            res = minimize(objective, pert_h, method='L-BFGS-B', bounds=bounds, constraints=cons,

                best_C = res.fun

            pass

    try:
        res = minimize(objective, best_h, method='trust-constr', bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'gtol': 1e-6})
        if res.success and res.fun < best_C:
            best_h = res.x.copy()
            best_C = res.fun
    except:
        pass
    
    return best_h.copy(), best_C, n_points
