import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    # Optimization loop with multiple restarts using different strategies
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts with different perturbation strategies
    max_restarts = 50
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Use different perturbation strategies
        if restart_count % 3 == 0:
            # Small random noise
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.03, n_points), 0, 1)
        elif restart_count % 3 == 1:
            # Medium perturbations
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.1, n_points), 0, 1)
        else:
            # Larger perturbations with more structure
            noise = np.random.normal(0, 0.15, n_points)
            # Add a trend to break symmetry
            trend = np.linspace(-0.1, 0.1, n_points)
            h_perturbed = np.clip(h_current + noise + trend, 0, 1)
        
        # re-normalize
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            # Try SLSQP optimizer first
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-8}
            )
            
            if not res.success:
                # Fallback to COBYLA if SLSQP fails
                res = minimize(
                    objective,
                    h_perturbed,
                    method='COBYLA',
                    constraints=cons,
                    options={'maxiter': 300}
                )
            
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
        
        # Occasionally try a smoothness constraint
        if restart_count % 7 == 0:
            # Apply local smoothing to current best
            h_smooth = np.convolve(best_h, [0.2, 0.6, 0.2], mode='same')
            h_current = np.clip(h_smooth, 0, 1)
    
    # Final computation with smoothing if needed
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)
