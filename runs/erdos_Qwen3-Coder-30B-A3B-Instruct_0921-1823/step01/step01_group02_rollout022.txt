import numpy as np
from scipy.optimize import minimize
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Make it a step function
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        
    dx = 2.0 / n_points
    
    def evaluate_overlap(h_vec):
        # Evaluate all overlaps for max
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def penalty_objective(h_vec, penalty_shifts, penalty_weights, current_max):
        # Compute overlap for all shifts and add penalty for the problematic ones
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = corr * dx
        max_overlap = np.max(overlaps)
        penalty = 0.0
        for i, (shift, weight) in enumerate(zip(penalty_shifts, penalty_weights)):
            if shift < len(overlaps):
                penalty += weight * max(0, overlaps[shift] - current_max)**2
        return max_overlap + penalty

    # Adaptive gradient-based optimization
    best_h = h0.copy()
    best_c5 = evaluate_overlap(best_h)
    
    # Initial feasibility check
    if np.abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        # Re-normalize
        best_h = best_h * (1.0 / (np.sum(best_h) * dx)) 

    # Iterative refinement with adaptive penalties
    max_iter = 500
    patience = 0
    min_patience = 10
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Keep track of latest improvement
        old_c5 = best_c5
        current_h = best_h.copy()
        
        # Run optimization with adaptive penalties
        # Identify problematic shifts
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx
        max_shift = np.argmax(overlaps)
        max_overlap = np.max(overlaps)
        
        # Add a strong penalty to max_shift
        penalty_weights = np.zeros(len(overlaps))
        penalty_weights[max_shift] = 100.0 * (1.0 + max_overlap)
        
        options = {'maxiter': 100, 'disp': False}
        
        # Use a higher learning rate for initial refinement
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(
                penalty_objective, 
                current_h,
                args=([max_shift], penalty_weights, max_overlap),
                method='L-BFGS-B',
                bounds=bounds,
                options=options
            )
            if res.success:
                proposed_h = res.x.copy()
                # Ensure feasibility
                if np.abs(np.sum(proposed_h) * dx - 1.0) > 1e-6:
                    proposed_h = proposed_h * (1.0 / (np.sum(proposed_h) * dx))
                new_c5 = evaluate_overlap(proposed_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = proposed_h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except:
            patience += 1
        
        if patience > min_patience:
            # Perturb slightly and restart to escape local optima
            noise = np.random.normal(0, 0.01, n_points)
            noise = np.clip(noise, -0.1, 0.1)
            best_h = np.clip(best_h + noise, 0, 1)
            # Re-normalize
            best_h = best_h * (1.0 / (np.sum(best_h) * dx))
            patience = 0
            
        # Re-evaluate best solution if time allows
        if time.time() - start_time > budget_s - 0.1:
            break
            
    # Final verification and CVXPY refinement
    try:
        import cvxpy as cp
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2.0)
        
        # Overlap constraints, focusing on most problematic shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlaps = corr * dx
        max_shift = np.argmax(overlaps)
        
        # Add some of the highest overlaps as constraints
        top_shifts = np.argsort(overlaps)[-5:]
        for shift in top_shifts:
            expr = 0
            for i in range(n_points):
                j = i + shift
                if j < n_points:
                    expr += h[i] * (1 - h[j])
            constraints.append(expr <= C * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                refined_h = h.value.copy()
                refined_c5 = prob.value
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except Exception as _:
            pass
    except:
        pass
        
    # Final normalization to satisfy sum constraint exactly
    best_h = best_h * (1.0 / (np.sum(best_h) * dx))
    
    # Ensure constraint
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6
    
    return best_h.copy(), best_c5, n_points
