import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization
    dx = 2.0 / n_points
    
    # Use initial construction if it exists

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize with variation around the current solution
    for i in range(n_particles):
        particles[i] = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
        # Normalize each particle
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Best positions for each particle
    pbest_positions = particles.copy()
    pbest_scores = np.array([evaluate_erdos_solution(p, n_points) for p in particles])
    
    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO iterations
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (pbest_positions[i] - particles[i]) + 
                            social * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            # Ensure sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_score = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        iter_count += 1
        
        # Early stopping
        if stagnation_count >= max_stagnation:
            break
            
        # Adapt inertia weight to prevent stagnation
        if stagnation_count > 50:
            inertia = min(0.9, inertia * 1.02)
    
    # Final evaluation
    final_h = gbest_position.copy()
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    # Ensure final constraint satisfaction
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)
