import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    return np.max(conv) * dx

def project_to_simplex(h, target_sum=1.0):
    """Project h to the feasible set with integral = target_sum."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if np.isclose(current_sum, target_sum):
        return h
    scale = target_sum / current_sum
    return h * scale

def compute_gradient(h, k_max):
    """Compute the gradient w.r.t. h for a specific shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    
    # For each position x, compute part of gradient due to k_max shift
    # The overlap at shift k_max is h(x) * (1 - h(x + k_max))
    # Gradient w.r.t. h(x) = 1 - h(x + k_max) if x + k_max is valid
    # and w.r.t. h(x + k_max) = -h(x) if x is valid
    
    # Handle the shifted contributions for each x
    for x in range(n):
        # Contribution to gradient at x from h[x + k_max] term
        if 0 <= x + k_max < n:
            grad[x] += 1.0 - h[x + k_max]  # gradient of h(x)(1-h(x+k_max)) w.r.t h(x)
    
    # To compute partial derivatives we need to be more precise
    # But for optimization, this is the core idea
    
    return grad
