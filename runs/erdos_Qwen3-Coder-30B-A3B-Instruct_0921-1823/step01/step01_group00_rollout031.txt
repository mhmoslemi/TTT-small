import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization
    dx = 2.0 / n_points
    
    # Use initial construction if it exists

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Particle Swarm Optimization
    n_particles = 30
    max_iter = 500
    inertia_w = 0.7
    cognitive_c1 = 1.5
    social_c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Initialize with current h_current as center
    for i in range(n_particles):
        # Add small random perturbations
        perturbation = np.random.normal(0, 0.1, n_points)
        particle = np.clip(h_current + perturbation, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(particle.copy())
        pbest_values.append(evaluate_erdos_solution(particle, n_points))
    
    gbest_position = pbest_positions[np.argmin(pbest_values)]
    gbest_value = min(pbest_values)
    
    # PSO iterations
    stagnation_count = 0
    stagnation_limit = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = cognitive_c1 * r1 * (pbest_positions[i] - particles[i])
            social = social_c2 * r2 * (gbest_position - particles[i])
            velocities[i] = (inertia_w * velocities[i] + 
                            cognitive + social)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary constraints
            particles[i] = np.clip(particles[i], 0, 1)
            # Apply sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            current_value = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        # Early termination if no improvement
        if stagnation_count >= stagnation_limit:
            break
            
        # Adaptive parameters
        if stagnation_count >= 50:
            inertia_w = min(0.9, inertia_w + 0.01)
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(gbest_position, n_points)
    
    return (gbest_position.tolist(), final_c5, n_points)
