import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    return np.max(conv) * dx

def project_to_simplex(h, target_sum=1.0):
    """Project h to the feasible set with integral = target_sum."""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if np.isclose(current_sum, target_sum):
        return h
    scale = target_sum / current_sum
    return h * scale

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        n_points = 47
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        h = project_to_simplex(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Hyperparameters
    max_iter = 500000
    tolerance = 1e-6
    last_improvement = 0
    adaptive_lr = 0.1
    decay_factor = 0.9999
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Evaluate current C5
        c5 = evaluate_c5(h)
        
        # Keep track of best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Find the maximum overlap index
        conv = signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(conv)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient more accurately
        # Gradient for h[x] is sum of contributions from all positions where h[x+k] appears
        grad = np.zeros(n_points)
        
        # For each position in the convolution which achieves max overlap, 
        # compute the gradient contribution from that overlap
        # The gradient of integral h(x)(1-h(x+k)) w.r.t. h(x) is (1-h(x+k))
        # The full derivative over all k values can be approximated by the current max k
        
        # For each index in the convolution, compute gradient estimate
        # Create a proper gradient vector
        if max_k >= 0:
            # h[i] contributes to overlap with 1-h[i+max_k]
            for i in range(n_points - max_k):
                grad[i] += (1 - h[i + max_k])
        else:
            # h[i+(-max_k)] contributes to overlap with 1-h[i]
            for i in range(-max_k, n_points):
                grad[i] += (1 - h[i + max_k])
        
        # Make adjustment based on gradient
        h_new = h - adaptive_lr * grad
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to feasible simplex
        h_new = project_to_simplex(h_new)
        
        # Update with projected version
        h = h_new
        
        # Adaptive learning rate
        adaptive_lr *= decay_factor
        
        # Periodic restarts
        if iteration - last_improvement > 2000 and iteration > 10000:
            # Add noise to escape local optima
            noise = np.random.normal(0, 0.001, n_points)
            noise -= np.mean(noise)  # Ensure mean is 0
            h = h + noise
            h = np.clip(h, 0, 1)
            h = project_to_simplex(h)
            last_improvement = iteration
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
