from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points
    best_h = h0.copy()
    best_C = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    # Precompute the convolution indices for efficiency
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def gradient_overlap(h_vec, k):
        # Compute gradient of overlap for shift k
        # For shift k: overlap = sum(h[i] * (1 - h[i+k]))
        # Gradient w.r.t. h[i] = 1 - h[i+k] if i+k < n_points, 0 otherwise
        # But we do this for the maximum overlap case
        if k >= n_points:
            return np.zeros_like(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(n_points - k):
            grad[i] += (1 - h_vec[i + k])
        return grad
    
    # Stochastic gradient descent with cycling through critical shifts
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Find current critical shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_shift = np.argmax(corr)
        max_overlap = corr[max_shift] * dx
        
        # The actual shift k is max_shift - (n_points - 1) 
        # due to how correlate works, so shift in [0, 2*dx)
        # But we actually want to optimize at that specific shift
        shift_idx = max_shift - (n_points - 1)
        if shift_idx < 0: 
            shift_idx += n_points
        
        # Directional minimization along the critical shift
        # Try smooth changes with descent step size
        lr = 0.01
        steps = 200
        for s in range(steps):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute the gradient of the overlap for shift k
            # Using finite differences for stabilization (since shift doesn't have fixed form)
            # For simplicity: we're not computing subgradients but use local perturbation
            
            # While we can't easily get analytical gradient from shift, we can try
            # using the existing optimal solution structure and improve locally
            
            # Simple hill-climbing scheme
            # One approach is to optimize the signature that contributes to max overlap
            
            # Try finding the most active elements that contribute to peak correlation
            # this is cheaper than re-evaluating every shift
            if shift_idx < n_points - 1:
                # Approximately compute gradient at abscissa where maximum occurs
                # This is approximated: we use simple finite difference from current best
                h_test = best_h.copy()
                # Add small perturbation to the elements that cause high cross correlation
                for i in range(max(0, shift_idx), min(n_points, shift_idx + 10)):

                    h_test[i] += 0.001
                    if h_test[i] > 1.0:
                        h_test[i] = 1.0
                
                # Ensure sum constraint
                h_test = h_test * (n_points / 2.0) / np.sum(h_test)
                
                # New overlap
                test_overlap = compute_overlap(h_test)

                    # Accept the change
                    best_h = h_test.copy()
                    best_C = test_overlap
        
        # Periodic full recompute to prevent stagnation
        if iteration % 10 == 0:
            # Construct gradient direction manually based on max overlap
            # For shift k, correlation = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
            corr_val = np.correlate(best_h, 1 - best_h, mode='full')
            max_idx = np.argmax(corr_val)
            best_shift = max_idx - (n_points - 1)
            if best_shift < 0:
                best_shift += n_points
            
            # Just simple optimization of full function using nonlinear method
            # But save large steps for full evaluation
            
            # Use scipy minimize
            def obj(h): 
                return compute_overlap(h)
            
            def constraint(h):
                return np.sum(h) - n_points/2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint}
            
            try:
                x0 = best_h.copy()
                result = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                                  constraints=cons, options={'maxiter': 100, 'ftol': 1e-6})
                if result.success:
                    attempt_h = result.x.copy()
                    # Adjust for constraint
                    attempt_h = attempt_h * (n_points/2.0) / np.sum(attempt_h)
                    candidate_C = compute_overlap(attempt_h)
                    if candidate_C < best_C:
                        best_h = attempt_h
                        best_C = candidate_C
            except:
                pass

    # Final check for any last improvements
    corr_val = np.correlate(best_h, 1 - best_h, mode='full')
    best_C = np.max(corr_val) * dx
    
    return best_h.copy(), best_C, n_points
