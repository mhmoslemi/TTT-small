import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Create a structured initial guess
        n_points = 45
        h0 = np.zeros(n_points)
        dx = 2.0 / n_points
        
        # Pattern: alternating high/low segments
        segment_size = 5
        for i in range(n_points):
            if (i // segment_size) % 2 == 0:
                h0[i] = 0.6
            else:
                h0[i] = 0.2
        
        # Adjust to meet integral constraint
        current_sum = np.sum(h0) * dx
        h0 = h0 * (1.0 / current_sum)
    
    def compute_C5(h_vec):
        dx = 2.0 / n_points
        # Compute correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Particle Swarm Optimization approach
    def pso_optimize():
        # PSO parameters
        n_particles = 30
        n_iter = 200
        w = 0.7  # inertia weight
        c1 = 1.5  # cognitive parameter
        c2 = 1.5  # social parameter
        
        # Initialize particles
        particles = np.random.rand(n_particles, n_points) * 0.5 + 0.25
        velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
        
        # Adjust particles to satisfy integral constraint
        for i in range(n_particles):
            current_sum = np.sum(particles[i]) * dx
            particles[i] = particles[i] * (1.0 / current_sum)
        
        # Personal best positions
        pbest = particles.copy()
        pbest_fitness = np.array([compute_C5(p) for p in pbest])
        
        # Global best
        gbest_idx = np.argmin(pbest_fitness)
        gbest = pbest[gbest_idx].copy()
        gbest_fitness = pbest_fitness[gbest_idx]
        
        for _ in range(n_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply bounds
                particles[i] = np.clip(particles[i], 0, 1)
                
                # Adjust to constraint
                current_sum = np.sum(particles[i]) * dx
                particles[i] = particles[i] * (1.0 / current_sum)
                
                # Evaluate fitness
                fitness = compute_C5(particles[i])
                if fitness < pbest_fitness[i]:
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = fitness
                    if fitness < gbest_fitness:
                        gbest = particles[i].copy()
                        gbest_fitness = fitness
        
        return gbest, gbest_fitness
    
    # Run PSO
    best_h, best_C = pso_optimize()
    
    # Local refinement
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 300}
    
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                      options=options, callback=lambda x: None)
        if res.success:
            refined_h = res.x.copy()
            refined_C = compute_C5(refined_h)
            if refined_C < best_C:
                best_h = refined_h
                best_C = refined_C
    except:
        pass
    
    # Final validation
    final_C = compute_C5(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h.copy(), best_C, n_points
    
    return best_h.copy(), final_C, n_points
