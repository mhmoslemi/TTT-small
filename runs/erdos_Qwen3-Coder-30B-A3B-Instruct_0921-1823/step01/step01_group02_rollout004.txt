import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5  # Start with uniform distribution

    dx = 2.0 / n_points
    
    def evaluate_overlap(h_vec):
        # Evaluate all overlaps with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return max overlap times dx
        return np.max(corr) * dx

    def penalty_function(h_vec, shift_indices, penalty_weight=1.0):
        # Compute penalty for selected shifts (for training focus)
        penalty = 0
        for k in shift_indices:
            # Manual computation of convolution at shift k
            overlap = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    overlap += h_vec[i] * (1 - h_vec[j]) 
            penalty += penalty_weight * max(0, overlap * dx - 0.38085) ** 2
        return penalty

    # Quadratic penalty with adaptive weights
    def adaptive_objective(h_vec):
        max_overlap = evaluate_overlap(h_vec)
        # Apply penalty to the maximum overlap scenario
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)  # Middle element corresponds to shift 0
        # Broad penalty on overlap
        shifted_corr = corr - 0.38085
        penalty = np.sum(np.maximum(shifted_corr, 0) ** 2) * 0.1
        return max_overlap + penalty
        
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h0.copy()
    best_c5 = evaluate_overlap(best_h)
    
    # Iterative refinement with limited function calls
    last_improvement = 0
    iteration = 0
    inner_it = 0
    while time.time() - start_time < budget_s - 1:
        # Short cycle of L-BFGS-B for gradient-based update
        options = {'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-6}
        try:
            res = minimize(adaptive_objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, 
                           options=options)
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun
                last_improvement = iteration
        except:
            pass
        
        # Occasionally, fine-tune with a CVXPY solver
        if iteration % 10 == 0 and iteration > 0:
            try:
                h_cvx = cp.Variable(n_points)
                C = cp.Variable()
                
                constraints = [h_cvx >= 0, h_cvx <= 1]
                constraints.append(cp.sum(h_cvx) == n_points / 2.0) 
                
                # Use empirical maximum shift constraint
                corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
                max_shift_idx = np.argmax(corr_vals)
                max_shift = max_shift_idx - (n_points - 1)
                
                # Only include a few constraints that are most impactful
                for k in range(1, min(10, n_points)):
                    expr = 0
                    for i in range(n_points):
                        j = i + k
                        if j < n_points:
                            expr += h_cvx[i] * (1 - h_cvx[j])
                    constraints.append(expr <= C * dx)
                
                objective = cp.Minimize(C)
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    best_h = h_cvx.value.copy()
                    best_c5 = float(C.value)
            except:
                pass

        iteration += 1
        # Early stopping if no significant improvement
        if iteration - last_improvement > 20 and best_c5 < 0.385:
            break

    # Final check with full overlap evaluation 
    best_c5 = evaluate_overlap(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h.copy(), best_c5, n_points
        
    return best_h.copy(), best_c5, n_points
