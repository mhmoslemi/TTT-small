from scipy.optimize import minimize
import time

    start_time = time.time()

        # Create a piecewise constant function with alternating high/low segments
        # This should help reduce overlaps
        segment_length = n_points // 4
        h0[:segment_length] = 0.8
        h0[segment_length:2*segment_length] = 0.2
        h0[2*segment_length:3*segment_length] = 0.8
        h0[3*segment_length:] = 0.2
        
        # Adjust to satisfy constraint sum(h) * dx = 1
        current_sum = np.sum(h0)
        h0 = h0 * (n_points / 2.0) / current_sum
    
    def evaluate_overlap(h_vec):
        dx = 2.0 / n_points
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Local optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        return evaluate_overlap(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

    cons = {'type': 'eq', 'fun': constraint_sum}

    # Try several random restarts to avoid local minima
    best_h = h0.copy()
    best_C = evaluate_overlap(best_h)
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation for restart
        h_perturbed = h0 + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Normalize to satisfy constraint
        current_sum = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (n_points / 2.0) / current_sum

        # Local optimization
        try:
            res = minimize(objective, h_perturbed, method='L-BFGSB', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100})
            if res.success:
                C_test = evaluate_overlap(res.x)
                if C_test < best_C:
                    best_h = res.x.copy()
                    best_C = C_test
        except:
            pass

    # Final refinement
    if time.time() - start_time < budget_s - 1:
        try:
            res = minimize(objective, best_h, method='L-BFGSB', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200})
            if res.success:
                final_C = evaluate_overlap(res.x)
                if final_C < best_C:
                    best_h = res.x.copy()
                    best_C = final_C
        except:
            pass

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h.copy(), best_C, n_points
