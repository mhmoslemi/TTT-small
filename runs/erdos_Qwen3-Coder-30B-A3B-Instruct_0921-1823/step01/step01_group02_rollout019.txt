from scipy.optimize import minimize
import time

    start_time = time.time()

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

        # Compute gradient with respect to shift k
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = i + k
            if j < len(h):
                grad[i] += (1 - h[j])
            j = i - k
            if j >= 0:
                grad[i] += h[j]
        return grad

    # Main iterative optimization
    h_current = h0.copy()
    best_h = h_current.copy()
    best_C = compute_overlap(h_current)
    
    # Precompute shifts for performance
    shifts = list(range(1, n_points))
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Find critical shift
        max_overlap = 0
        critical_k = 1
        
        for k in shifts:
            overlap = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    overlap += h_current[i] * (1 - h_current[j])
            if overlap > max_overlap:
                max_overlap = overlap
                critical_k = k
        
        # Optimize for the critical shift using projected gradient
        def obj_and_grad(h):
            # Compute objective for critical shift
            obj = 0
            for i in range(n_points):
                j = i + critical_k
                if j < n_points:
                    obj += h[i] * (1 - h[j])
            
            # Compute gradient
            grad = np.zeros_like(h)
            for i in range(n_points):
                j = i + critical_k
                if j < n_points:
                    grad[i] += (1 - h[j])
                j = i - critical_k
                if j >= 0:
                    grad[i] += h[j]
            
            return obj, grad
        
        # Use simple gradient ascent with projection
        h_new = h_current.copy()
        step_size = 0.01
        
        for _ in range(100):  # inner iterations
            if time.time() - start_time > budget_s - 0.1:
                break
                
            obj, grad = obj_and_grad(h_new)
            h_new = h_new + step_size * grad
            
            # Project to feasible region [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project to unit integral constraint
            current_sum = np.sum(h_new)
            h_new = h_new * (n_points/2.0) / current_sum
            
            # Check for convergence
            if np.linalg.norm(grad) < 1e-8:
                break
        
        # Verify the new solution
        new_C = compute_overlap(h_new)
        if new_C < best_C:
            best_h = h_new.copy()
            best_C = new_C
            
        h_current = h_new.copy()
    
    # Final refinement with scipy
    def objective(h_vec):
        dx = 2.0 / n_points
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100}, callback=lambda x: None)
        if res.success:
            final_C = res.fun
            if final_C < best_C:
                best_h = res.x.copy()
                best_C = final_C
    except:
        pass
    
    return best_h.copy(), best_C, n_points
