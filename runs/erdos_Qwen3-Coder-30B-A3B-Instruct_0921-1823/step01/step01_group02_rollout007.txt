import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default construction with structured initialization
        n_points = 45
        dx = 2.0 / n_points
        # Create a periodic, symmetric pattern
        h0 = np.ones(n_points)
        # Alternate pattern: high-low-high-low...
        for i in range(n_points):
            if i % 4 < 2:
                h0[i] = 0.7
            else:
                h0[i] = 0.3
        # Scale to meet integral constraint
        h0 = h0 * (n_points / 2.0) / np.sum(h0)
    
    # Helper function to compute overlap integral
    def compute_overlap(h_vec):
        dx = 2.0 / n_points
        # Correlation of h with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Global optimization using PSO-like approach with L-BFGS-B
    best_h = h0.copy()
    best_C = compute_overlap(best_h)
    
    # Local refinement
    options = {'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
    
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        try:
            res = minimize(
                compute_overlap,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options=options,
                callback=lambda x: None
            )
            if res.success:
                new_h = res.x.copy()
                new_C = compute_overlap(new_h)
                if new_C < best_C:
                    best_h = new_h
                    best_C = new_C
        except:
            pass
        
        # Perturb and try again
        if time.time() - start_time > budget_s - 1:
            break
        perturbed_h = best_h + np.random.normal(0, 0.01, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Re-scale to satisfy constraint
        perturbed_h = perturbed_h * (n_points / 2.0) / np.sum(perturbed_h)
        new_C = compute_overlap(perturbed_h)
        if new_C < best_C:
            best_h = perturbed_h
            best_C = new_C
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h.copy(), best_C, n_points
    
    return best_h.copy(), best_C, n_points
