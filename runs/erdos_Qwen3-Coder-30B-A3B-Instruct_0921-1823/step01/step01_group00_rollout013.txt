import numpy as np
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    w = 0.7  # inertia weight
    c1 = c2 = 1.5  # cognitive and social parameters
    v_max = 0.1  # maximum velocity
    
    # Initialize particles
    particles = []
    velocities = []
    best_positions = []
    best_values = []
    
    # Initialize swarm around initial solution
    for i in range(n_particles):
        if i == 0:
            # First particle is the initial solution
            pos = h_current.copy()
        else:
            # Others with small random perturbation
            pos = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
        
        # normalize
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        best_positions.append(pos.copy())
        best_values.append(evaluate_erdos_solution(pos, n_points))
    
    # Global best
    global_best_idx = np.argmin(best_values)
    global_best = best_positions[global_best_idx].copy()
    global_best_value = best_values[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # For each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (best_positions[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Enforce sum constraint exactly
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            value = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if value < best_values[i]:
                best_values[i] = value
                best_positions[i] = particles[i].copy()
            
            # Update global best
            if value < global_best_value:
                global_best_value = value
                global_best = particles[i].copy()
    
    # Return the best solution found
    final_c5 = evaluate_erdos_solution(global_best, n_points)
    
    return (global_best.tolist(), final_c5, n_points)
