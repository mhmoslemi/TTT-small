import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # PSO parameters
    n_points = 60  # Increase discretization for better resolution
    n_particles = 30
    max_iter = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    dx = 2.0 / n_points
    
    # Initialize particles around the initial construction or uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Normalize to sum(h) * dx = 1 
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particle positions and velocities
    positions = np.empty((n_particles, n_points))
    velocities = np.empty((n_particles, n_points))
    pbest_positions = np.empty((n_particles, n_points))
    pbest_values = np.full(n_particles, np.inf)
    
    for i in range(n_particles):
        # Start from variations of the initial solution
        noise = np.random.normal(0, 0.05, n_points)
        pos = np.clip(h_init + noise, 0, 1)
        # Ensure constraint sum(h) * dx = 1  
        pos = pos * (n_points / 2) / np.sum(pos)
        
        positions[i] = pos
        pbest_positions[i] = pos.copy()
        pbest_values[i] = evaluate_erdos_solution(pos, n_points)
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations  
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            vel_cognitive = cognitive * r1 * (pbest_positions[i] - positions[i])
            vel_social = social * r2 * (gbest_position - positions[i])
            velocities[i] = inertia * velocities[i] + vel_cognitive + vel_social
            
            # Update position
            new_pos = positions[i] + velocities[i]
            
            # Apply bounds
            new_pos = np.clip(new_pos, 0, 1)
            
            # Ensure constraint sum(h) * dx = 1
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            # Evaluate
            new_value = evaluate_erdos_solution(new_pos, n_points)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest_values[i] = new_value
                pbest_positions[i] = new_pos.copy()
                
                # Update global best
                if new_value < gbest_value:
                    gbest_value = new_value
                    gbest_position = new_pos.copy()
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
        
        # Early stopping if no improvement  
        if iter_no_improve > 100:
            break
    
    # Final refinement
    final_h = np.clip(gbest_position, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)
