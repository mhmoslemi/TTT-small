import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization for better precision
    dx = 2.0 / n_points
    
    # Use initial construction if it exists

    else:

    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    # Optimization loop
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts with different strategies
    max_restarts = 50
    restart_count = 0
    
    # Initialize with different patterns for diversity
    patterns = [
        lambda x: 0.5 + 0.3 * np.sin(np.pi * x / 2),
        lambda x: 0.5 * (1 + np.tanh(3 * (x - 1))),
        lambda x: np.clip(1.5 * (1 - np.abs(x - 1)), 0, 1)
    ]
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Try different starting patterns
        if restart_count < len(patterns):
            x = np.linspace(0, 2, n_points)
            h_perturbed = np.clip(patterns[restart_count](x), 0, 1)

            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)

            h_perturbed = np.clip(h_current + np.random.normal(0, 0.03, n_points), 0, 1)

            optimizers = ['SLSQP', 'COBYLA']
            for opt_method in optimizers:
                if time.time() - start_time > budget_s * 0.95:
                    break
                res = minimize(
                    objective,
                    h_perturbed,
                    method=opt_method,
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 300, 'disp': False}
                )
                
                if res.success:
                    h_new = res.x
                    c5_new = obj(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        break  # Found a better solution, try next restart

        h_current = best_h.copy()

        if restart_count % 10 == 0 and restart_count > 0:

        )

            best_c5 = obj(best_h)
    except:
        pass
        
    # Final computation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
