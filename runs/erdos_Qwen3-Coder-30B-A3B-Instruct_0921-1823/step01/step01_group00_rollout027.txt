import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 60  # Increase discretization
    dx = 2.0 / n_points
    
    # Use initial construction if it exists
    if 'initial_h_values' in globals():

    else:

        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    # Optimization loop
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts with different strategies
    max_restarts = 50
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Try different perturbation strategies
        strategy = restart_count % 4
        
        if strategy == 0:
            # Small random perturbation
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.015, n_points), 0, 1)
        elif strategy == 1:
            # Medium perturbation with some structure
            pert = np.random.normal(0, 0.03, n_points)
            # Ensure smoothness by averaging neighbors
            pert = 0.3 * pert + 0.7 * np.convolve(pert, [0.25, 0.5, 0.25], mode='same')
            h_perturbed = np.clip(h_current + pert, 0, 1)
        elif strategy == 2:
            # Larger perturbation with some smoothing
            pert = np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_current + 0.5 * pert, 0, 1)
        else:
            # Random with symmetry consideration
            h_perturbed = np.random.rand(n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
        # Ensure constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            # Try multiple optimizers
            optimizers = ['SLSQP', 'COBYLA']
            best_result = None
            best_result_c5 = float('inf')
            
            for opt in optimizers:
                try:
                    res = minimize(
                        obj,
                        h_perturbed,
                        method=opt,
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 300, 'disp': False}
                    )
                    
                    if res.success:
                        c5_new = obj(res.x)
                        if c5_new < best_result_c5:
                            best_result_c5 = c5_new
                            best_result = res.x
                except:
                    continue
                    
            if best_result is not None and best_result_c5 < best_c5:
                best_h = best_result.copy()
                best_c5 = best_result_c5
                
        except Exception as e:
            pass

        h_current = best_h.copy()
        restart_count += 1
        
    # Final fine-tuning

        res = minimize(
            obj,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'disp': False}
        )
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
