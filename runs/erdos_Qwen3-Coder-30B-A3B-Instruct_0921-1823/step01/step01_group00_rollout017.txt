import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n_points = 60 for better resolution
    n_points = 60
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Multi-start optimization with progressive refinement
    max_restarts = 50
    restart_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Start with current best
        h_perturbed = best_h.copy()
        
        # Apply adaptive random perturbations with varying intensity
        intensity = max(0.01, 0.1 * (1 - restart_count / max_restarts))
        noise = np.random.normal(0, intensity, n_points)
        h_perturbed = np.clip(h_perturbed + noise, 0, 1)
        
        # Re-scale to meet constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            # Use L-BFGS-B for better convergence
            res = minimize(
                obj,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except:
            pass
        
        restart_count += 1
        
        # Early termination if improvement is minimal
        if restart_count > 10 and abs(best_c5 - obj(best_h)) < 1e-6:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
