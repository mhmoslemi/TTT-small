import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values, n_points):
    """Compute the C5 constant for given h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

    start_time = time.time()
    
    # parameters

    # Use initial construction if it exists

    else:

    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Helper functions for optimization
    def obj(h):
        return evaluate_erdos_solution(h, n_points)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_ineq(h):
        return np.minimum(h, 1.0) - h  # h <= 1
          
    # Wrapped objective function to satisfy constraints
    def objective(h):
        return obj(h)
        
    # Others
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_eq},
            {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}]
    
    # Optimization loop
    best_h = h_current.copy()
    best_c5 = obj(best_h)
    
    # Try multiple restarts
    max_restarts = 50
    restart_count = 0
    
    # Prepare for progressive refinement
    best_h_history = []
    best_c5_history = []
    
    while time.time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Apply structured perturbation
        if len(best_h_history) > 0 and restart_count % 5 == 0:
            # Gradual refinement from previous best
            h_perturbed = best_h_history[-1].copy()
            noise_scale = 0.02 / (1 + restart_count // 5)
            h_perturbed += np.random.normal(0, noise_scale, n_points)
        else:
            # Random perturbation
            h_perturbed = np.clip(h_current + np.random.normal(0, 0.02, n_points), 0, 1)
        
        # Re-normalize to satisfy integral constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            # Try multiple optimization methods
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000}
            )
            
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    best_h_history.append(best_h.copy())
                    best_c5_history.append(best_c5)
                
        except:
            pass
        
        # Next restart
        h_current = best_h.copy()
        restart_count += 1
        
        # Progress check
        if restart_count % 10 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.9:
                break
    
    # Final computation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
