import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum over all shifts (excluding 0 shift which is not meaningful)
    c5 = np.max(corr[n_points-1:])
    return c5 * dx

def construct_initial_h():
    """Construct initial h function based on the parent solution."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (2.0 / len(h))
        return h
    else:
        # Fallback to uniform distribution with some variance
        n = 45
        h = np.ones(n) * 0.5
        h[0] = 0.1
        h[1] = 0.9
        h[-1] = 0.9
        h[-2] = 0.1
        h = h / np.sum(h) * (2.0 / n)
        return h

def optimized_objective(h):
    """Objective function to minimize with constraints."""
    # Normalize h to satisfy ∫ h(x) dx = 1
    n = len(h)
    dx = 2.0 / n
    h = h * (1.0 / np.sum(h)) * (2.0 / dx)
    c5 = evaluate_c5(h)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Construct initial guess
    h0 = construct_initial_h()
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Set bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Ensure the integral constraint is respected
        h_normalized = h * (1.0 / np.sum(h)) * (2.0 / dx)
        return evaluate_c5(h_normalized)
    
    # Optimization using differential evolution
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            strategy='best1bin'
        )
        if result.success:
            h_final = result.x * (1.0 / np.sum(result.x)) * (2.0 / dx)
        else:
            h_final = h0
    except Exception as e:
        # Fallback to original if optimization fails
        h_final = h0
    
    # Final evaluation
    if time.time() - start_time > budget_s - 1:
        c5_best = evaluate_c5(h_final)
        return (h_final, c5_best, n_points)
    
    # Perform local optimization if time allows
    try:
        from scipy.optimize import minimize
        
        # We want to minimize c5 subject to ∫ h(x) dx = 1 and h in [0,1]
        # So we'll optimize directly with constraints
        
        def obj(h):
            # Normalize
            h_norm = h * (1.0 / np.sum(h)) * (2.0 / dx)
            return evaluate_c5(h_norm)
        
        def constraint_func(h):
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_func}
        bnds = [(0, 1) for _ in range(n_points)]
        
        # Try local optimization with the result so far
        result_local = minimize(
            obj,
            h_final,
            method='SLSQP',
            bounds=bnds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result_local.success:
            h_final = result_local.x * (1.0 / np.sum(result_local.x)) * (2.0 / dx)
    except Exception:
        pass
    
    c5_best = evaluate_c5(h_final)
    return (h_final, c5_best, n_points)
