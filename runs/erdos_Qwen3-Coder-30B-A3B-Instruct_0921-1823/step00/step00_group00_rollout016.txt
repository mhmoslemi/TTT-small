import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h_values, dx):
    """Evaluate the overlap integral C5 for given h_values."""
    h = np.array(h_values)
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value in the correlation
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_particles(n_particles, n_points, initial_h):
    """Initialize particles for PSO."""
    particles = []
    # First particle initialized from initial_h_values
    particles.append(initial_h.copy())
    
    # Other particles initialized randomly
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        # Scale to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        
    return particles

def project_to_bounds(h, n_points):
    """Project h to [0,1] bounds."""
    h = np.clip(h, 0, 1)
    return h

def compute_integral_constraint(h, dx):
    """Return integral of h."""
    return np.sum(h) * dx

def enforce_integral_constraint(h, dx, n_points):
    """Scale h to satisfy integral = 1."""
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (n_points / 2) / integral
    else:
        # If sum is 0, set to uniform distribution
        h = np.full(n_points, 1.0 / n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    n_particles = 50
    max_iters = 500
    
    # Initialize particles
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Create a basic initial construction
        initial_h = np.full(n_points, 1.0 / n_points)
    
    particles = initialize_particles(n_particles, n_points, initial_h)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Personal bests
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate_h(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    start_time = time.time()
    i = 0
    
    while time.time() - start_time < budget_s and i < max_iters:
        i += 1
        for j in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[j] = (w * velocities[j] + 
                             c1 * r1 * (pbest[j] - particles[j]) + 
                             c2 * r2 * (gbest - particles[j]))
            
            # Update position
            particles[j] = particles[j] + velocities[j]
            
            # Apply bounds
            particles[j] = project_to_bounds(particles[j], n_points)
            
            # Enforce integral constraint
            particles[j] = enforce_integral_constraint(particles[j], dx, n_points)
            
            # Evaluate
            value = evaluate_h(particles[j], dx)
            
            # Update personal best
            if value < pbest_values[j]:
                pbest[j] = particles[j].copy()
                pbest_values[j] = value
                
                # Update global best
                if value < gbest_value:
                    gbest = particles[j].copy()
                    gbest_value = value
                    
    # Final evaluation
    final_c5 = evaluate_h(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
