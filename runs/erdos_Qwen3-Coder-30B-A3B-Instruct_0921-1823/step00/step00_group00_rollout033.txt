import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    
    # Precompute shifts for full convolution
    max_k = 2 * n_points  # Max shift index
    shifts = np.arange(-n_points + 1, n_points)  # All possible shifts
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Function to compute C5 from h
    def compute_c5(h):
        # Ensure h is a numpy array
        h = np.array(h)
        # Compute convolution-like
        conv = np.correlate(h, 1 - h, mode='full')
        # Shifts correspond to indices -n+1 .. n-1
        # We want max over all valid (x, x+k) pairs
        # That is the maximum value among all cross-correlation values
        max_val = np.max(conv)
        return max_val * dx
    
    # Constraint: sum(h)*dx = 1
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Initialize from provided construction or set up a good starting point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Create a balanced construction
        h0 = np.full(n_points, 0.5)
    
    # Make sure it satisfies constraint
    current_sum = np.sum(h0)
    if abs(current_sum * dx - 1.0) > 1e-10:
        h0 = h0 * (1.0 / current_sum) * (n_points / 2.0)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iters = 500
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (1.0 / np.sum(h)) * (n_points / 2.0)
        # enforce bounds
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        val = compute_c5(h)
        pbest.append(h.copy())
        pbest_vals.append(val)
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx]
    gbest_val = pbest_vals[gbest_idx]
    
    # Timing
    start_time = time.time()
    
    # PSO iterations
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s:
            break
            
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity and position
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = w * v + c1 * r1 * (p - h) + c2 * r2 * (gbest - h)
            
            new_h = h + v
            
            # Enforce bounds
            new_h = np.clip(new_h, 0, 1)
            
            # Enforce integral constraint via renormalization
            h_sum = np.sum(new_h) * dx
            if abs(h_sum - 1.0) > 1e-10:
                new_h = new_h * (1.0 / h_sum) * (n_points / 2.0)
            
            # Evaluate
            new_val = compute_c5(new_h)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = new_h.copy()
                pbest_vals[i] = new_val
                
                if new_val < gbest_val:
                    gbest = new_h.copy()
                    gbest_val = new_val
                    
            # Update particle
            particles[i] = new_h.copy()
            velocities[i] = v.copy()
            
            if new_val < best_c5:
                best_c5 = new_val
                best_h = new_h.copy()
    
    if best_h is None:
        best_h = gbest.copy()
        best_c5 = gbest_val
    
    return (best_h.tolist(), best_c5, len(best_h))
