import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Fallback: create a simple step function
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    h = cvxpy.Variable(n_points)
    C = cvxpy.Variable()
    
    # Constraint: integral = 1
    integral_constraint = cvxpy.sum(h) == n_points / 2.0
    
    # Constraints for all shifts
    constraints = [h >= 0, h <= 1, integral_constraint, C >= 0]
    
    # Build overlap constraints for all shifts
    for k in range(1, n_points):
        # Compute overlap for shift k
        expr = 0
        for i in range(n_points):
            if i + k < n_points:
                expr += h[i] * (1 - h[i+k])
        constraints.append(expr <= C * dx)
    
    # Objective: minimize C
    objective = cvxpy.Minimize(C)
    
    # Create the problem and solve
    prob = cvxpy.Problem(objective, constraints)
    
    # Try solving with ECOS
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=5000)
        final_C = prob.value
        if final_C is not None and not np.isnan(final_C):
            h_opt = h.value
            h_opt = np.clip(h_opt, 0, 1)
        else:
            # Fallback: use scipy for non-convex optimization
            h_opt, final_C = scipy_optimize(n_points, h0, dx, start_time, budget_s)
    except Exception as e:
        # Fallback to scipy optimization
        h_opt, final_C = scipy_optimize(n_points, h0, dx, start_time, budget_s)
    
    return (h_opt, final_C, n_points)

def scipy_optimize(n_points, h0, dx, start_time, budget_s):
    # Simple optimization with scipy
    def objective(h):
        # Compute C5 = max_k integral of h(x) * (1 - h(x+k))
        C5 = 0
        # Sample values for each shift
        for k in range(1, n_points):
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h[i] * (1 - h[i + k])
            C5 = max(C5, overlap_sum * dx)
        return C5

    def constraint_integral(h):
        return np.sum(h) - n_points / 2.0  # Should be 0

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_integral}]
    
    # Optimization with bounds
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 10000, 'ftol': 1e-8, 'eps': 1e-6})
    
    elapsed = time.time() - start_time
    if elapsed > 0.9 * budget_s:
        # Relax to get something
        return result.x, objective(result.x)
    
    # Try different search strategies
    best_h = result.x.copy()
    best_C = objective(best_h)
    
    # Start multiple random walks
    for _ in range(5):
        if time.time() - start_time > 0.9 * budget_s:
            break
        h_start = np.random.rand(n_points) * 0.5 + 0.25
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2.0) / np.sum(h_start)
        try:
            res = minimize(objective, h_start, method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 1000, 'ftol': 1e-6})
            Val = objective(res.x)
            if Val < best_C:
                best_C = Val
                best_h = res.x
        except:
            pass

    return best_h, best_C
