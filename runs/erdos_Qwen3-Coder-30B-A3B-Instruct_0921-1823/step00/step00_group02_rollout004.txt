from scipy.optimize import minimize

import time

def evaluate_erdos_solution(h_values):
    # Computes C5 = max_k integral h(x)(1 - h(x+k)) dx
    n = len(h_values)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # Find maximum value among all shifts (the full correlation is 2n-1 long)
    max_val = np.max(conv) * dx
    return max_val
