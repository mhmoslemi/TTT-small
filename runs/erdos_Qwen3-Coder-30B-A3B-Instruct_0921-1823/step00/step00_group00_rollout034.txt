import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for step function h."""
    # Compute all correlations for shifts k = 0 to 2*dx
    corr = correlate(h, 1 - h, mode='full')
    # Only consider shifts up to 2 (full correlation range)  
    # Select the appropriate indices for the actual shifts
    max_k = len(h) - 1
    # Get the correlation values corresponding to k = 0 to 2*dx
    # We want: k from 0 to 2*dx, but dx*shifts are discrete
    # corr indices go from 0 to 2*len(h) - 2, which we map back to k
    ranges = np.arange(len(corr))
    k_vals = ranges - (len(h) - 1)  # Map indices to k values
    # Restrict to effective shifts (k in [0, 2])
    valid_inds = (k_vals >= 0) & (k_vals <= 2)
    corr_vals = corr[valid_inds]
    # Scale by dx for the integral
    return np.max(corr_vals * dx)

def initialize_particles(n_particles, n_points, initial_h=None):
    """Initialize particles for PSO."""
    dx = 2.0 / n_points
    particles = []
    velocities = []
    for i in range(n_particles):
        if i == 0 and initial_h is not None:
            # Start with initial solution
            h = initial_h.copy()
        else:
            # Random values
            h = np.random.rand(n_points)
            # Scale so integral is 1
            h = h * (1.0 / (np.sum(h) * dx))
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.rand(n_points) * 0.1)
    return particles, velocities

def apply_velocity(h, v, w, c1, c2, pbest_h, gbest_h, r1, r2):
    """Update particle position using velocity."""
    new_v = w * v + c1 * r1 * (pbest_h - h) + c2 * r2 * (gbest_h - h)
    new_h = h + new_v
    new_h = np.clip(new_h, 0, 1)
    return new_h, new_v

    np.random.seed(seed)
    n_points = 90  # Slightly above minimum needed for reasonable resolution
    dx = 2.0 / n_points
    n_particles = 50
    max_iters = 500
    w = 0.7
    c1 = 2
    c2 = 2
    timeout = time.time() + budget_s

    particles, velocities = initialize_particles(n_particles, n_points, initial_h_values if 'initial_h_values' in globals() else None)
    pbest = [None] * n_particles
    pbest_vals = [np.inf] * n_particles
    gbest = None
    gbest_val = np.inf

    # Main loop
    for iteration in range(max_iters):
        if time.time() > timeout:
            break
        for i in range(n_particles):
            h = particles[i]
            val = evaluate_h(h, dx)
            if val < pbest_vals[i]:
                pbest_vals[i] = val
                pbest[i] = h.copy()
            if val < gbest_val:
                gbest_val = val
                gbest = h.copy()
        
        # Update particles
        r1 = np.random.rand(n_points)
        r2 = np.random.rand(n_points)
        for i in range(n_particles):
            if i == 0 and initial_h_values is not None:
                continue  # Don't update the initial one
            particles[i], velocities[i] = apply_velocity(
                particles[i], velocities[i], w, c1, c2,
                pbest[i], gbest, r1, r2
            )
    
    return (gbest, gbest_val, n_points)
