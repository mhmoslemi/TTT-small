import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Fallback: create simple construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Normalize
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Let's do a simple gradient-based approach using scipy
    # We will optimize h to minimize C5 = max_k ∫ h(x)(1-h(x+k)) dx
    
    # Discretize all shifts for evaluation of overlaps
    max_shift = n_points - 1
    dx = 2.0 / n_points

    def compute_overlap(h_vals, shift):
        """Compute overlap integral for a given shift"""
        # h_vals is a list of length n_points
        # shift is an integer
        if shift < 0:
            return 0
        h1 = h_vals[:n_points-shift]
        h2 = h_vals[shift:]
        if len(h1) != len(h2):
            return 0
        return np.sum(np.array(h1) * (1.0 - np.array(h2))) * dx

    def compute_C5(h_vals):
        """Compute C5 as max over all shifts"""
        max_val = 0
        for shift in range(1, n_points):
            overlap = compute_overlap(h_vals, shift)
            if overlap > max_val:
                max_val = overlap
        return max_val

    def obj_func(h_vals):
        # Want to minimize C5
        return compute_C5(h_vals)
    
    # Constraints
    def integral_constraint(h_vals):
        return np.sum(h_vals) * dx - 1.0

    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial solution
    h_init = np.array(h0)
    
    # Optimize
    result = minimize(obj_func, h_init, bounds=bounds,
                      constraints={'type': 'eq', 'fun': integral_constraint},
                      method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-6})

    final_h = result.x
    final_c5 = compute_C5(final_h)
    
    return final_h, final_c5, n_points
