import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, dx):
    """Evaluate the C5 bound for given h values"""
    # Compute the overlap integral using full convolution
    n = len(h_values)
    # Create the flipped version of h for convolution
    flipped_h = np.flip(h_values)
    # Do the full convolution (corresponds to the overlap integral)
    correlation = np.correlate(h_values, flipped_h, mode='full')
    # The max value among all possible shifts k is what we want
    # Since we're looking at integrals with dx, we multiply by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

        h[-2] = 0.4
    
    dx = 2.0 / n_points
    
    # Ensure the constraint ∫ h(x) dx = 1 (sum = n_points/2)
    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum
    
    # Define the optimization objective
    def objective(h_vec):
        # Normalize h_vec to satisfy constraint sum * dx = 1
        h_scaled = h_vec * (n_points / 2) / np.sum(h_vec)
        return evaluate_h(h_scaled, dx)

    # Define bounds for each h[i]: [0, 1] 
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization using scipy minimize
    start_time = time.time()
    
    # We'll do several passes of local optimization, starting from different points
    best_h = x0.copy()
    best_c5 = objective(best_h)
    
    # Try multiple restarts to avoid local minima
    restarts = 0
    max_restarts = 50
    while time.time() - start_time < budget_s * 0.95 and restarts < max_restarts:
        # Perturb the current best solution a bit for random restart
        perturbed = best_h.copy()

        noise = np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed + noise, 0, 1)

        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)

            if new_c5 < best_c5:
                best_h = new_h

                # Reduce maxiter to speed up if we are close to solution  
        restarts += 1
    
    # Final check
    final_h = best_h.copy() * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_h(final_h, dx)
    
    # Return the best solution found
    return (final_h.tolist(), final_c5, n_points)
