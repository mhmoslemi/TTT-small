import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for a given step function h."""
    n = len(h)
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum value (excluding the zero-padding)
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

    # Define objective function
    def objective(h):
        return evaluate_c5(h, dx)
