import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from existing construction or create new one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable starting point
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    max_iters = 500
    w = 0.7
    c1 = 2
    c2 = 2
    
    # Precompute indices for efficient correlation
    shift_indices = np.arange(n_points)
    
    # Function to compute overlap integral (C5)
    def compute_overlap(h):
        # Compute integral of h(x)(1 - h(x+k)) for all k
        h_pad = np.concatenate([h, np.zeros(n_points-1)])
        h_rev = np.concatenate([np.zeros(n_points-1), h[::-1]])
        # Correlation with flipped h
        correlation = np.convolve(h, 1 - h[::-1], mode='full')
        # Take max of all shifts
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Penalty for constraint violations
    def penalty(h):
        p = 0
        # Bound constraints
        p += np.sum(np.maximum(0, h)) + np.sum(np.maximum(0, 1 - h))
        # Integral constraint (allowing small violations)
        integral_violation = np.abs(np.sum(h) * dx - 1)
        p += integral_violation * 1000
        return p
    
    def objective(h):
        return compute_overlap(h) + penalty(h)
    
    # Initialize particles
    particles = []
    velocities = []
    
    # One particle at initial construction, rest randomly initialized
    particles.append(h.copy())
    velocities.append(np.zeros_like(h))
    
    for i in range(1, n_particles):
        h_i = np.random.rand(n_points)
        h_i = h_i * (n_points / 2) / np.sum(h_i)
        particles.append(h_i)
        velocities.append(np.random.randn(n_points) * 0.1)
    
    # Initialize personal bests
    pbest_h = [p.copy() for p in particles]
    pbest_score = [objective(p) for p in particles]
    
    # Initialize global best
    best_idx = np.argmin(pbest_score)
    gbest_h = pbest_h[best_idx].copy()
    gbest_score = pbest_score[best_idx]
    
    # PSO iteration
    for iter_num in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_h[i] - particles[i]) + 
                            c2 * r2 * (gbest_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize to satisfy integral constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate function
            score = objective(particles[i])
            
            # Update personal best
            if score < pbest_score[i]:
                pbest_score[i] = score
                pbest_h[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest_h = particles[i].copy()
    
    return (gbest_h, gbest_score, n_points)
