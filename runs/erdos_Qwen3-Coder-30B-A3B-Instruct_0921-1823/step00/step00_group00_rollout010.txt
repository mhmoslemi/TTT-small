import numpy as np
from scipy import optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h with the provided starting construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        # Otherwise create a simple initial guess
        h0 = np.ones(n_points) * 0.5
        
    h = h0.copy()
    
    # Create a distance matrix for autocorrelation calculation
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        # Compute all shifted correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Normalize by dx
        corr = corr * dx
        # We want to maximize the minimum
        return np.max(corr)
    
    # Define the constraints and variable
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(cp.multiply(1 - h_var, h_var)))
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Define the problem
    prob = cp.Problem(objective, constraints)
    
    # Use a local solver
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
        h_opt = h_var.value
    except:
        # Fallback to less constrained optimization
        def obj(h_vec):
            return compute_C5(h_vec)
        
        # Wrap constraint functions
        def constraint_0(h_vec):
            return np.sum(h_vec) - n_points / 2
        def constraint_1(h_vec):
            return 1 - np.max(h_vec)
        def constraint_2(h_vec):
            return np.min(h_vec) - 0
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [optimize.NonlinearConstraint(constraint_0, 0, 0),
                       optimize.NonlinearConstraint(constraint_1, 0, np.inf),
                       optimize.NonlinearConstraint(constraint_2, 0, np.inf)]
        
        result = optimize.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 500})
        
        if not result.success:
            h_opt = h0
        else:
            h_opt = result.x
            # Clamp to [0,1]
            h_opt = np.clip(h_opt, 0, 1)
    
    # Final refinement
    C5_best = compute_C5(h_opt)
    
    # Enforce integral constraint accurately
    h_final = h_opt.copy()
    h_final = h_final / np.sum(h_final) * (n_points / 2)
    
    C5_final = compute_C5(h_final)
    
    return (h_final.tolist(), C5_final, n_points)
