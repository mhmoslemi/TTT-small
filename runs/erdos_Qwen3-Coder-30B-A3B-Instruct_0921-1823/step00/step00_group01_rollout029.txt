import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, n_points):
    dx = 2.0 / n_points
    # Compute the overlap integral C_5
    conv = np.correlate(h_values, 1 - h_values, mode="full")
    c5 = np.max(conv) * dx
    return c5

    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint
    def sum_constraint(h):
        return np.sum(h) * dx - 1.0

    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Optimize
    best_c5 = float('inf')
    best_h = h0.copy()
    
    start_time = time.time()
    
    # Use COBYLA method for constrained optimization
    options = {'maxiter': 1000, 'disp': False}
    
    # Run optimization with time limit
    try:
        while time.time() - start_time < budget_s - 5:
            result = minimize(
                lambda h: evaluate_h(h, n_points),
                h0,
                method='COBYLA',
                bounds=bounds,
                constraints=constraints,

            )
            
            if result.success:
                current_c5 = evaluate_h(result.x, n_points)

                    best_c5 = current_c5

                # Perturb h0 for next iteration
                h0 = result.x + np.random.normal(0, 0.01, n_points)

                # Normalize to have integral = 1
                h0 = h0 * (1.0 / np.sum(h0) / dx)

    # Final evaluation
