import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, n_points):
    dx = 2.0 / n_points
    # Compute the overlap integral C_5
    conv = np.correlate(h_values, 1 - h_values, mode="full")
    c5 = np.max(conv) * dx
    return c5

        h0[-1] = 0.8
    
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Constraint function: sum(h) * dx = 1
    def constraint_sum(h):
        return np.sum(h) * dx - target_sum

    # Objective function
    def objective(h):
        return evaluate_h(h, n_points)

    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Constraints dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Time tracking
    start_time = time.time()

    # Initial value
    h_current = h0.copy()
    c5_current = evaluate_h(h_current, n_points)
    h_best = h_current.copy()
    c5_best = c5_current

    # Optimization parameters
    max_iter = 1000
    iter_count = 0

    # COBYLA optimizer with iterative process
    while time.time() - start_time < budget_s - 1.0 and iter_count < max_iter:
        # Optimize
        try:
            result = minimize(objective, h_current, method='COBYLA', bounds=bounds,
                              constraints=cons, options={'maxiter': 50, 'disp': False})
            if result.success:
                h_new = result.x
                c5_new = evaluate_h(h_new, n_points)
                
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
                    # Sometimes we may want to restart with a small perturbation
                    # But for now, we'll just keep updating
            else:
                # Fallback if optimization failed, adjust slightly
                h_new = h_current + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                # Re-normalize
                h_new = h_new * (target_sum / (np.sum(h_new) * dx))

                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
                    
        except Exception as e:
            # Occasionally, optimization might fail. Try a simple local search
            h_new = h_current.copy()
            # Simple random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
