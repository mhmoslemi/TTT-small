        new_val = np.clip(new_val, 0, 1)
        
        # Adjust other values to maintain sum constraint
        diff = new_val - old_val

        for j in range(n_points):
            if j != i:
                h[j] -= diff / (n_points - 1)
                h[j] = np.clip(h[j], 0, 1)

        h_sum = np.sum(h) * dx

    final_c5 = evaluate_c5(best_h, dx)

def evaluate_c5(h_values, dx):
    h = np.array(h_values)
    # Compute overlaps using FFT-based convolution
    h_rev = np.flip(h)
    overlaps = signal.convolve(h, h_rev, mode='full')
    overlaps = overlaps * dx  # Scale by dx
    
    # The overlap integral C5 is the maximum of these overlaps
    return np.max(overlaps)
