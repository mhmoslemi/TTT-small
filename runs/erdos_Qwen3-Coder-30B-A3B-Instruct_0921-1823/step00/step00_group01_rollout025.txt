import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h)
    except NameError:
        # Initialize with a simple construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.2
        h[-1] = 0.8
        h = h / (np.sum(h) * 2.0 / n_points)
    
    dx = 2.0 / n_points
    
    def evaluate_C5(h_vec):
        # Compute overlap integral using FFT for efficiency
        # Convolution with (1-h) including full mode
        h_pad = np.pad(h_vec, (0, n_points-1), mode='constant', constant_values=0)
        h1_pad = np.pad(1-h_vec, (0, n_points-1), mode='constant', constant_values=0)
        
        conv = ifft(fft(h_pad) * fft(h1_pad).conj()).real
        max_conv = np.max(conv[:n_points])
        return max_conv * dx
    
    def objective(h_vec):
        return evaluate_C5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def project_to_feasible(h_vec):
        # Ensure values are in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Adjust to make sum constraint hold
        current_sum = np.sum(h_vec) * dx
        if abs(current_sum - 1.0) > 1e-10:
            # Rescale to meet constraint
            h_vec = h_vec * (1.0 / current_sum)
            h_vec = np.clip(h_vec, 0, 1)
        return h_vec
    
    # Initial evaluation
    best_h = h.copy()
    best_C5 = objective(best_h)
    
    # Phase 1: Local optimization with random restarts
    max_iter = 5000
    iter_count = 0
    last_improvement = 0
    
    # Use COBYLA with constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Time tracking
    remaining_time = budget_s - (time.time() - start_time)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Perturb and optimize locally
        h_pert = best_h.copy()
        # Randomly modify some values
        for i in range(n_points):
            if np.random.random() < 0.1:
                change = np.random.normal(0, 0.05)
                h_pert[i] = h_pert[i] + change
        
        # Project to feasible and optimize
        h_feasible = project_to_feasible(h_pert)
        
        # Optimize with scipy minimize
        try:
            res = minimize(objective, h_feasible, method='L-BFGS-B', bounds=bounds, 
                          constraints=cons, tol=1e-6, options={'maxiter': 200})
            if res.success:
                new_h = project_to_feasible(res.x)
                new_C5 = objective(new_h)
                if new_C5 < best_C5:
                    best_h = new_h
                    best_C5 = new_C5
                    last_improvement = iter_count
        except:
            pass
        
        # Simple local adjustment with small noise
        if iter_count % 50 == 0:
            h_noisy = best_h + np.random.normal(0, 0.01, n_points)
            h_noisy = project_to_feasible(h_noisy)
            new_C5 = objective(h_noisy)
            if new_C5 < best_C5:
                best_h = h_noisy
                best_C5 = new_C5
                last_improvement = iter_count
        
        iter_count += 1
        
        # Early stopping if no improvement for a while
        if iter_count - last_improvement > 200 and iter_count > 300:
            break
    
    # Final polish with optimizer
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                      tol=1e-8, options={'maxiter': 1000})
        if res.success:
            final_h = project_to_feasible(res.x)
            final_C5 = objective(final_h)
            if final_C5 < best_C5:
                best_h = final_h
                best_C5 = final_C5
    except:
        pass
    
    # Final validation
    final_h = project_to_feasible(best_h)
    final_C5 = objective(final_h)
    
    # Normalize
    final_h = final_h / np.sum(final_h) * (2.0 / n_points)
    
    return (final_h.tolist(), float(final_C5), len(final_h))
