def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts in [0,2])
    start_idx = n - 1
    max_corr = np.max(corr[start_idx:])
    return max_corr * dx
