def evaluate_overlap(h, dx):

    # h is a 1D array of length n_points
    # We want max_k int h(x) (1 - h(x+k)) dx
    # This is equivalent to max_k correlate(h, 1-h, mode="full") * dx

    h_conj = 1 - h_vec
    corr = signal.correlate(h_vec, h_conj, mode='full')
    # The full correlation is of length 2*n_points - 1
    # The valid range of k is -n_points+1 to n_points-1 
    # But we want to focus on k from 0 to n_points-1 (or actually 0 to 2*n_points-2) 
    # The for k from 0 to n_points-1 we get the values corr[n_points-1:n_points-1+n_points] 
    # But also include k from n_points to 2*n_points-2, which are corr[n_points-1+n_points:2*n_points-2]
    corr_full = corr * dx
    # Maximize over all k
    max_val = np.max(corr_full)
    return max_val
