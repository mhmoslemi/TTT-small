import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.fftpack import fft, ifft
import time

def evaluate_c5(h, dx):
    # Compute convolution of h with (1-h) - this is equivalent to the overlap integral
    h_padded = np.pad(h, (0, len(h)), mode='constant', constant_values=0)
    one_minus_h = 1 - h
    one_minus_h_padded = np.pad(one_minus_h, (0, len(one_minus_h)), mode='constant', constant_values=0)
    conv = ifft(fft(h_padded) * np.conj(fft(one_minus_h_padded))).real
    # Take only the valid part of the convolution
    valid_conv = conv[len(h)-1:2*len(h)-1]
    # Return maximum value
    return np.max(valid_conv) * dx

    else:

        h = np.ones(n_points)
        h[:n_points//2] = 0.5
        
    # Compute initial dx
    dx = 2.0 / n_points
    
    # Optimization loop
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Helper function to project h onto feasible region
    def project_h(h_array):
        h_proj = np.clip(h_array, 0, 1)
        # Adjust to keep integral exactly 1
        current_sum = np.sum(h_proj) * dx
        if current_sum > 0:
            scale = 1.0 / current_sum
            h_proj *= scale
        return h_proj
    
    # Gradient based optimization (COBYLA)
    def obj_func(h_array):
        h_proj = project_h(h_array)
        return evaluate_c5(h_proj, dx)
    
    # Use a hybrid approach: explore with differential evolution, then local refinement
    max_iter = 500
    iter_count = 0
    
    # First phase: global search 
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a small subset of the space to explore
    if n_points > 100:
        sample_indices = np.random.choice(n_points, 50, replace=False)
        h_sample = h[sample_indices]
        bounds_sample = [(0, 1) for _ in range(50)]
        
        def obj_func_sample(h_arr):
            h_full = h.copy()
            h_full[sample_indices] = h_arr
            h_full = project_h(h_full)
            return evaluate_c5(h_full, dx)
        
        res = differential_evolution(obj_func_sample, bounds_sample, seed=seed, 
                                   maxiter=100, popsize=10, atol=1e-6)
        h[sample_indices] = res.x
        h = project_h(h)
        
    # Local refinement using COBYLA
    def constraint_func(h_array):
        return np.sum(h_array) * dx - 1.0
        
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Set up bounds
    bounds_opt = [(0.0, 1.0) for _ in range(n_points)]
    
    # Main optimization loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        # Refine using scipy minimize with constraints
        res = minimize(obj_func, h, method='COBYLA', bounds=bounds_opt, 
                       constraints=cons, options={'maxiter': 30, 'disp': False})
        
        if res.success:
            h = project_h(res.x)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
