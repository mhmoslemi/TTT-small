import time

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for a given step function h"""
    # Compute overlap integral using full convolution
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution scaled by dx
    c5 = np.max(conv) * dx
    return c5
