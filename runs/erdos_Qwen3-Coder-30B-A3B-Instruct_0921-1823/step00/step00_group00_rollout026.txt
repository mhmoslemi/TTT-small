    def objective(h):

            h = np.ones_like(h) * 0.5
        
        # Compute overlaps
        h_padded = np.concatenate([h, np.zeros(n_points)])

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using differential evolution
