import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute correlation of h with (1-h)
    correlation = np.correlate(h, 1-h, mode='full')
    # The maximum overlap is the maximum of correlation values
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Import initial construction if available

        # Normalize to sum = n_points/2  
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
    dx = 2.0 / n_points
    
    # Optimizer using scipy minimize with bounds
    def loss(h_vec):
        # Normalize h_vec to sum to n_points/2
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return evaluate_overlap(h_vec, dx)
    
    # Convert h0 to list
    h0_list = list(h0)

    # Try several random restarts
    n_restarts = 50

        if time.time() - start_time > budget_s - 1:
            break
            
        # Small random perturbation of h0
        h_pert = h0_list.copy()
        for j in range(n_points):
            # Add small random noise
            h_pert[j] += np.random.normal(0, 0.01)
            h_pert[j] = max(0, min(1, h_pert[j]))

        h_pert = np.array(h_pert)

        result = minimize(
