import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h, dx=2.0/45):
    # Computes C5 = max_k int h(x)(1 - h(x+k)) dx
    # This is equivalent to 1 - dx * min_k S_k where S_k = sum h(x)*h(x+k)
    # Using full convolution: 
    corr = np.correlate(h, 1-h, mode='full')
    # Take the middle part (overlaps)
    n = len(h)
    middle = corr[n-1:2*n-1]
    return 1 - dx * np.min(middle)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    # Initial guess
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5

    # Ensure sum constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Variables
    h = Variable(n_points)
    
    start_time = time.time()
    
    # To find C5 = 1 - dx * min(S_k), we minimize max(S_k) 
    # This is a QCQP. Instead, treat this as minimizing C5 = max_k int h(x)(1-h(x+k))
    # We will optimize the constraint representation
    
    # More direct approach:
    # Constraint: sum(h) = n_points/2 (pre-scaled)
    # Constraint: 0 <= h[i] <= 1
    # Objective: minimize max_k int h(x)(1 - h(x+k)) dx
    
    # Use sequential quadratic programming approach with bounds.
    # First, define the optimization. Maximize min(S_k)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
    
    # Try to directly optimize using scipy
    def objective(h_vals):
        return -evaluate_erdos_solution(h_vals, dx)  # negative because we minimize
    
    best_h = None
    best_c5 = float('inf')
    
    # Start with initial solution
    res = minimize(objective, h0, bounds=bounds, constraints=constraints, 
                   method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-8})
    
    if res.success:
        h_val = np.maximum(0, np.minimum(1, res.x))
        c5_bound = evaluate_erdos_solution(h_val, dx) 
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_val.copy()
    
    # Try one more refinement step if time allows
    if time.time() - start_time < 0.9 * budget_s:
        # Add more random perturbations to escape local optima
        np.random.seed(int(time.time()))
        for _ in range(50):
            pert = np.random.normal(0, 0.05, n_points)
            h_test = h0 + pert
            h_test = np.maximum(0, np.minimum(1, h_test))
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            try:
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
            except:
                pass

    # Final check
    final_c5 = best_c5
    final_h = best_h
    
    # Final evaluation
    return (final_h, final_c5, n_points)
