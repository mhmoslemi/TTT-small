import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_h(h, dx):
    # Use FFT for efficient convolution
    n = len(h)
    # Zero-pad for full convolution
    pad_n = 2 * n - 1
    h_padded = np.pad(h, (0, n-1), mode='constant')
    one_minus_h_padded = np.pad(1 - h, (0, n-1), mode='constant')
    
    # Compute convolution
    conv = ifft(fft(h_padded) * fft(one_minus_h_padded)).real
    
    # The overlap integral values for all k shifts
    # We take the maximum of these
    max_overlap = np.max(conv[:n])
    return max_overlap * dx

    cons = {'type': 'eq', 'fun': constraint_eq}

    result = minimize(objective, initial_h, method='COBYLA',

                      options={'maxiter': 1000, 'disp': False})
    
    best_h = result.x
    final_c5 = objective(best_h)
    
    return best_h, final_c5
