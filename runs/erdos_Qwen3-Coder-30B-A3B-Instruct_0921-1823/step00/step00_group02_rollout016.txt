import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    # h is assumed to be normalized to sum to 1
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum is over all possible shifts k
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use existing initial construction if available
    try:
        global initial_h_values
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        # Fallback construction - simple step function
        n_points = 45
        initial_h = np.ones(n_points) * 0.5
        # Make it a step function
        mid = n_points // 2
        initial_h[:mid] = 1.0
        initial_h[mid:] = 0.0
        # Normalize
        initial_h = initial_h * (n_points / (2 * np.sum(initial_h)))
    
    def objective(h):
        # Normalize h to satisfy constraint
        h = np.abs(h)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / (2 * h_sum))
        else:
            h = h + 1e-8
            h = h * (n_points / (2 * np.sum(h)))
        return evaluate_c5(h)
    
    # Define bounds for each variable (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using differential evolution
    # Set maxiter based on time budget
    maxiter = max(100, int(budget_s * 5))
    
    result = differential_evolution(objective, bounds, seed=seed, maxiter=maxiter, 
                                   popsize=15, strategy='best1bin',
                                   atol=1e-6, rtol=1e-6, workers=1)
    
    final_h = np.abs(result.x)
    h_sum = np.sum(final_h)
    if h_sum > 0:
        final_h = final_h * (n_points / (2 * h_sum))
    
    c5_bound = evaluate_c5(final_h)
    
    # Check if we're close to target
    if c5_bound < 0.3807:
        return (final_h.tolist(), c5_bound, n_points)
    
    # Try gradient-based refinement if time permits
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        # Use L-BFGS-B for refinement but with bounds
        from scipy.optimize import minimize
        
        def objective_bounded(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / (2 * h_sum))
            return evaluate_c5(h)
        
        # Slightly perturb the solution for better conditioning
        x0 = np.clip(final_h + np.random.normal(0, 0.01, n_points), 0, 1)
        h_sum = np.sum(x0)
        if h_sum > 0:
            x0 = x0 * (n_points / (2 * h_sum))
            
        # Refine with L-BFGS-B
        bounds_ref = [(0, 1) for _ in range(n_points)]
        result_ref = minimize(objective_bounded, x0, method='L-BFGS-B', bounds=bounds_ref,
                              options={'ftol': 1e-9, 'gtol': 1e-9})
        
        refined_h = np.clip(result_ref.x, 0, 1)
        refined_h_sum = np.sum(refined_h)
        if refined_h_sum > 0:
            refined_h = refined_h * (n_points / (2 * refined_h_sum))
        
        refined_c5 = evaluate_c5(refined_h)
        
        if refined_c5 < c5_bound:
            return (refined_h.tolist(), refined_c5, n_points)
    
    return (final_h.tolist(), c5_bound, n_points)
