import numpy as np

def run(seed=42, budget_s=600, **kwargs):

        h0[30:35] = 1.0
        
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Keep track of last improvements
    last_improvement = 0
    patience = 10
    
    # Iteration counter
    iteration = 0
    
    try:
        # Coordinate descent approach
        while time.time() - start_time < budget_s - 1:
            improved = False
            # Random order update
            indices = np.random.permutation(n_points)
            
            for i in indices:
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Try small perturbations
                original = best_h[i]
                step_sizes = np.array([-0.01, -0.005, 0.005, 0.01])
                current_c5 = evaluate_c5(best_h, dx)
                
                for step in step_sizes:
                    if (original + step < 0) or (original + step > 1):
                        continue

                    new_h[i] = original + step
                    
                    # Project to maintain sum constraint
                    diff = np.sum(new_h) - target_sum
                    if abs(diff) > 1e-10:
                        # Adjust all other points to compensate
                        adjustment = -diff / (n_points - 1)
                        new_h[new_h != original + step] += adjustment
                        # Ensure bounds
                        new_h = np.clip(new_h, 0, 1)
                        # Re-adjust the modified point to be exact
                        new_h[i] = original + step
                        new_h = np.clip(new_h, 0, 1)

                    if new_c5 < best_c5:
                        best_h = new_h

    except Exception as e:

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate the overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    h = np.array(h)
    # Create flipped version 1-h
    flipped = 1 - h
    # Full convolution
    conv = np.correlate(h, flipped, mode="full") * dx
    # Return maximum value
    return np.max(conv)
