import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute C5 using full convolution
    # Correlate h with (1-h) and find maximum
    conv = correlate(h, 1 - h, mode='full')
    # Adjust for discretization
    max_val = np.max(conv) * dx
    return max_val

def project_to_feasible(h, dx):
    # Ensure h[i] in [0,1] and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if total > 0:
        h = h / total * (2.0 / dx)
    return h
