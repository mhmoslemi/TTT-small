import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the Erdős constant C5 for given h values."""
    dx = 2.0 / n_points
    # Compute correlations for all shifts
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Take the maximum correlation
    max_corr = np.max(correlations)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    try:
        h0 = initial_h_values
    except:
        # Create initial guess
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[0:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Optimization variables for h values
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    # Integral constraint
    constraints.append(sum(h) * dx == 1.0)
    # Overlap constraints for each shift 1 to n_points-1
    for k in range(1, n_points):
        # For each shift k, we must have: 
        # sum h[i] * (1 - h[i+k]) <= C * dx
        cor = 0
        for i in range(n_points - k):
            cor += h[i] * (1 - h[i + k])
        constraints.append(cor <= C * dx)
    
    # Objective
    objective = Minimize(C)
    
    # Create the problem and solve
    prob = Problem(objective, constraints)
    
    # Try solving with different methods
    try:
        # First, try using ECOS solver
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        if prob.status == 'optimal':
            optimized_h = np.array(h.value).flatten()
            c5_bound = prob.value
        else:
            # Fallback to scipy optimization if CVXPY doesn't work
            optimized_h = h0.copy()
            c5_bound = evaluate_erdos_solution(optimized_h, n_points)
            
            # Simple gradient descent on the initial solution within box constraints
            def fun(h_vec):
                return evaluate_erdos_solution(h_vec, n_points)
            
            def jac(h_vec):
                # Finite difference gradient
                eps = 1e-6
                grad = np.zeros_like(h_vec)
                base_val = fun(h_vec)
                for i in range(len(h_vec)):
                    h_vec_eps = h_vec.copy()
                    h_vec_eps[i] += eps
                    grad[i] = (fun(h_vec_eps) - base_val) / eps
                return grad
            
            # Projected gradient descent
            lr = 0.01
            max_iter = 10000
            for i in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                g = jac(optimized_h)
                optimized_h -= lr * g
                # Project to [0,1] and integral constraint
                optimized_h = np.clip(optimized_h, 0, 1)
                # Scale to satisfy integral constraint
                optimized_h *= (n_points / 2) / np.sum(optimized_h)
                # Recompute C5
                c5_bound = fun(optimized_h)
                
    except:
        # Fallback to scipy optimization
        optimized_h = h0.copy()
        c5_bound = evaluate_erdos_solution(optimized_h, n_points)
        
        def fun(h_vec):
            return evaluate_erdos_solution(h_vec, n_points)
        
        # Use bounded L-BFGS-B to optimize within bounds [0,1]
        bounds = [(0, 1) for _ in range(n_points)]
        # Weighted sum constraint
        def constraint_func(x):
            return np.sum(x) - n_points / 2
        
        from scipy.optimize import minimize
        
        res = minimize(fun, optimized_h, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 10000, 'ftol': 1e-9})
        if res.success:
            optimized_h = res.x
        # Ensure integral constraint
        optimized_h *= (n_points / 2) / np.sum(optimized_h)
        c5_bound = fun(optimized_h)
    
    # Final evaluation 
    final_c5 = evaluate_erdos_solution(optimized_h, n_points)
    
    return (optimized_h, final_c5, n_points)
