import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        if np.all(h0 >= 0) and np.all(h0 <= 1):
            best_h = h0.copy()
            corr = np.correlate(best_h, 1 - best_h, mode="full")
            best_c5 = np.max(corr) * dx
        else:
            h0 = np.ones(n_points) * 0.5
            h0 = h0 / np.sum(h0) * (n_points / 2)
            best_h = h0.copy()
    else:
        # Create a random initial vector
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        best_h = h0.copy()
    
    # Objective function using direct correlation calculation
    def objective(h):
        if np.any(h < 0) or np.any(h > 1):
            return 1e10  # penalty for constraint violation
        # Normalize integral constraint
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Differential evolution with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def optimize_step():
        nonlocal best_h, best_c5
        # Run optimization
        res = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            tol=1e-6,
            recombination=0.7,
            seed=seed
        )
        
        if res.success:
            h_opt = res.x
            h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            c5_opt = np.max(corr) * dx
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        return best_h, best_c5
    
    # Run optimization over time budget
    while time.time() - start_time < budget_s - 1:
        try:
            optimize_step()
        except:
            break
    
    # Final evaluation
    if best_h is not None:
        h_final = best_h.copy()
        corr = np.correlate(h_final, 1 - h_final, mode="full")
        c5_final = np.max(corr) * dx
        return (h_final.tolist(), c5_final, n_points)
    else:
        return (np.ones(n_points) * 0.5, 0.5, n_points)
