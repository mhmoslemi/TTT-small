import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h, dx=2.0/45):
    """Evaluate the C5 constant for a given step function h"""
    # Ensure h is properly normalized
    h = np.array(h)
    h = h * (1.0 / np.sum(h) * 45 * dx)  # Scale to integral = 1
    
    # Compute overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The full correlation includes extra values
    # We want the overlap values for k in [0,2] 
    # For a sequence of length n, the valid range is from -n+1 to n-1
    # Our full correlation is of length 2*n - 1
    # The middle n-1 elements are the valid overlaps for k > 0
    # The correct way to select the relevant values
    n = len(h)
    # The overlap for shift k is the correlation at index n-1+k
    # So we take elements from index n-1 to 2*n-2 (inclusive)
    valid_overlaps = corr[n-1:] 
    # But the corresponding k values are 0 to 2*n-2-(n-1) = n-1
    # So we map index i to k = i - (n-1)
    # The shift k ranges from 0 to n-1, so the overlap at index n-1+k corresponds to k
    # Actually, for a function h defined on [0,2], shift k means h(x) vs h(x+k)
    # And the correlation is computed over all x where both are valid
    
    # For our implementation with dx = 2/n, 
    # k = i * dx = i * 2/n for shift i = 0, ..., n-1
    # The overlap integral for k=i*2/n is the value at corr[n-1+i] * dx
    # As the original value of negative indices refers to earlier values of h,
    # which is not what we want.
    # That is, when element j of h is patched with (1 - element j+i) of (1-h),
    # we ensure the sequence h and its shifted copy do not exceed domain bounds.
    
    # So let us simplify using np.correlate properly:
    # For h function values of [h[0], h[1], ..., h[n-1]]
    # The overlap integral for shift k can be obtained from shifted correlation
    # For the absolute overlap, we compute the correlation with 1-h, 
    # compute all values, and then get the maximum value among valid shifts.
    
    # Since our domain is [0,2], and we sample step function with n=45 points,
    # a shift of k = [0, 2) corresponds to index deltas of [0, n)
    # But also the corresponding correlation must be adjusted for domain,
    # and overlap is one integral per shift, which with the duality:
    # integral h(x)(1 - h(x+k)) dx
    
    # Here's an implementation using actual indexing for clean counters:
    
    # For shift k=0 it is h[0] * (1-h[0]), h[1]*(1-h[1]), etc.
    # which is higher in coefficient when we do the full correlation
    # As we compute integral we compute actual overlaps for all valid shifts.
    # To match domain: shift k = i * dx -> k = i * 2/n = 0 to 2
    # So from a *discrete* view, we compute the overlap integrals for shift values that span [0,2]
    # which means i from 0 to n-1 corresponds to k from 0 to 2
    # And the correlation index that gives that shift overlaps is the right part (n-1 to 2n-2) 
    # of the correlation array, corresponding to shift = 0 to n-1.
    # At index n-1 corresponds to k=0, index n corresponds to k=1, etc.
    # We take the last n values, which give the overlap for k = 0 to n-1. 
    # This is the right stock, so the correlation was already designed for this behavior.

    # However, the natural interpretation override: 
    # In the IDL that is np.correlate(h, 1-h, mode="full"), we get 2n-1. 
    # The overlaps are at index n-1 to 2n-2. Same as before.
    # But since shift k \in [0,2], then we step from 0 to 2, with step of 2/n 
    # So k = i * (2/n)  
    # The correlation gives the values using continuous indices of k = 0,1,...,n-1
    
    # So let's be clear:
    # h needs to be written in such a way that indexing index j by k corresponds to: 
    # overlap of shift k=2*j/n = 0 to 2
    # So for a delta dx=2/n, the shift k corresponds to correlation index i = k*n/2
    # That's why : 
    # i_corr = k_int / dx = k_int * n / 2  (hence k_int = i_corr * 2 / n)  
    # if k_int in [0, n-1]
    
    # So overlap at index i = n-1 + k_int 
    # Actually we need not do this analysis on top of 
    # correlation values directly.

    # A simpler working usage: 
    # Evaluate the overlap for shift k, where k = i * 2/n
    # For i from 0 to n-1; -> corr[n-1+i] * dx is the zeroth overlap normalized properly

    n = len(h)
    overlaps = corr[n-1:] * dx  # All valid overlaps for shift k from 0 to 2
    return np.max(overlaps)

def objective(h, n_points=45):
    """Objective to minimize: negative of the maximum overlap (C5)"""
    h = np.array(h)
    # Normalize the integral to 1:
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (1.0 / np.sum(h) * n_points * 2 / n_points)  # dx = 2/n_points
    # Call the overlap evaluator
    return -evaluate_erdos_solution(h, 2.0 / n_points)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45  # Fixed number of points
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Start from initial heuristic if available; otherwise initialize a simple one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Normalize to have integral=1 
        h0 = h0 * (1.0 / np.sum(h0) * n_points * dx)
    else:
        # Initialize with a simple non-negative ramp
        h0 = np.ones(n_points) * (0.5)
        # Make small tweaks to get integral=1
        h0 = h0 * (1.0 / np.sum(h0) * n_points * dx)
    
    # Prepare bounds [0,1] for each point
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the success callback to track optimization
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, dx)
    
    def callback(x, convergence):
        nonlocal best_h, best_c5
        h = np.array(x)
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * (1.0 / np.sum(h) * n_points * dx)
        c5_val = evaluate_erdos_solution(h, dx)
        if -c5_val < -best_c5:  # smaller is better
            best_h = h.copy()
            best_c5 = -c5_val
    
    # Perform optimization using differential evolution
    # It is robust and can handle the constraint of integral = 1 via the objective function
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            args=(n_points,),
            seed=seed,
            callbacks=[callback],
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-6,
            rtol=1e-6
        )
        
        if result.success:
            h = np.array(result.x)
            h = np.clip(h, 0, 1)
            if np.sum(h) > 0:
                h = h * (1.0 / np.sum(h) * n_points * dx)
        else:
            h = best_h.copy()
    except:
        # fallback if optimization has issues
        h = best_h.copy()
    
    # Ensure final normalization 
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (1.0 / np.sum(h) * n_points * dx)
    
    # Final evaluation
    c5_bound = evaluate_erdos_solution(h, dx)
    
    # Return the best solution found within budget
    return (list(h), -c5_bound, n_points)
