import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time
import random

def evaluate_C5(h, dx):
    """Compute C5 using FFT for efficiency"""
    n = len(h)
    # Zero-pad for full convolution without wraparound
    pad = n - 1
    h_padded = np.pad(h, (0, pad), mode='constant')
    ih_padded = np.pad(1 - h, (0, pad), mode='constant')
    conv = ifft(fft(h_padded) * fft(ih_padded).conj()).real
    # Extract valid overlaps (full convolution)
    valid_overlaps = conv[:2*n - 1]
    max_overlap = np.max(valid_overlaps) * dx
    return max_overlap

def project_to_constraints(h, dx):
    """Ensure ∫h = 1 and 0 ≤ h ≤ 1"""
    h = np.clip(h, 0, 1)
    # Adjust to meet integral constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-8:
        # Simple adjustment: scale to correct sum
        h = h * (1.0 / current_sum)
        h = np.clip(h, 0, 1)
    return h

def genetic_optimize(initial_h, n_points, dx, budget_s):
    """Optimize using genetic algorithm with local refinement"""
    best_h = initial_h.copy()
    best_C5 = evaluate_C5(best_h, dx)
    
    start_time = time.time()
    population_size = 20
    max_generations = 1000
    mutation_rate = 0.1
    elite_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    population = [project_to_constraints(initial_h + np.random.normal(0, 0.05, n_points), dx) 
                  for _ in range(population_size)]
    fitness = [evaluate_C5(h, dx) for h in population]
    
    for gen in range(max_generations):
        # Check timeout
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection (tournament)
        new_pop = [population[i] for i in range(elite_size)]
        
        while len(new_pop) < population_size:
            # Tournament selection (size 3)
            indices = random.sample(range(population_size), 3)
            selected_fitness = [fitness[i] for i in indices]
            winner_idx = indices[np.argmin(selected_fitness)]
            
            parent1 = population[winner_idx]
            parent2 = population[random.choice([i for i in range(population_size) if i != winner_idx])]
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = random.sample(range(n_points), max(1, int(0.1 * n_points)))
                for i in mutation_indices:
                    child[i] += random.gauss(0, 0.05)
            
            child = project_to_constraints(child, dx)
            new_pop.append(child)
            
        population = new_pop
        fitness = [evaluate_C5(h, dx) for h in population]
        
    return best_h, best_C5
