import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        dx = 2.0 / n_points
        h0 = np.ones(n_points) * 0.5  # Start with uniform distribution
        
    # CVXPY QCQP optimization
    def optimize_h(h_init):
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2.0)
        
        # Overlap constraints for all shifts
        dx = 2.0 / n_points
        for k in range(1, n_points):
            # Sliding window product
            expr = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    expr += h[i] * (1 - h[j])
            constraints.append(expr <= C * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Try different solvers
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.GLPK, verbose=False)
                except:
                    return h_init.copy(), 1e6  # fallback to initial
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return h.value.copy(), float(C.value)
        return h_init.copy(), 1e6  # fallback to initial

    # Initial optimization
    best_h, best_C = optimize_h(h0)
    
    # Local refinement with scipy
    def objective(h_vec):
        # Normalized by dx
        dx = 2.0 / n_points
        # Compute all correlations: correlate h with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        C = np.max(corr) * dx
        return C

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])

    # Optimize with scipy
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    options = {'maxiter': 500}
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options=options, callback=lambda x: None)
        if res.success and res.fun < best_C:
            best_h = res.x.copy()
            best_C = res.fun
    except:
        pass
    
    # Final QCQP optimization
    try:
        final_h, final_C = optimize_h(best_h)
        if final_C < best_C:
            best_h = final_h
            best_C = final_C
    except:
        pass
    
    # Track time
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h.copy(), best_C, n_points
    
    return best_h.copy(), best_C, n_points
