import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default: use a simple construction for n=45
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[-1] = 0.0
        h0[1] = 1.0
        h0[-2] = 1.0
    
    # Ensure integral constraint
    dx = 2.0 / n_points
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Start with initial solution
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simple gradient ascent approach with constraints
    max_iter = 10000
    step_size = 0.01
    decay = 0.9995
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient estimate via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[j] = (c5_plus - best_c5) / eps
            
        # Update step with constraints
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check for improvement
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        step_size *= decay
    
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Fast evaluation
    n = len(h)
    # Compute all correlations and find max
    corr = correlate(h, 1-h, mode='full')
    # Correlation is defined for lags from -n+1 to n-1
    # We take the negative part corresponding to k in [0,2]
    # k = 0 to 2 means shift 0 to n-1 in shifted array
    # shift = k/dx = k * n/2 = k*n/2
    # max_shift = 2 * n/2 = n
    max_k = n
    # but we only look at the part where 0 <= k <= 2, so from negative to positive shifts
    # for shift = 0, k = 0, corr[n-1] (offset from -to+ shift)
    # for shift = n, k = 2
    # corr indices map to k by index - (n-1) 
    # i: 0 to 2*n-2 
    # k: (i - (n-1)) * dx 
    # we want 0 <= k <= 2
    # 0 <= (i-(n-1)) * dx <= 2
    # 0 <= i - (n-1) <= 2*n/2 = n
    # n-1 <= i <= n-1 + n = 2*n-1 
    # but i goes from 0 to 2*n-2, so i from n-1 to 2*n-2
    positions = slice(n-1, 2*n-1)
    # We are mostly interested in negative values of i-(n-1), i.e. in lag we consider:
    # k = 0 -> i = n-1
    # k = 2 -> i = n+1  
    # but we want all valid k in [0,2], i in [n-1, 2*n-2]
    # This seems within bounds of our correlation computation
    vals = corr[positions] * dx
    return np.min(vals)
