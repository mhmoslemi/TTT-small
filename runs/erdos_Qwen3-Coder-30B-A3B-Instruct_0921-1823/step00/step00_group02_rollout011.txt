import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Try different discretizations
    n_points_list = [45, 60, 80, 100]
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n_points in n_points_list:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Initialize h from the initial construction if available
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values)
            h0 = h0[:n_points]
            if len(h0) < n_points:
                h0 = np.pad(h0, (0, n_points - len(h0)), 'constant', constant_values=0)
        else:
            # Create a reasonable initial guess
            h0 = np.ones(n_points) * 0.5
            dx = 2.0 / n_points
            # Adjust to satisfy integral constraint
            h0 = h0 * (1.0 / (np.sum(h0) * dx))
        
        # Create optimization variables
        h = cp.Variable(n_points)
        dx = 2.0 / n_points
        
        # CVXPY constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1]
        
        # Objective variable
        # We need to compute max_k integral h(x)(1-h(x+k)) dx
        # This is equivalent to max_k correlation of h with (1-h)
        # The correlation for each shift k is computed as:
        # sum_{i=0}^{n-1} h[i] * (1 - h[(i+k) % n])
        
        # Use a parameter for the max constraint
        max_overlap = cp.Variable()
        
        # Add constraints for each shift k
        for k in range(n_points):
            # Compute the correlation for shift k
            # This is sum_{i=0}^{n-1} h[i] * (1 - h[(i+k) % n])
            corr = cp.sum(cp.multiply(h, 1 - cp.hstack([h[k:], h[:k]])))
            constraints.append(corr <= max_overlap)
        
        # Objective is to minimize the maximum overlap
        objective = cp.Minimize(max_overlap)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, eps=1e-5, max_iters=5000)
        except:
            # If SCS fails, try with different solver
            try:
                prob.solve(solver=cp.OSQP, eps_rel=1e-5, eps_abs=1e-5)
            except:
                try:
                    prob.solve(solver=cp.GLPK, verbose=False)
                except:
                    continue  # Skip if optimization fails
        
        if prob.status == 'optimal':
            h_sol = h.value
            h_sol = np.array(h_sol)
            
            # Compute C5 value
            # This is the maximum of all the correlations
            correls = []
            for k in range(n_points):
                shifted_h = np.roll(h_sol, k)
                corr = np.sum(h_sol * (1 - shifted_h)) * dx
                correls.append(corr)
            
            max_corr = max(correls)
            
            if max_corr < best_c5:
                best_c5 = max_corr
                best_h = h_sol.copy()
                best_n = n_points
                
        # Early exit check
        if time() - start_time > budget_s * 0.95:
            break
    
    if best_h is None:
        # Fallback to initial construction or simple construction
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            best_h = np.ones(45) * 0.5
            dx = 2.0 / 45
            best_h = best_h * (1.0 / (np.sum(best_h) * dx))
        best_n = 45
        # Compute approximate C5
        dx = 2.0 / best_n
        correls = []
        for k in range(best_n):
            shifted_h = np.roll(best_h, k)
            corr = np.sum(best_h * (1 - shifted_h)) * dx
            correls.append(corr)
        best_c5 = max(correls)
    
    return (best_h.tolist(), best_c5, best_n)
