import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant for given h"""
    n = len(h)
    # Compute all correlations for shifts k = 0 to 2*dx
    correlation = np.correlate(h, 1 - h, mode='full')
    # The overlap integral for shifts k from 0 to 2
    # We only care about k in [0,2], so we extract the right values
    # correlation corresponds to shift k = 0 ... 2*n-2, but we want 0 to 2
    # The shift k is in the range [0, 2] with step dx
    valid_indices = range(n-1, 2*n-1)
    shift_values = np.array([i*dx - (n-1)*dx for i in valid_indices])
    valid_correlations = correlation[valid_indices]
    # All valid k values in [0, 2] correspond to shift indices from n-1 to 2*n-2
    c5 = np.max(valid_correlations * dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / np.sum(h) * 2  # To make integral 1

    # Precompute dx
    dx = 2.0 / n_points
    
    # Normalize our starting h to satisfy the constraint
    h = h / np.sum(h) * 2  # integral should be 1
    
    # Use scipy minimize with bounds and constraints
    # Optimization will minimize negative of C5 and maximize C5, so we
    # minimize -C5 to maximize C5, but that's not what we want.
    # We want to minimize C5 directly, so we'll use a heuristic approach:
    # Anneal using gradient descent with projected gradient on simplex
    def objective(h_vec):
        h_current = np.copy(h_vec)
        return evaluate_erdos_solution(h_current, dx)
    
    # Use a local optimization approach
    best_c5 = 1e10
    best_h = np.copy(h)
    max_steps = 10000
    step = 0
    restart_count = 0
    
    while time.time() - start_time < budget_s - 2:
        # Each optimization step
        x0 = np.copy(h)
        # Project to constraints and optimize
        # We'll do gradient descent implicitly while maintaining constraints
        
        # Simple local search with small perturbations
        for i in range(500):
            if time.time() - start_time > budget_s - 2:
                break
            h_current = np.copy(x0)
            
            # Small random change
            h_perturbed = np.copy(h_current)
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = h_perturbed + noise
            # Keep values in [0, 1]
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Re-normalize
            h_perturbed = h_perturbed / np.sum(h_perturbed) * 2
            
            # Compute current C5
            c5_current = objective(h_perturbed)
            
            # Accept if lower
            if c5_current < best_c5:
                best_h = np.copy(h_perturbed)
                best_c5 = c5_current
                if best_c5 < 0.3807:
                    break
                x0 = np.copy(h_perturbed)
            else:
                # Accept with low probability if not much worse
                if np.random.rand() < 0.01:
                    x0 = np.copy(h_perturbed)
            
            step += 1
            if step > max_steps:
                break

        # Try another random restart
        restart_count += 1
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * 2
        x0 = h_rand
        # If very small improvement, end
        if best_c5 < 0.3807 and restart_count > 5:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, len(best_h))
