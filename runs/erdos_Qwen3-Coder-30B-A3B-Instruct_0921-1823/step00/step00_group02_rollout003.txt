def evaluate_h(h, dx):
    """Evaluate the C_5 constant for a given step function h"""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')

    # which corresponds to k in [0, 2]
    mid = len(corr) // 2
    # Use the maximum value (it should be in range [0,2])
    max_val = np.max(corr[mid:])  # Only including non-negative shifts (k >= 0)
    return max_val * dx
