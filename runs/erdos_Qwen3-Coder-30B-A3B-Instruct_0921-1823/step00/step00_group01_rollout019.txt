import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, n_points):
    dx = 2.0 / n_points
    # Compute the maximum overlap integral
    conv = np.correlate(h_values, 1 - h_values, mode="full")
    max_conv = np.max(conv) * dx
    return max_conv

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:
        # Create a default starting point
        n_points = 47
        h0 = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    # Ensure the initial solution satisfies the constraint
    h0 = h0 * (1.0 / np.sum(h0) * n_points * dx)
    
    # Optimization variables: h[i] for i=0..n_points-1
    # Bounds for h[i]: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize (C5)
    def objective(h):
        return evaluate_h(h, n_points)
    
    # Start timing
    start_time = time.time()
    
    # Use COBYLA method, iterative optimization
    # Due to budget constraints, we limit the iterations
    max_iter = 5000
    iteration = 0
    
    # Keep track of best solution
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h, n_points)
    
    # Define callback to track best result
    def callback(h):
        nonlocal best_h, best_c5, iteration
        iteration += 1
        current_c5 = evaluate_h(h, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if iteration >= max_iter:
            return True  # Stop optimization
    
    # Run optimization
    try:
        result = minimize(
            objective,
            h0,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            callback=callback,
            options={'maxiter': max_iter, 'disp': False}
        )
        
        if not result.success:
            # If optimization failed, still return best found
            pass
        else:
            current_c5 = evaluate_h(result.x, n_points)
            if current_c5 < best_c5:
                best_h = result.x.copy()
                best_c5 = current_c5
                
    except Exception:
        pass
    
    # Final check of best solution
    final_c5 = evaluate_h(best_h, n_points)
    
    # Ensure constraint is met
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-10
    
    return (best_h.tolist(), float(final_c5), int(n_points))
