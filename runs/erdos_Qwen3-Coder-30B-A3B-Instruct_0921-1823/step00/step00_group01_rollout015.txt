import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_C5(h, dx):
    # Compute full convolution of h with (1-h) and return max
    h_vec = np.array(h)
    kernel = 1 - h_vec
    # Use FFT for efficiency
    n = len(h_vec)
    # Full convolution - pad with zeros to avoid circular effects
    padded_len = 2 * n - 1
    H = fft(np.pad(h_vec, (0, n-1), 'constant'))
    K = fft(np.pad(kernel, (0, n-1), 'constant'))
    conv = ifft(H * K).real
    # Crop to meaningful range of correlations
    # correlations for shifts -n+1 to n-1, we need the positive shifts
    # just take max of all non-zero lags in the overlap integral:
    # Overlap = integral h(x)(1-h(x+k)) dx, so we look for good k values
    # max value over all k from 0 to 2*dx = return the peak from full conv
    return np.max(conv[n-1:])

def project_to_feasible(h, dx):
    # Ensure sum = 1 via rescaling
    h_sum = np.sum(h) * dx
    if abs(h_sum - 1.0) > 1e-10:
        h = h / h_sum * 1.0 / dx  # Scale to target sum
    # Project onto [0,1] box constraints
    h = np.clip(h, 0, 1)
    return h
