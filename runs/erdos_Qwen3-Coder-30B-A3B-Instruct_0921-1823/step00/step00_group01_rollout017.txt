import numpy as np
from scipy.optimize import minimize
from scipy import fft
import math
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for a step function h."""
    # Compute convolution of h with (1-h) using FFT
    h_padded = np.pad(h, (0, len(h)), mode='constant')
    ones_padded = np.pad(1 - h, (0, len(h)), mode='constant')
    conv = fft.irfft(fft.rfft(h_padded) * fft.rfft(ones_padded))
    # Extract full convolution (first 2*len(h)-1 elements)
    conv_full = conv[:2*len(h)-1]
    # Find maximum value scaled by dx
    return np.max(conv_full) * dx

def project_onto_constraints(h, dx):
    """Project h onto feasible region: [0,1] and sum(h)*dx = 1."""
    # Bound h to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly 1/dx (which is n_points/2)
    h_sum = np.sum(h)
    target_sum = 1 / dx
    if h_sum != 0:
        h *= target_sum / h_sum
    return h
