from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        # Normalize to integral = 1
        h = h * 2.0 / n_points
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Define helper functions
    def compute_c5(h_vec, step):
        # Using FFT for faster convolution
        h_fft = np.fft.fft(h_vec)
        one_minus_h = 1.0 - h_vec
        one_minus_h_fft = np.fft.fft(one_minus_h)
        conv_result = np.fft.ifft(h_fft * np.conj(one_minus_h_fft)).real
        # Take maximum over all shifts (full convolution)
        max_val = np.max(conv_result)
        # Scale by dx
        return max_val * step
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def project_to_feasible(h_vec):
        # Maintain boundary constraints and integral constraint
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec) * dx
        if abs(current_sum) > 1e-10:
            h_vec = h_vec * (1.0 / current_sum)
        return np.clip(h_vec, 0, 1)
    
    # Gradient-based optimization with constraints
    def obj_func(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_func(h_vec):
        return constraint_sum(h_vec)
    
    # Check if we have a reasonable starting point
    penalty = 1e5
    def penalty_obj(h_vec):
        val = obj_func(h_vec)
        # Add penalty for constraint violation
        con_val = constraint_sum(h_vec)
        return val + penalty * (abs(con_val) ** 2)
    
    # Optimization loop
    max_iter = 2000
    iter_count = 0
    
    # Simulated Annealing style with random mutations
    temp = 0.1
    decay_rate = 0.995
    lastbest = best_c5
    patience = 0
    patience_limit = 50
    
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        h_new = h.copy()
        # Slightly perturb
        idx = np.random.randint(0, n_points)
        delta = (np.random.rand() - 0.5) * temp
        h_new[idx] += delta
        # Project
        h_new = project_to_feasible(h_new)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            h = h_new.copy()
        
        temp *= decay_rate
        
        # Check for stagnation
        if abs(lastbest - best_c5) < 1e-6:
            patience += 1
        else:
            patience = 0
            lastbest = best_c5
            
        if patience >= patience_limit:
            patience = 0
            temp = max(temp * 0.9, 1e-6)
            
        if iter_count % 100 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 10:
                break

    desired_sum = 1.0
    
    def obj(h_vals):
        h_vec = np.array(h_vals)
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (desired_sum / (np.sum(h_vec) * dx))
        return compute_c5(h_vec, dx)
    
    res = minimize(obj, best_h, method='Powell', 
                   bounds=[(0, 1) for _ in range(n_points)],
                   options={'maxiter': 1000})
    
    if res.success:
        best_h = np.clip(res.x, 0, 1)
        best_h = best_h * (desired_sum / (np.sum(best_h) * dx))
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
