import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    # Compute correlation using FFT to speed up convolution
    H = fft(np.concatenate([h, [0] * (2*n-1)]))
    H_conj = np.conj(H)
    h_conj = fft(np.concatenate([h[::-1], [0] * (2*n-1)]))
    overlap_fft = ifft(H * h_conj)
    overlap = np.real(overlap_fft[:2*n-1])
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h):
    """Project h to satisfy sum constraint and bounds"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (len(h) / 2) / h_sum
    return h
