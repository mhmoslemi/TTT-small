    h_vec = h_init.copy()
    
    # Compute overlap integral manually
    def compute_C5(h_vals):
        n = len(h_vals)
        dx = 2.0 / n
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Take the relevant parts (with shifts) and compute max
        # For full convolution, we have 2*n - 1 elements, representing shifts from -(n-1) to (n-1)
        # The full convolution results from step functions on [0,2]

        # We can calculate the valid range and get the maximum value
        k_range = 2 * n - 1
        shift_k = np.arange(-(n-1), n)
        # Shifts in range [0,2] corresponding to h_vals starting at 0
        # k corresponds to shift in the range we care about
        # But we want the specific case of convolution on [0,2] for valid k in [0,2]
        # Let's identify valid integral values using original method
        # Shift k = shift_k in [-(n-1), (n-1)]

        # This maps to full correlation, but the maximal shift in our domain that

        # Actual maximization over all discrete values

        C5 = 1.0 - dx * np.min(corr)
        return C5

    def objective(h_vec):
        return compute_C5(h_vec)

    return (best_h.tolist(), best_C5, len(best_h))
