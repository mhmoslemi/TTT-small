import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum value of the convolution
    max_conv = np.max(conv)
    return max_conv * dx

def run(seed=42, budget_s=600, **kwargs):

    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize
    def obj_func(x):
        return evaluate_c5(x)
    
    # Optimization parameters
    max_iter = 5000
    start_time = time.time()
    
    # Run optimization
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Use COBYLA which handles constraints well
    result = minimize(obj_func, h0, method='COBYLA', bounds=bounds,

    # If optimization was successful and found better solution
    if result.success:
        final_h = result.x
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Get the final result
    return (best_h.tolist(), best_c5, n_points)
