import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a reasonable starting point
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        h_init[0] = 0.2
        h_init[1] = 0.8
        h_init[-1] = 0.2
        h_init[-2] = 0.8

    # Ensure integral constraint is met
    dx = 2.0 / n_points
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define optimization variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C >= 0]
        
    # Compute all correlations for all shifts
    # The overlap integral is: max_k int h(x)(1 - h(x+k)) dx
    # For discrete case, we compute sum(h[i]*(1 - h[i+k])) for all valid k
    
    # Minimize C over all possible shifts
    # We'll compute it directly
    for k in range(1, n_points):
        corr = 0
        for i in range(n_points - k):
            corr += h[i] * (1 - h[i + k])
        constraints.append(corr <= C * dx)
    
    # Objective
    objective = Minimize(C)
        
    # Problem
    prob = Problem(objective, constraints)
    
    # Try to solve with ECOS
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = np.array(h.value).flatten()
            c5_value = C.value
        else:
            # Fall back to less constrained solving with scipy
            c5_value = minimize_overlap_scipy(h_init, n_points, dx, start_time, budget_s)
            h_opt = h_init.copy()
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    except:
        # Fallback to scipy
        c5_value = minimize_overlap_scipy(h_init, n_points, dx, start_time, budget_s)
        h_opt = h_init.copy()
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
    return (h_opt, c5_value, n_points)

def minimize_overlap_scipy(h_init, n_points, dx, start_time, budget_s):
    def overlap_objective(h):
        # Compute the maximum overlap integral
        max_overlap = 0
        for k in range(1, n_points):
            overlap = 0
            for i in range(n_points - k):
                overlap += h[i] * (1 - h[i + k])
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_integral})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use SLSQP with bounds and constraint
    res = minimize(overlap_objective, h_init, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6})
    
    if time.time() - start_time > budget_s * 0.95:
        return 0.5  # Conservative fallback
    
    # If we got a solution, recompute overlap
    if res.success:
        return overlap_objective(res.x)
    else:
        # Fallback estimate
        return 0.4  # conservative value
