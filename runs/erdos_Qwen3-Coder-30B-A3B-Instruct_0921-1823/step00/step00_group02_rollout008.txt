from scipy.optimize import differential_evolution
import time

def evaluate_c5(h):
    # h is a numpy array of length n
    # Compute the full convolution of h with (1 - h)
    # and return the maximum value
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    conv = np.correlate(h, h_rev, mode='full')
    conv = conv[len(conv)//2:]  # Take the positive lags part
    # Compute 1 - h for each element
    one_minus_h = 1 - h

    # Do the convolution with 1-h
    conv2 = np.correlate(h, one_minus_h_rev, mode='full')
    conv2 = conv2[len(conv2)//2:]
    # The integral value will be scaled by dx
    # The maximum of these convolutions over all lags
    max_val = np.max(conv2 * dx)
    return max_val
