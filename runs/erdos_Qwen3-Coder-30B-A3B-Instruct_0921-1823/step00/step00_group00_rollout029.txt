import numpy as np
import cvxpy as cp
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Normalize to have integral = 1
    h0 = h0 * (1.0 / np.sum(h0) * n_points * dx)
    
    # Build the optimization problem
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) == n_points / 2.0)
    
    # Add quadratic constraints for all shifts k
    # We'll do it in a smarter way: for each k, compute the correlation
    # Evaluate correlations
    max_k = 2 * n_points
    correlations = []
    for k in range(n_points):
        # For each shift k, we compute the correlation between h and (1 - h_k)
        # This is equivalent to: sum(h[i] * (1 - h[i + k])) for valid i
        # We'll broadcast this computation
        
        # Left side of h
        left_len = n_points - k
        if left_len > 0:
            corr = cp.sum(cp.multiply(h[:left_len], 1 - h[k:k+left_len]))
            correlations.append(corr)
        else:
            break
    
    # Create constraint for the maximum of correlations
    for corr in correlations:
        constraints.append(C5 >= 1 - corr * dx)
    
    # Objective function
    objective = cp.Minimize(C5)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with some time limit
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = C5.value
        else:
            h_opt = h0
            c5_opt = float('inf')
    except:
        h_opt = h0
        c5_opt = float('inf')
    
    # If we didn't get a better solution, try numerical optimization
    if c5_opt >= 0.38085 or time.time() - start_time > budget_s * 0.8:
        # Try a simpler gradient based approach (gradient descent, not very accurate)
        def objective_func(h_vals):
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2.0) / np.sum(h_vals)
            # Compute correlations
            correlations = []
            for k in range(n_points):
                if k == 0:
                    left_len = n_points
                else:
                    left_len = n_points - k
                if left_len > 0:
                    corr = np.sum(h_vals[:left_len] * (1 - h_vals[k:k+left_len]))
                    correlations.append(corr)
            return 1.0 - (1.0/len(correlations)) * np.mean(correlations)  # Minimize this

        # Optimize
        bounds = [(0, 1) for _ in range(n_points)]
        res = optimize.minimize(objective_func, h0, method='L-BFGS-B', bounds=bounds, 
                                tol=1e-6, options={'maxiter': 10000})
        if res.success and time.time() - start_time < budget_s:
            h_opt = res.x
            c5_opt = objective_func(h_opt)

    # Final evaluation
    final_h = np.clip(h_opt, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    correlations = []
    for k in range(n_points):
        if k == 0:
            left_len = n_points
        else:
            left_len = n_points - k
        if left_len > 0:
            corr = np.sum(final_h[:left_len] * (1 - final_h[k:k+left_len]))
            correlations.append(corr)
    final_c5 = 1.0 - np.max(correlations) * dx
    
    return (final_h.tolist(), final_c5, n_points)
