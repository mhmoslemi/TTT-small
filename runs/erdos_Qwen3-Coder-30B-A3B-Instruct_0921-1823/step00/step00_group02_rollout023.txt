import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    n = len(h)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the convolution
    max_overlap = np.max(conv) * dx
    return max_overlap

    def objective(h):
