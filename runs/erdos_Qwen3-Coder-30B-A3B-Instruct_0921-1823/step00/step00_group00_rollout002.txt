import numpy as np
from scipy import optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Fallback: create a simple initial guess
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h0 = np.array(h0)
    h0 = h0 * (n_points / 2) / np.sum(h0)  # scale to sum = n_points/2
    
    # Define optimization variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Compute the full convolution with (1-h), and add constraints
    # We will add one constraint for each shift k
    for k in range(n_points):
        # convolution at shift k
        conv_val = 0
        for i in range(n_points):
            if i+k < n_points:
                conv_val += h[i] * (1 - h[i+k])
        constraints.append(C5 >= conv_val * dx)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Create the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cp.SCS, eps=1e-5, max_iters=5000, time_limit=budget_s)
        h_opt = h.value
        c5_bound = C5.value
    except Exception as e:
        # If optimization fails, return the initial solution
        h_opt = h0
        conv_vals = []
        for k in range(n_points):
            corr = 0
            for i in range(n_points):
                if i+k < n_points:
                    corr += h0[i] * (1 - h0[i+k])
            conv_vals.append(corr * dx)
        c5_bound = max(conv_vals)
    
    # Ensure the solution is valid before returning
    if not np.all(h_opt >= 0) or not np.all(h_opt <= 1):
        h_opt = np.clip(h_opt, 0, 1)
    if abs(np.sum(h_opt) * dx - 1.0) > 1e-5:
        h_opt = h_opt * (1.0 / (np.sum(h_opt) * dx))  # project to correct sum
    
    return (h_opt, c5_bound, n_points)
