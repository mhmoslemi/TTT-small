import numpy as np
from scipy.optimize import minimize
import time

def evaluate(h, n_points):
    # Compute the overlap integral for given h
    dx = 2.0 / n_points
    # Correlate h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum of these correlations corresponds to the maximum overlap
    max_overlap = np.max(corr) * dx
    return max_overlap

        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
    # Define constraints
    def constraint_eq(hv):
        return np.sum(hv) * (2.0 / n_points) - 1.0
    
    # We'll optimize the h values using scipy.optimize
    def objective(hv):
        return evaluate(hv, n_points)
        
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
