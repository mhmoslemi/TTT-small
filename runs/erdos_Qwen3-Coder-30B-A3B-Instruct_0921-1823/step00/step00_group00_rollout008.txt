import time
from scipy.signal import correlate

    start_time = time.time()

    n_points = 100
    domain_length = 2.0
    dx = domain_length / n_points
    
    # Initialize particles
    def initialize_particle():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Scale to meet constraint
        h = np.clip(h, 0, 1)  # Enforce bounds
        return h

        # Flip h for correlation computation
        flipped_h = np.flip(h)
        # Compute full correlation with (1-h)
        correlation = correlate(h, flipped_h, mode="full")
        # Only keep relevant part
        correlation = correlation[len(h)-1:]
        # Scale by dx
        correlation *= dx
        # Return maximum value
        return np.max(correlation)

    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iters = 500

    particles = [initialize_particle() for _ in range(n_particles)]

    # Personal best initialization
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_C5(p) for p in particles]
    
    # PSO main loop
    for iteration in range(max_iters):

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)

            scale_factor = (n_points / 2) / np.sum(particles[i])
            particles[i] *= scale_factor

            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness

                if fitness < global_best_fitness:
