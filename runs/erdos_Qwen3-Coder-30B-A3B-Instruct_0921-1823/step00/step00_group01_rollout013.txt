def evaluate_erdos_solution(h, dx=2.0/len(h)):
    """Evaluate the overlap integral C5 for a given step function h."""
    # Compute the convolution of h with (1-h)
    corr = signal.correlate(h, 1-h, mode='full')
    # Extract the relevant part (shifts from -1 to 1)
    start_idx = len(h) - 1
    end_idx = 2 * len(h) - 1
    overlap = corr[start_idx:end_idx]
    return np.max(overlap) * dx
