from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    """Compute C5 bound for given h"""
    n = len(h)
    # Compute full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum correlation gives us C5 (scaled by dx)
    return np.max(corr) * dx
