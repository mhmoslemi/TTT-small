import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate C5 for given h values."""
    # Compute full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # Take the maximum value, then multiply by dx
    return np.max(conv) * dx

def project_to_feasible(h, dx):
    """Project h to feasible region: [0,1] and sum = 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to sum to 1 (exact)
    h_sum = np.sum(h) * dx
    if h_sum != 0:
        h = h / h_sum * (2.0 / dx)  # make sum = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable starting point (simple step function)
        n_points = 47
        h = np.ones(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 1.5
        h = project_to_feasible(h, 2.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        old_c5 = best_c5
        # Make a random update
        # Pick a random index
        i = np.random.randint(0, n_points)
        # Perturb h[i]
        delta = np.random.uniform(-0.1, 0.1) 
        h[i] += delta
        
        # Project to feasible set
        h = project_to_feasible(h, dx)
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        else:
            # Restore previous h if no improvement
            h = best_h.copy()
        
        # Stop if no improvement in last 100 iterations
        if iteration - last_improvement > 100:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
