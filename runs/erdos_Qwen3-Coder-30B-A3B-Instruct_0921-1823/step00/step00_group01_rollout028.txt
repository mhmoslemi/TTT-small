import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C₅ for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum value is our C₅
    c5 = np.max(conv) * dx
    return c5

        h0 *= (1.0 / np.sum(h0)) * (1.0 / dx)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for integral = 1
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Initial C5 value
    best_c5 = objective(h0)
    best_h = h0.copy()

    start_time = time.time()
    
    # Run optimization with iteration limit to save time
    try:
        # For initial exploration, let's do one pass with COBYLA
        res = minimize(
            objective,
            h0,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2000, 'disp': False}
        )
        if res.success:
            c5 = objective(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x.copy()
        
        # Try another small local optimization
        h_temp = best_h + np.random.normal(0, 0.01, n_points)

            h_new = np.clip(h_new, 0, 1)
            h_new *= (1.0 / np.sum(h_new)) / dx
            h_temp = h_new
            
    except Exception as e:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
