import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, n_points):
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h)
    conv = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum value in the convolution corresponds to the C5 value
    c5 = np.max(conv) * dx
    return c5

        h_init[:n_points//2] = 0.0
        h_init[n_points//2:] = 1.0
    
    # Ensure the integral constraint
    dx = 2.0 / n_points
    h_init = h_init * (1.0 / (np.sum(h_init) * dx))  # normalize
    
    # Store best solution
    best_h = h_init.copy()
    best_c5 = evaluate_h(h_init, n_points)
    
    # Set up constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize C5
    def objective(x):
        return evaluate_h(x, n_points)
    
    # Optimization parameters
    maxiter = 1000
    options = {'maxiter': maxiter, 'disp': False}
    
    start_time = time.time()
    
    # Run optimization with bounds and constraints
    try:
        result = minimize(objective, h_init, method='COBYLA', 
                          bounds=bounds, constraints=constraints, 
                          options=options, callback=lambda x: None)
        if result.success:
            fit_h = result.x
            fit_c5 = evaluate_h(fit_h, n_points)
            if fit_c5 < best_c5:
                best_c5 = fit_c5
                best_h = fit_h.copy()
    except Exception as e:
        pass
    
    # Further optimization using a few iterations of gradient-free method

    # We'll try a simple local search using random perturbations
    n_iter = 0

        # Renormalize
        h_trial = h_trial * (1.0 / (np.sum(h_trial) * dx))
        
        # Evaluate
        c5_trial = evaluate_h(h_trial, n_points)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        n_iter += 1
    
    return (best_h.tolist(), best_c5, n_points)
