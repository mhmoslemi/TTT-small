def evaluate_h(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the C5 bound
    c5 = np.max(corr) * dx
    return c5
