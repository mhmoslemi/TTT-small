import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default construction: 45 points
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    # Discretization
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h0.copy()
    best_C = float('inf')
    
    # Try to improve starting point with QCQP
    solver = 'ECOS'
    
    # Setup QCQP
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,  # integral constraint
        C >= 0
    ]
    
    # Add overlap constraints for all shifts k
    for k in range(1, n_points):
        # Compute overlap integral for shift k
        # This is equivalent to convolution-like constraint
        # sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k]) <= C * dx
        expr = cp.sum(cp.multiply(h[:-k], 1 - h[k:]))
        constraints.append(expr <= C * dx)
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            # Extract solution
            h_solution = h.value
            C_value = C.value
            if C_value < best_C:
                best_C = C_value
                best_h = h_solution.copy()
    except:
        pass
    
    # If QCQP didn't work, try gradient based method with bounds
    # Initialize with best solution so far
    h_current = best_h.copy()
    C_current = best_C
    
    # Switch to iterative refinement with scipy.optimize if needed
    from scipy.optimize import minimize
    
    # Function to compute overlap integral
    def compute_C(h_vec):
        h_vec = np.array(h_vec)
        # Length of the shifted h vector for each k
        overlaps = []
        for k in range(1, n_points):
            # Compute sum of h[i] * (1 - h[i+k])
            if k >= n_points:
                break
            overlap = np.sum(h_vec[:-k] * (1 - h_vec[k:]))
            overlaps.append(overlap)
        if not overlaps:
            return 0
        return max(overlaps)
    
    # Objective function for minimization with constraints
    def obj_func(h_vec):
        C_val = compute_C(h_vec)
        return C_val
    
    # Constraints in the form required by scipy.optimize
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points/2
    
    # Define constraints for boundary values
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Do gradient-based optimization if time allows
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.7:
        try:
            result = minimize(obj_func, h_current, method='L-BFGS-B',
                              bounds=bounds, constraints=constraints,
                              options={'ftol': 1e-8, 'gtol': 1e-6})
            if result.success:
                h_opt = result.x
                C_opt = compute_C(h_opt)
                if C_opt < C_current:
                    C_current = C_opt
                    h_current = h_opt
        except:
            pass
    
    # Final evaluation
    final_C = compute_C(h_current)
    
    return (h_current.tolist(), final_C, n_points)
