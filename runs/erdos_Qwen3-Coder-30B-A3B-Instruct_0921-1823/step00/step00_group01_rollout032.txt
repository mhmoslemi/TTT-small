def evaluate_c5(h):

    max_k = n - 1

    # So we're interested in the k = 0 case, then k = dx, 2*dx, etc.
    # For now, just use the max of the full correlation to get a bound
    # There's an easier and accurate way - compute all full overlaps
    # h(x) * (1 - h(x+k)) for all shifts k, then take max
    # Here, we do it by evaluating corr at appropriate lags
    # The valid lags go from -n+1 to n-1
    # Max overlap is when we slide h over (1-h) and get max contribution
    # corr[k + n - 1] corresponds to shift k, where k = 0..n-1
    # Normalize the convolution
    return np.max(corr) * dx

def project_h(h):
    """Project h onto [0,1]^n, maintain sum = 1."""

    n = len(h)
    dx = 2.0 / n
    total = np.sum(h) * dx
    if total > 0:
        h = h / total * 1.0
    return h
