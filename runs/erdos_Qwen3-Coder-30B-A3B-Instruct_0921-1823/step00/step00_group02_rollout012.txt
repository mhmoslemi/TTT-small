import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given step function h"""
    n = len(h)
    # Compute all correlations for shifts k in [0, 2] with step dx
    # This is a simplified version that samples k values at dx intervals
    # For the full computation, we compute the maximum over all k
    # The convolution directly gives us the integral for each shift
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is scaled by dx
    max_corr = np.max(corr) * dx
    return max_corr

        h[-1] = 0.0
        h = h / np.sum(h) * 2.0  # normalize to integral 1 over [0,2]
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Optimization loop
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Random perturbation
        h_new = h.copy() + np.random.normal(0, 0.01, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * 2.0  # Renormalize
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Use the best solution so far
        h = best_h.copy()
        
        # Occasionally add more structure
        if i % 100 == 0 and i > 0:
            # Local optimization
            def obj(h_vals):
                h_here = h_vals.copy()
                dx_here = 2.0 / n_points
                return evaluate_h(h_here, dx_here)
            
            # Constraint that integral equals 1
            def constraint(h_vals):
                return np.sum(h_vals) * dx - 1.0
            
            cons = {'type': 'eq', 'fun': constraint}
            
            # Minimize using a local optimizer
            try:

                pass  # Skip if optimization fails
            
            # Update best if improved
            current_c5 = evaluate_h(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
