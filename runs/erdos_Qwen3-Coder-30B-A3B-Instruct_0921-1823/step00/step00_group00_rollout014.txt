import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define the C5 optimization problem
    def compute_C5(h):
        # Compute all correlations for shifts
        correlations = np.correlate(h, 1 - h, mode='full')
        # Only take the valid ones (positive shifts)
        valid_corr = correlations[n_points-1:]
        return np.max(valid_corr) * dx
    
    # Define constraint for integral
    def integral_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # For avoiding local minima, we'll try multiple restarts with 
    # slightly perturbed initial points from the starting solution
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Try more direct optimization using scipy
    def objective(h):
        return compute_C5(h)
    
    def constraint_func(h):
        return integral_constraint(h)
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Run multiple optimizations with different random noise
    for i in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        # Slightly perturb initial guess
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            result = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-6},
                tol=1e-6
            )
            
            if result.success:
                h_opt = result.x
                C5_val = compute_C5(h_opt)
                if C5_val < best_C5:
                    best_h = h_opt
                    best_C5 = C5_val
        except:
            continue
    
    # Finally try one major optimization with CVXPY if time allows
    if time.time() - start_time < budget_s - 5:
        try:
            # Formulate as QCQP problem
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Objective is to minimize C
            objective = cp.Minimize(C)
            
            # Constraint for integral
            integral_con = cp.sum(h_vars) == n_points / 2
            
            # Constraint for each k  
            constraints = [h_vars >= 0, h_vars <= 1, integral_con]
            
            # For each shift
            for k in range(1, n_points):
                # Compute Correlation
                corr = 0
                for i in range(n_points - k):
                    corr += h_vars[i] * (1 - h_vars[i + k])
                constraints.append(corr <= C / dx)
            
            # Solve QCQP problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                C5_cvx = C.value
                if C5_cvx < best_C5:
                    best_h = h_cvx
                    best_C5 = C5_cvx
                    
        except:
            pass
            
    return (best_h.tolist(), best_C5, n_points)
