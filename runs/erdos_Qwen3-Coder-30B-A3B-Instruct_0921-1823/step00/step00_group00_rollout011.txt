import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Fallback: create a reasonable initial guess
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[-1] = 1.0
        h0 /= np.sum(h0) * 2.0 / n_points
    
    # Refine the initial solution using convex optimization
    # Variables: h[i] ∈ [0,1] and sum(h) = n/2
    h = cp.Variable(n_points, name="h")
    
    # Objective: minimize max_k int h(x)(1 - h(x+k)) dx = max_k (h * (1-h))[k]
    # So minimize the maximum of the cross-correlation of h and 1-h
    # Since we will compute full convolution, we consider only k=0,1,2..(2n-2)
    # and then find max over all of those
    dx = 2.0 / n_points
    
    # We'll compute the overlap integral by evaluating the full convolution
    # of h with (1-h) to get a vector of overlaps
    # The overlaps are stored in full mode interpretation
    
    # Constraint: integral of h equals 1
    # sum(h)*dx = 1 => sum(h) = n/2
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
    
    # Since the objective is a maximum of a set of quadratic terms,
    # we model the problem differently:
    # Introduce a variable C for the upper bound
    # C >= int h(x)(1 - h(x+k)) dx for all k
    
    # For quick construction, let's discretize deeply
    # But in QCQP setup, directly model with one constraint per k
    
    # Because of matrix-vector constraints, we will limit to reasonable time
    # and directly solve simpler QCQP with explicit constraints
    
    # Instead, we try gradient descent or proximal methods for refinement
    # Let's adopt a stochastic gradient approach due to time constraints
    
    # Initialize h to current solution
    h_vals = h0.copy()
    
    # Hyperparameters
    step_size = 0.01
    max_iter = 10000
    total_iter = 0
    
    def compute_C5(h_values):
        shifted = 1 - h_values
        # Use scipy's full cross-correlation
        corr = correlate(h_values, shifted, mode='full')
        # Normalize to get the C5 bound estimate
        # Shift into range index 0 to n-1 of overlap values
        # Full operation: n-1 to 2*n-2 for positive shifts
        # We want the maximum over all shifts
        # Since we care about the actual C5 constant,
        # which corresponds to the max integral across all valid shifts
        # Which corresponds to max value among all full convolution:
        # Shift indices 0 to 2*n-2 = 2*n-1 terms
        # Maximum of the correlation
        max_corr = np.max(corr)
        # this represents the C5 constant as:
        # C5 = max_k int h(x)(1 - h(x+k)) dx = max_k (h * (1-h))[k]
        # The C5 we want is thus just max_corr * dx
        return max_corr * dx
    
    # Loss function: minimize max(overlap integral) which is max_convolution_value * dx
    current_C5 = compute_C5(h_vals)
    
    # Optimization loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 2: # leave margin
            break
            
        # Compute gradient of overlap integral with respect to h
        shifted = 1 - h_vals
        
        # Manually compute Jacobian / gradient estimation
        # Using finite differences, or one analytical direction
        #    ∂/∂h_i int h(x) (1 - h(x+k)) dx = (1 - h(x+i)) - h(x+k)
        # For large systems, that's computationally costly.
        # Instead use just numerical gradient for all colors in parallel:
        eps = 1e-6
        
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            # Approximate gradient:
            h_plus = h_vals.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_minus = h_vals.copy()
            h_minus[i] -= eps
            if h_minus[i] < 0:
                h_minus[i] = 0
            
            loss_plus = compute_C5(h_plus)
            loss_minus = compute_C5(h_minus)
            
            grad[i] = (loss_plus - loss_minus) / (2*eps)
         
        # Update
        h_vals -= step_size * grad
        
        # Project back to constraints [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Enforce sum = n/2
        h_vals *= (n_points / 2) / np.sum(h_vals)
        
        # Check for improvement
        new_C5 = compute_C5(h_vals)
        if new_C5 < current_C5:
            current_C5 = new_C5
        else:
            # If no improvement, reduce step size
            step_size *= 0.995
            
        total_iter += 1
    
    # Return the final iteration's result
    return (h_vals, current_C5, n_points)
