import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h, dx):
    """Evaluate the C5 constant for a given step function h"""
    # Compute the maximum overlap integral
    n = len(h)
    # Create the flipped function (1 - h)
    flipped = 1 - h
    # Compute full convolution using FFT for efficiency
    conv = np.convolve(h, flipped, mode='full')
    # The overlap integral is the maximum value of the convolution
    # when properly normalized
    max_overlap = np.max(conv) * dx
    return max_overlap

def constraint_sum(h, dx):
    """Enforce integral constraint: sum(h) * dx = 1"""
    return np.sum(h) * dx - 1.0

def penalty_function(h, dx, penalty_weight=1e6):
    """Add penalty for constraint violations"""
    penalty = 0
    # Constraint violation penalty for sum
    sum_violation = abs(constraint_sum(h, dx))
    penalty += penalty_weight * sum_violation
    # Boundary constraint penalty
    for val in h:
        if val < 0:
            penalty += penalty_weight * (0 - val)
        elif val > 1:
            penalty += penalty_weight * (val - 1)
    return penalty

def objective_function(h, dx):
    """Objective function to minimize: C5 constant with penalties"""
    penalty = penalty_function(h, dx)
    try:
        value = evaluate_h(h, dx)
        return value + penalty
    except Exception:
        return 1e10 + penalty  # Return large value on error

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize with a simple start point
    h0 = np.ones(n_points) * 0.5
    
    # Use differential evolution for global optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with time control
    def optimized_objective(h):
        return objective_function(h, dx)
    
    result = differential_evolution(
        optimized_objective,
        bounds,
        maxiter=500,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: time.time() - start_time > budget_s/2 if time.time() - start_time > budget_s/2 else False
    )
    
    # Refine solution if time allows
    best_h = result.x.copy()
    best_value = result.fun
    
    # Additional optimization attempts
    for i in range(3):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate a new candidate solution
        new_h = best_h + np.random.normal(0, 0.05, n_points)
        # Clip to [0, 1] range
        new_h = np.clip(new_h, 0, 1)
        # Rescale to satisfy constraint
        scaled_sum = np.sum(new_h) * dx
        new_h = new_h / scaled_sum * 1.0
        new_h = np.clip(new_h, 0, 1)
        
        # Evaluate and accept if better
        new_value = objective_function(new_h, dx)
        if new_value < best_value:
            best_h = new_h
            best_value = new_value
            
    # Final check and adjustment
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = best_h / final_sum * 1.0
        
    c5_bound = evaluate_h(best_h, dx)
    
    return (best_h, c5_bound, n_points)
