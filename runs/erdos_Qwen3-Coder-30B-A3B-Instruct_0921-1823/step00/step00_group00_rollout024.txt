import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iters = 500
    n_particles = 50
    w = 0.7
    c1 = 2
    c2 = 2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # One particle at initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Use a simple construction
        h0 = np.ones(n_points) * 0.5
    
    # Normalize to integral = 1
    h0 = h0 * (1.0 / np.sum(h0) * dx)
    particles.append(h0.copy())
    velocities.append(np.zeros_like(h0))
    
    # Rest are random
    for i in range(1, n_particles):
        h = np.random.rand(n_points)
        h = h * (1.0 / np.sum(h) * dx)
        particles.append(h.copy())
        velocities.append(np.random.randn(n_points) * 0.1)
    
    # Best ever
    best_particle = particles[0].copy()
    best_value = compute_c5(particles[0], dx)
    
    # PSO loop
    start_time = time.time()
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            p_best = best_particle
            v = (w * velocities[i] + 
                 c1 * r1 * (p_best - particles[i]) + 
                 c2 * r2 * (best_particle - particles[i]))
            
            # Update particle
            new_h = particles[i] + v
            
            # Apply bounds
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize
            new_h = new_h * (1.0 / np.sum(new_h) * dx)
            
            # Evaluate
            new_value = compute_c5(new_h, dx)
            
            # Update personal best
            if new_value < compute_c5(particles[i], dx):
                particles[i] = new_h.copy()
                velocities[i] = v.copy()
                
                # Update global best
                if new_value < best_value:
                    best_value = new_value
                    best_particle = new_h.copy()
    
    return (best_particle.tolist(), best_value, n_points)

def compute_c5(h, dx):
    # Compute overlap integral for all shifts and return max
    h = np.array(h)
    n = len(h)
    # Correlate h with (1-h) - full mode
    corr = signal.correlate(h, 1-h, mode="full")
    # Extract the overlapping parts (all shifts from -1 to 1)
    # The full conv is of length 2n-1
    # We only care about shifts in [-1, 1]
    # shift k from -1 to 1 corresponds to indices n-1-k to n-1+k
    # So indices n-1 to 2n-2 for positive k
    # and indices 0 to n-2 for negative k
    
    # But actually we want all valid shifts (k from -1 to 1)
    # Which corresponds to correlation indices from 0 to 2n-2
    # And valid overlapping range is approximately k in [-1,1]
    # which is correlation indices from n-1 to n+1 inclusive
    
    # The maximum valid 'k' in the problem is from -1 to 1
    # and we should check from index (n-1) to (n+1) in the full corr
    
    # but we want the maximum k_in [-1,1] 
    # The full correlation gives us shift values from - (n-1) to (n-1)
    # But our range k goes from -1 to 1 only. So we only care about corr[n-1:2n-2] 
    
    # Actually given our discretization, we should compute overlap
    # for each possible shift k in [-1,1] 

    # Each shift k corresponds to correlation index n-1+k 
    # So shift k=0 -> index n-1
    # Shift k=1 -> index n
    # Shift k=-1 -> index n-2
    
    # But we're indexing to n-1+k in corr = [0, 2n-2]
    # Hence valid k are from -(n-1) to (n-1)
    # But since we're analyzing overlap on [0,2], shift k=1 gives h(x) and h(x+1)
    # which is a meaningful shift.
    
    # We will evaluate all shifts from -1 to 1
    # Interval [0,2], so shift can range from -1 to 1, actual shift values
    # In discretized version, shifts are from -n+1 to n-1  (so if n=45, -44 to 44)
    # But we're interested in plausible adjustments: sliding the function
    # Shifts k = -1, -0.9, ..., 0.9, 1 (in terms of interval length, not index)
    # Most useful k values can be in [-0.5, 0.5] (switching region)
    # In fact let's just compute for indices 0 to 2n-2 and extract reasonable shifts
    
    # Our approach: we're rather computing overlap over the range,
    # and maximum shift range overlaps should be about 
    # [-1/2, 1/2] scaled by dx.
    # Since our domain is [0,2], shift from -1 to 1 is valid.
    
    # We take full correlation, and we extract the overlapping part:
    # if n=45, correlation is of length 89
    # Take indices n-1 to 2n-2 = 44 to 88 to get shift range from 0 to k_max
    # 
    # But we actually don't have a good way to map the correlation index to k.
    # Let's do: 
    # In overlap integral \int h(x) (1 - h(x+k)) dx
    # The k can vary from 0 to 2, so we're computing overlap for all
    # possible shifts from (k in [-1,1]) domain.
    # if h is 45 points (dx=2/45), k=1 means you shift by dx*45 = 2
    # So shift in [0,2] corresponds to k in [0,2] 
    # and a proper k from the problem is from -1 to 1.
    # Ok, so correlation from -1 to 1 = 90 values (if we have 45 points).
    # When we do correlation of h with (1-h) in full mode, we get a vector of size 89.
    # The middle element corresponds to k=0, indices are from 0 to 88.
    # So, k can be from -2 to 2 with dx=2/45 
    # so shift -1 translates to corr index 45-1 = 44
    # shift 1 gives index 45+1 = 46 
    # shift 0 gives index 45
    # We want k = -1, 0, 1 -> indices 44, 45, 46 in the corr vector
    
    # we want to extract the valid k in [-1,1]
    # To get that, consider the correlation indices:
    # for shift k in [-1,1], we map k to index = n-1+k
    # So k ranges from -1 to 1 -> index ranges from n-2 to n
    # Let's check if that is right.  Let's visualize:
    # if h_index = i, to compute overlap with (1-h) shifted by k, 
    # you have h[x] * (1-h[x+k]) -> multiply h[i]  * (1-h[i+k/dx])  
    # 
    # But simpler: just use the standard overlap.
    overlap = corr[n-1:]
    # overlap[0] corresponds to k = 0 
    # and indices 2, 3, ... for k = 1, 2, ...
    # overlap array has 2*n-1 elements, 0 to 2*n-2
    # Hence shift k ranges from -(n-1) to (n-1)

    # The current implementation uses full correlation, then we recall what it means
    # Based on derivation:
    # Full correlation of h with (1-h) gives values that represent overlap for shift k
    # from - (n -1) to (n-1)
    # Expected k range in [-1,1], so we're looking at
    # k = -1, 0, 1 => indices n-2, n-1, n in the array
    
    # However, overlapping is only well-defined for k ∈ [-1,1], which gives the range

    # The key is that k values range from -1 to 1 (on interval [0,2] which has length 2)
    # With n = 45, dx=2/45, 
    # k = 1 corresponds to shifting by dx * (n-1) = (2/45)*(44) =

    # The key is that we measure shift values k in relative scale of interval length
    # that is: \int h(x)(1-h(x+k)) dx for k ∈ [-1,1] is what is meant.
    
    # So for k from -1 to 1 on interval [0,2] with n points, the corresponding discrete k = -1,0,1 correct  
    # Positions are indices n-1+(-1), n-1, n-1+1 
    # Therefore the desired overlap values are corr[n-2], corr[n-1], corr[n]
    
    # Cluster of 3 values: shift -1, 0, 1
    max_overlap = np.max(overlap[n-2:n+1])  # or corr[n-2:n+1]
    
    # However, not all points may matter to calculate)
    # So we do full computation from all shifts
    # that should yield the right result
    
    # to compute overlap for any shift k:
    # int h(x) (1-h(x+k)) dx
    # We sample it at points x=ix*dx, x=0 to 2-dx  
    # so the integral sum over i of h[i] * (1 - h[i-shifts])
    # For shifting we need to pad. 
    # Alternatively, use actual convolution
    # Return max of all possible k from -1 to 1 with dx steps
    
    # Let us re-calculate more clearly.
    
    # Let’s do total overlap computation with correct indices for shift range.
    # Shift k from -1 to 1, dx = 2/n -> so shift k = -1 to 1 corresponds to indices -n+1 to n-1
    # To be precise, 1 gcd shift on (0,2) interval maps to dx * (change in x)
    
    # Not to overthink I'll just pass simple values of full correlation
    return np.max(overlap)
