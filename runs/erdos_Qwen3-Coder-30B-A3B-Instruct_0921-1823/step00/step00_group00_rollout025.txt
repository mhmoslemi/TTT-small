import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 45
    dx = 2.0 / n_points
    target_integral = 1.0
    
    # Initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h / np.sum(initial_h) * target_integral / dx
    else:
        # Create a reasonable initial guess
        initial_h = np.ones(n_points) * 0.5
        initial_h[10:35] = 1.0  # Heuristic: some high values indicating overlap constraints
        
    # Normalize to meet integral constraint
    initial_h = initial_h / np.sum(initial_h) * target_integral / dx
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        # Reflect h to get 1-h
        reflected_h = 1 - h
        # Compute full convolution
        conv = np.correlate(h, reflected_h, mode='full')
        # Take the maximum value and scale
        max_val = np.max(conv) * dx
        return max_val
    
    # Penalty for constraint violations
    def penalty(h):
        # Bounds
        penalty_bounds = np.sum(np.maximum(0, h) + np.maximum(0, 1-h))
        # Integral constraint
        integral = np.sum(h) * dx
        penalty_integral = abs(integral - target_integral)
        return 10000 * (penalty_bounds + penalty_integral)
    
    # Objective function for optimization
    def objective(h):
        h = np.clip(h, 0, 1)
        return compute_C5(h) + penalty(h)
    
    # PSO-related stuff
    def pso_optimize():
        # PSO parameters
        n_particles = 50
        w = 0.7
        c1 = 2.0
        c2 = 2.0
        max_iters = 500
        
        # Initialize particles
        particles = []
        velocities = []
        pbest_positions = []
        pbest_scores = []
        
        for i in range(n_particles):
            if i == 0:
                h = initial_h.copy()
            else:
                h = np.random.rand(n_points) * 0.5
            # Normalize
            h = h / np.sum(h) * target_integral / dx
            particles.append(h.copy())
            velocities.append(np.random.randn(n_points) * 0.1)
            pbest_positions.append(h.copy())
            pbest_scores.append(objective(h))
        
        gbest_idx = np.argmin(pbest_scores)
        gbest_score = pbest_scores[gbest_idx]
        gbest_position = pbest_positions[gbest_idx].copy()
        
        # Timing
        start_time = time.time()
        
        for it in range(max_iters):
            if time.time() - start_time > budget_s - 5:
                break
                
            for i in range(n_particles):
                h = particles[i]
                v = velocities[i]
                
                # Update velocity
                r1, r2 = np.random.rand(2)
                v = w * v + c1 * r1 * (pbest_positions[i] - h) + c2 * r2 * (gbest_position - h)
                
                # Update position
                new_h = h + v
                new_h = np.clip(new_h, 0, 1)
                
                # Adjust to make integral correct
                new_h = new_h / np.sum(new_h) * target_integral / dx
                
                # Evaluate
                new_score = objective(new_h)
                
                # Update pBest
                if new_score < pbest_scores[i]:
                    pbest_scores[i] = new_score
                    pbest_positions[i] = new_h.copy()
                    
                    if new_score < gbest_score:
                        gbest_score = new_score
                        gbest_position = new_h.copy()
                
                # Update particles
                particles[i] = new_h.copy()
                velocities[i] = v.copy()
        
        return gbest_position, gbest_score
    
    # Run PSO
    best_h, best_score = pso_optimize()
    
    # Final refinement with differential evolution
    # Keep final best_h normalized by sum
    best_h = best_h / np.sum(best_h) * target_integral / dx
    
    # Differential evolution
    def de_objective(h):
        h = np.clip(h, 0, 1)
        return compute_C5(h) + penalty(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        result = differential_evolution(
            de_objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed,
            strategy='best1bin',
            disp=False
        )
        if result.success:
            final_h = result.x.copy()
            final_h = final_h / np.sum(final_h) * target_integral / dx
            final_score = compute_C5(final_h)
            if final_score < best_score:
                best_h = final_h
                best_score = final_score
    except:
        pass
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
