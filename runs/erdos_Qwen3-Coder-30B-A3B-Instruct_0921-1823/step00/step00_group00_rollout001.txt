import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        # Default starting point
        n_points = 45

        h0[0] = 0.0
        h0[-1] = 1.0
        
    dx = 2.0 / n_points
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Add the overlap constraints
    for k in range(n_points):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute the overlap integral with shift k
        # We have to be careful because correlation needs full
        # We'll calculate using direct sum indices
        if k == 0:
            # h(x) * (1 - h(x))
            integral = cp.sum(cp.multiply(h, 1 - h)) * dx
        else:
            # h(x) * (1 - h(x+k)) -> will be a partial overlap
            # Indexing x from 0 to n-k-1, see what's available
            # 1D correlation can be computed directly
            
            # Simplified: compute the sum for indices 0 to n-k-1 for h(x+k)  
            # Use numpy-style full convolution to make sure we have a maximum
            # h[i] hits 1-h[i+k] for indices consistent
            
            # Build the shifted part for k
            # integral = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) * dx
            
            integral = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:])) * dx

        constraints.append(C5 >= integral)

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers, with time limit
    solver = 'CBC'
    try:
        prob.solve(solver=solver, timeout=budget_s - (time.time() - start_time) - 1)
    except:
        try:
            prob.solve(solver='SCIP', timeout=budget_s - (time.time() - start_time) - 1)
        except:
            try:
                prob.solve(solver='ECOS', timeout=budget_s - (time.time() - start_time) - 1)
            except:
                # last resort
                pass

    # Extract results
    if prob.status != cp.OPTIMAL:
        h_final = h0

        h_vals = h_final

            if i == 0:
                integral = np.sum(h_vals * (1 - h_vals)) * dx
