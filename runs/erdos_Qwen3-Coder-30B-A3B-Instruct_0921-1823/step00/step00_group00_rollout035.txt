import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.4
        h0[-1] = 0.6
    
    # Normalize to ensure integral = 1
    h0 = h0 * (1.0 / np.sum(h0) * dx)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        # Compute the overlap integral for all shifts
        h_flip = 1 - h
        # Use convolution to compute all overlaps efficiently
        corr = np.correlate(h, h_flip, mode="full")
        # The full correlation has length 2*n_points - 1
        # We only need the shifts in [0, 2), which are to the right
        # indices are from n_points-1 to 2*n_points-2
        # The shift amount is given by k = t - (n_points - 1)
        # which gives k in [0, 2)
        # But actually, we're interested in t from n_points-1 to 2*n_points-2
        # so k = t - (n_points - 1) ranges from 0 to n_points-1 (shifts up to 1)
        # and we want t from n_points-1 to 2*n_points - 2  
        # This gives us shifts: 0 to n_points - 1 (but we only want those up to 2) 
        # Actually, the shifts are such that it consider the overlap for x+k
        # Let's use a more careful approach: 
        # For shift of k, we integrate h(x) * (1 - h(x + k))
        # compute all possible shifts from 0 to 2 (step of dx)
        # But using the correlation approach:
        # corr_val[t] = sum over x of h[x] * (1 - h[x + (t - (n_points-1))])
        # shift = t - (n_points - 1) = k
        # So shift k = t - (n_points - 1)
        # t ranges from 0 to 2*n_points-2
        # So k ranges from -n_points+1 to n_points-1
        # We're interested in k in [0,2] 
        start_k = n_points - 1
        overlaps = []
        for t in range(start_k, min(2*n_points - 1, start_k + int(2/dx) + 1)):
            k = t - (n_points - 1)
            if k < 0 or k > 2:
                continue
            overlaps.append(corr[t] * dx)
        if not overlaps:
            return 1e10
        return max(overlaps)
    
    # Boundary constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for better global optimization
    start_time = time.time()
    
    # Get remaining time for the next optimization call
    time_left = budget_s - (time.time() - start_time)
    if time_left < 1:
        return (h0, objective(h0), n_points)
    
    # Run with constraints
    try:
        # Rescale h0 to meet integral constraint
        h0_rescaled = h0 * (1.0 / np.sum(h0) * dx)
        res = differential_evolution(
            objective, bounds, 
            maxiter=500 if time_left > 10 else 50, 
            seed=seed,
            disp=False,
            popsize=15,
            tol = 1e-6 if time_left > 50 else 1e-4,
            mutation=(0.5, 1.0),
            recombination=0.7,
            polish=True
        )
        h_sol = res.x
        h_sol = np.clip(h_sol, 0, 1)
        # Normalize again to ensure integral = 1 (but not needed, it should be)
        c5_bound = objective(h_sol)
        return (h_sol, c5_bound, n_points)
    except Exception as e:
        # Default fallback if anything goes wrong
        return (h0, objective(h0), n_points)

# If the script were run directly, simulate the call
if __name__ == "__main__":
    h, c5, n = run()
    print(f"Result: h = {h}, C5 = {c5}, n = {n}")
