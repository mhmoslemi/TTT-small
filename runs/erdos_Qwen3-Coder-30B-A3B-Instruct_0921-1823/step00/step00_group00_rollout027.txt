    constraints = [h >= 0, h <= 1]
    
    # Constraint: integral = 1
    constraints.append(cp.sum(h) * dx == 1)

    # We need to handle all shifts k from 0 to n_points-1
    # For each shift k, the overlap integral for h and 1-h is:
    # int h(x) (1 - h(x+k)) dx
    # This corresponds to a convolution: h * (1-h) from the right
    
    # Compute all correlation values as constraints
    max_c5 = cp.Variable()
    # Create the full correlation vector
    # For now, let's try a more direct approach with optimization
    
    # Try solving a smooth approximation:
    # The constraint is that for each shift k:
    # int h(x) (1 - h(x+k)) dx <= C5 for all k
    # So we want max_k int h(x) (1 - h(x+k)) dx
    
    # We will solve this using scipy's minimize with custom constraints
    def compute_C5(h_vals):
        # Convolving h with (1-h)
        h_array = np.array(h_vals)
        h_flip = 1 - h_array
        # Full convolution
        corr = np.correlate(h_array, h_flip, mode='full')
        # This gives us all the shifted summations
        # Since we have dx spacing, we scale
        # But shifts could be from -n+1 to n-1
        # The valid range is from 0 to 2*n-2 indices
        # And the proper shifts are from 0 to n_points-1
        idx_offset = n_points - 1  # This centers the shift
        # We're interested in the correlation of h with 1-h
        # i.e. for shift k (from 0 to n_points-1):
        # int h(x) (1-h(x+k)) dx = sum h[i] * (1-h[i+k]) 
        # But we can just use the convolution for shifts from 0 to n_points-1
        # The indices are [0, 1, ..., 2*n - 2]
        # We want indices from n-1 to 2*n-2 which correspond to shifts 0, 1, ..., n-1
        shifts = corr[n_points-1:]
        return np.max(shifts) * dx
        
    def objective(h_vals):
        # We want to minimize max (h(x) * (1-h(x+k))) = minimize max(overlap) 
        return compute_C5(h_vals)
    
    def constraint_func(h_vals):
        return 1 - np.sum(h_vals) * dx  # Should be close to zero
        
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    def my_obj(h_vals):
        return objective(h_vals)
    
    # Use L-BFGS-B as it's more robust for this kind of constrained
    # Let's simply try a smart initial guess and use optimization
    # Let's also try using Scipy.minimize with trust-constr which is good for bounds
    res = optimize.minimize(my_obj, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                            options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9})
    
    # Extract result
    h_opt = res.x
    c5_value = compute_C5(h_opt)
    
    # Return the solution
    return (h_opt.tolist(), float(c5_value), n_points)
