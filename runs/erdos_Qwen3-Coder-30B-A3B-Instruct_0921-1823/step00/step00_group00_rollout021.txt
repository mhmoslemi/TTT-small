import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Fallback: create initial step function
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
    
    # Normalize to integral = 1
    h0 = h0 * (n_points / (2 * np.sum(h0)))
    
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # We ignore the first (n_points-1) elements and take the rest
        c5_vals = corr[len(h)-1:] * dx
        return np.max(c5_vals)
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_func(h):
        return np.sum(h) * (2.0 / len(h)) - 1.0
    
    # Boundaries
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimizer settings
    method = 'SLSQP'
    options = {'maxiter': 5000}
    
    # Record best solution
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # First try with initial solution
    current_c5 = evaluate_c5(h0)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h0.copy()
    
    # Optimization loop with timing control
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 100:
        # Random perturbation for exploration (independent of previous)
        h_trial = h0.copy()
        # Perturb a few random indices
        idx = np.random.choice(n_points, size=min(10, n_points//4), replace=False)
        for i in idx:
            # Add small random change
            h_trial[i] = max(0, min(1, h_trial[i] + np.random.normal(0, 0.05)))
        
        # Re-normalize to satisfy integral constraint
        h_trial = h_trial * (n_points / (2 * np.sum(h_trial)))
        
        # Optimize the perturbation
        res = minimize(objective, h_trial, method=method, bounds=bounds, 
                       constraints=constraints, options=options, 
                       tol=1e-6)
        
        if res.success:
            optimized_h = res.x
            new_c5 = evaluate_c5(optimized_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = optimized_h.copy()
        
        iter_count += 1
    
    # Final refinement
    res = minimize(objective, best_h, method=method, bounds=bounds, 
                   constraints=constraints, options=options, tol=1e-7)
    if res.success:
        final_h = res.x
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    return (best_h, best_c5, n_points)
