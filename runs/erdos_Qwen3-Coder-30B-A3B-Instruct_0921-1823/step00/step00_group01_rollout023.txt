import numpy as np
from scipy import signal
import time

def evaluate_c5(h, dx):
    """Evaluate C5 using FFT for efficiency"""
    # Compute convolution of h with (1-h) using full mode
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take maximum value and multiply by dx
    c5 = np.max(conv) * dx
    return c5

def project_h(h, dx):
    """Project h to satisfy sum constraint"""
    target_sum = 1.0
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h_proj = h * scale
    return np.clip(h_proj, 0, 1)
