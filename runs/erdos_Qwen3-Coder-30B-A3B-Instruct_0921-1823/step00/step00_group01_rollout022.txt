import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the maximum overlap integral C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum correlation is over all possible shifts [0, 2]
    # The scale factor is dx for the integral approximation
    return np.max(corr) * dx
