import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_c5(h, dx):
    # Compute convolution of h(x) and (1-h(x+k)) for all k
    h_folded = np.concatenate([h, np.zeros_like(h)])
    one_minus_h = 1 - h
    one_minus_h_folded = np.concatenate([one_minus_h, np.zeros_like(one_minus_h)])
    conv = ifft(fft(h_folded) * np.conj(fft(one_minus_h_folded))).real
    # Trim to relevant size (full convolution)
    conv = conv[:len(h)]
    return np.max(conv) * dx

def constraint_sum(h, target_sum):
    return np.sum(h) - target_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.3
        h[3] = 0.7
        h[4] = 0.2

    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Optimization
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    def obj(h_vec):
        # Ensure bounds
        h_clipped = np.clip(h_vec, 0, 1)
        # Restore sum constraint
        current_sum = np.sum(h_clipped)
        h_scaled = h_clipped * target_sum / current_sum
        # Evaluate
        c5 = evaluate_c5(h_scaled, dx)
        return c5
    
    def obj_grad(h_vec):
        # Simple finite difference gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_val = obj(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (obj(h_plus) - base_val) / eps
        return grad
    
    def optimize_step():
        nonlocal h, best_h, best_c5
        
        # Start with local optimization
        h_start = h.copy()
        
        # Use scipy minimize with COBYLA for constrained optimization
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, target_sum)}]
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            result = minimize(obj, h_start, method='COBYLA', constraints=cons, bounds=bounds,
                              options={'maxiter': 50, 'disp': False})
            if not result.success:
                return
            h_new = result.x
            c5_new = obj(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except:
            pass

    # Adaptive coarse-to-fine strategy
    scales = [32, 47, 64, 97, 128]
    current_scale = 0
    
    while time.time() - start_time < budget_s - 5:
        if current_scale < len(scales):
            new_n_points = scales[current_scale]
            if new_n_points > n_points:
                # Interpolate to new size
                old_x = np.linspace(0, 2, n_points)
                new_x = np.linspace(0, 2, new_n_points)
                h = np.interp(new_x, old_x, h)
                n_points = new_n_points
                dx = 2.0 / n_points
            current_scale += 1
        else:
            # Increase resolution incrementally
            new_n_points = min(n_points + 10, 256)
            if new_n_points > n_points:
                old_x = np.linspace(0, 2, n_points)
                new_x = np.linspace(0, 2, new_n_points)
                h = np.interp(new_x, old_x, h)
                n_points = new_n_points
                dx = 2.0 / n_points
        
        optimize_step()
        
        # Occasionally do a random perturbation to escape local minima
        if np.random.rand() < 0.2 and time.time() - start_time < budget_s - 10:
            noise_magnitude = 0.05
            perturbation = np.random.normal(0, noise_magnitude, n_points)
            h = np.clip(h + perturbation, 0, 1)
            # Renormalize
            current_sum = np.sum(h)
            h = h * target_sum / current_sum
            optimize_step()
            
        # Early stopping for last few seconds
        if time.time() - start_time > budget_s - 10:
            break

    # Final optimization
    try:
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, target_sum)}]
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(obj, best_h, method='COBYLA', constraints=cons, bounds=bounds, 
                          options={'maxiter': 100, 'disp': False})
        if result.success:
            best_h = result.x
            best_c5 = obj(best_h)
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
