import numpy as np
import cvxpy as cp
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except NameError:
        # Fallback to reasonable initial construction
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        h_init[0] = 0.0  # Start with lower value
        h_init[-1] = 0.0  # End with lower value

    # Normalize initial guess to satisfy constraint
    dx = 2.0 / n_points
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Set up optimization variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # For the QCQP, we'll minimize C5 such that
    # for all k: integral h(x)(1 - h(x+k)) dx <= C5
    
    # We compute the autocorrelation terms:
    # For small k, we'll fold this into the problem
    for k in range(n_points):
        # This is the k-th shifted correlation
        # Indexing: h[i] * (1 - h[(i+k) % n_points])
        # Due to circular correlation, this is a complex conv. pattern
        # Simplified approach: For each k, create the correlation constraint
        expr = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h[i] * (1 - h[j])
        expr *= dx  # Scale by dx
        # We need expression <= C5 for all k
        constraints.append(expr <= C5)
    
    # Minimize C5 subject to constraints
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Initialize with starting guess
    h.value = h_init.copy()
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, time_limit=budget_s - (time.time() - start_time))
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            best_h = h.value
            best_c5 = float(C5.value)
        else:
            # Fallback: use the initial solution
            best_h = h_init
            best_c5 = float(np.max([np.sum(best_h[i:] * (1 - np.roll(best_h, -i)[:n_points-i])) * dx
                                   for i in range(n_points)]))
    except Exception as e:
        # If solver fails or times out, use initial solution
        best_h = h_init
        best_c5 = float(np.max([np.sum(best_h[i:] * (1 - np.roll(best_h, -i)[:n_points-i])) * dx
                               for i in range(n_points)]))
    
    return (best_h, best_c5, n_points)
