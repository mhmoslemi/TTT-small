import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Fallback: construct simple initial solution
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Normalize to integral constraint ∫h = 1
    h0 = h0 * (1.0 / np.sum(h0)) * (n_points / 2)
    
    # QCQP solver approach
    def solve_qcqp(h_init, max_time=100):
        if max_time < 5:
            return h_init.copy(), 1e10
        
        # Variables
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Compute correlations
        for k in range(1, n_points):
            corr = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr += h[i] * (1 - h[i + k])
            constraints.append(corr <= C * dx)
        
        # Minimize C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h.value
                c5_val = C.value
            else:
                h_opt = h_init.copy()
                c5_val = 1e10
        except:
            h_opt = h_init.copy()
            c5_val = 1e10
            
        return h_opt, c5_val
    
    # Local optimization around the initial solution
    h_best = h0.copy()
    c5_best = 1e10
    
    def objective(h):
        # Evaluate C_5 = max_k ∫ h(x)(1 - h(x+k)) dx
        h = np.array(h)
        correlations = np.correlate(h, 1-h, mode="full")
        c5 = np.max(correlations) * dx
        return c5
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1-h])
    
    # BFGS optimization
    cons = {'type': 'eq', 'fun': constraint_integral}
    bounds = [(0, 1) for _ in range(n_points)]
    iterations = 0
    
    while time.time() - start_time < budget_s - 5:
        try:
            result = minimize(objective, h_best, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100})
            if result.success:
                h_new = result.x
                c5_new = objective(h_new)
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
            else:
                break
        except:
            break
        iterations += 1
    
    # QCQP refinement step
    if time.time() - start_time < budget_s - 10:
        h_final, c5_refined = solve_qcqp(h_best, budget_s - (time.time() - start_time) - 10)
        if c5_refined < c5_best:
            h_best = h_final
            c5_best = c5_refined
    
    # Final check of constructed solution
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (n_points / 2) / np.sum(h_best)
    c5_final = objective(h_best)
    
    return (h_best.tolist(), c5_final, n_points)
