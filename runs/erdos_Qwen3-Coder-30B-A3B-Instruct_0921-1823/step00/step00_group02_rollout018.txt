def evaluate_c5(h_vals, dx):
    h = np.array(h_vals)
    # Evaluate the overlap integral C₅ = max_k ∫ h(x)(1 - h(x+k)) dx
    # We can evaluate this efficiently by computing
    # np.correlate(h, 1-h, mode="full") and taking max
    h_minus = 1 - h
    corr = np.correlate(h, h_minus, mode="full")
    c5 = np.max(corr) * dx
    return c5
