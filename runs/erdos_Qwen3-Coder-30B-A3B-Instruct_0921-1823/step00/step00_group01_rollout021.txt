import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_solution(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    # Compute correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value, which corresponds to the maximum overlap
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to feasible(h, dx):
    """Project h to satisfy the constraints."""
    # Ensure h[i] ∈ [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to make integral = 1
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback construction
        n_points = 50
        h = np.full(n_points, 0.5)
        h[:n_points//2] = 1.0
        h = project_to_feasible(h, 2.0 / n_points)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_solution(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    iter_count = 0
    same_best_count = 0
    patience = 100
    
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        # Compute current correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        
        # The shift k that achieves maximum overlap
        # For full mode, indices [0, 2*n-2] map to shifts [-n+1, n-1]
        k = max_idx - (n_points - 1)
        
        # Create deviation vector with small changes
        dev = np.zeros(n_points)
        if k >= 0:
            # Positive shift: h[i] affects h[i+k] in correlation
            # We want to reduce contribution of h[i+k] to (1-h[i])
            # This amounts to reducing h[i+k] by some epsilon
            for i in range(n_points - k):
                dev[i] += 0.01  # Small increase in h[i+k] 
                dev[i + k] -= 0.01
        else:
            # Negative shift: shift by abs(k) into the future
            for i in range(-k, n_points):
                dev[i] += 0.01
                dev[i + k] -= 0.01
                
        # Apply small changes to h
        h_new = h + dev
        h_new = project_to_feasible(h_new, dx)
        
        # Evaluate new solution
        c5_new = evaluate_solution(h_new, dx)
        
        # Accept improvement or with some probability
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            same_best_count = 0
        else:
            same_best_count += 1
            
        iter_count += 1
        
        # Early termination if no significant improvements
        if same_best_count > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
