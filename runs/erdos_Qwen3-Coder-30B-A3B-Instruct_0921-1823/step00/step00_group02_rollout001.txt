import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Start with a reasonable number of points
        n_points = 45
        h0 = np.ones(n_points) * 0.5  # Initial guess
    
    dx = 2.0 / n_points
    # Ensure integral constraint is satisfied
    h0_sum = np.sum(h0)
    if h0_sum > 0:
        h0 = h0 * (n_points / 2) / h0_sum
    
    # Optimize
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iteratively refine with different numbers of points
    for n in [45, 60, 80, 100, 120]:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            h_opt = optimize_h(n, dx, budget_s - (time.time() - start_time))
            if h_opt is not None:
                c5_val = compute_c5(h_opt, dx)
                if c5_val < best_c5:
                    best_h = h_opt
                    best_c5 = c5_val
        except Exception:
            continue
    
    return (best_h.tolist(), best_c5, len(best_h))

def optimize_h(n_points, dx, remaining_time):
    if remaining_time < 0.1:
        return None
    
    # Initialize variables
    h = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2  # integral constraint
    ]
    
    # Objective: minimize maximum overlap
    # This is a bit tricky in cvxpy - we will compute the full correlation
    # and use a second-order cone constraint
    h_rev = h[::-1]
    corr = cp.convolve(h_rev, 1 - h, mode='full')
    
    # The overlap C_5 is max of all values in the convolution
    C5_var = cp.Variable()
    constraints.append(corr <= C5_var)
    constraints.append(C5_var >= 0)
    
    # Objective: minimize C5
    objective = cp.Minimize(C5_var)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=500, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = np.array(h.value).flatten()
            return h_solution
    except:
        pass
    
    # Fallback to grid search or local optimization if required
    return None

def compute_c5(h, dx):
    n_points = len(h)
    # Compute all correlations (full convolution)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
