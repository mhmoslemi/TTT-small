import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_h(h):
    # Compute overlap integral C5 using FFT for efficiency
    n = len(h)
    dx = 2.0 / n
    # pad for full convolution
    h_padded = np.concatenate([h, np.zeros(n-1)])
    one_minus_h = 1 - h
    one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n-1)])
    
    conv_result = ifft(fft(h_padded) * np.conj(fft(one_minus_h_padded))).real
    # extract the full convolution
    c5_vals = conv_result[n-1:] * dx
    c5 = np.max(c5_vals)
    return c5

def constraint_sum(h):
    n = len(h)
    dx = 2.0 / n
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Generate a reasonable starting point
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.3
        h[-1] = 0.3
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    prev_c5 = best_c5
    tolerance = 1e-5
    same_count = 0
    
    # Basic gradient descent helper
    def optimize_step(h, step_size=0.01):
        n = len(h)
        dx = 2.0 / n
        # make a copy
        h_new = h.copy()
        
        # compute gradient approximation
        grad = np.zeros(n)
        eps = 1e-4
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            c5_plus = evaluate_h(h_plus)
            c5_minus = evaluate_h(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # apply gradient step
        h_new = h_new - step_size * grad
        
        # apply constraints
        h_new = np.clip(h_new, 0, 1)
        # adjust to satisfy sum constraint
        current_sum = np.sum(h_new) * dx
        if abs(current_sum - 1.0) > 1e-6:
            scale_factor = 1.0 / current_sum
            h_new *= scale_factor
        
        return h_new
    
    # Gradient descent loop
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        h_new = optimize_step(best_h)
        
        c5_new = evaluate_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            same_count = 0
        else:
            same_count += 1
            if same_count > 5:
                # decrease learning rate or terminate
                break
                
        if abs(prev_c5 - best_c5) < tolerance:
            same_count += 1
        else:
            same_count = 0
        prev_c5 = best_c5
        if same_count > 10:
            break
    
    # Post optimization with scipy
    def obj(h_flat):
        h_vec = h_flat.reshape(-1)
        return evaluate_h(h_vec)
    
    def constraint(h_flat):
        h_vec = h_flat.reshape(-1)
        return np.sum(h_vec) * (2.0 / len(h_vec)) - 1.0
    
    bounds = [(0., 1.)] * len(best_h)
    cons = {'type': 'eq', 'fun': constraint}
    
    h_flat = best_h.reshape(-1)
    res = minimize(obj, h_flat, bounds=bounds, constraints=cons, 
                   method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200})
    
    if res.success or not np.isnan(res.fun):
        final_h = res.x.reshape(-1)
        final_c5 = res.fun
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
            
    return (best_h, float(best_c5), len(best_h))
