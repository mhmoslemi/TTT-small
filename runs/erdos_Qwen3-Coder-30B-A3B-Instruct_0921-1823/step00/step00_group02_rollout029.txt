import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    max_iter = 1000
    tol = 1e-6
    
    # Start with the initial solution
    h_current = h0.copy()
    h_current = h_current / np.sum(h_current) / dx  # Normalize to integral = 1
    h_current = np.clip(h_current, 0, 1)  # Clip to [0,1]
    
    # Compute initial C5
    corr = np.correlate(h_current, 1 - h_current, mode='full')
    c5_val = np.max(corr) * dx
    best_c5 = min(best_c5, c5_val)
    best_h = h_current.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use CVX as initial warm start for smaller optimization
        try:
            # Create optimization problem
            h = cp.Variable(n_points)
            
            # Constraint: 0 ≤ h[i] ≤ 1
            constraints = [h >= 0, h <= 1]
            
            # Constraint: integral = 1
            constraints.append(cp.sum(h) * dx == 1)
            
            # Compute objective using correlation
            # C5 = max_k ∫ h(x)(1-h(x+k)) dx
            # This corresponds to max of cross-correlation of h and (1-h)
            corr_vec = np.correlate(h.value, 1 - h.value, mode='full')
            objective = cp.Minimize(cp.max(corr_vec))
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_opt = h.value
                h_opt = np.clip(h_opt, 0, 1)
                corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(corr_opt) * dx
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            # Fallback to gradient-based approach or break
            pass
            
        # Simple gradient descent update (just to avoid local minimums)
        h_trial = h_current.copy()
        h_trial += np.random.normal(0, 0.01, n_points) * (1 - h_trial) * h_trial  # bounded gradient
        h_trial = np.clip(h_trial, 0, 1)

        c5_trial = np.max(corr_trial) * dx
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            
        h_current = h_trial
    
    # Ensure integral is exactly 1
    h_final = best_h / np.sum(best_h) / dx
    
    return (h_final.tolist(), best_c5, n_points)
