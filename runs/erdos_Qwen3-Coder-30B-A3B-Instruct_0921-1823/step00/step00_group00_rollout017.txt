import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a basic initialization
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h_init = h_init * (n_points / 2.0) / np.sum(h_init)
    
    # Constraint: 0 <= h[i] <= 1, sum(h) * dx = 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    def evaluate_h(h):
        # Normalize to satisfy constraint
        h = np.array(h)
        h = h * (n_points / 2.0) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        # Compute overlap integral
        h_full = np.concatenate([h, np.zeros(n_points-1)])
        h_rev = np.concatenate([np.zeros(n_points-1), 1-h])
        corr = np.correlate(h_full, h_rev, mode='full')[:-n_points+1]
        
        # Scale by dx
        max_corr = np.max(corr) * dx
        
        return max_corr
    
    # Optimization
    result = None
    best_h = None
    best_c5 = float('inf')
    
    # Use differential evolution with bounds
    while time.time() - start_time < budget_s - 1:
        try:
            # Perform optimization with given bounds
            res = differential_evolution(
                evaluate_h,
                bounds,
                maxiter=50,
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                callback=lambda x, convergence: None
            )
            
            if res.success:
                h_final = np.array(res.x)
                h_final = h_final * (n_points / 2.0) / np.sum(h_final)
                h_final = np.clip(h_final, 0, 1)
                c5 = evaluate_h(h_final)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_final.copy()
        except Exception as e:
            break
    
    # Final refinement
    if best_h is not None:
        # Use a fine-grained gradient-like approach to possibly improve
        # Just a simple iterative local search
        h_best = best_h.copy()
        learning_rate = 0.01
        max_iter = 1000
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            c5_old = evaluate_h(h_best)
            
            # Simple gradient approximation via finite differences
            eps = 1e-4
            grad = np.zeros_like(h_best)
            for i in range(n_points):
                h_plus = h_best.copy()
                h_plus[i] += eps
                if h_plus[i] <= 1:
                    c5_plus = evaluate_h(h_plus)
                    grad[i] = (c5_plus - c5_old) / eps
                else:
                    h_plus[i] = 1
                    c5_plus = evaluate_h(h_plus)
                    grad[i] = (c5_plus - c5_old) / eps
            
            # Update with gradient
            h_new = h_best - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Project onto integral constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            h_best = h_new
            
            c5_new = evaluate_h(h_best)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_best.copy()
    
    # Return final result
    if best_h is None:
        h_final = h_init
        c5_final = evaluate_h(h_final)
    else:
        h_final = best_h
        c5_final = best_c5
    
    return (h_final.tolist(), c5_final, n_points)
