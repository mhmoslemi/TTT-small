def evaluate_c5(h, dx):
    # Using FFT-based convolution to compute overlap
    n = len(h)
    # Pad to avoid circular convolution effects
    padded_len = 2 * n - 1
    h_padded = np.pad(h, (0, n-1), 'constant')
    one_minus_h = np.pad(1 - h, (0, n-1), 'constant')
    conv = ifft(fft(h_padded) * fft(one_minus_h).conj()).real
    # Extract the relevant portion (full convolution)
    # We want the maximum value among all shifts
    max_overlap = np.max(conv[n-1:])
    return max_overlap * dx

def project_to_constraint(h, dx):
    # Project h to satisfy integral constraint
    h_sum = np.sum(h) * dx
    if abs(h_sum - 1.0) < 1e-10:
        return h
    # Adjust all points equally to fix sum
    adjustment = (1.0 - h_sum) / len(h)
    h_new = h + adjustment
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    return h_new
