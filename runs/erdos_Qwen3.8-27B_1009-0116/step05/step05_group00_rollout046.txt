import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_opt(h0, alpha_start, alpha_end, n_iter, lr_start, lr_end, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        if t % 500 == 0:
            x = x + 0.005 * rng.randn(n)
        g = _grad(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _binary_sa(h0, n, rng, n_iters, deadline):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)

    best_c = c_cur

    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / T_sa):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
        T_sa *= 0.99995
    return best_x, best_c

def _paired_cd(h, n, rng, deadline, scale_init):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()

    scale = scale_init

                best_x = x.copy()
            scale = max(scale * 0.999, 1e-8)
        else:
            x = best_x.copy()
            c_cur = best_c
            scale *= 0.95
    return best_x, best_c

def _periodic_opt(h, T, alpha, maxiter):
    n = len(h)
    if n % T != 0:
        return h.copy()
    rep = n // T
    hp = h[:T].copy()
    def _tobj(x):
        hf = np.tile(x, rep)
        return _obj(hf, alpha)
    def _tgrad(x):
        hf = np.tile(x, rep)
        gf = _grad(hf, alpha)
        return gf[:T] * rep
    def _t_se(x):
        return np.tile(x, rep).sum() - n * 0.5
    def _t_sj(x):
        return np.ones(T) * rep
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _t_se, 'jac': _t_sj}]
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return np.tile(_repair(res.x, T), rep)
    except Exception:
        return h.copy()
