import numpy as np
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def compute_a(h, j):
    """Compute a[lag] = 1 - h[j+lag] - h[j-lag] for all lags"""
    n = len(h)
    a = np.ones(2*n-1)  # for each lag

        val = 1.0
        if 0 <= j + lag < n:
            val -= h[j + lag]
        if 0 <= j - lag < n:
            val -= h[j - lag]
        a[k] = val
    return a
