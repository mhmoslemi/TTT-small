def _grad_lag(h, k_idx, dx):
    """Gradient of c[k_idx] (single lag) with respect to h, scaled by dx."""
    n = len(h)
    g = np.zeros(n)
    k = k_idx - (n - 1)  # actual lag
    if k >= 0:
        # c = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
        # ∂/∂h[j] = (1-h[j-k]) * I[k<=j<=n-1] - h[j+k] * I[0<=j<=n-1-k]
        upper = n - 1 - k
        idx1 = np.arange(k, n)  # k to n-1
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, upper + 1)  # 0 to n-1-k
        g[idx2] -= h[idx2 + k]
    else:
        la = -k
        # c = sum_{n=la}^{n-1} h[n-la]*(1-h[n])
        # ∂/∂h[j] = (1-h[j+la]) * I[0<=j<=n-1-la] - h[j-la] * I[la<=j<=n-1]
        upper = n - 1 - la
        idx1 = np.arange(0, upper + 1)  # 0 to n-1-la
        g[idx1] += (1.0 - h[idx1 + la])
        idx2 = np.arange(la, n)  # la to n-1
        g[idx2] -= h[idx2 - la]
    return g * dx

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_lag(h, k_idx, dx):
    n = len(h)
    g = np.zeros(n)
    k = k_idx - (n - 1)
    if k >= 0:
        upper = n - 1 - k
        if upper >= 0:
            i1 = np.arange(k, n)
            g[i1] += (1.0 - h[i1 - k])
            i2 = np.arange(0, upper + 1)
            g[i2] -= h[i2 + k]
    else:
        la = -k
        upper = n - 1 - la
        if upper >= 0:
            i1 = np.arange(0, upper + 1)
            g[i1] += (1.0 - h[i1 + la])
            i2 = np.arange(la, n)
            g[i2] -= h[i2 - la]
    return g * dx

def _worst_lag_grad(h, dx, n_top):
    """Gradient of sum of top-n_top lags."""
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[-n_top:]
    g = np.zeros(len(h))
    for idx in top:
        g += _grad_lag(h, idx, dx)
    return g, c

def _opt_pass(h, dx, n_top, n_iters, step_base, t0, budget_s, rng):
    """Optimize using active-set gradient descent."""
    n = len(h)
    target = n * 0.5
    c5 = _eval(h)
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 5:
            break
        g, c = _worst_lag_grad(h, dx, n_top)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = step_base * mult
            h_new = _proj_sum(h - step * g, target)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                improved = True
                break
        if not improved:
            # Perturb and continue
            h = _repair(h + 0.003 * rng.randn(n), n)
            c5 = _eval(h)
    return h, c5

def _pairwise(h, n_iters, scale, t0, budget_s, rng):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iters):
        if time.time() - t0 > budget_s - 3:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
