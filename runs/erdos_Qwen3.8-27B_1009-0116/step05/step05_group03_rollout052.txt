import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _move_delta(h, i1, i2, K):
    """Compute b[j] for top K lag indices j.
    Move: h[i1] -= d, h[i2] += d
    Delta C[j] = d * b[j]
    """
    n = len(h)
    g = 1.0 - h
    mid = n - 1
    C = _corr(h)
    top = np.argsort(C)[::-1][:K]
    b = np.zeros(K)
    for t, j in enumerate(top):
        # b[j] = g[i2+j-mid] - g[i1+j-mid] - h[i2-j+mid] + h[i1-j+mid]
        # with 0 for out-of-range indices
        v1 = g[i2 + j - mid] if 0 <= i2 + j - mid < n else 0.0
        v2 = g[i1 + j - mid] if 0 <= i1 + j - mid < n else 0.0
        v3 = h[i2 - j + mid] if 0 <= i2 - j + mid < n else 0.0
        v4 = h[i1 - j + mid] if 0 <= i1 - j + mid < n else 0.0
        b[t] = v1 - v2 - v3 + v4
    return top, b, C[top]

def _minimax_d(C_top, b, d_max):
    """Find d in [0, d_max] minimizing max_j (C_top[j] + d * b[j])"""
    n_lines = len(C_top)
    if n_lines == 1:
        if b[0] < 0:
            return d_max
        return 0.0
    # Collect candidate d values: 0, d_max, and intersections
    candidates = [0.0, d_max]
    for i in range(n_lines):
        for j in range(i+1, n_lines):
            db = b[i] - b[j]
            if abs(db) > 1e-15:
                d_cross = (C_top[j] - C_top[i]) / db
                if 0 <= d_cross <= d_max:
                    candidates.append(d_cross)
    best_d, best_val = 0.0, np.inf
    for d in candidates:
        val = np.max(C_top + d * b)
        if val < best_val:
            best_val = val
            best_d = d
    return best_d

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def con_eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': con_eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _minimax_d(C_top, b, d_max):
    nL = len(C_top)
    if nL == 1:
        return d_max if b[0] < 0 else 0.0
    cands = [0.0, d_max]
    for i in range(nL):
        for j in range(i + 1, nL):
            db = b[i] - b[j]
            if abs(db) > 1e-15:
                dc = (C_top[j] - C_top[i]) / db
                if -1e-12 <= dc <= d_max + 1e-12:
                    cands.append(max(0.0, min(d_max, dc)))
    best_d, best_v = 0.0, 1e18
    for d in cands:
        v = float(np.max(C_top + d * b))
        if v < best_v:
            best_v = v
            best_d = d
    return best_d

def _exact_local(h, dx, rng, t_end, K=10, n_trials=200, cooling=0.9997):

    tgt = n / 2.0

                v1 = g[a1] if 0 <= a1 < n else 0.0

                continue
            nv = float(np.max(C_top + d_opt * b))

def _fine_refine(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c = _c5(h, dx)
                if c < best:
                    best = c
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), n / 2.0)
