def _lp_step(h, n, top_k=30):
    c = _corrs(h)
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix for top lags
    # For each lag k_idx, the gradient g_k = df_k/dh at current h
    A_rows = []
    b_vals = []
    for k_idx in top_idx:
        g = _grad_lag(k_idx, h)  # already normalized by 2/n
        # f_k(h_new) ≈ c[k_idx] + g @ (h_new - h)
        # = (c[k_idx] - g @ h) + g @ h_new
        b = c[k_idx] - g @ h
        A_rows.append(g)
        b_vals.append(b)
    
    A = np.array(A_rows)  # (K, n)
    b = np.array(b_vals)  # (K,)
    
    # LP: min t s.t. A @ h - t <= -b, sum(h) = n/2, 0 <= h <= 1
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [A @ x - t <= -b, cp.sum(x) == n/2, x >= 0, x <= 1]
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, warm_start=True)
    
    if x.value is not None:
        return _repair(np.array(x.value), n), t.value
    return h, c.max()

        # d/dt f_k(h + t*d) = g_k @ d

from scipy.interpolate import interp1d

_b = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(k_idx, h):
    n = len(h)
    s = k_idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:n-s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * (2.0 / n)

def _smooth_obj(h, alpha):
    c = _corrs(h)
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted - shifted.max()))) / alpha

def _smooth_grad(h, alpha):
    c = _corrs(h)
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    g = np.zeros(len(h))
    for k in range(2 * len(h) - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(k, h)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    target = n * 0.5
    h = h.copy()

    step = step0

        g /= topm
        found = False
        for _ in range(35):

            step *= 0.5

def _equalize(h, top_k, max_iter, lam):
    """Equalize top-k lags via least-squares direction."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _ in range(max_iter):
        if time.time() - _t0 > _b - 20:
            break
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
        mean_v = vals.mean()
        residual = vals - mean_v
        if np.max(np.abs(residual)) < 1e-6:
            break
        G = np.array([_grad_lag(k, h) for k in top_idx])
        # Project G to sum-zero subspace
        G = G - G.mean(axis=0, keepdims=True)
        M = G.T @ G + lam * (np.eye(n) - np.ones((n,n))/n)
        d = np.linalg.solve(M, G.T @ residual)
        d -= d.mean()
        step = 1.0
        for _ in range(25):

            f_new = _eval(h_new)
            if f_new < f - 1e-10:

            step *= 0.5
        else:

    return h, f

def _pairwise(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _b - 8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):

    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)

    return _repair(np.clip(f(x_new), 0, 1), n * factor)
