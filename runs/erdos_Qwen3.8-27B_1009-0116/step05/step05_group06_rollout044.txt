import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _penal_obj(h, alpha, lam):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    f = m + np.log(np.sum(np.exp(s))) / alpha
    n = len(h)
    f += lam * (h.sum() - 0.5 * n) ** 2
    return f


def _penal_grad(h, alpha, lam):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * c - alpha * m, -500, 500)
    wn = np.exp(s - s.max())
    wn /= wn.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if wn[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = wn[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:n])
            g[sh:n] -= wk * h[:cnt]
        else:
            p = -sh
            g[p:n] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:n]
    g += 2.0 * lam * (h.sum() - 0.5 * n)
    return g


def _lbfgs(h, alpha, lam, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penal_obj, h, jac=_penal_grad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)


def _upsample(h, f):
    n = len(h)
    return np.repeat(h, f)


def _pair_local(h, nsteps, scale, rng, nc):
    c5c = _eval(h)
    for _ in range(nsteps):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c


def _coord_min(h, j, alpha, n_passes):
    n = len(h)
    v = h[j]
    c = _corr(h)
    coefs = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        a = (1.0 - h[j + sh]) if 0 <= j + sh < n else 0.0
        b = h[j - sh] if 0 <= j - sh < n else 0.0
        coefs[k] = a - b
    for _ in range(n_passes):
        m_best = 1e18
        t_best = v
        for _it in range(8):
            lo, hi = 0.0, 1.0
            for _gs in range(20):
                m1 = lo + (hi - lo) * 0.382
                m2 = hi - (hi - lo) * 0.382
                c1 = c + (m1 - v) * coefs
                m1v = c1.max()
                s1 = np.clip(alpha * (c1 - m1v), -500, 500)
                o1 = m1v + np.log(np.sum(np.exp(s1))) / alpha
                c2 = c + (m2 - v) * coefs
                m2v = c2.max()
                s2 = np.clip(alpha * (c2 - m2v), -500, 500)
                o2 = m2v + np.log(np.sum(np.exp(s2))) / alpha
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_new = (lo + hi) * 0.5
            c_new = c + (t_new - v) * coefs
            m_new = c_new.max()
            s_new = np.clip(alpha * (c_new - m_new), -500, 500)
            o_new = m_new + np.log(np.sum(np.exp(s_new))) / alpha
            if o_new < m_best:
                m_best = o_new
                t_best = t_new
            break
        h[j] = t_best
        v = t_best
        c = c + (t_best - v) * coefs
    return h


def _multires_opt(h, n_target, alphas, iters_per_alpha, t_deadline):
    n = len(h)
    while n < n_target:
        n = min(n * 2, n_target)
        h = _upsample(h, max(1, n // len(h)))
        if len(h) != n:
            h = h[:n]
        h = _repair(h, n)
        for a in alphas:
            if time.time() > t_deadline:
                break
            lam = 1.0
            for _ in range(3):
                h = _lbfgs(h, a, lam, iters_per_alpha)
                viol = abs(h.sum() - 0.5 * n)
                if viol < 1e-8:
                    break
                lam *= 10.0
            h = _repair(h, n)
    return h
