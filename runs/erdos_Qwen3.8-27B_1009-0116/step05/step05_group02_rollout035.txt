    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(80):

            s2 = np.clip(h - mid, 0, 1).sum()
            if s2 > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)

        diff = target - h.sum()
        if abs(diff) > 1e-12:

        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g * dx

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_phase(h, n, t0, deadline, rng):
    best_h = h.copy()
    best_c5 = _eval(h)
    gap = 0.005
    for it in range(300000):
        if time.time() > deadline:
            break
        c = _eval_c(h)
        c5 = c.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        dx = 2.0 / n
        active = np.where(c >= c5 - 0.3 * dx)[0]
        g = np.zeros(n)
        cnt = 0
        for k in active[:12]:
            g += _grad_lag(h, k, n)
            cnt += 1
        g /= cnt
        gnrm = np.linalg.norm(g)
        if gnrm < 1e-14:
            break
        lr = min(gap / (gnrm * gnrm + 1e-12), 0.3)
        h_new = _proj_sum(h - lr * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            gap = max(c5_new - best_c5 + 0.001, 1e-5) * 0.6
        else:
            lr2 = lr * 0.05
            h2 = _proj_sum(h - lr2 * g, n)
            if _eval(h2) < c5:
                h = h2
            elif it % 200 == 0:
                h = _repair(h + 0.001 * rng.randn(n), n)
                gap = max(gap * 0.95, 1e-6)
    return best_h, best_c5

def _pairwise_phase(h, n, t0, deadline, rng, scale, max_iters):
    best_h = h.copy()
    c5_best = _eval(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    s = np.arange(-(n - 1), n)
    tn1 = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        delta = np.zeros(tn1)
        v1 = (i + s >= 0) & (i + s < n)
        if v1.any():
            delta[v1] += d * (1.0 - h[i + s[v1]])
        v2 = (i - s >= 0) & (i - s < n)
        if v2.any():
            delta[v2] -= d * h[i - s[v2]]
        v3 = (j + s >= 0) & (j + s < n)
        if v3.any():
            delta[v3] -= d * (1.0 - h[j + s[v3]])
        v4 = (j - s >= 0) & (j - s < n)
        if v4.any():
            delta[v4] += d * h[j - s[v4]]
        cm_old = c.max()
        cm_new = (c + delta).max()
        if cm_new < cm_old - 1e-14:
            h[i], h[j] = hi, hj
            c += delta
            c5_new = cm_new * dx
            if c5_new < c5_best:
                c5_best = c5_new
                best_h = h.copy()
        if it % 5000 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
    return best_h, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at base resolution
    for alpha in [1000, 5000, 20000, 80000]:
        if rem() < 180:
            break
        h_new = _slsqp(best_h, alpha, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Subgradient refinement
    if rem() > 130:
        h_sg, c5_sg = _subgrad_phase(best_h, best_n, t0, t0 + budget_s - 110, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Upsample to 2x and optimize
    if rem() > 280:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [5000, 30000, 150000]:
            if rem() < 200:
                break
            h_hi = _slsqp(h_hi, alpha, 2500)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase 4: Subgradient at current resolution
    if rem() > 130:
        h_sg, c5_sg = _subgrad_phase(best_h, best_n, t0, t0 + budget_s - 100, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 5: Coarse pairwise (fast incremental)
    if rem() > 80:
        h_pw, c5_pw = _pairwise_phase(best_h, best_n, t0, t0 + budget_s - 55, rng, 0.003, 999999)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    # Phase 6: Medium pairwise
    if rem() > 50:
        h_pw, c5_pw = _pairwise_phase(best_h, best_n, t0, t0 + budget_s - 30, rng, 0.0005, 999999)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    # Phase 7: Fine pairwise
    if rem() > 25:
        h_pw, c5_pw = _pairwise_phase(best_h, best_n, t0, t0 + budget_s - 12, rng, 5e-5, 999999)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    # Phase 8: Final SLSQP polish
    if rem() > 12:
        h_new = _slsqp(best_h, 500000, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 9: Ultra-fine pairwise
    if rem() > 5:
        h_pw, c5_pw = _pairwise_phase(best_h, best_n, t0, t0 + budget_s - 1, rng, 1e-5, 999999)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
