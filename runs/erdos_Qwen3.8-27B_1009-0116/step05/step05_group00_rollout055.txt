    return np.ones(len(x))

def _slsqp(h_full, alpha, maxiter, ftol):
    n = len(h_full)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h_full, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h_full, n)

def _slsqp_periodic(p_init, T, n, alpha, maxiter, ftol):
    m = n // T
    p = _repair(p_init.copy(), T)

    def p_obj(x):
        return _soft_obj(np.tile(x, m), alpha)

    def p_grad(x):
        gf = _soft_grad(np.tile(x, m), alpha)
        return gf.reshape(m, T).sum(axis=0)

    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(p_obj, p, jac=p_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.tile(_repair(res.x, T), m)
    except Exception:
        return np.tile(p, m)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _block_pattern(T):
    h = np.zeros(T)
    h[:T // 2] = 1.0
    return h

def _alternating(T):
    h = np.zeros(T)
    h[::2] = 1.0
    half = T // 2
    if T % 2 == 1 and h.sum() > half:
        h[-1] = 0.0
    return _repair(h, T)

def _get_divisors(n, max_d):
    divs = []
    for d in range(2, min(max_d, n // 2) + 1):
        if n % d == 0:
            divs.append(d)
    return divs

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP polish at original resolution
    for alpha in [10000, 100000, 500000]:
        if time.time() - t0 > 70:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Systematic n-sweep with periodic optimization
    n_list = [240, 280, 300, 320, 336, 360, 420, 480, 504, 540, 600, 720]
    for n_try in n_list:
        if time.time() - t0 > budget_s - 250:
            break
        if n_try == best_n:
            continue
        h_res = _repair(_resample(best_h, n_try), n_try)
        h_rand = _random_binary(n_try, rng)
        divs = _get_divisors(n_try, 50)
        divs.sort()
        for T in divs[:10]:
            if time.time() - t0 > budget_s - 250:
                break
            starts = [h_res[:T], h_rand[:T], _block_pattern(T), _alternating(T)]
            for h_src in starts:
                if time.time() - t0 > budget_s - 250:
                    break
                p_init = _repair(h_src.copy(), T)
                for alpha in [30000, 150000]:
                    if time.time() - t0 > budget_s - 250:
                        break
                    h_p = _slsqp_periodic(p_init, T, n_try, alpha, 600, 1e-16)
                    c5 = _eval(h_p)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_p.copy()
                        best_n = n_try

    # Phase 3: Full SLSQP on best found
    if time.time() - t0 < budget_s - 120:
        for alpha in [50000, 300000, 1000000]:
            if time.time() - t0 > budget_s - 120:
                break
            h_new = _slsqp(best_h, alpha, 2000, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase 4: Coordinate descent on high-overlap region
    if time.time() - t0 < budget_s - 40:
        nc = len(best_h)
        h_t = best_h.copy()
        c5_t = best_c5
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0004]:
            if time.time() - t0 > budget_s - 40:
                break
            for it in range(4000):
                if time.time() - t0 > budget_s - 40:
                    break
                i_idx = rng.randint(nc)
                j_idx = rng.randint(nc)
                if i_idx == j_idx:
                    continue
                d = scale * rng.randn()
                ni = h_t[i_idx] + d
                nj = h_t[j_idx] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h_t[i_idx] = ni
                h_t[j_idx] = nj
                c5_new = _eval(h_t)
                if c5_new < c5_t - 1e-15:
                    c5_t = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_t.copy()
                else:
                    h_t[i_idx] -= d
                    h_t[j_idx] += d

    # Phase 5: Simulated annealing with targeted restarts
    if time.time() - t0 < budget_s - 15:
        nc = len(best_h)
        h_a = best_h.copy()
        c5_a = best_c5
        n_anneal = 4000
        T_start = 0.03
        for i in range(n_anneal):
            if time.time() - t0 > budget_s - 15:
                break
            T = T_start * (0.001) ** (float(i) / float(n_anneal))
            i_idx = rng.randint(nc)
            j_idx = rng.randint(nc)
            if i_idx == j_idx:
                continue
            d = rng.uniform(-0.06, 0.06)
            ni = h_a[i_idx] + d
            nj = h_a[j_idx] - d
            if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
                continue
            h_a[i_idx] = np.clip(ni, 0, 1)
            h_a[j_idx] = np.clip(nj, 0, 1)
            h_a = _repair(h_a, nc)
            c5_new = _eval(h_a)
            delta = c5_new - c5_a
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                c5_a = c5_new
                if c5_a < best_c5:
                    best_c5 = c5_a
                    best_h = h_a.copy()
            else:
                h_a[i_idx] = np.clip(ni - d, 0, 1)
                h_a[j_idx] = np.clip(nj + d, 0, 1)
                h_a = _repair(h_a, nc)

    # Phase 6: Final high-alpha SLSQP polish
    if time.time() - t0 < budget_s - 3:
        h_fin = _slsqp(best_h, 2000000, 2000, 1e-20)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
