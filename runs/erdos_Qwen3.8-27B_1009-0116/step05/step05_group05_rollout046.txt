    g_prev = None
    for it in range(len(alphas)):
        alpha = alphas[it]
        obj, g = _obj_grad(h, alpha)
        if g_prev is not None:
            dh = h - h_prev
            dg = g - g_prev
            s1 = float(dg @ dg)
            s2 = float(dh @ dg)
            if s2 > 1e-15:
                step = s1 / s2
                step = float(np.clip(step, 1e-6, 0.05))
        h_prev = h.copy()
        g_prev = g
        improved = False
        for fac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj(h - step * fac * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
                improved = True
                break
        if not improved:
            step *= 0.7
    return h, c5

def _active_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for wk, k in zip(weights, idx):
            L = k - (n - 1)
            i_lo = max(0, -L)
            i_hi = min(n - 1, n - 1 - L)
            if i_lo > i_hi:
                continue
            idxv = np.arange(i_lo, i_hi + 1)
            contrib = 1.0 - h[idxv + L]
            sec = idxv - L
            vs = (sec >= 0) & (sec < n)
            if np.any(vs):
                contrib[vs] -= h[sec[vs]]
            g[idxv] += wk * dx * contrib
        improved = False
        for step in [0.03, 0.01, 0.004, 0.001, 0.0003, 0.0001]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_grad, h, jac=True, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fourier_init(nn, freq, amp, ph, rng):
    t = np.arange(nn) / nn
    h = 0.5 + amp * np.sin(2 * np.pi * freq * t + ph)
    return _repair(h, nn)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: PGD with alpha annealing at base n
    alphas_A = np.concatenate([np.full(40, 500), np.full(80, 5000),
                               np.full(150, 50000), np.full(200, 200000)])
    if _el() < 180:
        h1, c5 = _pgd(best_h.copy(), alphas_A, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase B: SLSQP polish
    for alpha in [100000, 500000, 2000000]:
        if _el() > 320:
            break
        h_new = _slsqp_opt(best_h, alpha, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase C: Active set descent
    if _el() < 400:
        h_new, c5 = _active_descent(best_h.copy(), 3000, 3)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase D: Multi-start exploration
    n_cur = best_n
    if _el() < budget_s - 300:
        for trial in range(40):
            if _el() > budget_s - 280:
                break
            strat = trial % 6
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _fourier_init(n_cur, rng.randint(1, 5), 0.25 + 0.2 * rng.rand(), rng.rand() * 6.28, rng)
            elif strat == 2:
                hp = _fourier_init(n_cur, rng.randint(3, 10), 0.15 + 0.2 * rng.rand(), rng.rand() * 6.28, rng)
            elif strat == 3:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n_cur), n_cur)
            elif strat == 4:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 4)):
                    bs = rng.randint(5, max(6, n_cur // 8))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            else:
                hp = _repair(np.clip(best_h + 0.08 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur)
            alphas_short = np.full(250, 100000)
            h_new, c5 = _pgd(hp, alphas_short, rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()
                best_n = n_cur

    # Phase E: Upsample to 2x
    if _el() < budget_s - 350:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        alphas_E = np.concatenate([np.full(80, 5000), np.full(150, 50000), np.full(200, 200000)])
        h_new, c5 = _pgd(h_hi, alphas_E, rng)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n1
        if _el() < budget_s - 100:
            h_new = _slsqp_opt(best_h, 500000, 2000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()
            h_new, c5 = _active_descent(best_h.copy(), 2000, 3)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase F: Second upsample to 4x from original
    if _el() < budget_s - 200:
        n2 = n0 * 4
        h_hi2 = np.repeat(best_h[:n0] if len(best_h) == n0 else _repair(best_h, n0), 4)
        h_hi2 = _repair(h_hi2, n2)
        alphas_F = np.concatenate([np.full(100, 10000), np.full(200, 100000)])
        h_new, c5 = _pgd(h_hi2, alphas_F, rng)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n2

    # Phase G: Final active set + SLSQP
    if _el() < budget_s - 60:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
    if _el() < budget_s - 20:
        h_new = _slsqp_opt(best_h, 5000000, 5000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase H: Ultra-fine active set
    if _el() < budget_s - 5:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
