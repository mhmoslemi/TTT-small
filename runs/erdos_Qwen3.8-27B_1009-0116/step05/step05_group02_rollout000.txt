        inits.append(_repair(0.5 + 0.2*rng.randn(n0), n0))

            if elapsed() > budget_s - 200: break
            h_new = _fista(h_init, 20000, 1000, topk=15)
            h_new = _slsqp_lse(h_new, 50000, 800, topk=10)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase D: FISTA at higher resolution (560)
    n1 = n0 * 2
    if elapsed() < budget_s - 200:
        h_hi = _repair(np.repeat(best_h, 2), n1)
        for alpha in [5000, 50000, 200000]:
            if elapsed() > budget_s - 150: break
            h_hi = _fista(h_hi, alpha, 1200, topk=20)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_hi.copy(); n0 = n1

    # Phase E: Hotspot descent (target max correlation specifically)
    if elapsed() < budget_s - 100:
        h_tmp = best_h.copy()
        h_tmp, c5tmp = _hotspot_descent(h_tmp, 300, [0.1, 0.03, 0.01, 0.003, 0.001])
        if c5tmp < best_c5:
            best_c5 = c5tmp; best_h = h_tmp.copy()

    # Phase F: Pairwise refinement
    if elapsed() < budget_s - 30:
        for sc in [0.01, 0.003, 0.001, 0.0003]:
            if elapsed() > budget_s - 20: break
            passes = 8000 if sc > 0.01 else 15000
            best_h, best_c5 = _pairwise_refine(best_h.copy(), passes, sc)

    # Phase G: Final SLSQP polish
    if elapsed() < budget_s - 5:
        h_new = _slsqp_lse(best_h, 200000, 2000, topk=5)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
