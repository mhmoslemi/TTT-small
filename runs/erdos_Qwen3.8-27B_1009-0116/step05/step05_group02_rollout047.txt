    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = s.max()
    return m + np.log(np.sum(np.exp(s - smax))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-14:
            continue
        sh = (n - 1) - k_idx
        wk = w[k_idx] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topm_grad(h, m):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k_idx in idx:
        sh = (n - 1) - k_idx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= h[:cnt]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g / m

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i*factor:(i+1)*factor] = h[i]
    return h_hi

def _run_slq_phases(h, alphas, maxiter, ftol, best_c5, best_h):
    n = len(h)
    for alpha in alphas:
        h_new = _slsqp(h, alpha, maxiter, ftol)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return h, best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _ok(rem=20):
        return time.time() - t0 < budget_s - rem

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP continuation at base resolution
    alphas_p1 = [300, 1000, 5000, 20000, 80000, 200000, 500000]
    h, best_c5, best_h = _run_slq_phases(h, alphas_p1, 4000, 1e-18, best_c5, best_h)

    # Phase 2: Top-m gradient descent with backtracking
    nc = len(best_h)
    dx = 2.0 / nc
    if _ok(80):
        h_cur = best_h.copy()
        c5_cur = _eval(h_cur)
        for m in [1, 2, 3, 5]:
            if not _ok(80):
                break
            for _it in range(300):
                if not _ok(80):
                    break
                g = _topm_grad(h_cur, m)
                gn = np.linalg.norm(g)
                if gn < 1e-14:
                    break
                improved = False
                for s in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4]:
                    h_try = _repair(h_cur - s * g / gn * gn, nc)
                    c5_try = _eval(h_try)
                    if c5_try < c5_cur - 1e-15:
                        h_cur = h_try
                        c5_cur = c5_try
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h_cur.copy()
                        break
                if not improved:
                    h_cur = _repair(h_cur + 0.002 * rng.randn(nc), nc)
                    c5_cur = _eval(h_cur)
                    if c5_cur > best_c5 + 1e-6:
                        h_cur = best_h.copy()
                        c5_cur = best_c5

    # Phase 3: SLSQP at 2x resolution
    if _ok(250):
        n1 = n0 * 2
        h2 = _repair(_upsample(best_h, 2), n1)
        alphas_p3 = [1000, 5000, 30000, 100000, 300000]
        h2, bc5, bh = _run_slq_phases(h2, alphas_p3, 3000, 1e-18, best_c5, best_h)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh.copy()
            best_n = n1

    # Phase 4: Multi-restart SLSQP with diverse perturbations
    nc = best_n
    if _ok(200):
        for _r in range(30):
            if not _ok(200):
                break
            strat = _r % 6
            if strat == 0:
                hp = _repair(best_h + 0.008 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                idx = rng.choice(nc, size=nc // 4, replace=False)
                hp[idx] += 0.015 * rng.randn(len(idx))
                hp = _repair(hp, nc)
            elif strat == 2:
                f = rng.randint(2, nc // 4 + 1)
                t_arr = np.arange(nc)
                hp = _repair(best_h + 0.004 * np.sin(2 * np.pi * f * t_arr / nc + rng.rand() * 6.28), nc)
            elif strat == 3:
                hp = _repair(1.0 - best_h + 0.003 * rng.randn(nc), nc)
            elif strat == 4:
                hp = _repair(best_h * rng.uniform(0.92, 1.08) + 0.005 * rng.randn(nc), nc)
            else:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 4)):
                    bs = rng.randint(10, max(11, nc // 6))
                    st = rng.randint(0, nc - bs)
                    hp[st:st+bs] += 0.01 * rng.randn(bs)
                hp = _repair(hp, nc)
            h_new = _slsqp(hp, 200000, 2500, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Extended active-lag equalization
    nc = best_n
    dx = 2.0 / nc
    if _ok(150):
        h_cur = best_h.copy()
        c5_cur = _eval(h_cur)
        for m in [3, 5, 7]:
            if not _ok(150):
                break
            for _it in range(400):
                if not _ok(150):
                    break
                c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                top_idx = np.argsort(c)[-m:]
                top_vals = c[top_idx]
                top_mean = top_vals.mean()
                # Gradient: push down top lags, plus a small variance term
                g = np.zeros(nc)
                for k_idx in top_idx:
                    sh = (nc - 1) - k_idx
                    if sh >= 0:
                        cnt = nc - sh
                        g[:cnt] += (1.0 - h_cur[sh:sh+cnt])
                        if sh > 0:
                            g[sh:] -= h_cur[:cnt]
                    else:
                        sp = -sh
                        g[sp:] += (1.0 - h_cur[:nc-sp])
                        g[:nc-sp] -= h_cur[sp:]
                g /= m
                gn = np.linalg.norm(g)
                if gn < 1e-15:
                    break
                improved = False
                for s in [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4, 2e-4]:
                    h_try = _repair(h_cur - s * g, nc)
                    c5_try = _eval(h_try)
                    if c5_try < c5_cur - 1e-15:
                        h_cur = h_try
                        c5_cur = c5_try
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h_cur.copy()
                        break
                if not improved:
                    h_cur = _repair(h_cur + 0.0008 * rng.randn(nc), nc)
                    c5_cur = _eval(h_cur)
                    if c5_cur > best_c5 + 5e-6:
                        h_cur = best_h.copy()
                        c5_cur = best_c5

    # Phase 6: Gradient-guided coordinate search
    nc = best_n
    if _ok(60):
        h_cur = best_h.copy()
        c5_cur = _eval(h_cur)
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_lag = c.argmax()
        sh = (nc - 1) - top_lag
        g_lag = np.zeros(nc)
        if sh >= 0:
            cnt = nc - sh
            g_lag[:cnt] += (1.0 - h_cur[sh:sh+cnt])
            if sh > 0:
                g_lag[sh:] -= h_cur[:cnt]
        else:
            sp = -sh
            g_lag[sp:] += (1.0 - h_cur[:nc-sp])
            g_lag[:nc-sp] -= h_cur[sp:]
        for scale in [0.003, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
            if not _ok(60):
                break
            for _it in range(10000):
                if not _ok(60):
                    break
                # Gradient-guided pair selection
                i = int(rng.randint(nc))
                # Bias j selection toward correlated partner
                if sh >= 0 and 0 <= i + sh < nc:
                    j = i + sh
                elif sh < 0 and 0 <= i - sh < nc:
                    j = i - sh
                else:
                    j = int(rng.randint(nc))
                if i == j:
                    j = (i + 1) % nc
                d = scale * rng.randn()
                ni, nj = h_cur[i] + d, h_cur[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h_cur[i], h_cur[j] = ni, nj
                c5n = _eval(h_cur)
                if c5n < c5_cur - 1e-16:
                    c5_cur = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = ni - d, nj + d

    # Phase 7: Final SLSQP polish with very high alpha
    if _ok(15):
        h_new = _slsqp(best_h, 800000, 6000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 8: Ultra-fine touch
    nc = best_n
    if _ok(3):
        h_cur = best_h.copy()
        c5_cur = _eval(h_cur)
        for _it in range(30000):
            if not _ok(3):
                break
            i = int(rng.randint(nc))
            j = int(rng.randint(nc))
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_cur[i], h_cur[j] = ni, nj
                c5n = _eval(h_cur)
                if c5n < c5_cur - 1e-17:
                    c5_cur = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
