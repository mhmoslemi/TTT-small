import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lse(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _obj_lse(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _cons_fun(x, target):
    return x.sum() - target

def _cons_jac(x, target):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(_cons_fun, target=target),
             'jac': partial(_cons_jac, target=target)}]
    try:
        res = minimize(_obj_lse, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _de_search(h_init, n, pop_size, n_gen, alpha, deadline, rng):
    F, CR = 0.6, 0.85
    pop = np.zeros((pop_size, n))
    pop[0] = h_init.copy()
    for i in range(1, pop_size):
        pop[i] = _repair(h_init + 0.08 * rng.randn(n), n)
    fitness = np.array([_eval(pop[i]) for i in range(pop_size)])
    best_c = fitness.min()
    best_h = pop[fitness.argmin()].copy()
    for gen in range(n_gen):
        now = time.time()
        if now > deadline:
            break
        bi = fitness.argmin()
        for i in range(pop_size):
            if now > deadline:
                break
            if i == bi:
                continue
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, 3, replace=False)
            trial = pop[r1].astype(float) + F * (pop[r2].astype(float) - pop[r3].astype(float))
            j_rand = rng.randint(n)
            cm = rng.rand(n) < CR
            cm[j_rand] = True
            trial[~cm] = pop[i][~cm]
            trial = _repair(trial, n)
            fit = _eval(trial)
            if fit <= fitness[i]:
                pop[i] = trial
                fitness[i] = fit
                if fit < best_c:
                    best_c = fit
                    best_h = trial.copy()
        gi = fitness.argmin()
        if fitness[gi] < best_c - 1e-15:
            best_c = fitness[gi]
            best_h = pop[gi].copy()
    return best_h, best_c

def _nesterov_opt(h, alpha, steps, step0, deadline):
    n = len(h)
    target = n * 0.5
    f = _obj_lse(h, alpha)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    h_prev = h.copy()
    for _s in range(steps):
        if time.time() > deadline:
            break
        h_eval = _proj_sum(h + mu * (h - h_prev), target)
        g = _grad_lse(h_eval, alpha)
        h_prev = h.copy()
        h_new = _proj_sum(h - step * g, target)
        f_new = _obj_lse(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 4)
        else:
            step *= 0.5
            if step < 1e-8:
                step = step0 * 0.5
            h = _proj_sum(h_prev - step * g, target)
            f = _obj_lse(h, alpha)
    return h

def _lag_target_descent(h, deadline, max_rounds=300):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_w = c.argmax()
        sh = (n - 1) - k_w
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
        g *= dx
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _pairwise_refine(h, deadline, rng):
    n = len(h)
    c5c = _eval(h)
    while time.time() < deadline:
        batch = max(1, int((deadline - time.time()) / 0.002))
        batch = min(batch, 30000)
        for _it in range(batch):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = 5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c
