    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_gradients(h, alpha, w):
    n = len(h)
    g = np.zeros(n)
    M = len(w)
    for k in range(M):
        if w[k] < 1e-14:
            continue
        tau = k - (n - 1)
        wk = w[k]
        if tau >= 0:
            if tau > 0:
                g[:tau] += -wk * h[tau:]
            if n - tau > tau:
                g[tau:n - tau] += wk * (1.0 - h[:n - 2 * tau])
            if n - 2 * tau > 0:
                g[tau:n - tau] += -wk * h[2 * tau:n]
            if n - tau < n:
                g[n - tau:] += wk * (1.0 - h[2 * tau - tau:])
        else:
            st = -tau
            if st > 0:
                g[n - st:] += -wk * h[:n - st]
            if n - st > st:
                g[st:n - st] += wk * (1.0 - h[n - st:])
            if n - 2 * st > 0:
                g[st:n - st] += -wk * h[:n - 2 * st]
            if st < n:
                g[:st] += wk * (1.0 - h[st:st + st])
    return g

def _nesterov_run(h, alpha_start, alpha_end, n_levels, iters_per_level, eta0, t0, time_limit):
    n = len(h)
    c5 = _eval(h)
    for lvl in range(n_levels):
        alpha = alpha_start * (alpha_end / alpha_start) ** (lvl / max(1, n_levels - 1))
        for it in range(iters_per_level):
            if time.time() - t0 > time_limit:
                break
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            cm = c.max()
            s = np.clip(alpha * (c - cm), -500, 500)
            w = np.exp(s - s.max())
            wsum = w.sum()
            w /= wsum
            g = _smooth_gradients(h, alpha, w)
            g *= 2.0 / n
            h_new = _proj_to_sum(h - eta0 * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
            else:
                eta0 *= 0.6
    return h, c5

def _active_descent(h, max_iters, n_top, rng, t0, time_limit):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    step0 = 0.05
    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[::-1][:n_top]
        cvals = c[top]
        temp = 0.01
        sw = np.exp((cvals - cvals.max()) / temp)
        weights = sw / sw.sum()
        g = np.zeros(n)
        for w, k in zip(weights, top):
            tau = k - (n - 1)
            if tau >= 0:
                if tau > 0:
                    g[:tau] += -w * h[tau:]
                if n - tau > tau:
                    g[tau:n - tau] += w * (1.0 - h[:n - 2 * tau])
                if n - 2 * tau > 0:
                    g[tau:n - tau] += -w * h[2 * tau:n]
                g[n - tau:] += w * (1.0 - h[2 * tau:])
            else:
                st = -tau
                if st > 0:
                    g[n - st:] += -w * h[:n - st]
                if n - st > st:
                    g[st:n - st] += w * (1.0 - h[n - st:])
                if n - 2 * st > 0:
                    g[st:n - st] += -w * h[:n - 2 * st]
                g[:st] += w * (1.0 - h[st:2 * st])
        g *= dx
        improved = False
        for fr in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = step0 * fr
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                improved = True
                step0 = min(step * 1.5, 0.1)
                break
        if not improved:
            step0 *= 0.7
            if step0 < 1e-6:
                break
    return h, c5c

def _fine_pairwise(h, n_iter, scale, rng, t0, time_limit):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    DEADLINE = t0 + budget_s

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: Nesterov smoothed optimization at n0
    h_a, c5a = _nesterov_run(best_h.copy(), 200, 2e6, 12, 40, 0.5, t0, t0 + 200)
    if c5a < best_c5:
        best_c5, best_h = c5a, h_a.copy()

    # Phase B: Active set descent
    h_b, c5b = _active_descent(best_h.copy(), 4000, 7, rng, t0, t0 + 400)
    if c5b < best_c5:
        best_c5, best_h = c5b, h_b.copy()
    h_b, c5b = _active_descent(best_h.copy(), 3000, 4, rng, t0, t0 + 600)
    if c5b < best_c5:
        best_c5, best_h = c5b, h_b.copy()

    # Phase C: Multi-start exploration
    n_cur = best_n
    if _el() < budget_s - 350:
        for r in range(15):
            if _el() > budget_s - 300:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(np.clip(1.0 - best_h, 0, 1) + 0.01 * rng.randn(n_cur), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(3, max(4, n_cur // 8))
                    st = rng.randint(0, max(1, n_cur - bs))
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(np.clip(hp, 0, 1), n_cur)
            elif strat == 3:
                hp = best_h.copy()
                t = np.arange(n_cur)
                for _f in range(4):
                    f = rng.randint(1, n_cur // 2 + 1)
                    hp += 0.005 * np.sin(2 * np.pi * f * t / n_cur + rng.rand() * 6.28)
                hp = _repair(hp, n_cur)
            elif strat == 4:
                hp = _repair(best_h + 0.06 * (rng.rand(n_cur) - 0.5), n_cur)
            else:
                hp = _repair(np.clip(best_h[::1].copy()[:n_cur], 0, 1), n_cur)
                m = n_cur // 4
                hp[m:] = 1.0 - best_h[:n_cur - m]
                hp = _repair(hp, n_cur)
            h_try, c5_try = _nesterov_run(hp, 500, 5e6, 8, 30, 0.4, t0, t0 + budget_s - 310)
            h_try, c5_try = _active_descent(h_try, 1500, 5, rng, t0, t0 + budget_s - 290)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                best_n = n_cur

    # Phase D: Upsample to 2x
    if _el() < budget_s - 300:
        n1 = best_n * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        h_hi, c5_hi = _nesterov_run(h_hi, 500, 5e6, 10, 30, 0.4, t0, t0 + budget_s - 220)
        h_hi, c5_hi = _active_descent(h_hi, 2000, 5, rng, t0, t0 + budget_s - 160)
        if c5_hi < best_c5:
            best_c5, best_h, best_n = c5_hi, h_hi.copy(), n1

    # Phase E: Fine pairwise refinement (multi-scale)
    if _el() < budget_s - 150:
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]:
            if _el() > budget_s - 100:
                break
            h_try, c5_try = _fine_pairwise(best_h.copy(), 6000, scale, rng, t0, t0 + budget_s - 90)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()

    # Phase F: Final active descent with very small top
    if _el() < budget_s - 60:
        h_try, c5_try = _active_descent(best_h.copy(), 8000, 3, rng, t0, t0 + budget_s - 40)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()

    # Phase G: Ultra-fine pairwise
    if _el() < budget_s - 15:
        for scale in [1e-5, 5e-6, 2e-6, 1e-6]:
            if _el() > budget_s - 8:
                break
            h_try, c5_try = _fine_pairwise(best_h.copy(), 8000, scale, rng, t0, t0 + budget_s - 5)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
