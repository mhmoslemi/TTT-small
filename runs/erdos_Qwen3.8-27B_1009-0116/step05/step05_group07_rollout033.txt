```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_lag(k_idx, h):
    n = len(h)
    s = k_idx - (n - 1)
    g = np.zeros(n)
    dx = 2.0 / n
    if s >= 0:
        if s == 0:
            g[:] = (1.0 - h) * dx
        else:
            g[:n-s] += (1.0 - h[s:]) * dx
            g[s:] -= h[:n-s] * dx
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:]) * dx
        g[sp:] -= h[:n-sp] * dx
    return g

def _proj_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} via bisection on threshold."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj(h, alpha, pen):
    c = _corrs(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    base = m + np.log(np.sum(np.exp(s))) / alpha
    sm = h.sum() - len(h) * 0.5
    return base + pen * sm * sm

def _smooth_grad(h, alpha, pen):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-10:
            g += w[k] * _grad_lag(k, h)
    sm = h.sum() - n * 0.5
    g += pen * sm
    return g

def _lbfgsb(h, alpha, pen, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _active_descent(h, topm, max_iter, step0=0.005):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 20:
            break
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, h)
        g /= topm
        # Line search
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-7 * step * np.dot(g, h_new - h):
                h = h_new; f = f_new; found = True
                break
            step *= 0.5
        if not found:
            break
        step = min(step * 1.05, step0)
    return h, f

def _fw_step(h, g, topm):
    """Frank-Wolfe step: find v minimizing <g,v> s.t. 0<=v<=1, sum(v)=n/2"""
    n = len(h)
    target = n // 2
    # v[i]=1 for target indices with smallest g[i]
    idx = np.argpartition(g, target)[:target]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def _nesterov_fw(h, topm, max_iter):
    """Frank-Wolfe with momentum for active set."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    y = h.copy()  # momentum variable
    f = _eval(h)
    beta = 0.9
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 20:
            break
        t = 2.0 / (it + 2.0)
        c = _corrs(y)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, y)
        g /= topm
        # FW step
        v = np.zeros(n)
        idx = np.argpartition(g, int(target))[:int(target)]
        v[idx] = 1.0
        # Update
        y_new = (1 - t) * y + t * v
        h_new = _repair(y_new, n)
        f_new = _eval(h_new)
        if f_new < f:
            f = f_new
            h = h_new

            if fn < f - 1e-15:

def _coord_descent(h, max_iter):
    """Cyclic coordinate descent with line search on pairs."""
    n = len(h)
    h = h.copy()
    f = _eval(h)

            break

            # Try to adjust h[i] and h[j] while preserving sum
            old_i, old_j = h[i], h[j]
            for d in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_lag(k_idx, h):
    n = len(h)
    s = k_idx - (n - 1)
    g = np.zeros(n)
    dx = 2.0 / n
    if s >= 0:
        g[s:] += (1.0 - h[:n-s]) * dx
        g[:n-s] -= h[s:] * dx
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:]) * dx
        g[sp:] -= h[:n-sp] * dx
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _sobj(h, alpha, pen):
    c = _corrs(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    sm = h.sum() - len(h) * 0.5
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * sm * sm

def _sgrad(h, alpha, pen):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-10:
            g += w[k] * _grad_lag(k, h)
    sm = h.sum() - n * 0.5
    g += pen * sm
    return g

def _lbfgsb(h, alpha, pen, maxiter):
    n = len(h)
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _active_desc(h, topm, max_iter, step0=0.005):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 20:
            break
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, h)
        g /= topm
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-7 * step * np.dot(g, h_new - h):
                h = h_new; f = f_new; found = True
                break
            step *= 0.5
        if not found:
            break
        step = min(step * 1.06, step0)
    return h, f

def _fw_momentum(h, topm, max_iter):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    y = h.copy()
    f = _eval(h)
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 15:
            break
        t = 2.0 / (it + 2.0)
        c = _corrs(y)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, y)
        g /= topm
        v = np.zeros(n)
        idx = np.argpartition(g, int(target))[:int(target)]
        v[idx] = 1.0
        y = (1.0 - t) * y + t * v
        h_new = _repair(y, n)
        f_new = _eval(h_new)
        if f_new < f:
            f = f_new
            h = h_new
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _pairwise(h, n_iter, scale, sd):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(sd)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f
```
