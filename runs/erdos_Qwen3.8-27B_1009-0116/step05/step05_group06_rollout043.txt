def _repair(h, n):
    # Clip and fix sum
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_max_lag(h, k_idx):
    # Gradient of c_k with respect to h
    n = len(h)
    dx = 2.0 / n
    shift = (n - 1) - k_idx
    g = np.zeros(n)
    if shift >= 0:
        i1 = np.arange(n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        if shift > 0:
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:

        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

        if c5 == best_f and np.allclose(h, best_h):

import numpy as np
from scipy.optimize import minimize
import time

_N_GLOBAL = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    shift = (n - 1) - k_idx
    g = np.zeros(n)
    if shift >= 0:
        i1 = np.arange(n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        if shift > 0:
            g[shift:n] -= h[:n - shift]
    else:
        sp = -shift
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _corr_vals(h)
    idx = np.argsort(c)[::-1][:topk]
    cv = c[idx]
    m = cv.max()
    s = alpha * (cv - m)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for j in range(topk):
        if w[j] < 1e-14:
            continue
        g += w[j] * _grad_lag(h, idx[j])
    return g

def _obj_topk(h, alpha, topk):
    c = _corr_vals(h)
    idx = np.argsort(c)[::-1][:topk]
    cv = c[idx]
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, topk, nsteps, lr0, mu, t0, tmax):
    n = len(h)
    f = _obj_topk(h, alpha, topk)
    v = np.zeros(n)
    lr = lr0
    for i in range(nsteps):
        if time.time() - t0 > tmax:
            break
        g = _grad_topk(h, alpha, topk)
        v = mu * v - lr * g
        improved = False
        for _bt in range(25):
            ht = _proj_sum(h - lr * v, n * 0.5)
            fn = _obj_topk(ht, alpha, topk)
            if fn < f - 1e-5 * lr * np.dot(v, ht - h):
                h, f = ht, fn
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.4
            if lr < 1e-15:
                break
        lr = min(lr * 1.03, lr0 * 4.0)
    return h

def _active_set(h, n_iter, topk, lr_base, t0, tmax):
    n = len(h)
    c5 = _eval(h)
    for it in range(n_iter):
        if time.time() - t0 > tmax:
            break
        c = _corr_vals(h)
        idx = np.argsort(c)[::-1][:topk]
        cv = c[idx]
        w = cv / max(cv.sum(), 1e-15)
        g = np.zeros(n)
        for j in range(topk):
            if w[j] < 1e-13:
                continue
            g += w[j] * _grad_lag(h, idx[j])
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        lr = lr_base
        for _ls in range(25):
            ht = _proj_sum(h - lr * g, n * 0.5)
            c5t = _eval(ht)
            if c5t < c5:
                h, c5 = ht, c5t
            else:
                lr *= 0.6
    return h, c5

def _up(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 1, len(h))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h)

def _cons_fun(x):
    return x.sum() - _N_GLOBAL * 0.5

def _cons_jac(x):
    return np.ones(_N_GLOBAL)

def _slsqp_obj(h, alpha, topk):
    return _obj_topk(h, alpha, topk)

def _slsqp_grad(h, alpha, topk):
    return _grad_topk(h, alpha, topk)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

            al = [5000, 20000, 80000][rng.randint(3)]
            hn = _proj_grad(hp, al, 12, 600, 0.002, 0.85, t0, budget_s - 350)
