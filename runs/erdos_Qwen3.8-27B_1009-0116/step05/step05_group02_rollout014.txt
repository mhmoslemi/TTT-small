    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] = (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] = (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _momentum_pgd_lse(h, alpha, n, t_budget, lr_init=0.003, mu=0.92):
    """Momentum projected gradient on LSE-smoothed max."""
    t0 = time.time()
    h_cur = h.copy()
    v = np.zeros(n)
    target = n * 0.5
    best_c5 = _eval(h_cur)
    best_h = h_cur.copy()
    lr = lr_init
    it = 0
    while time.time() - t0 < t_budget:
        c = _corr(h_cur)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-14:
                continue
            g += w[k] * _grad_corr(h_cur, k, n)
        lr = lr_init * (1.0 + it * 0.002) ** (-0.7)
        v = mu * v + lr * g
        h_new = _proj_sum(h_cur - v, target)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        it += 1
    return best_h, best_c5

def _topk_pgd(h, n, t_budget, n_top=5, lr=0.002, mu=0.9):
    """Projected gradient focusing on top-k correlations only."""
    t0 = time.time()
    h_cur = h.copy()
    v = np.zeros(n)
    target = n * 0.5
    best_c5 = _eval(h_cur)
    best_h = h_cur.copy()
    it = 0
    while time.time() - t0 < t_budget:
        c = _corr(h_cur)
        top_k = np.argsort(-c)[:n_top]
        weights = c[top_k]
        weights -= weights.min()
        weights += 1e-10
        weights /= weights.sum()
        g = np.zeros(n)
        for i, k in enumerate(top_k):
            g += weights[i] * _grad_corr(h_cur, k, n)
        v = mu * v + lr * g
        h_new = _proj_sum(h_cur - v, target)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        it += 1
    return best_h, best_c5

def _bisection_tau(h, n, t_budget, tau_lo, tau_hi, n_iter=800):
    """Bisection on tau using violation PGD feasibility check."""
    t0 = time.time()
    best_h = h.copy()
    best_c5 = _eval(best_h)
    while tau_hi - tau_lo > 1e-6 and time.time() - t0 < t_budget:
        if time.time() - t0 > t_budget - 5:
            break
        tau_mid = (tau_lo + tau_hi) * 0.5
        h_feas = _violation_pgd(h, tau_mid, n, t_budget=15)
        c5_feas = _eval(h_feas)
        c = _corr(h_feas)
        max_v = (c.max() * 2.0 / n) - tau_mid
        if max_v < 0.005:
            tau_hi = tau_mid
            if c5_feas < best_c5:
                best_c5 = c5_feas
                best_h = h_feas.copy()
        else:
            tau_lo = tau_mid
            if c5_feas < best_c5:
                best_c5 = c5_feas
                best_h = h_feas.copy()
    return best_h, best_c5, tau_hi

def _violation_pgd(h, tau, n, t_budget, lr=0.002, mu=0.85):
    """PGD on violation sum to drive all c_k*dx below tau."""
    t0 = time.time()
    h_cur = h.copy()
    v = np.zeros(n)
    target = n * 0.5
    dx = 2.0 / n
    while time.time() - t0 < t_budget:
        c = _corr(h_cur)
        exc = np.maximum(c * dx - tau, 0.0)
        if exc.max() < 1e-8:
            break
        g = np.zeros(n)
        active = np.where(exc > 1e-10)[0]
        for k in active:
            w = 2.0 * exc[k] * dx
            g += w * _grad_corr(h_cur, k, n)
        v = mu * v + lr * g
        h_cur = _proj_sum(h_cur - v, target)
    return h_cur

def _slsqp_lse(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            g += (w[k] * dx) * _grad_corr(x, k, n)
        return g
    def eqfun(x):
        return x.sum() - n / 2.0
    def eqjac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eqfun, 'jac': eqjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _proj_sum(res.x, n / 2.0)
    except Exception:
        return h

def _upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _proj_sum(np.array(initial_h_values, dtype=float), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: LSE momentum PGD at n0
    for alpha in [200, 1000, 5000, 20000]:
        if time.time() - t0 > budget_s * 0.15:
            break
        alloc = min(60, max(10, (budget_s * 0.15 - (time.time() - t0)) // 4))
        h_new, c5_new = _momentum_pgd_lse(best_h, alpha, best_n, alloc)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Top-k focused PGD
    if time.time() - t0 < budget_s * 0.25:
        alloc = min(80, int(budget_s * 0.08))
        h_new, c5_new = _topk_pgd(best_h, best_n, alloc, n_top=3, lr=0.001, mu=0.88)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new, c5_new = _topk_pgd(best_h, best_n, alloc, n_top=7, lr=0.0015, mu=0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: SLSQP polish at n0
    if time.time() - t0 < budget_s * 0.35:
        for alpha in [5000, 50000]:
            if time.time() - t0 > budget_s * 0.35:
                break
            h_new = _slsqp_lse(best_h, alpha, 300, 1e-16)
            c5_new = _eval(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Bisection on tau
    if time.time() - t0 < budget_s * 0.50:
        tau_lo = best_c5 * 0.95
        tau_hi = best_c5 * 1.001
        alloc = int((budget_s * 0.50) - (time.time() - t0))
        h_new, c5_new, _ = _bisection_tau(best_h, best_n, max(alloc, 30), tau_lo, tau_hi)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Multi-start with structured perturbations
    if time.time() - t0 < budget_s * 0.65:
        nc = best_n
        for _r in range(40):
            if time.time() - t0 > budget_s * 0.65:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = best_h + 0.015 * rng.randn(nc)
            elif strat == 1:
                f = rng.randint(2, nc // 3 + 1)
                amp = 0.01 * rng.choice([-1, 1])
                hp = best_h + amp * np.sin(2 * np.pi * f * np.arange(nc) / nc + rng.rand() * 6.28)
            elif strat == 2:
                bs = rng.randint(10, nc // 4)
                st = rng.randint(0, nc - bs)
                hp = best_h.copy()
                hp[st:st + bs] += 0.02 * rng.randn(bs)
            elif strat == 3:
                hp = _proj_sum(best_h * rng.uniform(0.9, 1.1), nc / 2.0) + 0.005 * rng.randn(nc)
            else:
                f1, f2 = rng.randint(2, nc // 4 + 1), rng.randint(nc // 4 + 2, nc // 2 + 1)
                hp = best_h + 0.005 * (np.sin(2 * np.pi * f1 * np.arange(nc) / nc) + 0.5 * np.cos(2 * np.pi * f2 * np.arange(nc) / nc))
            hp = _proj_sum(hp, nc / 2.0)
            h_new, c5_new = _momentum_pgd_lse(hp, 5000, nc, 20, lr_init=0.002, mu=0.9)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h_new2, c5_new2 = _topk_pgd(h_new, nc, 10, n_top=3, lr=0.001, mu=0.88)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2

    # Phase 6: Upsample to 2x
    if time.time() - t0 < budget_s * 0.80:
        n1 = best_n * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _proj_sum(h_hi, n1 / 2.0)
        for alpha in [2000, 10000, 50000]:
            if time.time() - t0 > budget_s * 0.80:
                break
            h_new, c5_new = _momentum_pgd_lse(h_hi, alpha, n1, 40, lr_init=0.002, mu=0.9)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n1
        h_new, c5_new = _topk_pgd(best_h, best_n, 40, n_top=5, lr=0.001, mu=0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: SLSQP on higher resolution
    if time.time() - t0 < budget_s * 0.90:
        h_new = _slsqp_lse(best_h, 20000, 500, 1e-17)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 8: Fine pairwise coordinate search
    if time.time() - t0 < budget_s - 5:
        h = best_h.copy()
        nc = best_n
        c5c = best_c5
        for scale in [0.003, 0.001, 0.0002]:
            n_iter = 15000 if scale > 0.001 else 30000
            for _it in range(n_iter):
                if time.time() - t0 > budget_s - 5:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 9: Final ultra-fine
    if time.time() - t0 < budget_s - 1:
        h = best_h.copy()
        nc = best_n
        c5c = best_c5
        for _it in range(30000):
            if time.time() - t0 > budget_s - 1:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _proj_sum(best_h, best_n / 2.0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
