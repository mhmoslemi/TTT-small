Actually, for np.correlate(a, v, mode='full'), the output c[k] = sum_i a[i+k-(len(v)-1)] * v[i] (with appropriate bounds).

For a = h, v = 1-h, both of length n:
c[k] = sum_i h[i+k-(n-1)] * (1-h[i])

For k and 2n-2-k:
c[2n-2-k] = sum_i h[i+(2n-2-k)-(n-1)] * (1-h[i]) = sum_i h[i+n-1-k] * (1-h[i])

And c[k] = sum_i h[i+k-n+1] * (1-h[i])

These are not the same in general (c[k] ≠ c[2n-2-k]) unless h is symmetric. So the correlations are NOT symmetric in general.

But the "positive lag" and "negative lag" versions ARE the same:
- c at "shift +s" (k = n-1+s): sum_i (1-h[i])*h[i+s]
- c at "shift -s" (k = n-1-s): sum_i (1-h[i+s])*h[i] = sum_j (1-h[j])*h[j+s] (substituting j=i+s)

So c(n-1+s) = c(n-1-s). The correlations ARE symmetric about k = n-1!

This means I only need to consider n-1 distinct lags (s = 0, 1, ..., n-1), each appearing twice in the full correlation (except s=0 which appears once).

This is useful for efficiency: I only need to track n-1 correlations instead of 2n-1.

import numpy as np
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(s, h):
    n = len(h)
    if s == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[s:] += (1.0 - h[:n - s])
    g[:n - s] -= h[s:]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _elapsed():
    return time.time() - _t0

def _pgd(h, m, max_iter, step_init, time_limit, anneal=0.0):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step_init
    t_start = time.time()
    rng_local = np.random.RandomState(int(time.time() * 1000) % 100000)
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(c, -m)[-m:]
        g = np.zeros(n)
        for k in top_idx:
            s = abs(k - (n - 1))
            g += _grad_lag(s, h)
        g /= m
        g *= (2.0 / n)
        found = False
        for _ in range(45):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-8 * step:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            if anneal > 0 and rng_local.rand() < anneal:
                h_new = _proj_sum(h - step * g, target)
                f_new = _eval(h_new)
                if f_new < f * 1.001:
                    h = h_new
                    f = f_new
            step *= 0.7
            if step < 1e-12:
                break
        else:
            step = min(step * 1.03, step_init)
    return h, f

def _coord_descent(h, max_iter, time_limit):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    t_start = time.time()
    rng_local = np.random.RandomState(123)
    order = np.arange(n)
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        np.random.RandomState(it).shuffle(order)
        for j in order:
            if time.time() - t_start > time_limit:
                break
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                old = h[j]
                new = old + delta
                if new < 0 or new > 1:
                    continue
                # Need to adjust another point to maintain sum
                h[j] = new
                # Find partner to adjust
                j2 = (j + 1) % n
                adj = old - new
                old2 = h[j2]
                new2 = old2 + adj
                if new2 < 0 or new2 > 1:
                    h[j] = old
                    continue
                h[j2] = new2
                f_new = _eval(h)
                if f_new < f - 1e-15:
                    f = f_new
                else:
                    h[j] = old
                    h[j2] = old2
                    break
    return h, f

def _multi_dir_pgd(h, m, max_iter, step_init, time_limit):
    """PGD cycling through different top-m sets and random subsets."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step_init
    t_start = time.time()
    rng_local = np.random.RandomState(777)
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if it % 3 == 0:
            top_idx = np.argpartition(c, -m)[-m:]
        elif it % 3 == 1:
            top_idx = np.argsort(c)[::-1][:m // 2]
        else:
            top_idx = rng_local.choice(2 * n - 1, size=m, replace=False)
        g = np.zeros(n)
        for k in top_idx:
            s = abs(k - (n - 1))
            g += _grad_lag(s, h)
        g /= len(top_idx)
        g *= (2.0 / n)
        found = False
        for _ in range(40):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-9:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            step *= 0.6
            if step < 1e-12:
                break
        else:
            step = min(step * 1.04, step_init)
    return h, f
