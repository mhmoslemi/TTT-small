    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 5000000]
    for alpha in alphas:
        if _el() > 350:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase C: Active descent
    if _el() < budget_s - 400:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase D: Structured pairwise
    if _el() < budget_s - 350:
        h_new, c5 = _structured_pair(best_h.copy(), 8000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase E: Multi-start
    n_cur = best_n
    if _el() < budget_s - 300:
        for _r in range(30):
            if _el() > budget_s - 250:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.005 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.005 * rng.randn(n_cur), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(3, max(4, n_cur // 8))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st + bs] += 0.02 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            elif strat == 3:
                hp = best_h.copy()
                t = np.arange(n_cur)
                for _f in range(4):
                    f = rng.randint(1, n_cur // 2 + 1)
                    hp += 0.002 * rng.randn() * np.sin(2 * np.pi * f * t / n_cur)
                hp = _repair(hp, n_cur)
            elif strat == 4:
                hp = _repair(np.clip(best_h + 0.03 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur)
            else:
                hp = _repair(np.clip(best_h + 0.01 * rng.randn(n_cur), 0, 1), n_cur)
            h_new = _slsqp(hp, 500000, 1000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                n_cur = len(h_new)
                # Quick polish from improved point
                if _el() < budget_s - 200:
                    h_new2, c52 = _active_descent(best_h.copy(), 2000, 4)
                    if c52 < best_c5:
                        best_c5 = c52
                        best_h = h_new2.copy()

    # Phase F: Upsample to 2x
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [5000, 50000, 500000, 5000000]:
            if _el() > budget_s - 280:
                break
            h_hi = _slsqp(h_hi, alpha, 1500, 1e-20)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _el() < budget_s - 150:
            h_new, c5 = _active_descent(best_h.copy(), 3000, 5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)
        if _el() < budget_s - 100:
            h_new, c5 = _structured_pair(best_h.copy(), 4000, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase G: Long active descent + structured
    if _el() < budget_s - 150:
        h_new, c5 = _active_descent(best_h.copy(), 8000, 3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    if _el() < budget_s - 100:
        h_new, c5 = _structured_pair(best_h.copy(), 8000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase H: Multi-scale pairwise
    if _el() < budget_s - 60:
        for scale in [0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4, 5e-5]:
            if _el() > budget_s - 30:
                break
            h_new, c5 = _pair_refine(best_h.copy(), 5000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase I: Final SLSQP + active descent
    if _el() < budget_s - 20:
        h_new = _slsqp(best_h, 10000000, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    if _el() < budget_s - 3:
        h_new, c5 = _active_descent(best_h.copy(), 10000, 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
