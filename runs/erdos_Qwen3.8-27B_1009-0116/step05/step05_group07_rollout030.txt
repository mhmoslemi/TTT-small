def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = 2.0, -2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    """Full gradient of smooth max (all lags weighted by softmax)."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    return c

def _cgrad_smoth(h, alpha):
    """Gradient of log-sum-exp smoothing."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    m = c.max()
    s = np.clip(alpha*(c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        sh = k - (n-1)
        wk = w[k]*(2.0/n)
        if sh == 0:
            g += wk*(1.0-2.0*h)
        elif sh > 0:

            g[sh:] -= wk*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _slsqp_solve(h, alpha, maxiter=2000, ftol=1e-16, time_limit=60.0):

    def fun(x):

        return _cgrad_smoth(x, alpha)
    def cfun(x):
        return x.sum() - n*0.5

        return np.ones(x.shape)

                       constraints=[{'type':'eq','fun':cfun,'jac':cjac}],
                       options={'maxiter':maxiter, 'ftol':ftol, 'disp':False})
        return _repair(res.x, n)
    except Exception:

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = 2.0, -2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = k - (n - 1)
        wk = w[k] * (2.0 / n)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov_topm(h, topm, n_iter, step_init, stop_time):
    n = len(h)
    target = n * 0.5
    scale = 2.0 / n
    x = _repair(h.copy(), n)
    y = x.copy()
    t = 1.0
    step = step_init
    f_best = _eval(x)
    for i in range(n_iter):
        if time.time() > stop_time:
            break
        c = np.correlate(y, 1.0 - y, mode='full')
        idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in idx:
            sh = k - (n - 1)
            if sh == 0:
                g += (1.0 - 2.0 * y)
            elif sh > 0:
                g[:n-sh] += (1.0 - y[sh:])
                g[sh:] -= y[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - y[:n-sp])
                g[:n-sp] -= y[sp:]
        g *= scale / topm
        z = y - step * g
        x_new = _proj_sum(z, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _proj_sum(y, target)
        x = x_new
        t = t_new
        if i % 3000 == 0:
            fn = _eval(x)
            if fn < f_best:
                f_best = fn
            if i > 10000:
                step *= 0.999
    return x, _eval(x)

def _coord_descend(h, n_passes, stop_time):

                    a[n - 1 - s] -= h[j + s]
            best_t, best_v = hv, f

                    improved = True

                    h[j] = hv

        if not improved:
            break
    return h, f

def _slsqp_solve(h, alpha, maxiter, stop_time):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cfun(x):
        return x.sum() - n * 0.5
    def _cjac(x):
        return np.ones(x.shape)
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'disp': False})
        if time.time() > stop_time:
            return h
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _downsample(h, factor):
    n = len(h)
    n_new = n // factor
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n_new)
