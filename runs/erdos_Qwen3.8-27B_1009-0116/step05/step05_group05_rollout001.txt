    n = len(h)
    dx = 2.0 / n

    step = step_init

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cfull(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _sm_obj(h, alpha):
    n = len(h)
    c = _cfull(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    return m + sm / alpha + np.log((np.exp(s - sm)).sum()) / alpha

def _sm_grad(h, alpha):
    n = len(h)
    c = _cfull(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    ws = w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14 * ws:
            continue
        wk = w[k] / (ws * alpha)
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descend(h, max_iters, n_top, step_init):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    step = step_init
    stall = 0
    for _ in range(max_iters):
        c = _cfull(h) * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmin = cvals.min()
        spread = cvals.max() - cmin + 1e-15
        weights = (cvals - cmin + 1e-10) / (n_top * spread)
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += w * (1.0 - h[sh:n])
                if sh > 0:
                    g[sh:] -= w * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n - sp])
                g[:n - sp] -= w * h[sp:]
        g *= dx
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_new = _proj(h - s * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                if s < step:
                    step = max(s, 1e-7)
                stall = 0
                break
        if not improved:
            step *= 0.5
            stall += 1
            if stall > 20 or step < 1e-9:
                break
    return h, c5c

def _subgrad(h, max_iters, n_top):
    """Subgradient method with Polyak step for the non-smooth max."""
    n = len(h)
    dx = 2.0 / n
    h_best = h.copy()
    c5_best = _eval(h)
    c5_lb = 0.2  # lower bound (trivial)
    for _ in range(max_iters):
        c = _cfull(h) * dx
        idx = np.argsort(c)[::-1][:n_top]
        # Average subgradient over top shifts
        g = np.zeros(n)
        cnt = 0
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:n])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
            cnt += 1
        g *= dx / cnt
        gnorm2 = np.dot(g, g)
        if gnorm2 < 1e-30:
            break
        # Polyak step
        c5_cur = _eval(h)
        step = (c5_cur - c5_lb) / gnorm2
        step = np.clip(step, 1e-8, 1.0)
        h_new = _proj(h - step * g, n * 0.5)
        c5_new = _eval(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        h = h_new
        # Shrink lower bound slowly
        c5_lb += 0.001
    return h_best, c5_best
