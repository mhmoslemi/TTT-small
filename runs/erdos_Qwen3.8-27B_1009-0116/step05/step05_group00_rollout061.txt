from scipy.optimize import minimize
import time
from itertools import combinations

def _eval_fast(h):
    n = len(h)
    N = 2 * n + 1
    H = np.fft.rfft(h, N)
    IH = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(IH), N)[:2*n-1]
    return float(np.max(corr) * 2.0 / n)

def _eval_full(h):
    """Original evaluation using np.correlate for verification"""
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _slsqp_optimize(h, alpha, maxiter=2000, ftol=1e-15):
    n = len(h)
    def _obj(x):
        N = 2 * n + 1
        H = np.fft.rfft(x, N)
        IH = np.fft.rfft(1.0 - x, N)
        corr = np.fft.irfft(H * np.conj(IH), N)[:2*n-1]
        m = corr.max()
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    
    def _sum_eq(x):
        return x.sum() - n * 0.5
    
    def _sum_jac(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return h.copy()
