                        h[i1] += d
                        h[i2] -= d
                else:
                    h[i1] += d
                    h[i2] -= d
    return h, best

def _fine_local(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    for scale in [0.004, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            h[i] += d
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 4), replace=False)
        hp[idx] += 0.08 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.03 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n / 2.0)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: SLSQP warmup with increasing alpha
    for alpha in [200, 1000, 4000, 12000]:
        if time.time() - t0 > budget_s - 300:
            break
        hh, cc = _slsqp(best_h, dx, alpha, 200)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 2: Gradient-guided local search
    hh, cc = _grad_local(best_h, dx, rng, t0 + 200, n_top=5)
    if cc < best_c5:
        best_h, best_c5 = hh, cc

    # Phase 3: Higher-alpha SLSQP
    for alpha in [30000, 80000, 200000]:
        if time.time() - t0 > budget_s - 250:
            break
        hh, cc = _slsqp(best_h, dx, alpha, 250)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 4: Targeted lag reduction
    hh, cc = _targeted_lag(best_h, dx, rng, t0 + 350, n_lags=4)
    if cc < best_c5:
        best_h, best_c5 = hh, cc

    # Phase 5: Second gradient round (more lags)
    hh, cc = _grad_local(best_h, dx, rng, t0 + 500, n_top=7)
    if cc < best_c5:
        best_h, best_c5 = hh, cc

    # Phase 6: Very high alpha SLSQP
    if time.time() - t0 < budget_s - 120:
        for alpha in [300000, 800000]:
            hh, cc = _slsqp(best_h, dx, alpha, 200)
            if cc < best_c5:
                best_h, best_c5 = hh, cc

    # Phase 7: Multi-start perturbations
    t7 = t0 + 700
    for r in range(15):
        if time.time() > t7 or time.time() - t0 > budget_s - 180:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _grad_local(hp, dx, rng, min(t7 - 3, time.time() + 12), n_top=5)
        hp, c = _slsqp(hp, dx, 20000, 150)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hp, c = _targeted_lag(hp, dx, rng, min(t7 - 2, time.time() + 8), n_lags=3)
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 8: Second targeted + gradient polish
    hh, cc = _targeted_lag(best_h, dx, rng, t0 + 800, n_lags=5)
    if cc < best_c5:
        best_h, best_c5 = hh, cc
    hh, cc = _grad_local(best_h, dx, rng, t0 + 860, n_top=4)
    if cc < best_c5:
        best_h, best_c5 = hh, cc

    # Phase 9: Fine local search
    best_h, best_c5 = _fine_local(best_h, dx, rng, t0 + 970)

    # Phase 10: Final SLSQP + micro-search
    if time.time() - t0 < budget_s - 25:
        hh, cc = _slsqp(best_h, dx, 500000, 150)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = _fine_local(best_h, dx, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
