from scipy.optimize import minimize
from scipy.fft import fft, ifft

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, n):
    """Gradient of c[k] with respect to h, where k is index in full correlate."""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        # c[k] = sum_{i=0}^{n-1-sh} h[i] * (1 - h[i+sh])
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        # c[k] = sum_{i=sp}^{n-1} h[i] * (1 - h[i-sp])
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _proj_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _active_set_opt(h, n, top_k=8, n_iter=200, step_scale=0.1):
    """Optimize using active-set: only consider worst 'top_k' lags."""
    h = h.copy()
    c5 = _eval(h)
    dx = 2.0 / n
    
    for it in range(n_iter):
        c = _corr(h)
        # Get top_k worst lags
        idx = np.argsort(c)[::-1][:top_k]
        c_max = c.max() * dx
        
        # Weighted gradient: average gradient of top lags
        # Weight by how close each lag is to the max
        w = np.zeros(len(c))
        w[idx] = 1.0
        # Scale: lags within 5% of max get higher weight
        w[c > c_max * 0.95] = max(w[c > c_max * 0.95], 2.0)
        w /= w.sum()
        
        g = np.zeros(n)
        for k in np.where(w > 0)[0]:
            g += w[k] * _grad_lag(h, k, n)
        
        # Adaptive step size (backtracking)
        step = step_scale
        improved = False

                c5 = c5_new

            step *= 0.5

            h = _repair(h + 0.005 * np.random.randn(n), n)
            c5 = _eval(h)

def _slsqp_lse(h, alpha, maxiter=2000, ftol=1e-19):

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)

        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-15:
                continue
            g += w[k] * _grad_lag(h_in, k, n)
        return g
    
    # This is too slow for large n due to the loop. Use vectorized version.
    # Actually let's just use the loop but limit iterations
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    
    try:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval_c5(h):
    n = len(h)
    return float(np.max(_corr_full(h)) * 2.0 / n)

def _grad_one_lag(h, k_idx, n):
    """Gradient of correlation[k_idx] * (2/n) w.r.t. h."""
    sh = (n - 1) - k_idx
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _active_descent(h, n, top_m=6, n_steps=300):
    """Gradient descent targeting the worst 'top_m' lags."""
    h = h.copy()
    best = _eval_c5(h)
    dx = 2.0 / n
    
    for it in range(n_steps):
        c = _corr_full(h)
        c_scaled = c * dx
        # Weight top lags exponentially
        thresh = c_scaled.max() * 0.92
        active = np.where(c_scaled > thresh)[0]
        if len(active) == 0:
            active = [np.argmax(c_scaled)]
        # Weight by excess
        w = (c_scaled[active] - thresh) + 0.01
        w /= w.sum()
        
        g = np.zeros(n)
        for ki in active:
            g += w[list(active).index(ki)] * _grad_one_lag(h, ki, n)
        
        step = 0.15
        for _ in range(15):
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval_c5(h_new)
            if c5n < best - 1e-14:
                h = h_new
                best = c5n
                break
            step *= 0.5
        else:
            h = _repair(h + 0.003 * np.random.randn(n), n)
            best = _eval_c5(h)
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _grad_lag(h, k_idx, n):
    sh = (n - 1) - k_idx
    g = np.zeros(n)
    dx = 2.0 / n
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lse_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / len(h)

def _lse_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(h, k, n)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lse_obj, h, jac=lambda x: _lse_grad(x, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, n, top_m, n_steps, momentum=0.3):
    h = h.copy()
    best = _eval(h)
    v = np.zeros(n)
    for it in range(n_steps):
        c = _corr(h) * (2.0 / n)
        cmax = c.max()
        thresh = cmax * 0.90
        active = np.where(c > thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        w = (c[active] - thresh + 0.005)
        w /= w.sum()
        g = np.zeros(n)
        for ii, ki in enumerate(active):
            g += w[ii] * _grad_lag(h, ki, n)

        else:

            best = _eval(h)
            v *= 0.5
    return h, best

def _power_mean_opt(h, alpha, n_steps):
    n = len(h)
    h = h.copy()
    best = _eval(h)
    dx = 2.0 / n
    for it in range(n_steps):
        c = _corr(h) * dx
        cmax = c.max()
        p = alpha
        obj = np.mean(c**p) ** (1.0 / p)
        # Gradient of power mean: (1/n) * c^(p-1) * (c - obj) / obj^(p-1) ... complex
        # Use LSE-like weight instead
        s = np.exp(p * (c - cmax))
        s /= s.sum()
        g = np.zeros(n)

            g += s[k] * _grad_lag(h, k, n)

            c5n = _eval(h_new)
            if c5n < best - 1e-14:

                best = c5n
                break
            step *= 0.5
        else:

            best = _eval(h)
    return h, best

def _pairwise_polish(h, n, n_iters, scale):
    rng = np.random.RandomState(12345)
    h = h.copy()
    best = _eval(h)
    for _ in range(n_iters):
        i = rng.randint(n)

        d = scale * rng.randn()

        s = h.sum()
        if abs(s - n * 0.5) > 1e-8:

        if c5n < best - 1e-15:
            best = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, best
