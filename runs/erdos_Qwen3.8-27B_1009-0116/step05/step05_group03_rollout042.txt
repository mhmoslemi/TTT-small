def _repair(h, target):

        if abs(d) < 1e-14:
            break

    d = target - h.sum()

        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _csolve(h, dx, alpha, maxiter=500):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def _eq(x):
        return x.sum() - tgt
    try:
        r = minimize(_obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': _eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _adv_grad(h, dx, t_end, top_k=12, lr_init=0.006):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = lr_init
    c = _corr(h)
    for _ in range(8000000):
        if time.time() > t_end:
            break
        top = np.argsort(c)[::-1][:top_k]
        w = np.exp(90.0 * (c[top] - c[top].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            if wi < 1e-7:
                continue
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            g = np.zeros(n)
            g[s:n] += 1.0 - h[:m]
            g[:m] -= h[s:n]
            grad += wi * g
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = _repair(np.clip(h - lr * grad / gn, 0, 1), tgt)
        c = _corr(h_new)
        c_new = float(c.max() * dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.004, 0.018)
        else:
            lr *= 0.91
            if lr < 1e-7:
                break
    return h, best

def _grad_swap(h, dx, rng, t_end, n_iter=120000):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    c = _corr(h)
    cnt = 0
    while time.time() < t_end and cnt < n_iter:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        cnt += 1
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            # Gradient-guided single swap
            for _ in range(5):
                a = rng.randint(m)
                i1, i2 = a + s, a
                gi1 = (1.0 - h[i1 + (n - 1 - s)]) if i1 + s < n else 0.0
                gi2 = -h[i2 - s] if i2 - s >= 0 else 0.0
                gi1 -= h[i1 - s] if i1 - s >= 0 else 0.0
                gi2 += (1.0 - h[i2 + s]) if i2 + s < n else 0.0
                pref = gi1 - gi2
                if rng.random() < 0.3:
                    d = rng.uniform(-0.012, 0.012)
                else:
                    d = 0.012 * (1.0 if pref > 0 else -1.0) * rng.uniform(0.5, 1.0)
                h[i1] += d; h[i2] -= d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[i1] -= d; h[i2] += d
                else:
                    h[i1] -= d; h[i2] += d
            # 2-swap using two different top lags
            if len(top) >= 2:
                ki2 = top[1 + rng.randint(max(1, len(top) - 1))]
                s2 = abs(int(ki2 - (n - 1)))
                if s2 != 0:
                    m2 = n - s2
                    a2 = rng.randint(min(m2, n))
                    i3 = min(a2 + s2, n - 1)
                    i4 = a2
                    if len(set([i1, i2, i3, i4])) == 4:
                        d1 = rng.uniform(-0.008, 0.008)
                        d2 = rng.uniform(-0.008, 0.008)
                        h[i3] += d1; h[i4] -= d1
                        h[(a+s) % n] += d2; h[a] -= d2
                        ok = all(0 <= h[x] <= 1 for x in [i3, i4, (a+s) % n, a])
                        if ok:
                            cn = _c5(h, dx)
                            if cn < best:
                                best = cn
                            else:
                                h[i3] -= d1; h[i4] += d1
                                h[(a+s) % n] -= d2; h[a] += d2
                        else:
                            h[i3] -= d1; h[i4] += d1
                            h[(a+s) % n] -= d2; h[a] += d2
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.025 * (2.0 * rng.random(n) - 1.0)

        idx = rng.choice(n, max(5, n // 4), replace=False)
        hp[idx] += 0.05 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.018 * np.sin(2*np.pi*rng.uniform(1,6)*x + rng.uniform(0,6.28))
        hp += 0.015 * np.cos(4*np.pi*rng.uniform(1,3)*x + rng.uniform(0,6.28))
    elif mode == 3:
        idx = rng.choice(n, max(6, n // 5), replace=False)
        hp[idx] = 1.0 - hp[idx]
    elif mode == 4:
        hp = 1.0 - hp[::-1].copy()
    else:
        k = rng.randint(2, 8)
        hp += 0.012 * np.sin(k * np.pi * np.linspace(0, 2, n) * rng.uniform(0.5, 1.5))
    return _repair(np.clip(hp, 0, 1), tgt)

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)
