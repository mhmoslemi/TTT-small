def _binary_local_search(h, n, max_rounds, rng):
    """Local search on binary sequences: swap 1s and 0s."""
    h = (h > 0.5).astype(float)  # project to binary
    h = _repair(h, n)
    best_c5 = _eval(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for r in range(max_rounds):
        improved = False
        # Try random swaps
        for _ in range(200):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5 = _eval(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                # Update sets
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _smoothed_grad(x, alpha):
    n = len(x)
    c = _corr_full(x)
    m = c.max()

    s -= s.max()

    wsum = w.sum()
