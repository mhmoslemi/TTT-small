```python
def _optimal_d(c_current, delta, d_lo, d_hi):
    """Find d in [d_lo, d_hi] that minimizes max_s (delta[s]*d + c_current[s])"""
    # The function f(d) = max_s (delta[s]*d + c_current[s]) is convex piecewise linear
    # Its minimum on [d_lo, d_hi] is at:
    # - d_lo or d_hi (endpoints)
    # - A vertex where two lines cross and the slope changes sign
    
    # Simple approach: ternary-like search (since convex)
    # Or: evaluate at all intersection points within [d_lo, d_hi]
    
    # For n=559 lines, computing all intersections is O(n²) = 312K. Too slow per iteration.
    # Use binary search on the subgradient instead.
    
    # f(d) is convex. Its subgradient at d is:
    # {sum of delta_s for s where delta_s*d + c_s = f(d)}
    # The minimum is where 0 is in the subgradient.
    
    # Binary search: find d where the slope changes from negative to positive.
    # slope(d) = {positive if the max is achieved by lines with positive delta, 
    #            negative if by lines with negative delta}
    
    # Actually for a convex PWL function, we can binary search:
    # f(d_lo) and f(d_hi) give bounds
    # The minimum d* satisfies: all lines with delta < 0 have f_s(d*) <= min, 
    # and all lines with delta > 0 have f_s(d*) <= min
    
    # Simpler: since f is convex PWL with at most 559 pieces,
    # binary search on d using the evaluation is O(log(range/eps) * n)
    
    lo, hi = d_lo, d_hi
    for _ in range(60):
        mid = (lo + hi) / 2
        # Compute f(mid) and subgradient sign
        vals = delta * mid + c_current
        m = vals.max()
        # Subgradient: weighted average of delta where vals = m
        active = vals >= m - 1e-12
        if active.sum() == 0:
            break
        sg = delta[active].mean()  # approximate subgradient
        if sg > 1e-10:
            hi = mid
        elif sg < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _grad_lse(h, alpha, top_k=50):
    """Gradient of LSE objective. Only uses top_k shifts for speed."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    # Softmax (numerically stable)
    s_max = s.max()
    w = np.exp(s - s_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # For large alpha, only top_k weights matter
    if alpha > 1000:
        idx = np.argsort(w)[-top_k:]
        w_masked = np.zeros_like(w)
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]
    return g

def _proj_bisect(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} via bisection on offset."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _fista(h_init, alpha, n_iter, t_end, top_k=50):
    n = len(h_init)
    target = n * 0.5
    x = h_init.copy()
    x_prev = h_init.copy()
    t = 1.0
    # Lipschitz constant estimate
    L = alpha * 2.0  # conservative
    step = 1.0 / L
    best_x = x.copy()
    best_f = np.inf
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, -0.5, 1.5)  # keep in reasonable range
        g = _grad_lse(y, alpha, top_k)
        x_new = y - step * g
        x_new = _proj_bisect(x_new, target)
        
        # Track best
        f = _eval(x_new)
        if f < best_f:
            best_f = f
            best_x = x_new.copy()
        
        x_prev = x
        x = x_new
        t = t_new
    
    return best_x

def _pairwise_descent(h, n_iter, t_end, rng):
    """Exact 1D pairwise coordinate descent."""
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Compute delta_s = dc/dh[i] - dc/dh[j] for all shifts
        # dc/dh[i] for shift s: (1-h[i-s])*[valid] - h[i+s]*[valid]
        di = np.zeros(2*n-1)
        dj = np.zeros(2*n-1)
        for s in range(-(n-1), n):
            # shift s means correlation at index (n-1+s)
            # h[i] contributes: (1-h[i-s]) if 0<=i-s<n, minus h[i+s] if 0<=i+s<n
            if 0 <= i - s < n:
                di[n-1+s] += (1.0 - h[i-s])
                if i == i - s + s:  # always true, this is the h[j+s] term
                    pass
            if 0 <= i + s < n:
                di[n-1+s] -= h[i + s]
            if 0 <= j - s < n:
                dj[n-1+s] += (1.0 - h[j-s])
            if 0 <= j + s < n:
                dj[n-1+s] -= h[j + s]
        
        delta = di - dj
        # Valid range for d: h[i]+d in [0,1], h[j]-d in [0,1]
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi <= d_lo:
            continue
        
        # Minimize max_s (delta[s]*d + c[s]) over d in [d_lo, d_hi]
        # Binary search on 1D convex function
        lo, hi = d_lo, d_hi
        for _ in range(50):
            mid = (lo + hi) * 0.5
            vals = delta * mid + c
            m = vals.max()
            active = (vals >= m - 1e-12)
            if active.any():
                sg = delta[active].mean()
            else:
                break
            if sg > 1e-10:
                hi = mid
            elif sg < -1e-10:
                lo = mid
            else:
                break
        d_opt = (lo + hi) * 0.5
        
        # Apply
        h[i] += d_opt
        h[j] -= d_opt
        # Recompute c
        c = np.correlate(h, 1.0 - h, mode='full')
        c5_new = c.max() * 2.0 / n
        if c5_new > c5:  # safety: revert if we got worse (numerical)
            h[i] -= d_opt
            h[j] += d_opt
            c = np.correlate(h, 1.0 - h, mode='full')
        else:
            c5 = c5_new
    
    return h
```

```python
    di = np.zeros(2*n-1)
    # s >= 0: s from 0 to n-1-i
    s_pos = np.arange(0, min(n-i, n))  # s values where i+s < n

    if len(s_pos) > 0:
        di[n-1+s_pos] -= h[i + s_pos]
    
    # s < 0: sp from 1 to n-1
    sp = np.arange(1, n)
    sp_valid_h = sp[sp < n - i]  # i+sp < n
    if len(sp_valid_h) > 0:

    sp_valid_h2 = sp[sp <= i]  # i>=sp
    if len(sp_valid_h2) > 0:
        di[n-1-sp_valid_h2] -= h[i - sp_valid_h2]
    
    return di
```
