    c = _corr(x) * dx
    cm = c.max()
    s = alpha * (c - cm)
    s = s - s.max()
    return cm + np.log(np.sum(np.exp(s))) / alpha + mu * (x.sum() - tgt) ** 2

    c = _corr(x) * dx
    cm = c.max()
    s = alpha * (c - cm)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    nh = n - 1

            grad[:m] += wi * (1.0 - x[k:n])

        else:

            grad[k:n] += wi * (1.0 - x[:m])
            grad[:m] -= wi * x[k:n]

def _lbfgs(h, dx, alpha, mu, maxiter=500):

    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(_lse_val, h, jac=lambda x: _lse_grad(x, dx, alpha, mu, tgt),
                     args=(dx, alpha, mu, tgt),
                     method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

        hc, c = _lbfgs(best_h, dx, a, mu, maxiter)
        if c < best_c:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _lse_val(x, dx, alpha, mu, tgt):
    c = _corr(x) * dx
    cm = c.max()
    s = alpha * (c - cm)
    s = s - s.max()
    return cm + np.log(np.sum(np.exp(s))) / alpha + mu * (x.sum() - tgt) ** 2

def _lse_grad(x, dx, alpha, mu, tgt):
    n = len(x)
    c = _corr(x) * dx
    cm = c.max()
    s = alpha * (c - cm)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    nh = n - 1
    active = np.where(w > 1e-8)[0]
    for ki in active:
        wi = w[ki] * dx
        k = int(ki - nh)
        if k >= 0:
            m = n - k
            grad[:m] += wi * (1.0 - x[k:n])
            grad[k:n] -= wi * x[:m]
        else:
            k = -k
            m = n - k
            grad[k:n] += wi * (1.0 - x[:m])
            grad[:m] -= wi * x[k:n]
    grad += 2.0 * mu * (x.sum() - tgt)
    return grad

def _lbfgs(h, dx, alpha, mu, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(_lse_val, h,
                     jac=lambda x: _lse_grad(x, dx, alpha, mu, tgt),
                     args=(dx, alpha, mu, tgt),
                     method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs_multi(h, dx, t_end, alphas, mu=30.0, maxiter=250):
    best_h, best_c = h, _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc, c = _lbfgs(best_h, dx, a, mu, maxiter)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def _perturb(h, rng, mode):

    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        k = max(5, n // 4)
        idx = rng.choice(n, k, replace=False)
        hp[idx] += 0.12 * (2.0 * rng.random(k) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1.5, 7.0)
        ph = rng.uniform(0, 6.28)
        hp += 0.05 * np.sin(2 * np.pi * f * x + ph)
    elif mode == 3:
        k = max(4, n // 6)
        idx = rng.choice(n, k, replace=False)
        hp[idx] = 1.0 - hp[idx]
    else:
        start = rng.randint(0, max(1, n - 25))
        length = rng.randint(12, max(16, n // 3))
        hp[start:start + length] = 1.0 - hp[start:start + length]
    return _repair(np.clip(hp, 0, 1), tgt)

def _crossover(h1, h2, rng):
    n = len(h1)
    tgt = n / 2.0
    w = rng.uniform(0.25, 0.75)
    hc = w * h1 + (1.0 - w) * h2
    return _repair(np.clip(hc, 0, 1), tgt)

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 5e-4, 2e-4, 1e-4, 4e-5]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best
