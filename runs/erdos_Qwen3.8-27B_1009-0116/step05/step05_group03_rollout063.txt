    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

        if s_val < n:

import time

def _corr(h):

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h
