Actually, one more thought: the targeted search in the previous code does random pair modifications. A more structured approach would be:
1. Compute the correlation c
2. Find the top-K lags (say K=10)
3. For each top lag s, compute the gradient of that lag with respect to h
4. Find the pair (i,j) that most reduces the worst lag
5. Make that move

This is more directed than random search.

For lag s (shift), the contribution is:
f_s = sum_i h[i] * (1-h[i-s]) (for valid i)
or equivalently for the other direction:
f_{-s} = sum_i h[i] * (1-h[i+s])

The gradient df_s/dh[j]:
= (1-h[j-s]) * I[0≤j-s<n] - h[j+s] * I[0≤j+s<n]

To reduce f_s, we want to decrease h where the gradient is positive, and increase h where the gradient is negative.

A simple greedy approach:
- Find j* = argmax grad_s[j] (most positive gradient → most to decrease)
- Find j** = argmin grad_s[j] (most negative gradient → most to increase)
- Transfer mass from j* to j**: h[j*] -= d, h[j**] += d
- Choose d to maintain constraints

This is O(n) per step, very fast.

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def _repair(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-13:
                break
    return h

def _obj_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    e = np.exp(s)
    Z = float(e.sum())
    w = e / Z
    obj = m + float(np.log(Z)) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        si = k - (n - 1)
        a1 = max(0, si)
        b1 = min(n - 1, n - 1 + si)
        if a1 <= b1:
            grad[a1:b1 + 1] += wk * (1.0 - h[a1 - si:b1 - si + 1])
        a2 = max(0, -si)
        b2 = min(n - 1, n - 1 - si)
        if a2 <= b2:
            grad[a2:b2 + 1] -= wk * h[a2 + si:b2 + si + 1]
    return obj, grad

                     constraints=[{'type': 'eq', 'fun': _ef, 'jac': _eg}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, n / 2.0)
        return hc, _c5(hc, 2.0 / n)
    except Exception:
        return h, _c5(h, 2.0 / n)

    dx = 2.0 / n
    best_h, best_c = h.copy(), _c5(h, dx)
    h = h.copy()

        cn = _c5(hn, dx)

            h = hn

            if lr < 1e-7:

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

_alpha = 100.0

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-13:
                break
    return h

def _obj_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    e = np.exp(s)
    Z = float(e.sum())
    w = e / Z
    obj = m + np.log(Z) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        si = k - (n - 1)
        a1 = max(0, si)
        b1 = min(n - 1, n - 1 + si)
        if a1 <= b1:
            grad[a1:b1 + 1] += wk * (1.0 - h[a1 - si:b1 - si + 1])
        a2 = max(0, -si)
        b2 = min(n - 1, n - 1 - si)
        if a2 <= b2:
            grad[a2:b2 + 1] -= wk * h[a2 + si:b2 + si + 1]
    return obj, grad

def _wo(x):
    return _obj_grad(x, _alpha)[0]

def _wg(x):
    return _obj_grad(x, _alpha)[1]

def _ef(x):
    return x.sum() - len(x) / 2.0

def _eg(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    global _alpha
    _alpha = alpha
    n = len(h)
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(_wo, h, jac=_wg, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': _ef, 'jac': _eg}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, n / 2.0)
        return hc, _c5(hc, 2.0 / n)
    except Exception:
        return h, _c5(h, 2.0 / n)

def _pgrad(h, alpha, t_end, lr0):
    n = len(h)
    dx = 2.0 / n
    best_h, best_c = h.copy(), _c5(h, dx)
    h = h.copy()
    lr = lr0
    while time.time() < t_end:
        _, g = _obj_grad(h, alpha)
        hn = _repair(h - lr * g, n / 2.0)
        cn = _c5(hn, dx)
        if cn < best_c:
            best_h, best_c = hn.copy(), cn
            h = hn
            lr = min(lr * 1.008, 0.05)
        else:
            lr *= 0.86
            if lr < 1e-7:
                break
    return best_h, best_c

def _lag_grad(h, si):
    n = len(h)
    g = np.zeros(n)
    a1 = max(0, si)
    b1 = min(n - 1, n - 1 + si)
    if a1 <= b1:
        g[a1:b1 + 1] = 1.0 - h[a1 - si:b1 - si + 1]
    a2 = max(0, -si)
    b2 = min(n - 1, n - 1 - si)
    if a2 <= b2:
        g[a2:b2 + 1] -= h[a2 + si:b2 + si + 1]
    return g

def _targeted(h, rng, t_end, K):
    n = len(h)
    dx = 2.0 / n
    best = _c5(h, dx)
    while time.time() < t_end:
        c = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(c)[::-1][:K]
        for ki in top:
            si = ki - (n - 1)
            g = _lag_grad(h, si)
            id_ = int(np.argmax(g))
            ii_ = int(np.argmin(g))
            if g[ii_] >= g[id_] - 1e-12:
                continue
            gap = g[id_] - g[ii_]
            for fs in [0.008, 0.003, 0.001]:
                d = min(fs * gap, h[id_], 1.0 - h[ii_])
                if d < 1e-10:
                    continue
                h[id_] -= d
                h[ii_] += d
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                    break
                h[id_] += d
                h[ii_] -= d
        for _ in range(2):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.004, 0.004)
            h[i] += d
            h[j] -= d
            ok = 0 <= h[i] <= 1 and 0 <= h[j] <= 1
            if ok:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best

                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 5), replace=False)
        hp[idx] += 0.05 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)

    else:
        idx = rng.choice(n, max(2, n // 8), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), n / 2.0)

def _interp(h, nn):
    return np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, len(h)), h)
