import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    diff = target - s
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-12:
            break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _all_pos_changes(h, n):
    pc = np.zeros((n, 2 * n - 1))
    for k in range(n):
        sh1 = np.arange(n - k)
        k1 = (n - 1) - sh1
        pc[k, k1] += (1.0 - h[k:k + n - k])
        sh2 = np.arange(k + 1)
        k2 = (n - 1) - sh2
        pc[k, k2] -= h[k::-1][:k + 1]
        sp = np.arange(1, n)
        k3 = (n - 1) + sp
        m1 = sp <= k
        if m1.any():
            pc[k, k3[m1]] += (1.0 - h[k - sp[m1]])
        m2 = sp < (n - k)
        if m2.any():
            pc[k, k3[m2]] -= h[k + sp[m2]]
    return pc

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    diff = target - s
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-12:
            break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _all_pos_changes(h, n):
    pc = np.zeros((n, 2 * n - 1))
    for k in range(n):
        sh1 = np.arange(n - k)
        k1 = (n - 1) - sh1
        pc[k, k1] += (1.0 - h[k:k + n - k])
        sh2 = np.arange(k + 1)
        k2 = (n - 1) - sh2
        pc[k, k2] -= h[k::-1][:k + 1]
        sp = np.arange(1, n)
        k3 = (n - 1) + sp
        m1 = sp <= k
        if m1.any():
            pc[k, k3[m1]] += (1.0 - h[k - sp[m1]])
        m2 = sp < (n - k)
        if m2.any():
            pc[k, k3[m2]] -= h[k + sp[m2]]
    return pc

def _binary_search(h, n, rng, deadline):
    c = _corr(h)
    c5 = c.max() * 2.0 / n
    best_h = h.copy()
    best_c5 = c5
    T = 0.015
    cooling = 0.9997
    while time.time() < deadline:
        pc = _all_pos_changes(h, n)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            base = c - pc[i]
            new_maxs = (base[None, :] + pc[zeros]).max(axis=1)
            deltas = new_maxs * 2.0 / n - c5
            idx = int(np.argmin(deltas))
            if deltas[idx] < best_delta - 1e-14:
                best_delta = float(deltas[idx])
                best_i = int(i)
                best_j = int(zeros[idx])
        if best_i < 0:
            if T > 1e-7 and rng.rand() < 0.3:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                dc = -pc[i] + pc[j]
                nm = float((c + dc).max())
                delta = nm * 2.0 / n - c5
                if delta < 0 or rng.rand() < np.exp(-delta / T):
                    best_i, best_j, best_delta = i, j, delta
                else:
                    T *= cooling
                    continue
            else:
                T *= cooling
                if T < 1e-9:
                    break
                continue
        elif best_delta > 0:
            if T > 1e-9 and rng.rand() > np.exp(-best_delta / T):
                T *= cooling
                continue
        h[best_i] = 1.0 - h[best_i]
        h[best_j] = 1.0 - h[best_j]
        c = c - pc[best_i] + pc[best_j]
        T *= cooling
        new_c5 = c.max() * 2.0 / n
        c5 = new_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _relax(h, n, rng, deadline):
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    while time.time() < deadline:
        n_p = rng.randint(1, 8)
        idxs = rng.choice(n, n_p, replace=False)
        deltas = rng.uniform(-0.08, 0.08, n_p)
        h_new = h.copy()
        h_new[idxs] += deltas
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def _grad_refine(h, n, deadline):
    alpha = 50000.0
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    step = 0.005
    while time.time() < deadline:
        corr = _corr(h)
        m = corr.max()
        s = np.clip(alpha * (corr - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w[k] * (1.0 - h[i1 + sh])
                i2 = np.arange(sh, n)
                g[i2] -= w[k] * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w[k] * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[k] * h[i2 + sp]
        improved = False
        for _ in range(30):
            h_new = h - step * g
            h_new = _repair(h_new, n)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-12:
                h = h_new
                c5 = c5_new
                improved = True
                break
            step *= 0.5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            step = min(step * 2.0, 0.005)
            break
    return best_h, best_c5

    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)

    # Phase 1: Binary search at n0

    if c5_b1 < best_c5:

    # Phase 1b: Try golden-ratio binary pattern
    if time.time() < t0 + budget_s * 0.40:

        h_gold = np.array([1.0 if (i * phi) % 1.0 > 0.5 else 0.0 for i in range(n0)])
        h_gold = _repair(h_gold, n0)
        h_gb, c5_gb = _binary_search(h_gold, n0, rng, t0 + budget_s * 0.40)
        if c5_gb < best_c5:
            best_h, best_c5, best_n = h_gb.copy(), c5_gb, n0

    # Phase 2: Upsample to 2*n0, binary search
    n1 = n0 * 2
    h_up = np.repeat(best_h, 2)
    h_up = _repair(h_up, n1)
    dl2 = t0 + budget_s * 0.65
    h_b2, c5_b2 = _binary_search(h_up, n1, rng, dl2)
    if c5_b2 < best_c5:
        best_h, best_c5, best_n = h_b2.copy(), c5_b2, n1

    # Phase 3: Gradient refinement on best
    if time.time() < t0 + budget_s * 0.85:

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
