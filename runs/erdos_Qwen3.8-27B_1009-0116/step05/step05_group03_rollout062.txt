```python
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on lambda."""
    n = len(h)
    lo, hi = h.min() - 1, h.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0, 1)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d/n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(h) if d > 0 else np.argsort(-h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0-h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h); tgt = n/2.0
    bds = [(0.0, 1.0)]*n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha*(c-m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s))))/alpha
    def eq(x): return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type':'eq','fun':eq}],
                     options={'maxiter':maxiter,'ftol':1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except: return h, _c5(h, dx)

def _coord_desc(h, t_end, n_samp=51):
    n = len(h); dx = 2.0/n; tgt = n/2.0
    h = h.copy(); C = _corr(h)
    best = float(np.max(C)*dx)
    ts = np.linspace(0, 1, n_samp)
    lags = np.arange(2*n-1) - (n-1)
    while time.time() < t_end:
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            hi = h[i]
            ip = i + lags; im = i - lags
            vp = (ip>=0)&(ip<n); vm = (im>=0)&(im<n)
            slope = np.zeros(2*n-1)
            slope[vp] = 1.0 - h[np.clip(ip[vp],0,n-1)]
            slope[vm] -= h[np.clip(im[vm],0,n-1)]
            # 1D convex min: min_t max_k(C[k] + slope[k]*t)
            vals = C + slope*ts[:,None].T  # shape (n_samp, 2n-1)
            mv = np.max(vals, axis=1)
            tb = ts[int(np.argmin(mv))]
            if 0 < tb < 1:
                tf = np.linspace(max(0,tb-0.005), min(1,tb+0.005), 21)
                vf = C + slope*tf[:,None].T
                mf = np.max(vf, axis=1)
                tb = tf[int(np.argmin(mf))]
            if abs(tb - hi) > 1e-9:
                C += (tb-hi)*slope
                h[i] = tb
                improved += 1
        if improved == 0: break
        h = _repair(h, tgt)
        C = _corr(h)
        c = float(np.max(C)*dx)
        if c < best: best = c
    return h, best

def _binary_sa(h, rng, t_end):
    n = len(h); dx = 2.0/n; tgt = n/2.0
    h = (h > 0.5).astype(float)
    if h.sum() != tgt:
        # fix
        diff = int(tgt - h.sum())
        if diff > 0:
            idx = np.where(h==0)[0]
            h[rng.choice(idx, min(diff,len(idx)),replace=False)] = 1
        elif diff < 0:
            idx = np.where(h==1)[0]
            h[rng.choice(idx, min(-diff,len(idx)),replace=False)] = 0
    best = _c5(h, dx)
    h = h.copy()
    T = 0.05
    t0 = time.time()
    while time.time() < t_end:
        T *= 0.9995
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
        c = _c5(h, dx)
        if c < best or rng.random() < np.exp(-(c-best)/max(T,1e-8)):
            best = min(best, c)
        else:
            h[i] = 1.0 - h[i]
        if int(time.time()-t0) % 3 > 1:  # every ~3 seconds, unquantize
            break
    return h, best

def _targeted(h, dx, rng, t_end):

    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        for ki in top:
            s = abs(int(ki - (n-1)))
            if s == 0: continue
            m = n - s
            for _ in range(5):

                d = rng.uniform(0.002, 0.015)

def _fine(h, dx, rng, t_end):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(h) if d > 0 else np.argsort(-h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h); tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coord_desc(h, t_end, n_samp=81):
    n = len(h); dx = 2.0 / n; tgt = n / 2.0
    h = h.copy()
    C = _corr(h)
    best = float(np.max(C) * dx)
    ts = np.linspace(0, 1, n_samp)
    lags = np.arange(2 * n - 1) - (n - 1)
    ge = 0
    while time.time() < t_end:
        improved = 0
        for i in range(n):
            hi = h[i]
            ip = i + lags; im = i - lags
            vp = (ip >= 0) & (ip < n); vm = (im >= 0) & (im < n)
            slope = np.zeros(2 * n - 1)
            slope[vp] = 1.0 - h[np.clip(ip[vp], 0, n - 1)]
            slope[vm] -= h[np.clip(im[vm], 0, n - 1)]
            vals = C + slope * ts[:, None].T
            mv = np.max(vals, axis=1)
            tb = ts[int(np.argmin(mv))]
            if 0 < tb < 1:
                lo = max(0.0, tb - 0.004); hi2 = min(1.0, tb + 0.004)
                tf = np.linspace(lo, hi2, 31)
                vf = C + slope * tf[:, None].T
                mf = np.max(vf, axis=1)
                tb = tf[int(np.argmin(mf))]
            if abs(tb - hi) > 1e-10:
                C += (tb - hi) * slope
                h[i] = tb
                improved += 1
        if improved == 0:
            ge += 1
            if ge > 5:
                break
        else:
            ge = 0
        h = _repair(h, tgt)
        C = _corr(h)
        c = float(np.max(C) * dx)
        if c < best:
            best = c
    return h, best

def _binary_sa(h, rng, t_end):

        else:

    return hb, best

def _targeted(h, dx, rng, t_end):

                else:
                    h[i] -= d; h[j] += d

    return h, best

def _alt_start(n):

    h[:n // 2] = 1.0
    return h, _repair(h, n / 2.0)
```
