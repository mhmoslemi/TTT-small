        h_new = _proj_grad(best_h, 80000, 300.0, 400, 0.005)
        h_new = _repair(h_new, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Multi-resolution (downsample, optimize, upsample, refine)
    if _el() < budget_s * 0.55:
        for n_sub in [n0 // 7, n0 // 4]:
            if _el() > budget_s * 0.55:
                break
            if n_sub < 16 or n0 % n_sub != 0:
                continue
            h_sub = _downsample(best_h, n0 // n_sub)
            h_sub = _repair(h_sub, n_sub)
            for alpha in [1000, 10000, 60000]:
                if _el() > budget_s * 0.58:
                    break
                h_sub = _lbfgs(h_sub, alpha, 30.0, 3000)
                h_sub = _repair(h_sub, n_sub)
            h_up = _upsample(h_sub, n0 // n_sub)
            h_up = _repair(h_up, n0)
            h_up = _lbfgs(h_up, 50000, 150.0, 2000)
            h_up = _repair(h_up, n0)
            c5 = _eval(h_up)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_up.copy(), n0

    # Phase 4: Upsample to 2x and optimize
    if _el() < budget_s * 0.65:
        n2 = best_n * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n2)
        for alpha in [10000, 80000, 300000]:
            if _el() > budget_s * 0.72:
                break
            h_hi = _lbfgs(h_hi, alpha, 100.0, 2500)
            h_hi = _repair(h_hi, n2)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n2
        if _el() < budget_s * 0.78:
            h_hi = _proj_grad(best_h, 200000, 500.0, 300, 0.002)
            h_hi = _repair(h_hi, best_n)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()

    # Phase 5: Multi-start perturbations
    if _el() < budget_s * 0.82:
        nc = best_n
        n_starts = 12
        for _r in range(n_starts):
            if _el() > budget_s * 0.82:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = best_h + 0.015 * rng.randn(nc)
            elif strat == 1:
                hp = 1.0 - best_h + 0.01 * rng.randn(nc)
            elif strat == 2:
                hp = best_h.copy()
                bs = max(4, nc // 10)
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.04 * rng.randn(bs)
            elif strat == 3:
                t_arr = np.arange(nc)
                hp = best_h.copy()
                for f in rng.randint(1, nc // 3 + 1, size=5):
                    hp += 0.006 * rng.randn() * np.sin(2 * np.pi * f * t_arr / nc + rng.rand() * 2 * np.pi)
            else:
                hp = best_h.copy()
                hp[rng.randint(nc, size=nc // 5)] += 0.03 * rng.randn(nc // 5)
            hp = _repair(hp, nc)
            alpha_pick = [50000, 200000, 800000][rng.randint(3)]
            hp = _lbfgs(hp, alpha_pick, 300.0, 1500)
            hp = _repair(hp, nc)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hp.copy(), nc

    # Phase 6: Simulated annealing pair search
    if _el() < budget_s - 80:
        h_work = best_h.copy()
        nc = best_n
        for scale, temp in [(0.004, 0.001), (0.0015, 0.0003), (0.0004, 0.0)]:
            if _el() > budget_s - 80:
                break
            h_work, c5_ws = _pair_search(h_work, 5000, scale, rng, temp)
            h_work = _repair(h_work, nc)
            c5_ws = _eval(h_work)
            if c5_ws < best_c5:
                best_c5, best_h, best_n = c5_ws, h_work.copy(), nc

    # Phase 7: Ultra-fine local search
    if _el() < budget_s - 15:
        h_work = best_h.copy()
        nc = best_n
        for scale in [2e-4, 5e-5, 1e-5]:
            if _el() > budget_s - 15:
                break
            h_work, c5_ws = _pair_search(h_work, 6000, scale, rng)
            h_work = _repair(h_work, nc)
            c5_ws = _eval(h_work)
            if c5_ws < best_c5:
                best_c5, best_h, best_n = c5_ws, h_work.copy(), nc

    # Phase 8: Final L-BFGS polish with very high alpha
    if _el() < budget_s - 5:
        h_final = _lbfgs(best_h, 1000000, 1000.0, 3000)
        h_final = _repair(h_final, best_n)
        c5_f = _eval(h_final)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_final.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
