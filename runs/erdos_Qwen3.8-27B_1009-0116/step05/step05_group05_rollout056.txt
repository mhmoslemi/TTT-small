import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n); g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp); g[i2] -= wk * h[i2 + sp]
    return g

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_one(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n); g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n); g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp); g[i2] -= h[i2 + sp]
    return g

def _active_descent(h, max_iters, n_top, t_end):
    n = len(h); dx = 2.0 / n
    c5c = _eval(h)
    t0 = time.time()
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        spread = cvals[0] - cvals[-1]
        if spread < 1e-14:
            weights = np.ones(n_top) / n_top
        else:
            weights = (cvals - cvals[-1] + 1e-12) ** 2
            weights /= weights.sum()
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            g += w * _grad_one(h, int(k), n)
        g *= dx
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new; c5c = c5n; improved = True
                break
        if not improved:
            break
    return h, c5c

def _pairwise_grad(h, i, j, n):
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        gi = 0.0; gj = 0.0
        if sh >= 0:
            if i + sh < n: gi += (1.0 - h[i + sh])
            if sh > 0 and i - sh >= 0: gi -= h[i - sh]
            if j + sh < n: gj += (1.0 - h[j + sh])
            if sh > 0 and j - sh >= 0: gj -= h[j - sh]
        else:
            sp = -sh
            if i - sp >= 0: gi += (1.0 - h[i - sp])
            if i + sp < n: gi -= h[i + sp]
            if j - sp >= 0: gj += (1.0 - h[j - sp])
            if j + sp < n: gj -= h[j + sp]
        a[k] = gi - gj
    return a

    h_hi = np.zeros(n * factor)

        if _el() < budget_s - 40:
            h_new, c5 = _pairwise_optimize(best_h.copy(), 30000, rng, t0 + budget_s - 40)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy(); best_n = n1

    # Phase 7: Final SLSQP polish
    if _el() < budget_s - 10:
        h_new = _slsqp(best_h, 1000000, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy(); best_n = len(h_new)

    # Phase 8: Ultra-fine active set
    if _el() < budget_s - 3:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 2, t0 + budget_s - 2)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
