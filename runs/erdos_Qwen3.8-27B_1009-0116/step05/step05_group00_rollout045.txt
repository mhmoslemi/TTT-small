    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _slsqp(h, alpha, maxiter=3000):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh == 0:
                g[:] += wk * (1.0 - x)
            elif sh > 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    def sum_eq(x):
        return x.sum() - n * 0.5
    def sum_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    n = len(h)

    c5 = _eval(h)

    for base in [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        improved = 0
        for _ in range(steps_per_phase):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = base * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = np.correlate(h, 1.0 - h, mode='full')
            c5n = c.max() * dx
            if c5n < c5 - 1e-16:
                c5 = c5n
                improved += 1
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if improved < 3:
            break
    return h, c5
