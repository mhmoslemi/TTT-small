import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h = np.clip(h * target / s, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lse(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[c.argmax()] = 1.0
    else:
        w /= ws
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _grad_worst(h, nw):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -nw)[-nw:]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return g / float(nw)

def _lam_lse(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lam_lse_jac(x, alpha):
    return _grad_lse(x, alpha)

def _lam_sum(x):
    return x.sum() - len(x) * 0.5

def _lam_sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lam_lse, h, jac=_lam_lse_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_loop(h, grad_fn, n_iter, lr, t_end):
    n = len(h)
    bm, bv = np.zeros(n), np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    h = h.copy()
    best_h, best_v = h.copy(), _eval(h)
    stuck = 0
    for t in range(1, n_iter + 1):
        if time.time() > t_end:
            break
        g = grad_fn(h)
        bm = b1 * bm + (1 - b1) * g
        bv = b2 * bv + (1 - b2) * g * g
        mhat = bm / (1 - b1 ** t)
        vhat = bv / (1 - b2 ** t)
        h_new = _proj(h - lr * mhat / (np.sqrt(vhat) + eps), n * 0.5)
        v = _eval(h_new)
        if v < best_v:
            best_v = v
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 8000:
            lr *= 0.6
            stuck = 0
    return best_h, best_v
