            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _freeze_and_optimize(h, freeze_eps):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    frozen = (h < freeze_eps) | (h > 1.0 - freeze_eps)
    frozen_vals = h[frozen].copy()
    free_idx = np.where(~frozen)[0]
    if len(free_idx) < 10:
        return h
    h[frozen] = np.where(frozen_vals < 0.5, 0.0, 1.0)
    h2 = _repair(h, n)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget, _cur_n
    _t0 = time.time()
    _budget = budget_s
    _cur_n = 0
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: SLSQP alpha sweep
    for alpha in [1000, 3000, 10000, 50000, 200000]:
        if time.time() - _t0 > 100:
            break
        hn = _slsqp(best_h, alpha, 2500, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase B: p-norm SLSQP (different smooth max)
    if time.time() - _t0 < budget_s - 300:
        for p in [30, 60]:
            if time.time() - _t0 > budget_s - 250:
                break
            hn = _slsqp_pnorm(best_h, p, 2000, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase C: Leveling descent with momentum
    if time.time() - _t0 < budget_s - 250:
        for _pass in range(3):
            if time.time() - _t0 > budget_s - 200:
                break
            hn, fc = _leveling(best_h, 4000, 0.004, 0.75)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase D: Round-robin on individual lags
    if time.time() - _t0 < budget_s - 200:
        for _rr in range(10):
            if time.time() - _t0 > budget_s - 150:
                break
            hn, fc = _round_robin(best_h, 60, 10)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase E: Binary push
    if time.time() - _t0 < budget_s - 150:
        hn, fc = _binary_push(best_h, 30000, rng)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase F: High-alpha SLSQP polish
    if time.time() - _t0 < budget_s - 120:
        for alpha in [500000, 1000000, 2000000]:
            if time.time() - _t0 > budget_s - 80:
                break
            hn = _slsqp(best_h, alpha, 3000, 1e-19)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase G: Multi-start with structured perturbations
    nc = len(best_h)
    if time.time() - _t0 < budget_s - 100:
        for _r in range(10):
            if time.time() - _t0 > budget_s - 60:
                break
            hp = best_h.copy()
            if _r % 4 == 0:
                hp += 0.012 * rng.randn(nc)
            elif _r % 4 == 1:
                t = np.linspace(0, 4.0 * np.pi, nc)
                hp += 0.025 * np.sin(t + _r * 0.7)
            elif _r % 4 == 2:
                bp = rng.randint(0, nc // 2)
                bp2 = bp + nc // 4 + rng.randint(0, nc // 8)
                hp[bp:bp2] += 0.02
            else:
                hp[::2] += 0.015 * rng.randn(nc // 2)
            hp = _repair(hp, nc)
            hn = _slsqp(hp, 50000, 1500, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase H: Fine round-robin polishing
    if time.time() - _t0 < budget_s - 60:
        for _pass in range(4):
            if time.time() - _t0 > budget_s - 35:
                break
            hn, fc = _round_robin(best_h, 200, 6)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase I: Stochastic local search (decreasing scale)
    for scale, nit in [(0.005, 15000), (0.002, 20000), (0.0005, 25000), (0.0001, 20000), (0.00003, 15000)]:
        if time.time() - _t0 > budget_s - 20:
            break
        hn, fc = _stoch_local(best_h, nit, scale, seed + int(scale * 1e8))
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase J: Final SLSQP + round-robin
    if time.time() - _t0 < budget_s - 15:
        hn = _slsqp(best_h, 1000000, 2000, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()
        if time.time() - _t0 < budget_s - 3:
            hn, fc = _round_robin(best_h, 50, 8)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
