                c5 = _eval(h_p)
                update(h_p, c5)
                base = h_p[:p].copy()

    # Phase 3: Multi-start from random binary patterns
    if elapsed() < budget_s - 150:
        n_s = min(best_n, 100)
        for trial in range(8):
            if elapsed() > budget_s - 150: break
            h_r = np.where(rng.rand(n_s) < 0.5, 1.0, 0.0)
            h_r = _repair(h_r, n_s)
            h_r, c5r = _binary_ls(h_r, 2, rng)
            h_r = _slsqp(h_r, 50000, 1500)
            c5r = _eval(h_r)
            update(h_r, c5r)

    # Phase 4: Top-k targeted descent on best
    if elapsed() < budget_s - 80:
        h4 = best_h.copy()
        for tk in [1, 2, 3, 5, 8, 12]:
            if elapsed() > budget_s - 80: break
            h4, c5 = _topk_descent(h4, tk, 60, rng)
            update(h4, c5)

    # Phase 5: Pair-flip local search
    if elapsed() < budget_s - 40:
        h5 = np.where(best_h > 0.5, 1.0, 0.0)
        h5 = _repair(h5, best_n)
        h5, c5 = _pair_flip_ls(h5, 5, rng)
        update(h5, c5)

    # Phase 6: Final high-alpha SLSQP
    if elapsed() < budget_s - 10:
        h_fin = _slsqp(best_h, 500000, 5000)
        update(h_fin, _eval(h_fin))
        h_fin = _slsqp(1.0 - best_h, 500000, 5000)
        update(h_fin, _eval(h_fin))

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
