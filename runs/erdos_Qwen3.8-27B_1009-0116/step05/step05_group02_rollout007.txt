import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lse_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam_opt(h0, alphas, iters, lr, t0, t_stop):
    n = len(h0)
    h = h0.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    best = h.copy()
    best_c5 = _eval(h)
    for alpha in alphas:
        for _ in range(iters):
            if time.time() - t0 > t_stop:
                return best, best_c5
            g = _lse_grad(h, alpha)
            t += 1
            m1 = 0.9 * m1 + 0.1 * g
            m2 = 0.999 * m2 + 0.001 * g * g
            mh = m1 / (1 - 0.9 ** t)
            vh = m2 / (1 - 0.999 ** t)
            h = _proj_sum(h - lr * mh / (np.sqrt(vh) + 1e-8), n * 0.5)
            c5 = _eval(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best = h.copy()
    return best, best_c5

def _sgd_active(h0, t0, t_stop, n_it=50000):
    n = len(h0)
    h = h0.copy()
    c5 = _eval(h)
    dx = 2.0 / n
    lr = 0.01
    for it in range(n_it):
        if time.time() - t0 > t_stop:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-5:]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += 1.0 - h[sh:]
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += 1.0 - h[:n - sp]
                g[:n - sp] -= h[sp:]
        g *= dx / len(top_k)
        h = _proj_sum(h - lr * g, n * 0.5)
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            lr *= 0.95
            h = h0 if lr < 1e-6 else h
            if lr < 1e-6:
                h = _proj_sum(h0 + 0.005 * np.random.randn(n), n * 0.5)
                c5 = _eval(h)
                lr = 0.01
    return h, c5

def _local_search(h0, n_it, scale, t0, t_stop):
    n = len(h0)
    h = h0.copy()
    c5 = _eval(h)
    rng = np.random.RandomState(77)
    for _ in range(n_it):
        if time.time() - t0 > t_stop:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

def _up(h, f):
    n = len(h)
    out = np.zeros(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def _sum_obj(x, n, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _sum_jacob(x, n, alpha):
    return _lse_grad(x, alpha)

def _slsqp(h0, alpha, maxiter, t0, t_stop):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _sum_fun(x, n), 'jac': lambda x: _sum_jac(x, n)}]
    def obj(x):
        return _sum_obj(x, n, alpha)
    def jac(x):
        return _sum_jacob(x, n, alpha)
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)
