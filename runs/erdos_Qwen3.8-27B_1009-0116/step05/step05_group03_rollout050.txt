import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _lse_obj(x, alpha, k_top):

    m = c.max()
    idx = np.argsort(c)[::-1][:k_top]
    s = alpha * (c[idx] - m)
    sm = s.max()
    return m + (sm + float(np.log(np.exp(s - sm).sum()))) / alpha

def _lse_grad(x, alpha, k_top):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[::-1][:k_top]
    s = alpha * (c[idx] - m)
    sm = s.max()
    p = np.exp(s - sm)
    p /= p.sum()

    # Actually: df/dx = sum_i p_raw_i * dc_i/dx where p_raw_i = exp(alpha*c_i)/sum(exp(alpha*c_j))
    # = sum_i p_i * dc_i/dx where p_i is normalized softmax over top-K, times... 
    # Let me redo: f = (1/alpha)*log(sum exp(alpha*c_i))
    # df/dx = (1/alpha)*sum_i exp(alpha*c_i)*alpha*(dc_i/dx)/sum exp(alpha*c_j)
    #       = sum_i [exp(alpha*c_i)/sum exp(alpha*c_j)] * dc_i/dx

    # df/dx ≈ sum_{i in topK} w_i * dc_i/dx
    # where w_i = exp(alpha*(c_i-m)) / sum_{j in topK} exp(alpha*(c_j-m))
    # (the m factor cancels in softmax)
    grad = np.zeros(n)
    for ki, wi in zip(idx, p / alpha):  # p/alpha gives the correct weight
        si = int(ki) - (n - 1)
        if si > 0:

            grad[:n - si] -= wi * x[si:]
        elif si < 0:
            t = -si
            grad[:n - t] += wi * (1.0 - x[t:])
            grad[t:] -= wi * x[:n - t]
        else:
            grad += wi * (1.0 - 2.0 * x)
    return grad

def _slsqp_opt(h, alpha, k_top, maxiter=400):

    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return _lse_obj(x, alpha, k_top)
    def jac(x):
        return _lse_grad(x, alpha, k_top)
    def eq(x):
        return x.sum() - tgt
    def eqjac(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eqjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
