import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_lag(h, s):
    """Exact gradient of lag-s overlap w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    # First appearance: h[i] in h[i]*(1-h[i+s]) for i < n-s
    if s > 0 and s < n:
        g[:n-s] += (1.0 - h[s:n])
    # Second appearance: h[i] in h[i-s]*(1-h[i]) for i >= s
    if s > 0 and s < n:
        g[s:] -= h[:n-s]
    return g

def _combo_grad(h, lags, weights):
    """Combined gradient for weighted top lags."""
    g = np.zeros_like(h)
    for s, w in zip(lags, weights):
        if s == 0:
            continue
        g += w * _grad_lag(h, s)
    g -= g.mean()  # Project to preserve sum
    return g

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _continuation(h, dx, t_end, alphas):
    """SLSQP with continuation on alpha."""
    for a in alphas:
        if time.time() > t_end:
            break
        h, c = _slsqp(h, dx, a, 300)
    return h, _c5(h, dx)

def _directed_search(h, dx, rng, t_end, top_k=5):
    """Directed local search using exact gradients for top lags."""
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.01
    while time.time() < t_end:
        c = _corr(h)
        # Get top lags (excluding 0)
        c_lags = c.copy()
        c_lags[n-1] = -1  # exclude lag 0
        top_idx = np.argsort(c_lags)[::-1][:top_k]
        # Weights: softmax over top values
        top_c = c_lags[top_idx]
        w = np.exp(50.0 * (top_c - top_c.max()))
        w /= w.sum()
        # Compute combined gradient
        lags = [abs(int(ki - (n-1))) for ki in top_idx]
        g = np.zeros(n)
        for wi, s in zip(w, lags):
            if s == 0:
                continue
            g += wi * _grad_lag(h, s)
        g -= g.mean()
        
        # Greedy: find best (i1, i2) pair
        # g[i] > 0 means increasing h[i] helps (reduces objective)
        # So move mass from negative g to positive g
        i2 = np.argmax(g)  # increase here
        i1 = np.argmin(g)  # decrease here
        
        if g[i2] - g[i1] < 1e-10:
            # Try random pairs
            for _ in range(20):
                a, b = rng.randint(n), rng.randint(n)
                if a == b:

                d = scale * (2.0 * rng.random() - 1.0)
                h[a] += d; h[b] -= d
                if 0 <= h[a] <= 1 and 0 <= h[b] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[a] -= d; h[b] += d
                else:
                    h[a] -= d; h[b] += d
            continue
        
        # Move mass from i1 to i2
        max_d = min(h[i1], 1.0 - h[i2], scale)
        d = max_d * (0.5 + 0.5 * rng.random())
        h[i1] -= d; h[i2] += d
        cn = _c5(h, dx)
        if cn < best:
            best = cn
        else:

def _subgrad(h, dx, t_end, max_iter=50000):
    """Subgradient with exact gradients for top lags."""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.003
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        c_lags = c.copy()
        c_lags[n-1] = -1
        top_idx = np.argsort(c_lags)[::-1][:4]
        top_c = c_lags[top_idx]
        w = np.exp(80.0 * (top_c - top_c.max()))
        w /= w.sum()
        
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            s = abs(int(ki - (n-1)))
            if s == 0:
                continue
            g += wi * _grad_lag(h, s)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _repair(h - lr * g / gn, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.005, 0.02)
        else:
            lr *= 0.92
            if lr < 1e-7:
                break
    return h, best

def _corr(h):

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a

def _continuation(h, dx, t_end, alphas):
    for a in alphas:
        if time.time() > t_end:
            break
        h, c = _slsqp(h, dx, a, 250)
    return h, _c5(h, dx)

def _subgrad(h, dx, t_end, max_iter=80000):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.004
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        _, lags, w, _ = _top_lags(h, 5)
        g = _combined_grad(h, lags, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _repair(h - lr * g / gn, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.005, 0.02)
        else:
            lr *= 0.93
            if lr < 1e-7:
                break
    return h, best

def _directed(h, dx, rng, t_end, top_k=5):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.008
    while time.time() < t_end:
        _, lags, w, c_work = _top_lags(h, top_k)
        g = _combined_grad(h, lags, w)
        i2 = int(np.argmax(g))
        i1 = int(np.argmin(g))
        if g[i2] - g[i1] < 1e-10:
            # Random fallback
            for _ in range(10):
                a, b = int(rng.randint(n)), int(rng.randint(n))
                if a == b:
                    continue
                d = scale * 0.5 * (2.0 * rng.random() - 1.0)
                h[a] += d; h[b] -= d
                if 0 <= h[a] <= 1 and 0 <= h[b] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[a] -= d; h[b] += d
                else:
                    h[a] -= d; h[b] += d
            scale = max(scale * 0.98, 1e-5)
            continue
        max_d = min(h[i1], 1.0 - h[i2], scale)
        if max_d < 1e-8:
            scale *= 0.95
            scale = max(scale, 1e-5)
            i1_r, i2_r = int(rng.randint(n)), int(rng.randint(n))
            if i1_r != i2_r:
                d = scale * 0.3 * (2.0 * rng.random() - 1.0)
                h[i1_r] += d; h[i2_r] -= d
                if 0 <= h[i1_r] <= 1 and 0 <= h[i2_r] <= 1:
                    if _c5(h, dx) < best:
                        best = _c5(h, dx)
                    else:
                        h[i1_r] -= d; h[i2_r] += d
                else:
                    h[i1_r] -= d; h[i2_r] += d
            continue
        d = max_d * (0.4 + 0.6 * rng.random())
        h[i1] -= d; h[i2] += d
        cn = _c5(h, dx)
        if cn < best:
            best = cn
        else:
            h[i1] += d; h[i2] -= d
        if rng.random() < 0.003:
            scale *= 0.97
            scale = max(scale, 1e-5)
    return h, best

def _single_lag(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.004
    while time.time() < t_end:
        _, lags, _, c_work = _top_lags(h, 1)
        s = lags[0]
        if s == 0:
            break
        g = _grad_lag(h, s)
        g -= g.mean()
        i2 = int(np.argmax(g))
        i1 = int(np.argmin(g))
        if g[i2] - g[i1] < 1e-10:
            break
        max_d = min(h[i1], 1.0 - h[i2], scale)
        if max_d < 1e-10:
            scale *= 0.95
            if scale < 1e-6:
                break
            continue
        d = max_d * rng.uniform(0.3, 1.0)
        h[i1] -= d; h[i2] += d
        cn = _c5(h, dx)
        if cn < best:
            best = cn
        else:
            h[i1] += d; h[i2] -= d
        if rng.random() < 0.002:
            scale *= 0.98
            scale = max(scale, 1e-6)
    return h, best

def _perturb(h, rng, mode, n):
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.04 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 5), replace=False)
        hp[idx] += 0.08 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1, 6)
        hp += 0.03 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 8), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # P1: Continuation SLSQP
    h, c = _continuation(best_h, dx, t0 + 130, [100, 300, 1000, 4000, 15000, 60000, 200000])
    if c < best_c5:
        best_h, best_c5 = h, c

    # P2: Subgradient with correct exact gradients
    h, c = _subgrad(best_h, dx, t0 + 280)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P3: Directed gradient search
    h, c = _directed(best_h, dx, rng, t0 + 480, top_k=5)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P4: SLSQP refinement with high alpha
    h, c = _continuation(best_h, dx, t0 + 580, [20000, 80000, 300000, 1000000])
    if c < best_c5:
        best_h, best_c5 = h, c

    # P5: Single worst lag focused
    h, c = _single_lag(best_h, dx, rng, t0 + 700)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P6: Multi-start with diverse perturbations
    t_end6 = t0 + 830
    for r in range(10):
        if time.time() > t_end6 or budget_s - (time.time() - t0) < 270:
            break
        hp = _perturb(best_h, rng, r % 4, n)
        hp, c = _subgrad(hp, dx, min(t_end6 - 15, time.time() + 12))
        hp, c = _slsqp(hp, dx, 50000, 200)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hp, c = _directed(hp, dx, rng, min(t_end6 - 3, time.time() + 6), top_k=4)
            if c < best_c5:
                best_h, best_c5 = hp, c

    # P7: 2x resolution
    t_end7 = t0 + 940
    if budget_s - (time.time() - t0) > 310:
        n2, dx2 = 2 * n, 1.0 / n
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _subgrad(h2, dx2, min(t_end7 - 100, time.time() + 70))
        h2, c2 = _slsqp(h2, dx2, 10000, 200)
        h2, c2 = _directed(h2, dx2, rng, min(t_end7 - 30, time.time() + 50), top_k=4)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # P8: Final directed on best
    nb, dxb = best_n, 2.0 / best_n
    best_h, best_c5 = _directed(best_h, dxb, rng, t0 + 1030, top_k=4)

    # P9: Final SLSQP
    if time.time() < t0 + 1070:
        h, c = _slsqp(best_h, dxb, 500000, 200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P10: Micro single-lag polish
    if budget_s - (time.time() - t0) > 8:
        best_h, best_c5 = _single_lag(best_h, dxb, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
