    dc[ks1[m1]] += d * (1.0 - h[js[m1]])
    ks2 = n - 1 + i - js
    m2 = (ks2 >= 0) & (ks2 < 2 * n - 1)
    dc[ks2[m2]] -= d * h[js[m2]]
    return dc

def _sa_cont(h, c, rng, n_iter, T0, dx, n):
    c5c = c.max() * dx
    T = T0
    cool = (1e-5 / T0) ** (1.0 / max(n_iter, 1))
    dc_buf = np.zeros(2 * n - 1)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = T * 80.0 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            if d > 0:
                d = min(d, 1.0 - h[i], h[j])
            else:
                d = max(d, h[i] - 1.0, -(1.0 - h[j]))
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1 or d == 0:
                T *= cool
                continue
        np.add(_pos_dc(h, i, d, n), _pos_dc(h, j, -d, n), out=dc_buf)
        nc5 = np.max(c + dc_buf) * dx
        if nc5 < c5c or rng.random() < np.exp(-(nc5 - c5c) / max(T, 1e-12)):
            h[i] = ni
            h[j] = nj
            c += dc_buf
            c5c = nc5
        T *= cool
    return h, c, c5c

def _sa_binary(h, rng, n_iter, T0, dx, n):
    hb = (h > 0.5).astype(float)
    s = hb.sum()
    c = _corr(hb)
    c5c = c.max() * dx
    T = T0
    cool = (1e-6 / T0) ** (1.0 / max(n_iter, 1))
    dc_buf = np.zeros(2 * n - 1)
    ones = np.where(hb > 0.5)[0]
    zeros = np.where(hb < 0.5)[0]
    for _ in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        d = 1.0
        c1 = _pos_dc(hb, i, -d, n)
        c2 = _pos_dc(hb, j, d, n)
        np.add(c1, c2, out=dc_buf)
        nc5 = np.max(c + dc_buf) * dx
        if nc5 < c5c or rng.random() < np.exp(-(nc5 - c5c) / max(T, 1e-12)):
            hb[i] = 0.0
            hb[j] = 1.0
            c += dc_buf
            c5c = nc5
            ones = np.where(hb > 0.5)[0]
            zeros = np.where(hb < 0.5)[0]
        T *= cool
    return hb, c5c

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    sf = lambda x: x.sum() - n * 0.5
    sj = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': sf, 'jac': sj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    dx0 = 2.0 / n0

    # Phase 1: SLSQP polish
    for alpha in [5000, 50000, 200000]:
        if _el() > 60:
            break
        hn = _slsqp(best_h, alpha, 1500, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Intensive continuous SA (multi-restart with cooling)
    c_work = _corr(best_h)
    for restart in range(5):
        if _el() > 250:
            break
        h_sa = best_h.copy()
        c_sa = _corr(h_sa)
        n_iter = 3000000
        T0 = 0.015 / (restart + 1)
        h_sa, c_sa, c5_sa = _sa_cont(h_sa, c_sa, rng, n_iter, T0, dx0, n0)
        h_sa = _repair(h_sa, n0)
        c5 = _eval(h_sa)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sa.copy()

    # Phase 3: Binary SA
    if _el() < budget_s - 300:
        hb, c5_b = _sa_binary(best_h, rng, 2000000, 0.05, dx0, n0)
        if c5_b < best_c5:
            best_c5, best_h, best_n = c5_b, hb.copy(), n0
        # Also try from a random binary start
        for _r in range(3):
            if _el() > 250:
                break
            hr = np.zeros(n0)
            hr[rng.choice(n0, n0 // 2, replace=False)] = 1.0
            hr, c5_br = _sa_binary(hr, rng, 500000, 0.03, dx0, n0)
            if c5_br < best_c5:
                best_c5, best_h, best_n = c5_br, hr.copy(), n0

    # Phase 4: Upsample to 2x and optimize
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        dx1 = 2.0 / n1
        for alpha in [10000, 100000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp(h_hi, alpha, 1000, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        c_hi = _corr(best_h)
        h_hi2, c_hi2, _ = _sa_cont(best_h.copy(), c_hi, rng, 2000000, 0.01, dx1, n1)
        h_hi2 = _repair(h_hi2, n1)
        c5 = _eval(h_hi2)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_hi2.copy(), n1

    # Phase 5: Greedy pairwise (only improving moves, many iterations)
    if _el() < budget_s - 100:
        h_g = best_h.copy()
        c_g = _corr(h_g)
        n_g = len(h_g)
        dx_g = 2.0 / n_g
        c5_g = c_g.max() * dx_g
        dc_buf = np.zeros(2 * n_g - 1)
        n_steps = 5000000
        for scale in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0003]:
            if _el() > budget_s - 100:
                break
            for _ in range(n_steps // 7):
                i = rng.randint(n_g)
                j = rng.randint(n_g)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni = h_g[i] + d
                nj = h_g[j] - d
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                np.add(_pos_dc(h_g, i, d, n_g), _pos_dc(h_g, j, -d, n_g), out=dc_buf)
                nc5 = np.max(c_g + dc_buf) * dx_g
                if nc5 < c5_g - 1e-15:
                    h_g[i] = ni
                    h_g[j] = nj
                    c_g += dc_buf
                    c5_g = nc5
        h_g = _repair(h_g, n_g)
        c5 = _eval(h_g)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_g.copy(), n_g

    # Phase 6: Final ultra-fine SA on best
    if _el() < budget_s - 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        h_f = best_h.copy()
        c_f = _corr(h_f)
        h_f, c_f, _ = _sa_cont(h_f, c_f, rng, 2000000, 0.003, dx_f, n_f)
        h_f = _repair(h_f, n_f)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_f.copy(), n_f

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
