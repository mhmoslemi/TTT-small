def _repair(h, n):

        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                    diff -= a

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_topk(h, alpha, top_k=30):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    idx = np.argsort(w)[::-1][:top_k]
    w2 = np.zeros_like(w)
    w2[idx] = w[idx]
    w2 /= w2.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    nsh = 2 * n - 1
    for k in np.nonzero(w2 > 1e-12)[0]:
        wk = w2[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:

                g[i2] -= wk * h[i2 - sh]
        else:

            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

        y = h + 0.9 * v
        y = y - lr * g
        h_new = _repair(y, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            v = 0.9 * (h_new - h)
            h = h_new
            c5c = c5n
        else:
            v *= 0.5
            lr = max(lr * 0.9, 1e-8)
    return h, c5c

def _binary_search(h, n, max_rounds, budget_fn):

    if abs(h.sum() - n/2) > 1:
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        need = int(round(n/2 - h.sum()))
        if need > 0:
            for i in idx1[:need]:
                h[i] = 0
        elif need < 0:
            for i in idx0[:abs(need)]:
                h[i] = 1
    h = _repair(h, n)
    c5c = _eval(h)
    ones = np.where(h == 1)[0]

def _interval_init(n, q, offset=0):
    h = np.zeros(n)
    w = n // (2 * q)

        s = offset + i * (n // q)
        e = min(s + w, n)
