def _mseq_init(n, rng):

    mlen = 1 << (n.bit_length()) - 1
    reg = np.ones(np.log2(mlen + 1).astype(int), dtype=int)
    taps = np.log2(mlen + 1).astype(int) - 1
    taps = (taps, 0)
    seq = np.zeros(mlen, dtype=int)
    reg = np.ones(len(taps) + 2, dtype=int)
    for i in range(mlen):
        seq[i] = reg[-1]
        fb = np.sum(reg[list(taps)]) % 2
        reg = np.roll(reg, 1)
        reg[0] = fb
    # Convert to real [0,1] perturbation
    f = seq.astype(float) * 2.0 - 1.0
    if n > mlen:
        f = np.resize(f, n)
    else:
        f = f[:n]
    f = f - f.mean()
    return f
