def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_weighted(h, w, active):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        wk = w[k]
        if wk < 1e-12:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g * dx

def _self_consistent(h, n_iter, T0, T1, step0, t_deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iter - 1))
        c_all = _eval_all(h)
        s = c_all / max(T, 1e-10)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        active = np.where(w > 1e-8)[0]
        g = _grad_weighted(h, w, active)
        step = step0
        improved = False
        for _bs in range(12):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c_new = _eval(h_new)
            c_old = _eval(h)
            if c_new < c_old - 1e-15:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            step = step0 * 0.1
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            if _eval(h_new) < _eval(h):
                h = h_new
    return best_h, best_c

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / len(h)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sa_refine(h, t0, t1, n_iter, rng, t_deadline):
    n = len(h)
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    for i in range(n_iter):
        if time.time() > t_deadline:
            break
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        blk = rng.randint(2, max(3, n // 10))
        st = rng.randint(0, n - blk)
        delta = rng.randn(blk) * (0.004 + 0.035 * t)
        h_new = h.copy()
        h_new[st:st + blk] += delta
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _pairwise_refine(h, n_tries, rng, delta_base, t_deadline):
    n = len(h)
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        if time.time() > t_deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = delta_base * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c_new = _eval(h)
        if c_new < c_cur - 1e-16:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t_start

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Self-consistent min-max at base res (two passes, cooling)
    dl1 = t_start + (budget_s - 350) * 0.35
    h_sc, c_sc = _self_consistent(h, 2500, 0.25, 0.008, 0.006, dl1)
    if c_sc < best_c5:
        best_c5 = c_sc
        best_h = h_sc.copy()

    dl2 = t_start + (budget_s - 350) * 0.55
    h_sc, c_sc = _self_consistent(best_h, 2000, 0.08, 0.002, 0.003, dl2)
    if c_sc < best_c5:
        best_c5 = c_sc
        best_h = h_sc.copy()

    # Phase 2: SLSQP polish at base res
    if _el() < budget_s - 300:
        for alpha in [10000, 100000, 500000]:
            if _el() > budget_s - 300:
                break
            h_c = _slsqp(best_h, alpha, 2500, 1e-16)
            c5 = _eval(h_c)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c.copy()

    # Phase 3: Upsample to 2x and optimize
    if _el() < budget_s - 200:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        c5 = _eval(h_hi)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_hi.copy()
            best_n = n1
        if _el() < budget_s - 150:
            dl3 = t_start + (budget_s - 130) * 0.7
            h_hi, c_hi = _self_consistent(h_hi, 1500, 0.15, 0.005, 0.004, dl3)
            if c_hi < best_c5:
                best_c5 = c_hi
                best_h = h_hi.copy()
                best_n = n1
            h_hi = _slsqp(h_hi, 200000, 1500, 1e-16)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase 4: SA at best resolution
    nc = best_n
    if _el() < budget_s - 150:
        t_rem = (budget_s - 150) - _el()
        n_iter = min(int(t_rem / 0.035), 12000)
        dl4 = t_start + 150
        h_sa, c_sa = _sa_refine(best_h, 0.003, 1e-6, n_iter, rng, dl4)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa.copy()

    # Phase 5: Multi-start SLSQP
    nc = best_n
    if _el() < budget_s - 130:
        for _r in range(8):
            if _el() > budget_s - 130:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.008 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = rng.randint(nc // 6, nc // 3)
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] = 1.0 - hp[st:st + bs]
                hp = _repair(hp, nc)
            elif strat == 2:
                hp = _repair(best_h + 0.015 * rng.randn(nc), nc)
            else:
                hp = best_h.copy()
                for _b in range(6):
                    i = rng.randint(nc - 1)
                    d = 0.012 * rng.randn()
                    ni, nj = hp[i] + d, hp[i + 1] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        hp[i], hp[i + 1] = ni, nj
                hp = _repair(hp, nc)
            h_new = _slsqp(hp, 300000, 1200, 1e-16)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = nc

    # Phase 6: Pairwise fine-tune (two passes with decreasing delta)
    if _el() < budget_s - 30:
        t_rem = (budget_s - 30) - _el()
        n_tries = min(int(t_rem * 0.6 / 0.002), 40000)
        dl6 = t_start + 30 + n_tries * 0.002 * 0.4
        h_pf, c_pf = _pairwise_refine(best_h, n_tries, rng, 3e-5, dl6)
        if c_pf < best_c5:
            best_c5 = c_pf
            best_h = h_pf.copy()
        if _el() < budget_s - 10:
            t_rem2 = (budget_s - 10) - _el()
            n_tries2 = min(int(t_rem2 / 0.002), 30000)
            h_pf, c_pf = _pairwise_refine(best_h, n_tries2, rng, 1e-5, t_start + 10)
            if c_pf < best_c5:
                best_c5 = c_pf
                best_h = h_pf.copy()

    # Phase 7: Final SLSQP with very high alpha
    if _el() < budget_s - 8:
        h_c = _slsqp(best_h, 2000000, 3000, 1e-18)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c
            best_n = len(h_c)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
