            lo = mid
        else:
            hi = mid

def _grad_topm(h, m=5):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += (1.0 - h[:cnt])
            g[:cnt] -= h[sp:sp+cnt]
    return g * dx

    return np.sum(np.power(c, p)) ** (1.0/p)

    dx = 2.0 / n
    c = _corr(h)
    # grad of (sum c_k^p)^(1/p) = (1/p) * (sum c_k^p)^(1/p - 1) * p * c_k^(p-1) * dc_k/dh
    # = (sum c_k^p)^(1/p - 1) * c_k^(p-1) * dc_k/dh

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag_into(h, m_idx, g):
    n = len(h)
    k = m_idx - (n - 1)
    if k >= 0:
        cnt = n - k
        g[k:] += (1.0 - h[:cnt])
        if cnt > 0:
            g[:cnt] -= h[k:]
    else:
        sp = -k
        cnt = n - sp
        g[:cnt] += (1.0 - h[sp:])
        g[sp:] -= h[:cnt]

def _grad_topm(h, m_top):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-m_top:]
    g = np.zeros(n)
    for mi in idx:
        _grad_lag_into(h, mi, g)
    return g * dx

def _ps_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    s = np.sum(c ** p)
    alpha = s ** (1.0 / p - 1.0)
    w = alpha * c ** (p - 1) * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        kk = k - (n - 1)
        if kk >= 0:
            cnt = n - kk
            g[kk:] += w[k] * (1.0 - h[:cnt])
            g[:cnt] -= w[k] * h[kk:]
        else:
            sp = -kk
            cnt = n - sp
            g[:cnt] += w[k] * (1.0 - h[sp:])
            g[sp:] -= w[k] * h[:cnt]
    return g

def _ps_obj(h, p):
    c = _corr(h)
    return float(np.sum(c ** p) ** (1.0 / p))

def _topm_desc(h, n, m_top, max_iters, t0, t_end):
    c5 = _eval(h)
    bh, bc5 = h.copy(), c5
    step = 0.04
    stuck = 0
    for _ in range(max_iters):
        if time.time() - t0 > t_end:
            break
        g = _grad_topm(h, m_top)
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1, 0.0001]:
            h_new = _repair(h - s * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h, c5 = h_new, c5n
                step = min(step * 1.05, 0.4)
                stuck = 0
                if c5 < bc5:
                    bc5, bh = c5, h.copy()
                improved = True
                break
        if not improved:
            step *= 0.5
            stuck += 1
            if stuck > 30:
                break
    return bh, bc5

def _ps_desc(h, n, p, max_iters, t0, t_end):
    c5 = _eval(h)
    bh, bc5 = h.copy(), c5
    step = 0.02
    stuck = 0
    for _ in range(max_iters):
        if time.time() - t0 > t_end:
            break
        g = _ps_grad(h, p)
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1, 0.0001]:
            h_new = _repair(h - s * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h, c5 = h_new, c5n
                step = min(step * 1.03, 0.3)
                stuck = 0
                if c5 < bc5:
                    bc5, bh = c5, h.copy()
                improved = True
                break
        if not improved:
            step *= 0.5
            stuck += 1
            if stuck > 20:
                break
    return bh, bc5

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_ps(h, p, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_ps_obj, h, jac=_ps_grad, args=(p,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return h

def _pairwise(h, n, iters, scale, t0, t_end, rng_seed):
    r = np.random.RandomState(rng_seed)
    c5 = _eval(h)
    best = h.copy()
    for _ in range(iters):
        if time.time() - t0 > t_end:
            break
        i, j = r.randint(n), r.randint(n)
        if i == j:
            continue
        d = scale * r.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
            best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best

def _make_structured(n, rng):
    t = np.arange(n) / n
    p = rng.randint(5)
    if p == 0:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t + rng.rand() * 6)
    elif p == 1:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * 5 * t) * np.cos(2 * np.pi * 2 * t)
    elif p == 2:
        h = 0.5 + 0.35 * np.tanh(3 * np.sin(2 * np.pi * 4 * t))
    elif p == 3:
        h = np.where(np.sin(2 * np.pi * 6 * t) > 0.3, 0.8, 0.2)
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) * rng.randint(2, 8) / n)
    return _repair(np.clip(h + 0.02 * rng.randn(n), 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)

    def _remaining():

    for m in [3, 5, 8, 5]:
        if _remaining() < 80:
            break
        h2, c5 = _topm_desc(bh, n0, m, 5000, t0, time.time() + _remaining() * 0.12)
        if c5 < bc5:
            bh, bc5 = h2, c5

    # Phase 2: Power-sum descent
    for p in [6, 10, 16]:
        if _remaining() < 60:
            break
        h2, c5 = _ps_desc(bh, n0, p, 3000, t0, time.time() + _remaining() * 0.08)
        if c5 < bc5:
            bh, bc5 = h2, c5

    # Phase 3: SLSQP refinement at base res
    if _remaining() > 120:
        h2 = _slsqp_ps(bh, 10, 400)
        c5 = _eval(h2)
        if c5 < bc5:
            bh, bc5 = h2, c5

    # Phase 4: Upsample to 2x and optimize
    if _remaining() > 250:
        n1 = n0 * 2
        h1 = _repair(np.repeat(bh, 2), n1)
        for m in [3, 5, 8]:
            if _remaining() < 100:
                break
            h2, c5 = _topm_desc(h1, n1, m, 3000, t0, time.time() + _remaining() * 0.1)
            if c5 < bc5:
                bh, bc5, bn = h2, c5, n1
        for p in [8, 16]:
            if _remaining() < 80:
                break
            h2, c5 = _ps_desc(bh, bn, p, 2000, t0, time.time() + _remaining() * 0.08)
            if c5 < bc5:
                bh, bc5 = h2, c5
        if _remaining() > 100:
            h2 = _slsqp_ps(bh, 10, 300)
            c5 = _eval(h2)
            if c5 < bc5:
                bh, bc5 = h2, c5

    # Phase 5: Multi-start with diverse initializations
    if _remaining() > 120:
        starts = []
        for _ in range(4):
            starts.append(_repair(bh + 0.03 * rng.randn(bn), bn))
        for _ in range(3):
            starts.append(_make_structured(bn, rng))
        for _ in range(3):

            h2, c5 = _topm_desc(hp, bn, 5, 2000, t0, time.time() + _remaining() * 0.04)
            if c5 < bc5:
                bh, bc5 = h2, c5

    # Phase 6: Fine pairwise search
    if _remaining() > 30:
        for scale in [0.004, 0.001, 0.0003, 0.0001]:
            if _remaining() < 15:
                break
            h2 = _pairwise(bh.copy(), bn, 6000, scale, t0, time.time() + _remaining() * 0.05, seed + 7 * scale * 1e6)
            c5 = _eval(h2)
            if c5 < bc5:
                bh, bc5 = h2, c5

    # Phase 7: Final SLSQP + top-m polish
    if _remaining() > 15:
        h2 = _slsqp_ps(bh, 16, 250)
        c5 = _eval(h2)
        if c5 < bc5:
            bh, bc5 = h2, c5
        h2, c5 = _topm_desc(bh, bn, 3, 2000, t0, time.time() + _remaining() * 0.3)
        if c5 < bc5:
            bh, bc5 = h2, c5

    # Phase 8: Ultra-fine pairwise
    if _remaining() > 5:
        h2 = _pairwise(bh.copy(), bn, 15000, 5e-5, t0, time.time() + _remaining() * 0.5, 99999)
        c5 = _eval(h2)
        if c5 < bc5:
            bh, bc5 = h2, c5

    bh = _repair(bh, bn)
    bc5 = _eval(bh)
    return (bh, bc5, bn)
