def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Bisection projection
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo+hi)*0.5, 0, 1)
        # Fine-tune
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    omh = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            G[k, :n - lag] = omh[lag:]
            G[k, lag:] -= h[:n - lag]
        else:
            m = -lag
            G[k, m:] = omh[:n - m]
            G[k, :n - m] -= h[m:]
    return G

def _solve_lp(h, c, G, max_step):
    n = len(h)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A1 = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b1 = -c.copy()
    A2 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b2 = np.minimum(1.0 - h, max_step).copy()
    A3 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b3 = np.minimum(h, max_step).copy()
    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    if res.status == 0 and res.x is not None:
        return res.x[:n]
    return None

def _incr_pair(h, c, G, rng, n_iter, scale, n):
    c5c = c.max()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = d * (G[:, i] - G[:, j])
        new_max = (c + dc).max()
        if new_max < c5c - 1e-14:
            c += dc
            h[i] = ni
            h[j] = nj
            c5c = new_max
            omh = 1.0 - h
            for k in range(2 * n - 1):
                lag = k - (n - 1)
                vi = 0.0
                if 0 <= i + lag < n:
                    vi += omh[i + lag]
                if 0 <= i - lag < n:
                    vi -= h[i - lag]
                G[k, i] = vi
                vj = 0.0
                if 0 <= j + lag < n:
                    vj += omh[j + lag]
                if 0 <= j - lag < n:
                    vj -= h[j - lag]
                G[k, j] = vj
    return h, c

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    G = _grad_matrix(h)
    return G.T @ w

def _con_fun(x, n):
    return x.sum() - n * 0.5

def _con_jac(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    c = _corr(h)
    G = _grad_matrix(h)

    # Phase 1: LP-based active-set optimization
    stall = 0
    max_step = 0.05
    while time.time() - t0 < budget_s - 250:
        d_star = _solve_lp(h, c, G, max_step)
        if d_star is None or np.linalg.norm(d_star) < 1e-12:
            stall += 1
            if stall > 20:
                break
            if stall % 5 == 0:
                h = _repair(h + 0.002 * rng.randn(n), n)
                c = _corr(h)
                G = _grad_matrix(h)
            continue
        improved = False
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05]:
            h_new = h + alpha * d_star
            if h_new.min() < -1e-9 or h_new.max() > 1.0 + 1e-9:
                continue
            h_new = _repair(h_new, n)
            c_new = _corr(h_new)
            if c_new.max() < c.max() - 1e-13:
                h = h_new
                c = c_new
                G = _grad_matrix(h)
                improved = True
                break
        if improved:
            c5 = c.max() * 2.0 / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
            max_step = min(0.05, max_step * 1.05)
        else:
            stall += 1
            max_step *= 0.7
            if max_step < 0.001:
                break

    # Phase 2: Incremental pairwise (coarse to fine)
    if time.time() - t0 < budget_s - 100:
        h_pg = best_h.copy()
        c_pg = _corr(h_pg)
        G_pg = _grad_matrix(h_pg)
        for scale in [0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0001]:
            if time.time() - t0 > budget_s - 100:
                break
            n_it = 40000 if scale > 0.001 else 60000
            h_pg, c_pg = _incr_pair(h_pg, c_pg, G_pg, rng, n_it, scale, n)
        h_pg = _repair(h_pg, n)
        c5 = _eval(h_pg)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pg.copy()

    # Phase 3: SLSQP with soft-max
    if time.time() - t0 < budget_s - 60:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: _con_fun(x, n),
                 'jac': lambda x: _con_jac(x, n)}]
        for alpha in [200000, 500000, 1000000]:
            if time.time() - t0 > budget_s - 60:
                break
            try:
                res = minimize(lambda x: _smooth_obj(x, alpha), best_h,
                              jac=lambda x: _smooth_grad(x, alpha),
                              method='SLSQP', bounds=bnds, constraints=cons,
                              options={'maxiter': 1500, 'ftol': 1e-19})
                h_sl = _repair(res.x, n)
                c5 = _eval(h_sl)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sl.copy()
            except Exception:
                pass

    # Phase 4: More LP iterations with smaller steps
    if time.time() - t0 < budget_s - 40:
        h_lp2 = best_h.copy()
        c_lp2 = _corr(h_lp2)
        G_lp2 = _grad_matrix(h_lp2)
        for ms in [0.01, 0.005, 0.002, 0.001]:
            if time.time() - t0 > budget_s - 40:
                break
            for _it in range(200):
                if time.time() - t0 > budget_s - 40:
                    break
                d_s = _solve_lp(h_lp2, c_lp2, G_lp2, ms)
                if d_s is None or np.linalg.norm(d_s) < 1e-12:
                    break
                h_t = h_lp2 + d_s
                if h_t.min() < -1e-9 or h_t.max() > 1.0 + 1e-9:
                    break
                h_t = _repair(h_t, n)
                c_t = _corr(h_t)
                if c_t.max() < c_lp2.max() - 1e-14:
                    h_lp2 = h_t
                    c_lp2 = c_t
                    G_lp2 = _grad_matrix(h_lp2)
        c5 = _eval(h_lp2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_lp2.copy()

    # Phase 5: Final ultra-fine incremental
    if time.time() - t0 < budget_s - 5:
        h_f = best_h.copy()
        c_f = _corr(h_f)
        G_f = _grad_matrix(h_f)
        for scale in [5e-5, 2e-5, 1e-5, 5e-6]:
            if time.time() - t0 > budget_s - 5:
                break
            h_f, c_f = _incr_pair(h_f, c_f, G_f, rng, 50000, scale, n)
        h_f = _repair(h_f, n)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
