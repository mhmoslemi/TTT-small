import numpy as np
import time
from itertools import combinations

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def _fix_binary(h, n):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones == target:
        return h
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    if n_ones > target:
        for k in range(n_ones - target):
            h[ones_idx[k]] = 0.0
    else:
        for k in range(target - n_ones):
            h[zeros_idx[k]] = 1.0
    return h

def _binary_ls(h, max_passes, t_limit, rng):
    n = len(h)
    h = _fix_binary(h, n)
    best_c5 = _eval(h)
    for _p in range(max_passes):
        if time.time() > t_limit:
            break
        improved = False
        ones_list = list(np.where(h == 1.0)[0])
        zeros_list = list(np.where(h == 0.0)[0])
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        for i in ones_list:
            for j in zeros_list:
                h[i], h[j] = 0.0, 1.0
                c5 = _eval(h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5

def _periodic_best(n, max_p, t_limit):
    best_h = None
    best_c5 = np.inf
    for p in range(2, max_p + 1, 2):
        if n % p != 0:
            continue
        k = p // 2
        for combo in combinations(range(p), k):
            if time.time() > t_limit:
                return best_h, best_c5
            pat = np.zeros(p)
            pat[list(combo)] = 1.0
            h = np.tile(pat, n // p)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

    nc = len(h)
    h_s = h.copy()
    c5_s = _eval(h_s)
    best = c5_s
    best_h = h_s.copy()
    T = 0.0008
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        T *= 0.9997
        ones = np.where(h_s == 1.0)[0]
        zeros = np.where(h_s == 0.0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h_s[i], h_s[j] = 0.0, 1.0
        c5_new = _eval(h_s)
        delta = c5_new - c5_s
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            c5_s = c5_new
            if c5_new < best:
                best = c5_new
                best_h = h_s.copy()
        else:
            h_s[i], h_s[j] = 1.0, 0.0
    return best_h, best

def _random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _resample_binary(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    vals = np.interp(xn, xo, h_old)
    return _fix_binary(vals, n_new)

def _targeted_search(h, top_k, n_iter, t_limit, rng):
    nc = len(h)
    h_t = h.copy()
    c5_t = _eval(h_t)
    for _ in range(n_iter):
        if time.time() > t_limit:
            break
        ov = np.correlate(h_t, 1.0 - h_t, mode='full')
        top_idx = np.argsort(ov)[-top_k:]
        worst_shift = int(top_idx[-1])
        sh = (nc - 1) - worst_shift
        ones = np.where(h_t == 1.0)[0]
        zeros = np.where(h_t == 0.0)[0]
        if sh >= 0 and sh < nc:
            cand_zeros = zeros[(zeros + sh) < nc]
            if len(cand_zeros) == 0:
                cand_zeros = zeros
        else:
            sp = -sh
            cand_zeros = zeros[(zeros + sp) >= 0]
            if len(cand_zeros) == 0:
                cand_zeros = zeros
        i = int(rng.choice(ones))
        j = int(rng.choice(cand_zeros))
        h_t[i], h_t[j] = 0.0, 1.0
        c5_new = _eval(h_t)
        if c5_new < c5_t - 1e-15:
            c5_t = c5_new
        else:
            h_t[i], h_t[j] = 1.0, 0.0
    return h_t, c5_t
