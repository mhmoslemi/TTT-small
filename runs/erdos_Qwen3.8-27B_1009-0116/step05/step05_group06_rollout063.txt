import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sum_con_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_con_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

    f = _obj(h, alpha)
    v = np.zeros(n)
    step = step0

        v = mu * v - step * g
        best_lam, best_f = 0.0, f
        lam = 1.0
        for _ in range(25):
            h_try = _repair(h - lam * v, n)
            fn = _obj(h_try, alpha)
            if fn < f - 1e-6 * lam * np.dot(v, h_try - h):
                best_lam, best_f = lam, fn
                break
            lam *= 0.5
        if best_lam > 0:
            h = _repair(h - best_lam * v, n)
            f = best_f
            step = min(step * 1.015, step0 * 4.0)
        else:
            step *= 0.95
        if step < 1e-15:
            break
    return h

    n = len(h)
