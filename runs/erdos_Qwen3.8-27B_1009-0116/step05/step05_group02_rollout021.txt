                best_h = h_up.copy()
                best_n = n1
        if tl() > 180:
            for k_top in [1, 3, 5]:
                if tl() < 140:
                    break
                h_up = _topk_descent(best_h, best_n, k_top, time.time() + min(25, tl() - 135), rng)
                c5 = _eval(h_up)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_up.copy()
        if tl() > 100:
            h = best_h.copy()
            c5c = _eval(h)
            for scale in [0.005, 0.001, 3e-4, 1e-4]:
                if tl() < 60:
                    break
                h, c5c = _pairwise(h, c5c, scale, time.time() + min(25, tl() - 55), rng)
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h.copy()

    # Phase 5: Multi-start exploration
    nc = best_n
    if tl() > 200:
        for _r in range(40):
            if tl() < 100:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, nc // 6))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 2:
                f = rng.randint(1, nc // 3 + 1)
                amp = 0.01 * rng.randn()
                hp = best_h + amp * np.sin(2 * np.pi * f * np.arange(nc) / nc)
                hp = _repair(hp, nc)
            else:
                hp = _repair(best_h * rng.uniform(0.9, 1.1) + 0.005 * rng.randn(nc), nc)
            h_new = _nesterov(hp, nc, 50000, time.time() + min(15, tl() - 95))
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Final fine polish
    if tl() > 40:
        h = best_h.copy()
        c5c = _eval(h)
        for scale in [3e-5, 1e-5, 3e-6]:
            if tl() < 15:
                break
            h, c5c = _pairwise(h, c5c, scale, time.time() + min(12, tl() - 12), rng)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
