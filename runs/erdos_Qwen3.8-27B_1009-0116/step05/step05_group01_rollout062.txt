import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _ps_obj(x, p):
    n = len(x)
    c = np.clip(np.correlate(x, 1.0 - x, mode='full') * (2.0 / n), 0, 2)
    return float(np.sum(c ** p) ** (1.0 / p))

def _ps_grad(x, p):
    n = len(x); dx = 2.0 / n
    c = np.clip(np.correlate(x, 1.0 - x, mode='full') * dx, 0, 2)
    w = c ** (p - 1); s = np.sum(c ** p)
    if s < 1e-300: return np.zeros(n)
    f = s ** (1.0 / p); g = np.zeros(n)
    for k in range(n):
        wj = w[n - 1 + k]
        if wj < 1e-25: continue
        g[k:] += wj * (1.0 - x[:n - k])
        g[:n - k] -= wj * x[k:]
    for kk in range(1, n):
        wj = w[n - 1 - kk]
        if wj < 1e-25: continue
        g[:n - kk] += wj * (1.0 - x[kk:])
        g[kk:] -= wj * x[:n - kk]
    return f ** (1.0 - p) * g

def _sc(x): return x.sum() - len(x) * 0.5
def _sj(x): return np.ones(len(x))

def _solve(h, p, mi, ft):
    n = len(h)
    try:
        r = minimize(_ps_obj, h, jac=_ps_grad, args=(p,), method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': _sc, 'jac': _sj}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _local_srch(h, nit, rng, scale0, scale1, use_grad):
    n = len(h); bc = _eval(h)
    ci, cj = 0, 1; gval = np.zeros(n)
    for it in range(nit):
        fr = it / max(nit, 1)
        sc = scale0 * (scale1 / scale0) ** fr
        if use_grad and it % 40 == 0:
            gval = _ps_grad(h, 60); gval -= gval.mean()
            ci, cj = int(np.argmin(gval)), int(np.argmax(gval))
        if use_grad and gval[cj] - gval[ci] > 1e-11 and rng.rand() > 0.15:
            i, j = ci, cj
            d = sc * 1.2 * rng.uniform(0.3, 1.5)
        else:
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j: continue
            d = sc * rng.randn()
        a, b = h[i], h[j]
        ni, nj = a + d, b - d
        if not (0 <= ni <= 1 and 0 <= nj <= 1):
            d *= 0.3; ni, nj = a + d, b - d
            if not (0 <= ni <= 1 and 0 <= nj <= 1): continue
        h[i], h[j] = ni, nj
        v = _eval(h)
        if v >= bc - 1e-15: h[i], h[j] = a, b
        else: bc = v
    return h, bc

def _anneal(h, nit, rng, t_start, t_end):
    n = len(h); cc = _eval(h); bh, bc = h.copy(), cc
    for it in range(nit):
        fr = it / max(nit, 1)
        T = t_start * (t_end / t_start) ** fr
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j: continue
        d = 0.35 * T * rng.randn()
        a, b = h[i], h[j]
        ni, nj = a + d, b - d
        if not (0 <= ni <= 1 and 0 <= nj <= 1): continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn <= cc or rng.rand() < np.exp(-(cn - cc) / max(T * 5e-5, 1e-10)):
            cc = cn
            if cn < bc: bc, bh = cn, h.copy()
        else:
            h[i], h[j] = a, b
    return bh, bc
