import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= w * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w * h[i2 + sp]
        g *= dx
        improved = False
        for step in [0.03, 0.015, 0.007, 0.003, 0.001, 0.0003, 0.0001]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new; c5c = c5n; improved = True; break
        if not improved:
            break
    return h, c5c

def _coord_desc(h, n_sweeps, n_grid=31, t_end=None, t0=None):
    n = len(h)
    nv = 2 * n - 1
    dx = 2.0 / n
    k0 = n - 1
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = c.max() * dx
    grid = np.linspace(0.0, 1.0, n_grid)
    k_arr = np.arange(nv)
    for sweep in range(n_sweeps):
        if t0 is not None and time.time() - t0 > t_end:
            break
        any_imp = False
        for i in range(n):
            a = h[i]
            idx1 = i + k_arr - (n - 1)
            idx2 = i - k_arr + (n - 1)
            I1 = (idx1 >= 0) & (idx1 < n)
            I2 = (idx2 >= 0) & (idx2 < n) & (idx2 != i)
            s = np.zeros(nv)
            if I1.any():
                s[I1] += (1.0 - h[idx1[I1]])
            if I2.any():
                s[I2] -= h[idx2[I2]]
            best_b = a
            best_val = c.max()
            for b in grid:
                if abs(b - a) < 1e-10:
                    continue
                c_new = c + (b - a) * s
                c_new[k0] = c[k0] + (b - a) * (1.0 - a - b)
                val = c_new.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_b = b
            if best_b != a:
                db = best_b - a
                c += db * s
                c[k0] -= db * s[k0]
                c[k0] += db * (1.0 - a - best_b)
                h[i] = best_b
                if best_val * dx < best_c5 - 1e-15:
                    best_c5 = best_val * dx
                    any_imp = True
        if not any_imp:
            break
    return h, best_c5

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi
