        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        h = np.clip(h, 0.0, 1.0)

            best_c5, best_h, best_n = c5, h_pol.copy(), len(h_pol)

    # Phase 6: Multi-start exploration
    if _el() < budget_s - 150:
        nc = len(best_h)
        for _r in range(8):
            if _el() > budget_s - 150:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 4)):
                    bs = rng.randint(3, max(4, nc // 8))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            else:
                hp = _repair(np.roll(best_h, rng.randint(1, nc // 4)), nc)
            h_ms = _slsqp_pm(hp, 200, 800, 1e-18)
            c_ms = _corr(h_ms)
            h_ms, c_ms, _ = _pair_sa(h_ms, c_ms, rng, 20000, 0.003, nc, 0.001)
            h_ms = _repair(h_ms, nc)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_ms, nc

    # Phase 7: Final ultra-fine greedy pairwise
    if _el() < budget_s - 8:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        n_fin = len(h_fin)
        for scale in [5e-5, 2e-5, 1e-5, 5e-6]:
            if _el() > budget_s - 8:
                break
            h_fin, c_fin, _ = _pair_sa(h_fin, c_fin, rng, 60000, scale, n_fin, 0.0)
        h_fin = _repair(h_fin, n_fin)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin, n_fin

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
