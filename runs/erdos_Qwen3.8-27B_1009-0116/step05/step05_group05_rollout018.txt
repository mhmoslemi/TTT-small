import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):

    c = _corr(h)
    m = c.max()

    dx = 2.0 / n
    c = _corr(h) * dx
    m = c.max()
    return m + np.sum(np.maximum(0, c - m - m * 0.01)) / beta

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def f(x):
        return _smooth_obj(x, alpha)
    def jf(x):
        return _smooth_grad(x, alpha)
    def cf(x):
        return x.sum() - n * 0.5
    def jcf(x):
        return np.ones(n)
    try:
        res = minimize(f, h, jac=jf, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': jcf}],
                       options={'maxiter': maxiter, 'ftol': ftol})

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

    if _el() < budget_s - 280:
        for _r in range(40):
            if _el() > budget_s - 220:
                break
            r = rng.rand()
            if r < 0.25:
                hp = _repair(best_h + 0.008 * rng.randn(n_cur), n_cur)
            elif r < 0.5:
                hp = _repair(np.clip(best_h + 0.025 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur)
            elif r < 0.75:
                hp = best_h.copy()
                t = np.arange(n_cur)
                for _f in range(2):
                    f = rng.randint(1, n_cur // 3 + 1)
                    hp += 0.004 * rng.randn() * np.sin(2 * np.pi * f * t / n_cur)
                hp = _repair(hp, n_cur)
            else:
                hp = best_h.copy()
                bs = rng.randint(4, max(5, n_cur // 10))
                st = rng.randint(0, max(1, n_cur - bs))
                hp[st:st + bs] += 0.02 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            h_new = _slsqp(hp, 500000, 800, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
                best_n = n_cur

    # Phase F: Refinement cycle
    if _el() < budget_s - 150:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 3, 0.9)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
    if _el() < budget_s - 100:
        h_new, c5 = _structured_pair(best_h.copy(), 6000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
    if _el() < budget_s - 70:
        h_new, c5 = _active_descent(best_h.copy(), 3000, 5, 0.8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase G: Multi-scale pairwise
    if _el() < budget_s - 50:
        for scale in [0.004, 0.001, 0.0003, 1e-4, 3e-5, 1e-5]:
            if _el() > budget_s - 25:
                break
            h_new, c5 = _pairwise(best_h.copy(), 4000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase H: Final SLSQP polish
    if _el() < budget_s - 15:
        h_new = _slsqp(best_h, 3000000, 6000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase I: Ultra-fine active set
    if _el() < budget_s - 3:
        h_new, c5 = _active_descent(best_h.copy(), 8000, 2, 0.92)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
