    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if abs(h.sum() - target) < 1e-14:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _lag_grad(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        idx = np.arange(n - sh)
        g[idx] += (1.0 - h[idx + sh])
        if sh > 0:
            idx2 = np.arange(sh, n)
            g[idx2] -= h[idx2 - sh]
    else:
        sp = -sh
        idx = np.arange(sp, n)
        g[idx] += (1.0 - h[idx - sp])
        idx2 = np.arange(n - sp)
        g[idx2] -= h[idx2 + sp]
    return g * dx

def _multi_grad(h, n, frac):
    c = _corr(h)
    cmax = c.max()
    thr = cmax - frac
    hot = np.where(c > thr)[0]
    if len(hot) == 0:
        hot = np.array([c.argmax()])
    w = c[hot] - thr
    w = w / max(w.sum(), 1e-15)
    g = np.zeros(n)
    for i in range(len(hot)):
        g += w[i] * _lag_grad(h, n, hot[i])
    return g

def _proj_step(h, n, g, a0=0.05):
    c5c = _eval(h)
    a = a0
    for _ in range(15):
        hnew = _project(h - a * g, n)
        c5n = _eval(hnew)
        if c5n < c5c - 1e-17:
            return hnew, c5n, True
        a *= 0.5
    return h, c5c, False

def _lp_solver(h, n):
    w = 1.0 - h
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        if sh >= 0:
            A[k, :n - sh] = w[sh:]
        else:
            sp = -sh
            A[k, sp:] = w[:n - sp]
    vh = cp.Variable(n)
    vt = cp.Variable()
    cons = [vh >= 0, vh <= 1, cp.sum(vh) == n / 2.0, A @ vh <= vt]
    prob = cp.Problem(cp.Minimize(vt * (2.0 / n)), cons)
    try:
        prob.solve(solver=cp.ECOS, max_iter=20000, reltol=1e-10, abstol=1e-12)
    except Exception:
        try:
            prob.solve()
        except Exception:
            return h
    if vh.value is not None and np.all(np.isfinite(vh.value)):
        return np.clip(vh.value, 0.0, 1.0)
    return h

def _slsqp_lse(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def jac(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            g += w[k] * _lag_grad(x, n, k)
        return g
    try:
        res = minimize(obj, h, jac=jac,
                       method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _project(res.x, n)
    except Exception:
        return _project(h, n)

    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-14:
            g += w[k] * _lag_grad(x, n, k)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _project(res.x, n)
    except Exception:
        return _project(h.copy(), n)
