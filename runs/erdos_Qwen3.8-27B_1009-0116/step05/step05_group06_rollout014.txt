            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp_inner(x, n, alpha):
    return _smooth_obj(x, alpha)

def _slsqp_jac(x, n, alpha):
    return _smooth_grad(x, alpha)

def _con_fun(x, n):
    return x.sum() - n * 0.5

def _con_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_inner, h, jac=_slsqp_jac, args=(n, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / max(topk, 1)

def _topk_descent(h, nsteps, step0, topk_init, t_end):
    n = len(h)
    c5 = _eval(h)
    step = step0
    for it in range(nsteps):
        if time.time() > t_end:
            break
        tk = max(2, int(topk_init * (1.0 - 0.5 * it / nsteps)))
        g = _topk_grad(h, tk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        st = step
        for _ in range(28):
            h_try = _proj_sum(h - st * g, n * 0.5)
            c5n = _eval(h_try)
            if c5n < c5 - 1e-15:
                h, c5 = h_try, c5n
                improved = True
                break
            st *= 0.6
        if not improved:
            step *= 0.8
            if step < 1e-12:
                break
        else:
            step = min(step * 1.03, step0 * 1.5)
    return h, c5

def _binary_push(h, nsteps, t_end):
    n = len(h)
    c5 = _eval(h)
    strength = 0.008
    for _ in range(nsteps):
        if time.time() > t_end:
            break
        mask = (h > 0.08) & (h < 0.92)
        if not mask.any():
            break
        h_new = h.copy()
        h_new[mask] = np.where(h_new[mask] > 0.5,
                               np.minimum(h_new[mask] + strength, 1.0),
                               np.maximum(h_new[mask] - strength, 0.0))
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5:
            h, c5 = h_new, c5n
        else:
            strength *= 0.6
            if strength < 1e-7:
                break
    return h, c5

def _upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP at base resolution with increasing alpha
    for alpha in [3000, 20000, 100000, 400000]:
        if _el() > 150:
            break
        h_new = _slsqp(best_h, alpha, 1200, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Top-k gradient descent
    for topk in [4, 8, 15]:
        if _el() > 250:
            break
        h_new, c5 = _topk_descent(best_h, 1500, 0.004, topk, t0 + min(_el() + 60, budget_s - 500))
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 3: Binary push
    if _el() < budget_s - 450:
        h_new, c5 = _binary_push(best_h, 400, t0 + budget_s - 450)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 4: Upsample to 2x and optimize
    if _el() < budget_s - 380:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [8000, 40000, 150000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp(h_hi, alpha, 800, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 200:
            h_hi, c5 = _topk_descent(best_h, 1000, 0.002, 8, t0 + budget_s - 200)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase 5: Multi-start local optimization
    nc = best_n
    if _el() < budget_s - 220:
        for _r in range(35):
            if _el() > budget_s - 220:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.025 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 2:
                shift = rng.randint(1, max(2, nc // 10))
                hp = _repair(np.roll(best_h, shift), nc)
            elif strat == 3:
                hp = best_h.copy()
                bs = rng.randint(4, max(5, nc // 7))
                st = rng.randint(0, nc - bs)
                hp[st:st+bs] = np.clip(hp[st:st+bs] + 0.05*rng.randn(bs), 0, 1)
                hp = _repair(hp, nc)
            elif strat == 4:
                hp = best_h.copy()
                ta = np.arange(nc)
                for _ in range(3):
                    f = rng.randint(2, nc // 3 + 1)
                    hp += 0.012 * rng.randn() * np.sin(2*np.pi*f*ta/nc + rng.rand()*2*np.pi)
                hp = _repair(hp, nc)
            else:
                hp = best_h.copy()
                idx = rng.choice(nc, size=nc//5, replace=False)
                hp[idx] = rng.rand(len(idx))
                hp = _repair(hp, nc)
            alpha_pick = [10000, 40000, 150000][rng.randint(3)]
            h_new = _slsqp(hp, alpha_pick, 500, 1e-17)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase 6: Top-k descent on best
    if _el() < budget_s - 120:
        h_new, c5 = _topk_descent(best_h, 2000, 0.003, 6, t0 + budget_s - 120)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), best_n

    # Phase 7: Fine coordinate descent
    h = best_h.copy()
    nc = len(h)
    c5c = _eval(h)
    t_end = t0 + budget_s - 20
    scale = 3e-5
    for sc in [3e-5, 1e-5, 4e-6, 2e-6]:
        scale = sc
        for _it in range(30000):
            if time.time() > t_end:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h, best_n = c5c, h.copy(), nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        if time.time() > t_end:
            break

    # Phase 8: Final SLSQP polish
    if _el() < budget_s - 10:
        h_fin = _slsqp(best_h, 500000, 1000, 1e-19)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin.copy(), len(h_fin)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
