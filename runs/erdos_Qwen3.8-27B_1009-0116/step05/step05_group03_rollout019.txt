import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp_soft(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    
    def eq(x):
        return x.sum() - tgt
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _simulated_anneal(h, dx, rng, t_start, t_end, t0_temp=0.05, cool_rate=0.999):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    curr_c = best
    temp = t0_temp
    
    while time.time() < t_end:
        # Choose move type
        r = rng.random()
        h_new = h.copy()
        
        if r < 0.4:
            # Nudge: move mass from i to j
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = temp * 0.3 * (2.0 * rng.random() - 1.0)
            h_new[i] -= d
            h_new[j] += d
        elif r < 0.7:
            # Targeted: move mass along worst lag
            c = _corr(h)
            top_k = np.argmax(c)
            s = abs(int(top_k - (n - 1)))
            if s == 0:
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = temp * 0.2 * (2.0 * rng.random() - 1.0)
                h_new[i] -= d
                h_new[j] += d
            else:
                m = n - s
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = temp * 0.2 * (2.0 * rng.random() - 1.0)
                h_new[i1] -= d
                h_new[i2] += d
        else:
            # Block perturbation
            blen = rng.randint(3, max(4, n // 10))
            start = rng.randint(n - blen)
            block = h_new[start:start+blen].copy()
            h_new[start:start+blen] = h_new[start+blen:start+2*blen] if start + 2*blen <= n else np.roll(block, rng.randint(1, blen))
        
        h_new = np.clip(h_new, 0, 1)
        h_new = _repair(h_new, tgt)
        
        new_c = _c5(h_new, dx)
        delta = new_c - curr_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            h = h_new
            curr_c = new_c
            if curr_c < best:
                best = curr_c
        else:
            pass  # reject
        
        temp = max(temp * cool_rate, 1e-5)
    
    return h, best

def _coord_descent(h, dx, rng, t_end, n_iter_max=50000):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    
    for it in range(n_iter_max):
        if time.time() > t_end:
            break
        
        c = _corr(h)
        # Focus on top 3 lags
        top3 = np.argsort(c)[::-1][:3]
        
        improved = False
        for ki in top3:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            # For this lag, overlap = sum h[a]*(1-h[a+s]) for a in range(m)
            # Gradient w.r.t. h[a]: (1-h[a+s]) for the first term, -h[a-s] or h[a+s] for the other
            # To reduce this lag, increase h[a+s] where h[a] is high, or decrease h[a] where h[a+s] is low
            
            # Try adjusting the worst individual contribution
            products = h[:m] * (1.0 - h[s:s+m])
            worst_idx = np.argmax(products)
            
            # Move mass: reduce h[worst_idx] or increase h[worst_idx + s]
            a = worst_idx
            d = rng.uniform(0.001, 0.01)
            
            # Option 1: reduce h[a], increase h[a+s]
            if h[a] > d and h[a+s] < 1.0:
                h[a] -= d
                h[a+s] += d
                h = _repair(h, tgt)
                nc = _c5(h, dx)
                if nc < best:
                    best = nc
                    improved = True
                else:
                    h[a+s] -= d
                    h[a] += d
                    h = _repair(h, tgt)
            
            # Option 2: increase h[a], decrease h[a+s] (if h[a+s] is contributing)
            # Actually for h[a]*(1-h[a+s]), we want to either decrease h[a] or increase h[a+s]
            # h[a+s]*(1-h[a]) also contributes if we consider the other direction
            
            if not improved:
                # Try random pair within this lag structure
                a = rng.randint(m)
                for sign in [1, -1]:
                    d = rng.uniform(0.002, 0.015) * sign
                    h[a] += d
                    h[a+s] -= d
                    if 0 <= h[a] <= 1 and 0 <= h[a+s] <= 1:

                        nc = _c5(h, dx)
                        if nc < best:

                            improved = True
                            break
                        else:
                            h[a+s] += d

                            h = _repair(h, tgt)
        
        if not improved and it % 10 == 0:

                h[i] += d
                h[j] -= d
                h = _repair(np.clip(h, 0, 1), tgt)

    best_h, best_c = h, _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc, c = _slsqp_soft(best_h, dx, a, maxiter)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h0)
    dx = 2.0 / n
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # P1: SLSQP warm (45s)
    h, c = _multi_slsqp(best_h, dx, t0 + 45, [400, 2000, 8000], 250)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P2: Subgradient top-12 (110s)
    h, c = _subgrad(best_h, dx, t0 + 155, K=12, lr_init=0.015)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P3: Simulated annealing (230s)
    h, c = _sa(best_h, dx, rng, t0 + 385, t_init=0.04, cool=0.9992)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P4: SLSQP after SA (70s)
    h, c = _multi_slsqp(best_h, dx, t0 + 455, [1500, 6000, 25000], 250)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P5: Lag coordinate descent (130s)
    h, c = _lag_cd(best_h, dx, rng, t0 + 585)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P6: Second SA, lower temp (130s)
    h, c = _sa(best_h, dx, rng, t0 + 715, t_init=0.018, cool=0.9996)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P7: Multi-start SLSQP (130s)
    for r in range(30):
        if time.time() > t0 + 845:
            break
        hp = best_h.copy()
        mode = r % 3
        if mode == 0:
            idx = rng.choice(n, max(5, n // 5), replace=False)
            hp[idx] += 0.05 * (2.0 * rng.random(len(idx)) - 1.0)
        elif mode == 1:
            hp[:] = 1.0 - hp
        else:
            x = np.linspace(0, 2, n)
            hp += 0.03 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
        hp = _repair(np.clip(hp, 0, 1), n / 2.0)
        hc, c = _slsqp_soft(hp, dx, 6000, 200)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # P8: Subgradient + CD polish (90s)
    h, c = _subgrad(best_h, dx, t0 + 935, K=15, lr_init=0.008)
    if c < best_c5:
        best_h, best_c5 = h, c
    h, c = _lag_cd(best_h, dx, rng, t0 + 960)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P9: Final SLSQP + micro CD (35s)
    if time.time() < t0 + budget_s - 18:
        hc, c = _slsqp_soft(best_h, dx, 50000, 150)
        if c < best_c5:
            best_h, best_c5 = hc, c
    if time.time() < t0 + budget_s - 3:
        h, c = _lag_cd(best_h, dx, rng, t0 + budget_s - 2)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = _repair(best_h, n / 2.0)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
