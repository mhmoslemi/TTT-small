import numpy as np
import time as _time

_T = 0.0
_B = 1100.0

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode='full') * 2.0/n

def _proj(h, target):
    h = h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _lg(p, h):
    n = len(h)
    g = np.zeros(n)
    if p == 0:
        return 1.0 - 2.0*h
    g[p:] += 1.0 - h[:n-p]
    g[:n-p] -= h[p:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n*0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s-target) > 1e-10:
        h *= target/s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0-h) if diff>0 else np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-12: break
            d = min(abs(diff), (1.0-h[i]) if diff>0 else h[i])
            h[i] += d if diff>0 else -d
            diff -= d if diff>0 else -d
    return h

def _grad_max(h, n, tol_frac=0.003):
    """Subgradient of max overlap: avg gradient over near-max lags"""
    c = _corrs(h)
    m = c.max()
    active = np.where(c >= m*(1.0-tol_frac))[0]
    if len(active) == 0:
        active = np.array([c.argmax()])
    g = np.zeros(n)
    for k in active:
        g += _lg(abs(k-(n-1)), h)
    g /= len(active)
    g *= 2.0/n
    return g, c, m

def _smooth_grad(h, n, alpha):
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s-s.max()); w /= w.sum()
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        g += w[k]*_lg(abs(k-(n-1)), h)
    g *= 2.0/n
    return g

def _descend(h, n, target, t_limit, alpha=20000, max_it=80000):
    f = _eval(h)
    step = 0.006
    for it in range(max_it):
        if _time.time()-_T > t_limit: break
        g = _smooth_grad(h, n, alpha)
        found = False
        for s in [step, step*0.4, step*0.15]:
            h2 = _proj(h-s*g, target)
            f2 = _eval(h2)
            if f2 < f-1e-13:
                h = h2; f = f2
                step = min(step*1.035, 0.018)
                found = True
                break
        if not found:
            step *= 0.6
            if step < 1e-8: break
    return h, f

def _subgrad_descend(h, n, target, t_limit, max_it=100000):
    f = _eval(h)
    step = 0.004
    for it in range(max_it):
        if _time.time()-_T > t_limit: break
        g, c, m = _grad_max(h, n, 0.002)
        found = False
        for s in [step, step*0.5, step*0.2, step*0.08]:
            h2 = _proj(h-s*g, target)
            f2 = _eval(h2)
            if f2 < f-1e-14:
                h = h2; f = f2
                step = min(step*1.03, 0.012)
                found = True
                break
        if not found:
            step *= 0.55
            if step < 1e-9: break
    return h, f

def _binary_push(h, n, target, t_limit):
    best = h.copy()
    bf = _eval(best)
    rng = np.random.RandomState(777)
    for r in range(500):
        if _time.time()-_T > t_limit: break
        improved = False
        for gamma in [1.3, 1.7, 2.2, 3.0]:
            h2 = 0.5 + (best-0.5)*gamma
            h2 = _proj(h2, target)
            f2 = _eval(h2)
            if f2 < bf - 1e-13:
                best = h2; bf = f2; improved = True
                break
        if not improved:
            for _tr in range(30):

                h2 = best.copy()
                if h2[i] > 0.5:

                else:
                    h2[i] = max(0.0, h2[i] - 0.15*h2[i])
                h2 = _proj(h2, target)
                f2 = _eval(h2)
                if f2 < bf - 1e-14:
                    best = h2; bf = f2; improved = True
                    break

    return best, bf

def _pairwise(h, n, target, t_limit, scale, n_iter, seed_v):

    rng = np.random.RandomState(seed_v)

            if f2 < bf - 1e-15:
