def _grad_descent(h, n, n_iters, rng, t_end):
    c = _corr(h)
    dx = 2.0 / n
    g_arr = np.zeros(n)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        idx = np.argsort(c)[::-1][:5]
        diffs = c[idx] - c[idx[0]]
        w = np.exp(diffs / (0.05 * c[idx[0]] + 1e-10))
        w /= w.sum()
        g = np.zeros(n)
        for wk, k in zip(w, idx):
            m = k - (n - 1)
            p = np.arange(n)
            t1 = np.zeros(n)
            mk1 = (p + m >= 0) & (p + m < n)
            t1[mk1] = (1.0 - h[p[mk1] + m])
            t2 = np.zeros(n)
            mk2 = (p - m >= 0) & (p - m < n)
            t2[mk2] = h[p[mk2] - m]
            g += wk * (t1 - t2)
        g *= dx
        lr = 0.005 * (1.0 + 0.5 * rng.rand())
        h_new = _proj(h - lr * g, n)
        c5n = _eval(h_new)
        c5c = float(c.max() * dx)
        if c5n < c5c:
            h = h_new
            c = _corr(h)
        else:
            j = rng.randint(n)
            d = 0.005 * rng.randn()
            if 0 < h[j] + d < 1:
                h[j] += d
                h = _proj(h, n)
                c = _corr(h)
    return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _proj(np.array(initial_h_values, dtype=float), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Intensive local search (coarse to fine)
    t1 = t0 + budget_s * 0.30
    best_h, best_c5 = _local_search(best_h, n0, t1, rng, s0=0.04, s1=0.001, max_restarts=3)

    # Phase 2: SLSQP polish
    if time.time() < t0 + budget_s * 0.40:
        for alpha in [2000, 10000, 50000]:
            if time.time() > t0 + budget_s * 0.40:
                break
            hn = _slsqp(best_h, alpha, 2000, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn

    # Phase 3: Finer local search
    t3 = t0 + budget_s * 0.58
    best_h, best_c5 = _local_search(best_h, n0, t3, rng, s0=0.008, s1=0.00005, max_restarts=5)

    # Phase 4: Gradient descent rounds
    if time.time() < t0 + budget_s * 0.68:
        hn, c5n = _grad_descent(best_h, n0, 50000, rng, t0 + budget_s * 0.68)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = hn
        best_h, best_c5 = _local_search(best_h, n0, t0 + budget_s * 0.72, rng, s0=0.005, s1=0.00003, max_restarts=3)

    # Phase 5: Try 2x resolution
    if time.time() < t0 + budget_s * 0.78:
        n1 = n0 * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _proj(h_hi, n1)
        best_hi, c5_hi = _local_search(h_hi, n1, t0 + budget_s * 0.87, rng, s0=0.015, s1=0.0001, max_restarts=2)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = best_hi

    # Phase 6: SLSQP on best
    if time.time() < t0 + budget_s * 0.92:
        hn = _slsqp(best_h, 100000, 3000, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn

    # Phase 7: Final ultra-fine local search
    t7 = t0 + budget_s - 3
    best_h, best_c5 = _local_search(best_h, len(best_h), t7, rng, s0=0.001, s1=0.000002, max_restarts=2)

    best_h = _proj(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
