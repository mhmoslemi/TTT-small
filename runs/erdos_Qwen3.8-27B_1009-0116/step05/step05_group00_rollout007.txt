import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _objective(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s2 = s - s.max()
    return m + np.log(float(np.sum(np.exp(s2)))) / alpha

def _gradient(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_optimize(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_objective, h, jac=_gradient, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_sa(h, n, max_rounds, rng, t_end, block_sizes=(1,2,3,5,7)):
    h = np.clip(np.round(h), 0, 1).astype(float)
    h = _repair(h, n)
    c5c = _eval_c5(h)
    T = 0.5
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        T *= 0.995
        improved = False
        for _ in range(300):
            move = rng.choice(len(block_sizes))
            bs = block_sizes[move]
            if bs == 1:
                i = rng.randint(n)
                h[i] = 1.0 - h[i]
            else:
                i = rng.randint(n - bs)
                if h[i:i+bs].sum() > bs / 2:
                    h[i:i+bs] = 1.0
                else:
                    h[i:i+bs] = 0.0
            h2 = _repair(h.copy(), n)
            c2 = _eval_c5(h2)
            if c2 < c5c:
                h = h2
                c5c = c2
                improved = True
            elif rng.rand() < np.exp(-(c2 - c5c) / max(T, 1e-6)):
                h = h2
                c5c = c2
                improved = True
            else:
                h[i:i+bs] = 1.0 - h[i:i+bs] if bs > 1 else 1.0 - h[i]
                h = _repair(h, n)
        if not improved and T < 0.01:
            break
    return _repair(h, n), c5c

def _swap_search(h, n, max_rounds, rng, t_end):
    h = np.clip(np.round(h), 0, 1).astype(float)
    h = _repair(h, n)
    c5c = _eval_c5(h)
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(500):
            a = rng.randint(n)
            b = rng.randint(n)
            if a == b or h[a] == h[b]:
                continue
            h[a], h[b] = h[b], h[a]
            c2 = _eval_c5(h)
            if c2 < c5c - 1e-15:
                c5c = c2
                improved = True
            else:
                h[a], h[b] = h[b], h[a]
        if not improved:
            break
    return h, c5c

def _fine_cd(h, n, rng, t_end, scales=(0.005, 0.002, 0.0008, 0.0003, 0.0001)):
    h = h.copy()
    c5c = _eval_c5(h)
    for scale in scales:
        if time.time() > t_end:
            break
        n_iter = 8000 if scale > 0.001 else 12000
        for _it in range(n_iter):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval_c5(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _arithmetic_start(n, step):
    h = np.zeros(n)
    for i in range(n // 2):
        pos = (i * step) % n
        h[pos] = 1.0
    return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
