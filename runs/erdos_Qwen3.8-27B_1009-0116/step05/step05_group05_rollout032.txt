def _spair(h, rounds, rng):
    n, dx = len(h), 2.0 / len(h)
    c5c = _eval(h)
    for _ in range(rounds):
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi: continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = cj = 0.0
            if sh >= 0:
                if i <= n - 1 - sh: ci += 1.0 - h[i + sh]
                if 0 <= i - sh and i - sh < n - sh: ci -= h[i - sh]
                if j <= n - 1 - sh: cj += 1.0 - h[j + sh]
                if 0 <= j - sh and j - sh < n - sh: cj -= h[j - sh]
            else:
                sp = -sh
                if sp <= i: ci += 1.0 - h[i - sp]
                if i + sp < n - sp: ci -= h[i + sp]
                if sp <= j: cj += 1.0 - h[j - sp]
                if j + sp < n - sp: cj -= h[j + sp]
            a[k] = ci - cj
        bd, bv = 0.0, np.inf
        for d in np.linspace(dlo, dhi, 17):
            v = np.max(c0 + dx * d * a)
            if v < bv:
                bv, bd = v, d
        if abs(bd) > 1e-10:
            h[i] += bd; h[j] -= bd
            h = np.clip(h, 0, 1)
            cn = _eval(h)
            if cn < c5c - 1e-16:
                c5c = cn
            else:
                h[i] -= bd; h[j] += bd
    return h, c5c
