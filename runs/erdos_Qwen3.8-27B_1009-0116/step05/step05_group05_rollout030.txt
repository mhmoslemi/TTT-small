import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pair_gradient(h, i, j):
    """Compute d_k = dc[k]/dh[i] - dc[k]/dh[j] for all k."""
    n = len(h)
    d = np.zeros(2 * n - 1)
    for idx in [i, j]:
        sign = 1.0 if idx == i else -1.0
        # s = 0: dc[n-1]/dh[idx] = 1 - 2*h[idx]
        d[n-1] += sign * (1.0 - 2.0 * h[idx])
        # s > 0: dc[n-1+s]/dh[idx] = (1-h[idx-s])*[idx>=s] - h[idx+s]*[idx<=n-1-s]
        for s in range(1, n):
            val = 0.0
            if idx >= s:
                val += (1.0 - h[idx - s])
            if idx <= n - 1 - s:
                val -= h[idx + s]
            if val != 0.0:
                d[n-1+s] += sign * val
        # s < 0 (i.e., s' > 0): dc[n-1-s']/dh[idx] = (1-h[idx+s'])*[idx<=n-1-s'] - h[idx-s']*[idx>=s']
        for sp in range(1, n):
            val = 0.0
            if idx <= n - 1 - sp:
                val += (1.0 - h[idx + sp])
            if idx >= sp:
                val -= h[idx - sp]
            if val != 0.0:
                d[n-1-sp] += sign * val
    return d
