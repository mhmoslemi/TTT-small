def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _thue_morse(n):
    """Generate Thue-Morse binary sequence of length n"""
    tm = [0]
    for i in range(n - 1):
        tm.append(tm[-1] ^ 1)
    # Actually, Thue-Morse is: t(n) = popcount(n) mod 2
    tm = [bin(i).count('1') % 2 for i in range(n)]
    return np.array(tm, dtype=float)

def _random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def _logsumexp_obj(x, alpha, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = s.max()
    return (m + np.log(float(np.sum(np.exp(s - smax))) + np.exp(-smax)) / alpha) * dx

def _logsumexp_grad(x, alpha, dx):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k

        if sh >= 0:

            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:

            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g * dx

    dx = 2.0 / n
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    # Use penalty for sum constraint

        pen = 100.0 * (x.sum() - target) ** 2
        return base + pen
    def grad_pen(x):

        pen = 200.0 * (x.sum() - target)
        return base + pen * np.ones(n)
    try:
        res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-12,
                                             'ftol': 1e-15, 'maxfun': maxiter * 5})

def _lbfgs_opt(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    
    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        smax = s.max()
        lse = (m + np.log(np.sum(np.exp(s - smax)) + np.exp(-smax)) / alpha) * dx
        pen = 50.0 * (x.sum() - target) ** 2
        return lse + pen
    
    def g(x):
        gf = _grad_lse(x, alpha)
        pen = 100.0 * (x.sum() - target)
        return gf + pen
    
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-10,
                                             'ftol': 1e-14, 'maxfun': maxiter*3})
        return _repair(res.x, n)
    except Exception:
        return h

            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Progressive L-BFGS-B at original resolution
    for alpha in [3000, 10000, 50000, 200000, 1000000]:
        if time.time() - t0 > 180:
            break
        h_new = _lbfgs_opt(best_h, alpha, 600)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Chase-the-max direct descent
    if time.time() - t0 < budget_s - 400:
        h2 = best_h.copy()
        for _ in range(300):
            if time.time() - t0 > budget_s - 400:
                break
            h2, c5 = _chase(h2, 15)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
            else:
                if _ > 50:
                    break

    # Phase 3: Top-k shift descent (target worst few shifts)
    if time.time() - t0 < budget_s - 250:
        for kt in [1, 2, 3, 5]:
            if time.time() - t0 > budget_s - 250:
                break
            h3, c5 = _topk_descent(best_h.copy(), kt, 150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()

    # Phase 4: Binary snap + local search + re-optimize
    if time.time() - t0 < budget_s - 200:
        h_bin = (best_h > 0.4).copy()
        h_bin[h_bin < 0.5] = 0.0
        h_bin[h_bin >= 0.5] = 1.0
        h_bin = _repair(h_bin, best_n)
        h_bin, c5b = _binary_ls(h_bin, 60, rng)
        c5b = _eval(h_bin)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()
        h_re = _lbfgs_opt(h_bin, 200000, 400)
        c5r = _eval(h_re)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_re.copy()
            best_n = len(h_re)

    # Phase 5: Higher resolution optimization
    if time.time() - t0 < budget_s - 200:
        for n_hi in [500, 800]:
            if time.time() - t0 > budget_s - 200:
                break
            h_hi = _resample(best_h, n_hi)
            h_hi = _lbfgs_opt(h_hi, 30000, 300)
            h_hi = _chase(h_hi, 50)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n_hi

    # Phase 6: Multi-start with structured inits
    if time.time() - t0 < budget_s - 150:
        n = best_n
        candidates = []
        tm = _thue_morse(n, rng)
        candidates.append(tm)
        for _ in range(4):
            h_r = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h_r[ones] = 1.0
            candidates.append(h_r)
        for hc in candidates:
            if time.time() - t0 > budget_s - 150:
                break
            hc = _repair(hc, n)
            ho = _lbfgs_opt(hc, 50000, 400)
            ho, c5 = _chase(ho, 30)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ho.copy()

    # Phase 7: Final high-alpha L-BFGS-B + chase
    if time.time() - t0 < budget_s - 30:
        h_f = _lbfgs_opt(best_h, 5000000, 500)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f.copy()
        h_f2, c5 = _chase(best_h.copy(), 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f2.copy()

    # Phase 8: Ultra-fine pairwise coordinate descent
    if time.time() - t0 < budget_s - 5:
        h8 = best_h.copy()
        nc = len(h8)
        c5c = _eval(h8)
        for scale in [0.002, 0.0005, 0.0001]:
            if time.time() - t0 > budget_s - 5:
                break
            for _it in range(3000):
                if time.time() - t0 > budget_s - 5:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h8.copy()
                else:
                    h8[i], h8[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
