```python
def _subgrad_max_cross(h, tol=1e-4):
    """Compute subgradient of max_k c_cross[k] where c_cross = correlate(h, 1-h)."""
    n = len(h)
    c_cross = np.correlate(h, 1.0 - h, mode='full')
    m = c_cross.max()
    # Active set: indices where c_cross is close to max
    active = np.where(c_cross >= m - tol * m)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_cross)])
    
    # For each active index, compute gradient
    g = np.zeros(n)
    w = c_cross[active] - (m - tol * m)  # weights (unnormalized)
    w = np.maximum(w, 0)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.ones(len(active)) / len(active)
    else:
        w /= wsum
    
    for wk, k in zip(w, active):
        sh = (n - 1) - k
        if sh >= 0:
            # c_cross[k] = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])
            # grad: (1-h[i+sh]) for i in [0, n-1-sh], -h[i-sh] for i in [sh, n-1]
            i1 = np.arange(0, n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            # c_cross[k] = sum_{i=0}^{n-1-sp} h[i+sp]*(1-h[i])
            # grad: h[i+sp] for i in [sp, n-1], -(1-h[i-sp]) for i in [0, n-1-sp]
            i1 = np.arange(sp, n)
            g[i1] += wk * h[i1 - sp]
            i2 = np.arange(0, n - sp)
            g[i2] -= wk * (1.0 - h[i2 + sp])
    
    return g, m
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cross_corr(h):
    """Compute full cross-correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_shift(h, sh):
    """Gradient of c_cross at a given shift sh (where sh = (n-1)-k)."""
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(0, n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += h[i1 - sp]
        i2 = np.arange(0, n - sp)
        g[i2] -= (1.0 - h[i2 + sp])
    return g

def _subgrad_max(h, tol_frac=0.001):
    """Subgradient of max_k c_cross[k]. Returns (gradient, max_value)."""
    n = len(h)
    c = _cross_corr(h)
    m = c.max()
    thresh = m - tol_frac * max(m, 1.0)
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    w = np.maximum(c[active] - thresh, 0.0)
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(len(active)) / len(active)
    else:
        w /= ws
    g = np.zeros(n)
    for wk, k in zip(w, active):
        sh = (n - 1) - k
        g += wk * _grad_at_shift(h, sh)
    return g, m

def _nesterov_subgrad(h, max_iters, tol_frac=0.001):
    """Nesterov accelerated projected subgradient method."""
    n = len(h)
    dx = 2.0 / n
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _eval(h)
    beta = 0.0
    
    for it in range(max_iters):
        # Nesterov momentum
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = h + ((1.0 - beta_new) / (1.0 + beta)) * (h - y) if it > 0 else h.copy()
        beta = beta_new
        
        g, m = _subgrad_max(y, tol_frac)
        
        # Step size: 2/(L*(it+1)) where L is Lipschitz constant (roughly 2*n)
        L = 2.0 * n
        lr = 2.0 / (L * (it + 1.0))
        
        # Try a few step sizes
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            h_new = _proj_to_sum(y - scale * lr * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_h = h_new.copy()
                best_c5 = c5n
                h = h_new
                improved = True
                break
        if not improved:
            # Try smaller
            for scale in [0.05, 0.01, 0.005]:
                h_new = _proj_to_sum(y - scale * lr * g, n * 0.5)
                c5n = _eval(h_new)
                if c5n < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5n
                    h = h_new
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c5
```

```python
def _pair_delta(h, i, j):
    """Compute Δc[k] = ∂c[k]/∂h[i] - ∂c[k]/∂h[j] for all k."""
    n = len(h)
    m = 2*n - 1
    dc = np.zeros(m)
    for k in range(m):
        sh = (n-1) - k
        ci = 0.0
        cj = 0.0
        if sh >= 0:
            if i <= n-1-sh:
                ci += 1.0 - h[i+sh]
            if i >= sh:
                ci -= h[i-sh]
            if j <= n-1-sh:
                cj += 1.0 - h[j+sh]
            if j >= sh:
                cj -= h[j-sh]
        else:
            sp = -sh
            if i >= sp:
                ci += 1.0 - h[i-sp]
            if i <= n-1-sp:
                ci -= h[i+sp]
            if j >= sp:
                cj += 1.0 - h[j-sp]
            if j <= n-1-sp:
                cj -= h[j+sp]
        dc[k] = ci - cj
    return dc
```

```python
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    return (m + np.log(np.sum(np.exp(s - sm))) / alpha + sm / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        g += w[k] * _grad_shift(h, sh)
    return g * (2.0 / n)

    return x.sum() - len(x) * 0.5

    return np.ones(len(x))

                       options={'maxiter': maxiter, 'ftol': ftol})

def _active_descent(h, max_iters, tol_frac, t_start):

    dx = 2.0 / n
    c5c = _eval(h)
    for it in range(max_iters):
        if time.time() - t_start > 10:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        thresh = m - tol_frac * max(m, 1.0)
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        w = np.maximum(c[active] - thresh, 0.0)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(active)) / len(active)
        else:
            w /= ws
        g = np.zeros(n)
        for wk, k in zip(w, active):
            sh = (n - 1) - k
            g += wk * _grad_shift(h, sh)
        g *= dx
        improved = False
        for step in [0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)

def _pair_delta(h, i, j):
    n = len(h)
    m = 2 * n - 1
    dc = np.zeros(m)
    for k in range(m):
        sh = (n - 1) - k
        ci = 0.0
        cj = 0.0

                ci += 1.0 - h[i + sh]
            if i >= sh:
                ci -= h[i - sh]
            if j <= n - 1 - sh:
                cj += 1.0 - h[j + sh]
            if j >= sh:
                cj -= h[j - sh]
        else:
            sp = -sh
            if i >= sp:
                ci += 1.0 - h[i - sp]
            if i <= n - 1 - sp:
                ci -= h[i + sp]
            if j >= sp:
                cj += 1.0 - h[j - sp]
            if j <= n - 1 - sp:
                cj -= h[j + sp]
        dc[k] = ci - cj
    return dc
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(abs(diff), h[idx])
                h[idx] += a if diff > 0 else -a

    return h
```
