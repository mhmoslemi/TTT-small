def _sgd(h, dx, t_end, top_k=12, lr_init=0.003):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = lr_init
    stalls = 0
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        ct = c[top]
        w = np.exp(60.0 * (ct - ct.max()))
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top):
            s = abs(int(ki - (n - 1)))
            if s == 0:
                g += wi * (1.0 - 2.0 * h)
            else:
                m = n - s
                g[s:n] += wi * (1.0 - h[:m])
                g[:m] -= wi * h[s:n]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        hp = _repair(np.clip(h - lr * g / gn, 0, 1), tgt)
        cn = _c5(hp, dx)
        if cn < best:
            best = cn
            h = hp
            lr = min(lr * 1.005, 0.02)
            stalls = 0
        else:
            lr *= 0.93
            stalls += 1
            if stalls > 300 or lr < 1e-7:
                break
    return h, best

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h._c5(h, dx) if hasattr(h, '_c5') else (h, _c5(h, dx))

def _smart_swap(h, s, n, d):
    m = n - s
    # Score: h[a+s] * (1-h[a]) - we want to maximize this
    # After swap: h[a+s]-d and h[a]+d
    # New term: (h[a+s]-d)*(1-h[a]-d) = h[a+s]*(1-h[a]) - d*(1-h[a]) - d*h[a+s] + d^2
    # Change: -d*(1-h[a]+h[a+s]) + d^2
    # We want this to be most negative: minimize (1-h[a]+h[a+s]) - d
    # i.e., minimize (1 - h[a] + h[a+s])
    # = 1 + h[a+s] - h[a]
    # So we want h[a+s] small and h[a] large? That's the opposite!
    # Wait, we're DECREASING h[a+s] and INCREASING h[a].
    # The change in c is: -d*(1-h[a]) - d*h[a+s] + d^2
    # = -d*(1 - h[a] + h[a+s]) + d^2
    # For this to be negative (reducing c), we need (1-h[a]+h[a+s]) > d
    # To maximize the reduction, we want to maximize (1-h[a]+h[a+s])
    # = 1 + (h[a+s] - h[a])
    # So we want h[a+s] large and h[a] small.
    
    scores = h[s:n] - h[:m]  # h[a+s] - h[a]
    best_a = np.argmax(scores)
    # Move mass from best_a+s to best_a
    d_actual = min(d, h[best_a+s], 1.0 - h[best_a])
    if d_actual > 0:
        h[best_a+s] -= d_actual
        h[best_a] += d_actual
    return d_actual
```

    return h, best

def _smoothed_grad(h, alpha, top_k=20):
    n = len(h)
    c = _corr(h)
    m = c.max()
    # Only consider top_k lags
    top = np.argsort(c)[::-1][:top_k]
    ct = c[top]
    s = np.clip(alpha * (ct - m), -300, 300)
    w = np.exp(s)

            continue

        else:
            mm = n - s_abs
            grad[s_abs:n] += wi * (1.0 - h[:mm])
            grad[:mm] -= wi * h[s_abs:n]
    return grad

If I think of h as a binary-like function (mostly 0s and 1s), then the overlap at lag s is counting the number of positions where h[i+s]=1 but h[i]=0 (or approximately that, for non-binary h).

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(400):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _subgrad(h, top_k=12):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:top_k]
    ct = c[top]
    w = np.exp(60.0 * (ct - ct.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            m = n - s
            grad[s:n] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[s:n]
    grad -= grad.mean()
    return grad

def _sgd(h, dx, t_end, top_k=12, lr_init=0.002, max_stalls=300):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = lr_init
    stalls = 0
    while time.time() < t_end:
        g = _subgrad(h, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        hp = _repair(np.clip(h - lr * g / gn, 0, 1), tgt)
        cn = _c5(hp, dx)
        if cn < best:
            best = cn; h = hp
            lr = min(lr * 1.005, 0.015)
            stalls = 0
        else:
            lr *= 0.94
            stalls += 1
            if stalls > max_stalls or lr < 5e-8:
                break
    return h, best

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _grad_line_search(h, dx, alpha, t_end, top_k=20):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    step = 0.008
    while time.time() < t_end:
        c = _corr(h)
        m = c.max()
        top = np.argsort(c)[::-1][:top_k]
        ct = c[top]
        sw = np.exp(np.clip(alpha * (ct - m), -300, 300))
        sw /= sw.sum()
        g = np.zeros(n)
        for wi, ki in zip(sw, top):
            if wi < 1e-10:
                continue
            s = abs(int(ki - (n - 1)))
            if s == 0:
                g += wi * (1.0 - 2.0 * h)
            else:
                mm = n - s
                g[s:n] += wi * (1.0 - h[:mm])
                g[:mm] -= wi * h[s:n]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        sp = step
        for _ in range(25):
            hp = _repair(np.clip(h - sp * g / gn, 0, 1), tgt)
            cn = _c5(hp, dx)
            if cn < best:
                best = cn; h = hp
                step = min(step * 1.02, 0.02)
                improved = True
                break
            sp *= 0.65
        if not improved:
            step *= 0.6
            if step < 1e-9:
                break
    return h, best

def _targeted(h, dx, rng, t_end, top_k=8):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.01
    stall_count = 0
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        found = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            for _ in range(6):
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = rng.uniform(0.001, scale)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; found = True
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
        for _ in range(4):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-scale, scale)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn; found = True
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
        if not found:
            stall_count += 1
            scale *= 0.88
            if scale < 3e-6:
                break
        else:
            stall_count = 0
            scale = min(scale * 1.01, 0.018)
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 0.0003, 0.0001, 3e-05, 1e-05, 3e-06]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb(h, rng, mode, n):
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] += 0.05 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.02 * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
