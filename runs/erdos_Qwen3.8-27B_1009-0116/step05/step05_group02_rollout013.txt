def _slsqp_lse(h, alpha, maxiter, ftol, n):
    bounds = [(0.0, 1.0)] * n
    def _f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -350, 350)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _g(x):
        return _grad_lse(x, alpha, n)
    def _cf(x):
        return x.sum() - n * 0.5
    def _cg(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cg}]
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _proj(res.x, n * 0.5)
    except Exception:
        return h

def _proj_grad_lse(h, alpha, n, max_iter, t0, tlimit, step0=0.04, mom=0.85):
    step = step0
    m_arr = np.zeros(n)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(max_iter):
        if time.time() - t0 > tlimit:
            break
        g = _grad_lse(h, alpha, n)
        m_arr = mom * m_arr - step * g
        h_new = _proj(h + m_arr, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5 - 1e-16:
            h = h_new
            c5 = c5n
            step = min(step * 1.02, 0.15)
        else:
            step *= 0.75
            m_arr *= 0.5
            if step < 1e-9:
                break
    return h, c5

def _worst_k_descent(h, n, max_iter, t0, tlimit, topk=5):
    h = h.copy()
    c5 = _eval(h)
    dx = 2.0 / n
    for _ in range(max_iter):
        if time.time() - t0 > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk_idx = np.argsort(corr)[-topk:]
        g = np.zeros(n)
        for k in topk_idx:
            g += _grad_corr(h, k, n)
        g *= (dx / topk)
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            h = _proj(h + 0.003 * np.random.randn(n), n * 0.5)
            c5 = _eval(h)
    return h, c5

def _coord_descent(h, n, max_iter, t0, tlimit, scale, rng):
    h = h.copy()
    c5 = _eval(h)
    for _ in range(max_iter):
        if time.time() - t0 > tlimit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _proj(np.array(initial_h_values, dtype=float).copy(), n0 * 0.5)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP with moderate alpha on n=280
    if time.time() - t0 < budget_s - 300:
        h_new = _slsqp_lse(best_h, 1000, 800, 1e-15, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Projected gradient LSE at n=280 with increasing alpha
    for alpha in [2000, 5000, 10000, 30000]:
        if time.time() - t0 > budget_s - 250:
            break
        h_tmp, c5 = _proj_grad_lse(best_h, alpha, best_n, 20000, t0, budget_s - 250, 0.03, 0.9)
        if c5 < best_c5:
            best_c5, best_h = c5, h_tmp.copy()

    # Phase 3: Worst-k descent at n=280
    if time.time() - t0 < budget_s - 200:
        h_tmp, c5 = _worst_k_descent(best_h, best_n, 50000, t0, budget_s - 200, topk=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h_tmp.copy()

    # Phase 4: SLSQP high-alpha polish at n=280
    if time.time() - t0 < budget_s - 180:
        h_new = _slsqp_lse(best_h, 20000, 1500, 1e-17, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: Upsample to 2x and optimize
    if time.time() - t0 < budget_s - 350:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2).astype(float)
        h_hi = _proj(h_hi, n1 * 0.5)
        for alpha in [2000, 8000, 20000]:
            if time.time() - t0 > budget_s - 200:
                break
            h_hi, c5 = _proj_grad_lse(h_hi, alpha, n1, 15000, t0, budget_s - 200, 0.02, 0.88)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if time.time() - t0 < budget_s - 150:
            h_hi, c5 = _worst_k_descent(best_h, best_n, 40000, t0, budget_s - 150, topk=5)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()
        if time.time() - t0 < budget_s - 120:
            h_new = _slsqp_lse(best_h, 10000, 1200, 1e-17, best_n)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 6: Coordinate descent fine-tuning
    for scale in [0.005, 0.001, 0.0003, 0.0001]:
        if time.time() - t0 > budget_s - 30:
            break
        h_tmp, c5 = _coord_descent(best_h, best_n, 60000, t0, budget_s - 30, scale, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h_tmp.copy()

    # Phase 7: Final SLSQP
    if time.time() - t0 < budget_s - 10:
        h_new = _slsqp_lse(best_h, 50000, 2000, 1e-18, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 8: Ultra-fine coordinate
    if time.time() - t0 < budget_s - 2:
        h_tmp, c5 = _coord_descent(best_h, best_n, 40000, t0, budget_s - 2, 5e-5, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h_tmp.copy()

    best_h = _proj(best_h, best_n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
