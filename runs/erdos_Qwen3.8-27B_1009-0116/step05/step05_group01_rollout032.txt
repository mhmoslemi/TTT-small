    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _pair_search(h, c, rng, n_iter, scale, n, temperature=0.0):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        delta = new_c5 - c5c
        if delta < -1e-15 or (temperature > 0 and rng.random() < np.exp(-delta / temperature)):
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
    return h, c, c5c

def _binarize(h, n, rng):
    # Push values toward 0 or 1
    h = h.copy()
    for _ in range(200):
        # Pick a random position and push toward 0 or 1
        idx = rng.randint(n)
        if h[idx] > 0.5:
            h[idx] = min(1.0, h[idx] + 0.1)
        else:
            h[idx] = max(0.0, h[idx] - 0.1)

def _binary_init(n, rng):
    # Create a random binary solution with n/2 ones
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _pair_search(h, c, rng, n_iter, scale, n, temp=0.0):
    dx = 2.0 / n
    c5c = c.max() * dx
    dc_buf = np.zeros(2 * n - 1)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc_buf[:] = 0
        np.add(dc_buf, _pos_change(h, i, d, n), out=dc_buf)
        dc_j = _pos_change(h, j, -d, n)
        dc_buf += dc_j
        new_max = (c + dc_buf).max() * dx
        delta = new_max - c5c
        if delta < -1e-15 or (temp > 0 and rng.random() < np.exp(-delta / max(temp, 1e-12))):
            h[i] = ni
            h[j] = nj
            c += dc_buf
            c5c = new_max
    return h, c, c5c

def _binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n - sp])
            g[:n - sp] -= w[k] * h[sp:]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP on parent
    for alpha in [5000, 50000, 200000]:
        if _el() > 90:
            break
        hn = _slsqp(best_h, alpha, 2000, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Binarize + pairwise
    if _el() < budget_s - 600:
        hb = best_h.copy()
        for t in np.linspace(0.3, 1.0, 40):
            hb = np.where(hb > 0.5 + 0.005 * t, np.minimum(1.0, hb + 0.03),
                          np.where(hb < 0.5 - 0.005 * t, np.maximum(0.0, hb - 0.03), hb))
        hb = _repair(hb, best_n)
        cb = _corr(hb)
        for scale in [0.15, 0.06, 0.025, 0.01, 0.005, 0.002]:
            if _el() > budget_s - 600:
                break
            hb, cb, c5c = _pair_search(hb, cb, rng, 100000, scale, best_n)
        hb = _repair(hb, best_n)
        c5 = _eval(hb)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Phase 3: Main extensive pairwise on best
    if _el() < budget_s - 180:
        hp = best_h.copy()
        cp = _corr(hp)
        np_ = best_n
        scales = [0.04, 0.02, 0.01, 0.005, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
        for scale in scales:
            if _el() > budget_s - 180:
                break
            n_it = 250000 if scale >= 0.001 else 200000
            hp, cp, c5c = _pair_search(hp, cp, rng, n_it, scale, np_)
        hp = _repair(hp, np_)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hp.copy(), np_

    # Phase 4: Try alternative n with binary init
    for n_alt in [200, 140, 220]:
        if _el() > budget_s - 150:
            break
        for trial in range(3):
            if _el() > budget_s - 150:
                break
            ha = _binary_init(n_alt, rng)
            ca = _corr(ha)
            for scale in [0.2, 0.1, 0.04, 0.015, 0.006, 0.003, 0.001, 0.0005]:
                if _el() > budget_s - 150:
                    break
                ha, ca, _ = _pair_search(ha, ca, rng, 120000, scale, n_alt)
            ha = _repair(ha, n_alt)
            c5 = _eval(ha)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, ha.copy(), n_alt

    # Phase 5: SLSQP polish on best
    if _el() < budget_s - 80:
        for alpha in [300000, 1000000]:
            if _el() > budget_s - 80:
                break
            hpol = _slsqp(best_h, alpha, 3000, 1e-20)
            c5 = _eval(hpol)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hpol.copy(), len(hpol)

    # Phase 6: Annealed pairwise (escape local optima)
    if _el() < budget_s - 40:
        ha = best_h.copy()
        ca = _corr(ha)
        na = best_n
        for temp in [0.008, 0.003, 0.001, 0.0003]:
            if _el() > budget_s - 40:
                break
            for scale in [0.015, 0.006, 0.002]:
                if _el() > budget_s - 40:
                    break
                ha, ca, _ = _pair_search(ha, ca, rng, 80000, scale, na, temp=temp)
        ha = _repair(ha, na)
        c5 = _eval(ha)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, ha.copy(), na

    # Phase 7: Final ultra-fine pairwise
    if _el() < budget_s - 5:
        hf = best_h.copy()
        cf = _corr(hf)
        nf = best_n
        for scale in [5e-5, 2e-5, 1e-5, 5e-6]:
            if _el() > budget_s - 5:
                break
            hf, cf, _ = _pair_search(hf, cf, rng, 200000, scale, nf)
        hf = _repair(hf, nf)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hf.copy(), nf

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
