      ```python
      def repair(h, n):
          h_flat = np.clip(h, 0, 1).copy()
          target = n/2.0
          s = np.sum(h_flat)
          if abs(s - target) > 1e-9:
              h_flat *= target / s
              h_flat = np.clip(h_flat, 0, 1)
              diff = target - np.sum(h_flat)
              order = np.argsort(1.0 - h_flat)
              for idx in order:
                  if abs(diff) < 1e-12:
                      break
                  if diff > 0:
                      a = min(diff, 1.0 - h_flat[idx])
                      h_flat[idx] += a
                      diff -= a
                  else:
                      a = -min(diff, h_flat[idx])
                      h_flat[idx] -= a
                      diff += a
          return h_flat
      ```
