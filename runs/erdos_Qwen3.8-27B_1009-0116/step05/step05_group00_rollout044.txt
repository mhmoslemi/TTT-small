def _smooth_obj(x, alpha, pen):

The optimal solution balances the overlaps across all shifts. This is why the solution tends to be "smooth" or have a particular frequency content.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(x, alpha, pen):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = np.clip(alpha * (c - m), -500, 500)
    cm = m + np.log(np.sum(np.exp(sh))) / alpha
    pen_term = pen * (x.sum() - n / 2.0) ** 2
    return cm + pen_term

def _smooth_grad(x, alpha, pen):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = (n - 1) - k
        wk = w[k] * dx
        if s >= 0:
            g[:n - s] += wk * (1.0 - x[s:])
            if s > 0:
                g[s:] -= wk * x[:n - s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    g += 2.0 * pen * (x.sum() - n / 2.0)
    return g

def _lbfgs(h, alpha, pen, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _binary_eval_fast(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * (2.0 / n), c

def _binary_hill_climb(h_bin, max_rounds, rng, time_limit):
    n = len(h_bin)
    h = h_bin.copy()
    c5 = _eval(h)
    t_start = time.time()
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_trials = min(2000, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c2 = _eval(h)
            if c2 < c5 - 1e-15:
                c5 = c2
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, c5

def _binary_sa(h_bin, max_rounds, rng, time_limit, t0=1.0, t1=0.01):
    n = len(h_bin)
    h = h_bin.copy()
    c5 = _eval(h)
    best_c5 = c5
    best_h = h.copy()
    t_start = time.time()
    total_iters = max_rounds * 500
    for it in range(total_iters):
        if time.time() > time_limit:
            break
        frac = min(1.0, it / total_iters)
        temp = t0 * (t1 / t0) ** frac
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c2 = _eval(h)
        delta = c2 - c5
        if delta < 0 or rng.rand() < np.exp(-delta / (temp * 0.01)):
            c5 = c2
            if c2 < best_c5:
                best_c5 = c2
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def _worst_shift_fix(h, top_k, steps, rng):
    n = len(h)
    c5 = _eval(h)
    for _ in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_k:]
        for k in top_idx:
            s = (n - 1) - k
            wk = 0.01
            if s >= 0:
                for _ in range(20):
                    i = rng.randint(n - s)
                    if h[i] < 0.5 and h[i + s] > 0.5:
                        d = min(h[i + s] - 0.01, 0.01)
                        h[i + s] -= d
                        h[i] += d
                        break
            else:
                sp = -s
                for _ in range(20):
                    i = rng.randint(sp, n)
                    if h[i - sp] < 0.5 and h[i] > 0.5:
                        d = min(h[i] - 0.01, 0.01)
                        h[i] -= d
                        h[i - sp] += d
                        break
        h = _repair(h, n)
        c2 = _eval(h)
        if c2 >= c5 + 0.001:
            break
        c5 = c2
    return h, c5

def _make_periodic(pattern, n):
    T = len(pattern)
    reps = n // T
    h = np.tile(pattern, reps)
    return h[:n]

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
