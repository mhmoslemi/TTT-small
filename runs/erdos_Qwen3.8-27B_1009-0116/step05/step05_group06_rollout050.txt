        y_new = _proj_to_sum(y - step0 * g, target)
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        y_mom = y_new + ((t_k - 1) / t_new) * (y_new - y)

            y = y_new
            t_k = 1.0
        else:
            y = y_mom
            t_k = t_new
        if k % 200 == 199:
            y = _repair(y, n)
    return _repair(y, n)

                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5, best_h, best_n = c5c, h.copy(), nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
