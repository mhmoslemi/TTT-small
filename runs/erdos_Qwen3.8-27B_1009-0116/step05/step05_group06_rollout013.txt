def _topk_grad_all(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _cobj_fun(x):
    return x.sum() - x.shape[0] * 0.5

def _cobj_jac(x):
    return np.ones(x.shape[0])

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _topk_grad(h, K):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    if K >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / K

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cobj_fun, 'jac': _cobj_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fista(h, nsteps, K, step0, t_end):
    n = len(h)
    target = n * 0.5
    y = h.copy()
    t = 1.0
    h = _repair(h, n)
    for it in range(nsteps):
        g = _topk_grad(y, K)
        gn = np.linalg.norm(g)
        step = step0 / (1.0 + gn * 50)
        x_new = _proj_to_sum(y - step * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - h)
        y = np.clip(y, 0, 1)
        h = x_new
        t = t_new
    return _repair(h, n), _eval(h)

def _pair_cd(h, n_pairs, rng, precision):
    n = len(h)
    c5c = _eval(h)
    ndx = 2 * n - 1
    k_arr = np.arange(ndx)
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        c = _corr(h)
        idx1i = i + k_arr - (n - 1)
        idx2i = i - k_arr + (n - 1)
        v1i = (idx1i >= 0) & (idx1i < n)
        v2i = (idx2i >= 0) & (idx2i < n)
        gi = np.zeros(ndx)
        gi[v1i] = 1.0 - h[idx1i[v1i]]
        gi[v2i] -= h[idx2i[v2i]]
        idx1j = j + k_arr - (n - 1)
        idx2j = j - k_arr + (n - 1)
        v1j = (idx1j >= 0) & (idx1j < n)
        v2j = (idx2j >= 0) & (idx2j < n)
        gj = np.zeros(ndx)
        gj[v1j] = 1.0 - h[idx1j[v1j]]
        gj[v2j] -= h[idx2j[v2j]]
        slope = gi - gj
        tlo = max(0.0, h[j] - 1.0)
        thi = min(1.0, h[j])
        if thi - tlo < 1e-12:
            continue
        a, b = tlo, thi
        cp = a + (1.0 - gr) * (b - a)
        dp = a + gr * (b - a)
        fc = np.max(c + slope * cp)
        fd = np.max(c + slope * dp)
        for _ in range(precision):
            if fc < fd:
                b = dp; dp = cp; fd = fc
                cp = a + (1.0 - gr) * (b - a)
                fc = np.max(c + slope * cp)
            else:
                a = cp; cp = dp; fc = fd
                dp = a + gr * (b - a)
                fd = np.max(c + slope * dp)
        t_opt = (a + b) * 0.5
        if abs(t_opt) < 1e-10:
            continue
        h[i] += t_opt
        h[j] -= t_opt
        c5n = _eval(h)
        if c5n > c5c + 1e-13:
            h[i] -= t_opt
            h[j] += t_opt
        else:
            c5c = c5n
    return _repair(h, n), c5c
