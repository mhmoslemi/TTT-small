import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_corr(k, h, n):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * (2.0 / n)

def _pen_obj(h, tgt, mu):
    c = _corrs(h)
    ex = np.maximum(0.0, c - tgt)
    return tgt + mu * np.dot(ex, ex)

def _pen_grad(h, tgt, mu):
    n = len(h)
    c = _corrs(h)
    ex = np.maximum(0.0, c - tgt)
    g = np.zeros(n)
    for k in range(len(c)):
        if ex[k] > 1e-13:
            g += (1.0 + 2.0 * mu * ex[k]) * _grad_corr(k, h, n)
    return g

def _slsqp(h, obj, grad, args, maxit, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': ftol})
        if res.fun < obj(h, *args) + 1e-12:
            return _fix_sum(res.x, n)
    except Exception:
        pass
    return _fix_sum(h, n)

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            g += w[k] * _grad_corr(k, h, n)
    return g

        step = min(step * 1.12, step0)
    return h, f

def _coordinate_polish(h, n_rounds, n_cand):
    n = len(h)
    h = h.copy()

                fn = _eval(h)
                if fn < f:
                    f = fn
                else:
                    h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):

    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _fix_sum(np.clip(f(x_new), 0, 1), n * factor)
