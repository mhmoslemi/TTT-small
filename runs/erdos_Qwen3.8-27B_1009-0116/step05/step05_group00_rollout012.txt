import numpy as np
from scipy.optimize import minimize
import time
from itertools import combinations

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _softobj(x, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    soft = m + np.log(float(np.sum(np.exp(s)))) / alpha
    penalty = lam * (x.sum() - len(x) * 0.5) ** 2
    return soft + penalty

def _softgrad(x, alpha, lam):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)

            i2 = np.arange(n - sp)

    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

                                'gtol': pgtol, 'maxfun': maxiter * 3})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _periodic_enum(P, n):
    """Enumerate all binary sequences of length P with P/2 ones, evaluate best"""
    if P % 2 != 0 or n % P != 0:
        return None, 1e9
    half = P // 2
    best_c5 = 1e9
    best_pat = None
    for combo in combinations(range(P), half):
        pat = np.zeros(P)
        pat[list(combo)] = 1.0
        h = np.tile(pat, n // P)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_pat = pat.copy()
    return best_pat, best_c5

def _coord_descent(h, rng, n_iter, scale):
    nc = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

                c5c = c2
