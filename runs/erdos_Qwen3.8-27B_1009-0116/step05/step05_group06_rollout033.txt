            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 4: Block coordinate refinement (new: optimizes small blocks)
    if _el() < budget_s - 200:
        h_blk, c5_blk = _block_refine(best_h, best_n, t0, budget_s - 200)
        if c5_blk < best_c5:
            best_c5, best_h = c5_blk, h_blk.copy()

    # Phase 5: Pairwise simulated annealing (structured, multi-scale)
    if _el() < budget_s - 60:
        h_sa, c5_sa = _pairwise_search(best_h.copy(), best_n, t0, budget_s - 60, 0.005, 8000)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 15:
        h_fin = _slsqp(best_h, 400000, 2000, 1e-19)
        c5_fin = _eval(h_fin)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()

    # Phase 7: Ultra-fine local search
    if _el() < budget_s - 3:
        rng = np.random.RandomState(seed + 1)
        h_ultra = best_h.copy()
        c5_ultra = _eval(h_ultra)
        for _ in range(8000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = 5e-5 * rng.randn()
            ni, nj = h_ultra[i] + d, h_ultra[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_ultra[i], h_ultra[j] = ni, nj
                c5n = _eval(h_ultra)
                if c5n < c5_ultra:
                    c5_ultra = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h_ultra.copy()
                else:
                    h_ultra[i], h_ultra[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
