import numpy as np
from scipy.optimize import minimize
import time

_sum_target = 0

def _cons_fun(x):
    return x.sum() - _sum_target

def _cons_jac(x):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pnorm_obj(h, p, dx):
    c = _corr(h); m = len(c)
    cp = np.maximum(c, 0) ** p
    return (cp.sum() / m) ** (1.0 / p) * dx

def _pnorm_grad(h, p, dx):
    c = _corr(h); n = len(h); m = len(c)
    cpos = np.maximum(c, 0)
    base = (cpos.sum() ** p / m) ** (1.0/p - 1.0) if cpos.sum() > 0 else 0
    w = base * cpos ** (p - 1) / m * dx if p > 1 else cpos * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += w[k] * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= w[k] * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return g

def _lse_obj(h, alpha, dx):
    c = _corr(h); m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha * dx

def _lse_grad(h, alpha, dx):
    c = _corr(h); m = c.max(); n = len(h)
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _slsqp(h, obj_fn, grad_fn, n, args, maxiter, ftol):
    global _sum_target
    _sum_target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=args,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, obj_fn, grad_fn, n, nsteps, step0, mu):
    f = obj_fn(h)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        g = grad_fn(h)
        v = mu * v - step * g
        h_try = _proj_sum(h - v, n * 0.5)
        fn = obj_fn(h_try)
        if fn < f:
            h, f = h_try, fn
        else:
            v *= 0.5; step *= 0.7
    return h

def _coord_descent(h, n, nsweeps, npts):
    c5c = _eval(h)
    trials = np.linspace(0, 1, npts)
    for _ in range(nsweeps):
        for i in range(n):
            saved = h[i]
            best_v, best_c = saved, c5c
            for tv in trials:
                h[i] = tv
                c5t = _eval(h)
                if c5t < best_c: best_c, best_v = c5t, tv
            h[i] = best_v; c5c = best_c
    return h, c5c

def _fine_coord(h, n, iters):
    c5c = _eval(h)
    for it in range(iters):
        i = it % n
        saved = h[i]
        lo = max(0.0, saved - 0.008)
        hi = min(1.0, saved + 0.008)
        if hi - lo < 1e-8: continue
        trials = np.linspace(lo, hi, 17)
        best_v, best_c = saved, c5c
        for tv in trials:
            h[i] = tv
            c5t = _eval(h)
            if c5t < best_c: best_c, best_v = c5t, tv
        h[i] = best_v; c5c = best_c
    return h, c5c

def _pair_coord(h, n, iters, gap, npts):
    c5c = _eval(h)
    trials = np.linspace(0, 1, npts)
    for it in range(iters):
        i = (it * gap) % n
        j = (i + gap) % n
        si, sj = h[i], h[j]
        best_c = c5c; bi, bj = si, sj
        for ti in trials:
            h[i] = ti
            for tj in trials:
                h[j] = tj
                c5t = _eval(h)
                if c5t < best_c: best_c = c5t; bi, bj = ti, tj
        h[i], h[j] = bi, bj; c5c = best_c
    return h, c5c

    t0 = time.time()

    n0 = len(initial_h_values)

        c5 = _eval(h)

    # Phase 2: LSE SLSQP

        if _el() > budget_s - 450: break

        c5 = _eval(h)
        if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 3: Nesterov
    if _el() < budget_s - 300:
        h_new = _nesterov(best_h, _lse_obj, _lse_grad, n0, 2500, 0.002, 0.85)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new.copy()

    # Phase 4: Block + single coordinate descent
    if _el() < budget_s - 200:
        h_new, _ = _pair_coord(best_h, n0, 8000, 3, 9)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new.copy()
    if _el() < budget_s - 150:
        h_new, _ = _coord_descent(best_h, n0, 3, 41)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new.copy()

    # Phase 5: Multi-start
    if _el() < budget_s - 250:
        for r in range(18):
            if _el() > budget_s - 250: break

            if strat == 0:

    return (best_h, best_c5, n0)
