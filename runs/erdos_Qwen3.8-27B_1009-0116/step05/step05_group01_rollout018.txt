```python
def _optimal_d(corr, delta, d_lo, d_hi, n_iter=40):
    # Find d in [d_lo, d_hi] minimizing max_k (corr[k] + d*delta[k])
    # f(d) is convex
    lo, hi = d_lo, d_hi
    for _ in range(n_iter):
        mid = (lo + hi) / 2
        vals = corr + mid * delta
        active = np.where(vals == vals.max())[0]
        subgrad = delta[active]
        if subgrad.min() > 1e-14:
            hi = mid  # need to decrease d
        elif subgrad.max() < -1e-14:
            lo = mid  # need to increase d
        else:
            break
    d_opt = (lo + hi) / 2
    # Evaluate at boundaries too
    best_d = d_opt
    best_f = np.max(corr + d_opt * delta)
    for d_test in [d_lo, d_hi]:
        f_test = np.max(corr + d_test * delta)
        if f_test < best_f:
            best_f = f_test
            best_d = d_test
    return best_d, best_f
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a

        elif sg.max() < -1e-14:
            lo = mid
        else:
            break
    d_opt = (lo + hi) * 0.5
    best_d, best_f = d_opt, np.max(corr + d_opt * dc)
    for d_t in (d_lo, d_hi):
        f_t = np.max(corr + d_t * dc)
        if f_t < best_f:
            best_f, best_d = f_t, d_t
    return best_d, best_f

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _pair_search(h, corr, n, max_sweeps=50, time_limit=None, t0=0):
    dx = 2.0 / n
    c5 = corr.max() * dx
    moves_since_recompute = 0
    dc_all = [np.zeros(2*n-1) for _ in range(n)]
    for i in range(n):
        dc_all[i] = _delta_corr(h, i, n)
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                dc = dc_all[i] - dc_all[j]
                d_opt, f_opt = _optimal_d(corr, dc, d_lo, d_hi)
                if f_opt * dx < c5 - 1e-16:
                    h[i] = min(max(h[i] + d_opt, 0.0), 1.0)
                    h[j] = min(max(h[j] - d_opt, 0.0), 1.0)
                    corr = corr + d_opt * dc
                    c5 = corr.max() * dx
                    improved = True
                    moves_since_recompute += 1
                    dc_all[i] = _delta_corr(h, i, n)
                    dc_all[j] = _delta_corr(h, j, n)
                    if moves_since_recompute >= 80:
                        corr = _corr(h)
                        c5 = corr.max() * dx
                        for ii in range(n):
                            dc_all[ii] = _delta_corr(h, ii, n)
                        moves_since_recompute = 0
        if not improved:
            break
    return h, corr, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP optimization at original resolution
    for alpha in [3000, 30000, 150000, 500000]:
        if _el() > 90:
            break
        h_new = _slsqp(h, alpha, 3000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        h = h_new

    # Phase 2: Systematic pair search (main improvement)
    if _el() < budget_s - 200:
        h_ps = best_h.copy()
        corr_ps = _corr(h_ps)
        tl = min(budget_s - 150, _el() + 400)
        h_ps, corr_ps, c5_ps = _pair_search(h_ps, corr_ps, n, max_sweeps=200, time_limit=tl, t0=t0)
        h_ps = _repair(h_ps, n)
        c5 = _eval(h_ps)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ps.copy()
            best_n = n

    # Phase 3: SLSQP polish on the pair-search result
    if _el() < budget_s - 120:
        h_pol = _slsqp(best_h, 300000, 2000, 1e-19)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h = c5, h_pol.copy()
        h_pol2 = _slsqp(best_h, 800000, 2000, 1e-19)
        c5 = _eval(h_pol2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_pol2.copy()

    # Phase 4: More pair search on polished result
    if _el() < budget_s - 60:
        h_ps2 = best_h.copy()
        corr_ps2 = _corr(h_ps2)
        tl4 = min(budget_s - 50, _el() + 200)
        h_ps2, corr_ps2, c5_ps2 = _pair_search(h_ps2, corr_ps2, n, max_sweeps=200, time_limit=tl4, t0=t0)
        h_ps2 = _repair(h_ps2, n)
        c5 = _eval(h_ps2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ps2.copy()

    # Phase 5: Try 1-h symmetry and perturbed starts
    if _el() < budget_s - 100:
        h_sym = _repair(1.0 - best_h, n)
        c5_sym = _eval(h_sym)
        if c5_sym < best_c5:
            best_c5, best_h = c5_sym, h_sym.copy()
        h_sym_ps = best_h.copy()
        corr_sym = _corr(h_sym_ps)
        tl5 = min(budget_s - 80, _el() + 60)
        h_sym_ps, corr_sym, c5_sym2 = _pair_search(h_sym_ps, corr_sym, n, max_sweeps=50, time_limit=tl5, t0=t0)
        c5 = _eval(_repair(h_sym_ps, n))
        if c5 < best_c5:
            best_c5, best_h = c5, _repair(h_sym_ps, n).copy()

    # Phase 6: Final SLSQP + pair polish
    if _el() < budget_s - 20:
        h_fin = _slsqp(best_h, 500000, 2000, 1e-20)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    # Phase 7: Ultra-fine pair search
    if _el() < budget_s - 3:
        h_f = best_h.copy()
        c_f = _corr(h_f)
        tl7 = max(t0 + _el() + 0.001, time.time() + 2)
        h_f, c_f, c5_f = _pair_search(h_f, c_f, n, max_sweeps=100, time_limit=budget_s - 2, t0=t0)
        h_f = _repair(h_f, n)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
