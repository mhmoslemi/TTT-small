def _equilibrate(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    h_work = h.copy()
    while time.time() < t_end:
        c = _corr(h_work)
        top = np.argsort(c)[::-1][:3]
        # Get gradients for top lags
        grads = [_grad_single_lag(h_work, top[i]) for i in range(len(top))]
        # Find direction that decreases max lag while keeping others constant
        # Solve: G d = b where G is 3xn, b = [-1, 0, sum_constraint]
        # Minimum norm solution: d = G^T (G G^T)^{-1} b
        G = np.array([grads[0], grads[1], np.ones(n)])
        b = np.array([-1.0, 0.0, 0.0])
        try:
            GGt = G @ G.T
            d = G.T @ np.linalg.solve(GGt, b)
        except:
            break
        # Scale d
        dn = np.linalg.norm(d)
        if dn < 1e-10:
            break
        d = d / dn * 0.01
        h_try = _repair(h_work + d, tgt)
        c_try = _c5(h_try, dx)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            h_work = h_try
        else:
            h_try = _repair(h_work - d, tgt)
            c_try = _c5(h_try, dx)
            if c_try < best_c:

                best_h = h_try.copy()

                h_work[idx[1]] -= d2

    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha):
    return _grad_lse(x, alpha)

def _eq_con(x, tgt):
    return x.sum() - tgt

def _slsqp_solve(h0, alpha, maxiter=300):
    n = len(h0)
    tgt = n / 2.0
    dx = 2.0 / n
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(_lse_obj, h0, args=(alpha,), jac=_lse_grad, method='SLSQP',
                     bounds=bds, constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (tgt,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _proj_grad(h0, dx, t_end, alpha0=50, alpha_mult=1.03, max_iter=200000):
    n = len(h0)
    tgt = n / 2.0
    best_h = h0.copy()
    best_c = _c5(h0, dx)
    h = h0.copy()
    alpha = alpha0
    step = 0.02
    stall = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = _grad_lse(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_try = _repair(h - step * g / gn, tgt)
        c_try = _c5(h_try, dx)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
            h = h_try
            step = min(step * 1.03, 0.15)
            alpha *= alpha_mult
            stall = 0
        else:
            step *= 0.75
            stall += 1
            if stall > 30:
                step = 0.02
                alpha *= 2.0
                if alpha > 50000:
                    break
                stall = 0
    return best_h, best_c

def _targeted(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:5]
        improved = False
        for ki in top:
            p = abs(int(ki - (n - 1)))
            if p == 0:
                continue
            L = n - p
            for _ in range(8):
                a = rng.randint(L)
                i1, i2 = a + p, a if p > 0 else a + p
                j1, j2 = (a + p, a) if p > 0 else (a, a + p)
                d = rng.uniform(0.003, 0.025) * rng.choice([-1, 1])
                h[j1] -= d; h[j2] += d
                if 0 <= h[j1] <= 1 and 0 <= h[j2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best_c:
                        best_c = cn; best_h = h.copy(); improved = True
                    else:
                        h[j1] += d; h[j2] -= d
                else:
                    h[j1] += d; h[j2] -= d
        if not improved:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.015, 0.015)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return best_h, best_c

def _equilibrate(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:3]
        grads = [_grad_single(h, top[i]) for i in range(len(top))]
        G = np.array([grads[0], grads[1], np.ones(n)])
        b = np.array([-1.0, 0.0, 0.0])
        try:
            GGt = G @ G.T
            d = G.T @ np.linalg.solve(GGt, b)
        except Exception:
            d = grads[0] - grads[1]
        dn = np.linalg.norm(d)
        if dn < 1e-10:
            break
        d = d / dn * 0.015
        for sign in [1, -1]:
            h_try = _repair(h + sign * d, tgt)
            cn = _c5(h_try, dx)
            if cn < best_c:
                best_c = cn; best_h = h_try.copy(); h = h_try
                break
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                dd = rng.uniform(-0.02, 0.02)
                h[i] += dd; h[j] -= dd
                h = _repair(h, tgt)
    return best_h, best_c

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(5, n // 4), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: Projected gradient with analytic LSE gradient
    h, c = _proj_grad(best_h, dx, t0 + 120, alpha0=50, alpha_mult=1.025)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 2: Targeted search
    h, c = _targeted(best_h, dx, rng, t0 + 220)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 3: Equilibration
    h, c = _equilibrate(best_h, dx, rng, t0 + 300)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 4: SLSQP with moderate alpha
    for a in [500, 2000, 8000]:
        if time.time() > t0 + 370:
            break
        hc, cc = _slsqp_solve(best_h, a, 200)
        if cc < best_c5:
            best_h, best_c5 = hc, cc

    # Phase 5: Multi-start
    t_end5 = t0 + 550
    for r in range(20):
        if time.time() > t_end5 or budget_s - (time.time() - t0) < 280:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _proj_grad(hp, dx, min(t_end5 - 5, time.time() + 12), alpha0=200, alpha_mult=1.05)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hp, c = _targeted(hp, dx, rng, min(t_end5 - 1, time.time() + 5))
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 6: Higher resolution
    t_end6 = t0 + 750
    if budget_s - (time.time() - t0) > 300:
        n2 = 2 * n
        dx2 = 2.0 / n2
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _proj_grad(h2, dx2, min(t_end6 - 60, time.time() + 60), alpha0=50, alpha_mult=1.03)
        h2, c2 = _targeted(h2, dx2, rng, min(t_end6 - 30, time.time() + 40))
        h2, c2 = _equilibrate(h2, dx2, rng, t_end6)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 7: Final targeted + equilibrate
    nb, dxb = best_n, 2.0 / best_n
    best_h, best_c5 = _targeted(best_h, dxb, rng, t0 + 880)
    best_h, best_c5 = _equilibrate(best_h, dxb, rng, t0 + 920)

    # Phase 8: Final SLSQP high-alpha
    if time.time() < t0 + 970:
        for a in [20000, 80000]:
            if time.time() > t0 + 965:
                break
            hc, cc = _slsqp_solve(best_h, a, 150)
            if cc < best_c5:
                best_h, best_c5 = hc, cc

    # Phase 9: Fine local search
    if time.time() < t0 + budget_s - 5:
        scales = [0.003, 0.001, 5e-04, 2e-04, 1e-04, 5e-05]
        h_f = best_h.copy()
        c_f = best_c5
        for sc in scales:
            for _ in range(3000):
                if time.time() > t0 + budget_s - 2:
                    break
                i, j = rng.randint(best_n), rng.randint(best_n)
                if i == j:
                    continue
                d = sc * (2.0 * rng.random() - 1.0)
                h_f[i] += d; h_f[j] -= d
                if 0 <= h_f[i] <= 1 and 0 <= h_f[j] <= 1:
                    cn = _c5(h_f, dxb)
                    if cn < c_f:
                        c_f = cn
                    else:
                        h_f[i] -= d; h_f[j] += d
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
