                i, j = rng.randint(ncc), rng.randint(ncc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c:
                        c5c = c5n
                        if c5c < best_c5:
                            best_c5, best_h, best_n = c5c, h.copy(), ncc
                    else:
                        h[i], h[j] = ni - d, nj + d

    # Phase 7: Final L-BFGS-B polish with very high p
    if _el() < budget_s - 3:
        h = _lsbfsb(best_h, 2048, 0.3, maxiter=4000)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
