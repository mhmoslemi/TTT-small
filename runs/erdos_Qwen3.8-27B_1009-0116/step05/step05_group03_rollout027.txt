def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break

                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:

def _grad_lag(h, s):
    """Gradient of lag-s correlation wrt h, for s in -(n-1)..(n-1)."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):

            g[i + s] += 1.0 - h[i]

        sp = -s
        for i in range(sp, n):

            g[i - sp] -= h[i]

        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _greedy_local(h, dx, t_end, rng, max_iter=800000):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, dx)
    stuck = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        full_c = _corr(h)
        worst_k = int(full_c.argmax())
        s = worst_k - (n - 1)
        g = _grad_lag(h, s)
        a = int(g.argmax())
        b = int(g.argmin())
        if g[a] - g[b] < 1e-12:
            break
        delta_max = min(h[a], 1.0 - h[b])
        improved = False
        for frac in [0.7, 0.35, 0.15, 0.06, 0.02]:
            d = delta_max * frac
            if d < 1e-14:
                break
            h[a] -= d; h[b] += d
            cn = _c5(h, dx)
            if cn < best:
                best = cn; improved = True; stuck = 0
                break
            h[a] += d; h[b] -= d
        if not improved:
            improved2 = False
            for _ in range(10):
                a2 = rng.randint(n); b2 = rng.randint(n)
                if a2 == b2:
                    continue
                sign = 1.0 if rng.random() < 0.5 else -1.0
                d = rng.uniform(0.002, 0.015) * sign
                if 0 <= h[a2] + d <= 1 and 0 <= h[b2] - d <= 1:
                    h[a2] += d; h[b2] -= d
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; improved2 = True; stuck = 0; break
                    h[a2] -= d; h[b2] += d
            if not improved2:
                stuck += 1
                if stuck > 80:
                    break
    return _repair(h, tgt), best

def _anneal(h, dx, t_end, rng, iters=300000):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, dx)
    best_h = h.copy()
    T = 0.03
    for it in range(iters):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.06, 0.06)
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d; h[j] -= d
            cn = _c5(h, dx)
            if cn < best:
                best = cn; best_h = h.copy()
            elif rng.random() < np.exp(-(cn - best) / max(T, 1e-9)):
                pass
            else:
                h[i] -= d; h[j] += d
        T *= 0.99995
        if T < 1e-7:
            break
    return _repair(best_h, tgt), best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _multi_slsqp(h, dx, t_end, alphas, flat_beta=0.0):
    best_h, best_c = h.copy(), _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc, c = _slsqp(best_h, dx, a, 250, flat_beta)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 4), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(1.5, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # P1: SLSQP cleanup
    h, c = _multi_slsqp(best_h, dx, t0 + 50, [500, 2000, 8000])
    if c < best_c5:
        best_h, best_c5 = h, c

    # P2: Exact greedy local search
    h, c = _greedy_local(best_h, dx, t0 + 180, rng)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P3: SLSQP with flatness penalty
    h, c = _multi_slsqp(best_h, dx, t0 + 270, [5000, 20000], flat_beta=0.1)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P4: Greedy again
    h, c = _greedy_local(best_h, dx, t0 + 380, rng)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P5: Annealing for diversity
    h, c = _anneal(best_h, dx, t0 + 520, rng, 250000)
    if c < best_c5:
        best_h, best_c5 = h, c
    h, c = _greedy_local(best_h, dx, t0 + 600, rng)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P6: Multi-start perturbations
    t_end6 = t0 + 730
    for r in range(8):
        if time.time() > t_end6:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _greedy_local(hp, dx, min(t_end6 - 5, time.time() + 12), rng)
        hp, c = _slsqp(hp, dx, 10000, 200)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # P7: 2x resolution
    if budget_s - (time.time() - t0) > 180:
        n2, dx2 = 2 * n, 1.0 / n
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _multi_slsqp(h2, dx2, min(t0 + 780, time.time() + 70), [2000, 10000, 50000])
        h2, c2 = _greedy_local(h2, dx2, min(t0 + 850, time.time() + 50), rng)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # P8: Final SLSQP + greedy on best
    nb, dxb = best_n, 2.0 / best_n
    h, c = _multi_slsqp(best_h, dxb, t0 + 930, [30000, 100000])
    if c < best_c5:
        best_h, best_c5 = h, c
    h, c = _greedy_local(best_h, dxb, t0 + 1020, rng)
    if c < best_c5:
        best_h, best_c5 = h, c

    # P9: Final SLSQP + greedy
    if budget_s - (time.time() - t0) > 40:
        h, c = _slsqp(best_h, dxb, 200000, 200)
        if c < best_c5:
            best_h, best_c5 = h, c
    if budget_s - (time.time() - t0) > 25:
        best_h, best_c5 = _greedy_local(best_h, dxb, t0 + budget_s - 3, rng)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
