-d*(1-h[i+s]) + h[i]*d = d*(h[i] - 1 + h[i+s]) = d*(h[i] + h[i+s] - 1)

When h[i] + h[i+s] < 1, this is negative, so O(s) decreases. 

But this might increase O(s') for other lags s'. That's the tradeoff.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _slsqp(h, dx, alpha, maxiter=500):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _directed(h, dx, rng, t_end, top_k=5):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            for _ in range(12):
                a = rng.randint(m)
                i1, i2 = a, a + s
                if h[i1] + h[i2] < 0.95:
                    d = rng.uniform(0.004, 0.02)
                    h[i1] -= d; h[i2] += d
                    if 0 <= h[i1] and h[i2] <= 1:
                        cn = _c5(h, dx)
                        if cn < best:
                            best = cn
                        else:
                            h[i1] += d; h[i2] -= d
                    else:
                        h[i1] += d; h[i2] -= d
            for _ in range(8):
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = rng.uniform(0.004, 0.02)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] and h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
        for _ in range(15):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.01, 0.01)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 4e-4, 1.5e-4, 6e-5, 2e-5, 8e-6]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(4, n // 6), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        for i in range(n):
            if rng.random() < 0.3:
                hp[i] = 1.0 - hp[i]
    else:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1, 8)
        hp += 0.04 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    return _repair(np.clip(hp, 0, 1), tgt)

def _binary_nudge(h, rng, strength=0.08):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    for i in range(n):
        if hp[i] < 0.4:
            hp[i] = max(0.0, hp[i] - strength * rng.random())
        elif hp[i] > 0.6:
            hp[i] = min(1.0, hp[i] + strength * rng.random())
    return _repair(hp, tgt)
