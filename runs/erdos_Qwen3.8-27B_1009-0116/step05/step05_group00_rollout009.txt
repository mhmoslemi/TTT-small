import numpy as np
import time

def _project(h, target):
    n = len(h)
    h = np.clip(h, -3.0, 3.0)
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def _topk_grad(h, k_list, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_list:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g * dx / len(k_list)

def _pg_optimize(h, n, top_k, max_iter, t0, budget):
    c5 = _eval(h)
    step = 0.08
    prev_g = None
    prev_s = None
    for it in range(max_iter):
        if time.time() - t0 > budget - 60:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        g = _topk_grad(h, top_idx, n)
        # BB stepsize
        if prev_g is not None:
            y = g - prev_g
            dy = y @ y
            sd = prev_s @ y
            if dy > 1e-15 and abs(sd) > 1e-15 and sd > 0:
                step_bb = sd / dy
                step = 0.3 * step + 0.7 * min(max(step_bb, 0.001), 1.0)
        h_new = _project(h - step * g, n * 0.5)
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-13:
            prev_s = h_new - h
            prev_g = g
            h = h_new
            c5 = c5_new
            step = min(step * 1.05, 1.0)

            if step < 0.0005:

    return h, c5

def _coord_descend(h, n, n_iter, rng, scale=0.02):
    c5 = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = hi - d, hj + d
    return h, c5

    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)

    """Local search on bang-bang solutions: swap a 0 and a 1."""
    h = (h > 0.5).astype(float)
    h = _project(h, n * 0.5)
    c5 = _eval(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

            break
        oi = rng.choice(ones)
        zi = rng.choice(zeros)

        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
        else:
            h[oi], h[zi] = 1.0, 0.0
    return h, c5
