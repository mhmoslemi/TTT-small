import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    dx = 2.0 / n
    c5c = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(max_sweeps):
        any_improved = False
        n_pairs = min(200, n * n // 4)
        for _ in range(n_pairs):
            j = rng.randint(n)
            m = rng.randint(n)
            if j == m:
                continue
            # Compute eff[k] for shift s = k-(n-1)
            eff = np.zeros(2 * n - 1)
            for idx_s in range(2 * n - 1):
                s = idx_s - (n - 1)
                e = 0.0
                if 0 <= m + s < n: e += h[m + s]
                if 0 <= m - s < n: e += h[m - s]
                if 0 <= j + s < n: e -= h[j + s]
                if 0 <= j - s < n: e -= h[j - s]
                eff[idx_s] = e
            tmin = max(-h[j], h[m] - 1.0)
            tmax = min(1.0 - h[j], h[m])
            if tmin >= tmax - 1e-12:
                continue
            # Candidate t values
            top_idx = np.argsort(c)[::-1][:n_top]
            cand_t = np.array([tmin, tmax, 0.0])
            for ii in range(n_top):
                for jj in range(ii + 1, n_top):
                    ki, kj = top_idx[ii], top_idx[jj]
                    de = eff[ki] - eff[kj]

            best_t = 0.0

def _subgrad(h, max_iters, rng):

    dx = 2.0 / n
    c5c = _eval(h)
    momentum = np.zeros(n)
    lr = 0.02
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        # Soft top-k
        top_k = max(3, n // 10)
        thresh = np.sort(c)[::-1][top_k - 1]
        w = np.maximum(c - thresh, 0)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= wsum
        # Gradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            s = (n - 1) - k  # shift

            if s >= 0:

                g[i1] += wk * (1.0 - h[i1 + s])

            else:
                sp = -s
                g[sp:] += wk * (1.0 - h[:n - sp])

        # Adam-like update

        h_new = _proj(h - lr * momentum, n * 0.5)

        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:

    return h, c5c

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

    target = n / 2.0

    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:

                    diff += a
