        res = minimize(obj_fun, h, jac=grad_fun, args=(),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 1, n, endpoint=False) + 0.5 / n
    x_new = np.linspace(0, 1, n * factor, endpoint=False) + 0.5 / (n * factor)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP at n=280
    for alpha in [200, 1000, 5000]:
        if time.time() - t0 > 60:
            break
        h_new = _slsqp_lse(best_h, alpha, 800, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Upsample to 2x
    if time.time() - t0 < budget_s - 300:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        c5_hi = _eval(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n1

    # Phase 3: Nesterov at higher resolution
    nc = best_n
    deadline3 = t0 + budget_s * 0.55
    for alpha, step0, iters in [(300, 0.003, 8000), (1000, 0.001, 6000), (5000, 0.0003, 4000), (20000, 0.0001, 3000)]:
        if time.time() > deadline3:
            break
        h_head = best_h.copy()
        h_res, obj_res = _nesterov_opt(h_head, alpha, iters, 0.9, step0, deadline3)
        c5 = _eval(h_res)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res.copy()
            best_n = len(h_res)

    # Phase 4: SLSQP polish
    deadline4 = t0 + budget_s * 0.75
    for alpha, iters in [(50000, 2000), (200000, 1500), (500000, 1000)]:
        if time.time() > deadline4:
            break
        h_new = _slsqp_lse(best_h, alpha, iters, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    # Phase 5: More Nesterov with fine alpha
    deadline5 = t0 + budget_s * 0.85
    for alpha, step0, iters in [(100000, 0.00005, 5000), (500000, 0.00002, 3000)]:
        if time.time() > deadline5:
            break
        h_res, _ = _nesterov_opt(best_h, alpha, iters, 0.95, step0, deadline5)
        c5 = _eval(h_res)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res.copy()

    # Phase 6: Coordinate descent
    h = best_h.copy()
    nc = len(h)
    c5c = _eval(h)
    deadline6 = t0 + budget_s - 30
    for scale in [0.002, 0.0005, 0.0001, 0.00003]:
        if time.time() > deadline6:
            break
        for _it in range(30000):
            if time.time() > deadline6:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    # Phase 7: Final SLSQP polish
    if time.time() < t0 + budget_s - 10:
        h_new = _slsqp_lse(best_h, 500000, 2000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 8: Ultra-fine
    h = best_h.copy()
    nc = len(h)
    c5c = _eval(h)
    for _it in range(50000):
        if time.time() > t0 + budget_s - 2:
            break
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
