import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h):
    """Gradient of c_k with respect to h, before multiplying by dx."""
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        if s > 0:
            i2 = np.arange(n - s)
            g[i2] -= h[i2 + s]
    else:
        sp = -s
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(n - sp, n)
        g[i2] -= h[i2 - sp]
    return g

def _subgrad_max(h, topm):
    """Subgradient of max_k c_k: average gradient of top m lags."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[::-1][:topm]
    g = np.zeros(n)
    for k in top_idx:
        g += _grad_lag(k, h)
    g /= topm
    g *= (2.0 / n)  # multiply by dx
    return g, c.max() * (2.0 / n)

def _projected_descent(h, topm, max_iter, step0, cool=0.98):
    """Projected subgradient descent with backtracking."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    for it in range(max_iter):
        if time.time() - _t0 > _budget - 15:

        g, _ = _subgrad_max(h, topm)
        # Backtracking line search
        found = False
        for _bt in range(40):

            f_new = _eval(h_new)
            if f_new < f - 1e-10:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:

    return h, f

def _equioscillation_obj(h, topm, alpha):
    """Objective: max(c) + alpha * var(top_m lags)"""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    top = np.sort(c)[::-1][:topm]
    return c.max() + alpha * top.var()

    """One active-set step: linearize and solve LP."""
    n = len(h)
    dx = 2.0 / n
    c_all = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c_all)[::-1][:topm]
    c_active = c_all[top_idx]
    
    # Build Jacobian rows for active lags
    J = np.zeros((topm, n))
    for i, k in enumerate(top_idx):
        J[i] = _grad_lag(k, h) * dx

    #   δ ≥ -h  → -δ ≤ h  → [[-I, 0]] @ [δ; t] ≤ h  (n inequalities)

    A_eq[0, :n] = 1.0

        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_step(h, topm, relax=0.0):
    n = len(h)
    dx = 2.0 / n
    c_all = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c_all)[::-1][:topm]
    c_act = c_all[top_idx] - relax
    J = np.zeros((topm, n))
    for i, k in enumerate(top_idx):
        J[i] = _grad_lag(k, h) * dx
    c_obj = np.zeros(n + 1); c_obj[-1] = 1.0
    A_ub = np.zeros((topm + 2 * n, n + 1))
    b_ub = np.zeros(topm + 2 * n)
    A_ub[:topm, :n] = J
    A_ub[:topm, n] = -1.0
    b_ub[:topm] = -c_act
    A_ub[topm:topm+n, :n] = -np.eye(n)
    b_ub[topm:topm+n] = h
    A_ub[topm+n:topm+2*n, :n] = np.eye(n)
    b_ub[topm+n:topm+2*n] = 1.0 - h
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)]*(n+1), method='highs')
    if res.success:
        return res.x[:n]
    return None

def _lp_descent(h, topm, max_iter, relax_init=0.0):
    n = len(h)
    h = h.copy(); f = _eval(h); relax = relax_init
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 12: break
        delta = _lp_step(h, topm, relax)
        if delta is None:
            relax += 1e-4
            if relax > 0.3: break
            continue
        found = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _repair(h + frac * delta, n)
            f_new = _eval(h_new)
            if f_new < f - 1e-14:
                h = h_new; f = f_new
                relax = max(0.0, relax - 5e-5)
                found = True; break
        if not found: break
    return h, f

def _slsqp(h, alpha, maxiter, ftol=1e-18):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max())))/alpha
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for k in np.where(w > 1e-10)[0]:
            g += w[k] * _grad_lag(k, x) * (2.0/n)
        return g
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n/2,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _downsample(h, factor):
    n = len(h)
    nk = n // factor
    hd = np.array([h[i*factor:(i+1)*factor].mean() for i in range(nk)])
    return _repair(hd, nk)

def _pairwise(h, n_iter, scale, sv):
    n = len(h); h = h.copy(); f = _eval(h)
    rng = np.random.RandomState(sv)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 8: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            if _eval(h) < f - 1e-15: f = _eval(h)
            else: h[i], h[j] = ni-d, nj+d
    return h, f

def _subgrad_descent(h, topm, max_iter, step0=0.003):
    n = len(h); h = h.copy(); f = _eval(h); step = step0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 10: break
        c = np.correlate(h, 1.0-h, mode='full')
        ti = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in ti: g += _grad_lag(k, h)
        g /= topm; g *= 2.0/n
        found = False
        for _b in range(35):
            hn = _proj_sum(h - step*g, n*0.5)
            fn = _eval(hn)
            if fn < f - 1e-12:
                h = hn; f = fn; found = True; break
            step *= 0.5
        if not found: break
        step = min(step*1.03, step0)
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time(); _budget = budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h); best_n = n0
    def _el(): return time.time() - _t0

    # Phase 1: SLSQP at n0
    if _el() < 40:
        best_h = _slsqp(best_h, 8000, 600)
        best_c5 = _eval(best_h)

    # Phase 2: Subgradient at n0
    if _el() < _budget - 300:
        for tm in [4, 7]:
            if _el() > _budget - 280: break
            hn, fc = _subgrad_descent(best_h, tm, 1500, 0.004)
            if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase 3: Upsample to 2x, SLSQP
    if _el() < _budget - 250:
        n1 = n0 * 2
        h2 = _upsample(best_h, 2)
        h2 = _slsqp(h2, 15000, 1000)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2; best_n = n1

    nc = len(best_h)

    # Phase 4: LP active-set descent
    if _el() < _budget - 200:
        for tm in [4, 6, 10]:
            if _el() > _budget - 170: break
            hn, fc = _lp_descent(best_h, tm, 1200)
            if fc < best_c5: best_c5 = fc; best_h = hn

    # Phase 5: SLSQP high-alpha
    if _el() < _budget - 120:
        hn = _slsqp(best_h, 80000, 1200)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5 = c5; best_h = hn
        if _el() < _budget - 80:
            hn = _slsqp(best_h, 200000, 800)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn

    # Phase 6: More LP descent with varied topm
    if _el() < _budget - 80:
        for tm in [5, 8, 12, 16]:
            if _el() > _budget - 50: break
            hn, fc = _lp_descent(best_h, tm, 800)
            if fc < best_c5: best_c5 = fc; best_h = hn

    # Phase 7: Multi-start perturbations
    nc = len(best_h)
    if _el() < _budget - 50:
        for r in range(6):
            if _el() > _budget - 30: break
            hp = best_h + 0.004 * rng.randn(nc)
            hp = _repair(hp, nc)
            hn = _slsqp(hp, 40000, 500)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn
            if _el() < _budget - 25:
                hn2, fc = _lp_descent(best_h, 6, 500)
                if fc < best_c5: best_c5 = fc; best_h = hn2

    # Phase 8: Final LP + subgradient polish
    if _el() < _budget - 25:
        hn, fc = _lp_descent(best_h, 5, 1000)
        if fc < best_c5: best_c5 = fc; best_h = hn
    if _el() < _budget - 15:
        hn, fc = _subgrad_descent(best_h, 6, 1000, 0.002)
        if fc < best_c5: best_c5 = fc; best_h = hn

    # Phase 9: SLSQP final
    if _el() < _budget - 12:
        hn = _slsqp(best_h, 300000, 600, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5 = c5; best_h = hn

    # Phase 10: Pairwise fine-tune
    for sc, ni in [(0.002, 4000), (0.0005, 6000), (0.0001, 8000)]:
        if _el() > _budget - 5: break
        hn, fc = _pairwise(best_h, ni, sc, seed + int(sc*1e7))
        if fc < best_c5: best_c5 = fc; best_h = hn

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
