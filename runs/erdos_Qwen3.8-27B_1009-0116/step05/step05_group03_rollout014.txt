                d = rng.uniform(0.001, 0.05)
                i1, i2 = a+s, a
                if h[i1] < d or h[i2] > 1-d: continue
                h[i1] -= d; h[i2] += d
                cn = _c5(h, dx)
                if cn < best:
                    best = cn; improved = True
                else:
                    h[i1] += d; h[i2] -= d
                # Also try reverse
                if h[i1] > d and h[i2] < 1-d:
                    h[i1] += d; h[i2] -= d
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; improved = True
                    else:
                        h[i1] -= d; h[i2] += d
        
        # Random pair moves
        for _ in range(100):
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = rng.uniform(-0.03, 0.03)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn; improved = True
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
        
        if not improved:
            # Try smaller steps
            for _ in range(200):
                i, j = rng.randint(n), rng.randint(n)
                if i == j: continue
                d = rng.uniform(-0.005, 0.005)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; improved = True
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
            if not improved:
                # Accept a small worsening move (simulated annealing)
                i, j = rng.randint(n), rng.randint(n)
                d = rng.uniform(-0.02, 0.02)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best * 1.001:  # accept small worsening
                        pass  # keep the move
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
    
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _smoobj(h, dx, alpha):
    n = len(h)
    tgt = n / 2.0
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs(h, dx, alpha, pen, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha + pen * (x.sum() - tgt) ** 2
    try:
        r = minimize(obj, h, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coord_max(h, dx, t_end, rng, scales=None):
    n = len(h)
    h = h.copy()
    best = _c5(h, dx)
    if scales is None:
        scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]
    c = _corr(h)
    for sc in scales:
        while time.time() < t_end:
            c = _corr(h)
            top = np.argsort(c)[::-1][:4]
            improved = False
            for ki in top:
                s = abs(int(ki) - (n - 1))
                if s == 0:
                    continue
                m = n - s
                for _ in range(80):
                    a = rng.randint(m)
                    i1, i2 = a + s, a
                    d = sc * rng.uniform(0.3, 1.0)
                    for sgn in [1, -1]:
                        h[i1] -= sgn * d
                        h[i2] += sgn * d
                        if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                            cn = _c5(h, dx)
                            if cn < best:
                                best = cn; improved = True
                            else:
                                h[i1] += sgn * d; h[i2] -= sgn * d
                        else:
                            h[i1] += sgn * d; h[i2] -= sgn * d
            for _ in range(40):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = sc * (2.0 * rng.random() - 1.0)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; improved = True
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
            if not improved and sc <= 1e-4:
                break
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    x = np.linspace(0, 2, n)
    hp = h.copy()

    elif mode % 7 == 3:

    elif mode % 7 == 4:

        hp += 0.015 * rng.normal(0, 1, n) * np.sin(np.pi * x)
    elif mode % 7 == 5:

    return _repair(np.clip(hp, 0, 1), tgt)
