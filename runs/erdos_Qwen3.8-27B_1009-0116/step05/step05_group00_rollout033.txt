import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            od = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for idx in od:
                if d > 0:
                    a = min(d, 1.0 - h[idx])
                    h[idx] += a; d -= a
                else:
                    a = min(-d, h[idx])
                    h[idx] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _ov(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)

        if w[k] < 1e-12:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _sm_weights(ov, alpha):
    m = ov.max()
    s = np.clip(alpha * (ov - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _bb_descent(h, t_end, alpha_start, alpha_end, n_iter):
    n = len(h)
    c5 = _eval(h)
    prev_h = h.copy()
    prev_g = None
    for it in range(n_iter):
        if time.time() > t_end:
            break
        frac = it / max(n_iter, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        ov = _ov(h)
        w = _sm_weights(ov, alpha)
        g = _grad(h, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        if prev_g is not None:
            sv = h - prev_h
            yv = g - prev_g
            sy = np.dot(sv, yv)
            yy = np.dot(yv, yv)
            ss = np.dot(sv, sv)
            if yy > 1e-16:
                if sy > 1e-8:
                    step = ss / sy
                else:
                    step = 1.0 / (gn + 1e-12)
            else:
                step = 1.0 / (gn + 1e-12)
        else:
            step = 1.0 / (gn + 1e-12)
        step = np.clip(step, 1e-10, 10.0)
        accepted = False
        for _ in range(6):
            h_new = _repair(h - step * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5 * (1 - 1e-10) - 1e-15:
                h = h_new
                c5 = c5_new
                accepted = True
                break
            step *= 0.5
        prev_h = h.copy()
        prev_g = g
    return h, c5

def _topk_descent(h, t_end, k, n_iter):
    n = len(h)
    c5 = _eval(h)
    # Find current worst k
    for it in range(n_iter):
        if time.time() > t_end:
            break
        ov = _ov(h)
        top_idx = np.argsort(ov)[-k:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / k
        g = _grad(h, w)
        gn = np.linalg.norm(g)

            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-15:

                c5 = c5_new
                break
    return h, c5

def _coord_descent(h, t_end, n_iter, scale):
    n = len(h)
    c5 = _eval(h)
    rng = np.random.RandomState(0)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = hi - d, hj + d
    return h, c5

    xn = np.linspace(0, 2.0, n_new, endpoint=False)

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            od = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for idx in od:
                if d > 0:
                    a = min(d, 1.0 - h[idx])
                    h[idx] += a; d -= a
                else:
                    a = min(-d, h[idx])
                    h[idx] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _ov(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _sm_w(ov, alpha):
    m = ov.max()
    s = np.clip(alpha * (ov - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _bb_descent(h, t_end, a_lo, a_hi, n_iter):
    n = len(h)
    c5 = _eval(h)
    ph = h.copy()
    pg = None
    for it in range(n_iter):
        if time.time() > t_end:
            break
        fr = it / max(n_iter, 1)
        al = a_lo * (a_hi / a_lo) ** fr
        ov = _ov(h)
        w = _sm_w(ov, al)
        g = _grad(h, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        if pg is not None:
            sv = h - ph
            yv = g - pg
            sy = np.dot(sv, yv)
            yy = np.dot(yv, yv)
            ss = np.dot(sv, sv)
            if yy > 1e-16 and sy > 1e-10:
                st = ss / sy
            elif yy > 1e-16:
                st = 1.0 / (gn + 1e-12)
            else:
                st = 1.0 / (gn + 1e-12)
        else:
            st = 1.0 / (gn + 1e-12)
        st = np.clip(st, 1e-10, 10.0)
        for _ in range(8):
            hn = _repair(h - st * g, n)
            c5n = _eval(hn)
            if c5n < c5 - 1e-15:
                h = hn
                c5 = c5n
                break
            st *= 0.4
        ph = h.copy()
        pg = g
    return h, c5

def _topk_desc(h, t_end, kk, n_iter):
    n = len(h)
    c5 = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        ov = _ov(h)
        ti = np.argsort(ov)[-kk:]
        w = np.zeros(2 * n - 1)
        w[ti] = 1.0 / kk
        g = _grad(h, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for st in [5e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
            hn = _repair(h - st * g, n)
            c5n = _eval(hn)
            if c5n < c5 - 1e-15:
                h = hn
                c5 = c5n
                break
    return h, c5

def _coord_desc(h, t_end, n_iter, sc, seed):
    n = len(h)
    c5 = _eval(h)
    rng = np.random.RandomState(seed)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = hi - d, hj + d
    return h, c5

def _resample(h, nn):
    no = len(h)
    xo = np.linspace(0.0, 2.0, no, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return np.interp(xn, xo, h)

def _shake_reopt(h, rng, t_end, n_iter):
    n = len(h)
    c5 = _eval(h)
    for _ in range(6):
        if time.time() > t_end:
            break
        h2 = h + 0.01 * rng.randn(n)
        h2 = _repair(h2, n)
        h2, c52 = _bb_descent(h2, t_end, 500, 50000, n_iter)
        if c52 < c5:
            h, c5 = h2, c52
    return h, c5
