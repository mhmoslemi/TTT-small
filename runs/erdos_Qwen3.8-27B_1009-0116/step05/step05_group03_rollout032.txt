import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_lse(h, alpha, top_k=20):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 0)
    p = np.exp(s)
    p /= p.sum()
    g = np.zeros(n)
    order = np.argsort(p)[::-1][:top_k]
    for j in order:
        w = p[j]
        if w < 1e-12:
            break
        s_val = abs(int(j - (n - 1)))
        if s_val == 0:
            g += w * (1.0 - 2.0 * h)
        else:
            m2 = n - s_val
            g[:m2] += w * (1.0 - h[s_val:])
            g[s_val:] -= w * h[:m2]
    return g

def _slsqp_solve(h_init, dx, alpha, maxiter=400):
    n = len(h_init)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n

    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 0)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha

    def obj_grad(x):
        return _grad_lse(x, alpha, top_k=30)

    def eq(x):
        return x.sum() - tgt

    def eq_jac(x):
        return np.ones(n)

    try:
        r = minimize(obj, h_init, method='SLSQP', bounds=bds,
                     jac=obj_grad,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        try:
            r = minimize(obj, h_init, method='SLSQP', bounds=bds,
                         constraints=[{'type': 'eq', 'fun': eq}],
                         options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
            hc = _repair(r.x, tgt)
            return hc, _c5(hc, dx)
        except Exception:
            return h_init, _c5(h_init, dx)

def _psgd(h, dx, t_end, lr0=0.006, top_k=6):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = lr0
    stalled = 0
    while time.time() < t_end:
        g = _grad_lse(h, 30.0, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _repair(np.clip(h - lr * g / gn, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best - 1e-15:
            best = c_new
            h = h_new
            lr = min(lr * 1.015, 0.04)
            stalled = 0
        else:
            lr *= 0.88
            stalled += 1
            if lr < 1e-6 or stalled > 800:
                break
    return h, best

def _coord_desc(h, dx, rng, t_end, n_per_scale=4000):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    c = _corr(h)
    top_lags = np.argsort(c)[::-1][:10]
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]
    for sc in scales:
        for _ in range(n_per_scale):
            if time.time() > t_end:
                return h, best
            ki = top_lags[rng.randint(len(top_lags))]
            s = abs(int(ki - (n - 1)))
            if s == 0:
                i = rng.randint(n)
                d = sc * (2 * rng.random() - 1)
                h[i] += d
                if 0 <= h[i] <= 1:
                    if _c5(h, dx) < best:
                        best = _c5(h, dx)
                    else:
                        h[i] -= d
                else:
                    h[i] -= d
            else:
                m = n - s
                a = rng.randint(m)
                i1, i2 = a + s, a
                d = sc * (2 * rng.random() - 1)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    if _c5(h, dx) < best:
                        best = _c5(h, dx)
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
            if _ % 200 == 199:
                c = _corr(h)
                top_lags = np.argsort(c)[::-1][:10]
    return h, best

def _interp_rt(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    x = np.linspace(0, 2, n)
    if mode == 0:
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(2, 7) * x + rng.uniform(0, 6.28))
    elif mode == 1:
        nb = rng.randint(3, 8)
        bs = n // nb
        for b in range(nb):
            s, e = b * bs, min((b + 1) * bs, n)
            hp[s:e] += 0.05 * (2 * rng.random(e - s) - 1)
    elif mode == 2:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] += 0.08 * (2 * rng.random(len(idx)) - 1)
    elif mode == 3:
        s = rng.randint(0, n // 3)
        ln = rng.randint(n // 10, n // 5)
        hp[s:s + ln] = 1.0 - hp[s:s + ln]
    else:
        hp += 0.03 * (2 * rng.random(n) - 1)
    return _repair(np.clip(hp, 0, 1), tgt)
