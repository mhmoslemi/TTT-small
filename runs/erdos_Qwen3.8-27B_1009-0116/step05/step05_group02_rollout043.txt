        else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _grad_lse(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]
    return g, c

def _grad_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k = c.argmax()
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:cnt]
    else:
        sp = -sh
        cnt = n - sp
        g[sp:] += (1.0 - h[:cnt])
        g[:cnt] -= h[sp:]
    return g

def _pgd_loop(h, t_start, t_end, alphas, iters):
    n = len(h)
    for alpha in alphas:
        lr = 0.5 / np.sqrt(alpha + 1.0)
        for _ in range(iters):
            if time.time() - t_start > t_end:
                return h
            g, _ = _grad_lse(h, alpha)
            for trial in range(10):
                h_new = _repair(h - lr * g, n)
                if _eval(h_new) < _eval(h) + 1e-16:
                    h = h_new
                    break
                lr *= 0.5
            if lr < 1e-14:
                break
    return h

def _slsqp_lse(h, alpha, maxiter):
    n = len(h)
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad_fn(x):
        g, _ = _grad_lse(x, alpha)
        return g
    def sfun(x):
        return x.sum() - n * 0.5
    def sjac(x):
        return np.ones(n)
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sfun, 'jac': sjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lag_descent(h, n_rounds, t_start, t_end):
    n = len(h)
    c5 = _eval(h)
    dx = 2.0 / n
    for _r in range(n_rounds):
        if time.time() - t_start > t_end:
            break
        g = _grad_worst(h) * dx
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            h = _repair(h + 0.001 * np.random.randn(n), n)
            c5 = _eval(h)
    return h, c5

def _coord_desc(h, n_iters, scale, rng, t_start, t_end):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iters):
        if time.time() - t_start > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

def _upsample(h, factor):
    return np.repeat(h, factor)
