    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s = s - s.max()
    obj = m + np.log(np.sum(np.exp(s))) / alpha
    ssum = x.sum() - n / 2.0
    return obj + pen * ssum * ssum

def _lse_jac(x, alpha, n, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    ssum = x.sum() - n / 2.0
    g += 2.0 * pen * ssum
    return g

def _lbfgs(h0, n, alpha, maxiter, pen):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_lse_obj, h0, jac=_lse_jac, args=(alpha, n, pen),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 40})
    return _repair(res.x, n)

def _lag_grad(h, k_w, n):
    dx = 2.0 / n
    sh = (n - 1) - k_w
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _upsample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, n_old), h)

def _upsample_block(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 0.9999)[0]
        else:
            idx = np.where(h > 0.0001)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    return h

def _lse_grad(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400.0, 400.0)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    obj = m + np.log(w.sum() / w.sum()) / alpha  # simplified: log(1)=0, use logsumexp
    sm = s.max()
    logsum = sm + np.log(np.sum(np.exp(s - sm)))
    obj = m + logsum / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
            else:
                g[i1] -= wk * h[i1]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return obj, g

def _proj_grad_opt(h, n, alpha, max_iter, t0, budget_s, reserve):
    h = h.copy()
    step = 0.02
    best_val = _eval(h)
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > budget_s - reserve:
            break
        _, g = _lse_grad(h, alpha, n)
        improved = False
        for trial in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = _repair(h - trial * g, n)
            val = _eval(h_new)
            if val < best_val - 1e-15:
                h = h_new
                best_val = val
                step = min(step * 1.15, 0.08)
                stall = 0
                improved = True
                break
        if not improved:
            step *= 0.6
            stall += 1
            if stall > 30:
                break
            if stall % 10 == 0:
                h = _repair(h + 0.002 * np.random.randn(n), n)
                best_val = _eval(h)
    return h, best_val

def _lag_grad(h, k_w, n):
    dx = 2.0 / n
    sh = (n - 1) - k_w
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
        else:
            g[i1] -= h[i1]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _lbfgs_opt(h0, n, alpha, maxiter, pen, t0, budget_s, reserve):
    def _obj(x):
        o, _ = _lse_grad(x, alpha, n)
        ss = x.sum() - n / 2.0
        return o + pen * ss * ss
    def _jac(x):
        _, g = _lse_grad(x, alpha, n)
        ss = x.sum() - n / 2.0
        return g + 2.0 * pen * ss
    try:
        res = minimize(_obj, h0, jac=_jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 30})
        if time.time() - t0 < budget_s - reserve:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h0, n)

def _up_lin(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, n_old), h)

def _up_block(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Projected gradient at n0 with annealing alpha
    for alpha in [100, 400, 1500, 6000, 25000, 100000, 400000]:
        if time.time() - t0 > budget_s - 350:
            break
        h_new, _ = _proj_grad_opt(best_h, n0, alpha, 8000, t0, budget_s, 350)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Multi-resolution
    res_list = sorted(set([n0 * 2, n0 * 3, 420, 560, 700]))
    res_list = [r for r in res_list if 140 <= r <= 900 and r % 2 == 0]
    for tn in res_list:
        if time.time() - t0 > budget_s - 250:
            break
        if tn >= best_n and best_n > n0:
            continue
        for method in [0, 1]:
            if time.time() - t0 > budget_s - 200:
                break
            if method == 0:
                h_up = _up_lin(best_h, tn)
            else:
                fac = max(1, tn // n0)
                h_up = _up_block(best_h, fac)
                if len(h_up) != tn:
                    h_up = _up_lin(h_up, tn)
            h_up = _repair(h_up, tn)
            for alpha in [500, 5000, 50000, 500000]:
                if time.time() - t0 > budget_s - 180:
                    break
                h_up, _ = _proj_grad_opt(h_up, tn, alpha, 6000, t0, budget_s, 180)
                c5 = _eval(h_up)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_up.copy()
                    best_n = tn

    # Phase 3: Lag-targeted optimization
    if time.time() - t0 < budget_s - 130:
        h = best_h.copy()
        nc = best_n
        c5c = _eval(h)
        for _it in range(800):
            if time.time() - t0 > budget_s - 100:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            k_w = int(c.argmax())
            g = _lag_grad(h, k_w, nc)
            improved = False
            for st in [0.08, 0.04, 0.02, 0.01, 0.004, 0.001, 0.0004]:
                h_new = _repair(h - st * g, nc)
                v = _eval(h_new)
                if v < c5c - 1e-15:
                    h = h_new
                    c5c = v
                    improved = True
                    if v < best_c5:
                        best_c5 = v
                        best_h = h.copy()
                    break
            if not improved:
                h = _repair(h + 0.004 * rng.randn(nc), nc)
                c5c = _eval(h)

    # Phase 4: Multi-start projected gradient
    if time.time() - t0 < budget_s - 80:
        nc = best_n
        for _r in range(20):
            if time.time() - t0 > budget_s - 55:
                break
            strat = _r % 5
            if strat == 0:
                hp = _repair(best_h + 0.012 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                bl = rng.randint(4, max(5, nc // 6))
                st = rng.randint(0, nc - bl)
                hp[st:st + bl] = 1.0 - hp[st:st + bl]
                hp = _repair(hp, nc)
            elif strat == 2:
                f = rng.randint(2, nc // 4 + 1)
                amp = 0.008 * rng.choice([-1.0, 1.0])
                t_arr = np.arange(nc)
                hp = _repair(best_h + amp * np.sin(2.0 * np.pi * f * t_arr / nc + rng.rand() * 6.28), nc)
            elif strat == 3:
                hp = _repair(1.0 - best_h + 0.008 * rng.randn(nc), nc)
            else:
                hp = _repair(np.clip(best_h * rng.uniform(0.85, 1.15) + 0.005 * rng.randn(nc), 0, 1), nc)
            for alpha in [5000, 50000]:
                if time.time() - t0 > budget_s - 55:
                    break
                h_new, _ = _proj_grad_opt(hp, nc, alpha, 4000, t0, budget_s, 55)
                c5 = _eval(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

    # Phase 5: L-BFGS polish at best resolution
    if time.time() - t0 < budget_s - 30:
        h_new = _lbfgs_opt(best_h, best_n, 200000, 2000, 1e4, t0, budget_s, 30)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 6: Fine coordinate descent
    if time.time() - t0 < budget_s - 10:
        h = best_h.copy()
        nc = best_n
        c5c = _eval(h)
        for scale in [0.002, 0.0005, 0.0001, 3e-5, 1e-5]:
            if time.time() - t0 > budget_s - 8:
                break
            n_iter = min(20000, int((budget_s - 8 - (time.time() - t0)) * 400))
            for _it in range(n_iter):
                if time.time() - t0 > budget_s - 8:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni = h[i] + d
                nj = h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 7: Final projected gradient micro-polish
    if time.time() - t0 < budget_s - 3:
        h_new, _ = _proj_grad_opt(best_h, best_n, 2000000, 5000, t0, budget_s, 3)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
