def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

                if diff > 0:

                    diff -= a

                    diff += a

def _eval(h):
