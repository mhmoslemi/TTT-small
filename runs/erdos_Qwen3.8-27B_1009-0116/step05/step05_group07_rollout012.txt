        g = np.zeros(n)
        for k in top_k:
            g += _grad_lag(k, h)
        g /= n_lags
        vel = momentum * vel + (1.0 - momentum) * g
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * vel, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-12:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            step *= 0.7
            if step < 1e-9:
                break
        else:
            step = min(step * 1.03, step0)
    return h, f

def _proj_descent_soft(h, alpha, n_iter, step0, momentum):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    vel = np.zeros(n)
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 25:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        s = np.clip(alpha * (c - c.max()), -400, 400)
        s = s - s.max()
        w = np.exp(s)
        ws = w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12 * ws:
                continue
            g += (w[k] / ws) * _grad_lag(k, h)
        vel = momentum * vel + (1.0 - momentum) * g
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * vel, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-12:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            step *= 0.7
            if step < 1e-9:
                break
        else:
            step = min(step * 1.02, step0)
    return h, f

def _cfun(x, n):
    return x.sum() - n * 0.5

def _cjac(x, n):
    return np.ones(len(x))

def _slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s = s - s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s = s - s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        g += w[k] * _grad_lag(k, x)
    return g

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(n, alpha),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.isfinite(_eval(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _pairwise_refine(h, n_steps, seed_val, scales):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    np_scales = np.array(scales)
    for _ in range(n_steps):
        if time.time() - _t0 > _budget - 10:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        scale = np_scales[rng.randint(len(np_scales))]
        d = scale * rng.randn()
        best_val = f
        best_d = 0.0
        for dd in [d, -d]:
            ni, nj = h[i] + dd, h[j] - dd
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                fn = _eval(h)
                if fn < best_val:
                    best_val = fn
                    best_d = dd
                h[i], h[j] = h[i] - dd, h[j] + dd
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            f = best_val
    return h, f

def _alternating_lag(h, n_rounds, steps_per_lag):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _r in range(n_rounds):
        if time.time() - _t0 > _budget - 20:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        k_star = c.argmax()
        step = 0.005
        for _s in range(steps_per_lag):
            if time.time() - _t0 > _budget - 20:
                break
            g = _grad_lag(k_star, h)
            found = False
            for _bt in range(35):
                h_new = _proj_sum(h - step * g, target)
                f_new = _eval(h_new)
                if f_new < f - 1e-11:
                    h = h_new
                    f = f_new
                    found = True
                    break
                step *= 0.5
            if not found:
                break
            step = min(step * 1.05, 0.005)
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    rng = np.random.RandomState(seed)

    def _tl():
        return _budget - (time.time() - _t0)

    # Phase 1: Soft projected gradient (broad optimization)
    for alpha in [5000, 20000, 80000]:
        if _tl() < 50:
            break
        hn, fn = _proj_descent_soft(best_h, alpha, 1500, 0.005, 0.1)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 2: Top-k projected gradient (focused)
    for nl in [3, 5, 10, 20, 40]:
        if _tl() < 40:
            break
        hn, fn = _proj_descent(best_h, nl, 2500, 0.005, 0.08)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 3: Alternating single-lag refinement
    if _tl() > 60:
        hn, fn = _alternating_lag(best_h, 8, 30)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 4: SLSQP polish at base res
    if _tl() > 80:
        for alpha in [50000, 200000]:
            if _tl() < 40:
                break
            hn = _slsqp_opt(best_h, alpha, 800)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 5: Upsample 2x and optimize
    if _tl() > 100:
        h2 = _upsample(best_h, 2)
        n2 = len(h2)
        for alpha in [10000, 50000]:
            if _tl() < 50:
                break
            hn, fn = _proj_descent_soft(h2, alpha, 1200, 0.004, 0.1)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()
                best_n = n2
                h2 = hn.copy()
        for nl in [5, 10, 25, 50]:
            if _tl() < 40:
                break
            hn, fn = _proj_descent(h2, nl, 1200, 0.004, 0.08)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()
                best_n = n2
                h2 = hn.copy()
        if _tl() > 50:
            hn = _slsqp_opt(best_h, 100000, 1000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 6: Pairwise fine-tuning
    if _tl() > 30:
        hn, fn = _pairwise_refine(best_h, 60000, seed, [0.004, 0.0015, 0.0005, 0.0002])
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 7: Multi-start with structured perturbations
    nc = len(best_h)
    n_trials = int(min(12, max(2, _tl() // 45)))
    for _r in range(n_trials):
        if _tl() < 35:
            break
        amp = 0.003 + 0.004 * (_r / max(1, n_trials))
        hp = best_h + amp * rng.randn(nc)
        hp = _repair(hp, nc)
        hn, fn = _proj_descent(hp, 8, 1200, 0.004, 0.05)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 8: Final polish
    if _tl() > 15:
        hn, fn = _proj_descent(best_h, 12, 1500, 0.002, 0.05)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()
        if _tl() > 8:
            hn, fn = _alternating_lag(best_h, 4, 20)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()
        if _tl() > 5:
            hn, fn = _pairwise_refine(best_h, 15000, seed + 99, [0.001, 0.0003, 0.0001])
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
