    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha, center):
    c = _corr(h) - center
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha, center):
    n = len(h)
    c = _corr(h) - center
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):

        else:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

            elif strat == 2:
                t_arr = np.arange(n0, dtype=float)
                hp = 0.5 + 0.3 * np.sin(2 * np.pi * rng.randint(2, 8) * t_arr / n0 + rng.randn() * 3)
                hp = _repair(hp, n0)
            elif strat == 3:
                hp = best_h.copy()
                nb = rng.randint(3, n0 // 5)
                st = rng.randint(0, n0 - nb)
                hp[st:st + nb] = rng.rand(nb)
                hp = _repair(hp, n0)
            else:
                hp = _repair(np.roll(best_h, rng.randint(1, n0 // 10)), n0)
            al = [20000, 80000, 200000][rng.randint(3)]
            h_new = _slsqp(hp, al, _eval(hp) - 0.005, 1200, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 5: Upsample to 2x and refine
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for al in [20000, 80000, 300000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp(h_hi, al, _eval(h_hi) - 0.005, 2000, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 180:
            h_hi = _proj_grad_descent(best_h, 400, 0.002, 5, 0.85)
            h_hi = _repair(h_hi, best_n)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h = c5, h_hi.copy()

    # Phase 6: Try downsampled (coarser) resolution
    if _el() < budget_s - 250 and best_n == n0:
        n2 = n0 // 2
        h_coarse = _repair(_downsample(best_h, 2), n2)
        for al in [5000, 30000, 100000]:
            if _el() > budget_s - 200:
                break
            h_coarse = _slsqp(h_coarse, al, _eval(h_coarse) - 0.005, 1500, 1e-18)
            c5 = _eval(h_coarse)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_coarse.copy(), n2

    # Phase 7: Fine local search (multi-scale)
    if _el() < budget_s - 80:
        for sc in [0.008, 0.003, 0.001, 0.0003]:
            if _el() > budget_s - 50:
                break
            h_tmp = best_h.copy()
            h_tmp, _ = _local_search(h_tmp, best_n, rng, 5000, sc)
            c5 = _eval(h_tmp)
            if c5 < best_c5:
                best_c5, best_h = c5, h_tmp.copy()

    # Phase 8: Final SLSQP polish with very high alpha
    if _el() < budget_s - 15:
        h_new = _slsqp(best_h, 500000, _eval(best_h) - 0.003, 3000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 9: Ultra-fine coordinate search
    if _el() < budget_s - 3:
        h_tmp = best_h.copy()
        nc = len(h_tmp)
        c5c = _eval(h_tmp)
        for _ in range(20000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni, nj = h_tmp[i] + d, h_tmp[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_tmp[i], h_tmp[j] = ni, nj
                c5n = _eval(h_tmp)
                if c5n < c5c:
                    c5c = c5n
                else:
                    h_tmp[i], h_tmp[j] = ni - d, nj + d
        if c5c < best_c5:
            best_c5, best_h = c5c, h_tmp.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
