import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    print(f"Init: C5={best_c5:.6f} n={n}")

    # P1: SLSQP escalating alpha
    t1 = t0 + 140
    for a in [500, 3000, 15000, 80000]:
        if time.time() > t1 - 20:
            break
        h, c = _slsqp(best_h, dx, a, 200)
        if c < best_c5:
            best_h, best_c5 = h, c
    print(f"P1: C5={best_c5:.6f}")

    # P2: Targeted worst-lag
    t2 = t0 + 320
    h, c = _targeted(best_h, dx, rng, t2, 6, 12)
    if c < best_c5:
        best_h, best_c5 = h, c
    print(f"P2: C5={best_c5:.6f}")

    # P3: High-alpha SLSQP
    t3 = t0 + 430
    for a in [50000, 200000, 800000]:
        if time.time() > t3 - 15:
            break
        h, c = _slsqp(best_h, dx, a, 180)
        if c < best_c5:
            best_h, best_c5 = h, c
    print(f"P3: C5={best_c5:.6f}")

    # P4: Binary-inspired search
    t4 = t0 + 480
    h, c = _binary_search(best_h, dx, rng, t4, 200)
    if c < best_c5:
        best_h, best_c5 = h, c
    # Re-optimize after binary search
    h, c = _slsqp(best_h, dx, 100000, 150)
    if c < best_c5:
        best_h, best_c5 = h, c
    print(f"P4: C5={best_c5:.6f}")

    # P5: Multi-start
    t5 = t0 + 650
    for r in range(10):
        if time.time() > t5 - 15:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _targeted(hp, dx, rng, min(time.time() + 12, t5 - 10), 5, 8)
        if time.time() < t5 - 8:
            hp, c = _slsqp(hp, dx, 80000, 120)
        if c < best_c5:
            best_h, best_c5 = hp, c
    print(f"P5: C5={best_c5:.6f}")

    # P6: 2x resolution
    t6 = t0 + 850
    if budget_s - (time.time() - t0) > 220:
        n2 = 2 * n
        dx2 = 2.0 / n2
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _slsqp(h2, dx2, 5000, 150)
        h2, c2 = _targeted(h2, dx2, rng, min(t6 - 20, time.time() + 50), 5, 8)
        if time.time() < t6 - 15:
            h2, c2 = _slsqp(h2, dx2, 50000, 100)
        if c2 < best_c5:
            best_h, best_c5, n, dx = h2, c2, n2, dx2
            print(f"P6: C5={best_c5:.6f} n={n2}")

    # P7: Extended targeted
    t7 = t0 + 1000
    h, c = _targeted(best_h, dx, rng, t7, 8, 14)
    if c < best_c5:
        best_h, best_c5 = h, c
    print(f"P7: C5={best_c5:.6f}")

    # P8: Final SLSQP
    if time.time() < t0 + budget_s - 20:
        h, c = _slsqp(best_h, dx, 500000, 150)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P9: Ultra-fine local search
    if time.time() < t0 + budget_s - 5:
        h, c = _fine(best_h, dx, rng, t0 + budget_s - 2)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P10: Last targeted micro-adjust
    if time.time() < t0 + budget_s - 2:
        h, c = _targeted(best_h, dx, rng, t0 + budget_s - 1, 5, 6, 0.0005, 0.005)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = _repair(best_h, n / 2.0)
    best_c5 = _c5(best_h, 2.0 / n)
    print(f"FINAL: C5={best_c5:.6f} n={n}")
    return (best_h, best_c5, n)
