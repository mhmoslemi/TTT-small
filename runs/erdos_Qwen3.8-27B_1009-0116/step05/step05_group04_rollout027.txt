    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov(h, alphas, steps_each, step0, mu, t_limit):
    n = len(h)
    t_start = time.time()
    for alpha in alphas:
        v = np.zeros(n)
        f = _smooth_obj(h, alpha)
        step = step0
        for _s in range(steps_each):
            if time.time() - t_start > t_limit:
                return _repair(h, n)
            g = _smooth_grad(h, alpha)
            h_prev = h.copy()
            v = mu * v + (1 - mu) * g
            h_new = _proj_sum(h - step * v, n * 0.5)
            f_new = _smooth_obj(h_new, alpha)
            if f_new <= f:
                h = h_new
                f = f_new
                step = min(step * 1.015, step0 * 5)
            else:
                step *= 0.5
                h = _proj_sum(h_prev - step * v, n * 0.5)
                f = _smooth_obj(h, alpha)
            if step < 1e-8:
                break
    return _repair(h, n)

def _worst_lag_descent(h, n_rounds, t_limit):

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_w = c.argmax()

        sh = (n - 1) - k_w
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
        g *= dx
        c_cur = _eval(h)
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < c_cur - 1e-14:
                h = h_new
                c_cur = c_new
                improved = True
                break
        if not improved:
            break

            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _pairwise(h, n_tries, rng, base_step=2e-5):
    n = len(h)
    c_cur = _eval(h)
    for _ in range(n_tries):

        if c_new < c_cur - 1e-16:
            c_cur = c_new
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c_cur

def _slsqp(h, alpha, maxiter):

    n = len(h)

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:

                hp[idx] = 1.0
                hp = _repair(hp, n0)
            else:
                hp = _repair(best_h + 0.04 * rng.randn(n0), n0)
            t_per = max(0.5, (budget_s * 0.8 - _el()) / max(1, n_starts - _r))
            h_new = _nesterov(hp, [30000, 200000], 150, 0.0008, 0.9, t_per)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 6: SA at best resolution
    nc = best_n
    if _el() < budget_s * 0.9:
        t_rem = budget_s * 0.9 - _el()
        n_it = min(int(t_rem / 0.035), 15000)
        bs, cs = _sa(best_h, nc, 0.003, 1e-5, n_it, rng)
        if cs < best_c5:
            best_c5, best_h = cs, bs.copy()

    # Phase 7: Pairwise + worst-lag alternating
    nc2 = best_n
    if _el() < budget_s - 15:
        h_pw = best_h.copy()
        c_pw = best_c5
        t_rem = budget_s - 15 - _el()
        n_tr = min(int(t_rem * 0.6 / 0.003), 80000)
        h_pw, c_pw = _pairwise(h_pw, n_tr, rng)
        if c_pw < best_c5:
            best_c5, best_h = c_pw, h_pw.copy()
        # Alternate with worst-lag
        if _el() < budget_s - 10:
            h_wl2 = _worst_lag_descent(best_h, 200, max(0.5, budget_s - 10 - _el()))
            c5 = _eval(h_wl2)
            if c5 < best_c5:
                best_c5, best_h = c5, h_wl2.copy()

    # Phase 8: Final SLSQP polish
    if _el() < budget_s - 5:
        h_f = _slsqp(best_h, 3000000, 2000)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()
        h_f2 = _worst_lag_descent(best_h, 100, max(0.5, budget_s - 2 - _el()))
        c5 = _eval(h_f2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
