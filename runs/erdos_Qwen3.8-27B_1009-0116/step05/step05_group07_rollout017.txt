import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(shift, h):
    n = len(h)
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        g[:n - shift] += (1.0 - h[shift:])
        g[shift:] -= h[:n - shift]
    else:
        sp = -shift
        g[:n - sp] -= h[sp:]
        g[sp:] += 1.0 - h[:n - sp]
    return g * (2.0 / n)

def _topk_grad(h, top_k):
    n = len(h)
    c = _corrs(h)
    idx = np.argsort(c)[::-1][:top_k]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(k - (n - 1), h)
    g /= top_k
    return g

def _mom_descend(h0, n_iter, top_k, alpha, beta, seed_val=0):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    v = np.zeros(n)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    stall = 0
    for t in range(n_iter):
        if time.time() - _t0 > _budget - 30:
            break
        g = _topk_grad(h, top_k)
        v = beta * v - alpha * g
        h_new = _proj_feasible(h + v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-15:
            h, f = h_new, f_new
            stall = 0
            if f < best_f:
                best_f, best_h = f, h.copy()
            alpha = min(alpha * 1.003, 0.04)
        else:
            v *= 0.5
            stall += 1
            if stall % 30 == 0:
                alpha *= 0.92
                if alpha < 1e-6:
                    break
    return best_h, best_f

def _coord_descent(h0, n_passes, top_k):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    for p in range(n_passes):
        if time.time() - _t0 > _budget - 30:
            break
        c = _corrs(h)
        active = np.argsort(c)[::-1][:top_k]
        for k in active:
            shift = k - (n - 1)
            g = _grad_lag(shift, h)

                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h, f = h_new, f_new
                    break
    return h, f

def _local_search(h0, n_moves, scale, seed_val=0):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_moves):
        if time.time() - _t0 > _budget - 15:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    ff = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(ff(x_new), 0, 1), n * factor)

def _slsqp_polish(h0, alpha, maxiter):
    n = len(h0)
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = _corrs(hx)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def jac(x):
        hx = np.clip(x, 0, 1)
        c = _corrs(hx)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            g += w[k] * _grad_lag(k - (n - 1), hx)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _block_search(h0, block_sz, seed_val=0):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(5000):

        i = rng.randint(0, max(1, n - block_sz))

        h_new = h.copy()

        h_new = _proj_feasible(h_new, target)
        fn = _eval(h_new)
        if fn < f - 1e-15:
            h, f = h_new, fn
    return h, f
