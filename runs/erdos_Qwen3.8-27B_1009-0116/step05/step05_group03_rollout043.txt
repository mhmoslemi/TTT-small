    return h

def _grad_smax(h, alpha, topk=0):
    """Analytical gradient of smooth max of correlations."""
    n = len(h)
    c = _corr(h)
    # softmax weights
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 50)
    w = np.exp(s)
    wsum = w.sum()
    w /= wsum
    if topk > 0:
        # zero out small weights, renormalize
        idx = np.argsort(w)[:(len(w) - topk)]
        w[idx] = 0
        w /= w.sum()
    # Gradient: sum_k w_k * dc_k/dh
    grad = np.zeros(n)
    # For each lag s = |k - (n-1)|:
    # dc_k/dh[p] = (1-h[p+s]) if p < n-s, plus -h[p-s] if p >= s
    # Vectorized: for all s simultaneously is expensive, but topk is small
    for ki in np.where(w > 1e-10)[0]:
        wi = w[ki]
        s_val = abs(int(ki - (n - 1)))
        if s_val == 0:
            continue
        m = n - s_val
        # First part: grad[0:m] += (1-h[s_val:s_val+m])
        grad[0:m] += wi * (1.0 - h[s_val:s_val+m])
        # Second part: grad[s_val:n] += -h[0:m]
        grad[s_val:n] -= wi * h[0:m]
    return grad

    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 50)

    bds = [(0.0, 1.0)] * n
    def fun(x):
        return _smax_fun(x, alpha)
    def jac(x):
        return _grad_smax(x, _corr(x), alpha)
    def eqf(x):
        return x.sum() - tgt
    try:
        r = minimize(fun, h0.copy(), jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eqf, 'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(np.clip(r.x, 0, 1), tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _alpha_sweep(h, dx, t_end, alphas, maxiter=300):
    best_h, best_c = h, _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc, c = _slsqp(best_h, dx, a, maxiter)
        if c < best_c:
            best_h, best_c = hc, c
        # second pass at same alpha
        if time.time() < t_end - 2:
            hc2, c2 = _slsqp(best_h, dx, a, maxiter // 2)
            if c2 < best_c:
                best_h, best_c = hc2, c2
    return best_h, best_c

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(4, n // 4), replace=False)
        hp[idx] += 0.06 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1.5, 6.0)
        ph = rng.uniform(0, 6.28)
        hp += 0.025 * np.sin(2 * np.pi * f * x + ph)
    elif mode == 3:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    elif mode == 4:
        hp += 0.05 * rng.standard_normal(n)
    else:
        idx = rng.choice(n, n // 2, replace=False)
        hp[idx] += 0.04 * (2.0 * rng.random(len(idx)) - 1.0)
    return _repair(np.clip(hp, 0, 1), tgt)

def _local_search(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    # Adaptive step sizes
    steps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for step in steps:
        stuck = 0
        while time.time() < t_end and stuck < 800:
            c = _corr(h)
            top = np.argsort(c)[::-1][:5]
            improved_this = False
            for ki in top:
                sv = abs(int(ki - (n - 1)))
                if sv == 0:
                    continue
                m = n - sv
                for _ in range(3):
                    a = rng.randint(m)
                    p = a + sv  # index in "tail"
                    q = a      # index in "head"
                    d = step * (2.0 * rng.random() - 1.0)
                    hp = h.copy()
                    hp[p] -= d
                    hp[q] += d
                    if 0 <= hp[p] <= 1 and 0 <= hp[q] <= 1:
                        hp = _repair(hp, tgt)
                        cn = _c5(hp, dx)
                        if cn < best:
                            h, best, stuck = hp, cn, 0
                            improved_this = True
                            break
            if not improved_this:
                # Random pair
                i, j = rng.randint(n), rng.randint(n)
                if i != j:
                    d = step * (2.0 * rng.random() - 1.0)
                    hp = h.copy()
                    hp[i] += d; hp[j] -= d
                    if 0 <= hp[i] <= 1 and 0 <= hp[j] <= 1:
                        hp = _repair(hp, tgt)
                        cn = _c5(hp, dx)
                        if cn < best:
                            h, best, stuck = hp, cn, 0
                            improved_this = True
            if not improved_this:
                stuck += 1
    return h, best

def _coord_descent(h, dx, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    imps = 0
    while time.time() < t_end and imps < 200:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            old = h[i]
            # Binary search for best h[i] in [0,1] keeping sum
            # Adjust one other index to maintain sum
            best_v, best_c = old, best
            for frac in np.linspace(0, 1, 11):
                v = old + frac * (1.0 - old) * 0.3
                v = min(max(v, 0.0), 1.0)
                dv = v - old
                if abs(dv) < 1e-12:
                    continue
                hp = h.copy()
                hp[i] = v
                # find j to compensate
                if dv > 0:
                    j = int(np.argmin(hp))
                else:
                    j = int(np.argmax(hp))
                if j == i or abs(hp[j] - (-dv * (1 if dv > 0 else 1))) > 1:
                    continue
                hp[j] += -dv
                if 0 <= hp[j] <= 1:
                    hp = _repair(hp, tgt)
                    cn = _c5(hp, dx)
                    if cn < best_c:
                        best_v, best_c = v, cn
            if best_c < best and abs(best_v - old) > 1e-10:
                h[i] = best_v
                hp = _repair(h.copy(), tgt)
                h = hp
                best = best_c
                improved = True
                imps += 1
                break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: Coarse SLSQP alpha sweep
    h, c = _alpha_sweep(best_h, dx, t0 + 80, [80, 200, 500, 1500], 250)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 2: Medium alpha refinement
    h, c = _alpha_sweep(best_h, dx, t0 + 200, [3000, 8000, 20000], 300)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 3: Multi-start with perturbations
    t_end3 = t0 + 500
    for r in range(20):
        rem = budget_s - (time.time() - t0)
        if time.time() > t_end3 or rem < 350:
            break
        hp = _perturb(best_h, rng, r % 6)
        hp, c = _slsqp(hp, dx, 5000, 250)
        if c < best_c5:
            best_h, best_c5 = hp, c
            hp, c = _slsqp(hp, dx, 20000, 200)
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 4: Very high alpha SLSQP
    t_end4 = t0 + 700
    h, c = _alpha_sweep(best_h, dx, t_end4, [50000, 100000, 200000], 250)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 5: Local search
    t_end5 = t0 + 900
    h, c = _local_search(best_h, dx, rng, t_end5)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 6: Final SLSQP + local
    if time.time() < t0 + budget_s - 80:
        h, c = _slsqp(best_h, dx, 500000, 200)
        if c < best_c5:
            best_h, best_c5 = h, c
    h, c = _local_search(best_h, dx, rng, t0 + budget_s - 30)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 7: Final polish
    if budget_s - (time.time() - t0) > 15:
        h, c = _slsqp(best_h, dx, 1000000, 150)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
