    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    S = compute_S(h, n)
    return S.T @ w

def proj_sum(h, n):
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def slsqp_phase(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_obj(x, alpha) + 1e4 * (x.sum() - n * 0.5) ** 2
    def grad(x):
        return lse_grad(x, alpha) + 2e4 * (x.sum() - n * 0.5) * np.ones(n)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except:
        return repair(h, n)

def paired_optimize(h, n, rng, budget_s, n_cands=5, n_iters=45):
    C = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = C.max()
    S = compute_S(h, n)
    t_end = time.time() + budget_s
    stall = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            cands = rng.choice(n, size=n_cands, replace=False)
            bd, bf, bj = 0.0, C.max(), -1
            for j in cands:
                if j == i:
                    continue
                dlo = max(-h[i] + 1e-15, -1.0)
                dhi = min(1.0 - h[j] - 1e-15, 1.0)
                if dlo >= dhi:
                    continue
                sd = S[i] - S[j]
                ds = min_1d(C, sd, dlo, dhi)
                fs = (C + sd * ds).max()
                if fs < bf - 1e-14:
                    bf, bd, bj = fs, ds, j
            if bj >= 0 and abs(bd) > 1e-14:
                h[i] += bd
                h[bj] -= bd
                C += bd * (S[i] - S[bj])
                S = compute_S(h, n)
                if C.max() < best_c5:
                    best_c5 = C.max()
                improved = True
        if not improved:
            stall += 1
            if stall > 5:
                break
            # Kick: small random perturbation
            idx = rng.randint(n)
            jdx = rng.randint(n)
            if idx != jdx:
                d = 0.01 * rng.randn()
                h[idx] = np.clip(h[idx] + d, 0, 1)
                h[jdx] = np.clip(h[jdx] - d, 0, 1)
                h = repair(h, n)
                C = np.correlate(h, 1.0 - h, mode='full')
                S = compute_S(h, n)
        else:
            stall = 0
    return repair(h, n), best_c5

def fine_random(h, n, rng, budget_s):
    C = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = C.max()
    t_end = time.time() + budget_s
    for scale in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        while time.time() < t_end:
            iters = 5000
            for _ in range(iters):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                C = np.correlate(h, 1.0 - h, mode='full')
                cm = C.max()
                if cm < best_c5:
                    best_c5 = cm
                else:
                    h[i], h[j] = ni - d, nj + d
                    C = C - C + np.correlate(h, 1.0 - h, mode='full')  # rebuild
    return repair(h, n), best_c5
