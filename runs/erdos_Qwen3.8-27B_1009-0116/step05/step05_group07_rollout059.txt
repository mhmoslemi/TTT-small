    if s >= 0:
        i1 = np.arange(n - s)
        g[i1] += (1.0 - h[i1 + s])
        if s > 0:
            i2 = np.arange(s, n)
            g[i2] -= h[i2 - s]
    else:
        sp = -s
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _momentum_descend(h, topm, n_iter, step0, mu, recompute_every=30):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    v = np.zeros(n)
    f = _eval(h)
    step = step0
    c = _corrs_raw(h)
    top_idx = np.argpartition(c, -topm)[-topm:]
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 25:
            break
        if it % recompute_every == 0:
            c = _corrs_raw(h)
            top_idx = np.argpartition(c, -topm)[-topm:]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_single(int(k), h)
        g /= len(top_idx)
        g *= (2.0 / n)
        v = mu * v + g
        improved = False
        for _bt in range(20):
            h_new = _proj_to_sum(h - step * v, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-13:
                h = h_new; f = f_new; improved = True
                break
            step *= 0.65
        if not improved:
            step *= 0.4
            if step < 1e-10:
                break
        else:
            step = min(step * 1.04, step0)
    return h, f

def _transfer_flip(h, n_moves, topm, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    c = _corrs_raw(h)
    top_idx = np.argpartition(c, -topm)[-topm:]
    for it in range(n_moves):
        if time.time() - _t0 > _budget - 20:
            break
        if it % 400 == 0:
            c = _corrs_raw(h)
            top_idx = np.argpartition(c, -topm)[-topm:]
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        di = 1.0 - h[i]
        dj = h[j]
        if di < 0.005 or dj < 0.005:
            continue
        amt = min(di, dj, 0.04 * (0.5 + 0.5 * rng.rand()))
        h[i] += amt
        h[j] -= amt
        f_new = _eval(h)
        if f_new < f - 1e-14:
            f = f_new
        else:
            h[i] -= amt
            h[j] += amt
    return h, f

def _slsqp_smooth(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        cx = _corrs_raw(x) * (2.0 / n)
        mm = cx.max()
        s = np.clip(alpha * (cx - mm), -300, 300)
        return mm + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def grad(x):
        cx = _corrs_raw(x) * (2.0 / n)
        mm = cx.max()
        s = np.clip(alpha * (cx - mm), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            g += w[k] * _grad_single(k, x) * (2.0 / n)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - _t0

    # Phase 1: Quick SLSQP warmup
    if _el() < 80:
        hn = _slsqp_smooth(best_h, 15000, 400)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn

    # Phase 2: Momentum descent, increasing focus
    schedules = [
        (3, 2500, 0.002, 0.75), (5, 2500, 0.002, 0.82),
        (7, 2000, 0.0015, 0.87), (10, 2000, 0.001, 0.90),
        (15, 1500, 0.0008, 0.92), (20, 1500, 0.0005, 0.94)
    ]
    for topm, nit, st, mu in schedules:
        if _el() > 380:
            break
        hn, fc = _momentum_descend(best_h, topm, nit, st, mu)
        if fc < best_c5:
            best_c5 = fc; best_h = hn

    # Phase 3: Transfer flips (different move type)
    if _el() < _budget - 280:
        for sf in [0.04, 0.02, 0.01, 0.005]:
            if _el() > _budget - 260:
                break
            hn, fc = _transfer_flip(best_h, 25000, 5, seed + 100 * int(sf * 1e4))
            if fc < best_c5:
                best_c5 = fc; best_h = hn

    # Phase 4: Fine momentum
    fine_schedules = [(5, 4000, 0.0004, 0.93), (8, 3000, 0.0003, 0.94),
                      (12, 2500, 0.0002, 0.95), (15, 2000, 0.0001, 0.96)]
    for topm, nit, st, mu in fine_schedules:
        if _el() > _budget - 180:
            break
        hn, fc = _momentum_descend(best_h, topm, nit, st, mu)
        if fc < best_c5:
            best_c5 = fc; best_h = hn

    # Phase 5: Multi-start exploration
    for _r in range(10):
        if _el() > _budget - 100:
            break
        amp = 0.003 + 0.003 * rng.rand()
        hp = best_h + amp * rng.randn(n0)
        hp = _repair(hp, n0)
        hn, fc = _momentum_descend(hp, 5, 1500, 0.001, 0.85)
        if fc < best_c5:
            best_c5 = fc; best_h = hn

    # Phase 6: SLSQP polish
    if _el() < _budget - 60:
        hn = _slsqp_smooth(best_h, 40000, 250)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn

    # Phase 7: Ultra-fine final pass
    if _el() < _budget - 25:
        hn, fc = _momentum_descend(best_h, 8, 4000, 0.0001, 0.96, recompute_every=20)
        if fc < best_c5:
            best_c5 = fc; best_h = hn
        if _el() < _budget - 10:
            hn, fc = _transfer_flip(best_h, 15000, 3, seed + 9999)
            if fc < best_c5:
                best_c5 = fc; best_h = hn

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)

def run(seed=42, budget_s=1100, **kwargs):
    return _run(seed=seed, budget_s=budget_s, **kwargs)
