        g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _subgrad_corr(h, k)
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_opt(h, n_iter, step0, tol):
    n = len(h)
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _eval(best_h)
    step = step0
    for it in range(n_iter):
        c = _corr(h)
        cmax = c.max()
        c5c = cmax * 2.0 / n
        if c5c < best_c5:
            best_c5, best_h = c5c, h.copy()
        active = np.where(c >= cmax - tol)[0]
        g = np.zeros(n)
        for k in active:
            g += _subgrad_corr(h, k)
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, target)
            c5_new = _eval(h_new)
            if c5_new < c5c - 1e-15:
                h, c5c = h_new, c5_new
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.6
        else:
            step = min(step * 1.15, step0)
    return best_h, best_c5

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _pair_search(h, c, rng, n_iter, scale, n, block_size=1):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i], nj_v = ni, nj
            h[j] = nj_v
            c, c5c = new_c, new_c5
    return h, c, c5c

def _block_search(h, c, rng, n_iter, scale, n, bsize):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(max(1, n - bsize))
        j = rng.randint(max(1, n - bsize))
        if abs(i - j) <= bsize:
            continue
        block_i = np.arange(i, i + bsize)
        block_j = np.arange(j, j + bsize)
        d = scale * rng.randn(bsize)
        new_bi = h[block_i] + d
        new_bj = h[block_j] - d
        if new_bi.min() < -1e-10 or new_bi.max() > 1.0 + 1e-10:
            continue
        if new_bj.min() < -1e-10 or new_bj.max() > 1.0 + 1e-10:
            continue
        dc = np.zeros(2 * n - 1)
        for idx in range(bsize):
            dc += _pos_change(h, block_i[idx], d[idx], n)
            dc += _pos_change(h, block_j[idx], -d[idx], n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[block_i] = new_bi
            h[block_j] = new_bj
            c, c5c = new_c, new_c5
    return h, c, c5c

def _nm_polish(h, n, maxiter):
    def obj(x):
        return _eval(_repair(x, n))
    try:
        res = minimize_nelder_me(obj, h, maxiter=maxiter, xatol=1e-9, fatol=1e-12)
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP with increasing alpha
    for alpha in [200, 500, 1000, 5000, 20000, 80000, 200000, 500000]:
        if _el() > 90:
            break
        h_new = _slsqp_opt(best_h, alpha, 4000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Subgradient with active set
    if _el() < budget_s - 200:
        for tol in [0.01, 0.005, 0.002, 0.001]:
            if _el() > budget_s - 200:
                break
            h_sg, c5_sg = _subgrad_opt(best_h, 500, 0.01, tol)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 3: SLSQP polish after subgradient
    if _el() < budget_s - 180:
        for alpha in [100000, 300000, 1000000]:
            if _el() > budget_s - 180:
                break
            h_new = _slsqp_opt(best_h, alpha, 3000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 4: Pairwise + block local search
    if _el() < budget_s - 100:
        h_ps = best_h.copy()
        c_ps = _corr(h_ps)
        n_ps = len(h_ps)
        for scale, nit in [(0.05, 20000), (0.02, 30000), (0.01, 40000), (0.005, 50000), (0.002, 60000), (0.001, 80000)]:
            if _el() > budget_s - 100:
                break
            h_ps, c_ps, _ = _pair_search(h_ps, c_ps, rng, nit, scale, n_ps)
        # Block moves
        for obj_scale, nit in [(0.02, 15000), (0.01, 20000), (0.005, 25000)]:
            if _el() > budget_s - 100:
                break
            h_ps, c_ps, _ = _block_search(h_ps, c_ps, rng, nit, obj_scale, n_ps, 3)
        h_ps = _repair(h_ps, n_ps)
        c5 = _eval(h_ps)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ps.copy()

    # Phase 5: Nelder-Mead polish
    if _el() < budget_s - 60:
        h_nm = _nm_polish(best_h.copy(), best_n, 2000)
        c5 = _eval(h_nm)
        if c5 < best_c5:
            best_c5, best_h = c5, h_nm.copy()

    # Phase 6: Fine pairwise search
    if _el() < budget_s - 30:
        h_f = best_h.copy()
        c_f = _corr(h_f)
        n_f = len(h_f)
        for scale, nit in [(5e-4, 60000), (2e-4, 80000), (1e-4, 100000), (5e-5, 120000)]:
            if _el() > budget_s - 30:
                break
            h_f, c_f, _ = _pair_search(h_f, c_f, rng, nit, scale, n_f)
        h_f = _repair(h_f, n_f)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    # Phase 7: Multi-start exploration
    if _el() < budget_s - 100:
        nc = len(best_h)
        n_starts = 4
        for r in range(n_starts):
            if _el() > budget_s - 100:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.03 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                s_idx = rng.randint(nc // 3)
                e_idx = min(nc, s_idx + nc // 3)
                hp[s_idx:e_idx] += 0.04 * rng.randn(e_idx - s_idx)
                hp = _repair(hp, nc)
            else:
                hp = _repair(best_h, nc)
                for _ in range(5):
                    bs = rng.randint(2, 4)
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.02 * rng.randn(bs)
                hp = _repair(hp, nc)
            h_ms = _slsqp_opt(hp, 50000, 2000, 1e-19)
            c_ms = _corr(h_ms)
            h_ms, c_ms, _ = _pair_search(h_ms, c_ms, rng, 30000, 0.01, nc)
            h_ms = _repair(h_ms, nc)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h = c5, h_ms.copy()

    # Phase 8: Final ultra-fine polish
    if _el() < budget_s - 5:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        n_fin = len(h_fin)
        for scale, nit in [(2e-5, 100000), (1e-5, 150000)]:
            if _el() > budget_s - 5:
                break
            h_fin, c_fin, _ = _pair_search(h_fin, c_fin, rng, nit, scale, n_fin)
        h_fin = _repair(h_fin, n_fin)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
