    """Project v onto [0,1]^n ∩ {sum = n/2}"""
    # First clip to [0,1]
    h = np.clip(v, 0, 1)
    # Then adjust to make sum = n/2
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Find threshold t such that sum(clip(v_i + t, 0, 1)) = target
    # Use bisection on t
    lo, hi = -1.0, 1.0
    for _ in range(60):

        if s < target:

        else:
            hi = mid

    # Fix any rounding

        else:

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(v, n):
    """Euclidean projection onto [0,1]^n ∩ {sum=n/2}"""
    target = n / 2.0
    v = np.asarray(v, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(v + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0, 1)
        else:
            mask = h > 0.0
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0, 1)
    return h

def _grad_smooth(h, alpha):
    """Gradient of smooth max of overlaps"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    s = np.clip(alpha * (c - c.max()), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            m = n - sh
            i1 = np.arange(m)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(v + mid, 0, 1).sum() < target: lo = mid
        else: hi = mid
    h = np.clip(v + (lo + hi) / 2, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            m = h < 1.0
            if m.any(): h[m] = np.clip(h[m] + diff / m.sum(), 0, 1)
        else:
            m = h > 0
            if m.any(): h[m] = np.clip(h[m] + diff / m.sum(), 0, 1)
    return h

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    s = np.clip(alpha * (c - c.max()), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            m = n - sh
            i1 = np.arange(m)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g * dx

def _pg(h, alpha, iters, deadline):
    n = len(h)
    best = _eval(h)
    h_prev = h.copy()
    g_prev = _grad(h, alpha)
    step = 0.005
    for _ in range(iters):
        if time.time() > deadline: break
        g = _grad(h, alpha)
        s = h - h_prev; yg = g - g_prev
        ys = s @ yg; gg = yg @ yg
        if gg > 1e-18 and abs(ys) > 1e-18:
            step = max(0.0005, min(0.05, ys / gg))
        h_new = _proj(h - step * g, n)
        c = _eval(h_new)
        if c < best:

            h_prev, h, g_prev = h, h_new, g
        else:
            step *= 0.6
            h_prev, g_prev = h, g
    return h, best

def _slsqp(h, alpha, iters, deadline):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(s.sum() + 1e-30) / alpha

        return _grad(x, alpha)

    def _eqj(x):
        return np.ones(n)
