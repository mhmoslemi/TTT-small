            h_ps = _proj(h_ps, n1 * 0.5)
            c5 = _eval(h_ps)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_ps.copy(), n1

    # Phase 5: Multi-start exploration
    if _el() < budget_s - 180:
        nc = len(best_h)
        for r in range(6):
            if _el() > budget_s - 100:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = best_h + 0.025 * rng.randn(nc)
            elif strat == 1:
                hp = 1.0 - best_h + 0.01 * rng.randn(nc)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(4, max(5, nc // 8))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
            else:
                perm = rng.permutation(nc)
                hp = best_h[perm]
                # Restore approximate structure
                hp = _proj(hp, nc * 0.5)
                hp = best_h * 0.7 + hp * 0.3
            hp = _proj(hp, nc * 0.5)
            dl = t0 + min(50, budget_s - _el() - 50)
            h_ms, _ = _adam(hp, 50000, 800, 0.001, dl)
            h_ms = _proj(h_ms, nc * 0.5)
            c_ms = _corr(h_ms)
            c5m = c_ms.max() * 2.0 / nc
            dl2 = t0 + min(25, budget_s - _el() - 30)
            h_ms, c_ms, c5m = _pair_search(h_ms, c_ms, c5m, rng, 25000, 0.005, dl2)
            h_ms = _proj(h_ms, nc * 0.5)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_ms.copy(), nc

    # Phase 6: Ultra-fine final polish
    if _el() < budget_s - 20:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        c5_fin = c_fin.max() * 2.0 / best_n
        for scale in [5e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
            if _el() > budget_s - 5:
                break
            dl = t0 + min(15, budget_s - _el() - 3)
            h_fin, c_fin, c5_fin = _pair_search(h_fin, c_fin, c5_fin, rng, 60000, scale, dl)
        h_fin = _proj(h_fin, best_n * 0.5)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin.copy(), best_n

    best_h = _proj(best_h, best_n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
