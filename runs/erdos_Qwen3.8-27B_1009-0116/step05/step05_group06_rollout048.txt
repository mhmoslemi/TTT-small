def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        mx = c.max()
        s = np.clip(alpha*(c-mx), -500, 500)
        return mx + np.log(np.sum(np.exp(s)))/alpha
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full')
        mx = c.max()
        s = np.clip(alpha*(c-mx), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        dx = 2.0/n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15: continue
            sh = (n-1)-k
            wk = w[k]*dx
            if sh >= 0:
                g[:n-sh] += wk*(1.0-x[sh:])
                if sh > 0: g[sh:] -= wk*x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk*(1.0-x[:n-sp])
                g[:n-sp] -= wk*x[sp:]
        return g
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except:
        return _repair(h.copy(), n)

def _coorddescent(h, n_rounds, rng, step_scale=0.02):
    n = len(h)
    c5 = _eval_raw(h)
    dx = 2.0/n
    for rnd in range(n_rounds):
        order = rng.permutation(n)
        improved = False
        for j in order:
            h0 = h[j]
            best_v, best_c = h0, c5
            for d in [step_scale, -step_scale, step_scale*0.4, -step_scale*0.4, step_scale*0.1, -step_scale*0.1]:
                nv = h0 + d
                if nv < 0 or nv > 1: continue
                h[j] = nv
                c = np.max(np.correlate(h, 1.0-h, mode='full')) * dx
                if c < best_c - 1e-14:
                    best_c, best_v = c, nv
            h[j] = best_v
            if best_v != h0:
                c5 = best_c
                improved = True
        if not improved and rnd > 10:
            step_scale *= 0.7
        if c5 < 0.3808:
            break
    return h, c5

def _es_pop(h_best, c5_best, pop_size, n_gen, rng, scale):
    n = len(h_best)
    pop = [h_best.copy()]
    for _ in range(pop_size-1):
        p = _repair(h_best + scale*rng.randn(n), n)
        pop.append(p)
    for gen in range(n_gen):
        # Mutate top-half
        for i in range(pop_size//2):
            base = pop[i].copy()
            # DE-like mutation
            a, b = rng.choice(pop_size, 2, replace=False)
            mutant = base + 0.5*(pop[a]-pop[b]) + 0.01*rng.randn(n)
            pop[i] = _repair(mutant, n)
        # Evaluate
        c5s = [_eval_raw(p) for p in pop]
        # Select top pop_size
        idx = np.argsort(c5s)[:pop_size]
        pop = [pop[i].copy() for i in idx]
        c5_best_gen = c5s[idx[0]]
        if c5s[idx[0]] < c5_best:
            c5_best = c5s[idx[0]]
            h_best = pop[0].copy()
        scale *= 0.995
    return h_best, c5_best

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h
