def _grad_lag(h, k, n):
    g = np.zeros(n)
    s = n - 1 - k
    if s >= 0:
        i1 = np.arange(n - s)
        g[i1] += (1.0 - h[i1 + s])
        if s > 0:
            g[s:] -= h[:n-s]

        p = -s
        g[p:] += (1.0 - h[:n-p])

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, ci, n):
    g = np.zeros(n)
    s = ci - (n - 1)
    if s >= 0:
        g[s:] += 1.0
        if s > 0:
            g[s:] -= h[:n - s]
        g[:n - s] -= h[s:]
    else:
        p = -s
        g[p:] += 1.0
        g[p:] -= h[:n - p]
        g[:n - p] -= h[p:]
    return g * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > target:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a if d > 0 else -a
            d = d - a if d > 0 else d + a
            if abs(d) < 1e-12:
                break
    return h

def _nesterov(h, n, t_end, top_k=1, mu=0.92, s0=0.08):

    f = _eval(h)
    best = h.copy()
    bf = f
    v = np.zeros(n)
    step = s0
    stall = 0
    while time.time() < t_end:
        c = _corr(h)
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = c[idx] - c.min() + 1e-8
        w /= w.sum()
        g = np.zeros(n)

            g += wi * _grad_lag(h, ki, n)
        v = mu * v + (1.0 - mu) * g

            h = h2

    for tk in [1, 3, 7]:
        te = t_start + min(el() + 20, 50)
        h1, f1 = _nesterov(best_h, best_n, te, top_k=tk)
        if f1 < best_c5:
            best_c5, best_h = f1, h1.copy()

    # Phase 2: Active set descent
    te2 = t_start + min(el() + 25, 70)
    h2, f2 = _active_desc(best_h, best_n, te2)
    if f2 < best_c5:
        best_c5, best_h = f2, h2.copy()

    # Phase 3: SLSQP polish at original res
    for alpha in [2000, 20000, 100000]:
        if el() > budget_s - 350:
            break
        te3 = t_start + min(el() + 30, budget_s - 350)
        h3 = _slsqp_fast(best_h, best_n, alpha, te3)
        f3 = _eval(h3)
        if f3 < best_c5:
            best_c5, best_h = f3, h3.copy()

    # Phase 4: Upsample to 2x and optimize
    if el() < budget_s - 250:
        n2 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _fix(h_hi, n2)
        for tk in [1, 3]:
            te4 = t_start + min(el() + 25, budget_s - 200)
            h4, f4 = _nesterov(h_hi, n2, te4, top_k=tk)
            if f4 < best_c5:
                best_c5, best_h, best_n = f4, h4.copy(), n2
        if el() < budget_s - 120:
            te5 = t_start + min(el() + 40, budget_s - 120)
            h5 = _slsqp_fast(best_h, best_n, 50000, te5)
            f5 = _eval(h5)
            if f5 < best_c5:
                best_c5, best_h = f5, h5.copy()
        if el() < budget_s - 80:
            h6, f6 = _active_desc(best_h, best_n, t_start + budget_s - 80)
            if f6 < best_c5:
                best_c5, best_h = f6, h6.copy()

    # Phase 5: SA + CD in remaining time
    t_remain = budget_s - 10 - el()
    if t_remain > 30:
        half = t_remain * 0.5
        h7, f7 = _block_sa(best_h, best_n, t_start + el() + half, rng)
        if f7 < best_c5:
            best_c5, best_h = f7, h7.copy()
        h8, f8 = _pair_cd(best_h, best_n, t_start + budget_s - 5, rng)
        if f8 < best_c5:
            best_c5, best_h = f8, h8.copy()

    # Final micro-polish
    if el() < budget_s - 3:
        h9, f9 = _active_desc(best_h, best_n, t_start + budget_s - 3)
        if f9 < best_c5:
            best_c5, best_h = f9, h9.copy()

    best_h = _fix(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
