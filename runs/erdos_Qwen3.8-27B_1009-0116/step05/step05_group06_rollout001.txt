import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_shift(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh + m])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _nesterov_subgrad(h, nsteps, topk, step0, mu, restart_every):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(nsteps):
        if it % 200 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(c, -topk)[-topk:]
        g = np.zeros(n)
        for k in idx:

        g = g * dx / topk
        v = mu * v + step * g
        h_new = _proj(h - v, n)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h, f = h_new, f_new
        else:
            step *= 0.6
        step = min(step * 1.005, step0 * 4.0)
        if it > 0 and it % restart_every == 0:
            v = np.zeros(n)
            step = step0
    return h

def _coord_descent(h, nsteps, rng):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(nsteps):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        for d in (0.01, -0.01, 0.005, -0.005, 0.002, -0.002):
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5 - 1e-15:
                    c5 = c5n
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
                    break
    return h

def _slsqp_smooth(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            g += w[k] * _grad_shift(x, k, n)
        return g * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
