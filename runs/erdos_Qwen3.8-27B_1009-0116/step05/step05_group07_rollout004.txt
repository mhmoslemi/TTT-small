import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        if sh == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            i1 = np.arange(n - sh)
            g[i1] += 1.0 - h[i1 + sh]
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            i1 = np.arange(sp, n)
            g[i1] += 1.0 - h[i1 - sp]
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g

def _smooth_obj_grad(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s - s.max()
    w = np.exp(sm)
    w /= w.sum()
    obj = m + np.log(w.sum()) / alpha
    g = np.zeros(n)
    n2n = 2 * n - 1
    for k in range(n2n):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        g += w[k] * _grad_lag(h, sh, n)
    return obj, g

def _nested(h, m, iterations=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.9
    lr = 0.01
    for it in range(iterations):
        if time.time() - _t0 > _budget - 30:
            break
        _, g = _smooth_obj_grad(h, m)
        v = mu * v + (1 - mu) * g
        found = False
        for _ in range(20):
            h_new = _proj_sum(h - lr * v, target)
            f_new = _eval(h_new)
            if f_new < f + 1e-14:
                h = h_new; f = f_new; found = True
                break
            lr *= 0.5
        if found:
            lr = min(lr * 1.05, 0.05)
            mu = min(0.9 + it * 0.001, 0.95)
        else:
            break
    return h, f

def _block_cd(h, win_size, n_sweeps, seed_val):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    order = list(range(n - win_size + 1))
    for sw in range(n_sweeps):
        if time.time() - _t0 > _budget - 30:
            break
        rng.shuffle(order)
        for start in order:
            if time.time() - _t0 > _budget - 30:
                break
            end = start + win_size
            w_sum = h[start:end].sum()
            w_target = win_size / 2.0
            h_loc = h[start:end].copy()

            def _block_obj(u):
                h_loc2 = h_loc.copy()
                h_loc2[:] = np.clip(0.5 + u, 0, 1)
                adj = (w_target - h_loc2.sum()) / win_size
                h_loc2 = np.clip(h_loc2 + adj, 0, 1)
                h[start:end] = h_loc2
                return _eval(h)

            n_free = win_size - 1
            u0 = h_loc[:n_free] - 0.5
            try:
                res = minimize(_block_obj, u0, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-7, 'fatol': 1e-10})
                f_new = _eval(h)
                if f_new < f - 1e-14:
                    f = f_new
                else:
                    h[start:end] = np.clip(0.5 + u0, 0, 1)
                    adj = (w_target - h[start:end].sum()) / win_size
                    h[start:end] = np.clip(h[start:end] + adj, 0, 1)
            except Exception:
                pass
    h = _repair(h, n)
    return h, _eval(h)

def _active_lag_polish(h, topm, n_iter, step0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    f = _eval(h)
    step = step0
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            sh = (n - 1) - k
            g += _grad_lag(h, sh, n)
        g /= topm
        found = False
        for _ in range(30):
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-7 * step * np.dot(g, h_new - h):
                h = h_new; f = f_new; found = True
                break
            step *= 0.4
        if not found:
            break
        step = min(step * 1.03, step0)
    return h, f

def _pairwise_local(h, n_iter, scale, seed_val):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    idx = np.arange(n)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 10:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)
