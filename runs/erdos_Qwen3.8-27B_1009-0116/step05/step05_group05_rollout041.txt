import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, idx, dx):
    n = len(h)
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _min_1d_convex(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        kmax = vals.max()
        active = np.where(np.abs(vals - kmax) < 1e-10)[0]
        slopes = a[active]
        if slopes.max() <= 1e-14:
            hi = mid
        elif slopes.min() >= -1e-14:
            lo = mid
        else:
            lo = hi = mid
            break
    return (lo + hi) * 0.5

def _active_descent(h, max_iters, n_top, momentum):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= ws
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= w * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w * h[i2 + sp]
        g *= dx
        v = momentum * v + g
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4, 5e-5]:
            h_new = _proj_sum(h - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                improved = True

                c5n = _eval(h_new)
                if c5n < c5c - 1e-16:
                    h = h_new
                    c5c = c5n
                    improved = True
                    break
            if not improved:
                break
    return h, c5c

def _coord_descent(h, sweeps):
    n = len(h)
    c5c = _eval(h)
    for _ in range(sweeps):
        for i in range(n):
            old = h[i]
            base = np.correlate(h, 1.0 - h, mode='full')
            slopes = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                sh = (n - 1) - k
                slope = 0.0
                if sh >= 0:
                    if i <= n - 1 - sh:
                        slope += (1.0 - h[i + sh])
                    if sh > 0 and 0 <= i - sh < n - sh:
                        slope -= h[i - sh]
                else:
                    sp = -sh
                    if sp <= i < n:
                        slope += (1.0 - h[i - sp])
                    if 0 <= i + sp < n - sp:
                        slope -= h[i + sp]
                slopes[k] = slope
            intercepts = base - slopes * old
            v_opt = _min_1d_convex(slopes, intercepts)
            h[i] = v_opt
        h = _proj_sum(h, n * 0.5)
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
    return h, c5c

def _smooth_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons, args=(alpha,),
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
