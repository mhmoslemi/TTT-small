        combined = np.zeros(n)
        for w, ki in zip(weights, top):
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            g = np.zeros(n)
            g[:m] += 1.0 - h[s:n]
            g[s:] -= h[:m]
            combined += w * g
        combined -= combined.mean()
        dec_cands = np.argsort(combined)[::-1][:10]
        inc_cands = np.argsort(combined)[:10]
        found = False
        for di in dec_cands:
            if found:
                break
            for ii in inc_cands:
                if di == ii:
                    continue
                d = min(h[di], 1.0 - h[ii], step)
                if d < 1e-8:
                    continue
                h[di] -= d
                h[ii] += d
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                    found = True
                    no_improve = 0
                else:
                    h[di] += d
                    h[ii] -= d
        if not found:
            for _ in range(15):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = step * (2.0 * rng.random() - 1.0)
                h[i] += d
                h[j] -= d
                if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                        found = True
                        no_improve = 0
                        break
                    else:
                        h[i] -= d
                        h[j] += d
            if not found:
                step *= 0.87
                no_improve += 1
                if no_improve > 80:
                    break
        else:
            step = min(step * 1.015, 0.015)
            no_improve = 0
    return _repair(h, tgt), best

def _fine_search(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, dx)
    scales = [0.003, 0.0015, 0.0007, 0.0003, 0.0001, 5e-05, 2e-05]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d
            h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
        if time.time() < t_end:
            h2, c2 = _slsqp(h, dx, 100000, 100)
            if c2 < best:
                h, best = _repair(h2, tgt), c2
    return _repair(h, tgt), best

def _multi_slsqp(h, dx, t_end, alphas, maxiter=300):
    best_h, best_c = h, _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc = _slsqp(best_h, dx, a, maxiter)
        c = _c5(hc, dx)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)

    # Phase 1: SLSQP warm-up
    h, c = _multi_slsqp(best_h, dx, t0 + 60, [500, 2000, 8000], 350)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 2: Intensive lag-balance local search
    h, c = _lag_balance(best_h, dx, rng, t0 + 400, n_lags=3)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 3: SLSQP sharpening
    h, c = _multi_slsqp(best_h, dx, t0 + 500, [30000, 80000], 250)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 4: Second lag-balance pass
    h, c = _lag_balance(best_h, dx, rng, t0 + 700, n_lags=4)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 5: High-alpha SLSQP
    h, c = _multi_slsqp(best_h, dx, t0 + 800, [200000, 500000], 200)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 6: Fine random search
    h, c = _fine_search(best_h, dx, rng, t0 + budget_s - 30)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 7: Final SLSQP + micro
    if budget_s - (time.time() - t0) > 10:
        hc = _slsqp(best_h, dx, 300000, 150)
        c = _c5(hc, dx)
        if c < best_c5:
            best_h, best_c5 = hc, c

    best_h = _repair(best_h, n / 2.0)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
