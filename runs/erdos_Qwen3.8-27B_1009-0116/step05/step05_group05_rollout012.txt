import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _so_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float((alpha * m + np.log(np.sum(np.exp(s)))) / alpha * 2.0 / n)

def _so_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    w = np.exp(sm)
    w /= w.sum()
    return _corr_grad(h, w) * 2.0 / n

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_so_obj, h, jac=_so_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_desc(h, n_inner, topk, alpha_smooth):
    """Projected gradient descent with top-k active set."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    lr = 0.02
    for _it in range(n_inner):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:topk]
        cvals = c[idx]
        sw = np.exp(alpha_smooth * (cvals - cvals.max()))
        sw /= sw.sum()
        w = np.zeros(2 * n - 1)
        w[idx] = sw
        g = _corr_grad(h, w) * dx
        improved = False
        for _ in range(30):
            h_new = _proj_to_sum(h - lr * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new; c5c = c5n; improved = True; break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break
        lr = min(lr * 1.05, 0.05)
    return h, c5c

def _active_descent(h, max_iters, n_top):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        a = 300.0
        sw = np.exp(a * (cvals - cvals.max()))
        sw /= sw.sum()
        w = np.zeros(2 * n - 1)
        w[idx] = sw
        g = _corr_grad(h, w) * dx
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new; c5c = c5n; improved = True; break
        if not improved:
            break
    return h, c5c

def _pairwise_refine(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)
