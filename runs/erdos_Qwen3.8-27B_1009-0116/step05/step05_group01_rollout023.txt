Now the main algorithm. Let me think about what's different from before:

Previous: SLSQP → Upsample → SLSQP → Projected gradient → Incremental (random) → SLSQP → Incremental (fine) → Multi-start → Incremental (ultra-fine)

My new approach:
1. Start from initial_h_values, repair
2. Quick SLSQP (moderate alpha, few iterations) to get close to local optimum
3. **Aggressive gradient-guided incremental search**:
   - Compute gradient
   - For many steps: select best pair, find optimal step, apply
   - Recompute gradient every 5000 steps
   - Adaptive step size cooling
4. SLSQP polish (high alpha)
5. More incremental search (finer)
6. Final SLSQP

Key innovation: **Gradient-guided optimal pair step**

For each iteration:
- Find i = argmax |g_i| (the index most "wanting" to change)
- Find j ≠ i that minimizes the objective change (greedy)
  Actually, just pick j = argmax_{j≠i} |g_j| with sign(g_j) ≠ sign(g_i)
- Direction: d has sign = -sign(g_i) (so that h[i] moves in -g direction)
- Step: bisect to find d that minimizes max(c + d*dc_i - d*dc_j)
  where dc_i = _pos_change(h, i, 1, n) and dc_j = _pos_change(h, j, 1, n)
  new_c = c + d*(dc_i - dc_j)
  We want to minimize max(new_c) subject to 0<=h[i]+d<=1, 0<=h[j]-d<=1

The bisection: the function max(c + d*delta) where delta = dc_i - dc_j is convex in d (max of linear functions). We can find the minimum by bisection on the derivative, or just try a few points.

Actually, simpler: just try a few candidate step sizes and pick the best.

Let me simplify: for each selected pair, try steps: ±scale, ±scale/2, ±scale/4, ±scale/8. Pick the one that gives the lowest max(c_new).

This is O(n * 8) per step (8 evaluations of max over 2n-1 elements). For n=280, that's ~4500 ops per step. At 100k steps/sec, that's ~450M ops/sec which is within numpy's capability.

Actually, let me just do: try step = scale * sign(-g_i), then if it doesn't help, try half, quarter, etc. This is the "backtracking line search" approach.

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.5, 1.5
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0, 1).sum() > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    s = np.arange(-(n - 1), n)
    dc += di
    m1 = (s >= i - (n - 1)) & (s <= i)
    dc[m1] -= di * h[i - s[m1]]
    m2 = (s >= -i) & (s <= n - 1 - i)
    dc[m2] -= di * h[i + s[m2]]
    return dc

    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    m = c.max()

    w = np.exp(s - s.max())

    return _smooth_grad(x, alpha)

    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sq_sum_fun, 'jac': _sq_sum_jac}]
    try:
        res = minimize(_sq_p_fun, h, jac=_sq_p_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_incremental(h, c, rng, n_steps, scale, n, recompute_every=5000):
    """Gradient-guided incremental pairwise search."""
    alpha = 50000.0
    g = _smooth_grad(h, alpha)
    c5c = c.max() * 2.0 / n
    accepted = 0

        if step_idx % recompute_every == 0:

        if gi > 0:

        j = int(candidates[j_idx])

            h[j] -= best_d
            c += best_d * delta
            c5c = best_val
            g[i] = 0  # Invalidate
            g[j] = 0
            accepted += 1

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.5, 1.5
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0, 1).sum() > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    s = np.arange(-(n - 1), n)
    m1 = (s >= i - (n - 1)) & (s <= i)
    dc[m1] += di * (1.0 - h[i - s[m1]])
    m2 = (s >= -i) & (s <= n - 1 - i)
    dc[m2] -= di * h[i + s[m2]]
    dc[n - 1] -= di * di
    return dc

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sq_fun(x, alpha):
    return _smooth_obj(x, alpha)

def _sq_jac(x, alpha):
    return _smooth_grad(x, alpha)

def _sq_c_fun(x):
    return x.sum() - len(x) * 0.5

def _sq_c_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sq_c_fun, 'jac': _sq_c_jac}]
    try:
        res = minimize(_sq_fun, h, jac=_sq_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
