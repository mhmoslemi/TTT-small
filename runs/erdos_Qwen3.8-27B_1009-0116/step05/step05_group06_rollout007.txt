    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, topk):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def _fista(h0, nsteps, topk, step0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    step = step0
    for it in range(nsteps):
        g = _grad_topk(h, topk)
        h_new = _proj_to_sum(h - step * g, target)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        h = h_new + ((t - 1) / t_new) * (h_new - h_prev)
        h = _proj_to_sum(h, target)
        h_prev = h_new
        t = t_new
        step = min(step * 1.003, step0 * 4.0)
    return h

def _slsqp_opt(h, alpha, maxiter, ftol, fw=0.0):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = _corr(x)
        m = c.max() * dx
        s = np.clip(alpha * (c * dx - m), -500, 500)
        lse = m + np.log(np.sum(np.exp(s))) / alpha
        if fw > 0:
            topk = max(5, n // 5)
            ct = np.sort(c)[::-1][:topk]
            cm = ct.mean()
            lse += fw * dx * np.sum((ct - cm) ** 2)
        return lse
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _worst_shift_descent(h, nrounds, step_size):
    nc = len(h)
    dx = 2.0 / nc
    c5c = _eval(h)
    for _ in range(nrounds):
        c = _corr(h)
        top3 = np.argsort(c)[::-1][:3]
        g = np.zeros(nc)
        for k in top3:
            sh = (nc - 1) - k
            if sh >= 0:
                i1 = np.arange(nc - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, nc)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, nc)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(nc - sp)
                g[i2] -= h[i2 + sp]
        g *= dx / 3.0
        h_new = _proj_to_sum(h - step_size * g, nc * 0.5)
        c5n = _eval(h_new)
        if c5n < c5c:
            h, c5c = h_new, c5n
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase A: Fine alpha schedule SLSQP at base resolution
    alphas = [300, 1500, 8000, 40000, 150000, 500000]
    fws = [0.0, 0.0, 0.0, 0.005, 0.02, 0.05]
    for idx_a, (a, fw) in enumerate(zip(alphas, fws)):
        if _el() > 150:
            break
        h_new = _slsqp_opt(best_h, a, 4000, 1e-18, fw)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase B: FISTA with decreasing topk
    if _el() < budget_s - 350:
        for topk, steps, s0 in [(12, 1000, 0.003), (6, 2000, 0.0025), (3, 3000, 0.002)]:
            if _el() > budget_s - 350:
                break
            h_new = _fista(best_h, steps, topk, s0)
            h_new = _repair(h_new, n0)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase C: Worst-shift targeted descent
    if _el() < budget_s - 250:
        h_new, c5n = _worst_shift_descent(best_h, 500, 0.004)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase D: 2x upsample and optimize
    if _el() < budget_s - 300:
        n1 = n0 * 2
        h_hi = np.interp(np.linspace(0, 1, n1, endpoint=False) * n0,
                         np.arange(n0), best_h)
        h_hi = _repair(h_hi, n1)
        for a in [20000, 100000, 400000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp_opt(h_hi, a, 2500, 1e-18, 0.0)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 200:
            h_hi = _fista(h_hi, 2000, 5, 0.001)
            h_hi = _repair(h_hi, n1)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
            h_hi, _ = _worst_shift_descent(h_hi, 300, 0.003)
            h_hi = _repair(h_hi, n1)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase E: Structured restarts at best resolution
    nc = best_n
    if _el() < budget_s - 200:
        t_arr = np.arange(nc)
        for _r in range(15):
            if _el() > budget_s - 200:
                break
            hp = best_h.copy()
            strat = rng.randint(5)
            if strat == 0:
                hp += 0.008 * rng.randn(nc)
            elif strat == 1:
                f = rng.randint(2, nc // 4 + 1)
                hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * t_arr / nc)
            elif strat == 2:
                hp = 1.0 - best_h + 0.004 * rng.randn(nc)
            elif strat == 3:
                bs = rng.randint(5, max(6, nc // 8))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.02 * rng.randn(bs)
            else:
                f1 = rng.randint(1, nc // 3 + 1)
                f2 = rng.randint(1, nc // 3 + 1)
                hp += 0.006 * np.sin(2 * np.pi * f1 * t_arr / nc)
                hp += 0.004 * np.cos(2 * np.pi * f2 * t_arr / nc)
            hp = _repair(hp, nc)
            a_pick = [30000, 100000, 300000][rng.randint(3)]
            h_new = _slsqp_opt(hp, a_pick, 2000, 1e-18, 0.0)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase F: Gradient-informed fine local search
    if _el() < budget_s - 40:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(12000):
            if _el() > budget_s - 40:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 8e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5, best_h, best_n = c5c, h.copy(), nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase G: Ultra-fine pair search
    if _el() < budget_s - 15:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(15000):
            if _el() > budget_s - 15:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5, best_h, best_n = c5c, h.copy(), nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase H: Final SLSQP polish
    if _el() < budget_s - 10:
        h = _slsqp_opt(best_h, 800000, 3000, 1e-19, 0.0)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
