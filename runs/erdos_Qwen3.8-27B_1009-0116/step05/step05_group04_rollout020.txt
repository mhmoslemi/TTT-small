def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _lag_grad(h, j):
    """Gradient of lag-j integral w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - j
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[sp:])
        g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _grad_topm(h, m, p):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[-m:]
    ct = c[top]
    if ct.max() < 1e-18:
        return np.zeros(n)
    base = np.sum(ct**p) ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for ii in range(m):
        j = top[ii]
        wj = base * c[j] ** (p - 1)
        g += wj * _lag_grad(h, j)
    return g
