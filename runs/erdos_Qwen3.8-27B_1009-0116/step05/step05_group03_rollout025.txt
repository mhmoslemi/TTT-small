import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _worst_lag_grad(h):
    """Gradient of the worst-lag overlap."""
    n = len(h)
    c = _corr(h)
    k = int(np.argmax(c))
    s = abs(k - (n - 1))
    if s == 0:
        return np.zeros(n), k
    m = n - s
    g = np.zeros(n)
    # c[k] = sum_{i=0}^{m-1} h[i]*(1-h[i+s])
    # dh[i] in c[k]: (1-h[i+s]) for i < m, and -h[i-s] for i >= s
    g[:m] += (1.0 - h[s:s+m])
    g[s:n] -= h[:m]
    return g, k

def _slsqp_powmean(h, dx, p, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(x)
        cm = c.max()
        if cm < 1e-15:
            return 0.0
        ratio = c / cm
        return cm * float(np.sum(ratio**p)) ** (1.0/p) / (len(c) ** (1.0/p))
    
    def eq(x):
        return x.sum() - tgt
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(np.clip(r.x, 0, 1), tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _worst_grad(h):
    n = len(h)
    c = _corr(h)
    k = int(np.argmax(c))
    s = abs(k - (n - 1))
    if s == 0:
        return np.zeros(n), k
    m = n - s
    g = np.zeros(n)
    g[:m] += (1.0 - h[s:s + m])
    g[s:n] -= h[:m]
    return g, k
