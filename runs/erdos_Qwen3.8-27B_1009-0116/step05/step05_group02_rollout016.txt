    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[s:n] += (1.0 - h[0:n-s])
            g[0:n-s] -= h[s:n]
    else:
        t = -s
        if t < n:
            g[0:n-t] += (1.0 - h[t:n])
            g[t:n] -= h[0:n-t]
    return g

def _gradient_step(h, c, dx, K, power, alpha, rng):

    idx = np.argpartition(-c, K - 1)[:K - 1]
    cm = c.max()

        c5_new = _eval(h_new)
        c5_old = c.max() * dx
        if c5_new < c5_old - 1e-15:
            return h_new, f, True
    return h, alpha, False

def _gradient_optimizer(h, budget_end, rng, alpha_init=0.3, K=8, power=2.0):
