    # a: change per unit d (2n-1 array)  
    # Find d in [dlo, dhi] minimizing max_k (c[k] + d*a[k])
    # Try boundary points and intersection points of top lines
    candidates = [dlo, dhi, 0.0]
    # Get top 20 lags
    idx = np.argsort(c)[::-1][:20]
    for i in range(min(10, len(idx))):
        for j in range(i+1, min(10, len(idx))):
            if abs(a[idx[i]] - a[idx[j]]) > 1e-15:
                d = (c[idx[j]] - c[idx[i]]) / (a[idx[i]] - a[idx[j]])
                if dlo <= d <= dhi:
                    candidates.append(d)
    best_d = 0.0
    best_v = np.inf
    for d in candidates:
        v = np.max(c + d * a)
        if v < best_v:
            best_v = v
            best_d = d
    return best_d, best_v

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _pair_delta(h, i, j):
    n = len(h)
    sh = np.arange(-(n - 1), n)
    a = np.zeros(2 * n - 1)
    v1 = (i + sh >= 0) & (i + sh < n)
    a[v1] += 1.0 - h[i + sh[v1]]

    a[v1j] -= 1.0 - h[j + sh[v1j]]
    v2j = (j - sh >= 0) & (j - sh < n)
