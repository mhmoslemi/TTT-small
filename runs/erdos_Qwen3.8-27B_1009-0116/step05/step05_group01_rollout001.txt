                    diff -= a

                    a = min(-diff, h[idx])

    vp_i = i + lags
    vq_i = i - lags
    vp_j = j + lags
    vq_j = j - lags
    s = np.zeros(2 * n - 1)
    v1 = (vp_i >= 0) & (vp_i < n)
    s[v1] += (1.0 - h[vp_i[v1]])
    v2 = (vq_i >= 0) & (vq_i < n)
    s[v2] -= h[vq_i[v2]]
    v3 = (vp_j >= 0) & (vp_j < n)
    s[v3] -= (1.0 - h[vp_j[v3]])
    v4 = (vq_j >= 0) & (vq_j < n)

    d_lo = max(-h[i], h[j] - 1.0)

        return 0.0
    lo, hi_ = d_lo, d_hi
    for _ in range(nb):

        vals = c_vec + mid * s
        mv = vals.max()
        active = vals >= (mv - 1e-13)
        if not active.any():
            active = vals == mv
        sg_lo = s[active].min()
        sg_hi = s[active].max()
        if sg_lo <= 0.0 and sg_hi >= 0.0:
            lo = mid
            hi_ = mid
        elif sg_lo > 0.0:
            hi_ = mid
        else:
            lo = mid
    return 0.5 * (lo + hi_)

def _pos_delta(h, i, d, n):

    dc = np.zeros(2 * n - 1)
    vp = i + lags
    vq = i - lags
    v1 = (vp >= 0) & (vp < n)
    dc[v1] += d * (1.0 - h[vp[v1]])
    v2 = (vq >= 0) & (vq < n)
    dc[v2] -= d * h[vq[v2]]
    return dc

def _pairwise_opt(h, c_vec, rng, n_iter, n):

    for _ in range(n_iter):
        i = rng.randint(n)

        c_new = c_vec + dc

    m = c_vec.max()
    s_exp = np.clip(alpha * (c_vec - m), -500, 500)

    wsh = s_exp.max()
    w = np.exp(s_exp - wsh)
    w /= w.sum()
