    if time.time() - t0 < budget_s * 0.95 and bc5 > 0.3808:
        for _r in range(5):
            if time.time() - t0 > budget_s * 0.94:
                break
            if rng.rand() < 0.5:
                hp = _rpr(bh + 0.01 * rng.randn(bn), bn)
            else:
                hp = _rpr(0.5 + 0.3 * np.sin(2 * np.pi * rng.randint(2, 6) * np.arange(bn) / bn + rng.rand() * 6.28) * 0.4 + 0.05 * rng.randn(bn), bn)
            hp, c5r = _pg(hp, 5000, 1.0, 3000, 20)
            if c5r < bc5:
                bc5 = c5r
                bh = hp.copy()
    bh = _rpr(bh, bn)
    bc5 = _ev(bh)
    return (bh, bc5, bn)
