import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _corr_raw(h):
    return np.correlate(h, 1.0-h, mode='full')

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _repair(h, n):
    return _proj(np.clip(h, 0, 1), n*0.5)

def _lsm_obj(h, alpha):
    c = _corr_raw(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -300, 300)
    return m + np.log(np.sum(np.exp(s)))/alpha

def _lsm_grad(h, alpha):
    n = len(h)
    c = _corr_raw(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1)-k
        wk = w[k]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1-h[sh:])
            if sh > 0:
                g[sh:] -= wk*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]

def _lsm_g(h, alpha):

        if sh >= 0:
            g[:n-sh] += wk*(1.0-h[sh:])
            if sh > 0: g[sh:] -= wk*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _slsqp(h, alpha, mi=2000):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:
        r = minimize(_lsm, h.copy(), jac=_lsm_g, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter':mi,'ftol':1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nest(h, alpha, steps, s0=0.003, mu=0.85):
    n = len(h); f = _lsm(h, alpha)
    v = np.zeros(n); step = s0
    for _ in range(steps):
        g = _lsm_g(h, alpha)
        v = mu*v - step*g
        bh, bf = h, f; lam = 1.0
        for _ in range(30):
            ht = _proj(h-lam*v, n*0.5)
            fn = _lsm(ht, alpha)
            if fn < f: bh, bf = ht, fn; break
            lam *= 0.5
        h, f = bh, bf
        step = min(step*1.02, s0*5.0)
    return h

def _pairwise(h, n, ntry, rng):
    cnt = 2*n-1
    C = np.correlate(h, 1.0-h, mode='full').copy()
    ks = np.arange(cnt)
    contrib = np.zeros((n, cnt))
    for p in range(n):
        bj = p+ks-n+1; ai = p-ks+n-1
        vb = (bj>=0)&(bj<n); va = (ai>=0)&(ai<n)
        contrib[p,vb] = 1.0-h[bj[vb]]
        contrib[p,va] -= h[ai[va]]
    c5 = float(np.max(C)*2.0/n)
    ls = np.arange(n)
    for _ in range(ntry):
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j: continue
        d = contrib[i]-contrib[j]
        lo = max(h[j]-1.0, -h[i]); hi = min(h[j], 1.0-h[i])
        if hi-lo < 1e-13: continue
        bd, bv = 0.0, c5
        for f in (0.0,0.2,0.4,0.5,0.6,0.8,1.0):
            delta = lo+f*(hi-lo)
            nv = float(np.max(C+delta*d)*2.0/n)
            if nv < bv: bd, bv = delta, nv
        if abs(bd) < 1e-14: continue
        C += bd*d; h[i]+=bd; h[j]-=bd; c5=bv
        for p in (i,j):
            bj = p+ks-n+1; ai = p-ks+n-1
            vb = (bj>=0)&(bj<n); va = (ai>=0)&(ai<n)
            contrib[p, :] = 0.0
            contrib[p,vb] = 1.0-h[bj[vb]]
            contrib[p,va] -= h[ai[va]]
        excl = (ls!=i)&(ls!=j)
        for kk, sg in [(n-1+i, -1), (n-1+j, 1), (-n+1+i, -1), (-n+1+j, 1)]:
            for s2 in ([0] if sg==-1 else [0]):
                pass
        # Vectorized update for other rows
        for offset, sg in [(i, -1.0), (j, 1.0)]:
            kA = n-1+offset-ls
            kB = -n+1+offset+ls
            mA = excl&(kA>=0)&(kA<cnt)
            contrib[ls[mA], kA[mA]] += sg*bd
            mB = excl&(kB>=0)&(kB<cnt)
            contrib[ls[mB], kB[mB]] += sg*bd
    return h, c5

def _upsample(h, fac):
    n = len(h)
    return np.repeat(h, fac)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time()-t0
    
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh, bc5, bn = h.copy(), _eval(h), n0
    
    # Phase 1: SLSQP
    for alpha in [500, 5000, 50000, 200000]:
        if _el() > 120: break
        h_new = _slsqp(bh, alpha, 3000)
        c5 = _eval(h_new)
        if c5 < bc5: bc5, bh, bn = c5, h_new.copy(), n0
    
    # Phase 2: Nesterov
    if _el() < budget_s-400:
        h_new = _nest(bh, 30000, 400, 0.003, 0.85)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < bc5: bc5, bh = c5, h_new.copy()
    
    # Phase 3: Pairwise
    if _el() < budget_s-300:
        h_new, c5 = _pairwise(bh.copy(), n0, 100000, rng)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < bc5: bc5, bh = c5, h_new.copy()
