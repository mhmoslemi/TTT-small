def _repair(h, n):

                h[idx] += a
                diff -= a

                diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _soft_obj(x, alpha, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    obj = m + np.log(float(np.sum(np.exp(s)))) / alpha
    n = len(x)
    obj += pen * (x.sum() - 0.5 * n) ** 2
    return obj

def _soft_grad(x, alpha, pen):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    g += 2.0 * pen * (x.sum() - 0.5 * n)
    return g

def _slsqp(h, alpha, pen, nit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha, pen),
                     method='SLSQP', bounds=bnds, options={'maxiter': nit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def _bin_steepest(b, t_end):

    c5 = _eval(b)
    for _ in range(500):

        ones = np.where(b == 1)[0]

        best_c5 = c5

        for i in ones:
            for j in zeros:

                b[j] = 1.0
                cv = _eval(b)

        if best_i >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            c5 = best_c5
        else:

    return b, c5

def _interp(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b
