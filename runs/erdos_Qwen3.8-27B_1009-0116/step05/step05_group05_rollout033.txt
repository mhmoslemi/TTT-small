import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_c(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    return val

def _smooth_g(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _lbfgs_phase(h, t0, tmax):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    best = h.copy()
    best_c = _eval(best)
    for alpha in [500, 2000, 10000, 50000, 200000, 500000, 1000000]:
        if time.time() - t0 > tmax:
            break
        pen = 100.0
        def f(x):
            return _smooth_c(x, alpha) + pen * (x.sum() - n*0.5)**2
        def g(x):
            return _smooth_g(x, alpha) + 2*pen*(x.sum()-n*0.5)*np.ones(n)
        try:
            res = minimize(f, best, jac=g, method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12})
            best = _repair(res.x, n)
            c = _eval(best)
            if c < best_c:
                best_c = c
        except:
            pass
    return best, best_c

def _active_descent(h, max_iters, n_top, t0, tmax):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        if time.time() - t0 > tmax:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += w * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= w * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n-sp])
                g[:n-sp] -= w * h[sp:]
        g *= dx
        improved = False
        step = 0.5
        for _ in range(20):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, c5c

def _coord2d(h, n_rounds, t0, tmax, rng):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for r in range(n_rounds):
        if time.time() - t0 > tmax:
            break
        if r % 3 == 0:
            i = (r * 7 + 3) % n
            j = (i + n // 2) % n
        elif r % 3 == 1:
            i = (r * 11 + 1) % n
            j = (r * 13 + 5) % n
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0) + 1e-12
        dhi = min(1.0 - h[i], h[j]) - 1e-12
        if dlo >= dhi:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        # Compute delta vector
        delta = np.zeros(2*n-1)
        for k in range(2*n-1):
            sh = (n-1) - k
            di = 0.0
            dj = 0.0
            if sh >= 0:
                if i < n - sh:
                    di += (1.0 - h[i + sh])
                if i >= sh:
                    di -= h[i - sh]
                if j < n - sh:
                    dj += (1.0 - h[j + sh])
                if j >= sh:
                    dj -= h[j - sh]
            else:
                sp = -sh
                if i >= sp:
                    di += (1.0 - h[i - sp])
                if i + sp < n:
                    di -= h[i + sp]
                if j >= sp:
                    dj += (1.0 - h[j - sp])
                if j + sp < n:
                    dj -= h[j + sp]
            delta[k] = dx * (di - dj)
        # Minimize max(c0 + d * delta) over d in [dlo, dhi]
        # Ternary search (convex PL function)
        a_t, b_t = dlo, dhi
        for _ in range(25):
            m1 = a_t + (b_t - a_t) / 3.0
            m2 = b_t - (b_t - a_t) / 3.0
            if np.max(c0 + m1 * delta) < np.max(c0 + m2 * delta):
                b_t = m2
            else:
                a_t = m1
        best_d = (a_t + b_t) * 0.5
        if abs(best_d) < 1e-12:
            continue
        h[i] += best_d
        h[j] -= best_d
        h = np.clip(h, 0.0, 1.0)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i] -= best_d
            h[j] += best_d
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out

- For sh = 0 to i: e[k] -= h[i-sh]  [the "second" term, when i-sh ≥ 0]
- For sh = -(n-1-i) to -1 (i.e., sp = 1 to n-1-i): e[k] -= h[i+sp]  [the "second" term for negative shifts, when i+sp < n]
- For sp = 1 to i: e[k] += (1-h[i-sp])  [the "first" term for negative shifts, when i ≥ sp]

So:
Positive shifts (sh = 0 to n-1):
  k = n-1+sh
  First term: if sh ≤ n-1-i, add (1-h[i+sh]) at k = n-1+sh
  Second term: if sh ≤ i, subtract h[i-sh] at k = n-1+sh

Negative shifts (sh = -(n-1) to -1, i.e., sp = 1 to n-1):
  k = n-1-sh = n-1-sp
  First term: if sp ≤ i, add (1-h[i-sp]) at k = n-1-sp
  Second term: if sp ≤ n-1-i, subtract h[i+sp] at k = n-1-sp

So:
```python
def _effect_vec(h, i):
    n = len(h)
    dx = 2.0/n
    e = np.zeros(2*n-1)
    # Positive shifts: k = n-1+sh, sh=0..n-1
    for sh in range(n):
        k = n-1+sh
        if sh <= n-1-i:
            e[k] += (1.0 - h[i+sh])
        if sh <= i:
            e[k] -= h[i-sh]
    # Negative shifts: k = n-1-sp, sp=1..n-1
    for sp in range(1, n):
        k = n-1-sp
        if sp <= i:
            e[k] += (1.0 - h[i-sp])
        if sp <= n-1-i:
            e[k] -= h[i+sp]
    return e * dx
```

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_val(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lbfgs_smooth(h, t0, tlimit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    best_c = _eval(h)
    for alpha in [800, 3000, 15000, 60000, 300000, 800000, 2000000]:
        if time.time() - t0 > tlimit:
            break
        pen = 50.0
        def f(x):
            return _smooth_val(x, alpha) + pen * (x.sum() - n * 0.5) ** 2
        def g(x):
            return _smooth_grad(x, alpha) + 2 * pen * (x.sum() - n * 0.5) * np.ones(n)
        try:
            res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': 50000})
            h_new = _repair(res.x, n)
            c_new = _eval(h_new)
            if c_new < best_c:
                h = h_new
                best_c = c_new
        except Exception:
            pass
    return h, best_c

def _active_descent(h, max_iters, n_top, t0, tlimit):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        if time.time() - t0 > tlimit:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += w * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= w * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - h[:n - sp])
                g[:n - sp] -= w * h[sp:]
        g *= dx
        improved = False
        step = 1.0
        for _ in range(25):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h, c5c = h_new, c5n
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, c5c

def _coord2d(h, n_rounds, t0, tlimit, rng):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for r in range(n_rounds):
        if time.time() - t0 > tlimit:
            break
        if r % 4 == 0:
            i = (r * 7 + 3) % n
            j = (i + n // 3) % n
        elif r % 4 == 1:
            i = (r * 11 + 1) % n
            j = (i + n // 2) % n
        elif r % 4 == 2:
            i = (r * 13 + 2) % n
            j = (i + n // 5) % n
        else:
            i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0) + 1e-12
        dhi = min(1.0 - h[i], h[j]) - 1e-12
        if dlo >= dhi - 1e-10:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        delta = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            di = dj = 0.0
            if sh >= 0:
                if i < n - sh:
                    di += 1.0 - h[i + sh]
                if i >= sh:
                    di -= h[i - sh]
                if j < n - sh:
                    dj += 1.0 - h[j + sh]
                if j >= sh:
                    dj -= h[j - sh]
            else:
                sp = -sh
                if i >= sp:
                    di += 1.0 - h[i - sp]
                if i + sp < n:
                    di -= h[i + sp]
                if j >= sp:
                    dj += 1.0 - h[j - sp]
                if j + sp < n:
                    dj -= h[j + sp]
            delta[k] = dx * (di - dj)
        a_t, b_t = dlo, dhi
        for _ in range(22):
            m1 = a_t + (b_t - a_t) / 3.0
            m2 = b_t - (b_t - a_t) / 3.0
            if np.max(c0 + m1 * delta) <= np.max(c0 + m2 * delta):
                b_t = m2
            else:
                a_t = m1
        best_d = (a_t + b_t) * 0.5
        if abs(best_d) < 1e-11:
            continue
        h[i] += best_d
        h[j] -= best_d
        h = np.clip(h, 0.0, 1.0)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i] -= best_d
            h[j] += best_d
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    n0 = len(initial_h_values)

            hp = _repair(best_h + 0.015 * rng.randn(n_cur), n_cur)
        elif r == 1:
            hp = _repair(1.0 - best_h + 0.01 * rng.randn(n_cur), n_cur)
        elif r == 2:

            for _ in range(3):

    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
