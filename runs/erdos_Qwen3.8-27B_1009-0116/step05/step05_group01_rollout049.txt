        g[(h <= 1e-10) & (g < 0)] = 0.0
        g[(h >= 1.0 - 1e-10) & (g > 0)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _repair(h - step * g, n)
            if np.array_equal(h_new, h):
                continue
            c_new = _corr(h_new)
            v = c_new.max()
            if v < prev_peak - 1e-13:
                h = h_new
                c = c_new
                prev_peak = v
                improved = True
                break
        if not improved:
            break
    return h

def _multi_point_search(h, c, c5c, rng, n, n_iter, scales):
    dx = 2.0 / n
    n_active = len(c)
    for _ in range(n_iter):
        kp = rng.choice([2, 3, 4], p=[0.5, 0.3, 0.2])
        idxs = rng.choice(n, kp, replace=False)
        peak_k = c.argmax()
        m_pk = peak_k - (n - 1)
        g_sub = np.array([_grad_lag(h, m_pk, n)[i] for i in idxs])
        d = -g_sub * 0.01
        d -= d.mean()
        scale = rng.choice(scales)
        d *= scale / (np.abs(d).max() + 1e-15)
        for i, idx in enumerate(idxs):
            d[i] = np.clip(d[i], -h[idx] + 1e-12, 1.0 - h[idx] - 1e-12)
        if np.all(np.abs(d) < 1e-13):
            continue
        new_h = h.copy()
        new_h[idxs] += d
        new_c = _corr(new_h)
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-14:
            h = new_h
            c = new_c
            c5c = new_c5
    return h, c, c5c

def _pair_fine(h, c, c5c, rng, n, n_iter, scale):
    dx = 2.0 / n
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        h[i], h[j] = ni, nj
        new_c = _corr(h)
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-14:
            c = new_c
            c5c = new_c5
        else:
            h[i], h[j] = h[j] - (h[j] - nj), nj  # revert
            h[i], h[j] = ni - d, nj + d  # back to original
            h[i], h[j] = h[j] + d, ni - d  # this is getting confused
            # Just recompute
    return h, c, c5c

def _pair_incremental(h, c, c5c, rng, n, n_iter, scale):
    dx = 2.0 / n
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        orig_i, orig_j = h[i], h[j]
        d = scale * rng.randn()
        h[i], h[j] = orig_i + d, orig_j - d
        if h[i] < -1e-12 or h[i] > 1.0 + 1e-12 or h[j] < -1e-12 or h[j] > 1.0 + 1e-12:
            h[i], h[j] = orig_i, orig_j
            continue
        new_c = _corr(h)
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-14:
            c = new_c
            c5c = new_c5
        else:
            h[i], h[j] = orig_i, orig_j
    return h, c, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_start = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_start.copy()
    best_c5 = _eval_c5(best_h)
    best_n = n0

    # Phase 1: SLSQP refinement
    for alpha in [5000, 50000, 300000]:
        if _el() > 80:
            break
        hn = _slsqp_run(best_h, alpha, 2500, 1e-19)
        c5 = _eval_c5(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Exact gradient descent on peak
    if _el() < budget_s - 500:
        n_g = len(best_h)
        hg = _grad_descend(best_h.copy(), n_g, n_iter=400, top_lags=3)
        c5g = _eval_c5(hg)
        if c5g < best_c5:
            best_c5, best_h = c5g, hg

    # Phase 3: Multi-point search (coarse to fine)
    if _el() < budget_s - 300:
        n = len(best_h)
        hm = best_h.copy()
        cm = _corr(hm)
        c5m = cm.max() * 2.0 / n
        for scales in [[0.08, 0.04, 0.02], [0.02, 0.01, 0.005], [0.005, 0.002, 0.001]]:
            if _el() > budget_s - 300:
                break
            hm, cm, c5m = _multi_point_search(hm, cm, c5m, rng, n, 50000, scales)
        c5m = _eval_c5(hm)
        if c5m < best_c5:
            best_c5, best_h = c5m, hm

    # Phase 4: SLSQP polish
    if _el() < budget_s - 150:
        hp = _slsqp_run(best_h, 800000, 4000, 1e-20)
        c5p = _eval_c5(hp)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 5: Fine pair search
    if _el() < budget_s - 50:
        n = len(best_h)
        hf = best_h.copy()
        cf = _corr(hf)
        c5f = cf.max() * 2.0 / n
        for sc in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if _el() > budget_s - 50:
                break
            hf, cf, c5f = _pair_incremental(hf, cf, c5f, rng, n, 60000, sc)
        c5f = _eval_c5(hf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    # Phase 6: Try coarser resolution (n=140)
    if _el() < budget_s - 200:
        n_coarse = 140
        # Downsample by averaging
        block = len(best_h) // n_coarse
        if block >= 1:
            hc = np.array([best_h[i * block:(i + 1) * block].mean()
                           for i in range(n_coarse)])
            if block * n_coarse < len(best_h):
                hc = np.append(hc, best_h[block * n_coarse:].mean())
            hc = hc[:n_coarse]
            hc = _repair(hc, n_coarse)
            c5c = _eval_c5(hc)
            # Optimize at coarse resolution
            if _el() < budget_s - 150:
                hc = _slsqp_run(hc, 50000, 2000, 1e-18)
                hc = _grad_descend(hc, n_coarse, 200, 2)
                c5c = _eval_c5(hc)
            hc, cc, c5cc = _pair_incremental(hc, _corr(hc), _eval_c5(hc), rng, n_coarse, 40000, 0.001)
            c5c = _eval_c5(hc)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, hc, n_coarse

    # Phase 7: Final gradient + pair polish on best
    if _el() < budget_s - 20:
        n = len(best_h)
        hf = _grad_descend(best_h.copy(), n, 200, 2)
        c5f = _eval_c5(hf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf
        if _el() < budget_s - 5:
            hf = _pair_incremental(best_h, _corr(best_h), best_c5, rng, n, 40000, 5e-5)
            c5f = _eval_c5(f if np.size(f) else best_h) if False else _eval_c5(best_h)
            # just re-evaluate
            hf = best_h
            c5f = _eval_c5(hf)
            if c5f < best_c5:
                best_c5, best_h = c5f, hf

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, len(best_h))
