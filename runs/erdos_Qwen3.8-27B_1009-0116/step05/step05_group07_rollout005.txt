            m = c.max()
            s = np.clip(alpha * (c - m), -500, 500)
            w = np.exp(s - s.max())
            w /= w.sum()
            for kk in range(2 * n - 1):
                if w[kk] < 1e-14:
                    continue
                g += w[kk] * _grad_vec(kk, y)
            t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2))
            step = 0.003
            for bt in range(10):
                y_new = _proj_sum(y - step * g, target)
                h_new = (y_new + (t_k - 1.0) / t_k * (y_new - h))
                h_new = _proj_sum(h_new, target)
                fn = _eval(h_new)
                if fn < f + 1e-15:
                    f = fn
                    h = h_new
                    y = y_new
                    break
                step *= 0.5
            else:
                break
            if f < best_f:
                best_f = f
                best_h = h.copy()
        h, f = best_h, best_f
        y = h.copy()
    return h, f

def _active_descent(h0, topm, max_iter, step0):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    step = step0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 30:
            break
        c = _corrs_s(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_vec(k, h)
        g /= topm
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, target)
            fn = _eval(h_new)
            if fn < f - 1e-10:
                h = h_new
                f = fn
                found = True
                break
            step *= 0.5
        if not found:
            step *= 1.1
            if step > step0:
                step = step0
            continue
        step = min(step * 1.15, step0)
    return h, f

def _pairwise_cd(h0, n_iter, scale, seed_val):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    c = _corrs_s(h)
    top_idx = np.argsort(c)[::-1][:20]
    g_active = np.zeros(n)
    for k in top_idx:
        g_active += _grad_vec(k, h)
    g_active /= len(top_idx)
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 20:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        fn = _eval(h)
        if fn < f - 1e-15:
            f = fn
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, f

def _steepest_pair(h0, max_iter):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    c = _corrs_s(h)
    top_idx = np.argsort(c)[::-1][:5]
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 25:
            break
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_vec(k, h)
        g /= len(top_idx)
        best_delta = 0.0
        i_max = 0
        j_max = 1
        for i in range(n):
            for j in range(i + 1, n):
                delta = g[i] - g[j]
                if delta > best_delta:
                    best_delta = delta
                    i_max, j_max = i, j
        if best_delta < 1e-10:
            break
        d = 0.02
        for bt in range(15):
            ni, nj = h[i_max] + d, h[j_max] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                d *= 0.5
                continue
            h[i_max], h[j_max] = ni, nj
            fn = _eval(h)
            if fn < f:
                f = fn
                break
            h[i_max], h[j_max] = ni - d, nj + d
            d *= 0.5
    return h, f

def _upsample_cubic(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n * factor, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _el():
        return time.time() - _t0

    # Phase 1: Multiple SLSQP passes with increasing alpha at n0
    for alpha in [2000, 8000, 30000, 80000, 200000]:
        if _el() > 60:
            break
        hn = _slsqp(best_h, alpha, 2000, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 2: Nesterov PGD with moderate alphas
    if _el() < budget_s - 200:
        alphas_pgd = [5000, 20000, 50000]
        hn, fc = _nesterov_pgd(best_h, alphas_pgd, 300)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 3: Active-set descent with various top-m
    if _el() < budget_s - 250:
        for topm in [3, 5, 8, 10, 15]:
            if _el() > budget_s - 200:
                break
            hn, fc = _active_descent(best_h, topm, 5000, 0.003)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 4: SLSQP polish with very high alpha
    if _el() < budget_s - 150:
        for alpha in [500000, 1000000]:
            if _el() > budget_s - 120:
                break
            hn = _slsqp(best_h, alpha, 3000, 1e-20)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 5: Steepest pair descent (structured, not random)
    if _el() < budget_s - 100:
        hn, fc = _steepest_pair(best_h, 200)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 6: More active descent + SLSQP alternation
    if _el() < budget_s - 80:
        for _alt in range(4):
            if _el() > budget_s - 60:
                break
            hn, fc = _active_descent(best_h, 12, 3000, 0.002)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()
            hn = _slsqp(best_h, 300000, 1500, 1e-19)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 7: Pairwise coordinate descent (fine-scale)
    for scale, nit in [(0.005, 10000), (0.002, 15000), (0.0005, 20000), (0.0001, 15000)]:
        if _el() > budget_s - 30:
            break
        hn, fc = _pairwise_cd(best_h, nit, scale, seed + int(scale * 1e7))
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 8: Try upsample to 560 and quick SLSQP
    n1 = n0 * 2
    if _el() < budget_s - 60:
        h_up = _upsample_cubic(best_h, 2)
        hn = _slsqp(h_up, 50000, 1500, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()
            best_n = n1

    # Phase 9: Final active descent polish
    if _el() < budget_s - 15:
        hn, fc = _active_descent(best_h, 8, 2000, 0.001)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 10: Final SLSQP
    if _el() < budget_s - 5:
        hn = _slsqp(best_h, 500000, 2000, 1e-20)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
