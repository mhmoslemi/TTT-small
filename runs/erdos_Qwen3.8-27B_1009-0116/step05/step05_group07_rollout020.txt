        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_f_g(h, alpha):
    n = len(h)
    c = _corrs_scaled(h)
    m = c.max()
    scaled = np.clip(alpha * (c - m), -500, 500)
    val = m + np.log(np.exp(scaled - scaled.max()).sum()) / alpha
    w = np.exp(scaled - scaled.max())
    w_rep = w.copy()
    w_rep[scaled - scaled.max() < -400] = 0
    w /= w.sum()
    g = np.zeros(n)
    significant = np.where(w > 1e-12)[0]
    for k in significant:
        s = k - (n - 1)
        g += w[k] * _grad_lag(s, h)
    return val, g

def _fista(h, alpha, n_iter, t0_offset):
    n = len(h)
    target = n * 0.5
    h = _repair(h.copy(), n)
    y = h.copy()
    h_prev = h.copy()
    t = 1.0
    f_curr = _eval(h)
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 30:
            break
        _, grad = _smooth_f_g(y, alpha)
        L = 4.0 * (2.0 * n) / n  # rough Lipschitz
        step = 1.0 / L
        y_new = y - step * grad
        h_new = _proj_sum(y_new, target)
        mu = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t - 1.0) / mu / (mu / (t + 1.0)) * (h_new - h_prev)
        y = _proj_sum(y, target)
        h_prev = h_new
        h_new = y
        t = mu
        f_new = _eval(h)
        if f_new < f_curr - 1e-15:
            f_curr = f_new
            h = h_new
    return h, f_curr

def _active_descent_v2(h, topm, max_iter, step0):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    mom = 0.8
    v = np.zeros(n)
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 25:
            break
        c = _corrs_scaled(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k - (n - 1), h)
        g /= topm
        v = mom * v + (1 - mom) * g
        found = False
        for _bt in range(40):
            h_new = _proj_sum(h - step * v, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-10:
                h = h_new; f = f_new; found = True
                break
            step *= 0.5
        if not found:
            step *= 1.05
            step = min(step, step0)
            if step < 1e-10:
                break
    return h, f

def _coord_descent(h, n_iter, topm):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs_scaled(h)
        active_lags = np.argsort(c)[::-1][:topm]
        improved = False
        for k in active_lags:
            s = k - (n - 1)
            if s == 0:
                continue
            sp = abs(s)
            pairs = range(n - sp) if s > 0 else range(n - sp)
            for t in range(0, n - sp, max(1, (n - sp) // 50)):
                i, j = t, t + sp
                hi, hj = h[i], h[j]
                for d in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                    ni, nj = hi + d, hj - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        h = _repair(h, n)
                        fn = _eval(h)
                        if fn < f - 1e-14:
                            f = fn; improved = True
                            break
                else:
                    h[i], h[j] = hi, hj
                    h = _repair(h, n)
        if not improved:
            break
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_f_g(x, alpha)[0]
    def jac(x):
        return _smooth_f_g(x, alpha)[1]
    def cons_f(x):
        return x.sum() - n * 0.5
    def cons_j(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cons_f, 'jac': cons_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - _t0

    # Phase 1: FISTA at n0 with varying alpha
    for alpha in [3000, 10000, 50000]:
        if _el() > 60:
            break
        hn, fc = _fista(best_h, alpha, 2000, 0)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase 2: SLSQP polish at n0
    if _el() < budget_s - 200:
        for alpha in [20000, 100000]:
            if _el() > budget_s - 180:
                break
            hn = _slsqp_polish(best_h, alpha, 800)
            fc = _eval(hn)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase 3: Active descent at n0
    if _el() < budget_s - 150:
        for tm in [3, 5, 10, 20]:
            if _el() > budget_s - 130:
                break
            hn, fc = _active_descent_v2(best_h, tm, 5000, 0.004)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase 4: Upsample to 2*n0 and optimize
    if _el() < budget_s - 120:
        n1 = min(n0 * 2, 560)
        h1 = _upsample(best_h, n1 // n0) if n1 != n0 else best_h.copy()
        if n1 != n0:
            for alpha in [5000, 30000]:
                if _el() > budget_s - 100:
                    break
                hn, fc = _fista(h1, alpha, 3000, 0)
                if fc < best_c5:
                    best_c5 = fc; best_h = hn.copy()
            if _el() < budget_s - 80:
                hn, fc = _active_descent_v2(best_h, 8, 3000, 0.003)
                if fc < best_c5:
                    best_c5 = fc; best_h = hn.copy()

    # Phase 5: Coordinate descent on best resolution
    if _el() < budget_s - 40:
        nc = len(best_h)
        hn, fc = _coord_descent(best_h, 2000, 8)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase 6: Multi-restart with moderate perturbations
    nc = len(best_h)
    if _el() < budget_s - 30:
        for _r in range(8):
            if _el() > budget_s - 20:
                break
            scale = 0.005 + 0.01 * _r
            hp = best_h + scale * rng.randn(nc)
            hp = _repair(hp, nc)
            hn, fc = _fista(hp, 30000, 1500, 0)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()
                hn, fc2 = _active_descent_v2(best_h, 5, 2000, 0.003)
                if fc2 < best_c5:
                    best_c5 = fc2; best_h = hn.copy()

    # Phase 7: Final active + FISTA polish
    if _el() < budget_s - 10:
        hn, fc = _active_descent_v2(best_h, 12, 3000, 0.002)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        hn, fc = _fista(best_h, 200000, 1000, 0)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase 8: SLSQP final polish
    if _el() < budget_s - 5:
        hn = _slsqp_polish(best_h, 500000, 1000)
        fc = _eval(hn)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
