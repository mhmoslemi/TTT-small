import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_budget = 0.0

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corrs(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s)) + 1e-300) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corrs(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 5e-5:
            continue
        sh = k - (n - 1)
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:sp] -= wk * h[sp:]
    return g

def _proj_sum_box(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _lbfgs(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-16,
                                'maxfun': maxiter * 3, 'maxls': 40, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_mom(h, n_iter, topm, step0, beta):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    f = _eval(h)
    mom = np.zeros(n)
    step = step0
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        c = _corrs(h) * (2.0 / n)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            sh = k - (n - 1)
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:sp] -= h[sp:]
        g *= (2.0 / n) / topm
        mom = beta * mom + (1.0 - beta) * g
        h_new = _proj_sum_box(h - step * mom, target)
        f_new = _eval(h_new)
        if f_new <= f + 1e-14:
            h = h_new
            f = f_new
            step = min(step * 1.015, step0)
        else:
            h = h_new
            step *= 0.75
        if it > 4000:
            step = max(step, step0 * 0.3)
    return h, f

def _pairwise(h, n_iter, scale, rng):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.randn() * scale
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _block_swap(h, rng, block_size):
    n = len(h)
    h = h.copy()
    if n < 2 * block_size:
        block_size = max(2, n // 4)
    a = rng.randint(0, n - block_size)
    b = rng.randint(0, n - block_size)
    if abs(a - b) < block_size:
        return h
    h[a:a+block_size], h[b:b+block_size] = h[b:b+block_size].copy(), h[a:a+block_size].copy()
    return h

def _try_binarize(h, n):
    target = n / 2.0
    h_sorted = np.sort(h)[::-1]
    binary = np.where(h_sorted > 0.5, 1.0, 0.0)
    s = binary.sum()
    if abs(s - target) < n * 0.3:
        h_bin = np.zeros(n)
        h_bin[np.argsort(h)[::-1][:int(round(target))]] = 1.0
        h_bin = _repair(h_bin, n)
        return h_bin
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _tl():
        return _budget - (time.time() - _t0)

    # Phase 1: L-BFGS-B alpha annealing (fast optimizer)
    alphas = [80, 250, 800, 2500, 8000, 25000, 80000]
    for alpha in alphas:
        if _tl() < 45:
            break
        hn = _lbfgs(best_h, alpha, 2500, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 2: Subgradient momentum (fast iterations, focus on top lags)
    if _tl() > 50:
        for topm in [3, 5, 8, 12]:
            if _tl() < 40:
                break
            hn, fc = _subgrad_mom(best_h, 12000, topm, 0.004, 0.82)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 3: L-BFGS polish after subgradient
    if _tl() > 40:
        for alpha in [50000, 200000]:
            if _tl() < 30:
                break
            hn = _lbfgs(best_h, alpha, 1800, 1e-19)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 4: Randomized restarts (affordable with L-BFGS-B)
    nc = len(best_h)
    for r in range(8):
        if _tl() < 40:
            break
        if r % 3 == 0:
            hp = best_h + 0.012 * rng.randn(nc)
        elif r % 3 == 1:
            hp = _block_swap(best_h, rng, max(4, nc // 20))
        else:
            hp = _try_binarize(best_h, nc)
            hp = hp + 0.005 * rng.randn(nc)
        hp = _repair(hp, nc)
        for alpha in [3000, 30000]:
            if _tl() < 25:
                break
            hn = _lbfgs(hp, alpha, 1200, 1e-17)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()

    # Phase 5: Fine subgradient
    if _tl() > 40:
        hn, fc = _subgrad_mom(best_h, 25000, 5, 0.0015, 0.92)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 6: Final L-BFGS+B polish
    if _tl() > 30:
        hn = _lbfgs(best_h, 500000, 1500, 1e-20)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 7: Pairwise refinement (multiple scales)
    for scale, nit in [(0.004, 4000), (0.0015, 6000), (0.0004, 8000), (0.0001, 8000)]:
        if _tl() < 12:
            break
        hn, fc = _pairwise(best_h, nit, scale, rng)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase 8: Final micro subgradient
    if _tl() > 10:
        hn, fc = _subgrad_mom(best_h, 8000, 3, 0.0005, 0.95)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
