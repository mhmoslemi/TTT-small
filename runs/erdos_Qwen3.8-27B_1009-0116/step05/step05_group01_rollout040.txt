from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            od = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in od:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sobj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n - sp])
            g[:n - sp] -= w[k] * h[sp:]
    return g

def _sf(x): return x.sum() - len(x) * 0.5
def _sj(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _lag_grad(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += 1.0 - h[sh:]
        g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def _subgrad_cyc(h, n, n_rounds, step0):
    for _ in range(n_rounds):
        c = _corr(h)
        kmax = c.argmax()
        f0 = c.max()
        step = step0

                step = min(step * 1.1, step0)
            else:

    return h

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0: return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)

    sp = np.arange(1, n)

    m1 = sp <= i
    if m1.any(): dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any(): dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _incr_pair(h, c, rng, n_iter, scale, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10: continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i], h[j], c, c5c = ni, nj, new_c, new_c5
    return h, c, c5c

def _block_opt(h, rng, n, n_blocks, alpha):
    for _ in range(n_blocks):
        bl = rng.randint(8, min(30, n // 4))
        st = rng.randint(0, n - bl)
        block_idx = list(range(st, st + bl))
        fixed_sum = h.sum() - h[st:st + bl].sum()
        t_block = n * 0.5 - fixed_sum
        x0 = h[st:st + bl].copy()
        def bobj(xb):
            hx = h.copy()
            hx[st:st + bl] = xb
            return _sobj(hx, alpha)
        def bgrad(xb):
            hx = h.copy()
            hx[st:st + bl] = xb
            return _sgrad(hx, alpha)[st:st + bl]
        def bsum(xb):
            return xb.sum() - t_block
        def bsumj(xb):
            return np.ones(len(xb))
        try:
            res = minimize(bobj, x0, jac=bgrad, method='SLSQP',
                           bounds=[(0.0, 1.0)] * bl,
                           constraints=[{'type': 'eq', 'fun': bsum, 'jac': bsumj}],
                           options={'maxiter': 200, 'ftol': 1e-15})
            if res.fun < bobj(x0) - 1e-12:
                h[st:st + bl] = _repair(res.x, bl)
        except Exception:
            pass
    return h
