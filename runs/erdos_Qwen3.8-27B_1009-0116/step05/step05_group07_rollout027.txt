    return g

def _ps_obj(h, p):
    n = len(h); c = _corrs(h) * (2.0 / n)
    if p < 80:
        return float(np.sum(c**p) / (p - 1))
    m = c.max(); s = p * (c - m); s = np.clip(s, -300, 300)
    return float(m + np.log(np.sum(np.exp(s - s.max()))) / p)

def _ps_grad(h, p):
    n = len(h); c = _corrs(h) * (2.0 / n); g = np.zeros(n)
    if p < 80:
        for k in range(2 * n - 1):
            ck = c[k]
            if ck < 1e-12: continue
            w = p * ck**(p - 1) * (2.0 / n); g += w * _gsingle(k, h)
    else:
        m = c.max(); s = p * (c - m); s = np.clip(s, -300, 300)
        w = np.exp(s - s.max()); w /= w.sum()
        for k in range(2 * n - 1):
            if w[k] < 1e-13: continue
            g += w[k] * (2.0 / n) * _gsingle(k, h)
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_ps(h, p, maxiter, ftol):
    n = len(h); bnds = [(0.0, 1.0)] * n
    def _obj(x): return _ps_obj(x, p)
    def _jac(x): return _ps_grad(x, p)
    def _cfn(x): return x.sum() - n * 0.5
    def _cjc(x): return np.ones(len(x))
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfn, 'jac': _cjc}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _mom_grad(h, p, n_iter, lr, mu):
    n = len(h); tgt = n * 0.5; h = h.copy(); v = np.zeros(n)
    f = _eval(h)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 15: break
        y = h + mu * v
        v = mu * v - lr * _ps_grad(y, p)
        h = _proj_sum(h + v, tgt)
        fn = _eval(h)
        if fn >= f: v *= 0.5
        else: f = fn
    return h, f

def _active_descent(h, topm, max_it, step0):
    n = len(h); tgt = n * 0.5; h = h.copy(); f = _eval(h); step = step0
    for _ in range(max_it):
        if time.time() - _t0 > _budget - 20: break
        c = _corrs(h); ti = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in ti: g += _gsingle(k, h)
        g /= topm; g *= (2.0 / n)
        found = False
        for _ in range(30):
            hn = _proj_sum(h - step * g, tgt)
            fn = _eval(hn)
            if fn < f: h = hn; f = fn; found = True; break
            step *= 0.5
        if not found: break
        step = min(step * 1.1, step0)
    return h, f

def _round_robin(h, nlags, spl):
    n = len(h); tgt = n * 0.5; h = h.copy(); f = _eval(h)
    for _ in range(nlags):
        if time.time() - _t0 > _budget - 20: break
        c = _corrs(h); ks = c.argmax()
        for _s in range(spl):
            g = _gsingle(ks, h) * (2.0 / n); found = False
            for st in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
                hn = _proj_sum(h - st * g, tgt); fn = _eval(hn)
                if fn < f - 1e-14: h = hn; f = fn; found = True; break
            if not found: break
    return h, f

def _upsample(h, fac):
    n = len(h); xo = np.linspace(0, 2, n); xn = np.linspace(0, 2, n * fac)
    fi = interp1d(xo, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(fi(xn), 0, 1), n * fac)

def _pairwise(h, ni, scale, sv):
    n = len(h); h = h.copy(); f = _eval(h)
    rng = np.random.RandomState(sv)
    for _ in range(ni):
        if time.time() - _t0 > _budget - 8: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = scale * rng.randn(); ni2, nj2 = h[i] + d, h[j] - d
        if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
            h[i], h[j] = ni2, nj2; fn = _eval(h)
            if fn >= f: h[i], h[j] = ni2 - d, nj2 + d
            else: f = fn
    return h, f

def _struct_perturb(h, rng, mode):
    n = len(h); hp = h.copy()
    if mode == 0:
        sh = rng.randint(1, max(2, n // 10)); hp = np.roll(hp, sh)
    elif mode == 1:
        st = rng.randint(0, max(1, n - 20)); ln = rng.randint(10, min(40, n - st))
        hp[st:st + ln] = hp[st:st + ln][::-1]
    elif mode == 2:
        b1 = rng.randint(0, max(1, n - 20)); b2 = rng.randint(0, max(1, n - 20))
        l1 = rng.randint(5, 15); l2 = rng.randint(5, 15)
        tmp = hp[b1:b1 + l1].copy(); hp[b1:b1 + l1] = hp[b2:b2 + l2]; hp[b2:b2 + l2] = tmp
    else:
        al = 0.1 + 0.3 * rng.rand(); hp = al * np.roll(hp, rng.randint(-n // 10, n // 10 + 1)) + (1 - al) * hp
    return _repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time(); _budget = budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h); best_n = n0

    def _el(): return time.time() - _t0

    # Phase A: corrected-gradient SLSQP with power-sum, increasing p
    for p in [15, 30, 60, 120, 300]:
        if _el() > 70: break
        hn = _slsqp_ps(best_h, p, 2000, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase B: momentum gradient (fast many-iteration)
    if _el() < budget_s - 250:
        for p in [50, 120, 300]:
            if _el() > budget_s - 200: break
            for lr in [0.008, 0.003, 0.001]:
                if _el() > budget_s - 180: break
                hn, fc = _mom_grad(best_h, p, 2500, lr, 0.85)
                if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase C: upsample 2x, optimize
    n1 = n0 * 2
    h2 = _upsample(best_h, 2)
    if _el() < budget_s - 300:
        for p in [30, 80, 200]:
            if _el() > budget_s - 200: break
            hn = _slsqp_ps(h2, p, 2500, 1e-18)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn.copy(); best_n = n1
        if _el() < budget_s - 150:
            for lr in [0.005, 0.002]:
                if _el() > budget_s - 120: break
                hn, fc = _mom_grad(best_h, 150, 2000, lr, 0.85)
                if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase D: active-set descent (corrected gradient)
    if _el() < budget_s - 200:
        for tm in [3, 5, 8, 15, 25]:
            if _el() > budget_s - 160: break
            hn, fc = _active_descent(best_h, tm, 5000, 0.005)
            if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase E: round-robin on worst lags
    if _el() < budget_s - 130:
        for _ in range(10):
            if _el() > budget_s - 100: break
            hn, fc = _round_robin(best_h, 50, 10)
            if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase F: structured multi-start + local SLSQP
    nc = len(best_h)
    if _el() < budget_s - 120:
        for _r in range(25):
            if _el() > budget_s - 90: break
            hp = _struct_perturb(best_h, rng, _r % 4)
            hp += 0.004 * rng.randn(nc); hp = _repair(hp, nc)
            hn = _slsqp_ps(hp, 150, 800, 1e-17)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase G: high-p SLSQP polish
    if _el() < budget_s - 80:
        for p in [500, 1000]:
            if _el() > budget_s - 50: break
            hn = _slsqp_ps(best_h, p, 2000, 1e-19)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase H: final active + round-robin
    if _el() < budget_s - 40:
        hn, fc = _active_descent(best_h, 10, 4000, 0.003)
        if fc < best_c5: best_c5 = fc; best_h = hn.copy()
        if _el() < budget_s - 20:
            hn, fc = _round_robin(best_h, 30, 8)
            if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase I: momentum gradient final polish
    if _el() < budget_s - 25:
        hn, fc = _mom_grad(best_h, 500, 1500, 0.001, 0.8)
        if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase J: pairwise fine-tuning
    for sc, ni in [(0.002, 6000), (0.0005, 8000), (0.0001, 10000)]:
        if _el() > budget_s - 10: break
        hn, fc = _pairwise(best_h, ni, sc, seed + int(sc * 1e6))
        if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    # Phase K: very final round-robin
    if _el() < budget_s - 5:
        hn, fc = _round_robin(best_h, 20, 6)
        if fc < best_c5: best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
