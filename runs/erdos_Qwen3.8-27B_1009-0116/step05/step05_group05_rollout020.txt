import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return (m + np.log(np.sum(np.exp(s - s.max()))) / alpha + s.max()/alpha) * (2.0/n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1) - k; wk = w[k]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-h[sh:]); g[sh:] -= wk*h[:n-sh]
        else:
            sp=-sh; g[sp:] += wk*(1.0-h[:n-sp]); g[:n-sp] -= wk*h[sp:]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pair_moves(h, c, n, n_moves, rng, temp=0.0):
    """Exact 1D line search for pairwise moves. temp>0 enables SA acceptance."""
    dx = 2.0/n
    s_arr = np.arange(-(n-1), n)
    cmax = c.max()
    accepted = 0
    for _ in range(n_moves):
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        dlo = max(-h[i], h[j]-1.0); dhi = min(1.0-h[i], h[j])
        if dlo >= dhi - 1e-14: continue
        # Gradient g_s = dc(s)/dh[i] - dc(s)/dh[j]
        g = np.zeros(2*n-1)
        v1 = (s_arr >= i-n+1) & (s_arr <= i)
        v2 = (s_arr >= -i) & (s_arr < n-i)
        if v1.any(): g[v1] += 1.0 - h[i - s_arr[v1]]
        if v2.any(): g[v2] -= h[i + s_arr[v2]]
        v3 = (s_arr >= j-n+1) & (s_arr <= j)
        v4 = (s_arr >= -j) & (s_arr < n-j)
        if v3.any(): g[v3] -= 1.0 - h[j - s_arr[v3]]
        if v4.any(): g[v4] += h[j + s_arr[v4]]
        # Binary search for min_d max_s(c_s + d*g_s) on [dlo,dhi]
        pos = g > 1e-15; neg = g < -1e-15
        lo_v = c.copy()
        lo_v[pos] = c[pos] + dlo*g[pos]
        lo_v[neg] = c[neg] + dhi*g[neg]
        lo = lo_v.max()
        f_lo = np.max(c + dlo*g); f_hi = np.max(c + dhi*g)
        hi = min(f_lo, f_hi)
        if dlo <= 0 <= dhi:
            hi = min(hi, c.max())
        best_d = 0.0
        for _ in range(45):
            if hi - lo < 1e-15: break
            mid = (lo+hi)*0.5
            ub = dhi; lb = dlo
            if pos.any(): ub = min(ub, ((mid - c[pos])/g[pos]).min())
            if neg.any(): lb = max(lb, ((mid - c[neg])/g[neg]).max())
            if lb <= ub: hi = mid; best_d = (lb+ub)*0.5
            else: lo = mid
        best_val = hi
        # SA acceptance
        accept = best_val < cmax - 1e-13
        if not accept and temp > 0 and best_val - cmax < 0.01:
            if rng.rand() < np.exp(-(best_val - cmax)/max(temp, 1e-8)):
                accept = True
        if accept:
            d_opt = best_d
            if abs(d_opt) < 1e-14: continue
            h[i] = np.clip(h[i]+d_opt, 0, 1); h[j] = np.clip(h[j]-d_opt, 0, 1)
            c += d_opt * g
            # d^2 correction for cross terms
            sj = j - i; si = i - j
            if 0 <= sj <= n-1:
                c[(n-1)+sj] += d_opt*d_opt
            if 0 <= si <= n-1:
                c[(n-1)+si] -= d_opt*d_opt
            new_max = c.max()
            if new_max < cmax:
                cmax = new_max
            elif new_max > cmax and temp == 0:
                # revert
                h[i] = np.clip(h[i]-d_opt, 0, 1); h[j] = np.clip(h[j]+d_opt, 0, 1)
                c -= d_opt * g
                if 0 <= sj <= n-1: c[(n-1)+sj] -= d_opt*d_opt
                if 0 <= si <= n-1: c[(n-1)+si] += d_opt*d_opt
                cmax = c.max()
            else:
                cmax = new_max
            accepted += 1
    return h, c, cmax, accepted

def _active_descent(h, n_iters, top_k, n):
    dx = 2.0/n
    c5c = _eval(h)
    for _ in range(n_iters):
        c = np.correlate(h, 1.0-h, mode='full')*dx
        idx = np.argsort(c)[::-1][:top_k]
        cv = c[idx]; w = np.maximum(cv - cv.mean(), 0)
        if w.sum() < 1e-15: w = np.ones(top_k)/top_k
        else: w /= w.sum()
        g = np.zeros(n)
        for wk, k in zip(w, idx):
            sh = (n-1)-k
            if sh >= 0:
                g[:n-sh] += wk*(1.0-h[sh:]); g[sh:] -= wk*h[:-sh] if sh else 0
            else:
                sp=-sh; g[sp:] += wk*(1.0-h[:n-sp]); g[:n-sp] -= wk*h[sp:]
        g *= dx
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj_sum(h - step*g, n*0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new; c5c = c5n; improved = True; break
        if not improved: break
    return h, c5c

def _upsample(h, f):
    return np.repeat(h, f)
