                    diff += a

    """Compute coefficient A[k] for coordinate i in correlation"""
    sh = np.arange(2 * n - 1) - (n - 1)
    A = np.zeros(2 * n - 1)
    ip = i + sh
    im = i - sh
    mp = (ip >= 0) & (ip < n)
    mm = (im >= 0) & (im < n)
    A[mp] += (1.0 - h[ip[mp]])
    A[mm] -= h[im[mm]]
    return A

        for pi in range(0, n - 1, 2):

            A_b = _calc_A(h, b, n)

            if lo >= hi:

            for _ in range(bisect_iters):

            delta = 0.5 * (lo + hi)

            c_hi = c + hi * D

        c5 = np.max(c)
        if c5 < best_local:

Wait, I realize there might be an issue with the bisection. The function f(δ) = max_k(c_k + δ*D_k) is convex but NOT necessarily differentiable. The bisection I described finds where the "slope" changes from negative to positive, which gives the minimum. But I need to be careful:

- If D[k_star] < 0 at mid: the active constraint at mid has negative slope, meaning f is decreasing at mid. So the minimum is to the RIGHT of mid. Set lo = mid.
- If D[k_star] > 0: f is increasing at mid. Minimum is to the LEFT. Set hi = mid.
- If D[k_star] = 0: we're at the minimum (or a flat region).

This is correct. The bisection converges to the minimum in O(log(range/eps)) iterations.

def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        c = _corr(x)
        m = c.max()

        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha + s.max() / alpha * 0

        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        sm = s.max()
        return (m + sm / alpha) + np.log(np.sum(np.exp(s - sm))) / alpha

    def _grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        sm = s.max()
        w = np.exp(s - sm)
        w /= w.sum()
        g = np.zeros(n)
