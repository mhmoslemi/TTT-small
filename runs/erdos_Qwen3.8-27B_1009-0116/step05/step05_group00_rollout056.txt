def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Project onto sum constraint while maintaining [0,1]
        lo, hi = -2.0, 2.0
        for _ in range(100):
            mid = (lo + hi) / 2
            h_try = np.clip(h - mid, 0, 1)
            if h_try.sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2, 0, 1)
        # Fix remainder
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)

    g = np.zeros(n)

            continue

            if c5 < best_c5:

    return h, c5

def _binary_swap(h_bin, n, max_rounds=150, seed=0):
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    c5 = _eval(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for _it in range(len(ones) * 2):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c2 = _eval(h)
            if c2 < c5 - 1e-15:
                c5 = c2
                improved = True
                break
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, c5

def _genetic_binary(n, pop_size=24, gen=120, seed=42):
    rng = np.random.RandomState(seed)
    half = n // 2
    pop = []
    for _ in range(pop_size):
        h = np.zeros(n)
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
        pop.append(h)
    scores = [_eval(h) for h in pop]
    best_h = pop[np.argmin(scores)].copy()
    best_c5 = min(scores)
    for g in range(gen):
        order = np.argsort(scores)
        npx = [pop[order[0]].copy(), pop[order[1]].copy()]
        while len(npx) < pop_size:
            i = order[rng.randint(4)]
            j = order[rng.randint(4)]
            p1, p2 = pop[i], pop[j]
            if rng.rand() < 0.3:
                mask = rng.rand(n) < 0.5
                child = np.where(mask, p1, p2).copy()
            else:
                pt = rng.randint(1, n)
                child = np.concatenate([p1[:pt], p2[pt:]]).copy()
            if rng.rand() < 0.8:
                ones = np.where(child > 0.5)[0]
                zeros = np.where(child < 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    child[a], child[b] = 0.0, 1.0
            npx.append(child)
        pop = npx
        scores = [_eval(h) for h in pop]
        if min(scores) < best_c5:
            best_c5 = min(scores)
            best_h = pop[np.argmin(scores)].copy()
    return best_h, best_c5

def _fourier_opt(n, K, maxiter=3000, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False)
    dx = 2.0 / n
    params = rng.randn(2 * K) * 0.1
    def _h_from_params(p):
        h = np.full(n, 0.5)
        for k in range(1, K + 1):
            h += p[2*(k-1)] * np.cos(k * np.pi * x) + p[2*(k-1)+1] * np.sin(k * np.pi * x)
        return h
    best_h = _h_from_params(params)
    best_c5 = _eval(best_h)
    for _it in range(maxiter):
        h = _h_from_params(params)
        h = np.clip(h, 0, 1)
        h = _repair(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        alpha = 50000
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        gh = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                cnt = n - sh
                gh[:cnt] += wk * (1.0 - h[sh:sh + cnt])
                if sh > 0:
                    gh[sh:] -= wk * h[:cnt]
            else:
                sp = -sh
                gh[sp:] += wk * (1.0 - h[:n - sp])
                gh[:n - sp] -= wk * h[sp:]
        gp = np.zeros(2 * K)
        for k in range(1, K + 1):
            gp[2*(k-1)] = np.sum(gh * np.cos(k * np.pi * x)) * dx
            gp[2*(k-1)+1] = np.sum(gh * np.sin(k * np.pi * x)) * dx
        params -= 0.02 * gp
        h_new = _h_from_params(params)
        h_new = np.clip(h_new, 0, 1)
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov with annealed alpha
    if time.time() - t0 < budget_s - 350:
        sched = [
            ([300, 1000, 5000, 20000], 2500),
            ([20000, 80000, 300000], 3000),
            ([300000, 1000000, 3000000], 4000),
        ]
        alpha_sched = [(a, n // len(a)) for a, n in sched]
        h_new, c5 = _nesterov(best_h, alpha_sched, m=0.92, lr0=0.03)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 2: L-BFGS-B
    if time.time() - t0 < budget_s - 250:
        for alpha in [30000, 150000, 800000]:
            if time.time() - t0 > budget_s - 250:
                break
            h_new = _lbfgs_opt(best_h, alpha, 800)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 3: Water-filling on critical shifts
    if time.time() - t0 < budget_s - 200:
        for tk in [1, 2, 3, 5, 10]:
            if time.time() - t0 > budget_s - 200:
                break
            h_new, c5 = _waterfill(best_h, tk, 60)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 4: Binary search at smaller n
    if time.time() - t0 < budget_s - 120:
        for n_try in [60, 80, 100, 120, 156, 168, 200]:
            if time.time() - t0 > budget_s - 120:
                break
            h_r = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                           np.linspace(0, 2, best_n, endpoint=False), best_h)
            h_bin = (h_r > 0.4).astype(float)
            h_bin = _repair(h_bin, n_try)
            h_bin, c5 = _binary_swap(h_bin, n_try, 120, seed=seed)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_bin
                best_n = n_try

    # Phase 5: Genetic binary
    if time.time() - t0 < budget_s - 80:
        for n_try in [100, 120, 156, 200]:
            if time.time() - t0 > budget_s - 80:
                break
            h_g, c5 = _genetic_binary(n_try, 20, 100, seed=seed + n_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_g
                best_n = n_try

    # Phase 6: Fourier parameterization
    if time.time() - t0 < budget_s - 50:
        for K in [8, 12, 20]:
            if time.time() - t0 > budget_s - 50:
                break
            h_f, c5 = _fourier_opt(best_n, K, 2000, seed=seed)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_f
                best_n = len(h_f)

    # Phase 7: Final Nesterov polish
    if time.time() - t0 < budget_s - 15:
        h_new, c5 = _nesterov(best_h, [(100000, 1500), (500000, 1500), (2000000, 1500)], m=0.95, lr0=0.015)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
        best_n = len(best_h)

    # Phase 8: Coordinate descent
    if time.time() - t0 < budget_s - 3:
        h8 = best_h.copy()
        nc = len(h8)
        c5c = _eval(h8)
        for scale in [0.005, 0.002, 0.0005]:
            for _it in range(2000):
                if time.time() - t0 > budget_s - 3:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h8[i], h8[j] = ni, nj
                    c5n = _eval(h8)
                    if c5n < c5c - 1e-14:
                        c5c = c5n
                        if c5c < best_c5:
                            best_c5 = c5c
                            best_h = h8.copy()
                    else:
                        h8[i], h8[j] = h8[i] - d, h8[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
