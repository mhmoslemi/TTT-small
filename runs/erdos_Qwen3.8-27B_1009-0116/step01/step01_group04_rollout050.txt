```python
def haar_basis(n, m):
    """Return m Haar wavelet functions evaluated at n points"""
    bases = []
    x = np.arange(n, dtype=float)
    # DC
    bases.append(np.ones(n) / np.sqrt(n))
    # Haar levels
    level = 0
    while len(bases) < m:
        step = n // (2**(level+1))
        if step < 1:
            break
        for k in range(2**level):
            if len(bases) >= m:
                break
            w = np.zeros(n)
            start = k * step * 2
            w[start:start+step] = 1.0/np.sqrt(n)
            w[start+step:start+2*step] = -1.0/np.sqrt(n)
            # Handle boundary
            if start + 2*step > n:
                w = w[:n]
            bases.append(w)
        level += 1
    while len(bases) < m:
        bases.append(np.zeros(n))
    return np.array(bases)  # shape (m, n)
```

```python
        return compute_c5(h, 2.0 / n) + 50.0 * (np.sum(h) - n / 2.0) ** 2
    p0 = np.zeros(m_k - 1)
    for k in range(1, m_k):
        p0[k - 1] = 2.0 / n * np.sum((h0 - a0) * np.cos(np.pi * k * i / n))
    bd = [(-0.5, 0.5)] * (m_k - 1)
    best_p = p0.copy()
    best_v = dct_obj(p0)
    for method in ['Nelder-Mead', 'Powell']:
        if time.time() - t0 > time_limit - 5:
            break
        try:
            r = minimize(dct_obj, best_p, method=method, bounds=bd,
                         options={'maxiter': 3000, 'xatol': 1e-11, 'fatol': 1e-13})
            if r.fun < best_v:
                best_v = r.fun
                best_p = r.x
        except Exception:
            pass
    h_out = params_to_h(best_p)
    return h_out

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def perturb_sin(x, n, rng):
    k = rng.integers(2, n // 4 + 1)
    ph = rng.uniform(0, 2.0 * np.pi)
    amp = rng.uniform(0.05, 0.12)
    xs = 2.0 * np.arange(n) / n
    return repair(x + amp * np.sin(np.pi * k * xs + ph), n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    rng = np.random.default_rng(seed)
    try:
        h0_raw = np.array(initial_h_values, dtype=float)
    except Exception:
        h0_raw = np.full(191, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)
    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    # Phase 1: DCT subspace optimization (15% budget)
    if budget_s > 60:
        p1_budget = budget_s * 0.12
        m_k = min(50, n0 // 3)
        h_dct = dct_subspace_opt(h0, n0, m_k, p1_budget)
        v_dct = compute_c5(h_dct, dx0)
        if v_dct < best_val:
            best_val = v_dct
            best_h = h_dct

    # Phase 2: Multi-restart SLSQP at base grid (40%)
    p2_limit = t_start + budget_s * 0.50
    restarts = [best_h.copy(), perturb(best_h, n0, 0.1, rng),
                perturb(best_h, n0, 0.15, rng), perturb_sin(best_h, n0, rng),
                perturb_sin(best_h, n0, rng)]
    alloc_per = (budget_s * 0.40) / len(restarts)
    for xs in restarts:
        if time.time() > p2_limit:
            break
        alloc = min(alloc_per + 5, p2_limit - time.time())
        if alloc < 12:
            break
        xb, v = refine_taus(xs, n0, dx0, alloc)
        if v < best_val:
            best_val = v
            best_h = xb
            best_n = n0

    # Phase 3: Alternate grid sizes (25%)
    p3_limit = t_start + budget_s * 0.78
    for n_try in [200, 250]:
        if time.time() > p3_limit:
            break
        dx_try = 2.0 / n_try
        xs = np.linspace(0, 2, best_n, endpoint=False)
        xd = np.linspace(0, 2, n_try, endpoint=False)
        xp = interp1d(xs, best_h, kind='linear')(xd)
        x_p = repair(xp, n_try)
        alloc = min(budget_s * 0.12, p3_limit - time.time())
        if alloc < 12:
            break
        xb, v = refine_taus(x_p, n_try, dx_try, alloc)
        if v < best_val:
            best_val = v
            best_h = xb
            best_n = n_try
        else:
            # Also try perturb at this grid
            xp2 = perturb(x_p, n_try, 0.06, rng)
            xp2 = perturb_sin(xp2, n_try, rng)
            alloc2 = min(alloc * 0.6, p3_limit - time.time())
            if alloc2 > 10:
                xb2, v2 = refine_taus(xp2, n_try, dx_try, alloc2)
                if v2 < best_val:
                    best_val = v2
                    best_h = xb2
                    best_n = n_try

    # Phase 4: Final polish (remainder)
    remaining = budget_s - (time.time() - t_start) - 3
    if remaining > 15:
        dx_best = 2.0 / best_n
        for sc in [0.03, 0.015]:
            rem2 = budget_s - (time.time() - t_start) - 2
            if rem2 < 10:
                break
            xp = perturb(best_h, best_n, sc, rng)
            xb, v = refine_taus(xp, best_n, dx_best, rem2 * 0.85)
            if v < best_val:
                best_val = v
                best_h = xb
                best_n = best_n

    # Final repair and evaluate
    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
```
