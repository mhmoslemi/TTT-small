import time

_state = {'target': 0.0, 'alpha': 100.0, 'dx': 0.02, 'n': 70, 'K': 5}

def _eq_fun(x):
    return np.sum(x) - _state['target']

def _eq_jac(x):
    return np.ones_like(x)

def _obj_slack(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -1.0, 0.0)
    a = _state['alpha']
    return (m + np.log(np.sum(np.exp(a * sh))) / a) * _state['dx']

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        m = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(m))
        if cnt > 0:
            h[m] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _fourier_to_h(coeffs, n):
    K = len(coeffs) // 2
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += coeffs[2*(k-1)] * np.cos(2.0*np.pi*k*i/n)
        h += coeffs[2*(k-1)+1] * np.sin(2.0*np.pi*k*i/n)
    return h

def _fourier_obj(coeffs, n, alpha):
    h = _fourier_to_h(coeffs, n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    pen = 200.0 * (np.sum(h) - target)**2
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -1.0, 0.0)

    n = len(h)
    best = _c5(h)
    deltas = [0.005, 0.01, 0.02, 0.002, 0.03, 0.001, 0.05]
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for d in deltas:
                found = False
                for sign in (1, -1):
                    hi2 = h[i] + sign * d
                    if hi2 < 0 or hi2 > 1:
                        continue
                    for j in [(i+1) % n, (i-1+n) % n, (i + n//2) % n, (i + n//3) % n]:
                        if j == i:
                            continue
                        hj2 = h[j] - sign * d
                        if hj2 < 0 or hj2 > 1:
                            continue
                        h2 = h.copy()
                        h2[i] = hi2
                        h2[j] = hj2
                        c = _c5(h2)
                        if c < best - 1e-12:
                            h = h2
                            best = c
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
        if not improved:
            break
    return h, best

def _sinusoid_init(n, rng, target):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for _ in range(rng.randint(2, 5)):
        f = rng.randint(1, max(2, n // 3))
        p = rng.uniform(0, 2.0 * np.pi)
        a = rng.uniform(0.05, 0.3)
        h += a * np.sin(2.0 * np.pi * f * x / n + p)
    return _proj(h, target)

def _block_init(n, rng, target):
    nb = rng.randint(3, 9)
    bs = n // nb
    h = np.zeros(n)
    for k in range(nb):
        s = k * bs
        e = s + bs if k < nb - 1 else n
        h[s:e] = rng.uniform(0.0, 1.0)
    return _proj(h, target)

def _anti_init(n, rng, target):
    if n % 2 != 0:
        return _sinusoid_init(n, rng, target)
    half = n // 2
    h = np.zeros(n)
    h[:half] = rng.uniform(0.15, 0.85, half)
    h[half:] = 1.0 - h[:half]
    return _proj(h, target)

def _fm_exact(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
