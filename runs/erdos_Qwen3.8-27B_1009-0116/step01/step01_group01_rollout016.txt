    h = h.copy()

    """One step of subgradient descent on max of correlations."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    # Find the max lag(s)
    max_idx = np.argmax(c)
    lag = max_idx - (n - 1)
    # Compute gradient at this lag
    g = np.zeros(n)
    if lag >= 0:
        # out = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        for i in range(n - lag):
            g[i] += (1.0 - h[i + lag])
            if i + lag < n:
                g[i + lag] -= (1.0 - h[i])

        for i in range(lag2, n):
            g[i] += (1.0 - h[i - lag2])
            if i - lag2 >= 0:
                g[i - lag2] -= (1.0 - h[i])
    return g, cmax * dx

Wait, this subgradient computation is O(n) per step but with Python loops - that's slow for n=376. Let me vectorize it.

For lag >= 0:
g[i] += (1-h[i+lag]) for i in 0..n-1-lag
g[i+lag] -= (1-h[i]) for i in 0..n-1-lag

So:
g[:n-lag] += (1-h[lag:])
g[lag:] -= (1-h[:-lag])  # h[:-lag] has length n-lag, and we add to g[lag:n]

For lag < 0 (let p = -lag):
g[p:] += (1-h[:-p])  # h[i-p] for i in p..n-1
g[:-p] -= (1-h[p:])  # for i in 0..n-1-p, the term -(1-h[i+p]) but wait...

Let me re-derive for lag < 0, say lag = -p where p > 0:
out = sum_{i=p}^{n-1} h[i] * (1-h[i-p])

∂out/∂h[j]:
- j appears as i in sum: contributes (1-h[j-p]) if p≤j≤n-1
- j appears as i-p, i.e., i=j+p: contributes -(1-h[j+p]) if 0≤j≤n-1-p

So:
g[p:] += (1-h[:-p])  # for j in p..n-1: (1-h[j-p])
g[:-p] -= (1-h[p:])  # for j in 0..n-1-p: -(1-h[j+p])

This is vectorizable!

Let me also think about a "focal" optimization: if I know which lags are the maximum, I can optimize only to reduce those specific lags. This is like a game: minimize the worst-case lag while keeping others in check.

```python
def focal_optimize(h, n_iters, t0):
    """Optimize focusing on the critical lags."""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    for it in range(n_iters):
        c = np.correlate(h, 1-h, mode='full') * (2.0/n)
        # Find critical lags (within 10% of max)
        threshold = np.max(c) * 0.95
        critical = np.where(c >= threshold)[0]
        # Compute weighted gradient
        g = np.zeros(n)
        for idx in critical:
            lag = idx - (n-1)
            weight = c[idx] - np.min(c)  # weight by excess
            # Add gradient contribution
            if lag >= 0:
                g[:n-lag] += weight * (1.0 - h[lag:])
                if lag > 0:
                    g[lag:] -= weight * (1.0 - h[:n-lag])
            else:
                p = -lag
                g[p:] += weight * (1.0 - h[:n-p])
                g[:n-p] -= weight * (1.0 - h[p:])
        # Normalize gradient
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        # Step
        lr = t0 / (1.0 + it * 0.1)
        h = project_onto_constraint(h - lr * g, target)
    return h
```

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        idx = int(np.argmax(h)) if diff < 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= (1.0 - h[:n - lag])
    else:
        p = -lag
        if p < n:
            g[p:] += (1.0 - h[:n - p])
            g[:n - p] -= (1.0 - h[p:])
    return g


def soft_max_obj(h, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    shifted = (c - cmax) * t
    shifted = np.clip(shifted, -700.0, 700.0)
    return (cmax + np.log(np.sum(np.exp(shifted))) / t) * (2.0 / n)


def soft_max_grad(h, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    shifted = (c - cmax) * t
    shifted = np.clip(shifted, -700.0, 700.0)
    w = np.exp(shifted)
    w = w / w.sum()
    g = np.zeros(n)
    for idx, wi in enumerate(w):
        if wi < 1e-10:
            continue
        lag = idx - (n - 1)
        g += wi * grad_at_lag(h, lag)
    return g * (2.0 / n)


def subgradient_descent(h, n_iters, lr_init, decay):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        cur = float(np.max(c))
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        max_idx = int(np.argmax(c))
        lag = max_idx - (n - 1)
        g = grad_at_lag(h, lag)
        # Also add contribution from near-max lags
        threshold = c[max_idx] * 0.98
        for idx2 in np.where(c >= threshold)[0]:
            if idx2 != max_idx:
                lag2 = idx2 - (n - 1)
                w2 = (c[idx2] - c[0]) / max(c[max_idx], 1e-15)
                g += max(0.0, w2) * 0.3 * grad_at_lag(h, lag2)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            break
        lr = lr_init / (1.0 + decay * it)
        h = project_onto_constraint(h - lr * g / gnorm, target)
    return best_h, best_c


def l_bfgs_optimize(h, t, max_iter, target_sum):
    n = len(h)

    def obj(x):
        return soft_max_obj(x, t)

    def jac(x):
        return soft_max_grad(x, t)

    res = minimize(obj, h.copy(), method='L-BFGS-B', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    h_out = project_onto_constraint(res.x, target_sum)
    return h_out


def coordinate_descent(h, n, n_sweeps, n_candidates):
    target = n * 0.5
    h = h.copy()
    best_val = compute_c5(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            # Generate candidates: current, 0, 1, neighbors, random
            cands = [0.0, 1.0, old]
            if 0 < old < 1:
                cands.append(old * 0.5)
                cands.append(0.5 + (old - 0.5) * 0.3)
                cands.append(0.5 + (old - 0.5) * 0.7)
            cands += list(np.random.uniform(0, 1, n_candidates))
            best_cand = old
            best_v = best_val
            for cand in cands:
                if abs(cand - old) < 1e-9:
                    continue
                h[i] = cand
                diff = target - h.sum()
                # Distribute correction to a random non-saturated location
                mask = (h > 1e-8) & (h < 1.0 - 1e-8)
                if mask.sum() > 0:
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0.0, 1.0)
                val = compute_c5(h)
                if val < best_v - 1e-13:
                    best_v = val
                    best_cand = cand
            if best_cand != old:
                h[i] = best_cand
                diff = target - h.sum()
                mask = (h > 1e-8) & (h < 1.0 - 1e-8)
                if mask.sum() > 0:
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0.0, 1.0)
                best_val = best_v
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, best_val


def multi_start_search(h_start, n, n_starts, t, budget_per, t0_global):
    target = n * 0.5
    best_h = h_start.copy()
    best_c = compute_c5(h_start)
    for s in range(n_starts):
        if time.time() - t0_global > 1050:

        noise_scale = 0.15 / (1.0 + s * 0.3)

        h_try = project_onto_constraint(h_try, target)

        v = compute_c5(h_opt)
        if v < best_c:
