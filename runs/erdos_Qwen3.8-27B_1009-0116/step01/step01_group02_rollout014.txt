def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        # Final correction on free entries
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(free) > 0:
            adj = (target - s2) / np.sum(free)
            h[free] = np.clip(h[free] + adj, 0.0, 1.0)
    return h

    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def corr_vec(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        free = (h > 1e-5) & (h < 1.0 - 1e-5)
        cnt2 = int(np.sum(free))
        if cnt2 > 0:
            h[free] += (target - s2) / cnt2
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    scaled = beta * corr
    m = np.max(scaled)
    return float(m + np.log(np.sum(np.exp(scaled - m)))) / beta

def exact_obj(x, n):
    return compute_c5(x, n)

def sum_con(x, target):
    return np.sum(x) - target

def project_onto_simplex(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h > 0) & (h < 1)
    cnt = int(np.sum(mask))
    if cnt == 0:
        return np.full(n, 0.5)
    step = diff / cnt
    h2 = h.copy()
    h2[mask] += step
    if np.any(h2[mask] < 0) or np.any(h2[mask] > 1):
        lo = h[mask] > 0
        hi = h[mask] < 1
        if np.sum(lo) > 0:
            h2[mask[lo]] = np.clip(h2[mask[lo]], 0.0, 1.0)
        if np.sum(hi) > 0:
            h2[mask[hi]] = np.clip(h2[mask[hi]], 0.0, 1.0)
    h2 = np.clip(h2, 0.0, 1.0)
    s = np.sum(h2)
    if abs(s - target) > 1e-10:
        h3 = h2 * (target / s)
        h3 = np.clip(h3, 0.0, 1.0)
        return h3
    return h2

def subgradient_step(h, n):
    corr = corr_vec(h, n)
    idx_max = int(np.argmax(corr))
    k = idx_max - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i] += 1.0 - h[i + k]
            if i + k < n:
                grad[i + k] -= h[i]
    else:
        kk = -k
        for i in range(kk, n):
            grad[i] += 1.0 - h[i - kk]
            grad[i - kk] -= h[i]
    return -grad * dx, corr[idx_max]
