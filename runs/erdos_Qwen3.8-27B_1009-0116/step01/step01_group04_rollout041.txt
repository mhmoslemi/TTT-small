        r = minimize(_pen_obj, x0, method='L-BFGS-B', bounds=bd,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def perturb_gauss(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def perturb_blocks(x, n, nblocks, scale, rng):
    x2 = x.copy()
    bs = max(1, n // nblocks)
    for _ in range(nblocks):
        st = int(rng.integers(0, n - bs + 1))
        x2[st:st + bs] += scale * rng.standard_normal(bs)
    return repair(x2, n)

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)

def refine_slq(x0, n, dx, time_limit):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    schedule = [(0.5, 150), (0.2, 200), (0.1, 250), (0.05, 300),
                (0.02, 350), (0.01, 400), (0.005, 450), (0.002, 500),
                (0.001, 600), (0.0005, 700), (0.0002, 800), (0.0001, 900)]
    for tau, mi in schedule:
        if time.time() - t0 > time_limit - 2.0:
            break
        x = slsqp_step(x, n, dx, tau, mi)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
    return best, best_val

def refine_lbg(x0, n, dx, time_limit):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    schedule = [(0.3, 200, 100.0), (0.1, 300, 500.0), (0.05, 300, 2000.0),
                (0.02, 350, 5000.0), (0.01, 400, 10000.0),
                (0.005, 500, 50000.0), (0.002, 500, 100000.0),
                (0.001, 600, 500000.0)]
    for tau, mi, pen in schedule:
        if time.time() - t0 > time_limit - 2.0:
            break
        x = lbfgs_step(x, n, dx, tau, mi, pen)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
    return best, best_val

def make_structured_start(n, rng):
    """Create diverse structured starting points."""
    opts = []
    # Block pattern
    h_blk = np.zeros(n)
    h_blk[:n // 2] = 1.0
    h_blk = repair(h_blk + 0.1 * rng.standard_normal(n), n)
    opts.append(h_blk)
    # Alternating blocks
    h_alt = np.zeros(n)
    bv = n // 8
    for i in range(0, n, 2 * bv):
        h_alt[i:i + bv] = 1.0
    h_alt = repair(h_alt + 0.1 * rng.standard_normal(n), n)
    opts.append(h_alt)
    # Sinusoidal
    t = np.linspace(0, 4 * np.pi, n)
    h_sin = 0.5 + 0.4 * np.sin(t)
    h_sin = repair(h_sin + 0.05 * rng.standard_normal(n), n)
    opts.append(h_sin)
    return opts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(95, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    # Phase 1: Heavy SLSQP optimization on parent (40% budget)
    p1 = budget_s * 0.40
    x_opt, v = refine_slq(h0, n0, dx0, p1)
    if v < best_val:
        best_val = v
        best_h = x_opt

    # Phase 2: L-BFGS-B from current best (15% budget) - different landscape
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.58:
        p2 = budget_s * 0.15
        x_opt, v = refine_lbg(best_h, n0, dx0, p2)
        if v < best_val:
            best_val = v
            best_h = x_opt

    # Phase 3: SLSQP again from L-BFGS result (back and forth) (10%)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.70:
        p3 = budget_s * 0.10
        x_opt, v = refine_slq(best_h, n0, dx0, p3)
        if v < best_val:
            best_val = v
            best_h = x_opt

    # Phase 4: Structured restarts (15%)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.85:
        starts = make_structured_start(n0, rng)
        starts.append(perturb_blocks(best_h, n0, 6, 0.08, rng))
        starts.append(perturb_gauss(best_h, n0, 0.05, rng))
        alloc4 = min(budget_s * 0.15, budget_s * 0.85 - elapsed)
        per_start = alloc4 / len(starts)
        for xs in starts:
            if time.time() - t_start > budget_s * 0.87:
                break
            x_opt, v = refine_slq(xs, n0, dx0, per_start)
            if v < best_val:
                best_val = v
                best_h = x_opt

    # Phase 5: Final ultra-fine polish (10%)
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 5
    if remaining > 20:
        dx_b = 2.0 / best_n
        _g['tau'] = 0.00005
        _g['dx'] = dx_b
        _g['n'] = best_n
        _g['tgt'] = float(best_n) / 2.0
        bd = [(0.0, 1.0)] * best_n
        cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
        try:
            r = minimize(_obj, best_h, method='SLSQP', bounds=bd,
                         constraints=cs, options={'maxiter': 2000, 'ftol': 1e-16})
            x_f = repair(r.x, best_n)
            v = compute_c5(x_f, dx_b)
            if v < best_val:
                best_val = v
                best_h = x_f
        except Exception:
            pass

    # Phase 6: Try slightly finer grid if time permits
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 5
    if remaining > 60:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        x_p = project_grid(best_h, best_n, n1)
        p6 = remaining * 0.7
        if p6 > 30:
            x_opt, v = refine_slq(x_p, n1, dx1, p6)
            if v < best_val:
                best_val = v
                best_h = x_opt
                best_n = n1

    # Final validation
    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)

    return (best_h, final_val, best_n)
