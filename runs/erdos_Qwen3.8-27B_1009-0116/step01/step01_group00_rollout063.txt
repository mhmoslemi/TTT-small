```python
def _optimal_delta(c, s, h_i, h_j):
    """Find optimal δ for h[i]+=δ, h[j]-=δ.
    c: current correlation values (length 2n-1)
    s: slope array, s[k] = ∂c_k/∂h[i] - ∂c_k/∂h[j]
    h_i, h_j: current values
    Returns: optimal δ (clipped to valid range)
    """
    # Valid range for δ: 0 ≤ h_i+δ ≤ 1 and 0 ≤ h_j-δ ≤ 1
    delta_lo = max(-h_i, -(1-h_j))  # h_i+δ ≥ 0 and h_j-δ ≤ 1
    delta_hi = min(1-h_i, h_j)       # h_i+δ ≤ 1 and h_j-δ ≥ 0
    
    if delta_lo >= delta_hi:
        return 0.0
    
    # Find breakpoints
    n_c = len(c)
    candidates = set([delta_lo, delta_hi, 0.0])
    for a in range(n_c):
        for b in range(a+1, n_c):
            denom = s[a] - s[b]
            if abs(denom) > 1e-12:
                d = (c[b] - c[a]) / denom
                if delta_lo <= d <= delta_hi:
                    candidates.add(d)
    
    # Evaluate at each candidate
    best_d = 0.0
    best_val = np.inf
    for d in candidates:
        val = np.max(c + d * s)
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d
```

```python
    """Minimize f(t) = max(c + t*s) for t in [d_lo, d_hi] using ternary search."""
    for _ in range(50):
        m1 = d_lo + (d_hi - d_lo) / 3
        m2 = d_hi - (d_hi - d_lo) / 3
        f1 = np.max(c + m1 * s)
        f2 = np.max(c + m2 * s)
        if f1 < f2:
            d_hi = m2
        else:
            d_lo = m1
    return (d_lo + d_hi) / 2
```

```python
def _update_G_row(G, p, h, n):
    """Update row G[p] after h changes."""
    for k in range(2*n-1):
        lag = k - (n-1)
        if lag == 0:
            G[p, k] = 1.0 - 2.0*h[p]
        elif lag > 0:
            val = 0.0
            if lag <= p <= n-1:
                val += (1.0 - h[p-lag])
            if 0 <= p <= n-1-lag:
                val -= h[p+lag]
            G[p, k] = val
        else:
            m = -lag
            val = 0.0
            if 0 <= p <= n-1-m:
                val += (1.0 - h[p+m])
            if m <= p <= n-1:
                val -= h[p-m]
            G[p, k] = val
```

This is O(n) per call (iterating over 2n-1 lags). For each pair move, we call it twice (for i and j), so O(n) per move.

One concern: the ternary search assumes the function is convex. Is f(δ) = max_k (c_k + δ*s_k) convex in δ? Yes! It's the max of linear functions, hence convex. Good.

Another concern: after a move, the new corr values must still satisfy corr = correlate(h, 1-h). Since I update corr += δ*s and s was computed from the old G, is this exact? Let me verify:

c_k(h') where h' differs from h only in coordinates i and j:
c_k(h') = c_k(h) + δ*(∂c_k/∂h[i] - ∂c_k/∂h[j]) (evaluated at h)

This is exact because c_k is linear in h[i] (when other coords are fixed) and linear in h[j] (when other coords are fixed). The second-order terms (involving ∂²c_k/∂h[i]∂h[j]) are zero because c_k is at most linear in each individual coordinate.

Wait, is that true? c_k = sum_a h[a+k]*(1-h[a]). The second derivative ∂²c_k/∂h[i]∂h[j] for i≠j: this would be non-zero if h[i] and h[j] both appear in the same product term. For example, if i+k = j (in the positive lag case), then the term h[i+k]*(1-h[i]) = h[j]*(1-h[i]) has second derivative -1 w.r.t. h[i] and h[j].

So c_k IS quadratic in (h[i], h[j]) when j = i+k or i = j+k! The linear approximation corr += δ*s is NOT exact in general!

Let me re-examine. For lag k ≥ 0:
c_k = sum_{a=0}^{n-1-k} h[a+k]*(1-h[a])

The term involving both h[i] and h[j] (with i≠j):
- h[i] appears as h[a+k] when a = i-k, contributing h[i]*(1-h[i-k])
- h[j] appears as h[a] when a = j, contributing h[j+k]*(1-h[j])

These are the same term only if i-k = j, i.e., i = j+k. In that case, the term is h[i-k]*(1-h[i-k])... wait no.

If a = i-k = j, then the term is h[j+k]*(1-h[j]) = h[i]*(1-h[j]). This term has second derivative ∂²/∂h[i]∂h[j] = -1.

So c_k is NOT linear in h[i] and h[j] simultaneously! It's bilinear.

This means my "exact" update corr += δ*s is actually only a first-order approximation. The error is O(δ²). For small δ (which ternary search will give), this error is small. But it accumulates over many moves.

To fix this: after each sweep (or every few moves), recompute corr exactly using np.correlate. This adds O(n²) per sweep, which is negligible.

This should work well. The approximate corr during the sweep is good enough for the ternary search (small error), and we correct it at the end.

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h_p) if diff > 0 else np.argsort(h_p)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h_p[idx])
                h_p[idx] += d
                diff -= d
            else:
                d = min(-diff, h_p[idx])
                h_p[idx] -= d
                diff += d
            if abs(diff) < 1e-10:
                break
    return h_p

def _grad_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    corr_len = 2 * n - 1
    for k in range(corr_len):
        lag = k - (n - 1)
        if lag == 0:
            G[:, k] = 1.0 - 2.0 * h
        elif lag > 0:
            hi = np.arange(lag, n)
            G[hi, k] += (1.0 - h[hi - lag])
            li = np.arange(0, n - lag)
            G[li, k] -= h[li + lag]
        else:
            m = -lag
            li = np.arange(0, n - m)
            G[li, k] += (1.0 - h[li + m])
            hi = np.arange(m, n)
            G[hi, k] -= h[hi - m]
    return G

def _ternary_max(c, s, d_lo, d_hi, iterations=40):
    if d_lo >= d_hi:
        return 0.0
    for _ in range(iterations):
        m1 = d_lo + (d_hi - d_lo) / 3.0
        m2 = d_hi - (d_hi - d_lo) / 3.0
        f1 = np.max(c + m1 * s)
        f2 = np.max(c + m2 * s)
        if f1 < f2:
            d_hi = m2
        else:
            d_lo = m1
    return (d_lo + d_hi) / 2.0
