    best_n = n0
    print(f"Start: n={best_n}, C5={best_c5:.6f}")

    resolutions = sorted(set([n0, 80, 90, 100, 120]))
    rng_main = np.random.RandomState(seed)

    for n in resolutions:
        if time.time() > t_end - 60:
            break
        target = n / 2.0

        if n == best_n:
            h = best_h.copy()
        else:
            xo = np.linspace(0, 2, best_n, endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(xn, xo, best_h)
        h = _proj(h, target)
        cx = _c5(h)

        # SLSQP with multiple alpha values
        for alpha in [800, 2000, 5000, 12000, 30000]:
            if time.time() > t_end - 20:
                break
            mn = 80 if n <= 100 else 50
            h_new = _slsqp_opt(h, n, target, alpha, mn)
            c_new = _c5(h_new)
            if c_new < cx:
                h = h_new
                cx = c_new

        # Subgradient refinement
        if time.time() > t_end - 15:
            pass
        else:
            h, cx = _subgrad(h, target, 300, 0.03, t_end - 10)

        # Local search
        if time.time() > t_end - 10:
            pass
        else:
            h, cx = _local_search(h, target, t_end - 5, seed + n * 7)

        # Perturbation restart (try once)
        if time.time() > t_end - 20 or n < best_n:
            pass
        else:
            for attempt in range(3):
                if time.time() > t_end - 8:
                    break
                h_pert = h + rng_main.normal(0, 0.02, n)
                h_pert = _proj(h_pert, target)
                h_pert = _slsqp_opt(h_pert, n, target, 5000, 60)
                cp = _c5(h_pert)
                if cp < cx:
                    h, cx = h_pert, cp
                    h, cx = _local_search(h, target, t_end - 5, seed + n * 13 + attempt)

        if cx < best_c5:
            best_c5 = cx
            best_h = h.copy()
            best_n = n
            print(f"n={n}: C5={cx:.6f}")

    # Final fine local search
    if time.time() < t_end - 5:
        target = best_n / 2.0
        h = best_h.copy()
        h, cx = _local_search(h, target, t_end, seed + 999, fine=True)
        if cx < best_c5:
            best_c5 = cx
            best_h = h.copy()

    h_ret = _proj(best_h, best_n / 2.0)
    c5f = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return h_ret, c5f, best_n
