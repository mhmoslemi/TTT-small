import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-9)[0]
    else:
        idx = np.where(h > 1e-9)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

def fix_binary(h, n):
    """Ensure h is binary with exactly n/2 ones."""
    n_ones = n // 2
    h = (h > 0.5).astype(float)
    current = h.sum()
    if current > n_ones:
        # Need to flip some 1s to 0s
        ones = np.where(h == 1)[0]
        to_flip = int(current - n_ones)
        h[ones[:to_flip]] = 0
    elif current < n_ones:
        zeros = np.where(h == 0)[0]
        to_flip = int(n_ones - current)
        h[zeros[:to_flip]] = 1
    return h

def slsqp_solve(h, n, ts, beta, maxiter=500):
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    
    res = minimize(smooth_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def pairwise_hill_climb(h, n, rng, n_trials, t_deadline):
    """Random pairwise swap hill climbing for binary h."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    for trial in range(n_trials):
        if trial % 1000 == 0 and time.time() > t_deadline:
            break
        # Pick random one and zero positions
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        a = rng.choice(ones)
        b = rng.choice(zeros)
        h_cur[a], h_cur[b] = h_cur[b], h_cur[a]
        c5 = compute_c5(h_cur)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_cur.copy()
        else:
            h_cur[a], h_cur[b] = h_cur[b], h_cur[a]
    
    return best_h, best_c5

def slsqp_solve(h, n, ts, beta, maxiter=400):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        return (m + np.log(np.sum(np.exp((corr - m) * beta))) / beta) * (2.0 / n)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def hill_climb(h, n, rng, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    ones = list(np.where(h_cur > 0.5)[0])
    zeros = list(np.where(h_cur < 0.5)[0])
    while time.time() < t_end:
        a = int(rng.integers(len(ones)))
        b = int(rng.integers(len(zeros)))
        ia, ib = ones[a], zeros[b]
        h_cur[ia], h_cur[ib] = h_cur[ib], h_cur[ia]
        c5 = compute_c5(h_cur)
        if c5 < cur_c5 - 1e-15:
            cur_c5 = c5
            ones[a], zeros[b] = ib, ia
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
        else:
            h_cur[ia], h_cur[ib] = h_cur[ib], h_cur[ia]
    return best_h, best_c5

def sa_search(h, n, rng, t_end, T0=0.12, Tmin=0.002):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    ones = list(np.where(h_cur > 0.5)[0])
    zeros = list(np.where(h_cur < 0.5)[0])
    total = 500000
    for trial in range(total):
        if time.time() > t_end:
            break
        frac = trial / total
        T = T0 * (Tmin / T0) ** frac
        a = int(rng.integers(len(ones)))
        b = int(rng.integers(len(zeros)))
        ia, ib = ones[a], zeros[b]
        h_cur[ia], h_cur[ib] = h_cur[ib], h_cur[ia]
        c5 = compute_c5(h_cur)
        delta = c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cur_c5 = c5
            ones[a], zeros[b] = ib, ia
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
        else:
            h_cur[ia], h_cur[ib] = h_cur[ib], h_cur[ia]
    return best_h, best_c5

def adjacent_moves(h, n, rng, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    while time.time() < t_end:
        i = int(rng.integers(n - 1))
        if abs(h_cur[i] - h_cur[i + 1]) < 0.1:
            continue
        h_cur[i], h_cur[i + 1] = h_cur[i + 1], h_cur[i]
        h_cur = project_feasible(h_cur, n / 2.0)
        c5 = compute_c5(h_cur)
        if c5 < cur_c5 - 1e-15:
            cur_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[i + 1] = h_cur[i + 1], h_cur[i]
    return best_h, best_c5

def continuous_refine(h, n, ts, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in [50, 150, 500, 1500, 5000, 15000]:
        if time.time() > t_end:
            break
        h_try = best_h.copy()
        h_try = slsqp_solve(h_try, n, ts, beta, maxiter=500)
        c5 = compute_c5(h_try)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_try.copy()
    for _ in range(5):
        if time.time() > t_end:
            break
        pert = best_h + rng.normal(0, 0.05, n)
        pert = slsqp_solve(pert, n, ts, 3000, maxiter=400)
        c5 = compute_c5(pert)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = pert.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def el():
        return time.time() - t0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 140
        h = np.zeros(n_init)
        h[rng.choice(n_init, n_init // 2, replace=False)] = 1.0

    n = len(h)
    ts = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP from initial
    for beta in [10, 100]:
        if el() > 25:
            break
        h = slsqp_solve(h, n, ts, beta, maxiter=300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Binary snap
    h_bin = fix_binary(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 3: Aggressive binary hill climbing (~250s)
    t_hc = t0 + budget_s * 0.32
    best_h, best_c5 = hill_climb(best_h, n, rng, t_hc)

    # Phase 4: SA (~150s)
    t_sa = t0 + budget_s * 0.46
    sa_h, sa_c5 = sa_search(best_h, n, rng, t_sa, T0=0.1, Tmin=0.003)
    if sa_c5 < best_c5:
        best_c5 = sa_c5
        best_h = sa_h.copy()

    # Phase 5: More hill climbing from best (~80s)
    t_hc2 = t0 + budget_s * 0.54
    best_h, best_c5 = hill_climb(best_h, n, rng, t_hc2)

    # Phase 6: Adjacent moves (~30s)
    t_adj = t0 + budget_s * 0.58
    best_h, best_c5 = adjacent_moves(best_h, n, rng, t_adj)

    # Phase 7: More hill climbing
    t_hc3 = t0 + budget_s * 0.64
    best_h, best_c5 = hill_climb(best_h, n, rng, t_hc3)

    # Phase 8: Continuous refinement (the key phase, ~350s)
    t_cont = t0 + budget_s * 0.93
    best_h, best_c5 = continuous_refine(best_h, n, ts, t_cont, rng)

    # Phase 9: Final polish
    if el() < budget_s - 10:
        h_fin = best_h.copy()
        h_fin = slsqp_solve(h_fin, n, ts, 10000, maxiter=600)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
