            a0 = np.log(h_curr / (1.0 - h_curr + 1e-12)) - np.log(h_curr / (1.0 - h_curr + 1e-12)).mean()
            a_local = a0.copy()
            c_sigmoid_best = c_curr
            for it in range(60):
                if time.time() > n_deadline - 12:
                    break
                a_P = a_local + rng.normal(0, 0.05, n)
                try:
                    r = minimize(obj_param, a_P, args=(n,), method='Nelder-Mead',
                                 options={'maxiter': 80, 'xatol': 1e-9, 'fatol': 1e-9})
                    h_test = h_from_a(r.x, n)
                    c_test = compute_c5(h_test, n)
                    if c_test < c_sigmoid_best - 1e-12:
                        a_local = r.x.copy()
                        c_sigmoid_best = c_test
                        h_curr = h_test
                        if c_test < best_c5:
                            best_c5, best_h, best_n = c_test, h_test.copy(), n
                except Exception:
                    pass

            # Phase B: Projected gradient on worst lag
            h_pg = repair(h_curr.copy(), n)
            c_pg = compute_c5(h_pg, n)
            for it in range(250):
                if time.time() > n_deadline - 10:
                    break
                grad, _ = gradient_worst_lag(h_pg, n)
                lr = 0.015 * (0.97 ** (it // 25))
                h_new = project(h_pg - lr * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new < c_pg - 1e-13:
                    h_pg, c_pg = h_new, c_new
                    if c_new < best_c5:
                        best_c5, best_h, best_n = c_new, h_new.copy(), n
                elif it % 10 == 0 and it > 30:
                    h_sm = project(h_pg - lr * 0.3 * grad, n)
                    c_sm = compute_c5(h_sm, n)
                    if c_sm < c_pg - 1e-13:
                        h_pg, c_pg = h_sm, c_sm
                        if c_sm < best_c5:
                            best_c5, best_h, best_n = c_sm, h_sm.copy(), n

            # Phase C: SLSQP in h-space
            if time.time() < n_deadline - 6:
                bounds = [(0.0, 1.0)] * n
                try:
                    # Use smooth surrogate first
                    beta = 20.0
                    def obj_smooth_h(x, n=n):
                        corr = np.correlate(x, 1.0 - x, mode='full')
                        sx = beta * corr * (2.0 / n)
                        m = sx.max()
                        return (m + np.log(np.sum(np.exp(sx - m)))) / beta
                    def eqc(x, n=n):
                        return np.sum(x) - n / 2.0
                    cons = [{'type': 'eq', 'fun': eqc}]
                    r1 = minimize(partial(obj_smooth_h, n=n), h_pg, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 200, 'ftol': 1e-13})
                    h1 = repair(r1.x, n)
                    c1 = compute_c5(h1, n)
                    if c1 < best_c5:
                        best_c5, best_h, best_n = c1, h1.copy(), n
                    if c1 < c_pg:
                        h_pg, c_pg = h1, c1

                    # Then exact max
                    r2 = minimize(partial(compute_c5, n=n), h_pg, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 300, 'ftol': 1e-14})
                    h2 = repair(r2.x, n)
                    c2 = compute_c5(h2, n)
                    if c2 < best_c5:
                        best_c5, best_h, best_n = c2, h2.copy(), n
                    if c2 < c_pg:
                        h_pg, c_pg = h2, c2
                except Exception:
                    pass

            # Phase D: Pairwise mass transfer
            if time.time() < n_deadline - 3:
                h_sw = h_pg.copy()
                c_sw = c_pg
                for _ in range(200):
                    if time.time() > n_deadline - 2:
                        break
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i == j:
                        continue
                    for eps in (0.02, 0.05, 0.1, 0.2):
                        hp = h_sw.copy()
                        hp[i] += eps
                        hp[j] -= eps
                        if hp[i] > 1.0 or hp[j] < 0.0:
                            continue
                        hp = project(hp, n)
                        c = compute_c5(hp, n)
                        if c < c_sw - 1e-13:
                            h_sw, c_sw = hp, c
                            if c_sw < best_c5:
                                best_c5, best_h, best_n = c_sw, h_sw.copy(), n
                            break
                if c_sw < c_pg:
                    h_pg, c_pg = h_sw, c_sw

    # Final polish
    if best_h is not None and time.time() < t0 + budget_s - 12:
        n_f = best_n
        bounds_f = [(0.0, 1.0)] * n_f
        try:
            def eqc_f(x, n=n_f):
                return np.sum(x) - n / 2.0
            cons_f = [{'type': 'eq', 'fun': eqc_f}]
            r_f = minimize(partial(compute_c5, n=n_f), best_h, method='SLSQP',
                           bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 1000, 'ftol': 1e-15})
            h_f = repair(r_f.x, n_f)
            c_f = compute_c5(h_f, n_f)
            if c_f < best_c5:
                best_h, best_c5 = h_f, c_f
        except Exception:
            pass

        # Gradient nudge
        if time.time() < t0 + budget_s - 8:
            grad_f, _ = gradient_worst_lag(best_h, best_n)
            for lr in [0.005, 0.002, 0.001]:
                h_t = project(best_h - lr * grad_f, best_n)
                c_t = compute_c5(h_t, best_n)
                if c_t < best_c5 - 1e-14:
                    best_h, best_c5 = h_t, c_t
                    break

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
