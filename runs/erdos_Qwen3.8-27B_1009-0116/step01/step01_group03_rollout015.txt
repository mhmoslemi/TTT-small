    dx = 2.0 / n

def compute_c5_val(h):
    return compute_c5(h)[0]

def project_feasible(h, ts):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    for _ in range(10):
        diff2 = ts - h.sum()
        if abs(diff2) < 1e-14:
            break
        if diff2 > 0:
            j = np.argmax(1.0 - h)
        else:
            j = np.argmin(h)
        adj = min(abs(diff2), abs(1.0 - h[j]) if diff2 > 0 else abs(h[j]))
        h[j] += np.sign(diff2) * adj
    return h

def subgrad_step(h, n, alpha, ts):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    # The correlation index k_star corresponds to lag offset
    # np.correlate(h, 1-h, 'full') has length 2n-1
    # Index k_star means: sum_j h[j] * (1-h)[j + k_star - (n-1)]
    # = sum_j h[j] * (1 - h[j + k_star - n + 1])
    # So the effective shift is s = k_star - (n-1)
    s = k_star - (n - 1)
    
    # grad[i] = (1-h[i+s]) - h[i-s] (with boundary = 0)
    grad = np.zeros(n)
    if s >= 0:
        # h[i+s] valid for i < n-s
        grad[:n-s] += (1.0 - h[s:])
        # h[i-s] not valid (negative index) for all i when s>0
        # Actually h[i-s] valid for i >= s
        if s > 0:
            grad[s:] -= h[:n-s]
    else:
        ss = -s
        # h[i+s] = h[i-ss] valid for i >= ss
        grad[ss:] += (1.0 - h[:n-ss])
        # h[i-s] = h[i+ss] valid for i < n-ss
        grad[:n-ss] -= h[ss:]
    
    h_new = h - alpha * grad
    h_new = project_feasible(h_new, ts)
    return h_new

        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        shifted = (corr - m) * beta
        val = (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
        return val
    
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': 10000})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

from scipy.optimize import minimize

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            j = np.argmax(1.0 - h)
            h[j] = min(h[j] + d, 1.0)
        else:
            j = np.argmin(h)
            h[j] = max(h[j] + d, 0.0)
    return h

def soft_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)

def lbfgs_opt(h, n, ts, beta, maxiter=1000):
    res = minimize(soft_max_obj, h, args=(n, beta), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfun': 20000,
                            'gtol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def coord_desc(h, n, ts, sweeps=100, tlimit=60, levels=(0.0, 0.5, 1.0)):
    best = compute_c5_val(h)
    t0 = time.time()
    lv = np.array(levels)

            break

        for i in range(n):
            ov = h[i]
            bt, bc = ov, best
            for tv in lv:
                if abs(tv - ov) < 1e-7:
                    continue
                h[i] = tv
                h = project_feasible(h, ts)
                c = compute_c5_val(h)
                if c < bc - 1e-13:
                    bc, bt = c, tv
            h[i] = bt
            h = project_feasible(h, ts)
            if bt != ov:
                best = bc
                imp = True
        if not imp:
            break
    return h, best

def pair_search(h, n, ts, rng, sweeps=50, tlimit=60):
    best = compute_c5_val(h)
    t0 = time.time()
    for sw in range(sweeps):
        if time.time() - t0 > tlimit:
            break

            j = rng.randint(n)
            if i == j:

            h[i], h[j] = oi + d, oj - d

                imp = True
            else:

        if not imp:
            break

def anneal_coord(h, n, ts, rng, tlimit=60, levels=(0.0, 0.25, 0.5, 0.75, 1.0)):

        else:

            h = project_feasible(h, ts)
        T *= 0.995
        if T < T_min:
            T = T_min
    return best_h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            j = int(np.argmax(1.0 - h))
            h[j] = min(h[j] + d, 1.0)
        else:
            j = int(np.argmin(h))
            h[j] = max(h[j] + d, 0.0)
    return h

def slsqp_opt(h, n, ts, beta, maxit=600):
    n_half = 0.5 * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        sh = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
    def con(x):
        return x.sum() - n_half
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxit, 'ftol': 1e-15})
    x = res.x if res.x is not None and np.all(np.isfinite(res.x)) else h
    return project_feasible(x, ts)

def lbfgs_pen(h, n, ts, beta, maxit=800):
    n_half = 0.5 * n
    pen = 50.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        sh = (corr - m) * beta
        v = (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
        v += pen * (x.sum() - n_half) ** 2
        return v
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxit, 'ftol': 1e-18, 'maxfun': 30000})
    x = res.x if res.x is not None and np.all(np.isfinite(res.x)) else h
    return project_feasible(x, ts)

def cd(h, n, ts, sw=80, tl=45, lv=None):
    if lv is None:
        lv = np.array([0.0, 1.0])
    best = compute_c5_val(h)
    t0 = time.time()
    for s in range(sw):
        if time.time() - t0 > tl:
            break
        imp = False
        for i in range(n):
            ov = h[i]
            bt, bc = ov, best
            for tv in lv:
                if abs(tv - ov) < 1e-8:
                    continue
                h[i] = tv
                h = project_feasible(h, ts)
                c = compute_c5_val(h)
                if c < bc - 1e-13:
                    bc, bt = c, tv
            if bt != ov:
                h[i] = bt
                h = project_feasible(h, ts)
                best, imp = bc, True
            else:
                h[i] = ov
                h = project_feasible(h, ts)
        if not imp:
            break
    return h, best

def pair_ls(h, n, ts, rng, sw=40, tl=30):
    best = compute_c5_val(h)
    t0 = time.time()
    for s in range(sw):
        if time.time() - t0 > tl:
            break
        imp = False
        for _ in range(3 * n):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            d = (rng.random() * 0.7 + 0.1) * (1.0 if h[i] > h[j] else -1.0)
            if h[i] + d < -0.01 or h[i] + d > 1.01:
                continue
            oi, oj = h[i], h[j]
            h[i], h[j] = oi + d, oj - d
            h = project_feasible(h, ts)
            c = compute_c5_val(h)
            if c < best - 1e-13:
                best, imp = c, True
            else:
                h[i], h[j] = oi, oj
                h = project_feasible(h, ts)
        if not imp:
            break
    return h, best

def sa_coord(h, n, ts, rng, tl=50, lv=None):
    if lv is None:
        lv = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    best = compute_c5_val(h)
    bh = h.copy()
    cur = best
    T = 0.008
    t0 = time.time()
    while time.time() - t0 < tl:
        i = int(rng.randint(n))
        ov = h[i]
        tv = lv[int(rng.randint(len(lv)))]
        if abs(tv - ov) < 1e-8:
            continue
        h[i] = tv
        h = project_feasible(h, ts)
        cn = compute_c5_val(h)
        dd = cn - cur
        if dd < 0 or (T > 1e-9 and rng.random() < np.exp(-dd / T)):
            cur = cn
            if cn < best - 1e-14:
                best, bh = cn, h.copy()
        else:
            h[i] = ov
            h = project_feasible(h, ts)
        T *= 0.996
    return bh, best
