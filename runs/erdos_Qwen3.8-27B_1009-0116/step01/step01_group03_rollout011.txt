import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    if not np.any(mask):
        return h
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def make_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def smooth_obj_and_grad(h, n, beta):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    c = corr * dx
    m = c.max()
    shifted = (c - m) * beta
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    Z = exp_s.sum()
    if Z < 1e-300:
        Z = 1e-300
    weights = exp_s / Z
    obj = m + np.log(Z) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        j = k - (n - 1)
        if j >= 0:
            mlo1, mhi1 = j, n
            m2lo, m2hi = 0, n - j
        else:
            mlo1, mhi1 = 0, n + j
            m2lo, m2hi = -j, n
        m1 = np.arange(mlo1, mhi1)
        grad[m1] += w * (1.0 - h[m1 - j])
        m2 = np.arange(m2lo, m2hi)
        grad[m2] -= w * h[m2 + j]
    return obj, grad

def subgradient_optimize(h, n, ts, n_iter=200000, time_limit=40):
    h = h.copy()
    t0 = time.time()
    vel = np.zeros(n)
    mu = 0.85
    alpha0 = 0.02
    gamma = 500
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_current = best_c5
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        g, _ = smooth_obj_and_grad(h, n, 100.0)
        alpha = alpha0 / (1.0 + it / gamma) ** 0.5
        vel = mu * vel - alpha * g
        h_new = h + vel
        h_new = project_feasible(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < c5_current:
            h = h_new
            c5_current = c5_new
        else:
            h = project_feasible(h - 0.3 * vel, ts)
            vel *= 0.5
    return best_h

def lbfgs_polish(h, n, ts, beta, time_limit=30):
    t0 = time.time()
    if time.time() - t0 > time_limit:
        return h
    def fun(x):
        return smooth_obj_and_grad(x, n, beta)[0]
    def fun_and_grad(x):
        return smooth_obj_and_grad(x, n, beta)
    res = minimize(fun_and_grad, h, method='L-BFGS-B',
                   jac=True, bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 3000, 'ftol': 1e-14,
                            'gtol': 1e-12, 'maxls': 50})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return h

def local_search(h, n, ts, time_limit=30):
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(200):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = np.arange(n)
        for j in order:
            if time.time() - t0 > time_limit:
                break
            for tv in (0.0, 1.0):
                if abs(tv - h[j]) < 1e-9:
                    continue
                h[j] = tv
                h = project_feasible(h, ts)
                c5t = compute_c5(h)
                if c5t < c5 - 1e-12:
                    c5 = c5t
                    improved = True
                    break
        # Continuous perturbations
        for j in order:
            if time.time() - t0 > time_limit:
                break
            for delta in (0.02, -0.02, 0.05, -0.05, 0.1, -0.1):
                hv = h.copy()
                hv[j] = np.clip(h[j] + delta, 0.0, 1.0)
                hv = project_feasible(hv, ts)
                c5v = compute_c5(hv)
                if c5v < c5 - 1e-12:
                    h = hv
                    c5 = c5v
                    improved = True
                    break
        if not improved:
            break
    return h

    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = make_random_init(140, seed)

    best_h = h_parent.copy()

    n_candidates = [70, 90, 110, 140, 160]

    for n in n_candidates:

            break
        ts = n / 2.0
        if n == len(h_parent):
            h = h_parent.copy()
        else:
            xo = np.linspace(0, 2, len(h_parent), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(xn, xo, h_parent, period=2)
            h = project_feasible(h, ts)

        # Phase 1: Subgradient with momentum
        h = subgradient_optimize(h, n, ts, n_iter=100000, time_limit=35)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Phase 2: L-BFGS-B polish with increasing beta
        for beta in [50.0, 200.0, 600.0]:
            if remaining() < 100:
                break
            h = lbfgs_polish(h, n, ts, beta, time_limit=25)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Phase 3: Local search (binary + continuous)
        if remaining() > 60:
            h_ls = h
            h_ls = local_search(h_ls, n, ts, time_limit=35)
            c5 = compute_c5(h_ls)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ls.copy()

        # Phase 4: Another subgradient round from different init
        if remaining() > 120:
            h2 = project_feasible(h + rng.normal(0, 0.05, n), ts)

            h2 = lbfgs_polish(h2, n, ts, 300.0, time_limit=15)

            if c5 < best_c5:
