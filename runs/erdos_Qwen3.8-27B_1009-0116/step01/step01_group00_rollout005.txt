def _corr_grad(h, k, n):
    """Gradient of corr[k] w.r.t. h, where k is index in full correlate (0 to 2n-2)."""
    shift = k - (n - 1)
    grad = np.zeros(n)
    i_min = max(0, -shift)
    i_max = min(n, n - shift)
    if i_min >= i_max:
        return grad
    # For i in [i_min, i_max): corr includes h[i] * (1 - h[i+shift])
    # d/dh[i] of h[i]*(1-h[i+shift]) = (1-h[i+shift])
    # d/dh[i+shift] of h[i]*(1-h[i+shift]) = -h[i]
    valid_i = np.arange(i_min, i_max)
    grad[valid_i] += (1.0 - h[valid_i + shift])
    grad[valid_i + shift] -= h[valid_i]

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_obj(h, alpha, active_idx=None):
    corr = _corr(h)
    if active_idx is not None:
        corr = corr[active_idx]
    m = float(np.max(corr))
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= 0:
                    break
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h


def _corr_grad_one(h, k, n):
    shift = k - (n - 1)
    grad = np.zeros(n)
    i_min = max(0, -shift)
    i_max = min(n, n - shift)
    if i_min >= i_max:
        return grad
    vi = np.arange(i_min, i_max)
    grad[vi] += (1.0 - h[vi + shift])
    grad[vi + shift] -= h[vi]
    return grad


def _subgrad_step(h, step_size, dx, n, top_k=5):
    corr = _corr(h)
    m = float(np.max(corr))
    active = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in active:
        grad += _corr_grad_one(h, int(k), n)
    grad /= float(len(active))
    h_new = h - step_size * grad / dx
    return _repair(h_new, n)


def _slsqp(h, n, alpha, maxiter, ftol, active_idx=None):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smooth_obj(x, alpha, active_idx)

    def eqfun(x):
        return float(np.sum(x) - target)

    cons = [{'type': 'eq', 'fun': eqfun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hc = _repair(res.x, n)
            if _c5(hc, dx) <= _c5(h, dx) + 1e-12:
                return hc
    except Exception:
        pass
    return h


def _partial_slsqp(h, n, free_idx, alpha, maxiter):
    dx = 2.0 / n
    fixed_sum = 0.0
    for i in range(n):
        if i not in free_idx:
            fixed_sum += h[i]
    target_free = n / 2.0 - fixed_sum
    m = len(free_idx)
    if m < 1:
        return h.copy()
    h0 = h[list(free_idx)].copy()
    bounds = [(0.0, 1.0)] * m
    hfull = h.copy()

    def obj(x):
        hfull[list(free_idx)] = x
        return _smooth_obj(hfull, alpha)

    def eqfun(x):
        return float(np.sum(x) - target_free)

    cons = [{'type': 'eq', 'fun': eqfun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hnew = h.copy()
            hnew[list(free_idx)] = np.clip(res.x, 0.0, 1.0)
            hnew = _repair(hnew, n)
            if _c5(hnew, dx) <= _c5(h, dx) + 1e-12:
                return hnew
    except Exception:
        pass
    return h
