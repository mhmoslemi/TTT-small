```python
    m = np.max(corr)
```

Looking at the evaluation: `np.correlate(h, 1-h, mode='full')` - this computes the full linear correlation (no wrapping). So shifts near the edges have fewer terms. This is the correct interpretation.

```python
import numpy as np
from scipy.optimize import minimize
import time

_ct0 = 0.0
_cbudget = 999.0

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def _norm(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def _lse_obj(h, alpha, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -40.0/alpha, 40.0/alpha)
    lse = m + np.log(np.sum(np.exp(alpha * sh))) / alpha
    pen = 500.0 * (np.sum(h) - target)**2
    return lse * (2.0/n) + pen

def _grad_corr_k(h, k):
    n = len(h)
    offset = k - (n - 1)
    grad = np.zeros(n)
    j1 = np.arange(n) - offset
    mask1 = (j1 >= 0) & (j1 < n)
    grad[mask1] += (1.0 - h[j1[mask1]])
    j2 = np.arange(n) + offset
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] -= h[j2[mask2]]
    return grad

def _subgrad_descent(h, target, max_iter=8000, lr0=0.08, top_m=5):
    n = len(h)
    h = h.copy().astype(float)
    best_c = _c5(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    v1, v2 = 0.9, 0.999
    
    for it in range(max_iter):
        if time.time() - _ct0 > _cbudget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_val = float(np.max(corr)) * (2.0/n)
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h.copy()
        
        top_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += _grad_corr_k(h, k)
        grad /= top_m
        
        momentum = v1 * momentum - lr0 / (1.0 + 0.005*it) * grad
        h = h + momentum
        h = _norm(h, target)
        if it % 500 == 0 and c_val - best_c > 0.001:
            h = best_h.copy()
            momentum *= 0.5
    
    return best_h, best_c

def _coord_descent(h, target, max_rounds=30):
    n = len(h)
    h = h.copy().astype(float)
    best_c = _c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - _ct0 > _cbudget:
            break
        improved = False
        for i in range(n):
            if time.time() - _ct0 > _cbudget:
                return h, _c5(h)
            old = h[i]
            for dv in (0.03, -0.03, 0.06, -0.06, 0.1, -0.1, 0.15, -0.15, 0.0, 1.0):
                trial = old + dv
                if trial < 0.0 or trial > 1.0:
                    continue
                h2 = h.copy()
                h2[i] = trial
                h2 = _norm(h2, target)
                c2 = _c5(h2)
                if c2 < best_c - 1e-12:
                    best_c = c2
                    h = h2
                    old = trial
                    improved = True
        if not improved:
            break
    return h, best_c

def _slsqp_lhs(h_init, target, maxiter=80):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    def eqf(x): return np.sum(x) - target
    def eqj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    best_x = h_init.copy()
    best_c = _c5(h_init)
    for alpha in (100, 300, 1000, 3000, 8000):
        def obj(x):
            return _lse_obj(x, alpha, target)
        try:
            res = minimize(obj, best_x, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
            hx = _norm(res.x, target)
            cx = _c5(hx)
            if cx < best_c:
                best_c = cx
                best_x = hx.copy()
        except Exception:
            pass
    return _norm(best_x, target), best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _ct0, _cbudget
    _ct0 = time.time()
    _cbudget = budget_s - 5.0
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)
    
    pn = len(parent)
    best_c5 = _c5(parent)
    best_h = parent.copy()
    best_n = pn
    print(f"Parent: n={pn}, C5={best_c5:.6f}")
    
    n_list = [100, 140, 180]
    for n in n_list:
        if budget_s - (time.time() - _ct0) < 40:
            break
        target = n * 0.5
        x_p = np.linspace(0, 2.0, pn, endpoint=False) + 1.0/pn
        x_n = np.linspace(0, 2.0, n, endpoint=False) + 1.0/n
        h_interp = np.interp(x_n, x_p, parent)
        h_interp = _norm(h_interp, target)
        
        starts = [h_interp, np.roll(h_interp, n//4), np.roll(h_interp, -n//4)]
        for r in range(3):
            if budget_s - (time.time() - _ct0) < 20:
                break
            hc = starts[r].copy()
            
            # Phase 1: SLSQP with log-sum-exp
            if budget_s - (time.time() - _ct0) > 25:
                hc, c1 = _slsqp_lhs(hc, target, maxiter=60)
                if c1 < best_c5:
                    best_c5 = c1
                    best_h = hc.copy()
                    best_n = n
                    print(f"  n={n} r={r} SLSQP: C5={c1:.6f}")
            
            # Phase 2: Subgradient with Nesterov momentum
            if budget_s - (time.time() - _ct0) > 20:
                hc, c2 = _subgrad_descent(hc, target, max_iter=6000, lr0=0.08, top_m=5)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = hc.copy()
                    best_n = n
                    print(f"  n={n} r={r} SGD: C5={c2:.6f}")
            
            # Phase 3: Coordinate descent polish
            if budget_s - (time.time() - _ct0) > 15:
                hc, c3 = _coord_descent(hc, target, max_rounds=20)
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = hc.copy()
                    best_n = n
                    print(f"  n={n} r={r} CD: C5={c3:.6f}")
            
            # Random perturbation + re-optimize
            if budget_s - (time.time() - _ct0) > 20:
                h_pert = hc + rng.randn(n) * 0.03
                h_pert = _norm(h_pert, target)
                h_p2, c4 = _slsqp_lhs(h_pert, target, maxiter=50)
                if c4 < best_c5:
                    best_c5 = c4
                    best_h = h_p2.copy()
                    best_n = n
                    print(f"  n={n} r={r} PERT: C5={c4:.6f}")
                if budget_s - (time.time() - _ct0) > 15:
                    h_p2, c5b = _coord_descent(h_p2, target, max_rounds=15)
                    if c5b < best_c5:
                        best_c5 = c5b
                        best_h = h_p2.copy()
                        best_n = n
                        print(f"  n={n} r={r} PERT+CD: C5={c5b:.6f}")
    
    # Final polish on best solution
    if budget_s - (time.time() - _ct0) > 10:
        hn = best_h.copy()
        tgt = best_n * 0.5
        hn, cf = _coord_descent(hn, tgt, max_rounds=40)
        if cf < best_c5:
            best_c5 = cf
            best_h = hn
    
    # Ensure exact sum
    tgt = best_n * 0.5
    diff = tgt - np.sum(best_h)
    if abs(diff) > 1e-13:
        m = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(m))
        if cnt > 0:
            best_h[m] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
        d2 = tgt - np.sum(best_h)
        if abs(d2) > 1e-14:
            best_h -= d2 / best_n
            best_h = np.clip(best_h, 0.0, 1.0)
    
    c5f = _c5(best_h)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return best_h, c5f, best_n
```
