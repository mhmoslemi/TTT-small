    """Compute gradient matrix G where G[k, i] = dc[k]/dh[i]"""
    # lags ranges from -(n-1) to n-1, indexed 0 to 2n-2
    G = np.zeros((2*n-1, n))

        if lag >= 0:
            # j+lag in [0,n-1] means j in [0, n-1-lag]
            G[idx, :n-lag] = 1 - h[lag:]
            # Second term: -h[j-lag] for j-lag in [0,n-1] means j in [lag, n-1]

            G[idx, al:] = 1 - h[:n-al]
            # j-lag = j+al in [0,n-1] means j in [0, n-1-al]
