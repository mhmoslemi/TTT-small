                                 options={'maxiter': 600, 'ftol': 1e-14})
                    hc = repair(r2.x, n)
                    c = compute_c5(hc, n)
                    if c < c5 - 1e-13:
                        h, c5, stalls = hc, c, 0
                except Exception:
                    pass

            sigma_sa = max(0.02, sigma_sa * 0.997)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Final intensive polish on best solution
    if best_h is not None and time.time() - t0 < budget_s - 8:
        for _ in range(4):
            if time.time() - t0 > budget_s - 4:
                break
            try:
                rf = minimize(lambda x: compute_c5(x, best_n), best_h,
                             method='SLSQP', bounds=[(0, 1)] * best_n,
                             constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2.0}],
                             options={'maxiter': 1000, 'ftol': 1e-15})
                hc = repair(rf.x, best_n)
                c = compute_c5(hc, best_n)
                if c < best_c5:
                    best_h, best_c5 = hc, c
            except Exception:
                pass

        # One more LSE round with high beta
        if time.time() - t0 < budget_s - 5:
            try:
                r3 = minimize(lse_penalty, best_h, args=(best_n, 80.0, 500.0),
                             method='L-BFGS-B', bounds=[(0, 1)] * best_n,
                             options={'maxiter': 1000, 'ftol': 1e-15})
                hc = repair(r3.x, best_n)
                c = compute_c5(hc, best_n)
                if c < best_c5:
                    best_h, best_c5 = hc, c
            except Exception:
                pass

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_n = 70
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
