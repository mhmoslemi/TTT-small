    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def obj_smooth(h, n, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx

def subgrad_h(h, n, top_k=10):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx_top = np.argpartition(c, -top_k)[-top_k:]
    g = np.zeros(n)
    for m in idx_top:
        d = m - n + 1
        if d >= 0:
            if d == 0:
                g[:] += (1.0 - h)
            else:
                g[d:] += (1.0 - h[:n - d])
            if d < n:
                g[:n - d] -= h[d:]
        else:
            dd = -d
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] -= h[:n - dd]
    g /= float(top_k)
    return g, float(np.max(c) * (2.0 / n))

def subgrad_descent(h, n, max_iter=500, t0_budget=0.4):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.04
    patience = 0
    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > t0_budget:
            break
        g, c_cur = subgrad_h(h, n, top_k=10)
        h_new = project_sum(h - alpha * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            best_h = h_new.copy()
            patience = 0
            if c_new < c_cur * (2.0 / n) * 0.999:
                alpha = min(alpha * 1.02, 0.15)
        else:
            patience += 1
            if patience > 30:
                alpha *= 0.7
                patience = 0
            if alpha < 0.001:
                break
        h = h_new
    return best_h, best_c5

def coord_descent(h, n, max_sweeps=20, n_grid=13):
    h = repair(h.copy(), n)
    target = n * 0.5
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_i = h[i]
            j = (i + 1) % n
            old_j = h[j]
            grid = np.linspace(0.0, 1.0, n_grid)
            grid = np.unique(np.concatenate([[old_i], grid]))
            best_val = best
            best_x = old_i
            for x in grid:
                if abs(x - old_i) < 1e-10:
                    continue
                h[i] = x
                h[j] = np.clip(old_j + old_i - x, 0.0, 1.0)
                val = compute_c5(h)
                if val < best_val - 1e-13:
                    best_val = val
                    best_x = x
            if best_x != old_i:
                lo_r = max(0.0, best_x - 0.08)
                hi_r = min(1.0, best_x + 0.08)
                grid2 = np.linspace(lo_r, hi_r, 9)
                for x in grid2:
                    h[i] = x
                    h[j] = np.clip(old_j + old_i - x, 0.0, 1.0)
                    val = compute_c5(h)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_x = x
                h[i] = best_x
                h[j] = np.clip(old_j + old_i - best_x, 0.0, 1.0)
                best = best_val
                improved = True
            else:
                h[i] = old_i
                h[j] = old_j
        if not improved:
            break
    h = repair(h, n)
    return h, compute_c5(h)

def slsqp_optimize(h, n, t_param, max_iter=300):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5,
             'jac': lambda x, n=n: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_smooth, h.copy(), args=(n, dx, t_param),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return repair(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = repair(h0.copy(), n0)
    best_c5 = compute_c5(best_h)

    def time_ok(frac):
        return (budget_s * frac - (time.time() - t0)) > 10.0

    # Phase 1: Sub-gradient descent (primary new method)
    if time_ok(0.35):
        sg_budget = min(budget_s * 0.30 - (time.time() - t0), 300.0)
        h_sg, c_sg = subgrad_descent(best_h, n0, max_iter=1000, t0_budget=max(sg_budget, 5.0))
        if c_sg < best_c5:
            best_c5, best_h = c_sg, h_sg

    # Phase 2: SLSQP at multiple temperatures
    for temp in [500.0, 150.0, 50.0, 20.0]:
        if not time_ok(0.55):
            break
        t_rem = max(budget_s * 0.55 - (time.time() - t0), 10.0)
        maxiter = max(100, int(t_rem / 0.5))
        h_sls = slsqp_optimize(best_h, n0, temp, maxiter=min(maxiter, 500))
        v = compute_c5(h_sls)
        if v < best_c5:
            best_c5, best_h = v, h_sls

    # Phase 3: Coordinate descent (heavy)
    if time_ok(0.80):
        cd_budget = budget_s * 0.20 - (time.time() - t0)
        max_sweeps = max(5, int(cd_budget / 3.0))
        h_cd, c_cd = coord_descent(best_h, n0, max_sweeps=max_sweeps, n_grid=15)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd

    # Phase 4: Another sub-gradient round (finer)
    if time_ok(0.88):
        sg_budget2 = max(budget_s * 0.06 - (time.time() - t0), 5.0)
        g, _ = subgrad_h(best_h, n0, top_k=20)
        alpha = 0.01
        for it in range(200):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_try = project_sum(best_h - alpha * g, n0)
            v = compute_c5(h_try)
            if v < best_c5 - 1e-14:
                best_c5, best_h = v, h_try
                g, _ = subgrad_h(best_h, n0, top_k=20)
                alpha = min(alpha * 1.01, 0.04)
            else:
                alpha *= 0.85
                if alpha < 0.001:
                    break

    # Phase 5: Random perturbations + SLSQP
    if time_ok(0.94):
        for attempt in range(4):
            if not time_ok(0.94):
                break
            noise = np.random.randn(n0) * 0.05
            h_try = repair(best_h + noise, n0)
            t_rem = max(budget_s * 0.94 - (time.time() - t0), 5.0)
            maxiter = max(50, int(t_rem / 2.0))
            h_res = slsqp_optimize(h_try, n0, 80.0, maxiter=min(maxiter, 200))
            v = compute_c5(h_res)
            if v < best_c5:
                best_c5, best_h = v, h_res

    # Phase 6: Final coordinate descent polish
    if time_ok(0.99):
        h_final, c_final = coord_descent(best_h, n0, max_sweeps=5, n_grid=9)
        if c_final < best_c5:
            best_c5, best_h = c_final, h_final

    # Final cleanup
    best_h = repair(best_h, n0)
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n0)
