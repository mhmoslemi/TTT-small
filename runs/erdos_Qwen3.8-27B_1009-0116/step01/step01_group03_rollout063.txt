```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        h[:] = ts / len(h)
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_solve(h, n, ts, beta, maxiter=500):
    def obj(x):
        return smooth_obj(x, n, beta)
    def grad(x):
        # Numerical gradient (SLSQP will do this if not provided)
        return None
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except:
        pass
    return project_feasible(h, ts)

def binarize(h, ts, rng=None):
    """Convert to binary with exactly ts ones"""
    n = len(h)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:int(ts)]] = 1.0
    return h_bin

def sa_discrete(h, ts, t_start_val, t_end_val, max_iter, rng, deadline_fn, best_tracker):
    """Simulated annealing for binary h with exactly ts ones"""
    n = len(h)
    h = h.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    t = t_start_val
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for it in range(max_iter):
        if deadline_fn():
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / t):
            cur_c5 = new_c5
            # Update ones and zeros
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                if best_tracker:
                    best_tracker(cur_c5)
        else:
            h[i] = 1.0
            h[j] = 0.0
        t = max(t_end_val, t * 0.9997)
    
    return best_h, best_c5

def greedy_local_search(h, ts, rng, max_sweeps=50, deadline_fn=None):
    """Greedy: try all single swaps, keep the best improving one"""
    n = len(h)
    h = h.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if deadline_fn and deadline_fn():
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_delta = 0
        best_i, best_j = None, None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                delta = cur_c5 - c5
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0
        if best_i is not None:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c5 -= best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5
```

```python
from scipy.optimize import minimize
import time

_G = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        h[:] = ts / len(h)
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def _slo(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    s = (c - m) * beta
    return (m + np.log(np.sum(np.exp(s))) / beta) * (2.0 / n)

def _con(x):
    return x.sum() - _G

def slsqp(h, n, ts, beta, mi=300):
    global _G
    _G = ts
    try:
        r = minimize(_slo, h, args=(n, beta), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con}],
                     options={'maxiter': mi, 'ftol': 1e-12})
        if r.x is not None and np.all(np.isfinite(r.x)):
            return project_feasible(r.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def to_bin(h, ts, rng):
    n = len(h)
    k = int(round(ts))
    idx = np.argsort(-h)[:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa(h, ts, t0, tmin, mi, rng, t0wall, budget):
    n = len(h)
    h = to_bin(h, ts, rng)
    o = np.where(h > 0.5)[0]
    z = np.where(h < 0.5)[0]
    bc = compute_c5(h)
    cc = bc
    best = h.copy()
    for it in range(mi):
        if time.time() - t0wall > budget - 12:
            break
        t = max(tmin, t0 * (tmin / t0) ** (it / mi))
        i = o[rng.randint(len(o))]
        j = z[rng.randint(len(z))]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h)
        d = nc - cc
        if d < 0 or rng.random() < np.exp(-d / t):
            cc = nc
            o = np.where(h > 0.5)[0]
            z = np.where(h < 0.5)[0]
            if cc < bc:
                bc = cc
                best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best, bc
```

```python
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
```

```python
            break
        bd = 0.0
        bi = bj = -1
        for i in o:
            for j in z:
                h[i] = 0.0
                h[j] = 1.0
                d = cc - compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if d > bd:
                    bd = d
                    bi = i
                    bj = j
        if bi < 0:
            break
        h[bi] = 0.0
        h[bj] = 1.0
        o.remove(bi)
        o.append(bj)
        z.remove(bj)
        z.append(bi)
        cc -= bd
    return h, cc

def _make_start(n, kind, seed):
    if kind == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif kind == 1:
        h = np.zeros(n)
        h[np.random.RandomState(seed).choice(n, n // 2, replace=False)] = 1.0
    else:
        step = max(1, int(n * 0.6180339887))
        while gcd(step, n) != 1:
            step += 1
        h = np.zeros(n)
        for i in range(n // 2):
            h[(i * step) % n] = 1.0
    return h

def run(seed=42, budget_s=1100, **kw):
    tw = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        hi = np.array(initial_h_values, dtype=float)
    else:
        hi = np.zeros(140)
        hi[::2] = 1.0
    n = len(hi)
    ts = n / 2.0
    bH = hi.copy()
    bC = compute_c5(bH)

    # Phase 1: SLSQP from initial
    for bt in [5.0, 30.0, 100.0, 300.0]:
        if time.time() - tw > budget_s - 50:
            break
        h = slsqp(hi, n, ts, bt, 200)
        c = compute_c5(h)
        if c < bC:
            bC, bH = c, h.copy()

    # Phase 2: SA rounds on discrete space
    for rd in range(4):
        if time.time() - tw > budget_s - 110:
            break
        t0s = [0.02, 0.01, 0.005, 0.002][rd]
        mi = 500000 if rd < 2 else 350000
        hs, cs = sa(bH, ts, t0s, 0.0002, mi, rng, tw, budget_s)
        if cs < bC:
            bC, bH = cs, hs.copy()

    # Phase 3: SLSQP continuous polish
    for bt in [50.0, 200.0, 500.0, 1500.0]:
        if time.time() - tw > budget_s - 80:
            break
        hp = slsqp(bH, n, ts, bt, 300)
        cp = compute_c5(hp)
        if cp < bC:
            bC, bH = cp, hp.copy()

    # Phase 4: Greedy local search + polish
    if time.time() - tw < budget_s - 80:
        hg, cg = greedy(bH, ts, tw, budget_s, 100)
        if cg < bC:
            bC, bH = cg, hg.copy()
        if time.time() - tw < budget_s - 30:
            hp = slsqp(bH, n, ts, 500.0, 300)
            cp = compute_c5(hp)
            if cp < bC:
                bC, bH = cp, hp.copy()

    # Phase 5: SA re-optimization on polished
    if time.time() - tw < budget_s - 70:
        hs2, cs2 = sa(bH, ts, 0.006, 0.0002, 400000, rng, tw, budget_s)
        if cs2 < bC:
            bC, bH = cs2, hs2.copy()
            if time.time() - tw < budget_s - 30:
                hp2 = slsqp(bH, n, ts, 500.0, 300)
                cp2 = compute_c5(hp2)
                if cp2 < bC:
                    bC, bH = cp2, hp2.copy()

    # Phase 6: Alternative structured starts
    for kind in range(3):
        if time.time() - tw > budget_s - 70:
            break
        ht = _make_start(n, kind, seed + 100 * (kind + 1))
        ht, ct = sa(ht, ts, 0.01, 0.0002, 300000, rng, tw, budget_s)
        if ct < bC:
            bC, bH = ct, ht.copy()
            if time.time() - tw < budget_s - 30:
                htp = slsqp(bH, n, ts, 300.0, 200)
                ctp = compute_c5(htp)
                if ctp < bC:
                    bC, bH = ctp, htp.copy()
        elif time.time() - tw < budget_s - 40:
            htp = slsqp(ht, n, ts, 100.0, 150)
            ctp = compute_c5(htp)
            if ctp < bC:
                bC, bH = ctp, htp.copy()

    # Phase 7: Finer discretizations
    for nf in [200, 280]:
        if time.time() - tw > budget_s - 120:
            break
        tsf = nf / 2.0
        xo = np.linspace(0, 2, len(bH))
        xn = np.linspace(0, 2, nf)
        hf = np.interp(xn, xo, bH)
        hf = project_feasible(hf, tsf)
        cf = compute_c5(hf)
        for bt in [30.0, 200.0]:
            if time.time() - tw > budget_s - 30:
                break
            hf = slsqp(hf, nf, tsf, bt, 200)
            cf = compute_c5(hf)
        if time.time() - tw < budget_s - 50:
            hf, cf = sa(hf, tsf, 0.01, 0.0002, 300000, rng, tw, budget_s)
            msf = min(60, max(10, 15 - nf // 40))
            hfg, cfg = greedy(hf, tsf, tw, budget_s, msf)
            if cfg < cf:
                hf, cf = hfg, cfg
        if time.time() - tw < budget_s - 30:
            hf = slsqp(hf, nf, tsf, 500.0, 300)
            cf = compute_c5(hf)
        if cf < bC:
            bC, bH = cf, hf.copy()

    # Final SLSQP polish
    if time.time() - tw < budget_s - 15:
        tf = len(bH) / 2.0
        bH = slsqp(bH, len(bH), tf, 800.0, 250)
        bC = compute_c5(bH)

    bH = project_feasible(bH, len(bH) / 2.0)
    bC = compute_c5(bH)
    return bH, bC, len(bH)
```
