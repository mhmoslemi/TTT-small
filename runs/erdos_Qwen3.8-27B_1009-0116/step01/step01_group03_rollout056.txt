def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if mask.sum() == 0:
        return h
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining
    rem = target_sum - h.sum()
    if abs(rem) > 1e-12:
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + rem, 0, 1)
    return h

def smooth_obj_and_weights(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    shifted -= shifted.max()  # numerical stability
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    obj = (m + np.log(exps.sum()) / beta) * (2.0 / n)
    return obj, weights, corr, m

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    rem = target_sum - h.sum()
    if abs(rem) > 1e-12:
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + rem, 0, 1)
    return h

def grad_smoothed(h, beta, n_top=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    # Use top n_top lags
    top = np.argsort(w)[::-1][:n_top]
    sw = w[top].sum()
    w_use = np.zeros_like(w)
    w_use[top] = w[top] / sw
    g = np.zeros(n)
    for ki in top:
        wk = w_use[ki]
        if wk < 1e-10:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            k = lag
            if k == 0:
                # special case: c[n-1] = sum h[i](1-h[i])
                g += wk * (1 - 2*h)
            else:
                g[k:] += wk
                g[:n-k] -= wk * h[k:]
                g[k:] -= wk * h[:n-k]
        else:
            k = -lag
            if k == 0:
                g += wk * (1 - 2*h)
            else:
                g[:n-k] += wk
                g[:n-k] -= wk * h[k:]
                g[k:] -= wk * h[:n-k]

    return g

def smooth_obj(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    return (m + np.log(np.exp(sh).sum()) / beta) * (2.0 / n)

def gradient_descent(h, n, beta, lr_init, time_limit):

    lr = lr_init
    best = h.copy()

        g = grad_smoothed(h_cur, beta)
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, ts)
        f = smooth_obj(h_new, beta)
        if f < best_f:
            best_f = f
            best = h_new.copy()
        h_cur = h_new
        lr *= 0.999
    return best, best_f

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    rem = target_sum - h.sum()
    if abs(rem) > 1e-12:
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + rem, 0, 1)
    return h

def smooth_obj(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    return (m + np.log(np.exp(sh).sum()) / beta) * (2.0 / n)

def grad_smoothed(h, beta, n_top=12):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    top = np.argsort(w)[::-1][:n_top]
    sw = w[top].sum()
    w_use = np.zeros_like(w)
    w_use[top] = w[top] / sw
    g = np.zeros(n)
    for ki in top:
        wk = w_use[ki]
        if wk < 1e-10:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            k = lag
            g[k:] += wk
            g[:n - k] -= wk * h[k:]
            g[k:] -= wk * h[:n - k]
        else:
            k = -lag
            g[:n - k] += wk
            g[:n - k] -= wk * h[k:]
            g[k:] -= wk * h[:n - k]
    g *= (2.0 / n)
    return g

def gradient_descent(h, n, beta, lr_init, time_limit, top_lags=12):
    t0 = time.time()
    ts = n / 2.0
    lr = lr_init
    best = h.copy()
    best_f = smooth_obj(h, beta)
    h_cur = h.copy()
    stale = 0
    while time.time() - t0 < time_limit:
        g = grad_smoothed(h_cur, beta, n_top=top_lags)
        f_cur = smooth_obj(h_cur, beta)
        # Backtracking line search
        accepted = False
        for trial in range(8):
            h_new = project_feasible(h_cur - lr * g, ts)
            f_new = smooth_obj(h_new, beta)
            if f_new < f_cur - 1e-14:
                h_cur = h_new
                accepted = True
                if f_new < best_f:
                    best_f = f_new
                    best = h_new.copy()
                    stale = 0
                else:
                    stale += 1
                break
            lr *= 0.5
        if not accepted:
            h_cur = project_feasible(h_cur + 0.01 * np.random.randn(n), ts)
            lr *= 0.9
            stale += 1
        if stale > 200:
            lr = lr_init * 0.5
            stale = 0
        lr = min(lr * 1.002, lr_init)
    return best, best_f

def binary_sa(h, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    ts = n / 2.0
    h_b = (h >= 0.5).astype(float)
    if h_b.sum() > n // 2:
        ones = np.where(h_b == 1)[0]
        rng.shuffle(ones)
        h_b[ones[int(h_b.sum()) - n // 2:]] = 0.0
    elif h_b.sum() < n // 2:
        zeros = np.where(h_b == 0)[0]
        rng.shuffle(zeros)
        h_b[zeros[:n // 2 - int(h_b.sum())]] = 1.0
    best_c5 = compute_c5(h_b)
    cur_c5 = best_c5
    T = 1.5
    cooldown = 0.997
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        if it % 300 == 0:
            T *= cooldown
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        new_c5 = compute_c5(h_b)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
        else:
            h_b[i], h_b[j] = 1.0, 0.0
    return h_b, best_c5

def swap_local_search(h_b, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_c5 = compute_c5(h_b)
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        for i in range(n):
            if h_b[i] != 1.0:
                continue
            for j in range(i + 1, n):
                if h_b[j] != 0.0:
                    continue
                h_b[i], h_b[j] = 0.0, 1.0
                c = compute_c5(h_b)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
                else:
                    h_b[i], h_b[j] = 1.0, 0.0
    return h_b, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n = 140
        h_init = np.zeros(n)
        h_init[rng.choice(n, n // 2, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    n_candidates = sorted(set([n_init] + [c for c in [140, 160, 200, 280] if c % 2 == 0]))

    for n in n_candidates:
        if remaining() < 100:
            break
        ts = n / 2.0

        if n == n_init:
            h = h_init.copy()
        else:
            xo = np.linspace(0, 2, n_init)
            xn = np.linspace(0, 2, n)
            h = np.interp(xn, xo, h_init)
            h = project_feasible(h, ts)

        h_sa = h
        best_c5_n = best_c5
        best_h_n = best_h if len(best_h) == n else project_feasible(h, ts)

        if remaining() > 80:
            h_sa, c5_sa = binary_sa(h, n, min(50, remaining() * 0.3), seed + n)
            if c5_sa < best_c5_n:
                best_c5_n = c5_sa
                best_h_n = h_sa.copy()

        if remaining() > 40:
            h_ls, c5_ls = swap_local_search(h_sa.copy(), n, min(20, remaining() * 0.3), seed + 100 + n)
            if c5_ls < best_c5_n:
                best_c5_n = c5_ls
                best_h_n = h_ls.copy()

        if remaining() > 50:
            h_gd = best_h_n + rng.normal(0, 0.03, n)
            h_gd = project_feasible(h_gd, ts)
            for beta in [15, 40, 100, 250]:
                if remaining() < 20:
                    break
                alloc = min(30, remaining() * 0.3)
                h_gd, _ = gradient_descent(h_gd, n, beta, 0.15, alloc)
                c5_gd = compute_c5(h_gd)
                if c5_gd < best_c5_n:
                    best_c5_n = c5_gd
                    best_h_n = h_gd.copy()

        if remaining() > 20 and len(best_h_n) == n:
            h_pol = best_h_n + rng.normal(0, 0.01, n)
            h_pol = project_feasible(h_pol, ts)
            h_pol, _ = gradient_descent(h_pol, n, 400, 0.03, min(15, remaining() * 0.4), top_lags=15)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5_n:
                best_c5_n = c5_pol
                best_h_n = h_pol.copy()

        if best_c5_n < best_c5:
            best_c5 = best_c5_n
            best_h = best_h_n.copy()

    # Final restarts at best resolution
    n_best = len(best_h)
    if remaining() > 50:
        for attempt in range(5):
            if remaining() < 20:
                break
            noise_scale = 0.1 * (attempt + 1) / 4
            h_r = best_h + rng.normal(0, noise_scale, n_best)
            h_r = project_feasible(h_r, n_best / 2.0)
            h_r, _ = gradient_descent(h_r, n_best, 30, 0.1, min(12, remaining() * 0.35))
            h_r, _ = gradient_descent(h_r, n_best, 200, 0.04, min(10, remaining() * 0.3))
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
