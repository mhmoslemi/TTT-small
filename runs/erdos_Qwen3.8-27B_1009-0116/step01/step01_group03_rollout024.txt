def smooth_c5_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, 'full')
    
    # Softmax weights (shift for numerical stability)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / exp_c.sum()
    
    # F = (dx/β) * (log(sum(exp(β*c))) + β*c_max) = (dx/β) * log(sum(exp(β*c))) + dx*c_max
    # Actually: F = (dx/β) * log(Σ exp(β*c[m]))
    # = (dx/β) * (β*c_max + log(Σ exp(β*(c[m]-c_max))))
    # = dx*c_max + (dx/β)*log(Σ exp(β*(c[m]-c_max)))
    
    F = dx * (c_max + np.log(exp_c.sum()) / beta)
    
    # Gradient: dF/dh[k] = dx * Σ_m p[m] * (dc[m]/dh[k])
    grad = np.zeros(n)
    for m in range(2*n-1):
        if p[m] < 1e-15:
            continue
        if m <= n - 1:
            i_vals = np.arange(n-1-m, n)
            j_vals = np.arange(m+1)

            i_vals = np.arange(n-1-s)
            j_vals = i_vals + s + 1
        
        grad[i_vals] += p[m] * g[j_vals]
        grad[j_vals] -= p[m] * h[i_vals]
    
    grad *= dx
    return F, grad

import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        idx = np.arange(len(h))
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12 and len(idx) > 0:
        # Distribute to one or a few entries
        for i in idx:
            adj = min(max(diff2, -h[i], 1.0-h[i]), 1.0-h[i] if diff2>0 else h[i])
            h[i] += adj
            diff2 -= adj
            if abs(diff2) < 1e-14:
                break
    return h

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, 'full')
    
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / exp_c.sum()
    
    F = dx * (c_max + np.log(exp_c.sum()) / beta)
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if p[m] < 1e-12:
            continue
        if m <= n - 1:
            i_vals = np.arange(n-1-m, n)
            j_vals = np.arange(m+1)
        else:
            s = m - n
            count = n-1-s
            if count <= 0:

            i_vals = np.arange(count)
            j_vals = i_vals + s + 1
        
        grad[i_vals] += p[m] * g[j_vals]
        grad[j_vals] -= p[m] * h[i_vals]

    return F, grad

def proj_grad_descent(h, n, ts, betas, max_iter_per_beta=200, time_limit=30):
    t0 = time.time()
    h_cur = h.copy()
    y = h_cur.copy()  # Nesterov momentum
    
    for beta in betas:
        lr = 2.0 / (beta * (2.0/n))  # Lipschitz-based (rough)
        alpha = 0  # Nesterov parameter
        p_ = 1.0  # momentum
        
        for it in range(max_iter_per_beta):
            if time.time() - t0 > time_limit:
                return h_cur
            if time.time() - t0 > time_limit - 5:
                break
                
            F, grad = smooth_obj_grad(h_cur, beta)
            
            # Nesterov step
            y = h_cur - lr * grad
            t_new = (1 + np.sqrt(1 + 4*p_**2)) / 2
            alpha_new = (1 - t_new) / t_new
            y = y + alpha_new * (y - h_cur)
            p_ = t_new
            
            h_new = project_feasible(y, ts)
            
            # Check improvement
            F_new, _ = smooth_obj_grad(h_new, beta)
            if F_new > F + 1e-10:
                # Revert
                h_new = h_cur.copy()
                # Halve learning rate
                lr *= 0.5
            else:
                h_cur = h_new
            
        # Reset Nesterov between beta changes
        y = h_cur.copy()
        p_ = 1.0
    
    return h_cur

def local_search_flip(h, n, ts, max_time=45, seed=0):
    """Local search: try flipping pairs to improve C5."""
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    
    # Binary-like: round to nearest 0 or 1 where possible
    # Actually work with current fractional values
    
    for sweep in range(200):
        if time.time() - t0 > max_time:
            break
        
        improved = False
        # Try each coordinate: set to 0 or 1
        order = rng.permutation(n)
        for j in order:
            for target in [0.0, 1.0]:
                if abs(h_best[j] - target) < 1e-6:
                    continue
                h_trial = h_best.copy()
                h_trial[j] = target
                h_trial = project_feasible(h_trial, ts)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    h_best = h_trial
                    improved = True
                    break
        if not improved:
            break
    
    return h_best, best_c5

def pair_flip_search(h, n, ts, max_time=30, seed=0):
    """Local search: flip pairs (j: 0->1, k: 1->0) preserving sum exactly."""
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(100):
        if time.time() - t0 > max_time:
            break
        
        zero_mask = h_best < 0.1
        one_mask = h_best > 0.9
        if zero_mask.sum() == 0 or one_mask.sum() == 0:

        z_idx = np.where(zero_mask)[0]
        o_idx = np.where(one_mask)[0]
        rng.shuffle(z_idx)
        rng.shuffle(o_idx)
        
        improved = False
        for j in z_idx:
            if time.time() - t0 > max_time:
                break
            for k in o_idx[:30]:  # Limit for speed
                h_trial = h_best.copy()
                h_trial[j] = 1.0
                h_trial[k] = 0.0

                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    h_best = h_trial
                    improved = True
                    break
    return h_best, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        idx = np.arange(len(h))
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        # Distribute remaining to one entry
        if diff2 > 0:
            for i in idx:
                room = 1.0 - h[i]
                adj = min(room, diff2)
                h[i] += adj
                diff2 -= adj
                if abs(diff2) < 1e-14:
                    break
        else:
            for i in idx:
                room = h[i]
                adj = max(-room, diff2)
                h[i] += adj
                diff2 -= adj
                if abs(diff2) < 1e-14:
                    break
    return h

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, 'full')
    
    c_max = c.max()
    shifted = beta * (c - c_max)
    max_shifted = shifted.max()
    exp_c = np.exp(shifted - max_shifted)
    p = exp_c / exp_c.sum()
    
    F = dx * (c_max + (max_shifted + np.log(exp_c.sum())) / beta)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-14:
            continue
        pm = p[m]
        if m <= n - 1:
            cnt = m + 1
            i_vals = np.arange(n - 1 - m, n)
            j_vals = np.arange(cnt)
        else:
            s = m - n
            cnt = n - 1 - s
            if cnt <= 0:
                continue
            i_vals = np.arange(cnt)
            j_vals = i_vals + s + 1
        
        grad[i_vals] += pm * g[j_vals]
        grad[j_vals] -= pm * h[i_vals]
    
    grad *= dx
    return F, grad

def proj_grad_descent(h, ts, betas, iters_per_beta=200, time_limit=40):
    n = len(h)
    t0 = time.time()
    h_cur = h.copy()
    
    for beta in betas:
        lr = 0.05 / (1.0 + beta * 0.1)
        # Backtracking will adjust
        for it in range(iters_per_beta):
            if time.time() - t0 > time_limit:
                return h_cur
            if time.time() - t0 > time_limit - 2:
                break
            
            F, grad = smooth_obj_grad(h_cur, beta)
            
            # Gradient step with projection
            for _ in range(10):  # Backtracking
                h_new = project_feasible(h_cur - lr * grad, ts)
                F_new, _ = smooth_obj_grad(h_new, beta)
                if F_new < F + 1e-12:
                    break
                lr *= 0.5
            else:
                break
            
            h_cur = h_new
    
    return h_cur

def local_search(h, ts, max_time=30, seed=0):
    t0 = time.time()
    n = len(h)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(500):
        if time.time() - t0 > max_time:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 > max_time:
                break
            for target in (0.0, 1.0):
                if abs(h_best[j] - target) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[j] = target
                h_trial = project_feasible(h_trial, ts)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    h_best = h_trial
                    improved = True
                    break
        if not improved:
            break
    return h_best, best_c5

                h_trial[j] = 1.0

                    best_c5 = c5_t
                    h_best = h_trial

                    break
        if not improved:
            break
    return h_best, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        idx = np.arange(len(h))
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        for i in idx:
            if diff2 > 0:
                adj = min(1.0 - h[i], diff2)
            else:
                adj = max(h[i], diff2)  # negative
                adj = max(-(-adj), diff2)
                adj = min(h[i], -diff2) * -1
                adj = max(diff2, -h[i])
            h[i] += adj
            diff2 -= adj
            if abs(diff2) < 1e-14:
                break
    return h

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, 'full')
    c_max = c.max()
    shifted = beta * (c - c_max)
    ms = shifted.max()
    exp_c = np.exp(shifted - ms)
    p = exp_c / exp_c.sum()
    F = dx * (c_max + (ms + np.log(exp_c.sum())) / beta)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-13:
            continue
        pm = p[m]
        if m <= n - 1:
            cnt = m + 1
            iv = np.arange(n - 1 - m, n)
            jv = np.arange(cnt)
        else:
            s = m - n
            cnt = n - 1 - s
            if cnt <= 0:
                continue
            iv = np.arange(cnt)
            jv = iv + s + 1
        grad[iv] += pm * g[jv]
        grad[jv] -= pm * h[iv]
    grad *= dx
    return F, grad

def proj_grad_descent(h, ts, betas, iters_per_beta=150, time_limit=40):
    n = len(h)
    t0 = time.time()
    h_cur = h.copy()
    for beta in betas:
        lr = 0.02 / (1.0 + beta * 0.05)
        for it in range(iters_per_beta):
            if time.time() - t0 > time_limit - 1:
                return h_cur
            F, grad = smooth_obj_grad(h_cur, beta)
            ng = np.linalg.norm(grad)
            if ng < 1e-10:
                break
            for _ in range(15):
                h_new = project_feasible(h_cur - lr * grad, ts)
                F_new, _ = smooth_obj_grad(h_new, beta)
                if F_new <= F + 1e-14:
                    break
                lr *= 0.4
            else:
                break
            h_cur = h_new
    return h_cur

def local_search(h, ts, max_time=30, seed=0):
    t0 = time.time()
    n = len(h)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(2000):
        if time.time() - t0 > max_time:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 > max_time:
                break
            for target in (0.0, 1.0):
                if abs(h_best[j] - target) < 1e-8:
                    continue
                h_best[j] = target
                h_best = project_feasible(h_best, ts)
                c5_t = compute_c5(h_best)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    improved = True
                    break
                else:
                    h_best[j] = h_best[j]  # stays projected
        if not improved:
            break
    return h_best, best_c5

def pair_flip_search(h, ts, max_time=30, seed=0):
    t0 = time.time()
    n = len(h)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(2000):
        if time.time() - t0 > max_time:
            break
        zi = np.where(h_best < 0.2)[0]
        oi = np.where(h_best > 0.8)[0]
        if len(zi) == 0 or len(oi) == 0:
            break
        rng.shuffle(zi)
        rng.shuffle(oi)
        improved = False
        for j in zi:
            if time.time() - t0 > max_time:
                break
            for k in oi[:50]:
                h_best[j] = 1.0
                h_best[k] = 0.0
                c5_t = compute_c5(h_best)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    improved = True
                    break
                else:
                    h_best[j] = 0.0
                    h_best[k] = 1.0
        if not improved:
            break
    return h_best, best_c5

def sa_local_search(h, ts, max_time=40, seed=0):
    t0 = time.time()
    n = len(h)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    cur_c5 = best_c5
    rng = np.random.RandomState(seed)
    temp = 0.0005
    for it in range(200000):
        if time.time() - t0 > max_time:
            break
        temp *= 0.9997
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        if abs(h_best[j] - h_best[k]) < 0.1:
            continue
        h_best[j], h_best[k] = h_best[k], h_best[j]
        h_best = project_feasible(h_best, ts)
        c5_t = compute_c5(h_best)
        if c5_t < cur_c5 or rng.rand() < np.exp((cur_c5 - c5_t) / max(temp, 1e-10)):
            cur_c5 = c5_t
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
        else:
            h_best[j], h_best[k] = h_best[k], h_best[j]
            h_best = project_feasible(h_best, ts)
    return h_best, best_c5

def make_binary(h, ts):
    h_bin = (h > 0.5).astype(float)
    cs = h_bin.sum()
    if cs > ts:
        ones = np.where(h_bin == 1)[0]
        h_bin[ones[:int(cs - ts)]] = 0.0
    elif cs < ts:
        zeros = np.where(h_bin == 0)[0]
        h_bin[zeros[:int(ts - cs)]] = 1.0
    return project_feasible(h_bin, ts)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def tl():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(140)
        h0[:70] = 1.0
    n0 = len(h0)
    ts0 = n0 / 2.0
    h0 = project_feasible(h0, ts0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient descent
    if tl() > 100:
        h1 = proj_grad_descent(h0, ts0, betas=[3, 10, 30, 80, 200],
                               iters_per_beta=120, time_limit=min(50, tl()-10))
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Binary round + pair flip
    if tl() > 80:
        h_bin = make_binary(best_h, ts0)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5, best_h = c_bin, h_bin
        h_pf, c_pf = pair_flip_search(best_h, ts0, max_time=min(35, tl()-30), seed=seed)
        if c_pf < best_c5:
            best_c5, best_h = c_pf, h_pf

    # Phase 3: Coordinate local search
    if tl() > 50:
        h_ls, c_ls = local_search(best_h, ts0, max_time=min(25, tl()-20), seed=seed+10)
        if c_ls < best_c5:
            best_c5, best_h = c_ls, h_ls

    # Phase 4: SA local search
    if tl() > 50:
        h_sa, c_sa = sa_local_search(best_h, ts0, max_time=min(30, tl()-20), seed=seed+20)
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa

    # Phase 5: Gradient polish
    if tl() > 40:
        h_g2 = proj_grad_descent(best_h, ts0, betas=[50, 150, 400],
                                 iters_per_beta=100, time_limit=min(20, tl()-10))
        c_g2 = compute_c5(h_g2)
        if c_g2 < best_c5:
            best_c5, best_h = c_g2, h_g2

    # Phase 6: Pair flip again on polished
    if tl() > 30:
        h_pf2, c_pf2 = pair_flip_search(best_h, ts0, max_time=min(20, tl()-10), seed=seed+30)
        if c_pf2 < best_c5:
            best_c5, best_h = c_pf2, h_pf2

    # Phase 7: Try other resolutions
    for n_try in [70, 100, 160, 200, 280]:
        if tl() < 50:
            break
        ts_t = n_try / 2.0
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_try)
        h_t = np.interp(xn, xo, best_h)
        h_t = project_feasible(h_t, ts_t)
        h_t = proj_grad_descent(h_t, ts_t, betas=[20, 100, 300],
                                iters_per_beta=80, time_limit=min(18, tl()-25))
        h_tb = make_binary(h_t, ts_t)
        c_tb = compute_c5(h_tb)
        if c_tb < best_c5:
            best_c5, best_h = c_tb, h_tb
        h_t2, c_t2 = pair_flip_search(h_t, ts_t, max_time=min(12, tl()-15), seed=seed+n_try)
        if c_t2 < best_c5:
            best_c5, best_h = c_t2, h_t2

    # Phase 8: Final local search

        n_b = len(best_h)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
