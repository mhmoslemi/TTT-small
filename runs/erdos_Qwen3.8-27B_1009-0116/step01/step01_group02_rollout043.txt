        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt == 0:
            h += diff / n
        else:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
