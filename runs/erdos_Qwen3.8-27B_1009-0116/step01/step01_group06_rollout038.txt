import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smoothed_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    Z = np.sum(exp_c)
    w = exp_c / Z
    val = (max_c + np.log(Z) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        L = k - (n - 1)
        if L >= 0:
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] += wk * (-h[L:])
        else:
            Lp = -L
            grad[:n - Lp] += wk * (1.0 - h[Lp:])
            grad[Lp:] += wk * (-h[:n - Lp])
    return val, grad * dx

def lsqgb_smoothed(h, n, dx, beta_seq, maxiter=200):
    h = h.copy()
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    for beta in beta_seq:
        def obj(x):
            v, _ = smoothed_obj_grad(x, dx, beta)
            return v
        def grad(x):
            _, g = smoothed_obj_grad(x, dx, beta)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, n)
            c = eval_c5(h_new, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5

def slsqp_opt(h, n, dx, maxiter=500):
    target = float(n) / 2.0
    def obj(x):
        return eval_c5(x, dx)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def de_search(n, dx, seed, budget_s, popsize=30, maxiter=200):
    t0 = time.time()
    def obj(x):
        xp = project_feasible(x, n)
        return eval_c5(xp, dx)
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(obj, bounds, seed=seed, popsize=popsize,
                                      maxiter=maxiter, tol=1e-8,
                                      mutation=(0.3, 0.7), recombination=0.8,
                                      polish=True, polling=False,
                                      workers=1)
        if res.success or res.fun < float('inf'):
            ho = project_feasible(res.x, n)
            return ho, eval_c5(ho, dx)
    except Exception:

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smoothed_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    Z = np.sum(exp_c)
    w = exp_c / Z
    val = (max_c + np.log(Z) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        if L >= 0:
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] += wk * (-h[L:])
        else:
            Lp = -L
            grad[:n - Lp] += wk * (1.0 - h[Lp:])
            grad[Lp:] += wk * (-h[:n - Lp])
    return val, grad * dx

            target = float(n) / 2.0

            return v + penalty

            _, g = smoothed_obj_grad(x, dx, beta)
            target = float(n) / 2.0
            g = g + 100.0 * (x.sum() - target) * np.ones(n)

def slsqp_opt(h, n, dx, maxiter=500):
    target = float(n) / 2.0
    def obj_fn(x):
        return eval_c5(x, dx)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)
