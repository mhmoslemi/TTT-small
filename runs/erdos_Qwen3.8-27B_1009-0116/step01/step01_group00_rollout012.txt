import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _subgradient(h, n, top_m=5):
    c = _corr(h)
    if top_m >= len(c):
        idxs = np.arange(len(c))
        weights = np.full(len(c), 1.0 / len(c))
    else:
        idxs = np.argsort(c)[-top_m:]
        vals = c[idxs]
        w = np.exp(vals - vals.max())
        weights = w / w.sum()
    grad = np.zeros(n)
    for k, w in zip(idxs, weights):
        s = n - 1 - k
        if s >= 0:
            grad[0:n-s] += w * (1.0 - h[s:n])
            grad[s:n] -= w * h[0:n-s]
        else:
            sp = -s
            grad[sp:n] += w * (1.0 - h[0:n-sp])
            grad[0:n-sp] -= w * h[sp:n]
    return grad, c

def _proj_grad(h0, n, dx, lr_init=0.5, top_m=5, iters=200, momentum=0.7):
    h = h0.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stall = 0
    for it in range(iters):
        g, c = _subgradient(h, n, top_m)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        v = momentum * v - lr * g
        h_new = _repair(h + v, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = _repair(h - 0.5 * lr * g, n)
            c5_h = _c5(h, dx)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h.copy()
            stall += 1
            if stall % 10 == 0:
                lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c5

def _coord_search(h0, n, dx, rng, max_passes=40):

    best_local = _c5(h, dx)

                   constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
    hc = _repair(res.x, n)
