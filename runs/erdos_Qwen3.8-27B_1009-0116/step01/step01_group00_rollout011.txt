import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _soft_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, t / n)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h
