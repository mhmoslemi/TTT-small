    return _corr(h, dx).max()

def _proj(h, target):
    """Project onto {h in [0,1]^n : sum(h) = target}"""
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    cv = _corr(x, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _smograd(x, dx, n, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)

        vp = (jp >= 0) & (jp < n)

        # t is the last variable
        return x[-1]

        g[-1] = 1.0
        return g

        g[:n] = 1.0

            if time.time() > t_end - 4:
                break
            amp = rng.uniform(0.02, 0.08)
            hp = _proj(best_h + rng.normal(0, amp, n), target)
            for eps, nit in [(0.001, 200), (0.0002, 200)]:
                if time.time() > t_end - 2:
                    break
                try:
                    h5 = _slsqp(hp, dx, n, target, eps, nit)
                    c5 = _c5(h5, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h5.copy()
                    hp = h5
                except Exception:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_tmp = 70
        h_parent = 0.5 + 0.35 * np.sin(np.linspace(0, 4 * np.pi, n_tmp) + 0.5)
        h_parent = 0.5 + 0.25 * np.cos(np.linspace(0, 2 * np.pi, n_tmp) * 3)
        h_parent = np.clip(h_parent, 0.05, 0.95)

    n70 = 70
    dx70 = 2.0 / n70
    tgt70 = n70 / 2.0
    if len(h_parent) >= n70:
        h0 = h_parent[:n70].copy()
    else:
        h0 = np.interp(np.linspace(0, 1, n70), np.linspace(0, 1, len(h_parent)), h_parent)
    h0 = _proj(h0, tgt70)

    # Phase 1: Optimize n=70 (55% budget)
    t70_end = t0 + (deadline - t0) * 0.55
    best_h, best_c5 = _full_opt(h0, dx70, n70, tgt70, t70_end, rng)

    # Phase 2: Try n=90 (20% budget)
    if time.time() < deadline - 40:
        n90 = 90
        dx90 = 2.0 / n90
        tgt90 = n90 / 2.0
        h0_90 = np.interp(np.linspace(0, 1, n90), np.linspace(0, 1, n70), best_h)
        h0_90 = _proj(h0_90, tgt90)
        t90_end = t0 + (deadline - t0) * 0.78
        h90, c90 = _full_opt(h0_90, dx90, n90, tgt90, t90_end, rng)
        if c90 < best_c5:
            best_h, best_c5, n70, dx70, tgt70 = h90, c90, n90, dx90, tgt90

    # Phase 3: Try n=50 (10% budget)
    if time.time() < deadline - 20:
        n50 = 50
        dx50 = 2.0 / n50
        tgt50 = n50 / 2.0
        h0_50 = np.interp(np.linspace(0, 1, n50), np.linspace(0, 1, n70), best_h)
        h0_50 = _proj(h0_50, tgt50)
        t50_end = deadline - 8
        h50, c50 = _full_opt(h0_50, dx50, n50, tgt50, t50_end, rng)
        if c50 < best_c5:
            best_h, best_c5, n70, dx70, tgt70 = h50, c50, n50, dx50, tgt50

    # Phase 4: Final polish on best
    if time.time() < deadline - 3:
        try:
            hf = _slsqp(best_h, dx70, n70, tgt70, 0.00001, 400)
            cf = _c5(hf, dx70)
            if cf < best_c5:
                best_h, best_c5 = hf, cf
        except Exception:
            pass
    if time.time() < deadline - 1:
        try:
            hf = _nm_polish(best_h, dx70, n70, tgt70, 400, deadline)
            best_h = _proj(hf, tgt70)
            cf = _c5(best_h, dx70)
            if cf < best_c5:
                best_c5 = cf
        except Exception:
            pass
    if time.time() < deadline - 0.5:
        h6, c6 = _coord_desc(best_h, dx70, n70, tgt70, 4, deadline - 0.1)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    best_h = _proj(best_h, tgt70)
    best_c5 = _c5(best_h, dx70)
    return best_h, best_c5, n70
