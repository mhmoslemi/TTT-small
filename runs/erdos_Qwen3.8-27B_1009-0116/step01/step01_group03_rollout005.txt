def project_feasible(h, ts):
    """Project h to [0,1]^n with sum = ts using water-filling."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = ts - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass to elements < 1
        idx = np.where(h < 1.0)[0]
    else:
        # Need to remove mass from elements > 0
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    # Water filling
    step = diff / len(idx)
    h[idx] += step
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    residual = ts - h.sum()
    if abs(residual) > 1e-12:

            idx2 = np.where(h > 1e-15)[0]

    return h

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    r = ts - h.sum()
    if abs(r) > 1e-11 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + r, 0.0, 1.0)
    return h

def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = np.sum(np.exp(beta * (corr - m)))
    return (m + np.log(s) / beta) * dx

def slsqp_smooth(h, n, ts, beta, maxiter=2000, ftol=1e-15):
    def obj(x):
        return soft_max_obj(x, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

                v = old_val + delta
                if 0.0 <= v <= 1.0:

    k_idx = np.arange(2 * n - 1)[None, :].astype(int)
    lags = k_idx - (n - 1)
    val1 = i_idx + lags
    mask1 = (val1 >= 0) & (val1 < n)
    idx1 = np.clip(val1, 0, n - 1)
    val2 = i_idx - lags
    mask2 = (val2 >= 0) & (val2 < n)
    idx2 = np.clip(val2, 0, n - 1)
    return mask1, idx1, mask2, idx2

def proj_grad_descend(h, n, ts, beta, n_steps, lr_init, time_limit):
    mask1, idx1, mask2, idx2 = precompute_masks(n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    lr = lr_init
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = corr.max()
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        term1 = np.where(mask1, 1.0 - h[idx1], 0.0)
        term2 = np.where(mask2, h[idx2], 0.0)
        grad = (term1 @ w - term2 @ w) * dx
        h_new = project_feasible(h - lr * grad, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if step % 200 == 199:
            lr *= 0.7
    return best_h, best_c5

def pair_cd(h, n, ts, n_sweeps, time_limit, rng):
    best_c5 = compute_c5(h)
    t0 = time.time()
    best_improve_count = 0
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t0 > time_limit:
                break
            j = rng.randint(n)
            if j == i:
                continue
            hi, hj = h[i], h[j]
            d_lo = max(-hi, hj - 1.0)
            d_hi = min(1.0 - hi, hj)
            if d_hi - d_lo < 1e-12:
                continue
            deltas = np.linspace(d_lo, d_hi, 15)
            best_delta = 0.0
            best_local = best_c5
            for delta in deltas:
                h[i] = hi + delta
                h[j] = hj - delta
                c5 = compute_c5(h)
                if c5 < best_local - 1e-15:
                    best_local = c5
                    best_delta = delta
            if best_delta != 0.0:
                h[i] = hi + best_delta
                h[j] = hj - best_delta
                best_c5 = best_local
                improved = True
                best_improve_count += 1
            else:
                h[i] = hi
                h[j] = hj
        if not improved:
            break
    return h, best_c5

def binary_ls(h, n, ts, time_limit):
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(200):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            for v in (0.0, 1.0):
                if abs(v - h[j]) < 1e-8:
                    continue
                old = h[j]
                h[j] = v
                h = project_feasible(h, ts)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                else:
                    h[j] = old
                    h = project_feasible(h, ts)
        if not improved:
            break
    return h, best_c5

def fine_coord_descent(h, n, ts, time_limit, rng):
    """Try random single-coordinate moves with fine grid."""
    best_c5 = compute_c5(h)
    t0 = time.time()
    n_tries = 0
    for _ in range(200000):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        old = h[i]
        # Try 5 random candidates around current value
        candidates = rng.uniform(0, 1, 5)
        candidates = np.clip(candidates, 0, 1)
        candidates = np.unique(np.round(candidates, 6))
        best_local = best_c5
        best_v = old
        for v in candidates:
            if abs(v - old) < 1e-6:
                continue
            h[i] = v
            h = project_feasible(h, ts)
            c5 = compute_c5(h)
            if c5 < best_local - 1e-15:
                best_local = c5
                best_v = v
        if best_v != old:
            h[i] = best_v
            h = project_feasible(h, ts)
            best_c5 = best_local
        else:
            h[i] = old
            h = project_feasible(h, ts)
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def rem():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n_init = 140
        h_init = np.zeros(n_init)
        h_init[rng.choice(n_init, n_init // 2, replace=False)] = 1.0

    n = len(h_init)
    ts = n / 2.0
    best_h = project_feasible(h_init.copy(), ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing smoothness
    h = best_h.copy()
    for beta in [10.0, 30.0, 80.0, 200.0]:
        if rem() < 60:
            break
        h = slsqp_opt(h, n, ts, beta, maxiter=1200)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Projected gradient descent (fast, many steps)
    if rem() > 120:
        h_g = best_h.copy()
        h_g, c5_g = proj_grad_descend(h_g, n, ts, 300.0, 3000, 0.3, min(80, rem() - 60))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 3: Pair coordinate descent
    if rem() > 150:
        h_p = best_h.copy()
        h_p, c5_p = pair_cd(h_p, n, ts, 80, min(120, rem() - 80), rng)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: SLSQP polish at high beta
    if rem() > 100:
        for beta in [500.0, 1000.0]:
            if rem() < 50:
                break
            h = slsqp_opt(best_h.copy(), n, ts, beta, maxiter=800)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 5: Binary (0/1) local search
    if rem() > 60:
        h_b = best_h.copy()
        h_b, c5_b = binary_ls(h_b, n, ts, min(80, rem() - 30))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: Another round of pair CD + gradient
    if rem() > 100:
        h_p2 = best_h.copy()
        h_p2, c5_p2 = pair_cd(h_p2, n, ts, 60, min(60, rem() - 50), rng)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()
        if rem() > 50:
            h_g2, c5_g2 = proj_grad_descend(best_h.copy(), n, ts, 500.0, 2000, 0.1, min(40, rem() - 20))
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()

    # Phase 7: Random perturbations + re-optimize
    if rem() > 120:
        for _ in range(2):
            if rem() < 60:
                break
            h_r = best_h + rng.normal(0, 0.04, n)
            h_r = project_feasible(h_r, ts)
            h_r = slsqp_opt(h_r, n, ts, 400.0, maxiter=600)
            h_r, c5_r = binary_ls(h_r, n, ts, min(20, rem() - 20))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Final fine pair CD
    if rem() > 30:
        h_f, c5_f = pair_cd(best_h.copy(), n, ts, 30, min(25, rem() - 5), rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
