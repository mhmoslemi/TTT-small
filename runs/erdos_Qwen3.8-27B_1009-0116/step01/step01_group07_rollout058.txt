import time

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _make_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float)
    if len(h) != n:
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_old, h)
    # Sort by value, threshold
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _sa(h, n, iters, T0, T1, rng, time_limit):
    """Simulated annealing on binary sequence."""
    dx = 2.0 / n
    h = h.copy()
    cur = _c5(h)
    best = cur
    best_h = h.copy()
    T = T0
    cool = (T1 / T0) ** (1.0 / iters)

                best_h = h.copy()
