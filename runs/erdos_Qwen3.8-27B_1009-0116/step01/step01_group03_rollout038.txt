```python
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def soft_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    if m <= 0:
        return 0.0
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_solve(h, n, ts, beta, maxiter=600):
    def obj(x):
        return soft_obj(x, n, beta)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except:
        pass
    return project_feasible(h, ts)

def coord_descent(h, n, ts, rng, max_sweeps=100, time_limit=200):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            ov = h[j]
            cands = set()
            cands.add(0.0)
            cands.add(1.0)
            if 0 < ov < 1:
                cands.add(min(1.0, ov + 0.3))
                cands.add(max(0.0, ov - 0.3))
            cands = [c for c in cands if abs(c - ov) > 1e-8]
            bv, bc = ov, best_c5
            for cv in cands:
                h[j] = cv
                h = project_feasible(h, ts)
                c = compute_c5(h)

            if bc < best_c5 - 1e-15:

                improved = True
        if not improved:

    return h, best_c5
```

```python
def binary_sa(h, n, ts, rng, time_limit=300):
    """Simulated annealing on binary swaps"""
    t0 = time.time()
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, ts)
    # Ensure exactly n/2 ones
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    current_c5 = compute_c5(h_bin)
    best_c5 = current_c5
    best_h = h_bin.copy()
    
    T = 0.01  # initial temperature
    T_min = 1e-8
    cooling = 0.9995
    
    while time.time() - t0 < time_limit and T > T_min:
        # Pick a random swap: one 1 -> 0, one 0 -> 1
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        
        T *= cooling
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def soft_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_solve(h, n, ts, beta, maxiter=600):
    def obj(x):
        return soft_obj(x, n, beta)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except:
        pass
    return project_feasible(h, ts)

    h_sa = project_feasible(h_sa, ts)

    best_c5 = cur_c5

    T = 0.005

        ones = np.where(h_sa == 1.0)[0]
        zeros = np.where(h_sa == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_sa[i], h_sa[j] = 0.0, 1.0
        new_c5 = compute_c5(h_sa)
        delta = new_c5 - cur_c5

            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_sa.copy()
        else:
            h_sa[i], h_sa[j] = 1.0, 0.0
        T *= cooling
    return best_h, best_c5

    t0 = time.time()

    h = h.copy()
    for sweep in range(max_sweeps):

            if bc < best_c5 - 1e-15:

        if not improved:
            break
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def soft_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_solve(h, n, ts, beta, maxiter=600):
    def obj(x):
        return soft_obj(x, n, beta)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def binary_sa(h, n, ts, rng, time_limit=200):
    t0 = time.time()
    h_sa = (h > 0.5).astype(float)
    h_sa = project_feasible(h_sa, ts)
    cur_c5 = compute_c5(h_sa)
    best_c5 = cur_c5
    best_h = h_sa.copy()
    T = 0.004
    cooling = 0.99995
    ones = np.where(h_sa == 1.0)[0].tolist()
    zeros = np.where(h_sa == 0.0)[0].tolist()
    while time.time() - t0 < time_limit:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_sa[i] = 0.0
        h_sa[j] = 1.0
        new_c5 = compute_c5(h_sa)
        delta = new_c5 - cur_c5
        if delta < 0 or (delta / T < 600 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_sa.copy()
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
        else:
            h_sa[i] = 1.0
            h_sa[j] = 0.0
        T *= cooling
    return best_h, best_c5

def coord_descent(h, n, ts, rng, max_sweeps=80, time_limit=150):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in rng.permutation(n):
            ov = h[j]
            cands = [0.0, 1.0, max(0.0, ov - 0.25), min(1.0, ov + 0.25)]
            cands = [c for c in cands if abs(c - ov) > 1e-8]
            bv, bc = ov, best_c5
            for cv in cands:
                h[j] = cv
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < bc - 1e-15:
                    bc, bv = c, cv
            h[j] = bv
            h = project_feasible(h, ts)
            if bc < best_c5 - 1e-15:
                best_c5 = bc
                improved = True
        if not improved:
            break
    return h, best_c5

def fine_polish(h, n, ts, rng, max_sweeps=40, time_limit=80):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in rng.permutation(n):
            ov = h[j]
            bv, bc = ov, best_c5
            for d in (0.03, -0.03, 0.01, -0.01, 0.05, -0.05, 0.08, -0.08):
                cv = ov + d
                if cv < -0.01 or cv > 1.01:
                    continue
                h[j] = cv
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < bc - 1e-16:
                    bc, bv = c, cv
            h[j] = bv
            h = project_feasible(h, ts)
            if bc < best_c5 - 1e-16:
                best_c5 = bc
                improved = True
        if not improved:
            break
    return h, best_c5

            if rem() > 30:
```
