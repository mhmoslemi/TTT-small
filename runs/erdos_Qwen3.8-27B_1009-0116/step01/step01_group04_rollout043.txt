    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)

def slsqp_smooth(x0, n, dx, tau, top_k, maxiter=400):
    _G['n'] = n
    _G['dx'] = dx
    _G['tau'] = tau
    _G['top_k'] = top_k
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cons_eq_f, 'jac': _cons_eq_j}]
    try:
        r = minimize(_obj_smooth, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def slsqp_constrained(x0, n, dx, target, top_k, maxiter=300):
    _G['n'] = n
    _G['dx'] = dx
    _G['target'] = target
    _G['top_k'] = top_k
    bd = [(0.0, 1.0)] * n
    cs = [
        {'type': 'eq', 'fun': _cons_eq_f, 'jac': _cons_eq_j},
        {'type': 'ineq', 'fun': _ineq_fun}
    ]
    try:
        r = minimize(_obj_const, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def refine_schedule(x0, n, dx, time_limit):
    t0 = time.time()
    x = x0.copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    # Progressive smooth-max optimization
    schedule = [
        (0.3, 0, 300),
        (0.15, 0, 300),
        (0.08, 8, 300),
        (0.04, 8, 400),
        (0.02, 6, 400),
        (0.01, 6, 400),
        (0.005, 5, 400),
        (0.002, 5, 400),
        (0.001, 5, 300),
        (0.0005, 4, 300),
    ]
    for tau, tk, mi in schedule:
        if time.time() - t0 > time_limit - 2.0:
            break
        x = slsqp_smooth(x, n, dx, tau, tk, mi)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
    return best, best_val

def refine_constrained(x0, n, dx, time_limit):
    t0 = time.time()
    x = x0.copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    # Binary-search-like: tighten target
    current_target = best_val + 0.005
    for iteration in range(8):
        if time.time() - t0 > time_limit - 2.0:
            break
        target = best_val - 0.002 * (iteration + 1)
        if target < best_val - 0.02:
            target = best_val - 0.02
        x = slsqp_constrained(x, n, dx, target, 25, 250)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
            current_target = val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(95, 0.5)
    n_parent = len(h0_raw)
    dx_parent = 2.0 / n_parent

    best_h = h0_raw.copy()
    best_val = compute_c5(best_h, dx_parent)
    best_n = n_parent

    # Phase 1: Optimize at n=95 (project down if needed)
    n95 = 95
    dx95 = 2.0 / n95
    if n_parent != n95:
        h95 = project_grid(h0_raw, n_parent, n95)
    else:
        h95 = h0_raw.copy()
    h95 = repair(h95, n95)

    p1_budget = min(budget_s * 0.35, 350)
    n_restarts = 5
    per_r = p1_budget / n_restarts

    h95_best = h95.copy()
    h95_best_val = compute_c5(h95, dx95)

    for ri in range(n_restarts):
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.55:
            break
        alloc = min(per_r, budget_s * 0.55 - elapsed - 5)
        if alloc < 15:
            break
        if ri == 0:
            xs = h95.copy()
        elif ri == 1:
            xs = repair(h95 + 0.08 * rng.standard_normal(n95), n95)
        elif ri == 2:
            xs = repair(h95 + 0.15 * rng.standard_normal(n95), n95)
        elif ri == 3:
            xs = repair(h95 + 0.25 * rng.standard_normal(n95), n95)
        else:
            xs = h95_best.copy()

        x_best, val = refine_schedule(xs, n95, dx95, alloc)
        if val < h95_best_val:
            h95_best_val = val
            h95_best = x_best.copy()

    # Constrained refinement at n=95
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.60:
        alloc = min(budget_s * 0.12, budget_s * 0.60 - elapsed - 5)
        if alloc > 15:
            x_c, val_c = refine_constrained(h95_best, n95, dx95, alloc)
            if val_c < h95_best_val:
                h95_best_val = val_c
                h95_best = x_c.copy()

    # Update global best if n=95 is better
    if h95_best_val < best_val:
        best_val = h95_best_val
        best_h = h95_best.copy()
        best_n = n95

    # Phase 2: Try n=75 for easier optimization
    n75 = 75
    dx75 = 2.0 / n75
    h75 = project_grid(h95_best, n95, n75)
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.65:
        alloc = min(budget_s * 0.15, budget_s * 0.65 - elapsed - 5)
        if alloc > 15:
            x_best75, val75 = refine_schedule(h75, n75, dx75, alloc)
            if val75 < best_val:
                best_val = val75
                best_h = x_best75.copy()
                best_n = n75

    # Phase 3: Project best up to parent grid and refine
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.82 and best_n < n_parent:
        h_up = project_grid(best_h, best_n, n_parent)
        alloc = min(budget_s * 0.15, budget_s * 0.82 - elapsed - 5)
        if alloc > 20:
            x_up, val_up = refine_schedule(h_up, n_parent, dx_parent, alloc)
            if val_up < best_val:
                best_val = val_up
                best_h = x_up.copy()
                best_n = n_parent

    # Phase 4: Final polish on best grid
    remaining = budget_s - (time.time() - t_start) - 8
    if remaining > 20:
        dx_best = 2.0 / best_n
        # Very fine tau sweep
        for tau in [0.002, 0.001, 0.0005, 0.0003]:
            if time.time() - t_start > budget_s - 8:
                break
            best_h = slsqp_smooth(best_h, best_n, dx_best, tau, 4, 300)
            val = compute_c5(best_h, dx_best)
            if val < best_val:
                best_val = val

    # Phase 5: One more constrained pass
    remaining = budget_s - (time.time() - t_start) - 8
    if remaining > 25:
        dx_best = 2.0 / best_n
        best_h = slsqp_constrained(best_h, best_n, dx_best,
                                   best_val - 0.003, 20, 250)
        val = compute_c5(best_h, dx_best)
        if val < best_val:
            best_val = val

    # Final
    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
