def slp_step(h, n, max_iters=50):
    """Sequential Linear Programming: linearize max of L_p, solve LP, iterate"""
    h_cur = h.copy()
    for _ in range(max_iters):
        lags = compute_lags(h_cur, n)
        # Linearize each L_p at h_cur
        # L_p(h) ≈ L_p(h_cur) + g_p·(h - h_cur) where g_p = grad_lag(h_cur, n, p)
        # So: g_p·h ≤ t - L_p(h_cur) + g_p·h_cur
        #     g_p·h - t ≤ -L_p(h_cur) + g_p·h_cur
        #     g_p·h - t ≤ g_p·h_cur - L_p(h_cur)
        
        n_lags = 2*n - 1
        A = np.zeros((n_lags + 1, n + 1))  # n h-variables + 1 t-variable
        b = np.zeros(n_lags + 1)
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        for p in range(n_lags):
            g = grad_lag(h_cur, n, p)
            A[p, :n] = g
            A[p, n] = -1.0
            b[p] = g @ h_cur - lags[p]
        
        # Sum constraint: sum(h) = n/2 → use two inequalities
        # Actually linprog needs ≤ constraints. Use: sum(h) ≤ n/2 and -sum(h) ≤ -n/2
        # Or use bounds and the sum as a separate constraint
        # Let's add sum constraint as row n_lags
        A[n_lags, :n] = 0  # placeholder, will handle separately
        
        # Actually, let's use a different formulation. linprog supports A_ub, b_ub, A_eq, b_eq
        A_ub = np.zeros((n_lags, n + 1))
        b_ub = np.zeros(n_lags)
        for p in range(n_lags):
            g = grad_lag(h_cur, n, p)
            A_ub[p, :n] = g
            A_ub[p, n] = -1.0
            b_ub[p] = g @ h_cur - lags[p]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        A_eq[0, n] = 0.0
        b_eq = np.array([n / 2.0])
        
        bounds_list = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_list, method='highs')
            if res.success:
                h_new = repair(res.x[:n], n)
                return h_new
        except:
            pass
        # If LP fails, break
        break
    return h_cur

        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def _eq_con(x):
    return np.sum(x) - _opt_n[0] / 2.0

def _max_obj(x):
    n = _opt_n[0]
    return compute_c5(repair(x, n), n)

def _lse_obj(x):
    n = _opt_n[0]
    b = _opt_beta[0]
    h = repair(x, n)
    lags = compute_lags(h, n)
    m = np.max(lags)
    scaled = b * lags - b * m
    s = np.sum(np.exp(scaled))
    return m + np.log(s) / b

def slsqp_optimize(h, n, use_lse=False, beta=20.0, maxiter=300, ftol=1e-12):
    _opt_n[0] = n
    if use_lse:
        _opt_beta[0] = beta
        obj_fn = _lse_obj
    else:
        obj_fn = _max_obj
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    try:
        r = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return repair(r.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, max_passes=30, deadline=None):
    h = h.copy()
    c5 = compute_c5(h, n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            hi = h[i]
            if hi <= 1e-10:
                continue
            for j in range(i + 1, n):
                hj = h[j]
                if hj >= 1.0 - 1e-10:
                    continue
                # Try moving mass from i to j
                di = h[i]
                dj = h[j]
                best_c, best_d = c5, 0.0
                for frac in (0.3, 0.6, 1.0):
                    d = min(di, 1.0 - dj) * frac
                    if d < 1e-12:
                        continue
                    h[i], h[j] = di - d, dj + d
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c, best_d = c, d
                if best_d > 0:
                    h[i], h[j] = di - best_d, dj + best_d
                    c5 = best_c
                    improved = True
                else:
                    h[i], h[j] = di, dj
                # Try moving mass from j to i
                di, dj = h[i], h[j]
                if dj <= 1e-10 or di >= 1.0 - 1e-10:
                    continue
                best_c, best_d = c5, 0.0
                for frac in (0.3, 0.6, 1.0):
                    d = min(dj, 1.0 - di) * frac
                    if d < 1e-12:
                        continue
                    h[j], h[i] = dj - d, di + d
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c, best_d = c, d
                if best_d > 0:
                    h[j], h[i] = dj - best_d, di + best_d
                    c5 = best_c
                    improved = True
                else:
                    h[j], h[i] = dj, di
        if not improved:
            break
    return h, c5

def make_starts(n, rng, init_vals, init_n):
    starts = []
    if init_vals is not None:
        if n == init_n:
            starts.append(init_vals.copy())
        else:
            x_old = np.linspace(0, 2, init_n)
            x_new = np.linspace(0, 2, n)
            starts.append(np.interp(x_new, x_old, init_vals))
    x = np.linspace(0, 2, n)
    starts.append(0.5 + 0.45 * np.sin(np.pi * x))
    starts.append(0.5 + 0.45 * np.sin(2 * np.pi * x))
    starts.append(0.5 + 0.4 * np.cos(np.pi * x / 2))
    h_r = rng.random(n)
    starts.append(h_r)
    h_b = np.zeros(n)
    h_b[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(h_b)
    return starts
