from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def lse_objective(h, n, beta, lam):
    dx = 2.0 / n
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    lse = (m + np.log(exp_s.sum() + 1e-300) / beta) * dx
    penalty = lam * (h.sum() - ts) ** 2
    return lse + penalty

def lse_gradient(h, n, beta, lam):
    dx = 2.0 / n
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ps = p[s + n - 1]
        if ps < 1e-15:
            continue
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_end > i_start:
            grad[i_start:i_end] += ps * (1.0 - h[i_start+s:i_end+s])
        j_start = max(0, s)
        j_end = min(n, n + s)
        if j_end > j_start:
            grad[j_start:j_end] -= ps * h[j_start-s:j_end-s]
    
    grad *= dx
    grad += 2.0 * lam * (h.sum() - ts)
    return grad

    res = minimize(lse_objective, h, jac=lse_gradient,
                   args=(n, beta, lam), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def subgradient_step(h, n, step_size):
    dx = 2.0 / n
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)
    
    # Subgradient of max at h
    subgrad = np.zeros(n)
    i_start = max(0, -s)
    i_end = min(n, n - s)
    if i_end > i_start:
        subgrad[i_start:i_end] += (1.0 - h[i_start+s:i_end+s])
    j_start = max(0, s)
    j_end = min(n, n + s)
    if j_end > j_start:
        subgrad[j_start:j_end] -= h[j_start-s:j_end-s]
    subgrad *= dx
    
    h_new = h - step_size * subgrad
    h_new = project_feasible(h_new, ts)
    return h_new

                    improved = True
                    break
            # If no improvement, restore
            h[i] = old
            h = project_feasible(h, ts)
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def lse_obj(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    e = np.exp(sh)
    return (m + np.log(e.sum() + 1e-300) / beta) * dx + lam * (h.sum() - n / 2.0) ** 2

def lse_grad(h, n, beta, lam):
    dx = 2.0 / n
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    e = np.exp(sh)
    p = e / e.sum()
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        ps = p[s + n - 1]
        if ps < 1e-16:
            continue
        a = max(0, -s)
        b = min(n, n - s)
        if b > a:
            g[a:b] += ps * (1.0 - h[a + s:b + s])
        c = max(0, s)
        d = min(n, n + s)
        if d > c:
            g[c:d] -= ps * h[c - s:d - s]
    g *= dx
    g += 2.0 * lam * (h.sum() - ts)
    return g

def lbfgs(h, n, beta, lam, maxiter=400):
    ts = n / 2.0
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, beta, lam),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def subgrad_refine(h, n, max_time=5.0):
    ts = n / 2.0
    t0 = time.time()
    alpha = 0.05
    for _ in range(500):
        if time.time() - t0 > max_time:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        ki = np.argmax(corr)
        s = ki - (n - 1)
        sg = np.zeros(n)
        a, b = max(0, -s), min(n, n - s)
        if b > a:
            sg[a:b] += 1.0 - h[a + s:b + s]
        c, d = max(0, s), min(n, n + s)
        if d > c:
            sg[c:d] -= h[c - s:d - s]
        sg *= dx
        h = project_feasible(h - alpha * sg, ts)
        alpha *= 0.98
    return project_feasible(h, ts)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def lse_obj(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    e = np.exp(sh)
    return (m + np.log(e.sum() + 1e-300) / beta) * dx + lam * (h.sum() - n / 2.0) ** 2

def lse_grad(h, n, beta, lam):
    dx = 2.0 / n
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    sh -= sh.max()
    e = np.exp(sh)
    p = e / e.sum()
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        ps = p[s + n - 1]
        if ps < 1e-16:
            continue
        a, b = max(0, -s), min(n, n - s)
        if b > a:
            g[a:b] += ps * (1.0 - h[a + s : b + s])
        c, d = max(0, s), min(n, n + s)
        if d > c:
            g[c:d] -= ps * h[c - s : d - s]
    g *= dx
    g += 2.0 * lam * (h.sum() - ts)
    return g

def lbfgs(h, n, beta, lam, maxiter=400):
    ts = n / 2.0
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, beta, lam),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

    ts = n / 2.0

        sg *= dx
        h = project_feasible(h - alpha * sg, ts)
