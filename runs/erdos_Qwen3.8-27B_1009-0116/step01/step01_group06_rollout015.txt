    """Convert h to binary with exactly n/2 ones, minimizing L2 distance."""
    h = h.copy()
    target = n // 2
    # Sort by value, take top n/2
    idx = np.argsort(-h)[:target]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

import numpy as np
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def local_search(h, n, dx, time_limit=10, max_passes=200000):
    t0 = time.time()
    h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    best = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    stall = 0
    passes = 0
    while passes < max_passes and time.time() - t0 < time_limit:
        passes += 1
        moved = False
        for ii in range(len(ones)):
            i = ones[ii]
            for jj in range(len(zeros)):
                j = zeros[jj]
                h[i] = 0.0
                h[j] = 1.0
                v = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                h[i] = 1.0
                h[j] = 0.0
                if v < best - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    ones[ii] = j
                    zeros[jj] = i
                    best = v
                    moved = True
                    break
            if moved:
                break
        if moved:
            stall = 0
        else:
            stall += 1
            if stall > 3:
                break
    return h, best

def sa(h, n, dx, time_limit=60, seed=42, t_hi=0.02, t_lo=1e-4, kicks=10000, ksz=3):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    cur = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    best_c = cur
    best_h = h.copy()
    n_iter = 3000000
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        T = max(t_hi * (t_lo / t_hi) ** (it / n_iter), t_lo)
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i, j = ones[ii], zeros[jj]
        h[i] = 0.0
        h[j] = 1.0
        nv = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
        d = nv - cur
        if d < 0 or (T > t_lo * 1.2 and rng.random() < np.exp(-d / (T + 1e-16))):
            cur = nv
            ones[ii] = j
            zeros[jj] = i
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        if kicks > 0 and (it + 1) % kicks == 0:
            for _ in range(ksz):
                a = rng.randint(len(ones))
                b = rng.randint(len(zeros))
                h[ones[a]] = 0.0
                h[zeros[b]] = 1.0
                ones[a], zeros[b] = zeros[b], ones[a]
            cur = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    return best_h, best_c
