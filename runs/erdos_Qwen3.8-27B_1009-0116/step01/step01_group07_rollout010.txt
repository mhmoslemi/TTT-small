import time

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    return h

def _subgrad_step(h, step, top_k=30):
    """One projected subgradient step."""
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    peak = np.max(corr)
    active = np.zeros(2 * n - 1)
    thr = peak - 0.05 * peak
    active[corr > thr] = 1.0
    if np.sum(active) == 0:
        active[np.argmax(corr)] = 1.0
    nk = np.sum(active)
    g = np.zeros(n)
    for idx in np.where(active > 0)[0]:
        sk = idx - (n - 1)
        jp_arr = np.arange(n) + sk
        jm_arr = np.arange(n) - sk
        valid_p = (jp_arr >= 0) & (jp_arr < n)
        valid_m = (jm_arr >= 0) & (jm_arr < n)
        g[valid_p] += (1.0 - h[jp_arr[valid_p]])
        g[valid_m] += (-h[jm_arr[valid_m]])
    g /= nk
    h2 = h - step * g
    return _project(h2, target)

def _slsqp_polish(h, target, bounds, cons, dx, maxiter=100):
    n = len(h)
    alpha = 500.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(corr - m, -1.0, 1.0)
        return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return _project(res.x, target)

def _interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    return _project(h_dst, n_dst / 2.0)

def _structured_init(n, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = np.full(n, 0.5)
    x = np.arange(n) / n * 2.0
    for _ in range(rng.randint(2, 6)):
        f = rng.randint(1, 6)
        p = rng.uniform(0, 2.0 * np.pi)
        a = rng.uniform(0.1, 0.4)
        h += a * np.sin(2.0 * np.pi * f * x / 2.0 + p)
    h = _project(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)
    parent = _project(parent, len(parent) / 2.0)
    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = len(parent)

    n_list = [80, 100, 120, 150, 200, 250]
    for n in n_list:
        if budget_s - (time.time() - t0) < 30:
            break
        target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]

        h = _interp_to_n(parent, n)
        # Coordinate descent
        h = _cd_sweep(h, 4)
        c = _c5(h)
        if c < best_c5:
            best_c5, best_h, best_n = c, h.copy(), n
            print(f"n={n} CD: C5={c:.6f}")

        # Subgradient + CD alternating
        step = 0.05
        for it in range(80):
            if budget_s - (time.time() - t0) < 15:
                break
            h = _subgrad_step(h, step, top_k=20)
            step *= 0.97
            h = _cd_sweep(h, 1)
            c = _c5(h)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n
                if it % 20 == 0:
                    print(f"n={n} sub+CD it={it}: C5={c:.6f}")

        # SLSQP polish
        if budget_s - (time.time() - t0) > 20:
            h = _slsqp_polish(h, target, bounds, cons, dx, maxiter=80)
            h = _cd_sweep(h, 2)
            c = _c5(h)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n
                print(f"n={n} SLSQP: C5={c:.6f}")

        # Restart with structured init
        if budget_s - (time.time() - t0) > 60:
            for rs in range(3):
                if budget_s - (time.time() - t0) < 20:
                    break
                h2 = _structured_init(n, seed + n * 7 + rs)
                h2 = _cd_sweep(h2, 3)
                c = _c5(h2)
                step2 = 0.04
                for it2 in range(50):
                    if budget_s - (time.time() - t0) < 10:
                        break
                    h2 = _subgrad_step(h2, step2, top_k=15)
                    step2 *= 0.97
                    h2 = _cd_sweep(h2, 1)
                    c = _c5(h2)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h2.copy(), n
                    print(f"n={n} rs={rs}: C5={c:.6f}")
            # SLSQP on restart
            if budget_s - (time.time() - t0) > 20:
                h2 = _slsqp_polish(h2, target, bounds, cons, dx, maxiter=60)
                h2 = _cd_sweep(h2, 2)
                c = _c5(h2)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h2.copy(), n
                    print(f"n={n} rs+SLSQP: C5={c:.6f}")

    # Final polish on best
    if budget_s - (time.time() - t0) > 10:
        bn = best_n
        ht = _cd_sweep(best_h.copy(), 5)
        ct = _c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
        dxn = 2.0 / bn
        bnd = [(0.0, 1.0)] * bn
        cns = [{'type': 'eq', 'fun': lambda x: np.sum(x) - bn / 2.0,
                'jac': lambda x: np.ones(bn)}]
        ht = _slsqp_polish(best_h, bn / 2.0, bnd, cns, dxn, maxiter=50)
        ht = _cd_sweep(ht, 3)
        ct = _c5(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht
            best_n = bn

    # Ensure exact constraint
    ht = best_h.copy()
    tgt = best_n / 2.0
    diff = tgt - np.sum(ht)
    if abs(diff) > 1e-12:
        mask = (ht > 1e-8) & (ht < 1.0 - 1e-8)
        cnt = max(1, int(np.sum(mask)))
        ht[mask] += diff / cnt
        ht = np.clip(ht, 0.0, 1.0)
        diff2 = tgt - np.sum(ht)
        if abs(diff2) > 1e-14:
            ht += diff2 / best_n
            ht = np.clip(ht, 0.0, 1.0)
    c5f = _c5(ht)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return ht, c5f, best_n
