            strength = 0.25 + 0.06 * min(stall - 6, 12)
            h_spec = spectral_perturb(h_best, n, rng, strength)
            try:
                res = minimize(partial(_smooth_c5, n=n, beta=20.0), h_spec,
                              method='SLSQP', bounds=bounds, constraints=con,
                              options={'maxiter': 250, 'ftol': 1e-14})
                hc = repair(res.x, n)
                c = compute_c5(hc, n)
                if c < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc, c, True
                    stall = 0
            except Exception:
                pass

        # 2e: Binary quantization pressure (push mid-values to 0 or 1)
        if it % 4 == 2 and elapsed_ok(35):
            hp = h_best.copy()
            quant_improved = False
            for j in range(n):
                if 0.25 < hp[j] < 0.75:
                    for tv in (0.0, 1.0):
                        ht = hp.copy()
                        ht[j] = tv
                        ht = repair(ht, n)
                        c = compute_c5(ht, n)
                        if c < best_c5 - 1e-14:
                            hp = ht
                            best_c5 = c
                            quant_improved = True
                            improved = True
                            break
            if quant_improved:
                h_best = hp

        # 2f: Simulated annealing step (accept worse solutions occasionally)
        if it % 5 == 3 and elapsed_ok(30):
            temp = 0.002 * max(0.3, 1.0 - it / 80.0)
            h_cand = repair(h_best + rng.normal(0, 0.08, n), n)
            c_cand = compute_c5(h_cand, n)
            if c_cand < best_c5 or rng.random() < np.exp(-(c_cand - best_c5) / (temp * 0.01)):
                if c_cand < best_c5:
                    h_best, best_c5, improved = h_cand, c_cand, True
                else:
                    h_best = h_cand  # temporarily accept worse

        stall = 0 if improved else stall + 1

    # Phase 3: Final high-precision polishing
    if elapsed_ok(15):
        try:
            res = minimize(partial(_smooth_c5, n=n, beta=200.0), h_best,
                          method='SLSQP', bounds=bounds, constraints=con,
                          options={'maxiter': 800, 'ftol': 1e-16})
            hc = repair(res.x, n)
            c = compute_c5(hc, n)
            if c < best_c5:
                h_best, best_c5 = hc, c
        except Exception:
            pass

    # Phase 4: Exact objective polish
    if elapsed_ok(8):
        try:
            res = minimize(partial(_exact_c5, n=n), h_best,
                          method='SLSQP', bounds=bounds, constraints=con,
                          options={'maxiter': 600, 'ftol': 1e-16})
            hc = repair(res.x, n)
            c = compute_c5(hc, n)
            if c < best_c5:
                h_best, best_c5 = hc, c
        except Exception:
            pass

    # Phase 5: Ultra-fine local search
    if elapsed_ok(5):
        hc, c = local_pair_search(h_best, n, rng, max_rounds=50)
        if c < best_c5:
            h_best, best_c5 = hc, c

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
