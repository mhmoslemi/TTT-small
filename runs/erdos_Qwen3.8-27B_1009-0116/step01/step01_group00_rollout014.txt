        return np.full(n, 0.5)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

    hi_c5 = _c5(h, dx)
    t_end = time.time() + time_limit
    steps = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, -0.01, -0.02, -0.05, -0.1, -0.2, -0.3]
    for p in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            hi = h[i]
            for j in order:
                if j == i:
                    continue
                hj = h[j]
                for d in steps:
                    ni = hi + d
                    nj = hj - d
                    if ni < -1e-9 or ni > 1.0 + 1e-9:

                    if nj < -1e-9 or nj > 1.0 + 1e-9:

                        hi = ni
                        hj = nj
                        improved = True
                    else:
                        h[i] = hi
                        h[j] = hj
        if not improved:
