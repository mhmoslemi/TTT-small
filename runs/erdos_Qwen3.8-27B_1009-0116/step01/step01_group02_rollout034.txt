Now for the optimization algorithm:

**Projected Gradient Descent with Log-Sum-Exp:**
1. Start with h (repaired)
2. For beta in [5, 10, 20, 50, 100, 200, 500]:
   a. For iter in range(num_iters):
      - Compute gradient of f_beta
      - Take step: h_new = h - lr * grad
      - Project: h = repair(h_new, n)
      - Adaptive step size
   b. Check time budget
3. Periodic restarts
4. Final SLSQP polish

The key advantage over SLSQP: each gradient step is O(n^2) ≈ 5000 ops, so I can do thousands of steps quickly. SLSQP does line searches and function evaluations which are much more expensive.

Let me also add:
- **Coordinate descent** as a secondary method: for each coordinate j, find the optimal h[j] given all others fixed (it's a quadratic in h[j] for the max objective, or a rational function for log-sum-exp)
- **Block moves**: move mass between pairs of coordinates
- **Random restarts** with different structures (random, sawtooth, two-block, etc.)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-15:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def lse_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    total = np.sum(exps)
    weights = exps / total
    
    obj = (m + np.log(total)) / beta * dx
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for s in range(-(n-1), n):
            k = n - 1 + s
            if 0 <= j + s <= n - 1:
                g += weights[k] * (1.0 - h[j + s])
            if 0 <= j - s <= n - 1:
                g -= weights[k] * h[j - s]
        grad[j] = dx * g
    return obj, grad

def pgd_optimize(h, n, beta, max_iters, lr_init, t0, budget_s):
    """Projected gradient descent with log-sum-exp"""
    h = h.copy()
    lr = lr_init
    best_obj = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 30:
            break
        obj, grad = lse_obj_grad(h, n, beta)
        # Gradient step
        h_new = h - lr * grad
        h_new = repair(h_new, n)
        c5 = compute_c5(h_new, n)
        if c5 < best_obj:
            best_obj = c5
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_h, best_obj

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)
    
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def eq_con(x):
        return np.sum(x) - target_sum
    
    cons = [{'type': 'eq', 'fun': eq_con}]
    
    # Phase 1: Projected gradient with increasing beta
    betas = [5, 10, 20, 40, 80, 150, 300]
    for beta in betas:
        if time.time() - t0 > budget_s - 60:
            break
        h_best, c5 = pgd_optimize(h_best, n, beta, max_iters=500, lr_init=0.01, t0=t0, budget_s=budget_s)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h_best
        # Also try from perturbed
        h_pert = repair(h_best + rng.normal(0, 0.05, n), n)
        h_p, c5p = pgd_optimize(h_pert, n, beta, max_iters=300, lr_init=0.005, t0=t0, budget_s=budget_s)
        if c5p < best_c5:
            best_c5 = c5p
            h_best = h_p
    
    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s - 30:
        def obj_fn(x):
            return compute_c5(x, n)
        try:
            r = minimize(obj_fn, h_best, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-14})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5:
                h_best, best_c5 = hc, c
        except Exception:
            pass
    
    # Phase 3: Coordinate moves
    if time.time() - t0 < budget_s - 20:
        for _ in range(200):
            if time.time() - t0 > budget_s - 20:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for eps in (0.01, 0.02, 0.05, 0.1):
                hp = h_best.copy()
                hp[i] += eps
                hp[j] -= eps
                if 0 <= hp[i] <= 1 and 0 <= hp[j] <= 1:
                    c = compute_c5(hp, n)
                    if c < best_c5 - 1e-14:
                        h_best = hp
                        best_c5 = c
                        break
    
    # Phase 4: More SLSQP from slightly perturbed
    if time.time() - t0 < budget_s - 10:
        for sigma in (0.02, 0.05, 0.1):
            if time.time() - t0 > budget_s - 10:
                break
            try:
                hp = repair(h_best + rng.normal(0, sigma, n), n)
                r = minimize(obj_fn, hp, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 200, 'ftol': 1e-14})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5:
                    h_best, best_c5 = hc, c
            except Exception:
                pass
    
    # Final repair and evaluate
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-15:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    total = np.sum(exps)
    weights = exps / total
    obj = (m + np.log(total)) / beta * dx
    
    # Vectorized gradient via convolution
    conv1 = np.convolve(1.0 - h, weights, 'full')
    conv2 = np.convolve(h, weights[::-1], 'full')
    grad = dx * (conv1[n-1:2*n-1] - conv2[n-1:2*n-1])
    return obj, grad

def pgd_step(h, n, beta, lr, max_iters, t0, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = max(lr, 1e-12)
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        obj, grad = lse_obj_grad(h, n, beta)
        h_new = repair(h - lr * grad, n)
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            # Slowly increase learning rate
            lr *= 1.01
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
            h = h_new  # still move, just didn't improve
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)
    
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def eq_con(x):
        return np.sum(x) - target_sum
    cons = [{'type': 'eq', 'fun': eq_con}]
    
    def obj_fn(x):
        return compute_c5(x, n)
    
    # Phase 1: PGD with increasing beta
    betas = [3, 8, 15, 30, 60, 120, 250, 500]
    for beta in betas:
        deadline = min(t0 + budget_s - 80, time.time() + 100)
        h_best, c5 = pgd_step(h_best, n, beta, 0.01, 400, t0, deadline)
        if c5 < best_c5:
            best_c5 = c5
        if time.time() - t0 > budget_s - 100:
            break
    
    # Phase 2: Multiple PGD from perturbations
    for sigma in [0.03, 0.08, 0.15]:
        if time.time() - t0 > budget_s - 100:
            break
        deadline = min(t0 + budget_s - 60, time.time() + 50)
        hp = repair(h_best + rng.normal(0, sigma, n), n)
        for beta in [50, 150]:
            hp, c5p = pgd_step(hp, n, beta, 0.005, 300, t0, deadline)
            if c5p < best_c5:
                best_c5 = c5p
                h_best = hp
    
    # Phase 3: SLSQP on the exact max
    if time.time() - t0 < budget_s - 40:
        try:
            r = minimize(obj_fn, h_best, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 400, 'ftol': 1e-15})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5:
                h_best, best_c5 = hc, c
        except Exception:
            pass
    
    # Phase 4: Coordinate moves
    if time.time() - t0 < budget_s - 30:
        for _ in range(500):
            if time.time() - t0 > budget_s - 30:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for eps in (0.005, 0.01, 0.02, 0.05, 0.1):
                hp = h_best.copy()
                hp[i] += eps
                hp[j] -= eps
                if 0.0 <= hp[i] <= 1.0 and 0.0 <= hp[j] <= 1.0:
                    c = compute_c5(hp, n)
                    if c < best_c5 - 1e-14:
                        h_best = hp
                        best_c5 = c
                        break
    
    # Phase 5: Final SLSQP from multiple perturbations
    if time.time() - t0 < budget_s - 15:
        for sigma in [0.01, 0.03, 0.06]:
            if time.time() - t0 > budget_s - 15:
                break
            try:
                hp = repair(h_best + rng.normal(0, sigma, n), n)
                r = minimize(obj_fn, hp, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 300, 'ftol': 1e-15})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5:
                    h_best, best_c5 = hc, c
            except Exception:
                pass
    
    # Final: one more PGD at very high beta
    if time.time() - t0 < budget_s - 5:
        h_best, c5 = pgd_step(h_best, n, 1000, 0.001, 200, t0, t0 + budget_s - 2)
        if c5 < best_c5:
            best_c5 = c5
    
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
```
