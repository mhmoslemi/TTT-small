For the "full" correlation, we need to be careful about the range. The full correlation of h (length n) with (1-h) (length n) has 2n-1 elements. For lag k (where k ranges from -(n-1) to (n-1)):

- If k < 0: count of i from 0 to n-1+k where h[i+k]=1 and h[i]=0 = count of j from |k| to n-1 where h[j]=1 and h[j-|k|]=0

So for |k| = d:
- Positive side: count of i in [0, n-1-d] where h[i]=1, h[i+d]=0
- Negative side: count of j in [d, n-1] where h[j]=1, h[j-d]=0

The max over all k includes both sides.

For a binary sequence, let's denote the positions of 1s as S = {s_1, s_2, ..., s_{n/2}}.
Then count_10(d) for positive lag d = |{s ∈ S : s+d ∉ S and s+d < n}|
And for negative lag d = |{s ∈ S : s-d ∉ S and s-d ≥ 0}|

We want to minimize the max of these over all d.

This is a hard combinatorial problem, but for n ≤ 120 or so, we can try greedy/randomized approaches.

        if np.sum(free) > 0:
            h_cd[free] += diff / np.sum(free)
        h_cd = np.clip(h_cd, 0, 1)
        h_cd2, c_cd = _coord_descent(h_cd, best_n / 2.0, 100, seed + 2, t0 + rem() - 30, t0, best_c5)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd2.copy()
            print(f"CD n={best_n}: C5={c_cd:.6f}")

    # Phase 4: Try other n values
    for n_try in [60, 80, 90, 100, 120, 140]:
        if rem() < 35:
            break
        tgt_t = n_try / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = np.clip(h_interp, 0, 1)
        target_t_ones = n_try // 2
        h_bi = _make_binary(h_interp, target_t_ones)
        c_bi = _c5(h_bi)
        # Quick SA
        h_si, c_si, _ = _sa_binary(h_bi, target_t_ones, 100000, 0.0015, 0.00008,
                                     seed + n_try, t0 + rem() - 20, t0)
        c_best_local = min(c_bi, c_si)
        h_best_local = h_bi.copy() if c_bi < c_si else h_si.copy()
        if c_best_local < best_c5:
            best_c5 = c_best_local
            best_h = h_best_local.copy()
            best_n = n_try
            print(f"n={n_try}: C5={c_best_local:.6f}")
        # Try continuous refinement for this n
        if rem() > 25 and best_n == n_try:
            h_c = best_h.astype(float) + rng.uniform(-0.02, 0.02, n_try)
            h_c = np.clip(h_c, 0, 1)
            d2 = n_try / 2.0 - np.sum(h_c)
            fr = (h_c > 1e-6) & (h_c < 1.0 - 1e-6)
            if np.sum(fr) > 0:
                h_c[fr] += d2 / np.sum(fr)
            h_c = np.clip(h_c, 0, 1)
            h_c3, c_c = _coord_descent(h_c, n_try / 2.0, 50, seed + n_try + 100,
                                        t0 + rem() - 15, t0, best_c5)
            if c_c < best_c5:
                best_c5 = c_c
                best_h = h_c3.copy()
                print(f"CD n={n_try}: C5={c_c:.6f}")

    # Phase 5: Final greedy pass
    if rem() > 10:
        n_f = best_n
        tgt_f = int(n_f / 2)
        h_f = best_h.copy()
        if np.sum((h_f > 0.01) & (h_f < 0.99)) > 5:
            h_f = _make_binary(h_f, tgt_f)
        c_f = _c5(h_f)
        if c_f >= best_c5:
            h_f = best_h.astype(float)
            c_f = best_c5
        # Greedy: try all single-coordinate changes
        for _ in range(5):
            improved = False
            for i in range(n_f):
                for nv in [0.0, 1.0]:
                    d = nv - h_f[i]
                    if abs(d) < 1e-12:
                        continue
                    if d > 0:
                        cands = np.where(h_f < 1.0 - d - 1e-11)[0]
                    else:
                        cands = np.where(h_f > -d + 1e-11)[0]
                    if len(cands) == 0:
                        continue
                    for j in cands[:5]:
                        h2 = h_f.copy()
                        h2[i] = nv
                        h2[j] = h_f[j] - d
                        h2[j] = np.clip(h2[j], 0, 1)
                        c = _c5(h2)
                        if c < c_f - 1e-13:
                            h_f = h2
                            c_f = c
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    # Final: fix sum exactly
    tgt_final = best_n / 2.0
    diff = tgt_final - np.sum(best_h)
    if abs(diff) > 1e-13:
        free = (best_h > 1e-9) & (best_h < 1.0 - 1e-9)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
        best_h = np.clip(best_h, 0, 1)
        diff2 = tgt_final - np.sum(best_h)
        if abs(diff2) > 1e-13:
            best_h -= diff2 / best_n
            best_h = np.clip(best_h, 0, 1)

    final_c5 = _c5(best_h)
    print(f"FINAL: n={best_n}, C5={final_c5:.6f}")
    return best_h, final_c5, best_n
