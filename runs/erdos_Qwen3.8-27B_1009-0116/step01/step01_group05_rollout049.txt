                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': cf, 'jac': cjf}],
                 options={'maxiter': maxit, 'ftol': 1e-18})
    return _proj(r.x, target)

def _sa_binary(h, dx, n, target, rng, t_max, niter=80000):
    b = (h > 0.5).astype(float)
    diff = int(b.sum()) - int(target)
    if diff > 0:
        ones = np.where(b == 1)[0]
        for i in rng.choice(ones, diff, replace=False):
            b[i] = 0
    elif diff < 0:
        zeros = np.where(b == 0)[0]
        for i in rng.choice(zeros, -diff, replace=False):
            b[i] = 1
    best = b.copy()
    bc = _c5(b, dx)
    cc = bc
    T = 0.04
    for it in range(niter):
        if time.time() > t_max:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        b[i1], b[i0] = 0.0, 1.0
        cn = _c5(b, dx)
        d = cn - cc
        T *= 0.99985
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-8)):
            cc = cn
            if cn < bc:
                bc = cn
                best = b.copy()
        else:
            b[i1], b[i0] = 1.0, 0.0
    return best, bc

def _targeted_flip(h, dx, n, target, rng, t_max, niter=30000):
    b = (h > 0.5).astype(float)
    diff = int(b.sum()) - int(target)
    if diff > 0:
        ones = np.where(b == 1)[0]
        for i in rng.choice(ones, diff, replace=False):
            b[i] = 0
    elif diff < 0:
        zeros = np.where(b == 0)[0]
        for i in rng.choice(zeros, -diff, replace=False):
            b[i] = 1
    best = b.copy()
    bc = _c5(b, dx)
    for it in range(niter):
        if time.time() > t_max:
            break
        cv = _corr(b, dx)
        worst = np.argmax(cv)
        s = worst - (n - 1)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        b[i1], b[i0] = 0.0, 1.0
        cn = _c5(b, dx)
        if cn < bc:
            bc = cn
            best = b.copy()
        else:
            b[i1], b[i0] = 1.0, 0.0
    return best, bc

def _optimize(h0, dx, n, target, rng, t_end):
    bh = h0.copy()
    bc = _c5(h0, dx)
    h = h0.copy()
    eps_sched = [0.1, 0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]
    for eps in eps_sched:
        if time.time() > t_end - 50:
            break
        try:
            hn = _slsqp(h, dx, n, target, eps, 500)
        except Exception:
            break
        c = _c5(hn, dx)
        if c < bc:
            bc, bh = c, hn.copy()
        h = hn
    if time.time() < t_end - 40:
        tm = time.time() + min(150.0, t_end - time.time() - 30)
        bn, bcn = _sa_binary(bh, dx, n, target, rng, tm)
        if bcn < bc:
            bc, bh = bcn, bn.copy()
    if time.time() < t_end - 25:
        tm2 = time.time() + min(60.0, t_end - time.time() - 20)
        bt, bct = _targeted_flip(bh, dx, n, target, rng, tm2)
        if bct < bc:
            bc, bh = bct, bt.copy()
    if time.time() < t_end - 3:
        for eps in [0.0002, 0.00005]:
            if time.time() > t_end - 2:
                break
            try:
                h3 = _slsqp(bh, dx, n, target, eps, 400)
            except Exception:
                break
            c3 = _c5(h3, dx)
            if c3 < bc:
                bc, bh = c3, h3.copy()
    return bh, bc

def _make_random_binary(n, rng):
    h = rng.uniform(0, 1, n)
    return _proj(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

            except Exception:

            if cf < bco:
                bco, bho = cf, hf
    return bho, bco, len(bho)
