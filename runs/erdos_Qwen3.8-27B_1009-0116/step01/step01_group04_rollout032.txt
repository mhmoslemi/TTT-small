def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n, target):
    """Project x onto {0 <= x_i <= 1, sum(x) = target}"""
    # Bisection for lambda: sum(clip(x + lambda, 0, 1)) = target
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    for _ in range(80):

        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj(x, dx, tau):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    if tau < 1e-10:
        return m * dx
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    logsum = np.log(np.sum(np.exp(sh)))
    return (m + tau * logsum) * dx

def grad_softmax(x, dx, tau):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    ew = np.exp(sh)
    w = ew / np.sum(ew)  # softmax weights, length 2n-1
    
    # Compute gradient: grad[j] = dx * sum_k w[k] * ((1-x[j-d_k])*I - x[j+d_k])*I)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    n = len(x)
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def softmax_obj_grad(x, dx, tau):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    ew = np.exp(sh)
    ew_sum = ew.sum()
    ws = ew / ew_sum
    obj = (m + tau * np.log(ew_sum)) * dx
    d = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        i1 = j - d
        v1 = (i1 >= 0) & (i1 < n)
        if v1.any():
            grad[j] += np.dot(ws[v1], 1.0 - x[i1[v1]])
        i2 = j + d
        v2 = (i2 >= 0) & (i2 < n)
        if v2.any():
            grad[j] -= np.dot(ws[v2], x[i2[v2]])
    return obj, grad * dx

    best = x.copy()
    f_best = compute_c5(best, dx)
    f_cur, g = softmax_obj_grad(x, dx, tau)
    step = 1.0
    t_val = 1.0
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        t_new = 1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val) / 2.0
        mu = (t_val - 1.0) / t_new
        y = x + mu * (x - x_prev)
        fy, gy = softmax_obj_grad(y, dx, tau)
        ls = step
        y_trial = project_feasible(y - ls * gy, target)
        ft, _ = softmax_obj_grad(y_trial, dx, tau)
        c1 = 1e-4
        gg = np.dot(gy, gy)
        if gg < 1e-30:
            break
        trail = 0
        while ft > fy - c1 * ls * gg and ls > 1e-10 and trail < 30:
            ls *= 0.5
            y_trial = project_feasible(y - ls * gy, target)
            ft, _ = softmax_obj_grad(y_trial, dx, tau)
            trail += 1
        x_prev = x
        x = y_trial
        step = min(1.0, ls * 2.0)
        f_cur = ft
        t_val = t_new
        if it % 300 == 0:
            fv = compute_c5(x, dx)
            if fv < f_best:
                f_best = fv
                best = x.copy()
    fv = compute_c5(x, dx)
    if fv < f_best:
        f_best = fv
        best = x.copy()
    return best, f_best

def slsqp_polish(x0, dx, target, tau, max_iter=150):
    n = len(x0)
    def ofun(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(ofun, x0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-13})
        return project_feasible(r.x, target)
    except Exception:
        return project_feasible(x0.copy(), target)
