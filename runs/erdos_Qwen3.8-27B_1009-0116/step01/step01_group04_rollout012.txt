import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    # Iterative fix for sum after clipping
    for _ in range(10):
        s = np.sum(x)
        if abs(s - tgt) < 1e-10:
            break
        if s > tgt:
            # Reduce elements that are > 0
            mask = x > 0
            excess = s - tgt
            reducible = np.sum(x[mask])
            if reducible > 0:
                x[mask] -= excess * x[mask] / reducible
                x = np.clip(x, 0.0, 1.0)
        else:
            # Increase elements that are < 1
            mask = x < 1
            deficit = tgt - s
            increasable = np.sum(1 - x[mask])
            if increasable > 0:
                x[mask] += deficit * (1 - x[mask]) / increasable
                x = np.clip(x, 0.0, 1.0)
    return x

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    from scipy.interpolate import interp1d
    xp = interp1d(xs, x_src, kind='linear', bounds_error=False, fill_value=(x_src[0], x_src[-1]))
    x_new = xp(xd)
    return repair(x_new, n_dst)

def slsqp_optimize(x0, n, dx, tau, maxiter=500, ftol=1e-14):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    def cjac(x):
        return np.ones(n)
    
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': ftol})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def coordinate_descent(x, n, dx, n_iters, rng, step_base=0.1):
    x = x.copy()
    best_val = compute_c5(x, dx)
    
    for it in range(n_iters):
        corr = np.correlate(x, 1.0 - x, mode="full")
        curr_val = np.max(corr) * dx
        
        if curr_val < best_val:
            best_val = curr_val
        
        # Identify active lags (top few)
        top_k = np.argsort(corr)[-5:]
        weights = np.exp((corr[top_k] - corr[top_k].max()) / 0.05)
        weights /= weights.sum()
        
        # Compute blended gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            # Lag k in 'full' mode: offset is (2*n-2) - k for the correlation
            # Actually, in np.correlate(a, v, 'full'), the result has length 2*n-1
            # Index k corresponds to lag (k - (n-1))
            lag = k - (n - 1)
            # ∂c/∂h[j] = 1 - h[j+lag] - h[j-lag] (with bounds checking)
            for j in range(n):
                g = 1.0
                if 0 <= j + lag < n:
                    g -= x[j + lag]
                if 0 <= j - lag < n:
                    g -= x[j - lag]
                grad[j] += w * g
        
        # Step
        step = step_base / (1.0 + 0.01 * it)
        x_new = x - step * grad
        x_new = repair(x_new, n)
        
        new_val = compute_c5(x_new, dx)
        if new_val < curr_val:
            x = x_new
        else:
            step_base *= 0.9
    
    return x, best_val

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(x)
        d = tgt - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = x < 1.0
            cap = np.sum(1.0 - x[mask])
            if cap > 1e-12:
                x[mask] += d * (1.0 - x[mask]) / cap
            else:
                break
        else:
            mask = x > 0.0
            floor = np.sum(x[mask])
            if floor > 1e-12:
                x[mask] += d * x[mask] / floor
            else:
                break
        x = np.clip(x, 0.0, 1.0)
    return x

def grad_for_lag(x, n, lag):
    """Gradient of c_lag(h) w.r.t. h, where c_lag = sum_i h[i+lag]*(1-h[i])"""
    grad = np.zeros(n)
    L = abs(lag)
    if L == 0:
        return 1.0 - 2.0 * x
    if lag > 0:
        # c = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
        # d/dh[j]: (1-h[j-L])*1[j>=L] - h[j+L]*1[j<n-L]
        grad[L:] += (1.0 - x[:n-L])
        grad[:n-L] -= x[L:]
    else:
        # c = sum_{i=L}^{n-1} h[i-L]*(1-h[i])
        # d/dh[j]: (1-h[j+L])*1[j<n-L] - h[j-L]*1[j>=L]
        grad[:n-L] += (1.0 - x[L:])
        grad[L:] -= x[:n-L]
    return grad

def slsqp_step(x0, n, dx, tau, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': cfun, 'jac': np.ones(n)}]
    
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x, n)
    except:
        return x0.copy()

def proj_grad_descend(x, n, dx, max_iters, rng, top_m=3, temp=0.03):
    """Projected gradient descent with exact max (softmax-weighted)"""
    x = x.copy()
    best_val = compute_c5(x, dx)
    alpha = 0.5
    n_corr = 2 * n - 1
    
    for it in range(max_iters):
        corr = np.correlate(x, 1.0 - x, mode="full")
        curr = np.max(corr)
        
        # Softmax weights over top-m lags
        idx = np.argsort(corr)[-top_m:]
        w = np.exp((corr[idx] - curr) / temp)
        w /= w.sum()
        
        # Blended gradient
        grad = np.zeros(n)
        for wi, ii in zip(w, idx):
            lag = ii - (n - 1)
            grad += wi * grad_for_lag(x, n, lag)
        
        # Step
        x_new = repair(x - alpha * grad, n)
        v_new = compute_c5(x_new, dx)
        
        if v_new < curr * dx:
            x = x_new
            if v_new < best_val:
                best_val = v_new
            alpha = min(alpha * 1.05, 2.0)
        else:
            alpha *= 0.7
        
        if alpha < 1e-8:
            break
    
    return x, best_val

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    f = interp1d(xs, x_src, kind='linear', bounds_error=False,
                 fill_value=(x_src[0], x_src[-1]))
    return repair(f(xd), n_dst)

    t0 = time.time()

    # Get initial

        iters = 300 if n <= 200 else 150

        if v < best_val:

            best_h = x.copy()
    
    # Restart with perturbation
    if elapsed() < p1_end - 10:
        x2 = repair(h0 + 0.15 * rng.standard_normal(n), n)
        for tau in [0.05, 0.02, 0.01, 0.005]:
            if elapsed() > p1_end:
                break
            x2 = slsqp_step(x2, n, dx, tau, 200)
            v = compute_c5(x2, dx)
            if v < best_val:
                best_val = v
                best_h = x2.copy()
    
    # Phase 2: Project to finer grid (25% time)
    p2_end = min(t0 + budget_s * 0.65, p1_end + budget_s * 0.25)
    if elapsed() < p2_end - 5:
        n2 = 2 * n + 1
        dx2 = 2.0 / n2
        x_p = project_grid(best_h, n, n2)
        
        for tau in [0.05, 0.02, 0.01, 0.005]:
            if elapsed() > p2_end:
                break
            x_p = slsqp_step(x_p, n2, dx2, tau, 200)
            v = compute_c5(x_p, dx2)
            if v < best_val:
                best_val = v
                best_h = x_p.copy()
                n = n2
                dx = dx2
    
    # Phase 3: Projected gradient descent (25% time)
    p3_end = min(t0 + budget_s * 0.90, p2_end + budget_s * 0.25)
    if elapsed() < p3_end - 3:
        n_pgd = len(best_h)
        dx_pgd = 2.0 / n_pgd
        max_iters = 50000
        x_pgd, v_pgd = proj_grad_descend(best_h, n_pgd, dx_pgd, max_iters, rng)
        if v_pgd < best_val:
            best_val = v_pgd
            best_h = x_pgd
            n = n_pgd
            dx = dx_pgd
    
    # Phase 4: Final SLSQP polish (10% time)

            best_h = x_f
    
    # Final
    best_h = repair(best_h, len(best_h))
    final_val = compute_c5(best_h, 2.0 / len(best_h))
    
    return (best_h, final_val, len(best_h))

    # Here tau = 1/beta, so smaller tau = sharper max approximation
    # Objective: tau * log(sum(exp(corr/tau))) * dx ≈ max(corr) * dx
    # Gradient: sum_k w_k * grad_k * dx where w_k = softmax(corr/tau)
    
    def obj(x):

        m = np.max(corr)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s > 1e-14:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    for _ in range(30):
        s = np.sum(x)
        d = tgt - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = x < 1.0
            cap = np.sum(1.0 - x[m])
            if cap > 1e-14:
                x[m] += d * (1.0 - x[m]) / cap
            else:
                break
        else:
            m = x > 0.0
            fl = np.sum(x[m])
            if fl > 1e-14:
                x[m] += d * x[m] / fl
            else:
                break
        x = np.clip(x, 0.0, 1.0)
    return x

def grad_for_lag(x, n, lag):
    grad = np.zeros(n)
    L = abs(lag)
    if L == 0:
        return 1.0 - 2.0 * x
    if lag > 0:
        grad[L:] += (1.0 - x[:n - L])
        grad[:n - L] -= x[L:]
    else:
        grad[:n - L] += (1.0 - x[L:])
        grad[L:] -= x[:n - L]
    return grad

def slsqp_opt(x0, n, dx, tau, maxiter=400):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

    def grad_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        w = np.exp(sh)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            g += w[k] * grad_for_lag(x, n, lag)
        return g * dx

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bd, jac=grad_fn,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def pgd_refine(x, n, dx, max_iters, rng, top_m=4, temp=0.02):
    x = x.copy()
    best_val = compute_c5(x, dx)
    alpha = 0.3
    stall = 0
    for it in range(max_iters):
        corr = np.correlate(x, 1.0 - x, mode="full")
        curr = np.max(corr) * dx
        if curr < best_val:
            best_val = curr
            stall = 0
        else:
            stall += 1
        if stall > 2000 and alpha < 0.01:
            break
        idx = np.argsort(corr)[-top_m:]
        w = np.exp((corr[idx] - np.max(corr)) / temp)
        w /= w.sum()
        g = np.zeros(n)
        for wi, ii in zip(w, idx):
            lag = ii - (n - 1)
            g += wi * grad_for_lag(x, n, lag)
        x_new = repair(x - alpha * g, n)
        v_new = compute_c5(x_new, dx)
        if v_new < curr:
            x = x_new
            alpha = min(alpha * 1.03, 1.5)
        else:
            alpha *= 0.8
            if alpha < 1e-7:
                break
    return x, best_val

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    f = interp1d(xs, x_src, kind='linear', bounds_error=False,
                 fill_value=(x_src[0], x_src[-1]))
    return repair(f(xd), n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(191, 0.5)

    n = len(h0)
    dx = 2.0 / n
    h0 = repair(h0, n)
    best_h = h0.copy()
    best_val = compute_c5(h0, dx)
    best_n = n

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with analytical grad on base grid (35%)
    p1_end = t0 + budget_s * 0.35
    taus = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    x = h0.copy()
    for tau in taus:
        if elapsed() > p1_end:
            break
        n_iter = 400 if n <= 200 else 200
        x = slsqp_opt(x, n, dx, tau, n_iter)
        v = compute_c5(x, dx)
        if v < best_val:
            best_val = v
            best_h = x.copy()
            best_n = n

    # Phase 2: Perturbed restart (20%)
    p2_end = t0 + budget_s * 0.55
    if elapsed() < p2_end - 10:
        x2 = repair(h0 + 0.12 * rng.standard_normal(n), n)
        for tau in [0.04, 0.02, 0.01, 0.005, 0.002]:
            if elapsed() > p2_end:
                break
            x2 = slsqp_opt(x2, n, dx, tau, 300)
            v = compute_c5(x2, dx)
            if v < best_val:
                best_val = v
                best_h = x2.copy()
                best_n = n

    # Phase 3: Finer grid (25%)
    p3_end = t0 + budget_s * 0.80
    if elapsed() < p3_end - 10:
        n2 = 2 * best_n + 1
        dx2 = 2.0 / n2
        x3 = project_grid(best_h, best_n, n2)
        for tau in [0.03, 0.015, 0.008, 0.004, 0.002]:
            if elapsed() > p3_end:
                break
            n_it = 250 if n2 <= 400 else 120
            x3 = slsqp_opt(x3, n2, dx2, tau, n_it)
            v = compute_c5(x3, dx2)
            if v < best_val:
                best_val = v
                best_h = x3.copy()
                best_n = n2

    # Phase 4: PGD refinement on best grid (20%)
    remaining = t0 + budget_s - elapsed() - 8
    if remaining > 15:
        nf = len(best_h)
        dxf = 2.0 / nf
        max_it = int(remaining * 8000)
        x4, v4 = pgd_refine(best_h, nf, dxf, max_it, rng)
        if v4 < best_val:
            best_val = v4
            best_h = x4.copy()
            best_n = nf

    # Phase 5: Final SLSQP polish if time remains
    remaining2 = t0 + budget_s - elapsed() - 3
    if remaining2 > 20:
        nf = len(best_h)
        dxf = 2.0 / nf
        x5 = slsqp_opt(best_h, nf, dxf, 0.001, 200)
        v5 = compute_c5(x5, dxf)
        if v5 < best_val:
            best_val = v5
            best_h = x5.copy()
            best_n = nf

    best_h = repair(best_h, len(best_h))
    final_val = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, final_val, len(best_h))
