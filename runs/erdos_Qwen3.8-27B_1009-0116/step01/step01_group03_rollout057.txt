def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to increase sum; increase where h < 1
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    step = diff / len(idx)
    h[idx] = np.clip(h[idx] + step, 0, 1)
    # fix residual
    residual = ts - h.sum()
    if abs(residual) > 1e-10:
        # distribute to first few
        for i in idx:
            adj = min(residual, 1.0 - h[i]) if residual > 0 else max(residual, -h[i])
            h[i] += adj
            residual -= adj
            if abs(residual) < 1e-12:
                break
    return h

    # corr[k] = sum over valid (i,j) pairs: h[j] * (1-h[i]) where j - i = k - (n-1)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    step = diff / len(idx)
    h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
    residual = ts - h.sum()
    if abs(residual) > 1e-10:
        for i in idx:
            if residual > 0:
                adj = min(residual, 1.0 - h[i])
            else:
                adj = max(residual, -h[i])
            h[i] += adj
            residual -= adj
            if abs(residual) < 1e-12:

    return h

def slsqp_smooth(h, n, ts, beta, maxiter=500):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        s = (c - m) * beta
        return (m + np.log(np.sum(np.exp(s))) / beta) * (2.0 / n)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

import numpy as np
from scipy.optimize import minimize
import time

_TS = 70.0

def _constraint(x):
    return x.sum() - _TS

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    step = diff / len(idx)
    h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
    residual = ts - h.sum()
    if abs(residual) > 1e-10:
        for i in idx:
            if residual > 0:
                adj = min(residual, 1.0 - h[i])
            else:
                adj = max(residual, -h[i])
            h[i] += adj
            residual -= adj
            if abs(residual) < 1e-12:
                break
    return h

def slsqp_smooth(h, n, ts, beta, maxiter=500):
    global _TS
    _TS = ts
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        s = (c - m) * beta
        return (m + np.log(np.sum(np.exp(s))) / beta) * (2.0 / n)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _constraint}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def subgradient_optimize(h, ts, max_iter=15000, t_limit=120):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step_size = 0.04
    no_improve = 0
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        corr = compute_corr(best_h)
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            i_v = np.arange(0, n - shift)
            j_v = i_v + shift
            grad[j_v] += (1.0 - best_h[i_v])
            grad[i_v] += (-best_h[j_v])
        else:
            i_v = np.arange(-shift, n)
            j_v = i_v + shift
            grad[j_v] += (1.0 - best_h[i_v])
            grad[i_v] += (-best_h[j_v])
        h_new = best_h - step_size * grad
        h_new = project_feasible(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 40:
                step_size *= 0.8
                no_improve = 0
                if step_size < 0.0003:
                    break
    return best_h, best_c5

def continuous_local_search(h, ts, rng, max_iter=5000, t_limit=80):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = 0.2

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
            step = max(0.003, step * 0.997)
        else:
            step = max(0.003, step * 0.98)

                h_b = h_t
                improved = True
            else:
                h_b = h_t
        if not improved:
