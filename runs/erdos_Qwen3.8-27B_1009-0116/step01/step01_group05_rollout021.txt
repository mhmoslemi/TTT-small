            h_bin = (best_h > 0.5).astype(float)
            h_bin[rng.choice(n, n // 7, replace=False)] = 0.5
            h_pert = _proj(h_bin + rng.normal(0, 0.3, n), target)
        c_pert = _c5(h_pert, dx)
        if c_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c_pert
        remaining = t2 - time.time()
        if remaining > 4:
            sub_end = min(time.time() + min(18.0, remaining * 0.85), t2)
            h_opt, c_opt = _optimize(h_pert, dx, n, target, rng, sub_end)
            if c_opt < best_c5:
                best_h = h_opt
                best_c5 = c_opt
    # Phase 3: Intensive polish (15% time)
    t3 = deadline
    remaining = t3 - time.time()
    if remaining > 20:
        h_ann, c_ann = _anneal(best_h, dx, n, rng, time.time() + remaining * 0.5)
        if c_ann < best_c5:
            best_h, best_c5 = h_ann, c_ann
    if time.time() < t3 - 10:
        h_fin, c_fin = _optimize(best_h, dx, n, target, rng, t3)
        if c_fin < best_c5:
            best_h, best_c5 = h_fin, c_fin
    return best_h, best_c5, n
