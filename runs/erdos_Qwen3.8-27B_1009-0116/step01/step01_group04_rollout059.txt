    sh = np.clip((corr - m) / tau, -50.0, 0.0)

    _G_TGT = n / 2.0

                     bounds=bd, constraints=cs,
                     options={'maxiter': maxiter, 'ftol': ftol})

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    if n_src == n_dst:
        return x_src.copy()
    f = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')
    return repair(f(xd), n_dst)

def perturb_gauss(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def perturb_swap(x, n, n_swaps, rng):
    x = x.copy()
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, 2)
        if i != j:
            x[i], x[j] = x[j], x[i]
    return repair(x, n)

def temperature_schedule(start, end, n_steps):
    return list(np.geomspace(start, end, n_steps))
