def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):

    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:

                    if (diff2 > 0 and h[i] < 1.0) or (diff2 < 0 and h[i] > 0.0):

        w[int(np.argmax(c))] = 1.0
    else:
        w /= wsum

    return val, w
