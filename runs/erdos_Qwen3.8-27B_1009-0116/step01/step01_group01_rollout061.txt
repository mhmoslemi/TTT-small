def proj_sum_box(y, n):

def smooth_obj_grad(h, n, t):
    dx = 2.0 / n

    return fval, g * dx

def linsolve_bfgs(h, n, t, maxiter):

    x0 = h.copy()
    res = minimize(lambda x: smooth_obj_grad(x, n, t)[0],
                   x0,
                   jac=lambda x: smooth_obj_grad(x, n, t)[1],
                   method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair(res.x, n)

    elif n_ones < target:
        zeros_idx = np.where(b == 0)[0]
        for i in range(target - n_ones):
            b[zeros_idx[i]] = 1

    ones = set(np.where(b == 1)[0])
    zeros = set(np.where(b == 0)[0])

                else:

def compute_flip_effects(b, n):
    """Compute flip_effect[i][m] for all i, m.
    flip_effect[i][m] = change in corr[m] when b[i] goes 0->1."""
    n_corr = 2*n - 1
    flip_effects = np.zeros((n, n_corr))
    k_lags = np.arange(n_corr) - (n-1)

                val += -b[i+k]

def repair(h, n):

def smooth_grad(x, n, t):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    w = np.exp((c - cmax) * t)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w[np.argmax(w)] = 1.0
        return np.zeros(n)
    w /= w_sum
    g = np.zeros(n)
    active_m = np.where(w > 1e-14)[0]
    for m in active_m:
        k = m - (n - 1)
        if k >= 0:
            g[k:] += w[m] * (1.0 - x[:n - k])
            g[:n - k] += w[m] * (-x[k:])
        else:
            k2 = -k
            g[:n - k2] += w[m] * (1.0 - x[k2:])
            g[k2:] += w[m] * (-x[:n - k2])
