from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    return np.clip(x, 0.0, 1.0)

def repair_sum(x, tgt_sum):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if s > 1e-12:
        x = x * (tgt_sum / s)
    return np.clip(x, 0.0, 1.0)

def project_periodic(h_full, n_full):
    n_half = n_full // 2
    h_half = 0.5 * (h_full[:n_half] + h_full[n_half:])
    return repair_sum(h_half, n_half / 2.0)

def make_full(h_half, n_half):
    return np.concatenate([h_half, h_half])

def smooth_max(vals, tau):
    m = np.max(vals)
    sh = np.clip((vals - m) / tau, -50.0, 0.0)
    return m + tau * np.log(np.sum(np.exp(sh)))

def _per_obj(x, n_half, tau):
    n_full = 2 * n_half
    dx = 2.0 / n_full
    h = np.concatenate([x, x])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr * dx, tau)

def _per_cjac(x, n_half):

    except Exception:
        return x0
