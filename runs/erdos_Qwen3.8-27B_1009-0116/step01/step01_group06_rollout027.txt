import numpy as np
from scipy.optimize import minimize
import time

_Ng = 70
_bx = (0.0, 1.0)

def _sf(x):
    dx = 2.0 / _Ng
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(c) * dx)

def _cf(x):
    return float(x.sum() - _Ng * 0.5)

def _cj(x):
    return np.ones(_Ng)

def eval_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def proj(h, n):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    tgt = n * 0.5
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-12:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += d * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = tgt - h.sum()
        if abs(d2) > 1e-12:
            i = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[i] = float(np.clip(h[i] + d2, 0.0, 1.0))
    return h

def smgrad(h, dx, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    ec = np.exp(alpha * (c - mx))
    s = ec.sum()
    w = ec / s
    obj = (mx + np.log(s)) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        wd = wk * dx
        a, b = max(0, -off), min(n, n - off)
        if a < b:
            g[a:b] += wd * (1.0 - h[a + off:b + off])
        a2, b2 = max(0, off), min(n, n + off)
        if a2 < b2:
            g[a2:b2] -= wd * h[a2 - off:b2 - off]
    return obj, g

def nesterov(h, n, dx, al0, al1, niter, lr, mu, tlim):
    t0 = time.time()
    h = proj(h, n).copy()
    hp = h.copy()
    bh = h.copy()
    bc = eval_c5(h, dx)
    best_it = -200
    for it in range(niter):
        if time.time() - t0 > tlim:
            break
        frac = it / max(1, niter - 1)
        al = al0 * (al1 / al0) ** frac
        mu_cur = mu - 0.15 * frac
        y = np.clip(h + mu_cur * (h - hp), 0.0, 1.0)
        _, g = smgrad(y, dx, al)
        ng = np.linalg.norm(g)
        step = lr / max(ng, 1e-10)
        step = min(step, 2.0)
        hp = h.copy()
        h = proj(y - step * g, n)
        if (it + 1) % 300 == 0:
            c5 = eval_c5(h, dx)
            if c5 < bc:
                bc = c5
                bh = h.copy()
                best_it = it
            elif it - best_it > 8000:
                hp = h.copy()
                best_it = it
    c5 = eval_c5(h, dx)
    if c5 < bc:
        bc = c5
        bh = h.copy()
    return bh, bc

def slsqp_polish(h, n, dx, mit, ftol, tlim):
    global _Ng
    _Ng = n
    bnds = [_bx] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_sf, proj(h, n), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': mit, 'ftol': ftol})
        ho = proj(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        hp = proj(h, n)
        return hp, eval_c5(hp, dx)

def interp_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def hard_subgrad_step(h, n, dx, lr):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    act = np.where(np.abs(c - mx) < 1e-6)[0]
    g = np.zeros(n)
    for k in act:
        off = int(k) - (n - 1)
        a, b = max(0, -off), min(n, n - off)
        if a < b:
            g[a:b] += (1.0 - h[a + off:b + off])
        a2, b2 = max(0, off), min(n, n + off)
        if a2 < b2:
            g[a2:b2] -= h[a2 - off:b2 - off]
    g /= max(1, len(act))
    return proj(h - lr * dx * g, n)
