```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # p-norm approximation to max
```

```python
def adam_optimize(h, n, t=300.0, lam=1000.0, steps=2000, lr=0.01):
    h = h.copy()
    target = n * 0.5
    dx = 2.0 / n
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for step in range(steps):
        # Compute objective gradient
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(c)
        e = np.exp((c - cmax) * t)
        w = e / np.sum(e)
        
        s = h.sum() - target
        grad = 2.0 * lam * s * np.ones(n)
        
        # Correlation gradient
        for k in range(2*n - 1):
            if w[k] < 1e-14:
                continue
            m_shift = k - n + 1
            gk = np.zeros(n)
            lo = max(0, -m_shift)
            hi = min(n, n - m_shift)
            if hi > lo:
                gk[lo:hi] += (1.0 - h[lo+m_shift:hi+m_shift])
            lo2 = max(0, m_shift)
            hi2 = min(n, n + m_shift)
            if hi2 > lo2:
                gk[lo2:hi2] -= h[lo2-m_shift:hi2-m_shift]
            grad += dx * w[k] * gk
        
        # Adam update
        step_num = step + 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**step_num)
        v_hat = v_v / (1 - beta2**step_num)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Fix sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    
    return h
```

def obj_and_grad(h, t, lam):
    n = len(h)
