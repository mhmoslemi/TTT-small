```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    # Fix remaining error
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute to elements with room
        indices = np.arange(n)
        room = np.where((x < 1.0) & (diff > 0)) | np.where((x > 0.0) & (diff < 0))
        room = room[0]
        if len(room) > 0:
            per = diff / len(room)
            x[room] += per
        x = np.clip(x, 0.0, 1.0)
    return x

def project_feasible(x, n):
    """Project x onto {0 <= h <= 1, sum(h) = n/2}"""
    x = np.array(x, dtype=float)
    tgt = n / 2.0
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(y)
        if s > tgt:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Final fix
    diff = tgt - np.sum(y)
    if abs(diff) > 1e-12:
        idx = np.where((y < 1.0) & (diff > 0) | (y > 0.0) & (diff < 0))[0]
        if len(idx) > 0:
            y[idx] += diff / len(idx)
            y = np.clip(y, 0.0, 1.0)
    return y

def compute_grad_max(h, n, dx):
    """Compute gradient of max_k f_k(h) w.r.t. h.
    f_l = dx * sum_i h[i]*(1-h[i-l]) for valid indices, corresponding to corr index k = l + n - 1
    """
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    M = np.max(corr)
    # Active set: lags close to max
    active = np.where(corr >= M - 1e-6)[0]
    
    grad = np.zeros(n)
    for k in active:
        l = k - (n - 1)  # lag

        # ∂f_l/∂h[j]:
        #   from h[j] term: (1 - h[j-l]) if j-l in [0,n)
        #   from (1-h[i-l]) where i-l=j, i=j+l: -h[j+l] if j+l in [0,n)
        for j in range(n):
            jl = j - l
            jpl = j + l
            g = 0.0
            if 0 <= jl < n:
                g += (1.0 - h[jl])
            if 0 <= jpl < n:
                g -= h[jpl]
            grad[j] += g
    grad *= dx / len(active)
    return grad, M

def projected_subgrad(h, n, dx, steps, lr0=0.01, decay=0.999, momentum=0.9):
    """Projected subgradient method with momentum for the max problem."""
    h = h.copy()
    v = np.zeros(n)
    best = h.copy()
    best_val = compute_c5(h, dx)
    
    for step in range(steps):
        grad, M = compute_grad_max(h, n, dx)
        lr = lr0 * (decay ** step)
        v = momentum * v + (1 - momentum) * grad
        h_new = project_feasible(h - lr * v, n)
        val = compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            best = h_new.copy()
        h = h_new
    return best, best_val
```

```python
def obj_t(x, n, dx, lam, mu):
    # x = [h_0, ..., h_{n-1}, t]
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    pen = np.maximum(0.0, corr - t)
    obj = t + lam * np.sum(pen**2) + mu * (np.sum(h) - n/2.0)**2
    return obj

def obj_t_grad(x, n, dx, lam, mu):
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    pen = np.maximum(0.0, corr - t)
    
    # d(obj)/dt = 1 - lam * sum(2*pen)
    dt = 1.0 - 2.0 * lam * np.sum(pen)
    
    # d(obj)/dh[j] = sum_k 2*lam*pen[k]*d(corr[k])/dh[j] + 2*mu*(sum(h)-n/2)
    # d(corr[k])/dh[j] where corr[k] = dx * sum_i h[i]*(1-h[i + k - n + 1])
    # Actually let me just use numerical or the analytical formula
    
    # The gradient of corr[k] w.r.t. h:
    # corr[k] = dx * sum_i h[i] * (1 - h[i - (k-n+1)])
    # Let l = k - (n-1)
    # d(corr[k])/dh[j] = dx * [(1-h[j-l]) - h[j+l]] (with bounds)
    
    dcorr = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        l = k - (n-1)
        for j in range(n):
            g = 0.0
            jl = j - l
            if 0 <= jl < n:
                g += (1.0 - h[jl])
            jpl = j + l
            if 0 <= jpl < n:
                g -= h[jpl]
            dcorr[k, j] = g
    dcorr *= dx
    
    sum_h = np.sum(h)
    dh = 2.0 * lam * (pen @ dcorr) + 2.0 * mu * (sum_h - n/2.0)
    
    return np.concatenate([dh, [dt]])
```

Actually, I can avoid storing the full dcorr matrix. Let me compute the gradient more efficiently:

dh[j] = 2*lam * sum_k pen[k] * dcorr[k,j] + 2*mu*(sum_h - n/2)

For each k with pen[k] > 0:
  l = k - (n-1)
  For j: contribution to dh[j] from this k:
    If 0 <= j-l < n: += 2*lam*pen[k]*dx*(1-h[j-l])
    If 0 <= j+l < n: -= 2*lam*pen[k]*dx*h[j+l]

So:
for k where pen[k] > 0:
  l = k - (n-1)
  w = 2*lam*pen[k]*dx
  # j-l in [0,n): j in [max(0,l), min(n, n+l))
  if l >= 0:
    j_lo, j_hi = l, n
  else:
    j_lo, j_hi = 0, n + l
  if j_hi > j_lo:
    dh[j_lo:j_hi] += w * (1 - h[j_lo-l:j_hi-l])
  
  # j+l in [0,n): j in [max(0,-l), min(n, n-l))
  if l >= 0:
    j_lo2, j_hi2 = 0, n - l
  else:
    j_lo2, j_hi2 = -l, n
  if j_hi2 > j_lo2:
    dh[j_lo2:j_hi2] -= w * h[j_lo2+l:j_hi2+l]

This is O(n * number_of_active_k) which is much better.

```python
import numpy as np
from scipy.optimize import minimize, minimize_slsqp
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        # Add to elements with room
        if diff > 0:
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
        x = np.clip(x, 0.0, 1.0)
    return x

def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    tgt = n / 2.0
    # Bisection on additive shift to satisfy sum constraint while maintaining [0,1] bounds
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if np.sum(y) > tgt:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-10)[0]
        else:
            idx = np.where(x > 1e-10)[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
        x = np.clip(x, 0.0, 1.0)
    return x

def _gdx, _gtau, _gn:  # These can't be module-level like this in a function
    pass

# Use a class or module-level variables for SLSQP objective
_gdx = 1.0
_gtau = 0.1
_gn = 0

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx

def _slsqp_cfun(x):
    return np.sum(x) - _gn / 2.0

def _slsqp_cjac(x):
    return np.ones(_gn)

def slsqp_optimize(x0, n, dx, tau, maxiter=300, ftol=1e-15):
    global _gdx, _gtau, _gn
    _gn = n
    _gdx = dx
    _gtau = tau
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except:
        return x0.copy()

def lbfgsb_t_optimize(h0, n, dx, lam, mu, maxiter=2000, tol=1e-14):
    """L-BFGS-B on [h, t] variables with hinge penalty."""
    x0 = np.concatenate([h0, [compute_c5(h0, dx)]])
    
    def obj(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        pen = np.maximum(0.0, corr - t)
        return t + lam * np.sum(pen * pen) + mu * (np.sum(h) - n / 2.0) ** 2
    
    def grad(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        pen = np.maximum(0.0, corr - t)
        
        dt = 1.0 - 2.0 * lam * np.sum(pen)
        
        dh = np.zeros(n)
        active = pen > 1e-14
        for k in np.where(active)[0]:
            l = k - (n - 1)
            w = 2.0 * lam * pen[k] * dx
            # h[j] term: j-l in [0,n)
            if l >= 0:
                jlo, jhi = l, n
            else:
                jlo, jhi = 0, n + l
            if jhi > jlo:
                dh[jlo:jhi] += w * (1.0 - h[jlo - l:jhi - l])
            # (1-h[i-l]) term: i-l=j means i=j+l, need j+l in [0,n)
            if l >= 0:
                jlo2, jhi2 = 0, n - l
            else:
                jlo2, jhi2 = -l, n
            if jhi2 > jlo2:
                dh[jlo2:jhi2] -= w * h[jlo2 + l:jhi2 + l]
        
        dh += 2.0 * mu * (np.sum(h) - n / 2.0)
        
        return np.concatenate([dh, [dt]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-12})
    h_opt = repair(res.x[:n], n)
    return h_opt, compute_c5(h_opt, dx)

def subgrad_polish(h0, n, dx, steps=50000, lr0=0.005):
    """Projected subgradient with momentum."""
    h = h0.copy()
    v = np.zeros(n)
    best = h.copy()
    best_val = compute_c5(h, dx)
    beta = 0.8
    
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        M = np.max(corr)
        if M < best_val:
            best_val = M
            best = h.copy()
        
        # Subgradient of max
        active = np.where(corr >= M - 1e-8)[0]
        grad = np.zeros(n)
        for k in active:
            l = k - (n - 1)
            if l >= 0:
                jlo, jhi = l, n
            else:
                jlo, jhi = 0, n + l
            if jhi > jlo:
                grad[jlo:jhi] += (1.0 - h[jlo - l:jhi - l])
            if l >= 0:
                jlo2, jhi2 = 0, n - l
            else:
                jlo2, jhi2 = -l, n
            if jhi2 > jlo2:
                grad[jlo2:jhi2] -= h[jlo2 + l:jhi2 + l]
        grad *= dx / len(active)
        
        # Momentum
        v = beta * v + (1 - beta) * grad
        lr = lr0 / (1.0 + 0.01 * step)
        h = project_feasible(h - lr * v, n)
    
    return best, best_val

def perturb(h, n, scale, rng):
    return repair(h + scale * rng.standard_normal(n), n)

def project_grid(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    from scipy.interpolate import interp1d
    f = interp1d(xs, h_src, kind='linear', fill_value='extrapolate')
    return repair(f(xd), n_dst)

def refine_multi(x0, n, dx, time_limit, rng):
    """Multi-stage refinement: SLSQP -> L-BFGS-B -> subgradient"""
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    
    # Stage 1: SLSQP with decreasing tau
    if time.time() - t0 < time_limit * 0.4:
        for tau in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            if time.time() - t0 > time_limit * 0.4:
                break
            alloc_iters = 200
            x_new = slsqp_optimize(x, n, dx, tau, alloc_iters, 1e-14)
            v = compute_c5(x_new, dx)
            if v < best_val:
                best_val = v
                best = x_new.copy()
            x = x_new
    
    # Stage 2: L-BFGS-B with t-variables
    if time.time() - t0 < time_limit * 0.75:
        for lam in [100.0, 50.0, 20.0, 10.0]:
            if time.time() - t0 > time_limit * 0.75:

            if v < best_val:
                best_val = v
                best = x_new.copy()
            x = x_new
    
    # Stage 3: Subgradient polish
    if time.time() - t0 < time_limit * 0.95:

        steps = max(steps, 5000)
        x_new, v = subgrad_polish(x, n, dx, steps, 0.003)
        if v < best_val:
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gtau = 0.1
_gn = 0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0).copy()
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
        else:
            idx = np.where(x > 1e-12)[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
        x = np.clip(x, 0.0, 1.0)
    return x

def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    tgt = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if np.sum(y) > tgt:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-10)[0]
        else:
            idx = np.where(x > 1e-10)[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
        x = np.clip(x, 0.0, 1.0)
    return x

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx

def _slsqp_cfun(x):
    return np.sum(x) - _gn / 2.0

def _slsqp_cjac(x):
    return np.ones(_gn)

def slsqp_opt(x0, n, dx, tau, maxiter=400, ftol=1e-15):
    global _gdx, _gtau, _gn
    _gn, _gdx, _gtau = n, dx, tau
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return x0.copy()

def lbfgsb_t_opt(h0, n, dx, lam, mu, maxiter=3000, tol=1e-15):
    x0 = np.concatenate([h0, [compute_c5(h0, dx) + 0.01]])
    def obj(x):
        h, t = x[:n], x[n]
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        pen = np.maximum(0.0, corr - t)
        return t + lam * np.dot(pen, pen) + mu * (np.sum(h) - n / 2.0) ** 2
    def grad(x):
        h, t = x[:n], x[n]
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        pen = np.maximum(0.0, corr - t)
        dt = 1.0 - 2.0 * lam * np.sum(pen)
        dh = np.zeros(n)
        active = np.where(pen > 1e-14)[0]
        for k in active:
            l = k - (n - 1)
            w = 2.0 * lam * pen[k] * dx
            if l >= 0:
                jlo, jhi = l, n
            else:
                jlo, jhi = 0, n + l
            if jhi > jlo:
                dh[jlo:jhi] += w * (1.0 - h[jlo - l:jhi - l])
            if l >= 0:
                jlo2, jhi2 = 0, n - l
            else:
                jlo2, jhi2 = -l, n
            if jhi2 > jlo2:
                dh[jlo2:jhi2] -= w * h[jlo2 + l:jhi2 + l]
        dh += 2.0 * mu * (np.sum(h) - n / 2.0)
        return np.concatenate([dh, [dt]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-14})
        h_opt = repair(res.x[:n], n)
        return h_opt, compute_c5(h_opt, dx)
    except Exception:
        return h0.copy(), compute_c5(h0, dx)

def subgrad_polish(h0, n, dx, steps, lr0=0.004, beta=0.7):
    h = h0.copy()
    v = np.zeros(n)
    best, best_val = h.copy(), compute_c5(h, dx)
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        M = np.max(corr)
        if M < best_val:
            best_val, best = M, h.copy()
        active = np.where(corr >= M - 1e-9)[0]
        grad = np.zeros(n)
        for k in active:
            l = k - (n - 1)
            if l >= 0:
                jlo, jhi = l, n
            else:
                jlo, jhi = 0, n + l
            if jhi > jlo:
                grad[jlo:jhi] += (1.0 - h[jlo - l:jhi - l])
            if l >= 0:
                jlo2, jhi2 = 0, n - l
            else:
                jlo2, jhi2 = -l, n
            if jhi2 > jlo2:
                grad[jlo2:jhi2] -= h[jlo2 + l:jhi2 + l]
        grad *= dx / len(active)
        v = beta * v + (1.0 - beta) * grad
        lr = lr0 * lr0 / (lr0 + step * 0.0001)
        h = project_feasible(h - lr * v, n)
    return best, best_val

def perturb(h, n, scale, rng):
    return repair(h + scale * rng.standard_normal(n), n)

def project_grid(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    f = interp1d(xs, h_src, kind='linear', fill_value='extrapolate')
    return repair(f(xd), n_dst)
```
