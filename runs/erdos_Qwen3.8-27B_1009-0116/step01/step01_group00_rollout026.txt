    offset = k - (len(h) - 1)
    n = len(h)
    g = np.zeros(n)
    # Term 1: (1 - h[j - offset]) where 0 <= j - offset < n
    lo1 = max(0, offset)

    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 + offset:hi2 + offset]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Scale to meet sum constraint
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a

                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

    """Compute gradient of sum of top-m correlation values w.r.t. h"""
    corr = np.correlate(h, 1.0 - h, mode='full')

        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_val = best
        for o in ones:
            for z in zeros:
                h[o], h[z] = 0.0, 1.0
                v = _c5(h, dx)
                if v < best_val - 1e-14:
                    best_val = v
                    best_swap = (o, z)
                h[o], h[z] = 1.0, 0.0
        if best_swap:
            o, z = best_swap
            h[o], h[z] = 0.0, 1.0
            best = best_val
            improved = True
        else:
            break
    return h, best

def _binary_sa(h, n, rng, deadline, n_iter=500000):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    T = 0.005
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        o = ones[rng.randint(len(ones))]
        z = zeros[rng.randint(len(zeros))]
        h[o], h[z] = 0.0, 1.0
        c5 = _c5(h, dx)
        if c5 < best_c5 or rng.rand() < np.exp(-(c5 - best_c5) / T):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[o], h[z] = 1.0, 0.0
        T = max(0.0002, T * 0.99998)
    return best_h, best_c5

def _subgradient(h, n, deadline, n_steps=30000):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    lr = 0.004
    m = max(5, n // 8)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g, corr = _grad_top(h, n, m)
        gn = np.linalg.norm(g)
        if gn > 1e-12:
            h_new = _repair(h - lr * g / gn, n)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        lr = max(1e-6, lr * 0.9995)
    return best_h, best_c5

def _coord_descent(h, n, rng, deadline, n_passes=50000):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05,
                       -0.002, -0.005, -0.01, -0.02, -0.05])
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        ni, nj = h[i] + d, h[j] - d
        if -1e-12 <= ni <= 1.0 + 1e-12 and -1e-12 <= nj <= 1.0 + 1e-12:
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c5

def _make_binary(n, rng, seed_mode=0):
    h = np.zeros(n)
    if seed_mode == 0:
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    elif seed_mode == 1:
        h[:n // 2] = 1.0
    else:
        for i in range(n):
            h[i] = 1.0 if (i * (n // 2)) % n < n // 2 else 0.0
        h[rng.permutation(n)] = h
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)

    # Phase 1: Binary descent from initial
    h_bin = (h0 >= 0.5).astype(float)
    s = h_bin.sum()
    while abs(s - n // 2) > 1e-10:
        if s > n // 2:
            h_bin[rng.choice(np.where(h_bin == 1)[0])] = 0.0
            s -= 1
        else:
            h_bin[rng.choice(np.where(h_bin == 0)[0])] = 1.0
            s += 1
    dl = min(t0 + 80, t0 + budget_s - 500)
    h_bin, c5_bin = _binary_descent(h_bin, n, dl)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin.copy()

    # Phase 2: Binary SA for escape
    dl = min(t0 + 300, t0 + budget_s - 300)
    h_sa, c5_sa = _binary_sa(h_bin, n, rng, dl)
    if c5_sa < best_c5:
        best_c5, best_h = c5_sa, h_sa.copy()
    h_bd2, c5_bd2 = _binary_descent(h_sa, n, dl)
    if c5_bd2 < best_c5:
        best_c5, best_h = c5_bd2, h_bd2.copy()

    # Phase 3: Multiple binary restarts with descent+SA
    dl = min(t0 + 600, t0 + budget_s - 200)
    for r in range(15):
        if time.time() > dl:
            break
        h_r = _make_binary(n, rng, seed_mode=r % 3)
        h_r, c5_r = _binary_descent(h_r, n, dl)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()
        if r % 3 == 2:
            h_sa2, c5_sa2 = _binary_sa(h_r, n, rng, min(dl, t0 + budget_s - 300),
                                       n_iter=100000)
            if c5_sa2 < best_c5:
                best_c5, best_h = c5_sa2, h_sa2.copy()

    # Phase 4: Subgradient continuous refinement
    dl = min(t0 + 850, t0 + budget_s - 100)
    h_sg, c5_sg = _subgradient(best_h, n, dl)
    if c5_sg < best_c5:
        best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 5: Coordinate descent polish
    dl = min(t0 + 1000, t0 + budget_s - 40)
    h_cd, c5_cd = _coord_descent(best_h, n, rng, dl)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 6: Another subgradient round with finer lr
    if time.time() < t0 + budget_s - 30:
        h_sg2, c5_sg2 = _subgradient(best_h, n, min(t0 + budget_s - 20, t0 + 1060),
                                     n_steps=10000)
        if c5_sg2 < best_c5:
            best_c5, best_h = c5_sg2, h_sg2.copy()

    # Phase 7: Final SLSQP polish
    if time.time() < t0 + budget_s - 8:
        try:
            alpha = 300
            def smooth_obj(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                m = c.max()
                s = np.clip(alpha * (c - m), -200, 200)
                return m + np.log(np.sum(np.exp(s))) / alpha
            def eq_con(x):
                return np.sum(x) - n / 2.0
            res = minimize(smooth_obj, best_h, method='SLSQP',
                          bounds=[(0, 1)] * n,
                          constraints=[{'type': 'eq', 'fun': eq_con}],
                          options={'maxiter': 150, 'ftol': 1e-14})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
