    best_c5 = _c5(h, dx)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Try moving from j to i and from i to j
                for direction in [(i, j), (j, i)]:
                    src, dst = direction
                    if h[src] < 1e-10 or h[dst] > 1 - 1e-10:

**Greedy swap descent:**
1. Compute all c_k
2. Find the worst shift s*
3. For shift s*, the "cost" of position i is (1-h[i])*h[i+s] (if s>0) or (1-h[i-s])*h[i] (if s<0)
4. The "most expensive" positions are where this cost is highest
5. Try to reduce cost by moving mass: from i+s to i (if s>0) or from i-s to i (if s<0)
6. This directly reduces c(s*) but might increase c(s*') for other shifts

import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def _cv(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return float(_cv(h).max() * dx)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sobj(x):
    dx, eps = _OPT['dx'], _OPT['eps']
    cv = _cv(x) * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad(x):
    dx, eps, n = _OPT['dx'], _OPT['eps'], _OPT['n']
    cv = _cv(x) * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wt = w[k] * dx
        a, b = max(0, s), min(n - 1, n - 1 + s)
        if a <= b:
            g[a:b+1] += wt * (1.0 - x[a - s:b - s + 1])
        a, b = max(0, -s), min(n - 1, n - 1 - s)
        if a <= b:
            g[a:b+1] -= wt * x[a + s:b + s + 1]
    return g

def _feq(x):
    return x.sum() - _OPT['target']

def _feq_jac(x):
    return np.ones(len(x))

def _slsqp(h0, dx, n, target, eps, maxiter=400):
    _OPT['dx'] = dx
    _OPT['n'] = n
    _OPT['eps'] = float(eps)
    _OPT['target'] = float(target)
    res = minimize(_sobj, h0, jac=_sgrad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _feq, 'jac': _feq_jac}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-15})
    return _proj(res.x, target)
