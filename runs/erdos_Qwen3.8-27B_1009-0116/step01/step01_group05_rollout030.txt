import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj_binary(h, n_ones):
    """Project to binary with exactly n_ones ones"""
    idx = np.argsort(-h)
    out = np.zeros(len(h))
    out[idx[:n_ones]] = 1.0
    return out

def _proj_sum(h, target):
    """Project to [0,1] with sum = target"""
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _smooth_obj(x, dx, n, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _smooth_grad(x, dx, n, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _slsqp_opt(h0, dx, n, target, eps, maxiter=300):

        return _smooth_obj(x, dx, n, eps)
    def jac(x):
        return _smooth_grad(x, dx, n, eps)
    def eqfun(x):
        return x.sum() - target
    def eqjac(x):

    res = minimize(

from scipy.optimize import minimize
import time

_DX = 2.0 / 70
_N = 70
_TGT = 35.0
_EPS = 0.01

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj_sum(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sm_obj(x):
    cv = _corr(x, _DX)
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))

def _sm_grad(x):
    cv = _corr(x, _DX)
    mx = cv.max()
    w = np.exp((cv - mx) / _EPS)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _eq_fun(x):
    return x.sum() - _TGT

def _eq_jac(x):
    return np.ones(_N)

def _slsqp(h0, eps, maxiter=400):
    global _EPS
    old_eps = _EPS
    _EPS = float(eps)
    try:
        res = minimize(
            _sm_obj, h0, jac=_sm_grad,
            bounds=[(0.0, 1.0)] * _N,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
    except Exception:
        res = None
    _EPS = old_eps
    if res is not None and res.x is not None:
        return _proj_sum(res.x, _TGT)
    return h0.copy()

def _binary_sa(h_start, dx, n, n_ones, rng, t_end, iters_per_sec=50000):
    """Simulated annealing on binary sequences"""
    h = np.zeros(n)
    ones = np.zeros(n, dtype=bool)
    # Initialize from h_start
    thresh = np.partition(h_start, n - n_ones)[n - n_ones]
    ones = h_start > thresh
    # Ensure exactly n_ones

        # Remove the least "wanted" ones
        vals = h_start[excess]
        order = np.argsort(vals)[:ones.sum() - n_ones]
        ones[excess[order]] = False
    elif ones.sum() < n_ones:
        deficit = np.where(~ones)[0]
        vals = h_start[deficit]
        order = np.argsort(-vals)[:n_ones - ones.sum()]
        ones[deficit[order]] = True
    
    h[ones] = 1.0
    cur_c5 = _c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.1
    T_min = 0.001
    cool = 0.9999
    
    total_iters = int((t_end - time.time()) * iters_per_sec)
    for it in range(total_iters):
        if it % 10000 == 0 and time.time() > t_end:
            break
        # Pick random swap
        one_idx = np.where(ones)[0]
        zero_idx = np.where(~ones)[0]
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        if i == j:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        ones[i] = False
        ones[j] = True
        
        new_c5 = _c5(h, dx)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Undo
            h[i], h[j] = h[j], h[i]
            ones[i] = True
            ones[j] = False
        
        T = max(T * cool, T_min)
    
    return best_h, best_c5
