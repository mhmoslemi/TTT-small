import numpy as np
import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _project_sum(h, target):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target:
            hi = mid
        else:
            lo = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-10:
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _obj_grad(h, alpha):
    N = len(h)
    dx = 2.0 / N
    R = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(R)
    sh = np.clip(R - m, -50.0/alpha, 10.0/alpha)
    ew = np.exp(alpha * sh)
    w = ew / ew.sum()
    obj = (m + np.log(ew.sum()) / alpha) * dx
    grad = np.zeros(N)
    N1 = N - 1
    for j in range(N):
        s = 0.0
        k1lo = max(0, N1 - j)
        k1hi = min(2*N1, 2*N1 - j)
        for k in range(k1lo, k1hi + 1):
            s += w[k] * (1.0 - h[j + k - N1])
        k2lo = max(0, j)
        k2hi = min(2*N1, j + N1)
        for k in range(k2lo, k2hi + 1):
            s -= w[k] * h[j - k + N1]
        grad[j] = dx * s
    return obj, grad

def _local_refine(h, target, max_passes=15, t0=None, budget=1100):
    N = len(h)
    bc = _c5(h)
    deltas = (0.05, -0.05, 0.02, -0.02, 0.1, -0.1, 0.01, -0.01)
    for p in range(max_passes):
        if t0 is not None and budget - (time.time() - t0) < 5:
            break
        moved = False
        for i in range(N):
            for d in deltas:
                h2 = h.copy()
                hi_new = h[i] + d
                if hi_new < 0 or hi_new > 1:
                    continue
                h2[i] = hi_new
                # Move to nearest feasible neighbor
                found = False
                for offset in (1, -1, 2, -2, 3, -3, 4, -4, N//3, -N//3, N//2, -N//2):
                    j = (i + offset) % N
                    hj_new = h[j] - d
                    if 0 <= hj_new <= 1 and j != i:
                        h2[j] = hj_new
                        found = True
                        break
                if not found:
                    continue
                c = _c5(h2)
                if c < bc - 1e-12:
                    h = h2
                    bc = c
                    moved = True
                    break
            if moved:
                break
        if not moved:
            break
    return h, bc

def _init_parent_to_n(parent, n):
    np_orig = len(parent)
    x_orig = np.linspace(0, 2, np_orig, endpoint=False) + 1.0/np_orig
    x_new = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    hc = np.interp(x_new, x_orig, parent)
    return _project_sum(hc, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    best_n = len(parent)

        if budget_s - (time.time() - t0) < 30:
            break
        target = n / 2.0
        dx = 2.0 / n
        n_restarts = 3 if n <= 90 else 2

            if budget_s - (time.time() - t0) < 15:
                break
            
            # Initialize

                else:

            elif r == 1:

                    p = rng.uniform(0, 2 * np.pi)

            alpha = 50.0

            obj_cur = _obj_grad(hc, alpha)[0]

            for step in range(max_steps):

                    break
                if step % 500 == 499:
                    alpha *= 2
                    lr *= 0.5
                
                _, g = _obj_grad(hc, alpha)
                gnorm = np.linalg.norm(g)
                if gnorm < 1e-8:

                h_new = _project_sum(h_new, target)

                if obj_new < obj_cur:
                    hc = h_new
                    obj_cur = obj_new
                    lr = min(lr * 1.1, 2.0)

    tgt = best_n / 2.0

        h_ret = np.clip(h_ret, 0.0, 1.0)
    
    c5f = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return h_ret, c5f, best_n

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _project_sum(h, target):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-10:
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _obj_grad(h, alpha):
    N = len(h)
    dx = 2.0 / N
    R = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(R)
    sh = np.clip(R - m, -50.0 / alpha, 10.0 / alpha)
    ew = np.exp(alpha * sh)
    sw = ew.sum()
    w = ew / sw
    obj = (m + np.log(sw) / alpha) * dx
    N1 = N - 1
    grad = np.zeros(N)
    for j in range(N):
        s = 0.0
        k1lo = max(0, N1 - j)
        k1hi = min(2 * N1, 2 * N1 - j)
        for k in range(k1lo, k1hi + 1):
            s += w[k] * (1.0 - h[j + k - N1])
        k2lo = j
        k2hi = min(2 * N1, j + N1)
        for k in range(k2lo, k2hi + 1):
            s -= w[k] * h[j - k + N1]
        grad[j] = dx * s
    return obj, grad

            break

            if moved:
                break

                hi2 = h[i] + d

                    continue
                for offset in (1, -1, 2, -2, 3, -3, N // 4, -N // 4, N // 2, -N // 2):
