import numpy as np
from scipy.optimize import minimize
import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return np.max(_corr_arr(h)) * dx

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _qpso_update(particle, gbest, beta, rng, n):
    mid = 0.5 * (particle + gbest)
    spread = 0.5 * np.abs(particle - gbest)
    r = -np.log(rng.rand(n))
    return mid + beta * spread * (2.0 * r - 1.0)

def _directed_local(h, n, dx, rng, deadline, t_start):
    best_local = _compute_c5(h, dx)
    h = h.copy()
    scales = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    for scale in scales:
        stalled = 0
        for _pass in range(200):
            if time.time() >= deadline:
                return h, best_local
            improved = False
            for _ in range(800):
                if time.time() >= deadline:
                    return h, best_local
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c5 = _compute_c5(h, dx)
                if c5 < best_local - 1e-11:
                    best_local = c5
                    improved = True
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
            if improved:
                stalled = 0
            else:
                stalled += 1
                if stalled >= 3:
                    break
    return h, best_local

def _slsqp_refine(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    try:
        res = minimize(
            lambda x, a=alpha: _smooth_obj(x, a), h,
            method='SLSQP', bounds=bounds, constraints=[eq_con],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t_start)

    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Phase 1: SLSQP warm-start from initial (120s)
    t_end = t_start + 120
    for alpha in [30, 80, 200]:
        if time.time() > t_end or tl() < 500:
            break
        best_h, best_c5 = _slsqp_refine(best_h, n, dx, bounds, eq_con, alpha, 500, t_end)

    # Phase 2: QPSO global exploration (250s)
    t_end = t_start + 370
    swarm_size = 18
    particles = [best_h.copy()]
    for _ in range(swarm_size - 1):
        particles.append(_repair(best_h + 0.35 * rng.randn(n), n))
    gbest = best_h.copy()
    gbest_c5 = best_c5
    it = 0
    while time.time() < t_end and tl() > 350:
        it += 1
        beta = 1.8 - 0.6 * min(1.0, it / 120.0)
        for pi in range(swarm_size):
            if time.time() > t_end:
                break
            x = _qpso_update(particles[pi], gbest, beta, rng, n)
            x = _repair(x, n)
            c5 = _compute_c5(x, dx)
            particles[pi] = x
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest = x.copy()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x.copy()

    # Phase 3: SLSQP multi-restart refinement (300s)
    t_end = t_start + 670
    for alpha in [50, 100, 200, 400, 800, 1500]:
        if time.time() > t_end or tl() < 150:
            break
        for restart in range(2):
            if time.time() > t_end or tl() < 150:
                break
            if restart == 0:
                start = best_h.copy()
            else:
                start = _repair(best_h + 0.04 * rng.randn(n), n)
            best_h, best_c5 = _slsqp_refine(start, n, dx, bounds, eq_con, alpha, 800, t_end)

    # Phase 4: Intensive directed local search (200s)
    if tl() > 80:
        ls_deadline = time.time() + min(200, tl() - 40)
        best_h, best_c5 = _directed_local(best_h, n, dx, rng, ls_deadline, t_start)

    # Phase 5: Final SLSQP polish + local search
    if tl() > 40:
        best_h, best_c5 = _slsqp_refine(best_h, n, dx, bounds, eq_con, 3000, 1500, time.time() + min(40, tl() - 15))
    if tl() > 20:
        ls_deadline = time.time() + min(15, tl() - 5)
        best_h, best_c5 = _directed_local(best_h, n, dx, rng, ls_deadline, t_start)

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
