from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
            else:
                h += diff / n
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
            else:
                h += diff / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def ternary_max_linear(a, d, lo=0.0, hi=1.0, iters=50):
    """Find t in [lo,hi] minimizing max_m (a[m] + d[m]*t)"""
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a + d * m1)
        f2 = np.max(a + d * m2)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, dx, max_passes=5000, tol=1e-12):
    """Coordinate descent for minimizing max correlation."""
    h = project_feasible(h.copy(), n)
    target = float(n) / 2.0
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    
    for p in range(max_passes):
        max_change = 0.0
        # Random coordinate order for this pass
        order = np.random.permutation(n)
        for i in order:
            # Compute 1D subproblem
            old = h[i]
            # Set h[i]=0, compute corr
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            # Set h[i]=1, compute corr
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            
            a = c0
            d = c1 - c0
            # Note: at lag 0 (index n-1), the true model is a + t - t^2
            # but d[n-1] = 0, so the linear model gives a[0] (constant)
            # The true value at t is a[n-1] + t - t^2
            # We'll refine after the linear solve
            
            t_opt = ternary_max_linear(a, d)
            
            # Refine around t_opt with the full (quadratic-aware) evaluation
            # Try a few candidates
            best_t = t_opt
            best_val = np.inf
            for dt in [0, 0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                t_try = float(np.clip(t_opt + dt, 0.0, 1.0))
                h[i] = t_try

                val = float(np.max(corr))
                if val < best_val:

                    best_t = t_try

        if c5 < best_c5:

            best_h = h.copy()

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
            else:
                h[0] += min(diff, 1.0 - h[0])
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
            else:
                h[0] += max(diff, -h[0])
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            if d2 > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + d2)
            else:
                idx = int(np.argmin(h))
                h[idx] = max(0.0, h[idx] + d2)
    return h

def solve_1d(a, d):
    """Minimize max_m(a[m] + d[m]*t) for t in [0,1]."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a + d * m1) <= np.max(a + d * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
