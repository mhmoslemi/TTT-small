                factor = -1.0
            else:
                factor = 1.0
            for j in range(n):
                if valid_h[j] > 1e-10:
                    valid_h[j] = min(1.0, max(0.0, valid_h[j] + factor * diff/ n))
            valid_h = np.clip(valid_h, 0.0, 1.0)
        best_c5 = compute_c5(valid_h)
        best_h = valid_h
    else:
        best_h = h_initial
        best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n

# Strategy optimization loop
def strategy_optimization(initial_func='parent', n0=70):
    """
    Execute the complete strategy optimization over multiple candidates and resolutions.
    
    Args:
        initial_func: If 'parent', use provided initial_h_values, or 'multi'
        n0: Base resolution
    """
    # Time budget management
    initial_time = time.time()

    best_c5_val = 1.0

        else:
            h0_70 = multi_sine(70)
            print("Using multi-sine for n=70 because parent size is invalid")
    else:
        h0_70 = multi_sine(70)
        print("Using multi-sine for n=70")

    best_candidate = cand
    best_c5_val = c5_val
    all_best = c5_val
    print(f"\nBase n=70: C5 = {c5_val}")
    
    # Set time spent
    time_spent = time.time() - initial_time
    remaining_time -= time_spent
    
    # Check remaining_time at start of n=80
    n_prog = [70, 80, 90, 100]
    n_best_so_far = 70
    candidate = h0_70.copy()
    best_scale = None  # Not used
    
    # Now, adjust for n increments
    if len(n_arr) > 0:
        for n_idx, n_next in enumerate(n_arr[1:]):
            if remaining_time < 0:
                break  # No time left
            prev_n = n_arr[n_idx]
            local_time = max(0.5, min(remaining_time/n_prog[-1], 100))
            # Generate candidate from previous best
            try:
                if n_prog[-1] == 70:
                    h_next = interpolate_solution(candidate, 70, n_next)
                else:
                    h_next = interpolate_solution(candidate, n_prog[n_idx], n_next)
                    # Generate second candidate independently
                    h_next_2 = multi_sine(n_next)
                    
                    # Now, optimize both candidates
                    h_best1, c5_best1, _ = resolve_optimization(h_next, n_next, local_time)
                    h_best2, c5_best2, _ = resolve_optimization(h_next_2, n_next, local_time)
                    
                    # Combine the best
                    if c5_best1 < c5_best2:
                        best_c5_val = c5_best1
                        candidate = h_best1
                    else:
                        best_c5_val = c5_best2
                        candidate = h_best2
                    
                    # Update all_best
                    if best_c5_val < all_best:
                        all_best = best_c5_val
                    
                    # Track best candidate for this n
                    print(f"n={n_next}, Best: {best_c5_val}")
                    
            except Exception as e:
                print(f"Error at n={n_next}: {str(e)}")
                # Try to fallback
                if 'n_initial_80' not in locals():
                    h80 = multi_sine(80)
                else:
                    h80 = n_initial_80
            
            # Update time tracking
            time_spent = time.time() - initial_time
            remaining_time -= time_spent
    
    # Final candidate and score validation
    candidate_final = np.clip(candidate, 0.0, 1.0)
    effective_sum = np.sum(candidate_final)
    if abs(effective_sum - (len(candidate_final)/2)) > 1e-5:
        # Scale if needed
        if abs(effective_sum) > 1e-10:
            scale_factor = (len(candidate_final)/2) / effective_sum
            candidate_final = candidate_final * scale_factor
        else:
            target_arr = np.array([1.0]*int(len(candidate_final)/2)  # Force ~n/2 ones
            perm = np.random.permutation(len(candidate_final))[:len(target_arr)]
            candidate_final = target_arr[perm]
            if len(candidate_final) % 2 != 0:
                candidate_final[-1] = 0.0
            # Ensure length match
            if len(candidate_final) > len(candidate):
                # Downhill to the correct length
                ratio = len(candidate) / len(candidate_final)
                candidate_final = candidate_final[:min(int(len(candidate_final)*ratio), len(candidate))]

    # Force final validation of integral
    final_sum = np.sum(candidate_final)
    target_sum = len(candidate_final)/2.0
    if abs(final_sum - target_sum) > 1e-5 and abs(final_sum) > 1e-10:
        scale_factor = target_sum / final_sum
        candidate_final = candidate_final * scale_factor
    
    # Recalculate C5 with final candidate
    best_c5_val = compute_c5(candidate_final)
    
    # Check if we were asked to use the parent and didn't
    if initial_func == 'parent' and 'initial_h_values' in globals():
        parent_used = globals()['initial_h_values']
    else:
        parent_used = None
    
    return (candidate_final, best_c5_val, len(candidate_final)), parent_used
