import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 1.0
_g_alpha = 100.0
_g_target = 35.0

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _norm_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full') * _g_dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(_g_alpha * (corr - m)))) / _g_alpha

def _eq_con(x):
    return np.sum(x) - _g_target

def _eq_jac(x):
    return np.ones_like(x)

def _period_obj(a, p, target_p):
    worst = 0.0
    for k in range(1, p):
        ac = float(np.sum(a * np.roll(a, k)))
        worst = max(worst, target_p - ac)
    return worst + 500.0 * (np.sum(a) - target_p) ** 2

def _make_periodic(n, p, seed):
    rng = np.random.RandomState(seed)
    target_p = p / 2.0
    a0 = rng.uniform(0.2, 0.8, p)
    try:
        res = minimize(_period_obj, a0, args=(p, target_p), method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-9, 'fatol': 1e-10})
        a = np.clip(res.x, 0.0, 1.0)
    except Exception:
        a = np.full(p, 0.5)
    a = a * target_p / max(float(np.sum(a)), 1e-12)
    a = np.clip(a, 0.0, 1.0)
    h = np.tile(a, n // p + 1)[:n]
    return _norm_h(h, n / 2.0)

def _slsqp_opt(h, n, t0, budget_s):
    global _g_dx, _g_alpha, _g_target
    dx = 2.0 / n
    target = n / 2.0
    _g_dx = dx
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    for alpha in [20, 50, 100, 200, 400, 800]:
        if time.time() - t0 > budget_s - 5:
            break
        _g_alpha = alpha
        try:
            maxit = 120 if n <= 80 else (80 if n <= 100 else 50)
            res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}],
                           options={'maxiter': maxit, 'ftol': 1e-13})
            h = _norm_h(res.x, target)
        except Exception:
            pass
    return h

def _targeted_cd(h, max_rounds, t0, budget_s):
    n = len(h)
    h = h.copy()
    best_c = _c5(h)
    rng = np.random.RandomState(31337)
    target = n / 2.0

    for rd in range(max_rounds):
        if time.time() - t0 > budget_s - 2:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        improved = False

        if lag != 0:
            for st in [0.005, 0.01, 0.02, 0.03]:
                if improved:
                    break
                for _ in range(12):
                    if improved:
                        break
                    if lag > 0:
                        i = rng.randint(0, n - lag)
                        j = i + lag
                        h2 = h.copy()
                        h2[j] -= st
                        h2[i] += st
                    else:
                        l = -lag
                        i = rng.randint(l, n)
                        j = i - l
                        h2 = h.copy()
                        h2[i] -= st
                        h2[j] += st
                    if h2[i] < -1e-12 or h2[i] > 1.0 + 1e-12 or h2[j] < -1e-12 or h2[j] > 1.0 + 1e-12:
                        continue
                    h2 = np.clip(h2, 0.0, 1.0)
                    if abs(h2.sum() - target) > 1e-10:
                        continue
                    c2 = _c5(h2)
                    if c2 < best_c - 1e-13:
                        h = h2
                        best_c = c2
                        improved = True

        if not improved:
            for _ in range(40):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.choice([-0.01, 0.01, -0.02, 0.02, -0.03, 0.03])
                h2 = h.copy()
                h2[i] += d
                h2[j] -= d
                if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                    continue
                if abs(h2.sum() - target) > 1e-10:
                    continue
                c2 = _c5(h2)
                if c2 < best_c - 1e-13:
                    h = h2
                    best_c = c2
                    improved = True
                    break

        if not improved:
            for _ in range(15):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                tmp = h[i]
                h[i] = h[j]
                h[j] = tmp
                c2 = _c5(h)
                if c2 < best_c - 1e-13:
                    best_c = c2
                    improved = True
                    break
                h[i] = tmp
                h[j] = h[i]
                h[i] = tmp
                h[j] = tmp

        if not improved:
            break
    return h, best_c

def _binary_anneal(n, seed, t0, budget_s, max_iters=200000):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    best_c = _c5(h)
    best_h = h.copy()
    T = 0.003
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 2:
            break
        oi = rng.choice(n, 2, replace=False)
        if h[oi[0]] == h[oi[1]]:
            continue
        h[oi[0]], h[oi[1]] = 1.0 - h[oi[0]], 1.0 - h[oi[1]]
        c_new = _c5(h)
        if c_new <= best_c + T * rng.uniform():
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[oi[0]], h[oi[1]] = 1.0 - h[oi[0]], 1.0 - h[oi[1]]
        if it % 1000 == 0:
            T = max(0.0005, T * 0.98)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = len(parent)
    print(f"Parent: n={best_n}, C5={best_c5:.6f}")

    n_list = [70, 80, 90, 100]

    for n in n_list:
        if time.time() - t0 > budget_s - 25:
            break
        target = n / 2.0
        inits = []

        if n == len(parent):
            inits.append(parent.copy())
        else:
            xp = np.linspace(0, 2, len(parent), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            inits.append(_norm_h(np.interp(xn, xp, parent, period=2.0), target))

        for p in [5, 7, 10, 14]:
            if n % p == 0 and time.time() - t0 < budget_s - 15:
                try:
                    inits.append(_make_periodic(n, p, seed + n * 17 + p))
                except Exception:
                    pass

        for r in range(2):
            rng = np.random.RandomState(seed + n * 131 + r * 7)
            hs = np.full(n, 0.5)
            for k in range(rng.randint(2, 5)):
                f = rng.randint(1, max(2, n // 5))
                ph = rng.uniform(0, 2.0 * np.pi)
                am = rng.uniform(0.08, 0.38)
                hs += am * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
            inits.append(_norm_h(hs, target))

        for idx, h0 in enumerate(inits):
            if time.time() - t0 > budget_s - 10:
                break
            h = h0.copy()
            h = _slsqp_opt(h, n, t0, budget_s)
            h, cv = _targeted_cd(h, 150, t0, budget_s)
            if cv < best_c5:
                best_c5 = cv
                best_h = h.copy()
                best_n = n
                print(f"  n={n} init{idx}: C5={cv:.6f}")

    # Binary anneal at best_n and a couple others
    for bn in [best_n, 80, 100]:
        if bn != best_n and time.time() - t0 > budget_s - 30:
            break
        if time.time() - t0 > budget_s - 5:
            break
        bh, bc = _binary_anneal(bn, seed + 999, t0, budget_s)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()
            best_n = bn
            print(f"  binary n={bn}: C5={bc:.6f}")
        # Also try CD from binary
        if time.time() - t0 > budget_s - 5:
            break
        bh2, bc2 = _targeted_cd(bh.astype(float), 100, t0, budget_s)
        if bc2 < best_c5:
            best_c5 = bc2
            best_h = bh2.copy()
            best_n = bn
            print(f"  binary+cd n={bn}: C5={bc2:.6f}")

    # Final polish
    if time.time() - t0 < budget_s - 4:
        best_h, best_c5 = _targeted_cd(best_h, 500, t0, budget_s)

    # Enforce exact constraint
    tgt = best_n / 2.0
    diff = tgt - np.sum(best_h)
    if abs(diff) > 1e-13:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
        diff2 = tgt - np.sum(best_h)
        if abs(diff2) > 1e-14:
            best_h -= diff2 / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    c5f = _c5(best_h)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return best_h, c5f, best_n
