    m = np.max(scaled)
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    return val * dx

def sum_con(x, n):
    return np.sum(x) - float(n) / 2.0

def sum_jac(x, n):
    return np.ones(n)

def grad_step(h, n, step):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr_raw))
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        idx1 = np.arange(n - k)
        grad[idx1] += (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= h[idx2 - k]
    else:
        m = -k
        idx1 = np.arange(m, n)
        grad[idx1] += (1.0 - h[idx1 - m])
        idx2 = np.arange(n - m)
        grad[idx2] -= h[idx2 + m]
    h_new = h - step * grad
    return repair(h_new, n)

def pair_search(h, n, max_rounds, time_limit):
    t_start = time.time()
    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            for j in range(i + 1, n):
                for eps in (0.005, 0.01, 0.02, -0.005, -0.01, -0.02):
                    if 0.0 <= h[i] + eps <= 1.0 and 0.0 <= h[j] - eps <= 1.0:
                        hp = h.copy()
                        hp[i] += eps
                        hp[j] -= eps
                        c = compute_c5(hp, n)
                        if c < best_c - 1e-15:
                            h, best_c, improved = hp, c, True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    best_h = None
    best_c5 = 1.0
    best_n = 0

    if init is not None:
        n0 = len(init)
        h0 = repair(np.array(init, dtype=float), n0)
        best_h = h0.copy()
        best_c5 = compute_c5(h0, n0)
        best_n = n0

    def polish(h_cur, n, remaining):
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
        c = compute_c5(h_cur, n)

        for obj_fn in [compute_c5, lse_obj, lse_obj2, compute_c5]:
            if time.time() - t0 > t0 + budget_s - 30:
                break
            for trial in range(2):
                if time.time() - t0 > t0 + budget_s - 30:
                    break
                try:
                    if trial == 0:
                        x0 = h_cur.copy()
                    else:
                        x0 = h_cur + rng.normal(0, 0.01, n)
                        x0 = repair(x0, n)
                    r = minimize(obj_fn, x0, method='SLSQP',
                                 bounds=bounds, constraints=cons, args=(n,),
                                 options={'maxiter': 500, 'ftol': 1e-15})
                    hc = repair(r.x, n)
                    cc = compute_c5(hc, n)
                    if cc < c:
                        h_cur, c = hc, cc
                except Exception:
                    pass

        step = 0.005
        for _ in range(200):
            if time.time() - t0 > t0 + budget_s - 20:
                break
            hg = grad_step(h_cur, n, step)
            cg = compute_c5(hg, n)
            if cg < c - 1e-16:
                h_cur, c = hg, cg
                step = max(0.0005, step * 0.95)
            else:
                step *= 0.5
                if step < 0.0001:
                    break
        return h_cur, c

    # Primary: polish initial solution
    if init is not None:
        n0 = len(init)
        h0 = repair(np.array(init, dtype=float), n0)
        h0, c0 = polish(h0, n0, budget_s * 0.4)
        if c0 < best_c5:
            best_h, best_c5, best_n = h0, c0, n0

    # Try other n values with various starts
    n_list = []
    for nv in [60, 80, 50, 90, 100, 70, 40]:
        if nv != best_n:
            n_list.append(nv)

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]

        for restart in range(4):
            if time.time() - t0 > budget_s - 80:
                break
            if restart == 0:
                if init is not None:
                    x_old = np.linspace(0, 2, len(init))
                    x_new = np.linspace(0, 2, n)
                    h_s = np.interp(x_new, x_old, np.array(init, dtype=float))
                else:
                    h_s = np.full(n, 0.5) + rng.normal(0, 0.1, n)
            elif restart == 1:
                h_s = np.linspace(1.0, 0.2, n)
            elif restart == 2:
                t = np.linspace(0, 4 * np.pi, n)
                h_s = 0.5 + 0.3 * np.sin(t) + 0.1 * np.cos(2.0 * t)
            else:
                h_s = np.full(n, 0.5) + rng.normal(0, 0.25, n)

            h_s = repair(h_s, n)
            try:
                r = minimize(compute_c5, h_s, method='SLSQP',
                           bounds=bounds, constraints=cons, args=(n,),
                           options={'maxiter': 300, 'ftol': 1e-14})
                h_s = repair(r.x, n)
            except Exception:
                pass

            try:
                r2 = minimize(lse_obj, h_s, method='SLSQP',
                            bounds=bounds, constraints=cons, args=(n,),
                            options={'maxiter': 200, 'ftol': 1e-13})
                h_l = repair(r2.x, n)
                if compute_c5(h_l, n) < compute_c5(h_s, n):
                    h_s = h_l
            except Exception:
                pass

            c_s = compute_c5(h_s, n)
            if c_s < best_c5 and time.time() - t0 < budget_s - 40:
                h_s, c_s = polish(h_s, n, 30)
                if c_s < best_c5:
                    best_h, best_c5, best_n = h_s.copy(), c_s, n

    # Final intensive polish on best
    if best_h is not None and time.time() - t0 < budget_s - 20:
        n = best_n
        h_final, c_final = polish(best_h, n, budget_s * 0.2)
        if c_final < best_c5:
            best_h, best_c5 = h_final, c_final

    # Pair transfer fine-tuning
    if best_h is not None and time.time() - t0 < budget_s - 10:
        remaining = budget_s - 10 - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = pair_search(best_h, best_n, 30, remaining)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
