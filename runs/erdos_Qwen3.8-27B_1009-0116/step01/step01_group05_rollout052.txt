import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _sobj(x):
    cv = _corr(x, _P['dx'])
    mx = cv.max()
    eps = _P['eps']
    t = (cv - mx) / eps
    ms = t.max()
    return mx + eps * (ms + np.log(np.sum(np.exp(t - ms))))

def _sgrad(x):
    cv = _corr(x, _P['dx'])
    mx = cv.max()
    eps = _P['eps']
    dx = _P['dx']
    t = (cv - mx) / eps
    ms = t.max()
    w = np.exp(t - ms)
    w /= w.sum()
    n = len(x)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _eq_fun(x):
    return x.sum() - _P['target']

def _eq_jac(x):
    return np.ones(len(x))

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp(h0, n, target, eps, maxiter):
    _P['dx'] = 2.0 / n
    _P['n'] = n
    _P['target'] = float(target)
    _P['eps'] = float(eps)
    try:
        res = minimize(
            _sobj, h0.copy(), jac=_sgrad,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
        )
    except Exception:
        return h0.copy()
    return _proj(res.x, target)

def _subgrad_step(h, dx, target, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-10)[0]
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    lr_cur = lr
    best_h = h.copy()
    best_c = mx
    for _ in range(10):
        h_new = _proj(h - lr_cur * g, target)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        lr_cur *= 0.5
    return best_h, best_c

def _optimize_phase(h_init, dx, n, target, rng, t_end):
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)
    h = h_init.copy()

    eps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for eps in eps_list:
        if time.time() > t_end - 15:
            break
        iters = 700 if eps > 0.005 else 500
        h_new = _slsqp(h, n, target, eps, maxiter=iters)
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new

    lr = 0.4
    for it in range(8000):
        if time.time() > t_end - 8:
            break
        h_new, c_new = _subgrad_step(h, dx, target, lr)
        c_cur = _c5(h, dx)
        if c_new < c_cur - 1e-14:
            lr = min(lr * 1.04, 4.0)
        else:
            lr = max(lr * 0.93, 0.005)
        h = h_new
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if it % 200 == 150:
            h = _proj(best_h + rng.normal(0, 0.025, n), target)

    if time.time() < t_end - 5:
        for eps in [0.00005, 0.00002, 0.00001, 0.000005]:
            if time.time() > t_end - 2:
                break
            h2 = _slsqp(best_h, n, target, eps, maxiter=400)
            c2 = _c5(h2, dx)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

    return best_h, best_c5

def _quick_opt(h_init, dx, n, target, t_end):
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)
    h = h_init.copy()
    for eps in [0.01, 0.003, 0.001, 0.0003, 0.0001]:
        if time.time() > t_end - 3:
            break
        h_new = _slsqp(h, n, target, eps, maxiter=300)
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    lr = 0.3
    for it in range(1500):
        if time.time() > t_end - 2:
            break
        h_new, c_new = _subgrad_step(h, dx, target, lr)
        c_cur = _c5(h, dx)
        if c_new < c_cur:
            lr = min(lr * 1.03, 3.0)
        else:
            lr = max(lr * 0.9, 0.01)
        h = h_new
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.RandomState(seed)

    n = 70
    dx = 2.0 / n
    target = n / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
    else:
        h0 = np.full(n, 0.5)
    h0 = _proj(h0, target)

    best_h, best_c5 = _optimize_phase(h0, dx, n, target, rng, deadline - 180)

    # Random restarts with structured perturbations
    restart_end = deadline - 25
    n_restarts = 0
    while time.time() < restart_end:
        n_restarts += 1
        roll = rng.random()
        if roll < 0.6:
            amp = rng.uniform(0.015, 0.12)
            h_pert = _proj(best_h + rng.normal(0, amp, n), target)
        elif roll < 0.8:
            # Structured: add small sinusoidal perturbation
            freq = rng.randint(2, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            h_pert = _proj(best_h + pert + rng.normal(0, 0.01, n), target)
        else:
            # Coarse perturbation
            amp = rng.uniform(0.08, 0.25)
            h_pert = _proj(best_h + rng.normal(0, amp, n), target)

        c_pert = _c5(h_pert, dx)
        if c_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c_pert

        sub_time = min(12.0, restart_end - time.time() - 3.0)
        if sub_time > 4:
            h2, c2 = _quick_opt(h_pert, dx, n, target, time.time() + sub_time)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    # Final polish
    if time.time() < deadline - 10:
        fh, fc = _optimize_phase(best_h, dx, n, target, rng, deadline - 2)
        if fc < best_c5:
            best_h, best_c5 = fh, fc

    return best_h, best_c5, n
