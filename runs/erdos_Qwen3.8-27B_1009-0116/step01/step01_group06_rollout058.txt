import numpy as np
from scipy.optimize import minimize
import time

NP = 70
DX = 2.0 / NP

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lse_target(x, n, dx, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    sh = np.clip(corr - mc, -400.0, 0.0)
    ev = np.exp(sh)
    return (mc + np.log(ev.sum())) / alpha

def lse_gradient(x, n, dx, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    sh = np.clip(corr - mc, -400.0, 0.0)
    ev = np.exp(sh)
    ss = ev.sum()
    if ss < 1e-300:
        return np.zeros(n)
    w = ev / ss / alpha
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-15:
            continue
        lg = ki - (n - 1)
        if lg >= 0:
            g[lg:] += wi * (1.0 - x[:n - lg])
            g[:n - lg] -= wi * x[lg:]
        else:
            ln = -lg
            g[:n - ln] += wi * (1.0 - x[ln:])
            g[ln:] -= wi * x[:n - ln]
    return g

def eqfun_fixed(x):
    return x.sum() - float(NP) / 2.0

def eqjac_fixed(x):
    return np.ones(NP)

def hardobj(x, n, dx):
    return float(np.max(np.correlate(x, 1.0 - x, mode='full')) * dx)

def subgrad_step(x, n, dx, alpha, lr):
    """One step of smoothed subgradient descent with projection."""
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    sh = np.clip(corr - mc, -400.0, 0.0)
    ev = np.exp(sh)
    w = ev / ev.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-15:
            continue
        lg = ki - (n - 1)
        if lg >= 0:
            g[lg:] += wi * (1.0 - x[:n - lg])
            g[:n - lg] -= wi * x[lg:]
        else:
            ln = -lg
            g[:n - ln] += wi * (1.0 - x[ln:])
            g[ln:] -= wi * x[:n - ln]
    g *= dx
    return x - lr * g
