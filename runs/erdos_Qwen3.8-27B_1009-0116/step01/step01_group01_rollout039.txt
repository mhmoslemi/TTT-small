def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()

        else:

            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()

                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def solve_1d_minimax(a, b):
    """minimize_{v in [0,1]} max_m (b[m] + a[m]*v)"""
    if np.all(a >= -1e-15):
        return 0.0
    if np.all(a <= 1e-15):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(b + a * m1) <= np.max(b + a * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def coord_descent_fast(h, n, max_sweeps=200, seed=0, deadline=None):
    h = h.astype(float).copy()
    best = compute_c5(h)
    rng = np.random.RandomState(seed)
    m_arr = np.arange(2 * n - 1)
    n1 = n - 1
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c_curr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        max_change = 0.0
        for j in order:
            old_v = h[j]
            idx1 = j + m_arr - n1
            idx2 = j - m_arr + n1
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2 * n - 1)
            a[valid1] += (1.0 - h[idx1[valid1]])
            a[valid2] -= h[idx2[valid2]]
            b = c_curr - a * old_v
            v_opt = solve_1d_minimax(a, b)
            if abs(v_opt - old_v) > 1e-10:
                h[j] = v_opt
                c_curr += a * (v_opt - old_v)
                max_change = max(max_change, abs(v_opt - old_v))
        h = repair(h, n)
        v = compute_c5(h)
        if v < best:
            best = v
        if max_change < 1e-9:
            break
    return h, best


def try_binary(h, n):
    best_h = h.copy()
    best_v = compute_c5(h)
    target_ones = int(np.round(n * 0.5))
    for thresh in [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
        h_bin = (h >= thresh).astype(float)
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            oi = np.where(h_bin == 1)[0]
            ord_ = oi[np.argsort(h[oi])]
            for idx in ord_[:n_ones - target_ones]:
                h_bin[idx] = 0.0
        elif n_ones < target_ones:
            zi = np.where(h_bin == 0)[0]
            ord_ = zi[np.argsort(-h[zi])]
            for idx in ord_[:target_ones - n_ones]:
                h_bin[idx] = 1.0
        v = compute_c5(h_bin)
        if v < best_v:
            best_v = v
            best_h = h_bin.copy()
    return best_h, best_v


def _smooth_obj(x, n, t_param, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(c)
    smooth = (cmax + np.log(np.sum(np.exp((c - cmax) * t_param))) / t_param) * dx
    pen = penalty * (x.sum() - n * 0.5) ** 2
    return smooth + pen


def _check(t0, budget_s, frac, buf=3.0):
    return (budget_s * frac - (time.time() - t0)) > buf
