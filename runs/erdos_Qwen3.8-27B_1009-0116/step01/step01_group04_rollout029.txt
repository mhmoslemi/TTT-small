def project_grid(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    # Actually, for step functions on [0,2] with n points, each point represents
    # an interval of width 2/n. Let me just use linear interpolation.
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = np.linspace(0.0, 2.0, n_src)
    x_interp = np.interp(xd, xp, h_src)
    return repair(x_interp, n_dst)

def subgrad_optimize(h0, n, dx, time_limit, rng, lr_init=0.1, tau_init=0.5, k_top=15, momentum=0.95):
    """Subgradient descent with momentum and softmax weighting."""
    t0 = time.time()
    h = repair(h0, n).copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    
    for it in range(200000):
        if time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        cm = np.max(corr)
        
        if it % 100 == 0:
            val = cm * dx
            if val < best_val:
                best_val = val
                best_h = h.copy()
        
        # Softmax weights on top-k shifts
        top_idx = np.argsort(corr)[-k_top:]
        tv = corr[top_idx]
        w = np.exp((tv - tv.max()) / tau)
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for ii in range(k_top):
            k = top_idx[ii]
            wk = w[ii]
            s = k - (n - 1)
            if s >= 0:
                grad[:n-s] += wk * (1.0 - h[s:n])
                grad[s:] -= wk * h[:n-s]
            else:
                t_ = -s
                grad[t_:] += wk * (1.0 - h[:n-t_])
                grad[:n-t_] -= wk * h[t_:]
        
        # Momentum step
        v = momentum * v - lr * grad
        h_new = h + v
        h = repair(h_new, n)
        
        # Periodic decay
        if it % 500 == 0:
            lr *= 0.97
            tau *= 0.98
            momentum = min(0.99, momentum + 0.001)
    
    # Final check
    val = compute_c5(h, dx)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    
    return best_h, best_val

def slsqq_step(x0, n, dx, tau=0.01, maxiter=200):
    """SLSQP step with log-sum-exp smoothing."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        if tau < 1e-6:
            return m * dx
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    
    def cfun(x):
        return np.sum(x) - n / 2.0
    
    def cjac(x):
        return np.ones(n)
    
    try:
        r = minimize(obj, x0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x, n)
    except Exception:
        return x0.copy()
