def _slsqp_opt(h0, dx, n, target, eps, maxiter=500):
    """Optimize using SLSQP with softmin objective."""
    def obj(x):
        cv = np.correlate(x, 1.0 - x, mode="full") * dx
        mx = cv.max()
        return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))
    
    def grad(x):
        cv = np.correlate(x, 1.0 - x, mode="full") * dx
        mx = cv.max()
        w = np.exp((cv - mx) / eps)
        w /= w.sum()
        g = np.zeros(n)
        j = np.arange(n)

            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():

            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= wt * x[jp[vp]]
        return g
    
    def constraint(x):
        return x.sum() - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    try:

import numpy as np
from scipy.optimize import minimize
import time

_S = {}

def _obj(x):
    dx, eps = _S['dx'], _S['eps']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _grad(x):
    dx, eps, n = _S['dx'], _S['eps'], _S['n']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    ja = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = ja - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = ja + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _cons(x):
    return x.sum() - _S['t']

def _cons_jac(x):
    return np.ones(_S['n'])

def _c5(h, dx):
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    try:
        r = minimize(_obj, h0, jac=_grad,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}],
                     method='SLSQP', options={'maxiter': iters, 'ftol': 1e-16})
        return _proj(r.x, tgt)
    except Exception:
        return _proj(h0, tgt)

def _substep(h, dx, tgt, lr):

    cv = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = cv.max()
    active = np.where(cv >= mx - 1e-9)[0]
    n = len(h)
    g = np.zeros(n)
    ja = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = ja - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = ja + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    return _proj(h - lr * g, tgt)

def _bin_search(h0, dx, n, rng, nit, tend):
    """Binary greedy + SA local search."""
    tgt = n // 2
    hb = np.where(h0 >= 0.5, 1.0, 0.0)

        cnt = hb.sum()
        if cnt == tgt:

            i = rng.choice(np.where(hb == 1.0)[0])

        else:
            i = rng.choice(np.where(hb == 0.0)[0])

            hb[i], hb[j] = 1.0, 0.0

    h = h.copy()
    bc = _c5(h, dx)
    for r in range(rounds):

                    bc = c
                    imp = True
                    break
            else:
                h[i] = old
                h = _proj(h, tgt)
        if not imp:
            break
    return h, bc

import numpy as np
from scipy.optimize import minimize
import time

_S = {}

def _obj(x):
    dx, eps = _S['dx'], _S['eps']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _grad(x):
    dx, eps, n = _S['dx'], _S['eps'], _S['n']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    ja = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = ja - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = ja + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _cons(x):
    return x.sum() - _S['t']

def _cons_jac(x):
    return np.ones(_S['n'])

def _c5(h, dx):
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def _proj(h, tgt):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp(h0, dx, n, tgt, eps, iters):
    _S.update(dx=dx, eps=eps, n=n, t=float(tgt))
    try:
        r = minimize(_obj, h0, jac=_grad,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}],
                     method='SLSQP', options={'maxiter': iters, 'ftol': 1e-16})
        return _proj(r.x, tgt)
    except Exception:
        return _proj(h0, tgt)

def _substep(h, dx, tgt, lr):
    cv = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = cv.max()
    active = np.where(cv >= mx - 1e-9)[0]
    n = len(h)
    g = np.zeros(n)
    ja = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = ja - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = ja + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    return _proj(h - lr * g, tgt)
