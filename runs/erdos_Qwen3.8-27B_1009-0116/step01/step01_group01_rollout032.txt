import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_feasible(h, target, n):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on shift."""
    h = h.copy()
    lo, hi = -max(abs(h).max(), 1.0) * 2, max(abs(h).max(), 1.0) * 2
    for _ in range(60):
        mid = (lo + hi) / 2
        h_test = np.clip(h + mid, 0.0, 1.0)
        s = h_test.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag, n):
    """Compute ∂c_lag/∂h for a given lag."""
    g = np.zeros(n)
    if lag >= 0:
        # c_lag = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
        # g[j] = (1-h[j+lag])*[0<=j<=n-1-lag] - h[j-lag]*[lag<=j<=n-1]
        valid1 = slice(0, n - lag)  # j from 0 to n-1-lag
        valid2 = slice(lag, n)  # j from lag to n-1
        g[valid1] += (1.0 - h[lag:n])  # (1-h[j+lag]) for j in [0, n-1-lag]
        g[valid2] -= h[0:n-lag]  # -h[j-lag] for j in [lag, n-1]
    else:
        m = -lag
        # c_lag = sum_{i=m}^{n-1} h[i](1-h[i-m])
        # g[j] = (1-h[j-m])*[m<=j<=n-1] - h[j+m]*[0<=j<=n-1-m]
        valid1 = slice(m, n)  # j from m to n-1
        valid2 = slice(0, n - m)  # j from 0 to n-1-m
        g[valid1] += (1.0 - h[0:n-m])  # (1-h[j-m]) for j in [m, n-1]
        g[valid2] -= h[m:n]  # -h[j+m] for j in [0, n-1-m]
    return g

def subgradient_step(h, n, alpha, K=5):
    """One step of projected subgradient using top-K lags."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top K indices
    topk = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in topk:
        lag = k - (n - 1)
        g += lag_gradient(h, lag, n)
    g /= K
    h_new = h - alpha * g
    h_new = project_onto_feasible(h_new, n * 0.5, n)
    return h_new, c

            if time_left(0.97) < 3:

            alpha = 0.15 / (1.0 + it / 1000.0) ** 0.5
            h, c = subgrad_step(h, n_f, alpha, 8)
            c5 = float(np.max(c) * 2.0 / n_f)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    best_h = repair(best_h.copy(), len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
