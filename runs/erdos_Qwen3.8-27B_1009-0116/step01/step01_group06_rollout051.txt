    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_diff = np.exp((corr - max_corr) / T)

            valid_m = range(s, n)
            grad[valid_m] += w[k] * (1.0 - h[:n-s])
            # m+s < n => m < n-s
            valid_m2 = range(0, n-s)
            grad[valid_m2] -= w[k] * h[s:]
        else:

            t = -s
            # m-s = m+t >= 0 always; m+t < n => m < n-t
            valid_m = range(0, n-t)
            grad[valid_m] += w[k] * (1.0 - h[t:])
            # m+s = m-t >= 0 => m >= t; m-t < n always
            valid_m2 = range(t, n)
            grad[valid_m2] -= w[k] * h[:n-t]
    
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = float(n) / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_eval(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return float((mc + T * np.log(np.sum(np.exp((corr - mc) / T)))) * dx)

def smooth_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ed = np.exp((corr - mc) / T)
    se = np.sum(ed)
    w = ed / se
    # Only use lags with significant weight
    thresh = 1e-8
    active = w > thresh
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        wk = w[k]
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n - t]
    return dx * grad

def slsqp_smooth(h, n, dx, T, maxiter=500):
    target = float(n) / 2.0
    def obj(x):
        return smooth_eval(x, dx, T)
    def obj_grad(x):
        return smooth_grad(x, dx, T)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def slsqp_exact(h, n, dx, maxiter=500):
    target = float(n) / 2.0
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_eval_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ed = np.exp((corr - mc) / T)
    se = ed.sum()

        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n - t]
    return val * dx, dx * grad

            idx_i = ones_idx.index(i)
            idx_j = zeros_idx.index(j)
            ones_idx[idx_i] = j
            zeros_idx[idx_j] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 70
    dx = 2.0 / n

    best_h = project_feasible(np.array(initial_h_values, dtype=float), n)
    best_c5 = eval_c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Smooth annealing SLSQP
    for T in [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]:
        if rem() < 40:
            break
        h, c = slsqp_smooth(best_h, n, dx, T, maxiter=300)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: Exact SLSQP polish
    if rem() > 25:
        h, c = slsqp_exact(best_h, n, dx, maxiter=1500)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Binary SA to find good discrete structure
    if rem() > 40:
        h_bin, c_bin = binary_sa(n, dx, best_h, max_iter=25000)
        h_ref = h_bin.copy()
        h_ref, c_ref = slsqp_smooth(h_ref, n, dx, 0.08, maxiter=200)
        h_ref, c_ref = slsqp_exact(h_ref, n, dx, maxiter=500)
        if c_ref < best_c5:
            best_h, best_c5 = h_ref, c_ref
        if c_bin < best_c5:
            best_h, best_c5 = h_bin, c_bin

    # Phase 4: Nelder-Mead polish
    if rem() > 25:
        h, c = nelder_polish(best_h, n, dx, maxiter=3000)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 5: Second round of smooth annealing from improved point
    if rem() > 60:
        for T in [0.3, 0.15, 0.06, 0.02]:
            if rem() < 20:
                break
            h, c = slsqp_smooth(best_h, n, dx, T, maxiter=250)
            if c < best_c5:
                best_h, best_c5 = h, c
        if rem() > 20:
            h, c = slsqp_exact(best_h, n, dx, maxiter=1000)
            if c < best_c5:
                best_h, best_c5 = h, c

    # Phase 6: Extensive random restarts
    attempt = 0
    while rem() > 4:
        attempt += 1
        if attempt <= 15:
            h0 = best_h + 0.08 * np.random.randn(n)
        elif attempt <= 35:
            h0 = best_h + 0.03 * np.random.randn(n)
        elif attempt <= 55:
            h0 = best_h + 0.15 * np.random.randn(n)
        else:
            h0 = np.random.rand(n)
        h0 = project_feasible(h0, n)
        h, c = slsqp_smooth(h0, n, dx, 0.04, maxiter=100)
        if c < best_c5 + 0.001:
            h, c = slsqp_exact(h, n, dx, maxiter=300)
        if c < best_c5:
            best_h, best_c5 = h, c
            # Re-run smooth polish from new best
            if rem() > 20:
                h2, c2 = slsqp_smooth(h, n, dx, 0.02, maxiter=200)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
