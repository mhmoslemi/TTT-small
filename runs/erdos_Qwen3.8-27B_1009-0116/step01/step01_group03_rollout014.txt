import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    shifted = np.clip((corr - m) * beta, -500, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_optimize(h, n, ts, beta, maxiter=500):
    def obj(x):
        return smooth_obj(x, n, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

    """Build matrix where row i is h[idx_i + shifts] + h[idx_i - shifts] with boundary=0."""
    m = len(idx_set)
    L = len(shifts)
    M = np.zeros((m, L))
    for i, j in enumerate(idx_set):
        js = j + shifts
        v1 = (js >= 0) & (js < n)
        M[i, v1] += h[js[v1]]
        js2 = j - shifts
        v2 = (js2 >= 0) & (js2 < n)
        M[i, v2] += h[js2[v2]]
    return M

        corr = np.correlate(h, 1.0 - h, mode="full")
        best_max = float(corr.max())
    return h, best_max * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    def elapsed():
        return time.time() - t_start
    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # === Stage 1: Optimize at n=140 ===
    n = n_init
    ts = n / 2.0
    h = h_init.copy()

    for beta in [5.0, 15.0, 40.0, 100.0, 300.0]:
        if remaining() < 200:
            break
        h = slsqp_optimize(h, n, ts, beta, maxiter=400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Binary local search
    if remaining() > 150:
        h_bin = np.round(h)
        h_ls, c5_ls = swap_local_search(h_bin, n, ts, max_time=50)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls.copy()

    # SLSQP polish after binary
    if remaining() > 100:
        for beta in [50.0, 200.0, 500.0]:
            if remaining() < 80:
                break
            h = slsqp_optimize(h, n, ts, beta, maxiter=300)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # More binary search with perturbation escapes
    if remaining() > 80:
        h_b = np.round(h)
        for attempt in range(3):
            if remaining() < 80:
                break
            # Perturb: random 5 swaps
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            for _ in range(5):
                j = int(rng.choice(ones))
                k = int(rng.choice(zeros))
                h_b[j], h_b[k] = 1.0 - h_b[j], 1.0 - h_b[k]
            h_ls, c5_ls = swap_local_search(h_b, n, ts, max_time=20)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h = h_ls.copy()

    # === Stage 2: Try n=280 ===
    if remaining() > 120:
        n2 = 280
        ts2 = n2 / 2.0
        xo = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        xn = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h2 = np.interp(xn, xo, h)
        h2 = project_feasible(h2, ts2)

        for beta in [10.0, 50.0, 200.0]:
            if remaining() < 80:
                break
            h2 = slsqp_optimize(h2, n2, ts2, beta, maxiter=300)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

        if remaining() > 80:
            h2_bin = np.round(h2)
            h2_ls, c5_2_ls = swap_local_search(h2_bin, n2, ts2, max_time=40)
            if c5_2_ls < best_c5:
                best_c5 = c5_2_ls
                best_h = h2_ls.copy()

        if remaining() > 30:
            h2 = slsqp_optimize(h2, n2, ts2, 300.0, maxiter=200)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # === Stage 3: Try n=100 (different resolution) ===
    if remaining() > 60:
        n3 = 100
        ts3 = n3 / 2.0
        h3 = np.zeros(n3)
        h3[rng.choice(n3, 50, replace=False)] = 1.0
        for beta in [20.0, 100.0]:
            if remaining() < 40:
                break
            h3 = slsqp_optimize(h3, n3, ts3, beta, maxiter=300)
        if remaining() > 30:
            h3_ls, c5_3 = swap_local_search(np.round(h3), n3, ts3, max_time=20)
            c5_3 = compute_c5(h3_ls)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3_ls.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
