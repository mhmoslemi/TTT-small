from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(g, target_sum):
    n = len(g)
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    w = np.exp(sh)
    w_sum = w.sum()
    w = w / w_sum
    val = (m + tau * np.log(w_sum)) * dx
    g = np.full(n, dx)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - (n - 1)
        js = max(0, lag)
        je = min(n, n + lag)
        if js < je:
            g[js:je] -= dx * wk * h[js - lag:je - lag]
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            g[js2:je2] -= dx * wk * h[js2 + lag:je2 + lag]
    return g, val, m * dx

def fista_optimize(h0, dx, tau, n_iter, t_limit):
    n = len(h0)
    target = n / 2.0
    x = project_constraint(h0, target)
    z = x.copy()
    t = 1.0
    step = max(tau / 6.0, 1e-4)
    best_x = x.copy()
    best_val = compute_c5(x, dx)
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > t_limit:
            break
        grad, val_smooth, val_exact = smooth_max_and_grad(z, dx, tau)
        x_new = project_constraint(z - step * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = x_new + ((t - 1.0) / t_new) * (x_new - x)
        z = project_constraint(z, target)
        x = x_new
        t = t_new
        if val_exact < best_val:
            best_val = val_exact
            best_x = x.copy()
    return best_x, best_val

def lbfgsb_optimize(h0, dx, tau, n_iter, t_limit):
    n = len(h0)
    target = n / 2.0
    pen = 50.0

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        val = (m + tau * np.log(np.sum(np.exp(sh)))) * dx
        val += pen * (np.sum(h) - target) ** 2
        return val

    t0 = time.time()
    bd = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h0, method='L-BFGS-B', bounds=bd,
                     options={'maxiter': n_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                              'maxfun': n_iter * 3})
        x = project_constraint(r.x, target)
    except Exception:
        x = project_constraint(h0, target)
    if time.time() - t0 > t_limit:
        x = project_constraint(h0, target)
    val = compute_c5(x, dx)
    return x, val

def slsqp_polish(h0, dx, tau, t_limit):
    n = len(h0)
    target = n / 2.0
    global _gN, _gdx, _gtau
    _gN = n
    _gdx = dx
    _gtau = tau

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
        return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx

    def _cfun(x):
        return np.sum(x) - target

    def _cjac(x):
        return np.ones(_gN)

    t0 = time.time()
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]

    try:

        x = project_constraint(h0, target)
