import numpy as np
from scipy.optimize import minimize
import time

_p_n = 0
_p_target = 0.0
_p_beta = 1.0

def c5_val(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * dx

def fix(h, n):
    t = n * 0.5
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-11) & (h < 1.0 - 1e-11)
        c = int(m.sum())
        if c == 0:
            break
        h[m] = np.clip(h[m] + d / c, 0.0, 1.0)
    return h

def _fmax(x):
    return c5_val(x, _p_n)

def _flse(x):
    dx = 2.0 / _p_n
    cr = np.correlate(x, 1.0 - x, mode='full')
    b = _p_beta
    s = b * cr
    mx = s.max()
    return (mx + np.log(np.exp(s - mx).sum())) / b * dx

def _feq(x):
    return x.sum() - _p_target

def _feqj(x):
    return np.ones_like(x)

def opt_slsqp(h0, n, beta, maxiter, ftol):
    global _p_n, _p_target, _p_beta
    _p_n = n
    _p_target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _feq, 'jac': _feqj}]
    f = _flse if beta > 0 else _fmax
    if beta > 0:
        _p_beta = beta
    r = minimize(f, h0, method='SLSQP', bounds=bounds, constraints=cons,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return fix(r.x, n)

def coord_descent(h, n, n_rounds, time_limit):
    t_end = time.time() + time_limit
    best = c5_val(h, n)
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for dv in (0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 1.0 - h[i], -h[i]):
                nv = h[i] + dv
                if nv < -1e-9 or nv > 1.0 + 1e-9:
                    continue
                hp = h.copy()
                hp[i] = np.clip(nv, 0.0, 1.0)
                hp = fix(hp, n)
                cv = c5_val(hp, n)
                if cv < best - 1e-14:
                    h, best, improved = hp, cv, True
                    break
        if not improved and rnd > 30:
