    return _corr(h, dx).max()

            hi = mid

def _sobj(x, dx, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad(x, dx, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(x)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _efun(x, target):
    return x.sum() - target

def _ejac(x, target):
    return np.ones(len(x))

def _slsqp_polish(h0, dx, n, target, eps, maxiter):
    def obj(x):
        return _sobj(x, dx, eps)
    def jac(x):
        return _sgrad(x, dx, eps)
    def cfun(x):
        return _efun(x, target)
    def cjac(x):
        return _ejac(x, target)
    try:
        res = minimize(obj, h0, jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _proj(res.x, target)
    except Exception:
        return h0

    return out

        h_p = _proj(h_p + rng.normal(0, amp * 0.5, n), target)
        c_p = _c5(h_p, dx)
        h_start = h_p if c_p < best_c5 else best_h.copy()
        sub_end = min(time.time() + 20.0, deadline - 5.0)
        h_q, c_q = _cd_run(h_start, dx, target, 200, sub_end, mu=0.0)
        if c_q < best_c5:
            best_c5 = c_q
            best_h = h_q.copy()
        eps_try = 0.001
        if time.time() < deadline - 8:
            h_q2 = _slsqp_polish(best_h, dx, n, target, eps_try, 150)
            c_q2 = _c5(h_q2, dx)
            if c_q2 < best_c5:
                best_c5 = c_q2
                best_h = h_q2.copy()

    # Phase 6: Final coordinate descent squeeze
    if time.time() < deadline - 3:
        h_final, c_final = _cd_run(best_h, dx, target, 300, deadline, mu=0.0)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    return best_h, best_c5, n
