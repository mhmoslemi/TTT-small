def project_feasible(h, n):

    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
            else:
                h[np.argmax(room)] += diff
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
            else:
                h[np.argmin(h)] += diff
        h = np.clip(h, 0.0, 1.0)
        # Final fix
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.where(h < 1.0)[0]
                frac = min(int(np.ceil(d2)), len(idx))

                    h[idx[i]] += adj

                for i in range(frac):
                    adj = min(-d2, h[idx[i]])
                    h[idx[i]] -= adj
                    d2 += adj

    m = np.max(corr)
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m)))) + m

def lse_gradient(h, dx, beta):

    m = np.max(corr)
    exps = np.exp(beta * (corr - m))

    grad = np.zeros(n)

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            if d2 > 0:
                idx = np.where(h < 1.0)[0]
                for i in idx[:50]:
                    a = min(d2, 1.0 - h[i])
                    h[i] += a
                    d2 -= a
                    if abs(d2) < 1e-12:
                        break
            else:
                idx = np.where(h > 0.0)[0]
                for i in idx[:50]:
                    a = min(-d2, h[i])
                    h[i] -= a
                    d2 += a
                    if abs(d2) < 1e-12:
                        break
    return h

def lse_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()
    obj = m + np.log(exps.sum()) / beta
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wk = w[lag + n - 1]
        if wk < 1e-14:
            continue
        if lag >= 0:
            neg = np.zeros(n)
            ngl = n - lag
            if ngl > 0:
                neg[lag:] = 1.0 - h[:ngl]
                pos = np.zeros(n)
                pos[:ngl] = h[lag:]
        else:
            L = -lag
            ngl = n - L
            if ngl > 0:
                neg[:ngl] = 1.0 - h[L:]
                pos = np.zeros(n)
                pos[L:] = h[:ngl]
        grad += wk * (neg - pos)
    grad *= dx
    return obj, grad

def slsqp_opt(h0, n, dx, maxiter, ftol, beta=50.0):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: lse_obj_grad(x, dx, beta)[0], h0.copy(), method='SLSQP',
                      jac=lambda x: lse_obj_grad(x, dx, beta)[1],
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h0, eval_c5(h0, dx)

def binary_swap_search(h, n, dx, max_rounds=300):
    target = n // 2
    hb = make_binary(h, n)
    best_hb = hb.copy()
    best_c5 = eval_c5(hb, dx)
    
    for rnd in range(max_rounds):
        curr_c5 = eval_c5(hb, dx)
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        best_delta = 1e-15
        best_pair = None
        for i in ones:
            for j in zeros:
                hb[i], hb[j] = 0.0, 1.0
                c = eval_c5(hb, dx)
                hb[i], hb[j] = 1.0, 0.0
                d = curr_c5 - c
                if d > best_delta:
                    best_delta = d
                    best_pair = (i, j)
        if best_pair is not None:
            hb[best_pair[0]] = 0.0
            hb[best_pair[1]] = 1.0
            best_hb = hb.copy()
            best_c5 = eval_c5(hb, dx)
        else:
            break
    return best_hb, best_c5

def sim_anneal_binary(h, n, dx, max_iter=10000):
    target = n // 2
    hb = make_binary(h, n)
    best_hb = hb.copy()
    best_c5 = eval_c5(hb, dx)
    curr_c5 = best_c5
    rng = np.random.RandomState(42)
    
    for it in range(max_iter):
        frac = it / max_iter
        T = 0.05 * (0.01) ** frac
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i], hb[j] = 0.0, 1.0
        new_c5 = eval_c5(hb, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_hb = hb.copy()
        else:
            hb[i], hb[j] = 1.0, 0.0
    return best_hb, best_c5
