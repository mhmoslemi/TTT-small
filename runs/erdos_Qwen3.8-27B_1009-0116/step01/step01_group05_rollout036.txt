from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _smooth_obj(x, dx, eps, target):
    cv = _corr(x, dx)
    mx = cv.max()
    penalty = 100.0 * (x.sum() - target) ** 2
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps))) + penalty

def _smooth_grad(x, dx, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(x)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _slsqp_run(h0, dx, n, target, eps, maxiter=500):
    def fun(x):
        return _smooth_obj(x, dx, eps, target)
    def jac(x):
        return _smooth_grad(x, dx, eps)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    res = minimize(fun, h0, jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   method='SLSQP',

            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _ones_jac(x):
    return np.ones(len(x))

def _g_cfun(x):
    return x.sum() - _G_TGT

def _g_fun(x):
    cv = _corr(x, _G_DX)
    mx = cv.max()
    return mx + _G_EPS * np.log(np.sum(np.exp((cv - mx) / _G_EPS)))

def _g_jac(x):
    cv = _corr(x, _G_DX)
    mx = cv.max()
    w = np.exp((cv - mx) / _G_EPS)
    w /= w.sum()
    n = _G_N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _G_DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _slsqp(h0, dx, n, target, eps, maxiter=400):
    global _G_DX, _G_N, _G_EPS, _G_TGT
    _G_DX = dx
    _G_N = n
    _G_EPS = float(eps)
    _G_TGT = float(target)
    try:
        res = minimize(_g_fun, h0, jac=_g_jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _g_cfun, 'jac': _ones_jac}],
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-15})
        return _proj(res.x, target)
    except Exception:
        return _proj(h0, target)

def _momentum_step(h, dx, n, target, lr, mom, t_end, max_iter=5000):
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    v = np.zeros(n)
    j = np.arange(n)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        cv = _corr(h, dx)
        mx = cv.max()
        active = np.where(cv >= mx - 1e-8)[0]
        g = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            wt = dx / max(len(active), 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g[vm] += wt * (1.0 - h[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= wt * h[jp[vp]]
        v = mom * v - lr * g
        h_new = _proj(h + v, target)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.9995, 0.002)
        if it % 500 == 400:
            v *= 0.5
    return best_h, best_c

def _anneal01(h_init, n, dx, target, rng, t_end):
    n_ones = int(round(target))
    h01 = np.zeros(n)
    h01[np.argsort(h_init)[-n_ones:]] = 1.0
    best = h01.copy()
    best_c = _c5(h01, dx)
    temp = 0.04
    step = 0
    while time.time() < t_end:
        step += 1
        temp = max(0.003, 0.04 * (0.9992 ** step))
        ones = np.where(h01 == 1.0)[0]
        zeros = np.where(h01 == 0.0)[0]
        a = rng.choice(ones)
        b = rng.choice(zeros)
        h01[a] = 0.0
        h01[b] = 1.0
        c_new = _c5(h01, dx)
        if c_new < best_c or rng.rand() < np.exp(-(c_new - best_c) / temp):
            if c_new < best_c:
                best = h01.copy()
                best_c = c_new
        else:
            h01[a] = 1.0
            h01[b] = 0.0
    return best, best_c

def _swap_search(h, dx, n, target, rng, t_end):
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    deltas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    while time.time() < t_end:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            for j in rng.choice(n, size=min(6, n - 1), replace=False):
                if i == j:
                    continue
                for d in deltas:
                    if h[i] - d < -1e-10 or h[j] + d > 1.0 + 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    c = _c5(h, dx)
                    h[i] += d
                    h[j] -= d
                    if c < best_c - 1e-12:
                        h[i] -= d
                        h[j] += d
                        best_c = c
                        best_h = h.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c

def _optimize_n(h0, dx, n, target, rng, t0, deadline):
    best_h = h0.copy()
    best_c = _c5(h0, dx)
    h = h0.copy()
    frac = (deadline - t0)
    t_slsqp_end = t0 + frac * 0.40
    t_mom_end = t0 + frac * 0.60
    t_anneal_end = t0 + frac * 0.80
    t_swap_end = deadline - 20

    for eps in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > t_slsqp_end - 5:
            break
        h = _slsqp(h, dx, n, target, eps, maxiter=350)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h = best_h.copy()

    if time.time() < t_mom_end - 5:
        h2, c2 = _momentum_step(h, dx, n, target, lr=0.08, mom=0.85, t_end=t_mom_end)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    if n == 70 and time.time() < t_anneal_end - 5:
        h01, c01 = _anneal01(best_h, n, dx, target, rng, t_anneal_end)
        if c01 < best_c:
            best_c = c01
            best_h = h01.copy()

    if time.time() < t_swap_end:
        h3, c3 = _swap_search(best_h, dx, n, target, rng, t_swap_end)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    if time.time() < deadline - 5:
        hf = _slsqp(best_h, dx, n, target, 0.0005, maxiter=200)
        cf = _c5(hf, dx)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)

    overall_best_h = None
    overall_best_c = float('inf')

    n_main = 70
    dx_main = 2.0 / n_main
    target_main = n_main / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0_main = np.array(initial_h_values, dtype=float).copy()
        if len(h0_main) != n_main:
            h0_main = np.interp(np.linspace(0, 1, n_main),
                                np.linspace(0, 1, len(h0_main)), h0_main)
    else:
        h0_main = rng.uniform(0.2, 0.8, n_main)
    h0_main = _proj(h0_main, target_main)

    t_n70_end = t0 + (deadline - t0) * 0.75
    h70, c70 = _optimize_n(h0_main, dx_main, n_main, target_main, rng, t0, t_n70_end)
    overall_best_c = c70
    overall_best_h = h70.copy()

    if time.time() < deadline - 60:
        n_alt = 50
        dx_alt = 2.0 / n_alt
        target_alt = n_alt / 2.0
        h0_alt = np.interp(np.linspace(0, 1, n_alt),
                           np.linspace(0, 1, n_main), overall_best_h)
        h0_alt = _proj(h0_alt, target_alt)
        t_alt_end = min(time.time() + (deadline - time.time()) * 0.5, deadline - 30)
        h50, c50 = _optimize_n(h0_alt, dx_alt, n_alt, target_alt, rng, time.time(), t_alt_end)
        if c50 < overall_best_c:
            overall_best_c = c50
            overall_best_h = h50.copy()

    if time.time() < deadline - 10:
        h0_r = _proj(overall_best_h + rng.normal(0, 0.06, len(overall_best_h)),
                     len(overall_best_h) / 2.0)
        n_r = len(overall_best_h)
        dx_r = 2.0 / n_r
        tgt_r = n_r / 2.0
        hr, cr = _optimize_n(h0_r, dx_r, n_r, tgt_r, rng, time.time(), deadline - 5)
        if cr < overall_best_c:
            overall_best_c = cr
            overall_best_h = hr.copy()

    return overall_best_h, overall_best_c, len(overall_best_h)
