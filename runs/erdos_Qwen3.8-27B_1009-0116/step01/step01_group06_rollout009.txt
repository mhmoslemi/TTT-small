```python
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_c5_grad(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = alpha * (corr - mc)
    exp_shifted = np.exp(shifted)
    wsum = exp_shifted.sum()
    weights = exp_shifted / wsum
    f = float(mc + np.log(wsum) / alpha) * dx  # Scale by dx
    
    g = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-16:
            continue
        lag = idx - (n - 1)
        for m in range(n):
            if 0 <= m - lag < n:
                g[m] += w * (1.0 - h[m - lag])
            if 0 <= m + lag < n:
                g[m] -= w * h[m + lag]
    g *= dx
    return f, g

def nesterov_projected(h, n, dx, alpha, max_iter=5000, lr=0.01):
    target = float(n) / 2.0
    h_cur = h.copy()
    h_new = h.copy()
    best_h = h.copy()
    best_f = eval_c5(h, dx)
    beta = 0.9
    
    for it in range(max_iter):
        # Nesterov step
        beta_it = (1 + np.sqrt(1 + 4 * beta**2)) / 2
        h_mom = h_new + (beta_it - 1) / (beta_it + 1) * (h_new - h_cur)
        h_mom = project_feasible(h_mom, n)
        
        _, g = smooth_c5_grad(h_mom, dx, n, alpha)
        h_new = project_feasible(h_mom - lr * g, n)
        h_cur = h_mom
        
        f = eval_c5(h_new, dx)
        if f < best_f - 1e-12:
            best_f = f
            best_h = h_new.copy()
    
    return best_h, best_f

def slsqp_smooth(h, n, dx, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    target = float(n) / 2.0
    
    def obj(x):
        f, _ = smooth_c5_grad(x, dx, n, alpha)
        return f
    
    def ceq(x):
        return [x.sum() - target]
    
    def cjac(x):
        return [np.ones(n)]
    
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except:
        return h, eval_c5(h, dx)

    return best_h, best_f
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = float(n) / 2.0
    d = t - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-12:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += d * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = t - h.sum()
        if abs(d2) > 1e-12:
            i = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[i] = float(np.clip(h[i] + d2, 0.0, 1.0))
    return h

def smooth_fg(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = alpha * (corr - mc)
    es = np.exp(sh)
    ws = es.sum()
    w = es / ws
    f = (mc + np.log(ws) / alpha) * dx
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-18:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            g[:n - lag] += wi * (1.0 - h[lag:n])
            if lag > 0:
                g[lag:] -= wi * h[:n - lag]
        else:
            a = -lag
            g[a:] += wi * (1.0 - h[:n - a])
            g[:n - a] -= wi * h[a:]
    return f, g

def nesterov_smooth(h, n, dx, alpha, max_iter=5000, lr=0.01):
    best_h = h.copy()
    best_f = eval_c5(h, dx)
    xp = h.copy()
    xc = h.copy()
    for k in range(max_iter):
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (k + 1) ** 2)) / 2.0
        t_km1 = (1.0 + np.sqrt(1.0 + 4.0 * k ** 2)) / 2.0 if k > 0 else 1.0
        y = xc + (t_k - 1.0) / t_k * (xc - xp)
        y = project_feasible(y, n)
        _, g = smooth_fg(y, dx, n, alpha)
        xn = project_feasible(y - lr * g, n)
        f = eval_c5(xn, dx)
        if f < best_f - 1e-13:
            best_f = f
            best_h = xn.copy()
        xp = xc
        xc = xn
    return best_h, best_f

def slsqp_opt(h, n, dx, use_smooth=False, alpha=100, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    t = float(n) / 2.0
    
    def obj(x):
        if use_smooth:
            f, _ = smooth_fg(x, dx, n, alpha)
            return f
        return eval_c5(x, dx)
    
    def obj_grad(x):
        if use_smooth:
            _, g = smooth_fg(x, dx, n, alpha)
            return g
        return None
    
    def ceq(x):
        return [x.sum() - t]
    
    def cjac(x):
        return [np.ones(n)]
    
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    kwargs = {'bounds': bounds, 'constraints': cons, 'options': {'maxiter': maxiter, 'ftol': 1e-14}}
    if use_smooth:
        kwargs['jac'] = obj_grad
    try:
        res = minimize(obj, h, method='SLSQP', **kwargs)
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def binary_local_search(h, n, dx, max_iter=5000):
    best_h = h.copy()
    best_f = eval_c5(h, dx)
    rng = np.random.default_rng(42)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    for it in range(max_iter):
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        h_new = project_feasible(h_new, n)
        f = eval_c5(h_new, dx)
        if f < best_f - 1e-14:
            best_h = h_new
            best_f = f
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
    return best_h, best_f
```

```python
import numpy as np
from scipy.optimize import minimize

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = float(n) / 2.0
    d = t - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-12:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += d * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = t - h.sum()
        if abs(d2) > 1e-12:
            i = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[i] = float(np.clip(h[i] + d2, 0.0, 1.0))
    return h

def smooth_fg(h, dx, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = alpha * (corr - mc)
    es = np.exp(sh)
    ws = es.sum()
    w = es / ws
    f = (mc + np.log(ws) / alpha) * dx
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-18:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            g[:n - lag] += wi * (1.0 - h[lag:n])
            if lag > 0:
                g[lag:] -= wi * h[:n - lag]
        else:
            a = -lag
            g[a:] += wi * (1.0 - h[:n - a])
            g[:n - a] -= wi * h[a:]
    return f, g

def nesterov_smooth(h, n, dx, alpha, max_iter=5000, lr=0.01):
    best_h = h.copy()
    best_f = eval_c5(h, dx)
    xp = h.copy()
    xc = h.copy()
    for k in range(max_iter):
        tk = (1.0 + np.sqrt(1.0 + 4.0 * (k + 1) ** 2)) / 2.0
        tkm1 = (1.0 + np.sqrt(1.0 + 4.0 * k ** 2)) / 2.0 if k > 0 else 1.0
        y = xc + (tk - 1.0) / tk * (xc - xp)
        y = project_feasible(y, n)
        _, g = smooth_fg(y, dx, n, alpha)
        xn = project_feasible(y - lr * g, n)
        f = eval_c5(xn, dx)
        if f < best_f - 1e-13:
            best_f = f
            best_h = xn.copy()
        xp = xc
        xc = xn
    return best_h, best_f

def slsqp_opt(h, n, dx, use_smooth=False, alpha=100, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    t = float(n) / 2.0
    def obj(x):
        if use_smooth:
            f, _ = smooth_fg(x, dx, n, alpha)
            return f
        return eval_c5(x, dx)
    def ceq(x):
        return [x.sum() - t]
    def cjac(x):
        return [np.ones(n)]
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    kw = {'bounds': bounds, 'constraints': cons, 'options': {'maxiter': maxiter, 'ftol': 1e-14}}
    if use_smooth:
        def og(x):
            _, gg = smooth_fg(x, dx, n, alpha)
            return gg
        kw['jac'] = og
    try:
        res = minimize(obj, h, method='SLSQP', **kw)
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)
```
