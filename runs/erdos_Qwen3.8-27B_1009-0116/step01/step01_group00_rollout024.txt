        improved = False
        i = rng.randint(n)
        for j in rng.choice(n, size=min(n, 35), replace=False):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                ni, nj = hi + d, hj - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c5 = _compute_c5(h, dx)[0]
                if c5 < best_local - 1e-11:
                    best_local = c5
                    improved = True
                else:
                    h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best_local


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / float(n)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _EqConstraint(target_sum)}]

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)[0]

    # Phase 1: Binary simulated annealing (find good structure)
    binary_limit = t_start + min(350, budget_s * 0.35)
    for restart in range(8):
        if time.time() > binary_limit:
            break
        h_b, c5_b = _binary_sa(rng, n, iters=4000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Continuous refinement from binary
        h_c = _repair(h_b + 0.15 * rng.randn(n), n)
        hc = _slsqp_optimize(h_c, n, 200, dx, target_sum, bounds, cons, 250)
        if hc is not None:
            c5 = _compute_c5(hc, dx)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 2: SLSQP from initial and current best with varying alpha
    slsqp_limit = t_start + min(600, budget_s * 0.6)
    starts = [h0, best_h]
    for alpha in [150, 300, 600, 1200]:
        if time.time() > slsqp_limit:
            break
        for s_h in starts:
            if time.time() > slsqp_limit:
                break
            hc = _slsqp_optimize(s_h.copy(), n, alpha, dx, target_sum, bounds, cons, 350)
            if hc is not None:
                c5 = _compute_c5(hc, dx)[0]
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
                    starts[1] = best_h.copy()

    # Phase 3: Active-set optimization
    active_limit = t_start + min(800, budget_s * 0.8)
    top_counts = [5, 8, 12, 15]
    for it in range(12):
        if time.time() > active_limit:
            break
        _, corr = _compute_c5(best_h, dx)
        tc = top_counts[it % len(top_counts)]
        active_idx = np.argsort(corr)[-tc:]
        hc = _slsqp_optimize(best_h.copy(), n, 400, dx, target_sum,
                             bounds, cons, 300, active_idx)
        if hc is not None:
            c5 = _compute_c5(hc, dx)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 4: Coordinate descent (multiple rounds)
    cd_limit = min(t_start + budget_s - 60, t_start + min(950, budget_s * 0.95))
    for round_i in range(4):
        if time.time() > cd_limit:
            break
        h_cd, c5_cd = _coord_descent(best_h.copy(), rng, n, dx,
                                     max_passes=300, t_limit=cd_limit)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: Final high-alpha refinement
    if time.time() < t_start + budget_s - 15:
        hc = _slsqp_optimize(best_h.copy(), n, 2000, dx, target_sum,
                             bounds, cons, 400)
        if hc is not None:
            c5 = _compute_c5(hc, dx)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Final coordinate polish
    if time.time() < t_start + budget_s - 5:
        h_cd, c5_cd = _coord_descent(best_h.copy(), rng, n, dx,
                                     max_passes=100, t_limit=t_start + budget_s - 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)[0]
    return (best_h, best_c5, n)
