from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:

    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-10:

        h = np.clip(h, 0.0, 1.0)

def constraint_sum(x):
    return np.sum(x) - len(x) / 2.0

def obj_fn(x):

def gradient_max_corr(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:

            grad[:n - k] -= h[k:]
    else:
        m = -k
        if n - m > 0:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]

            lr *= 0.85

    eps_list = (0.01, 0.02, 0.05, 0.1, 0.2)

            hp = repair(hp, n)
            c = compute_c5(hp, n)
            if c < best_c - 1e-14:

    return h, best_c

    if time.time() > deadline:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    try:

                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(r.x, n)

def make_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)

def make_smooth(n, rng):

    indices = np.argsort(corr)[::-1][:top_m]

    grad = np.zeros(n)
