    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def lsq_eval(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx


def exact_eval(x, n):
    return compute_c5(x, n)


class ObjLSQ:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
    def __call__(self, x):
        return lsq_eval(x, self.n, self.beta)


class ObjExact:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return exact_eval(x, self.n)


class EqSum:
    def __init__(self, target):
        self.target = target
    def __call__(self, x):
        return np.sum(x) - self.target


def make_inits(n, rng, init_vals):
    inits = []
    if init_vals is not None and len(init_vals) == n:
        inits.append(repair(np.array(init_vals, dtype=float), n))
    for m in [1, 2, 3, 4, 5, 6, 7, 8]:
        ph = rng.uniform(0, 2 * np.pi)
        h0 = 0.5 + 0.35 * np.sin(2.0 * np.pi * m * np.arange(n) / n + ph)
        inits.append(repair(h0, n))
    h0 = np.where(np.arange(n) % 2 == 0, 0.9, 0.1)
    inits.append(repair(h0, n))
    half = n // 2
    h_ap = np.zeros(n)
    h_ap[:half] = 0.5 + 0.4 * np.sin(np.pi * np.arange(half) / half)
    h_ap[half:] = 1.0 - h_ap[:n - half]
    inits.append(repair(h_ap, n))
    for bs in [2, 3, 4, 5, 7]:
        period = 2 * bs
        h0 = np.zeros(n)
        for i in range(n):
            h0[i] = 1.0 if (i % period) < bs else 0.0
        h0 = h0 + rng.normal(0, 0.05, n)
        inits.append(repair(h0, n))
    for _ in range(4):
        h0 = rng.rand(n) * 0.6 + 0.2
        inits.append(repair(h0, n))
    return inits


def optimize_from_start(h_init, n, t0, deadline, rng):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': EqSum(target_sum)}]
    h_cur = h_init.copy()
    c_cur = compute_c5(h_cur, n)
    betas = [6, 12, 25, 50, 100, 200, 400]
    for beta in betas:
        if time.time() > deadline - 5:
            break
        try:
            r = minimize(ObjLSQ(n, beta), h_cur, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 200, 'ftol': 1e-12})
            hc = repair(r.x, n)
            c_new = compute_c5(hc, n)
            if c_new < c_cur - 1e-13:
                c_cur = c_new
                h_cur = hc
        except Exception:
            pass
    if time.time() < deadline - 5:
        try:
            r = minimize(ObjExact(n), h_cur, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-14})
            hc = repair(r.x, n)
            c_new = compute_c5(hc, n)
            if c_new < c_cur - 1e-13:
                c_cur = c_new
                h_cur = hc
        except Exception:
            pass
    return h_cur, c_cur


def interpolate_h(h_coarse, n_coarse, n_fine):
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    x_fine = np.linspace(0, 2, n_fine, endpoint=False)
    return np.interp(x_fine, x_coarse, h_coarse)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    init = globals().get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')

    def update_best(h, c):
        nonlocal best_h, best_c5
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 1: Multi-scale coarse-to-fine
    # Start at small n, find structure, then interpolate up
    coarse_n = 20
    if time.time() < deadline - 60:
        for h_init in make_inits(coarse_n, rng, None)[:6]:
            if time.time() > deadline - 60:
                break
            hc, cc = optimize_from_start(h_init, coarse_n, t0, deadline - 60, rng)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()
                coarse_best = hc.copy()

        # Interpolate coarse best to medium and fine
        for target_n in [50, 70, 90]:
            if time.time() > deadline - 40:
                break
            h_interp = interpolate_h(best_h, coarse_n, target_n)
            h_interp = repair(h_interp, target_n)
            hc, cc = optimize_from_start(h_interp, target_n, t0, deadline - 40, rng)
            update_best(hc, cc)

    # Phase 2: Direct optimization at multiple n values with diverse inits
    n_values = [60, 70, 80]
    if init is not None:
        n_init = len(init)
        if n_init not in n_values:
            n_values.insert(0, n_init)

    time_per_n = max(60, (deadline - time.time() - 100) / len(n_values))

    for n in n_values:
        if time.time() > deadline - 30:
            break
        n_deadline = min(deadline - 30, time.time() + time_per_n)
        inits = make_inits(n, rng, init)
        for h_init in inits:
            if time.time() > n_deadline - 5:
                break
            hc, cc = optimize_from_start(h_init, n, t0, n_deadline, rng)
            update_best(hc, cc)

    # Phase 3: Iterated local search from best
    if time.time() < deadline - 30 and best_h is not None:
        n_best = len(best_h)
        for _ in range(30):
            if time.time() > deadline - 20:
                break
            sigma = 0.05 + 0.03 * rng.rand()
            h_pert = repair(best_h + rng.normal(0, sigma, n_best), n_best)
            hc, cc = optimize_from_start(h_pert, n_best, t0, deadline - 10, rng)
            update_best(hc, cc)

    # Phase 4: Final aggressive refinement
    if time.time() < deadline and best_h is not None:
        n_best = len(best_h)
        target_sum = n_best / 2.0
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq', 'fun': EqSum(target_sum)}]
        try:
            r = minimize(ObjExact(n_best), best_h, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 1000, 'ftol': 1e-15})
            hf = repair(r.x, n_best)
            cf = compute_c5(hf, n_best)
            if cf < best_c5:
                best_c5 = cf
                best_h = hf
        except Exception:
            pass

        # Try log-sum-exp with very high beta for final push
        if time.time() < deadline - 5:
            try:
                r2 = minimize(ObjLSQ(n_best, 500), best_h, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 800, 'ftol': 1e-15})
                hf2 = repair(r2.x, n_best)
                cf2 = compute_c5(hf2, n_best)
                if cf2 < best_c5:
                    best_c5 = cf2
                    best_h = hf2
            except Exception:
                pass

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
