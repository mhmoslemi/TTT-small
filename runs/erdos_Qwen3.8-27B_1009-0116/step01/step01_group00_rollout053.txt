def _smooth_obj(x, alpha):
    corr = _corr_full(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    # Phase 1: Multiple initializations
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    candidates = [h0]
    for i in range(8):
        f = 1.0 + i * 0.5
        a = 0.2 + 0.1 * (i % 3)
        p = i * 0.7
        o = 0.48 + 0.03 * (i % 2)
        hc = _gen_cosine(n, f, a, p, o, seed + i)
        candidates.append(_repair(hc, n))
    for i in range(4):
        hp = np.array(initial_h_values, dtype=float) + (0.1 + 0.05 * i) * rng.randn(n)
        candidates.append(_repair(hp, n))

    evals = [(_compute_c5(c, dx), c) for c in candidates]
    evals.sort(key=lambda x: x[0])
    best_h = evals[0][1].copy()
    best_c5 = evals[0][0]

    if time.time() - t0 > budget_s - 80:
        return (best_h, best_c5, n)

    # Phase 2: Subgradient descent with adaptive step
    h = best_h.copy()
    lr = 0.008
    best_local = best_c5
    for it in range(8000):
        if time.time() - t0 > budget_s - 250:
            break
        if it % 200 == 0 and it > 0:
            lr = max(0.001, lr * 0.92)
        corr = _corr_full(h)
        mx = np.max(corr)
        thresh = mx - 0.02 * mx if mx > 0 else 0
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-3:]
        g = _subgrad_for_active(h, active, n)
        g = g - np.mean(g)  # sum-zero for constraint
        h_new = h + lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, n)
        c_new = _compute_c5(h_new, dx)
        if c_new < best_local - 1e-12:
            best_local = c_new
            best_h = h_new.copy()
            h = h_new
            if c_new < best_c5:
                best_c5 = c_new
        else:
            h = h_new  # still accept for trajectory

    if time.time() - t0 > budget_s - 150:
        best_h = _repair(best_h, n)
        best_c5 = _compute_c5(best_h, dx)
        return (best_h, best_c5, n)

    # Phase 3: Targeted coordinate-pair search on active set
    h = best_h.copy()
    best_local = best_c5
    deltas = np.array([0.005, 0.01, 0.02, 0.03, 0.05, 0.08, -0.005, -0.01, -0.02, -0.03])
    stall = 0

        if time.time() - t0 > budget_s - 60:
            break

        mx = np.max(corr)
        active = np.argsort(corr)[-4:]
        improved = False

        for k_star in active:
            if improved:
                break
            ps = max(0, k_star - n + 1)
            pe = min(n - 1, k_star)
            if ps > pe:
                continue
            idxs = np.arange(ps, pe + 1)
            contrib = h[idxs] * (1.0 - h[k_star - idxs])
            top_harm = idxs[np.argsort(contrib)[-8:]]

            for i_max in top_harm:
                if improved:
                    break
                # Find best receiver: minimize increase in all active entries
                best_j_move = None
                best_d_val = None
                best_c_try = best_local
                for j in range(n):
                    if j == i_max:
                        continue
                    for d in deltas:
                        ni = h[i_max] + d
                        nj = h[j] - d
                        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                            continue
                        h[i_max] = ni
                        h[j] = nj
                        c_try = _compute_c5(h, dx)
                        if c_try < best_c_try:
                            best_c_try = c_try
                            best_j_move = (j, d)
                        h[i_max] = ni - d
                        h[j] = nj + d
                if best_j_move is not None and best_c_try < best_local - 1e-11:
                    j, d = best_j_move
                    h[i_max] += d
                    h[j] -= d
                    best_local = best_c_try
                    improved = True
                    stall = 0
                    if best_local < best_c5:
                        best_c5 = best_local
                        best_h = h.copy()

            stall += 1

                stall = 0
