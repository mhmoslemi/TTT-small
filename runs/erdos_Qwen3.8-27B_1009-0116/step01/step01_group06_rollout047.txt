    if abs(diff) > 1e-14:
        if diff > 0:

            s = room.sum()
            if s > 1e-14:

        else:

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_c5_val(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    return max_c + (1.0 / alpha) * np.log(np.sum(np.exp(shifted)))

def compute_smooth_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for L in range(-(n - 1), n):
        w = p[L + n - 1]
        if w < 1e-15:
            continue
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            k = -L
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:k] -= w * h[k:]
    return grad

def _penalty_smooth_obj(x, dx, alpha, penalty, target):
    f = smooth_c5_val(x, dx, alpha)
    f += penalty * (x.sum() - target) ** 2
    g = compute_smooth_grad(x, dx, alpha)
    g = g + 2.0 * penalty * (x.sum() - target)
    return f, g
