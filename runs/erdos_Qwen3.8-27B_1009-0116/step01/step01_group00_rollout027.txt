import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(alpha * (c - m), -400, 400)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _eq_con(x):
    return x.sum() - len(x) / 2.0

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-14:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        return h
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _pair_sweep(h, n, dx, deltas, max_passes, deadline, t0):
    best = _c5(h, dx)
    for _ in range(max_passes):
        if time.time() - t0 > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                for dl in deltas:
                    ni, nj = hi + dl, hj - dl
                    if ni < -1e-12 or ni > 1.0 + 1e-12:
                        continue
                    if nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i] = max(0.0, min(1.0, ni))
                    h[j] = max(0.0, min(1.0, nj))
                    c = _c5(h, dx)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, (alpha,), 'SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _eq_con}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
