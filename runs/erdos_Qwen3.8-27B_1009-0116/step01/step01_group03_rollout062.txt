```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        if diff2 > 0:
            idx2 = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx2 = np.where(h > 1e-10)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + diff2, 0, 1)
    return h

def lse_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) * beta, -50, 50)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / len(h))

def lse_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) * beta, -50, 50)
    w = np.exp(sh) / np.sum(np.exp(sh))
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wi = w[n-1-d]
        if wi < 1e-14:
            continue
        if d >= 0:
            jf = np.arange(0, n-d)
            jb = np.arange(d, n)
        else:
            jf = np.arange(-d, n)
            jb = np.arange(0, n+d)
        grad[jf] += wi * (1.0 - h[jf + d])
        grad[jb] -= wi * h[jb - d]
    return grad * (2.0 / n)

def nesterov_opt(h, ts, beta, steps=3000, lr0=0.005):
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_f = lse_obj(x, beta)
    for t in range(steps):
        lr = lr0 / (1.0 + 0.005 * t)
        y = np.clip(x + v * lr, 0, 1)
        g = lse_grad(y, beta)
        x_new = np.clip(x - lr * g, 0, 1)
        x_new = project_feasible(x_new, ts)
        v = x_new - x
        x = x_new
        f = lse_obj(x, beta)
        if f < best_f:
            best_f = f
            best_x = x.copy()
    return best_x, best_f

def binary_local_search(h, ts, rng, max_sweeps=50, time_limit=80):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        # Try all swaps
        for i in ones.copy():
            for j in zeros.copy():
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def remaining():
        return budget_s - (time.time() - t_start)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[:70] = 1.0
    
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at initial n
    ts = n_init / 2.0
    h = h_init.copy()
    
    # Nesterov with increasing beta
    for beta in [5.0, 15.0, 40.0, 100.0, 300.0]:
        if remaining() < 60:
            break
        h, _ = nesterov_opt(h, ts, beta, steps=2000, lr0=0.003)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Binarize and local search
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    num_ones = int(h_bin.sum())
    if num_ones > n_init // 2:
        # Remove excess ones (the ones with lowest h value)
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - n_init//2]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < n_init // 2:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:n_init//2 - num_ones]:
            h_bin[zero_idx[idx]] = 1
    
    h_bin, c5_ls = binary_local_search(h_bin, ts, rng, max_sweeps=30, time_limit=120)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_bin.copy()
    
    # SLSQP polish
    if remaining() > 60:
        def obj_f(x):
            return lse_obj(x, 200.0)
        res = minimize(obj_f, best_h, method='SLSQP',
                       bounds=[(0.0,1.0)]*n_init,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-ts}],
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_s = project_feasible(res.x, ts)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
    
    # Phase 2: Try other n values
    for n_try in [n_init-20, n_init+20, n_init+40]:
        if remaining() < 120 or n_try < 40:
            continue
        ts_t = n_try / 2.0
        # Interpolate
        xo = np.linspace(0, 2, n_init)
        xn = np.linspace(0, 2, n_try)
        h_t = np.interp(xn, xo, best_h)
        h_t = project_feasible(h_t, ts_t)
        
        for beta in [10.0, 50.0, 200.0]:
            if remaining() < 60:
                break
            h_t, _ = nesterov_opt(h_t, ts_t, beta, steps=1500, lr0=0.003)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
        
        # Local search on this n
        if remaining() > 60:
            h_bin_t = (h_t > 0.5).astype(float)
            num_o = int(h_bin_t.sum())
            if num_o > n_try // 2:
                oi = np.where(h_bin_t == 1)[0]
                for idx in np.argsort(h_t[oi])[:num_o - n_try//2]:
                    h_bin_t[oi[idx]] = 0
            elif num_o < n_try // 2:
                zi = np.where(h_bin_t == 0)[0]
                for idx in np.argsort(-h_t[zi])[:n_try//2 - num_o]:
                    h_bin_t[zi[idx]] = 1
            h_bin_t, c5_lt = binary_local_search(h_bin_t, ts_t, rng, max_sweeps=20, time_limit=60)
            if c5_lt < best_c5:
                best_c5 = c5_lt
                best_h = h_bin_t.copy()
    
    # Phase 3: Final SLSQP on best
    if remaining() > 30:
        n_best = len(best_h)
        ts_best = n_best / 2.0
        def obj_f2(x):
            return lse_obj(x, 500.0)
        res2 = minimize(obj_f2, best_h, method='SLSQP',
                        bounds=[(0.0,1.0)]*n_best,
                        constraints=[{'type':'eq','fun':lambda x: x.sum()-ts_best}],
                        options={'maxiter': 1000, 'ftol': 1e-15})
        if res2.x is not None and np.all(np.isfinite(res2.x)):
            h_f = project_feasible(res2.x, ts_best)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h)/2.0)
    best_c5 = compute_c5(best_h)
```

```python
def lse_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) * beta, -50, 50)
    w = np.exp(sh) / np.sum(np.exp(sh))
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wi = w[n-1+d]
        if wi < 1e-14:
            continue
        # (1-h[j-d]) for j where 0<=j-d<n
        # j-d >= 0 => j >= d; j-d < n => j < n+d
        j_start_f = max(0, d)
        j_end_f = min(n, n+d)
        if j_end_f > j_start_f:
            js = np.arange(j_start_f, j_end_f)
            grad[js] += wi * (1.0 - h[js - d])
        # -h[j+d] for j where 0<=j+d<n
        # j+d >= 0 => j >= -d; j+d < n => j < n-d
        j_start_b = max(0, -d)

        if j_end_b > j_start_b:
            js = np.arange(j_start_b, j_end_b)
            grad[js] -= wi * h[js + d]
    return grad * (2.0 / n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        if diff2 > 0:
            idx2 = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx2 = np.where(h > 1e-10)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + diff2, 0, 1)
    return h

def lse_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) * beta, -50, 50)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / len(h))

def lse_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) * beta, -50, 50)
    w = np.exp(sh) / np.sum(np.exp(sh))
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wi = w[n - 1 + d]
        if wi < 1e-14:
            continue
        js_f = np.arange(max(0, d), min(n, n + d))
        if len(js_f) > 0:
            grad[js_f] += wi * (1.0 - h[js_f - d])
        js_b = np.arange(max(0, -d), min(n, n - d))
        if len(js_b) > 0:
            grad[js_b] -= wi * h[js_b + d]
    return grad * (2.0 / n)

def nesterov_opt(h, ts, beta, steps=2000, lr0=0.004, time_limit=60):
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_f = lse_obj(x, beta)
    t0 = time.time()
    for t in range(steps):
        if time.time() - t0 > time_limit:
            break
        lr = lr0 / (1.0 + 0.008 * t) ** 0.5
        y = np.clip(x + v * lr, 0, 1)
        g = lse_grad(y, beta)
        x_new = np.clip(x - lr * g, 0, 1)
        x_new = project_feasible(x_new, ts)
        v = x_new - x
        x = x_new
        f = lse_obj(x, beta)
        if f < best_f:
            best_f = f
            best_x = x.copy()
    return best_x, best_f

def binarize(h, n):
    h_b = (h > 0.5).astype(float)
    target = n // 2
    num_o = int(h_b.sum())
    if num_o > target:
        oi = np.where(h_b == 1)[0]
        for idx in np.argsort(h[oi])[:num_o - target]:
            h_b[oi[idx]] = 0
    elif num_o < target:
        zi = np.where(h_b == 0)[0]
        for idx in np.argsort(-h[zi])[:target - num_o]:
            h_b[zi[idx]] = 1
    return h_b

def swap_local_search(h, ts, rng, max_sweeps=40, time_limit=90):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        best_swap = None
        best_val = best_c5
        ri = rng.permutation(len(ones))
        rj = rng.permutation(len(zeros))
        for k in range(len(ri)):
            for m in range(len(rj)):
                i, j = ones[ri[k]], zeros[rj[m]]
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                if c5 < best_val - 1e-15:
                    best_val = c5
                    best_swap = (i, j)
                else:
                    h[i], h[j] = h[j], h[i]
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_c5 = best_val
        else:
            break
    return h, best_c5

def slsqp_polish(h, ts, beta, maxiter=600):
    def obj_f(x):
        return lse_obj(x, beta)
    res = minimize(obj_f, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * len(h),
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h.copy(), ts)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[:70] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov at initial n with annealed beta
    ts = n_init / 2.0
    h = h_init.copy()
    for beta in [5.0, 15.0, 40.0, 100.0, 250.0]:
        if remaining() < 50:
            break
        h, _ = nesterov_opt(h, ts, beta, steps=2500, lr0=0.004, time_limit=45)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Binarize + swap local search
    h_bin = binarize(best_h, n_init)

    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_bin.copy()

    # Phase 3: SLSQP polish on continuous
    if remaining() > 50:
        h_p = slsqp_polish(best_h.copy(), ts, 300, maxiter=500)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

        if remaining() < 80:

        h_r = np.zeros(n_init)
        h_r[rng.choice(n_init, n_init // 2, replace=False)] = 1.0
        h_r, _ = nesterov_opt(h_r, ts, 30.0, steps=800, lr0=0.005, time_limit=20)
```
