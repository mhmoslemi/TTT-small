import numpy as np
from scipy.optimize import minimize
import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return np.max(_corr_arr(h)) * dx

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    w_sum = exp_s.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = exp_s[k] / w_sum
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - x[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * x[lo2 - shift:hi2 - shift]
    return grad / alpha

def _project_sum(h, target):
    lo, hi = -1.1, 1.1
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    return _project_sum(h, n / 2.0)

def _pgd(h, alpha, n, max_steps, deadline):

    h = h.copy()
    lr = 1.0
    prev_f = _smooth_obj(h, alpha)
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        g = _smooth_grad(h, alpha)
        g_norm = np.linalg.norm(g)

        step = lr * g / g_norm
        h_new = _project_sum(h - step, target)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < prev_f - 1e-15:
            h = h_new
            prev_f = f_new
            lr = min(lr * 1.1, 50.0)
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    return h

def _coord_search(h, dx, n, deadline, seed=42):

    h = h.copy()
    best_c5 = _compute_c5(h, dx)
    steps = np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1,
                      -0.001, -0.002, -0.005, -0.01, -0.02, -0.05, -0.1])
    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        for _j in range(min(n, 40)):
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                ni, nj = hi + d, hj - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i] = max(0.0, min(1.0, ni))
                h[j] = max(0.0, min(1.0, nj))
                c5 = _compute_c5(h, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    stall = 0
                else:
                    h[i], h[j] = hi, hj
                    stall += 1
        if stall > 500:
            break
    return h, best_c5

def _critical_shift_opt(h, alpha_top, n, dx, max_rounds, deadline):
    """Focus optimization on top shifts achieving near-max overlap."""
    h = h.copy()
    target = n / 2.0
    best_c5 = _compute_c5(h, dx)
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        corr = _corr_arr(h)
        order = np.argsort(-corr)
        top_k = order[:alpha_top]
        top_corr = corr[top_k]
        m = top_corr[0]
        s_arr = alpha_top * (top_corr - m)
        s_arr = np.clip(s_arr, -500, 500)
        weights = np.exp(s_arr) / np.exp(s_arr).sum()
        grad = np.zeros(n)
        for idx in range(len(top_k)):
            w = weights[idx]
            k = top_k[idx]
            shift = k - (n - 1)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            if lo1 < hi1:
                grad[lo1:hi1] += w * (1.0 - h[lo1 + shift:hi1 + shift])
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= w * h[lo2 - shift:hi2 - shift]
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        lr = 1.0

            break
