def _c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = int(np.sum(interior))
    if cnt > 0:
        h[interior] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        h += diff2 / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_smooth(x, alpha, target, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx + 1e4 * (np.sum(x) - target) ** 2

    return grad
