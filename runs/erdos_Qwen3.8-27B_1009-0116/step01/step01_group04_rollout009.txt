def compute_c5(h, dx):

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-10 and s2 > 1e-12:
        x = x * (tgt / s2)
        x = np.clip(x, 0.0, 1.0)
    return x

def proj_simplex_box(x, n):
    """Project onto {0<=x<=1, sum(x)=n/2} approximately."""
    x = np.clip(x, 0.0, 1.0)

        tgt = n / 2.0
        if abs(s - tgt) < 1e-12:
            break
        x = np.clip(x * (tgt / s), 0.0, 1.0)

    """Compute gradient of tau * log(sum(exp(corr/tau))) and the max value."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_c = np.exp((corr - m) / tau)
    w = exp_c / np.sum(exp_c)
    obj = m + tau * np.log(np.sum(exp_c))
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-8:

        s = idx - (n - 1)

        else:

        # -h[j+s] contribution
        if s >= 0:
            grad[s:] -= wi * h[:n-s]
        else:
            grad[:n+s] -= wi * h[-s:]
    
    return grad / tau, obj, m

    h = h0.copy()
    dx = 2.0 / n
    best = h.copy()
    best_val = compute_c5(h, dx)
    t0 = time.time()
    
    tau = tau_init

        c5_val = mval * dx

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-10 and s2 > 1e-12:
        x = x * (tgt / s2)
        x = np.clip(x, 0.0, 1.0)
    return x

def proj_simplex_box(x, n):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(x)
        tgt = n / 2.0
        if abs(s - tgt) < 1e-14:
            break
        x = np.clip(x * (tgt / s), 0.0, 1.0)
    return x

def smooth_max_and_grad(h, n, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_c = np.exp((corr - m) / tau)
    ec_sum = np.sum(exp_c)
    w = exp_c / ec_sum
    obj = m + tau * np.log(ec_sum)
    grad = np.zeros(n)
    omh = 1.0 - h
    thresh = 1e-9
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < thresh:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += wi * omh[s:]
            grad[s:] -= wi * h[:n - s]
        else:
            ss = -s
            grad[ss:] += wi * omh[:n - ss]
            grad[:n - ss] -= wi * h[ss:]
    grad /= tau
    return grad, obj, m

def pg_optimize(h0, n, budget, seed=0):

    h = h0.copy()
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    lr = 0.15
    n_iter = 0
    streak = 0
    total_iters = 8000
    tau_max = 2.0
    tau_min = 0.0005

    while time.time() - t0 < budget:
        n_iter += 1
        frac = min(n_iter / float(total_iters), 1.0)
        tau = tau_max * (tau_min / tau_max) ** frac
        tau = max(tau, tau_min)

        grad, obj, mval = smooth_max_and_grad(h, n, tau)
        c5 = mval * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h.copy()
            streak = 0
        else:
            streak += 1

        if streak > 800:
            h = repair(h + 0.05 * rng.standard_normal(n), n)
            lr = 0.1
            streak = 0
            continue

        gnorm = np.sqrt(np.mean(grad * grad) + 1e-12)
        lr_eff = lr * min(1.0, 0.3 / gnorm)
        h_new = proj_simplex_box(h - lr_eff * grad, n)

        obj_new, _, _ = smooth_max_and_grad(h_new, n, tau)
        if obj_new <= obj + 1e-14:
            h = h_new
            lr = min(lr * 1.02, 0.4)
        else:
            lr *= 0.6
            if lr < 1e-5:
                lr = 0.05
                h = proj_simplex_box(h - 0.05 * grad / gnorm, n)

        if n_iter % 200 == 0:
            h = repair(h, n)

    best = repair(best, n)
    fc5 = compute_c5(best, n)
    if fc5 < best_c5:
        best_c5 = fc5
    return best, best_c5

def slsqp_polish(h0, n, budget, seed=0):
    t0 = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h, n)
    taus = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for tau in taus:
        if time.time() - t0 > budget - 1:
            break
        gd, obj, mval = smooth_max_and_grad(h, n, tau)
        try:
            def obj_fn(x):
                _, o, _ = smooth_max_and_grad(x, n, tau)
                return o
            def obj_grad(x):
                g, _, _ = smooth_max_and_grad(x, n, tau)
                return g
            bd = [(0.0, 1.0)] * n
            cs = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                   'jac': lambda x: np.ones(n)}]
            r = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                         bounds=bd, constraints=cs,
                         options={'maxiter': 300, 'ftol': 1e-14})
            h = repair(r.x, n)
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        except Exception:
            pass
    return best, best_c5

def project_grid(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, h_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)
