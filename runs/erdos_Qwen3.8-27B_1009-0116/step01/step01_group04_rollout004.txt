                xb2, v2 = anneal(xp2, n1, dx1, p2_end - time.time() - 5)
                if v2 < best_v:
                    best_v = v2
                    best_h = xb2
                    best_n = n1

    # Phase 3: Further refinement (20% time)
    p3_end = t_start + budget_s * 0.83
    n2 = 2 * best_n + 1
    if n2 <= 1000 and time.time() < p3_end - 15:
        dx2 = 2.0 / n2
        xp3 = interp_to(best_h, best_n, n2)
        alloc3 = p3_end - time.time() - 5
        if alloc3 > 15:
            xb3, v3 = anneal(xp3, n2, dx2, alloc3)
            if v3 < best_v:
                best_v = v3
                best_h = xb3
                best_n = n2

    # Phase 4: Final polish at best grid (17% time)
    p4_end = t_start + budget_s * 0.96
    dx_b = 2.0 / best_n
    x4 = best_h.copy()
    fine_taus = [0.002, 0.001, 0.0005, 0.0003, 0.00015, 0.0001]
    for tau in fine_taus:
        if time.time() > p4_end:
            break
        x4 = _slsqp_step(x4, best_n, dx_b, tau, 400)
        v4 = compute_c5(x4, dx_b)
        if v4 < best_v:
            best_v = v4
            best_h = x4.copy()

    # Final repair and evaluation
    best_h = project_feasible(best_h, best_n)
    dx_f = 2.0 / best_n
    final_v = compute_c5(best_h, dx_f)
    elapsed = time.time() - t_start
    return (best_h, final_v, best_n)
