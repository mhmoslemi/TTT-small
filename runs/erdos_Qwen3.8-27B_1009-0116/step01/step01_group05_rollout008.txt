            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
        h0 = _proj(h0, target)
    else:
        h0 = _proj(rng.uniform(0.2, 0.8, n), target)

    t1 = t0 + (deadline - t0) * 0.55
    best_h, best_c5 = _optimize_phase(h0, dx, n, target, rng, t1)

    t2 = t0 + (deadline - t0) * 0.82
    while time.time() < t2:
        r = rng.rand()
        if r < 0.3:
            h_pert = _proj(best_h + rng.normal(0, rng.uniform(0.03, 0.12), n), target)
        elif r < 0.6:
            h_pert = _block_pert(best_h, n, target, rng, nb=rng.randint(3, 8), bs=rng.randint(3, 8))
        elif r < 0.8:
            hb = _binary_snap(best_h, n, target)
            h_pert = _proj(hb + rng.normal(0, 0.08, n), target)
        else:
            h_pert = _proj(1.0 - best_h + rng.normal(0, 0.05, n), target)
        sub_end = min(time.time() + min(18.0, t2 - time.time()), t2)
        if sub_end - time.time() > 4:
            h_new, c_new = _optimize_phase(h_pert, dx, n, target, rng, sub_end)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new

    if time.time() < deadline - 80:
        for n2 in [84, 98]:
            if time.time() >= deadline - 60:
                break
            dx2 = 2.0 / n2
            target2 = n2 / 2.0
            h_up = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, len(best_h)), best_h)
            h_up = _proj(h_up, target2)
            t3 = min(deadline - 20, time.time() + 45)
            h_new2, c_new2 = _optimize_phase(h_up, dx2, n2, target2, rng, t3)
            if c_new2 < best_c5:
                best_h = h_new2
                best_c5 = c_new2

    if time.time() < deadline - 5:
        nf = len(best_h)
        dxf = 2.0 / nf
        tf = nf / 2.0
        h_f, c_f = _optimize_phase(best_h, dxf, nf, tf, rng, deadline)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    return best_h, best_c5, len(best_h)
