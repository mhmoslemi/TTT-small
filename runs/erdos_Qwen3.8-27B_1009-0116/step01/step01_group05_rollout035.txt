```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return _corr(h).max() * dx

def _pmax(c, p):
    c = np.maximum(c, 1e-30)
    s = np.sum(c ** p)
    return s ** (1.0 / p)

def _pgrad(h, dx, p):
    n = len(h)
    cv = _corr(h)
    f = _pmax(cv, p)
    w = (np.maximum(cv, 0.0) / f) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n - 1, s + n - 1)
        if lo1 <= hi1:
            i1 = np.arange(lo1, hi1 + 1)
            g[i1] += wt * (1.0 - h[i1 - s])
        lo2 = max(0, -s)
        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            i2 = np.arange(lo2, hi2 + 1)
            g[i2] -= wt * h[i2 + s]
    return g

def _proj(v, target):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_desc(h0, dx, target, t_end, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    for p in [4, 8, 16, 32, 64, 128, 256, 512]:
        if time.time() > t_end - 10:
            break
        lr = 0.5 / (p * dx)  # adaptive lr
        for it in range(500):
            if time.time() > t_end - 5:
                break
            g = _pgrad(h, dx, float(p))
            # Project gradient to satisfy sum constraint
            g = g - g.mean()
            # Backtracking line search
            c_cur = _c5(h, dx)
            lr_cur = lr
            for _ in range(20):
                h_new = _proj(h - lr_cur * g, target)
                c_new = _c5(h_new, dx)
                if c_new <= c_cur + 1e-12:
                    break
                lr_cur *= 0.5
            else:
                h_new = h
```

```python
import numpy as np

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return _corr(h).max() * dx

def _pmax(c, p):
    c = np.maximum(c, 1e-30)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pgrad(h, dx, p):
    n = len(h)
    cv = _corr(h)
    f = _pmax(cv, p)
    if f < 1e-30:
        return np.zeros(n)
    w = (np.maximum(cv, 0.0) / f) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n - 1, s + n - 1)
        if lo1 <= hi1:
            i1 = np.arange(lo1, hi1 + 1)
            g[i1] += wt * (1.0 - h[i1 - s])
        lo2 = max(0, -s)
        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            i2 = np.arange(lo2, hi2 + 1)
            g[i2] -= wt * h[i2 + s]
    return g

def _proj(v, target):

            lo = mid
        else:
            hi = mid
```
