def c5_subgradient(h, n):
    """Compute C5 and subgradient (average over active set)"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    c5 = np.max(corr)
    
    # Active set: shifts within tolerance of max
    thresh = c5 * 0.999
    corr_shifted = corr  # already in "full" mode
    active = np.where(corr_shifted >= thresh)[0]
    
    # Compute gradient at active shifts
    # For full correlation at index idx: sum over i of h[i] * omh[i - (idx - n + 1)]
    # where idx goes from 0 to 2n-2
    # The lag is (idx - (n-1)), ranging from -(n-1) to (n-1)
    
    grad = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)  # shift amount
        # f_lag = sum_i h[i] * (1 - h[i + lag]) (with valid range)
        # d/dh[j] = (1 - h[j + lag]) if j+lag in [0,n-1], 0 otherwise
        #           - h[j - lag] if j-lag in [0,n-1], 0 otherwise
        if lag >= 0:
            # h[i] * (1-h[i+lag]) for i in [0, n-1-lag]
            # d/dh[j] for j in [0, n-1-lag]: (1 - h[j+lag])
            # d/dh[j] for j in [n-lag, n-1]: -h[j-lag]
            grad[:n-lag] += (1.0 - h[lag:n])

            grad[:lag2] -= h[lag2:]
