    """Project x onto {x in [0,1]^n : sum(x) = n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 2.0  # bounds for lambda
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    return proj

def compute_grad_smooth(corr, h, tau, dx, n):
    """Compute gradient of smooth max of corr * dx w.r.t. h"""
    cmax = corr.max()
    shifted = (corr - cmax) / tau
    shifted -= shifted.max()  # for numerical stability
    exps = np.exp(shifted)
    p = exps / exps.sum()
    
    grad = np.zeros(n)
    indices = np.arange(n)
    
    # Only process lags with significant weight
    thresh = 1e-8
    active = p > thresh
    active_idx = np.where(active)[0]
    
    for m in active_idx:
        s = m - (n - 1)
        w = p[m]
        # Term 1: (1 - h[j+s]) where 0 <= j+s < n
        j1 = indices[(indices + s >= 0) & (indices + s < n)]
        if len(j1) > 0:
            grad[j1] += w * (1.0 - h[j1 + s])
        # Term 2: -h[j-s] where 0 <= j-s < n
        j2 = indices[(indices - s >= 0) & (indices - s < n)]
        if len(j2) > 0:
            grad[j2] -= w * h[j2 - s]
    
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0).astype(float)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-10 and s2 > 1e-12:
        x += (tgt - s2) / n
        x = np.clip(x, 0.0, 1.0)
    return x

def project_onto_set(x, n):
    x = np.clip(x, -2.0, 3.0)
    target = n / 2.0
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def project_to_grid(h_src, n_src, n_dst):
    if n_src == n_dst:
        return repair(h_src.copy(), n_dst)
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, h_src, kind='linear', fill_value='extrapolate', bounds_error=False)(xd)
    return repair(xp, n_dst)

def smooth_max_obj(corr, dx, tau):
    cmax = np.max(corr)
    sh = (corr - cmax) / tau
    sh -= sh.max()
    f = dx * (cmax + tau * np.log(np.exp(sh).sum()))
    p = np.exp(sh)
    p /= p.sum()
    return f, p

def compute_grad(corr, x, p, dx, n):
    idx = np.arange(n)
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for m in active:
        s = m - (n - 1)
        w = p[m]
        fwd = (idx + s >= 0) & (idx + s < n)
        bwd = (idx - s >= 0) & (idx - s < n)
        if np.any(fwd):
            grad[fwd] += w * (1.0 - x[idx[fwd] + s])
        if np.any(bwd):
            grad[bwd] -= w * x[idx[bwd] - s]
    grad *= dx
    return grad

def _lbfgs_obj(x, n, dx, tau, penalty):
    corr = np.correlate(x, 1.0 - x, mode="full")
    cmax = np.max(corr)
    sh = (corr - cmax) / tau
    sh -= sh.max()
    f = dx * (cmax + tau * np.log(np.exp(sh).sum()))
    s = np.sum(x) - n / 2.0
    f += penalty * s * s
    p = np.exp(sh)
    p /= p.sum()
    grad = compute_grad(corr, x, p, dx, n)
    grad += 2.0 * penalty * s
    return f, grad

def optimize_lbfgs_stage(h_in, n, dx, tau, penalty, maxiter, t_limit, t0):

    bounds = [(0.0, 1.0)] * n

            return np.inf, np.zeros(n)
        return _lbfgs_obj(x, n, dx, tau, penalty)

                       options={'maxiter': maxiter, 'gtol': 1e-7, 'maxfun': maxiter * 3, 'maxls': 20})
        return project_onto_set(res.x, n)

        per_start = rem_budget / n_starts
        starts = [h_init]
        for i in range(n_starts - 1):
            sc = 0.12 + 0.06 * i
            starts.append(repair(h_init + sc * rng.standard_normal(n), n))
        for si, hs in enumerate(starts):
            elapsed = time.time() - t_start
            if elapsed > budget_s * 0.82:
                break
            t_alloc = min(per_start, budget_s * 0.82 - elapsed)
            if t_alloc < 12:
                break
            h_opt, val = optimize_one(hs, n, dx, t_alloc, rng)
            if val < best_val:
                best_val = val
                best_h = h_opt
                best_n = n

    # Hill climbing on best
    rem = budget_s - (time.time() - t_start) - 10
    if rem > 15:
        dx = 2.0 / best_n
        h_hc, val_hc = hill_climb(best_h, best_n, dx, rem * 0.6, rng)
        if val_hc < best_val:
            best_val = val_hc
            best_h = h_hc
        # Another L-BFGS polish
        rem2 = budget_s - (time.time() - t_start) - 5
        if rem2 > 15:
            h_pol, val_pol = optimize_one(best_h, best_n, 2.0 / best_n, rem2, rng)
            if val_pol < best_val:
                best_val = val_pol
                best_h = h_pol

    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, 2.0 / best_n)
    return (best_h, final_val, best_n)
