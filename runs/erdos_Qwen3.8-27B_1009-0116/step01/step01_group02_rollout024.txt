    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                a = min(1.0 - h[i], diff)
                h[i] += a; diff -= a
            elif diff < 0 and h[i] > 0.0:
                a = min(h[i], -diff)
                h[i] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _eq_con(x):
    return np.sum(x) - _N / 2.0

def _obj_max(x):
    return compute_c5(x, _N)

def _obj_power(x):
    s = compute_shifts(x, _N)
    return np.sum(s ** _P) ** (1.0 / _P)

def _obj_lse(x):
    s = compute_shifts(x, _N)
    b = 25.0
    m = np.max(b * s)
    return (m + np.log(np.sum(np.exp(b * s - m)))) / b

def slsqp_opt(h0, n, bounds, cons, obj, maxiter=250, ftol=1e-13):
    try:
        r = minimize(obj, h0.copy(), method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return repair(r.x, n)
    except Exception:
        return h0

def make_patterns(n, rng):
    pats = []
    x = np.arange(n)
    for k in (1, 2, 3, 4, 5, 6):
        pats.append(0.5 + 0.4 * np.sin(2 * np.pi * k * x / n + rng.uniform(0, 6.28)))
    for bl in (3, 4, 5, 6, 7, 10):
        if n % bl == 0:
            blk = np.zeros(bl)
            blk[:bl // 2] = 1.0
            pats.append(np.tile(blk, n // bl))
    pats.append((x % 7) / 7.0)
    for per in (5, 7, 9, 11):
        if n % per == 0:
            tri = np.abs(np.arange(per) - per / 2) / (per / 2)
            pats.append(np.tile(tri, n // per))
    return pats

def run(seed=42, budget_s=1100, **kwargs):
    global _N, _P
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 90
        h_best = np.full(n, 0.5)

    _N = n
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]

    # Initial SLSQP from parent
    h_best = slsqp_opt(h_best, n, bounds, cons, _obj_max, 300, 1e-13)
    best_c5 = compute_c5(h_best, n)

    # Try diverse patterns
    patterns = make_patterns(n, rng)
    for pat in patterns:
        if time.time() - t0 > budget_s - 80:
            break
        hp = repair(pat, n)
        hc = slsqp_opt(hp, n, bounds, cons, _obj_max, 150, 1e-12)
        c = compute_c5(hc, n)
        if c < best_c5:
            h_best, best_c5 = hc, c

    # Main loop
    stall = 0
    it = 0
    pop = [h_best.copy()]

    while time.time() - t0 < budget_s - 40:
        it += 1
        improved = False

        # Basin hop: big kick + optimize
        if it % 3 == 0 and time.time() - t0 < budget_s - 90:
            kick = 0.12 + 0.01 * min(stall, 25)
            h_kick = repair(h_best + rng.normal(0, kick, n), n)
            hc = slsqp_opt(h_kick, n, bounds, cons, _obj_max, 200, 1e-13)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5, improved = hc, c, True
            if time.time() - t0 < budget_s - 90:
                _P = 12 + 2 * (it % 4)
                hc = slsqp_opt(h_kick, n, bounds, cons, _obj_power, 150, 1e-12)
                c = compute_c5(hc, n)
                if c < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc, c, True

        # LSE smooth objective (every 2)
        if it % 2 == 0 and time.time() - t0 < budget_s - 80:
            h_pert = repair(h_best + rng.normal(0, 0.03, n), n)
            hc = slsqp_opt(h_pert, n, bounds, cons, _obj_lse, 200, 1e-12)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5, improved = hc, c, True

        # Power-sum (every 2, offset)
        if it % 2 == 1 and time.time() - t0 < budget_s - 80:
            _P = 8 + 2 * (it % 7)
            h_pert = repair(h_best + rng.normal(0, 0.04, n), n)
            hc = slsqp_opt(h_pert, n, bounds, cons, _obj_power, 180, 1e-12)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5, improved = hc, c, True

        # SA step (every iteration)
        if time.time() - t0 < budget_s - 20:
            progress = min(1.0, (time.time() - t0) / max(budget_s - 100, 1))
            T = 0.003 * (1.0 - progress) ** 2 + 0.0001
            nm = rng.randint(1, 4)
            idx = rng.choice(n, size=nm, replace=False)
            delta = rng.uniform(-0.12, 0.12, nm)
            h_cand = h_best.copy()
            h_cand[idx] += delta
            h_cand = repair(h_cand, n)
            c_cand = compute_c5(h_cand, n)
            dc = c_cand - best_c5
            if dc < 0 or rng.random() < np.exp(-dc / T):
                h_best, best_c5 = h_cand, c_cand
                if dc < 0:
                    improved = True

        # Coordinate descent when stalled
        if stall > 3 and time.time() - t0 < budget_s - 30:
            for _ in range(20):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for eps in (0.02, 0.05, 0.09):
                    hp = h_best.copy()
                    hp[i] += eps; hp[j] -= eps
                    if hp[i] > 1.0 or hp[j] < 0.0:
                        continue
                    hp = repair(hp, n)
                    c = compute_c5(hp, n)
                    if c < best_c5 - 1e-14:
                        h_best, best_c5, improved = hp, c, True
                        break

        # Periodic structure search (every 15)
        if it % 15 == 0 and time.time() - t0 < budget_s - 100:
            for period in (3, 4, 5, 6, 7, 8, 10, 12, 14, 15, 21, 30):
                if n % period != 0:
                    continue
                base = h_best[:period] + rng.normal(0, 0.06, period)
                pat = repair(np.tile(base, n // period), n)
                hc = slsqp_opt(pat, n, bounds, cons, _obj_max, 120, 1e-12)
                c = compute_c5(hc, n)
                if c < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc, c, True
                    break

        # Population crossover (every 5, if we have 2+)
        if len(pop) >= 2 and it % 5 == 0 and time.time() - t0 < budget_s - 80:
            a = pop[0]
            b = pop[1 % len(pop)]
            alpha = rng.uniform(0.3, 0.7)
            cross = alpha * a + (1.0 - alpha) * b
            cross = repair(cross, n)
            hc = slsqp_opt(cross, n, bounds, cons, _obj_max, 150, 1e-13)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5, improved = hc, c, True

        # Update population
        if improved:
            pop = [h_best.copy()] + pop[:2]
        elif len(pop) > 3:
            pop = pop[:3]

        stall = 0 if improved else stall + 1

    # Final polish
    if time.time() - t0 < budget_s - 5:
        h_best = slsqp_opt(h_best, n, bounds, cons, _obj_max, 600, 1e-15)
        best_c5 = compute_c5(h_best, n)
    if time.time() - t0 < budget_s - 3:
        _P = 24
        hc = slsqp_opt(h_best, n, bounds, cons, _obj_power, 300, 1e-14)
        c = compute_c5(hc, n)
        if c < best_c5:
            h_best, best_c5 = hc, c

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
