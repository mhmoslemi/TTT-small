import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_gradients(h, n):
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    grads[n - 1] = 1.0 - 2.0 * h
    for l in range(1, n):
        g = np.zeros(n)
        g[:n - l] = 1.0 - h[l:]
        g[l:] -= h[:n - l]
        grads[n - 1 + l] = g
        g2 = np.zeros(n)
        g2[l:] = 1.0 - h[:n - l]
        g2[:n - l] -= h[l:]
        grads[n - 1 - l] = g2
    return grads

def project_feasible(h, target):
    n = len(h)
    h = h.copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.clip(h + theta, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_obj(h_flat, n, beta):
    h = h_flat.copy()
    dx = 2.0 / n
    corr = get_corr(h, n)
    scaled = beta * corr
    m = np.max(scaled)
    lse = m + np.log(np.sum(np.exp(scaled - m)))
    return lse / beta * dx

def lse_grad(h, n, beta):
    dx = 2.0 / n
    corr = get_corr(h, n)
    scaled = beta * corr
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    grads = compute_all_gradients(h, n)
    return dx * (w @ grads)

def subgradient_phase(h, n, target, t_end, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    v = np.zeros(n)
    momentum = 0.85
    for it in range(50000):
        if time.time() > t_end:
            break
        corr = get_corr(h, n)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        beta = 8.0 + 4.0 * min(it, 500)
        scaled = beta * corr
        m = np.max(scaled)
        w = np.exp(scaled - m)
        w /= w.sum()
        grads = compute_all_gradients(h, n)
        g = w @ grads
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        lr = 0.8 / (1.0 + 0.01 * it)
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, target)
        h_new_c5 = compute_c5(h_new, n)
        if h_new_c5 < c5_val + 0.001 * lr:
            h = h_new
        else:
            v *= 0.5
            h = h
    return h, best_h, best_c5

def lbfbsb_phase(h, n, target, t_end, rng, beta_start=30.0, n_restarts=6):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    per_restart = (t_end - time.time()) / max(n_restarts, 1)
    for rs in range(n_restarts):
        if time.time() > t_end - 10:
            break
        if rs == 0:
            h0 = h.copy()
        else:
            sigma = 0.05 * (rs + 1)
            h0 = project_feasible(h + rng.normal(0, sigma, n), target)
        beta = beta_start + 10.0 * rs
        def obj(x):
            return lse_obj(x, n, beta)
        def obj_grad(x):
            return lse_grad(x, n, beta)
        res = fmin_l_bfgs_b(obj, h0, fprime=obj_grad,
                             pgtol=1e-14, maxiter=3000, maxfun=100000,
                             iprint=0)
        hc = project_feasible(res[0], target)
        c = compute_c5(hc, n)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()
    return best_h, best_c5

def active_set_slsqp(h, n, target, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    for attempt in range(12):
        if time.time() > t_end - 5:
            break
        corr = get_corr(best_h, n)
        max_val = np.max(corr)
        threshold = max_val - max(0.002, 5 * np.std(corr))
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        active = active[:8]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
        bounds = [(0.0, 1.0)] * n
        for k in active:
            def make_con(kk):
                def confunc(x):
                    c = get_corr(x, n)
                    return c[kk] * dx - max_val * dx
                return confunc
            cons.append({'type': 'eq', 'fun': make_con(int(k))})
        def obj(x):
            return lse_obj(x, n, 100.0)
        def obj_grad(x):
            return lse_grad(x, n, 100.0)
        try:
            res = minimize(obj, best_h, method='SLSQP', jac=obj_grad,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-15})
            hc = project_feasible(res.x, target)
            c = compute_c5(hc, n)
            if c < best_c5:
                best_c5 = c
                best_h = hc.copy()
        except Exception:
            pass
    return best_h, best_c5

def coord_descent(h, n, target, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(50000):
        if time.time() > t_end:
            break
        for j in range(n):
            if time.time() > t_end:
                break
            best_v = best_h[j]
            best_c = best_c5
            for delta in (-0.1, -0.05, -0.02, 0.02, 0.05, 0.1,
                          -0.01, 0.01, -0.03, 0.03, -0.07, 0.07):
                nv = best_h[j] + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                hp = best_h.copy()
                hp[j] = nv
                hp = project_feasible(hp, target)
                if np.any(np.abs(hp - best_h) > 1e-10):
                    continue
                c = compute_c5(hp, n)
                if c < best_c:
                    best_c = c
                    best_v = nv
            if best_v != best_h[j]:
                best_h[j] = best_v
                best_c5 = best_c
        if it % 50 == 0 and it > 0:
            pert = project_feasible(best_h + rng.normal(0, 0.03, n), target)
            cp = compute_c5(pert, n)
            if cp < best_c5:
                best_h = pert
                best_c5 = cp
    return best_h, best_c5
