import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def soft_max_c5(x, dx, tau):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    m = np.max(c)
    e = np.exp(np.clip(tau * (c - m), -40, 40))
    return m + np.log(np.sum(e)) / tau

def soft_max_grad(x, dx, tau):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    m = np.max(c)
    e = np.exp(np.clip(tau * (c - m), -40, 40))
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * x)
        elif lag > 0:
            t1 = np.zeros(n)
            t1[:n - lag] = 1.0 - x[lag:]
            t2 = np.zeros(n)
            t2[lag:] = -x[:-lag]
            grad += wk * dx * (t1 + t2)
        else:
            la = -lag
            t1 = np.zeros(n)
            t1[la:] = 1.0 - x[:n - la]
            t2 = np.zeros(n)
            t2[:n - la] = -x[la:]
            grad += wk * dx * (t1 + t2)
    return grad

def _obj_soft(x):
    return soft_max_c5(x, _opt['dx'], _opt['tau'])

def _obj_soft_grad(x):
    return soft_max_grad(x, _opt['dx'], _opt['tau'])

def _con_fun(x):
    return x.sum() - _opt['target']

def _con_jac(x):

def slsqp_optimize(h, n, dx, use_soft=False, tau=100.0, maxiter=3000, ftol=1e-14):
    _opt['dx'] = dx
    _opt['target'] = float(n) / 2.0
    _opt['n'] = n
    _opt['tau'] = tau
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    if use_soft:
        try:
            res = minimize(_obj_soft, h, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=_obj_soft_grad,
                           options={'maxiter': maxiter, 'ftol': ftol})
        except Exception:
            try:
                res = minimize(_obj_soft, h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            except Exception:
                return h, eval_c5(h, dx)
    else:
        try:
            res = minimize(_obj_direct, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        except Exception:
            return h, eval_c5(h, dx)
    ho = project_feasible(res.x, n)
    return ho, eval_c5(ho, dx)
