```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_solve(h, n, ts, beta, maxiter=500):
    res = minimize(lambda x: smooth_obj(x, n, beta), h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return np.clip(res.x, 0, 1)
    return np.clip(h, 0, 1)

def threshold_binary(h, target_sum):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    # Adjust to meet sum constraint
    diff = int(target_sum) - int(h_bin.sum())

                count += 1
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    se = np.sum(np.exp(shifted))
    return (m + np.log(se) / beta) * (2.0 / n)

def slsqp_step(h, n, ts, beta, maxiter=500):
    def obj(x):
        return smooth_obj(x, n, beta)
    def obj_grad(x):
        # Numerical gradient (or analytical)
        eps = 1e-7
        g = np.zeros_like(x)
        base = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - base) / eps
            x[i] -= eps
        return g
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.success and np.all(np.isfinite(res.x)):
        return np.clip(res.x, 0, 1)
    return np.clip(h, 0, 1)
```

Wait, I need to be more careful. f = M * 2/n + (2/n) * (1/β) * log(Σ exp(β(c_k - M)))

The second term is a smooth (log-sum-exp) of (c_k - M). Its gradient is:
(2/n) * Σ_k [exp(β(c_k-M)) / Σ exp(β(c_k-M))] * ∂(c_k - M)/∂h[j]
= (2/n) * Σ_k p_k * (∂c_k/∂h[j] - ∂M/∂h[j])

The first term M * 2/n has gradient (2/n) * ∂M/∂h[j] (subgradient).

Combined: (2/n) * [∂M/∂h[j] + Σ_k p_k * ∂c_k/∂h[j] - Σ_k p_k * ∂M/∂h[j]]
= (2/n) * [∂M/∂h[j] * (1 - Σ_k p_k) + Σ_k p_k * ∂c_k/∂h[j]]
= (2/n) * [0 + Σ_k p_k * ∂c_k/∂h[j]]  (since Σ p_k = 1)

So the gradient is just (2/n) * Σ_k p_k * ∂c_k/∂h[j]. The M term cancels out!

So the gradient of the smooth objective is:
∂f/∂h[j] = (2/n) * Σ_k p_k * [(1-h[j+ℓ_k])_{valid} - h[j-ℓ_k]_{valid}]

This is a weighted average of the per-position deltas, with weights p_k (softmax of beta*(c_k - M)).

This can be computed efficiently. For each j, the sum is over 2n-1 lags. Total: n * (2n-1) = O(n²) ≈ 20K. Fast!
