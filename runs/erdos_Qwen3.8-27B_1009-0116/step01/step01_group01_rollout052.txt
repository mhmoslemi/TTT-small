def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def subgradient_optimize(h_init, n, deadline, top_k=5, alpha0=0.04, mom=0.92):

    target = n * 0.5

    v = np.zeros(n)
    it = 0
    while time.time() < deadline:
        it += 1
        c = np.correlate(h, 1.0 - h, mode='full') * dx

        if cv < best:
            best = cv
            best_h = h.copy()
        top_idx = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        for s in top_idx:
            shift = int(s) - (n - 1)
            if shift >= 0:
                if n > shift:
                    g1 = np.arange(n - shift)
                    grad[g1] -= h[g1 + shift]
                g2 = np.arange(shift, n)
                grad[g2] += 1.0 - h[g2 - shift]
            else:
                sa = -shift
                if n > sa:
                    g1 = np.arange(n - sa)
                    grad[g1] += 1.0 - h[g1 + sa]
                g2 = np.arange(sa, n)
                grad[g2] -= h[g2 - sa]
        grad *= dx / top_k
        step = alpha0 / (1.0 + 0.001 * it) ** 0.6

        h = np.clip(h + v, 0.0, 1.0)
        d = target - h.sum()

            msk = h < 1.0 if d > 0 else h > 0.0
            if msk.sum() > 0:
                h[msk] += d / msk.sum()
                h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            idx = int(np.argmin(h)) if d2 > 0 else int(np.argmax(h))
            h[idx] = np.clip(h[idx] + d2, 0, 1)
    return best_h, best

def coord_descent(h_init, n, max_sweeps=40):
    h = h_init.copy()
    target = n * 0.5
    dx = 2.0 / n
    best = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = best
            best_c = old
            for cand in (0.0, 1.0, old * 0.5, 0.5 + (old - 0.5) * 0.5):
                if abs(cand - best_c) < 1e-12:
                    continue
                h[i] = cand
                d = target - h.sum()
                j = (i + 1) % n
                h[j] = np.clip(h[j] + d, 0.0, 1.0)
                v = compute_c5(h)
                if v < best_v:
                    best_v = v
                    best_c = cand
            if best_c != old:
                h[i] = best_c
                d = target - h.sum()
                j = (i + 1) % n
                h[j] = np.clip(h[j] + d, 0.0, 1.0)
                best = best_v
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, best

def slsqp_refine(h_init, n, t_param, maxiter):
    dx = 2.0 / n
    target = n * 0.5
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cm = np.max(c)
        return cm + np.log(np.sum(np.exp(t_param * (c - cm)))) / t_param
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init.copy(), method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x
