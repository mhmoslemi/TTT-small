    # R[i][m] where m corresponds to L = m - (n-1)
    R = np.zeros((n, 2*n-1))

        # P(i,L) = 1-h[i-L] if 0<=i-L<n
        idx = i - L
        valid_p = (idx >= 0) & (idx < n)
        R[i, valid_p] += 1 - h[idx[valid_p]]

        idx2 = i + L
        valid_q = (idx2 >= 0) & (idx2 < n)
        R[i, valid_q] -= h[idx2[valid_q]]
    return R

def binarize(h, n):
    """Convert to binary with exactly n//2 ones."""
    h = (h >= 0.5).astype(float).copy()
    target = n // 2
    cnt = int(h.sum())
    if cnt > target:
        # Turn some 1s to 0s (the ones that are closest to 0.5)
        ones = np.where(h == 1)[0]
        # Sort by original value (closest to 0.5 first)
        order = np.argsort(h[ones])  # lowest first (closest to 0.5)
        for idx in order[:cnt - target]:
            h[ones[idx]] = 0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-h[zeros])  # highest first (closest to 0.5)
        for idx in order[:target - cnt]:
            h[zeros[idx]] = 1
    return h

def compute_R(h, n):
    L = np.arange(-(n-1), n)
    R = np.zeros((n, 2*n-1))
    for i in range(n):
        idx_p = i - L
        valid_p = (idx_p >= 0) & (idx_p < n)
        R[i, valid_p] += 1.0 - h[idx_p[valid_p]]
        idx_q = i + L
        valid_q = (idx_q >= 0) & (idx_q < n)
        R[i, valid_q] -= h[idx_q[valid_q]]
    return R

def greedy_local_search(h, n, t0, time_limit):
    dx = 2.0 / n
    c = compute_c5_val(h)
    best_c5 = c.max() * dx
    R = compute_R(h, n)
    n_corr = 2 * n - 1
    
    for sweep in range(10000):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        cur_max = c.max()
        best_gain = 1e-12
        best_swap = None
        
        for i in zeros:
            Ri = R[i]
            for j in ones:
                nm = c + Ri - R[j]
                nm_max = nm.max()
                gain = cur_max - nm_max
                if gain > best_gain:
                    best_gain = gain
                    best_swap = (i, j)
                    # Optional: early termination if we found a perfect swap
                    if nm_max * dx < best_c5 - 0.001:
                        break
            if best_swap and nm_max * dx < best_c5 - 0.001:

        if best_swap is None:

        i, j = best_swap
        h[i] = 1.0
        h[j] = 0.0
        R = compute_R(h, n)
        c = compute_c5_val(h)
        best_c5 = c.max() * dx
    
    return h, best_c5

        if T < T_min:

        zeros = np.where(h == 0)[0]
        i = rng.integers(len(zeros))
        j = rng.integers(len(ones))
        i_idx = zeros[i]
        j_idx = ones[j]

    L = np.arange(-(n - 1), n)
    R = np.zeros((n, 2 * n - 1))

        vm = (idx_minus >= 0) & (idx_minus < n)

        idx_plus = i + L
