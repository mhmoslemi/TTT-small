    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode="full")  # length 2n-1, index 0 to 2n-2
    # corr[index] corresponds to lag m = index - (n-1)
    
    t0 = h[j]
    A = np.zeros(2*n-1)
    for idx in range(2*n-1):
        m = idx - (n-1)
        a = 0.0
        # h[j] as first factor: i+m = j → i = j-m, valid if 0<=j-m<n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to coordinates with room
        order = np.argsort(1.0 - h)  # most slack first
    else:
        # Need to decrease sum: remove from coordinates with value
        order = np.argsort(h)  # least value first (will hit 0 first)
    
    h_new = h.copy()
    remaining = diff
    for idx in order:
        if remaining > 0:
            add = min(1.0 - h_new[idx], remaining)
            h_new[idx] += add
            remaining -= add
        else:
            sub = min(h_new[idx], -remaining)
            h_new[idx] -= sub
            remaining += sub
        if abs(remaining) < 1e-14:
            break
    return h_new

def soft_max_obj(h, n, beta):
    """Smoothed max of correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    if beta < 1:
        return float(m * dx)
    shifted = (corr - m) * beta
    # shift for numerical stability
    shifted -= shifted.max()
    result = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    return float(result)

def slsqp_optimize(h, n, beta, maxiter=300):
    """SLSQP with smooth objective."""
    ts = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts}]
    res = minimize(lambda x: soft_max_obj(x, n, beta), h,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_h(res.x, ts)
    return project_h(h, ts)

def coord_sweep(h, n, max_sweeps=50, time_limit=60):
    """Coordinate descent with sum constraint via compensation."""
    ts = n / 2.0
    t0 = time.time()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            old_hj = h[j]
            # Try different values for h[j]
            # After changing, project to maintain sum
            candidates = [0.0, 1.0, old_hj * 0.5, old_hj * 1.5,
                         max(0, old_hj - 0.1), min(1, old_hj + 0.1)]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            
            best_val = best_c5
            best_t = old_hj
            
            for t in candidates:
                if abs(t - old_hj) < 1e-10:
                    continue
                h[j] = t
                h = project_h(h, ts)
                c5 = compute_c5(h)
                if c5 < best_val - 1e-12:
                    best_val = c5
                    best_t = t
                    # Keep h at this value
                else:
                    h[j] = old_hj  # revert (approximately)
                    h = project_h(h, ts)
            
            if best_t != old_hj:
                h[j] = best_t
                h = project_h(h, ts)
                best_c5 = best_val
                improved = True
        
        if not improved:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, ts):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(30):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def soft_max_c5(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = (corr - m) * beta
    s -= s.max()
    return float((m + np.log(np.sum(np.exp(s))) / beta) * dx)

def slsqp_solve(h, n, beta, maxiter=250):
    ts = n / 2.0
    res = minimize(lambda x: soft_max_c5(x, n, beta), h,
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_h(res.x, ts)
    return project_h(h, ts)

def subgrad_fine_tune(h, n, time_limit=30):
    ts = n / 2.0
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.005
    
    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        thresh = max_val - 0.02
        active = np.where(corr > thresh)[0]
        
        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j - m < n:
                    g += (1.0 - h[j - m])
                if 0 <= j + m < n:
                    g -= h[j + m]
                grad[j] += g
        grad /= max(1, len(active))
        
        h_new = project_h(h - lr * grad, ts)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.2, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_h, best_c5

def local_search(h, n, time_limit=90):
    ts = n / 2.0
    t0 = time.time()
    best_c5 = compute_c5(h)
    rng_ls = np.random.RandomState(12345)
    
    for sweep in range(300):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng_ls.permutation(n)
        for j in order:
            old = h[j]
            cands = [0.0, 1.0,
                     max(0.0, min(1.0, old - 0.15)),
                     max(0.0, min(1.0, old + 0.15)),
                     max(0.0, min(1.0, old - 0.05)),
                     max(0.0, min(1.0, old + 0.05)),
                     max(0.0, min(1.0, old - 0.02)),
                     max(0.0, min(1.0, old + 0.02))]
            best_t = old
            best_v = best_c5
            for t in cands:
                if abs(t - old) < 1e-9:
                    continue
                h[j] = t
                h = project_h(h, ts)
                v = compute_c5(h)
                if v < best_v - 1e-13:
                    best_v = v
                    best_t = t
                else:
                    h[j] = old
                    h = project_h(h, ts)
            if best_t != old:
                h[j] = best_t
                h = project_h(h, ts)
                best_c5 = best_v
                improved = True
        if not improved:
            break
    return h, best_c5

def block_move_search(h, n, time_limit=40):
    ts = n / 2.0
    t0 = time.time()
    best_c5 = compute_c5(h)
    rng_bm = np.random.RandomState(999)
    
    for _ in range(5000):
        if time.time() - t0 > time_limit:
            break
        j = rng_bm.randint(n)
        k = rng_bm.randint(n)
        if abs(j - k) < 2:
            continue
        old_j, old_k = h[j], h[k]
        for delta in [0.05, -0.05, 0.1, -0.1]:
            new_j = old_j + delta
            new_k = old_k - delta
            if 0 <= new_j <= 1 and 0 <= new_k <= 1:
                h[j], h[k] = new_j, new_k
                v = compute_c5(h)
                if v < best_c5 - 1e-12:
                    best_c5 = v
                else:
                    h[j], h[k] = old_j, old_k
        h = project_h(h, ts)
    return h, best_c5
