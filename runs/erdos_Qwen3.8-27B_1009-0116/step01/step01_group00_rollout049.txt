def pair_swap_optimize(h, i, j, c):
    # c = current correlation values (length 2n-1)
    # Returns optimal d for h[i]+=d, h[j]-=d
    n = len(h)
    k_offset = n - 1
    
    # Compute s_k = effect of unit swap on c_k
    s = np.zeros(2*n-1)
    for k in range(2*n-1):
        shift = k - k_offset  # actual shift
        # c_k = sum_m h[m]*(1-h[m+shift])
        # d(c_k)/d(h[i]) = (1-h[i+shift]) - h[i-shift] (for valid indices)
        # d(c_k)/d(h[j]) = (1-h[j+shift]) - h[j-shift]
        # total effect of h[i]+=d, h[j]-=d:
        # d * [(1-h[i+shift]) - h[i-shift] - (1-h[j+shift]) + h[j-shift]]
        
        val = 0.0
        ip = i + shift
        im = i - shift
        jp = j + shift
        jm = j - shift
        if 0 <= ip < n:
            val += 1.0 - h[ip]
        if 0 <= im < n:
            val -= h[im]
        if 0 <= jp < n:
            val -= (1.0 - h[jp])
        if 0 <= jm < n:
            val += h[jm]
        s[k] = val
    
    # Now minimize max_k (c[k] + d*s[k]) over feasible d
    d_min = -h[i]
    d_max = 1.0 - h[i]
    # Also h[j] - d >= 0 => d <= h[j]
    # h[j] - d <= 1 => d >= h[j] - 1
    d_min = max(d_min, h[j] - 1.0)
    d_max = min(d_max, h[j])
    
    if d_min >= d_max:
        return 0.0
    
    # Ternary search
    for _ in range(60):
        m1 = d_min + (d_max - d_min) / 3.0
        m2 = d_max - (d_max - d_min) / 3.0
        f1 = np.max(c + m1 * s)
        f2 = np.max(c + m2 * s)
        if f1 < f2:
            d_max = m2
        else:
            d_min = m1
    
    d_opt = (d_min + d_max) / 2.0
    return d_opt
```

def compute_s(h, i, j, n):
    shifts = np.arange(2*n-1) - (n-1)  # shift_k for k=0..2n-2
    # f_i[k] = (1-h[i+shift_k]) - h[i-shift_k] with boundary = 0
    s = np.zeros(2*n-1)
    ip = i + shifts
    im = i - shifts
    jp = j + shifts
    jm = j - shifts
    
    mask_ip = (ip >= 0) & (ip < n)
    mask_im = (im >= 0) & (im < n)
    mask_jp = (jp >= 0) & (jp < n)
    mask_jm = (jm >= 0) & (jm < n)
    
    s += np.where(mask_ip, 1.0 - h[ip[mask_ip]], 0.0)
    s -= np.where(mask_im, h[im[mask_im]], 0.0)
    s -= np.where(mask_jp, 1.0 - h[jp[mask_jp]], 0.0)
    s += np.where(mask_jm, h[jm[mask_jm]], 0.0)
    return s
```

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _c5_from_corr(c, dx):
    return np.max(c) * dx

def _lps_obj(h, alpha, target, mu):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    smooth = m + np.log(np.sum(np.exp(s))) / alpha
    penalty = mu * (np.sum(h) - target) ** 2
    return smooth + penalty

def _pair_swap_s(h, i, j, n):
    """Compute effect vector s for pair swap h[i]+=d, h[j]-=d"""
    shifts = np.arange(2 * n - 1) - (n - 1)
    s = np.zeros(2 * n - 1)
    ip = i + shifts
    im = i - shifts
    jp = j + shifts
    jm = j - shifts
    m_ip = (ip >= 0) & (ip < n)
    m_im = (im >= 0) & (im < n)
    m_jp = (jp >= 0) & (jp < n)
    m_jm = (jm >= 0) & (jm < n)
    s[m_ip] += 1.0 - h[ip[m_ip]]
    s[m_im] -= h[im[m_im]]
    s[m_jp] -= 1.0 - h[jp[m_jp]]
    s[m_jm] += h[jm[m_jm]]
    return s

def _optimal_d(h, i, j, c, s, n):
    """Find optimal d for pair swap via ternary search"""
    d_lo = max(-h[i], h[j] - 1.0) + 1e-12
    d_hi = min(1.0 - h[i], h[j]) - 1e-12
    if d_lo >= d_hi:
        return 0.0
    for _ in range(80):
        m1 = d_lo + (d_hi - d_lo) / 3.0
        m2 = d_hi - (d_hi - d_lo) / 3.0
        if np.max(c + m1 * s) < np.max(c + m2 * s):
            d_hi = m2
        else:
            d_lo = m1
    return (d_lo + d_hi) / 2.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    h = h * (target / s)

    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
            else:
                a = min(-diff, h[i])
            h[i] += a if diff > 0 else -(-a)
            diff -= a if diff > 0 else a
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(h, alpha, target, mu):

def _pair_s(h, i, j, n):
    shifts = np.arange(2 * n - 1) - (n - 1)
    s = np.zeros(2 * n - 1)

    im = i - shifts
    jp = j + shifts
    jm = j - shifts

    m2 = (im >= 0) & (im < n)
    m3 = (jp >= 0) & (jp < n)

    s[m1] += 1.0 - h[ip[m1]]
    s[m2] -= h[im[m2]]
    s[m3] -= 1.0 - h[jp[m3]]
    s[m4] += h[jm[m4]]
    return s

        t4_end = t_start + budget_s - 5
        h, local_c5 = _pair_swap_phase(h, dx, target, n, rng, t4_end, max_passes=3000)
        if local_c5 < best_c5:
            best_c5 = local_c5
            best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _c5(np.correlate(best_h, 1.0 - best_h, mode='full'), dx)
    return (best_h, best_c5, n)
