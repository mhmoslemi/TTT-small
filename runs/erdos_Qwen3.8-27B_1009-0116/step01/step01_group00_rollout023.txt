            h_relax = _slsqp_refine(h_relax, n, dx, 100, 400)
            c_relax = _c5(h_relax, dx)
            if c_relax < best_c5:
                best_c5 = c_relax
                best_h = h_relax

    # Phase 3: Intensive SA on best solution rounded to 0-1
    if time.time() - t_start < budget_s - 200:
        h01 = _round_to_01(best_h, n, rng)
        t_rem = budget_s - 250 - (time.time() - t_start)
        if t_rem > 10:
            h01, cost01 = _sa_01(h01, rng, t_start, time.time() + t_rem * 0.7)
            h01, cost01 = _greedy_01(h01, max_sweeps=60)
            c5_01 = cost01 * dx
            if c5_01 < best_c5:
                best_c5 = c5_01
                best_h = h01.copy()
            # Relax
            if time.time() - t_start < budget_s - 100:
                h_r = _repair(h01 + 0.03 * rng.randn(n), n)
                h_r = _slsqp_refine(h_r, n, dx, 200, 500)
                c_r = _c5(h_r, dx)
                if c_r < best_c5:
                    best_c5 = c_r
                    best_h = h_r

    # Phase 4: Final SLSQP refinement from best
    if time.time() - t_start < budget_s - 30:
        h_new = _slsqp_refine(best_h, n, dx, 500, 600)
        c_new = _c5(h_new, dx)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Fine coordinate search
    if time.time() - t_start < budget_s - 10:
        h = best_h.copy()
        bl = best_c5
        steps = [0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05]
        for _p in range(20000):
            if time.time() - t_start > budget_s - 10:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in steps:
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bl - 1e-10:
                    bl = c
                else:
                    h[i], h[j] = ni - d, nj + d
        if bl < best_c5:
            best_c5 = bl
            best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
