def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_box_sum(v, n):
    target = n * 0.5
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0

        mid = (lo + hi) * 0.5

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    grad = np.zeros(n)

        if d >= 0:
            if d < n:
                grad[:n - d] += pk * (1.0 - h[d:])
                grad[d:] -= pk * h[:n - d]
        else:

                grad[:n - dd] -= pk * h[dd:]
    grad += 2.0 * rho * (float(np.sum(h)) - n * 0.5)

def projected_subgrad(h, n, steps, tau_init, tau_decay):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(steps):
        t = tau_init * (tau_decay ** it)
        c = compute_corr(h)
        # Use top-3 lags for richer subgradient
        top_idx = np.argsort(c)[-3:]
        dk = c[top_idx]
        weights = np.exp((dk - dk[-1]) * 50.0)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            d = int(k) - (n - 1)
            if d >= 0:
                if d < n:
                    grad[:n - d] += w * (1.0 - h[d:])
                    grad[d:] -= w * h[:n - d]
            else:
                dd = -d
                if dd < n:
                    grad[dd:] += w * (1.0 - h[:n - dd])
                    grad[:n - dd] -= w * h[dd:]
        grad *= 2.0 / n
        h = project_box_sum(h - t * grad, n)
        c5 = compute_c5(h)
        if c5 < best_c5:

def coord_descent(h, n, max_sweeps):
    h = h.copy()
    target = n * 0.5
    best = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_val = best
            best_cand = old
            for cand in (0.0, 1.0):
                if abs(cand - old) < 1e-10:
                    continue
                h[i] = cand
                h = project_box_sum(h, n)
                val = compute_c5(h)

                    best_cand = cand
            if best_cand != old:
                h[i] = best_cand
                h = project_box_sum(h, n)
                best = best_val
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, compute_c5(h)
