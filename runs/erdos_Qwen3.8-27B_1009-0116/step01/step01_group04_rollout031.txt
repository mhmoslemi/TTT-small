import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    # If still not matching, adjust
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-8:
        # Add/subtract to elements closest to 0.5
        idx = np.argsort(np.abs(x - 0.5))[:int(abs(diff)) + 1]
        sign = 1 if diff > 0 else -1
        x[idx[:int(abs(diff))]] += sign * 0.5  # crude
        x = np.clip(x, 0, 1)
    # Final scaling
    s2 = np.sum(x)
    if s2 > 1e-12:
        x = x * (tgt / s2)
        x = np.clip(x, 0.0, 1.0)
    return x

def project_onto_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    tgt = n / 2.0
    lo, hi = -2.0, 2.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2.0
        xp = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(xp)
        if s > tgt:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune
    s = np.sum(xp)
    if abs(s - tgt) > 1e-10:
        # Distribute residual
        residual = tgt - s
        # Add to elements with most room
        if residual > 0:
            order = np.argsort(1.0 - xp)
        else:
            order = np.argsort(xp)
        for i in order:
            add = min(residual, 1.0 - xp[i]) if residual > 0 else max(residual, -xp[i])
            xp[i] += add
            residual -= add
            if abs(residual) < 1e-12:
                break
    return xp

def compute_grad(h, w, n):
    """Compute gradient of smooth max correlation w.r.t. h."""
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[s + n - 1]
        if ws < 1e-14:
            continue
        # First term: (1-h[j-s]) for j where 0<=j-s<n
        j_lo = max(0, s)
        j_hi = min(n-1, s + n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += ws * (1.0 - h[j_lo - s : j_hi + 1 - s])
        # Second term: -h[j+s] for j where 0<=j+s<n
        j_lo2 = max(0, -s)
        j_hi2 = min(n-1, n-1 - s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= ws * h[j_lo2 + s : j_hi2 + 1 + s]
    return grad

def soft_max_weights(corr, tau):
    """Compute softmax weights for the correlation values."""
    vals = corr - np.max(corr)
    ev = np.exp(vals / tau)
    return ev / np.sum(ev)

def pgd_step(h, lr, tau, n, binarize=0.0):
    """One projected gradient descent step."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = soft_max_weights(corr, tau)
    grad = compute_grad(h, w, n)
    h_new = h - lr * grad * (2.0 / n)  # scale by dx
    # Binarization push
    if binarize > 0:
        h_new = h_new + binarize * (h_new - 0.5)
    h_new = project_onto_feasible(h_new, n)
    return h_new

    best_val = compute_c5(h0, dx0)
    best_n = n0
    
    def time_left():

    # Phase 1: PGD on parent grid
    taus_p1 = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]

import numpy as np
import time
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    tgt = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    xp = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    resid = tgt - xp.sum()
    if abs(resid) > 1e-13:
        if resid > 0:
            idx = np.argsort(1.0 - xp)[::-1]
        else:
            idx = np.argsort(xp)
        for i in idx:
            room = (1.0 - xp[i]) if resid > 0 else xp[i]
            step = min(abs(resid), room)
            if resid > 0:
                xp[i] += step
                resid -= step
            else:
                xp[i] -= step
                resid += step
            if abs(resid) < 1e-14:
                break
    return xp

def compute_grad(h, w, n):
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        if ws < 1e-14:
            continue
        a, b = max(0, s), min(n - 1, s + n - 1)
        if a <= b:
            g[a:b + 1] += ws * (1.0 - h[a - s:b + 1 - s])
        a2, b2 = max(0, -s), min(n - 1, n - 1 - s)
        if a2 <= b2:
            g[a2:b2 + 1] -= ws * h[a2 + s:b2 + 1 + s]
    return g

def fista_run(h0, n, tau, lr, niter, tl=None):
    x = h0.copy()
    y = x.copy()
    t = 1.0
    for it in range(niter):
        if tl is not None and it % 200 == 0 and tl() < 5:
            break
        c = np.correlate(y, 1.0 - y, mode="full")
        v = c - c.max()
        e = np.exp(v / tau)
        w = e / e.sum()
        g = compute_grad(y, w, n)
        xn = project_feasible(y - lr * g, n)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = xn + ((t - 1.0) / tn) * (xn - x)
        t = tn
        x = xn
    return x

def binarize(h, n):
    b = (h >= 0.5).astype(float)
    if n % 2 == 0:
        tgt = n // 2
        s = int(b.sum())
        if s > tgt:
            ones = np.where(b == 1)[0]
            order = np.argsort(h[ones])
            b[ones[order[:s - tgt]]] = 0
        elif s < tgt:
            zeros = np.where(b == 0)[0]
            order = np.argsort(-h[zeros])
            b[zeros[order[:tgt - s]]] = 1
    else:
        tgt = n / 2.0
        b = project_feasible(b, n)
    return b

def local_search_binary(b, n, rng, max_sweeps, tl=None, kick_every=50, kick_sz=4):
    dx = 2.0 / n
    best = b.copy()
    bf = compute_c5(best, dx)
    for sw in range(max_sweeps):
        if tl is not None and tl() < 10:
            break
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-15
        best_pair = None
        npairs = len(ones) * len(zeros)
        if npairs <= 25000:
            for ii in range(len(ones)):
                i = ones[ii]
                for jj in range(len(zeros)):
                    j = zeros[jj]
                    best[i] = 0
                    best[j] = 1
                    f = compute_c5(best, dx)
                    d = bf - f
                    if d > best_delta:
                        best_delta = d
                        best_pair = (i, j)
                    best[i] = 1
                    best[j] = 0
        else:
            for _ in range(20000):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                best[i] = 0
                best[j] = 1
                f = compute_c5(best, dx)
                d = bf - f
                if d > best_delta:
                    best_delta = d
                    best_pair = (i, j)
                best[i] = 1
                best[j] = 0
        if best_pair is not None:
            i, j = best_pair
            best[i] = 0
            best[j] = 1
            bf -= best_delta
        else:
            if (sw + 1) % kick_every == 0 and len(ones) >= 2 and len(zeros) >= 2:
                ko = rng.choice(ones, kick_sz // 2, replace=False)
                kz = rng.choice(zeros, kick_sz // 2, replace=False)
                best[ko] = 0
                best[kz] = 1
                f = compute_c5(best, dx)
                if f >= bf:
                    best[ko] = 1
                    best[kz] = 0
    return best, bf

def make_time_fn(t_start, budget):
    tl_val = [budget - (time.time() - t_start)]
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    def tl():
        return budget_s - (time.time() - t_start) - 5

    # Get initial values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(96, 0.5)
    n0 = len(h0)

    best_h = h0.copy()
    best_val = compute_c5(h0, 2.0 / n0)
    best_n = n0

    # Helper to update best
    def update(x, n, val):
        nonlocal best_h, best_val, best_n
        if val < best_val:
            best_val = val
            best_h = x.copy()
            best_n = n

    # Phase 1: FISTA on parent grid
    if tl() > 30:
        x = project_feasible(h0, n0)
        for tau in [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            if tl() < 15:
                break
            x = fista_run(x, n0, tau, 2.0, 1500, tl)
            v = compute_c5(x, 2.0 / n0)
            update(x, n0, v)

    # Phase 2: Try even grids (binary-friendly)
    for n_even in [96, 192, 48]:
        if tl() < 40:
            break
        dx_e = 2.0 / n_even
        # Initialize by interpolating best_h
        xs = np.linspace(0, 2, best_n)
        xd = np.linspace(0, 2, n_even)
        f_interp = interp1d(xs, best_h, kind='linear', fill_value='extrapolate')
        xe = project_feasible(f_interp(xd), n_even)

        # FISTA continuous optimization
        if tl() > 30:
            for tau in [0.3, 0.15, 0.07, 0.03, 0.01, 0.005]:
                if tl() < 10:
                    break
                xe = fista_run(xe, n_even, tau, 2.0, 1000, tl)
                v = compute_c5(xe, dx_e)
                update(xe, n_even, v)

        # Binarize and local search
        if tl() > 20:

            vb = compute_c5(b, dx_e)
            update(b, n_even, vb)

            # Local search - main optimization
            if tl() > 30:
                b, vb = local_search_binary(b, n_even, rng, 5000, tl,
                                           kick_every=40, kick_sz=4)
                update(b, n_even, vb)

            # Additional restarts from perturbed binary
            for _ in range(3):
                if tl() < 15:
                    break
                b2 = b.copy()
                ones = np.where(b2 == 1)[0]
                zeros = np.where(b2 == 0)[0]
                nk = min(6, len(ones) // 2, len(zeros) // 2)
                if nk >= 2:
                    ko = rng.choice(ones, nk, replace=False)
                    kz = rng.choice(zeros, nk, replace=False)
                    b2[ko] = 0
                    b2[kz] = 1
                b2, v2 = local_search_binary(b2, n_even, rng, 3000, tl,
                                             kick_every=30, kick_sz=6)
                update(b2, n_even, v2)

    # Phase 3: Random binary initialization for even grids
    for n_r in [96, 192]:
        if tl() < 30:
            break
        dx_r = 2.0 / n_r
        br = np.zeros(n_r)
        br[rng.choice(n_r, n_r // 2, replace=False)] = 1.0
        for _ in range(5):
            if tl() < 10:
                break
            br, vbr = local_search_binary(br, n_r, rng, 3000, tl,
                                          kick_every=30, kick_sz=4)
            update(br, n_r, vbr)
            # Kick for next restart
            ones = np.where(br == 1)[0]
            zeros = np.where(br == 0)[0]
            if len(ones) >= 3 and len(zeros) >= 3:
                ko = rng.choice(ones, 3, replace=False)
                kz = rng.choice(zeros, 3, replace=False)
                br[ko] = 0
                br[kz] = 1

    # Phase 4: Polish best with FISTA (slight unbinarization)
    if tl() > 20:
        x_pol = best_h + 0.05 * rng.standard_normal(best_n)
        x_pol = project_feasible(x_pol, best_n)
        for tau in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if tl() < 8:
                break
            x_pol = fista_run(x_pol, best_n, tau, 3.0, 800, tl)
            v = compute_c5(x_pol, 2.0 / best_n)
            update(x_pol, best_n, v)

    # Phase 5: Final local search on best (if binary-ish)
    if tl() > 20:
        b_final = binarize(best_h, best_n) if best_n % 2 == 0 else project_feasible(best_h, best_n)
        if best_n % 2 == 0:
            b_final, vf = local_search_binary(b_final, best_n, rng, 5000, tl,
                                              kick_every=30, kick_sz=4)
            update(b_final, best_n, vf)

    # Final evaluation
    dx_f = 2.0 / best_n
    best_h = project_feasible(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    if final_val < best_val:
        best_val = final_val

    return (best_h, best_val, best_n)
