from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_constraint(v, n):
    target = n * 0.5
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        interior = (p > 1e-12) & (p < 1.0 - 1e-12)
        if interior.sum() > 0:
            p[interior] += diff / interior.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lbfgs_phase(h, n, t, lam, maxiter):
    def obj(x):
        c = _corr_vals(x)
        m = c.max()
        s = m + np.log(np.sum(np.exp((c - m) * t))) / t
        return s + lam * (x.sum() - n * 0.5) ** 2

    def grad(x):
        c = _corr_vals(x)
        m = c.max()
        e = np.exp((c - m) * t)
        w = e / e.sum()
        dx = 2.0 / n
        lags = np.arange(2 * n - 1) - (n - 1)
        g = np.zeros(n)
        for idx in range(2 * n - 1):
            if w[idx] < 1e-10:
                continue
            l = lags[idx]
            ws = w[idx]
            js, je = max(0, -l), min(n, n - l)
            if je > js:
                g[js:je] += ws * (1.0 - x[js + l:je + l])
            js2, je2 = max(0, l), min(n, n + l)
            if je2 > js2:
                g[js2:je2] -= ws * x[js2 - l:je2 - l]
        g *= dx
        g += 2.0 * lam * (x.sum() - n * 0.5)
        return g

    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
    return project_constraint(res.x, n)

def subgrad_method(h, n, iters, lr0):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(iters):
        lr = lr0 / (1.0 + 0.003 * it)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        mi = np.argmax(c)
        lag = mi - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[0:n - lag] += (1.0 - h[lag:n])
            g[lag:] -= h[:n - lag]
        else:
            g[-lag:] += (1.0 - h[0:n + lag])  # h[j+lag] for j in [-lag, n-1]
            g[0:n + lag] -= h[-lag:]  # h[j-lag] for j in [0, n-1-l]
        g *= dx
        h_new = h - lr * g
        h_new = project_constraint(h_new, n)
        v = compute_c5(h_new)
        if v < best:
            best = v
            best_h = h_new.copy()
        h = h_new
    return best_h, best

        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        interior = (p > 1e-12) & (p < 1.0 - 1e-12)
        if interior.sum() > 0:
            p[interior] += diff / interior.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _obj_fn(x, t, lam, n):
    c = _corr_vals(x)
    m = c.max()
    s = m + np.log(np.sum(np.exp((c - m) * t))) / t
    return s + lam * (x.sum() - n * 0.5) ** 2

def _grad_fn(x, t, lam, n):
    c = _corr_vals(x)
    m = c.max()
    e = np.exp((c - m) * t)
    w = e / e.sum()
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-10:
            continue
        l = lags[idx]
        ws = w[idx]
        js = max(0, -l)
        je = min(n, n - l)
        if je > js:
            g[js:je] += ws * (1.0 - x[js + l:je + l])
        js2 = max(0, l)
        je2 = min(n, n + l)
        if je2 > js2:
            g[js2:je2] -= ws * x[js2 - l:je2 - l]
    g *= dx
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _lbfgs(h, n, t, lam, maxiter):
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj_fn, h, args=(t, lam, n),
                   method='L-BFGS-B', jac=_grad_fn, bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16,
                            'gtol': 1e-9, 'maxls': 20})
    return project_constraint(res.x, n)

def _subgrad(h, n, iters, lr0):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(iters):
        lr = lr0 / (1.0 + 0.003 * it)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        mi = int(np.argmax(c))
        lag = mi - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[0:n - lag] += (1.0 - h[lag:n])
            g[lag:] -= h[:n - lag]
        else:
            g[-lag:] += (1.0 - h[0:n + lag])
            g[0:n + lag] -= h[-lag:]
        g *= dx
        h_new = project_constraint(h - lr * g, n)
        v = compute_c5(h_new)
        if v < best:
            best = v
            best_h = h_new.copy()
        h = h_new
    return best_h, best
