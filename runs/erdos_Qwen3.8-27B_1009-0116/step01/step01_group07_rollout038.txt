    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _subgrad(corr, h, n):
    """Compute subgradient of max corr at the maximizing lag(s)"""
    max_val = np.max(corr)
    # Weight lags near the max
    thresh = max_val - 0.01 * max_val  # include lags within 1% of max
    mask = corr >= thresh
    count = np.sum(mask)
    grad = np.zeros(n)
    for lag_idx in np.where(mask)[0]:
        k = lag_idx - (n - 1)  # actual lag from -(n-1) to (n-1)
        if k >= 0:
            for i in range(n - k):
                grad[i] += (1.0 - h[i + k])
                grad[i + k] += (-h[i])
        else:
            for i in range(n + k):
                grad[i] += (-h[i - k])
                grad[i - k] += (1.0 - h[i])
    grad /= count
    return grad

                    h2[i] += d

                    if not (0.0 <= h2[i] <= 1.0 and 0.0 <= h2[j] <= 1.0):
                        continue
                    c = _c5(h2)
                    if c < best - 1e-13:
                        h = h2
                        best = c
                        improved = True
                        break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _G_target, _G_dx, _G_alpha
    t0 = time.time()

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    pn = len(parent)
    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = pn

    n_list = [pn, 60, 80, 90, 100, 120]
    n_list = sorted(set(n_list))

    for n in n_list:
        if budget_s - (time.time() - t0) < 20:
            break
        target = n / 2.0
        dx = 2.0 / n
        _G_target = target
        _G_dx = dx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        n_restarts = 3 if n <= 80 else 2
        for r in range(n_restarts):
            if budget_s - (time.time() - t0) < 12:
                break
            if r == 0 and n == pn:
                hc = parent.copy()
            elif r == 0:
                hc = np.full(n, 0.5)
                rng0 = np.random.RandomState(seed + n)
                hc += 0.25 * np.sin(2.0 * np.pi * 2.0 * np.arange(n) / n)
                hc = _project(hc, target)
            else:
                hc = _sine_init(n, seed + n * 97 + r * 31)

            # PGD
            if budget_s - (time.time() - t0) > 15:
                h_sg, c_sg = _run_pgd(hc, target, 2000, 0.012, 0.9994, 0.06)
                if c_sg < best_c5:
                    best_c5, best_h, best_n = c_sg, h_sg, n
                    print(f"  PGD n={n} r={r}: C5={c_sg:.6f}")
                hc = h_sg

            # SLSQP with increasing alpha
            for al in (30, 60, 120, 300, 600, 1200):
                if budget_s - (time.time() - t0) < 8:
                    break
                _G_alpha = al
                try:
                    mx_it = 80 if n <= 80 else 50
                    res = minimize(_obj_slsqp, hc, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': mx_it, 'ftol': 1e-12})
                    hn = _project(res.x, target)
                    cn = _c5(hn)
                    if cn < best_c5:
                        best_c5, best_h, best_n = cn, hn, n
                        print(f"  SLSQP n={n} a={al} r={r}: C5={cn:.6f}")
                    hc = hn
                except Exception:
                    pass

            # Local search
            if budget_s - (time.time() - t0) > 8:
                hc, cg = _local_search(hc, target, 25)
                if cg < best_c5:
                    best_c5, best_h, best_n = cg, hc, n
                    print(f"  Loc1 n={n} r={r}: C5={cg:.6f}")
                hc, cg = _pair_local(hc, target, 8)
                if cg < best_c5:
                    best_c5, best_h, best_n = cg, hc, n
                    print(f"  Loc2 n={n} r={r}: C5={cg:.6f}")

    # Final refinement on best
    if budget_s - (time.time() - t0) > 10:
        bn = best_n
        bt = bn / 2.0
        _G_target = bt
        _G_dx = 2.0 / bn
        for al in (200, 500, 1000, 2000):
            if budget_s - (time.time() - t0) < 6:
                break
            _G_alpha = al
            try:
                res = minimize(_obj_slsqp, best_h, method='SLSQP',
                               bounds=[(0.0, 1.0)] * bn, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-13})
                hn = _project(res.x, bt)
                cn = _c5(hn)
                if cn < best_c5:
                    best_c5, best_h = cn, hn
                    print(f"  Fin a={al}: C5={cn:.6f}")
            except Exception:
                pass
        best_h, best_c5 = _local_search(best_h, bt, 30)
        best_h, best_c5 = _pair_local(best_h, bt, 10)

    # Enforce exact sum
    h_ret = best_h.copy()
    tgt = best_n / 2.0
    diff = tgt - np.sum(h_ret)
    if abs(diff) > 1e-13:
        m = (h_ret > 1e-9) & (h_ret < 1.0 - 1e-9)
        cnt = int(np.sum(m))
        if cnt > 0:
            h_ret[m] += diff / cnt
        h_ret = np.clip(h_ret, 0.0, 1.0)
        diff2 = tgt - np.sum(h_ret)
        if abs(diff2) > 1e-14:
            h_ret -= diff2 / best_n
            h_ret = np.clip(h_ret, 0.0, 1.0)

    c5f = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return h_ret, c5f, best_n
