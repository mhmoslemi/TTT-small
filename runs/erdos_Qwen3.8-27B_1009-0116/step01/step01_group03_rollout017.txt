```python
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

    """Project v onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Bisection on offset
    lo, hi = -max(v), 1 - min(v)  # offsets that could work
    # Actually simpler: find t such that sum(clip(v-t, 0, 1)) = target_sum

    dx = 2.0 / n
    corr = full_corr(h)
    k_star = np.argmax(corr)
    # Gradient of corr[k_star] w.r.t. h

        valid1 = np.arange(0, n - shift)  # j values where h[j+shift] is valid
        grad[valid1] += (1.0 - h[shift:])  # (1-h[j+shift]) for j in valid1
        valid2 = np.arange(shift, n)  # i values where h[i-shift] is valid

        grad[valid2 + s] -= h[:n-s]  # -h[j]
    
    h_new = h - alpha * grad * dx
    return project_box_sum(h_new, n / 2.0)
```

```python
def compute_c5(h):

    return float(np.max(corr) * dx)

    v = np.asarray(v, dtype=float)
    # Quick check
    s = v.sum()
    if abs(s - target) < 1e-10:
        return np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0

            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(v - t, 0.0, 1.0)

def corr_and_grad(h):
    """Return (corr_array, gradient_of_max_corr)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    shift = k_star - n + 1
    grad = np.zeros(n)

    elif shift > 0:
        grad[:n-shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] += (1.0 - h[:-s])
        grad[:n-s] -= h[s:]
    return corr, grad * dx, k_star

def subgrad_momentum(h, n, n_iters, alpha0, momentum):
    """Subgradient method with momentum and projection."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    for t in range(n_iters):
        alpha = alpha0 / (1.0 + t * 0.01)
        _, grad, _ = corr_and_grad(h)
        v = momentum * v + grad
        h_new = project_box_sum(h - alpha * v, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[len(idx)//2]] = np.clip(h[idx[len(idx)//2]] + diff2, 0, 1)
    return h

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_polish(h, n, ts, beta, maxiter=500):
    from scipy.optimize import minimize
    def obj(x):
        return smooth_obj(x, n, beta)
    def con_fun(x):
        return x.sum() - ts
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def binary_ls(h, n, ts, max_sweeps=200, time_limit=120):
    """Binary local search with single flips and pair swaps."""
    t0 = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        # Single flips
        for j in range(n):
            orig = h[j]
            for t_val in (0.0, 1.0):
                if abs(t_val - orig) < 1e-8:
                    continue
                h[j] = t_val
                h = project_feasible(h, ts)
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    improved = True
                    orig = h[j]  # update orig to accepted value
                else:
                    h[j] = orig  # RESTORE
                    h = project_feasible(h, ts)
        # Pair swaps (for binary h)
        if time.time() - t0 > time_limit:

        ones = np.where(h > 0.5)[0]

    return h, best_c5

def subgrad_descend(h, n, ts, n_iters=300, alpha0=0.5):
    """Subgradient descent with projection."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.7
    for t in range(n_iters):
        alpha = alpha0 / (1.0 + 0.05 * t)
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        s = k_star - n + 1
        dx = 2.0 / n
        grad = np.zeros(n)
        if s == 0:
            grad = 1.0 - 2.0 * h
        elif s > 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            ss = -s
            grad[ss:] += (1.0 - h[:n-ss])
            grad[:n-ss] -= h[ss:]
        grad *= dx
        v = momentum * v + grad
        h_new = project_feasible(h - alpha * v, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def upsample(h, n_new):
    """Upsample h to n_new points via interpolation and reproject."""
    n_old = len(h)
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    h_new = np.interp(xn, xo, h)

def make_binary(h, ts):

    return h_bin
```

- Fix all coordinates except i
- f as function of h[i]: c_i * h[i] - Q_{ii} * h[i]^2 - 2 * h[i] * (sum_{j≠i} Q_{ij} h[j])
- This is a concave quadratic in h[i] (since Q_{ii} > 0), so the minimum is at h[i]=0 or h[i]=1.

```python
def weighted_sum_ls(h, n, weights, max_sweeps=100):
    """Coordinate descent on weighted sum of correlations."""
    dx = 2.0 / n
    h = h.copy()
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Try h[i] = 0 and h[i] = 1, pick the one that minimizes weighted sum
            orig = h[i]
            best_val = np.inf
            best_h = orig
            for val in (0.0, 1.0):
                if abs(val - orig) < 1e-8:
                    continue
                h[i] = val
                # Compute weighted sum
                ws = 0.0
                for k, w in enumerate(weights):
                    s = k - n + 1  # shift
                    if s == 0:
                        d = np.sum(h[:n] * (1-h[:n]))
                    elif s > 0:
                        d = np.sum(h[:n-s] * (1-h[s:]))
                    else:
                        d = np.sum(h[-s:] * (1-h[:n+s]))
                    ws += w * d
                if ws < best_val:
                    best_val = ws
                    best_h = val
            h[i] = best_h
            if best_h != orig:
                improved = True
        if not improved:
            break
    return h
```

This is O(n * n * n) = O(n^3) per sweep (for each of n coordinates, compute n correlations of length n). For n=140, that's 2.7M per sweep. Fast enough.

```python
def weighted_objective(h, n, weights):
    """Compute sum_k w_k * d_k(h) where d_k is the shift-k correlation."""
    dx = 2.0 / n
    total = 0.0
    for s in range(-(n-1), n):
        k = s + n - 1  # index into weights
        if k < 0 or k >= 2*n-1:
            continue
        w = weights[k]
        if w < 1e-15:
            continue
        if s == 0:
            d = np.sum(h * (1-h))
        elif s > 0:
            d = np.sum(h[:n-s] * (1-h[s:]))
        else:
            d = np.sum(h[-s:] * (1-h[:n+s]))
        total += w * d
    return total * dx
```

This is O(n^2) per call. For coordinate descent with n coordinates, each sweep is O(n^3). For n=140, that's about 2.7M operations per sweep. Very fast.

One more thought: the EXP3 approach. After converging the weighted sum to its minimum (a binary point), the effective d_k values at that point tell us which shifts are "hard". We update weights to emphasize those shifts and solve again. This is like a no-regret algorithm for the minimax problem.

4. Normalize w
5. Repeat

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[len(idx) // 2]] = np.clip(h[idx[len(idx) // 2]] + diff2, 0, 1)
    return h

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_polish(h, n, ts, beta, maxiter=500):
    from scipy.optimize import minimize
    def obj(x):
        return smooth_obj(x, n, beta)
    def cons(x):
        return x.sum() - ts
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def binary_swap_ls(h, n, ts, max_sweeps=200, time_limit=120):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for a in range(len(ones)):
            if time.time() - t0 > time_limit:
                break
            for b in range(len(zeros)):
                ia = ones[a]
                ib = zeros[b]
                h[ia] = 0.0
                h[ib] = 1.0
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    a = range(len(ones))[0] if len(ones) > 0 else 0
                    break
                else:
                    h[ia] = 1.0
                    h[ib] = 0.0
        if not improved:
            break
    return h, best_c5

def exp3_minimax(h, n, ts, n_iters=50, tau0=5.0, time_limit=120):
    t0 = time.time()
    weights = np.ones(2 * n - 1)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        # Coordinate descent on weighted sum
        for sweep in range(20):
            if time.time() - t0 > time_limit:
                break
            changed = False
            for i in range(n):
                orig = h[i]
                best_val = np.inf
                best_bit = orig
                for val in (0.0, 1.0):
                    if abs(val - orig) < 1e-8:
                        continue
                    h[i] = val
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    ws = np.dot(weights, corr)
                    if ws < best_val:
                        best_val = ws
                        best_bit = val
                h[i] = best_bit
                if best_bit != orig:
                    changed = True
            if not changed:
                break
        # Evaluate and update weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        tau = tau0 * (0.9 ** it)
        weights = weights * np.exp(corr / tau)
        weights = weights / weights.sum()
    return best_h, best_c5

def make_binary(h, ts):
    target = int(round(ts))
    order = np.argsort(-h)
    h_bin = np.zeros(len(h))
    h_bin[order[:target]] = 1.0
    return h_bin

def upsample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n0 = 140
        h_init = np.zeros(n0)
        h_init[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([n_init]))
    for c in [60, 80, 100, 120, 140, 160, 180, 200, 240, 280]:
        if c % 2 == 0:
            n_list.add(c) if hasattr(n_list, 'add') else None
    n_list = sorted(set([n_init] + [c for c in [60,80,100,120,140,160,180,200,240,280] if c % 2 == 0]))

    for n in n_list:
        if remaining() < 50:
            break
        ts = n / 2.0

        if n == n_init:
            h = h_init.copy()
        elif n > n_init:
            h = upsample(h_init, n)
            h = project_feasible(h, ts)
        else:
            xo = np.linspace(0, 2, n_init)
            xn = np.linspace(0, 2, n)
            h = np.interp(xn, xo, h_init)
            h = project_feasible(h, ts)

        # SLSQP with annealed beta
        for beta in [5.0, 20.0, 80.0, 300.0]:
            if remaining() < 25:
                break
            h = slsqp_polish(h, n, ts, beta, maxiter=400)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # EXP3 minimax (key differentiator)
        if remaining() > 60:
            h_bin = make_binary(best_h if len(best_h) == n else h, ts)
            h_e, c5_e = exp3_minimax(h_bin, n, ts, n_iters=40, tau0=4.0,
                                     time_limit=min(100, remaining() * 0.4))
            if c5_e < best_c5:
                best_c5 = c5_e
                best_h = h_e.copy()

        # Binary swap local search
        if remaining() > 40:
            h_b = make_binary(best_h if len(best_h) == n else h, ts)
            h_b, c5_b = binary_swap_ls(h_b, n, ts, max_sweeps=150,
                                       time_limit=min(80, remaining() * 0.3))
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

        # Final SLSQP polish on best for this n
        if remaining() > 20 and len(best_h) == n:
            hp = slsqp_polish(best_h.copy(), n, ts, 500.0, maxiter=500)
            c5_p = compute_c5(hp)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = hp

    # Extended final search on best solution
    if remaining() > 50:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_b = make_binary(best_h, ts_b)
        # Two EXP3 passes with different tau
        h_e2, c5_e2 = exp3_minimax(h_b, n_b, ts_b, n_iters=60, tau0=3.0,
                                   time_limit=min(80, remaining() * 0.4))
        if c5_e2 < best_c5:
            best_c5 = c5_e2
            best_h = h_e2.copy()
        # Final binary swap
        if remaining() > 30:
            h_fs = make_binary(best_h, ts_b)
            h_fs, c5_fs = binary_swap_ls(h_fs, n_b, ts_b, max_sweeps=300,
                                         time_limit=min(60, remaining() * 0.4))
            if c5_fs < best_c5:
                best_c5 = c5_fs
                best_h = h_fs.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
