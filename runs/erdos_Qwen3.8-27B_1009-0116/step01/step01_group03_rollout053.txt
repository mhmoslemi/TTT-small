    delta = new - old
    # corr[k] = dx * sum_i h[i]*(1-h[i+k])
    # Changing h[j] from old to new:
    # Term i=j: delta*(1-h[j+k]) for all k where j+k valid
    # Term i=j-k: h[j-k]*(-delta) for all k where j-k valid
    # So corr[k] += dx * delta * (1-h[j+k] - h[j-k]) (with boundary checks)
    shifted = corr.copy()
    for k in range(-(n-1), n):
        val = 0.0
        jp = j + k
        jm = j - k
        if 0 <= jp < n:
            val += (1 - h[jp])
        if 0 <= jm < n:
            val -= h[jm]
        shifted[k + n - 1] += delta * val * dx
    return shifted

The key issue with the previous code: it uses SLSQP which is a general-purpose solver that might not be optimal for this specific problem. Also, the smooth max with beta up to 800 might not be smooth enough.

Let me try a different approach: **use the structure of the problem more directly**.

Key observation: The objective is max_k corr_k where corr_k = dx * ∑_i h[i](1-h[i+k]).

At the optimum, we'd like all corr_k to be as equal as possible (minimax). This suggests a "water-filling" or "balancing" approach.

**Algorithm**: 
1. Start with a feasible h
2. Compute all corr_k
3. The "active" k values are those with corr_k ≈ max
4. For the active k, the gradient w.r.t. h[j] is dx*(1-h[j+k]-h[j-k])
5. We want to move h in a direction that decreases the active corr_k while not increasing others
6. This is like a linear program in the direction: find direction d minimizing max_k (grad_k · d) subject to ||d|| ≤ 1
7. Take a step in that direction

This is essentially a **subgradient method with multiple active constraints**.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    d = ts - h.sum()
    if abs(d) < 1e-13:
        return h
    if d > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    d2 = ts - h.sum()
    if abs(d2) > 1e-12 and cnt > 0:
        idx = np.where(mask)[0]
        h[idx[0]] = np.clip(h[idx[0]] + d2, 0, 1)
    return h

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = corr.max()
    s = np.minimum((corr - m) * beta, 400.0)
    return m + np.log(np.exp(s).sum()) / beta

def slsqp_opt(h, n, ts, beta, maxiter=400):
    def obj(x):
        return smooth_obj(x, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': 8000})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project(res.x, ts)
    return project(h, ts)

def subgrad_descent(h, n, ts, n_iter=500, time_limit=100):
    t0 = time.time()
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_c = corr.max()
        active = np.where(corr > max_c - 0.015)[0]
        grad = np.zeros(n)
        for ki in active:
            k = ki - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j + k < n:
                    g += 1.0 - h[j + k]
                if 0 <= j - k < n:
                    g -= h[j - k]
                grad[j] += g
        grad *= dx / len(active)
        step = 0.06 * (0.98 ** it)
        for _ in range(12):
            h_new = project(h - step * grad, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h, c5 = h_new, c5_new
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
                break
            step *= 0.5
        if step < 1e-9:
            break
    return best_h, best_c5

def coord_descent(h, n, ts, max_sweeps=25, time_limit=70):
    t0 = time.time()
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            old = h[j]
            cands = [0.0, 1.0]
            if 0.01 < old < 0.99:
                cands += [max(0.0, old - 0.04), min(1.0, old + 0.04),
                          max(0.0, old - 0.01), min(1.0, old + 0.01)]
            best_c, best_v = old, c5
            for c in cands:
                if abs(c - old) < 1e-11:
                    continue
                h_backup = h.copy()
                h[j] = c
                h = project(h, ts)
                v = compute_c5(h)
                if v < best_v - 1e-14:
                    best_v, best_c = v, c
                else:
                    h = h_backup
            h[j] = best_c
            h = project(h, ts)
            if best_v < c5:
                c5 = best_v
                improved = True
        if not improved:
            break
    return h, c5

def fine_coord(h, n, ts, max_sweeps=15, time_limit=50):
    t0 = time.time()
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            old = h[j]
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                c = np.clip(old + delta, 0.0, 1.0)
                if abs(c - old) < 1e-12:
                    continue
                h_backup = h.copy()
                h[j] = c
                h = project(h, ts)
                v = compute_c5(h)
                if v < c5 - 1e-15:
                    c5 = v
                    improved = True
                else:
                    h = h_backup
        if not improved:
            break
    return h, c5

def binary_sa(h, n, ts, n_iter=4000, time_limit=80, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    t0 = time.time()
    hr = np.where(h > 0.5, 1.0, 0.0)
    hr = project(hr, ts)
    # Fix to exactly n/2 ones
    n_ones = int(np.sum(hr > 0.5))
    if n_ones != n // 2:
        ones_idx = np.where(hr > 0.5)[0]
        zeros_idx = np.where(hr < 0.5)[0]
        if n_ones > n // 2:
            for i in ones_idx[n//2:]:
                hr[i] = 0.0
                if len(zeros_idx) > 0:
                    hr[zeros_idx[0]] = 1.0
                    zeros_idx = zeros_idx[1:]
        else:
            for i in zeros_idx[:n//2 - n_ones]:
                hr[i] = 1.0
        hr = project(hr, ts)
    c5 = compute_c5(hr)
    best_hr, best_c5 = hr.copy(), c5
    T = 0.004
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        ones_idx = np.where(hr > 0.5)[0]
        zeros_idx = np.where(hr < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        hr[i], hr[j] = 0.0, 1.0
        c5_new = compute_c5(hr)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5, best_hr = c5, hr.copy()
        else:
            hr[i], hr[j] = 1.0, 0.0
        T = max(T * 0.9996, 5e-6)
    return best_hr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n0 = len(h0)

    best_c5 = compute_c5(best_h)

    n_list = sorted(set([n for n in range(n0 - 4, n0 + 14, 2) if n >= 40]))

    for n in n_list:
        if remaining() < 130:
            break
        ts = n / 2.0

        if n == n0:
            h = h0.copy()
        else:
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, n)
            h = np.interp(xn, xo, h0)
            h = project(h, ts)

        # Phase 1: SLSQP with annealed beta
        for beta in [15.0, 80.0, 400.0, 2000.0, 8000.0]:
            if remaining() < 45:
                break
            h = slsqp_opt(h, n, ts, beta, maxiter=350)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

        # Phase 2: Subgradient descent
        if remaining() > 90:
            h_sg, c5_sg = subgrad_descent(h, n, ts, n_iter=400, time_limit=75)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg.copy()

        # Phase 3: Coordinate descent (coarse)

            h_cd, c5_cd = coord_descent(h, n, ts, max_sweeps=20, time_limit=55)
            if c5_cd < best_c5:

        # Phase 4: Binary SA for exploration

                best_c5, best_h = c5_p, h_pol.copy()

    # Final projection and evaluation
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
