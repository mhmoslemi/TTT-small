    shifted = beta * (corr - max_c)

    weights /= weights.sum()

    for k in range(m):

        if w < 1e-12:
            continue

        # valid i: max(0,-lag) <= i < min(n, n-lag)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        if i_max <= i_min:
            continue
        # Gradient wrt h[m] for m in [i_min, i_max):
        # d/dh[m] = (1 - h[m+lag]) - h[m-lag]*1{m-lag in [i_min,i_max)}
        # Wait: d/dh[m] of sum_i h[i]*(1-h[i+lag])
        # = (1-h[m+lag]) * 1{m in [i_min,i_max)} - sum_i h[i] * d/dh[m](1-h[i+lag])
        # = (1-h[m+lag]) * 1{m valid} - h[m-lag] * 1{m-lag valid AND m valid for the h[i+lag] term}
        
        # The first term: h[m] appears as h[i] when i=m
        #   Contribution: (1 - h[m+lag]) if m in [i_min, i_max)
        # The second term: h[m] appears as h[i+lag] when i+lag=m, i.e., i=m-lag
        #   Contribution: -h[m-lag] if m-lag in [i_min, i_max) AND m in [0,n)
        #   (m is always in [0,n) since it's our index)
        
        # So: g[m] += w * (1-h[m+lag]) for m in [i_min, i_max)
        #     g[m] -= w * h[m-lag] for m-lag in [i_min, i_max), i.e., m in [i_min+lag, i_max+lag)
        
        # Term 1: m in [i_min, i_max), h[m+lag]
        #   m+lag in [i_min+lag, i_max+lag)
        #   For these m, m+lag is in [0,n) (guaranteed by validity)
        g[i_min:i_max] += w * (1.0 - h[i_min + lag:i_max + lag])

        #   m-lag in [i_min, i_max)

        m_start = max(0, i_min + lag)
        m_end = min(n, i_max + lag)
        if m_end > m_start:
            g[m_start:m_end] -= w * h[m_start - lag:m_end - lag]
    
    return g * dx

def project_feasible(h, n):

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def softmax_obj(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted -= np.max(shifted)
    return (max_c + np.log(np.sum(np.exp(shifted))) / beta) * dx

def softmax_grad(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    weights /= weights.sum()
    m = len(corr)
    g = np.zeros(n)
    for k in range(m):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        if i_max <= i_min:
            continue
        g[i_min:i_max] += w * (1.0 - x[i_min + lag:i_max + lag])
        ms = max(0, i_min + lag)
        me = min(n, i_max + lag)
        if me > ms:
            g[ms:me] -= w * x[ms - lag:me - lag]
    return g * dx

def _eq_fun(x):
    return x.sum() - 35.0

def _eq_jac(x):
    return np.ones(70)

def _slsqp_soft(x0, dx, beta, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * 70
    try:
        res = minimize(softmax_obj, x0, args=(dx, beta), jac=softmax_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        ho = project_feasible(res.x, 70)
        return ho, eval_c5(ho, dx)
    except Exception:
        return x0, eval_c5(x0, dx)

def _slsqp_exact(x0, dx, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * 70
    try:
        res = minimize(eval_c5, x0, args=(dx,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        ho = project_feasible(res.x, 70)
        return ho, eval_c5(ho, dx)
    except Exception:
        return x0, eval_c5(x0, dx)

def _anneal(x0, n, dx, maxiter_each=250):
    best_h = x0.copy()
    best_c5 = eval_c5(x0, dx)
    betas = [5, 15, 50, 150, 500, 1500, 5000]
    for beta in betas:
        h, c = _slsqp_soft(best_h, dx, beta, maxiter_each)
        if c < best_c5:
            best_h, best_c5 = h, c
    return best_h, best_c5

def _de_search(x0, n, dx, budget, seed=0):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    pop_size = 45
    F = 0.6
    CR = 0.85
    pop = np.zeros((pop_size, n))
    pop[0] = x0
    for i in range(1, pop_size):
        scale = 0.15 + 0.1 * rng.rand()
        pop[i] = x0 + scale * rng.randn(n)
        pop[i] = project_feasible(pop[i], n)
    for i in range(pop_size):
        if not np.isfinite(pop[i]).all():
            pop[i] = project_feasible(rng.rand(n), n)
    scores = np.array([eval_c5(pop[i], dx) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_c5 = scores[best_idx]
    gen = 0
    stagnation = 0
    while time.time() - t_start < budget:
        gen += 1
        F_cur = min(0.9, F + 0.05 * stagnation / 50.0)
        for i in range(pop_size):
            idxs = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(idxs, size=3, replace=False)
            v = pop[a] + F_cur * (pop[b] - pop[c])
            r = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[r] = True
            v = np.where(mask, pop[i], v)
            v = project_feasible(v, n)
            sv = eval_c5(v, dx)
            if sv < scores[i]:
                pop[i] = v
                scores[i] = sv
                if sv < best_c5 - 1e-10:
                    best_c5 = sv
                    best_h = v.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1
        if stagnation > 200:
            for i in range(1, pop_size, 3):
                pop[i] = x0 + 0.3 * rng.randn(n)
                pop[i] = project_feasible(pop[i], n)
                scores[i] = eval_c5(pop[i], dx)
            stagnation = 0
    return best_h, best_c5
