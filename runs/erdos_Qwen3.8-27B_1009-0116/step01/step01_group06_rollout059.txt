from scipy.optimize import minimize

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def corr_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    gc = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):

        if L >= 0:

def softmax_slsqp(h0, dx, n, temp, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = temp * (corr - mc)
        ev = np.exp(sh)
        return (mc + np.log(ev.sum()) / temp) * dx
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = temp * (corr - mc)
        ev = np.exp(sh)
        wts = ev / ev.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wt = wts[k]
            if wt < 1e-10:
                continue
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wt * (1.0 - x[:n - L])
                g[:n - L] -= wt * x[L:]
            else:
                p = -L
                g[:n - p] += wt * (1.0 - x[p:])
                g[p:] -= wt * x[:n - p]
        return dx * g
    def ceq(x):
        return [x.sum() - n / 2.0]
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h = project_feasible(res.x, n)
        return h, eval_c5(h, dx)
    except Exception:
        h = project_feasible(h0, n)
        return h, eval_c5(h, dx)

    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return [x.sum() - n / 2.0]
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
