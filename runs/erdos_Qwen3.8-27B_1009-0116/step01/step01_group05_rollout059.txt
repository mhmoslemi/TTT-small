import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return np.max(_corr(h, dx))

def _proj_sum_box(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _softmax_obj(x, dx, n, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _softmax_grad(x, dx, n, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    return _wgrad(x, dx, n, w)

def _wgrad(x, dx, n, w):
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            g[lo1:hi1] += wt * (1.0 - x[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, s + n)
        if hi2 > lo2:
            g[lo2:hi2] -= wt * x[lo2 - s:hi2 - s]
    return g

def _eq_fun(x, target):
    return x.sum() - target

def _eq_jac(x, target):
    return np.ones(len(x))

def _slsqp_opt(h0, dx, n, target, eps, maxiter):
    res = minimize(
        lambda x: _softmax_obj(x, dx, n, eps),
        h0,

from scipy.optimize import minimize
import time

# Global params for SLSQP
_P = {}

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return np.max(_corr(h, dx))

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def _obj(x):
    cv = _corr(x, _P['dx'])
    mx = cv.max()
    e = _P['eps']
    return mx + e*np.log(np.sum(np.exp((cv-mx)/e)))

def _grad(x):
    cv = _corr(x, _P['dx'])
    mx = cv.max()
    e = _P['eps']
    w = np.exp((cv-mx)/e)
    w /= w.sum()
    dx = _P['dx']
    n = _P['n']
    g = np.zeros(n)
    for idx in range(2*n-1):

        hi1 = min(n, n-s)
        if hi1 > lo1:

        lo2 = max(0, s)
        hi2 = min(n, s+n)
        if hi2 > lo2:
            g[lo2:hi2] -= wt*x[lo2-s:hi2-s]
    return g

    return x.sum() - _P['target']

    return np.ones(len(x))

                   method='SLSQP',
