import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + diff2, 0.0, 1.0)
    return h

def sa_coordinate_descent(h, n, ts, rng, t_start, deadline, best_c5_init):
    best_c5 = best_c5_init
    best_h = h.copy()
    c5_cur = best_c5
    T = 0.015
    cool = 0.99997
    T_min = 0.00005
    it = 0
    while time.time() < deadline:
        j = rng.randint(n)
        r = rng.random()
        if r < 0.3:
            new_val = 0.0
        elif r < 0.55:
            new_val = 1.0
        elif r < 0.75:
            new_val = h[j] + (1.0 - h[j]) * rng.uniform(0.3, 1.0) * rng.choice([-1, 1])
        else:
            new_val = rng.random()
        new_val = float(np.clip(new_val, 0.0, 1.0))
        if abs(new_val - h[j]) < 1e-10:
            it += 1
            if it % 1000 == 0:
                T = max(T * cool, T_min)
            continue
        old = h[j]
        h[j] = new_val
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[j] = old
            h = project_feasible(h, ts)
        it += 1
        if it % 2000 == 0:
            T = max(T * cool, T_min)
    return best_h, best_c5, h

def swap_sa(h, n, ts, rng, t_start, deadline, best_c5_init):
    best_c5 = best_c5_init
    best_h = h.copy()
    c5_cur = best_c5
    T = 0.01
    cool = 0.99996
    T_min = 0.00003
    while time.time() < deadline:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = ones[rng.randint(len(ones))]
        k = zeros[rng.randint(len(zeros))]
        if j == k:
            continue
        h[j], h[k] = h[k], h[j]
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[j], h[k] = h[k], h[j]
            h = project_feasible(h, ts)
        if T > T_min:
            T = max(T * cool, T_min)
    return best_h, best_c5, h

def binary_local_search(h, n, ts, rng, t_start, deadline, best_c5_init):
    best_c5 = best_c5_init
    h = h.copy()
    c5_cur = best_c5
    improved_any = True
    max_sweeps = 500
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if not improved_any:
            break
        improved_any = False
        order = rng.permutation(n)
        for j in order:
            for t_val in (0.0, 1.0):
                if abs(t_val - h[j]) < 1e-10:
                    continue
                old = h[j]
                h[j] = t_val
                h = project_feasible(h, ts)
                c5_t = compute_c5(h)
                if c5_t < c5_cur - 1e-14:
                    c5_cur = c5_t
                    improved_any = True
                    if c5_t < best_c5:
                        best_c5 = c5_t
                else:
                    h[j] = old
                    h = project_feasible(h, ts)
    return h, best_c5

def slsqp_polish(h, n, ts, dx, beta, maxiter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def make_structured_start(n, seed, pattern_type):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    m = n // 2
    if pattern_type == 0:
        for i in range(n):
            h[i] = 1.0 if (i % 2 == 0) else 0.0
    elif pattern_type == 1:
        period = max(3, n // m) if m > 0 else 2
        ones_per_period = max(1, m * period // n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < ones_per_period else 0.0
    elif pattern_type == 2:
        h[rng.choice(n, m, replace=False)] = 1.0
        for _ in range(200):
            i = rng.randint(n)
            if h[i] > 0.5:
                j = rng.randint(n)
                if h[j] < 0.5 and j != i:
                    h[i], h[j] = h[j], h[i]
    else:
        h[rng.choice(n, m, replace=False)] = 1.0
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_structured_start(140, seed, 2)

    n = len(h_init)
    ts = n / 2.0
    dx = 2.0 / n

    best_h = project_feasible(h_init.copy(), ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: SA coordinate descent on initial
    deadline1 = t_start + budget_s * 0.25
    h_w, c5_1, h_w = sa_coordinate_descent(best_h.copy(), n, ts, rng, t_start, deadline1, best_c5)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h_w

    # Phase 2: Swap-based SA
    deadline2 = t_start + budget_s * 0.45
    h_w, c5_2, h_w = swap_sa(best_h.copy(), n, ts, rng, t_start, deadline2, best_c5)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h_w

    # Phase 3: Binary local search
    deadline3 = t_start + budget_s * 0.65
    h_ls, c5_3 = binary_local_search(best_h, n, ts, rng, t_start, deadline3, best_c5)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h_ls

    # Phase 4: SLSQP polish
    if time.time() < t_start + budget_s * 0.78:
        h_pol = slsqp_polish(best_h.copy(), n, ts, dx, 150.0, 400)
        c5_4 = compute_c5(h_pol)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_pol

    # Phase 5: SA from multiple structured starts
    for pt in [0, 1]:
        if time.time() > t_start + budget_s * 0.92:
            break
        h_alt = make_structured_start(n, seed + pt * 1000, pt)
        # Quick SA from this start
        dl = min(time.time() + 30, t_start + budget_s * 0.93)
        h_a, c5_a, _ = sa_coordinate_descent(h_alt, n, ts, rng, t_start, dl, best_c5)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a

    # Phase 6: Final binary LS + polish
    if time.time() < t_start + budget_s * 0.97:
        h_ls, c5_f = binary_local_search(best_h, n, ts, rng, t_start, t_start + budget_s * 0.97, best_c5)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_ls

    if time.time() < t_start + budget_s - 5:
        h_pol = slsqp_polish(best_h.copy(), n, ts, dx, 300.0, 200)
        c5_p = compute_c5(h_pol)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_pol

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
