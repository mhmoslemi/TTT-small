def smooth_grad(h, t, dx, idx_p, idx_q, mask_q):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) * t)
    p = exp_c / np.sum(exp_c)
    grad = p[idx_p] @ (1.0 - h) - (p[idx_q] * mask_q) @ h
    return grad * dx

    cmax = np.max(c)
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx

                            max_iter=5000, tol=1e-14, time_limit=None, t0=None):

    step = 0.01
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        g = smooth_grad(h, t, dx, idx_p, idx_q, mask_q)
        # Backtracking line search
        f_old = smooth_obj_val(h, t, dx)
        # Use Armijo condition
        c1 = 1e-4
        s = step
        for _ in range(30):
            h_trial = h - s * g
            h_trial = np.clip(h_trial, 0.0, 1.0)
            diff = target - h_trial.sum()
            if abs(diff) > 1e-10:
                mask = h_trial < 1.0 - 1e-12 if diff > 0 else h_trial > 1e-12
                if mask.sum() > 0:
                    h_trial[mask] += diff / mask.sum()
                    h_trial = np.clip(h_trial, 0.0, 1.0)
            f_new = smooth_obj_val(h_trial, t, dx)
            if f_new <= f_old - c1 * s * np.dot(g, g):
                break
            s *= 0.5
        else:
            s = step * 1e-6
        
        h = h_trial
        step = s * 1.05  # slight increase
        
        # Check actual C5
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        
        if np.linalg.norm(g) < tol:
            break
    
    return h, best_c5
