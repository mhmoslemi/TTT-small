        i_max = min(n, n-s)
        result[s+n-1] = np.dot(h[i_min:i_max], h1[i_min+s:i_max+s]) * dx

    mx = cv.max()

    def grad(x): return _soft_grad(x, dx, eps)
    def cons_fun(x): return x.sum() - target
    def cons_jac(x): return np.ones(n)

                   bounds=[(0,1)]*n,

        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            cur = h[j]
            best_t = cur
            best_val = cv.max()
            for t in grid:
                if abs(t - cur) < 1e-8:
                    continue
                h[j] = t
                cv_t = _corr(h, dx)
                val = cv_t.max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
            if best_t != cur:
                h[j] = best_t
                cv = _corr(h, dx)
                improved = True
                if cv.max() < best:
                    best = cv.max()
        if not improved:
            break
    return h, _c5_val(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()

    best_h, best_c5, best_n = None, 1.0, 0

    n_list = [50, 70, 90]
    n_alloc = {50: 0.25, 70: 0.35, 90: 0.25}
    elapsed_frac = 0.0

    for n in n_list:
        if time.time() > deadline - 40:
            break
        dx = 2.0 / n
        target = n / 2.0
        phase_time = (deadline - t0) * n_alloc.get(n, 0.25)
        phase_end = t0 + phase_time

        h0 = None
        if parent is not None and len(parent) > 0:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            h0 = rng.uniform(0.3, 0.7, n)
        h0 = _proj(h0, target)

        h_opt, c5_opt = _optimize(h0, dx, n, target, rng, phase_end)
        h_cd, c5_cd = _coord_descent(h_opt, dx, target, max_sweeps=60,
                                     time_limit=min(time.time() + 30, deadline - 20))
        if c5_cd < c5_opt:
            h_opt, c5_opt = h_cd, c5_cd

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n

    # Random restarts and refinements on best_n
    n = best_n
    dx = 2.0 / n
    target = n / 2.0

    attempts = 0
    while time.time() < deadline - 45:
        attempts += 1
        amp = 0.03 + 0.08 * rng.rand()
        h_pert = _proj(best_h + rng.normal(0, amp, n), target)
        sub_end = min(time.time() + 40, deadline - 30)
        h_try, c5_try = _optimize(h_pert, dx, n, target, rng, sub_end)
        h_cd, c5_cd = _coord_descent(h_try, dx, target, max_sweeps=40,
                                     time_limit=min(time.time() + 20, deadline - 10))
        if c5_cd < c5_try:
            h_try, c5_try = h_cd, c5_cd
        if c5_try < best_c5:
            best_h, best_c5 = h_try.copy(), c5_try

    # Final coordinate descent polish
    if time.time() < deadline - 10:
        h_fin, c5_fin = _coord_descent(best_h, dx, target, max_sweeps=80,
                                       time_limit=deadline - 3)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    # Verify integral constraint
    best_h = _proj(best_h, target)
    best_c5 = _c5_val(best_h, dx)

    return best_h, best_c5, n
