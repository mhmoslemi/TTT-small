import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(150):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    val = 0.5 * (lo + hi)
    return np.clip(h - val, 0.0, 1.0)

def _weighted_grad(h, w, dx, n):
    """Gradient of sum_s w[s]*cv[s] where cv = correlate(h,1-h,'full')*dx"""
    g = np.zeros(n)
    for s in range(2*n-1):
        if w[s] < 1e-14:
            continue
        wt = w[s] * dx
        d = s - (n-1)
        if d >= 0:
            if n > d:
                g[d:n] += wt * (1.0 - h[0:n-d])
                g[0:n-d] -= wt * h[d:n]
        else:
            e = -d
            g[0:n-e] += wt * (1.0 - h[e:n])
            g[e:n] -= wt * h[0:n-e]
    return g

def _exp_weights_solve(h_init, dx, n, target, t_end, rng):
    """Exponential weights (follow-the-leader) for minimax."""
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)
    h = h_init.copy()
    n_s = 2*n - 1
    w = np.ones(n_s) / n_s
    tau = 1.0
    
    for outer in range(200):
        if time.time() > t_end - 5:
            break
        
        cv = _corr(h, dx)
        w = np.exp(cv / tau)
        w /= w.sum()
        tau = max(0.05, tau * 0.92)
        
        # Minimize weighted objective with projected gradient
        lr = 0.1
        for inner in range(200):
            if time.time() > t_end - 5:
                break
            g = _weighted_grad(h, w, dx, n)
            h_new = _proj(h - lr * g, target)
            f_new = np.dot(w, _corr(h_new, dx))
            f_cur = np.dot(w, _corr(h, dx))
            if f_new <= f_cur * (1 + 1e-12) + 1e-15:
                h = h_new
                lr = min(lr * 1.03, 1.0)
            else:
                lr *= 0.5
                if lr < 0.001:
                    break
        
        c = _c5(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    return best_h, best_c5

def _slsqp_polish(h0, dx, n, target, eps, maxiter, t_end):
    """SLSQP polish with smooth max."""
    if time.time() > t_end - 5:
        return h0, _c5(h0, dx)
    
    def _sobj(x):
        cv = _corr(x, dx)
        m = cv.max()
        return m + eps * np.log(np.sum(np.exp((cv - m) / eps)))
    
    def _sgrad(x):
        cv = _corr(x, dx)
        m = cv.max()
        w = np.exp((cv - m) / eps)
        w /= w.sum()
        return _weighted_grad(x, w, dx, len(x))
    
    def _efun(x):
        return x.sum() - target
    
    def _ejac(x):
        return np.ones(len(x))
    
    try:
        res = minimize(_sobj, h0, jac=_sgrad,
                      bounds=[(0.0, 1.0)]*n,
                      constraints=[{'type':'eq','fun':_efun,'jac':_ejac}],
                      method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h = _proj(res.x, target)
        return h, _c5(h, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _coord_descend(h, dx, n, target, t_end, rng):
    """Coordinate descent: optimize one coordinate at a time."""
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    n_s = 2*n - 1
    cv_best = _corr(best_h, dx).copy()
    
    for it in range(2000):
        if time.time() > t_end - 3:
            break
        j = it % n  # or rng.randint(n)
        
        # Compute A_s for coordinate j
        A = np.zeros(n_s)
        for s in range(n_s):
            d = s - (n-1)
            a = 0.0
            if d >= 0:
                if 0 <= j-d < n:
                    a += dx * (1.0 - h[j-d])
                if 0 <= j+d < n:
                    a -= dx * h[j+d]
            else:
                e = -d
                if 0 <= j+e < n:
                    a += dx * (1.0 - h[j+e])
                if 0 <= j-e < n:
                    a -= dx * h[j-e]
            A[s] = a
        
        # Minimize max_s (cv_s + y*A_s) over y ∈ [-(h[j]), 1-h[j]]
        y_lo = -h[j]
        y_hi = 1.0 - h[j]
        
        # Evaluate at several points
        best_y = 0.0
        best_val = np.inf
        for y in np.linspace(y_lo, y_hi, 30):
            val = np.max(cv_best + y * A)
            if val < best_val:
                best_val = val
                best_y = y
        
        new_j = h[j] + best_y
        if abs(best_y) > 1e-10:
            best_h = best_h.copy()
            best_h[j] = new_j
            best_h = _proj(best_h, target)
            cv_best = _corr(best_h, dx)
            c = cv_best.max()
            if c < best_c5 - 1e-12:
                best_c5 = c
                best_h = best_h
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    phase_time = (p1_end - t0) / len(n_list)

        dx = 2.0 / n
        target = n / 2.0
        p_end = min(time.time() + phase_time, p1_end)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _wgrad(h, w, dx, n):
    g = np.zeros(n)
    for s in range(2 * n - 1):
        if w[s] < 1e-14:
            continue
        wt = w[s] * dx
        d = s - (n - 1)
        if d >= 0:
            if n > d:
                g[d:n] += wt * (1.0 - h[0:n-d])
                g[0:n-d] -= wt * h[d:n]
        else:
            e = -d
            g[0:n-e] += wt * (1.0 - h[e:n])
            g[e:n] -= wt * h[0:n-e]
    return g

def _exp_weights(h_init, dx, n, target, t_end):
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)
    h = h_init.copy()
    n_s = 2 * n - 1
    w = np.ones(n_s) / n_s
    tau = 1.0
    for outer in range(80):
        if time.time() > t_end - 5:
            break
        cv = _corr(h, dx)
        w = np.exp(cv / tau)
        w /= w.sum()
        tau = max(0.03, tau * 0.90)
        lr = 0.05
        for inner in range(80):
            if time.time() > t_end - 5:
                break
            g = _wgrad(h, w, dx, n)
            h_new = _proj(h - lr * g, target)
            f_new = np.dot(w, _corr(h_new, dx))
            f_cur = np.dot(w, _corr(h, dx))
            if f_new <= f_cur + 1e-15:
                h = h_new
                lr = min(lr * 1.04, 0.8)
            else:
                lr *= 0.5
                if lr < 5e-4:
                    break
        c = _c5(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5

def _slsqp(h0, dx, n, target, eps, maxiter, t_end):
    if time.time() > t_end - 3:
        return h0, _c5(h0, dx)
    def _sobj(x):
        cv = _corr(x, dx)
        m = cv.max()
        return m + eps * np.log(np.sum(np.exp((cv - m) / eps)))
    def _sgrad(x):
        cv = _corr(x, dx)
        m = cv.max()
        w = np.exp((cv - m) / eps)
        w /= w.sum()
        return _wgrad(x, w, dx, len(x))
    def _efun(x):
        return x.sum() - target
    def _ejac(x):
        return np.ones(len(x))
    try:
        res = minimize(_sobj, h0, jac=_sgrad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                       method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h = _proj(res.x, target)
        return h, _c5(h, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _coord_descend(h, dx, n, target, t_end):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    n_s = 2 * n - 1
    cv_cur = _corr(best_h, dx)
    for it in range(3000):
        if time.time() > t_end - 3:
            break
        j = it % n
        A = np.zeros(n_s)
        for s in range(n_s):
            d = s - (n - 1)
            a = 0.0
            if d >= 0:
                if 0 <= j - d < n:
                    a += dx * (1.0 - best_h[j - d])
                if 0 <= j + d < n:
                    a -= dx * best_h[j + d]
            else:
                e = -d
                if 0 <= j + e < n:
                    a += dx * (1.0 - best_h[j + e])
                if 0 <= j - e < n:
                    a -= dx * best_h[j - e]
            A[s] = a
        y_lo = -best_h[j]
        y_hi = 1.0 - best_h[j]
        if y_hi - y_lo < 1e-10:
            continue
        best_y = 0.0
        best_val = np.inf
        for y in np.linspace(y_lo, y_hi, 25):
            v = np.max(cv_cur + y * A)
            if v < best_val:
                best_val = v
                best_y = y
        if abs(best_y) < 1e-10:
            continue
        trial = best_h.copy()
        trial[j] += best_y
        trial = _proj(trial, target)
        cv_t = _corr(trial, dx)
        c_t = cv_t.max()
        if c_t < best_c5 - 1e-12:
            best_c5 = c_t
            best_h = trial
            cv_cur = cv_t
    return best_h, best_c5

def _optimize_start(sp, dx, n, target, t_end):
    h1, c1 = _exp_weights(sp, dx, n, target, t_end - 15)
    h2, c2 = _slsqp(h1, dx, n, target, 0.05, 150, t_end - 5)
    h3, c3 = _slsqp(h2, dx, n, target, 0.01, 200, t_end)
    candidates = [(c1, h1), (c2, h2), (c3, h3)]
    candidates.sort(key=lambda x: x[0])
    return candidates[0][1], candidates[0][0]

            starts.append(_proj(hp, target))

            h_r, c_r = _optimize_start(sp, dx, n, target, sub_end)

        target = n / 2.0
        p2_end = deadline - 80
