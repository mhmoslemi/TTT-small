from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    return np.clip(x, 0.0, 1.0)

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    xp = np.interp(xd, xs, x_src, period=None)
    # Handle wrap-around: extend
    xs_ext = np.concatenate([xs - 2.0, xs, xs + 2.0])
    xv_ext = np.concatenate([x_src, x_src, x_src])
    xp = np.interp(xd, xs_ext, xv_ext)
    return repair(xp, n_dst)

def softmax_obj(x, dx, tau):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

def slsqp_optimize(x0, n, dx, tau, maxiter=400, ftol=1e-15):
    bd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: softmax_obj(x, dx, tau), x0, method='SLSQP',
                     bounds=bd, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return repair(r.x, n)
    except:
        return x0.copy()

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = (tgt / s) * x
    return np.clip(x, 0.0, 1.0)

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    xs_ext = np.concatenate([xs - 2.0, xs, xs + 2.0])
    xv_ext = np.concatenate([x_src, x_src, x_src])
    try:
        cs = CubicSpline(xs_ext, xv_ext)
        xp = cs(xd)
    except Exception:
        xp = np.interp(xd, xs_ext, xv_ext)
    return repair(xp, n_dst)

def soft_obj(x, dx, tau):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

def slsqp_opt(x0, n, dx, tau, maxiter=400):
    bd = [(0.0, 1.0)] * n
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(lambda x: soft_obj(x, dx, tau), x0, method='SLSQP',
                     bounds=bd, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def refine(x0, n, dx, tlim, taus=None):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    bv = compute_c5(x, dx)
    if taus is None:
        taus = [0.3, 0.15, 0.07, 0.035, 0.018, 0.009, 0.005, 0.0025, 0.0012, 0.0006]
    for tau in taus:
        if time.time() - t0 > tlim - 1.0:
            break
        x = slsqp_opt(x, n, dx, tau)
        v = compute_c5(x, dx)
        if v < bv:
            bv = v
            best = x.copy()
    return best, bv

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def pert_sin(x, n, freq, amp, rng):
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    ph = rng.uniform(0, 2 * np.pi)
    return repair(x + amp * np.sin(2 * np.pi * freq * t + ph), n)

def pert_mix(x, n, scales, rng):
    out = x.copy()
    for f, a in scales:
        out = pert_sin(out, n, f, a, rng)
    return repair(out, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = repair(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    # Phase 1: Base grid with diverse starts (32%)
    p1_end = t_start + budget_s * 0.32
    starts = [
        h0.copy(),
        perturb(h0, n0, 0.06, rng),
        pert_mix(h0, n0, [(3, 0.07), (5, 0.04)], rng),
        pert_mix(h0, n0, [(7, 0.05), (2, 0.06)], rng),
        perturb(h0, n0, 0.10, rng),
        pert_sin(h0, n0, 4, 0.08, rng),
    ]
    for i, xs in enumerate(starts):
        if time.time() > p1_end - 3:
            break
        alloc = min(p1_end - time.time() - 2, budget_s * 0.055)
        if alloc < 4:
            break
        xb, v = refine(xs, n0, dx0, alloc)
        if v < best_val:
            best_val = v
            best_h = xb
            best_n = n0

    # Phase 2: Project to finer grid (30%)
    p2_end = t_start + budget_s * 0.62
    if time.time() < p2_end - 10:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        xp = project_grid(best_h, best_n, n1)
        alloc = p2_end - time.time() - 5
        if alloc > 12:
            xb, v = refine(xp, n1, dx1, alloc)
            if v < best_val:
                best_val = v
                best_h = xb
                best_n = n1
        # Second restart on fine grid
        if time.time() < p2_end - 8:
            xp2 = perturb(xp, n1, 0.04, rng)
            alloc2 = p2_end - time.time() - 3
            if alloc2 > 8:
                xb2, v2 = refine(xp2, n1, dx1, alloc2)
                if v2 < best_val:
                    best_val = v2
                    best_h = xb2
                    best_n = n1

    # Phase 3: Even finer grid (25%)
    p3_end = t_start + budget_s * 0.87
    if time.time() < p3_end - 10:
        n2 = 2 * best_n + 1
        if n2 < 1000:
            dx2 = 2.0 / n2
            xp = project_grid(best_h, best_n, n2)
            alloc = p3_end - time.time() - 5
            if alloc > 12:
                xb, v = refine(xp, n2, dx2, alloc)
                if v < best_val:
                    best_val = v
                    best_h = xb
                    best_n = n2
            if time.time() < p3_end - 8:
                xp2 = perturb(xp, n2, 0.03, rng)
                alloc2 = p3_end - time.time() - 3
                if alloc2 > 8:
                    xb2, v2 = refine(xp2, n2, dx2, alloc2)
                    if v2 < best_val:
                        best_val = v2
                        best_h = xb2
                        best_n = n2

    # Phase 4: Final polish at best grid (remaining)
    remaining = budget_s - (time.time() - t_start) - 3
    if remaining > 8:
        dx_b = 2.0 / best_n
        xp = perturb(best_h, best_n, 0.02, rng)
        alloc = remaining * 0.5
        xb, v = refine(xp, best_n, dx_b, alloc)
        if v < best_val:
            best_val = v
            best_h = xb
        remaining2 = budget_s - (time.time() - t_start) - 3
        if remaining2 > 6:
            xp2 = pert_sin(best_h, best_n, 6, 0.03, rng)
            xb2, v2 = refine(xp2, best_n, dx_b, remaining2 * 0.6)
            if v2 < best_val:
                best_val = v2
                best_h = xb2

    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
