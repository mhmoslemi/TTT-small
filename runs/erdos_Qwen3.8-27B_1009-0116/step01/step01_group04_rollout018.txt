from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    tgt = n / 2.0
    lo = -np.max(x) - 2.0
    hi = 1.0 - np.min(x) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < tgt:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(x + t, 0.0, 1.0)

def subgrad_corr(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] += -h[k:]
    else:
        kp = -k
        g[:n-kp] += (1.0 - h[kp:])
        g[kp:] += -h[:n-kp]

def softmax_fn(x, tau, dx, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / tau, -50.0, 0.0)
    lse = m + tau * np.log(np.sum(np.exp(shifted)))
    pen = 1000.0 * (np.sum(x) - n * 0.5) ** 2
    return (lse + pen) * dx

def softmax_grad(x, tau, dx, n):

    m = np.max(corr)
    shifted = np.clip((corr - m) / tau, -50.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))

            continue
        k = k_idx - (n - 1)
        g += weights[k_idx] * subgrad_corr(x, n, k)
    g += 2000.0 * (np.sum(x) - n * 0.5)
    return g * dx

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    tgt = n / 2.0
    lo = -float(np.max(x)) - 2.0
    hi = 1.0 - float(np.min(x)) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < tgt:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(x + t, 0.0, 1.0)

def subgrad_corr(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
    else:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] += -h[:n - kp]
    return g

def _soft_obj(x, tau, dx, n, pen):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    lse = m + tau * np.log(np.sum(np.exp(sh)))
    p = pen * (np.sum(x) - n * 0.5) ** 2
    return (lse + p) * dx

def _soft_grad(x, tau, dx, n, pen):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    w = np.exp(sh)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    nz = np.where(w > 1e-8)[0]
    g = np.zeros(n)
    for ki in nz:
        k = int(ki) - (n - 1)
        g += w[ki] * subgrad_corr(x, n, k)
    g += 2.0 * pen * (np.sum(x) - n * 0.5)
    return g * dx

def lbfgsb_step(h0, n, tau, dx, maxiter=200):
    bd = [(0.0, 1.0)] * n
    pen = 500.0
    try:
        res = minimize(_soft_obj, h0, args=(tau, dx, n, pen),
                       jac=lambda x: _soft_grad(x, tau, dx, n, pen),
                       method='L-BFGS-B', bounds=bd,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)
