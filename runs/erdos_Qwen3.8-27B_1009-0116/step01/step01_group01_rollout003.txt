import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h, j):
    """Gradient of c_j = sum_i h[i]*(1-h[i+j]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if j >= 0:
        g[:n-j] += (1.0 - h[j:])
        g[j:] -= h[:n-j]
    else:
        jo = -j
        g[jo:] += (1.0 - h[:n-jo])
        g[:n-jo] -= h[jo:]
    return g

def project_polytope(y, target):
    """Project y onto {h : 0<=h_i<=1, sum(h)=target}"""
    n = len(y)
    # Bisection on lambda: h_i = clip(y_i - lambda, 0, 1)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

    c = compute_corr(h)
    cmax = c.max()
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx

def c5_subgradient(h, n, dx, K=1, T=1.0):
    """Return (objective_value, gradient) using top-K lags"""
    c = compute_corr(h)
    cmax = c.max()
    # Top K indices
    if K >= len(c):
        idx = np.arange(len(c))
        w = np.ones(len(c)) / len(c)
    else:
        idx = np.argsort(c)[-K:][::-1]
        w = np.exp(c[idx] * T)
        w /= w.sum()
    # Weighted gradient
    g = np.zeros(n)
    for k_idx in idx:
        j = k_idx - (n - 1)  # convert to lag
        g += w[list(idx).index(k_idx)] * corr_gradient(h, j)
    return cmax * dx, g * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad(h, j):
    n = len(h)
    g = np.zeros(n)
    if j >= 0:
        if j < n:
            g[:n-j] += (1.0 - h[j:])
            g[j:] -= h[:n-j]
    else:
        jo = -j
        if jo < n:
            g[jo:] += (1.0 - h[:n-jo])
            g[:n-jo] -= h[jo:]
    return g

def project_polytope(y, target):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(y - lam, 0.0, 1.0)

def obj_smooth(h, n, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = (c - cmax) * temp
    shifted[shifted > 50] = 50  # prevent overflow
    return (cmax + np.log(np.exp(shifted).sum()) / temp) * dx

def subgrad_descent(h, n, dx, target, n_steps, alpha0, decay, momentum, K, T, t_start, t_deadline):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_steps):
        if time.time() > t_deadline:
            break
        c = compute_corr(h)
        cmax = c.max()
        c5_val = cmax * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Top-K lags
        if K >= len(c):
            topk = np.arange(len(c))
            weights = np.ones(len(c)) / len(c)
        else:
            topk = np.argsort(c)[-K:][::-1]
            cw = c[topk] * T
            cw -= cw.max()
            weights = np.exp(cw)
            weights /= weights.sum()
        
        # Weighted gradient
        g = np.zeros(n)
        for i, k in enumerate(topk):
            j = k - (n - 1)
            if -n < j < n:
                g += weights[i] * corr_grad(h, j)
        g *= dx

        alpha = alpha0 / (1.0 + decay * it)
        v = momentum * v - alpha * g
        h = project_polytope(h + v, target)

        best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def corr_grad(h, j):
    n = len(h)
    g = np.zeros(n)
    if j >= 0 and j < n:
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    elif j < 0 and -j < n:
        jo = -j
        g[jo:] += (1.0 - h[:n - jo])
        g[:n - jo] -= h[jo:]
    return g


def project_polytope(y, target):
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)


def obj_smooth(h, n, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    s = np.clip((c - cmax) * temp, -500, 0)
    return (cmax + np.log(np.exp(s).sum()) / temp) * dx


def subgrad_run(h_in, n, dx, target, deadline, alpha0, mom, K, T):
    h = h_in.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    it = 0
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        c5v = cmax * dx
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        if K >= len(c):
            idx = np.arange(len(c))
            w = np.full(len(c), 1.0 / len(c))
        else:
            idx = np.argsort(c)[-K:][::-1]
            cw = c[idx] * T
            cw -= cw.max()
            w = np.exp(cw)
            w /= w.sum()
        g = np.zeros(n)
        for i, k in enumerate(idx):
            j = k - (n - 1)
            if -n < j < n:
                g += w[i] * corr_grad(h, j)
        g *= dx
        alpha = alpha0 / (1.0 + 0.002 * it)
        v = mom * v - alpha * g
        h = project_polytope(h + v, target)
        it += 1
        if it % 500 == 0 and np.random.random() < 0.02:
            h = project_polytope(h + 0.02 * np.random.randn(n), target)
    if compute_c5(h) < best_c5:
        best_c5 = compute_c5(h)
        best_h = h.copy()
    return best_h, best_c5


def slsqp_solve(h_in, n, dx, target, deadline, temp=300.0, maxiter=150):
    if time.time() >= deadline:
        return h_in, compute_c5(h_in)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_smooth, h_in.copy(), args=(n, dx, temp),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_p = project_polytope(res.x, target)
    return h_p, compute_c5(h_p)
