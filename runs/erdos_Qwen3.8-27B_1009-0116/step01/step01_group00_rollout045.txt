import numpy as np
from scipy.optimize import minimize
import time

_GS = 0.0

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            fr = np.where(h < 1.0 - 1e-14)[0]
        else:
            fr = np.where(h > 1e-14)[0]
        if fr.size == 0:
            break
        h[fr] += d / fr.size
        h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            ordr = np.argsort(1.0 - h)
        else:
            ordr = np.argsort(h)
        for idx in ordr:
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a; d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _obj_lse(x, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + mu * (x.sum() - _GS) ** 2

def _obj_pn(x, p, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(c)
    if cm < 1e-15:
        return mu * (x.sum() - _GS) ** 2
    cn = c / cm
    pn = cm * (np.sum(cn ** p)) ** (1.0 / p)
    return pn + mu * (x.sum() - _GS) ** 2

def _refine_lse(h, n, dx, alpha, mu, maxiter):
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj_lse, h, args=(alpha, mu), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                         'gtol': 1e-12, 'maxls': 50})
    return _repair(res.x, n)

def _refine_pn(h, n, dx, p, mu, maxiter):
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj_pn, h, args=(p, mu), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                         'gtol': 1e-12, 'maxls': 50})
    return _repair(res.x, n)

def _coord_descent(h, n, dx, rng, max_iter, t0, t_end):
    best = _c5(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05,
                       -0.002, -0.005, -0.01, -0.02, -0.05])
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                continue
            h[i] = ni
            h[j] = nj
            c5 = _c5(h, dx)
            if c5 < best - 1e-13:
                best = c5
            else:
                h[i] -= d
                h[j] += d
    return h, best
