def subgrad_step(h, n, dx, lr, n_active=1):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    tol = max(1e-6, 1e-4 * mx)
    active = np.where(corr >= mx - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    n_act = min(n_active, len(active))
    idx = np.argsort(corr[active])[-n_act:]
    active = active[idx]
    g = np.zeros(n)
    for ki in active:
        lag = ki - (n - 1)
        for mi in range(n):
            if 0 <= mi - lag < n:
                g[mi] += (1.0 - h[mi - lag])
            if 0 <= mi + lag < n:
                g[mi] -= h[mi + lag]
    g /= float(len(active))
    h_new = h - lr * dx * g

                h2, c2 = slsqp_max(h, n, dx, maxiter=1500)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2
                h2, c2 = slsqp_topk(h, n, dx, k=5, maxiter=600)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2
        else:
            stall_count += 1
            if stall_count > 30 and mode >= 3:
                h, c = nm_opt(best_h, n, dx, maxiter=300)
                if c < best_c5:
                    best_h, best_c5 = h, c
                    stall_count = 0

    # Final: try n=100 if time permits
    if remaining() > 100:
        n2 = 100
        dx2 = 2.0 / n2
        # Interpolate best_h to n2 points
        x_orig = np.linspace(0, 2, n, endpoint=False) + dx / 2
        x_new = np.linspace(0, 2, n2, endpoint=False) + dx2 / 2
        h_interp = np.interp(x_new, x_orig, best_h)
        h2 = project_feasible(h_interp, n2)
        c2 = eval_c5(h2, dx2)
        if c2 < best_c5:
            h2, c2 = slsqp_max(h2, n2, dx2, maxiter=1500)
            if c2 < best_c5:
                best_h, best_c5, n = h2, c2, n2
                h2, c2 = slsqp_smooth(h2, n2, dx2, alpha=400, maxiter=600)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2
                h2, c2 = slsqp_max(h2, n2, dx2, maxiter=2000)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2
                h2, c2 = slsqp_topk(h2, n2, dx2, k=5, maxiter=800)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2

    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
