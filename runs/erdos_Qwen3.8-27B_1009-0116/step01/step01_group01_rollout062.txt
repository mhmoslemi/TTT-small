    target = n * 0.5

                h[idx] += diff2

def cd_sweep(h, n, c, dx):

    c is the current correlation (length 2n-1), updated in place.
    Returns (h, c, best_c5)."""
    target = n * 0.5
    n_full = 2 * n - 1
    best = np.max(c) * dx
    h = h.copy()
    
    for j in np.random.permutation(n):
        old = h[j]
        # Affected lags
        k_lo = max(0, n - 1 - j)
        k_hi = min(n_full - 1, n - 1 + j)
        
        # g[k] = (1-h[j+k-n+1]) - h[j-k+n-1]
        ks = np.arange(k_lo, k_hi + 1)
        idx1 = j + ks - n + 1  # partner for term 1
        idx2 = j - ks + n - 1  # partner for term 2
        g = (1.0 - h[idx1]) - h[idx2]
        
        # We want to find new_val ∈ [0,1] minimizing max(c) after update
        # c_new[k] = c[k] + (new_val - old) * g[k-lo] for k in [k_lo, k_hi]
        # c_new[k] = c[k] for other k
        
        c_affected = c[k_lo:k_hi+1].copy()
        c_unaffected_max = np.max(np.delete(c, np.arange(k_lo, k_hi+1))) if n_full > (k_hi-k_lo+1) else -np.inf
        
        # 1D optimization: min_t max(c_unaffected_max, max(c_affected + t*g)) for t ∈ [0-old, 1-old]
        t_lo = -old
        t_hi = 1.0 - old
        
        # Golden section / ternary search (function is convex in t)
        # Actually, since it's max of linear functions, it's convex piecewise linear
        # Use golden section for speed
        gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
        
        def f1d(t):
            return max(c_unaffected_max, np.max(c_affected + t * g))
        
        # First check endpoints
        f_lo = f1d(t_lo)
        f_hi = f1d(t_hi)
        f_mid0 = f1d(0.0)
        
        best_f = min(f_lo, f_hi, f_mid0)
        best_t = 0.0
        if f_lo < best_f:
            best_t, best_f = t_lo, f_lo
        if f_hi < best_f:
            best_t, best_f = t_hi, f_hi
        
        # Golden section on [t_lo, t_hi]
        # Since f is convex, golden section should work
        a, b = t_lo, t_hi
        c_pt = b - gr * (b - a)
        d_pt = a + gr * (b - a)
        fc = f1d(c_pt)
        fd = f1d(d_pt)
        
        for _ in range(30):  # 30 iterations gives precision ~ (0.618)^30 ≈ 1e-7
            if fc < fd:
                b = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = b - gr * (b - a)
                fc = f1d(c_pt)
            else:
                a = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = a + gr * (b - a)
                fd = f1d(d_pt)
        
        t_opt = (a + b) / 2.0
        f_opt = f1d(t_opt)
        if f_opt < best_f:
            best_t, best_f = t_opt, f_opt
        
        if abs(best_t) > 1e-12 and best_f < best - 1e-14:
            new_val = old + best_t
            new_val = np.clip(new_val, 0.0, 1.0)
            # Update h and c
            delta = new_val - h[j]
            h[j] = new_val
            c[k_lo:k_hi+1] += delta * g
            # Fix sum: adjust a free coordinate
            # Actually, we'll handle the constraint separately
            best = max(np.max(c) * dx, best_f * dx)
    
    return h, c, best

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    shifted = (c - cmax) * beta
    shifted -= shifted.max()
    return cmax + np.log(np.exp(shifted).sum()) / beta


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    shifted = (c - cmax) * beta
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        ks = k - (n - 1)
        lo1 = max(0, -ks)
        hi1 = min(n, n - ks)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            grad[idx] += wk * (1.0 - h[idx + ks])
        lo2 = max(0, ks)
        hi2 = min(n, n + ks)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= wk * h[idx - ks]
    return grad * dx


def cd_sweeps(h, n, n_sweeps, t_start, t_limit):
    dx = 2.0 / n
    target = n * 0.5
    n_full = 2 * n - 1
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(c) * dx)
    rng = np.random.RandomState(123)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.time() > t_limit:
            break
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            k_lo = max(0, n - 1 - j)
            k_hi = min(n_full - 1, n - 1 + j)
            n_affected = k_hi - k_lo + 1
            if n_affected == 0:
                continue
            ks_arr = np.arange(k_lo, k_hi + 1)
            idx1 = j + ks_arr - (n - 1)
            idx2 = j - ks_arr + (n - 1)
            g_coeffs = (1.0 - h[idx1]) - h[idx2]
            c_aff = c[k_lo:k_hi + 1]
            c_rest_max = float(np.max(np.concatenate([c[:k_lo], c[k_hi + 1:]]))) if n_full > n_affected else 0.0

            def f1d(t):
                return max(c_rest_max, float(np.max(c_aff + t * g_coeffs)))

            t_lo = -old
            t_hi = 1.0 - old
            if t_hi - t_lo < 1e-14:
                continue
            f_lo = f1d(t_lo)
            f_hi = f1d(t_hi)
            f_0 = f1d(0.0)
            best_f = min(f_lo, f_hi, f_0)
            best_t = 0.0
            if f_lo < best_f:

            if f_hi < best_f:
                best_t, best_f = t_hi, f_hi
            a, b = t_lo, t_hi
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)

                    a = c1
                    c1 = d1
                    fc1 = fd1
                    d1 = a + gr * (b - a)
                    fd1 = f1d(d1)
            t_opt = (a + b) / 2.0

                        c = np.correlate(h, 1.0 - h, mode='full')

    return h, best_c5


def slsqp_optimize(h, n, beta, maxiter, t_limit):
    dx = 2.0 / n

    def obj(x):
        return smooth_max_val(x, beta)

    def obj_grad(x):
        return smooth_max_grad(x, beta)

    def cons_fun(x, nn=n):
        return x.sum() - nn * 0.5

    def cons_jac(x, nn=n):
        return np.ones(nn)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bnds = [(0.0, 1.0)] * n
    t_rem = max(1.0, t_limit - time.time())
    mi = int(max(20, min(maxiter, t_rem / 0.05)))
    res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': mi, 'ftol': 1e-13})
    h_new = repair(res.x, n)
    return h_new, compute_c5(h_new)


def subgrad_project(h, n, n_iters, lr_init, t_limit):
    dx = 2.0 / n
    target = n * 0.5
    h = h.copy()
    best = compute_c5(h)
    lr = lr_init
    n_full = 2 * n - 1
    for it in range(n_iters):
        if time.time() > t_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(c)
        tol = 0.02 * cmax + 1e-10
        active = np.where(c >= cmax - tol)[0]
        grad = np.zeros(n)
        for k in active:
            ks = int(k) - (n - 1)
            lo1 = max(0, -ks)
            hi1 = min(n, n - ks)
            if hi1 > lo1:
                idx = np.arange(lo1, hi1)
                grad[idx] += (1.0 - h[idx + ks])
            lo2 = max(0, ks)
            hi2 = min(n, n + ks)
            if hi2 > lo2:
                idx = np.arange(lo2, hi2)
                grad[idx] -= h[idx - ks]
        grad *= dx / len(active)
        h_new = h - lr * grad
        # Project onto {0<=h<=1, sum(h)=target}
        lo_lam, hi_lam = h_new.min() - 1.0, h_new.max() + 1.0
        for _ in range(50):
            mid = (lo_lam + hi_lam) / 2.0
            hv = np.clip(h_new - mid, 0.0, 1.0)
            if hv.sum() > target:
                lo_lam = mid
            else:
                hi_lam = mid
        h_new = np.clip(h_new - (lo_lam + hi_lam) / 2.0, 0.0, 1.0)
        h_new = repair(h_new, n)
        v = compute_c5(h_new)
        if v < best:
            best = v
            h = h_new
        else:
            lr *= 0.7
        if lr < 1e-8:
            break
    return h, best
