        for _ in range(swaps_per_round):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            v = compute_c5(h)
            if v < best - 1e-12:
                best = v
                improved = True
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                break
            else:
                h[i], h[j] = 1.0, 0.0
                ones[i if i < len(ones) else 0] if False else None
        if not improved:
            if rd > 50:
                break
    return h, best


def pgd_optimize(h, n, max_iter, t0, lr0, use_full, time_limit):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    mu = 0.9
    t = t0
    lr = lr0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        g = grad_full(h, t) if use_full else grad_topk(h, 4)
        v = mu * v - lr * g
        h_new = project_onto(h + v, n)
        c5 = compute_c5(h_new)
        if c5 < best:
            best = c5
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 199:
            t *= 0.7
            lr *= 0.92
    return best_h, best


def slsqp_polish(h, n, max_iter=200):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    t = 150.0

    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        cm = np.max(cc)
        return (cm + np.log(np.sum(np.exp(t * (cc - cm)))) / t) * dx

    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5,
             'jac': lambda x, n=n: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return project_onto(res.x, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def el():
        return time.time() - t0

    def tl(frac):
        return budget_s * frac - el()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: PGD at n0 with full gradient
    if tl(0.20) > 5:
        h1, c1 = pgd_optimize(best_h, n0, 4000, 60.0, 0.02, True, t0 + budget_s * 0.18)
        h1 = project_onto(h1, n0)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Binarize + local search at n0
    if tl(0.35) > 5:
        hb = binarize(best_h, n0)
        hb, cb = local_search_bin(hb, n0, max_rounds=100, swaps_per_round=300)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 3: SLSQP polish on continuous
    if tl(0.50) > 20:
        hp = slsqp_polish(best_h, n0, max_iter=200)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp

    # Phase 4: Upsample to 2*n0
    if tl(0.75) > 30:
        n1 = n0 * 2
        x0 = np.linspace(0, 2, len(best_h))
        x1 = np.linspace(0, 2, n1)
        h_up = np.interp(x1, x0, best_h)
        h_up = project_onto(h_up, n1)
        h2, c2 = pgd_optimize(h_up, n1, 3000, 40.0, 0.01, False, t0 + budget_s * 0.70)
        h2 = project_onto(h2, n1)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

        if tl(0.75) > 10:
            h2b = binarize(h2, n1)
            h2b, c2b = local_search_bin(h2b, n1, max_rounds=80, swaps_per_round=200)
            if c2b < best_c5:
                best_c5, best_h = c2b, h2b

    # Phase 5: Random restarts + re-optimize at n0
    for attempt in range(3):
        if tl(0.90) < 15:
            break
        noise_scale = 0.15 * (1 - attempt * 0.3)
        hn = best_h + np.random.randn(n0) * noise_scale
        hn = project_onto(hn, n0)
        h3, c3 = pgd_optimize(hn, n0, 2000, 40.0, 0.015, True, t0 + budget_s * 0.88)
        h3 = project_onto(h3, n0)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
        # Also try binarized version
        if tl(0.90) > 5:
            h3b = binarize(h3, n0)
            h3b, c3b = local_search_bin(h3b, n0, max_rounds=50, swaps_per_round=150)
            if c3b < best_c5:
                best_c5, best_h = c3b, h3b

    # Phase 6: Final SLSQP + local search
    if tl(0.98) > 10:
        nf = len(best_h)
        if nf <= 500:
            hf = slsqp_polish(best_h, nf, max_iter=150)
            cf = compute_c5(hf)
            if cf < best_c5:
                best_c5, best_h = cf, hf
        hb = binarize(best_h, nf)
        hb, cb = local_search_bin(hb, nf, max_rounds=60, swaps_per_round=200)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    best_h = project_onto(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
