import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _binary_round(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _binary_local_search(h, dx, n, max_passes=200, t_limit=60.0, t0=0.0):
    h = _binary_round(h, n)
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    for _ in range(max_passes):
        if time.time() - t0 > t_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = -1e-15
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c = _c5(h, dx)
                h[j] = 0
                h[i] = 1
                if c - best_c5 < best_delta:

                    best_pair = (i, j)

            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
            best_c5 += best_delta
            best_h = h.copy()
        else:
            break
    return best_h, best_c5

    cv = _corr(x, cx[0])
    mx = cv.max()
    eps = cx[1]
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

    cv = _corr(x, cx[0])
    mx = cv.max()
    eps = cx[1]
    w = np.exp((cv - mx) / eps)

        method='SLSQP',

        if not improved_any:
            stall += 1
            if stall > 2:
                break
        else:
            stall = 0
            if rnd > 0 and rnd % 3 == 0:
                kswap = rng_state.randint(2, 6)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(kswap):
                    a = ones[rng_state.randint(len(ones))]
                    b = zeros[rng_state.randint(len(zeros))]
                    h[a], h[b] = h[b], h[a]
    return best_h, best_c5


def _smooth_obj(x, dx, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))


def _smooth_grad(x, dx, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(x)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g


def _efun(x, target):
    return x.sum() - target


def _ejac(x, target):
    return np.ones(len(x))


def _slsqp(h0, dx, n, target, eps, maxiter=250):
    res = minimize(
        _smooth_obj, h0, args=(dx, eps),
        jac=_smooth_grad,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': _efun, 'args': (target,), 'jac': _ejac}],
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return _proj(res.x, target)


rng_state = np.random.RandomState(42)


def _optimize_phase(h0, dx, n, target, t_end):
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = h0.copy()
    eps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for eps in eps_list:
        if time.time() > t_end - 15:
            break
        try:
            h_new = _slsqp(h, dx, n, target, eps, maxiter=300)
            c_new = _c5(h_new, dx)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break
    # Binary round and local search
    if time.time() < t_end - 20:
        h_bin = _binary_round(h, n)
        t_rem = min(35.0, t_end - time.time() - 5)
        if t_rem > 5:
            h_bo, c_bo = _binary_local_search(h_bin, dx, n,
                                              max_passes=120,
                                              t_start=time.time(),
                                              t_limit=t_rem)
            if c_bo < best_c5:
                best_c5 = c_bo
                best_h = h_bo.copy()
                h = h_bo + rng_state.normal(0, 0.01, n)
                h = _proj(h, target)
    # Polish from best
    if time.time() < t_end - 8:
        for eps in [0.0005, 0.0002, 0.0001]:
            if time.time() > t_end - 3:
                break
            try:
                h2 = _slsqp(best_h, dx, n, target, eps, maxiter=200)
                c2 = _c5(h2, dx)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()
            except Exception:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global rng_state
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng_state = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = None

    n_list = [70]
    if h_init is not None and len(h_init) != 70:
        h_init = np.interp(np.linspace(0, 1, 70),
                           np.linspace(0, 1, len(h_init)), h_init)
    for nn in [60, 80, 50, 90]:
        if nn % 2 == 0:
            n_list.append(nn)

    # Phase 1: Optimize at n=70 (primary)
    n = 70
    dx = 2.0 / n
    target = n / 2.0
    if h_init is not None:
        h0 = _proj(h_init, target)
    else:
        h0 = rng_state.uniform(0.2, 0.8, n)
        h0 = _proj(h0, target)
    t_end1 = t0 + (deadline - t0) * 0.50
    h_best_70, c_best_70 = _optimize_phase(h0, dx, n, target, t_end1)
    best_h = h_best_70.copy()
    best_c5 = c_best_70

    # Phase 2: Restart from perturbed versions at n=70
    t_end2 = t0 + (deadline - t0) * 0.72
    for _ in range(8):
        if time.time() > t_end2:
            break
        amp = rng_state.uniform(0.03, 0.15)
        h_pert = _proj(best_h + rng_state.normal(0, amp, n), target)
        sub_end = min(time.time() + min(25.0, t_end2 - time.time()), t_end2)
        if sub_end - time.time() > 4:
            hp, cp = _optimize_phase(h_pert, dx, n, target, sub_end)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()

    # Phase 3: Try other n values
    for nn in n_list[1:]:
        if time.time() > deadline - 40:
            break
        if nn == n:
            continue
        dxn = 2.0 / nn
        tgn = nn / 2.0
        if best_h is not None:
            hn = np.interp(np.linspace(0, 1, nn),
                           np.linspace(0, 1, len(best_h)), best_h)
        else:
            hn = rng_state.uniform(0.3, 0.7, nn)
        hn = _proj(hn, tgn)
        t_end_n = min(time.time() + 80.0, deadline - 30)
        if time.time() >= t_end_n:
            continue
        hn_opt, cn_opt = _optimize_phase(hn, dxn, nn, tgn, t_end_n)
        if cn_opt < best_c5:
            best_c5 = cn_opt
            best_h = hn_opt.copy()

    # Phase 4: Final polish on the best solution
    if best_h is not None and time.time() < deadline - 15:
        nb = len(best_h)
        dxb = 2.0 / nb
        tgb = nb / 2.0
        t_end_f = deadline - 3
        # Binary search on best
        h_b = _binary_round(best_h, nb)
        t_bl = min(20.0, t_end_f - time.time() - 5)
        if t_bl > 5:
            h_bo, c_bo = _binary_local_search(h_b, dxb, nb,
                                              max_passes=100,
                                              t_start=time.time(),
                                              t_limit=t_bl)
            if c_bo < best_c5:
                best_c5 = c_bo
                best_h = h_bo.copy()
        # Continuous polish
        if time.time() < t_end_f - 5:
            for eps in [0.0002, 0.0001, 0.00005]:
                if time.time() > t_end_f - 2:
                    break
                try:
                    h2 = _slsqp(best_h, dxb, nb, tgb, eps, maxiter=200)
                    c2 = _c5(h2, dxb)
                    if c2 < best_c5:
                        best_c5 = c2
                        best_h = h2.copy()
                except Exception:
                    break

    if best_h is None:
        best_h = _proj(rng_state.uniform(0.3, 0.7, 70), 35.0)
        best_c5 = _c5(best_h, 2.0 / 70)

    return best_h, best_c5, len(best_h)
