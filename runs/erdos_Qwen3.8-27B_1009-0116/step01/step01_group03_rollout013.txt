```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def soft_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    lse = max_s + np.log(np.sum(exp_s))
    return (m + lse / beta) * dx

def lgbfs_optimize(h, n, ts, beta, maxiter=200):
    def obj(x):
        return soft_obj(x, n, beta)
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    d2 = ts - h.sum()
    if abs(d2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + d2, 0, 1)
    return h

def soft_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    s = (corr - m) * beta
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / beta) * dx

    n = len(h)
    res = minimize(lambda x: soft_obj(x, beta), h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})

        return project_feasible(res.x, ts)
    return project_feasible(h, ts)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, ts):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    d2 = ts - h.sum()
    if abs(d2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + d2, 0, 1)
    return h

def soft_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    s = (corr - m) * beta
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / beta) * dx

def slsqp_opt(h, ts, beta, maxiter=600):
    n = len(h)
    res = minimize(lambda x: soft_obj(x, beta), h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def binarize(h, ts):
    h_b = (h >= 0.5).astype(float)
    h_b = project_feasible(h_b, ts)
    n = len(h_b)
    n_ones = int(h_b.sum())
    if n_ones != n // 2:
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if n_ones > n // 2:
            for k in range(n_ones - n // 2):
                h_b[ones[k]] = 0.0
        else:
            for k in range(n // 2 - n_ones):
                h_b[zeros[k]] = 1.0
    return h_b

def pair_swap_hill(h, ts, max_sweeps=5000, tlimit=30):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlimit:
            break
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5

def pair_swap_sa(h, ts, tlimit=50, T0=0.5, T1=0.005, seed=0):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t0 = time.time()
    rng = np.random.RandomState(seed)
    while time.time() - t0 < tlimit:
        elapsed = time.time() - t0
        frac = elapsed / tlimit
        T = T0 * (T1 / T0) ** frac
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-13:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def rem():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Stage 1: Continuous SLSQP on initial n
    ts_init = n_init / 2.0
    for beta in [10, 30, 100, 300, 800]:
        if rem() < 40:
            break
        h_s = slsqp_opt(best_h, ts_init, beta, maxiter=1000)
        c5 = compute_c5(h_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s

    # Stage 2: Binary optimization across multiple n
    n_targets = sorted(set([n_init, 142, 138, 144, 136, 146, 134, 148, 150]))
    for n in n_targets:
        if rem() < 70:
            break
        ts = n / 2.0

        # Build candidate initial binary for this n
        if n == n_init:
            h_cand = binarize(best_h, ts)
        else:
            h_int = interp_to_n(best_h, n)
            h_cand = binarize(h_int, ts)
        # Also try a random start
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0

        for h_start in [h_cand, h_rand]:
            if rem() < 50:
                break
            # Hill climb
            tlimit_hc = min(35, max(10, rem() - 30))
            h_hs, c5_hs = pair_swap_hill(h_start, ts, max_sweeps=8000, tlimit=tlimit_hc)
            if c5_hs < best_c5:
                best_c5 = c5_hs
                best_h = h_hs

            # SA ref-start
            if rem() > 90:
                tlimit_sa = min(45, max(10, rem() - 20))
                h_sa, c5_sa = pair_swap_sa(h_hs, ts, tlimit=tlimit_sa, seed=seed + n)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa

                # Hill climb after SA
                if rem() > 50:
                    h_hc2, c5_hc2 = pair_swap_hill(h_sa, ts, max_sweeps=5000, tlimit=min(20, rem() - 10))
                    if c5_hc2 < best_c5:
                        best_c5 = c5_hc2
                        best_h = h_hc2

        # Continuous polish on best binary for this n
        if rem() > 35:
            h_pol = best_h.copy() if len(best_h) == n else h_hs.copy()
            for beta in [100, 400, 1000]:
                if rem() < 20:
                    break
                h_pol = slsqp_opt(h_pol, ts, beta, maxiter=700)
                c5_p = compute_c5(h_pol)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pol

    # Stage 3: Final intensive polish
    if rem() > 20:
        n_best = len(best_h)
        ts_best = n_best / 2.0
        for beta in [300, 800, 1500, 3000]:
            if rem() < 15:
                break
            h_f = slsqp_opt(best_h, ts_best, beta, maxiter=800)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Final projection and check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
