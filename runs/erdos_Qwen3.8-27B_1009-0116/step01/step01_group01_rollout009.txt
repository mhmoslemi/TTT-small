import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx, dx

def compute_subgradient(h, m, dx):
    n = len(h)
    g = np.zeros(n)
    # m is the lag offset (can be negative)
    # c[m_offset] = sum_i h[i+m] * (1-h[i]) for valid i
    # dc/da[j]: from h[i+m] term: (1-h[i]) when i=j-m, valid if 0<=j-m<n
    #           from (1-h[i]) term: -h[i+m] when i=j, valid if 0<=j+m<n
    j = np.arange(n)
    # First term: h[i+m] with i = j-m
    valid1 = (j - m >= 0) & (j - m < n) & (j >= 0) & (j < n)
    # Second term: (1-h[i]) with i = j
    valid2 = (j >= 0) & (j < n) & (j + m >= 0) & (j + m < n)
    g[valid1] += (1.0 - h[j[valid1] - m]) 
    g[valid2] -= h[j[valid2] + m]
    return g * dx

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if diff > 1e-14:
        mask = h < 1.0
        if mask.sum() > 0:
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        # If still not enough, bump up the ones that are exactly 1
        diff2 = target - h.sum()
        if diff2 > 1e-14:
            h[np.argmin(h)] += diff2
    elif diff < -1e-14:
        mask = h > 0.0
        if mask.sum() > 0:
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
        diff2 = target - h.sum()
        if diff2 < -1e-14:
            h[np.argmax(h)] += diff2
    return h

def subgradient_optimize(h0, n, max_iter=10000, top_m=5, base_lr=0.01):
    """Subgradient method with adaptive step size for the min-max problem."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(max_iter):
        # Compute all lags
        c, _ = compute_all_lags(h)
        cmax = np.max(c)
        current_c5 = cmax
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find top m lags
        # c has length 2n-1, index n-1+m corresponds to lag m
        # We want the top m values
        top_indices = np.argsort(c)[-top_m:]
        
        # Compute subgradient as average of active lags
        g = np.zeros(n)
        for idx in top_indices:
            m = idx - (n - 1)  # actual lag offset
            g += compute_subgradient(h, m, dx)
        g /= top_m
        
        # Adaptive step size (Zink发表 method)
        # alpha_t = c_t / ||g||^2
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break
        alpha = current_c5 / g_norm_sq
        alpha = min(alpha, base_lr)
        
        # Gradient step
        h_new = h - alpha * g
        
        # Project
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def softmax_smooth_objective(h, n, dx, temperature):
    """Smooth approximation of max using soft-max."""
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    # Soft max: cmax + (1/t) * log(sum(exp(t*(c-cmax))))
    t = temperature
    shifted = t * (c - cmax)
    # For numerical stability
    log_sum = np.log(np.sum(np.exp(shifted - np.max(shifted))) + 1e-30) + np.max(shifted)
    return (cmax + log_sum / t)
