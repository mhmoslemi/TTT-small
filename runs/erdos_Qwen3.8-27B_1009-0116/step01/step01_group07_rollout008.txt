import numpy as np
from scipy.optimize import minimize
import time

def c5_val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def c5_lag_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sh = np.clip(corr - m, -80.0 / alpha, 80.0 / alpha)
    return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * (2.0 / len(h))

def lag_focus_obj(h, top_indices, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[top_indices]
    m = vals.max()
    sh = np.clip(vals - m, -20.0 / alpha, 20.0 / alpha)
    return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * (2.0 / len(h))

def norm_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def eq_fun(x, target):
    return x.sum() - target

def eq_jac(x):
    return np.ones_like(x)

def coord_search(h, n_iter=30, fine=False):
    n = len(h)
    target = n / 2.0
    best_c = c5_val(h)
    if fine:
        deltas = (0.005, 0.01, 0.02, 0.04)
    else:
        deltas = (0.02, 0.05, 0.1, 0.2, 0.4)
    for _ in range(n_iter):
        moved = False
        for i in range(n):
            if moved:
                break
            for j in range(n):
                if j == i or moved:
                    continue
                for d in deltas:
                    for sign in (1, -1):
                        h2 = h.copy()
                        hi2 = h[i] + sign * d
                        hj2 = h[j] - sign * d
                        if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                            h2[i] = hi2
                            h2[j] = hj2
                            c = c5_val(h2)
                            if c < best_c - 1e-11:
                                h = h2
                                best_c = c
                                moved = True
                                break
        if moved:
            h = norm_h(h, target)
            best_c = c5_val(h)
        else:
            break
    return h, best_c

def random_hill(h, n_iter=100, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    best_c = c5_val(h)
    rng = np.random.RandomState(12345)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            continue
        d = rng.choice((0.01, 0.03, 0.06, 0.1, 0.15, 0.25, 0.4))
        sign = rng.choice((1, -1))
        hi2 = h[i] + sign * d
        hj2 = h[j] - sign * d
        if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
            h2 = h.copy()
            h2[i] = hi2
            h2[j] = hj2
            c = c5_val(h2)
            if c < best_c - 1e-12:
                h = h2
                best_c = c
    return h, best_c

def try_slqp(hc, n, bounds, cons, target, alphas, maxiter_hi, t0, budget_s, label):
    best_loc = None
    best_c_loc = c5_val(hc)
    for al in alphas:
        if budget_s - (time.time() - t0) < 6:
            break
        try:
            res = minimize(smooth_obj, hc, args=(al,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_hi, 'ftol': 1e-13})
            hn = norm_h(res.x, target)
            cn = c5_val(hn)
            if cn < best_c_loc:
                best_c_loc = cn
                best_loc = hn.copy()
                print(f"  {label} al={al}: C5={cn:.6f}")
            hc = hn
        except Exception:
            pass
    if best_loc is not None:
        return best_loc, best_c_loc
    return hc, best_c_loc
