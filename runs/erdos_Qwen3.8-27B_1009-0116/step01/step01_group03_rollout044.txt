                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})

            return project_feasible(res.x, ts)
    except Exception:
        pass

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    ts = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:ts]] = 1.0
    return b

def sa_binary(h, n, rng, t_deadline, T_init=0.015, max_iters=300000):
    dx = 2.0 / n
    h = h.astype(float).copy()
    ones_idx = np.where(h == 1.0)[0].copy()

    c5 = float(corr.max())
    best_c5 = c5
    best_h = h.copy()
    T = T_init
    T_final = 0.0005
    cool = (T_final / T_init) ** (1.0 / max(1, max_iters))

            break

        j = zeros_idx[zi]
        h[i] = 0.0
        h[j] = 1.0

        d = c5n - c5
