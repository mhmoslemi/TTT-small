import time

def compute_corrs(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def get_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad(h, corr, n, top_m):
    """Average subgradient for top-M worst lags."""
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        shift = k_idx - (n - 1)
        # Term 1: (1-h[j+shift]) for j where 0<=j<n, 0<=j+shift<n
        lo1 = max(0, -shift)
        hi1 = min(n-1, n-1-shift)
        if hi1 >= lo1:
            grad[lo1:hi1+1] += (1.0 - h[lo1+shift:hi1+1+shift])
        # Term 2: -h[j-shift] for j where 0<=j-shift<n, 0<=j<n
        lo2 = max(0, shift)
        hi2 = min(n-1, n-1+shift)
        if hi2 >= lo2:
            grad[lo2:hi2+1] -= h[lo2-shift:hi2+1-shift]
    return grad / top_m

def coord_descent(h, n, max_rounds=50):
    """Simple coordinate descent: for each j, try a few values."""
    best_val = get_c5(h, n)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            orig = h[j]
            for cand in [orig - 0.1, orig - 0.05, orig, orig + 0.05, orig + 0.1]:
                if cand < 0 or cand > 1:
                    continue
                h[j] = cand
                h = repair(h, n)
                c = get_c5(h, n)
                if c < best_val - 1e-14:
                    best_val = c
                    improved = True
                    break
                else:
                    h[j] = orig
        if not improved:
            break
        h = repair(h, n)
    return h, best_val
