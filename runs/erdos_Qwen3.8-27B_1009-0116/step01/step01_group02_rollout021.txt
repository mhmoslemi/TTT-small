    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:

    h = np.clip(h, 0.0, 1.0)

        cnt = int(np.sum(mask))
        if cnt == 0:

            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
