```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_detail(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr, float(np.max(corr))

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    # Scale to meet target
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix any remaining discrepancy
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    if m > 700:
        return 1e10  # Overflow protection
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def binary_round(h, n):
    """Round h to binary, maintaining sum = n/2."""
    n_ones = n // 2
    # Sort by value, take top n_ones as 1s
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n_ones]] = 1.0
    return b

def binary_local_search(h_bin, n, max_iters=500, rng=None):
    """Local search on binary solution: swap adjacent or random pairs."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.copy()
    best = compute_c5(h, n)
    
    ones = set(np.where(h == 1.0)[0])
    zeros = set(np.where(h == 0.0)[0])
    
    for it in range(max_iters):
        if not zeros:
            break
        # Try a random swap
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h, n)
        if c < best:
            best = c
            ones.discard(i); ones.add(j)
            zeros.discard(j); zeros.add(i)
        else:
            h[i], h[j] = 1.0, 0.0
    
    return h, best
```
