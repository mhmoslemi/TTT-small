import numpy as np
from scipy.optimize import minimize
import time

_G_dx = 2.0 / 70
_G_target = 35.0
_G_alpha = 100.0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _so(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(corr - m, -100.0 / _G_alpha, 100.0 / _G_alpha)
    return (m + np.log(np.sum(np.exp(_G_alpha * s))) / _G_alpha) * _G_dx + 2000.0 * (np.sum(x) - _G_target) ** 2

def _eo(x):
    return np.sum(x) - _G_target

def _ej(x):
    return np.ones_like(x)

def _fix_sum(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    m = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = int(np.sum(m))
    if cnt == 0:
        h -= diff / len(h)
        return np.clip(h, 0, 1)
    h[m] += diff / cnt
    h = np.clip(h, 0, 1)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-14:
        h -= diff2 / len(h)
        h = np.clip(h, 0, 1)
    return h

def _local_refine(h, target, t_limit, rng_seed=7):
    n = len(h)
    t0 = time.time()
    best = _c5(h)
    h = h.copy()
    rng = np.random.RandomState(rng_seed)
    stalls = 0
    step_list = (0.003, 0.007, 0.015, 0.03, 0.06, 0.1, 0.15)
    while time.time() - t0 < t_limit:
        i = rng.randint(n)
        j = rng.randint(n - 1)
        j += (j >= i)
        moved = False
        for ss in step_list:
            for sgn in (1, -1):
                d = sgn * ss
                v1, v2 = h[i] + d, h[j] - d
                if v1 < -1e-12 or v1 > 1 + 1e-12 or v2 < -1e-12 or v2 > 1 + 1e-12:
                    continue
                h2 = h.copy()
                h2[i] = v1
                h2[j] = v2
                c = _c5(h2)
                if c < best - 5e-14:
                    h = h2
                    best = c
                    moved = True
                    break
            if moved:
                break
        if not moved:
            stalls += 1
            if stalls > 800:
                break
        else:
            stalls = 0
    return h, best

def _cyclic_local(h, target, n_passes=5):
    n = len(h)
    best = _c5(h)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in (0.02, -0.02, 0.05, -0.05):
                    v1, v2 = h[i] + d, h[j] - d
                    if v1 < 0 or v1 > 1 or v2 < 0 or v2 > 1:
                        continue
                    h2 = h.copy()
                    h2[i] = v1
                    h2[j] = v2
                    c = _c5(h2)
                    if c < best - 1e-13:
                        h = h2
                        best = c
                        improved = True
                        break
                if not improved and i > n - 3:
                    break
        if not improved:
            break
    return h, best

def _slsqp_phase(h0, n, target, alphas, maxiter_base=50):
    global _G_dx, _G_target, _G_alpha
    _G_dx = 2.0 / n
    _G_target = target
    dx = 2.0 / n
    best = h0.copy()
    best_c5 = _c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eo, 'jac': _ej}]
    for alpha in alphas:
        _G_alpha = alpha
        mi = maxiter_base if n <= 80 else maxiter_base - 10
        try:
            res = minimize(_so, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': mi, 'ftol': 1e-12})
            hn = _fix_sum(res.x, target)
            cn = _c5(hn)
            if cn < best_c5:
                best_c5 = cn
                best = hn.copy()
        except Exception:
            pass
    return best, best_c5

def _lbfbs_phase(h0, n, target, alphas, maxiter=60):

def _gen_init(n, target, rng, kind, parent=None):
    if kind == 0 and parent is not None and len(parent) != n:
        xp = np.linspace(0, 1, len(parent))
        x = np.linspace(0, 1, n)
        h = np.interp(x, xp, parent)

    if kind == 0 and parent is not None:

    if kind == 1:

        h += rng.uniform(-0.1, 0.1, n)
        return _fix_sum(np.clip(h, 0, 1), target)
    if kind == 2:

        return _fix_sum(np.clip(h, 0, 1), target)
    h = np.full(n, 0.5) + rng.uniform(-0.35, 0.35, n)
    return _fix_sum(np.clip(h, 0, 1), target)
