            g[i1] += wk * (1.0 - x[i1 - sp])

def _obj_soft(x, alpha):

    m = c.max()

def _sum_jac(x):

def _proj_sum(h, T):
    target = T / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / T
    return np.clip(h, 0.0, 1.0)

def _nm_31(p0, m, budget_inner=60):

    n = 31 * m
    def f(p):

    bnds = [(0.0, 1.0)] * 31
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 15.5,
             'jac': lambda x: np.ones(31)}]
    # Use penalty for Nelder-Mead (it doesn't handle constraints)

def _pgd_period(h_p, T, n_full, alphas, max_iters, lr_init=0.004, seed=0, nesterov=True):
    m = n_full // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    momentum = np.zeros(T)
    beta = 0.9
    for alpha in alphas:
        iters = max_iters // len(alphas)
        stall = 0
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            mm = c.max()
            s = np.clip(alpha * (c - mm), -500, 500)
            w = np.exp(s - s.max())
            w /= w.sum()
            dx = 2.0 / n_full
            g_full = np.zeros(n_full)
            for k in range(2 * n_full - 1):
                if w[k] < 1e-12:
                    continue
                sh = (n_full - 1) - k
                wk = w[k] * dx
                if sh >= 0:
                    cnt = n_full - sh
                    i1 = np.arange(cnt)
                    g_full[i1] += wk * (1.0 - h_full[i1 + sh])
                    if sh > 0:
                        i2 = np.arange(sh, n_full)
                        g_full[i2] -= wk * h_full[i2 - sh]
                else:
                    sp = -sh
                    i1 = np.arange(sp, n_full)
                    g_full[i1] += wk * (1.0 - h_full[i1 - sp])
                    i2 = np.arange(n_full - sp)
                    g_full[i2] -= wk * h_full[i2 + sp]
            g_p = np.zeros(T)
            for p_idx in range(m):
                g_p += g_full[p_idx * T:(p_idx + 1) * T]
            g_p /= m
            if nesterov:
                momentum = beta * momentum + (1.0 - beta) * g_p
                grad_eff = g_p + beta * momentum
            else:
                momentum = beta * momentum + (1.0 - beta) * g_p
                grad_eff = momentum
            h_new = _proj_sum(h_p - lr * grad_eff, T)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                stall = 0
                if lr < 0.1:
                    lr *= 1.05
            else:
                stall += 1
                lr *= 0.5
                if lr < 1e-7:
                    lr = lr_init
                    momentum = np.zeros(T)
            if stall > max_iters // len(alphas) // 5 and it > 100:
                break
    return best, best_c

def _coord_descent(h, n_iter, scales, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(len(h))
            j = rng.randint(len(h))
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):

    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_soft(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    log_sum = float(np.log(np.sum(np.exp(s))))
    return (2.0 / n) * (m + log_sum / alpha)

def _grad_soft(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    s = alpha * c
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[sp:])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _proj_sum(h, T):
    target = T / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / T
    return np.clip(h, 0.0, 1.0)

def _slsqp(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_soft, h0, jac=_grad_soft, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _nm_pattern(p0, m):
    n_full = 31 * m
    def f_p(x):
        h = _proj_sum(x, 31)
        return _eval(np.tile(h, m))
    def f_pen(x):
        return f_p(x) + 5.0 * abs(x.sum() - 15.5)
    try:
        res = minimize(f_pen, _proj_sum(p0, 31), method='Nelder-Mead',
                       options={'maxiter': 15000, 'xatol': 1e-15, 'fatol': 1e-17,
                                'adaptive': True, 'initial_simplex_size': 0.08})
        return _proj_sum(res.x, 31)
    except Exception:
        return _proj_sum(p0, 31)

def _pgd_period(h_p, T, n_full, alphas, max_iters, lr_init=0.005, seed=0, beta=0.9):
    m = n_full // T
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    moment = np.zeros(T)
    for alpha in alphas:
        iters = max_iters // len(alphas)
        stall = 0
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            mm = c.max()
            s = np.clip(alpha * (c - mm), -500, 0)
            w = np.exp(s)
            w /= w.sum()
            dx = 2.0 / n_full
            g_full = np.zeros(n_full)
            for k in range(2 * n_full - 1):
                if w[k] < 1e-12:
                    continue
                sh = (n_full - 1) - k
                wk = w[k] * dx
                if sh >= 0:
                    cnt = n_full - sh
                    g_full[:cnt] += wk * (1.0 - h_full[sh:sh + cnt])
                    if sh > 0:
                        g_full[sh:] -= wk * h_full[:cnt]
                else:
                    sp = -sh
                    g_full[sp:] += wk * (1.0 - h_full[sp:])
                    g_full[:n_full - sp] -= wk * h_full[sp:]
            g_p = np.zeros(T)
            for p_idx in range(m):
                g_p += g_full[p_idx * T:(p_idx + 1) * T]
            g_p /= m
            moment = beta * moment + (1.0 - beta) * g_p
            grad_eff = g_p + beta * moment
            h_new = _proj_sum(h_p - lr * grad_eff, T)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                stall = 0
                if lr < 0.15:
                    lr *= 1.04
            else:
                stall += 1
                lr *= 0.5
                if lr < 1e-7:
                    lr = lr_init
                    moment = np.zeros(T)
            if stall > iters // 4 and it > 50:
                break
    return best, best_c

def _coord_desc(h, n_iter, scales, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    nn = len(h)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(nn)
            j = rng.randint(nn)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pairwise_cd31(p, m, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(np.tile(p, m))
    for _ in range(n_iter):

        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            p[i], p[j] = oi, oj
    return p, c5c
