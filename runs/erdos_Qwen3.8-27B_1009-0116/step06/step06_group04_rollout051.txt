import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

def _sa_binary(hb, n, rng, n_iters, deadline, T0=0.01):
    """Simulated annealing with 2-swap moves for binary sequences"""
    x = hb.copy()
    ones_pos = list(np.where(x > 0.5)[0])
    zeros_pos = list(np.where(x <= 0.5)[0])
    c_cur = _binary_eval(x)
    best_x = x.copy()
    best_c = c_cur
    T = T0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # 2-swap: pick one 1 and one 0, flip them
        if not ones_pos or not zeros_pos:
            break
        oi = ones_pos[rng.randint(len(ones_pos))]
        zi = zeros_pos[rng.randint(len(zeros_pos))]
        
        x[oi] = 0.0
        x[zi] = 1.0
        
        c_new = _binary_eval(x)
        delta = c_new - c_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c_cur = c_new
            ones_pos.remove(oi)
            ones_pos.append(zi)
            zeros_pos.remove(zi)
            zeros_pos.append(oi)
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
        else:
            x[oi] = 1.0
            x[zi] = 0.0
        
        T *= 0.9999
        if T < 0.0001:
            T = 0.0001
    
    return best_x, best_c

    try:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_logsumexp(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s_vals = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s_vals - s_vals.max())
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    def _o(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sv = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(sv)))) / alpha
    def _g(x):
        return _grad_logsumexp(x, alpha)
    def _se(x):
        return x.sum() - n * 0.5
    def _sj(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_o, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
