            hb, _ = _subgrad(hb, 10000, 0.5, 0.002, n0, 3, 3000)
            update(hb, n0)
            if elapsed() > budget_s - 200:
                break
            hb = _smooth_opt(hb, 200000, 8000, 200)
            update(hb, n0)

        # Try different resolutions
        for n_try in [250, 400, 500]:
            if elapsed() > budget_s - 200:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _smooth_opt(h_r, 50000, 5000, 200)
            h_r, _ = _subgrad(h_r, 5000, 0.5, 0.002, n_try, 3, 2000)
            h_r = _smooth_opt(h_r, 200000, 8000, 200)
            update(h_r, n_try)

    # Phase 6: Simulated annealing on binary + polish (150s)
    if elapsed() < budget_s - 100:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _repair(h_bin, len(h_bin))
        t_rem = max(budget_s - 100 - elapsed(), 10)
        n_steps = max(int(t_rem * 600), 1000)
        h_a, _ = _anneal(h_bin, 0.3, 0.0005, n_steps, rng)
        h_a = _smooth_opt(h_a, 100000, 5000, 200)
        h_a, _ = _subgrad(h_a, 5000, 0.3, 0.003, len(h_a), 3, -1)
        h_a = _smooth_opt(h_a, 300000, 10000, 200)
        update(h_a, len(h_a))

    # Phase 7: Intensive subgradient on best (100s)
    if elapsed() < budget_s - 50:
        h7 = best_h.copy()
        n7 = len(h7)
        t_rem = max(budget_s - 50 - elapsed(), 5)
        n_it = max(int(t_rem * 800), 2000)
        h7, c5_7 = _subgrad(h7, n_it, 0.3, 0.002, n7, 5, 3000)
        update(h7, n7)
        h7 = _smooth_opt(h7, 500000, 10000, 300)
        update(h7, n7)

    # Phase 8: Final coordinate descent + micro-polish (40s)
    if elapsed() < budget_s - 3:
        h8 = best_h.copy()
        nc = len(h8)
        h8, _ = _coord_descent(h8, 5000, rng)
        update(h8, nc)
        h8 = _smooth_opt(h8, 1000000, 20000, 200)
        update(h8, nc)

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    assert best_h.shape == (best_n,)
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(best_h.sum() - best_n / 2.0) < 1.0
    return (best_h, best_c5, best_n)
