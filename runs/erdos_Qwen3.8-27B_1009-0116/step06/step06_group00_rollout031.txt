    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lbfgs_lp(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 15:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_pnorm(h, p, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 15:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_opt(h, steps, step0, tol, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.02 * i)
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / len(active)
        h = _proj(h - t * g, n)
    return h

def _top_lag(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 20:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_lags
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5]:
            h_new = _proj(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _local_2pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _local_3pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 6)) % n
        k = (i + rng.randint(7, 14)) % n
        d1 = 3e-5 * rng.randn()
        d2 = 3e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _up(h, f):
    return np.repeat(h, f).copy()

def _down(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS with adaptive alpha (log-sum-exp)
    h = h0.copy()
    for alpha in [500, 2000, 10000, 50000, 200000, 800000]:
        if time.time() > deadline - 400:
            break
        h = _lbfgs_lp(h, alpha, 4000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: p-norm optimization (different smoothing structure)
    h = best_h.copy()
    for p in [10, 30, 100, 300]:
        if time.time() > deadline - 300:
            break
        h = _lbfgs_pnorm(h, p, 3000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Subgradient on exact max (no smoothing bias)
    if time.time() < deadline - 200:
        h = _subgrad_opt(best_h, 800, 0.02, 1e-8, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Top-lag descent multiple configurations
    if time.time() < deadline - 150:
        for nl in [3, 5, 2, 4]:
            if time.time() > deadline - 150:
                break
            h_ml, c5_ml = _top_lag(best_h.copy(), nl, 500, deadline)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()

    # Phase 5: Downsample to half, optimize, upsample back
    if time.time() < deadline - 120:
        n_half = n0 // 2
        h_half = _down(best_h, 2)
        h_half = _repair(h_half, n_half)
        for alpha in [2000, 20000, 100000]:
            if time.time() > deadline - 120:
                break
            h_half = _lbfgs_lp(h_half, alpha, 3000, deadline)
        h_up = _repair(_up(h_half, 2), n0)
        c5v = _eval(h_up)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_up.copy()
            # Further refine at full res
            h = _lbfgs_lp(h_up, 200000, 5000, deadline)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 6: Final L-BFGS with very high alpha
    if time.time() < deadline - 80:
        h = _lbfgs_lp(best_h, 2000000, 5000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 7: Subgradient refinement with tight tol
    if time.time() < deadline - 50:
        h = _subgrad_opt(best_h, 2000, 0.005, 1e-10, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 8: Top-lag pass again
    if time.time() < deadline - 30:
        h_ml, c5_ml = _top_lag(best_h.copy(), 3, 300, deadline)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 9: 3-point local search
    if time.time() < deadline - 5:
        h_ls, c5_ls = _local_3pt(best_h, 30000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 10: 2-point ultra-fine polish
    if time.time() < deadline - 2:
        h_ls, c5_ls = _local_2pt(best_h, 80000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
