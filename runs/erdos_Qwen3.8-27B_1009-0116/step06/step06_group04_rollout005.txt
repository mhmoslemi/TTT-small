def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _subgrad(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        cnt = n - sh
        i1 = np.arange(cnt)
        g[i1] += dx * (1.0 - h[i1 + sh])
        if sh > 0:
            g[sh:] -= dx * h[:sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:n - sp])
        g[:n - sp] -= dx * h[sp:]
    return g

def _topk_grad(h, tk):
    c = _corr(h)
    ks = np.argsort(c)[-tk:]
    g = np.zeros(len(h))
    for k in ks:
        g += _subgrad(h, int(k))
    return g / tk

def _nesterov(h0, n_iter, deadline, lr0, lr1, mu, tk):
    n = len(h0)
    x = h0.copy()
    vx = np.zeros(n)
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        lr = lr0 * (lr1 / lr0) ** (i / max(n_iter - 1, 1))
        xl = np.clip(x + mu * vx, 0.0, 1.0)
        g = _topk_grad(xl, tk)
        vx = mu * vx - lr * g
        x = _repair(x + vx, n)
        c = _eval(x)
        if c < best_c:
            best_c, best_x = c, x.copy()
    return best_x, best_c

def _fourier_eval(coeffs, n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, coeffs[0])
    K = (len(coeffs) - 1) // 2
    for k in range(1, K + 1):
        a = coeffs[2 * k - 1]; b = coeffs[2 * k]
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return np.clip(h, 0.0, 1.0)

def _fo_obj(coeffs, n):
    h = _repair(_fourier_eval(coeffs, n), n)
    return _eval(h)

def _nm_fourier(h0, n, K, maxiter, deadline):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    coeffs = np.zeros(2 * K + 1)
    coeffs[0] = np.mean(h0)
    for k in range(1, K + 1):
        coeffs[2 * k - 1] = np.mean(h0 * np.cos(np.pi * k * x))
        coeffs[2 * k] = np.mean(h0 * np.sin(np.pi * k * x))
    try:
        res = minimize(lambda c: _fo_obj(c, n), coeffs, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-7, 'fatol': 1e-10, 'adaptive': True})
        if time.time() > deadline:
            return h0.copy(), _eval(h0)
        h_new = _repair(_fourier_eval(res.x, n), n)
        return h_new, _eval(h_new)
    except Exception:
        return h0.copy(), _eval(h0)

def _resample(h, n_new):
    xo = np.linspace(0.0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s - s.max()))) + 1e-30) / alpha

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            g += w[k] * _subgrad(x, k)
    return g

def _slq_obj(x, n, alpha):
    return _smooth_obj(x, n, alpha)

def _slq_jac(x, n, alpha):
    return _smooth_grad(x, n, alpha)

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_slq_obj, h, jac=_slq_jac, args=(n, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cyclic_cd(h, deadline, n_pass=4):
    n = len(h)
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    for _ in range(n_pass):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for d in [0.002, -0.002, 0.005, -0.005, 0.001, -0.001]:
                x2 = x.copy(); x2[i] = min(1.0, max(0.0, x2[i] + d))
                x2 = _repair(x2, n)
                c2 = _eval(x2)
                if c2 < c_cur - 1e-13:
                    x, c_cur = x2, c2
                    if c2 < best_c:
                        best_c, best_x = c2, x.copy()
                    break
    return best_x, best_c

def _binary_opt(h, n, rng, n_iter, deadline):
    x = (h > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    T = 0.004
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / T):
            c_cur = c_new
            if c_new < best_c:
                best_c, best_x = c_new, x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
        T *= 0.9999
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_c5 = _eval(best_h)
    best_n = n0

    dl_a = t0 + 130

        h_n, c_n = _nm_fourier(best_h, best_n, K, 3000, dl_a)

    dl_b = t0 + 420
    params = [(0.012, 0.001, 0.85, 3, 5000),
              (0.008, 0.0005, 0.9, 2, 5000),

              (0.003, 0.0002, 0.85, 3, 5000)]
    for (l0, l1, mu, tk, ni) in params:
        if time.time() > dl_b:
            break
        h_n, c_n = _nesterov(best_h, ni, dl_b, l0, l1, mu, tk)
        if c_n < best_c5:
            best_c5, best_h = c_n, h_n.copy()

    dl_c = t0 + 650
    for n_try in [best_n - 10, best_n + 10, best_n + 20, 280, 340, 360, 380]:
        if time.time() > dl_c:
            break
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 60 or n_try > 700:
            continue
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_n, c_n = _nesterov(h_r, 3000, dl_c, 0.01, 0.0005, 0.85, 3)
        if c_n < best_c5:
            best_c5, best_h, best_n = c_n, h_n.copy(), n_try

    dl_d = t0 + 880
    for alpha in [200000, 1000000, 5000000]:
        if time.time() > dl_d:
            break
        h_s = _slsqp(best_h, best_n, alpha, 1500)
        c_s = _eval(h_s)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s.copy()

    dl_e = t0 + 1020
    h_n, c_n = _cyclic_cd(best_h, dl_e, n_pass=3)
    if c_n < best_c5:
        best_c5, best_h = c_n, h_n.copy()

    dl_f = t0 + 1080
    h_b, c_b = _binary_opt(best_h, best_n, rng, 200000, dl_f)
    if c_b < best_c5:
        best_c5, best_h = c_b, h_b.copy()
    if time.time() < dl_f:
        h_n2, c_n2 = _nesterov(best_h, 3000, dl_f, 0.003, 0.0001, 0.85, 2)
        if c_n2 < best_c5:
            best_c5, best_h = c_n2, h_n2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
