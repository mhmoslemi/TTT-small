import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(h, alpha):
    c = _correlation(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _correlation(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smap(h, alpha, maxiter, ftol, gtol=1e-12):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return np.array([x.sum() - 0.5 * n])
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_grad(h, top_k):
    n = len(h)
    c = _correlation(h)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _nesterov_subgrad(h, n_iter, step0, top_k, momentum=0.9):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    y = h.copy()
    v = np.zeros(n)
    step = step0
    stuck = 0
    for it in range(n_iter):
        # Compute gradient at y
        g = _topk_grad(y, top_k)
        # Update momentum
        v = momentum * v + g
        # Step

        y_new = _proj(y_new, 0.5 * n)
        c_new = _eval(y_new)
        if c_new < best_c:

            y = _proj(y, 0.5 * n)
            best = y_new.copy()
            best_c = c_new
            step = min(step * 1.05, 0.3)
            stuck = 0

                best_c = _eval(best)
                v = np.zeros(n)
                y = best.copy()
                step = step0
                stuck = 0
    return best, best_c

        F[0] = 0
        mag = np.abs(F)
        # Target: flatten the spectrum
        target = np.mean(mag[1:])
        # Smooth transition

        h_new = 0.5 + np.fft.ifft(F_new).real
        h_new = np.clip(h_new, 0, 1)
        h = _repair(h_new, n)
    return h

def _pairwise_refine(h, n_tries, rng, top_k=5):
    """Refine by trying pairwise transfers that reduce the max correlation"""
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    
    # Precompute which lags are worst
    corr = _correlation(h)
    worst_lags = np.argsort(corr)[-top_k:]
    
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 2e-4 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i] = ni
        h[j] = nj
        cn = _eval(h)
        if cn < c_cur - 1e-15:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
                # Update worst lags periodically
        else:
            h[i] = ni - d
            h[j] = nj + d
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smap(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return np.array([x.sum() - 0.5 * n])
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_grad(h, top_k):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _nesterov_subgrad(h, n_iter, step0, top_k):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    y = h.copy(); v = np.zeros(n)
    step = step0; stuck = 0
    for it in range(n_iter):
        g = _topk_grad(y, top_k)
        v = 0.85 * v + g
        y_new = _proj(best - step * v, 0.5 * n)
        c_new = _eval(y_new)
        if c_new < best_c - 1e-16:
            y = y_new + 0.85 * (y_new - best)
            y = _proj(y, 0.5 * n)
            best = y_new.copy(); best_c = c_new
            step = min(step * 1.03, 0.25); stuck = 0
        else:
            step *= 0.75; stuck += 1
            if stuck > 60:
                rng_r = np.random.RandomState(it + 7)
                best = _repair(best + 0.008 * rng_r.randn(n), n)
                best_c = _eval(best)
                v = np.zeros(n); y = best.copy()
                step = step0; stuck = 0
    return best, best_c

def _spectral_refine(h, n_iter):
    n = len(h)
    for _ in range(n_iter):
        f = h - 0.5
        F = np.fft.rfft(f, n=2 * n - 1)
        F[0] = 0
        mag = np.abs(F)
        target = np.mean(mag[1:])
        blend = 0.15
        new_mag = blend * target + (1 - blend) * mag
        F_new = new_mag * np.exp(1j * np.angle(F))
        F_new[0] = 0
        h_new = 0.5 + np.fft.irfft(F_new, n=2 * n - 1)[:n]
        h = _repair(np.clip(h_new, 0, 1), n)
    return h

def _pairwise_refine(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:

        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c
