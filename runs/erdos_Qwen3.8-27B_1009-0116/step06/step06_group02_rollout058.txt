def _slsqp_unconstrained(h, alpha, maxiter, ftol):
    n = len(h)
    target = n * 0.5
    # Parameterize: optimize h[0:n-1], h[n-1] = target - sum(h[0:n-1])
    x0 = h[:n-1].copy()
    
    def extend(x):
        x_full = np.empty(n)
        x_full[:n-1] = x
        x_full[n-1] = target - x.sum()
        return x_full
    
    def obj_and_grad(x):
        x_full = extend(x)
        # Ensure feasibility (for objective computation)
        x_full = np.clip(x_full, 0, 1)
        c = _corr(x_full)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        
        # Objective
        val = m + np.log(float(np.sum(np.exp(s)))) / alpha
        
        # Gradient w.r.t. x_full
        dx = 2.0 / n
        gf = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            wk = w[k] * dx
            sh = k - (n - 1)
            if sh == 0:
                gf += wk * (1.0 - 2.0 * x_full)
            elif sh > 0:
                gf[:n-sh] += wk * (1.0 - x_full[sh:]); gf[sh:] -= wk * x_full[:n-sh]
            else:
                sp = -sh; gf[sp:] += wk * (1.0 - x_full[:n-sp]); gf[:n-sp] -= wk * x_full[sp:]
        
        # Chain rule: dx_full/dx[j] = e_j for j < n-1, and dx_full[n-1]/dx[j] = -1
        g = gf[:n-1] - gf[n-1]
        return val, g
    
    bnds = [(0.0, 1.0)] * (n-1)
    try:
        res = minimize(obj_and_grad, x0, jac=True,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        h_new = extend(res.x)
        return _repair(h_new, n)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _soft_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

    c = _corr(x)

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _subgrad(h, top_m, max_iter, t_end):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    step_base = 0.08
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx:
            sh = k - (n - 1)
            if sh == 0:
                g += (1.0 - 2.0 * h)
            elif sh > 0:
                g[:n-sh] += (1.0 - h[sh:]); g[sh:] -= h[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n-sp]); g[:n-sp] -= h[sp:]
        g /= float(top_m)
        step = step_base * (2.0 / n)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:

            ct = _eval(ht)

                step_base = min(step_base * 1.03, 2.0)
                break

            if step_base < 1e-8:
                break
    return h, cur
