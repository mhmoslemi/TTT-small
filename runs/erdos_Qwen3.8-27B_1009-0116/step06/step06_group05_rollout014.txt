    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

def _subgrad_step(h, step, n_worst=5):

    nc = len(h)

    c = _corr_vec(h) * (2.0 / len(h))

def _power_obj_grad(h, p):

    dx = 2.0 / nc
    c = _corr_vec(h) * dx
    c_p1 = c ** (p-1)  # p/1 * c^(p-1) / c_sum^(p-1)
    s = np.sum(c**p)

    g = np.zeros(nc)
    for k in range(2*nc - 1):
        ck = c[k]

        else:

            g[sp:] += w * (1.0 - h[:nc-sp])

    return g

def _subgrad(h, step, nw):
    nc = len(h)
    dx = 2.0 / nc
    c = _corr_vec(h)
    idx = np.argsort(c)[-nw:][::-1]

        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:nc-sp])
            g[:nc-sp] -= dx * h[sp:]
    g /= nw
    return _repair(h - step * g, nc)

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
