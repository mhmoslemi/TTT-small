            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter=1500, ftol=1e-17):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_softmax_obj, h, jac=_softmax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_adaptive(h, a_start, a_end, steps, step0):
    n = len(h)
    f = _softmax_obj(h, a_start)
    v = np.zeros(n)
    mu = 0.83
    step = step0
    ratio = (a_end / a_start) ** (1.0 / max(steps, 1))
    for i in range(steps):
        a = a_start * (ratio ** i)
        g = _softmax_grad(h, a)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _softmax_obj(h_new, a)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 6.0)
        else:
            step *= 0.45
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _softmax_obj(h, a)
    return _repair(h, n)

def _perturb(h, rng, scale, mode):
    n = len(h)
    h_new = h.copy()
    if mode == 0:
        h_new += scale * rng.randn(n)
    elif mode == 1:
        bsz = max(4, n // 16)
        for _ in range(6):
            st = rng.randint(0, n - bsz)
            h_new[st:st + bsz] += scale * rng.randn(bsz)
    else:
        bsz = max(4, n // 10)
        st = rng.randint(0, n - bsz)
        h_new[st:st + bsz] = 1.0 - h_new[st:st + bsz]
    return _repair(h_new, n)

def _local_search(h, n_tries, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 4e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Coarse SLSQP at n/4
    n_low = max(40, n0 // 4)
    f = n0 // n_low
    h_low = h[:n_low * f].reshape(n_low, f).mean(axis=1)
    h_low = _repair(h_low, n_low)
    for alpha in [500, 5000, 50000]:
        if _el() > 40:
            break
        h_low = _slsqp(h_low, alpha, 800, 1e-16)
    h_up = _repair(np.repeat(h_low, f)[:n0], n0)
    c5u = _eval(h_up)
    if c5u < best_c5:
        best_c5 = c5u
        best_h = h_up.copy()
        h = h_up
    else:
        h = best_h.copy()

    # Phase 2: SLSQP at full res with increasing alpha
    for alpha in [10000, 100000, 1000000, 5000000]:
        if _el() > 250:
            break
        h_c = _slsqp(h, alpha, 2000, 1e-17)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_c.copy()
            best_n = len(h_c)
        h = h_c

    # Phase 3: Adaptive Nesterov with geometric alpha
    if _el() < budget_s - 350:
        h_nr = _nesterov_adaptive(best_h, 500, 2000000, 600, 0.004)
        c5n = _eval(h_nr)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_nr.copy()
            best_n = len(h_nr)

    # Phase 4: Diverse restarts
    for trial in range(10):
        if _el() > budget_s - 250:
            break
        mode = trial % 3
        scale = [0.08, 0.04, 0.02][trial % 3]
        h_t = _perturb(best_h, rng, scale, mode)
        h_t = _slsqp(h_t, 30000, 1200, 1e-16)
        h_t = _slsqp(h_t, 500000, 1200, 1e-17)
        c5t = _eval(h_t)
        if c5t < best_c5:
            best_c5 = c5t
            best_h = h_t.copy()
            best_n = len(h_t)

    # Phase 5: Second Nesterov pass on best
    if _el() < budget_s - 150:
        h_nr2 = _nesterov_adaptive(best_h, 20000, 3000000, 400, 0.002)
        c5n2 = _eval(h_nr2)
        if c5n2 < best_c5:
            best_c5 = c5n2
            best_h = h_nr2.copy()
            best_n = len(h_nr2)

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 80:
        h_f = _slsqp(best_h, 8000000, 2500, 1e-18)
        c5f = _eval(h_f)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = h_f.copy()
            best_n = len(h_f)

    # Phase 7: Fine local search
    if _el() < budget_s - 5:
        t_rem = budget_s - 3 - _el()
        n_tries = min(int(t_rem / 0.003), 150000)
        best_h, best_c5 = _local_search(best_h, n_tries, rng, budget_s - 2)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
