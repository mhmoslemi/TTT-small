from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Bisection projection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        clipped = np.clip(h - mid, 0, 1)
        if clipped.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _grad_smoothed(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(2*n-1):

            if q > 0:
                g[q:] -= wk*h[:n-q]
        else:
            p = -q
            g[p:] += wk*(1-h[:n-p])
            g[:n-p] -= wk*h[p:]

    h = h.copy()

    obj = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:

        # Backtracking line search
        for _ in range(30):

            obj_new = _eval(h_new)
            if obj_new < obj - 1e-12*lr*np.dot(g,g):
                h = h_new
                obj = obj_new

        else:

            h[best_j] = 1.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        if clipped.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_smoothed(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        q = k - (n - 1)
        wk = w[k] * dx
        if q >= 0:
            g[:n - q] += wk * (1.0 - h[q:n])
            if q > 0:
                g[q:] -= wk * h[:n - q]
        else:
            p = -q
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] -= wk * h[p:]
    return g
