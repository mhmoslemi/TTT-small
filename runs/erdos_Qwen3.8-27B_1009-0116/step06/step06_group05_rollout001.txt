    if _elapsed() < budget_s - 50:
        h8 = _slsqp(best_h, 10000000, 3000, 1e-20)
        c5 = _eval(h8)
        if c5 < best_c5 - 1e-14:
            best_c5, best_h = c5, h8.copy()

    # Phase 9: Fine coordinate descent
    if _elapsed() < budget_s - 10:
        dl = t_start + budget_s - 10
        h9, c5_9 = _coord_descent(best_h, dl, rng)
        if c5_9 < best_c5 - 1e-16:
            best_c5, best_h = c5_9, h9.copy()

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
