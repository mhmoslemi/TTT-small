So the weight for each correlation value c_k is:
w_k = (sum |c|^p)^(1/p - 1) * |c_k|^(p-1) * sign(c_k)

And the gradient is sum_k w_k * dc_k/dx_j

dc_k/dx_j: c_k = sum_{valid i} h[i] * (1 - h[i + sh]) where sh = (n-1) - k

For a given k with shift sh = (n-1) - k:

- dc_k/dx_j = sum over terms involving x_j

If j is in the first factor: dc/dx_j = sum of (1-h[i+sh]) where i=j, i in [0, n-1-sh]
If j is in the second factor: dc/dx_j = -sum of h[i] where i+sh=j, i in [0, n-1-sh], i.e., j in [sh, n-1]

So for shift sh >= 0:
- dc_k/dx_j += (1 - h[j+sh]) for j in [0, n-1-sh] (first factor)
- dc_k/dx_j -= h[j-sh] for j in [sh, n-1] (second factor)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _sum_con_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_con_jac(x):
    return np.ones(len(x))

def _pnorm_obj(x, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    sp = np.sum(np.abs(c) ** p)
    return float(sp ** (1.0 / p))

def _pnorm_grad(x, p):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    sp = np.sum(np.abs(c) ** p)
    norm = sp ** (1.0 / p)
    if norm < 1e-30:
        return np.zeros(n)
    w = (norm ** (1 - p)) * np.abs(c) ** (p - 1) * np.sign(c)
    return _compute_grad(x, w)

def _compute_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if abs(weights[k]) < 1e-15:
            continue
        sh = (n - 1) - k
        wk = weights[k] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * h[i1]
        else:
            sp_ = -sh
            i1 = np.arange(sp_, n)
            g[i1] += wk * (1.0 - h[i1 - sp_])
            g[i1 - sp_] -= wk * h[i1]
    return g

def _slsqp_pnorm(h, p, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = ({"type": "eq", "fun": _sum_con_fun, "jac": _sum_con_jac},)
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if time.time() > t_end:
            return _repair(h, n)
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _reweight_cool(h, n_iters, t_end, top_m=5, t0_temp=20.0):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        c = _corr(h)
        idx = np.argsort(c)[-top_m:]
        cmin_v = c[idx].min(); cmax_v = c[idx].max()
        temp = t0_temp * (0.97 ** it)
        w = np.zeros(2 * n - 1)
        if cmax_v - cmin_v > 1e-12:
            w[idx] = np.exp((c[idx] - cmin_v) / (temp * 0.005))
        else:
            w[idx] = 1.0
        w /= w.sum()
        g = _compute_grad(h, w)
        moved = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-14:
                h = ht; cur = ct; moved = True
                break
        if not moved:
            break
    return h, cur

def _binary_sa(h, max_iters, t_end, t_start=40.0):
    n = len(h)
    rng = np.random.RandomState(99)
    h = h.copy()
    h[h > 0.5] = 1.0; h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    best_h = h.copy(); best_c = cur
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(max_iters):
        if time.time() > t_end:
            break
        temp = t_start * (0.005 / max_iters) ** (it / max(1, max_iters))
        move = rng.randint(3)
        if move == 0:
            if len(ones) < 1 or len(zeros) < 1:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0; h[j] = 1.0
            new_c = _eval(h)
            accept = new_c < cur or (temp > 1e-8 and rng.rand() < np.exp(-(new_c - cur) / temp))
            if not accept:
                h[i] = 1.0; h[j] = 0.0
            else:
                cur = new_c
                if cur < best_c:
                    best_c = cur; best_h = h.copy()
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
        elif move == 1:
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                continue
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                continue
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            new_c = _eval(h)
            accept = new_c < cur or (temp > 1e-8 and rng.rand() < np.exp(-(new_c - cur) / temp))
            if not accept:

            if accept:
                cur = new_c
                if cur < best_c:
                    best_c = cur; best_h = h.copy()
    return _repair(best_h, n), best_c

def _resample(h, n_new):
    n = len(h)
    xo = np.linspace(0, 2.0, n, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _perturb(h, n, sigma, rng):
    hp = h.copy() + sigma * rng.randn(n)
    return _repair(hp, n)
