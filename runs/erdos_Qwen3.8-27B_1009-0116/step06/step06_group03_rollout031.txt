import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    em = np.exp(s)
    return m + np.log(em.sum()) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        wka = wk * alpha
        if sh >= 0:
            g[:n - sh] += wka * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wka * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wka * (1.0 - h[:n - sp])
            g[:n - sp] -= wka * h[sp:]
    return g * 2.0 / n

    n = len(h)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _smax(x, alpha) + 0.5 * mu * (x.sum() - target) ** 2
    def grad(x):
        return _sgrad(x, alpha) + mu * (x.sum() - target) / n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',

        return _repair(res.x, n)
    except Exception:

    target = n / 2.0

        return [np.ones(n)]

def _coord_descent(h, n_iter, rng):
    """Coordinate descent with 2-point exchanges."""
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try to move mass from i to j (or vice versa)
        for d in [0.001, 0.0001, 0.00001]:
            ni, nj = best[i] + d, best[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best[i], best[j] = ni, nj
                c = _eval(best)
                if c < best_c:
                    best_c = c
                    break
                best[i], best[j] = ni - d, nj + d
            # Try reverse
            ni, nj = best[i] - d, best[j] + d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best[i], best[j] = ni, nj
                c = _eval(best)
                if c < best_c:
                    best_c = c
                    break
                best[i], best[j] = ni + d, nj - d
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    return val * 2.0 / len(h)

def _sgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        wkd = wk * dx
        if sh >= 0:
            g[:n - sh] += wkd * (1.0 - h[sh:])
            g[sh:] -= wkd * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wkd * (1.0 - h[:n - sp])
            g[:n - sp] -= wkd * h[sp:]
    return g

def _lbfgs(h, alpha, mu, maxiter):
    n = len(h)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _smax(x, alpha) + 0.5 * mu * (x.sum() - target) ** 2
    def grad(x):
        return _sgrad(x, alpha) + mu * (x.sum() - target)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return [x.sum() - target]
    def cj(x):
        return [np.ones(1, dtype=float)]
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, n_iter, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in (1e-4, 5e-5, 1e-5, 5e-6):
            improved = False
            ni, nj = best[i] + d, best[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                best[i], best[j] = ni, nj
                c = _eval(best)
                if c < best_c - 1e-18:
                    best_c = c
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
            if not improved:
                ni, nj = best[i] - d, best[j] + d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    best[i], best[j] = ni, nj
                    c = _eval(best)
                    if c < best_c - 1e-18:
                        best_c = c
                    else:
                        best[i], best[j] = ni + d, nj - d
    return best, best_c
