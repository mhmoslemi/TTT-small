                g += w[k] * _grad_lag(y, k, dx)
            x_new = np.clip(y - lr * g, 0.0, 1.0)
            x_new = _repair(x_new, n)
            c5n = _eval(x_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_x = x_new.copy()
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y = np.clip(y, 0.0, 1.0)
            x = x_new
            t = t_new
    return best_x, best_c5

def _subgrad_boost(h, n, niter, t0, momentum_decay=True):
    """Nesterov-accelerated subgradient on worst lag only."""
    dx = 2.0 / n
    x = h.copy()
    v = np.zeros(n)
    t = 1.0
    best_c5 = _eval(x)
    best_x = x.copy()
    for it in range(niter):
        if time.time() - t0 > budget_s - 20:
            break
        c = np.correlate(x, 1.0 - x, mode='full')
        worst_k = int(np.argmax(c))
        g = _grad_lag(x, worst_k, dx)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        basis = x + ((t - 1.0) / t_new) * v
        basis = np.clip(basis, 0.0, 1.0)
        gb = _grad_lag(basis, worst_k, dx)
        lr = 1.0 / (t_new * (norm_g + 1e-8))
        x_new = _repair(basis - lr * gb, n)
        c5n = _eval(x_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_x = x_new.copy()
        v = (x_new - x) * (momentum_decay and 0.8 or 1.0)
        x = x_new
        t = t_new
    return best_x, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            g += w[k] * _grad_lag(x, k, dx)
        return g
    def sum_fun(x):
        return x.sum() - n * 0.5
    def sum_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=grad, args=(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descent(h, n, rng, scales, iters_per_scale, t0):
    c5c = _eval(h)
    for scale in scales:
        for _ in range(iters_per_scale):
            if time.time() - t0 > budget_s - 15:
                return h, c5c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i] -= d
                h[j] += d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: FISTA with adaptive alpha at parent resolution
    if time.time() - t0 < budget_s - 200:
        h1, c5 = _fista_anneal(best_h, best_n, [500, 2000, 8000, 30000], 400, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h1

    # Phase 2: Subgradient boosting on worst lag
    if time.time() - t0 < budget_s - 180:
        h2, c5 = _subgrad_boost(best_h, best_n, 3000, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 3: SLSQP with high alpha (polish)
    if time.time() - t0 < budget_s - 150:
        for alpha in [50000, 200000]:
            if time.time() - t0 > budget_s - 60:
                break
            h3 = _slsqp(best_h, alpha, 3000, 1e-18)
            c5 = _eval(h3)
            if c5 < best_c5:
                best_c5, best_h = c5, h3

    # Phase 4: Subgradient boost again
    if time.time() - t0 < budget_s - 120:
        h4, c5 = _subgrad_boost(best_h, best_n, 5000, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h4

    # Phase 5: FISTA with very high alpha
    if time.time() - t0 < budget_s - 100:
        h5, c5 = _fista_anneal(best_h, best_n, [50000, 150000, 400000], 300, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 6: Multi-resolution exploration
    if time.time() - t0 < budget_s - 150:
        for n_try in [310, 400, 500, 790, 930]:
            if time.time() - t0 > budget_s - 60:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r, c5r = _fista_anneal(h_r, n_try, [2000, 10000, 50000], 250, t0)
            if c5r < best_c5:
                best_c5, best_h, best_n = c5r, h_r, n_try
                h_r, c5r = _subgrad_boost(h_r, n_try, 2000, t0)
                if c5r < best_c5:
                    best_c5, best_h = c5r, h_r

    # Phase 7: Back to best resolution, final FISTA + SLSQP
    if time.time() - t0 < budget_s - 80:
        if best_n != n0:
            best_h = _repair(_resample(best_h, n0), n0)
            best_c5 = _eval(best_h)
            best_n = n0
        h7, c5 = _fista_anneal(best_h, best_n, [100000, 500000], 200, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h7
        h7 = _slsqp(best_h, 300000, 2000, 1e-19)
        c5 = _eval(h7)
        if c5 < best_c5:
            best_c5, best_h = c5, h7

    # Phase 8: Structured sinusoidal perturbations + re-optimize
    if time.time() - t0 < budget_s - 80:
        for freq in [1, 2, 3, 5, 7]:
            if time.time() - t0 > budget_s - 30:
                break
            for amp in [0.03, 0.01]:
                if time.time() - t0 > budget_s - 30:
                    break
                x = np.linspace(0, 2.0, best_n, endpoint=False)
                pert = amp * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
                h_p = _repair(best_h + pert, best_n)
                h_p, c5 = _subgrad_boost(h_p, best_n, 1500, t0)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_p

    # Phase 9: Fine coordinate descent
    if time.time() - t0 < budget_s - 25:
        h9 = best_h.copy()
        h9, c5 = _coord_descent(h9, best_n, rng,
                                [0.005, 0.001, 0.0003, 1e-4], 3000, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h9

    # Phase 10: Final subgradient polish
    if time.time() - t0 < budget_s - 10:
        h10, c5 = _subgrad_boost(best_h, best_n, 3000, t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h10

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
