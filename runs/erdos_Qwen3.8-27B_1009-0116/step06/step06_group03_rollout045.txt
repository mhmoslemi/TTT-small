            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return np.array([x.sum() - 0.5 * n])
    def _cj(x):
        return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _top_grad(h, top_m):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    dx = 2.0 / n
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return g

def _subgrad(h, n_iter, step0, top_m=5):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stuck = 0
    for it in range(n_iter):
        g = _top_grad(best, top_m)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new
                best_c = c_new
                step = min(step * 1.15, 0.6)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.65
            stuck += 1
            if stuck > 80:
                best = _repair(best + 0.008 * np.random.RandomState(it).randn(n), n)
                best_c = _eval(best)
                step = step0
                stuck = 0
                v = np.zeros(n)
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        move_type = rng.randint(3)
        if move_type == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c_cur:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        elif move_type == 1:
            i = rng.randint(n - 3)
            d = 1.5e-5 * rng.randn(4)
            cand = h[i:i+4].copy() + d
            if np.any(cand < 0) or np.any(cand > 1):
                continue
            buf = h[i:i+4].copy()
            h[i:i+4] = cand
            cn = _eval(h)
            if cn < c_cur:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = h.copy()
            else:
                h[i:i+4] = buf
        else:
            i1 = rng.randint(n // 2)
            i2 = n // 2 + rng.randint(n // 2)
            d = 3e-5 * rng.randn()
            h[i1] += d
            h[i2] -= d
            if h[i1] < 0 or h[i1] > 1 or h[i2] < 0 or h[i2] > 1:
                h[i1] -= d
                h[i2] += d
                continue
            cn = _eval(h)
            if cn < c_cur:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = h.copy()
            else:
                h[i1] -= d
                h[i2] += d
    return best, best_c

def _block_refine(h, n_blocks, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for _ in range(n_blocks):
        bsize = rng.randint(3, 9)
        start = rng.randint(0, max(1, n - bsize))
        end = min(start + bsize, n)
        orig = best[start:end].copy()
        for trial in range(30):
            pert = 0.01 * rng.randn(bsize)
            cand = orig + pert
            if np.any(cand < 0) or np.any(cand > 1):
                continue
            test = best.copy()
            test[start:end] = cand
            # repair just this block's contribution to sum
            diff = best.sum() - test.sum()
            if abs(diff) < 1e-10:
                pass
            else:
                # redistribute
                test = _repair(test, n)
            cn = _eval(test)
            if cn < best_c:
                best = test
                best_c = cn
                orig = best[start:end].copy()
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c = _eval(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP with increasing alpha
    for alpha in [300, 1000, 5000, 20000, 80000]:
        if _el() > 120:
            break
        hc = _slsqp(best_h, alpha, 1500, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 2: Subgradient
    if _el() < budget_s - 350:
        n_it = min(5000, int((budget_s - 350 - _el()) / 0.02))
        best_h, best_c = _subgrad(best_h, n_it, 0.02, top_m=5)
        best_n = len(best_h)

    # Phase 3: Another SLSQP pass
    for alpha in [30000, 150000, 600000]:
        if _el() > budget_s - 280:
            break
        hc = _slsqp(best_h, alpha, 2000, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 4: Fine search (extended)
    if _el() < budget_s - 200:
        n_tr = min(int((budget_s - 200 - _el()) / 0.003), 60000)
        best_h, best_c = _fine_search(best_h, n_tr, rng)

    # Phase 5: Block refinement
    if _el() < budget_s - 120:
        n_blk = min(int((budget_s - 120 - _el()) / 0.015), 3000)
        best_h, best_c = _block_refine(best_h, n_blk, rng)

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 40:
        hc = _slsqp(best_h, 500000, 2000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 7: Micro fine-tuning
    if _el() < budget_s - 15:
        n_tr = min(int((budget_s - 15 - _el()) / 0.003), 30000)
        best_h, best_c = _fine_search(best_h, n_tr, rng)

    # Phase 8: Try multi-restart from best
    if _el() < budget_s - 10:
        for r in range(3):
            if _el() > budget_s - 8:
                break
            pert = 0.003 * rng.randn(best_n)
            hp = _repair(best_h + pert, best_n)
            cp = _eval(hp)
            n_tr = min(int(max(0, budget_s - 8 - _el()) / 0.004), 10000)
            if n_tr > 500:
                hp, cp = _fine_search(hp, n_tr, rng)
            if cp < best_c:
                best_c = cp
                best_h = hp

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
