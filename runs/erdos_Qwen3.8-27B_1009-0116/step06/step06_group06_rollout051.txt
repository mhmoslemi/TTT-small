        t_end = t0 + rem() - 45
        h_new = _multi_target_fista(best_h, 300, t_end, topk=20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Upsample to 2x and optimize
    if rem() > 250:
        n2 = n0 * 2
        h2 = _upsample(best_h, 2)
        h2 = _repair(h2, n2)
        for alpha in [2000, 10000, 40000, 120000]:
            if rem() < 220:
                break
            t_end = t0 + rem() - 210
            h2 = _fista(h2, alpha, 500, t_end, lr_init=0.006)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
                best_n = n2
        if rem() > 200:
            t_end = t0 + rem() - 195
            h2 = _multi_target_fista(best_h, 300, t_end, topk=12)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
                best_n = n2

    # Phase 4: Try perturbations and re-optimize
    if rem() > 100:
        for restart in range(6):
            if rem() < 85:
                break
            scale = 0.02 + 0.01*restart
            h_try = best_h + scale*rng.randn(best_n)
            h_try = _proj(h_try, best_n*0.5)
            t_end = t0 + rem() - 80
            h_opt = _fista(h_try, 50000, 200, t_end, lr_init=0.005)
            c5_opt = _eval(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Block swap exploration
    if rem() > 60:
        for _swap in range(8):
            if rem() < 50:
                break
            h_try = best_h.copy()
            bl = rng.randint(best_n//12, best_n//5)
            st = rng.randint(0, best_n-bl)
            h_try[st:st+bl] = 1.0 - h_try[st:st+bl]
            h_try = _proj(h_try, best_n*0.5)
            t_end = t0 + rem() - 48
            h_opt = _fista(h_try, 30000, 150, t_end, lr_init=0.004)
            c5_opt = _eval(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Final FISTA polish with very high alpha
    if rem() > 40:
        t_end = t0 + rem() - 35
        h_new = _fista(best_h, 500000, 400, t_end, lr_init=0.003)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 7: Coordinate descent micro-polish
    if rem() > 20:
        h_cur = best_h.copy()
        nc = len(h_cur)
        c5_cur = _eval(h_cur)
        budget_iters = int(min(rem()*60, 150000))
        for _it in range(budget_iters):
            if rem() < 8:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni = h_cur[i] + d
            nj = h_cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_cur[i], h_cur[j] = ni, nj
            c5n = _eval(h_cur)
            if c5n < c5_cur:
                c5_cur = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
