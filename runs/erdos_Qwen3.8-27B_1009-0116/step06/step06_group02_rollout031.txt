```python
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    p = 1
    while p < n:
        p *= 2
    return p

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _eval_full(h, N):
    n = len(h)
    h_pad = np.zeros(N)
    h_pad[:n] = h
    H = np.fft.fft(h_pad)
    auto_circ = np.fft.ifft(H*H).real
    auto_full = np.concatenate([auto_circ[N-(n-1):N], auto_circ[0:n]])
    ifs = np.arange(-(n-1), n)
    i_s = np.maximum(0, -ifs)
    i_e = np.minimum(n-1, n-1-ifs)
    hs_ext = np.concatenate([[0.0], h.cumsum()])
    valid = hs_ext[i_e+1] - hs_ext[i_s]
    return (valid - auto_full) * (2.0/n)

def _grad_fft(h, w, N):
    n = len(h)
    h_pad = np.zeros(N)
    h_pad[:n] = h
    H = np.fft.fft(h_pad)
    w_pad = np.zeros(N)
    w_pad[N-(n-1):N] = w[:n-1]
    w_pad[0:n] = w[n-1:]
    w_rev = w_pad.copy()
    w_rev[1:] = w_pad[:0:-1]
    A = np.fft.ifft(np.fft.fft(w_rev) * H).real[:n]
    B = np.fft.ifft(np.fft.fft(w_pad) * H).real[:n]
    return (1.0 - A - B) * (2.0/n)

def _project(h, S):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo+hi)*0.5
        s = np.clip(h-mid, 0, 1).sum()
        if s > S: lo = mid
        else: hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)
```

```python
            hr = _project(_resample(best_h, n_try), n_try * 0.5)
            hr, c5 = _multistage_gd(hr, t0 + rem() - 50)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hr.copy()

    # Phase 3: Binary optimization
    if rem() > 80:
        hb = _project((best_h > 0.5).astype(float), len(best_h) * 0.5)
        hb, c5 = _binary_swap(hb, t0 + rem() - 20, 500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hb.copy()
        # Also try binary at a nearby resolution
        if rem() > 60:
            nb = len(best_h) - 20 if len(best_h) > 100 else len(best_h) + 20
            hb2 = _project(_resample(best_h, nb), nb * 0.5)
            hb2 = _project((hb2 > 0.5).astype(float), nb * 0.5)
            hb2, c5 = _binary_swap(hb2, t0 + rem() - 20, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hb2.copy()

    # Phase 4: High-alpha LSE polish on best solution
    if rem() > 80:
        h4, c5 = _ppt_gd(best_h, 100000, 6000, t0 + rem() - 30, lr0=0.005, mom=0.9)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h4.copy()
        if rem() > 40:
            h4b, c5 = _reweight_gd(best_h, 1, 5000, t0 + rem() - 15, lr0=0.003, mom=0.8)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h4b.copy()

    # Phase 5: Pair coordinate descent for fine-tuning
    if rem() > 40:
        h5, c5 = _pair_cd(best_h.copy(), 8000, t0 + rem() - 10, [0.005, 0.001, 0.0003])
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5

    # Phase 6: Final LSE at very high alpha
    if rem() > 30:
        h6, c5 = _ppt_gd(best_h, 500000, 4000, t0 + rem() - 10, lr0=0.002, mom=0.92)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h6.copy()

    # Final projection and evaluation
    best_h = _project(np.clip(best_h, 0.0, 1.0), len(best_h) * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
```
