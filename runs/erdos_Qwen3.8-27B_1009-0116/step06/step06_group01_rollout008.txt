import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _overlap_grad_k(h, k, n):
    """Gradient of the overlap at correlation index k with respect to h."""
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0 and sh < n:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    elif sh < 0:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:]
    return g

def _active_descent(h, n, max_rounds, t0, time_limit):

    dx = 2.0 / n
    c5 = _eval(h)
    active = [int(np.argmax(np.correlate(h, 1.0 - h, mode='full')))]
    
    for r in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        
        over = np.correlate(h, 1.0 - h, mode='full')
        # Update active set: top 3-5 lags
        top_k = min(5, 2*n-1)
        active = list(np.argsort(over)[-top_k:])
        
        # Sum gradients over active lags
        g = np.zeros(n)
        for k in active:
            g += _overlap_grad_k(h, k, n)
        g *= dx

        if gn < 1e-15:
            break
        g /= gn

                best_h = h_try.copy()
                if step < 0.001:

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_k(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0 and sh < n:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    elif 0 < -sh < n:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:]
    return g

def _active_descent(h, n, max_rounds, t0, deadline, top_m=5):
    dx = 2.0 / n
    c5 = _eval(h)
    no_improve = 0
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        over = np.correlate(h, 1.0 - h, mode='full')
        active = np.argsort(over)[-top_m:]
        g = np.zeros(n)
        for k in active:
            g += _grad_k(h, int(k), n)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        step = 0.06
        improved = False
        for _ls in range(35):
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < c5 - 1e-14:
                h, c5 = ht, ct
                improved = True
                if step < 0.0005:
                    break
                step = min(step * 1.25, 0.5)
            else:
                step *= 0.55
        if not improved:
            k_w = int(over.argmax())
            g1 = _grad_k(h, k_w, n) * dx
            g1 /= (np.linalg.norm(g1) + 1e-30)
            step = 0.02
            for _ls in range(25):
                ht = _repair(h - step * g1, n)
                ct = _eval(ht)
                if ct < c5 - 1e-14:
                    h, c5 = ht, ct
                    improved = True
                else:
                    step *= 0.5
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
    return h, c5

def _pair_balance(h, n, t0, deadline):
    """Balance top-2 lags simultaneously."""
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(300):
        if time.time() > deadline:
            break
        over = np.correlate(h, 1.0 - h, mode='full')
        top2 = np.argsort(over)[-2:]
        k1, k2 = int(top2[0]), int(top2[1])
        g1 = _grad_k(h, k1, n) * dx
        g2 = _grad_k(h, k2, n) * dx
        g = g1 + g2
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        step = 0.04
        for _ls in range(25):
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < c5 - 1e-14:
                h, c5 = ht, ct
                step = min(step * 1.2, 0.4)
            else:
                step *= 0.55
    return h, c5

def _coord_descent(h, n, rng, t0, deadline, scales=None):
    if scales is None:
        scales = [0.008, 0.003, 0.001, 0.0003, 0.00008]
    c5 = _eval(h)
    for sc in scales:
        if time.time() > deadline:
            break
        n_iter = 6000
        stuck = 0
        for _ in range(n_iter):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5 - 1e-15:
                c5 = cn
                stuck = 0
            else:
                h[i], h[j] = ni - d, nj + d
                stuck += 1
                if stuck > 4000:
                    break
    return h, c5

def _slsqp_opt(h, n, alpha, maxiter, t0, deadline):
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

    def gfun(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-11:
                continue
            g += w[k] * _grad_k(x, k, n)
        return g

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=gfun, method='SLSQP', bounds=bnds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        out = _repair(res.x, n)
        if _eval(out) <= _eval(h) + 1e-12:
            return out
    except Exception:
        pass
    return _repair(h, n)

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
