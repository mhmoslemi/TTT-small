                hr, c5r = _pg_solve(hr, alpha, 3000, t0 + rem() - 50)
            if c5r < best_c5:
                best_c5 = c5r; best_h = hr.copy(); best_n = n_try
            if rem() > 60:
                h_bin2 = (hr > 0.5).astype(float)
                h_bin2 = _repair(h_bin2, n_try)
                h_bin2, c5b2 = _binary_ls(h_bin2, n_try, t0 + rem() - 30, max_rounds=200)
                if c5b2 < best_c5:
                    best_c5 = c5b2; best_h = h_bin2.copy(); best_n = n_try

    # Phase 5: Perturbation refinement
    if rem() > 50:
        rng = np.random.RandomState(seed)
        h_try, c5 = _perturb_refine(best_h.copy(), best_n, rng, t0 + rem() - 20, n_iter=5000)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_try.copy()

    # Phase 6: Final PG polish with very high alpha
    if rem() > 30:
        h_try, c5 = _pg_solve(best_h, 5000000, 5000, t0 + rem() - 10)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_try.copy()

    # Phase 7: Final binary touch-up
    if rem() > 15:
        h_fin = (best_h > 0.5).astype(float)
        h_fin = _repair(h_fin, best_n)
        c5f = _eval(h_fin)
        if c5f < best_c5:
            best_c5 = c5f; best_h = h_fin.copy()
        else:
            best_c5 = _eval(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
