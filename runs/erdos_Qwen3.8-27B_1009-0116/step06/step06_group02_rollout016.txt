def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    # Proportional scaling with clipping
    scale = target / s
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_weighted(h, w, n):
    """Gradient of sum_k w[k] * c[k] where c = correlate(h, 1-h)"""
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        if sh >= 0:
            # c_k = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _adam_step(h, g, m, v, t, lr, beta1, beta2, eps, n):
    m[t] = beta1 * m[t-1] + (1 - beta1) * g
    v[t] = beta2 * v[t-1] + (1 - beta2) * g**2
    mhat = m[t] / (1 - beta1**t)
    vhat = v[t] / (1 - beta2**t)
    step = lr * mhat / (np.sqrt(vhat) + eps)
    h_new = h - step
    return _repair(h_new, n), m[t], v[t]

    v = [None] * (max_iter + 1)
    m[0] = np.zeros(n)
    v[0] = np.zeros(n)

        threshold = np.percentile(c, 70)  # top 30%
        w = np.where(c > threshold, np.exp(3.0 * (c - threshold) / (c.max() - threshold + 1e-15)), 0.0)
        if w.max() > 0:

        # Decay learning rate
        lr_t = lr / (1.0 + 0.01 * t)
        h_new = _repair(h - lr_t * g, n)
        c_new = _eval(h_new)
        if c_new < cur - 1e-15:

                c_try = _eval(h_try)
                if c_try < cur - 1e-15:
                    h = h_try
                    cur = c_try
                    break
    
    return h, cur

def _repair(h, n):

def _eval(h):

def _grad_weighted(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            valid1 = slice(0, n - d)
            g[valid1] += wk * (1.0 - h[d:])
            g[d:] -= wk * h[:n - d]
        else:
            s = -d
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
    return g

def _grad_descent(h, n, t_end, lr0=0.04, beta_top=0.7, focus=0.25):

    cur = _eval(h)
    dx = 2.0 / n
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n); v = np.zeros(n)
    t = 0
    while time.time() < t_end:
        t += 1
        c = np.correlate(h, 1.0 - h, mode='full')
        thresh = np.percentile(c, 100 * (1 - focus))
        spread = c.max() - thresh + 1e-15
        w = np.where(c > thresh, np.exp(beta_top * (c - thresh) / spread), 0.0)
        if w.sum() > 0:
            w /= w.sum()
        else:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        g = _grad_weighted(h, w, n)
        lr = lr0 / (1 + 0.02 * t)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        mhat = m / (1 - beta1 ** t)
        vhat = v / (1 - beta2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + 1e-8)
        h_new = _repair(h - step, n)
        c_new = _eval(h_new)
        if c_new < cur - 1e-15:
            h = h_new; cur = c_new
        else:

                h_try = _repair(h - step * f, n)
                c_try = _eval(h_try)
                if c_try < cur - 1e-15:
