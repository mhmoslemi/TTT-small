def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(x, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n * 0.5
    if x.sum() == target and (x >= 0).all() and (x <= 1).all():
        return x.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    """FISTA optimization with homotopy in alpha."""
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _eval(h)
    
    alphas = np.geomspace(alpha_start, alpha_end, alpha_steps)

        h_prev = h.copy()

            h_new = _project(y - eta * g, n)

            bt = 0
            while c_new > c_old * (1 + 1e-12) and bt < 10:
                eta *= 0.5
                h_new = _project(y - eta * g, n)
                c_new = _eval(h_new)
                bt += 1
            
            h_prev = h
            h = h_new
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

            h = np.clip(h, 0.0, 1.0)
    return h

def _grad_lag_k(h, kk, dx, n):
    g = np.zeros(n)
    shift = kk - (n - 1)
    if shift >= 0:
        s = shift
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    else:
        s = -shift
        if s > 0:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g * dx

def _grad_smoothed(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    a_dx = alpha * dx
    m = c.max()
    shifted = a_dx * (c - m)
    ex = np.exp(shifted)
    w = ex / ex.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-11:
            continue
        g += w[k] * _grad_lag_k(h, k, dx, n)
    return g

def _fista(h0, n, t_budget, alpha_lo, alpha_hi, n_alpha):
    dx = 2.0 / n
    h = h0.copy()
    best_h, best_c5 = h.copy(), _eval(h)
    alphas = np.geomspace(alpha_lo, alpha_hi, n_alpha)
    time_per = t_budget / n_alpha
    for alpha in alphas:
        t_start = time.time()
        t = 1.0
        h_prev = h.copy()
        eta = 2.0 / (alpha * dx * n)
        while time.time() - t_start < time_per:
            g = _grad_smoothed(h, alpha)
            y = h + (t - 1.0) / (t + 1.0) * (h - h_prev)
            h_new = _project(y - eta * g, n)
            c_old = max(np.correlate(h, 1.0 - h, 'full')) * dx
            c_new = max(np.correlate(h_new, 1.0 - h_new, 'full')) * dx
            bt = 0
            while c_new > c_old + 1e-14 and bt < 8:
                eta *= 0.4
                h_new = _project(y - eta * g, n)
                c_new = max(np.correlate(h_new, 1.0 - h_new, 'full')) * dx
                bt += 1
            h_prev, h = h, h_new
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            c5 = c_new
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def _toplag_descent(h0, n, t_budget, n_lags=3):
    dx = 2.0 / n
    h = h0.copy()
    best_h, best_c5 = h.copy(), _eval(h)
    t_start = time.time()
    step = 0.05
    while time.time() - t_start < t_budget:
        c = np.correlate(h, 1.0 - h, 'full')
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for kk in top_k:
            g += _grad_lag_k(h, kk, dx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for _ in range(30):
            h_new = _project(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-15:
                best_c5, best_h, h = c5n, h_new.copy(), h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.7
            if step < 1e-7:
                break
        else:
            step = min(step * 1.1, 0.08)
    return best_h, best_c5

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return _project(np.interp(xn, xo, h_old), n_new)

def _fourier_init(n, K, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        h += rng.uniform(-0.25, 0.25) * np.cos(np.pi * j * x)
        h += rng.uniform(-0.25, 0.25) * np.sin(np.pi * j * x)
    return _project(h, n)

def _binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return _project(h + rng.randn(n) * 0.1, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _project(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)
    best_n = n0

    # Phase 1: FISTA homotopy from parent
    if tl() > 250:
        h1, c5 = _fista(best_h, best_n, min(200, tl() - 250), 500, 200000, 4)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Top-lag descent
    if tl() > 150:
        h2, c5 = _toplag_descent(best_h, best_n, min(120, tl() - 150), n_lags=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: FISTA with sharper alpha
    if tl() > 150:
        h3, c5 = _fista(best_h, best_n, min(120, tl() - 150), 50000, 1000000, 3)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 4: More top-lag descent (5 lags)
    if tl() > 120:
        h4, c5 = _toplag_descent(best_h, best_n, min(100, tl() - 120), n_lags=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h4.copy()

    # Phase 5: Try other resolutions
    if tl() > 250:
        for n_try in [400, 500, 800, 1000]:
            if tl() < 60:
                break
            hr = _resample(best_h, n_try)
            hr, c5 = _fista(hr, n_try, min(40, tl() - 60), 1000, 100000, 3)
            hr, c5 = _toplag_descent(hr, n_try, min(30, tl() - 40), n_lags=3)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hr.copy(), n_try

    # Phase 6: Fourier multi-start
    if tl() > 200:
        for K in [4, 7, 11]:
            if tl() < 50:
                break
            for _tr in range(3):
                hf = _fourier_init(best_n, K, rng)
                hf, c5 = _fista(hf, best_n, min(25, tl() - 50), 500, 50000, 2)
                hf, c5 = _toplag_descent(hf, best_n, min(20, tl() - 30), n_lags=3)
                if c5 < best_c5:
                    best_c5, best_h = c5, hf.copy()

    # Phase 7: Binary perturbation
    if tl() > 150:
        for _rep in range(4):
            if tl() < 40:
                break
            hb = _binary_init(best_n, rng)
            hb = _project(best_h + hb * 0.3 + rng.randn(best_n) * 0.05, best_n)
            hb, c5 = _fista(hb, best_n, min(30, tl() - 40), 1000, 80000, 2)
            hb, c5 = _toplag_descent(hb, best_n, min(25, tl() - 20), n_lags=3)
            if c5 < best_c5:
                best_c5, best_h = c5, hb.copy()

    # Phase 8: Fine pairwise coordinate descent
    if tl() > 60:
        h8 = best_h.copy()
        nc = len(h8)
        c5c = _eval(h8)
        for scale in [0.002, 0.0005, 0.0001]:
            if tl() < 15:
                break
            for _it in range(8000):
                if tl() < 12:
                    break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h8[i], h8[j] = ni - d, nj + d
            if c5c < best_c5:
                best_c5, best_h = c5c, h8.copy()

    # Phase 9: Final top-lag polish
    if tl() > 15:
        h9, c5 = _toplag_descent(best_h, best_n, min(15, tl() - 5), n_lags=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h9.copy()

    best_h = _project(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
