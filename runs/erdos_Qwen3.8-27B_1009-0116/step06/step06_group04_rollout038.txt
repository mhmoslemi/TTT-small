def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _gc(x, w):
    """Gradient of sum_k w[k]*c[k] for c=correlate(x,1-x,full)"""
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-20:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            if sh == 0:
                g += wk * (1.0 - x)
            else:
                g[:n-sh] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

    stall = 0

        if time.time() > dl: break
        t += 1
        f = i / max(niter - 1, 1)

        c = np.correlate(x, 1.0 - x, mode='full')

        w = cp ** (p - 1) * (sump ** ((1 - p) / p))
        g = _gc(x, w)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        x -= lr_t * (m / (1 - b1**t)) / (np.sqrt(v / (1 - b2**t)) + eps)
        x = _repair(x, n)
        cv = _eval(x)
        if cv < bc:
            bc, bx = cv, x.copy(); stall = 0
        else:
            stall += 1
        if stall > 800 and i % 800 == 0:
            x = bx + 0.004 * rng.randn(n)
            x = _repair(x, n)
            stall = 0
    return bx, bc

def _slsqp(h, p, mit, ftol):
    n = len(h)
    def _f(x):
        c = np.correlate(x, 1.0-x, mode='full')
        cp = c - c.min() + 1.0
        return (np.sum(cp**p))**(1/p) + c.min()
    def _g(x):
        c = np.correlate(x, 1.0-x, mode='full')
        cp = c - c.min() + 1.0
        sp = np.sum(cp**p)
        return _gc(x, cp**(p-1) * sp**((1-p)/p))
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,
             'jac':lambda x: np.ones(n)}]
    try:
        r = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter':mit,'ftol':ftol})
        return _repair(r.x, n)
    except:
        return _repair(h, n)

def _rs(h, nn):

        cn = _eval(x)
        if cn < cc - 1e-15:

            if cn < bc: bc, bx = cn, x.copy()
            sc = max(sc * 0.9995, 1e-10)
        else:
            x[i], x[j] = oi, oj
            sc *= 0.95
    return bx, bc
