import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lssm(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0 and sh < n:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        elif sh < 0:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _se(x): return x.sum() - len(x) * 0.5
def _sj(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol, t_end=None):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > t_end: return _repair(h, n)
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, T, t_end, lr0=0.01):
    n = len(h)
    x = h.copy(); y = h.copy()
    best = h.copy(); best_c = _eval(best)
    t_mom = 1.0
    for it in range(T):
        if time.time() > t_end: break
        g = _lssm_grad(y, alpha)
        g = g - g.mean()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        lr = lr0 / (1.0 + 0.02 * it)
        y_new = _repair(y - lr * g, n)
        x_new = _repair(y_new + beta * (y_new - y), n)
        c_new = _eval(x_new)
        if c_new < best_c:
            best_c = c_new; best = x_new.copy()
        x = x_new; t_mom = t_new
    return best, best_c

def _worst_shift_grad(h, t_end, rounds=500):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    h = h.copy()
    for _ in range(rounds):
        if time.time() > t_end: break
        c = _corr(h)
        ki = int(np.argmax(c))
        sh = (n - 1) - ki
        g = np.zeros(n)
        if sh >= 0 and sh < n:
            g[:n-sh] += dx * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= dx * h[:n-sh]
        elif sh < 0:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n-sp])
            g[:n-sp] -= dx * h[sp:]
        g = g - g.mean()
        moved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-14:
                h, c5 = h_new, c5n; moved = True; break
        if not moved: break
    return h, c5

def _binary_targeted_swap(h, t_end, rng, max_iter=50000):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5 = _eval(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    stall = 0
    for _ in range(max_iter):
        if time.time() > t_end: break
        if len(ones) == 0 or len(zeros) == 0: break
        c = _corr(h)
        k_worst = int(np.argmax(c))
        s_eff = (n - 1) - k_worst
        candidates = []
        if s_eff >= 0 and s_eff < n:
            for i in range(n - s_eff):
                if h[i] > 0.5 and h[i+s_eff] < 0.5:
                    candidates.append((i + s_eff, i))
                elif h[i] < 0.5 and h[i+s_eff] > 0.5:
                    candidates.append((i, i + s_eff))
        elif s_eff < 0:
            sp = -s_eff
            for i in range(sp, n):
                if h[i] > 0.5 and h[i-sp] < 0.5:
                    candidates.append((i - sp, i))
                elif h[i] < 0.5 and h[i-sp] > 0.5:
                    candidates.append((i, i - sp))
        if not candidates:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            candidates = [(j, i)]
        rng.shuffle(candidates)
        best_delta = 1e-15
        best_pair = None
        trial_count = min(len(candidates), 80)
        for jj, ii in candidates[:trial_count]:
            if h[jj] < 0.5 and h[ii] > 0.5:
                h[jj] = 1.0; h[ii] = 0.0
                c5n = _eval(h)
                delta = c5 - c5n
                if delta > best_delta:
                    best_delta = delta; best_pair = (jj, ii)
                h[jj] = 0.0; h[ii] = 1.0
        if best_pair is not None and best_delta > 1e-14:
            jj, ii = best_pair
            h[jj] = 1.0; h[ii] = 0.0
            c5 = _eval(h)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            stall = 0
        else:
            stall += 1
            if stall > 200:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0.0, 1.0
                h = _repair(h, n)
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                stall = 0
    return h, c5

def _binary_anneal(h, t_end, rng):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5 = _eval(h); best = c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    t_arr = np.linspace(0.3, 0.005, 4)
    for ti, t_val in enumerate(t_arr):
        n_steps = max(int((t_end - time.time()) / (4 - ti)), 1000)
        for _ in range(n_steps):
            if time.time() > t_end: break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5 or rng.rand() < np.exp(-(c5n - c5) / max(t_val, 1e-6)):
                c5 = c5n
                if c5n < best: best = c5n
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                h[i], h[j] = 1.0, 0.0
                h = _repair(h, n)
    return h, best

def _coord_descent(h, t_end, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for scale in [0.01, 0.003, 0.001, 0.0003]:
        for _ in range(20000):
            if time.time() > t_end: break
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
