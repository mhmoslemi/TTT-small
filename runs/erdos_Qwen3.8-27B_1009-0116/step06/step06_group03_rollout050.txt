import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad(h, w, dx):
    """Compute gradient given correlation weights w."""
    n = len(h)

        else:
            sp = -sh
            g[:n - sp] -= wk * h[sp:n]
            g[sp:] += wk * (1.0 - h[:n - sp])
    return g

So for S=[0,1]: max_k measure = max(max_{k∈(0,1)} (1-k), max_{k∈[1,2]} (2-k)) = 1 (at k→0 or k→2).

That's bad. We need a set with small boundary for all shifts simultaneously.

What about S = [0,0.5] ∪ [1.5, 2]? Measure = 1. For shift k:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _correct_grad(h, w, dx):
    """Compute gradient of soft-max objective with CORRECT gradient."""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            # c[k] = sum_{n=0}^{n-1-sh} h[n+sh]*(1-h[n])
            # d/dh[j] = (1-h[j-sh]) for j in [sh, n-1]
            #          + (-h[j+sh]) for j in [0, n-1-sh]
            if sh > 0:
                g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            # c[k] = sum_{n=sp}^{n-1} h[n-sp]*(1-h[n])
            # Let m=n-sp: sum_{m=0}^{n-1-sp} h[m]*(1-h[m+sp])
            # d/dh[j] = (1-h[j+sp]) for j in [0, n-1-sp]
            #          + (-h[j-sp]) for j in [sp, n-1]
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _softmax_obj_and_weights(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    return m, w

Let me also think about whether I should use a different optimization approach entirely. 

**Alternative: Use the fact that the objective is a max of quadratics.**

The problem is: min_h max_k f_k(h) subject to constraints.
where f_k(h) = Σ h[i+lag]*(1-h[i]) * dx.

This is a min-max problem. One approach: alternate between:
1. Fix the weights w (dual variables), solve the convex problem: min_h Σ w_k * f_k(h)
2. Update w to put more weight on the currently largest f_k.

This is the Frank-Wolfe / mirror descent approach for min-max. But f_k is non-convex (it's quadratic with negative h*h terms), so step 1 is non-convex.

However! If we restrict h to be "close" to the current iterate, the quadratic is approximately linear, and we can use a local solver. This is essentially what the gradient descent is doing.

**Another idea: Use the Dinkelbach method or bisection on C₅.**

Can we check: does there exist h with C₅ ≤ t? This is:
min_h max_k f_k(h) ≤ t
⇔ ∃h: f_k(h) ≤ t for all k, sum(h)=n/2, 0≤h≤1

Each f_k(h) ≤ t is a quadratic constraint: Σ h[i+lag]*(1-h[i]) * dx ≤ t.
This is a non-convex quadratic constraint (the -h*h terms). So bisection doesn't directly help.
