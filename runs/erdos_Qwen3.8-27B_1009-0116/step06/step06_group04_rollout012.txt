6. Coordinate descent fine-tuning

Let me also add a key innovation: **shift-localized coordinate descent**. Instead of random coordinate pairs, I'll focus on coordinates that affect the worst shifts. If shift k is the worst, then the coordinates h[max(0,k-n+1):min(n, n+k-n+1)] are the ones that matter.

Actually, for a given shift index i (in 0..2n-2), the relevant coordinates are:
- For d = i-(n-1) >= 0: h[0:n-d] and h[d:n]
- For d < 0: h[0:n+d'] and h[d':n] where d'=-d

So if I know the top-k worst shifts, I can focus coordinate descent on the union of their relevant coordinate sets.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _wgrad(x, w):
    n = len(x)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g * dx

def _pen_weights(c, T):
    w = 2.0 * np.maximum(c - T, 0)
    if w.sum() < 1e-14:
        idx = np.argmax(c)
        w = np.zeros_like(c); w[idx] = 1.0
    return w

def _softmax_weights(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _adam_pen(h0, T_lo, T_hi, n_iter, lr, seed, deadline):
    n = len(h0)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        T = T_hi + (T_lo - T_hi) * frac
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        w = _pen_weights(c, T)
        g = _wgrad(x, w)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c2 = _eval(x)
        if c2 < best_c:
            best_c = c2; best_x = x.copy()
        if t % 2000 == 0:
            x = x + 0.003 * rng.randn(n)
            x = _repair(x, n)
    return best_x, best_c

def _lbfgs_pen(h0, T, maxit):
    n = len(h0)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        viol = np.maximum(c - T, 0)
        return float(np.sum(viol ** 2))
    def grd(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        w = _pen_weights(c, T)
        return _wgrad(x, w)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grd, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-22, 'gtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _slsqp_max(h0, alpha, maxit):
    n = len(h0)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m * dx + np.log(float(np.sum(np.exp(s)))) * dx / alpha
    def grd(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        w = _softmax_weights(c, alpha)
        return _wgrad(x, w)
    def se(x):
        return x.sum() - n * 0.5
    def sj(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': se, 'jac': sj}]
    try:
        res = minimize(obj, h0, jac=grd, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _resample(h, n_new):
    xo = np.linspace(0.0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _binary_swap(h, n, rng, deadline, n_iters=500000):
    x = (h > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j or x[i] == x[j]:
            continue
        x[i], x[j] = x[j], x[i]
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
        else:
            x[i], x[j] = x[j], x[i]
            x = _repair(x, n)
    return best_x, best_c

def _coord_desc(h, n, rng, deadline, scale=0.01):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    for _ in range(300000):
        if time.time() > deadline:
            break
        if scale < 1e-8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        xi = x[i]; xj = x[j]
        x[i] = np.clip(xi + d, 0, 1)
        x[j] = np.clip(xj - d, 0, 1)
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < c_cur - 1e-15:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
            scale = max(scale * 0.9995, 1e-9)
        else:
            x = best_x.copy(); c_cur = best_c
            scale *= 0.95
    return best_x, best_c
