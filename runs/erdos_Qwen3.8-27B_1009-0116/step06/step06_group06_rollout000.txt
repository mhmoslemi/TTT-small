    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:tk]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ki in range(tk):
        k = idx[ki]; wk = w[ki]
        if wk < 1e-14: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= 2.0 / n
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def proj_bisect(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def lbf_opt(h, alpha, pen, maxiter, t_end, tk=20):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smax_full, h, jac=smax_grad, args=(alpha, pen, tk),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        return repair(res.x, n)
    except:
        return repair(h, n)

def grad_desc(h, alpha, n_steps, t_end, tk=20):
    n = len(h)
    h_arr = h.copy()
    v = np.zeros(n)
    f_prev = eval_c5(h_arr)
    step = 0.04
    for _ in range(n_steps):
        if time.time() > t_end: break
        g = smax_grad(h_arr, alpha, 0.0, tk)
        gnv = np.linalg.norm(g)
        if gnv < 1e-14: break
        v = 0.7 * v + 0.3 * g
        v /= np.linalg.norm(v)
        ok = False
        for _ in range(12):
            h_new = proj_bisect(h_arr - step * v, n * 0.5)
            f_new = eval_c5(h_new)
            if f_new < f_prev - 1e-15:
                h_arr = h_new; f_prev = f_new; ok = True
                step = min(step * 1.08, 0.25); break
            step *= 0.5
        if not ok: step *= 0.3
    return h_arr

def sa_refine(h, t_end, rng, n_iter=None):
    n = len(h)
    if n_iter is None:
        n_iter = min(int((t_end - time.time()) / 0.0025), 100000)
    h_cur = h.copy(); best_h = h_cur.copy()
    c_cur = eval_c5(h_cur); best_c = c_cur
    t0v, t1v = 0.005, 5e-7
    for i in range(n_iter):
        if time.time() > t_end: break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n); i2 = rng.randint(n)
        if i1 == i2: continue
        d = rng.randn() * (0.002 + 0.04 * t)
        o1, o2 = h_cur[i1], h_cur[i2]
        h_cur[i1], h_cur[i2] = o1 + d, o2 - d
        h_cur = repair(h_cur, n)
        c_new = eval_c5(h_cur)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c: best_c = c_cur; best_h = h_cur.copy()
        else:
            h_cur[i1], h_cur[i2] = o1, o2; h_cur = repair(h_cur, n)
    return best_h, best_c

def multi_refine(h, alphas, pens, t_end, tk=18, maxit=600):
    n = len(h)
    h_cur = h.copy(); best_c = eval_c5(h_cur)
    for alpha in alphas:
        if time.time() > t_end: break
        for pen in pens:
            if time.time() > t_end: break
            h_new = lbf_opt(h_cur, alpha, pen, maxit, t_end, tk)
            c = eval_c5(h_new)
            if c < best_c: best_c = c; h_cur = h_new
    return h_cur, best_c

def upsample(h, f):
    n = len(h)
    hh = np.zeros(n * f)
    hh[::f] = h
    for i in range(1, f):
        hh[i::f] = 1.0 / f * (f - i) * h + 1.0 / f * i * np.roll(h, -1)
    return repair(hh, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def rem(): return budget_s - (time.time() - t0)
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy(); best_c5 = eval_c5(best_h); best_n = n0

    # Phase 1: Core L-BFGS-B refinement with progressive alpha
    h_r, c_r = multi_refine(best_h, [100, 400, 1500, 6000, 25000, 80000],
                            [10.0, 100.0], t0 + rem() - 200, tk=18, maxit=800)
    if c_r < best_c5: best_c5 = c_r; best_h = h_r

    # Phase 2: Projected gradient to escape SLSQP-like stagnation
    if rem() > 180:
        h_pg = grad_desc(best_h, 5000, 400, t0 + rem() - 160, tk=25)
        c_pg = eval_c5(h_pg)
        if c_pg < best_c5: best_c5 = c_pg; best_h = h_pg

    # Phase 3: Alternative starting points
    if rem() > 320:
        x = np.linspace(0, 2, n0, endpoint=False)
        starts = [np.roll(best_h, n0 // 7), 1.0 - best_h,
                  repair(best_h + 0.04 * rng.randn(n0), n0),
                  repair(np.sin(3 * np.pi * x) * 0.15 + 0.5 + 0.1 * np.sin(7 * np.pi * x), n0)]
        for sh in starts:
            if rem() < 260: break
            h_try, c_try = multi_refine(sh, [400, 3000, 15000], [50.0],
                                        t0 + rem() - 240, tk=15, maxit=500)
            if c_try < best_c5: best_c5 = c_try; best_h = h_try

    # Phase 4: Upsample to 2x and refine
    if rem() > 320:
        n1 = n0 * 2
        h_hi = upsample(best_h, 2)
        h_hi, c_hi = multi_refine(h_hi, [100, 500, 3000, 12000], [10.0, 100.0],
                                  t0 + rem() - 200, tk=20, maxit=600)
        if c_hi < best_c5: best_c5 = c_hi; best_h = h_hi; best_n = n1
        if rem() > 180:
            h_pg = grad_desc(best_h, 8000, 300, t0 + rem() - 160, tk=30)
            c_pg = eval_c5(h_pg)
            if c_pg < best_c5: best_c5 = c_pg; best_h = h_pg

    # Phase 5: SA refinement
    if rem() > 80:
        h_sa, c_sa = sa_refine(best_h, t0 + rem() - 50, rng)
        if c_sa < best_c5: best_c5 = c_sa; best_h = h_sa

    # Phase 6: Final L-BFGS-B polish at high alpha
    if rem() > 50:
        h_fin = lbf_opt(best_h, 150000, 500.0, 1500, t0 + rem() - 25, tk=25)
        c_fin = eval_c5(h_fin)
        if c_fin < best_c5: best_c5 = c_fin; best_h = h_fin

    # Phase 7: Ultra-fine SA
    if rem() > 25:
        h_sa2, c_sa2 = sa_refine(best_h, t0 + rem() - 8, rng, n_iter=30000)
        if c_sa2 < best_c5: best_c5 = c_sa2; best_h = h_sa2

    # Phase 8: Final coordinate micro-adjustments
    if rem() > 15:
        h_cur = best_h.copy(); nc = len(h_cur); c5c = eval_c5(h_cur)
        nt = min(int((rem() - 10) / 0.0012), 50000)
        for _ in range(nt):
            if rem() < 10: break
            i = rng.randint(nc); j = rng.randint(nc)
            if i == j: continue
            d = 3e-5 * rng.randn()
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h_cur[i], h_cur[j] = ni, nj
            cn = eval_c5(h_cur)
            if cn < c5c - 1e-17:
                c5c = cn
                if cn < best_c5: best_c5 = cn; best_h = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d

    best_h = repair(best_h, best_n)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, best_n)
