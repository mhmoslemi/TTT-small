import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sm_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, n_iter, step0, bc):
    n = len(h)
    f = _sm_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(n_iter):
        if bc():
            break
        g = _sm_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _sm_obj(h_new, alpha)
        if f_new < f - 1e-16:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.7
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _sm_obj(h, alpha)
    return _repair(h, n)

def _topk_descent(h, topk, n_iter, bc):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    stall = 0
    for _ in range(n_iter):
        if bc():
            break
        c = _corr(h)
        top = np.argsort(c)[-topk:]
        g = np.zeros(n)
        for k in top:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= dx / topk
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)

                stall = 0

            if stall > 20:
                h = _repair(h + 0.002 * np.random.randn(n), n)
                c5 = _eval(h)
                stall = 0
    return h, c5

    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()
