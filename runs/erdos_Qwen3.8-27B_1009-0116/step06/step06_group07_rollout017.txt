def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:

        h3, c5 = _cd(h3, 3000, seed + 500 + trial * 13, t_end - 8, t0)
        if c5 < best_c5:
            best_c5 = c5; best_h = h3.copy()

    # Phase 7: Try nearby n
    if time.time() < t_end - 20:
        for dn in [10, -10, 0, 20, -20]:
            if time.time() > t_end - 10:
                break
            nt = best_n + dn
            if nt < 50 or nt > 600 or nt % 2 != 0:
                continue
            ht = _repair(_resample(best_h, nt), nt)
            ht, c5 = _cd(ht, 2000, seed + 900, t_end - 8, t0)
            if c5 < best_c5:
                best_c5 = c5; best_h = ht.copy(); best_n = nt

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
