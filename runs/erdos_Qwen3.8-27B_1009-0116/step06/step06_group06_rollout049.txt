    lo, hi = -5.0, 5.0

            lo = mid
        else:
            hi = mid

def _smoother_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s = s - s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(h, alpha):

import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _soft_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _lag_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])

    return g

def _proj_grad(h, alpha, nsteps, tend):
    n = len(h)
    hc = h.copy()
    fp = _soft_obj(hc, alpha)
    v = np.zeros(n)
    mu = 0.75

            break

def _slsqp(h, alpha, maxit, tend):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return x.sum() - n * 0.5
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    fs = _soft_obj(h, alpha)
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-17, 'disp': False})
        if time.time() < tend and res.fun < fs - 1e-13:
            return np.clip(res.x, 0, 1)
    except Exception:
        pass
    return h

def _coord_desc(h, niters, tend, rng):
    n = len(h)
    hc = h.copy()

    for _ in range(niters):

        j = rng.randint(n)
        if i == j:

        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        c5n = _eval(hc)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            hc[i], hc[j] = ni - d, nj + d
    return hc, c5c

    n = len(h)
    hc = h.copy()
    cc = _eval(hc)
    bh, bc = hc.copy(), cc
    t0v, t1v = 0.003, 1e-7

            break
        t = t0v * (t1v / t0v) ** (i / max(1, niters - 1))
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.002 + 0.012 * t)
        o1, o2 = hc[i1], hc[i2]
        hc[i1] = min(1, max(0, o1 + d))
        hc[i2] = min(1, max(0, o2 - d))
        diff = n * 0.5 - hc.sum()
        if abs(diff) > 1e-12:
            hc[i1] += 0.5 * diff
            hc[i2] -= 0.5 * diff
        hc = np.clip(hc, 0, 1)
        cn = _eval(hc)
        if cn < cc or rng.rand() < np.exp(-(cn - cc) / max(t, 1e-9)):
            cc = cn
            if cn < bc:
                bc, bh = cn, hc.copy()
        else:
            hc[i1], hc[i2] = o1, o2
    return bh, bc

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for j in idx:
            if d > 0:
                a = min(d, 1.0 - h[j])
                h[j] += a
                d -= a
            else:
                a = min(-d, h[j])
                h[j] -= a
                d += a
            if abs(d) < 1e-12:
                break
    return h
