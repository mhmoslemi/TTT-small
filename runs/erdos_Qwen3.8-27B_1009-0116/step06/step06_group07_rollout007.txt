def _coord_descent(h, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d

        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:

        else:
            h[i], h[j] = ni - d, nj + d

    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    T = 31
    m = n0 // T

    # Phase 1: L-BFGS-B on full vector (explore non-periodic)
    for alpha, pen in [(3000, 100), (10000, 200), (50000, 500)]:
        if elapsed() > budget_s * 0.08:
            break
        h_new = _lbfgsb_full(best_h.copy(), alpha, pen, 800)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Periodic PGD on T=31 with detailed alpha sweep
    h_p = best_h.reshape(m, T).mean(axis=0)
    h_p = _repair(h_p, T)
    alpha_sets = [
        [1000, 3000, 10000, 30000, 100000],
        [5000, 15000, 50000, 150000, 500000],
        [20000, 60000, 200000, 600000, 1500000],
    ]
    for ai, aset in enumerate(alpha_sets):
        if elapsed() > budget_s * 0.35:
            break
        h_p, c5 = _pgd_period(h_p, T, n0, aset, 12000, lr_init=0.004, seed=seed + ai)
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.tile(h_p, m).copy()

    # Phase 3: SLSQP on period-31 (31 vars, very fast)
    if elapsed() < budget_s * 0.55:
        h_p = best_h[:T].copy() if best_n == n0 else _repair(_resample(best_h, n0).reshape(m, T).mean(axis=0), T)
        for alpha in [30000, 100000, 400000, 1500000]:
            if elapsed() > budget_s * 0.60:
                break
            h_p = _slsqp_period(h_p, T, n0, alpha, 5000)
            c5 = _eval(np.tile(h_p, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()

    # Phase 4: L-BFGS-B on full 310 from periodic solution
    if elapsed() < budget_s * 0.78:
        for alpha, pen in [(50000, 500), (200000, 1000), (500000, 2000)]:
            if elapsed() > budget_s * 0.80:
                break
            h_new = _lbfgsb_full(best_h.copy(), alpha, pen, 1500)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Multi-start perturbation + periodic re-opt
    if elapsed() < budget_s * 0.90:
        for trial in range(6):
            if elapsed() > budget_s * 0.92:
                break
            scale = 0.03 + 0.02 * trial
            h_pert = best_h + scale * rng.randn(best_n)
            h_pert = _repair(h_pert, best_n)
            if best_n % T == 0:
                h_p = h_pert.reshape(m, T).mean(axis=0)
                h_p = _repair(h_p, T)
                h_p, c5 = _pgd_period(h_p, T, best_n, [50000, 200000, 800000], 6000, lr_init=0.003, seed=seed + 50 + trial)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.tile(h_p, m).copy()
                h_p = _slsqp_period(h_p, T, best_n, 500000, 3000)
                c5 = _eval(np.tile(h_p, m))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.tile(h_p, m).copy()

    # Phase 6: Fine SLSQP polish
    if elapsed() < budget_s - 30:
        h_new = _slsqp_full(best_h.copy(), 2000000, 2000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 7: Coordinate descent polish
    if elapsed() < budget_s - 10:
        for scale in [0.002, 0.0005, 0.0001]:
            if elapsed() > budget_s - 5:
                break
            h_cd, c5 = _coord_descent(best_h.copy(), 5000, scale, seed=seed + 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd.copy()

    # Phase 8: Try n=300 and n=320
    if elapsed() < budget_s - 20:
        for nt in [300, 320, 310]:
            if elapsed() > budget_s - 10:
                break
            if nt == best_n:
                continue
            h_r = _repair(_resample(best_h, nt), nt)
            h_new = _lbfgsb_full(h_r, 100000, 500, 1000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = nt

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
