import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff += -a if diff > 0 else a
            if abs(diff) < 1e-14:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha, mu):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = m + np.log(np.sum(np.exp(s))) / alpha
    return smax * (2.0 / n) + mu * (h.sum() - n * 0.5) ** 2

def _smax_grad(h, alpha, mu):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    g += 2.0 * mu * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, alpha, mu, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 3})
        if res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _subgrad(h, n_topk, step):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[::-1][:n_topk]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    g *= dx / n_topk
    return _proj(h - step * g, n * 0.5)

def _mom_subgrad(h, n_topk, n_steps, t_end):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    step = 0.03
    v = np.zeros(n)
    mu_m = 0.75
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        h_new = _subgrad(h_cur, n_topk, step)
        c5_new = _eval(h_new)
        if c5_new < c5_cur:
            v = mu_m * v + (1 - mu_m) * (h_new - h_cur)
            h_cur = h_new
            c5_cur = c5_new
            step = min(step * 1.01, 0.06)
        else:
            h_revert = _subgrad(h_cur, n_topk, step * 0.5)
            c5_rev = _eval(h_revert)
            if c5_rev < c5_cur:
                v = mu_m * v + (1 - mu_m) * (h_revert - h_cur)
                h_cur = h_revert
                c5_cur = c5_rev
            step *= 0.85
    return h_cur, c5_cur

def _coord_cd(h, n_iter, t_end, rng):
    n = len(h)
    c_cur = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c_new = _eval(h)
        if c_new < c_cur:
            c_cur = c_new
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c_cur

def _sa(h, n_iter, t_end, rng):
    n = len(h)
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    for i in range(n_iter):

        t = 0.03 * (1e-4 / 0.03) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        if rng.rand() < 0.3:
            h[i1] = 1.0 - h[i1]
            h[i2] = 1.0 - h[i2]
        else:
            d = rng.randn() * (0.005 + 0.04 * t)
            h[i1], h[i2] = h[i1] + d, h[i2] - d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-10)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            # revert logic: just re-repair
            pass
    return best_h, best_c

def _try_alt_start(n, rng, method):
