import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_corr(h, k):
    """Gradient of correlation at shift k w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(0, n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(0, n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _grad_smoothed_max(h, alpha):
    """Gradient of log-sum-exp smoothing of max correlation"""
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -100, 100)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * dx * _grad_corr(h, k)
    return g

def _active_set_grad(h, top_m, alpha):
    """Gradient focusing on top-m shifts with exponential weighting"""
    n = len(h)
    c = _corr(h)
    # Get top m shifts
    idx = np.argpartition(c, -top_m)[-top_m:]
    c_top = c[idx]
    c_min = c_top.min()
    # Exponential weighting within top-m
    if c_top.max() - c_min > 1e-14:
        w_top = np.exp(alpha * (c_top - c_min))
    else:
        w_top = np.ones(top_m)
    w_top /= w_top.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w_top[i] * dx * _grad_corr(h, k)
    return g

def _momentum_descent(h, n, n_iter, top_m, alpha, lr_init, t_end, momentum=0.9):
    """Momentum-based gradient descent on active-set smoothed max"""
    h = h.copy()
    v = np.zeros(n)
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    for t in range(n_iter):
        if time.time() > t_end:
            break
        g = _active_set_grad(h, top_m, alpha)
        lr = lr_init / (1.0 + t * 0.01)
        # Momentum update
        v = momentum * v - lr * g
        h_new = _repair(h + v, n)
        c_new = _eval(h_new)
        if c_new < cur:
            h = h_new
            cur = c_new
            if cur < best_c:

                best_h = h.copy()
        else:

                if c_try < cur:
                    h = h_try
                    cur = c_try
                    v = momentum * v
                    found = True
                    break
            if not found:
                v *= 0.5
    return best_h, best_c

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(0, n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(0, n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -100, 100)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            g += w[k] * dx * _grad_shift(h, k)
    return g

def _grad_active(h, top_m, walpha):

    c = _corr(h)
    idx = np.argpartition(c, -top_m)[-top_m:]
    ct = c[idx]
    rng = ct.max() - ct.min()
    if rng > 1e-14:
        wt = np.exp(walpha * (ct - ct.min()))
    else:
        wt = np.ones(top_m)
    wt /= wt.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(top_m):
        g += wt[i] * dx * _grad_shift(h, idx[i])
    return g
