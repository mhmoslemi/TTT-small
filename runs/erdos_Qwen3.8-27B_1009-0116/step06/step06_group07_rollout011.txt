            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()

    # Phase 3: PGD with adaptive alpha
    if time.time() - t0 < budget_s * 0.65:
        h_p = best_h[:T].copy()
        for alpha in [30000, 100000, 400000]:
            if time.time() - t0 > budget_s * 0.66:
                break
            h_p, c5 = _pgd_refine(h_p, T, n0, alpha, 2000, 0.003, seed + 50)
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()

    # Phase 4: Random restarts + DE (escape local minima)
    if time.time() - t0 < budget_s * 0.82:
        for trial in range(4):
            if time.time() - t0 > budget_s * 0.84:
                break
            scale = 0.08 * (1.0 + 0.5 * trial)
            h_pert = best_h[:T] + scale * rng.randn(T)
            h_pert = _repair(h_pert, T)
            h_test = _de_optimize(h_pert, T, n0, seed + 200 + trial * 7,
                                  maxiter=120, popsize=18)
            c5 = _eval(np.tile(h_test, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_test, m).copy()
                # Polish immediately
                h_pol = _slsqp_optimize(h_test, T, n0, 200000, maxiter=5000)
                c5p = _eval(np.tile(h_pol, m))
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = np.tile(h_pol, m).copy()

    # Phase 5: Try alternative n values
    if time.time() - t0 < budget_s * 0.92:
        for n_alt, T_alt in [(300, 30), (330, 33), (320, 16)]:
            if time.time() - t0 > budget_s * 0.93:
                break
            m_alt = n_alt // T_alt
            if n_alt % T_alt != 0:
                continue
            xo = np.linspace(0, 2, best_n, endpoint=False)
            xn = np.linspace(0, 2, n_alt, endpoint=False)
            h_alt = np.interp(xn, xo, best_h)
            h_alt = _repair(h_alt, n_alt)
            h_pa = h_alt.reshape(m_alt, T_alt).mean(axis=0)
            h_pa = _repair(h_pa, T_alt)
            h_pa = _de_optimize(h_pa, T_alt, n_alt, seed + 500, maxiter=150, popsize=18)
            h_pa = _slsqp_optimize(h_pa, T_alt, n_alt, 100000, maxiter=5000)
            c5 = _eval(np.tile(h_pa, m_alt))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_pa, m_alt).copy()
                best_n = n_alt

    # Phase 6: Final SLSQP polish on best
    if time.time() - t0 < budget_s - 15:
        T_best = T if best_n == n0 else (best_n // 30 if best_n % 30 == 0 else best_n)
        if best_n % T_best == 0 and T_best < 60:
            h_pb = best_h[:T_best].copy()
            h_pb = _slsqp_optimize(h_pb, T_best, best_n, 500000, maxiter=10000)
            c5 = _eval(np.tile(h_pb, best_n // T_best))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_pb, best_n // T_best).copy()

    # Final: small coordinate descent polish
    if time.time() - t0 < budget_s - 5:
        h_cd = best_h.copy()
        c5_cd = best_c5
        for _ in range(2000):
            if time.time() - t0 > budget_s - 3:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = 0.001 * rng.randn()
            ni, nj = h_cd[i] + d, h_cd[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_cd[i], h_cd[j] = ni, nj
            c = _eval(h_cd)
            if c < c5_cd - 1e-16:
                c5_cd = c
            else:
                h_cd[i], h_cd[j] = ni - d, nj + d
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
