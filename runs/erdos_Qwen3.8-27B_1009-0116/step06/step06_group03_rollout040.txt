import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(h, alpha):
    c = _corr(h)
    m = c.max()
    d = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(d))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    d = np.clip(alpha * (c - m), -500, 0)
    dm = d.max()
    w = np.exp(d - dm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter=500, ftol=1e-14):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return np.array([x.sum() - 0.5 * n])
    def cj(x):
        return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _topk_grad(h, k):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-k:]
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in idx:
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * dx / k

def _subgrad(h, n_iter, step0, topk):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stuck = 0
    for it in range(n_iter):
        g = _topk_grad(best, topk)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_sum(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-16:
                best = h_new
                best_c = c_new
                step = min(step * 1.15, 0.3)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.7
            stuck += 1
            if stuck > 80:
                best = _repair(best + 0.02 * np.random.RandomState(it).randn(n), n)
                best_c = _eval(best)
                step = step0
                stuck = 0
                v = np.zeros(n)
    return best, best_c

def _coord_descent(h, passes, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for p in range(passes):
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            best_local = old
            best_c_local = c_cur
            for trial_step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                for sign in [1, -1]:
                    new_val = old + sign * trial_step
                    if new_val < 0 or new_val > 1:
                        continue
                    d = sign * trial_step
                    h[i] = new_val
                    diff = -d
                    others = np.delete(np.arange(n), i)
                    j = rng.randint(len(others))
                    jj = others[j]
                    adj = diff
                    if h[jj] + adj < 0 or h[jj] + adj > 1:
                        h[i] = old
                        break
                    h[jj] += adj
                    cn = _eval(h)
                    if cn < best_c_local - 1e-16:
                        best_c_local = cn
                        best_local = new_val
                    h[i] = old
                    h[jj] -= adj
                if best_local != old:
                    h[i] = best_local
                    others = np.delete(np.arange(n), i)
                    j = others[rng.randint(len(others))]
                    h[j] += old - best_local
                    c_cur = _eval(h)
            if c_cur < best_c:
                best_c = c_cur
                best = h.copy()
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    step = 0.01
    for t in range(n_tries):
        if t % 2000 == 1999:
            step *= 0.85
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur - 1e-16:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c
