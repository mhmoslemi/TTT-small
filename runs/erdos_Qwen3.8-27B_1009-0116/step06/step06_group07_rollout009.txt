                    diff -= a

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    c = _correlate(x)

    s = np.clip(alpha * (c - m), -500, 500)

    n = len(h)
    c5c = _eval(h)

        j = rng.randint(n)

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d

        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-18:

        else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _obj(x, alpha, mu):
    c = _cor(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha + mu * (x.sum() - len(x) * 0.5) ** 2

def _grad(x, alpha, mu):
    n = len(x)
    c = _cor(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def _lbfgsb(x0, alpha, mu, maxiter, tol=1e-16):
    n = len(x0)
    bnds = [(0.0, 1.0)] * n

        res = minimize(_obj, x0, jac=_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,

                                'maxls': 80, 'maxcor': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(x0, n)

def _subgrad(h, lr):
    n = len(h)
    dx = 2.0 / n
    c = _cor(h)
    top_k = np.argsort(c)[-3:]
    g = np.zeros(n)
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += dx * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= dx * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += dx * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= dx * h[i2 + sp]
    return _repair(h - lr * g, n)

def _coord(h, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):

        j = rng.randint(n)

            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d

        if c5n < c5c - 1e-18:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _proj_sum(h, n):

    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
