        (5000000, 50000000, 4000, 0.002, 0.00003, 8, 600),
    ]:
        if time.time() > d6: break
        x, c = _adam(bh, a0, a1, ni, l0, l1, tk, seed + sd, d6)
        if c < bc5: bc5, bh = c, x.copy()

    # Phase 7: Final SLSQP (920-1030s)
    d7 = t0 + 1020
    if time.time() < d7:
        hf = _slsqp(bh, 20000000, 6, 4000)
        c5 = _eval(hf)
        if c5 < bc5: bc5, bh = c5, hf.copy()

    # Phase 8: Micro-tune (1030-1085s)
    d8 = t0 + 1080
    if time.time() < d8:
        x, c = _adam(bh, 20000000, 100000000, 3000, 0.001, 0.00002, 5, seed + 700, d8)
        if c < bc5: bc5, bh = c, x.copy()
    if time.time() < d8:
        hf2 = _slsqp(bh, 50000000, 5, 2000)
        c5 = _eval(hf2)
        if c5 < bc5: bc5, bh = c5, hf2.copy()

    bh = _repair(bh, bn)
    bc5 = _eval(bh)
    return (bh, bc5, bn)
