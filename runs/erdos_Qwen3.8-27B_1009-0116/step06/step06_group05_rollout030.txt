def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute difference
    if diff > 0:
        order = np.argsort(1.0 - h)  # most room first
    else:
        order = np.argsort(h)  # most to remove first
    remaining = abs(diff)
    for idx in order:
        if remaining < 1e-12:
            break
        if diff > 0:
            a = min(remaining, 1.0 - h[idx])
            h[idx] += a
            remaining -= a
        else:
            a = min(remaining, h[idx])
            h[idx] -= a
            remaining -= a
    return h

def _proj_feasible(y, n):
    """Project y onto {x : 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: clip(y_i - lambda, 0, 1) sums to target
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _subgrad_topk(h, k=5):
    """Compute subgradient using top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    # Get top-k indices
    topk = np.argsort(c)[-k:]
    g = np.zeros(n)
    for ki in topk:
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n-sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n-sp])
            g[:n-sp] -= dx * h[sp:]
    # Weight by correlation values (normalization)
    w = c[topk]
    w = w / w.sum()
    # Recompute with weights
    g2 = np.zeros(n)
    for idx, ki in enumerate(topk):
        sh = (n - 1) - ki
        wk = w[idx] * dx
        if sh >= 0:
            g2[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g2[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g2[sp:] += wk * (1.0 - h[:n-sp])
            g2[:n-sp] -= wk * h[sp:]
    return g2

def _nesterov_subgrad(h, n_steps, k_shifts=5, base_lr=0.01):
    """Nesterov-accelerated subgradient with projection."""
    n = len(h)
    h_curr = h.copy()
    h_next = h.copy()
    y = h.copy()  # momentum variable
    best_h = h.copy()
    best_c5 = _eval(h)
    for t in range(n_steps):
        lr = base_lr / (1.0 + 0.1 * t / n_steps)
        # Nesterov step
        y = _proj_feasible(h_next - lr * _subgrad_topk(h_next, k_shifts), n)
        h_new = _proj_feasible(0.5 * (h_curr + y) + 0.5 * (y - h_curr) * (1 - (t+1)/(t+2)), n)
        # Simpler: just use y as the next point
        h_new = y
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_next = h_curr  # swap
        h_curr = h_new
    return best_h, best_c5

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            rem = abs(diff)
            for idx in order:
                if rem < 1e-12: break
                if diff > 0:
                    a = min(rem, 1.0 - h[idx]); h[idx] += a; rem -= a
                else:
                    a = min(rem, h[idx]); h[idx] -= a; rem += a
    return h

def _proj_box_sum(y, n):
    """Project y onto {x: 0<=x<=1, sum(x)=n/2} via bisection."""
    target = n * 0.5
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target: lo = mid
        else: hi = mid
    return np.clip(y - 0.5*(lo+hi), 0.0, 1.0)

def _lssm_obj(z, h0, alpha, mu):
    n = len(h0)
    h = np.clip(z, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    sm = s - s.max()
    val = m + np.log(np.sum(np.exp(sm))) / alpha
    val += mu * (h.sum() - n*0.5)**2
    return val

def _subgrad_shift(h, sh):
    """Gradient of corr[sh-shift] w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0: g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _nesterov(h0, n_steps, topk=5, lr0=0.05, decay=0.999):
    n = len(h0)
    dx = 2.0/n
    x = _proj_box_sum(h0, n)
    y = x.copy()
    best = x.copy()
    best_c5 = _eval(x)
    lr = lr0
    for t in range(n_steps):
        c = _corr_full(y)
        # Top-k weighted subgradient
        idx = np.argpartition(c, -topk)[-topk:]
        w = c[idx]
        w = np.maximum(w, 0)
        if w.sum() < 1e-15: w = np.ones(topk)/topk
        else: w /= w.sum()
        g = np.zeros(n)
        for ii, ki in enumerate(idx):
            sh = (n-1) - ki
            g += w[ii] * _subgrad_shift(y, sh)
        g *= dx
        # Nesterov
        theta = (1 + np.sqrt(1 + 4*t*t)) / (2*(t+1))
        y_new = _proj_box_sum(y - lr*g, n)
        x_new = theta * y_new + (1-theta) * (y_new - x)
        x_new = _proj_box_sum(x_new, n)
        c5_new = _eval(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = x_new.copy()
        x = x_new
        y = y_new
        lr *= decay
    return best, best_c5

def _lbfgsb(h0, alpha, mu, maxiter=2000):

        s = alpha*(c - m)

def _coord_descent(h, n_iter, rng, scales=None):

    h = h.copy()

import numpy as np

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            rem = abs(diff)
            for idx in order:
                if rem < 1e-12:
                    break
                if diff > 0:
                    a = min(rem, 1.0 - h[idx]); h[idx] += a; rem -= a
                else:
                    a = min(rem, h[idx]); h[idx] -= a; rem += a
    return h

def _proj_feas(y, n):
    target = n * 0.5
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _subgrad_shift(h, sh):
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _topk_grad(x, topk):
    n = len(x)
    c = _corr_full(x)
    idx = np.argpartition(c, -topk)[-topk:]
    w = np.maximum(c[idx], 0.0)
    if w.sum() < 1e-15:
        w = np.ones(topk) / topk
    else:
        w /= w.sum()
    g = np.zeros(n)
    for ii, ki in enumerate(idx):
        sh = (n - 1) - ki
        g += w[ii] * _subgrad_shift(x, sh)
    return g * (2.0 / n)

def _lbfgsb(h0, alpha, mu, maxiter):

        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        s = alpha * (c - m)
        sm = s - s.max()
        return m + np.log(np.sum(np.exp(sm))) / alpha + mu * (h.sum() - n * 0.5) ** 2

def _anneal(h0, t0, t1, n_steps, rng):

    h = (h0 > 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = ones[rng.randint(len(ones))]

        h[i], h[j] = 0.0, 1.0

    n = len(h)

    h = h.copy()
    c5c = _eval(h)
    scales = [0.005, 0.002, 0.0005, 0.0001]
    for sc in scales:
        for _ in range(n_iter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _cont_anneal(h0, n_steps, rng, step_scale):
    n = len(h0)
    x = h0.copy()
    c5c = _eval(x)
    best, best_c5 = x.copy(), c5c
    for s in range(n_steps):
        t = step_scale * (1.0 - s / n_steps) + 0.0001
        p = max(1, int(n * 0.02))
        idx = rng.choice(n, p, replace=False)
        x[idx] += t * rng.randn(p)
        x = _proj_feas(x, n)
        c5n = _eval(x)
        if c5n < c5c or rng.rand() < np.exp(-(c5n - c5c) / max(t * 0.01, 1e-7)):
            c5c = c5n
            if c5c < best_c5:
                best_c5, best = c5c, x.copy()
        else:
            x = _repair(h0 if s == 0 else x, n)
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: L-BFGS-B quick polish
    for alpha in [2000, 10000, 50000]:
        if elapsed() > 100:
            break
        h_new = _lbfgsb(best_h, alpha, 1e6, 1500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Momentum subgradient
    if elapsed() < budget_s - 80:
        h_m, c5_m = _momentum_subgrad(best_h, 400, 5, 0.08, 0.85)
        if c5_m < best_c5:
            best_c5, best_h = c5_m, h_m.copy()

    # Phase 3: Intensive L-BFGS-B with increasing alpha
    for alpha in [100000, 500000, 2000000, 10000000]:
        if elapsed() > 350:
            break
        h_new = _lbfgsb(best_h, alpha, 1e8, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        hc = _repair(1.0 - best_h, best_n)
        hc = _lbfgsb(hc, alpha, 1e8, 1500)
        c5c = _eval(hc)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc.copy()

    # Phase 4: Different n exploration
    if elapsed() < budget_s - 200:
        for n_try in [200, 250, 350, 400, 500]:
            if elapsed() > budget_s - 200:
                break
            ht = _repair(_resample(best_h, n_try), n_try)
            ht = _lbfgsb(ht, 100000, 1e7, 2000)
            c5 = _eval(ht)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, ht.copy(), n_try
                ht2, c5m = _momentum_subgrad(ht, 200, 5, 0.05, 0.85)
                if c5m < best_c5:
                    best_c5, best_h = c5m, ht2.copy()

    # Phase 5: Binary pattern + optimize
    if elapsed() < budget_s - 180:
        for pat in range(7):
            if elapsed() > budget_s - 180:
                break
            hb = _make_binary(best_n, rng, pat)
            hb = _lbfgsb(hb, 50000, 1e6, 2000)
            c5 = _eval(hb)
            if c5 < best_c5:
                best_c5, best_h = c5, hb.copy()

    # Phase 6: Nesterov cycles
    if elapsed() < budget_s - 120:
        for _ in range(3):
            if elapsed() > budget_s - 120:
                break
            h_m2, c5_2 = _momentum_subgrad(best_h, 300, 3, 0.04, 0.9)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h_m2.copy()
            h_lb = _lbfgsb(best_h, 5000000, 1e9, 3000)
            c5_lb = _eval(h_lb)
            if c5_lb < best_c5:
                best_c5, best_h = c5_lb, h_lb.copy()

    # Phase 7: Continuous annealing
    if elapsed() < budget_s - 80:
        t_rem = max(budget_s - 80 - elapsed(), 5)
        ns = int(t_rem * 500)
        h_ca, c5_ca = _cont_anneal(best_h, ns, rng, 0.1)
        h_ca = _lbfgsb(h_ca, 2000000, 1e9, 2000)
        c5_ca = _eval(h_ca)
        if c5_ca < best_c5:
            best_c5, best_h = c5_ca, h_ca.copy()

    # Phase 8: Binary anneal + polish
    if elapsed() < budget_s - 50:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _repair(h_bin, best_n)
        t_rem = max(budget_s - 50 - elapsed(), 5)
        ns = int(t_rem * 500)
        h_a, c5_a = _anneal(h_bin, 0.3, 0.001, ns, rng)
        h_a = _lbfgsb(h_a, 5000000, 1e9, 2000)
        c5_a = _eval(h_a)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()

    # Phase 9: Final coordinate descent
    if elapsed() < budget_s - 5:
        h_cd, c5_cd = _coord_desc(best_h, 3000, rng)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
