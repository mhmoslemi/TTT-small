But for the soft-max, it's also convex and smooth in h[j], so we can minimize it with a few Newton steps or bisection.

This gives us a **cyclic coordinate descent** with exact 1D minimization, which should converge faster than random coordinate descent.

```python
def _coord_1d(h, j, alpha, n_corrs_to_check=50):
    """Optimize coordinate j by 1D minimization"""
    n = len(h)
    c_base = np.correlate(h, 1.0 - h, mode='full')
    m_base = c_base.max()
    
    # For coordinate j, c[k] changes linearly:
    # c_new[k] = c_base[k] + h_new[j] * a[k] + b[k]
    # where a[k] and b[k] depend on j
    # Actually: c[k] = sum_i h[i]*(1-h[i+k])
    # Term with h[j]: if j+k < n: h[j]*(1-h[j+k]) contributes to c[k]
    #                  if j-k >= 0: h[j-k]*(1-h[j]) contributes to c[k]
    
    # Let's compute the effect of changing h[j] from old to new:
    old = h[j]
    # delta_c[k] = (new - old) * contrib[k]
    # where contrib[k] is the derivative of c[k] w.r.t. h[j]
    
    contrib = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = (n-1) - k
        # c[k] = sum over valid i of h[i]*(1-h[i+sh])
        # d(c[k])/d(h[j]) = (1-h[j+sh]) if j+sh valid, minus h[j-sh] if j-sh valid
        deriv = 0.0
        if 0 <= j + sh < n:
            deriv += (1.0 - h[j + sh])
        if 0 <= j - sh < n:
            deriv -= h[j - sh]
        contrib[k] = deriv
    
    # Now minimize soft-max of (c_base[k] + (new-old)*contrib[k])
    # over new ∈ [0,1] while maintaining sum constraint
    # The sum constraint: new = old + (target - sum) / ... 
    # Actually, if we change h[j], we need to adjust another coordinate to maintain sum
    # For simplicity, ignore sum constraint in 1D step (it's a small perturbation)
    
    # Bisection on new ∈ [0,1]
    def f(new_val):
        c_new = c_base + (new_val - old) * contrib
        m = c_new.max()
        s = np.clip(alpha * (c_new - m), -500, 500)
        return np.log(float(np.sum(np.exp(s)))) / alpha  # + m
    
    # Actually the objective is m + log(sum(exp(alpha*(c-m))))/alpha
    # = (max_k c[k]) + (1/alpha)*log(sum_k exp(alpha*(c[k]-m)))
    # This is the soft-max approximation
    
    # Bisection
    lo, hi = 0.0, 1.0
    f_lo = f(lo)
    f_hi = f(hi)
    # Need to handle the non-convexity... actually soft-max of linear functions IS convex
    # Use ternary search or bisection on derivative
    # Simpler: just try a few values
    best_val = old
    best_f = f(old)
    for trial in np.linspace(0, 1, 21):
        fv = f(trial)
        if fv < best_f:
            best_f = fv
            best_val = trial
    # Refine around best
    lo2 = max(0, best_val - 0.05)
    hi2 = min(1, best_val + 0.05)
    for trial in np.linspace(lo2, hi2, 21):
        fv = f(trial)
        if fv < best_f:
            best_f = fv
            best_val = trial
    
    return best_val
```

This is O(n) per evaluation (computing c_base change is O(1) per k since it's just linear), and we do 42 evaluations per coordinate. For n=310, that's 42 * 619 ≈ 26K operations per coordinate, and 310 coordinates, so about 8M per full sweep. That's fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _soft_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_sum(h, n):
    target = n / 2.0
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _pgd_nest(h_p, T, n_full, alphas, iters_total, lr_init, seed):
    m = n_full // T
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    x = h_p.astype(float)
    v = np.zeros(T)
    rng = np.random.RandomState(seed)
    lr = lr_init
    iters_per = max(1, iters_total // len(alphas))
    for alpha in alphas:
        beta = 0.75
        for it in range(iters_per):
            y = x + beta * v
            y = np.clip(y, 0.0, 1.0)
            y = _proj_sum(y, len(y))
            h_full = np.tile(y, m)
            g = _soft_grad(h_full, alpha)
            g_p = np.mean(g.reshape(m, T), axis=0)
            v = beta * v - lr * g_p
            x_new = _proj_sum(x + v, T)
            c_new = _eval(np.tile(x_new, m))
            if c_new < best_c:
                best_c = c_new
                best = x_new.copy()
            x = x_new
            beta = min(0.9, 0.7 + 0.2 * it / iters_per)
            if it % 100 == 99 and it < iters_per - 1:
                c_test = _eval(np.tile(x, m))
                if c_test > best_c + 0.001:
                    x = best.copy()
                    v = np.zeros(T)
                    lr *= 0.5
    return best, best_c

        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        return _soft_grad(x, alpha)
    def sum_eq(x):
        return x.sum() - n * 0.5
    def sum_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n

    try:

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return _repair(h0, n)
```
