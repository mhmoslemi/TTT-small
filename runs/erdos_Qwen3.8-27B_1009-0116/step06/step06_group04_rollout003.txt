import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(h, alpha, top_k=None):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    if top_k is not None and top_k < 2*n-1:
        idx = np.argpartition(-w, top_k)[:top_k]
        wk = np.zeros(2*n-1)
        wk[idx] = w[idx]
        wk /= wk.sum()
        w = wk
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        sh = (n-1) - k
        if sh >= 0:
            g[:n-sh] += w[k]*(1.0 - h[sh:])
            if sh > 0: g[sh:] -= w[k]*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k]*(1.0 - h[:n-sp])
            g[:n-sp] -= w[k]*h[sp:]
    return g

def _lbfgs(h0, alpha, n_iter, deadline, top_k=200, lam=1e4):
    n = len(h0)
    def f(x):
        return _sm(x, alpha) + 0.5*lam*(x.sum()-n*0.5)**2
    def g(x):
        return _sm_grad(x, alpha, top_k) + lam*(x.sum()-n*0.5)
    bnds = [(0,1)]*n
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': n_iter, 'ftol': 1e-20, 'gtol': 1e-10, 'maxls': 50})
        return _repair(res.x, n)
    except:
        return _repair(h0, n)

def _adam(h0, alpha_start, alpha_end, n_iter, lr, seed, deadline, top_k=300):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x, best_c = x.copy(), _eval_c5(x)
    for i in range(n_iter):
        if time.time() > deadline: break
        t += 1
        frac = i/max(n_iter-1,1)
        alpha = alpha_start*(alpha_end/alpha_start)**frac
        _sm_grad(x, alpha, top_k)  # to get gradient (waste)
        g = _sm_grad(x, alpha, top_k)
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g*g
        mhat = m/(1-b1**t); vhat = v/(1-b2**t)
        x = x - lr*mhat/(np.sqrt(vhat)+eps)
        x = _repair(x, n)
        c = _eval_c5(x)
        if c < best_c: best_c, best_x = c, x.copy()
    return best_x, best_c

def _paired_cd(h, n, rng, deadline, scale):
    x = h.copy()
    c_cur = _eval_c5(x)
    best_x, best_c = x.copy(), c_cur
    for _ in range(500000):
        if time.time() > deadline or scale < 1e-9: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = scale*rng.randn()
        x[i] += d; x[j] -= d
        x = _repair(x, n)
        c_new = _eval_c5(x)
        if c_new < c_cur:
            c_cur = c_new
            if c_new < best_c: best_c, best_x = c_new, x.copy()
            scale = max(scale*0.9995, 1e-10)
        else:
            x = best_x.copy(); c_cur = best_c
            scale *= 0.93
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h), n_new)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(h, alpha, top_k=None):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argpartition(-w, top_k)[:top_k]
        wk = np.zeros(2 * n - 1)
        wk[idx] = w[idx]
        wk /= wk.sum()
        w = wk
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n - sp])
            g[:n - sp] -= w[k] * h[sp:]
    return g

def _lbfgs_opt(h0, alpha, n_iter, deadline, top_k=400, lam=1e5):
    n = len(h0)
    def f(x):
        return _sm_val(x, alpha) + 0.5 * lam * (x.sum() - n * 0.5) ** 2
    def g(x):
        return _sm_grad(x, alpha, top_k) + lam * (x.sum() - n * 0.5)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': n_iter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _adam_opt(h0, a0, a1, n_iter, lr0, lr1, seed, deadline, top_k=400):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)

    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x, best_c = x.copy(), _eval_c5(x)
    for i in range(n_iter):

            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = a0 * (a1 / a0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if t % 800 == 0:

        g = _sm_grad(x, alpha, top_k)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval_c5(x)
        if c < best_c:
            best_c, best_x = c, x.copy()
    return best_x, best_c

    c_best = _eval_c5(x)
    h_best = x.copy()

        if sh >= 0:
            g[:n - sh] += (1.0 - x[sh:])

        else:
            sp = -sh
            g[sp:] += (1.0 - x[:n - sp])
            g[:n - sp] -= x[sp:]
        lr_i = lr * (1 - i / n_steps) ** 1.5
        x = x - lr_i * g
        x = _repair(x, n)
        c = _eval_c5(x)
        if c < c_best:
            c_best, h_best = c, x.copy()
    return h_best, c_best

            if c_new < best_c:

            scale = max(scale * 0.9997, 1e-11)
        else:
            x = best_x.copy(); c_cur = best_c
            scale *= 0.94
    return best_x, best_c

    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h), n_new)

    rng = np.random.RandomState(seed)
    return _repair((rng.rand(n) < 0.5).astype(float), n)
