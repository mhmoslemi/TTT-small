def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:

            hi = mid

def _grad_lag(h, k):
    """Gradient of c_k with respect to h."""
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _subgrad_weighted(h, n_top=5, eps_frac=0.01):
    """Weighted subgradient using top lags."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    C = c.max()
    # Select top lags
    idx = np.argsort(c)[-n_top:]

    vals = c[idx]
    w = np.exp(-(C - vals) / (C * eps_frac)) if C > 1e-10 else np.ones(n_top)

    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w[i] * _grad_lag(h, k)
    return g * (2.0 / n), C

    def _obj(x):

        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha

        m = c.max()

Actually, for fixed lambda, the problem becomes:
minimize_t sum_mu nu h: t(1 - sum lambda_k) + sum_k lambda_k c_k(h) + nu(sum(h) - n/2) + penalties

If sum lambda_k = 1, the t term vanishes and we minimize a quadratic in h. This is a QP!

So the algorithm is:
1. Start with lambda_k = 1/(2n-1) for all k
2. Solve the QP: minimize sum_k lambda_k c_k(h) + penalties, subject to 0≤h≤1, sum(h)=n/2
3. Update lambda_k ∝ lambda_k * exp(alpha * c_k(h)) (or similar)
4. Repeat

For the QP in step 2, c_k(h) = sum_i h[i](1-h[i+k]) which is quadratic. So sum_k lambda_k c_k(h) is a quadratic form in h. We can write it as h^T A h + b^T h where A and b depend on lambda.

c_k(h) for k = n-1+sh (sh≥0):
= sum_{i=0}^{n-1-sh} h[i+sh] * (1-h[i])
= sum_{i=0}^{n-1-sh} h[i+sh] - sum_{i=0}^{n-1-sh} h[i+sh]*h[i]

The linear part: sum_{i=0}^{n-1-sh} h[i+sh] = sum_{j=sh}^{n-1} h[j]
The quadratic part: -sum_{i=0}^{n-1-sh} h[i]*h[i+sh]

So c_k = (linear in h) - (quadratic in h)

The full objective sum_k lambda_k * c_k is:
= sum_k lambda_k * (linear_k) - sum_k lambda_k * (quadratic_k)

The quadratic part is: -sum_k lambda_k * sum_i h[i]*h[i+sh_k]

This is a quadratic form h^T Q h where Q is symmetric (since h[i]*h[j] = h[j]*h[i]).

For positive lag sh: the term h[i]*h[i+sh] contributes to Q[i, i+sh] and Q[i+sh, i].
For negative lag -sh: the term h[i]*h[i-sh] contributes to Q[i, i-sh] and Q[i-sh, i].

So Q is a banded matrix with bandwidth at most max_sh.

This is solvable as a QP! And CVXPY with GLOP or SCS can handle it.

def _mm_fw(h, n_mm=20, n_fw=5, alpha_mm=10):
    """Multiplicative weights with Frank-Wolfe inner loop."""
    n = len(h)
    n_lags = 2 * n - 1
    lam = np.ones(n_lags) / n_lags
    best = h.copy()
    best_c = _eval(best)
    
    for mm_it in range(n_mm):
        # Run FW inner loop
        for fw_it in range(n_fw):
            c = np.correlate(h, 1.0 - h, mode='full')
            # Compute gradient of weighted sum
            g = np.zeros(n)
            for k in range(n_lags):
                if lam[k] < 1e-12:
                    continue
                sh = (n - 1) - k
                if sh >= 0:
                    g[:n-sh] += lam[k] * (1.0 - h[sh:])
                    if sh > 0:
                        g[sh:] -= lam[k] * h[:n-sh]
                else:
                    sp = -sh
                    g[sp:] += lam[k] * (1.0 - h[:n-sp])
                    g[:n-sp] -= lam[k] * h[sp:]
            # FW vertex
            idx = np.argsort(g)[:n // 2]
            l = np.zeros(n)
            l[idx] = 1.0
            gamma = 2.0 / (fw_it + 2)
            h = (1 - gamma) * h + gamma * l
            h = _repair(h, n)
        
        c = _eval(h)
        if c < best_c:
            best_c = c
            best = h.copy()
        
        # Update lambda
        c_full = np.correlate(h, 1.0 - h, mode='full')
        lam *= np.exp(alpha_mm * (c_full - c_full.max()) / n_lags)
        lam /= lam.sum()
    
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _subgrad_top(h, n_top=5):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-n_top:]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(h, k)
    return g * (2.0 / n), c.max() * 2.0 / n

def _proj_grad(h, n_iter, step0=0.02, n_top=5, n_lag_top=8):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stuck = 0
    for it in range(n_iter):
        g, C = _subgrad_top(best, n_top)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new; best_c = c_new
                step = min(step * 1.1, 0.12)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.65
            stuck += 1
            if stuck > 30:
                rng = np.random.RandomState(it)
                best = _repair(best + 0.003 * rng.randn(n), n)
                best_c = _eval(best)
                step = step0; stuck = 0; v = np.zeros(n)
    return best, best_c

def _fw_opt(h, n_iter, gamma0=0.25, decay=0.999, n_top=5):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    gamma = gamma0
    for it in range(n_iter):
        g, C = _subgrad_top(best, n_top)
        idx = np.argsort(g)[:n // 2]
        l = np.zeros(n)
        l[idx] = 1.0
        h_new = (1 - gamma) * best + gamma * l
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-16:
            best = h_new; best_c = c_new
        gamma = max(0.01, gamma * decay)
    return best, best_c

def _slsqp_smooth(h, alpha, maxiter=2000):
    n = len(h)
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-13:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += w[k] * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= w[k] * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += w[k] * (1.0 - x[:n - sp])
                g[:n - sp] -= w[k] * x[sp:]
        return g * (2.0 / n)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - 0.5 * n
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _mm_weights(h, n_mm=15, n_fw=4, alpha_wt=8.0, n_active=10):
    n = len(h)
    n_lags = 2 * n - 1
    lam = np.ones(n_lags) / n_lags
    best = h.copy()
    best_c = _eval(best)
    for mm in range(n_mm):
        for fw in range(n_fw):
            g = np.zeros(n)
            c_all = np.correlate(h, 1.0 - h, mode='full')
            top_idx = np.argsort(c_all)[-n_active:]
            for k in top_idx:
                wt = lam[k] * (n_lags / max(1.0, np.sum(lam[top_idx])))
                if wt < 1e-12:
                    continue
                g += wt * _grad_lag(h, k)
            idx_v = np.argsort(g)[:n // 2]
            l = np.zeros(n)
            l[idx_v] = 1.0
            gamma = 2.0 / (fw + 2)
            h = (1 - gamma) * h + gamma * l
            h = _repair(h, n)
        c_now = _eval(h)
        if c_now < best_c:
            best_c = c_now
            best = h.copy()
        c_all = np.correlate(h, 1.0 - h, mode='full')
        lam = np.exp(alpha_wt * (c_all - c_all.max()))
        lam /= lam.sum()
    return best, best_c

def _pairwise_polish(h, n_tries, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1, 1]) * rng.uniform(1e-5, 3e-4)
        ni, nj = best[i] + d, best[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best[i], best[j] = ni, nj
            cn = _eval(best)
            if cn < best_c - 1e-16:
                best_c = cn
            else:
                best[i], best[j] = ni - d, nj + d
    return best, best_c

            best_h = h_fw; best_c = c_fw

    # Phase 5: Upsample to 2n and optimize
    if _el() < budget_s - 200:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        h_hi, c_hi = _proj_grad(h_hi, 1200, step0=0.01, n_top=5)
        for alpha in [5000, 50000]:
            if _el() > budget_s - 80:
                break
            h_hi = _slsqp_smooth(h_hi, alpha, 1200)
            c = _eval(h_hi)
            if c < best_c:
                best_c = c; best_h = h_hi; best_n = n1
        if _el() < budget_s - 60:
            h_hi, c_hi = _pairwise_polish(h_hi, 30000, rng)
            if c_hi < best_c:
                best_c = c_hi; best_h = h_hi; best_n = n1

    # Phase 6: Very high alpha SLSQP on best_h
    if _el() < budget_s - 80:
        hc = _slsqp_smooth(best_h, 2000000, 3000)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc; best_n = len(hc)

    # Phase 7: Final projected subgradient polish
    if _el() < budget_s - 40:
        h_pg2, c_pg2 = _proj_grad(best_h, 1500, step0=0.005, n_top=3)
        if c_pg2 < best_c:
            best_h = h_pg2; best_c = c_pg2

    # Phase 8: Pairwise polish
    if _el() < budget_s - 5:
        n_tr = min(int((budget_s - 5 - _el()) / 0.004), 50000)
        best_h, best_c = _pairwise_polish(best_h, n_tr, rng)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
