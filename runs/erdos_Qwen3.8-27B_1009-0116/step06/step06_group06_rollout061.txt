import time

def _repair(h, n):

                diff += a

def _soft_grad(h, alpha=800.0):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()

    g = np.zeros(n)

        else:

            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

    bnds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return np.array([x.sum() - n * 0.5])
    def cjac(x):
        return np.ones((1, n))
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    f0 = _soft_obj(h0, alpha)
    try:

                if c_new < c_cur - 1e-15:
                    c_cur = c_new
                    improved = True

            break

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _soft_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += w[k] * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= w[k] * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return g * dx

def _proj_simple(h, n):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > n * 0.5:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h0, alpha, niter, t_end):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - n * 0.5])
    def _cj(x): return np.ones((1, n))
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    f0 = _soft_obj(h0, alpha)
    try:
        res = minimize(_soft_obj, h0, jac=lambda x: _soft_grad(x, alpha),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': niter, 'ftol': 1e-15, 'disp': False})
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h0, n)

def _sa_search(h_init, n, t_end, rng, T0=0.008, T1=5e-6):
    h = h_init.copy()
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    g = _soft_grad(h, 500.0)
    stale = 0
    it = 0
    while time.time() < t_end - 3:
        it += 1
        t = T0 * (T1 / T0) ** (min(it, 500000) / 500000.0)
        r = rng.rand()
        undo = None

            i1, i2 = rng.randint(n), rng.randint(n)
            if i1 == i2: continue

            if rng.rand() < 0.5:
                d = rng.randn() * sc + 0.15 * (g[i2] - g[i1]) * sc * 6
            else:
                d = rng.randn() * sc
            undo = (0, i1, i2, h[i1], h[i2])
            h[i1] += d; h[i2] -= d
        elif r < 0.46:
            L = int(rng.randint(3, min(12, n // 8)))
            i1, i2 = rng.randint(0, n - L), rng.randint(0, n - L)
            if i1 == i2: continue
            d = rng.randn() * (0.002 + 0.01 * t)
            undo = (1, i1, i2, L, h[i1:i1+L].copy(), h[i2:i2+L].copy())
            h[i1:i1+L] += d; h[i2:i2+L] -= d
        elif r < 0.58:
            L = int(rng.randint(3, min(10, n // 10)))
            i1, i2 = rng.randint(0, n - L), rng.randint(0, n - L)

            h[i1:i1+L], h[i2:i2+L] = h[i2:i2+L].copy(), h[i1:i1+L].copy()
        elif r < 0.72:

            i1 = rng.randint(0, n - L)

            i1, i2 = rng.randint(n), rng.randint(n)

        h = _repair(h, n)

            elif kind in (1, 2):
                h[undo[1]:undo[1]+undo[3]] = undo[4]

            c_cur = _eval(h)
        if it % 4000 == 0:
            g = _soft_grad(h, 500.0)
    return best_h, best_c

def _multi_lag(h, n_rounds, t_end):
    n = len(h); dx = 2.0 / n
    h = h.copy()
    c5 = _eval(h)
    for _r in range(n_rounds):
        if time.time() > t_end - 5: break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n-sh] += (1.0 - h[sh:])
                if sh > 0: g[sh:] -= h[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n-sp])
                g[:n-sp] -= h[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14: break
        g /= gn
        improv = False
        for step in [0.06, 0.03, 0.015, 0.007, 0.003, 0.001]:
            hnew = _proj_simple(h - step * g, n)
            c5n = _eval(hnew)
            if c5n < c5 - 1e-15:
                h = hnew; c5 = c5n; improv = True; break
        if not improv: break
    return h, c5

def _cd_refine(h, n, t_end, rng):

                c_new = _eval(h)
                if c_new < c_cur - 1e-16:
                    c_cur = c_new; improved = True; break
        if not improved: break

def _upsample(h, factor):
    n = len(h)
    out = np.zeros(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out
