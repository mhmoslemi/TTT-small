import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _fwh_to_h(coeffs, n):
    x = np.arange(n) * (2.0 / n)
    h = np.full(n, 0.5)
    K = len(coeffs) // 2
    for j in range(1, K + 1):
        h += coeffs[2 * (j - 1)] * np.cos(np.pi * j * x)
        h += coeffs[2 * (j - 1) + 1] * np.sin(np.pi * j * x)
    return h

def _fwh_obj_from(coeffs, n):
    h = _fwh_to_h(coeffs, n)
    if np.any(h < -0.2) or np.any(h > 1.2):
        return 1.5
    h = _repair(h, n)
    return _eval(h)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            ii = np.arange(n - sh)
            g[ii] += (1.0 - h[ii + sh])
            if sh > 0:
                ii2 = np.arange(sh, n)
                g[ii2] -= h[ii2 - sh]
    else:

            ii = np.arange(sp, n)
            g[ii] += (1.0 - h[ii - sp])
            ii2 = np.arange(n - sp)
            g[ii2] -= h[ii2 + sp]
    return g * dx

def _subgrad_step(h, n, dx, temperature):
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    w = np.exp((ov - ov.max()) / temperature)
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(len(ov)):
        if w[kk] > 1e-8:
            g += w[kk] * _grad_lag(h, kk, dx)
    return g, ov.max()

def _subgrad_descent(h, n, n_iter, t0, budget):

            break
        temp = temps[min(it // (n_iter // 4), 3)]
        g, _ = _subgrad_step(h, n, dx, temp)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

    return h, c5

def _slsqp_lse(x, alpha, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m * dx + dx / alpha * np.log(float(np.sum(np.exp(s))))

def _slsqp_grad(x, alpha, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(len(c)):
        if w[kk] > 1e-13:
            g += w[kk] * _grad_lag(x, kk, dx)
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_slsqp_lse, h, args=(alpha, n, dx), jac=lambda x: _slsqp_grad(x, alpha, n, 2.0/len(x)),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_coeffs(h, n, K):
    x = np.arange(n) * (2.0 / n)
    c = np.zeros(2 * K)
    for j in range(1, K + 1):
        c[2*(j-1)] = (2.0/n) * np.sum(h * np.cos(np.pi * j * x))
        c[2*(j-1)+1] = (2.0/n) * np.sum(h * np.sin(np.pi * j * x))
    return c

                   bounds=bnds, options={'maxiter': max_iter, 'xtol': 1e-12, 'ftol': 1e-14})
    return res.x, float(res.fun)
