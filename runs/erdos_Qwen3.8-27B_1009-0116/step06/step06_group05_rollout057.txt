def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def top_k_subgrad(h, k):
    """Subgradient of (1/k)*sum of top-k correlation values."""
    n = len(h)
    c = corr_values(h)
    # Find top k indices
    if k >= len(c):
        k = len(c)
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    lambda_thresh = top_vals.min()
    
    g = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)  # physical shift

        if s >= 0:

            if s > 0:
                g[s:] -= h[:n-s]
        else:
            p = -s
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] -= h[p:]
    return g / k, c

import numpy as np

import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    thresh = c[top_idx].min() + 1e-15
    g = np.zeros(n)
    cnt = 0
    for ki in top_idx:
        if c[ki] <= thresh:
            continue
        s = ki - (n - 1)
        if s >= 0:
            g[:n - s] += (1.0 - h[s:])
            if s > 0:
                g[s:] -= h[:n - s]
        else:
            p = -s
            g[p:] += (1.0 - h[:n - p])
            g[:n - p] -= h[p:]
        cnt += 1
    if cnt > 0:
        g /= cnt
    return g, c

def lse_obj(h, alpha):
    n = len(h)
    c = corr(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha * (2.0 / n)

def lse_grad(h, alpha):
    n = len(h)
    c = corr(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s_val = k - (n - 1)
        if s_val >= 0:
            g[:n - s_val] += w[k] * (1.0 - h[s_val:])
            if s_val > 0:
                g[s_val:] -= w[k] * h[:n - s_val]
        else:
            p = -s_val
            g[p:] += w[k] * (1.0 - h[:n - p])
            g[:n - p] -= w[k] * h[p:]
    return g * (2.0 / n)

    return h.sum() - len(h) / 2.0

    return np.ones(len(h))

def slsqp_lse(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)
