    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):

        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pmax(c, alpha):
    m = c.max()
    s = np.clip((c - m) * alpha, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _pmax_w(c, alpha):
    m = c.max()
    s = np.clip((c - m) * alpha, -500, 500)
    e = np.exp(s - s.max())
    return e / e.sum()

def _grad_pmax(h, alpha):
    n = len(h)
    c = _corr_vals(h)
    w = _pmax_w(c, alpha)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        j = k - (n - 1)
        if j == 0:
            g += wk * (1.0 - 2.0 * h)
        elif j > 0:
            g[j:] += wk * (1.0 - h[:n - j])
            g[:n - j] -= wk * h[j:]
        else:
            jj = -j
            g[:n - jj] += wk * (1.0 - h[jj:])
            g[jj:] -= wk * h[:n - jj]
    return g * dx

def _proj_bs(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    return _proj_bs(h, target)

def _momentum_descend(h, alpha, t_end, lr=0.02, mu=0.85, max_iter=5000):
    n = len(h)
    target = n / 2.0
    hc = h.copy()
    f_prev = _pmax(_corr_vals(hc), alpha)
    v = np.zeros(n)
    step = lr
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = _grad_pmax(hc, alpha)
        v = mu * v + (1.0 - mu) * g
        accepted = False
        for _bt in range(12):
            ht = _proj_bs(hc - step * v, target)
            ft = _pmax(_corr_vals(ht), alpha)
            if ft <= f_prev + 1e-13:
                hc = ht
                f_prev = ft
                step = min(step * 1.08, 0.15)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 0.25
            v *= 0.5
    return hc

def _slsqp_smooth(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _pmax(_corr_vals(x), alpha)
    def jac(x):
        return _grad_pmax(x, alpha)
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    f0 = obj(h)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})
        if res.fun < f0 - 1e-12:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _coord_polish(h, n, t_end, n_max=200000):
    dx = 2.0 / n
    hc = h.copy()
    c0 = _eval(hc)
    rng = np.random.RandomState(123)
    for _ in range(n_max):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni = hc[i] + d
        nj = hc[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        cn = _eval(hc)
        if cn < c0 - 1e-16:
            c0 = cn
        else:
            hc[i], hc[j] = hc[i] - d, hc[j] + d
    return hc, c0

def _active_refine(h, t_end):
    n = len(h)
    dx = 2.0 / n
    hc = h.copy()
    c0 = _eval(hc)
    for _round in range(50):
        if time.time() > t_end:
            break
        c = _corr_vals(hc)
        top = np.argsort(c)[::-1][:4]
        g = np.zeros(n)
        for kk in top:
            j = kk - (n - 1)
            if j == 0:
                g += (1.0 - 2.0 * hc)
            elif j > 0:
                g[j:] += (1.0 - hc[:n - j])
                g[:n - j] -= hc[j:]
            else:
                jj = -j
                g[:n - jj] += (1.0 - hc[jj:])
                g[jj:] -= hc[:n - jj]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            ht = _proj_bs(hc - st * g, n * 0.5)
            ct = _eval(ht)
            if ct < c0 - 1e-15:
                hc = ht
                c0 = ct
                improved = True
                break
        if not improved:
            break
    return hc, c0

    best_h = h_init.copy()
    best_c = _eval(best_h)
    # Downsample to n/4
    n_c = max(40, n_init // 4)

    h_up = _repair(h_up, n_init)
    c_up = _eval(h_up)
    if c_up < best_c:
        best_h = h_up.copy()
        best_c = c_up
    return best_h, best_c

def _slsqp_polish(h, t_end):

    for alpha in [5000, 20000, 80000, 300000]:

        h = _slsqp_smooth(h, alpha, 1500, t_end - 30)
