    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_adaptive(h, steps, alpha0, alpha_mult, step0, el, budget_s):
    n = len(h)
    target = n * 0.5
    f = _smooth_obj(h, alpha0)
    v = np.zeros(n)
    mu = 0.875
    step = step0
    alpha = alpha0
    for _s in range(steps):
        if el() > budget_s - 20:
            break
        alpha *= alpha_mult
        g = _smooth_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, target)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 5)
        else:
            step *= 0.5
            h = _proj_sum(h_prev - step * v, target)
            f = _smooth_obj(h, alpha)
    return _repair(h, n)

def _multi_lag_descent(h, n_top, max_rounds, el, budget_s):
    n = len(h)
    target = n * 0.5
    c5_cur = _eval(h)
    for _r in range(max_rounds):
        if el() > budget_s - 15:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= (2.0 / n) / n_top
        improved = False
        for step in [0.04, 0.015, 0.007, 0.003, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj_sum(h - step * g, target)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _guided_local_search(h, rng, el, budget_s, n_max):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_max):
        if el() > budget_s - 5:
            break
        mode = rng.randint(3)
        if mode == 0:
            i1 = rng.randint(n)
            i2 = (i1 + 1 + rng.randint(0, 10)) % n
            d = rng.randn() * 4e-5
        elif mode == 1:
            i1 = rng.randint(n)
            i2 = (i1 + 1 + rng.randint(0, 4)) % n
            d = rng.randn() * 1e-5
        else:
            i1 = rng.randint(n)
            i2 = (i1 + 1 + rng.randint(0, 20)) % n
            i3 = (i1 + 1 + rng.randint(0, 40)) % n
            d = rng.randn() * 2e-5
            h[i1] += d
            h[i2] -= 0.7 * d
            h[i3] -= 0.3 * d
            if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1 and 0 <= h[i3] <= 1:
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i1] -= d
                    h[i2] += 0.7 * d
                    h[i3] += 0.3 * d
            else:
                h[i1] -= d
                h[i2] += 0.7 * d
                h[i3] += 0.3 * d
            continue
        h[i1] += d
        h[i2] -= d
        if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i1] -= d
                h[i2] += d
        else:
            h[i1] -= d
            h[i2] += d
    return best_h, best_c

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Low-res multi-start
    n_low = 140
    h_low = _downsample(h0, n0 // n_low)
    h_low = _repair(h_low, n_low)
    for phase_frac in [0, 0.12, 0.25, -0.12]:
        if el() > 45:
            break
        shift = int(phase_frac * n_low)
        h_p = np.roll(h_low, shift) if shift != 0 else h_low.copy()
        h_p = _repair(h_p, n_low)
        h_opt = _slsqp_opt(h_p, 8000, 400, 1e-14)
        c5v = _eval(h_opt)
        h_hi = _repair(_upsample(h_opt, n0 // n_low), n0)
        c5h = _eval(h_hi)
        if c5h < best_c5:
            best_c5 = c5h
            best_h = h_hi.copy()
            best_n = n0

    # Phase 2: Multi-start at full res via phase shifts
    for phase_frac in [0, 0.05, 0.15, -0.08]:
        if el() > 120:
            break
        shift = int(phase_frac * n0)
        h_p = np.roll(best_h, shift) if shift != 0 else best_h.copy()
        h_p = _repair(h_p, n0)
        h_opt = _slsqp_opt(h_p, 10000, 600, 1e-15)
        c5v = _eval(h_opt)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt.copy()

    # Phase 3: Adaptive Nesterov rounds
    for rnd in range(4):
        if el() > budget_s - 250:
            break
        a0 = 3000 * (1 + rnd * 2)
        h_nr = _nesterov_adaptive(best_h.copy(), 350, a0, 1.008, 0.003, el, budget_s)
        c5v = _eval(h_nr)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_nr.copy()

    # Phase 4: Multi-lag descent with various top-k
    for n_top in [3, 5, 2, 7, 4]:
        if el() > budget_s - 150:
            break
        h_ml, c5_ml = _multi_lag_descent(best_h.copy(), n_top, 400, el, budget_s)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 5: SLSQP high alpha
    for alpha in [50000, 200000, 600000]:
        if el() > budget_s - 100:
            break
        h_sp = _slsqp_opt(best_h, alpha, 1500, 1e-16)
        c5sp = _eval(h_sp)
        if c5sp < best_c5:
            best_c5 = c5sp
            best_h = h_sp.copy()

    # Phase 6: Another Nesterov burst
    if el() < budget_s - 90:
        h_nr2 = _nesterov_adaptive(best_h.copy(), 300, 80000, 1.004, 0.002, el, budget_s)
        c5nr = _eval(h_nr2)
        if c5nr < best_c5:
            best_c5 = c5nr
            best_h = h_nr2.copy()

    # Phase 7: Guided local search
    if el() < budget_s - 30:
        n_tries = min(int((budget_s - 30 - el()) / 0.004), 40000)
        best_h, best_c5 = _guided_local_search(best_h, rng, el, budget_s, n_tries)
        best_n = len(best_h)

    # Phase 8: Ultra-fine polish
    if el() < budget_s - 3:
        h = best_h.copy()
        n = len(h)
        c5c = _eval(h)
        n_pol = min(int((budget_s - 3 - el()) / 0.002), 15000)
        for _it in range(n_pol):
            if el() > budget_s - 3:
                break
            i1 = rng.randint(n)
            i2 = (i1 + 1 + rng.randint(0, 3)) % n
            d = 1.5e-6 * rng.randn()
            h[i1] += d
            h[i2] -= d
            if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                c5n = _eval(h)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                        best_n = n
                else:
                    h[i1] -= d
                    h[i2] += d
            else:
                h[i1] -= d
                h[i2] += d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
