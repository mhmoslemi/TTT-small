    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

            diff -= a

        else:

    for alpha in [400, 1500, 6000, 25000, 100000, 400000]:
        if rem() < 260:
            break
        hc = _lbfgs(best_h, alpha, 5000)
        cc = _eval(hc)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 2: Top-k subgradient descent
    if rem() > 240:
        hc, cc = _topk_descent(best_h, 500, t0 + rem() - 235, rng, k_top=8)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 3: Binary snap + re-optimize (explores different basin)
    if rem() > 190:
        hb = _snap_binary(best_h)
        for alpha in [2000, 15000, 80000]:
            if rem() < 170:
                break
            hc = _lbfgs(hb, alpha, 3000)
            cc = _eval(hc)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()
        if rem() > 140:
            hc, cc = _topk_descent(best_h, 250, t0 + rem() - 135, rng, k_top=5)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    # Phase 4: Structural restarts (Thue-Morse, random binary, roll)
    if rem() > 150:
        starts = []
        tm = _thue_morse(best_n)
        starts.append(tm)
        for _rs in range(3):
            starts.append(_random_binary(best_n, rng))
        sh = rng.randint(1, best_n // 8)
        starts.append(_repair(np.roll(best_h, sh), best_n))
        for sp in starts:
            if rem() < 120:
                break
            hc = _lbfgs(sp, 30000, 2000)
            cc = _eval(hc)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()
            if rem() < 110:
                break
            hc, cc = _topk_descent(best_h if cc >= best_c else hc, 150, t0 + rem() - 105, rng, k_top=5)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    # Phase 5: Perturbation + re-optimize (escape local optima)
    if rem() > 120:
        for _tr in range(6):
            if rem() < 100:
                break
            scale = 0.02 * (1 + 0.3 * _tr)
            hp = best_h + scale * rng.randn(best_n)
            hp = _repair(hp, best_n)
            hc = _lbfgs(hp, 50000, 2000)
            cc = _eval(hc)
            if cc < best_c:
                best_c, best_h = cc, hc.copy()

    # Phase 6: High-alpha L-BFGS polish
    if rem() > 70:
        hc = _lbfgs(best_h, 1000000, 6000)
        cc = _eval(hc)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 7: Final top-k descent
    if rem() > 50:
        hc, cc = _topk_descent(best_h, 300, t0 + rem() - 45, rng, k_top=5)
        if cc < best_c:
            best_c, best_h = cc, hc.copy()

    # Phase 8: Micro coordinate descent (ultra-fine)
    if rem() > 15:
        hc = best_h.copy()
        nc = len(hc)
        cc = _eval(hc)
        n_it = min(int((rem() - 12) / 0.002), 50000)
        for _ in range(n_it):
            if rem() < 12:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 1.2e-5 * rng.randn()
            ni, nj = hc[i] + d, hc[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            hc[i], hc[j] = ni, nj
            cn = _eval(hc)
            if cn < cc - 1e-18:
                cc = cn
                if cn < best_c:
                    best_c, best_h = cn, hc.copy()
            else:
                hc[i] -= d
                hc[j] += d

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
