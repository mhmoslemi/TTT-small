    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    sm = x.sum() - n * 0.5
    g += 2.0 * mu * sm
    return g

def _lbfgsb(h, alpha, mu, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-5, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

    def _grad(x):
        n2 = len(x)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n2
        g = np.zeros(n2)
        for k in range(2 * n2 - 1):
            if w[k] < 1e-10:
                continue
            sh = (n2 - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n2 - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n2)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n2)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n2 - sp)
                g[i2] -= wk * x[i2 + sp]
        return g

    def _cfun(x):
        return x.sum() - len(x) * 0.5

    def _cjac(x):
        return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _focused_descent(h, nc, max_rounds, top_k=1, use_momentum=True):
    c5c = _eval(h)
    dx = 2.0 / nc
    momentum = np.zeros(nc)
    beta = 0.7
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        n_top = min(top_k, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        if use_momentum:
            momentum = beta * momentum - g
            direction = momentum
        else:
            direction = -g
        bn = np.linalg.norm(direction)
        if bn < 1e-14:
            break
        direction /= bn
        best_c5 = c5c
        best_h = h
        step = 0.04
        for _ in range(18):
            h_new = _repair(h + step * direction, nc)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                best_h = h_new
            else:
                step *= 0.5
        if best_c5 >= c5c - 1e-15:
            momentum *= 0.5
            if np.linalg.norm(momentum) < 1e-12:
                break
        h = best_h
        c5c = best_c5
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_init(K, n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-0.2, 0.2)
        b = rng.uniform(-0.2, 0.2)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)

def _sawtooth_init(n, rng, period):
    x = np.arange(n)
    phase = rng.uniform(0, 1)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * (x / n * period + phase))
    h += rng.randn(n) * 0.05
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS-B polish (fast)
    if tl() > 180:
        h1 = _lbfgsb(best_h, 5000, 200, 2000, 1e-18)
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Focused momentum descent on worst lag
    if tl() > 180:
        h2 = best_h.copy()
        h2, c5 = _focused_descent(h2, best_n, 400, top_k=1, use_momentum=True)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: SLSQP with equality constraint (more accurate)
    if tl() > 250:
        h3 = _slsqp(best_h, 15000, 1500, 1e-18)
        c5 = _eval(h3)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()
        h3, c5 = _focused_descent(h3, best_n, 200, top_k=1)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 4: L-BFGS-B high temperature + focused
    if tl() > 200:
        for alpha in [50000, 200000]:
            if tl() < 70:
                break
            h4 = _lbfgsb(best_h, alpha, 500, 2000, 1e-18)
            h4, c5 = _focused_descent(h4, best_n, 250, top_k=1)
            if c5 < best_c5:
                best_c5, best_h = c5, h4.copy()

    # Phase 5: Multi-resolution search
    if tl() > 250:
        for n_try in [450, 550, 750, 900]:
            if tl() < 70:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _lbfgsb(h_r, 10000, 300, 1500, 1e-17)
            h_r, c5 = _focused_descent(h_r, n_try, 250, top_k=1)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 6: Multi-start with structured inits
    if tl() > 200:
        starts = []
        starts.append(_fourier_init(4, best_n, rng))
        starts.append(_fourier_init(7, best_n, rng))
        starts.append(_sawtooth_init(best_n, rng, 3))
        starts.append(_sawtooth_init(best_n, rng, 5))
        for h_s in starts:
            if tl() < 50:
                break
            h_s = _lbfgsb(h_s, 8000, 100, 1500, 1e-17)
            h_s, c5 = _focused_descent(h_s, best_n, 200, top_k=1)
            if c5 < best_c5:
                best_c5, best_h = c5, h_s.copy()

    # Phase 7: Perturbation escape from local min
    if tl() > 200:
        for ns in [0.04, 0.02, 0.01]:
            if tl() < 50:
                break
            h_p = _repair(best_h + rng.randn(best_n) * ns, best_n)
            h_p = _lbfgsb(h_p, 30000, 500, 1500, 1e-18)
            h_p, c5 = _focused_descent(h_p, best_n, 250, top_k=1)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 8: SLSQP final polish on best
    if tl() > 80:
        h8 = _slsqp(best_h, 80000, 2000, 1e-18)
        h8, c5 = _focused_descent(h8, best_n, 200, top_k=1)
        if c5 < best_c5:
            best_c5, best_h = c5, h8.copy()

    # Phase 9: Coordinate descent micro-polish
    if tl() > 20:
        h9 = best_h.copy()
        nc = len(h9)
        c5c = _eval(h9)
        for scale in [0.003, 0.0008, 0.0002]:
            if tl() < 10:
                break
            n_it = 4000
            for _it in range(n_it):
                if tl() < 5:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h9[i] + d, h9[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h9[i], h9[j] = ni, nj
                    c5n = _eval(h9)
                    if c5n < c5c - 1e-16:
                        c5c = c5n
                    else:
                        h9[i], h9[j] = ni - d, nj + d
        if c5c < best_c5:
            best_c5, best_h = c5c, h9.copy()

    # Final: one more focused descent
    if tl() > 5:
        hf, c5 = _focused_descent(best_h.copy(), best_n, 100, top_k=1)
        if c5 < best_c5:
            best_c5, best_h = c5, hf.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
