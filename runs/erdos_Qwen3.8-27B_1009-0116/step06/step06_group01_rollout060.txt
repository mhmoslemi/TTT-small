def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k):
    n = len(h)
    dx = 2.0 / n
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        upper = n - lag
        g[:upper] += (1.0 - h[lag:])
        if lag > 0:
            g[lag:] -= h[:n-lag]
    else:
        ell = -lag
        g[ell:] += (1.0 - h[:n-ell])
        g[:n-ell] -= h[ell:]
    return g * dx

def _subgrad_descent(h, n_top, max_iter, t0, budget):

    c5 = _eval(h)
    momentum = np.zeros(len(h))
    lr = 0.05

            break
        lags = _full_lags(h)
        # Top K lags
        top = np.argsort(lags)[-n_top:]

            h = h_new
            c5 = c5_new
            lr = min(lr * 1.02, 0.3)

            if lr < 1e-6:
                lr = 0.05
                momentum *= 0.5

            c5_snap = _eval(h_snap)

    c5 = _eval(h)

    T = 0.5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_lags(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_k(h, k):
    n = len(h)
    dx = 2.0 / n
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        upper = n - lag
        g[:upper] += (1.0 - h[lag:])
        if lag > 0:
            g[lag:] -= h[:n - lag]
    else:
        ell = -lag
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g * dx

def _subgrad(h, n_top, max_iter, t0, rem):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    mom = np.zeros(n)
    lr = 0.04
    for it in range(max_iter):
        if time.time() - t0 > rem - 20:
            break
        lags = _full_lags(h)
        top = np.argsort(lags)[-n_top:]
        g = np.zeros(n)
        for kk in top:
            g += _grad_k(h, kk)
        g /= n_top
        g -= g.mean()
        mom = 0.6 * mom + 0.4 * g
        h_new = _repair(h - lr * mom, n)
        c5n = _eval(h_new)
        if c5n < c5 - 1e-15:
            h, c5 = h_new, c5n
            lr = min(lr * 1.03, 0.25)
        else:
            lr *= 0.65
            if lr < 1e-5:
                lr, mom = 0.04, mom * 0.3
        if it % 400 == 399:
            hs = np.round(h).astype(float)
            hs = _repair(hs, n)
            cs = _eval(hs)
            if cs < c5:
                h, c5 = hs, cs
                mom *= 0.5
    return h, c5

def _binary_sa(hb, max_iter, t0, rem, rng):
    n = len(hb)
    h = hb.copy()
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    T = 0.3
    for it in range(max_iter):
        if time.time() - t0 > rem - 20:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5n = _eval(h)
        d = c5n - c5
        if d < 0 or rng.random() < np.exp(-d / T):
            c5 = c5n
            ones = np.concatenate([ones[ones != i], [j]])
            zeros = np.concatenate([zeros[zeros != j], [i]])
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        if it % 5000 == 4999:
            T = max(0.002, T * 0.92)
    return best_h, best_c5

def _ls_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _ls_jac(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_k(x, k)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol, t0, rem):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_ls_obj, h, jac=_ls_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _ensure_binary(h, n, rng):
    h = np.round(h).astype(float)
    h = _repair(h, n)
    n_ones = int(np.sum(h > 0.5))
    tgt = n // 2
    diff = tgt - n_ones
    if diff > 0:
        zp = np.where(h < 0.5)[0]
        h[zp[:diff]] = 1.0
    elif diff < 0:
        op = np.where(h > 0.5)[0]
        h[op[:diff]] = 0.0
    return h

def _pairwise_cd(h, rng, t0, rem):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for scale in [0.004, 0.001, 0.0002]:
        if time.time() - t0 > rem - 15:
            break
        for _ in range(8000):
            if time.time() - t0 > rem - 15:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c
