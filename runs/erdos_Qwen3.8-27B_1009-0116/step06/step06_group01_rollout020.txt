    s = h.sum()

                else:

                    h[idx] -= a; diff += a

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_alllags(h, dx, top_m):
    """Compute weighted gradient over top-m lags"""
    n = len(h)
    c = _corr(h)
    # Get top-m lags
    idx = np.argsort(c)[-top_m:]
    w = c[idx] - c[idx[0]] + 1e-10  # weights (all positive, largest gets most weight)
    w = w ** 4  # sharpen
    w /= w.sum()
    g = np.zeros(n)
    for j, k in enumerate(idx):

        if sh >= 0:
            if sh < n:

                g[i1] += w[j] * (1.0 - h[i1 + sh])

                g[i1] += w[j] * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[j] * h[i2 + sp]

For the discrete problem with n points at positions x_i = i*2/n, i=0,...,n-1:
The overlap integral for shift k (in grid units) is:
∫ h(x)(1-h(x+k*2/n)) dx ≈ (2/n) * Σ_i h[i] * (1-h[(i+k) mod n])

Wait, is it circular or linear? The problem says h: [0,2] → [0,1]. If we extend h periodically (h(x+2) = h(x)), then the shift wraps around. But the problemdoesn't explicitly say periodic extension.

Looking at the evaluation: C₅ = max(np.correlate(h, 1-h, mode='full') * dx)

np.correlate with mode='full' does a LINEAR correlation (no wrap-around). The result has 2n-1 elements.

So it's NOT circular. The shift k goes from -(n-1) to (n-1), and we only compute the overlap where both h and shift exist.

This means for large shifts, the overlap integral is computed over a shorter interval. For shift s (in grid units), the overlap is computed over n-|s| points:
c[n-1+s] = Σ_{i=0}^{n-1-|s|} h[i] * (1-h[i+|s|]) (for s ≥ 0)
c[n-1-s] = Σ_{i=|s|}^{n-1} h[i] * (1-h[i-|s|]) (for s > 0)

And C₅ = max over all these * (2/n).

OK so this is a linear (non-circular) correlation. This is important because it means the edges of the interval matter.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _grad_focused(h, dx, top_m=5, power=4):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-top_m:]
    w = c[idx] - c[idx[0]] + 1e-8
    w = w ** power
    w /= w.sum()
    g = np.zeros(n)
    for j, k in enumerate(idx):
        g += w[j] * _grad_lag(h, k, dx)
    return g, float(c.max()) * dx

        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def _fista(h0, n, max_iter=5000, top_m=5, power=4, step_init=0.02):
    dx = 2.0 / n
    h = h0.copy()
    hp = h0.copy()
    t = 1.0
    step = step_init
    for it in range(max_iter):
        t_old = t
        y = h + (t_old - 1.0) / (t_old + 1.0) * (h - hp)
        y = np.clip(y, 0, 1)
        g, cmax = _grad_focused(y, dx, top_m, power)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        hp = h.copy()
        candidate = _project_constr(y - step * g, n)
        c_new = _eval(candidate)
        if c_new > cmax + 1e-12:
            step *= 0.5
            candidate = _project_constr(y - step * g, n)
        h = candidate
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
    return h, _eval(h)

def _lsexp_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(x, k, dx)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _focused_descent(h, nc, max_rounds, top_m=3, power=4):
    c5c = _eval(h)
    dx = 2.0 / nc
    for r in range(max_rounds):
        g, _ = _grad_focused(h, dx, top_m, power)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = 0.05
        improved = False
        for _ in range(20):
            h_new = _project_constr(h - step * g, nc)

            if c5n < c5c - 1e-14:
                h = h_new; c5c = c5n; improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, c5c
