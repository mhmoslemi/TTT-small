from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += dx * (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= dx * h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += dx * (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= dx * h[i2 + sp]
    return g

def _top_lags(h, n_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-n_top:]
    return idx, c

def _max_lag_descent(h, n_top, steps, step0, budget_check):
    n = len(h)
    cur = _eval(h)
    step = step0
    stall = 0
    for _s in range(steps):
        if budget_check():
            break
        idx, _ = _top_lags(h, n_top)
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, k)
        g /= n_top
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        accepted = False
        for f in [1.0, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:
            h_new = _proj(h - step * f * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < cur - 1e-14:
                h = h_new
                cur = c5n
                accepted = True
                stall = 0
                break
        if not accepted:
            step *= 0.8
            stall += 1
            if stall > 30:
                break
        else:
            step = min(step * 1.06, step0 * 3)
    return h, cur

def _nesterov(h, alpha, steps, step0, budget_check):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _s in range(steps):
        if budget_check():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        s = np.clip(alpha * (c - c.max()), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-8:
                continue
            g += w[k] * _grad_lag(h, k)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f - 1e-15:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 5)
        else:
            step *= 0.5
            h = _proj(h_prev - step * v, n * 0.5)
            f = _eval(h)
        if step < 1e-10:
            break
    return h, f

def _slsqp(h, alpha, maxiter, ftol, budget_check):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def _cfg(x):
        return x.sum() - n * 0.5
    def _cjg(x):
        return np.ones(n)
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _grd(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            g += w[k] * _grad_lag(x, k)
        return g
    cons = [{'type': 'eq', 'fun': _cfg, 'jac': _cjg}]
    try:
        res = minimize(_obj, h, jac=_grd,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, n_iter, step, rng, budget_check):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    jump = max(1, n // 200)
    for _it in range(n_iter):
        if budget_check():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, jump)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c
