def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_k(k, h, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh+m])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _subgrad_top(h, n, m, dx):
    """Subgradient using soft weights on top-m shifts"""
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-m:]
    c_top = c[top_idx]
    # Soft weights: exp scaling
    temp = max(0.01 * (c_top.max() - c_top.min()), 1e-6)
    w = np.exp((c_top - c_top.max()) / temp)
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        g += w[i] * _grad_k(k, h, n)
    return g * dx

        c_try = _eval(h_try)

        else:

            h_try = _repair(h - step * vel, n)
            c_try = _eval(h_try)
            if c_try < cur:
                h = h_try
                cur = c_try
    
    return best_local, best_local_c

    dx = 2.0 / n

        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

        s = np.clip(alpha * (c - m), -500, 500)

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_k(k, h, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh + m])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _subgrad_h(h, n, m, temp_frac):
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[-m:]
    c_top = c[top]
    rng_c = c_top.max() - c_top.min()
    temp = max(temp_frac * rng_c, 1e-8)
    w = np.exp((c_top - c_top.max()) / temp)
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(top):
        g += w[i] * _grad_k(k, h, n)
    return g * (2.0 / n)

def _momentum_sub(h, n_iter, t_end, m=10, lr=0.06, decay=0.995):
    n = len(h)
    h = h.copy()
    vel = np.zeros(n)
    cur = _eval(h)
    best_h, best_c = h.copy(), cur
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _subgrad_h(h, n, m, 0.05)
        vel = 0.85 * vel + 0.15 * g
        step = lr * (decay ** (it // 50))
        ht = _repair(h - step * vel, n)
        ct = _eval(ht)
        if ct < cur - 1e-15:
            cur, h = ct, ht
            if cur < best_c:
                best_c, best_h = cur, h.copy()
        else:
            ht = _repair(h - 0.3 * step * vel, n)
            ct = _eval(ht)
            if ct < cur - 1e-15:
                cur, h = ct, ht
                if cur < best_c:
                    best_c, best_h = cur, h.copy()
    return best_h, best_c

def _lse_opt(h, alpha, maxiter, t_end):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            g += w[k] * _grad_k(k, x, n)
        return g * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_swap(h, n, max_rounds, t_end):
    h = h.copy()
    h[h > 0.5] = 1.0; h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    rng = np.random.RandomState(99)
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        best_val, best_pair = cur, None
        n_try = min(len(ones) * len(zeros), 8000)
        for _s in range(n_try):
            if len(ones) * len(zeros) <= 30000:
                i = ones[_s // len(zeros)] if _s < len(ones)*len(zeros) else ones[0]
                j = zeros[_s % len(zeros)]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0; h[j] = 1.0
            c2 = _eval(h)
            if c2 < best_val:
                best_val, best_pair = c2, (i, j)
            h[i] = 1.0; h[j] = 0.0
        if best_pair is not None:
            h[best_pair[0]] = 0.0; h[best_pair[1]] = 1.0
            cur = best_val
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            break
    return h, cur

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _perturb_restart(h, best_c, n, rng, t_end):
    hp = h.copy()
    n_swap = max(3, n // 50)
    idx = rng.choice(n, size=2 * n_swap, replace=False)
    for i in range(0, 2 * n_swap, 2):
        a, b = hp[idx[i]], hp[idx[i+1]]
        hp[idx[i]], hp[idx[i+1]] = b, a
    hp = _repair(hp, n)
    cp = _eval(hp)
    if cp < best_c + 0.005:
        hp = _lse_opt(hp, 50000, 3000, t_end - 30)
        cp = _eval(hp)
        if cp < best_c:
            return hp, cp
    return None, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: LSE sweep
    for alpha in [2000, 10000, 50000, 250000]:
        if time.time() > t_end - 380:
            break
        hn = _lse_opt(best_h, alpha, 8000, t_end - 370)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hn.copy(), len(hn)

    # Phase 2: Momentum subgradient (escape local min)
    if time.time() < t_end - 250:
        hn, c5 = _momentum_sub(best_h.copy(), 10000, t_end - 230, m=8, lr=0.05)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hn.copy(), len(hn)

    # Phase 3: Perturbation restarts
    if time.time() < t_end - 400:
        for attempt in range(4):
            if time.time() > t_end - 350:
                break
            hp, cp = _perturb_restart(best_h, best_c5, best_n, rng, t_end - 50)
            if hp is not None and cp < best_c5:
                best_c5, best_h, best_n = cp, hp.copy(), best_n

    # Phase 4: Multi-resolution
    if time.time() < t_end - 350:
        for n_try in [300, 400, 550, 700]:
            if time.time() > t_end - 280:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            for alpha in [30000, 200000]:
                if time.time() > t_end - 270:
                    break
                hr = _lse_opt(hr, alpha, 5000, t_end - 260)
            c5 = _eval(hr)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hr.copy(), n_try
            if time.time() < t_end - 240:
                hr2, c5 = _momentum_sub(hr.copy(), 4000, t_end - 230, m=12, lr=0.04)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, hr2.copy(), n_try

    # Phase 5: High-alpha LSE polish
    if time.time() < t_end - 120:
        for alpha in [800000, 2000000]:
            if time.time() > t_end - 100:
                break
            hn = _lse_opt(best_h, alpha, 12000, t_end - 90)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), len(hn)

    # Phase 6: Binary optimization
    if time.time() < t_end - 70:
        hb = _repair(_resample(best_h, best_n), best_n)
        hb, c5 = _binary_swap(hb, best_n, 800, t_end - 30)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hb.copy(), best_n

    # Phase 7: Final subgradient + LSE
    if time.time() < t_end - 25:
        hn, c5 = _momentum_sub(best_h.copy(), 4000, t_end - 18, m=5, lr=0.03)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hn.copy(), len(hn)
    if time.time() < t_end - 10:
        hf = _lse_opt(best_h, 4000000, 12000, t_end - 4)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hf.copy(), len(hf)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
