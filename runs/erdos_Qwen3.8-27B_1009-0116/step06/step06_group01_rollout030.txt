import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _all_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _subgrad_loop(h, nc, t0, budget, eta0, K_start, K_end, n_iters, momentum):
    dx = 2.0 / nc
    best_c5 = _eval(h)
    best_h = h.copy()
    v = np.zeros(nc)
    for it in range(n_iters):
        if time.time() - t0 > budget - 30:
            break
        ov = _all_overlaps(h)
        K = min(K_start + int((K_end - K_start) * it / max(n_iters, 1)), len(ov))
        worst = np.argsort(ov)[-K:]
        g = np.zeros(nc)
        for kk in worst:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = momentum * v - eta0 * g
        v_n = np.linalg.norm(v)
        if v_n < 1e-14:
            break
        v /= v_n
        h_try = _repair(h - v, nc)
        c5_try = _eval(h_try)
        if c5_try < best_c5 - 1e-16:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
            eta0 = min(eta0 * 1.01, 0.15)
        else:
            eta0 *= 0.85
            if eta0 < 0.001:
                break
    return best_h, best_c5

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        g += w[k] * _grad_lag(x, k, dx)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pair_flip_local(h, nc, rng, n_trials):
    c5c = _eval(h)
    ov = _all_overlaps(h)
    worst_k = int(np.argmax(ov))
    dx = 2.0 / nc
    sh = (nc - 1) - worst_k
    if sh < 0:
        sh = 0
    for _ in range(n_trials):
        i = rng.randint(0, nc - sh if sh < nc else nc)
        j = i + sh if sh > 0 else i
        if i == j:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return h, c5c

def _fourier_init(K, n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-0.25, 0.25)
        b = rng.uniform(-0.25, 0.25)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)
