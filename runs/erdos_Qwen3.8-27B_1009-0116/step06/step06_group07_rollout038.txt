def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

                    h[idx] += a
                    diff -= a
                else:

                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

def _periodic_obj(p, T, m, alpha, pen):
    h = np.tile(p, m)
    c = np.correlate(h, 1.0 - h, mode='full')
    mm = c.max()
    s = np.clip(alpha * (c - mm), -500, 500)
    val = (mm + np.log(float(np.sum(np.exp(s)))) / alpha) * 2.0 / (T * m)
    val += pen * (p.sum() - T * 0.5) ** 2
    return val

def _periodic_grad(p, T, m, alpha, pen):
    n = T * m
    h = np.tile(p, m)
    c = np.correlate(h, 1.0 - h, mode='full')
    mm = c.max()
    s = np.clip(alpha * (c - mm), -500, 500)

        if w[k] < 1e-14: continue

        if sh >= 0:

            if sh > 0: g_full[sh:] -= w[k] * h[:n-sh]
        else:
            sp = -sh
            g_full[sp:] += w[k] * (1.0 - h[:n-sp])

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            idx = int(np.argmax(h))
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(x)
    g = np.zeros(n)

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = _grad_c(x, w) * (2.0 / n)
    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    return _lse_fun(h, alpha, pen)

    c = np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    while abs(d) > 1e-12:
        if d > 0:
            i = int(np.argmax(1.0 - h))
            a = min(d, 1.0 - h[i])
            h[i] += a
            d -= a
        else:
            i = int(np.argmax(h))
            a = min(-d, h[i])
            h[i] -= a
            d += a
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_corr(x, w):
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            g[off:n] += w[k] * (1.0 - x[:n - off])
            if off < n:
                g[:n - off] -= w[k] * x[off:n]
        else:
            p = -off
            g[:n - p] += w[k] * (1.0 - x[p:n])
            g[p:n] -= w[k] * x[:n - p]
    return g

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w, m

def _lse_fun(x, alpha, pen):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    w, m = _make_weights(c, alpha)
    return (m + np.log(float(np.sum(np.exp(np.clip(alpha*(c-m), -500, 500)))) / alpha)) * (2.0/n) + pen*(x.sum()-n*0.5)**2

    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    w, _ = _make_weights(c, alpha)
    g = _grad_corr(x, w) * (2.0/n)
    g += 2.0*pen*(x.sum()-n*0.5)
    return g

    return _lse_fun(np.tile(x, m), alpha, pen)

def _pgrad(x, T, m, alpha, pen):
    n = T*m
    h = np.tile(x, m)
    c = np.correlate(h, 1.0-h, mode='full')
    w, _ = _make_weights(c, alpha)
    gf = _grad_corr(h, w)*(2.0/n) + 2.0*pen*(h.sum()-n*0.5)
    gp = np.zeros(T)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    while abs(d) > 1e-12:
        if d > 0:
            i = int(np.argmax(1.0 - h))
            a = min(d, 1.0 - h[i]); h[i] += a; d -= a
        else:
            i = int(np.argmax(h))
            a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _eval(h):

def _gc(x, w):
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        o = k - (n - 1)

            g[o:] += w[k] * (1.0 - x[:n-o])

        else:
            p = -o
            g[:n-p] += w[k] * (1.0 - x[p:])
            g[p:] -= w[k] * x[:n-p]
    return g
