```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _grad_top(h, top_indices, dx):
    """Gradient of sum of c_k for k in top_indices"""
    n = len(h)
    g = np.zeros(n)
    for k in top_indices:
        sh = (n - 1) - k
        if sh == 0:
            g += dx * (1.0 - 2.0 * h)
        elif sh > 0:
            g[0:n-sh] += dx * (1.0 - h[sh:n])
            g[sh:n] -= dx * h[0:n-sh]
        else:
            sp = -sh
            g[sp:n] += dx * (1.0 - h[0:n-sp])
            g[0:n-sp] -= dx * h[sp:n]
    return g

def _subgrad_descent(h, n_iter, top_m, dx, t_end, step_init=0.1):
    """Projected subgradient descent on top-m correlations"""
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    step = step_init
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[-top_m:]
        g = _grad_top(h, top_idx, dx)
        
        # Backtracking line search
        best_h = h.copy()
        best_c = cur
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * factor
            h_try = h - s * g
            h_try = _repair(h_try, n)
            c_try = _eval(h_try)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
        h = best_h
        cur = best_c
        if _eval(h) == cur:  # no improvement
            step *= 0.7
    
    return h, cur
```

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

        hr, c5 = _subgrad(hr, 100000, 5, t_end - 100,
                          step_init=0.05, decay=0.98)
        if c5 < best_c5:
            best_c5 = c5; best_h = hr; best_n = n_try

    # Phase 4: Coordinate descent polish
    if time.time() < t_end - 60:
        h4, c5 = _coord_descent(best_h, 3, t_end - 50)
        if c5 < best_c5:
            best_c5 = c5; best_h = h4; best_n = len(h4)

    # Phase 5: Final SLSQP high-alpha
    if time.time() < t_end - 25:
        hf = _slsqp(best_h, 2000000, 5000, t_end - 20)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5 = c5; best_h = hf; best_n = len(hf)

    # Phase 6: Tiny-step subgradient
    if time.time() < t_end - 10:
        h6, c5 = _subgrad(best_h, 50000, 2, t_end - 5,
                          step_init=0.005, decay=0.995, use_avg=True)
        if c5 < best_c5:
            best_c5 = c5; best_h = h6

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
