import time

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

    if abs(s - target) > 1e-9:

        diff = target - h.sum()
        if abs(diff) > 1e-12:

            for idx in order:

                else:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lssm(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + (float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha)

def _lssm_g(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

import time

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lssm(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + (float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha)

def _lssm_g(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

            A_ub[k, sp:] = g[:n-sp]
        A_ub[k, n] = -1.0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lssm(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + (float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha)

def _lssm_g(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, n, maxiter=5000):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lssm, h, jac=_lssm_g, args=(alpha, n),
                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lp_solve(A, b_ub, A_eq, b_eq, n):
    nvar = n + 1
    c = np.zeros(nvar); c[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_opt_h(g, n):
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        sh = (n-1) - k
        if sh >= 0:
            A[k, :n-sh] = g[sh:n]
        else:
            A[k, -sh:] = g[:n-sp] if (sp := -sh) > 0 else g
        A[k, n] = -1.0
    b = np.zeros(2*n-1)
    Ae = np.zeros((1, n+1)); Ae[0, :n] = 1.0
    return _lp_solve(A, b, Ae, np.array([n/2.0]), n)

def _lp_opt_g(h, n):
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        sh = (n-1) - k
        if sh >= 0:
            A[k, sh:] = h[:n-sh]
        else:
            sp = -sh
            A[k, :n-sp] = h[sp:]
        A[k, n] = -1.0
    b = np.zeros(2*n-1)
    Ae = np.zeros((1, n+1)); Ae[0, :n] = 1.0
    return _lp_solve(A, b, Ae, np.array([n/2.0]), n)

def _alt_lp(h, n, max_rounds=300, time_limit=50):
    h = _repair(h.copy(), n)
    g = 1.0 - h
    best_c5 = _eval(h)
    best_h = h.copy()
    t0 = time.time()
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit: break
        h_new = _lp_opt_h(g, n)
        if h_new is None: break
        h_new = _repair(h_new, n)
        g_new = _lp_opt_g(h_new, n)
        if g_new is None: break
        g_new = _repair(g_new, n)
        h2 = _repair(1.0 - g_new, n)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()
        h = h2
        g = 1.0 - h
    return best_h, best_c5

def _anneal_binary(h, t0, t1, n_steps, rng):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5 = _eval(h)
    best_c5 = c5
    best_h = h.copy()
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        if h[i] == h[j]: continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c5n = _eval(h)
        if c5n < c5 or rng.rand() < np.exp(-(c5n - c5) / max(t, 1e-9)):
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5; best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return best_h, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _shake_refine(h, n, rng, alpha, n_kicks=5):
    best_c5 = _eval(h)
    best = h.copy()
    for _ in range(n_kicks):
        mask = rng.random(n) < 0.08
        h_k = h.copy()
        h_k[mask] = rng.random(mask.sum())
        h_k = _repair(h_k, n)

        c5 = _eval(h_k)
        if c5 < best_c5:
            best_c5 = c5; best = h_k.copy()
    return best, best_c5
