                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _overlap_scaled(h):

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _softmax_grad(h, temp):
    """Compute gradient with softmax weights over lags."""
    n = len(h)
    ov = _overlap_scaled(h)
    # Softmax weights
    max_ov = ov.max()
    log_w = (ov - max_ov) / temp
    log_w = np.clip(log_w, -500, 0)
    w = np.exp(log_w)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(ov)):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_lag(h, k, dx)
    return g, w

    """Armijo backtracking line search."""
    n = len(h)
    step = 0.5
    c1 = 1e-4
    for _ in range(max_iter):

        c5n = _eval(h_new)
        g_dot = np.dot(g, h_new - h)
        if c5n <= c5c + c1 * step * g_dot:

        step *= 0.5
    return h, c5c

    """Log-sum-exp objective and gradient."""
    n = len(x)
    ov = _overlap(x)  # already scaled by dx
    m = ov.max()
    s = np.clip(alpha * (ov - m), -500, 500)
    obj = m + np.log(float(np.sum(np.exp(s)))) / alpha
    # Gradient
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(ov)):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            if sp < n:
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
    return obj, g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj_and_grad, h, jac=True, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

The overlap at shift s is:

Note: c[s] + c[-s] = Σ_{i=0}^{s-1} h[i+s]*(1-h[i]) + Σ_{i=s}^{n-1} h[i+s]*(1-h[i]) + Σ_{i=0}^{s-1} h[i]*(1-h[i+s])

c[-s] counts pairs (i, i+s) with h[i]=1, h[i+s]=0 for i=0..s-1

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _overlap(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-11:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]

            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < _lsexp_obj(h, alpha):
            return _repair(res.x, n)
    except Exception:
        pass

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _topk_descent(h, nc, K_max, n_iter, use_momentum=True):
    c5c = _eval(h)
    dx = 2.0 / nc
    v = np.zeros(nc) if use_momentum else None
    for it in range(n_iter):
        ov = _overlap(h)
        thr = ov.max() - 0.002
        sig = np.where(ov > thr)[0]
        K = min(K_max, len(sig))
        if K == 0:
            break
        idx = np.argsort(ov)[-K:]
        g = np.zeros(nc)
        for kk in idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        if use_momentum and v is not None:
            v = 0.85 * v - g
            direction = v
        else:
            direction = -g
        step = 0.12
        improved = False
        for _ in range(30):
            h_new = _repair(h + step * direction, nc)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
            step *= 0.6
        if not improved:
            if use_momentum and v is not None:
                v *= 0.5
            else:
                break
    return h, c5c

    h = _repair(h_init, nb)
    h = (h > 0.5).astype(float)
    h = _repair(h, nb)
    c5c = _eval(h)
    temp = 0.005

        h2 = _repair(h, nb)
        c5n = _eval(h2)
        if c5n < c5c or (c5n - c5c) < temp * rng.randn():
            c5c = c5n
            h = h2
        else:
            h[i] = 1.0 - h[i]
    return h, c5c

def _fine_cd(h, nc, rng, scales, n_iter_total):
    c5c = _eval(h)
    n_iter = max(n_iter_total // len(scales), 500)
    for scale in scales:
        for _ in range(n_iter):
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:

            d = scale * rng.randn()
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _binary_snap(h, nc):
    h2 = h.copy()
    h2[h2 > 0.92] = 1.0
    h2[h2 < 0.08] = 0.0
    h2 = _repair(h2, nc)
    return h2
