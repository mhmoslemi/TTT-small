            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new
            step *= 1.2  # try larger step
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h, best_c5

def _focused_descent(h, n, max_rounds, top_k_start=3, top_k_max=7):
    c5c = _eval(h)
    dx = 2.0 / n
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        # Adaptive top-k
        frac = (r + 1) / max_rounds
        top_k = min(top_k_start + int(frac * (top_k_max - top_k_start)), top_k_max, len(ov))
        top_idx = np.argsort(ov)[-top_k:]
        g = np.zeros(n)
        for kk in top_idx:
            g += _lag_grad(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new, c5n = _line_search(h, g, n, c5c)
        if c5n >= c5c - 1e-15:
            # Try single worst
            g1 = _lag_grad(h, top_idx[-1], dx)
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14:
                break
            g1 /= gn1
            h_new, c5n = _line_search(h, g1, n, c5c)
            if c5n >= c5c - 1e-15:
                # Try top-2
                if top_k >= 2:
                    g2 = _lag_grad(h, top_idx[-1], dx) + _lag_grad(h, top_idx[-2], dx)
                    gn2 = np.linalg.norm(g2)
                    if gn2 < 1e-14:
                        break
                    g2 /= gn2
                    h_new, c5n = _line_search(h, g2, n, c5c)
                    if c5n >= c5c - 1e-15:
                        break
        h = h_new
        c5c = c5n
    return h, c5c

def _lsexp_obj(x, alpha):

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s_shifted = s - s.max()
    w = np.exp(s_shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            if sp < n:
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
    return g

    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):

    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _micro_cd(h, n, rng, n_iter):
    """Very fine coordinate descent."""
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        for delta in [0.001, -0.001, 0.003, -0.003]:
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
            # Fix sum
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                # Distribute to a random other point
                j = rng.randint(n)
                if j != i:
                    h[j] = np.clip(h[j] + diff, 0, 1)
                    h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                break
            else:
                h[i] -= delta
        else:
            continue
    return h, c5c

            h_r = _slsqp(h_r, 20000, 2000, 1e-17)
            h_r, c5 = _focused_descent(h_r, n_try, 300, 4, 8)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 9: Fourier multi-start
    if tl() > 250:
        for K in [4, 7, 10, 14]:
            if tl() < 55:
                break
            for _tr in range(3):
                h_f = _fourier_init(K, best_n, rng)
                h_f = _slsqp(h_f, 12000, 1200, 1e-15)
                h_f, c5 = _focused_descent(h_f, best_n, 200, 3, 7)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_f.copy()

    # Phase 10: Perturbation restarts
    if tl() > 200:
        for _rep in range(6):
            if tl() < 45:
                break
            noise = rng.randn(best_n) * 0.006
            h_p = _repair(best_h + noise, best_n)
            h_p, c5 = _focused_descent(h_p, best_n, 200, 3, 7)
            h_p = _slsqp(h_p, 80000, 1500, 1e-18)
            c5 = _eval(h_p)
            h_p, c5 = _focused_descent(h_p, best_n, 200, 5, 10)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 11: Binary init + optimize
    if tl() > 120:
        h_bb = (best_h > 0.5).astype(float)
        h_bb = _repair(h_bb, best_n)
        h_bb = _slsqp(h_bb, 40000, 2000, 1e-17)
        h_bb, c5 = _focused_descent(h_bb, best_n, 250, 4, 9)
        if c5 < best_c5:
            best_c5, best_h = c5, h_bb.copy()

    # Phase 12: Pairwise CD polish
    if tl() > 40:
        for scale in [0.004, 0.001, 0.0003]:
            if tl() < 15:
                break
            h_cd, c5 = _pairwise_cd(best_h.copy(), best_n, rng, scale, 5000)
            if c5 < best_c5:
                best_c5, best_h = c5, h_cd.copy()

    # Phase 13: Final focused descent
    if tl() > 15:
        h_fin, c5 = _focused_descent(best_h.copy(), best_n, 150, 5, 10)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    # Phase 14: Final SLSQP
    if tl() > 10:
        h_fin = _slsqp(best_h, 150000, 1500, 1e-18)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
