def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    if budget_s - (time.time() - t0) > 20:
        hf = _slsqp(bh, 2000000, 8000, 1e-21)
        c5 = _eval(hf)
        if c5 < bc:
            bc, bh, bn = c5, hf.copy(), len(hf)
        if budget_s - (time.time() - t0) > 15:
            hf2 = bh.copy()
            hf2, c5 = _reweight(hf2, 2000, 25.0, t0 + budget_s - (time.time() - t0) - 5)
            if c5 < bc:
                bc, bh, bn = c5, hf2.copy(), len(hf2)

    bh = _repair(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)
