    n = len(h)
    c = _corr(h)
    idx = np.argpartition(c, -topK)[-topK:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * dx

def _sm_grad(h, alpha, dx):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n - sp])
            g[:n - sp] -= w[k] * h[sp:]
    return g * dx

def _nesterov_opt(h, n, t_end, topK=15, n_steps=3000, kick_every=500, kick_scale=0.005):
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    mu = 0.8
    step = 0.015
    c5_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    stuck_count = 0
    for s in range(n_steps):
        if time.time() > t_end:
            break
        if s > 0 and s % kick_every == 0:
            h_cur += kick_scale * np.random.randn(n)
            h_cur = _repair(h_cur, n)
            v *= 0.3
            c5_cur = _eval(h_cur)
        tk = min(topK, topK // 2 + (topK - topK // 2) * (s / max(1, n_steps)))
        g = _topk_grad(h_cur, max(3, int(tk)), dx)
        v = mu * v + (1 - mu) * g
        # Nesterov lookahead
        h_look = _proj(h_cur - step * v, n * 0.5)
        g_look = _topk_grad(h_look, max(3, int(tk)), dx)
        v = mu * v + (1 - mu) * g_look
        gn = np.linalg.norm(v)
        if gn < 1e-15:
            break
        accepted = False
        for _trial in range(12):
            h_new = _proj(h_cur - step * v, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-17:
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                step = min(step * 1.03, 0.25)
                stuck_count = 0
                accepted = True
                break
            step *= 0.6
        if not accepted:
            step *= 0.4
            stuck_count += 1
            if stuck_count > 100:
                h_cur += 0.02 * np.random.randn(n)
                h_cur = _repair(h_cur, n)
                v *= 0.1
                c5_cur = _eval(h_cur)
                step = 0.01
                stuck_count = 0
    return best_h, best_c5

def _slsqp_opt(h, alpha, t_end, maxit=3000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    def objg(x):
        return _sm_grad(x, alpha, 2.0 / n)
    try:
        f0 = obj(h)
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-18, 'disp': False})
        if time.time() > t_end:
            return _repair(h, n)
        hx = _repair(res.x, n)
        return hx
    except Exception:
        return _repair(h, n)

def _sa_coord(h, n, t_end, n_iter, rng):
    best_h = h.copy()
    best_c5 = _eval(h)
    h_cur = h.copy()
    c5_cur = best_c5
    for i in range(n_iter):
        if time.time() > t_end:
            break
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        frac = i / max(1, n_iter - 1)
        scale = 0.002 * (1.0 - frac) + 0.0002
        d = rng.randn() * scale
        h_cur[i1] += d
        h_cur[i2] -= d
        h_cur = _repair(h_cur, n)
        c5_new = _eval(h_cur)
        t_ann = 0.003 * (0.00005 / 0.003) ** frac
        if c5_new < c5_cur or rng.rand() < np.exp(-(c5_new - c5_cur) / max(t_ann, 1e-12)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        else:
            h_cur[i1] -= d
            h_cur[i2] += d
            h_cur = _repair(h_cur, n)
    return best_h, best_c5

def _perturb(h, n, rng, scale):
    return _repair(h + scale * rng.randn(n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov momentum, progressive top-k
    h_opt, c5 = _nesterov_opt(best_h, best_n, t0 + rem() - 320, topK=20, n_steps=4000, kick_every=600, kick_scale=0.004)
    if c5 < best_c5:
        best_c5, best_h = c5, h_opt.copy()

    # Phase 2: SLSQP with increasing alpha
    for alpha in [500, 2000, 10000, 40000, 120000]:
        if rem() < 260:
            break
        h_new = _slsqp_opt(best_h, alpha, t0 + rem() - 259, maxit=4000)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()

    # Phase 3: Nesterov with different topK values
    for tk in [8, 35, 60]:
        if rem() < 220:
            break
        h_opt, c5 = _nesterov_opt(best_h, best_n, t0 + rem() - 219, topK=tk, n_steps=2500, kick_every=400, kick_scale=0.003)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 4: Upsample to 2x and optimize
    if rem() > 380:
        n1 = best_n * 2
        fac = n1 // best_n
        h_hi = np.repeat(best_h, fac)
        h_hi = np.convolve(h_hi, np.ones(fac) / fac, mode='same')
        h_hi = _repair(h_hi, n1)
        for alpha in [500, 5000, 30000]:
            if rem() < 340:
                break
            h_hi = _slsqp_opt(h_hi, alpha, t0 + rem() - 339, maxit=2500)
            c5n = _eval(h_hi)
            if c5n < best_c5:
                best_c5, best_h, best_n = c5n, h_hi.copy(), n1
        if rem() > 300:
            h_opt, c5 = _nesterov_opt(best_h, best_n, t0 + rem() - 299, topK=25, n_steps=2000, kick_every=400, kick_scale=0.003)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt.copy()

    # Phase 5: Perturbed restarts
    if rem() > 160:
        for _r in range(6):
            if rem() < 145:
                break
            for sc in [0.03, 0.08, 0.15]:
                if rem() < 140:
                    break
                h_p = _perturb(best_h, best_n, rng, sc)
                h_opt, c5 = _nesterov_opt(h_p, best_n, t0 + rem() - 139, topK=20, n_steps=800, kick_every=300, kick_scale=0.002)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_opt.copy()

    # Phase 6: SA coordinate
    if rem() > 90:
        n_it = min(int((rem() - 80) / 0.003), 50000)
        h_sa, c5_sa = _sa_coord(best_h, best_n, t0 + rem() - 80, n_it, rng)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()

    # Phase 7: Final SLSQP polish
    if rem() > 55:
        h_new = _slsqp_opt(best_h, 200000, t0 + rem() - 50, maxit=6000)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()

    # Phase 8: Final Nesterov with small top-k
    if rem() > 30:
        h_opt, c5 = _nesterov_opt(best_h, best_n, t0 + rem() - 25, topK=10, n_steps=3000, kick_every=800, kick_scale=0.002)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt.copy()

    # Phase 9: Ultra-fine coordinate descent
    if rem() > 15:
        h_cur = best_h.copy()
        nc = len(h_cur)
        c5c = _eval(h_cur)
        n_tries = min(int((rem() - 12) / 0.0015), 120000)
        for _it in range(n_tries):
            if rem() < 12:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_cur[i], h_cur[j] = ni, nj
            c5n = _eval(h_cur)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h_cur.copy()
            else:
                h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
