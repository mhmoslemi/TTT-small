import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(x, target):
    n = len(x)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_k(h, k, n, dx):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(max(0, n - sh))
        g[i1] = (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] = (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        if w[kk] < 1e-14:
            continue
        sh = (n - 1) - kk
        wk = w[kk] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_pen(x, n, mu):
    return _lse_obj(x[0], x[1], n) + mu * (x[2:].sum() - n * 0.5) ** 2

def _lbfgs_opt(h, alpha, n, maxiter, t_end):
    n = len(h)
    mu = 1e4
    def obj(x):
        return _lse_obj(x, alpha, n) + mu * (x.sum() - n * 0.5) ** 2
    def grad(x):
        return _lse_grad(x, alpha, n) + 2.0 * mu * (x.sum() - n * 0.5)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-6,
                                'maxfun': maxiter * 2, 'iprint': 0})
        if time.time() > t_end:
            return h
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _worst_descent(h, n_iter, t_end, top_m=1):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur = _eval(h)
    lr = 0.08
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for kk in top_idx:
            g += _grad_k(h, kk, n, dx)
        g /= top_m
        moved = False
        for s in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            ht = _project(h - s * g, n * 0.5)
            ct = _eval(ht)
            if ct < cur - 1e-15:
                h, cur = ht, ct
                moved = True
                break
        if not moved:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h, cur

def _reweight_descent(h, n_iter, top_m, t_end):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur = _eval(h)
    lr = 0.06
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_m:]
        cvals = c[top_idx]
        rng = cvals.max() - cvals.min()
        if rng > 1e-12:
            w = np.exp(8.0 * (cvals - cvals.min()) / rng)
        else:
            w = np.ones(top_m)
        w /= w.sum()
        g = np.zeros(n)
        for i, kk in enumerate(top_idx):
            g += w[i] * _grad_k(h, kk, n, dx)
        moved = False
        for s in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            ht = _project(h - s * g, n * 0.5)
            ct = _eval(ht)
            if ct < cur - 1e-15:
                h, cur = ht, ct
                moved = True
                break
        if not moved:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pair_swap(h, n, max_rounds, t_end):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h >= 1.0 - 1e-9)[0]
    zeros = np.where(h <= 1e-9)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, cur
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in ones.copy():
            for j in zeros.copy():
                h[i] = 0.0; h[j] = 1.0
                c2 = _eval(h)
                if c2 < cur - 1e-13:
                    cur = c2
                    improved = True
                    ones = np.where(h >= 1.0 - 1e-9)[0]
                    zeros = np.where(h <= 1e-9)[0]
                    break
                else:
                    h[i] = 1.0; h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur
