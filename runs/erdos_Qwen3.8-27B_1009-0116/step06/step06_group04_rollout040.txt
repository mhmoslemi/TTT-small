def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')

def _top_shifts(h, K=10):

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -K)[-K:]
    idx = idx[np.argsort(c[idx])[::-1]]
    return idx, c

        wk = w[ki] * dx

    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

    m = n // 2

    for k in range(1, n):

        ones_idx = np.where(x_binary == 1)[0]

        order = np.argsort(-x[zeros_idx])
        for idx in order[:target - n_ones]:
            x_binary[idx] = 1
    return x_binary

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n), c

def _eval_only(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _focused_grad(h, c, K, dx):
    n = len(h)
    m = c.max()
    top = np.argpartition(c, -K)[-K:]
    w = np.exp(-500.0 * (c[top] - m))
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(K):
        if w[ki] < 1e-10:
            continue
        k = top[ki]
        sh = (n - 1) - k
        wk = w[ki] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _adam_focused(h0, n_iter, lr_s, lr_e, K, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy()
    mv = np.zeros(n); vv = np.zeros(n)
    t = 0; b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy(); best_c = _eval_only(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        lr = lr_s * (lr_e / lr_s) ** frac
        c_val = np.correlate(x, 1.0 - x, mode='full')
        g = _focused_grad(x, c_val, K, 2.0 / n)
        mv = b1 * mv + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        mhat = mv / (1 - b1**t)
        vhat = vv / (1 - b2**t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c_now = _eval_only(x)
        if c_now < best_c:
            best_c = c_now; best_x = x.copy()
    return best_x, best_c

def _binary_swap(h0, n, rng, n_iters, deadline, focus_top=6):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    c_cur, c_arr = _eval(x)
    best_x = x.copy(); best_c = c_cur
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    # Identify positions contributing to top shifts
    top_k = np.argpartition(c_arr, -focus_top)[-focus_top:]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Rebuild focus positions
        top_k = np.argpartition(c_arr, -focus_top)[-focus_top:]
        focus_pos = set()
        for k_idx in top_k:
            sh = (n - 1) - k_idx
            if sh >= 0:
                for ii in range(n - sh):
                    if x[ii] == 1 and x[ii + sh] == 0:
                        focus_pos.add(ii)
            else:
                sp = -sh
                for ii in range(sp, n):
                    if x[ii] == 1 and x[ii - sp] == 0:
                        focus_pos.add(ii)
        best_swap = None; best_new_c = c_cur
        n_try = min(10, max(3, n // 40))
        for _ in range(n_try):
            # 60% focus, 40% random
            if focus_pos and rng.random() < 0.6:
                i = list(focus_pos)[rng.randint(len(focus_pos))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            x[i] = 0; x[j] = 1
            c_new, c_a = _eval(x)
            if c_new < best_new_c:
                best_new_c = c_new; best_swap = (i, j); best_c_a = c_a
            else:
                c_a = None
            x[i] = 1; x[j] = 0
        if best_swap is not None:
            i, j = best_swap
            x[i] = 0; x[j] = 1
            c_cur = best_new_c
            if c_cur < best_c:
                best_c = c_cur; best_x = x.copy()
            c_arr = best_c_a
            ones = np.where(x == 1)[0]
            zeros = np.where(x == 0)[0]
    return best_x, best_c

def _paired_cd(h, n, rng, deadline, scale_init):
    x = h.copy()
    c_cur = _eval_only(x)
    best_x = x.copy(); best_c = c_cur
    scale = scale_init
    for _ in range(300000):
        if time.time() > deadline:
            break
        if scale < 1e-9:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        x[i] += d; x[j] -= d
        x = _repair(x, n)
        c_new = _eval_only(x)
        if c_new < c_cur - 1e-16:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
            scale = max(scale * 0.9995, 1e-10)
        else:
            x[i] -= min(d, x[i]); x[j] += min(d, x[i])
            x = _repair(x, n)
            c_cur = _eval_only(x)
            scale *= 0.93
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _spectral_binary(n, seed):
    rng = np.random.RandomState(seed)
    spectrum = np.zeros(n, dtype=complex)
    for k in range(1, n):
        ang = rng.uniform(0, 2 * np.pi)
        r = rng.uniform(0.3, 1.0)
        spectrum[k] = r * np.exp(1j * ang)
    x = np.real(np.fft.ifft(spectrum))
    xb = (x > np.median(x)).astype(float)
    n_ones = int(xb.sum())
    target = n // 2
    if n_ones > target:
        oi = np.where(xb == 1)[0]
        order = np.argsort(x[oi])
        for idx in order[:n_ones - target]:
            xb[idx] = 0.0
    elif n_ones < target:
        zi = np.where(xb == 0)[0]
        order = np.argsort(-x[zi])
        for idx in order[:target - n_ones]:
            xb[idx] = 1.0
    return xb

def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n); dx = 2.0 / n
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k; wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    def _se(x): return x.sum() - n * 0.5
    def _sj(x): return np.ones(n)
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':_se,'jac':_sj}],
                       options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
