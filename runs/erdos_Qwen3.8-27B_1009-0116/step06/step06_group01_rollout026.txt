from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0 and sh < n:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    elif -sh > 0 and -sh < n:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _momentum_descent(h, n_iters, lr_init, decay, momentum, top_k, seed):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    v = np.zeros(n)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    lr = lr_init
    for it in range(n_iters):
        cov = np.correlate(h, 1.0 - h, mode='full')
        # Top-k lags
        ksel = np.argsort(cov)[-top_k:]
        g = np.zeros(n)
        for kk in ksel:
            g += _grad_lag(h, kk, dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        # Nesterov
        v = momentum * v + lr * g
        h_new = _repair(h - v, n)
        c5_new = _eval(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
