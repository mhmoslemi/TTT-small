Actually, let me reconsider the Nesterov method. The standard Nesterov accelerated gradient is:
1. y_n = x_n + (t_n - 1)/(t_{n+1}) * (x_n - x_{n-1})
2. x_{n+1} = proj(y_n - step * grad(y_n))
3. t_{n+1} = (1 + sqrt(1 + 4*t_n^2)) / 2

This is more complex than simple momentum. Let me use a simpler "heavy ball" method:
1. v_{n+1} = mu * v_n - step * grad(x_n)
2. x_{n+1} = proj(x_n + v_{n+1})

This is simpler and works well in practice.

For the step size, I'll use a schedule: step = c / (1 + i * decay_rate) or just a constant with occasional restarts.

    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-tm:]
    return _sgw(h, idx, np.ones(tm) * (2.0 / n))

def _cf(x, n):
    return np.array([x.sum() - 0.5 * n])

def _cj(x, n):
    return np.ones((1, len(x)))

def _slsqp(h, a, mi, ft):
    n = len(h)
    bnd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        r = minimize(_sm, h, jac=_sg, args=(a,),
                     method='SLSQP', bounds=bnd, constraints=cons,
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _heavyball(h, a, tlim, t0, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    x = h.copy(); v = np.zeros(n)
    mu = 0.8; step = 0.05; stuck = 0
    for i in range(200000):
        if time.time() - t0 > tlim:
            break
        g = _sg(x, a)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        v = mu * v - step * g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            xn = _proj(x + f * v, 0.5 * n)
            cn = _eval(xn)
            if cn < best_c - 1e-15:
                best = xn; best_c = cn; x = xn
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.7; stuck += 1
            if stuck > 80:
                best = _repair(best + 0.005 * rng.randn(n), n)
                best_c = _eval(best)
                x = best.copy(); v = np.zeros(n)
                step = 0.05; stuck = 0
    return best, best_c

def _lsearch(h, tlim, t0, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    step = 0.005; stuck = 0
    for i in range(150000):
        if time.time() - t0 > tlim:
            break
        if i % 5 == 0:
            g = _tg(best, 5)
            ok = False
            for f in [1.0, 0.5, 0.25, 0.1]:
                xn = _proj(best - step * f * g, 0.5 * n)
                cn = _eval(xn)
                if cn < best_c - 1e-15:
                    best = xn; best_c = cn; stuck = 0; ok = True
                    break
            if not ok:
                stuck += 1
        elif i % 5 in (1, 2):
            a1, a2 = rng.randint(n), rng.randint(n)
            if a1 == a2:
                continue
            d = step * rng.randn()
            na, nb = best[a1] + d, best[a2] - d
            if 0 <= na <= 1 and 0 <= nb <= 1:
                best[a1], best[a2] = na, nb
                cn = _eval(best)
                if cn < best_c - 1e-15:
                    best_c = cn; stuck = 0
                else:
                    best[a1], best[a2] = na - d, nb + d
        else:
            a1 = rng.randint(n)
            d = step * 0.5 * rng.randn()
            b1 = best[a1] + d
            if 0 <= b1 <= 1:
                best[a1] = b1
                h2 = _repair(best, n)
                cn = _eval(h2)
                if cn < best_c - 1e-15:
                    best = h2; best_c = cn; stuck = 0
                else:
                    best[a1] = b1 - d
        if i % 1000 == 500:
            h2 = best.copy()
            lo = (h2 > 0.05) & (h2 < 0.5)
            hi = (h2 >= 0.5) & (h2 < 0.95)
            h2[lo] = np.maximum(0, h2[lo] - 0.008)
            h2[hi] = np.minimum(1, h2[hi] + 0.008)
            h2 = _repair(h2, n)
            c2 = _eval(h2)
            if c2 < best_c - 1e-15:
                best = h2; best_c = c2; stuck = 0
        if stuck > 300:
            best = _repair(best + 0.002 * rng.randn(n), n)
            best_c = _eval(best); step = 0.005; stuck = 0
    return best, best_c

def _upsample(h, f):
    n = len(h)
    nn = n * f
    return np.interp(np.linspace(0, n, nn, endpoint=False), np.arange(n), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh = h.copy(); bc = _eval(bh); bn = n0

    for a in [500, 2000, 10000, 50000]:
        if time.time() - t0 > 100:
            break
        hc = _slsqp(bh, a, 1200, 1e-16)
        c = _eval(hc)
        if c < bc:
            bc = c; bh = hc

    if time.time() - t0 < budget_s - 350:
        bh, bc = _heavyball(bh, 30000, budget_s - 350, t0, rng)
        bn = len(bh)

    for a in [30000, 100000, 300000]:
        if time.time() - t0 > budget_s - 300:
            break
        hc = _slsqp(bh, a, 1500, 1e-16)
        c = _eval(hc)
        if c < bc:
            bc = c; bh = hc

    if time.time() - t0 < budget_s - 150:
        bh, bc = _lsearch(bh, budget_s - 150, t0, rng)
        bn = len(bh)

    if time.time() - t0 < budget_s - 180:
        h2 = _upsample(bh, 2)
        n2 = len(h2)
        h2 = _repair(h2, n2)
        for a in [2000, 15000, 80000]:
            if time.time() - t0 > budget_s - 80:
                break
            h2 = _slsqp(h2, a, 800, 1e-16)
            c2 = _eval(h2)
            if c2 < bc:
                bc = c2; bh = h2.copy(); bn = n2
        if time.time() - t0 < budget_s - 60:
            bh, bc = _lsearch(bh, budget_s - 60, t0, rng)
            bn = len(bh)

    for _ in range(3):
        if time.time() - t0 > budget_s - 20:
            break
        hk = _repair(bh + 0.008 * rng.randn(bn), bn)
        hc = _slsqp(hk, 50000, 500, 1e-16)
        c = _eval(hc)
        if c < bc:
            bc = c; bh = hc; bn = len(hc)

    if time.time() - t0 < budget_s - 5:
        hc = _slsqp(bh, 1500000, 1200, 1e-18)
        c = _eval(hc)
        if c < bc:
            bc = c; bh = hc; bn = len(hc)

    if time.time() - t0 < budget_s - 1:
        bh, bc = _lsearch(bh, budget_s - 1, t0, rng)
        bn = len(bh)

    bh = _repair(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)
