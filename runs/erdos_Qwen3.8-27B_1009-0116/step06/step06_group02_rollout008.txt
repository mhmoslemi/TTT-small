def _repair(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    # Binary search for lambda
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _corr(h):

    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old, left=0.0, right=0.0)

def _slsqp_quick(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons_sum = n * 0.5

    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

    def grad(x):
        g = np.zeros(n)
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        g = _weighted_grad(x, np.arange(len(c)), w)
        return g

                 'jac': lambda x: np.ones(n)}]

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
