import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _grad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            e = n - sh
            g[:e] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:e]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _adam_opt(h, n_iter, t_end, lr=0.01, top_m=5, sharpen=5.0):
    n = len(h)
    h = h.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    bc = _eval(h)
    for t in range(1, n_iter + 1):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(c, -top_m)[-top_m:]
        vals = c[idx]
        vmin, vmax = vals.min(), vals.max()
        w = np.zeros(2 * n - 1)
        if vmax - vmin > 1e-12:
            w[idx] = np.exp(sharpen * (vals - vmin) / (vmax - vmin))
        else:
            w[idx] = 1.0 / top_m
        w /= w.sum()
        g = _grad(h, w)
        b1, b2 = 0.9, 0.999
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1**t)
        m2h = m2 / (1 - b2**t)
        nu = np.sqrt(m2h) + 1e-8
        for scale in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            hn = np.clip(h - scale * m1h / nu, 0.0, 1.0)
            hn = _repair(hn, n)
            vn = _eval(hn)
            if vn < bc:
                h = hn
                bc = vn
                break
    return h, bc

def _slsqp_obj_fn(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(max(float(s.sum()), 1e-300)) / alpha

def _slsqp_grad_fn(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= max(w.sum(), 1e-300)
    return _grad(x, w)

def _ctr_fun(x):
    return x.sum() - len(x) * 0.5

def _ctr_jac(x):
    return np.ones(len(x))

def _slsqp_run(x0, alpha, maxiter=3000, ftol=1e-18):
    from scipy.optimize import minimize
    n = len(x0)
    bnd = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _ctr_fun, 'jac': _ctr_jac}]
    try:
        r = minimize(_slsqp_obj_fn, x0, jac=_slsqp_grad_fn, args=(alpha,),
                     method='SLSQP', bounds=bnd, constraints=con,
                     options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(r.x, n)
    except Exception:
        return x0.copy()

def _resample(h, nn):
    n = len(h)
    if n == nn:
        return h.copy()
    xo = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh = h0.copy()
    bc = _eval(bh)
    bn = n0
    rng = np.random.RandomState(seed)

    # Phase 1: Adam at native resolution, varying focus
    for tm in [10, 5, 3, 1]:
        if _rem() < 200:
            break
        for sh in [3.0, 8.0, 20.0]:
            if _rem() < 180:
                break
            ha, ca = _adam_opt(bh.copy(), 4000, t0 + _rem() - 30,
                               lr=0.003, top_m=tm, sharpen=sh)
            if ca < bc:
                bc = ca
                bh = ha.copy()

    # Phase 2: SLSQP polish
    if _rem() > 150:
        for alpha in [10000, 50000, 200000]:
            if _rem() < 130:
                break
            hs = _slsqp_run(bh, alpha, 3000, 1e-18)
            cs = _eval(hs)
            if cs < bc:
                bc = cs
                bh = hs.copy()
                bn = len(hs)

    # Phase 3: Multi-resolution
    for nt in [400, 500, 600, 800, 1000]:
        if _rem() < 250:
            break
        hr = _repair(_resample(bh, nt), nt)
        for tm in [5, 2]:
            if _rem() < 220:
                break
            hr, cr = _adam_opt(hr, 3000, t0 + _rem() - 30,
                               lr=0.002, top_m=tm, sharpen=10.0)
        if _rem() > 200:
            hr = _slsqp_run(hr, 50000, 2500, 1e-18)
        cr = _eval(hr)
        if cr < bc:
            bc = cr
            bh = hr.copy()
            bn = nt

    # Phase 4: Lower resolutions
    for nt in [300, 350, 250]:
        if _rem() < 150:
            break
        hr = _repair(_resample(bh, nt), nt)
        hr, cr = _adam_opt(hr, 3000, t0 + _rem() - 20,
                           lr=0.003, top_m=3, sharpen=8.0)
        if _rem() > 130:
            hr = _slsqp_run(hr, 30000, 2000, 1e-18)
        cr = _eval(hr)
        if cr < bc:
            bc = cr
            bh = hr.copy()
            bn = nt

    # Phase 5: Perturbation restarts
    if _rem() > 80:
        for i in range(5):
            if _rem() < 50:
                break
            amp = 0.005 + 0.005 * i
            hp = bh + rng.randn(bn) * amp
            hp = _repair(hp, bn)
            for tm in [3, 1]:
                if _rem() < 30:
                    break
                hp, cp = _adam_opt(hp, 2500, t0 + _rem() - 10,
                                   lr=0.004, top_m=tm, sharpen=15.0)
            cp = _eval(hp)
            if cp < bc:
                bc = cp
                bh = hp.copy()

    # Phase 6: Final SLSQP
    if _rem() > 40:
        hf = _slsqp_run(bh, 100000, 5000, 1e-19)
        cf = _eval(hf)
        if cf < bc:
            bc = cf
            bh = hf.copy()
            bn = len(hf)

    bh = _repair(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _grad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            e = n - sh
            g[:e] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:e]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:n])
            g[sp:] -= wk * h[:n - sp]
    return g

def _adam_opt(h, n_iter, t_end, lr=0.01, top_m=5, sharpen=5.0):
    n = len(h)
    h = h.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    bc = _eval(h)
    for t in range(1, n_iter + 1):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(c, -top_m)[-top_m:]
        vals = c[idx]
        vmin, vmax = vals.min(), vals.max()
        w = np.zeros(2 * n - 1)
        if vmax - vmin > 1e-12:
            w[idx] = np.exp(sharpen * (vals - vmin) / (vmax - vmin))
        else:
            w[idx] = 1.0 / top_m
        w /= w.sum()
        g = _grad(h, w)
        b1, b2 = 0.9, 0.999
        m1 = b1 * m1 + (1.0 - b1) * g
        m2 = b2 * m2 + (1.0 - b2) * g * g
        m1h = m1 / (1.0 - b1**t)
        m2h = m2 / (1.0 - b2**t)
        nu = np.sqrt(m2h) + 1e-8
        taken = False
        for scale in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            hn = np.clip(h - scale * m1h / nu, 0.0, 1.0)
            hn = _repair(hn, n)
            vn = _eval(hn)
            if vn < bc:
                h = hn
                bc = vn
                taken = True
                break
    return h, bc

def _slsqp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(max(float(s.sum()), 1e-300)) / alpha

def _slsqp_grad(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= max(w.sum(), 1e-300)
    return _grad(x, w)

def _ctr_fun(x):
    return x.sum() - len(x) * 0.5

def _ctr_jac(x):
    return np.ones(len(x))

def _slsqp_run(x0, alpha, maxiter=3000, ftol=1e-18):
    from scipy.optimize import minimize
    n = len(x0)
    bnd = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _ctr_fun, 'jac': _ctr_jac}]
    try:
        r = minimize(_slsqp_obj, x0, jac=_slsqp_grad, args=(alpha,),
                     method='SLSQP', bounds=bnd, constraints=con,
                     options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(r.x, n)
    except Exception:
        return x0.copy()

def _resample(h, nn):
    n = len(h)
    if n == nn:
        return h.copy()
    xo = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(xn, xo, h)

        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)

                bn = len(hs)

        hr = _repair(_resample(bh, nt), nt)
        for tm in [5, 2]:
            if _rem() < 220:

            hr, cr = _adam_opt(hr, 4000, t0 + _rem() - 30,
