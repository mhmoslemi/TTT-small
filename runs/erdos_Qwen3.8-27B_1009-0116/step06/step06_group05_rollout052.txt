import numpy as np
import time

def _correlate_fast(h):
    """Compute correlation of h with (1-h) using FFT."""
    n = len(h)
    # Use FFT for circular correlation
    # corr_k = sum_i h[i] * (1-h[(i+k) % n]) -- circular
    # But we need linear (non-circular) correlation
    # Actually np.correlate(h, 1-h, mode='full') is linear
    # For speed, we can zero-pad and use FFT
    N = 2 * n - 1
    Nfft = 1
    while Nfft < N:
        Nfft *= 2
    H = np.fft.rfft(h, Nfft)
    O = np.fft.rfft(1.0 - h, Nfft)
    # Cross-correlation via FFT
    corr = np.fft.irfft(H * np.conj(O), Nfft)[:N]
    return corr

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    c = _correlate_fast(h)
    return float(np.max(c) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:
            if diff > 0:
                for idx in np.argsort(1.0 - h):
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                    if abs(diff) < 1e-12: break
            else:

                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                    if abs(diff) < 1e-12: break

def _eval(h):

def _smooth_max(h, alpha):
    """Log-sum-exp approximation to max of correlation * dx."""
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    sm = s - s.max()
    return (m + np.log(np.sum(np.exp(sm))) / alpha) * dx

def _smooth_max_grad(h, alpha):
    """Gradient of smooth max."""
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    dx = 2.0 / n
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-16: continue

        else:

            g[sp:] += wk * (1.0 - h[:n-sp])

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s - s.max()
    return (m + np.log(np.sum(np.exp(sm))) / alpha) * 2.0 / n

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _sum_eq(h):
    return h.sum() - len(h) * 0.5

def _sum_jac(h):
    return np.ones(len(h))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _perturb(h, sigma, rng):
    h = h + sigma * rng.randn(len(h))
    return _repair(h, len(h))

def _coord_descent(h, n_passes, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_passes):
        for _ in range(3000):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for scale in [0.005, 0.002, 0.0005, 0.0001]:
                d = scale * rng.randn()
                ni = h[i] + d
                nj = h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n <= c5c:
                    c5c = c5n
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
    return h, c5c
