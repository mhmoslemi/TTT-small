        if s >= 0:
            g[s:] += dx * (1.0 - h[:n - s])
            g[:n - s] -= dx * h[s:]
        else:
            p = -s
            g[:n - p] += dx * (1.0 - h[p:])
            g[p:] -= dx * h[:n - p]
    return g / k

def adam_opt(h, alpha, steps, lr, b1=0.9, b2=0.999, eps=1e-8, topk=0, lr_decay=1.0):
    n = len(h)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    h = h.copy()
    for t in range(1, steps + 1):
        if topk > 0:
            g = grad_topk(h, topk)
        else:
            g = grad_soft(h, alpha)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        m_hat = m_arr / (1 - b1**t)
        v_hat = v_arr / (1 - b2**t)
        cur_lr = lr * (lr_decay ** t)
        h = repair(h - cur_lr * m_hat / (np.sqrt(v_hat) + eps), n)
    return h

def nesterov_opt(h, n_iter, lr, mu=0.9, topk=3):
    n = len(h)
    v = np.zeros(n)
    best = h.copy()
    best_c = eval_c5(best)
    stuck = 0
    for t in range(n_iter):
        g = grad_topk(best, topk)
        v = mu * v + g
        h_new = repair(best - lr * v, n)
        c = eval_c5(h_new)
        if c < best_c:
            best = h_new
            best_c = c
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                v = np.zeros(n)
                best = repair(best + 0.02 * np.random.randn(n), n)
                best_c = eval_c5(best)
                stuck = 0
    return best, best_c

def coord_descent(h, n_iter, rng, step_size=5e-4):
    n = len(h)
    best = h.copy()
    best_c = eval_c5(best)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step_size * rng.randn()
        ni, nj = best[i] + d, best[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        best[i], best[j] = ni, nj
        c = eval_c5(best)
        if c < best_c:
            best_c = c
        else:
            best[i], best[j] = ni - d, nj + d
    return best, best_c

def sa_refine(h, n_iter, rng, t_start=0.02, t_end=0.002):
    n = len(h)
    best = h.copy()
    best_c = eval_c5(best)
    cur = h.copy()
    cur_c = best_c
    for t in range(n_iter):
        temp = t_start * (t_end / t_start) ** (t / max(n_iter - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = temp * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        c_new = eval_c5(cur)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (temp * 0.01)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best = cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def el():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c = eval_c5(best_h)
    best_n = n0

    # Phase 1: Adam with alpha annealing (main optimization)
    for a in [100, 500, 2000, 10000, 40000]:
        if el() > budget_s * 0.3:
            break
        h = adam_opt(h, a, 350, lr=0.015, lr_decay=0.9995)
        c = eval_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        h2 = adam_opt(best_h, a, 200, lr=0.004, lr_decay=0.9998)
        c2 = eval_c5(h2)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 2: Nesterov on top-k
    if el() < budget_s * 0.48:
        n_it = min(3000, int((budget_s * 0.48 - el()) / 0.06))
        h_ng, c_ng = nesterov_opt(best_h, max(n_it, 500), lr=0.015, topk=3)
        if c_ng < best_c:
            best_c, best_h, best_n = c_ng, h_ng, len(h_ng)
        h_ng2, c_ng2 = nesterov_opt(best_h, max(n_it // 2, 300), lr=0.008, topk=5)
        if c_ng2 < best_c:
            best_c, best_h, best_n = c_ng2, h_ng2, len(h_ng2)

    # Phase 3: Coordinate descent
    if el() < budget_s * 0.65:
        n_cd = min(60000, int((budget_s * 0.65 - el()) / 0.003))
        best_h, best_c = coord_descent(best_h, n_cd, rng, step_size=5e-4)
        best_n = len(best_h)
        best_h, best_c = coord_descent(best_h, min(n_cd // 2, 30000), rng, step_size=1e-5)

    # Phase 4: Try half resolution (n/2)
    n_half = n0 // 2
    h_half = np.empty(n_half)
    for i in range(n_half):
        h_half[i] = 0.5 * (best_h[2 * i] + best_h[2 * i + 1])
    h_half = repair(h_half, n_half)
    if eval_c5(h_half) < best_c:
        best_c = eval_c5(h_half)
        best_h, best_n = h_half.copy(), n_half
    if el() < budget_s * 0.80:
        for a in [300, 2000, 10000]:
            h_half = adam_opt(h_half, a, 250, lr=0.02, lr_decay=0.9995)
            c = eval_c5(h_half)
            if c < best_c:
                best_c, best_h, best_n = c, h_half.copy(), n_half
        h_half, c = coord_descent(h_half, 20000, rng, step_size=5e-4)
        if c < best_c:
            best_c, best_h, best_n = c, h_half, n_half

    # Phase 5: Iterated local search on best
    if el() < budget_s * 0.92:
        n_ls = min(5, int((budget_s * 0.92 - el()) / 30))
        for _ in range(max(n_ls, 2)):
            if el() > budget_s * 0.92:
                break
            h_pert = repair(best_h + 0.03 * rng.randn(best_n), best_n)
            h_pert = adam_opt(h_pert, 5000, 150, lr=0.01)
            h_pert, c_pert = coord_descent(h_pert, 8000, rng, step_size=2e-5)
            if c_pert < best_c:
                best_c, best_h, best_n = c_pert, h_pert, best_n

    # Phase 6: Final refinement
    if el() < budget_s - 40:
        best_h = adam_opt(best_h, 50000, 200, lr=0.002, lr_decay=0.9999)
        c = eval_c5(best_h)
        if c < best_c:
            best_c = c
        n_cd = min(40000, int((budget_s - 40 - el()) / 0.003))
        best_h, best_c = coord_descent(best_h, n_cd, rng, step_size=1e-5)
        best_n = len(best_h)

    best_h = repair(best_h, best_n)
    best_c = eval_c5(best_h)
    return (best_h, best_c, best_n)
