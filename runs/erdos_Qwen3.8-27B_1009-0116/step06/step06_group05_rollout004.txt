def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection"""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-10:

        nfree = free.sum()
        if nfree > 0:

import numpy as np
import time

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nfree = free.sum()
        if nfree > 0:
            h[free] += diff / nfree
            h = np.clip(h, 0.0, 1.0)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _shift_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:]) * dx
        if sh > 0:
            g[sh:] -= h[:n - sh] * dx
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp]) * dx
        g[:n - sp] -= h[sp:] * dx
    return g

def _pgd(h, n_steps, top_k, lr_init, t_end):

    h = h.copy()

            break
        c = np.correlate(h, 1.0 - h, mode='full')

            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]

            c5 = c5_new
            stagn = 0
            if c5 < best:

        else:
            lr *= 0.75
            stagn += 1
            if lr < 1e-6 or stagn > 300:
                break
    return best_h, best

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _binary_local_search(h, n_iter, rng, t_end):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = _project(h, n)
    c5 = _eval(h)
    best = c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        h2 = _project(h, n)
        c5n = _eval(h2)
        if c5n < c5 - 1e-14:
            c5 = c5n
            if c5n < best:
                best = c5n
                best_h = h2.copy()
            ones = np.where(h2 > 0.5)[0]
            zeros = np.where(h2 < 0.5)[0]
            h = h2
        else:
            h[i], h[j] = 1.0, 0.0
            h = _project(h, n)
    return best_h, best

def _coord_descent(h, n_steps, scale, rng, t_end):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for it in range(n_steps):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        d = scale * rng.randn()
        ni = h[i] + d
        if ni < 0 or ni > 1:
            continue
        h[i] = ni
        h = _project(h, n)
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i] = h[i] - (h[i] - (h[i] - d))  # revert
            h[i] = np.clip(ni - d, 0.0, 1.0)
            h = _project(h, n)
    return h, c5c
