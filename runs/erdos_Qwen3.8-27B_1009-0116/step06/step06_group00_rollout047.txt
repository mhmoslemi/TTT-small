import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _top_grad(h, n_act):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-n_act:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g * dx / n_act

def _smooth_grad(h, alpha):

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.minimum(alpha * (c - m), 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _nesterov_smooth(h, alpha, max_steps, step0, t_limit):
    n = len(h)
    target = n * 0.5
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    t0 = time.time()
    for _ in range(max_steps):
        if time.time() > t_limit:
            break
        g = _smooth_grad(h, alpha)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, target)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            h_new = _proj(h - step * v, target)
            f_new = _smooth_obj(h_new, alpha)
            if f_new < f:
                h = h_new
                f = f_new
    return _repair(h, n)

def _top_lag_descent(h, n_act, max_rounds, t_limit):
    n = len(h)
    target = n * 0.5
    c5 = _eval(h)
    for _ in range(max_rounds):

            break

        gn = np.linalg.norm(g)
        if gn < 1e-14:
