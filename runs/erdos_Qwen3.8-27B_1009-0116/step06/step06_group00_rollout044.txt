    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h

def _obj_fn(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_fn(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad_top(h, alpha, steps, step0, top_k):
    n = len(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        dx_ = 2.0 / n
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(active):
            if w[ii] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[ii] * dx_
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.06, step0 * 6.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.5, step0 * 6.0)
    return _repair(h, n)

def _block_opt(h, bsz, maxit):
    n = len(h)
    nb = n // bsz
    for _pass in range(4):
        for b in range(nb):
            sl = slice(b * bsz, (b + 1) * bsz)
            x0 = h[sl].copy()
            bsum = x0.sum()
            def _ob(x, _x0=x0, _sl=sl, _h=h):
                ht = _h.copy()
                ht[_sl] = x
                return _eval(ht)
            def _oc(x, _bs=bsum):
                return x.sum() - _bs
            def _oj(x, _bs=bsum):
                return np.ones(len(x))
            try:
                cons = [{'type': 'eq', 'fun': _oc, 'jac': _oj, 'args': (bsum,)}]
                res = minimize(_ob, x0, method='SLSQP',
                             bounds=[(0,1)]*bsz, constraints=cons,
                             options={'maxiter': maxit, 'ftol': 1e-15})
                h[sl] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _local3(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    stagnation = 0
    for _ in range(n_iter):
        if c5cur < best_c - 1e-15:
            stagnation = 0
        else:
            stagnation += 1
        step = base_step * (1.0 + 0.5 * (stagnation // 5000))
        step = min(step, base_step * 4.0)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 5 + rng.randint(0, 8)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution coarse search
    for ratio in [4, 8, 16]:
        nc = max(32, n // ratio)
        hc = _repair(_downsample(h, n // nc), nc)
        for alpha in [500, 3000, 15000]:
            if _el() > 80:
                break
            hc = _slsqp(hc, alpha, 400, 1e-14)
        h_up = _repair(_upsample(hc, n // nc), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: SLSQP alpha sweep at full res
    alpha_seq = [2000, 8000, 30000, 100000, 300000, 1000000, 3000000]
    for alpha in alpha_seq:
        if _el() > 350:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Block coordinate optimization
    if _el() < budget_s - 250:
        for bsz in [14, 22, 34]:
            if _el() > budget_s - 120:
                break
            h_new = _block_opt(best_h.copy(), bsz, 80)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Top-lag projected gradient
    if _el() < budget_s - 200:
        for alpha, steps, s0, tk in [(3000, 600, 0.004, 4),
                                      (15000, 500, 0.003, 6),
                                      (60000, 400, 0.002, 8),
                                      (200000, 300, 0.001, 5)]:
            if _el() > budget_s - 130:
                break
            h_new = _proj_grad_top(best_h.copy(), alpha, steps, s0, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Another SLSQP pass at very high alpha
    if _el() < budget_s - 100:
        for alpha in [5000000, 20000000]:
            if _el() > budget_s - 50:
                break
            h_new = _slsqp(best_h, alpha, 800, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 6: Extensive 3-point local search
    if _el() < budget_s - 8:
        ntries = min(int((budget_s - 8 - _el()) / 0.0012), 800000)
        best_h, best_c5 = _local3(best_h, ntries, rng, 6e-6)

    # Phase 7: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 2 - _el()) / 0.0008), 400000)
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 3)) % nc2
            d = 2e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
