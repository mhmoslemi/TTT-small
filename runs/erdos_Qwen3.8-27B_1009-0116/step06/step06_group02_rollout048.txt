    """Gradient of shift k's correlation wrt h"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        i1 = np.arange(cnt)
        g[i1] += dx * (1.0 - h[i1 + sh])

            g[i2] -= dx * h[i2 - sh]

        g[i1] += dx * (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= dx * h[i2 + sp]

        w = np.zeros(n_w)
        w[idx] = wv

        w = np.exp(s - s.max())
        ws = w.sum()

            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h + ((t_k - 1.0) / t_new) * (h - h_prev)
        h_prev = h.copy()
        
        c_y = np.correlate(y, 1.0 - y, mode='full')
        m_y = c_y.max()
        w = _build_weights(c_y, m_y, alpha, top_m)
        g = _grad_lse(y, n, w)
        
        h_new = _project(y - lr * g, n)
        h = h_new
        t_k = t_new
        
        c_curr = _eval(h)
        if c_curr < best_c - 1e-15:
            best_c = c_curr
            best_h = h.copy()
    return best_h, best_c
