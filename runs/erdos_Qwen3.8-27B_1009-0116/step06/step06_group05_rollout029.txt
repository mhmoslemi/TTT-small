def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

def _worst_lag_descent(h, n_rounds, n=310):
    """Focus on the worst lag specifically."""
    dx = 2.0 / n
    for _ in range(n_rounds):
        c = _bcorr(h)
        ki = int(np.argmax(c))
        s = ki - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[:n-s] += dx * (1.0 - h[s:])
            if s > 0:
                g[s:] -= dx * h[:n-s]

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
        c5_old = _eval(h)
        for step in [0.03, 0.01, 0.005, 0.001]:
            h_new = _repair(h - step * g, n)
            if _eval(h_new) < c5_old - 1e-15:
                h = h_new
                break
    return h

def _make_pattern(n, pat, rng):
    h = np.zeros(n)
    if pat == 0:
        h[::2] = 1.0
    elif pat == 1:
        for i in range(n):
            h[i] = 1.0 if (i % 3) < 2 else 0.0
    elif pat == 2:
        for i in range(n):
            h[i] = 1.0 if (i % 5) < 3 else 0.0
    elif pat == 3:
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    elif pat == 4:
        phi = (1.0 + 5.0 ** 0.5) / 2.0
        for i in range(n):
            h[i] = 1.0 if ((i * phi) % 1.0) < 0.5 else 0.0
    else:
        g = int(n ** 0.5)
        for i in range(n):
            r, c_ = divmod(i, g)
            h[i] = 1.0 if (r + c_) % 2 == 0 else 0.0
    return _to_binary(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_descent(h, n_iters, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _left():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Binary search from initial (250s)
    if _left() > 250:
        h_bin = _to_binary(best_h, best_n)
        h_bin, c5b = _bsearch(h_bin, int(_left() * 300), rng, 0.3, 0.0005, patience=3000)
        c5b = _eval(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

    # Phase 2: SLSQP polish multi-alpha (150s)
    if _left() > 150:
        for alpha in [5000, 50000, 200000, 1000000]:
            if _left() < 40:
                break
            h_new = _slsqp(best_h, alpha, 3000, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()
            hc = _slsqp(_repair(1.0 - best_h, best_n), alpha, 2000, 1e-17)
            c5c = _eval(hc)
            if c5c < best_c5:
                best_c5, best_h = c5c, hc.copy()

    # Phase 3: Worst-lag descent (60s)
    if _left() > 60:
        h3 = _worst_descent(best_h.copy(), int(_left() * 150), best_n)
        c5_3 = _eval(h3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Binary search on polished (250s)
    if _left() > 250:
        h_bin = _to_binary(best_h, best_n)
        h_bin, c5b = _bsearch(h_bin, int(_left() * 400), rng, 0.15, 0.0001, patience=2000)
        c5b = _eval(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

    # Phase 5: Multi-pattern + multi-n exploration (200s)
    for n_try in [best_n, best_n - 5, best_n + 5, best_n - 10, best_n + 10]:
        if _left() < 30 or n_try < 100:
            break
        for pat in [0, 1, 3, 4]:
            if _left() < 15:
                break
            hp = _make_pattern(n_try, pat, rng)
            hp, c5p = _bsearch(hp, 8000, rng, 0.1, 0.005, patience=1500)
            c5p = _eval(hp)
            if c5p < best_c5:
                best_c5, best_h, best_n = c5p, hp.copy(), n_try
            hp2 = _slsqp(hp, 100000, 1500, 1e-16)
            c5p2 = _eval(hp2)
            if c5p2 < best_c5:
                best_c5, best_h, best_n = c5p2, hp2.copy(), n_try

    # Phase 6: Resample exploration
    for n_r in [best_n + 20, best_n - 20]:
        if _left() < 30 or n_r < 100:
            break
        hr = _repair(_resample(best_h, n_r), n_r)
        hr = _slsqp(hr, 100000, 1500, 1e-16)
        c5r = _eval(hr)
        if c5r < best_c5:
            best_c5, best_h, best_n = c5r, hr.copy(), n_r

    # Phase 7: Final intensive binary + polish (remaining)
    if _left() > 60:
        h_bin = _to_binary(best_h, best_n)
        h_bin, c5b = _bsearch(h_bin, int(_left() * 500), rng, 0.05, 0.00005, patience=1500)
        c5b = _eval(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

    if _left() > 30:
        hf = _slsqp(best_h, 2000000, 5000, 1e-20)
        c5f = _eval(hf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    # Phase 8: Coordinate descent micro-adjust
    if _left() > 10:
        for sc in [0.002, 0.0005, 0.0001]:
            if _left() < 3:
                break
            best_h, best_c5 = _coord_descent(best_h, 2000, sc, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
