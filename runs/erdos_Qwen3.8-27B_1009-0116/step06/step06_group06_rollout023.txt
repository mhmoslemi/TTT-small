                break
            step *= 0.5
        if not accepted:
            step *= 0.5
        h_pool = h_cur.copy()
        g_prev = _smax_grad(h_cur, alpha)
    return h_cur

def _active_descent(h, n_rounds, t_end, k_active=5):
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    h_cur = h.copy()
    c5 = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:k_active]
        g = np.zeros(n)
        for kk in top_k:
            g += _lag_grad(h_cur, kk)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - step * g, target)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h_cur = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            # Try adding more lags
            top_k2 = np.argsort(c)[::-1][:min(15, 2 * n - 1)]
            g2 = np.zeros(n)
            for kk in top_k2:
                g2 += _lag_grad(h_cur, kk)
            g2 *= dx
            gn2 = np.linalg.norm(g2)
            if gn2 < 1e-14:
                break
            g2 /= gn2
            for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
                h_new = _proj(h_cur - step * g2, target)
                c5n = _eval(h_new)
                if c5n < c5 - 1e-16:
                    h_cur = h_new
                    c5 = c5n
                    improved = True
                    break
        if not improved:
            break
    return h_cur, c5

def _coord_descent(h, n_pairs, t_end, rng):
    n = len(h)
    target = n * 0.5
    h_cur = h.copy()
    c5 = _eval(h_cur)
    for _it in range(n_pairs):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = 2e-4 + 1e-5 * rng.rand()
        d = scale * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5 - 1e-18:
            c5 = c5n
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return h_cur, c5

def _perturb_and_refine(h, rng, t_end, penalty, alpha_lbgfs):
    n = len(h)
    target = n * 0.5
    h_pert = h + rng.randn(n) * 0.01
    h_pert = _proj(h_pert, target)
    h_pert = _lbgfs(h_pert, alpha_lbgfs, penalty, 300, t_end)
    h_pert, c5 = _active_descent(h_pert, 100, t_end, k_active=5)
    return h_pert, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n0 / 2.0
    h = _proj(h, target)
    best_c5 = _eval(h)
    best_h = h.copy()

    def rem():
        return budget_s - (time.time() - t0)

    def maybe_update(h_new, c5_new):
        nonlocal best_c5, best_h
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 1: L-BFGS-B with increasing alpha (fast)
    for alpha in [200, 1000, 5000, 20000, 100000]:
        if rem() < 150:
            break
        h_cand = _lbgfs(best_h, alpha, 1e8, 5000, t0 + rem() - 149)
        h_cand = _proj(h_cand, target)
        c5 = _eval(h_cand)
        maybe_update(h_cand, c5)

    # Phase 2: BB projected gradient
    if rem() > 120:
        h_cand = _bb_descent(best_h, 50000, 800, t0 + rem() - 119)
        h_cand = _proj(h_cand, target)
        c5 = _eval(h_cand)
        maybe_update(h_cand, c5)

    # Phase 3: Active-set descent
    if rem() > 100:
        h_cand, c5 = _active_descent(best_h, 500, t0 + rem() - 99, k_active=5)
        maybe_update(h_cand, c5)

    # Phase 4: More L-BFGS-B rounds with highest alpha
    if rem() > 100:
        h_cand = _lbgfs(best_h, 200000, 1e9, 8000, t0 + rem() - 99)
        h_cand = _proj(h_cand, target)
        c5 = _eval(h_cand)
        maybe_update(h_cand, c5)

    # Phase 5: Active descent with more lags
    if rem() > 80:
        h_cand, c5 = _active_descent(best_h, 400, t0 + rem() - 79, k_active=10)
        maybe_update(h_cand, c5)

    # Phase 6: Perturbation restarts
    if rem() > 60:
        for _p in range(6):
            if rem() < 55:
                break
            h_pert, c5_p = _perturb_and_refine(best_h, rng, t0 + rem() - 54, 1e8, 50000)
            maybe_update(h_pert, c5_p)

    # Phase 7: Fine coordinate descent
    if rem() > 30:
        n_pairs = min(int((rem() - 25) / 0.003), 80000)
        h_cand, c5 = _coord_descent(best_h, n_pairs, t0 + rem() - 25, rng)
        maybe_update(h_cand, c5)

    # Phase 8: Final active descent (very fine)
    if rem() > 20:
        h_cand, c5 = _active_descent(best_h, 300, t0 + rem() - 19, k_active=8)
        maybe_update(h_cand, c5)

    # Phase 9: Ultra-fine coordinate descent
    if rem() > 10:
        n_pairs = min(int((rem() - 8) / 0.002), 50000)
        h_cand, c5 = _coord_descent(best_h, n_pairs, t0 + rem() - 8, rng)
        maybe_update(h_cand, c5)

    # Final validation
    best_h = _proj(best_h, target)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
