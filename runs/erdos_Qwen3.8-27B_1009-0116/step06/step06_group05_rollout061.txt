```python
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_subgrad(h, alpha):
    nc = len(h)
    c = _corr(h)
    ac = alpha * c
    max_ac = ac.max()
    w = np.exp(ac - max_ac); w /= w.sum()
    dx = 2.0 / nc
    g = np.zeros(nc)
    for k in range(2*nc - 1):
        if w[k] < 1e-14: continue
        s = k - (nc-1)
        wk = w[k]
        if s >= 0:
            g[s:] += wk * (1.0 - h[:nc-s])
            g[:nc-s] -= wk * h[s:]
        else:
            t = -s
            g[:nc-t] += wk * (1.0 - h[t:])
            g[t:] -= wk * h[:nc-t]
    return g * dx

def _pg_solve(h0, alpha, max_iter, step0):
    """Projected gradient with backtracking on smoothed max"""
    nc = len(h0)
    h = h0.copy()
    f0 = _eval(h)
    step = step0
    g_avg = np.zeros(nc)
    M = 50  # averaging window
    for it in range(max_iter):
        g = _proj_subgrad(h, alpha)
        g_avg = (g_avg * min(it, M) + g) / (min(it, M) + 1)
        step = step0
        h_new = _repair(h - step * g_avg, nc)
        f_new = _eval(h_new)
        backtracks = 0
        while f_new > f0 - 1e-12 * step * np.abs(g_avg).max() and backtracks < 20:
            step *= 0.5
            h_new = _repair(h - step * g_avg, nc)
            f_new = _eval(h_new)
            backtracks += 1
        if f_new < f0 - 1e-16:
            h = h_new
            f0 = f_new
            step = min(step * 1.1, step0 * 2)
        else:
            step *= 0.5
        if step < 1e-10: break
    return h, _eval(h)

def _slsqp_smooth(h, alpha, maxiter, ftol):
    nc = len(h)
    def obj(x):
        c = _corr(x)
        ac = alpha * c
        mac = ac.max()
        return float((mac + np.log(np.sum(np.exp(ac - mac)))) / alpha)
    def grad(x):
        return _proj_subgrad(x, alpha)
    def cfun(x):
        return x.sum() - nc * 0.5
    def cjac(x):
        return np.ones(nc)
    bnds = [(0.0, 1.0)] * nc
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, nc)
    except:
        return _repair(h, nc)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_descent(h, n_iter, rng, scale=0.001):
    nc = len(h)
    f0 = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d

        fn = _eval(h)
        if fn < f0 - 1e-16:
            f0 = fn
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, f0
```

For the Nesterov method on the constrained problem:
- We want to minimize f(h) = max_k c_k * dx subject to h ∈ [0,1]^n, sum(h) = n/2
- The constraint set is a convex polytope P
- Projected gradient: h_{k+1} = proj_P(h_k - α_k * g_k)
- Nesterov: y_k = h_k + β_k (h_k - h_{k-1}), g_k = ∇f(y_k), h_{k+1} = proj_P(h_k - α_k * g_k)

The "projection" here is approximated by _repair (which does clip + scale + distribute).

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_grad(h, alpha):
    nc = len(h)
    c = _corr(h)
    ac = alpha * c
    mac = ac.max()
    w = np.exp(ac - mac); w /= w.sum()
    dx = 2.0 / nc
    g = np.zeros(nc)
    for k in range(2 * nc - 1):
        if w[k] < 1e-14: continue
        s = k - (nc - 1)
        wk = w[k]
        if s >= 0:
            g[s:] += wk * (1.0 - h[:nc-s])
            g[:nc-s] -= wk * h[s:]
        else:
            t = -s
            g[:nc-t] += wk * (1.0 - h[t:])
            g[t:] -= wk * h[:nc-t]
    return g * dx

def _smooth_obj(h, alpha):
    c = _corr(h)
    ac = alpha * c
    mac = ac.max()
    return float((mac + np.log(np.sum(np.exp(ac - mac)))) / alpha)
```

Wait, I realize there might be an issue with the alpha parameter. The log-sum-exp:
LSE(c, alpha) = (1/alpha) * log(sum(exp(alpha * c_j)))

As alpha → ∞, this approaches max(c_j). The gradient is:
∇_h LSE = (1/alpha) * sum_j alpha * exp(alpha*c_j) / sum_k exp(alpha*c_k) * ∇_h c_j
= sum_j w_j * ∇_h c_j
where w_j = exp(alpha*c_j) / sum_k exp(alpha*c_k)

And ∇_h c_j is what I computed above (the subgradient of c_j) * dx.

So the gradient of the smooth objective w.r.t. h is:
∇_h LSE = sum_j w_j * (subgradient of c_j) * dx
= [the vector I computed in _smooth_grad]

This is correct. And as alpha increases, w becomes more concentrated on the max, and the smooth gradient approaches the true subgradient of the max.

```python
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_grad(h, alpha):
    nc = len(h)
    c = _corr(h)
    ac = alpha * c
    mac = ac.max()
    w = np.exp(ac - mac); w /= w.sum()
    dx = 2.0 / nc
    g = np.zeros(nc)
    for k in range(2 * nc - 1):
        if w[k] < 1e-14: continue
        s = k - (nc - 1)
        wk = w[k]
        if s >= 0:

            g[:nc-t] += wk * (1.0 - h[t:])
            g[t:] -= wk * h[:nc-t]
    return g * dx

def _smooth_obj(h, alpha):
    c = _corr(h)
    ac = alpha * c
    mac = ac.max()
    return float((mac + np.log(np.sum(np.exp(ac - mac)))) / alpha)

def _se_fn(x):
    return x.sum() - len(x) * 0.5

def _se_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    nc = len(h)
    bnds = [(0.0, 1.0)] * nc
    cons = [{'type': 'eq', 'fun': _se_fn, 'jac': _se_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, nc)
    except Exception:
        return _repair(h, nc)

def _momentum_pg(h0, alphas, n_iter, lr0, beta=0.85):
    nc = len(h0)
    h = h0.copy()
    v = np.zeros(nc)
    best_f = _eval(h)
    for alpha in alphas:
        lr = lr0
        for it in range(n_iter):
            y = _repair(h + v, nc)
            g = _smooth_grad(y, alpha)
            v = beta * v - lr * g
            h_new = _repair(h + v, nc)
            fn = _eval(h_new)
            if fn < best_f - 1e-17:
                h = h_new
                best_f = fn
            else:
                v *= 0.3
                lr *= 0.5
            if lr < 1e-14:
                v = np.zeros(nc)
                lr = lr0
                break
        v = np.zeros(nc)
    return h, best_f

def _cyclic_cd(h, n_passes, scales):
    nc = len(h)
    best_f = _eval(h)
    for scale in scales:
        for p in range(n_passes):
            for i in range(0, nc - 1, 2):
                d = scale * (2.0 * (hash((i, p)) % 1000) / 1000.0 - 1.0)
                ni, nj = h[i] + d, h[i+1] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                h[i], h[i+1] = ni, nj
                fn = _eval(h)
                if fn < best_f - 1e-17:
                    best_f = fn
                else:
                    h[i], h[i+1] = ni - d, nj + d
    return h, best_f

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _anneal_continuous(h0, t0, t1, n_steps, rng):
    nc = len(h0)
    h = h0.copy()
    best = _eval(h)
    best_h = h.copy()
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i, j = rng.randint(nc), rng.randint(nc)
        if i == j: continue
        d = 0.05 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        h2 = _repair(h, nc)
        fn = _eval(h2)
        if fn < best or rng.rand() < np.exp(-(fn - best) / max(t, 1e-7)):
            h = h2
            if fn < best:
                best = fn
                best_h = h2.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            h = _repair(h, nc)
    return best_h, best
```
