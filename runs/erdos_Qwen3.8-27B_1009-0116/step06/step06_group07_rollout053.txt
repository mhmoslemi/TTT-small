import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(h, alpha, penalty):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = m + np.log(float(np.sum(np.exp(s)))) / alpha
    n = len(h)
    val += penalty * (h.sum() - n * 0.5) ** 2
    return val

def _smooth_grad(h, alpha, penalty):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            p = -sh
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

def _lqbfgsb(h0, alpha, penalty, max_iter, time_limit):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    t_start = time.time()
    def _tl():
        return time.time() - t_start > time_limit
    try:
        res = minimize(_smooth_obj, h0, jac=lambda x: _smooth_grad(x, alpha, penalty),
                       args=(), method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxls': 50, 'maxfun': max_iter * 5,
                                'callback': lambda: _tl() and True})
        if res.success or res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h0, n)

def _coord_descent(h, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    n = len(h)
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _proj_sum(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _pgd_period(h_p, T, n, alphas, max_iters, lr_init, seed):
    m = n // T
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    momentum = np.zeros(T)
    beta = 0.9
    for alpha in alphas:
        iters = max(1, max_iters // len(alphas))
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            mm = c.max()
            s = np.clip(alpha * (c - mm), -500, 500)
            w = np.exp(s - s.max())
            w /= w.sum()
            dx = 2.0 / n
            g_full = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k] * dx
                if wk < 1e-14:
                    continue
                sh = k - (n - 1)
                if sh >= 0:
                    g_full[sh:] += wk * (1.0 - h_full[:n - sh])
                    g_full[:n - sh] -= wk * h_full[sh:]
                else:
                    p = -sh
                    g_full[:n - p] += wk * (1.0 - h_full[p:])
                    g_full[p:] -= wk * h_full[:n - p]
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p * T:(p + 1) * T]
            g_p /= m
            momentum = beta * momentum + (1.0 - beta) * g_p
            h_new = _proj_sum(h_p - lr * momentum, T)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                if lr < 0.05:
                    lr *= 1.04
            else:
                lr *= 0.55
                if lr < 1e-7:
                    lr = lr_init
                    momentum = np.zeros(T)
    return best, best_c

                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.tile(h_p, m).copy()

    # Phase 6: Coordinate descent polish (multi-scale)
    if elapsed() < budget_s - 15:
        for scale in [0.005, 0.001, 0.0002, 0.00005]:
            if elapsed() > budget_s - 10:
                break
            h9, c5 = _coord_descent(best_h.copy(), 4000, scale, seed=seed + 100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h9.copy()

    # Phase 7: Try nearby n values
    if elapsed() < budget_s - 30:
        for dn in [10, -10, 20, -20, 31, -31, 5, -5]:
            if elapsed() > budget_s - 15:
                break
            n_try = best_n + dn
            if n_try < 50 or n_try > 800:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            tl = min(remaining() * 0.1, 60)
            if tl < 3:
                break
            h_r = _lqbfgsb(h_r, 200000, 100.0, 500, tl)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                best_n = n_try

    # Final repair and eval
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
