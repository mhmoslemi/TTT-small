import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        interior = (h > 0.005) & (h < 0.995)
        if interior.sum() > 0:
            h[interior] += (target - s) / interior.sum()
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(np.abs(h - 0.5))
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cgrad(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += dx * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx * h[:n - sh]
    else:
        sp = -sh
        g[:n - sp] += dx * (1.0 - h[sp:])
        g[sp:] -= dx * h[:n - sp]
    return g

def _sm_obj(h, a):
    c = _lags(h)
    m = c.max()
    s = np.clip(a * (c - m), -40, 40)
    return m + np.log(np.sum(np.exp(s - s.max()))) / a

def _sm_grad(h, a):
    c = _lags(h)
    m = c.max()
    s = np.clip(a * (c - m), -40, 40)
    w = np.exp(s - s.max())
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _cgrad(h, k, n)
    return g

def _slsqp(h, a, mx, ft, t0, bud):
    n = len(h)
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    def _ob(x):
        if time.time() - t0 > bud:
            return 1e10
        return _sm_obj(x, a)
    def _gr(x):
        return _sm_grad(x, a)
    bnds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        r = minimize(_ob, h, jac=_gr, method='SLSQP', bounds=bnds,
                     constraints=con, options={'maxiter': mx, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, a, steps, s0, t0, bud):
    n = len(h)
    f = _sm_obj(h, a)
    v = np.zeros(n)
    mu = 0.85
    step = s0
    for _ in range(steps):
        if time.time() - t0 > bud:
            break
        g = _sm_grad(h, a)
        hp = h.copy()
        v = mu * v + (1 - mu) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _sm_obj(hn, a)
        if fn < f:
            h, f, step = hn, fn, min(step * 1.03, s0 * 5)
        else:
            step *= 0.55
            h = _proj(hp - step * v, n * 0.5)
            f = _sm_obj(h, a)
    return _repair(h, n)

def _waterfill(h, nd, t0, bud):
    n = len(h)
    c5c = _eval(h)
    for _ in range(nd):
        if time.time() - t0 > bud:
            break
        c = _lags(h)
        kw = int(np.argmax(c))
        g = _cgrad(h, kw, n)
        imp = False
        for st in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
            hn = _proj(h - st * g, n * 0.5)
            cp = _eval(hn)
            if cp < c5c - 1e-15:
                h, c5c, imp = hn, cp, True
                break
        if not imp:
            break
    return h, c5c

def _mlag_w(h, tn, nd, bt, t0, bud):
    n = len(h)
    c5c = _eval(h)
    for _ in range(nd):
        if time.time() - t0 > bud:
            break
        c = _lags(h)
        top = np.argsort(c)[-tn:]
        ts = top[np.argsort(c[top])[::-1]]
        g = np.zeros(n)
        ws = sum(bt ** i for i in range(len(ts)))
        for i, k in enumerate(ts):
            g += (bt ** i / ws) * _cgrad(h, k, n)
        imp = False
        for st in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
            hn = _proj(h - st * g, n * 0.5)
            cp = _eval(hn)
            if cp < c5c - 1e-15:
                h, c5c, imp = hn, cp, True
                break
        if not imp:
            break
    return h, c5c

def _lsearch(h, ni, rng, st_sc, np_, t0, bud):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(ni):
        if time.time() - t0 > bud:
            break
        i = rng.randint(n)
        offs = rng.randint(1, max(4, n // 15), size=np_)
        idx = [(i + offs[j]) % n for j in range(np_)]
        if len(set(idx)) < np_:
            continue
        d = st_sc * rng.randn(np_)
        d -= d.mean()
        ok = True
        nv = []
        for j in range(np_):
            v = h[idx[j]] + d[j]
            if v < -1e-15 or v > 1 + 1e-15:
                ok = False
                break
            nv.append(np.clip(v, 0, 1))
        if not ok:
            continue
        for j in range(np_):
            h[idx[j]] = nv[j]
        cp = _eval(h)
        if cp < c5c - 1e-18:
            c5c = cp
            if cp < bc:
                bc, bh = cp, h.copy()
        else:
            for j in range(np_):
                h[idx[j]] -= d[j]
    return bh, bc

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()
