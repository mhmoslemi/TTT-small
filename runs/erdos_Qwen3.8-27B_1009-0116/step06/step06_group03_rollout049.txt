import time

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _topk_obj(grad_mode, h, top_m, dx):
    """Returns (objective_value, gradient) for sum of top-k correlation values."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    top_idx = np.argsort(c)[-top_m:]
    val = c[top_idx].sum() * dx
    if not grad_mode:
        return val, None
    g = np.zeros(n)
    for k in top_idx:
        sh = (n - 1) - k
        if sh >= 0:
            if n - sh > 0:
                g[:n-sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n-sp])
            g[:n-sp] -= dx * h[sp:]
    return val, g

def _slsqp_topk(h, top_m, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(c)[-top_m:]
        return c[top_idx].sum() * dx
    
    def _grad(x):
        _, g = _topk_obj(True, x, top_m, dx)
        return g
    
    def _con_fun(x):
        return [x.sum() - 0.5 * n]
    
    def _con_jac(x):
        return [np.ones(n)]
    
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_grad(h, n_iters, top_m, step0, momentum):
    """Nesterov accelerated projected gradient on top-k sum."""
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c = _eval(best)
    x = h.copy()
    v = np.zeros(n)
    
    for it in range(n_iters):
        # Nesterov step
        x_nr = x - momentum * v
        x_nr = _proj(x_nr, 0.5 * n)
        
        _, g = _topk_obj(True, x_nr, top_m, dx)
        
        # Gradient step
        v = momentum * v - step0 * g
        x_new = _proj(x + v, 0.5 * n)
        
        c_new = _eval(x_new)
        if c_new < best_c:
            best = x_new.copy()
            best_c = c_new
        
        x = x_new
    
    return best, best_c

def _local_search(h, n_tries, rng):
    """Thorough local search with adaptive steps."""
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    step_sizes = [1e-3, 5e-4, 1e-4, 5e-5, 1e-5, 5e-6, 1e-6]
    
    for t in range(n_tries):
        # Choose step size adaptively
        si = min(t // 5000, len(step_sizes) - 1)
        ds = step_sizes[si] * rng.uniform(0.5, 1.5)
        
        # Random move type
        r = rng.rand()
        if r < 0.5:
            # Pairwise move
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = ds * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
        elif r < 0.8:
            # Single move with spread
            i = rng.randint(n)
            d = ds * rng.randn()
            ni = h[i] + d
            if ni < 0 or ni > 1:
                continue
            others = np.delete(range(n), i)
            h[i] = ni
            adj = -d / (n - 1)
            h[others] += adj
            if h[others].min() < -1e-10 or h[others].max() > 1 + 1e-10:
                h[i] -= d
                h[others] -= adj

        else:
            # Block move
            b = rng.randint(1, max(2, n // 20))
            start = rng.randint(0, n - b)
            d = ds * rng.randn()
            h[start:start+b] += d

            h[rest] += comp
            if h.min() < -1e-10 or h.max() > 1 + 1e-10:
                h[start:start+b] -= d
                h[rest] -= comp

        if cn < c_cur - 1e-15:
            if cn < best_c:
                best_c = cn
                best = h.copy()
            c_cur = cn
        else:
            # Revert
            if r < 0.5:
                h[i], h[j] = h[i] - d, h[j] + d
            elif r < 0.8:
                h[i] -= d
                h[others] -= adj
            else:
                h[start:start+b] -= d
                h[rest] -= comp
    
    return best, best_c

    # Phase 1: SLSQP with progressive top-k
    for top_m, maxit in [(3, 300), (5, 400), (8, 300)]:
        if _el() > 120:
            break
        hc = _slsqp_topk(best_h, top_m, maxit, 1e-14)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc.copy()

    # Phase 2: Momentum optimization (Nesterov-style)
    if _el() < budget_s - 500:
        rem = budget_s - 500 - _el()
        n_it = min(6000, int(rem / 0.015))
        best_h, best_c = _momentum_opt(best_h, n_it, 5, 0.007, 0.9, 250, rng)
        best_n = len(best_h)

    # Phase 3: SLSQP refinement
    for top_m, maxit in [(5, 500), (10, 400), (5, 300)]:
        if _el() > budget_s - 400:
            break
        hc = _slsqp_topk(best_h, top_m, maxit, 1e-15)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc.copy()

    # Phase 4: Momentum with larger top-k
    if _el() < budget_s - 350:
        rem = budget_s - 350 - _el()
        n_it = min(6000, int(rem / 0.015))
        best_h, best_c = _momentum_opt(best_h, n_it, 8, 0.004, 0.92, 300, rng)
        best_n = len(best_h)

    # Phase 5: Upsample to 2x resolution
    if _el() < budget_s - 350:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for top_m, maxit in [(5, 1200), (8, 800)]:
            if _el() > budget_s - 180:
                break
            h_hi = _slsqp_topk(h_hi, top_m, maxit, 1e-15)
            c = _eval(h_hi)
            if c < best_c:
                best_c = c
                best_h = h_hi
                best_n = n1
        if _el() < budget_s - 120:
            rem = budget_s - 120 - _el()
            n_it = min(4000, int(rem / 0.03))
            best_h, best_c = _momentum_opt(best_h, n_it, 10, 0.002, 0.93, 300, rng)
            best_n = len(best_h)

    # Phase 6: Final SLSQP squeeze
    if _el() < budget_s - 80:
        hc = _slsqp_topk(best_h, 5, 800, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc
            best_n = len(hc)

    # Phase 7: Local search (adaptive step)
    if _el() < budget_s - 3:
        rem = budget_s - 3 - _el()
        n_tr = min(int(rem / 0.0015), 400000)
        best_h, best_c = _local_search(best_h, n_tr, rng)
        best_n = len(best_h)

    # Final mini SLSQP if time permits
    if _el() < budget_s - 1:
        hc = _slsqp_topk(best_h, 5, 200, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc
            best_n = len(hc)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
