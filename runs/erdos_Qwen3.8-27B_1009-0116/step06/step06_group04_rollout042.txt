def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, alpha, tk):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    if tk is not None and tk < len(c):
        idx = np.argpartition(c, -tk)[-tk:]
        cs = c[idx]
    else:
        cs = c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -100, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(x, alpha, tk):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    if tk is not None and tk < len(c):
        idx = np.argpartition(c, -tk)[-tk:]
        cs = c[idx]

        s = alpha * (cs - m)
        s -= s.max()
        w = np.exp(s) / np.exp(s).sum()
        wf = np.zeros(len(c))
        wf[idx] = w
    else:

        s = alpha * (c - m)
        s -= s.max()

    g = np.zeros(n)
