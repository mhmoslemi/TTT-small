If min subgradient > 0: f is increasing → move t down
If max subgradient < 0: f is decreasing → move t up
Otherwise: we're at the minimum.

Wait, I realize I should think about this more carefully. The coordinate descent finds the exact 1D optimum for each coordinate. But the issue is the sum constraint. After changing h[i] independently, the sum drifts. The repair at the end of each pass projects back to the correct sum, but this undoes some of the optimization.

A better approach: paired coordinate descent. For each pair (i, j), set h[i] = t, h[j] = h[j] + old_i - t. This preserves the sum exactly. But then the 1D problem becomes:
f(t) = max_k (c[k] + (t - old_i)*a_i[k] + (old_i - t)*a_j[k])
     = max_k (c[k] + (t - old_i)*(a_i[k] - a_j[k]))

This is still convex in t (max of linear functions)! And t is constrained to [max(0, old_i - h[j]), min(1, old_i + h[j])] to keep both h[i] and h[j] in [0,1].

This is a better formulation. Let me use this.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_coeff(h, i, n, dx):
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        sh = k - (n - 1)
        coeff = 0.0
        if sh >= 0:
            if i + sh < n:
                coeff += 1.0 - h[i + sh]
            if sh <= i:
                coeff += -h[i - sh]
        else:
            sp = -sh
            if sp <= i:
                coeff += 1.0 - h[i - sp]
            if i + sp < n:
                coeff += -h[i + sp]
        a[k] = coeff * dx
    return a

def _coord_descend_pair(h, n_passes, offset, budget_fn):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    for _p in range(n_passes):
        if budget_fn is not None and budget_fn():
            break
        for i in range(n):
            j = (i + offset) % n
            if i == j:
                continue
            old_i = h[i]
            old_j = h[j]
            a_i = _grad_coeff(h, i, n, dx)
            a_j = _grad_coeff(h, j, n, dx)
            d_a = a_i - a_j
            # f(t) = max_k (c[k] + (t - old_i) * d_a[k])
            intercepts = c - old_i * d_a
            lo_t = max(0.0, old_i - old_j)
            hi_t = min(1.0, old_i + old_j)
            if hi_t - lo_t < 1e-15:
                continue
            lo, hi = lo_t, hi_t
            for _b in range(40):
                mid = (lo + hi) * 0.5
                vals = d_a * mid + intercepts
                mv = vals.max()
                active = vals >= mv - 1e-14
                if not active.any():
                    active[np.argmax(vals)] = True
                mn = d_a[active].min()
                mx = d_a[active].max()
                if mn > 1e-12:
                    hi = mid
                elif mx < -1e-12:
                    lo = mid
                else:
                    break
            t_opt = (lo + hi) * 0.5
            if abs(t_opt - old_i) > 1e-14:
                h[j] = old_j + old_i - t_opt
                h[i] = t_opt
                c += (t_opt - old_i) * d_a
    return h

def _local_search_2pt(h, n_iter, rng, budget_fn):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_iter):
        if budget_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _slsqp_soft(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full') * dx
        m = cc.max()
        s = np.clip(alpha * (cc - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _jac(x):
        cc = np.correlate(x, 1.0 - x, mode='full') * dx
        m = cc.max()
        s = np.clip(alpha * (cc - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    from scipy.optimize import minimize
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_jac, args=(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
