            g[:n-sp] -= wk * x[sp:]
    return g

def _se(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old + 1)
    h_ext = np.concatenate([h_old, [h_old[0]]])
    xn = np.linspace(0, 2.0, n_new)
    return np.interp(xn, xo, h_ext)

def _fourier_params(h, M):
    n = len(h)
    t = np.linspace(0, 2.0, n, endpoint=False)
    dx = 2.0 / n
    params = np.zeros(2 * M)
    for m in range(1, M + 1):
        params[2*(m-1)] = np.sum((h - 0.5) * np.cos(np.pi * m * t)) * dx * 2
        params[2*(m-1)+1] = np.sum((h - 0.5) * np.sin(np.pi * m * t)) * dx * 2
    return params

def _fourier_to_h(params, M, n):
    t = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        h += params[2*(m-1)] * np.cos(np.pi * m * t)
        h += params[2*(m-1)+1] * np.sin(np.pi * m * t)
    return h

def _fourier_opt(h_start, M, max_time):
    n = len(h_start)
    params = _fourier_params(h_start, M)
    h_best = _repair(_fourier_to_h(params, M, n), n)
    c5_best = _eval(h_best)
    t_s = time.time()
    t_arr = np.linspace(0, 2.0, n, endpoint=False)
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() - t_s > max_time: break
        def f4(p, _M=M, _n=n, _t=t_arr, _a=alpha):
            hh = _repair(_fourier_to_h(p, _M, _n), _n)
            return _lssm(hh, _a)
        def g4(p, _M=M, _n=n, _t=t_arr, _a=alpha):
            hh = _fourier_to_h(p, _M, _n)
            gc = _lssm_grad(hh, _a)
            dx = 2.0 / _n
            gf = np.zeros(2 * _M)
            for m in range(1, _M + 1):
                gf[2*(m-1)] = np.sum(gc * np.cos(np.pi * m * _t)) * dx * 2
                gf[2*(m-1)+1] = np.sum(gc * np.sin(np.pi * m * _t)) * dx * 2
            return gf
        try:
            res = minimize(f4, params, jac=g4, method='SLSQP',
                          options={'maxiter': 1500, 'ftol': 1e-19})
            h_new = _repair(_fourier_to_h(res.x, M, n), n)
            c5_new = _eval(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                params = res.x.copy()
                h_best = h_new.copy()
        except Exception:
            pass
    return h_best, c5_best

def _topk_subgrad(h, K, n_rounds, t_limit):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    for r in range(n_rounds):
        if time.time() > t_limit: break
        c = _corr(h)
        top_k = np.argpartition(c, -K)[-K:]
        g = np.zeros(n)
        for ki in top_k:
            sh = (n - 1) - ki
            if sh >= 0:
                g[:n-sh] += dx * (1.0 - h[sh:])
                if sh > 0: g[sh:] -= dx * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += dx * (1.0 - h[:n-sp])
                g[:n-sp] -= dx * h[sp:]
        g /= float(K)
        lr = 0.05
        for _ in range(25):
            h_new = _repair(h - lr * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c = h_new, c5n
                break
            lr *= 0.5
    return h, c5c

def _jitter_reopt(h, scale, alpha, maxiter, rng):
    n = len(h)
    h_j = h + scale * rng.randn(n)
    h_j = _repair(h_j, n)
    h_j = _slsqp(h_j, alpha, maxiter, 1e-18)
    return h_j

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick SLSQP (15s)
    for alpha in [10000, 100000]:
        if time.time() - t0 > 15: break
        h_new = _slsqp(best_h, alpha, 1500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5

    # Phase 2: Multi-resolution (80s)
    for n_low in [30, 50, 70, 100]:
        if time.time() - t0 > 80: break
        h_low = _repair(_resample(best_h, n_low), n_low)
        for alpha in [2000, 20000, 200000, 2000000]:
            if time.time() - t0 > 80: break
            h_low = _slsqp(h_low, alpha, 500, 1e-18)
        h_up = _repair(_resample(h_low, n0), n0)
        h_up = _slsqp(h_up, 200000, 2000, 1e-19)
        c5 = _eval(h_up)
        if c5 < best_c5:
            best_h, best_c5 = h_up.copy(), c5

    # Phase 3: Fourier parameterized optimization (150s)
    t_ft = time.time() - t0
    if t_ft < budget_s - 350:
        for M in [6, 10, 15, 20, 25]:
            if time.time() - t0 > budget_s - 350: break
            h_f, c5_f = _fourier_opt(best_h, M, 25)
            if c5_f < best_c5:
                best_h, best_c5 = h_f.copy(), c5_f
                h_ref = _slsqp(best_h, 500000, 2000, 1e-19)
                c5r = _eval(h_ref)
                if c5r < best_c5:
                    best_h, best_c5 = h_ref.copy(), c5r

    # Phase 4: Top-K subgradient (120s)
    if time.time() - t0 < budget_s - 280:
        t_end4 = t0 + budget_s - 280
        h4, c5_4 = _topk_subgrad(best_h, 5, 800, t_end4)
        if c5_4 < best_c5:
            best_h, best_c5 = h4.copy(), c5_4
        h4 = _slsqp(best_h, 1000000, 2000, 1e-19)
        c5_4 = _eval(h4)
        if c5_4 < best_c5:
            best_h, best_c5 = h4.copy(), c5_4

    # Phase 5: Jitter escape attempts (150s)
    if time.time() - t0 < budget_s - 200:
        for scale in [0.04, 0.07, 0.12, 0.03, 0.06, 0.15]:
            if time.time() - t0 > budget_s - 200: break
            h_j = _jitter_reopt(best_h, scale, 200000, 2000, rng)
            c5j = _eval(h_j)
            if c5j < best_c5:
                best_h, best_c5 = h_j.copy(), c5j
                h_j2 = _jitter_reopt(best_h, scale * 0.5, 1000000, 2000, rng)
                c5j2 = _eval(h_j2)
                if c5j2 < best_c5:
                    best_h, best_c5 = h_j2.copy(), c5j2

    # Phase 6: Aggressive SLSQP sweep (120s)
    if time.time() - t0 < budget_s - 80:
        for alpha in [500000, 2000000, 10000000, 50000000, 200000000]:
            if time.time() - t0 > budget_s - 80: break
            h_new = _slsqp(best_h, alpha, 5000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    # Phase 7: Second round of top-K + Fourier (80s)
    if time.time() - t0 < budget_s - 40:
        h7, c5_7 = _topk_subgrad(best_h, 10, 400, t0 + budget_s - 50)
        if c5_7 < best_c5:
            best_h, best_c5 = h7.copy(), c5_7
        if time.time() - t0 < budget_s - 20:
            h7f, c5_7f = _fourier_opt(best_h, 20, 10)
            if c5_7f < best_c5:
                best_h, best_c5 = h7f.copy(), c5_7f
                h7f = _slsqp(best_h, 100000000, 2000, 1e-20)
                c5_7f = _eval(h7f)
                if c5_7f < best_c5:
                    best_h, best_c5 = h7f.copy(), c5_7f

    # Phase 8: Final micro coordinate descent (30s)
    if time.time() - t0 < budget_s - 5:
        h8 = best_h.copy()
        c5c = _eval(h8)
        n = len(h8)
        t_end8 = t0 + budget_s - 5
        for _scale in [0.003, 0.001, 0.0003, 0.0001]:
            if time.time() > t_end8: break
            for _it in range(8000):
                if time.time() > t_end8: break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j: continue
                d = _scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                h8[i], h8[j] = ni, nj
                h8 = _repair(h8, n)
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h8[i], h8[j] = h8[i] - d, h8[j] + d
                    h8 = _repair(h8, n)
        if c5c < best_c5:
            best_h, best_c5 = h8.copy(), c5c

    # Final polish
    if time.time() - t0 < budget_s - 1:
        hf = _slsqp(best_h, 500000000, 2000, 1e-21)
        cf = _eval(hf)
        if cf < best_c5:
            best_h, best_c5 = hf.copy(), cf

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
