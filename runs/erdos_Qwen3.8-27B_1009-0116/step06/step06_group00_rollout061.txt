import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_k(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    m = min(K, 2 * n - 1)
    if m < 2 * n - 1:
        ti = np.argpartition(c, -m)[-m:]
    else:
        ti = np.arange(2 * n - 1)
    s = np.clip(alpha * (c[ti] - cm), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for t in range(m):
        k = ti[t]
        if w[t] < 1e-14:
            continue
        wk = w[t] * dx
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nest(h, alpha, steps, lr, K, mu=0.85):
    n = len(h)
    tgt = 0.5 * n
    f = _obj(h, alpha)
    v = np.zeros(n)
    for _ in range(steps):
        g = _grad_k(h, alpha, K)
        hp = h.copy()
        v = mu * v + (1 - mu) * g
        h_new = _proj(hp - lr * v, tgt)
        f_new = _obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            lr = min(lr * 1.02, 5.0)
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
            h = _proj(hp - lr * v, tgt)
            f = _obj(h, alpha)
    return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

            lo = mid
        else:
