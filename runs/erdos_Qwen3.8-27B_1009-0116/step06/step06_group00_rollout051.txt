import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_soft(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:-sp])
            g[:-sp] -= w[k] * h[sp:]
    return g

def _grad_topk(h, tk):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-tk:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / tk
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[k] * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:-sp])
            g[:-sp] -= w[k] * h[sp:]
    return g

def _nesterov(h, mode, param, steps, step0, t0, budget_s):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    for _ in range(steps):
        if time.time() - t0 > budget_s - 20:
            break
        if mode == 'soft':
            g = _grad_soft(h, param)
        else:
            g = _grad_topk(h, param)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        improved = False
        for _ in range(15):
            h_new = _proj_to_sum(h - step * v, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-16:
                h = h_new
                f = f_new
                step = min(step * 1.05, step0 * 5.0)
                improved = True
                break
            step *= 0.5
        if not improved:
            step = max(step * 0.4, 1e-9)
            h = _proj_to_sum(h_prev - step * v, target)
            f = _eval(h)
    return _repair(h, n), f

def _local_search(h, n_iter, rng, t0, budget_s, scale):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    stall = 0
    for _it in range(n_iter):
        if time.time() - t0 > budget_s - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i] = ni
            h[j] = nj
            c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)
            if c5n < c5c - 1e-19:
                c5c = c5n
                stall = 0
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
                stall += 1
                if stall > 300:
                    nw = rng.randint(n, size=4)
                    for m in range(4):
                        h[nw[m]] += 5e-5 * rng.randn()
                    h -= h.mean() - n * 0.5 / n
                    h = _repair(h, n)
                    c5c = _eval(h)
                    stall = 0
    return best_h, best_c

def _local_3pt(h, n_iter, rng, t0, budget_s, scale):
