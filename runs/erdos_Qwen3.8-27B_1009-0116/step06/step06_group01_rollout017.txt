    best_c5 = _eval(x)

        if time.time() - t0 > budget - 30:
            break
        t2 = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        if it > 0:
            th = (t - 1.0) / t2
            y = x + th * (y - x)
            y = _proj(y, target)
        ov = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(ov)[-n_top:]
        sm = ov[top_idx] - ov[top_idx].max()
        w = np.exp(sm)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for i, kk in enumerate(top_idx):
            g += w[i] * _lag_g(y, kk, dx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 2.0 / (0.5 + t2 * gn)
        x_new = _proj(y - step * g, target)
        for _sr in range(25):
            c5n = _eval(x_new)
            if c5n <= ov.max() * dx + step * step * gn * gn * 0.5:
                break
            step *= 0.5
            x_new = _proj(y - step * g, target)
        x = x_new
        t = t2
        c5x = _eval(x)
        if c5x < best_c5:
            best_c5 = c5x
            best = x.copy()
    return best, best_c5

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s - s.max())))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _lag_g(x, k, dx, n)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, method):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method=method, bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_cd(h, n, rng, scales, iters_per_scale, t0, budget):
    c5c = _eval(h)
    for sc in scales:
        for _ in range(iters_per_scale):
            if time.time() - t0 > budget - 20:
                break
            i = rng.randint(n)
            d = sc * rng.randn()
            ni = h[i] + d
            if ni < 0 or ni > 1:
                continue
            h[i] = ni
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h = _repair(h, n)
                h[i] = np.clip(h[i] - d, 0, 1)
                h = _repair(h, n)
    return h, c5c

def _binary_flip(h_bin, n, rng, n_iter, t0, budget):
    c5c = _eval(h_bin)
    stuck = 0
    for _ in range(n_iter):
        if time.time() - t0 > budget - 15:
            break
        i = rng.randint(n)
        h_bin[i] = 1.0 - h_bin[i]
        c5n = _eval(h_bin)
        if c5n < c5c - 1e-14:
            c5c = c5n
            stuck = 0
        else:
            h_bin[i] = 1.0 - h_bin[i]
            stuck += 1
            if stuck > 2000:
                break
    return h_bin, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP with moderate alpha
    if _tl() > 200:
        h1 = _slsqp(best_h, 5000, 3000, 'SLSQP')
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Nesterov accelerated projected gradient (main workhorse)
    if _tl() > 200:
        h2, c5 = _accel(best_h, best_n, 30.0, 5, 3000, t0, budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: SLSQP with high alpha (TRF method)
    if _tl() > 180:
        h3 = _slsqp(best_h, 50000, 3000, 'TRF')
        c5 = _eval(h3)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 4: Accelerated gradient with sharper alpha
    if _tl() > 180:
        h4, c5 = _accel(best_h, best_n, 80.0, 8, 3000, t0, budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h4.copy()

    # Phase 5: SLSQP ultra-high alpha
    if _tl() > 150:
        h5 = _slsqp(best_h, 200000, 3000, 'SLSQP')
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Coordinate descent polish
    if _tl() > 120:
        h6, c5 = _coord_cd(best_h, best_n, rng, [0.01, 0.003, 0.0008, 0.0002, 0.00005], 3000, t0, budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h6.copy()

    # Phase 7: Try antipodal solution
    h_anti = 1.0 - best_h[::-1].copy()
    h_anti = _repair(h_anti, best_n)
    c5_anti = _eval(h_anti)
    if c5_anti < best_c5:
        best_c5, best_h = c5_anti, h_anti.copy()
        h_anti = _slsqp(best_h, 30000, 2000, 'TRF')
        c5 = _eval(h_anti)
        if c5 < best_c5:
            best_c5, best_h = c5, h_anti.copy()

    # Phase 8: Multi-resolution
    if _tl() > 200:
        for n_try in [300, 420, 500, 620, 750, 900]:
            if _tl() < 60:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _slsqp(h_r, 20000, 2000, 'SLSQP')
            h_r, c5 = _accel(h_r, n_try, 40.0, 5, 1500, t0, budget_s)
            h_r = _coord_cd(h_r, n_try, rng, [0.005, 0.001, 0.0002], 2000, t0, budget_s)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 9: Binary local search on coarse grid + interpolate
    if _tl() > 150:
        for n_bb in [60, 100, 150]:
            if _tl() < 50:
                break
            h_bb = _repair(_resample(best_h, n_bb), n_bb)
            h_bb = (h_bb >= 0.5).astype(float)
            h_bb = _repair(h_bb, n_bb)
            h_bb, c5c = _binary_flip(h_bb, n_bb, rng, 30000, t0, budget_s)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, h_bb.copy(), n_bb
            h_up = _repair(_resample(h_bb, 500), 500)
            h_up = _slsqp(h_up, 30000, 1500, 'SLSQP')
            h_up, c5 = _accel(h_up, 500, 30.0, 5, 1000, t0, budget_s)
            c5 = _eval(h_up)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_up.copy(), 500

    # Phase 10: Perturbation restarts
    if _tl() > 100:
        for _rep in range(4):
            if _tl() < 40:
                break
            noise = rng.randn(best_n) * 0.008
            h_p = _repair(best_h + noise, best_n)
            h_p = _slsqp(h_p, 30000, 2000, 'TRF')
            h_p, c5 = _accel(h_p, best_n, 50.0, 5, 1500, t0, budget_s)
            h_p, c5 = _coord_cd(h_p, best_n, rng, [0.002, 0.0005, 0.0001], 2000, t0, budget_s)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 11: Final fine polish
    if _tl() > 20:
        h_fin = _coord_cd(best_h, best_n, rng, [0.0003, 0.00005, 0.00001], 5000, t0, budget_s)
        if h_fin[1] < best_c5:
            best_c5, best_h = h_fin[1], h_fin[0].copy()
        h_fin = _slsqp(best_h, 500000, 2000, 'SLSQP')
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
