import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sm_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _coord_descend(h, n_steps, rng, el_fn, budget_s):
    n = len(h)
    sh_vals = (n - 1) - np.arange(2 * n - 1)
    best_c5 = _eval(h)
    for step in range(n_steps):
        if el_fn() > budget_s - 30:
            break
        r = rng.random()
        if r < 0.4:
            i = rng.randint(n)
            off = rng.randint(1, max(2, n // 8)) * (1 if rng.random() < 0.5 else -1)
            j = (i + off) % n
        elif r < 0.7:
            i = rng.randint(n)
            j = rng.randint(n)
            if j == i:
                j = (j + 1) % n
        else:
            hi_idx = np.argsort(h)[-n // 8:]
            lo_idx = np.argsort(h)[:n // 8:]
            if rng.random() < 0.5:
                i, j = rng.choice(hi_idx), rng.choice(lo_idx)
            else:
                i, j = rng.choice(lo_idx), rng.choice(hi_idx)
            if i == j:
                j = (j + 1) % n
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        ip = i + sh_vals
        im = i - sh_vals
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        bi = np.where(vp, 1.0 - h[np.clip(ip, 0, n - 1)], 0.0) - np.where(vm, h[np.clip(im, 0, n - 1)], 0.0)
        jp = j + sh_vals
        jm = j - sh_vals
        vq = (jp >= 0) & (jp < n)
        vq2 = (jm >= 0) & (jm < n)
        bj = np.where(vq, 1.0 - h[np.clip(jp, 0, n - 1)], 0.0) - np.where(vq2, h[np.clip(jm, 0, n - 1)], 0.0)
        b = bi - bj
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-14:
            continue
        lo, hi = d_lo, d_hi
        for _ in range(45):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(c + m1 * b) < np.max(c + m2 * b):
                hi = m2
            else:
                lo = m1
        d_opt = (lo + hi) * 0.5
        h[i] += d_opt
        h[j] -= d_opt
        new_c5 = float(np.max(c + d_opt * b))
        if new_c5 < best_c5:
            best_c5 = new_c5
    return h, best_c5

def _fine_polish(h, n_iter, rng, el_fn, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_iter):
        if el_fn() > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
