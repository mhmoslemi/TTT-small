def _lssm(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue

        else:

def _slsqp(h, alpha, maxiter, ftol):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):

def _shift_grad(h, ki):
    nc = len(h)
    sh = (nc - 1) - ki
    g = np.zeros(nc)
    if sh >= 0:
        g[:nc-sh] += (1.0 - h[sh:])
        if sh > 0: g[sh:] -= h[:nc-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:nc-sp])

    return g

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hr.copy(), n_try
                hr2, c5p = _proj_subgrad(hr, 5, 2000, 0.03, 0.001, seed + n_try)
                if c5p < best_c5: best_c5, best_h = c5p, hr2.copy()

    # Phase 6: Top-k push (direct pair manipulation)
    if elapsed() < budget_s - 120:
        hn, c5n = _topk_push(best_h, 5, 3000, seed + 200)
        if c5n < best_c5: best_c5, best_h = c5n, hn.copy()
        hn = _slsqp(hn, 5000000, 2000, 1e-20)
        c5n = _eval(hn)
        if c5n < best_c5: best_c5, best_h = c5n, hn.copy()

    # Phase 7: Binary annealing on best
    if elapsed() < budget_s - 80:
        hbin = (best_h > 0.5).astype(float); hbin = _repair(hbin, len(hbin))
        t_rem = max(budget_s - 80 - elapsed(), 10)
        ns = max(int(t_rem * 500), 3000)
        ha, ca = _anneal_binary(hbin, 0.3, 0.001, ns, rng)
        ha = _slsqp(ha, 200000, 1500, 1e-19)
        ca = _eval(ha)
        if ca < best_c5: best_c5, best_h = ca, ha.copy()

    # Phase 8: Final coordinate micro-descent
    if elapsed() < budget_s - 5:
        h8 = best_h.copy(); c5c = _eval(h8); nc = len(h8)
        for scale in [0.001, 0.0003, 0.0001]:
            if elapsed() > budget_s - 5: break
            for _ in range(3000):
                if elapsed() > budget_s - 5: break
                i = rng.randint(nc); j = rng.randint(nc)
                if i == j: continue
                d = scale * rng.randn()
                if h8[i]+d < 0 or h8[i]+d > 1 or h8[j]-d < 0 or h8[j]-d > 1: continue
                h8[i] += d; h8[j] -= d
                c5n = _eval(h8)
                if c5n < c5c - 1e-15: c5c = c5n
                else: h8[i] -= d; h8[j] += d
        if c5c < best_c5: best_c5, best_h = c5c, h8.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
