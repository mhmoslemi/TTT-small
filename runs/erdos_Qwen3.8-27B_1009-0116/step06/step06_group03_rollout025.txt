        # h[i] contributes to c[k] if (i+k-n+1) in [0,n-1] OR (i-k+n-1) in [0,n-1]
        # h[j] contributes to c[k] if (j+k-n+1) in [0,n-1] OR (j-k+n-1) in [0,n-1]
        
        mask_i = ((i + affected - (n-1)).clip(0, n-1) == i + affected - (n-1)) & \
                 ((i + affected - (n-1)) >= 0) & ((i + affected - (n-1)) <= n-1)
        mask_i2 = ((i - affected + (n-1)) >= 0) & ((i - affected + (n-1)) <= n-1)
        mask_j = ((j + affected - (n-1)) >= 0) & ((j + affected - (n-1)) <= n-1)
        mask_j2 = ((j - affected + (n-1)) >= 0) & ((j - affected + (n-1)) <= n-1)
        mask = mask_i | mask_i2 | mask_j | mask_j2
        affected = affected[mask]
        
        if len(affected) < 2:
            continue
        
        # Compute a[k] and b[k] for each affected k
        # c[k] = sum_l h[l]*(1-h[l+k-n+1])
        # When we change h[i] -> h[i]+d, h[j] -> h[j]-d:
        # The change in c[k] comes from terms where l=i or l+k-n+1=i or l=j or l+k-n+1=j
        
        a = np.zeros(len(affected))
        b = np.zeros(len(affected))
        
        for idx_k, k in enumerate(affected):
            sh = k - (n - 1)  # shift
            # Term where h[i] is the "first" factor: contribution is h[i]*(1-h[i+sh])
            # Term where h[i] is the "second" factor: h[i-sh]*(1-h[i]), contribution to h[i] part: -h[i-sh]*h[i]
            # Similarly for j.
            
            # h[i] as first factor (l=i): valid if i+sh in [0,n-1], i.e., i+sh >= 0 and i+sh < n
            if 0 <= i + sh < n:
                a[idx_k] += best[i] * (1 - best[i + sh])
                b[idx_k] += (1 - best[i + sh])  # d h[i] contributes this much
                # Also h[i+sh] as second factor... wait no, h[i] is only the first factor here
            # h[i] as second factor (l+sh=i, l=i-sh): valid if 0 <= i-sh < n
            if 0 <= i - sh < n:
                a[idx_k] += best[i - sh] * (1 - best[i])
                b[idx_k] -= best[i - sh]  # d h[i] contributes -h[i-sh]*d... 
                # Wait: term is h[i-sh]*(1-h[i]) = h[i-sh] - h[i-sh]*h[i]
                # d/dh[i] = -h[i-sh], so change is -h[i-sh]*d
            # h[j] as first factor (l=j): valid if 0 <= j+sh < n
            if 0 <= j + sh < n:
                a[idx_k] += best[j] * (1 - best[j + sh])
                b[idx_k] -= (1 - best[j + sh])  # h[j] changes by -d
            # h[j] as second factor (l+sh=j, l=j-sh): valid if 0 <= j-sh < n
            if 0 <= j - sh < n:
                a[idx_k] += best[j - sh] * (1 - best[j])
                b[idx_k] += best[j - sh]  # h[j] changes by -d, so -h[j-sh]*(-d) = h[j-sh]*d
        
        # Now c[k] = a[k] + b[k] * d for affected k
        # Unaffected k: c[k] unchanged
        # We want to minimize max(max(affected), max(unaffected)) over d
        
        # Unaffected max:
        unaffected_max = c[~np.isin(np.arange(2*n-1), affected)].max() if len(affected) < 2*n-1 else 0.0
        
        # The optimal d minimizes max(max(a + b*d), unaffected_max)
        # Subject to: 0 <= best[i] + d <= 1 and 0 <= best[j] - d <= 1
        # => d in [max(0, best[j]-1), min(1-best[i], best[j])]
        
        d_lo = max(0.0, best[j] - 1.0)
        d_hi = min(1.0 - best[i], best[j])
        
        if d_lo >= d_hi:
            continue
        
        # Binary search on t = target max
        # Feasible d for target t:
        # For affected k: a[k] + b[k]*d <= t => if b[k]>0: d <= (t-a[k])/b[k]; if b[k]<0: d >= (t-a[k])/b[k]; if b[k]=0: a[k] <= t
        # Also d in [d_lo, d_hi]
        # And unaffected_max <= t
        
        t_lo = max(unaffected_max, a.max() - np.abs(b).max() * max(d_hi, d_lo, 1.0))  # rough lower
        t_hi = max(c.max(), a.max() + np.abs(b).max() * max(d_hi, d_lo, 1.0))  # rough upper
        
        # Actually just use c.max() as upper and 0 as lower
        t_lo = 0.0
        t_hi = c.max()
        
        for _ in range(30):
            t_mid = 0.5 * (t_lo + t_hi)
            if t_mid < unaffected_max:
                t_lo = t_mid
                continue
            # Check feasibility
            lo_feas = d_lo
            hi_feas = d_hi
            feasible = True
            for kk in range(len(affected)):
                if b[kk] > 1e-15:
                    hi_feas = min(hi_feas, (t_mid - a[kk]) / b[kk])
                elif b[kk] < -1e-15:
                    lo_feas = max(lo_feas, (t_mid - a[kk]) / b[kk])
                else:
                    if a[kk] > t_mid + 1e-15:
                        feasible = False
                        break
            if not feasible or lo_feas > hi_feas:
                t_lo = t_mid
            else:
                t_hi = t_mid
        
        d_opt = 0.5 * (t_lo + t_hi)  # Not quite right, we want the d that achieves t_hi
        # Actually we want the d within [d_lo, d_hi] that's feasible for t = t_hi
        t_target = t_hi
        lo_feas = d_lo
        hi_feas = d_hi
        for kk in range(len(affected)):
            if b[kk] > 1e-15:
                hi_feas = min(hi_feas, (t_target - a[kk]) / b[kk])
            elif b[kk] < -1e-15:
                lo_feas = max(lo_feas, (t_target - a[kk]) / b[kk])
        
        if lo_feas <= hi_feas:
            d_new = 0.5 * (lo_feas + hi_feas)
            d_new = np.clip(d_new, d_lo, d_hi)
            best[i] += d_new
            best[j] -= d_new
            best = _repair(best, n)
            c_new = _eval(best)
            if c_new < best_c - 1e-16:
                best_c = c_new
        
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c
