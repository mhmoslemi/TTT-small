def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a
            else:
                a = min(-diff, h[idx]); h[idx] -= a
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corrs(h)) * 2.0 / len(h))

    n = len(h)

        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

    for kk in idx:
        _lag_into(g, h, int(kk))
    return g / len(idx)

def _soft_obj(h, alpha):
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _soft_grad(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        _lag_into(g, h, k)
        g *= 1.0  # no-op to avoid compiler优化, just structure
        g = w[k] * (g / w[k]) if w[k] > 0 else g
    # redo properly
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g2 = np.zeros(n)
        _lag_into(g2, h, k)
        g += w[k] * g2
    return g * (2.0 / n)

def _cs(x):
    return x.sum() - len(x) * 0.5

def _cj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    f0 = _soft_obj(h, alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cs, 'jac': _cj}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h.copy(), n)

def _nesterov(h, t_end, n_max):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    f_cur = _eval(h_cur)
    best_h, best_f = h_cur.copy(), f_cur
    v = np.zeros(n)
    beta = 0.9
    step = 0.04
    ks = [1, 1, 1, 2, 2, 3, 5, 5, 8, 10, 15, 25]
    for it in range(n_max):
        if time.time() > t_end:
            break
        K = ks[min(it * len(ks) // n_max, len(ks) - 1)]
        g = _topk_grad(h_cur, K) * dx
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        v = beta * v + (1.0 - beta) * g
        step = min(step * 1.02, 0.4 / (ng + 1e-8), 0.3)
        acc = False
        for _ in range(15):
            h_new = _proj(h_cur - step * v, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f_cur + 1e-15:
                h_cur, f_cur = h_new, f_new
                if f_cur < best_f:
                    best_f, best_h = f_cur, h_cur.copy()
                step = min(step * 1.04, 0.3)
                acc = True
                break
            step *= 0.4
        if not acc:
            step *= 0.3
        beta = min(beta * 1.0001, 0.94)
    return best_h, best_f

def _pair_cd(h, t_end, n_max, rng):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h, best_c = h_cur.copy(), c_cur
    for _ in range(n_max):
        if time.time() > t_end:
            break
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        val1, val2 = h_cur[i1], h_cur[i2]
        for d in [0.003, 0.001, 0.0005, 0.0002]:
            nv1, nv2 = val1 + d, val2 - d
            if not (0 < nv1 < 1 and 0 < nv2 < 1):
                nv1, nv2 = val1 - d, val2 + d
                if not (0 < nv1 < 1 and 0 < nv2 < 1):
                    continue
            h_cur[i1], h_cur[i2] = nv1, nv2
            ht = _repair(h_cur, n)
            ct = _eval(ht)
            if ct < c_cur - 1e-17:
                h_cur, c_cur = ht, ct
                if c_cur < best_c:
                    best_c, best_h = c_cur, h_cur.copy()
                break
            h_cur[i1], h_cur[i2] = val1, val2
    return best_h, best_c

def _fine_cd(h, t_end, n_max, rng):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h, best_c = h_cur.copy(), c_cur
    scale = 3e-5
    for _ in range(n_max):
        if time.time() > t_end:
            break
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        val1, val2 = h_cur[i1], h_cur[i2]
        d = scale * rng.randn()
        nv1, nv2 = val1 + d, val2 - d
        if 0 < nv1 < 1 and 0 < nv2 < 1:
            h_cur[i1], h_cur[i2] = nv1, nv2
            ht = _repair(h_cur, n)
            ct = _eval(ht)
            if ct < c_cur - 1e-18:
                h_cur, c_cur = ht, ct
                if c_cur < best_c:
                    best_c, best_h = c_cur, h_cur.copy()
                    scale *= 0.97
            else:
                h_cur[i1], h_cur[i2] = val1, val2
    return best_h, best_c

def _soft_grad_v2(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        tmp = np.zeros(n)
        _lag_into(tmp, h, k)
        g += wk * tmp
    return g * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    for alpha in [500, 5000, 30000, 150000, 800000]:
        if time.time() > t0 + budget_s - 380:
            break
        hc = _slsqp(best_h, alpha, 2000)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 310:
        hc, c5 = _nesterov(best_h, t0 + budget_s - 309, 5000)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 240:
        samples = [np.roll(best_h, 1), np.roll(best_h, -1)]
        samples.append(best_h + 0.008 * rng.randn(n0))
        bm = best_h.copy()
        bl = n0 // 6
        st = rng.randint(0, n0 - bl)
        bm[st:st + bl] = bm[st:st + bl][::-1]
        samples.append(bm)
        samples.append(best_h + 0.02 * rng.randn(n0))
        for s in samples:
            if time.time() > t0 + budget_s - 180:
                break
            sc = _repair(s, n0)
            hc = _slsqp(sc, 80000, 1500)
            c5 = _eval(hc)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 200:
        hc, c5 = _nesterov(best_h, t0 + budget_s - 199, 5000)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 140:
        rem_t = t0 + budget_s - 130 - time.time()
        ni = min(int(rem_t / 0.003), 60000)
        hc, c5 = _pair_cd(best_h, t0 + budget_s - 130, max(ni, 2000), rng)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 80:
        hc = _slsqp(best_h, 3000000, 4000)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 45:
        hc, c5 = _nesterov(best_h, t0 + budget_s - 44, 4000)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    if time.time() < t0 + budget_s - 12:
        rem_t = t0 + budget_s - 10 - time.time()
        ni = min(int(rem_t / 0.002), 80000)
        hc, c5 = _fine_cd(best_h, t0 + budget_s - 10, max(ni, 2000), rng)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
