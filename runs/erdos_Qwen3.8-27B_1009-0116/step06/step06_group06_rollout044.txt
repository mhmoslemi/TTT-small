from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lag_gradient(h, k):
    """Gradient of correlation at lag k (0 to 2n-2) w.r.t. h."""
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def topk_subgradient(h, K=10, eps=1e-6):
    """Subgradient of max correlation using top-K lags."""
    n = len(h)
    c = compute_correlations(h)
    cmax = c.max()
    # Find lags within eps of max
    mask = c >= cmax - eps
    if mask.sum() == 0:
        mask = c >= cmax - 1e-3
    indices = np.where(mask)[0]
    # Take top K at most
    if len(indices) > K:
        order = np.argsort(c[indices])[::-1][:K]
        indices = indices[order]
    g = np.zeros(n)
    for k in indices:
        g += lag_gradient(h, k)
    g /= len(indices)
    return g, cmax, indices

def proj_box_sum(h, target):
    """Project onto {0<=x<=1, sum(x)=target} via bisection on water-filling level."""
    lo, hi = -2.0, 2.0

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

                diff -= a
            else:

                h[idx] -= a

        g, cmax, active = topk_subgradient(h_cur, K=K, eps=1e-4)

                lr = min(lr * 1.02, 0.05)
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c5

def smax_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

    m = c.max()

    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        g += wk * lag_gradient(x, k)
    g *= dx

import time

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def lag_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return h
    if s < 1e-12:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def lbfgs_run(h, alpha, t_end, maxiter=200):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def constr_fun(x):
        return x.sum() - n * 0.5

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def lag_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def topk_pgrad(h, t_end, K=20, lr0=0.006):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = eval_c5(h_cur)
    best_h, best_c5 = h_cur.copy(), c5_cur
    lr = lr0
    stuck = 0
    while time.time() < t_end:
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cmax = c.max()
        mask = c >= cmax - 2e-5
        idx = np.where(mask)[0]
        if len(idx) > K:
            idx = idx[np.argsort(c[idx])[::-1][:K]]
        g = np.zeros(n)
        for k in idx:
            g += lag_grad(h_cur, k)
        g *= dx / max(len(idx), 1)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.04]:
            h_new = proj(h_cur - step * g, n * 0.5)
            c5_new = eval_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
                improved = True
                stuck = 0
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                lr = min(lr * 1.015, 0.04)
                break
        if not improved:
            lr *= 0.65
            stuck += 1
            if stuck > 80:
                break
    return best_h, best_c5

def _eq_fun(x):
    return x.sum() - _global_n * 0.5

def _eq_jac(x):
    return np.ones(_global_n)

def smax_f(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def smax_g(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * lag_grad(x, k)
    return g * dx

def lbfgs_run(h, alpha, t_end, maxiter=300):
    global _global_n
    _global_n = len(h)
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(smax_f, h, jac=smax_g, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def perturb_restart(h, n, rng, scale=0.03):
    h2 = h + scale * rng.randn(n)
    return repair(h2, n)

def coord_descend(h, t_end, rng, n_iter=200000):
    n = len(h)
    h_cur = h.copy()
    c5_cur = eval_c5(h_cur)
    best_h, best_c5 = h_cur.copy(), c5_cur
    for _ in range(n_iter):

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        c5_new = eval_c5(h_cur)
        if c5_new < c5_cur - 1e-17:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return best_h, best_c5

def dual_ascent(h, t_end, K=15, lr0=0.004):
    n = len(h)

    h_cur = h.copy()

        g = np.zeros(n)

            g += w[k] * lag_grad(h_cur, k)
        wsum = w[idx].sum()
        if wsum > 0:

        g /= gn
        improved = False

                lr = min(lr * 1.01, 0.03)
