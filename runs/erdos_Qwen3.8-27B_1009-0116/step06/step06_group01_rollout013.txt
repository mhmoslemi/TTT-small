from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0/n)

def _eval(h):
    return float(np.max(_corr(h)))

def _grad_lag(h, k, dx):
    n = len(h)
    s = k - (n-1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        if s > 0: g[s:] -= h[:n-s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g * dx

def _ls_exp(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _ls_exp_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-10: continue
        g += w[k] * _grad_lag(h, k, dx)
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    def _f(x): return _ls_exp(x, alpha)
    def _gf(x): return _ls_exp_grad(x, alpha)
    def _sum_eq(x): return x.sum() - n*0.5
    def _sum_jac(x): return np.ones(n)
    try:

                       bounds=bnds, constraints=[{'type':'eq','fun':_sum_eq,'jac':_sum_jac}],
                       options={'maxiter':maxiter, 'ftol':ftol})

    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()

    best_h = h_curr.copy()

    """Pairwise coordinate descent."""

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _grad_lag(h, k, dx):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        if s > 0: g[s:] -= h[:n - s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g * dx

def _lsf(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lsf_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-10: continue
        g += w[k] * _grad_lag(h, k, dx)
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lsf, h, jac=lambda x: _lsf_grad(x, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h_old), endpoint=False), h_old)

    best_c5, best_h = c5c, h
    step = step0
    for _ in range(30):
        h_new = _repair(h - step * g, nc)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:

            if step < 0.5 and _ < expand:
                step *= 1.3
                continue
        step *= 0.45
        if step < 1e-8: break
    return best_h, best_c5

    dx = 2.0 / nc
    for r in range(rounds):

        top = np.argsort(c)[-min(n_top, len(c)):]
        g = np.zeros(nc)

        gn = np.linalg.norm(g)

        g /= gn

    return h, c5c

        mu = max(0.4, mu * 0.9998)
        yprev, yc = yc, h_cand + mu * (h_cand - yprev)
