        if np.clip(h - mid, 0, 1).sum() > target:

        else:

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c.max() + np.log(np.sum(np.exp(np.clip(alpha * (c - c.max()), -500, 500)))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()

        h_try = _lbfgs_run(best_h.copy(), 2048, 1e8, 200)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
