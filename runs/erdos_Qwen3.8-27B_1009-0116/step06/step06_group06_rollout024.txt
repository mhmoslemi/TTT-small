def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 1e-12: break
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
        else:
            order = np.argsort(h)
            for idx in order:
                if -diff <= 1e-12: break
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_max_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s = s - s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s = s - s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': n_iter, 'ftol': 1e-18, 'disp': False})
        if time.time() < t_limit and res.fun < _smooth_max_obj(h0, alpha):
            return _repair(res.x, n)
    except:
        pass
    return _repair(h0, n)

def _proj_grad_step(h, alpha, n_iter, t_limit, step0=0.01):
    n = len(h)
    h_cur = h.copy()
    f_prev = _smooth_max_obj(h_cur, alpha)

                    break
                h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 1e-12: break
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
        else:
            order = np.argsort(h)
            for idx in order:
                if -diff <= 1e-12: break
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s); w /= w.sum()

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g * dx

def _slsqp(h0, alpha, n_iter, t_lim):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smax_obj, h0, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': n_iter, 'ftol': 1e-18, 'disp': False})
        if time.time() < t_lim and res.fun < _smax_obj(h0, alpha):
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h0, n)

def _proj_grad(h, alpha, n_iter, t_lim, step0=0.01):
    n = len(h); h_c = h.copy()
    f_p = _smax_obj(h_c, alpha); step = step0
    for _ in range(n_iter):

            if f_n < f_p:

            step *= 0.5
        if not acc: step *= 0.3
    return h_c

def _active_subgrad(h, n_iter, t_lim, n_lags=3):

        if c5n < c5: h_c = h_n; c5 = c5n; step = min(step * 1.05, 0.15)
        else: step *= 0.65
    return h_c, c5

def _block_init(m, rng, style):
    x = np.linspace(0, 2, m, endpoint=False)
    if style == 0:
        v = rng.uniform(0, 1, m)
    elif style == 1:

    elif style == 2:
        nb = rng.randint(2, 6); v = np.zeros(m)
        for b in range(nb):
            s, e = b * m // nb, (b + 1) * m // nb
            v[s:e] = rng.choice([0.0, 0.3, 0.7, 1.0])
    else:
        v = 0.5 + 0.5 * np.abs(np.sin(np.pi * rng.randint(1, 3) * x)) ** rng.uniform(0.5, 2.5)
    return _repair(v, m)

def _block_to_full(bv, n_full):
    out = np.zeros(n_full)
    for i in range(len(bv)):
        s, e = int(i * n_full / len(bv)), int((i + 1) * n_full / len(bv))
        out[s:e] = bv[i]
    return _repair(out, n_full)

def _coord_desc(h, n_iter, t_lim, rng):
    n = len(h); h_c = h.copy(); c5 = _eval(h_c)
    ds = [1e-4, 5e-5, 1e-5, 5e-6, 1e-6]
    for _ in range(n_iter):
        if time.time() > t_lim: break
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = ds[rng.randint(len(ds))] * (1 if rng.rand() < 0.5 else -1)
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_c[i], h_c[j] = ni, nj
            cn = _eval(h_c)
            if cn < c5: c5 = cn
            else: h_c[i], h_c[j] = h_c[i] - d, h_c[j] + d
    return h_c, c5
