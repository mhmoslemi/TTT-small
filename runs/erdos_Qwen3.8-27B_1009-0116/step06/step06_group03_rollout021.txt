def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(len(h), 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_top(h, top_k, alpha):

    n = len(h)
    c = _corrs(h)
    # Find top-k correlations
    idx_sorted = np.argsort(c)[::-1][:top_k]
    vals = c[idx_sorted]
    m = vals[0]
    s = np.clip(alpha * (vals - m), -100, 100)
    w = np.exp(s - s.max())
    w /= w.sum()
    
    dx = 2.0 / n
    g = np.zeros(n)
    for j, k in enumerate(idx_sorted):
        wk = w[j] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

        s = np.sum(np.clip(h - mid, 0, 1))
