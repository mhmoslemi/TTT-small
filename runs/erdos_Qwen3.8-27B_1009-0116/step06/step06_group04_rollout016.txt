import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12: break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_ls(x, alpha):
    """Gradient of log-sum-exp smoothed objective"""
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _grad_topk(x, K):
    """Subgradient using top-K worst shifts (equal weight)"""
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        wk = dx / K
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _nesterov(x0, n_iter, lr_sched, alpha_start, alpha_end, K_top, seed, deadline):
    n = len(x0)
    x = x0.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_c = _eval(x)
    rng = np.random.RandomState(seed)
    for i in range(n_iter):
        if time.time() > deadline: break
        t = i + 1
        frac = i / max(n_iter - 1, 1)
        lr = lr_sched(frac)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        if K_top is not None and K_top <= 2 * n - 1:
            g = _grad_topk(x, K_top)
        else:
            g = _grad_ls(x, alpha)
        v = 0.9 * v + g  # Nesterov-like momentum
        x = x - lr * v
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
        # Periodic perturbation to escape local optima
        if t % 1000 == 0:
            x = x + 0.003 * rng.randn(n)
            x = _repair(x, n)
    return best_x, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_ls(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _grad_topk(x, K):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        wk = dx / K
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _nestern_v(x0, n_iter, lr_init, lr_end, alpha_init, alpha_end, use_topk, K, seed, deadline):
    n = len(x0)
    x = x0.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_c = _eval(x)
    rng = np.random.RandomState(seed)
    mu = 0.9
    for i in range(n_iter):
        if time.time() > deadline:
            break
        frac = (i + 1) / n_iter
        lr = lr_init * (lr_end / lr_init) ** frac
        alpha = alpha_init * (alpha_end / alpha_init) ** frac
        if use_topk and K is not None:
            g = _grad_topk(x, K)
        else:
            g = _grad_ls(x, alpha)
        v = mu * v + (1.0 - mu) * g
        x = x - lr * v
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
        if (i + 1) % 800 == 0:
            x = x + 0.004 * rng.randn(n)
            x = _repair(x, n)
    return best_x, best_c

def _slsqp_opt(h, alpha, maxiter, deadline):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def jac(x):
        return _grad_ls(x, alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        if time.time() < deadline:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _pair_cd(h, n, rng, deadline, n_pairs):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    for _ in range(n_pairs):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        lo = max(0.0, -(x[i] - 0.0), -(x[j] - 1.0))
        hi = min(1.0 - x[i], x[j])
        if hi - lo < 1e-10:
            continue
        a, b = lo, hi
        for _ in range(25):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            x1 = x.copy(); x1[i] += m1; x1[j] -= m1; x1 = _repair(x1, n)
            x2 = x.copy(); x2[i] += m2; x2[j] -= m2; x2 = _repair(x2, n)
            if _eval(x1) < _eval(x2):
                b = m2
            else:
                a = m1
        d = (a + b) / 2.0
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_x = x.copy()
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    out = np.interp(xn, xo, h)
    return _repair(out, n_new)
