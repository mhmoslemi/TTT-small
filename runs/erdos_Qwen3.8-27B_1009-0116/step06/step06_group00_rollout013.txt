import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sobj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    g *= dx
    return g

def _topK_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    top = np.argsort(c)[-K:]
    g = np.zeros(n)
    for k in top:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    g *= dx / K
    return g

def _nesterov(h, t0, t_end):
    n = len(h)
    target = n / 2.0
    v = np.zeros(n)
    mu = 0.85
    step = 0.004
    f = _sobj(h, 500)
    restart = 0
    for i in range(200000):
        if time.time() - t0 > t_end:
            break
        frac = min(i / 5000.0, 1.0)
        alpha = 500.0 + frac * 499500.0
        if i % 300 == 0 and i > 0:
            v *= 0.3
        g = _sgrad(h, alpha)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, target)
        f_new = _sobj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.015, 0.06)
            restart = 0
        else:
            step *= 0.6
            h_new = _proj(h - step * v, target)
            f = _sobj(h_new, alpha)
            h = h_new
            restart += 1
            if restart > 20:
                v = np.zeros(n)
                step = 0.003
                restart = 0
    return _repair(h, n)

def _topK_descend(h, K, t0, t_end, max_rounds):
    n = len(h)
    target = n / 2.0
    c5c = _eval(h)
    for _ in range(max_rounds):
        if time.time() - t0 > t_end:
            break
        g = _topK_grad(h, K)
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4]:

            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _local3(h, rng, t0, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() - t0 < t_end:

        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2

            h[i], h[j], h[k] = ni, nj, nk

                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

    c5c = _eval(h)

    best_c = c5c

def _anneal_step(h, rng, temp, target):
    n = len(h)
    i = rng.randint(n)
    j = (i + rng.randint(1, 15)) % n

    ni, nj = h[i] + d, h[j] - d

        h_old_c5 = _eval(h)

            return _proj(h, target), h_new_c5
