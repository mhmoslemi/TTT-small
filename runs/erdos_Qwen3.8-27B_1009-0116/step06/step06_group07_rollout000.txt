                if hs > 1e-15:

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lsmax_obj(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.logsumexp(s) / alpha

def _lsmax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

        # Actually, let's use a fixed small lr with backtracking
        lr = 0.01 / alpha * 100
        for it in range(max_iters // len(alpha_list)):

            obj, g_full = _lsmax_obj_grad_full(h_full, alpha)
            # Average gradient over periods
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p*T:(p+1)*T]
            g_p /= m
            # FISTA step
            x_new = np.clip(y - lr * g_p, 0, 1)
            # Enforce sum constraint
            x_new = _repair(x_new, T)
            # Evaluate
            h_try = np.tile(x_new, m)
            c_try = _eval(h_try)
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - h_p)
            t_k = t_new
            h_p = x_new
            if c_try < best_c:
                best_c = c_try
                best = h_p.copy()
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h
                rs = room.sum()
                if rs > 1e-15:
                    h += diff * room / rs
            else:
                hs = h.sum()
                if hs > 1e-15:
                    h += diff * h / hs
        h = np.clip(h, 0.0, 1.0)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lsmax_og(h_full, alpha):
    n = len(h_full)
    c = np.correlate(h_full, 1.0 - h_full, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    obj = m + float(np.log(np.sum(np.exp(s)))) / alpha
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-11:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h_full[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h_full[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h_full[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h_full[i2 + sp]
    return obj, g

def _fista_period(h_p, T, n, alphas, iters_per, lr_scale=0.02, seed=0):
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(h_p, m))
    for ai, alpha in enumerate(alphas):
        xk = h_p.copy()
        y = h_p.copy()
        t_k = 1.0
        lr = lr_scale / (alpha * 0.01)
        for it in range(iters_per):
            h_full = np.tile(y, m)
            _, g_full = _lsmax_og(h_full, alpha)
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p * T:(p + 1) * T]
            g_p /= m
            gn = np.linalg.norm(g_p)
            if gn < 1e-14:
                break
            x_new = _repair(y - lr * g_p / max(gn, 1e-6) * min(gn, lr * 0.5), T)
            h_try = np.tile(x_new, m)
            c_try = _eval(h_try)
            if c_try < best_c:
                best_c = c_try
                best = x_new.copy()
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - xk)

    best_c = _eval(h)
    xk = h.copy()

        h = np.where(h > 0.5, 1.0, 0.0)

        c = _eval(h)
        if c < best_c:

        else:
            h[i], h[j] = h[j], h[i]
    return h, best_c

def _coord_descent(h, n_iter, seed=0):
    n = len(h)
    c5 = _eval(h)
    rng = np.random.RandomState(seed)
    for scale in [0.01, 0.003, 0.001, 0.0003]:

            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (rng.rand() * 2 - 1)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5

    return x.sum() - len(x) * 0.5

    return np.ones(len(x))

def _slsqp_period(h_p, T, n, alpha, maxiter):
    m = n // T
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(lambda x: _lsmax_og(np.tile(x, m), alpha)[0],
                       h_p, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = _repair(res.x, T)
        return h_new, _eval(np.tile(h_new, m))
    except Exception:
        return h_p, _eval(np.tile(h_p, m))
