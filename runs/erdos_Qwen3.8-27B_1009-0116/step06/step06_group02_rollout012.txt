                h = ht; cur = ct; moved = True
                break
        if not moved:
            break
    return h, cur

def _coord_cd(h, n, n_iter, t_end, rng_seed=42):
    h = h.copy()
    cur = _eval(h)
    rng = np.random.RandomState(rng_seed)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.randn() * 0.003
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < cur - 1e-15:
            cur = cn
        else:
            h[i], h[j] = ni - d, nj + di if False else (h[i] - d, h[j] + d)
    return h, cur

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _slsqp_polish(h0, alpha, maxiter, t_end):
    n = len(h0)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(s - s.max()))) / alpha) * dx

    def _obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        return _grad_w(x, w, n)

    def _cons_fun(x):
        return x.sum() - n * 0.5

    def _cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        if time.time() > t_end:
            return h0
        res = minimize(_obj, h0, jac=_obj_jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        return _repair(res.x, n)
    except:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_g = t0 + budget_s

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def rem():
        return t_end_g - time.time()

    # Phase 1: Nesterov at original resolution with increasing alpha
    for alpha in [3000, 15000, 80000, 300000]:
        if rem() < 120:
            break
        t_end = t0 + rem() - 80
        h_new, c5 = _nesterov(best_h, alpha, 8000, t_end, top_k=None, lr_init=0.015)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Reweighting on worst shifts
    if rem() > 100:
        for top_m in [8, 5, 3, 2]:
            if rem() < 70:
                break
            h_r, c5 = _reweight_opt(best_h, 3000, top_m, t0 + rem() - 30, alpha=50000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()

    # Phase 3: Multi-resolution exploration
    if rem() > 250:
        for n_try in [250, 350, 450, 550, 400]:
            if rem() < 180:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            for alpha in [10000, 60000, 250000]:
                if rem() < 150:
                    break
                hr, c5 = _nesterov(hr, alpha, 5000, t0 + rem() - 50, top_k=None, lr_init=0.012)
            c5 = _eval(hr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hr.copy()
                best_n = n_try

    # Phase 4: SLSQP polish
    if rem() > 150:
        h_s = _slsqp_polish(best_h, 200000, 4000, t0 + rem() - 60)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s
            best_n = len(h_s)

    # Phase 5: Top-k Nesterov again at best resolution
    if rem() > 80:
        for top_k in [6, 4, 2]:
            if rem() < 50:
                break
            h_n, c5 = _nesterov(best_h, 100000, 4000, t0 + rem() - 20, top_k=top_k, lr_init=0.008)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_n.copy()

    # Phase 6: Reweighting final pass
    if rem() > 50:
        h_r, c5 = _reweight_opt(best_h, 2000, 2, t0 + rem() - 15, alpha=100000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_r.copy()

    # Phase 7: Coordinate descent
    if rem() > 25:
        h_cd, c5 = _coord_cd(best_h, best_n, 8000, t_end_g - 8, rng_seed=99)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()

    # Phase 8: Final SLSQP
    if rem() > 20:
        h_f = _slsqp_polish(best_h, 500000, 5000, t_end_g - 5)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f
            best_n = len(h_f)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
