            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, steps, lr, mu, bc):
    n = len(h)
    f = _obj_lse(h, alpha)
    v = np.zeros(n)
    step = lr
    for _ in range(steps):
        if bc():
            break
        g = _grad_lse(h, alpha)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_lse(h_new, alpha)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, lr * 5)
        else:
            step *= 0.5
            h = _proj_to_sum(h - step * v, n * 0.5)
            f = _obj_lse(h, alpha)
    return _repair(h, n)

def _slsqp_obj(x, alpha):
    return _obj_lse(x, alpha)

def _slsqp_grad(x, alpha):
    return _grad_lse(x, alpha)

def _slsqp_cfun(x, n):
    return x.sum() - n * 0.5

def _slsqp_cjac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _bump_init(n, rng, nb=None):
    if nb is None:
        nb = rng.randint(2, 6)
    h = np.zeros(n)
    x = np.linspace(0, 2, n)
    for _ in range(nb):
        c = rng.uniform(0.15, 1.85)
        w = rng.uniform(0.06, 0.25)
        h += np.exp(-0.5 * ((x - c) / w) ** 2)
    return _repair(np.clip(h, 0, 1), n)

def _focus_descent(h, n_top, steps, lrs, bc):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for step in lrs:
        for _ in range(steps):
            if bc():
                return h, c5
            c = np.correlate(h, 1.0 - h, mode='full')
            top_idx = np.argsort(c)[::-1][:n_top]
            shifts = [(n - 1) - k for k in top_idx]
            g = np.zeros(n)
            for sh in shifts:
                if sh == 0:
                    g += (1.0 - h)
                elif sh > 0:
                    g[:n - sh] += (1.0 - h[sh:])
                    g[sh:] -= h[:n - sh]
                else:
                    sp = -sh
                    g[sp:] += (1.0 - h[:n - sp])
                    g[:n - sp] -= h[sp:]
            g *= dx / n_top
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h, c5 = h_new, c5n
            else:
                break
    return h, c5

def _polish(h, n_iter, rng, bc):
    n = len(h)
    c5 = _eval(h)
    best_h, best_c = h.copy(), c5
    for _ in range(n_iter):
        if bc():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 4e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if -1e-12 <= ni <= 1 + 1e-12 and -1e-12 <= nj <= 1 + 1e-12:
            h[i], h[j] = max(0.0, min(1.0, ni)), max(0.0, min(1.0, nj))
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni, nj
    return best_h, best_c

def _perturb_restart(h, rng, alpha, steps, bc):
    n = len(h)
    h_p = h.copy()
    mask = rng.rand(n) < 0.1
    h_p[mask] = np.clip(h_p[mask] + 0.01 * rng.randn(mask.sum()), 0, 1)
    h_p = _proj_to_sum(h_p, n * 0.5)
    h_o = _nesterov(h_p, alpha, steps, 0.003, 0.85, bc)
    return _repair(h_o, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 12

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Low-res with diverse starts
    for n_low in [80, 120]:
        if _el() > 100:
            break
        f_d = max(1, n0 // n_low)
        n_l = max(40, n0 // f_d)
        cands = [_downsample(h, f_d), _bump_init(n_l, rng), _bump_init(n_l, rng)]
        for hc in cands:
            hc = _repair(hc.copy(), n_l)
            for alpha in [500, 3000, 20000, 80000]:
                if _el() > 120:
                    break
                ho = _nesterov(hc, alpha, 120, 0.004, 0.85, _bc)
                hu = _repair(_upsample(ho, max(1, n0 // n_l)), n0)
                cu = _eval(hu)
                if cu < best_c5:
                    best_c5, best_h = cu, hu.copy()
                hc = ho

    # Phase 2: SLSQP at n0 with increasing alpha
    if _el() < budget_s - 300:
        for alpha in [3000, 20000, 80000]:
            if _el() > budget_s - 250:
                break
            hc = _slsqp(best_h.copy(), alpha, 1200, 1e-17)
            c = _eval(hc)
            if c < best_c5:
                best_c5, best_h = c, hc.copy()

    # Phase 3: Focused gradient descent on top lags
    if _el() < budget_s - 100:
        for ntop in [2, 3]:
            if _el() > budget_s - 80:
                break
            lrs = [0.03, 0.01, 0.005, 0.002, 0.001]
            hf, cf = _focus_descent(best_h.copy(), ntop, 300, lrs, _bc)
            if cf < best_c5:
                best_c5, best_h = cf, hf.copy()

    # Phase 4: Perturbation restarts
    if _el() < budget_s - 60:
        for _ in range(3):
            if _el() > budget_s - 50:
                break
            hp = _perturb_restart(best_h, rng, 30000, 100, _bc)
            c = _eval(hp)
            if c < best_c5:
                best_c5, best_h = c, hp.copy()

    # Phase 5: Nesterov high-alpha polish
    if _el() < budget_s - 40:
        hc = _nesterov(best_h, 200000, 150, 0.0015, 0.87, _bc)
        c = _eval(hc)
        if c < best_c5:
            best_c5, best_h = c, hc.copy()

    # Phase 6: 3-point local search
    if _el() < budget_s - 20:
        n = len(best_h)
        c5c = _eval(best_h)
        nt = min(int((budget_s - 20 - _el()) / 0.006), 25000)
        for _ in range(nt):
            if _bc():
                break
            i = rng.randint(n)
            j = (i + rng.randint(2, 7)) % n
            k = (i + rng.randint(8, 15)) % n
            d1 = 8e-6 * rng.randn()
            d2 = 8e-6 * rng.randn()
            ni, nj, nk = best_h[i] + d1, best_h[j] + d2, best_h[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                best_h[i], best_h[j], best_h[k] = ni, nj, nk
                c5n = _eval(best_h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                else:
                    best_h[i], best_h[j], best_h[k] = ni - d1, nj - d2, nk + d1 + d2

    # Phase 7: Fine 2-point polish
    if _el() < budget_s - 3:
        ntries = min(int((budget_s - 3 - _el()) / 0.002), 100000)
        best_h, best_c5 = _polish(best_h, ntries, rng, _bc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
