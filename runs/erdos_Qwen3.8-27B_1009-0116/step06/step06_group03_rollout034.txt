    x = h.copy()
    y = h.copy()
    best = h.copy()
    best_c = _eval(best)
    beta = 1.0

        y = x + (beta - 1) / beta_new * (x - y)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / len(h)

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    kss = np.where(w > 1e-10)[0]
    for k in kss:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.array([x.sum() - 0.5 * n]),
             'jac': lambda x: np.array([np.ones(n)])}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _momentum_pg(h, n_iter, alpha, step0, rng):
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c = _eval(best)
    mu = 0.9
    step = step0
    stuck = 0
    for t in range(n_iter):
        c_y = _corr(x)
        m_y = c_y.max()
        s_y = np.clip(alpha * (c_y - m_y), -500, 500)
        w_y = np.exp(s_y - s_y.max()); w_y /= w_y.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        kss = np.where(w_y > 1e-10)[0]
        for k in kss:
            sh = (n - 1) - k
            wk = w_y[k] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        v = 0.9 * v + 0.1 * g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            x_new = _proj(x - step * f * v, 0.5 * n)
            c_new = _eval(x_new)
            if c_new < best_c - 1e-14:
                x = x_new
                best_c = c_new
                best = x.copy()
                step = min(step * 1.1, 0.05)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.7
            stuck += 1
            if stuck > 80:
                x = _repair(x + 0.02 * rng.randn(n), n)
                best_c = _eval(best)
                step = step0
                stuck = 0
                v = np.zeros(n)
    return best, best_c

def _top_subgrad(h, n_iter, step0, top_m, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    step = step0
    stuck = 0
    for _ in range(n_iter):
        c = _corr(best)
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        dx = 2.0 / n
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - best[sh:])
                if sh > 0:
                    g[sh:] -= best[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - best[:n - sp])
                g[:n - sp] -= best[sp:]
        g *= dx / top_m
        v = 0.85 * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-14:
                best = h_new
                best_c = c_new
                step = min(step * 1.15, 0.04)
                improved = True
                stuck = 0
                break
        if not improved:

            stuck += 1
            if stuck > 60:
                best = _repair(best + 0.03 * rng.randn(n), n)
                best_c = _eval(best)
                step = step0
                v = np.zeros(n)
                stuck = 0
    return best, best_c

def _fine_tune(h, n_tries, rng, step_sizes):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        ss = rng.choice(step_sizes)
        d = ss * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        h[i] = np.clip(ni, 0, 1)
        h[j] = np.clip(nj, 0, 1)
        h = _repair(h, n)
        cn = _eval(h)
        if cn < c_cur - 1e-14:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i] = np.clip(h[i] - d, 0, 1)
            h[j] = np.clip(h[j] + d, 0, 1)
            h = _repair(h, n)
    return best, best_c
