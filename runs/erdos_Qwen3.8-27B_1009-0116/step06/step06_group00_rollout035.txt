    target = n / 2.0

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.9  # momentum
    eta = 0.01  # initial step size
    best_h = h.copy()
    best_f = f
    stall = 0
    
    for it in range(n_steps):
        if time_check():
            break
        
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = c.max()
        # Top lags

        top_k = np.where(c >= thresh)[0]
        if len(top_k) == 0:
            top_k = np.array([np.argmax(c)])
        if len(top_k) > s_params:
            top_k = np.argsort(c)[-s_params:]
        
        g = _grad_exact(h, top_k)
        
        # Nesterov step
        v = mu * v + (1 - mu) * g
        h_new = _proj_to_sum(h - eta * v, n * 0.5)

                stall = 0

                stall += 1

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0. 0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_top(h, top_k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g * (dx / len(top_k))

def _softmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, steps, step0, t0, budget):
    n = len(h)
    f = _softmax_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    best_h, best_f = h.copy(), _eval(h)
    for _ in range(steps):
        if time.time() - t0 > budget - 30:
            break
        g = _softmax_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _softmax_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 3)
            c5 = _eval(h)
            if c5 < best_f:

                best_h = h.copy()
        else:
            step *= 0.5
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _softmax_obj(h, alpha)
    return best_h, best_f

def _subgrad_loop(h, n_steps, top_n, t0, budget):

    f = _eval(h)
    v = np.zeros(n)
    mu = 0.9
    eta = 0.005
    best_h, best_f = h.copy(), f

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
