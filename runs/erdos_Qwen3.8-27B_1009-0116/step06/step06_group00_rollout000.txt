                stalls = 0
            else:
                stalls += 1
        else:
            stalls += 1
        if stalls > 500:
            i = rng.randint(0, n - 5)
            delta = rng.randn(5) * 0.05
            delta -= delta.mean()
            cur[i:i+5] += delta
            cur = _repair(cur, n)
            cur_c = _eval(cur)
            stalls = 0
    return best_h, best_c

def _block_swap(h, n, rng, t0, budget, n_iter, scale):
    best_h = h.copy()
    best_c = _eval(best_h)
    cur = h.copy()
    cur_c = best_c
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        B = 1 + rng.randint(0, 4)
        i = rng.randint(0, n - B)
        j = rng.randint(0, n - B)
        if abs(i - j) < B + 1:
            continue
        d = scale * rng.randn()
        ni = cur[i:i+B] + d
        nj = cur[j:j+B] - d
        if ni.min() >= 0 and ni.max() <= 1 and nj.min() >= 0 and nj.max() <= 1:
            cur[i:i+B] = ni
            cur[j:j+B] = nj
            c_new = _eval(cur)
            if c_new < cur_c - 1e-19:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = cur.copy()
            else:
                cur[i:i+B] -= d
                cur[j:j+B] += d
    return best_h, best_c

def _fine_search(h, n, rng, t0, budget, n_iter, scale):
    best_h = h.copy()
    best_c = _eval(best_h)
    cur = h.copy()
    cur_c = best_c
    for it in range(n_iter):
        if time.time() - t0 > budget - 2:
            break
        if rng.rand() < 0.55:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = scale * rng.randn()
            ni, nj = cur[i] + d, cur[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                cur[i], cur[j] = ni, nj
                c_new = _eval(cur)
                if c_new < cur_c - 1e-19:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = cur.copy()
                else:
                    cur[i], cur[j] = ni - d, nj + d
        else:
            i1 = rng.randint(n)
            i2 = (i1 + rng.randint(2, 7)) % n
            i3 = (i1 + rng.randint(8, 14)) % n
            d1 = scale * rng.randn()
            d2 = scale * rng.randn()
            nv1, nv2, nv3 = cur[i1]+d1, cur[i2]+d2, cur[i3]-d1-d2
            if 0 <= nv1 <= 1 and 0 <= nv2 <= 1 and 0 <= nv3 <= 1:
                cur[i1], cur[i2], cur[i3] = nv1, nv2, nv3
                c_new = _eval(cur)
                if c_new < cur_c - 1e-19:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = cur.copy()
                else:
                    cur[i1], cur[i2], cur[i3] = nv1-d1, nv2-d2, nv3+d1+d2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Quick gradient polish (top 4, 6, 8 lags)
    for tk in [4, 6, 8]:
        if _rem() < 80:
            break
        h, c = _grad_descent(best_h, n, rng, t0, budget_s, 3000, tk, 0.05)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Aggressive SA to escape local minimum
    if _rem() > 200:
        h, c = _sa_blocks(best_h, n, rng, t0, budget_s, 8000, 0.08, 0.002, 3, 12)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 3: Gradient after SA
    if _rem() > 100:
        for tk in [5, 8, 3, 10]:
            if _rem() < 80:
                break
            h, c = _grad_descent(best_h, n, rng, t0, budget_s, 4000, tk, 0.03)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    # Phase 4: Second SA round (cooler)
    if _rem() > 150:
        h, c = _sa_blocks(best_h, n, rng, t0, budget_s, 6000, 0.04, 0.001, 2, 8)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 5: Block swaps
    if _rem() > 80:
        for sc in [0.02, 0.01, 0.005]:
            if _rem() < 60:
                break
            h, c = _block_swap(best_h, n, rng, t0, budget_s, 15000, sc)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    # Phase 6: Fine local search (bulk of remaining time)
    if _rem() > 30:
        rem = _rem() - 15
        n_iter = int(min(rem / 0.00012, 5000000))
        h, c = _fine_search(best_h, n, rng, t0, budget_s, n_iter, 5e-5)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 7: Final gradient + ultra-fine
    if _rem() > 20:
        h, c = _grad_descent(best_h, n, rng, t0, budget_s, 2000, 3, 0.02)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
    if _rem() > 5:
        n_iter = int(min(_rem() / 0.00012, 3000000))
        h, c = _fine_search(best_h, n, rng, t0, budget_s, n_iter, 2e-5)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
