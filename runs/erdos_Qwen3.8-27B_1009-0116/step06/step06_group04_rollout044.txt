def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):

def _smoothmax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(s))) / alpha
    w = np.exp(s - s.max())
    w /= w.sum()
    return sm, w

def _sturmian_binary(n):
    """Create a balanced (0,1) sequence using a Sturmian/mechanical word"""
    n_ones = n // 2
    # Use the "golden ratio" placement
    positions = np.zeros(n_ones, dtype=int)
    for i in range(n_ones):
        positions[i] = int(i * n / n_ones + 0.5) % n  # Hmm, this is just every 2nd
    # Better: use van der Corput / low-discrepancy
    # Position i goes to floor(i * n / n_ones) for i = 0..n_ones-1
    # This gives 0, 2, 4, ... which is a checkerboard
    # For a better distribution, use:
    positions = np.array([int((i + 0.5) * n / n_ones) for i in range(n_ones)])
    positions = np.sort(positions)
    h = np.zeros(n)
    h[positions] = 1.0
    return h

                    x[j] = 0.0

                        best_x = x.copy()
                else:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    # Phase 3: Binary SA + re-Adam (250-420s)
    dl3 = t0 + 410
    xb, cb = _binary_sa(best_h, best_n, rng, 150000, dl3)
    if cb < best_c5:
        best_c5, best_h = cb, xb.copy()
    if time.time() < dl3 - 20:
        al2 = [5000, 50000, 500000, 5000000, 30000000]
        lr2 = [0.01, 0.005, 0.002, 0.0008, 0.0003]
        x2, c2 = _adam_run(best_h, al2, lr2, 5000, seed + 7, dl3, 500)
        if c2 < best_c5:
            best_c5, best_h = c2, x2.copy()

    # Phase 4: SLSQP polish (410-560s)
    dl4 = t0 + 550
    for a_slq in [200000, 2000000, 10000000, 50000000]:
        if time.time() > dl4:
            break
        hnew = _slsqp_run(best_h, a_slq, 2500)
        c5 = _eval(hnew)
        if c5 < best_c5:
            best_c5, best_h = c5, hnew.copy()

    # Phase 5: Multi-resolution (550-700s)
    dl5 = t0 + 690
    n_cands = sorted(set([best_n - 20, best_n - 10, best_n, best_n + 10,
                          best_n + 20, 300, 320, 350, 380, 400]))
    for nt in n_cands:
        if time.time() > dl5:
            break
        if nt < 60 or nt > 500 or nt % 2 != 0:
            continue
        hr = _repair(_resample_h(best_h, nt), nt)
        x5, c5 = _adam_run(hr, [1000, 10000, 100000, 1000000, 5000000],
                           [0.01, 0.005, 0.002, 0.0008, 0.0003],
                           4000, seed + nt, dl5, 400)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, x5.copy(), nt
        if time.time() < dl5 - 10:
            hs5 = _slsqp_run(x5, 5000000, 1500)
            c5s = _eval(hs5)
            if c5s < best_c5:
                best_c5, best_h, best_n = c5s, hs5.copy(), nt

    # Phase 6: Subgradient refinement (690-880s)
    dl6 = t0 + 870
    xsg, csg = _subgrad_refine(best_h, best_n, dl6, rng)
    if csg < best_c5:
        best_c5, best_h = csg, xsg.copy()
    if time.time() < dl6 - 30:
        al6 = [200000, 2000000, 20000000, 100000000]
        lr6 = [0.002, 0.001, 0.0004, 0.0001]
        x6, c6 = _adam_run(best_h, al6, lr6, 6000, seed + 42, dl6, 800)
        if c6 < best_c5:
            best_c5, best_h = c6, x6.copy()

    # Phase 7: Final SLSQP + subgradient (870-1060s)
    dl7 = t0 + 1050
    if time.time() < dl7 - 60:
        hf = _slsqp_run(best_h, 80000000, 4000)
        c5f = _eval(hf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()
    xsg2, csg2 = _subgrad_refine(best_h, best_n, dl7, rng)
    if csg2 < best_c5:
        best_c5, best_h = csg2, xsg2.copy()
    if time.time() < dl7 - 10:
        al7 = [5000000, 50000000, 200000000]
        lr7 = [0.001, 0.0003, 0.0001]
        x7, c7 = _adam_run(best_h, al7, lr7, 3000, seed + 99, dl7, 1000)
        if c7 < best_c5:
            best_c5, best_h = c7, x7.copy()

    # Final repair and evaluate
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
