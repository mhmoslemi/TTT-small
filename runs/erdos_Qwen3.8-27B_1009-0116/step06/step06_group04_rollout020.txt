def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _softmax_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _softmax_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    c = np.correlate(x, 1.0 - x, mode='full')

    s = np.clip(alpha * (c - m), -500, 500)

def _adam_opt(h0, a_lo, a_hi, n_iter, lr0, lr1, seed, deadline, noise_every=800):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n); v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t = i + 1
        frac = i / max(n_iter - 1, 1)
        alpha = a_lo * (a_hi / a_lo) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if t % noise_every == 0:
            x += 0.003 * rng.randn(n)
            x = _repair(x, n)
        g = _grad(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _nag_opt(h0, alpha, n_iter, lr, deadline):
    n = len(h0)
    x = h0.copy(); y = h0.copy()
    best_x = x.copy(); best_c = _eval(x)
    for t in range(1, n_iter + 1):
        if time.time() > deadline:
            break
        g = _grad(y, alpha)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = y + (beta - 1.0) / beta * (y - x)
        step = lr / (1.0 + 0.005 * t)
        xn = _repair(y - step * g, n)
        c = _eval(xn)
        if c < best_c:
            best_c = c; best_x = xn.copy()
        x = xn
    return best_x, best_c

def _fw_opt(h0, alpha, n_iter, deadline):
    n = len(h0)
    x = h0.copy().astype(float)
    best_x = x.copy(); best_c = _eval(x)
    for t in range(n_iter):
        if time.time() > deadline:
            break
        g = _grad(x, alpha)
        idx = np.argsort(g)
        s = np.zeros(n); s[idx[:n // 2]] = 1.0
        gamma = 2.0 / (t + 2)
        xn = _repair((1 - gamma) * x + gamma * s, n)
        c = _eval(xn)
        if c < best_c:
            best_c = c; best_x = xn.copy()
        x = xn
    return best_x, best_c

def _cd(h, n, rng, deadline, scale_init=0.002):
    x = h.copy(); c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    scale = scale_init
    for _ in range(300000):
        if time.time() > deadline or scale < 1e-9:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        x[i] += d; x[j] -= d
        x = _repair(x, n)
        cn = _eval(x)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best_x = x.copy()
            scale = max(scale * 0.999, 1e-9)
        else:
            x[i] -= d; x[j] += d
            x = _repair(x, n)
            scale *= 0.95
    return best_x, best_c

def _slsqp_polish(h, alpha, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _se(x):
        return x.sum() - n * 0.5
    def _sj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'disp': False, 'iprint': -1})
        if time.time() > deadline:
            return _repair(h, n)
        c = _eval(_repair(res.x, n))
        if np.isfinite(c) and c < _eval(h):
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _gen_starts(n, rng):
    starts = []
    for freq in [2, 3, 5, 7, 11]:
        x = np.arange(n)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + rng.randn())
        starts.append(_repair(h, n))
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    xr = (np.arange(n) * (gr - 1.0)) % 1.0
    starts.append(_repair((xr < 0.5).astype(float), n))
    for _ in range(2):
        h = 0.5 + 0.2 * rng.randn(n)
        starts.append(_repair(h, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Adam on parent + diverse starts (0-250s)
    dl1 = t0 + 250
    starts = [h_init] + _gen_starts(n0, rng)
    for si, hs in enumerate(starts):
        per = (dl1 - t0) / max(len(starts) - si, 1)
        if time.time() > dl1:
            break
        d1 = t0 + (si + 1) * per
        x, c = _adam_opt(hs, 50, 2000, 3000, 0.03, 0.002, seed + si, d1)
        x, c = _adam_opt(x, 2000, 80000, 3000, 0.01, 0.0005, seed + 100 + si, d1)
        if c < best_c5:
            best_c5, best_h, best_n = c, x.copy(), best_n

    # Phase 2: NAG + SLSQP polish on best (250-420s)
    dl2 = t0 + 420
    x, c = _nag_opt(best_h, 50000, 5000, 0.005, dl2)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl2:
        h2 = _slsqp_polish(best_h, 200000, 3000, dl2)
        c = _eval(h2)
        if c < best_c5:
            best_c5, best_h = c, h2.copy()

    # Phase 3: Multi-n exploration (420-620s)
    dl3 = t0 + 620
    n_cands = sorted(set([n0 - 20, n0 - 10, n0, n0 + 10, n0 + 20, n0 + 40,
                          260, 280, 320, 340, 360, 400]))
    for nt in n_cands:
        if time.time() > dl3:
            break
        if nt < 50 or nt > 600 or nt % 2 != 0:
            continue
        hr = _repair(_resample(best_h, nt), nt)
        x, c = _adam_opt(hr, 200, 50000, 2500, 0.015, 0.001, seed + nt, dl3)
        x, c = _nag_opt(x, 50000, 2000, 0.003, dl3)
        if c < best_c5:
            best_c5, best_h, best_n = c, x.copy(), nt
            h2 = _slsqp_polish(best_h, 500000, 2000, dl3)
            c2 = _eval(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 4: FW + NAG (620-780s)
    dl4 = t0 + 780
    x, c = _fw_opt(best_h, 20000, 5000, dl4)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    x, c = _nag_opt(best_h, 100000, 4000, 0.003, dl4)
    if c < best_c5:
        best_c5, best_h = c, x.copy()

    # Phase 5: CD polish (780-920s)
    dl5 = t0 + 920
    x, c = _cd(best_h, best_n, rng, dl5, 0.002)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl5:
        x, c = _nag_opt(best_h, 500000, 3000, 0.001, dl5)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    # Phase 6: SLSQP + CD final (920-1070s)
    dl6 = t0 + 1070
    h6 = _slsqp_polish(best_h, 5000000, 4000, dl6)
    c = _eval(h6)
    if c < best_c5:
        best_c5, best_h = c, h6.copy()
    x, c = _cd(best_h, best_n, rng, dl6, 0.0005)
    if c < best_c5:
        best_c5, best_h = c, x.copy()

    # Phase 7: Final tiny CD + repair (1070-1095s)
    dl7 = t0 + 1090
    x, c = _cd(best_h, best_n, rng, dl7, 0.0001)
    if c < best_c5:
        best_c5, best_h = c, x.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
