def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _ls_obj(x, alpha, pen=0.0):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = m + np.log(float(np.sum(np.exp(s)))) / alpha
    if pen > 0:
        sv = x.sum() - len(x) * 0.5
        val += pen * sv * sv
    return val

def _ls_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _adam_opt(h0, a0, a1, n_iter, lr0, lr1, seed, dl):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > dl:
            break
        t += 1
        fr = i / max(n_iter - 1, 1)
        alpha = a0 * (a1 / a0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        if t % 400 == 0:
            x = x + 0.004 * rng.randn(n)
        g = _ls_grad(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _resample(h, n_new):
    xo = np.linspace(0.0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _rand_bin(n, seed):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n); h[idx] = 1.0
    return h

def _slsqp(h, alpha, maxit, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x): return _ls_obj(x, alpha)
    def _g(x): return _ls_grad(x, alpha)
    def _c(x): return x.sum() - n * 0.5
    def _cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c, 'jac': _cj}]
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs(h, alpha, maxit, pen):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x): return _ls_obj(x, alpha, pen)
    def _g(x):
        g = _ls_grad(x, alpha)
        if pen > 0:
            g += 2.0 * pen * (x.sum() - n * 0.5)
        return g
    try:
        res = minimize(_f, h, jac=_g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _worst_desc(h, n, nsteps, lr, dl):
    x = h.copy(); best_x = x.copy(); best_c = _eval(x)
    for st in range(nsteps):
        if time.time() > dl:
            break
        x = best_x.copy()
        c = _corr(x); m = c.max()
        alpha = 3000
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            sh = (n - 1) - k
            wk = w[k] * (2.0 / n)
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    g[np.arange(sh, n)] -= wk * x[sh:]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        x = x - lr * g
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < best_c:
            best_c = c_new; best_x = x.copy()
        lr *= 0.999
    return best_x, best_c

def _pair_flip(h, n, rng, n_iters, dl):
    x = (h > 0.5).astype(float); x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    for it in range(n_iters):
        if time.time() > dl:
            break
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        x[i] = 0.0; x[j] = 1.0
        c_new = _eval(x)
        if c_new <= c_cur:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
        else:
            x[i] = 1.0; x[j] = 0.0
    return best_x, best_c
