    # Phase 3: Nelder-Mead on sensitive coords (100s)
    t_end = t0 + 540
    if time.time() < t_end:
        for ntop in [30, 50]:
            if time.time() > t_end:
                break
            hc, c = _local_nm(best_h, ntop, 3000)
            if c < best_c:
                best_c = c; best_h = hc

    # Phase 4: Subgradient (200s)
    t_end = t0 + 740
    n_it = min(6000, int((t_end - time.time()) / 0.02) + 1)
    best_h, best_c = _subgrad(best_h, n_it, 0.01, t_end)

    # Phase 5: SLSQP polish (120s)
    t_end = t0 + 860
    for alpha in [300000, 2000000, 5000000]:
        if time.time() > t_end:
            break
        hc = _slsqp(best_h, alpha, 3000, 1e-17)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 6: Coordinate hill climbing (120s)
    t_end = t0 + 980
    if time.time() < t_end:
        best_h, best_c = _coord_hill(best_h, t_end, rng)

    # Phase 7: Final SLSQP + fine search (115s)
    t_mid = t0 + budget_s - 60
    if time.time() < t_mid:
        hc = _slsqp(best_h, 3000000, 2500, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    t_end2 = t0 + budget_s - 3
    n_tr = min(int((t_end2 - time.time()) / 0.0015), 300000)
    if n_tr > 100:
        best_h, best_c = _fine_search(best_h, n_tr, rng, t_end2)

    best_h = _repair(best_h, n0)
    best_c = _eval(best_h)
    return (best_h, best_c, n0)
