def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP progressive at base res
    for alpha in [200, 1000, 5000, 20000, 80000, 300000]:
        if remaining() < 350:
            break
        h_cand = _slsqp_opt(best_h, alpha, 8, 3000, t0 + remaining() - 349)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand

    # Phase 2: Momentum projected gradient at base res
    if remaining() > 300:
        for alpha in [5000, 20000, 80000]:
            if remaining() < 280:
                break
            h_cand = _momentum_proj(best_h, alpha, 8, 3000, t0 + remaining() - 279)
            c5 = _eval(h_cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand

    # Phase 3: Multi-lag at base res
    if remaining() > 260:
        h_cand, c5_cand = _multi_lag(best_h, 500, t0 + remaining() - 259, topk=6)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand

    # Phase 4: Upsample to 2x and optimize
    if remaining() > 500:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [500, 3000, 15000, 60000, 200000]:
            if remaining() < 460:
                break
            h_hi = _slsqp_opt(h_hi, alpha, 8, 2500, t0 + remaining() - 459)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if remaining() > 400:
            h_hi = _momentum_proj(best_h, 50000, 10, 4000, t0 + remaining() - 399)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi
                best_n = n1
        if remaining() > 360:
            h_cand, c5_cand = _multi_lag(best_h, 600, t0 + remaining() - 359, topk=8)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand
                best_n = n1

    # Phase 5: SLSQP polish at current best
    if remaining() > 200:
        h_cand = _slsqp_opt(best_h, 500000, 12, 5000, t0 + remaining() - 199)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand

    # Phase 6: More momentum iterations
    if remaining() > 160:
        h_cand = _momentum_proj(best_h, 100000, 10, 5000, t0 + remaining() - 159)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand

    # Phase 7: SA pairs
    if remaining() > 120:
        n_iter = min(int((remaining() - 110) / 0.001), 80000)
        best_sa, c5_sa = _sa_pairs(best_h, t0 + remaining() - 110, n_iter, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = best_sa

    # Phase 8: Coordinate polish (most critical for final squeeze)
    if remaining() > 40:
        n_tries = min(int((remaining() - 35) / 0.0008), 200000)
        h_cur, c5_cur = _coord_polish(best_h, t0 + remaining() - 35, n_tries, rng)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur

    # Phase 9: Final tiny steps
    if remaining() > 15:
        for _ in range(min(int((remaining() - 12) / 0.001), 50000)):
            if remaining() < 12:
                break
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = 3e-6 * rng.randn()
            best_h[i] += d
            best_h[j] -= d
            if best_h[i] < 0 or best_h[i] > 1 or best_h[j] < 0 or best_h[j] > 1:
                best_h[i] -= d
                best_h[j] += d
                continue
            c5n = _eval(best_h)
            if c5n < best_c5 - 1e-16:
                best_c5 = c5n
            else:
                best_h[i] -= d
                best_h[j] += d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
