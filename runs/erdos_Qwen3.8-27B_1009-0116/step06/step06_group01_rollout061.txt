def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):

def _proj_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x_i<=1} via bisection on lambda."""
    h = np.clip(h, 0.0, 1.0)

        if s > target:

        else:
            lo = mid

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lag(h, k, n):
    dx = 2.0 / n
    s = (n - 1) - k
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(0, n - s)
        g[i1] += 1.0 - h[i1 + s]
        if s > 0:
            i2 = np.arange(s, n)
            g[i2] -= h[i2 - s]
    else:
        p = -s
        if p > 0:
            i1 = np.arange(0, n - p)
            g[i1] -= h[i1 + p]
            i2 = np.arange(p, n)
            g[i2] += 1.0 - h[i2 - p]
    return g * dx

def _top_desc(h, nc, top_k, max_iter, t_end):
    c5c = _eval(h)
    h = h.copy()
    mom = 0.7
    v = np.zeros(nc)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, nc)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        v = mom * v - 0.02 * g
        h_try = _proj_sum(h + v, nc / 2.0)
        c5n = _eval(h_try)
        if c5n < c5c - 1e-15:
            c5c = c5n
            h = h_try
            mom = min(mom + 0.01, 0.92)
        else:
            step = 0.01
            found = False
            for _ in range(20):
                h_try = _proj_sum(h - step * g, nc / 2.0)
                c5n = _eval(h_try)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    h = h_try
                    v = 0.9 * v
                    found = True
                    break
                step *= 0.5
            if not found:
                v *= 0.5
    return h, c5c

def _pair_cd(h, nc, max_sweeps, t_end, rng):
    h = h.copy()
    c5c = _eval(h)
    dx_val = 2.0 / nc
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(nc - 1):
            j = i + 1
            oi, oj = h[i], h[j]
            c = oi + oj
            lo = max(0.0, c - 1.0)
            hi = min(1.0, c)
            if hi - lo < 1e-11:
                continue
            a, b = lo, hi
            for _ in range(18):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[i], h[j] = m1, c - m1
                f1 = float(np.max(_corr(h)) * dx_val)
                h[i], h[j] = m2, c - m2
                f2 = float(np.max(_corr(h)) * dx_val)
                if f1 <= f2:
                    b = m2
                else:
                    a = m1
            t = 0.5 * (a + b)
            h[i], h[j] = t, c - t
            c5n = float(np.max(_corr(h)) * dx_val)
            if c5n < c5c - 1e-14:
                c5c = c5n
                improved = True
            else:
                h[i], h[j] = oi, oj
        if not improved and sweep > 3:
            break
    return h, c5c

def _rand_pair_cd(h, nc, max_sweeps, t_end, rng):
    h = h.copy()
    c5c = _eval(h)
    dx_val = 2.0 / nc
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        for _ in range(nc):
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            oi, oj = h[i], h[j]
            c = oi + oj
            lo = max(0.0, c - 1.0)
            hi = min(1.0, c)
            if hi - lo < 1e-11:
                continue
            a, b = lo, hi
            for _ in range(14):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[i], h[j] = m1, c - m1
                f1 = float(np.max(_corr(h)) * dx_val)
                h[i], h[j] = m2, c - m2
                f2 = float(np.max(_corr(h)) * dx_val)
                if f1 <= f2:
                    b = m2
                else:
                    a = m1
            t = 0.5 * (a + b)
            h[i], h[j] = t, c - t
            c5n = float(np.max(_corr(h)) * dx_val)
            if c5n < c5c - 1e-14:
                c5c = c5n
                improved += 1
            else:
                h[i], h[j] = oi, oj
        if improved == 0:
            break
    return h, c5c

def _lbfgs_smooth(h, nc, alpha, max_iter, t_end):
    target = nc / 2.0
    dx_val = 2.0 / nc
    nlags = 2 * nc - 1
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        ls = m + np.log(s.sum()) / alpha
        return ls * dx_val + 5e8 * (x.sum() - target) ** 2
    def grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(nc)
        for k in range(nlags):
            if w[k] < 1e-12:
                continue
            g += w[k] * _grad_lag(x, k, nc)
        g += 1e9 * (x.sum() - target)
        return g
    bnds = [(0.0, 1.0)] * nc
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': max_iter, 'ftol': 1e-20,
                    'gtol': 1e-14, 'maxfun': max_iter * 5, 'disp': False})
        return _repair(res.x, nc)
    except Exception:
        return _repair(h, nc)

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
