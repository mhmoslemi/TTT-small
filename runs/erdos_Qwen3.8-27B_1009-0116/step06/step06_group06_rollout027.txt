import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_g(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    f0 = _smax(h, alpha)
    try:
        res = minimize(_smax, h, jac=_smax_g, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _lbfgsb_proj(h, alpha, maxiter, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    f_prev = _smax(h_cur, alpha)
    g_prev = _smax_g(h_cur, alpha)
    v = np.zeros(n)
    mu = 0.7
    step = 0.1
    for _ in range(maxiter):
        if time.time() > t_end:
            break
        g = _smax_g(h_cur, alpha)
        v = mu * v + (1.0 - mu) * g
        dh = h_cur - h_prev if _ > 0 else np.zeros(n)
        dg = g - g_prev if _ > 0 else np.zeros(n)
        den = np.dot(dg, dg) + 1e-14
        step_bb = 0.5 * abs(np.dot(dh, dg) / den) if den > 1e-30 else step
        step = max(0.005, min(0.5, 0.7 * step + 0.3 * step_bb))
        accepted = False
        for _bt in range(12):
            h_new = _proj(h_cur - step * v, n * 0.5)
            f_new = _smax(h_new, alpha)
            if f_new <= f_prev + 1e-14:
                h_prev = h_cur.copy()
                h_cur = h_new
                f_prev = f_new
                g_prev = g.copy()
                accepted = True
                step = min(step * 1.08, 0.6)
                break
            step *= 0.5
        if not accepted:
            step *= 0.3
    return h_cur

def _multi_lag_descent(h, n_rounds, t_end):
    n = len(h)
    h_cur = h.copy()
    c5 = _eval(h_cur)
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:8]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= (2.0 / n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h_cur = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5

def _block_cd(h, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c5 = _eval(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5
    block_sizes = [3, 5, 8, 12, 16]
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    while time.time() < t_end:
        bs = rng.choice(block_sizes)
        ss = rng.choice(step_sizes)
        idx = rng.choice(n, size=min(bs, n), replace=False)
        perp = np.zeros(n)
        pts = idx.copy().astype(float)
        perp[idx] = 1.0
        perp -= perp.mean()
        perp /= np.linalg.norm(perp) + 1e-15
        improved = False
        for sign in [1.0, -1.0]:
            h_new = h_cur + sign * ss * perp
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h_cur = h_new
                c5 = c5n
                improved = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                break
    return best_h, best_c5

def _sa_block(h, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    t0v, t1v = 0.008, 1e-5
    n_iter = 80000
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        bs = rng.randint(2, 12)
        idx = rng.choice(n, size=min(bs, n), replace=False)
        delta = rng.randn(bs)
        delta -= delta.mean()
        delta *= (0.005 + 0.03 * t)
        old_vals = h_cur[idx].copy()
        h_cur[idx] += delta
        h_new = _repair(h_cur, n)
        c_new = _eval(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        else:
            h_cur = h_new.copy()
            h_cur[idx] = old_vals
            h_cur = _repair(h_cur, n)
    return best_h, best_c

def _fine_cd(h, t_end, rng):
    n = len(h)
    h_cur = h.copy()

    return best_h, best_c5

def _upsample(h, factor):

    return np.repeat(h, factor)
