def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

    if s < 1e-12:
        return np.full(n, 0.5)

                    h[idx] += a

def _grad_overlap(h, k_idx, n):
    """Gradient of the k-th overlap w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k_idx

        # dc/dh[i+sh] = -dx * h[i] for i in [0, n-1-sh]
        i1 = np.arange(n - sh)

        m = c.max()

        return m + np.log(float(np.sum(np.exp(s)))) / alpha

    best_c = _eval(best)

    for it in range(n_iter):

            h = h_new  # still move, but don't update best

    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += dx * (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= dx * h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)

        i2 = np.arange(n - sp)
        g[i2] -= dx * h[i2 + sp]

def _lp_step(h, n, active_idx, max_step=0.1):
    """Solve LP to find best step"""
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # minimize t
    # s.t. c[k] + g_k^T delta <= t  for k in active
    #      -h[i] <= delta[i] <= 1-h[i]
    #      sum(delta) = 0
    #      ||delta||_inf <= max_step (approx: -max_step <= delta[i] <= max_step)
    
    c_all = _correlate_vals(h)
    
    m = len(active_idx)
    # c[k] + g_k^T delta - t <= 0
    # i.e., g_k^T delta - t <= -c[k]
    
    # A_ub: m rows for active constraints + n+1 rows for upper bounds on delta
    #        + n rows for lower bounds on delta + n rows for max_step
    # b_ub: corresponding
    
    # Let's use: A_ub * [delta, t] <= b_ub
    # Row for active k: [g_k, -1] <= -c[k]
    # Row for delta[i] <= 1-h[i]: [e_i, 0] <= 1-h[i]
    # Row for -delta[i] <= h[i]: [-e_i, 0] <= h[i]
    # Row for delta[i] <= max_step: [e_i, 0] <= max_step
    # Row for -delta[i] <= max_step: [-e_i, 0] <= max_step
    
    # A_eq: sum(delta) = 0 -> [1,1,...,1, 0] = 0
    
    n_var = n + 1
    rows = []
    rhs = []
    
    for k in active_idx:
        g = _grad_k(h, k, n)
        row = np.zeros(n_var)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        rhs.append(-c_all[k])
    
    for i in range(n):
        row = np.zeros(n_var)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - h[i])
    
        row = np.zeros(n_var)
        row[i] = -1.0
        rows.append(row)
        rhs.append(h[i])
    
    # Step size constraint
    for i in range(n):
        row = np.zeros(n_var)
        row[i] = 1.0
        rows.append(row)
        rhs.append(max_step)
        
        row = np.zeros(n_var)
        row[i] = -1.0
        rows.append(row)
        rhs.append(max_step)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c_obj = np.zeros(n_var)
    c_obj[n] = 1.0  # minimize t
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=[(None, None)] * n_var, method='highs')
    
    if res.status != 0:
        return h
    
    delta = res.x[:n]

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

    return (bh, bc5, bn)
