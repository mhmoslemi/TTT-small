                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _top_grad(h, top_m=1):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n / top_m

def _adaptive_search(h, n_tries, rng, t0, time_limit):
    """Aggressive adaptive pairwise search with gradient guidance"""
    n = len(h)
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    g = _top_grad(h, 1)
    stuck = 0
    
    for t in range(n_tries):
        if time.time() - t0 > time_limit:
            break
        if (t + 1) % 500 == 0:
            g = _top_grad(h, 1)
        
        # Select indices
        r = rng.random()
        if r < 0.4:
            # Gradient-guided
            p = np.abs(g) + 1e-10
            p /= p.sum()
            i = rng.choice(n, p=p)
            j = rng.choice(n, p=p)
        elif r < 0.7:
            # Semi-random (near interior points)
            interior = np.where((h > 0.001) & (h < 0.999))[0]
            if len(interior) == 0:
                i = rng.randint(n); j = rng.randint(n)
            else:
                i = interior[rng.randint(len(interior))]
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        
        if i == j:
            continue
        
        # Determine direction
        dir_i = -np.sign(g[i]) if g[i] != 0 else rng.choice([-1, 1])
        if dir_i == 0:
            dir_i = 1
        
        # Adaptive step size
        base_step = 1e-4 * (0.999 ** (stuck // 100))
        d = base_step * rng.lognormal(0, 0.5)
        
        ni = h[i] + dir_i * d
        nj = h[j] - dir_i * d
        
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        
        h[i], h[j] = ni, nj
        cn = _eval(h)
        
        if cn < c_cur - 1e-15:
            c_cur = cn
            stuck = 0
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = ni - dir_i * d, nj + dir_i * d
            stuck += 1
            if stuck > 2000:
                # Restart from perturbation
                h = _repair(best + 0.005 * rng.randn(n), n)
                c_cur = _eval(h)
                g = _top_grad(h, 1)
                stuck = 0
    
    return best, best_c
