def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                a = min(abs(diff), max(0, (1.0 - h[idx]) if diff > 0 else h[idx]))
                h[idx] += a if diff > 0 else -a
                diff += -a if diff > 0 else a
                if abs(diff) < 1e-12:

    return h
