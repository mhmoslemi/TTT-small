import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _proj_sum_box(h, target):
    n = len(h)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
    return np.clip(h, 0.0, 1.0)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * x[i1]
        else:
            sp = -sh
            i2 = np.arange(sp, n)
            g[i2] += wk * (1.0 - x[i2 - sp])
            g[i2 - sp] -= wk * x[i2]
    return g

def _sm_obj_per(x, alpha, m):
    h = np.tile(x, m)
    return _sm_obj(h, alpha)

def _sm_grad_per(x, alpha, m):
    T = len(x)
    h = np.tile(x, m)
    g = _sm_grad(h, alpha)
    return g.reshape(m, T).sum(axis=0)

def _sum_eq(x, T):
    return x.sum() - T * 0.5

def _sum_jac(x, T):
    return np.ones(T)

def _slsqp_per(hp, m, alphas, maxiter, ftol):
    T = len(hp)
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (T,)}]
    for alpha in alphas:
        try:
            res = minimize(_sm_obj_per, hp, jac=_sm_grad_per,
                          args=(alpha, m), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            hp = _repair(res.x, T)
        except Exception:
            pass
    return hp

def _nesterov(h0, n, alpha, n_iter, lr_init, t_limit):
    t_s = time.time()
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    beta = 0.85
    best_h = h0.copy()
    best_c = _eval(h0)
    for it in range(n_iter):
        if time.time() - t_s > t_limit:
            break
        g = _sm_grad(h, alpha)
        v = beta * v - lr * g
        h_new = _proj_sum_box(h + v, target)
        c = _eval(h_new)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            if lr < 0.03:
                lr *= 1.03
        else:
            lr *= 0.55
            if lr < 1e-7:
                lr = lr_init
                v = np.zeros(n)
    return best_h, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _ils_per(hp, m, T, n_trials, n_inner, alpha, lr, seed):
    rng = np.random.RandomState(seed)
    best_h = hp.copy()
    best_c = _eval(np.tile(best_h, m))
    for tr in range(n_trials):
        h_try = best_h + 0.04 * rng.randn(T)
        h_try = _proj_sum_box(h_try, T * 0.5)
        for _ in range(n_inner):
            g = _sm_grad_per(h_try, alpha, m)
            h_try = _proj_sum_box(h_try - lr * g, T * 0.5)
        c = _eval(np.tile(h_try, m))
        if c < best_c:
            best_c = c
            best_h = h_try.copy()
    return best_h, best_c

def _binary_init(T, rng):
    k = int(T / 2)
    hp = np.zeros(T)
    idx = rng.choice(T, size=k, replace=False)
    hp[idx] = 1.0
    hp = hp + 0.1 * rng.randn(T)
    return _proj_sum_box(hp, T * 0.5)
