def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

    if s < 1e-12:
        return np.full(n, 0.5)

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())

    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue

        if sh >= 0:
            m2 = n - sh
            g[:m2] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:m2]
        else:
            sp = -sh
            m2 = n - sp
            g[sp:] += wk * (1.0 - x[:m2])
            g[:m2] -= wk * x[sp:]

    n = len(h)
    bnds = [(0.0, 1.0)] * n

                       method='SLSQP', bounds=bnds, constraints=cons,

    cur = _eval(h)
    best_h = h.copy()

    zeros = np.where(h < 0.5)[0]
    T = t_start
    cooling = (t_end / t_start) ** (1.0 / max(1, max_rounds))
    for it in range(max_rounds):
        if time.time() > time_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c2 = _eval(h)

                best_h = h.copy()

            zeros = np.where(h < 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
