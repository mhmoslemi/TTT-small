def _grad_index(h, idx, weights):

    c0 = n - 1
    for r in range(len(idx)):
        w = float(weights[r])

def _proj_sum(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    s = h.sum()
    if abs(s - target) <= 1e-13:
        return h
    shift = (s - target) / n
    h1 = np.clip(h - shift, 0.0, 1.0)
    s1 = h1.sum()
    if abs(s1 - target) <= 1e-10:
        return h1
    lo = min(float(np.min(h)) - 1.0, shift)
    hi = max(float(np.max(h)) + 1.0, shift)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_index(h, idx, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    c0 = n - 1
    scalar = np.isscalar(weights)
    for r in range(len(idx)):
        w = float(weights) if scalar else float(weights[r])
        if w == 0.0:
            continue
        sh = c0 - int(idx[r])
        if sh == 0:
            g += w * (1.0 - 2.0 * h)
        elif sh > 0:
            cnt = n - sh
            i = np.arange(cnt)
            g[i] += w * (1.0 - h[i + sh])
            g[i + sh] -= w * h[i]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += w * (1.0 - h[:cnt])
            g[:cnt] -= w * h[sp:]
    return g

def _lp_step(h, deadline, top_count=256, gap=1.0):
    n = len(h)
    m = 2*n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(c))
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)
    if top_count >= m:
        idx = np.arange(m)
    else:
        act = np.where(c >= cur - gap)[0]
        if len(act) == 0:
            act = np.arange(m)
        if len(act) > top_count:
            vals = c[act]
            cut = np.argpartition(vals, len(vals) - top_count)[len(vals) - top_count:]
            act = act[cut]
        idx = act
    G = np.empty((len(idx), n))
    c0 = n - 1
    for r, k in enumerate(idx):
        row = G[r]
        row.fill(0.0)
        sh = c0 - int(k)
        if sh == 0:
            row[:] = 1.0 - 2.0 * hc
        elif sh > 0:
            cnt = n - sh
            ii = np.arange(cnt)
            row[ii] += 1.0 - hc[ii + sh]
            row[ii + sh] -= hc[ii]
        else:
            sp = -sh
            cnt = n - sp
            row[sp:] += 1.0 - hc[:cnt]
            row[:cnt] -= hc[sp:]
    A = np.empty((len(idx), n + 1))
    A[:, :n] = G
    A[:, -1] = -1.0
    b = -c[idx]
    cost = np.zeros(n + 1)
    cost[-1] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    bounds = np.empty((n + 1, 2))
    bounds[:n, 0] = -hc
    bounds[:n, 1] = 1.0 - hc
    bounds[-1, 0] = 0.0
    bounds[-1, 1] = np.inf
    try:
        res = linprog(cost, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0, bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        return h, cur * dx
    if not res.success:
        return h, cur * dx
    if res.fun >= cur - 1e-9:
        return h, cur * dx
    d = np.clip(res.x[:n], -hc, 1.0 - hc)
    s = d.sum()
    if abs(s) > 1e-7:
        d = d - s / n
        d = np.clip(d, -hc, 1.0 - hc)
    if np.max(np.abs(d)) < 1e-13:
        return h, cur * dx
    best_count = cur
    best_h = h
    for a in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.03):
        if time.time() > deadline:
            break
        cand = h + a * d
        if cand.min() < -1e-10 or cand.max() > 1.0 + 1e-10:
            cand = np.clip(cand, 0.0, 1.0)
            if abs(cand.sum() - 0.5 * n) > 1e-7:
                cand = _proj_sum(cand, 0.5 * n)
        cc = float(np.max(np.correlate(cand, 1.0 - cand, mode='full')))
        if cc < best_count - 1e-12:
            best_count = cc
            best_h = cand
    return best_h, best_count * dx

import numpy as np
from scipy.optimize import linprog
import time

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def _proj_sum(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    s = h.sum()
    if abs(s - target) <= 1e-13:
        return h
    shift = (s - target) / n
    h1 = np.clip(h - shift, 0.0, 1.0)
    s1 = h1.sum()
    if abs(s1 - target) <= 1e-10:
        return h1
    lo = min(float(np.min(h)) - 1.0, shift)
    hi = max(float(np.max(h)) + 1.0, shift)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    t = 0.5 * len(h)
    if abs(h.sum() - t) > 1e-8:
        h = _proj_sum(h, t)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_index(h, idx, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    c0 = n - 1
    scalar = np.isscalar(weights)
    for r in range(len(idx)):
        w = float(weights) if scalar else float(weights[r])
        if w == 0.0:
            continue
        sh = c0 - int(idx[r])
        if sh == 0:
            g += w * (1.0 - 2.0 * h)
        elif sh > 0:
            cnt = n - sh
            i = np.arange(cnt)
            g[i] += w * (1.0 - h[i + sh])
            g[i + sh] -= w * h[i]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += w * (1.0 - h[:cnt])
            g[:cnt] -= w * h[sp:]
    return g

def _lp_step(h, deadline, top_count=256, gap=1.0):
    n = len(h)
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(np.max(c))
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)
    if top_count >= m:
        idx = np.arange(m)
    else:
        act = np.where(c >= cur - gap)[0]
        if len(act) == 0:
            act = np.arange(m)
        if len(act) > top_count:
            vals = c[act]
            cut = np.argpartition(vals, len(vals) - top_count)[len(vals) - top_count:]
            act = act[cut]
        idx = act
    G = np.empty((len(idx), n))
    c0 = n - 1
    for r, k in enumerate(idx):
        row = G[r]
        row.fill(0.0)
        sh = c0 - int(k)
        if sh == 0:
            row[:] = 1.0 - 2.0 * hc
        elif sh > 0:
            cnt = n - sh
            ii = np.arange(cnt)
            row[ii] += 1.0 - hc[ii + sh]
            row[ii + sh] -= hc[ii]
        else:
            sp = -sh
            cnt = n - sp
            row[sp:] += 1.0 - hc[:cnt]
            row[:cnt] -= hc[sp:]
    A = np.empty((len(idx), n + 1))
    A[:, :n] = G
    A[:, -1] = -1.0
    b = -c[idx]
    cost = np.zeros(n + 1)
    cost[-1] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    bounds = np.empty((n + 1, 2))
    bounds[:n, 0] = -hc
    bounds[:n, 1] = 1.0 - hc
    bounds[-1, 0] = 0.0
    bounds[-1, 1] = np.inf
    try:
        res = linprog(cost, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return h, cur * dx
    if not res.success or res.fun >= cur - 1e-9:
        return h, cur * dx
    d = np.clip(res.x[:n], -hc, 1.0 - hc)
    s = d.sum()
    if abs(s) > 1e-7:
        d = np.clip(d - s / n, -hc, 1.0 - hc)
    if np.max(np.abs(d)) < 1e-13:
        return h, cur * dx
    best = cur
    best_h = h
    for a in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.03):
        if time.time() > deadline:
            break
        cand = h + a * d
        if cand.min() < -1e-10 or cand.max() > 1.0 + 1e-10:
            cand = np.clip(cand, 0.0, 1.0)
        if abs(cand.sum() - 0.5 * n) > 1e-7:
            cand = _proj_sum(cand, 0.5 * n)
        cc = float(np.max(np.correlate(cand, 1.0 - cand, mode='full')))
        if cc < best - 1e-12:
            best = cc
            best_h = cand
    return best_h, best * dx

def _nesterov_topk(h, deadline, max_iters, k_start, k_max, step0):
    n = len(h)
    target = 0.5 * n
    m = 2 * n - 1
    K = max(1, min(int(k_start), m))
    Kmax = max(K, min(int(k_max), m))
    h_c = h.copy()
    v = np.zeros(n)
    beta = 0.7
    step = float(step0)
    best = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 25 and K < Kmax:
            K += 1
        c = np.correlate(h_c, 1.0 - h_c, mode='full')
        idx = np.arange(m) if K >= m else np.argpartition(c, m - K)[m - K:]
        g = _grad_index(h_c, idx, 1.0 / len(idx))
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        ht = _proj_sum(h_c - step * v, target)
        cc = float(np.max(np.correlate(ht, 1.0 - ht, mode='full')))
        if cc < best - 1e-12:
            best = cc
            h_c = ht
            stall = 0
            step = min(step * 1.03, 0.03)
        else:
            step *= 0.6
            stall += 1
            if step < 1e-9:
                break
    return h_c, best * (2.0 / n)

def _coord_search(h, deadline, max_iters, k_start, k_max):
    n = len(h)
    m = 2 * n - 1
    h_c = h.copy()
    best = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))
    K = max(1, min(int(k_start), m))
    Kmax = max(K, min(int(k_max), m))
    stall = 0
    fracs = (1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h_c, 1.0 - h_c, mode='full')
        cur = float(c.max())
        idx = np.arange(m) if K >= m else np.argpartition(c, m - K)[m - K:]
        g = _grad_index(h_c, idx, 1.0 / len(idx))
        p = int(np.argmin(g))
        q = int(np.argmax(g))
        moved = False
        if p != q:
            diff = g[p] - g[q]
            if diff < 0.0:
                dmax = min(1.0 - h_c[p], h_c[q])
                dsign = 1.0
            else:
                dmax = min(h_c[p], 1.0 - h_c[q])
                dsign = -1.0
            if dmax > 1e-10 and abs(diff) > 1e-12:
                oldi = h_c[p]
                oldj = h_c[q]
                bd = 0.0
                bcc = cur
                bhi = oldi
                bhj = oldj
                for f in fracs:
                    if time.time() > deadline:
                        break
                    d = dsign * f * dmax
                    hi = oldi + d
                    hj = oldj - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_c[p] = hi
                        h_c[q] = hj
                        cc = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))
                        if cc < bcc - 1e-12:
                            bcc = cc
                            bd = d
                            bhi = hi
                            bhj = hj
                        h_c[p] = oldi
                        h_c[q] = oldj
                if bd != 0.0:
                    h_c[p] = bhi
                    h_c[q] = bhj
                    best = bcc
                    moved = True
        if moved:
            stall = 0
        else:
            stall += 1
            if stall > 15 and K < Kmax:
                K = min(2 * K, Kmax)
                stall = 0
            elif stall > 120:
                break
    return h_c, best * (2.0 / n)

def _random_search(h, deadline, rng, max_iters, scale0):
    n = len(h)

    best = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))

            off = 1 + rng.randint(0, span2)
            j = (i + off) % n
            if j != i:
                if it % 5 == 0 and n > 2:
                    l = -1
                    for _ in range(20):
                        l = (j + 1 + rng.randint(0, span2)) % n
                        if l != i and l != j:
                            break
                    if l != i and l != j:
                        d1 = scale * rng.randn()
                        d2 = scale * rng.randn()
                        hi = h_c[i] + d1
                        hj = h_c[j] + d2
                        hl = h_c[l] - d1 - d2
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0 and 0.0 <= hl <= 1.0:
                            oi, oj, ol = h_c[i], h_c[j], h_c[l]
                            h_c[i] = hi
                            h_c[j] = hj
                            h_c[l] = hl
                            cc = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))
                            if cc < best - 1e-12:

                                h_c[i] = oi

                else:

                    hi = h_c[i] + d

    return h_c, best * (2.0 / n)
