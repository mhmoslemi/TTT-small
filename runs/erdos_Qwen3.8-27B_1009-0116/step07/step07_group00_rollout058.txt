import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _compute_c(h):

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

    obj = np.maximum(lin.max(axis=1), quad)
    t = ts[obj.argmin()]
    
    # Refine
    if 1 < obj.argmin() < 40:
        bi = obj.argmin()
        t2 = np.linspace(ts[bi-2], ts[bi+2], 16)
        lin2 = intercept[None,:] + t2[:,None]*slope[None,:]
        lin2[:, n-1] = -1e18
        quad2 = rest0 + t2 - t2*t2
        obj2 = np.maximum(lin2.max(axis=1), quad2)
        t = t2[obj2.argmin()]
    
    return t, slope, rest0

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _optimal_coord(c, h, i, n):
    nlag = 2 * n - 1
    shifts = np.arange(nlag) - (n - 1)
    nz = shifts != 0
    v1 = (i + shifts >= 0) & (i + shifts < n) & nz
    v2 = (i - shifts >= 0) & (i - shifts < n) & nz
    slope = np.zeros(nlag)
    slope[v1] += (1.0 - h[i + shifts[v1]])
    slope[v2] -= h[i - shifts[v2]]
    intercept = c - h[i] * slope
    rest0 = c[n - 1] - h[i] + h[i] * h[i]
    ts = np.linspace(0, 1, 41)
    lin = intercept[None, :] + ts[:, None] * slope[None, :]
    lin[:, n - 1] = -1e18
    quad = rest0 + ts - ts * ts
    obj = np.maximum(lin.max(axis=1), quad)
    bi = obj.argmin()
    t = ts[bi]
    if 2 <= bi <= 38:
        t2 = np.linspace(ts[bi - 2], ts[bi + 2], 17)
        lin2 = intercept[None, :] + t2[:, None] * slope[None, :]
        lin2[:, n - 1] = -1e18
        quad2 = rest0 + t2 - t2 * t2
        obj2 = np.maximum(lin2.max(axis=1), quad2)
        t = t2[obj2.argmin()]
    return np.clip(t, 0, 1), slope, rest0

def _coord_desc(h, deadline, rng, repair_every=5):
    n = len(h)
    c = _corr(h)
    best_h = h.copy()
    best_c5 = float(np.max(c) * 2.0 / n)
    order = np.arange(n).copy()
    sw = 0
    while time.time() < deadline:
        for i in order:
            if time.time() > deadline:
                break
            t, slope, rest0 = _optimal_coord(c, h, i, n)
            delta = t - h[i]
            if abs(delta) > 1e-13:
                c += delta * slope
                c[n - 1] = rest0 + t - t * t
                h[i] = t
        sw += 1
        if sw % repair_every == 0:
            h = _repair(h, n)
            c = _corr(h)
        c5 = float(np.max(c) * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if sw % 40 == 0 and sw > 0:
            order = rng.permutation(n)
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_phase(h, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    configs = [(5000, 25, 200), (20000, 15, 200), (80000, 10, 150), (400000, 8, 150)]
    for alpha, tk, mi in configs:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq',
                 'fun': lambda x, nn=n: x.sum() - nn * 0.5,
                 'jac': lambda x, nn=n: np.ones(nn)}]
        try:
            r = minimize(lambda x: _smax_obj(x, alpha, tk), h,
                         jac=lambda x: _smax_grad(x, alpha, tk),
                         method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=cons,
                         options={'maxiter': mi, 'ftol': 1e-16})
            h = _repair(r.x, n)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c5

def _anneal(h, deadline, rng, max_T=0.005, min_T=1e-10):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    c5c = best_c5
    blocks = [3, 5, 8, 12, 16, 24, 32, 48]
    it = 0
    while time.time() < deadline:
        T = max_T * (min_T / max_T) ** (it / 200000)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        it += 1
    return best_h, best_c5

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hp = h.copy()
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _finetune(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    c5c = best_c5
    while time.time() < deadline:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 3e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5
