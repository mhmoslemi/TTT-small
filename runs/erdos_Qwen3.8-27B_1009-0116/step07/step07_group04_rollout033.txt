        d2 = scale * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < cc:
                cc = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return bh, bc

def _local_2pt(h, ntries, rng, t_end, scale):
    n = len(h)
    cc = _eval(h)
    bh, bc = h.copy(), cc
    for _ in range(ntries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = scale * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cc:
                cc = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return bh, bc

def _up(h, f):
    return np.repeat(h, f).copy()

def _dn(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh, bc5 = h.copy(), _eval(h)

    # P1: Nesterov full softmax moderate (0-200s)
    h1, _ = _nesterov_soft(h.copy(), 50, 8000, 40000, 0.003, 0.72, t0+200)
    c = _eval(h1)
    if c < bc5:
        bc5, bh = c, h1.copy()

    # P2: SLSQP annealing (200-450s)
    for al in [20000, 80000, 300000, 1200000]:
        if time.time() > t0 + 440:
            break
        h2 = _slsqp_run(bh.copy(), al, n0, 600, t0+445)
        c = _eval(h2)
        if c < bc5:
            bc5, bh = c, h2.copy()

    # P3: Nesterov high-alpha (450-600s)
    h3, _ = _nesterov_soft(bh.copy(), 50000, 2000000, 30000, 0.0004, 0.82, t0+595)
    c = _eval(h3)
    if c < bc5:
        bc5, bh = c, h3.copy()

    # P4: Multi-res escape (600-750s)
    for nr in [n0//2, n0*3//2]:
        if time.time() > t0 + 740:
            break
        nr = max(60, nr)
        if nr != n0:
            if n0 > nr and n0 % nr == 0:
                hn = _dn(bh, n0//nr)
            elif nr > n0 and nr % n0 == 0:
                hn = _up(bh, nr//n0)
            else:
                hn = np.interp(np.linspace(0,1,nr), np.linspace(0,1,n0), bh)
            hn = _repair(hn, len(hn))
            hn, _ = _nesterov_soft(hn, 200, 30000, 15000, 0.002, 0.75, t0+730)
            if len(hn) != n0:
                hn = np.interp(np.linspace(0,1,n0), np.linspace(0,1,len(hn)), hn)
            hn = _repair(hn, n0)
            c = _eval(hn)
            if c < bc5:
                bc5, bh = c, hn.copy()

    # P5: SLSQP very high alpha (750-850s)
    if time.time() < t0 + 845:
        h5 = _slsqp_run(bh.copy(), 5000000, n0, 400, t0+845)
        c = _eval(h5)
        if c < bc5:
            bc5, bh = c, h5.copy()

    # P6: Local search multi-scale (850-1040s)
    for sc in [2e-4, 8e-5, 3e-5, 1e-5, 4e-6]:
        if time.time() > t0 + 1035:
            break
        nt = min(25000, max(100, int((t0+1035-time.time())/0.0025)))
        bh, bc5 = _local_3pt(bh, nt, rng, t0+1038, sc)

    # P7: Ultra-fine 2pt (1040-1090s)
    if time.time() < t0 + budget_s - 8:
        nt = min(50000, max(100, int((t0+budget_s-10-time.time())/0.0015)))
        bh, bc5 = _local_2pt(bh, nt, rng, t0+budget_s-10, 2e-6)

    # P8: Final SLSQP + tiny search
    if time.time() < t0 + budget_s - 5:
        hf = _slsqp_run(bh.copy(), 10000000, n0, 200, t0+budget_s-4)
        c = _eval(hf)
        if c < bc5:
            bc5, bh = c, hf.copy()
        if time.time() < t0 + budget_s - 1:
            bh, bc5 = _local_2pt(bh, 3000, rng, t0+budget_s-1, 1e-6)

    bh = _repair(bh, n0)
    bc5 = _eval(bh)
    return (bh, bc5, n0)
