import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_cvals(h).max())

def _grad_one(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_one(x, k)
    return g

def _nesterov_sub(h, steps, eta0, tol, top_frac, deadline):
    n = len(h)
    x = h.copy()
    x_prev = h.copy()
    beta = 1.0
    best_h = h.copy()
    best_c = _eval(h)
    for t in range(steps):
        if time.time() > deadline - 20:
            break
        th = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = x + (beta - 1.0) / th * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        c = _cvals(y)
        m = c.max()
        # Select top lags

        top_k = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in top_k:
            g += _grad_one(y, k)
        g /= len(top_k)
        eta = eta0 / (t + 1)
        x_new = _proj(x - eta * g, n)
        c_new = _eval(x_new)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        x_prev = x
        x = x_new
        beta = th
    return best_h, best_c

def _top_lag_descent(h, n_lags, max_rounds, deadline):

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_cvals(h).max())

def _grad_one(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_one(x, k)
    return g

    x = h.copy()

        y = x + (beta - 1.0) / th * (x - x_prev)

        m = c.max()

                x_prev = x
                x = x_new
                beta = 1.0
