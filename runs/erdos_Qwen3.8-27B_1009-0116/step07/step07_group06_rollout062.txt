The d² correction: when h[i] += d and h[j] -= d, the correlation c[k] changes by:
- Linear: d * (dc[k]/dh[i] - dc[k]/dh[j])
- Quadratic: d² for lags where both h[i] and h[j] appear in the product

When do both appear? h[i] and h[j] both appear in c[k] when:
- c[k] = sum h[a](1-h[b]) and {a,b} contains both i and j in some combination
- This happens when j-i = sh or i-j = sh for the shift sh of lag k
- So k = n-1+(j-i) or k = n-1-(j-i) = n-1+(i-j)

These are exactly k1 = n-1+(j-i) and k2 = n-1-(j-i).

The d² coefficient: c[k] contains the term h[i]*(1-h[j]) or h[j]*(1-h[i]) (depending on the shift). When we change h[i] += d, h[j] -= d:
- h[i]*(1-h[j]) → (h[i]+d)(1-h[j]+d) = h[i](1-h[j]) + d(1-h[j]) + d*h[i] + d²
  Change: d(1-h[j] + h[i]) + d²
- But we also have the other terms in the sum...

import numpy as np
import time

def _proj_to_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _c5_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sm_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    tc = tau * c_raw - (tau * c_raw).max()
    e = np.exp(tc)
    p = e / e.sum()
    grad = np.zeros(n)
    w0 = p[n - 1] * dx
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    for sh in range(1, n):
        w = (p[n - 1 + sh] + p[n - 1 - sh]) * dx
        if w < 1e-14:
            continue
        grad[sh:] += w * (1.0 - h[:-sh])
        grad[:n - sh] -= w * h[sh:]
    return grad

def _soft_gd(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    mom = np.zeros(n)
    beta = 0.85
    tau = 25.0
    lr = 0.015
    for it in range(2000):
        if time.time() > deadline:
            break
        g = _sm_grad(h, tau)
        mom = beta * mom - lr * g
        h_new = _proj_to_sum(h + mom, n * 0.5)
        c5n = _eval(h_new)
        if c5n < best_c:
            best_c = c5n
            best_h = h_new.copy()
        if c5n <= _eval(h) + 1e-12:
            h = h_new
        else:
            lr *= 0.9
            mom *= 0.5
        if it % 200 == 199 and tau > 5.0:
            tau *= 0.8
            lr = max(0.003, lr * 0.85)
    return best_h, best_c

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    g[n - 1] = 1.0 - 2.0 * h[p]
    if p > 0:
        mx = min(p, n - 1)
        sh = np.arange(1, mx + 1)
        g[n - 1 + sh] += (1.0 - h[p - sh])
        g[n - 1 - sh] += (1.0 - h[p - sh])
    if p < n - 1:
        mx = n - 1 - p
        sh = np.arange(1, mx + 1)
        g[n - 1 + sh] -= h[p + sh]
        g[n - 1 - sh] -= h[p + sh]
    return g

def _pair_cd(h0, rng, deadline, n_iter=100000):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    c = _c5_vec(h).copy()
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it > 3000 and stable > 5000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        gd = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-14:
            continue
        # Line search: minimize max of (c + d*gd + d² at k1,k2)
        for _ in range(40):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * gd
            vals[k1] += d_mid * d_mid
            vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = gd[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-12:
                d_hi = d_mid
            elif sg < -1e-12:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-12:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * gd
        c[k1] += d_opt * d_opt
        c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * gd
            c[k1] -= d_opt * d_opt
            c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _fine_polish(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(500000):

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 1e-5 * rng.randn()

                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
