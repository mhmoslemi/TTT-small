        h = _proj_sum(h, n / 2.0)
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_from_weights(h, w):
    """Gradient given softmax weights w over correlation lags."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]

    g = _grad_from_weights(h, w)

def _proj_sum(h, target):
