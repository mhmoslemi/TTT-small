        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = _proj_to_sum(h, target)
    return h

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corrs(h).max())

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k]
        if sh >= 0:
            idx = np.arange(n - sh)
            g[idx] += wk * (1.0 - h[idx + sh])
            if sh > 0:
                idx2 = np.arange(sh, n)
                g[idx2] -= wk * h[idx2 - sh]
        else:
            sp = -sh
            idx = np.arange(sp, n)
            g[idx] += wk * (1.0 - h[idx - sp])
            idx2 = np.arange(n - sp)
            g[idx2] -= wk * h[idx2 + sp]
    g *= dx
    return obj, g

def _nesterov(h, alpha, t_end, step0=0.003, n_steps=2000):
    n = len(h)
    f, _ = _smooth_obj_and_grad(h, alpha)
    v = np.zeros(n)
    step = step0
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        _, g = _smooth_obj_and_grad(h, alpha)
        v = 0.85 * v + 0.15 * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new, _ = _smooth_obj_and_grad(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.05, step0 * 12.0)
        else:
            step *= 0.45
            if step < 1e-13:
                break
            h = _proj_to_sum(h - step * v, n * 0.5)
            f, _ = _smooth_obj_and_grad(h, alpha)
    return h, _eval(h)

def _lbfgs_smooth(h_init, alpha, penalty, t_end, max_iter=250):
    n = len(h_init)
    def obj(v):
        hh = v
        c = _corrs(hh)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        e = np.exp(s - s.max())
        return m + np.log(e.sum()) / alpha + penalty * (hh.sum() - n * 0.5) ** 2
    def grad(v):
        _, g = _smooth_obj_and_grad(v, alpha)
        g = g + 2.0 * penalty * (v.sum() - n * 0.5)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                'maxfun': max_iter * 5, 'iprint': -1})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h_init.copy()

def _coord_descent(h, n, duration_s, top_K, rng, t_deadline):
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    gr = (np.sqrt(5) - 1) / 2
    no_improve = 0
    while time.time() < min(end_t, t_deadline - 4):
        c = _corrs(h)
        top_idx = np.argsort(c)[-top_K:]
        a_vals = c[top_idx].copy()
        g_active = np.zeros((top_K, n))
        for ki, k in enumerate(top_idx):
            sh = k - (n - 1)
            gk = np.zeros(n)
            if sh >= 0:
                idx = np.arange(n - sh)
                gk[idx] = 1.0 - h[idx + sh]
                if sh > 0:
                    idx2 = np.arange(sh, n)
                    gk[idx2] -= h[idx2 - sh]
            else:
                sp = -sh
                idx = np.arange(sp, n)
                gk[idx] = 1.0 - h[idx - sp]
                idx2 = np.arange(n - sp)
                gk[idx2] -= h[idx2 + sp]
            g_active[ki] = gk * (2.0 / n)
        improved = False
        n_pairs = n * 2
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_deadline - 4):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n - 1)) % n
            if i == j:
                continue
            delta = g_active[:, i] - g_active[:, j]
            if np.max(np.abs(delta)) < 1e-16:
                continue
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            lo, hi = d_lo, d_hi
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = (a_vals + delta * x1).max()
            f2 = (a_vals + delta * x2).max()
            for _ in range(28):
                if f1 < f2:
                    hi, x2 = x1, hi - gr * (hi - lo)
                    f2 = (a_vals + delta * x2).max()
                    x1 = hi - gr * (hi - lo)
                    f1 = (a_vals + delta * x1).max()
                else:
                    lo, x1 = x2, lo + gr * (hi - lo)
                    f1 = (a_vals + delta * x1).max()
                    x2 = lo + gr * (hi - lo)
                    f2 = (a_vals + delta * x2).max()
            d_opt = (lo + hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        new_c5 = _eval(h)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 25:
            h = _proj_to_sum(h + 0.003 * no_improve * rng.randn(n) / 10, n * 0.5)
            no_improve = 0
        h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _fine_search(h, n_tries, rng, t_end):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for scale in [5e-4, 1e-4, 3e-5, 1e-5, 3e-6]:
        for _ in range(n_tries):
            if time.time() > t_end:
                return best_h, best_c
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < best_c - 1e-18:
                    best_c = c_new
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_dead = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: L-BFGS with smooth max approximation
    if time.time() < t_dead - 200:
        for alpha, pen in [(3000, 5e6), (10000, 2e7), (50000, 1e8), (200000, 5e8)]:
            if time.time() > t_dead - 200:
                break
            h_c = best_h.copy()
            t_le = min(time.time() + 50, t_dead - 200)
            h_c = _lbfgs_smooth(h_c, alpha, pen, t_le, max_iter=200)
            h_c = _repair(h_c, n)
            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_c.copy()

    # Phase 2: Nesterov with high alpha
    if time.time() < t_dead - 120:
        for alpha in [100000, 500000, 2000000]:
            if time.time() > t_dead - 120:
                break
            h_c = best_h.copy()
            t_pe = min(time.time() + 45, t_dead - 120)
            h_c, c5v = _nesterov(h_c, alpha, t_pe)
            h_c = _repair(h_c, n)
            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_c.copy()

    # Phase 3: Coordinate descent with active set
    for top_K in [8, 15, 5, 12]:
        if time.time() > t_dead - 80:
            break
        h_c = best_h.copy()
        h_c, c5v = _coord_descent(h_c, n, 50, top_K, rng, t_dead)
        h_c = _repair(h_c, n)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_c.copy()

    # Phase 4: Structured perturbations + re-optimize
    if time.time() < t_dead - 50:
        for mode in [1, 2, 3, 5, 7]:
            if time.time() > t_dead - 50:
                break
            h_c = best_h.copy()
            pert = 0.02 * np.sin(2.0 * np.pi * mode * np.arange(n) / n)
            h_c = _proj_to_sum(h_c + pert, n * 0.5)
            h_c, c5v = _nesterov(h_c, 200000, min(time.time() + 20, t_dead - 50))
            h_c = _repair(h_c, n)
            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_c.copy()

    # Phase 5: More coordinate descent
    for top_K in [10, 6, 20]:
        if time.time() > t_dead - 30:
            break
        h_c = best_h.copy()
        h_c, c5v = _coord_descent(h_c, n, 20, top_K, rng, t_dead)
        h_c = _repair(h_c, n)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_c.copy()

    # Phase 6: Fine local search
    if time.time() < t_dead - 8:
        remaining = max(10, int((t_dead - 8 - time.time()) / 0.002))
        h_c = best_h.copy()
        best_h, best_c5 = _fine_search(h_c, remaining, rng, t_dead - 3)

    # Phase 7: Final coordinate polish
    if time.time() < t_dead - 3:
        h_c = best_h.copy()
        rem = max(2, t_dead - 4 - time.time())
        h_c, c5v = _coord_descent(h_c, n, rem, 10, rng, t_dead)
        h_c = _repair(h_c, n)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_c.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
