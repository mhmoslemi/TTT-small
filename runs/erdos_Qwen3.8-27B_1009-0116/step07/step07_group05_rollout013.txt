def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

            stall += 1
            continue
        c5n = _eval(h)
        if c5n < c5 - 1e-14:
            c5 = c5n
            step = min(step * 1.08, 0.15)
            stall = 0
        else:
            h[i] -= delta
            h[j] += delta
            step *= 0.65
            stall += 1
            if stall > 20:
                step = base_step
                stall = 0
    return h, c5

def _rand_2pt(h, n_iter, rng, c5, check_fn, scale=1e-4):
    n = len(h)
    best_c = c5
    best_h = h.copy()
    stall = 0
    for _ in range(n_iter):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                stall = 0
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                stall += 1
        else:
            stall += 1
        if stall > 500:
            scale *= 0.7
            if scale < 1e-8:
                break
            stall = 0
    return best_h, best_c

def _rand_3pt(h, n_iter, rng, c5, check_fn, scale=3e-4):
    n = len(h)
    best_c = c5
    best_h = h.copy()
    stall = 0
    for _ in range(n_iter):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 7 + rng.randint(0, 6)) % n
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                stall = 0
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
                stall += 1
        else:
            stall += 1
        if stall > 800:
            scale *= 0.6
            if scale < 1e-8:
                break
            stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _bc():
        return time.time() - t0 > budget_s - 20

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    n = n0

    # Phase 1: Subgradient descent with active lag tracking
    for kt, n_iter in [(2, 1200), (3, 800), (4, 500)]:
        if _bc():
            break
        step = 0.06
        hw = best_h.copy()
        c5w = _eval(hw)
        stall = 0
        for _ in range(n_iter):
            if _bc():
                break
            g = _grad_top(hw, kt)
            hn = _project(hw - step * g * (2.0 / n), n * 0.5)
            c5n = _eval(hn)
            if c5n < c5w - 1e-14:
                hw, c5w = hn, c5n
                step = min(step * 1.12, 0.4)
                stall = 0
            else:
                step *= 0.55
                stall += 1
                if stall > 25:
                    break
        if c5w < best_c5:
            best_c5 = c5w
            best_h = hw

    # Phase 2: SLSQP with smooth max
    if budget_s - (time.time() - t0) > 150:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j, 'args': (n,)}]
        for _ in range(4):
            if _bc():
                break
            try:
                res = minimize(_obj_s, best_h, args=(n,), jac=_obj_g,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-17})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 3: Gradient-informed local search
    if not _bc():
        rem = budget_s - 120 - (time.time() - t0)
        n3 = max(50, int(rem / 1.5)) if rem > 0 else 50
        n3 = min(n3, 5000)
        best_h, best_c5 = _grad_local(best_h, n3, _bc, base_step=0.02)

    # Phase 4: 3-point random (medium scale)
    if not _bc():
        rem = budget_s - 50 - (time.time() - t0)
        n4 = max(100, int(rem / 0.003)) if rem > 0 else 100
        n4 = min(n4, 150000)
        best_h, best_c5 = _rand_3pt(best_h, n4, rng, best_c5, _bc, scale=4e-4)

    # Phase 5: 2-point random (fine)
    if not _bc():
        rem = budget_s - 10 - (time.time() - t0)
        n5 = max(100, int(rem / 0.002)) if rem > 0 else 100
        n5 = min(n5, 400000)
        best_h, best_c5 = _rand_2pt(best_h, n5, rng, best_c5, _bc, scale=1.5e-4)

    # Phase 6: 2-point ultra-fine
    if not _bc():
        rem = budget_s - 3 - (time.time() - t0)
        n6 = max(50, int(rem / 0.001)) if rem > 0 else 50
        n6 = min(n6, 500000)
        best_h, best_c5 = _rand_2pt(best_h, n6, rng, best_c5, _bc, scale=5e-5)

    # Phase 7: Final micro-polish
    if not _bc():
        n7 = min(200000, max(50, int((budget_s - 1 - (time.time() - t0)) / 0.0005)))
        best_h, best_c5 = _rand_2pt(best_h, n7, rng, best_c5, _bc, scale=2e-5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
