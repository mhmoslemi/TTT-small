import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())

def _correlate_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_simplex_box(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_c_k(h, k, n, dx):
    """Gradient of c_k with respect to h, where k is the index in full correlation (0 to 2n-2)"""
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        # c = sum_{j=0}^{n-1-sh} h[j]*(1-h[j+sh]) * dx
        # d/dh[i]:
        #   if 0<=i<n-sh: dx*(1-h[i+sh])
        #   if sh<=i<n:   -dx*h[i-sh]
        if sh == 0:
            g = (1.0 - 2.0 * h) * dx
        else:
            g[:n-sh] += (1.0 - h[sh:]) * dx
            g[sh:] -= h[:n-sh] * dx
    else:
        sp = -sh
        # c = sum_{j=sp}^{n-1} h[j]*(1-h[j-sp]) * dx
        # d/dh[i]:
        #   if sp<=i<n:   dx*(1-h[i-sp])
        #   if 0<=i<n-sp: -dx*h[i+sp]
        g[sp:] += (1.0 - h[:n-sp]) * dx
        g[:n-sp] -= h[sp:] * dx
    return g

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_c_k(h, k, n, dx)
    return g

def _nesterov_smooth(h, alpha, n_steps, step0, t_end):
    n = len(h)
    target = n / 2.0
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.9
    step = step0
    for i in range(n_steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_simplex_box(h - step * v, target)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.05, step0 * 10)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_simplex_box(h_prev - step * v, target)
            f = _smooth_obj(h, alpha)
    return h

def _coord_descent_top(h, n, duration_s, top_T, rng, t_end):

        best_grad = np.max(grads, axis=0)  # subgradient of max over top lags...

        # But for a smoothed version, we want a weighted combination

        # Direction that reduces c_{k*}: move in direction -g_active[i] for i, +g_active[j] for j

        if s > target:
            lo = mid
        else:
            hi = mid

    sh = k - (n - 1)

        if sh > 0:
            g[sh:] -= h[:n - sh] * dx
    else:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:]) * dx
        g[:n - sh] -= h[:n - sh] * dx
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp]) * dx
        g[:n - sp] -= h[sp:] * dx
    return g

def _smooth_fg(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    ws = w.sum()
    f = m + np.log(ws) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += (w[k] / ws) * _grad_k(h, k, n, dx)
    return f, g

def _accel(h, alpha, steps, step0, mu, t_end):
    n = len(h)
    tgt = n * 0.5
    f, _ = _smooth_fg(h, alpha)
    v = np.zeros(n)
    st = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        _, g = _smooth_fg(h, alpha)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        hn = _proj(h - st * v, tgt)
        fn, _ = _smooth_fg(hn, alpha)
        if fn < f:
            h = hn
            f = fn
            st = min(st * 1.04, step0 * 12.0)
        else:
            st *= 0.5
            if st < 1e-13:
                break
            h = _proj(hp - st * v, tgt)
            f, _ = _smooth_fg(h, alpha)
    return h

def _fw_step(h, g, n):
    target = n * 0.5
    idx = np.argsort(g)
    v = np.zeros(n)
    k_full = int(target)
    v[idx[:k_full]] = 1.0
    frac = target - k_full
    if frac > 1e-12:
        v[idx[k_full]] = frac
    return v

def _frank_wolfe(h, alpha, n_iter, t_end):
    n = len(h)
    tgt = n * 0.5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        _, g = _smooth_fg(h, alpha)
        v = _fw_step(h, g, n)
        t = 2.0 / (it + 2)
        h = (1 - t) * h + t * v
        h = _proj(h, tgt)
    return h

def _coord(h, n, dur, top_T, rng, t_end):
    dx = 2.0 / n
    tgt = n * 0.5
    best_c5 = _eval(h)
    end_t = time.time() + dur
    no_imp = 0
    while time.time() < min(end_t, t_end - 4):
        c = _corr(h)
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        grads = np.array([_grad_k(h, k, n, dx) for k in top_idx])
        imp = False
        for _ in range(n * 4):
            if time.time() > min(end_t, t_end - 4):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-12
            d_hi = min(1.0 - h[i], h[j]) - 1e-12
            if d_lo >= d_hi:
                continue
            gr = (np.sqrt(5) - 1) / 2
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(28):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-12:
                h[i] += d_opt
                h[j] -= d_opt
                imp = True
        if imp:
            nc = _eval(h)
            if nc < best_c5 - 1e-15:
                best_c5 = nc
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > 50:
            h += 1e-4 * rng.randn(n)
            h = _proj(h, tgt)
            no_imp = 0
        if abs(h.sum() - tgt) > 1e-6:
            h = _proj(h, tgt)
    return h, best_c5

def _local_fine(h, rng, n_tries, t_end):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    bc = c5c
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-18:
                c5c = cn
                if cn < bc:
                    bc = cn
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, bc
