import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n / 2.0) > 1e-9:
        h = _proj_sum(h, n / 2.0)
    return h.copy()

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-18:
            continue
        sh = (n - 1) - k
        v = wk * dx
        if sh >= 0:
            g[sh:] += v * (1.0 - h[:n - sh])
            g[:n - sh] -= v * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += v * (1.0 - h[sp:])
            g[sp:] -= v * h[:n - sp]
    return g

def _pnorm_obj(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = np.sum(np.maximum(c, 0)**p)**(1.0 / p)
    val += pen * (h.sum() - n * 0.5)**2
    return val

def _pnorm_grad(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    ca = np.maximum(c, 1e-30)
    cp = ca**(p - 1)
    s = ca.sum() if p == 1 else np.sum(ca**p)
    w = cp / s**((p - 1) / p) if p > 1 else cp / s
    g = _grad_w(h, w)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _smobj(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    ex = np.exp(np.clip(beta * (c - m), -50, 50))
    return m + np.log(ex.sum()) / beta + pen * (h.sum() - n * 0.5)**2

def _smgrad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    ex = np.exp(np.clip(beta * (c - m), -50, 50))
    w = ex / ex.sum()
    g = _grad_w(h, w)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, objf, grf, args, nit):
    n = len(h)
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(objf, h, jac=grf, args=args, method='L-BFGS-B',
                     bounds=b, options={'maxiter': nit, 'ftol': 1e-18,
                                         'gtol': 1e-14, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, nit, k, st0, t0, budget):
    n = len(h)
    hc = h.copy()
    v = np.zeros(n)
    st = st0
    best = _eval(hc)
    bh = hc.copy()
    stall = 0
    for _ in range(nit):
        if time.time() - t0 > budget - 15:
            break
        c = _corrs(hc)
        tk = np.argsort(c)[-k:]
        w = np.zeros(2 * n - 1)
        w[tk] = 1.0
        g = _grad_w(hc, w) / k
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = 0.7 * v + 0.3 * (g / gn)
        ht = _proj_sum(hc - st * v, n * 0.5)
        ct = _eval(ht)
        if ct < best - 1e-16:
            hc = ht
            best = ct
            bh = hc.copy()
            stall = 0
            st = min(st * 1.02, 0.05)
        else:
            st *= 0.5
            stall += 1
            if st < 1e-9:
                break
    return bh, best

def _lsearch(h, nit, rng, t0, budget):
    n = len(h)
    hc = h.copy()
    best = _eval(hc)
    bh = hc.copy()
    sc = 5e-5
    for i in range(nit):
        if time.time() - t0 > budget - 15:
            break
        a = rng.randint(n)
        b2 = (a + 1 + rng.randint(0, 8)) % n
        d = rng.randn() * sc
        na, nb = hc[a] + d, hc[b2] - d
        if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:
            hc[a], hc[b2] = na, nb
            cn = _eval(hc)
            if cn < best:
                best = cn
                bh = hc.copy()
            else:
                hc[a], hc[b2] = na - d, nb + d
        if i % 12000 == 0 and sc > 1e-7:
            sc *= 0.55
    return bh, best

def _downsample(h, new_n):
    old_n = len(h)
    if new_n >= old_n:
        return h.copy()
    idx = np.linspace(0, old_n - 1, new_n).round().astype(int)
    return h[idx]
