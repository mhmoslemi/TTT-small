import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h); m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _sgrad(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        k = int(idx[ii]); sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lbfgsb_opt(h, alpha, topk, maxiter, penalty, deadline):
    n = len(h)
    def obj(x):
        if time.time() > deadline: return 1e10
        return _smax(x, alpha, topk) + penalty * (x.sum() - n * 0.5) ** 2
    def jac(x):
        if time.time() > deadline: return np.zeros(n)
        g = _sgrad(x, alpha, topk)
        g += 2 * penalty * (x.sum() - n * 0.5)
        return g
    try:
        r = minimize(obj, h.copy(), jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n); beta = 0.9; step = step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _sgrad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn; step = min(step * 1.015, step0 * 5.0)
        else:
            step *= 0.4
            h = _proj(hp - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _binary_anneal(h0, niter, rng, deadline, T0, T1):
