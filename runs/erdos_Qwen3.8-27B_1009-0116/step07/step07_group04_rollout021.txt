        val = max_c + sm / alpha

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, alpha, n, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = c.max()
    sh = np.clip(alpha * (c - mc), -500, 500)
    val = mc + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n * 0.5) ** 2
    return val

def _sm_grad(x, alpha, n, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = c.max()
    sh = np.clip(alpha * (c - mc), -500, 500)
    w = np.exp(sh)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        wk = w[idx] * dx
        if wk < 1e-14:
            continue
        if s >= 0:
            g[s:] += wk * (1.0 - x[:n - s])
            g[:n - s] -= wk * x[s:]
        else:
            p = -s
            g[:n - p] += wk * (1.0 - x[p:])
            g[p:] -= wk * x[:n - p]
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def _lbfgs_run(h, alpha, mu, maxiter, t0, limit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj_f(x):
        return _sm_obj(x, alpha, n, mu)
    def grad_f(x):
        return _sm_grad(x, alpha, n, mu)
    try:
        res = minimize(obj_f, h, jac=grad_f, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        if time.time() - t0 < limit:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _anneal(h, alpha_list, mu_list, maxiters, t0, limit):
    n = len(h)
    for i, (alpha, mu) in enumerate(zip(alpha_list, mu_list)):
        if time.time() - t0 > limit:
            break
        h = _lbfgs_run(h, alpha, mu, maxiters, t0, limit)
    return h

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.repeat(h, factor).copy()

def _fourier_start(n, seed):
    rng = np.random.RandomState(seed)
    x = np.arange(n)
    h = np.full(n, 0.5)
    for m in range(1, 8):
        freq = m * 2.0 * np.pi * x / n
        h += 0.04 / m * np.cos(freq + rng.randn() * 0.5)
    h = _repair(h, n)
    return h

def _local_search(h, n_tries, rng, t0, limit):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > limit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 10)) % n
        scale = max(5e-6, 0.01 / (1 + _it * 0.001))
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        d3 = -d1 - d2
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] + d3
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk - d3
    return best_h, best_c

def _ultra_fine(h, n_tries, rng, t0, limit):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > limit:
            break
        i = rng.randint(n)
        j = (i + 1) % n
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
