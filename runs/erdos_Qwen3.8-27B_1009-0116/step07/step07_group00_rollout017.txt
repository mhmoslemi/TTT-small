    return bh, bc

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(xo, h, kind='linear', fill_value='extrapolate', bounds_error=False)
    return _repair(f(xn), n_new)

def _slsqp_brief(h, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    c = _corr(h)
    k = 5
    idx = np.argpartition(c, -k)[-k:]
    def obj(x):
        cc = _corr(x)
        mx = cc.max()
        s = 5000.0 * (cc[idx] - mx)
        s = np.clip(s, -300, 300)
        return mx + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / 5000.0
    def jac(x):
        cc = _corr(x)
        mx = cc.max()
        s = 5000.0 * (cc[idx] - mx)
        s = np.clip(s, -300, 300)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            g += w[ii] * _grad_at(x, idx[ii])
        return g
    try:
        from scipy.optimize import minimize
        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0,1)]*n,
                     options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-8})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(h0)

    # Phase 1: Brief L-BFGS-B (fast convergence early)
    d1 = min(deadline, t0 + 120)
    h1 = _slsqp_brief(h0, 80, d1)
    c1 = _eval(h1)
    if c1 < best_c5: best_h, best_c5 = h1.copy(), c1

    # Phase 2: Top-k gradient descent (exact, adaptive k)
    d2 = min(deadline, t0 + 350)
    h2, c2 = _topk_descent(best_h, d2, 2, 10, 8000, 0.004)
    if c2 < best_c5: best_h, best_c5 = h2.copy(), c2

    # Phase 3: Coordinate descent (fast, many sweeps)
    d3 = min(deadline, t0 + 500)
    h3, c3 = _coord_desc(best_h, 200, d3, rng)
    if c3 < best_c5: best_h, best_c5 = h3.copy(), c3

    # Phase 4: Second top-k descent from improved point
    d4 = min(deadline, t0 + 700)
    h4, c4 = _topk_descent(best_h, d4, 3, 8, 6000, 0.002)
    if c4 < best_c5: best_h, best_c5 = h4.copy(), c4

    # Phase 5: Targeted 2-point search
    d5 = min(deadline, t0 + 900)
    h5, c5 = _targeted_2pt(best_h, 50000, rng, d5)
    if c5 < best_c5: best_h, best_c5 = h5.copy(), c5

    # Phase 6: Brief annealing
    d6 = min(deadline, t0 + budget_s - 30)
    h6, c6 = _anneal(best_h, 20000, rng, d6, 0.002, 1e-10)
    if c6 < best_c5: best_h, best_c5 = h6.copy(), c6

    # Phase 7: Final coordinate + targeted polish
    if time.time() < deadline:
        h7, c7 = _coord_desc(best_h, 50, deadline, rng)
        if c7 < best_c5: best_h, best_c5 = h7.copy(), c7
    if time.time() < deadline:
        h8, c8 = _targeted_2pt(best_h, 20000, rng, deadline)
        if c8 < best_c5: best_h, best_c5 = h8.copy(), c8

    # Quick check with n//2 (different discretization)
    n_half = n0 // 2
    if n_half >= 100 and time.time() < deadline - 60:
        d9 = time.time() + min(40, deadline - time.time() - 10)
        hh = _resample(best_h, n_half)
        hh, ch = _topk_descent(hh, d9, 2, 6, 3000, 0.005)
        if ch < best_c5:
            best_h, best_c5 = hh.copy(), ch

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
