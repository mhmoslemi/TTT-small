    m = c.max()
    sc = np.clip(a * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    return w, c.max()

            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_full(h, n, a):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g, m

def _grad_topk(h, n, k_top):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g / k_top, c.max()

def _gd_nesterov(h, n, rounds, mode, param, init_steps, check_fn):
    """Gradient descent with Nesterov momentum. mode='full' uses param as sharpness, mode='topk' uses param as k."""
    c5c = _eval(h)
    v = np.zeros(n)
    mu = 0.85  # momentum
    steps = list(init_steps)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        if mode == 'full':
            g, _ = _grad_full(h, n, param)
        else:
            g, _ = _grad_topk(h, n, param)
        # Nesterov: step in direction of v, then recompute
        v_new = mu * v + (1 - mu) * g
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * v_new, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                v = v_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.5] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 10:
                    steps = steps[:10]
                break
        if not improved:
            v *= 0.3
            steps = [s * 0.4 for s in steps]
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0
    return h, c5c

def _slsqp_fun(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()

                          [0.004, 0.002, 0.001, 5e-4, 2e-4, 1e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: 3-point local search
    if _el() < budget_s - 25:
        nt = min(int((budget_s - 25 - _el()) / 0.004), 100000)
        best_h, best_c5 = _local_3pt(best_h, n, nt, rng, best_c5, _bc)

    # Phase 8: 2-point ultra-fine polish
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.002), 200000)
        best_h, best_c5 = _local_2pt(best_h, n, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
