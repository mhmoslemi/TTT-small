import numpy as np

from scipy.optimize import linprog

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0

        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

def _lag_grad(h, s, n):

        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        sp = -s
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]

            for k in top:

                g = _lag_grad(h, s, n)
                rhs = c5v[k] - g @ h
                row = np.zeros(n + 1)
                row[:n] = g
                row[n] = -1.0
                A_rows.append(row)
                b_vals.append(rhs)
            c_obj = np.zeros(n + 1); c_obj[n] = 1.0
            A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
            b_eq = np.array([n / 2.0])

            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_vals),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if not res.success:

            h_lp = res.x[:n]
            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
