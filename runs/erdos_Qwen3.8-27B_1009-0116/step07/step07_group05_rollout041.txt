        h2, c52 = _grad_desc(best_h.copy(), 600, [0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 5e-4], kt, False, 0, deadline - 300)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 2: Soft-weighted gradient descent
    for sa in [80, 200, 500, 1500]:
        if time.time() > deadline - 200:
            break
        h2, c52 = _grad_desc(best_h.copy(), 400, [0.03, 0.015, 0.007, 0.003, 0.001, 5e-4], 5, True, sa, deadline - 200)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 3: SLSQP with multiple smoothness levels
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for a_val in [2000, 8000, 30000, 100000]:
        if time.time() > deadline - 150:
            break
        try:
            res = minimize(_smooth_obj, best_h, args=(a_val,), jac=_smooth_grad,
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-15})
            hr = _repair(res.x, n)
            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr
        except Exception:
            pass

    # Phase 4: Structured sinusoidal perturbations + gradient refinement
    t_arr = 2.0 * np.arange(n) / n
    for freq in [1, 2, 3, 5, 7, 11, 13]:
        if time.time() > deadline - 80:
            break
        for amp in [0.002, 0.006, 0.015]:
            if time.time() > deadline - 80:
                break
            phase = rng.rand() * 2.0 * np.pi
            perturb = amp * np.sin(2.0 * np.pi * freq * t_arr + phase)
            h_p = _repair(best_h + perturb, n)
            h2, c52 = _grad_desc(h_p, 300, [0.02, 0.01, 0.005, 0.002, 0.001], 3, False, 0, deadline - 80)
            if c52 < best_c5:
                best_c5, best_h = c52, h2

    # Phase 5: Very fine gradient polish with small steps
    for kt in [2, 3, 4, 3, 5]:
        if time.time() > deadline - 40:
            break
        h2, c52 = _grad_desc(best_h.copy(), 500, [0.003, 0.001, 5e-4, 2e-4, 1e-4], kt, False, 0, deadline - 40)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 6: Fine soft gradient polish
    if time.time() < deadline - 30:
        for sa in [3000, 8000]:
            if time.time() > deadline - 30:
                break
            h2, c52 = _grad_desc(best_h.copy(), 400, [0.002, 0.001, 5e-4, 2e-4], 5, True, sa, deadline - 30)
            if c52 < best_c5:
                best_c5, best_h = c52, h2

    # Phase 7: Ultra-fine 2-point local search
    nc = n
    remain = max(0, deadline - time.time() - 2)
    ntries = int(min(remain / 0.0025, 150000))
    for _ in range(ntries):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = 2.5e-6 * rng.randn()
        ni = best_h[i] + d
        nj = best_h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            oi, oj = best_h[i], best_h[j]
            best_h[i], best_h[j] = ni, nj
            c5n = _eval(best_h)
            if c5n < best_c5:
                best_c5 = c5n
            else:
                best_h[i], best_h[j] = oi, oj

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
