def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _exact_subgrad(h, kmax):
    n = len(h)
    tau = kmax - (n - 1)
    g = np.zeros(n)
    if tau >= 0:
        g[:n - tau] += (1.0 - h[tau:])
        if tau > 0:
            g[tau:] -= h[:n - tau]
    else:
        s = -tau
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def _smooth_obj(h, alpha, topk):
    c = _correl_scaled(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx] - m
    s = np.clip(alpha * ct, -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smooth_grad(h, alpha, topk):
    n = len(h)
    c = _correl_scaled(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s_raw = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s_raw - s_raw.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = int(idx[ii])
        tau = k - (n - 1)
        wk = w[ii] * dx
        if tau >= 0:
            g[:n - tau] += wk * (1.0 - h[tau:])
            if tau > 0:
                g[tau:] -= wk * h[:n - tau]
        else:
            s = -tau
            g[s:] += wk * (1.0 - h[:n - s])
            g[:n - s] -= wk * h[s:]
    return g

def _nesterov_mixed(h0, alpha, topk, steps, step0, deadline):
    n = len(h0)
    h = h0.copy()
    f = _smooth_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    best_f = f
    best_h = h.copy()
    for it in range(steps):
        if time.time() > deadline:
            break
        c = _correl_scaled(h)
        kmax = int(np.argmax(c))
        g_exact = _exact_subgrad(h, kmax)
        gn = np.sqrt(np.sum(g_exact ** 2))
        if gn > 1e-12:
            g_exact = g_exact / gn
        g_smooth = _smooth_grad(h, alpha, topk)
        gs = np.sqrt(np.sum(g_smooth ** 2))
        if gs > 1e-12:
            g_smooth = g_smooth / gs
        g = 0.4 * g_exact + 0.6 * g_smooth
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(h - step * v, n * 0.5)
        fn = _smooth_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            if fn < best_f:
                best_f = fn
                best_h = h.copy()
            step = min(step * 1.02, step0 * 8.0)
        else:
            step *= 0.5
            h = _proj_sum(hp - step * v, n * 0.5)
            f = _smooth_obj(h, alpha, topk)
    return best_h

def _pair_anneal(h0, niter, rng, deadline, T0, T1):
    n = len(h0)
    h = h0.copy()
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n // 3)) % n
        d = 0.008 * max(T, 1e-12) ** 0.5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _slsqp_refine(h0, alpha, topk, maxit, deadline, n):
    if time.time() > deadline:
        return h0
    target = n * 0.5
    ones = np.ones(n)
    def obj(x):
        return _smooth_obj(x, alpha, topk)
    def jac(x):
        return _smooth_grad(x, alpha, topk)
    def cerr(x):
        return x.sum() - target
    def cjac(x):
        return ones
    try:
        r = minimize(obj, h0, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cerr, 'jac': cjac}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h0

def _lag_target_descent(h0, deadline, nsteps):
    """Iteratively target the worst lag for reduction."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _eval(h)
    base_lr = 0.005
    for it in range(nsteps):
        if time.time() > deadline:
            break
        c = _correl_scaled(h)
        kmax = int(np.argmax(c))
        c5v = c[kmax]
        # Find all lags within 5% of max (cluster of near-max)
        thresh = c5v * 0.05
        near_max = np.where(c > c5v - thresh)[0]
        g = np.zeros(n)
        for k in near_max:
            g += _exact_subgrad(h, int(k))
        g /= len(near_max)
        gn = np.sqrt(np.sum(g ** 2))
        if gn < 1e-12:
            break
        g = g / gn
        lr = base_lr / (1 + it * 0.01)
        hn = _proj_sum(h - lr * g, n * 0.5)
        c5n = _eval(hn)
        if c5n < _eval(h):
            h = hn
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            base_lr *= 0.7
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
