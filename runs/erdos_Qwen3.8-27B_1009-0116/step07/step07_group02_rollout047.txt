            best_c5, best_h = c5, h.copy()

    # Phase 5: Active descent + pair again
    if time.time() < deadline - 80:
        ha, ca = _active_descent(best_h.copy(), 3, 500, deadline)
        if ca < best_c5:
            best_c5, best_h = ca, ha.copy()
        hp, cp = _pair_opt(best_h.copy(), 10, deadline)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    # Phase 6: Random restart
    if time.time() < deadline - 60:
        hr = _perturb_restart(best_h, rng, deadline)
        cr = _eval(hr)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()

    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 30:
        h = _lbfgs(best_h.copy(), 100000000, 8000, deadline)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 8: Active descent final
    if time.time() < deadline - 15:
        ha, ca = _active_descent(best_h.copy(), 2, 500, deadline)
        if ca < best_c5:
            best_c5, best_h = ca, ha.copy()

    # Phase 9: Fine local search
    if time.time() < deadline - 3:
        hl, cl = _local_fine(best_h.copy(), 300000, rng, deadline)
        if cl < best_c5:
            best_c5, best_h = cl, hl.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
