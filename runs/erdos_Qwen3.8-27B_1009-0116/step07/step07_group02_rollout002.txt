import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Bisection for threshold
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, n):

def _grad_lag(h, k, n):
    """Gradient of the overlap at a specific lag k (0-indexed in correlate output)."""
    dx = 2.0 / n
    shift = (n - 1) - k
    g = np.zeros(n)
    if shift >= 0:
        cnt = n - shift
        g[:cnt] += (1.0 - h[shift:])
        if shift > 0:
            g[shift:] -= h[:cnt]
    else:
        sp = -shift
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _subgrad_active(h, n_top, steps, step0, deadline):

        return c[top_idx].sum()

        return g

    except Exception:

    dx = 2.0 / n
    shift = (n - 1) - k

    if shift >= 0:

        g[:cnt] += (1.0 - h[shift:])

            g[shift:] -= h[:cnt]
    else:

        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, n):

def _grad_lag(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:cnt]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _lbfgs_topk(h, n_top, maxit, deadline):

        return g
    bnds = [(0.0, 1.0)] * n

                     options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-15, 'maxls': 30})
        return _repair(r.x, n)
    except:
        return h
