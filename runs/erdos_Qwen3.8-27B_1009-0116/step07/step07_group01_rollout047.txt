```python
def _grad_softmax(h, alpha, top_k=None):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    if top_k is not None:
        idx = np.argsort(c)[-top_k:]
        c_active = c[idx]
        s = np.clip(alpha*(c_active - c_active.max()), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
    else:
        idx = np.arange(2*n-1)
        s = np.clip(alpha*(c - c.max()), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
    
    dx = 2.0/n
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        if k < n:
            # j in [0, k]
            if k < n:
                js = np.arange(0, k+1)
                g[js] += wk * (1.0 - 2.0*h[k - js])
        else:
            # j in [k-n+1, n-1]
            js = np.arange(k-n+1, n)
            g[js] += wk * (1.0 - 2.0*h[k - js])
    return g
```

This is O(n) per active k, and with top_k active lags, O(top_k * n) total.

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_topk(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k < 2 * n - 1:
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(2 * n - 1)
    c_a = c[idx]
    s = np.clip(alpha * (c_a - c_a.max()), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k = idx[ii]
        wk = w[ii] * dx
        if k < n:
            js = np.arange(0, k + 1)
            g[js] += wk * (1.0 - 2.0 * h[k - js])
        else:
            js = np.arange(k - n + 1, n)
            g[js] += wk * (1.0 - 2.0 * h[k - js])
    return g

def _proj_sum(h, target):
    n = len(h)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pg_solve(h0, alpha, top_k, n_iter, step0, el_fn, budget_frac=0.7):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for it in range(n_iter):
        if el_fn() > budget_frac:
            break
        g = _grad_topk(h, alpha, top_k)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ls in range(15):
            h_new = _proj_sum(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 8.0)
                taken = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            step *= 0.5
        if not taken:
            step = max(step * 1.2, 1e-5)
    return best_h, best_c

def _local_search(h, n_iter, rng, step_base, el_fn, deadline):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        k = (i + 8 + rng.randint(0, 12)) % n
        d1 = step_base * rng.randn()
        d2 = step_base * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.repeat(h, factor)

def _slsqp_step(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        return _grad_topk(x, alpha, 2*n-1)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return _repair(res.x, n)
