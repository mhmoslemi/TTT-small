import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k_idx, n):
    """Gradient of the k-th correlation lag w.r.t. h"""
    g = np.zeros(n)
    dx = 2.0 / n
    sh = (n - 1) - k_idx  # the actual shift
    if sh >= 0:
        # c = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        # c = sum_{i=0}^{n-1-sp} h[i+sp]*(1-h[i])
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _nesterov_topk(h, k_top, steps, step0, t0, time_limit):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    best_f = f
    best_h = h.copy()
    for _s in range(steps):
        if time.time() - t0 > time_limit:
            break
        c = _eval_vec(h)
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = 1.0 / k_top
        for k in top_k:
            g += wk * _grad_lag(h, k, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
            step = min(step * 1.05, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return best_h, best_f

def _slsqp_smooth_max(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_smooth_max_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(x, k, n)
    return g

def _slsqp_opt(h, alpha, maxiter, ftol, t0, time_limit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    
    def obj_f(x):
        return _slsqp_smooth_max(x, alpha, n)
    
    def grad_f(x):
        return _slsqp_smooth_max_grad(x, alpha, n)
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_f, h, jac=grad_f,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, n_tries, rng, t0, time_limit, scale=1e-4):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, n//4)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _worst_lag_repair(h, n_iters, t0, time_limit):
    """Directly minimize the worst lag"""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        c = _eval_vec(best_h)
        k_worst = c.argmax()
        g = _grad_lag(best_h, k_worst, n)
        # Project gradient: remove component along constraint
        g = g - g.mean()
        # Line search
        step = 0.01
        for _ls in range(30):
            h_new = _proj_to_sum(best_h - step * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-12:
                best_h = h_new
                best_c = c_new
                break
            step *= 0.7
            if step < 1e-10:
                break
    return best_h, best_c

    elif s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
        g *= dx
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:n])
        g[t:] -= h[:n - t]
        g *= dx
    return g

def _nesterov(h, ktop, steps, step0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    bh, bf = h.copy(), f
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = _eval_vec(h)
        topk = np.argsort(c)[-ktop:]
        g = np.zeros(n)
        for idx in topk:
            g += _grad_one(h, idx, n)
        g /= ktop
        v = mu * v + (1.0 - mu) * g
        hp = h.copy()
        hn = _proj(h - step * v, n * 0.5)
        fn = _eval(hn)
        if fn < f:
            h, f = hn, fn
            if f < bf:
                bf, bh = f, h.copy()
            step = min(step * 1.04, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-11:
                break
            h = _proj(hp - step * v, n * 0.5)
            f = _eval(h)
    return bh, bf

def _sm_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sm_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-10:
            g += w[k] * _grad_one(x, k, n)
    return g

def _slsqp(h, alpha, maxit, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _sm_obj(x, alpha, n), h,
                     jac=lambda x: _sm_grad(x, alpha, n),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': ftol})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _worst_repair(h, niter, t_end):
    n = len(h)
    bh, bc = h.copy(), _eval(h)
    for _ in range(niter):
        if time.time() > t_end:
            break
        c = _eval_vec(bh)
        kw = c.argmax()
        g = _grad_one(bh, kw, n)
        g -= g.mean()
        step = 0.05
        ok = False
        for _ in range(50):
            hn = _proj(bh - step * g, n * 0.5)
            fn = _eval(hn)
            if fn < bc - 1e-14:
                bh, bc = hn, fn
                ok = True
                break
            step *= 0.6
            if step < 1e-12:
                break
        if not ok:
            break
    return bh, bc

def _local(h, ntries, rng, t_end, scale):
    n = len(h)
    fc = _eval(h)
    bh, bc = h.copy(), fc
    for _ in range(ntries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, n // 3)) % n
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            fn = _eval(h)
            if fn < fc - 1e-18:
                fc = fn
                if fn < bc:
                    bc, bh = fn, h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return bh, bc

def _resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Nesterov k=50 (broad gradient, 80s)
    h2, c2 = _nesterov(h, 50, 6000, 0.003, t0 + 80)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 2: Nesterov k=10 (sharper, 60s)
    h2, c2 = _nesterov(best_h.copy(), 10, 5000, 0.002, t0 + 150)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: SLSQP with increasing alpha
    for alpha, mit in [(20000, 800), (80000, 1000), (300000, 1000)]:
        if time.time() - t0 > 550:
            break
        h3 = _slsqp(best_h.copy(), alpha, mit, 1e-16)
        c3 = _eval(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Worst-lag directed repair
    if time.time() - t0 < budget_s - 280:
        best_h, best_c5 = _worst_repair(best_h, 300, budget_s - 200)

    # Phase 5: High-alpha SLSQP polish
    if time.time() - t0 < budget_s - 200:
        h5 = _slsqp(best_h.copy(), 800000, 800, 1e-17)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 6: Multi-scale local search
    t_ls_end = budget_s - 5
    if time.time() < t_ls_end:
        elapsed = time.time() - t0
        remain = t_ls_end - time.time()
        scales = [8e-5, 4e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        per_scale = remain / len(scales)
        for sc in scales:
            if time.time() >= t_ls_end:
                break
            nt = max(100, int(per_scale / 0.003))
            best_h, best_c5 = _local(best_h, nt, rng, t_ls_end, sc)

    # Phase 7: Try resampling to nearby n for potential improvement
    if time.time() - t0 < budget_s - 15:
        for n_try in [n0 - 40, n0 + 40, n0 - 80, n0 + 80]:
            if time.time() - t0 > budget_s - 8:
                break
            if n_try < 100:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            c_r = _eval(h_r)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
