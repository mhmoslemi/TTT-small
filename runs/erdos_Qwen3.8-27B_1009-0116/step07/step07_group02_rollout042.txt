import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        s = -sh
        g[s:] += (1.0 - h[:-s])
        g[:-s] -= h[s:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_k(x, k, n)
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 35:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-11, 'maxls': 60})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _toplag(h, n_lags, rounds, deadline):
    n = len(h)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 3e-5, 1e-5]
    c5 = _eval(h)
    for _ in range(rounds):
        if time.time() > deadline - 25:
            break
        c = _cvals(h)
        idx = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(h, k, n)
        g /= n_lags
        imp = False
        for t in steps:
            h2 = _proj(h - t * g, n)
            v = _eval(h2)
            if v < c5 - 1e-15:
                h, c5 = h2, v
                imp = True
                break
        if not imp:
            break
    return h, c5

def _local(h, iters, rng, dl, sc, span):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for _ in range(iters):
        if time.time() > dl - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, span)) % n
        d = sc * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0 <= a <= 1 and 0 <= b <= 1:
            h[i], h[j] = a, b
            v = _eval(h)
            if v < c5 - 1e-19:
                c5 = v
                if v < bc:
                    bc, bh = v, h.copy()
            else:
                h[i], h[j] = a - d, b + d
    return bh, bc

def _gen_cos(n, f, ph):
    x = np.linspace(0, 2, n, endpoint=False)
    return _repair(0.5 + 0.35 * np.cos(2 * np.pi * f * x + ph), n)

def _gen_humps(n, c, w, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for j in range(3):
        cj = (c + j * 0.35 + rng.randn() * 0.05) % 2.0
        h += 0.7 * np.exp(-((x - cj) / w) ** 2)
    return _repair(h, n)

def _gen_rand(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for j in range(6):
        c = rng.rand() * 2.0
        w = 0.08 + rng.rand() * 0.25
        a = 0.2 + rng.rand() * 0.6
        h += a * np.exp(-((x - c) / w) ** 2)
    return _repair(h, n)

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(h0)
    best_n = n0

    # --- Brief low-res exploration for diverse starts ---
    if time.time() < dl - 500:
        n_lo = 64
        f_up = n0 // n_lo
        starts = [_downsample(h0, f_up)]
        starts.append(_gen_cos(n_lo, 2, rng.rand() * 6.28))
        starts.append(_gen_cos(n_lo, 3, rng.rand() * 6.28))
        starts.append(_gen_cos(n_lo, 4, rng.rand() * 6.28))
        starts.append(_gen_humps(n_lo, rng.rand(), 0.15, rng))
        starts.append(_gen_rand(n_lo, rng))
        for sc in starts:
            if time.time() > dl - 400:
                break
            hs = sc.copy()
            for a in [2000, 50000]:
                hs = _lbfgs(hs, a, 1500, dl)
            hs, _ = _toplag(hs, 3, 100, dl)
            hu = _repair(_upsample(hs, f_up), n0)
            v = _eval(hu)
            if v < best_c5:
                best_c5 = v
                best_h = hu.copy()

    # --- Main intensive optimization: L-BFGS annealing ---
    h = best_h.copy()
    alphas = [300, 1000, 5000, 20000, 80000, 300000, 1000000, 4000000]
    for i, a in enumerate(alphas):
        if time.time() > dl - 100:
            break
        # More iterations for mid-range alphas
        mx = 4000 if a < 100000 else 2000
        h = _lbfgs(h, a, mx, dl)
        v = _eval(h)
        if v < best_c5:
            best_c5 = v
            best_h = h.copy()
        # Periodic top-lag interlude
        if i % 2 == 1:
            if time.time() < dl - 80:
                h2, c2 = _toplag(h, 3, 300, dl)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()
                    h = h2.copy()

    # --- Focused top-lag sweeps ---
    for nl in [2, 3, 4]:
        if time.time() > dl - 50:
            break
        h2, c2 = _toplag(best_h.copy(), nl, 800, dl)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # --- High-alpha L-BFGS final push ---
    if time.time() < dl - 40:
        h = _lbfgs(best_h, 8000000, 3000, dl)
        v = _eval(h)
        if v < best_c5:
            best_c5 = v
            best_h = h.copy()

    # --- Multi-scale local search ---
    scales = [(1e-4, 10, 30000), (1e-5, 6, 40000), (1e-6, 4, 50000), (1e-7, 3, 60000)]
    for sc, span, it in scales:
        if time.time() > dl - 8:
            break
        h2, c2 = _local(best_h.copy(), it, rng, dl, sc, span)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # --- Final micro-polish ---
    if time.time() < dl - 4:
        h2, c2 = _local(best_h, 50000, rng, dl, 5e-8, 2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
