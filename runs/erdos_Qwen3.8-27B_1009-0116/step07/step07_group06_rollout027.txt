                    diff -= a
                else:

                    diff += a

**Key insight**: The problem is min-max. At the optimum, multiple lags should achieve the maximum simultaneously. The CD method naturally drives towards this because it always reduces the current maximum. But it might oscillate between different lags being the maximum.

**Alternative approach**: Instead of pure CD, use a "water filling" approach:
- Identify the current top m lags
- Compute a weighted gradient that equalizes them
- Take a step in that direction
- Repeat

This is like the method of multipliers or a primal-dual approach.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0

        else:

def _grad_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    dx = 2.0 / n
    d_lo1 = max(-(n-1), p - n + 1)
    d_hi1 = min(n-1, p)
    if d_lo1 <= d_hi1:
        ds = np.arange(d_lo1, d_hi1 + 1)
        g[ds + n - 1] += (1.0 - h[p - ds])
    d_lo2 = max(-(n-1), -p)
    d_hi2 = min(n-1, n-1-p)
    if d_lo2 <= d_hi2:
        ds = np.arange(d_lo2, d_hi2 + 1)
        g[ds + n - 1] -= h[p + ds]
    return g * dx

def _grad_lag(d, h, n):
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    e = abs(d)
    if d >= 0:
        g[:n-e] += (1.0 - h[e:n])
        g[e:] -= h[:n-e]
    else:
        g[e:] += (1.0 - h[:n-e])
        g[:n-e] -= h[e:]
    return g
