def _mom_grad_desc(h, alpha, n_steps, lr0, beta1, beta2):
    """Momentum gradient descent with sum constraint"""
    n = len(h)
    target = n / 2.0
    m = np.zeros(n)
    v = np.zeros(n)
    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    for t in range(n_steps):
        g = _grad_smooth(h, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (m ** 2)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        lr = lr0 / (np.sqrt(t + 1) + 1e-8)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = _proj_sum(h_new, target)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _block_opt_fast(h, bsz, n_passes, rng):
    """Block coordinate descent"""
    n = len(h)
    nb = n // bsz
    best_c = _eval(h)
    for _pass in range(n_passes):
        order = rng.permutation(nb)
        for b in order:
            sl = slice(b * bsz, (b + 1) * bsz)
            x0 = h[sl].copy()
            bsum = x0.sum()
            # Try random perturbations of this block
            best_x = x0.copy()
            best_bc = best_c
            for _trial in range(20):
                pert = 0.02 * rng.randn(bsz)
                x_new = np.clip(x0 + pert, 0, 1)
                # Adjust to maintain block sum
                diff = bsum - x_new.sum()
                if abs(diff) < 1e-6:
                    x_new = x_new
                else:
                    x_new = x_new + diff / bsz
                    x_new = np.clip(x_new, 0, 1)
                    # Re-adjust
                    x_new = x_new * (bsum / x_new.sum() + 1e-15)
                    x_new = np.clip(x_new, 0, 1)

                    best_x = x_new.copy()

            best_c = best_bc
    return _repair(h, n), best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.repeat(h, f)
    return out

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _adam_desc(h, alphas, steps, lr0):
    n = len(h)
    target = n * 0.5
    m1 = np.zeros(n); m2 = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    t = 0
    for alpha in alphas:
        for _ in range(steps):
            t += 1
            g = _grad_smooth(h, alpha)
            b1, b2 = 0.9, 0.999
            m1 = b1 * m1 + (1 - b1) * g
            m2 = b2 * m2 + (1 - b2) * (g * g)
            m1h = m1 / (1 - b1 ** t)
            m2h = m2 / (1 - b2 ** t)
            lr = lr0 / (np.sqrt(t) + 1e-8)
            h = _proj_sum(h - lr * m1h / (np.sqrt(m2h) + 1e-8), target)
            c = _eval(h)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def _informed_local(h, n_iter, rng, base_step, elapsed_fn, budget_s):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    step = base_step
    stall = 0
    for _ in range(n_iter):
        if elapsed_fn() > budget_s - 5:
            break
        if c5cur < best_c - 1e-15:
            stall = 0
            step = min(step * 1.08, base_step * 6.0)
        else:
            stall += 1
            if stall > 500:
                step = max(step * 0.93, base_step * 0.05)
        # Get active lags for informed moves
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-5:]
        if rng.rand() < 0.5:
            # Informed: move at a location related to an active lag
            k_act = top_k[rng.randint(5)]
            sh = k_act - (n - 1)
            if sh > 0:
                i = rng.randint(0, n - sh)
                j = i + sh
            elif sh < 0:
                i = rng.randint(-sh, n)
                j = i + sh
            else:
                i = rng.randint(n)
                j = (i + 1) % n
            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5cur:
                    c5cur = c5n
                    if c5n < best_c:
                        best_c = c5n; best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        else:
            # Random 2-3 point move
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            if rng.rand() < 0.5:
                d = step * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5cur:
                        c5cur = c5n
                        if c5n < best_c:
                            best_c = c5n; best_h = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
            else:
                k = (i + 10 + rng.randint(0, 15)) % n
                d1 = step * rng.randn()
                d2 = step * rng.randn()
                ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
                if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                    old = (h[i], h[j], h[k])
                    h[i], h[j], h[k] = ni, nj, nk
                    c5n = _eval(h)
                    if c5n < c5cur:
                        c5cur = c5n
                        if c5n < best_c:
                            best_c = c5n; best_h = h.copy()
                    else:
                        h[i], h[j], h[k] = old
    return best_h, best_c

def _block_cd(h, bsz, n_passes, rng, elapsed_fn, budget_s):
    n = len(h)
    nb = n // bsz
    best_c = _eval(h)
    for _pass in range(n_passes):
        if elapsed_fn() > budget_s - 80:
            break
        order = rng.permutation(nb)
        for b in order:
            sl = slice(b * bsz, (b + 1) * bsz)
            x0 = h[sl].copy()
            bsum = x0.sum()
            best_x = x0.copy()
            best_bc = best_c
            for _trial in range(15):
                if elapsed_fn() > budget_s - 80:
                    break
                pert = 0.015 * rng.randn(bsz)
                xn = np.clip(x0 + pert, 0, 1)
                xn *= bsum / max(xn.sum(), 1e-15)
                xn = np.clip(xn, 0, 1)
                h[sl] = xn
                cn = _eval(h)
                if cn < best_bc:
                    best_bc = cn; best_x = xn.copy()
                else:
                    h[sl] = x0
            h[sl] = best_x
            best_c = best_bc
    return _repair(h, n), best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f)

        h_up = _repair(_upsample(hc, n // nc), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()
        if _el() > 70:
            break

    # Phase 2: Adam descent with alpha annealing at full res
    if _el() < budget_s - 400:
        h2, c2 = _adam_desc(best_h.copy(),
                            [100, 300, 1000, 3000, 10000, 30000, 100000, 500000],
                            800, 0.004)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Block coordinate descent
    if _el() < budget_s - 300:
        for bsz, nps in [(6, 4), (12, 3), (20, 2), (32, 2)]:
            if _el() > budget_s - 200:
                break
            h3, c3 = _block_cd(best_h.copy(), bsz, nps, rng, _el, budget_s)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

    # Phase 4: More Adam with very high alpha (exploitation)
    if _el() < budget_s - 150:
        for aseq, stp, lr in [([500000, 2000000], 400, 0.001),
                              ([5000000, 20000000], 200, 0.0005)]:
            if _el() > budget_s - 80:
                break
            h4, c4 = _adam_desc(best_h.copy(), aseq, stp, lr)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

    # Phase 5: Informed local search (main time sink)
    if _el() < budget_s - 10:
        remaining = budget_s - 10 - _el()
        ntries = int(remaining / 0.0018)
        best_h, best_c5 = _informed_local(best_h, ntries, rng, 6e-6, _el, budget_s)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 3:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        remaining = max(0, budget_s - 3 - _el())
        npol = min(int(remaining / 0.0009), 400000)
        for _ in range(npol):
            if _el() > budget_s - 3:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 6)) % nc2
            d = 1.5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
