            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 2)) % n
        d = s * rng.randn()
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d; h[j] -= d
        c = _eval(h)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            s = min(s * 1.03, scale)
        else:
            h[i] -= d; h[j] += d
            if rng.random() < 0.04:
                s = max(s * 0.92, scale * 0.1)
    return best_h, best_c

def _coord3(h, n, n_sweeps, check_fn):
    """3-coordinate descent: pick i, j, k and do a small line search."""
    best_h = h.copy()
    best_c = _eval(best_h)
    for sweep in range(n_sweeps):
        if check_fn():
            break
        imp = 0
        rng_i = np.random.RandomState(sweep * 7 + 3)
        for i in range(n):
            j = (i + 2 + rng_i.randint(0, 5)) % n
            k = (i + 8 + rng_i.randint(0, 10)) % n
            if i == j or i == k or j == k:
                continue
            # Perturb: h[i]+d, h[j]+e, h[k]-d-e (preserve sum)
            best_de = (0.0, 0.0)
            best_v = best_c
            s = 0.002
            for trial in range(4):
                d = s * rng_i.randn()
                e = s * rng_i.randn()
                if (0 <= h[i]+d <= 1 and 0 <= h[j]+e <= 1 and 0 <= h[k]-d-e <= 1):
                    h[i] += d; h[j] += e; h[k] -= d + e
                    c = _eval(h)
                    if c < best_v:
                        best_v = c; best_de = (d, e)
                    h[i] -= d; h[j] -= e; h[k] += d + e
                s *= 0.5
            if any(x != 0 for x in best_de):
                h[i] += best_de[0]; h[j] += best_de[1]; h[k] -= best_de[0] + best_de[1]
                imp += 1
                if best_v < best_c:
                    best_c = best_v
                    best_h = h.copy()
        if imp == 0:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0
    def _bc():
        return _el() > budget_s - 15

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Gradient warmup
    for alpha in [500, 5000, 50000, 200000]:
        if _bc():
            break
        steps = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
        h_new, c_new = _grad_phase(best_h.copy(), alpha, n, steps, 250, _bc)
        if c_new < best_c5:
            best_c5 = c_new; best_h = h_new

    # Phase 2: Coordinate descent (main workhorse) - multiple j offsets
    joffs = [1, 2, 3, 5, 7, 1, 2, 3]
    for joff in joffs:
        if _bc():
            break
        if _el() > budget_s - 120:
            break
        sweeps = 200
        best_h, best_c5 = _coord2(best_h, n, sweeps, joff, _bc)

    # Phase 3: Multi-restart to escape local minima
    for r in range(6):
        if _bc():
            break
        if _el() > budget_s - 80:
            break
        sc = 0.006 * (1.0 - 0.12 * r)
        hp = _repair(best_h + sc * rng.randn(n), n)
        h_new, c_new = _grad_phase(hp, 100000, n,
                                    [0.05, 0.02, 0.01, 0.005, 0.002, 0.001], 150, _bc)
        if _el() < budget_s - 40:
            h_new, c_new = _coord2(h_new, n, 80, r + 1, _bc)
        if c_new < best_c5:
            best_c5 = c_new; best_h = h_new

    # Phase 4: 3-coordinate descent for more complex interactions
    if _el() < budget_s - 60:
        for joff in [1, 2, 3]:
            if _bc():
                break
            best_h, best_c5 = _coord3(best_h, n, 30, _bc)

    # Phase 5: More 2-coordinate with all offsets
    for joff in [1, 2, 3, 4, 5, 8, 13]:
        if _bc():
            break
        if _el() > budget_s - 40:
            break
        best_h, best_c5 = _coord2(best_h, n, 100, joff, _bc)

    # Phase 6: Random local search - medium
    if _el() < budget_s - 25:
        rem = budget_s - 25 - _el()
        ntry = min(int(rem / 0.0004), 250000)
        best_h, best_c5 = _local_rand(best_h, n, ntry, 8e-5, rng, _bc)

    # Phase 7: Random local search - fine
    if _el() < budget_s - 10:
        rem = budget_s - 10 - _el()
        ntry = min(int(rem / 0.0004), 200000)
        best_h, best_c5 = _local_rand(best_h, n, ntry, 3e-5, rng, _bc)

    # Phase 8: Final coordinate polish
    if _el() < budget_s - 5:
        best_h, best_c5 = _coord2(best_h, n, 50, 1, _bc)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
