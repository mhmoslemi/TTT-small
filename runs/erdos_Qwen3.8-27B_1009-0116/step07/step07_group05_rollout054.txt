    sc = np.clip(temp * (c - c.max()), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        wk = w[k] * (2.0 / n)
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _mlgd_sm(h, rounds, temp, steps_init, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        g = _sm_grad(h, temp) * dx
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.6]
                if len(steps) > 12:
                    steps = steps[:12]
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            if min(steps) < 5e-7:
                break
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _mlgd_topk(h, rounds, k_top, steps_init, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in top_k:
            g += _grad_lag(h, k)
        g *= dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.6]
                if len(steps) > 12:
                    steps = steps[:12]
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            if min(steps) < 5e-7:
                break
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _mom_hc(h, steps_init, check_fn, rounds):
    """Momentum-based gradient with projection."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)
    lr = steps_init[0]
    mom = 0.7
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        g = _sm_grad(h, 300.0) * dx
        v = mom * v - lr * g
        h_new = _project(h + v, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            stall = 0
        else:
            v *= 0.5
            lr *= 0.7
            stall += 1
            if stall > 15:
                break
    return h, c5c

def _local2(h, ntry, rng, c5c, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    step_sizes = [2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    si = 0
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        d = step_sizes[si] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                    si = max(0, si - 1)
                else:
                    si = min(len(step_sizes) - 1, si + 1)
            else:
                h[i], h[j] = ni - d, nj + d
                si = min(len(step_sizes) - 1, si + 1)
    return best_h, best_c

def _local3(h, ntry, rng, c5c, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    step_sizes = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    si = 0
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        k = (i + 7 + rng.randint(0, 6)) % nc
        d = step_sizes[si] * rng.randn()
        f = rng.rand()
        ni, nj, nk = h[i] + d, h[j] - d * f, h[k] - d * (1.0 - f)
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                    si = max(0, si - 1)
                else:
                    si = min(len(step_sizes) - 1, si + 1)
            else:
                h[i], h[j], h[k] = ni - d, nj + d * f, nk + d * (1.0 - f)
                si = min(len(step_sizes) - 1, si + 1)
    return best_h, best_c

def _slsqp_solve(h, temp, n, bnds, cons, check_fn):
    try:
        res = minimize(_sm_obj, h, args=(temp,), jac=_sm_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 600, 'ftol': 1e-16})
        return _repair(res.x, n), _eval(_repair(res.x, n))
    except Exception:
        return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Soft-max gradient with decreasing temperature (coarse to fine)
    for temp in [200, 500, 1000, 500, 200]:
        if _bc():
            break
        h2, c52 = _mlgd_sm(best_h.copy(), 1000, temp,
                           [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Top-k focused gradient
    for kt in [3, 5, 7, 4, 6, 3, 5, 3]:
        if _bc():
            break
        h2, c52 = _mlgd_topk(best_h.copy(), 700, kt,
                             [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: Momentum gradient
    if _el() < budget_s - 150:
        h2, c52 = _mom_hc(best_h.copy(), [0.03], _bc, 2000)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: SLSQP with smooth max at multiple temperatures
    if _el() < budget_s - 180:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        for a in [30000.0, 80000.0, 200000.0]:
            if _bc():
                break
            hr, c5r = _slsqp_solve(best_h.copy(), a, n, bnds, cons, _bc)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr

    # Phase 5: Perturbation restarts
    for _ in range(4):
        if _bc():
            break
        h_p = best_h + 0.015 * rng.randn(n0)
        h_p = _repair(h_p, n0)
        h2, c52 = _mlgd_sm(h_p, 500, 500.0,
                           [0.04, 0.02, 0.01, 0.005, 0.002, 0.001], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
        h2, c52 = _mlgd_topk(h2, 300, 3,
                             [0.03, 0.015, 0.008, 0.004, 0.002, 0.001], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Fine top-k + soft-max alternating
    for kt, temp in [(2, 1000), (3, 500), (2, 2000), (3, 800), (4, 600)]:
        if _bc():
            break
        h2, c52 = _mlgd_topk(best_h.copy(), 600, kt,
                             [0.015, 0.008, 0.004, 0.002, 0.001, 5e-4, 2e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
        h2, c52 = _mlgd_sm(best_h.copy(), 400, temp,
                           [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: 3-point local search
    if _el() < budget_s - 30:
        nt = min(int((budget_s - 30 - _el()) / 0.005), 120000)
        best_h, best_c5 = _local3(best_h, nt, rng, best_c5, _bc)

    # Phase 8: 2-point ultra-fine polish
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.003), 250000)
        best_h, best_c5 = _local2(best_h, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
