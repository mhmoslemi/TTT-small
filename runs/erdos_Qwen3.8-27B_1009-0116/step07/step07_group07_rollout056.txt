```python
def _lse_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, 'full') * dx
    # softmax weights
    bc = beta * c - beta * c.max()  # numerical stability
    w = np.exp(bc)
    w /= w.sum()
    # gradient
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        sh = (n-1) - k
        if sh >= 0:
            # c_k = sum_{i=0}^{n-1-sh} h[i+sh](1-h[i])
            i1 = np.arange(0, n-sh)
            g[i1+sh] += wk * (1-h[i1])
            if sh > 0:
                i2 = np.arange(0, n-sh)
                g[i2] -= wk * h[i2+sh]
        else:
            sp = -sh
            i1 = np.arange(0, n-sp)
            g[i1] += wk * (1-h[i1+sp])
            i2 = np.arange(sp, n)
            g[i2] -= wk * h[i2-sp]
    gp = 2.0 * penalty * (h.sum() - n*0.5)
    return g + gp
```

```python
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_corr(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if abs(w) < 1e-18:
            continue

        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        g[i_lo:i_hi+1] += w * (1.0 - h[i_lo+off:i_hi+1+off])
        g[i_lo+off:i_hi+1+off] -= w * h[i_lo:i_hi+1]
    return g

def _obj_pnorm(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.maximum(c, 1e-30)
    val = (np.sum(cm**p))**(1.0/p)
    s = h.sum() - n * 0.5
    return val + pen * s * s

def _grad_pnorm(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, mode='full') * dx, 1e-30)
    s = np.sum(c**p)
    factor = (s**(1.0/p - 1.0) / s)  # = s^(1/p - 1) * c^(p-1) / c... 
    # Actually: d/dh (sum c^p)^(1/p) = (1/p)(sum c^p)^(1/p-1) * p * c^(p-1)
    # = (sum c^p)^(1/p-1) * c^(p-1)
    fac = (np.sum(c**p))**(1.0/p - 1.0)
    w = fac * c**(p-1)
    g = _grad_corr(h, w)
    gp = 2.0 * pen * (h.sum() - n * 0.5)
    return g + gp

def _obj_lse(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * c - beta * c.max()
    val = (1.0/beta) * np.log(np.sum(np.exp(bc)))
    s = h.sum() - n * 0.5
    return val + pen * s * s

def _grad_lse(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * c - beta * c.max()
    e = np.exp(bc)
    w = e / e.sum()  # softmax weights
    g = _grad_corr(h, w)
    gp = 2.0 * pen * (h.sum() - n * 0.5)
    return g + gp

def _lbfgs(h0, obj_fn, grad_fn, args, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-20,
                       'gtol': 1e-15, 'maxls': 50})

def _lse_momentum(h0, max_time, time_fn, stop_fn):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    beta = 5.0  # LSE temperature
```
