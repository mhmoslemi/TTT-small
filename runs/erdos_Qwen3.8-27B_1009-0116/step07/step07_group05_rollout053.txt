    lo, hi = -2.0, 2.0

            lo = mid
        else:

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lags(h, lags, n):
    g = np.zeros(n)
    for k in lags:
        s = k - (n - 1)
        if s >= 0:
            v = n - s
            g[:v] += (1.0 - h[s:s+v])
            if s > 0:
                g[s:] -= h[:n-s]
        else:
            sp = -s
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g / max(len(lags), 1)

    m = c.max()
    sc = np.clip(a * (c - m), -300, 0)
    return m + np.log(np.sum(np.exp(sc))) / a
