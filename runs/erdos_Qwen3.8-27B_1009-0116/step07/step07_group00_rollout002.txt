        hp = h.copy()
        hn = _proj(h - step * v, n * 0.5)
        hn = _repair(hn, n)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.015, step0 * 6.0)
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            hn = _repair(hn, n)
            f = _smax_obj(hn, alpha, topk)
            h = hn
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def _obj(x):
        return _smax_obj(x, alpha, topk)
    def _jac(x):
        return _smax_grad(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_obj, h, jac=_jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _directed_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        g, c = _active_grad(h, n, 3, 0.0)
        gi = np.argmax(g)
        gj = np.argmin(g)
        if gi == gj:
            break
        sgn = 1.0 if g[gi] > 0 else -1.0
        d = 1e-5 * rng.uniform(0.5, 1.5)
        ni = h[gi] - sgn * d
        nj = h[gj] + sgn * d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[gi], h[gj] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[gi], h[gj] = ni + sgn * d, nj - sgn * d
    return bh, bc

def _random_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, nc - 2)) % nc
        d = 1.5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP (quick start)
    dl1 = min(deadline, t0 + 120)
    for alpha, tk, mi in [(3000, 40, 200), (15000, 20, 200), (60000, 10, 150)]:
        if time.time() > dl1:
            break
        h = _slsqp(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Subgradient with backtracking (main workhorse)
    dl2 = min(deadline, t0 + 500)
    phases = [
        (30, 0.0, 0.05, 800),
        (15, 0.0, 0.02, 800),
        (8, 0.0, 0.008, 800),
        (5, 0.0, 0.004, 1000),
        (3, 500, 0.002, 1200),
        (3, 0.0, 0.001, 1000),
    ]
    for topk, soft, st0, ns in phases:
        if time.time() > dl2:
            break
        h = _repair(best_h.copy(), n)
        h, c5v = _subgrad_bt(h, n, topk, soft, st0, ns, dl2)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Nesterov with high alpha
    dl3 = min(deadline, t0 + 750)
    for alpha, tk, st0, ns in [(200000, 8, 0.003, 2500),
                               (1000000, 5, 0.001, 2500),
                               (5000000, 3, 0.0005, 2000)]:
        if time.time() > dl3:
            break
        h = _nesterov(best_h.copy(), alpha, tk, ns, st0, dl3)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Block anneal
    dl4 = min(deadline, t0 + 920)
    blocks = [1, 2, 3, 4, 5, 7, 11]
    h_a, c5_a = _block_anneal(best_h.copy(), 50000, rng, dl4, 0.004, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 5: Directed + random polish
    dl5 = deadline
    rem = max(0, dl5 - time.time())
    n_dir = min(int(rem * 0.55 / 0.001), 20000)
    best_h, best_c5 = _directed_polish(best_h, n_dir, rng, dl5)

    rem = max(0, dl5 - time.time())
    n_rnd = min(int(rem / 0.001), 15000)
    best_h, best_c5 = _random_polish(best_h, n_rnd, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
