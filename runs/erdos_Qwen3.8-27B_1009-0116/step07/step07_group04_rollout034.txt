            g[seg1] += aL * (1.0 - h[L:n]) - bL * h[L:n]
            # For j in [L, n): g[j] += (bL-aL) - (aL+bL)*h[j-L]
            seg2 = slice(L, n)
            g[seg2] += (bL - aL) - (aL + bL) * h[:n-L]
    return g * dx

def _nesterov_softmax(h, alpha_start, alpha_end, steps, t0, t_budget):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = 0.003
    alpha = alpha_start
    for s in range(steps):
        if time.time() > t_budget:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (s / max(steps, 1))
        g = _softmax_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj(h_prev - step * v, n * 0.5)
            f = _eval(h)

def _topk_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:

            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a

        h, f = _sgd_top1(best_h.copy(), 3000, 0.0008, t0 + budget_s * 0.82)
        if f < best_c5:
            best_c5, best_h = f, h.copy()

    h, f = _nesterov(best_h.copy(), tk(3), 2500, 0.0004, 0.9, t0 + budget_s * 0.9)
    if f < best_c5:
        best_c5, best_h = f, h.copy()

    if time.time() < t0 + budget_s * 0.92:
        h2, f2 = _binary_push(best_h, n)
        if f2 < best_c5:
            best_c5, best_h = f2, h2.copy()

    rem = budget_s - 15 - (time.time() - t0)
    if rem > 5:
        nt = min(int(rem * 0.7 / 0.003), 50000)
        best_h, best_c5 = _local_search(best_h, nt, rng, t0 + budget_s - 10)

    rem2 = budget_s - 3 - (time.time() - t0)
    if rem2 > 2:
        nt2 = min(int(rem2 / 0.002), 40000)
        best_h, best_c5 = _two_pt_search(best_h, nt2, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
