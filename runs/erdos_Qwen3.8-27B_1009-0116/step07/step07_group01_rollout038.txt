def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(60):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
       if diff >0: order = np.argsort(1.0-h)
       else: order = np.argsort(h)

          if diff >0:

    else:

    wv = np.exp(s)

def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = float(h.sum())
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        if abs(s - target) < 1e-13:
            break
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - float(h[idx]))
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, float(h[idx]))
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def _proj(h, target):
    lo = min(float(h.min()) - 1.0, -1.0)
    hi = max(float(h.max()) + 1.0, 1.0)
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _to_binary(h, n):
    out = np.zeros(n, dtype=float)
    order = np.argsort(-h)
    out[order[:n // 2]] = 1.0
    return out


def _resample(h, m):
    n = len(h)
    if m == n:
        return h.copy()
    if n % m == 0:
        f = n // m
        return h[:m * f].reshape(m, f).mean(axis=1).copy()
    if m % n == 0:
        f = m // n
        out = np.empty(m, dtype=float)
        for i in range(n):
            out[i * f:(i + 1) * f] = h[i]
        return out
    xs = np.linspace(0.0, 1.0, n, endpoint=False)
    xm = np.linspace(0.0, 1.0, m, endpoint=False)
    return np.interp(xm, xs, h)


def _make_binary_idx(n):
    ar = np.arange(n, dtype=np.int32)
    fi = (ar[:, None] - ar[None, :]) + (n - 1)
    si = (ar[None, :] - ar[:, None]) + (n - 1)
    return fi.astype(np.int32, copy=False), si.astype(np.int32, copy=False)


def _fill_swap_delta(delta, fi, si, h, h1m, i, j, l0):
    delta.fill(0.0)
    delta[fi[i]] -= h1m
    delta[si[i]] += h
    delta[fi[j]] += h1m
    delta[si[j]] -= h
    delta[(i - j) + l0] += 1.0
    delta[(j - i) + l0] += 1.0
    delta[l0] = 0.0


def _rebuild_binary_lists(h):
    n = len(h)
    ones = np.flatnonzero(h > 0.5).astype(np.int32)
    zeros = np.flatnonzero(h <= 0.5).astype(np.int32)
    po = np.full(n, -1, dtype=np.int32)
    pz = np.full(n, -1, dtype=np.int32)
    for p, i in enumerate(ones):
        po[i] = p
    for p, j in enumerate(zeros):
        pz[j] = p
    return ones, zeros, len(ones), len(zeros), po, pz


def _active_weights(c, alpha, K, l0):
    m = len(c)
    K = int(max(1, min(K, m)))
    if K == m:
        top = np.arange(m)
    else:
        top = np.argpartition(c, -K)[-K:]
    vals = c[top]
    mv = float(vals.max())
    s = np.clip(alpha * (vals - mv), -50.0, 50.0)
    s -= s.max()
    wv = np.exp(s)
    ws = float(wv.sum())
    if ws <= 1e-300:
        wv.fill(1.0)
        ws = float(len(wv))
    wv /= ws
    active = []
    aw = []
    for idx, wt in zip(top, wv):
        ii = int(idx)
        if ii != l0 and wt > 1e-8:
            active.append(ii)
            aw.append(float(wt))
    if not active:
        idx2 = l0 + 1
        if idx2 >= m:
            idx2 = l0 - 1
        active.append(int(idx2))
        aw.append(1.0)
    return np.asarray(active, dtype=np.int32), np.asarray(aw, dtype=float)


def _scores_active(h, active, aw, n):
    s1 = np.zeros(n, dtype=float)
    s2 = np.zeros(n, dtype=float)
    l0 = n - 1
    for a, wt in zip(active, aw):
        a = int(a)
        wt = float(wt)
        d = a - l0
        if d > 0:
            s1[d:n] += wt * (1.0 - h[:n - d])
            s2[:n - d] += wt * h[d:n]
        elif d < 0:
            e = -d
            s1[:n - e] += wt * (1.0 - h[e:n])
            s2[e:n] += wt * h[:n - e]
    return s1, s2
