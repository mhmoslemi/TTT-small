import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                room = (1.0 - h[idx]) if diff > 0 else h[idx]
                a = min(abs(diff), room)
                if a < 1e-15:
                    break
                h[idx] += a if diff > 0 else -a
                diff += (-a if diff > 0 else a)
                if abs(diff) < 1e-13:
                    break
    return h

def _c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    """Gradient of c_k with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _active_set_grad(h, c, frac):
    """Average gradient over top `frac` fraction of correlation values."""
    n = len(h)
    m = c.max()
    thresh = m - frac * (m - c.min())
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    g = np.zeros(n)
    for k in active:
        g += _grad_k(h, k, n)
    g /= len(active)
    return g

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                room = (1.0 - h[idx]) if diff > 0 else h[idx]
                a = min(abs(diff), room)
                if a < 1e-15:
                    break
                h[idx] += a * (1.0 if diff > 0 else -1.0)
                diff -= a * (1.0 if diff > 0 else -1.0)
                if abs(diff) < 1e-13:
                    break
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0, 1)

def _grad_lag(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _avg_grad_subset(h, c, thresh, n):
    m = c.max()
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    g = np.zeros(n)
    for k in active:
        g += _grad_lag(h, int(k), n)
    return g / len(active)

def _pgd_active(h, deadline, niter, step0, frac_init, frac_min):

    step = step0

    stall = 0

        if time.time() > deadline:

        thresh = f - frac * (f - c.min() + 1e-15)

                    stall = 0

def _nesterov_smooth(h, alpha, topk, niter, step0, deadline):
    n = len(h)
    target = n * 0.5
    def smax(x):
        c = _cvals(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def smax_grad(x):
        c = _cvals(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            k = int(idx[ii])
            sh = (n - 1) - k
            wk = w[ii] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    v = np.zeros(n)
    beta = 0.82
    step = step0
    f = smax(h)
    for _ in range(niter):
        if time.time() > deadline:
            break
        g = smax_grad(h)

        hn = _proj(h - step * v, target)

            step = min(step * 1.03, step0 * 10)
        else:

            f = smax(h)
    return h

    c5c = _c5(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):

            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        bs = blocks[rng.randint(len(blocks))]

def _slsqp_round(h, deadline):
    n = len(h)
    def obj(x):
        c = _cvals(x)
        m = c.max()
        topk = 8
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        al = 50000.0
        s = np.clip(al * (ct - m), -300, 300)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / al
    def obj_grad(x):
        c = _cvals(x)
        m = c.max()
        topk = 8
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        al = 50000.0
        s = np.clip(al * (ct - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue

            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
