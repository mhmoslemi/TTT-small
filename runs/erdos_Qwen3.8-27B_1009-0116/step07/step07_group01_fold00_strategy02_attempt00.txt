   def _repair_step(h, target_sum):
        n = len(h)
        current_sum = np.sum(h)
        target_sum_adj = target_sum
        # Add step correction
        for i in range(20):
            if np.abs(current_sum - target_sum_adj) < 1e-6:
                break
            diff = target_sum_adj - current_sum
            if diff > 0:
                # Increase sum by pushing elements toward 1
                sorted_idx = np.argsort(h)
                mask = (sorted_idx < int(diff * n)) | (sorted_idx >= n - int(diff * n))
                h[mask] += 0.01 * np.clip(diff, 0, 1)
            else:
                # Decrease sum by pushing elements toward 0
                sorted_idx = np.argsort(1 - h)
                mask = (sorted_idx < int(-diff * n)) | (sorted_idx >= n - int(-diff * n))
                h[mask] -= 0.01 * np.clip(diff, 0, 1)
        return np.clip(h, 0, 1)
