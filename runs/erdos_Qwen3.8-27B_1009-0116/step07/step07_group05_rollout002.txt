C₅(h) = max_k { corr_k * dx }

where corr_k = sum_i h[i] * (1-h[i+k]) (with appropriate bounds)

The smooth approximation: C_smooth = (1/γ) * log(sum_k exp(γ * corr_k * dx))

For γ large enough, this approximates the max well. The gradient is:
∂C/∂h[i] = (1/γ) * sum_k w_k * ∂(corr_k)/∂h[i]
where w_k = exp(γ * corr_k * dx) / sum_j exp(γ * corr_j * dx)

This is a softmax weighting over all lags.

For Nesterov, we maintain a momentum vector:
y_new = x + α * (y - x)  (extrapolate)
x_new = y_new - β * ∇f(y_new)

import numpy as np

def _repair(h, n):
    """Project h onto [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        # Bisection for projection
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        # Fix residual
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_max_val(corr, gamma):
    m = corr.max()
    sc = np.clip(gamma * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / gamma

def _smooth_max_grad_h(h, gamma):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    sc = np.clip(gamma * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g
