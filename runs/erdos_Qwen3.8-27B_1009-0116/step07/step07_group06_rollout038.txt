import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_at_pos(h, p, n, q_vals, q_scaled):
    g = np.zeros(2 * n - 1)
    idx1 = p + q_vals
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p - q_vals
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g * q_scaled

def _subgrad(h, deadline, m_top=8, max_iters=30000):
    n = len(h)
    dx = 2.0 / n
    q_vals = np.arange(2 * n - 1) - (n - 1)
    q_scaled = (2.0 / n)
    best_h = h.copy()
    best_c = _eval(h)
    mom = np.zeros(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = c.max()
        top_k = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top_k:
            d = k - (n - 1)
            if d >= 0:
                if d == 0:
                    g[:] += 1.0 - 2.0 * h
                else:
                    g[:n-d] += (1.0 - h[d:n])
                    g[d:] -= h[:n-d]
            else:
                dd = -d
                if dd == 0:
                    g[:] += 1.0 - 2.0 * h
                else:
                    g[dd:] += (1.0 - h[:n-dd])
                    g[:n-dd] -= h[dd:]
        g /= m_top
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        base = 0.4 / (gn + 1e-12)
        ok = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            mom = 0.5 * mom - base * mult * g
            h_new = _proj_sum(h + mom, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-14:
                h = h_new
                c5 = c5n
                ok = True
                break
        if not ok:
            mom *= 0.3
        if c5 < best_c:

    return best_h, best_c

def _cd(h, max_iter, rng, deadline):

    dx = 2.0 / n
    q_vals = np.arange(2 * n - 1) - (n - 1)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float) * dx
