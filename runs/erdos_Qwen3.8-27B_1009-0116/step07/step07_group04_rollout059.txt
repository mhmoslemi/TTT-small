import numpy as np
from scipy.optimize import minimize
import time

_N = 0
_ALPHA = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]

    wk = dx / k_top

def _fix_topk_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * omh[sh:]
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * omh[:n-sp]
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov(h, k_top, steps, step0, deadline, mu=0.84):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h = h.copy()
    best_f = f
    for _s in range(steps):
        if time.time() > deadline:
            break
        g = _fix_topk_grad(h, k_top)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
            step = min(step * 1.02, step0 * 4.0)
        else:
            step *= 0.5
            if step < 1e-11:
                break
    return best_h

def _cons_eq(x):
    return x.sum() - _N * 0.5

def _cons_jac(x):
    return np.ones(_N)

def _lse_obj_fn(x):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / _N)
    m = c.max()
    s = np.clip(_ALPHA * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / _ALPHA

def _lse_grad_fn(x):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / _N)
    m = c.max()
    s = np.clip(_ALPHA * (c - m), -400, 400)
    w = np.exp(s - s.max())
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(_N)
    w /= wsum
    n = _N
    g = np.zeros(n)
    omh = 1.0 - x
    dxv = 2.0 / n
    top_ids = np.argsort(w)[-25:]
    for k in top_ids:
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dxv
        if sh >= 0:
            g[:n-sh] += wk * omh[sh:]
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * omh[:n-sp]
            g[:n-sp] -= wk * x[sp:]
    return g

def _slsqp_run(x0, alpha, maxiter, deadline):
    global _N, _ALPHA
    _N = len(x0)
    _ALPHA = alpha
    n = _N
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    try:
        res = minimize(_lse_obj_fn, x0, jac=_lse_grad_fn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(x0, n)

def _local_search_3pt(h, n_tries, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, n // 3)) % n
        k = (j + 3 + rng.randint(0, n // 3)) % n
        sc = 5e-5
        d1 = sc * rng.randn()
        d2 = sc * rng.randn()
        dk = -d1 - d2
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] + dk
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk - dk
    return best_h, best_c

def _local_search_2pt(h, n_tries, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):

        i = rng.randint(n)

        d = 2.5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj

                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
