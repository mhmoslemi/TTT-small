        h_cos = _repair(0.5 * (1 + 0.6 * np.cos(4 * np.pi * xc)), nc)
        for hc in [h_c, h_cos]:
            if _el() > 125: break
            hc = _proj_grad(hc, 300, 0.012, 4, mu=0.7)
            hc = _slsqp(hc, 8000, 400, 1e-13)
            hc_up = _repair(_up(hc, f), nf)
            c5u = _eval(hc_up)
            if c5u < best_c5: best_c5 = c5u; best_h = hc_up.copy()

    # Phase 2: SLSQP alpha sweep
    for alpha in [2000, 8000, 30000, 120000, 500000, 2000000]:
        if _el() > 380: break
        h_new = _slsqp(best_h, alpha, 1200, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5: best_c5 = c5n; best_h = h_new.copy()

    # Phase 3: Top-k projected gradient
    if _el() < budget_s - 260:
        for steps, s0, tk, mu in [(400, 0.005, 3, 0.7), (300, 0.003, 5, 0.75),
                                    (250, 0.002, 7, 0.8), (200, 0.002, 4, 0.7)]:
            if _el() > budget_s - 160: break
            h_new = _proj_grad(best_h, steps, s0, tk, mu)
            c5n = _eval(h_new)
            if c5n < best_c5: best_c5 = c5n; best_h = h_new.copy()

    # Phase 4: Lag focus
    if _el() < budget_s - 200:
        for it, tk in [(500, 2), (400, 3), (300, 5)]:
            if _el() > budget_s - 130: break
            h_new = _lag_focus(best_h, it, tk)
            c5n = _eval(h_new)
            if c5n < best_c5: best_c5 = c5n; best_h = h_new.copy()

    # Phase 5: Block coordinate
    if _el() < budget_s - 160:
        for bsz, npass in [(12, 3), (20, 2), (30, 2)]:
            if _el() > budget_s - 90: break
            h_new = _block_opt(best_h.copy(), bsz, 50, npass)
            c5n = _eval(h_new)
            if c5n < best_c5: best_c5 = c5n; best_h = h_new.copy()

    # Phase 6: Very high alpha SLSQP
    if _el() < budget_s - 90:
        for alpha in [5000000, 20000000]:
            if _el() > budget_s - 45: break
            h_new = _slsqp(best_h, alpha, 500, 1e-16)
            c5n = _eval(h_new)
            if c5n < best_c5: best_c5 = c5n; best_h = h_new.copy()

    # Phase 7: 3-point local search
    if _el() < budget_s - 50:
        nt = min(int((budget_s - 50 - _el()) * 700), 400000)
        best_h, best_c5 = _local3(best_h, nt, rng, 5e-6)

    # Phase 8: 2-point ultra-fine
    if _el() < budget_s - 2:
        nt = min(int((budget_s - 2 - _el()) * 1000), 500000)
        best_h, best_c5 = _local2(best_h, nt, rng, 1.2e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
