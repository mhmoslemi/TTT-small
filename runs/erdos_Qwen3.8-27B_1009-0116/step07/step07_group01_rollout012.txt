    t = n // 2
    if kind == 0:
        p = max(3, n // 25)
        period = np.zeros(p)
        period[:p//2 + 1] = 1.0
        h = np.tile(period, n // p + 1)[:n].copy()
        ones = int(h.sum())
        if ones > t:
            idx = rng.permutation(np.where(h > 0.5)[0])[:ones - t]
            h[idx] = 0
        elif ones < t:
            idx = rng.permutation(np.where(h < 0.5)[0])[:t - ones]
            h[idx] = 1
    elif kind == 1:
        phi = (1.0 + 5.0**0.5) / 2.0
        fr = (np.arange(n) * phi) % 1.0
        h = (fr > 0.5).astype(float)
        ones = int(h.sum())
        if ones > t:
            idx = rng.permutation(np.where(h > 0.5)[0])[:ones - t]
            h[idx] = 0
        elif ones < t:
            idx = rng.permutation(np.where(h < 0.5)[0])[:t - ones]
            h[idx] = 1
    elif kind == 2:
        idx = rng.permutation(n)[:t]
        h[idx] = 1.0
    else:
        p1, p2 = 7, 11
        mask = (np.arange(n) % p1 < 3) & (np.arange(n) % p2 < 5)
        h = mask.astype(float)
        ones = int(h.sum())
        if ones > t:
            idx = rng.permutation(np.where(h > 0.5)[0])[:ones - t]
            h[idx] = 0
        elif ones < t:
            idx = rng.permutation(np.where(h < 0.5)[0])[:t - ones]
            h[idx] = 1
    return h

def _soft_threshold(h, beta):
    mid = h - 0.5
    return 1.0 / (1.0 + np.exp(-beta * mid))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase A1: Binarize initial and greedy
    h_bin = _binarize(h0, n)
    h_bin, bc1 = _greedy_bin(h_bin.copy(), rng, t0, 50)
    if bc1 < best_c5:
        best_c5 = bc1; best_h = h_bin.copy()

    # Phase A2: Diverse binary starts
    for kind in range(4):
        if _el() > 100: break
        hk = _make_pattern(n, kind, rng)
        hk, bk = _greedy_bin(hk.copy(), rng, t0, 35)
        if bk < best_c5:
            best_c5 = bk; best_h = hk.copy()

    # Phase A3: Multi-scale (downsample from best, optimize, upsample)
    for scale in [4, 2]:
        if _el() > 160: break
        nc = n // scale
        hc = _down(_repair(best_h, n), scale)
        hc = _binarize(_repair(hc, nc), nc)
        hk = _make_pattern(nc, 0, rng)
        if _eval(hk) < _eval(hc):
            hc = hk
        hc, hc_c = _greedy_bin(hc.copy(), rng, t0, 30)
        hu = _repair(_up(hc, scale), n)
        hu, hu_c = _greedy_bin(hu.copy(), rng, t0, 25)
        if hu_c < best_c5:
            best_c5 = hu_c; best_h = hu.copy()

    # Phase B: PGD continuous refinement
    h_work = _repair(best_h.copy(), n)
    specs = [
        (1500, 500, 0.006, 6, 0.0),
        (8000, 450, 0.004, 10, 0.0),
        (40000, 400, 0.003, 14, 0.005),
        (150000, 350, 0.002, 20, 0.02),
        (500000, 300, 0.001, 30, 0.05),
    ]
    for alpha, steps, s0, tk, bp in specs:
        if _el() > 550: break
        h_work, pw_c = _pgd_top(h_work.copy(), t0, 550, alpha, steps, s0, tk, bp)
        if pw_c < best_c5:
            best_c5 = pw_c; best_h = h_work.copy()
            h_work = best_h.copy()

    # Phase C: Soft binarization + re-optimize
    if _el() < 650:
        for beta in [20, 50, 100]:
            if _el() > 650: break
            hs = _soft_threshold(_repair(best_h, n), beta)
            hs = _repair(hs, n)
            hs, hs_c = _greedy_bin(_binarize(hs, n), rng, t0, 20)
            if hs_c < best_c5:
                best_c5 = hs_c; best_h = hs.copy()

    # Phase D: PGD again from current best
    if _el() < 750:
        h_work = _repair(best_h.copy(), n)
        for alpha, steps, s0, tk, bp in [(10000, 300, 0.003, 16, 0.0),
                                          (100000, 250, 0.002, 25, 0.03)]:
            if _el() > 750: break
            h_work, pw_c = _pgd_top(h_work.copy(), t0, 750, alpha, steps, s0, tk, bp)
            if pw_c < best_c5:
                best_c5 = pw_c; best_h = h_work.copy()

    # Phase E: Local search (3-point)
    p_end = budget_s - 60
    rem = p_end - _el()
    if rem > 5:
        niter = min(int(rem / 0.0014), 700000)
        best_h, best_c5 = _local3(best_h, niter, rng, 4e-6)

    # Phase F: 2-point micro polish
    rem = budget_s - 2 - _el()
    if rem > 3:
        niter = min(int(rem / 0.0007), 500000)
        best_h, best_c5 = _local2(best_h, niter, rng, 1.2e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
