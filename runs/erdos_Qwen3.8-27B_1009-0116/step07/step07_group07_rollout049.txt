import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _lse_obj(h, t, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    val = (m + np.log(np.sum(np.exp(c - m)))) / t
    return val + penalty * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, t, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    e = np.exp(t * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in np.where(w > 1e-10)[0]:
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    gp = 2.0 * penalty * (h.sum() - n * 0.5)
    return g + gp

def _lbfgs_lse(h, t, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(t, penalty), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_mom(h, n_iter, t, step0, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.72
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    for i in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        m = c.max()
        e = np.exp(t * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for k in np.where(w > 1e-10)[0]:
            wk = w[k] * 2.0 / n
            sh = (n - 1) - k
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wk * (1.0 - h_c[sh:sh + cnt])
                if sh > 0:
                    g[sh:] -= wk * h_c[:cnt]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h_c[:n - sp])
                g[:n - sp] -= wk * h_c[sp:]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.02, 0.06)
        else:
            step *= 0.55
            stall += 1
            if step < 1e-9:
                h_c = best_h + np.random.randn(n) * 1e-4
                h_c = _proj_sum(h_c, n * 0.5)
                step = step0 * 0.5
                stall = 0
            elif stall > 80:
                break
    return best_h, best_c

def _coord_search(h, n_iter, rng, el_fn, bc_fn):
