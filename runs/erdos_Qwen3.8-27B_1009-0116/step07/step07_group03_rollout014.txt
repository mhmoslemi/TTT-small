        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_step(h, v, alpha, step, dx, n):
    g = _smooth_grad(h, alpha, dx)
    v_new = 0.85 * v + 0.15 * g
    h_new = _proj_to_sum(h - step * v_new, n * 0.5)
    return h_new, v_new

def _nesterov(h, alpha, n_steps, step0, t_end):
    n = len(h)
    dx = 2.0 / n
    fv = _smooth_obj(h, alpha, dx)
    v = np.zeros(n)
    step = step0
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        h_new, v = _nesterov_step(h, v, alpha, step, dx, n)
        f_new = _smooth_obj(h_new, alpha, dx)
        if f_new < fv:
            h = h_new
            fv = f_new
            step = min(step * 1.04, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-11:
                break
            h, v = _nesterov_step(h, 0.85 * v, alpha, step, dx, n)
            fv = _smooth_obj(h, alpha, dx)
    return h

def _top_max_eval(h, i_val, j_val, i, j, top_idx, n, dx):
    old_i, old_j = h[i], h[j]
    h[i] = i_val
    h[j] = j_val
    mx = 0.0
    for k in top_idx:
        sh = k - (n - 1)
        if sh >= 0:
            val = np.dot(h[:n-sh], 1.0 - h[sh:]) * dx
        else:
            val = np.dot(h[-sh:], 1.0 - h[:-sh]) * dx
        if val > mx:
            mx = val
    h[i] = old_i
    h[j] = old_j
    return mx

def _golden_pair(h, i, j, top_idx, n, dx):
    d_lo = max(-h[i], h[j] - 1.0) + 1e-13
    d_hi = min(1.0 - h[i], h[j]) - 1e-13
    if d_lo >= d_hi:
        return 0.0
    gr = (np.sqrt(5) - 1) / 2
    x1 = d_hi - gr * (d_hi - d_lo)
    x2 = d_lo + gr * (d_hi - d_lo)
    f1 = _top_max_eval(h, h[i]+x1, h[j]-x1, i, j, top_idx, n, dx)
    f2 = _top_max_eval(h, h[i]+x2, h[j]-x2, i, j, top_idx, n, dx)
    for _ in range(14):
        if f1 < f2:
            d_hi, x2, f2 = x2, x1, f1
            x1 = d_hi - gr * (d_hi - d_lo)
            f1 = _top_max_eval(h, h[i]+x1, h[j]-x1, i, j, top_idx, n, dx)
        else:
            d_lo, x1, f1 = x1, x2, f2
            x2 = d_lo + gr * (d_hi - d_lo)
            f2 = _top_max_eval(h, h[i]+x2, h[j]-x2, i, j, top_idx, n, dx)
    return (d_lo + d_hi) / 2.0

def _grad_top(h, top_idx, n, dx):
    g = np.zeros(n)
    for k in top_idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh]) * dx
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh] * dx
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp]) * dx
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp] * dx
    return g

def _sa_search(h, n, t_end, rng, T_init, cooling, patience):
    best_h = h.copy()
    best_c5 = _eval(h)
    target = n * 0.5
    dx = 2.0 / n
    T = T_init
    T_min = 3e-8
    no_improve = 0
    S0 = T_init

    while time.time() < t_end:
        c_full = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_c5 = c_full.max()
        if cur_c5 < best_c5 - 1e-15:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            scale = 0.06 * min(1.0, S0 / max(T, T_min)) ** 0.3
            h = best_h + scale * rng.randn(n)
            h = _proj_to_sum(h, target)
            T = S0 * 0.4
            no_improve = 0
            cur_c5 = _eval(h)
            continue

        ratio = T / S0
        top_T = max(3, min(14, int(3 + 11 * ratio ** 0.4)))
        top_idx = np.argpartition(c_full, -top_T)[-top_T:]
        h_save = h.copy()
        r = rng.random()

        if r < 0.45:
            gv = _grad_top(h, top_idx, n, dx)
            pos = np.where(gv > 0.01)[0]
            neg = np.where(gv < -0.01)[0]
            if len(pos) > 0 and len(neg) > 0:
                i = neg[rng.randint(len(neg))]
                j = pos[rng.randint(len(pos))]
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            d_max = min(h[j], 1.0 - h[i])
            if d_max > 1e-6:
                d = 0.15 * d_max * rng.uniform(0.3, 1.2)
                h[i] += d
                h[j] -= d
                h = np.clip(h, 0, 1)
                if abs(h.sum() - target) > 1e-9:
                    h = _proj_to_sum(h, target)
        elif r < 0.72:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n // 6)) % n
            if i != j:
                d = _golden_pair(h, i, j, top_idx, n, dx)
                if abs(d) > 1e-12:
                    h[i] += d
                    h[j] -= d
                    h = np.clip(h, 0, 1)
        elif r < 0.82:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                d = _golden_pair(h, i, j, top_idx, n, dx)
                if abs(d) > 1e-12:
                    h[i] += d
                    h[j] -= d
                    h = np.clip(h, 0, 1)
        elif r < 0.90:
            blen = rng.choice([4, 8, 16, 32])
            if blen < n // 2:
                i = rng.randint(0, n - blen)
                for _ in range(12):
                    j = rng.randint(0, n - blen)
                    if abs(j - i) >= blen:
                        break
                else:
                    h = h_save
                    T = max(T * cooling, T_min)
                    continue
                h[i:i+blen], h[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
                h = _proj_to_sum(h, target)
        elif r < 0.96:
            scale = 0.012 * np.sqrt(ratio + 0.15)
            h = h + scale * rng.randn(n)
            h = _proj_to_sum(h, target)
        else:
            hf = np.fft.rfft(h)
            nz = (rng.randn(len(hf)) + 1j * rng.randn(len(hf))) * 0.012 * np.sqrt(ratio + 0.1)
            nz[0] = 0
            h = np.fft.irfft(hf + nz, n=n)
            h = _proj_to_sum(h, target)

        new_c5 = _eval(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
        else:
            h = h_save

        T = max(T * cooling, T_min)

    return best_h, best_c5

def _local_polish(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = _eval(h)
    while time.time() < t_end:
        c5_now = _eval(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
        i = rng.randint(n)
        j = (i + rng.randint(1, 6)) % n
        d = 1.5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n >= c5_now:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov (35s)
    for alpha in [80000, 400000]:
        t_e = min(time.time() + 17, t0 + 35)
        h = _nesterov(h, alpha, 250, 0.003, t_e)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Hot SA exploration (35% budget)
    t_e2 = min(time.time() + int(budget_s * 0.32), deadline - 150)
    if time.time() < t_e2:
        h = best_h.copy()
        h, c5v = _sa_search(h, n, t_e2, rng, 0.02, 0.999, 10000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 3: Nesterov refinement (50s)
    if time.time() < deadline - 50:
        h = best_h.copy()
        t_e3 = min(time.time() + 50, deadline - 50)
        for alpha in [300000, 1000000]:
            h = _nesterov(h, alpha, 200, 0.002, t_e3)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5, best_h = c5v, h.copy()

    # Phase 4: Cold SA exploitation (20% budget)
    if time.time() < deadline - 40:
        t_e4 = min(time.time() + int(budget_s * 0.2), deadline - 40)
        h = best_h.copy()
        h, c5v = _sa_search(h, n, t_e4, rng, 0.004, 0.9995, 12000)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 5: Local polish (remaining)
    if time.time() < deadline - 5:
        h = best_h.copy()
        h, c5v = _local_polish(h, n, deadline - 3, rng)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
