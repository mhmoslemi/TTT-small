    from scipy.optimize import linprog
    dx = 2.0 / n
    c = _corr(h)
    t0 = c[active_lags].max()
    
    # Variables: d[0..n-1], t
    # Minimize t (coefficient: 0 for d, 1 for t)
    # Constraints: c_k + grad_k · d <= t for each active k
    #              0 <= h_i + d_i <= 1  =>  -h_i <= d_i <= 1-h_i
    #              sum(d) = 0

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, n):
    dx = 2.0 / n
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        if m > 0:
            g[:m] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    dx = 2.0 / n
    def _obj(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _grad(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        gg = np.zeros(n)
        for k in np.where(w > 1e-12)[0]:
            sh = int(k) - (n - 1)
            wk = w[int(k)] * dx
            if sh >= 0:
                m2 = n - sh
                if m2 > 0:
                    gg[:m2] += wk * (1.0 - hh[sh:])
                    if sh > 0:
                        gg[sh:] -= wk * hh[:m2]
            else:
                sp = -sh
                gg[sp:] += wk * (1.0 - hh[:n - sp])
                gg[:n - sp] -= wk * hh[sp:]
        return gg
    f = _obj(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad(h)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj(h)
    return h

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    gr = (np.sqrt(5) - 1) / 2
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = _corr(h)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_grad_lag(h, int(k), n) for k in top_idx])
        a_vals = c[top_idx]
        improved = False
        for _ in range(n):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 6))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(28):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 50:
            h += 2e-5 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _bit_flip_search(h, n, duration_s, rng, t_end):
    """Local search on binary-valued structure, then relax."""
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        flipped = False
        for _ in range(200):
            if time.time() > min(end_t, t_end - 5):
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5n = _eval(h)
            if c5n < best_c5 - 1e-12:
                best_c5 = c5n
                no_improve = 0
                flipped = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
            else:
                h[i], h[j] = 1.0, 0.0
        no_improve += 1
        if no_improve > 30:
            for _ in range(5):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0.0, 1.0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            no_improve = 0
            best_c5 = min(best_c5, _eval(h))
    h = _repair(h, n)
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c
