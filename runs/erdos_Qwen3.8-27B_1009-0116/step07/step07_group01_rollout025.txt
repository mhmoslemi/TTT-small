    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _softmax_top(c, topk, alpha):
    """Compute soft-max of top 'topk' values with temperature 1/alpha"""
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    ct = ct - ct.max()
    e = np.exp(alpha * ct)
    return (ct.max() + np.log(e.sum()) / alpha)

def _grad_correlate(h, n, dx, active_lags):

    for k in active_lags:

        if shift >= 0:

The exact optimal d: we need to minimize max_k(c[k] + d * a_k) over d ∈ [d_min, d_max].

Since this is convex and piecewise linear, the minimum is at a "kink" point. But finding all kink points is expensive. Instead, I'll use bisection:

At the optimal d*, the subgradient contains 0, meaning there exist active k, k' with:
- c[k] + d* * a_k = c[k'] + d* * a_k' = max
- a_k ≤ 0 and a_k' ≥ 0 (or one of them is 0)

Equivalently: at d*, among the k achieving the max, some have a_k > 0 and some have a_k < 0.

So I can binary search on d:
- At d=0, compute which k are active and their a_k values
- If all active a_k > 0: the max is decreasing, so d* > 0. Increase d.
- If all active a_k < 0: the max is increasing, so d* < 0. Decrease d.
- If mixed: we're at (or near) the optimum.

But "compute which k are active" requires O(n) work, and bisection takes O(log(1/eps)) iterations. Total: O(n * 20) per pair.

For n=560 and 560*10 = 5600 pairs per pass: 5600 * 560 * 20 = 62.7M operations. In numpy with vectorized ops, this should be fast. In Python loops, maybe 30-60 seconds per pass.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smx_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smx_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sf(x, n):
    return x.sum() - n * 0.5

def _sj(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n,)}]
    try:
        res = minimize(_smx_obj, h, jac=_smx_grad, args=(n, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _two_point_sys(h, n, max_gap, stop_check):
    dx = 2.0 / n
    improved = 0
    c_cur = _eval(h)
    for gap in range(1, max_gap + 1):
        for i in range(n):
            if stop_check():
                return h, c_cur, improved
            j = (i + gap) % n
            hi, hj = h[i], h[j]
            d_lo = -hj
            d_hi = 1.0 - hi
            if d_hi - d_lo < 1e-12:
                continue
            best_d = 0.0
            best_c = c_cur
            ntr = 7
            for t in range(ntr):
                d = d_lo + (t + 0.5) * (d_hi - d_lo) / ntr
                h[i] = hi + d
                h[j] = hj - d
                cn = _eval(h)
                if cn < best_c:
                    best_c = cn
                    best_d = d
            h[i] = hi + best_d
            h[j] = hj - best_d
            if best_c < c_cur - 1e-15:
                c_cur = best_c
                improved += 1
            else:
                h[i] = hi
                h[j] = hj
    return h, c_cur, improved

def _random_local(h, n, n_iter, rng, base_step):
    c_cur = _eval(h)
    best_c = c_cur
    best_h = h.copy()
    for _ in range(n_iter):
        if c_cur < best_c - 1e-15:
            best_c = c_cur
            best_h = h.copy()
        i = rng.randint(n)
        r = rng.rand()
        if r < 0.6:
            j = (i + 1 + rng.randint(0, 4)) % n
            d = base_step * rng.randn()
            hi, hj = h[i], h[j]
            h[i] = hi + d
            h[j] = hj - d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _eval(h)
                if cn < c_cur:
                    c_cur = cn
                else:
                    h[i] = hi
                    h[j] = hj
            else:
                h[i] = hi
                h[j] = hj
        else:
            j = (i + 1 + rng.randint(0, 6)) % n
            k = (i + 7 + rng.randint(0, 10)) % n
            if k == i or k == j:
                continue
            d1 = base_step * rng.randn()
            d2 = base_step * rng.randn()
            hi, hj, hk = h[i], h[j], h[k]
            h[i] = hi + d1
            h[j] = hj + d2
            h[k] = hk - d1 - d2
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1 and 0 <= h[k] <= 1:
                cn = _eval(h)
                if cn < c_cur:
                    c_cur = cn
                else:
                    h[i], h[j], h[k] = hi, hj, hk
            else:
                h[i], h[j], h[k] = hi, hj, hk
    return best_h, best_c

def _proj(h, n):
    lo, hi = -5.0, 5.0
    target = n * 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, n, topk, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(topk):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g, c
