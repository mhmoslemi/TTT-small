import time

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        sh = (n - 1) - k
        wk *= dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g, m

def _nesterov_pg(h, n_steps, alpha0, alpha1, step0, mu0, mu1, mu_t):
    n = len(h)
    target = n / 2.0
    v = np.zeros(n)
    cur_c = _eval(h)
    step = step0
    for t in range(n_steps):
        frac = t / max(n_steps - 1, 1)
        alpha = alpha0 + (alpha1 - alpha0) * frac
        mu = mu0 + (mu1 - mu0) * min(frac / mu_t, 1.0)
        g, _ = _grad_smooth(h, alpha)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, target)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 8.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 8.0)
            if step < step0 * 0.01:
                step = step0
                v *= 0.5
    return _repair(h, n), cur_c

    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c

def _grad_coord(h, n_iter, rng, alpha):

    target = n / 2.0
    cur_c = _eval(h)
    best_h = h.copy()

        order = np.argsort(g)
        for j in order[:8]:

            for scale in [1e-3, 5e-4, 2e-4, 1e-4]:

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    half_n = n * 0.5
    def _ob(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _oj(x):
        return _grad_smooth(x, alpha)[0]
    def _es(x):
        return x.sum() - half_n
    def _ej(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _es, 'jac': _ej}]
    try:
        res = minimize(_ob, h, jac=_oj, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 7: Intensive random coordinate search
    if _el() < budget_s - 10:
        n_it = min(int((budget_s - 10 - _el()) / 0.001), 500000)
        h_new, c5n = _coord_desc(best_h.copy(), n_it, rng)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 8: Tiny 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 2 - _el()) / 0.0008), 300000)
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 4)) % nc2
            d = 1e-6 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
