import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _sum_con(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _soft_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_step(h, alpha, maxiter, tlim):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_soft_obj, h.copy(), method='SLSQP', args=(n, alpha),
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() < tlim:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _sa(h, rng, n_it, t0, t1, tlim, multi=False):
    n = len(h)
    cur = _eval(h)
    bc = cur
    bh = h.copy()
    for it in range(n_it):
        if time.time() > tlim:
            break
        t = t0 * (t1 / t0) ** (it / max(n_it - 1, 1))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        old = h.copy()
        r = rng.random()
        if not multi or r < 0.55:
            oi = ones[rng.randint(len(ones))]
            zi = zeros[rng.randint(len(zeros))]
            h[oi], h[zi] = 0.0, 1.0
        elif r < 0.85 and len(ones) >= 2 and len(zeros) >= 2:
            os_ = rng.choice(ones, 2, replace=False)
            zs_ = rng.choice(zeros, 2, replace=False)
            h[os_] = 0.0; h[zs_] = 1.0
        elif len(ones) >= 3 and len(zeros) >= 3:
            os_ = rng.choice(ones, 3, replace=False)
            zs_ = rng.choice(zeros, 3, replace=False)
            h[os_] = 0.0; h[zs_] = 1.0
        nc = _eval(h)
        if nc < cur or rng.random() < np.exp(-(nc - cur) / max(t, 1e-15)):
            cur = nc
            if nc < bc:
                bc = nc
                bh = h.copy()
        else:
            h[:] = old
    return bh, bc

def _kick(h, rng, k):
    n = len(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    return h

def _polish(h, rng, n_it, tlim, step):
    n = len(h)
    cur = _eval(h)
    bc = cur
    bh = h.copy()
    for _ in range(n_it):
        if time.time() > tlim:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 < ni < 1 and 0 < nj < 1:
            h[i], h[j] = ni, nj
            nc = _eval(h)
            if nc < cur:
                cur = nc
                if nc < bc:
                    bc = nc
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
