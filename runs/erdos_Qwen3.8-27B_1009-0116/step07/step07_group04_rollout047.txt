import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _nesterov(h, k_top, steps, step0, t0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    for _ in range(steps):
        if time.time() - t0 > t_end:
            break
        c = _corr(h)
        top = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = (2.0 / n) / k_top
        for kk in top:
            lag = kk - (n - 1)
            if lag >= 0:
                i1 = np.arange(n - lag)
                g[i1] += wk * (1.0 - h[i1 + lag])
                if lag > 0:
                    i2 = np.arange(lag, n)
                    g[i2] -= wk * h[i2 - lag]
            else:
                sp = -lag
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        hp = h.copy()
        v = mu * v + (1 - mu) * g
        hn = _proj_sum(h - step * v, n * 0.5)
        fn = _eval(hn)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.04, step0 * 6)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_sum(hp - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _coord_sweep(h, t0, t_end, max_sweeps=500, proj_every=3):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    for sw in range(max_sweeps):
        if time.time() - t0 > t_end:
            break
        for i in range(n):
            old = h[i]
            if old < 1e-14 or old > 1.0 - 1e-14:
                continue
            k = np.arange(-(n - 1), n)
            idx1 = i - k
            idx2 = i + k
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            h1 = h[np.where(v1, idx1, 0)]
            h2 = h[np.where(v2, idx2, 0)]
            s = (1.0 - h1) * v1.astype(float) - h2 * v2.astype(float)
            ims = np.argmax(s)
            imn = np.argmin(s)
            if abs(s[ims] - s[imn]) < 1e-14:
                ts = old
            else:
                ts = old + (c[imn] - c[ims]) / (s[ims] - s[imn])
                ts = max(0.0, min(1.0, ts))
            cands = (ts, 0.0, 1.0, old)
            bt, bv = old, np.max(c + (old - old) * s)
            for tc in cands:

                if val < bv - 1e-17:
                    bv, bt = val, tc

                c += (bt - old) * s
                h[i] = bt
        if (sw + 1) % proj_every == 0:
            h = _proj_sum(h, n * 0.5)
            c = _corr(h)
    h = _repair(h, n)
    return h, c, float(c.max())

def _slsqp_obj(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        if w[kk] < 1e-15:
            continue
        lag = kk - (n - 1)
        wk2 = w[kk] * 2.0 / n
        if lag >= 0:
            i1 = np.arange(n - lag)
            g[i1] += wk2 * (1.0 - x[i1 + lag])
            if lag > 0:
                i2 = np.arange(lag, n)
                g[i2] -= wk2 * x[i2 - lag]
        else:
            sp = -lag
            i1 = np.arange(sp, n)
            g[i1] += wk2 * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk2 * x[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, t0, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return x.sum() - n * 0.5
    def cons_jac(x):
        return np.ones(n)
    def obj(x):
        return _slsqp_obj(x, alpha, n)
    def grad(x):
        return _slsqp_grad(x, alpha, n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'iprint': 0})
        if time.time() - t0 > t_end:
            return _repair(h, n)
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_2pt(h, n_tries, rng, t0, t_end):

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c

        i = rng.randint(n)

        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-18:

            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.8, 1e-7)
    h = _repair(h, n)
