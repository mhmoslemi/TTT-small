import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax(h, alpha, topk):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:][:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            if n - sh > 0:
                g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            if n - sp > 0:
                g[:n-sp] -= wk * h[sp:]
    return g

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax_obj(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    tk = min(topk, len(c))
    idx = np.argpartition(c, -tk)[-tk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    tk = min(topk, len(c))
    idx = np.argpartition(c, -tk)[-tk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = dx * w[ii]
        if sh >= 0:
            if n - sh > 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            if n - sp > 0:
                g[sp:] += wk * (1.0 - h[:-sp])
            if n - sp > 0:
                g[:n - sp] -= wk * h[sp:]
    return g

def _nest(h, alpha, topk, nsteps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    best_h = h.copy()
    best_f = f
    for _ in range(nsteps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            if f < best_f:
                best_f = f
                best_h = h.copy()
            step = min(step * 1.04, step0 * 10.0)
        else:
            step *= 0.4
            if step < step0 * 1e-4:
                step = step0 * 0.5
                v *= 0.5
    return _repair(best_h, n)

def _perturb(h, n, rng, style):
    hp = h.copy()
    if style == 0:
        hp += 0.018 * rng.randn(n)
    elif style == 1:
        nb = rng.randint(3, 9)
        bs = max(1, n // nb)
        for b in range(nb):
            hp[b*bs:min((b+1)*bs, n)] += 0.012 * rng.randn()
    elif style == 2:
        f = rng.uniform(0.5, 10.0)
        ph = rng.uniform(0, 2 * np.pi)
        hp += 0.012 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif style == 3:
        i0 = rng.randint(0, max(1, n - 25))
        bl = rng.randint(8, 25)
        hp[i0:i0+bl] = 1.0 - hp[i0:i0+bl]
    else:
        i0 = rng.randint(0, max(1, n - 35))
        bl = rng.randint(15, 35)
        seg = hp[i0:i0+bl].copy()
        hp[i0:i0+bl] = seg[rng.permutation(bl)]
    return _repair(hp, n)

def _coord_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 8)) % n
        d = 1.2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _trio_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 5)) % n
        k = (i + rng.randint(6, 12)) % n
        d1 = 7e-6 * rng.randn()
        d2 = 7e-6 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return bh, bc

def _sine_sweep(h, n, rng, deadline):
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(60):
        if time.time() > deadline:
            break
        f = rng.uniform(0.3, 12.0)
        ph = rng.uniform(0, 2 * np.pi)
        eps = rng.uniform(0.001, 0.008)
        hp = h + eps * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        hp = _repair(hp, n)
        c5n = _eval(hp)
        if c5n < bc:
            bc = c5n
            bh = hp.copy()
            h = hp.copy()
    return bh, bc
