import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-9:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, top_k_indices, weight=None):
    n = len(h)
    g = np.zeros(n)
    n_top = len(top_k_indices)
    if weight is None:
        weight = np.ones(n_top) / n_top
    for idx, w in enumerate(weight):

            if sh > 0:

        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g

def _nesterov_adaptive(h, k_sched, alpha_sched, n_steps, step0, t0, t_end):
    """Nesterov with scheduled k and step size."""
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    it = 0
    n_k = len(k_sched)
    n_a = len(alpha_sched)
    step_per_k = max(n_steps // n_k, 100)
    for k_phase in range(n_k):
        k_top = k_sched[k_phase]
        steps_this = step_per_k
        for _s in range(steps_this):
            if time.time() > t_end:
                break
            it += 1
            # Compute correlation and find top-k
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top_idx = np.argsort(c)[-k_top:]
            top_idx = np.sort(top_idx)
            g = _grad_topk(h, top_idx)
            h_prev = h.copy()
            v = mu * v + (1.0 - mu) * g
            h_new = _proj_sum(h - step * v, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f - 1e-15:
                h = h_new
                f = f_new
                step = min(step * 1.02, step0 * 20.0)
            else:
                step *= 0.7
                if step < 1e-10:
                    break
                h = _proj_sum(h_prev - step * v, n * 0.5)
                f = _eval(h)
    return h, f

    target = n * 0.5
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        smax = s.max()
        return m + np.log(np.sum(np.exp(s - smax))) / alpha

        m = c.max()

        return x.sum() - target

        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-9:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, top_idx):
    n = len(h)
    g = np.zeros(n)
    w = 1.0 / len(top_idx)
    for k in top_idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g

def _nesterov(h, k_list, steps_total, step0, t0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0

        for _s in range(per_k):
            if time.time() > t_end:
                break
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top = np.sort(np.argsort(c)[-kt:])
            g = _grad_topk(h, top)
            mu = 0.82
            h_old = h.copy()
            v = mu * v + (1.0 - mu) * g
            h_new = _proj_sum(h - step * v, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f - 1e-16:
                h = h_new; f = f_new
                step = min(step * 1.025, step0 * 30.0)
            else:
                step *= 0.6
                if step < 1e-11:
                    break
                h = _proj_sum(h_old - step * v, n * 0.5)
                f = _eval(h)
    return h, f

def _slsqp(h, alpha, maxiter):
    n = len(h)
    target = n * 0.5
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s.max()
        return m + np.log(np.sum(np.exp(s - sm))) / alpha
    def grd(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s.max()
        w = np.exp(s - sm); w /= w.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * (2.0 / n)
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:n])
                if sh > 0:
                    g[sh:] -= wk * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    def cf(x): return x.sum() - target
    def cj(x): return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, jac=grd, method='SLSQP', bounds=bnds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

        best_h, best_c5 = _loc_search(best_h, ntr, rng, t0,
                                      time.time() + (budget_s - 15 - (time.time() - t0)), 1, 8, 5e-5)

    # Phase 9: Ultra-fine 2-point (span 1-4, step 2e-6)
    if time.time() - t0 < budget_s - 3:
        ntr = min(int((budget_s - 3 - (time.time() - t0)) * 250), 200000)
        best_h, best_c5 = _loc_search(best_h, ntr, rng, t0,
                                      time.time() + (budget_s - 2 - (time.time() - t0)), 1, 4, 2e-6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
