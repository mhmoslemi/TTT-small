import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Binary search for the right shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)

    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)

def _proj(h, n):
    """Project to simplex-like constraint: 0<=h<=1, sum=n/2"""
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):

def _lbfgs(h, obj_fn, grad_fn, arg, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 10:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=(arg,),
                       method='L-BFGS-B', bounds=bnds,

def _raised_cosine(n, width_frac=1.0, center=0.5):
    """Raised cosine bump"""
    x = np.linspace(0, 2, n)
    h = 0.5 * (1 + np.cos(2 * np.pi * (x - center) / (2 * width_frac)))
    h[x > center + width_frac] = 0
    h[x < center - width_frac] = 0
    return _repair(h, n)

def _trapezoid(n, base=1.2, top=0.6, center=1.0):
    """Trapezoid"""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    half_top = top / 2
    half_base = base / 2
    # Rise
    mask = (x >= center - half_base) & (x < center - half_top)
    h[mask] = (x[mask] - (center - half_base)) / (base - top)
    # Top
    mask = (x >= center - half_top) & (x < center + half_top)
    h[mask] = 1.0
    # Fall
    mask = (x >= center + half_top) & (x < center + half_base)
    h[mask] = ((center + half_base) - x[mask]) / (base - top)
    return _repair(h, n)

def _smooth_bump(n, sigma=0.5, center=1.0):
    """Gaussian-like bump (clipped)"""
    x = np.linspace(0, 2, n)
    h = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    h /= h.max()
    return _repair(h, n)

    h /= h.max()
    return _repair(h, n)

    h = np.clip(h, 0.0, 1.0).copy()

        return h
