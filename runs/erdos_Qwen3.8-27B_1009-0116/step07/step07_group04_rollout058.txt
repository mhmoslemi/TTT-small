def _local_search(h, n_tries, rng, t0, budget_s, step_scale=5e-5):

g = sum_k w[k] * grad_c[k]

where grad_c[k] is the gradient of the k-th correlation component.

- grad_c[k][j] = (1-h[j-s]) for j >= s (if s >= 0) or j >= -s (if s < 0)

grad_c[k] = (1-h[shifted]) - h[shifted_back]

g[j] = sum_k w[k] * [(1-h[(j-s_k) mod structure]) - h[(j+s_k) mod structure]]

This is essentially a weighted sum of shifted versions of (1-h) and h, which can be done with a single convolution-like operation if the weights are structured. But since w[k] is arbitrary (from softmax), we can't simplify it further.

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_grad(h, alpha, n, wthresh=1e-7):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()

def _nesterov(h, alpha, steps, step0, t0, t_end, mu=0.85, step_cap=10.0):

    f = _eval(h)
    v = np.zeros(n)
    step = step0
    target = n * 0.5

        if time.time() - t0 > t_end:
            break
        g = _smooth_grad(h, alpha, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h_prev - step * v, target)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * step_cap)
        else:
            step *= 0.45
            if step < 1e-11:
                break
            h = _proj_to_sum(h_prev - step * v, target)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_opt(h, alpha, maxiter, ftol):
