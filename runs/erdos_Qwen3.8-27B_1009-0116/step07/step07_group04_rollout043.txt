def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_one(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh + m])
        g[sh:] -= h[:m]
    else:
        p = -sh
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr_vec(h)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    top = np.where(w > 1e-10)[0]
    g = np.zeros(n)
    for k in top:
        g += w[k] * _grad_one(h, k)
    return g, float(m + np.log(w.sum()) / alpha)

            best_c5, best_h = c5v, h_cd.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
