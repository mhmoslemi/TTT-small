import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_one(h, k, n):
    dx = 2.0 / n
    kl = k - (n - 1)
    lo = max(0, kl)
    hi = min(n - 1, n - 1 + kl)
    g = np.zeros(n)
    if lo > hi:
        return g
    g[lo:hi+1] += 1.0 - h[lo-kl:hi+1-kl]
    jl = max(0, lo - kl)
    jh = min(n - 1, hi - kl)
    if jl <= jh:
        g[jl:jh+1] -= h[jl+kl:jh+1+kl]
    return g * dx

def _grad_max(h, n, topk=20, gap=0.002):
    c = _corr(h)
    m = c.max()
    idx = np.where(c >= m - gap)[0]
    if len(idx) > topk:
        idx = np.argsort(c)[-topk:]
    g = np.zeros(n)
    for k in idx:
        g += _grad_one(h, k, n)
    g /= len(idx)
    return g, m

def _nesterov(h, n, iters, deadline, lr0=0.01, beta=0.9, topk=20, gap=0.002):
    c5 = _eval(h)
    v = np.zeros(n)
    lr = lr0
    best_h = h.copy()
    best_c = c5
    stale = 0
    for it in range(iters):
        if time.time() > deadline:
            break
        g, m = _grad_max(h, n, topk, gap)
        v = beta * v + (1 - beta) * g
        hn = _proj(h - lr * v, n * 0.5)
        c5n = _eval(hn)
        if c5n < c5:
            h = hn; c5 = c5n
            lr = min(lr * 1.02, lr0 * 10)
            stale = 0
            if c5 < best_c:
                best_c = c5; best_h = h.copy()
        else:
            lr *= 0.7
            stale += 1
            if stale > 500:
                # Perturb and restart
                hp = h + 0.01 * np.random.randn(n)
                h = _proj(hp, n * 0.5)
                c5 = _eval(h)
                v = np.zeros(n)
                lr = lr0
                stale = 0
    return best_h, best_c

def _pair_search(h, n, iters, rng, deadline, base_step=0.005):
    c5 = _eval(h)
    best = h.copy()
    best_c = c5
    step = base_step
    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * rng.uniform(0.5, 1.5) * (1.0 if rng.random() < 0.5 else -1.0)
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:

                step = min(step * 1.01, base_step * 5)
                if c5 < best_c:
                    best_c = c5; best = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
                step *= 0.95
    return best, best_c

    return best, best_c
