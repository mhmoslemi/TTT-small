import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _project_h(h, n):
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, n, lags):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g * dx / len(lags)

def _smooth_cg(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sc = np.clip(alpha * (c - m), -300, 0)
    obj = m + np.log(np.sum(np.exp(sc))) / alpha
    w = np.exp(sc) / np.sum(np.exp(sc))
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-9:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return obj, g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _slsqp_obj(x, n, alpha):
    return _smooth_cg(x, n, alpha)[0]

def _slsqp_jac(x, n, alpha):
    return _smooth_cg(x, n, alpha)[1]

def _proj_grad(h, n, a_s, a_e, iters, lr0, t0, budget_s):
    steps = [lr0, lr0 * 0.5, lr0 * 0.25, lr0 * 0.125]
    c5c = _eval(h)
    stall = 0
    for it in range(iters):
        if time.time() - t0 > budget_s - 30:
            break
        frac = it / max(1, iters - 1)
        alpha = a_s * (a_e / a_s) ** frac
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        sc = np.clip(alpha * (c - c.max()), -300, 0)
        w = np.exp(sc)
        wsum = w.sum()
        w /= wsum
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-8:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[sh:] += wk * (1.0 - h[:n - sh])
                if sh > 0:
                    g[:n - sh] -= wk * h[sh:]
            else:
                sp = -sh
                g[:n - sp] += wk * (1.0 - h[sp:])
                g[sp:] -= wk * h[:n - sp]
        improved = False
        for si in range(len(steps)):
            h_new = _project_h(h - steps[si] * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.5] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _topk_desc(h, n, lags, iters, steps_in, t0, budget_s):
    c5c = _eval(h)
    steps = list(steps_in)
    stall = 0
    for _ in range(iters):
        if time.time() - t0 > budget_s - 30:
            break
        g = _topk_grad(h, n, lags)
        improved = False
        for si in range(len(steps)):
            h_new = _project_h(h - steps[si] * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.4] + [s for s in steps if s < steps[si] * 0.5]
                if len(steps) > 10:
                    steps = steps[:10]
                break
        if not improved:
            steps = [s * 0.4 for s in steps]
            stall += 1
            if stall > 10:

        else:
            stall = 0
    return h, c5c

def _coord_desc(h, n, ntries, rng, c5c, t0, budget_s):
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntries):
        if time.time() - t0 > budget_s - 10:
            break
        r = rng.random()
        if r < 0.55:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = rng.choice([-1.0, 1.0]) * rng.uniform(1e-5, 3e-4)
            hi, hj = h[i] + d, h[j] - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                c5n = _eval(h)
                if c5n < best_c - 1e-15:
                    best_c = c5n
                    best_h = h.copy()
                else:
                    h[i], h[j] = hi - d, hj + d
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            k = (i + 4 + rng.randint(0, 3)) % n
            d1 = rng.choice([-1.0, 1.0]) * rng.uniform(1e-5, 2e-4)
            d2 = rng.choice([-1.0, 1.0]) * rng.uniform(1e-5, 2e-4)
            hi, hj, hk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0 and 0.0 <= hk <= 1.0:
                h[i], h[j], h[k] = hi, hj, hk
                c5n = _eval(h)
                if c5n < best_c - 1e-15:
                    best_c = c5n
                    best_h = h.copy()
                else:
                    h[i], h[j], h[k] = hi - d1, hj - d2, hk + d1 + d2
    return best_h, best_c

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
