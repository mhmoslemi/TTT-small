            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr
        except Exception:
            pass

    # Phase 2: Nesterov momentum on smooth-max
    if _elapsed() < budget_s - 250:
        for (a_s, a_e, nit, mu, lr) in [(50, 500, 200, 0.9, 0.05),
                                         (500, 5000, 150, 0.85, 0.03),
                                         (5000, 50000, 100, 0.8, 0.01)]:
            if _bc():
                break
            h2, c52 = _nesterov_smooth(best_h, n, a_s, a_e, nit, mu, lr, _bc)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2

    # Phase 3: Multi-k projected gradient descent
    for kt in [3, 5, 8, 4, 3, 6, 5, 3]:
        if _bc():
            break
        steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]
        h2, c52 = _proj_gd(best_h, n, kt, steps, 350, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: Resolution exploration
    if _elapsed() < budget_s - 350:
        for n_try in [480, 640, 800]:
            if _bc():
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
            h_interp = _repair(h_interp, n_try)
            h2, c52 = _proj_gd(h_interp, n_try, 5, [0.03, 0.01, 0.005, 0.002, 0.001], 150, _bc)
            if c52 < best_c5 - 1e-5:
                best_c5 = c52
                best_h = h2
                n = n_try
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]

    # Phase 5: Second SLSQP round (if resolution changed or after gradient)
    if _elapsed() < budget_s - 200:
        for alpha in [20000, 100000]:
            if _bc():
                break
            try:
                res = minimize(_smooth_obj, best_h, args=(n, alpha), jac=_smooth_grad,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-16})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 6: Fine gradient polish
    for kt in [3, 4, 3, 5, 3]:
        if _bc():
            break
        steps = [0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4]
        h2, c52 = _proj_gd(best_h, n, kt, steps, 300, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: 3-point local search
    if _elapsed() < budget_s - 15:
        nt = min(int((budget_s - 15 - _elapsed()) / 0.003), 180000)
        best_h, best_c5 = _local_3pt(best_h, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
