    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_con(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _local_search(h, ntry, rng, npts, scale, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(ntry):
        if check_fn():
            break
        i0 = rng.randint(nc)
        idxs = []
        for _ in range(npts):
            j = (i0 + rng.randint(1, 12)) % nc
            if j not in idxs:
                idxs.append(j)
        if len(idxs) < 2:
            continue
        ds = rng.randn(len(idxs)) * scale
        ds -= ds.mean()
        old = h[idxs].copy()
        nv = old + ds
        if np.all(nv >= -1e-9) and np.all(nv <= 1.0 + 1e-9):
            h[idxs] = np.clip(nv, 0, 1)
            c5n = _eval(h)
            if c5n < best_c - 1e-18:
                best_c = c5n
                best_h = h.copy()
            else:
                h[idxs] = old
    return best_h, best_c

def _block_local(h, n, ntry, rng, check_fn):
    nc = n
    best_h = h.copy()
    best_c = _eval(h)
    bs = 5
    for _ in range(ntry):
        if check_fn():
            break
        start = rng.randint(0, nc - bs + 1)
        idxs = list(range(start, start + bs))
        old = h[idxs].copy()
        ds = rng.randn(bs) * 3e-6
        ds -= ds.mean()
        nv = old + ds
        if np.all(nv >= 0) and np.all(nv <= 1):
            h[idxs] = nv
            c5n = _eval(h)
            if c5n < best_c - 1e-18:
                best_c = c5n
                best_h = h.copy()
            else:
                h[idxs] = old
                ds2 = rng.randn(bs) * 3e-6
                ds2 -= ds2.mean()
                nv2 = old + ds2
                if np.all(nv2 >= 0) and np.all(nv2 <= 1):
                    h[idxs] = nv2
                    c5n = _eval(h)
                    if c5n < best_c - 1e-18:
                        best_c = c5n
                        best_h = h.copy()
                    else:
                        h[idxs] = old
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 25

    n0 = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float).copy()
    h = _repair(h, n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Momentum GD with cycling k_top
    for kt in [3, 5, 3, 7, 5, 3, 4]:
        if _bc():
            break
        h2, c52 = _gd_mom(best_h.copy(), n0, 600, kt, 0.06, 0.75, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: SLSQP multi-temperature
    bnds = [(0.0, 1.0)] * n0
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n0,)}]
    for a_val in [150, 2000, 20000]:
        if _bc():
            break
        if _el() > budget_s - 120:
            break
        try:
            res = minimize(_soft_obj, best_h, args=(n0, a_val),
                           jac=_soft_grad, method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-17})
            hr = _repair(res.x, n0)
            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr
        except Exception:
            pass

    # Phase 3: Finer GD
    for kt in [3, 5, 3, 4]:
        if _bc():
            break
        h2, c52 = _gd_mom(best_h.copy(), n0, 800, kt, 0.025, 0.8, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: SLSQP cold
    if _el() < budget_s - 100:
        for a_val in [50000, 100000]:
            if _bc():
                break
            try:
                res = minimize(_soft_obj, best_h, args=(n0, a_val),
                               jac=_soft_grad, method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-18})
                hr = _repair(res.x, n0)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 5: Block local search (5-point)
    if _el() < budget_s - 40:
        rem = budget_s - 40 - _el()
        nt = min(int(rem / 0.006), 80000)
        best_h, best_c5 = _block_local(best_h, n0, nt, rng, _bc)

    # Phase 6: Multi-point local search (3-point)
    if _el() < budget_s - 30:
        rem = budget_s - 30 - _el()
        nt = min(int(rem / 0.004), 120000)
        best_h, best_c5 = _local_search(best_h, nt, rng, 3, 4e-6, _bc)

    # Phase 7: Ultra-fine 2-point polish
    if _el() < budget_s - 5:
        rem = budget_s - 5 - _el()
        nt = min(int(rem / 0.003), 200000)
        best_h, best_c5 = _local_search(best_h, nt, rng, 2, 1.5e-6, _bc)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
