import numpy as np
from scipy.optimize import minimize
import time

def _fourier_h(theta, n, M):
    """Reconstruct h from Fourier coefficients."""
    x = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, M+1):
        h += theta[2*(k-1)] * np.cos(2*np.pi*k*x/n)
        h += theta[2*(k-1)+1] * np.sin(2*np.pi*k*x/n)
    return h

def _fourier_grad_map(grad_h, n, M):
    """Map gradient from h-space to theta-space."""
    x = np.arange(n)
    g = np.zeros(2*M)
    for k in range(1, M+1):
        g[2*(k-1)] = np.sum(grad_h * np.cos(2*np.pi*k*x/n))
        g[2*(k-1)+1] = np.sum(grad_h * np.sin(2*np.pi*k*x/n))
    return g

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _cvals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _smooth_obj(theta, n, M, alpha, penalty_w):
    h = _fourier_h(theta, n, M)
    # Penalty for box constraint
    pen = 0.0
    if h.min() < 0:
        pen += penalty_w * np.sum(np.maximum(-h, 0)**2)
    if h.max() > 1:
        pen += penalty_w * np.sum(np.maximum(h - 1, 0)**2)
    # Smooth max
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen

def _smooth_grad(theta, n, M, alpha, penalty_w):
    h = _fourier_h(theta, n, M)
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    # Gradient of smooth max w.r.t. h
    dx = 2.0 / n
    g_h = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g_h[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g_h[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g_h[sp:] += wk * (1.0 - h[:-sp])
            g_h[:-sp] -= wk * h[sp:]
    # Penalty gradient
    if h.min() < 0:

    if h.max() > 1:

    return _fourier_grad_map(g_h, n, M)

import numpy as np
from scipy.optimize import minimize
import time

def _fourier_h(theta, n, M):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, M+1):
        h += theta[2*(k-1)] * np.cos(2*np.pi*k*x/n)
        h += theta[2*(k-1)+1] * np.sin(2*np.pi*k*x/n)
    return h

def _fit_fourier(h, n, M):
    x = np.arange(n)
    A = np.zeros((n, 2*M))
    for k in range(1, M+1):
        A[:, 2*(k-1)] = np.cos(2*np.pi*k*x/n)
        A[:, 2*(k-1)+1] = np.sin(2*np.pi*k*x/n)
    # Least squares fit h - 0.5
    coeff, _, _, _ = np.linalg.lstsq(A, h - 0.5, rcond=None)
    return coeff

def _fourier_jacobian(n, M):
    """Returns the matrix J where h = 0.5 + J @ theta"""
    x = np.arange(n)
    J = np.zeros((n, 2*M))
    for k in range(1, M+1):
        J[:, 2*(k-1)] = np.cos(2*np.pi*k*x/n)
        J[:, 2*(k-1)+1] = np.sin(2*np.pi*k*x/n)
    return J

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _cvals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad_h(h, n, w):
    """Compute gradient of weighted sum of correlation values w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(theta, n, M, J, alpha, pw):
    h = 0.5 + J @ theta
    pen = 0.0
    neg = np.maximum(-h, 0)
    if neg.sum() > 0:
        pen += pw * (neg @ neg)
    pos = np.maximum(h - 1, 0)
    if pos.sum() > 0:
        pen += pw * (pos @ pos)
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha + pen)

def _smooth_grad(theta, n, M, J, alpha, pw):
    h = 0.5 + J @ theta
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g_h = _corr_grad_h(h, n, w)
    neg = np.maximum(-h, 0)
    if neg.sum() > 0:
        g_h -= 2 * pw * neg
    pos = np.maximum(h - 1, 0)
    if pos.sum() > 0:
        g_h += 2 * pw * pos
    return J.T @ g_h

def _top_lag(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval_c5(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 20:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        w = np.zeros(len(c))
        for k in top_k:
            w[k] = 1.0
        g = _corr_grad_h(h, n, w)
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5, 1e-5]:
            h_new = _repair(h - step * g, n)
            c5n = _eval_c5(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _subgrad_opt(h, steps, step0, tol, deadline):
    n = len(h)
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.02 * i)
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        g = _corr_grad_h(h, n, w)
        h = _repair(h - t * g, n)
    return h

def _local_2pt(h, n_iter, rng, deadline):

import numpy as np
from scipy.optimize import minimize
import time

def _fourier_h(theta, J):
    return 0.5 + J @ theta

def _fourier_J(n, M):
    x = np.arange(n)
    J = np.zeros((n, 2 * M))
    for k in range(1, M + 1):
        J[:, 2 * (k - 1)] = np.cos(2 * np.pi * k * x / n)
        J[:, 2 * (k - 1) + 1] = np.sin(2 * np.pi * k * x / n)
    return J

def _fit_fourier(h, J):
    coeff, _, _, _ = np.linalg.lstsq(J, h - 0.5, rcond=None)
    return coeff

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _corr_grad_h(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _fobj(theta, n, J, alpha, pw):
    h = _fourier_h(theta, J)
    pen = pw * (np.maximum(-h, 0) @ np.maximum(-h, 0) + np.maximum(h - 1, 0) @ np.maximum(h - 1, 0))
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha + pen)

def _fgrad(theta, n, J, alpha, pw):
    h = _fourier_h(theta, J)
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g_h = _corr_grad_h(h, n, w)
    g_h -= 2.0 * pw * np.maximum(-h, 0)
    g_h += 2.0 * pw * np.maximum(h - 1, 0)
    return J.T @ g_h

def _pnorm_obj(theta, n, J, p, pw):
    h = _fourier_h(theta, J)
    pen = pw * (np.maximum(-h, 0) @ np.maximum(-h, 0) + np.maximum(h - 1, 0) @ np.maximum(h - 1, 0))
    c = np.maximum(_cvals(h), 0.0)
    return float((np.sum(c ** p)) ** (1.0 / p) + pen)

def _pnorm_grad(theta, n, J, p, pw):

    c = np.maximum(_cvals(h), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        g_h = 2.0 * pw * (np.maximum(-h, 0) * -1 + np.maximum(h - 1, 0))
        return J.T @ g_h

    g_h -= 2.0 * pw * np.maximum(-h, 0)
    g_h += 2.0 * pw * np.maximum(h - 1, 0)
    return J.T @ g_h

            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new.copy()

    # Phase 4: Multi-start perturbation in Fourier space
    if time.time() < deadline - 80:
        M = 20
        J = _fourier_J(n0, M)
        theta_best = _fit_fourier(best_h, J)
        for _ in range(6):
            if time.time() > deadline - 80:
                break
            theta_try = theta_best + 0.005 * rng.randn(2 * M)
            res = minimize(_fobj, theta_try, jac=_fgrad, args=(n0, J, 50000, 100.0),
                          method='L-BFGS-B',
                          options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = _repair(_fourier_h(res.x, J), n0)
            c5v = _eval_c5(h_new)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_new.copy()

    # Phase 5: Final local search
    if time.time() < deadline - 30:
        h_tl, c5_tl = _top_lag(best_h.copy(), 3, 200, deadline)
        if c5_tl < best_c5:
            best_c5, best_h = c5_tl, h_tl.copy()
    if time.time() < deadline - 10:
        h_ls, c5_ls = _local_3pt(best_h, 15000, rng, deadline)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()
    if time.time() < deadline - 3:
        h_ls, c5_ls = _local_2pt(best_h, 60000, rng, deadline)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, n0)
