import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_topk(h, k):
    n = len(h)
    dx = 2.0 / n
    c = _eval_vec(h)
    top = np.argsort(c)[-k:]
    g = np.zeros(n)
    for kk in top:
        ms = kk - (n - 1)
        if ms >= 0:
            g[:n - ms] += (1.0 - h[ms:])
            if ms > 0:
                g[ms:] -= h[:n - ms]
        else:
            s = -ms
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    return g * dx / k

def _grad_softmax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _eval_vec(h)
    m = c.max()
    e = np.exp(np.clip(alpha * (c - m), -350, 0))
    w = e / e.sum()
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        if w[kk] < 1e-13:
            continue
        ms = kk - (n - 1)
        wk = w[kk]
        if ms >= 0:
            g[:n - ms] += wk * (1.0 - h[ms:])
            if ms > 0:
                g[ms:] -= wk * h[:n - ms]
        else:
            s = -ms
            g[s:] += wk * (1.0 - h[:n - s])
            g[:n - s] -= wk * h[s:]
    return g * dx

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_desc(h, n_iter, k_top, alpha, init_step, check_fn):
    n = len(h)
    target = n * 0.5
    best = h.copy()
    best_c = _eval(h)
    step = init_step
    stall = 0
    for _ in range(n_iter):
        if check_fn():
            break
        if alpha > 0:
            g = _grad_softmax(h, alpha)
        else:
            g = _grad_topk(h, k_top)
        h_new = _project(h - step * g, target)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-16:
            best = h_new.copy()
            best_c = c_new
            h = h_new
            step = min(step * 1.35, 0.5)
            stall = 0
        else:
            h_try = _project(h - step * 0.15 * g, target)
            c_try = _eval(h_try)
            if c_try < _eval(h) - 1e-16:
                h = h_try
                step *= 0.5
            else:
                step *= 0.3
                stall += 1
                if stall > 20:
                    break
    return best, best_c

def _sa_run(h, n_iter, T0, T1, rng, check_fn):
    n = len(h)
    best = h.copy()
    best_c = _eval(h)
    cur = h.copy()
    cur_c = best_c
    T = T0
    cool = (T1 / T0) ** (1.0 / max(n_iter, 1))
    for _ in range(n_iter):

            break
        T *= cool
        i = rng.randint(n)

        d = 0.08 * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            cur[i], cur[j] = ni, nj
            nc = _eval(cur)
            delta = nc - cur_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c = nc
                if nc < best_c:
                    best_c = nc
                    best = cur.copy()
            else:
                cur[i], cur[j] = ni - d, nj + d
    return best, best_c

def _sa_block(h, n_iter, T0, T1, rng, check_fn):
    n = len(h)
    best = h.copy()

            break
        T *= cool
        bs = 2 + rng.randint(0, 5)

        noise = rng.randn(bs) * 0.06

        old_sum = seg.sum()

        if new_sum > 1e-10:

            new_seg = np.clip(new_seg, 0, 1)
        cur[st:st + bs] = new_seg
        nc = _eval(cur)
        delta = nc - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c = nc
            if nc < best_c:
                best_c = nc
                best = cur.copy()
        else:
            cur[st:st + bs] = seg
    return best, best_c
