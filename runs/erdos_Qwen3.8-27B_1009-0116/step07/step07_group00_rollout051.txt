def _nesterov(h, alpha, topk, steps, step0, deadline):

def _adam(h, alpha, topk, nsteps, lr0, deadline):

        if time.time() > deadline: break
        g = _grad(h, alpha, topk)
        m_t = b1*m_t + (1-b1)*g
        v_t = b2*v_t + (1-b2)*g*g

        v_h = v_t/(1-b2**t)
        hn = _proj(h - lr0*m_h/(np.sqrt(v_h)+eps), n*0.5)

    h = np.clip(h, 0.0, 1.0).copy()

        if time.time() > dl5:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl5)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()
    h = _slsqp(best_h, 50000, 15, 300, dl5)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 6: Fine 2-point polish
    rem = deadline - 10 - time.time()
    if rem > 1:
        n_pol = min(int(rem * 0.6 / 0.001), 60000)
        best_h, best_c5 = _polish(best_h, n_pol, rng, deadline - 10, 2.5e-6)

    # Phase 7: 3-point search
    if time.time() < deadline - 8:
        rem2 = deadline - 8 - time.time()
        n_3 = min(int(rem2 * 0.7 / 0.0018), 30000)
        best_h, best_c5 = _polish3(best_h, n_3, rng, deadline - 3, 1.5e-6)

    # Phase 8: Ultra-fine 2-point
    if time.time() < deadline - 2:
        rem3 = deadline - 2 - time.time()
        n_pf = min(int(rem3 / 0.001), 20000)
        best_h, best_c5 = _polish(best_h, n_pf, rng, deadline - 1, 3e-7)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
