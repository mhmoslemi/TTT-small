            h, c5v = _coord_descent(h, n, budget_s * 0.04, top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5, best_h = c5v, h.copy()

    # Phase 6: Final local search
    if time.time() < t0 + budget_s - 4:
        remaining = (t0 + budget_s) - time.time() - 3
        n_tries = min(int(remaining / 0.003), 60000)
        if n_tries > 200:
            h = best_h.copy()
            best_h, best_c5 = _local_search(h, n_tries, rng, t0 + budget_s - 2)

    # Phase 7: Final coord polish
    if time.time() < t0 + budget_s - 2:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, max(1, budget_s - (time.time() - t0) - 2), 5, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
