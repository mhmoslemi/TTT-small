    cur_c = _eval(h)
    for _ in range(steps):
        g = _topk_grad(h, alpha, top_k)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 8.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 8.0)
    return _repair(h, n)

def _slsqp_smooth(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()

    def _obj(x, a=alpha, _mm=m):
        cc = np.correlate(x, 1.0 - x, mode='full') * (2.0 / len(x))
        mm = cc.max()
        s = np.clip(a * (cc - mm), -400, 400)
        s -= s.max()
        return mm + np.log(np.exp(s).sum()) / a

    def _objg(x, a=alpha):
        return _topk_grad(x, a, 2 * len(x) - 1)

    def _consf(x):
        return x.sum() - n * 0.5

    def _consg(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _consf, 'jac': _consg}]
    try:
        res = minimize(_obj, h, jac=_objg, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _sim_anneal(h, n_iter, rng, base_step, T0, T1):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter - 1, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 120)) % n
        d1 = base_step * rng.randn()
        d2 = base_step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5cur or (T > 1e-12 and rng.rand() < np.exp(-(c5n - c5cur) / T)):
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
    return best_h, best_c

def _downsample(h, target_n):
    n = len(h)
    f = n // target_n
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, multiplier):
    n = len(h)
    out = np.empty(n * multiplier)
    for i in range(n):
        out[i * multiplier:(i + 1) * multiplier] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution Adam
    for ratio in [8, 4, 2]:
        if _el() > 120:
            break
        nc = max(32, n // ratio)
        hc = _repair(_downsample(h, nc), nc)
        hc, _ = _adam_topk(hc, [2000, 10000, 50000], 200, 0.02, 8, 0.85, 0.999)
        h_up = _repair(_upsample(hc, n // nc), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Momentum top-k gradient (multiple configs)
    for alpha, steps, s0, tk, mu in [(2000, 800, 0.008, 5, 0.7),
                                      (8000, 600, 0.004, 8, 0.75),
                                      (30000, 500, 0.002, 12, 0.8),
                                      (100000, 400, 0.0015, 16, 0.8),
                                      (400000, 300, 0.001, 10, 0.85)]:
        if _el() > 350:
            break
        h_new = _momentum_topk(best_h.copy(), alpha, steps, s0, tk, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Adam at full resolution
    if _el() < budget_s - 300:
        for lr, tk in [(0.003, 8), (0.002, 12), (0.001, 16)]:
            if _el() > budget_s - 180:
                break
            h_ad, c5a = _adam_topk(best_h.copy(), [5000, 20000, 80000, 300000],
                                   300, lr, tk, 0.85, 0.999)
            if c5a < best_c5:
                best_c5 = c5a
                best_h = h_ad.copy()

    # Phase 4: SLSQP with smooth max
    if _el() < budget_s - 200:
        for alpha in [30000, 150000, 800000]:
            if _el() > budget_s - 120:
                break
            h_new = _slsqp_smooth(best_h.copy(), alpha, 400)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Short momentum pass after SLSQP
    if _el() < budget_s - 120:
        for alpha, steps, s0, tk in [(200000, 300, 0.0008, 6), (800000, 200, 0.0005, 4)]:
            if _el() > budget_s - 60:
                break
            h_new = _momentum_topk(best_h.copy(), alpha, steps, s0, tk, 0.85)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 6: Simulated annealing
    if _el() < budget_s - 40:
        ntr = min(int((budget_s - 40 - _el()) / 0.004), 400000)
        best_h, best_c5 = _sim_anneal(best_h, ntr, rng, 4e-5, 0.0008, 1e-8)

    # Phase 7: Ultra-fine 2-point polish
    if _el() < budget_s - 1:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 1 - _el()) / 0.0007), 300000)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 7)) % nc2
            d = 3e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
