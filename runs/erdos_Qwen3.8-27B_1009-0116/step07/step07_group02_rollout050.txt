Actually, let me reconsider the Nesterov subgradient. The standard Nesterov accelerated gradient for a non-smooth convex function f:
- x_0 = 0, y_0 = 0
- For k = 1, 2, ...:
  - t_k = (1 + sqrt(1+4t_{k-1}^2))/2
  - y_k = x_{k-1} + (t_{k-1}-1)/t_k * (x_{k-1} - x_{k-2})
  - x_k = proj(y_k - (2/t_k) * g(y_k))

But our function isn't convex (it's the max of non-convex quadratic functions). So Nesterov may not converge. However, in practice, momentum can still help escape local minima.

Let me use a simpler momentum approach:
- v = 0
- For each step:
  - v = mu * v - lr * g
  - h = proj(h + v)

Where mu is the momentum (e.g., 0.9) and lr is the learning rate.

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _wgrad(h, w):
    n = len(h); dx = 2.0 / n; g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _sls(x, a):
    c = _cvals(x); m = c.max()
    s = np.clip(a * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / a

def _slsg(x, a):
    c = _cvals(x); m = c.max()
    s = np.clip(a * (c - m), -400, 0)
    w = np.exp(s); w /= w.sum()
    return _wgrad(x, w)

def _pnorm(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnormg(x, p):
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30: return np.zeros(len(x))
    w = (c / fp) ** (p - 1); ws = w.sum()
    if ws < 1e-30: return np.zeros(len(x))
    w /= ws
    return _wgrad(x, w)

def _lbfgs(h, objf, gradf, args, maxit, dl):
    n = len(h)
    if time.time() > dl - 20: return h
    try:
        r = minimize(objf, h, jac=gradf, args=args, method='L-BFGS-B',
                     bounds=[(0,1)]*n, options={'maxiter':maxit,'ftol':1e-16,'gtol':1e-12,'maxls':50})
        return _repair(r.x, n)
    except: return h

def _mom_subgrad(h, steps, t0, mu, tol, dl):
    n = len(h); v = np.zeros(n)
    for i in range(steps):
        if time.time() > dl - 10: break
        t = t0 / (1.0 + 0.05 * i)
        c = _cvals(h); m = c.max()
        act = np.where(c >= m - tol)[0]
        w = np.zeros(2*n-1)
        w[act] = 1.0 / max(len(act), 1)
        g = _wgrad(h, w)
        v = mu * v - t * g
        h = _repair(h + v, n)
    return h

def _toplag(h, nl, mr, dl):
    n = len(h); c5c = _eval(h)
    for _ in range(mr):
        if time.time() > dl - 15: break
        c = _cvals(h)
        tk = np.argsort(c)[-nl:]
        w = np.zeros(2*n-1); w[tk] = 1.0/nl
        g = _wgrad(h, w)
        imp = False
        for st in [0.05,0.02,0.01,0.005,0.002,0.001,5e-4,1e-4,5e-5,1e-5]:
            hn = _repair(h - st*g, n)
            c5n = _eval(hn)
            if c5n < c5c - 1e-16:
                h = hn; c5c = c5n; imp = True; break
        if not imp: break
    return h, c5c

def _local2(h, iters, rng, dl):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(iters):
        if time.time() > dl - 2: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc: bc = c5n; bh = h.copy()
            else: h[i], h[j] = h[i]-d, h[j]+d
    return bh, bc

def _up(h, f): return np.repeat(h, f).copy()
def _down(h, f):
    n = len(h); m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _make_smooth(n, rng, seed):
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for freq in [1, 2, 3, 5, 7, 11]:
        h += rng.randn() * np.sin(2 * np.pi * freq * x / 2 + rng.randn() * 2)
    h = 0.5 + 0.3 * np.tanh(h)
    return _repair(h, n)

    n0 = len(initial_h_values)
