def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    best_h = h.copy()
    best_c = _eval(h_c)
    step = 0.01

        c = _corrs(h_c)

        order = np.argsort(c)[::-1]

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        wk = w[k] * dx
        s = k - (n - 1)
        if s >= 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            t = -s
            g[t:] += wk * (1.0 - h[:n-t])
            g[:n-t] -= wk * h[t:]
    g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

def _obj_lse(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    val = c_max + T * np.log(exp_c.sum())
    return val + penalty * (h.sum() - n * 0.5)**2

def _lbfgs(h, T, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_lse, h, jac=_grad_lse, args=(T, penalty),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-13, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, n_iter, bc_fn, k_frac, step0):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy().astype(float)
    best_h = h_c.copy()
    best_c = _eval(h_c)
    step = step0
    beta = 0.75
    v = np.zeros(n)
    stall = 0
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        k = max(3, int(len(c) * k_frac))
        order = np.argsort(c)[::-1][:k]
        w = np.exp(np.arange(k) * (-0.15))
        w /= w.sum()
        g = np.zeros(n)
        for ii, kk in enumerate(order):
            wk = w[ii] * dx
            s = kk - (n - 1)
            if s >= 0:
                g[:n-s] += wk * (1.0 - h_c[s:])
                g[s:] -= wk * h_c[:n-s]
            else:
                t = -s
                g[t:] += wk * (1.0 - h_c[:n-t])
                g[:n-t] -= wk * h_c[t:]
        v = beta * v + (1.0 - beta) * g
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-18:
            best_c = c_try
            best_h = h_try.copy()
            h_c = h_try
            stall = 0
            step = min(step * 1.015, 0.04)
        else:
            step *= 0.65
            stall += 1
            if step < 1e-10:
                break
            if stall > 200:
                step = step0 * 0.3
                v *= 0.5
                stall = 0
    return best_h, best_c

def _local2(h, n_iter, rng, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 8e-5
    for _ in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = rng.randn() * scale
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            cn = _eval(h_c)
            if cn < best_c:
                best_c = cn
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
        if _ % 8000 == 0:
            scale *= 0.65
    return best_h, best_c

def _local3(h, n_iter, rng, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 5e-5
    for _ in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 12)) % n
        m = (i + rng.randint(13, 30)) % n
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni, nj, nm = h_c[i]+d1, h_c[j]+d2, h_c[m]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nm <= 1:
            h_c[i], h_c[j], h_c[m] = ni, nj, nm
            cn = _eval(h_c)
            if cn < best_c:
                best_c = cn
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j], h_c[m] = ni-d1, nj-d2, nm+d1+d2
        if _ % 6000 == 0:
            scale *= 0.7
    return best_h, best_c

def _binary_flip(h, n_iter, rng, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    ones = np.where(h_c > 0.5)[0]
    zeros = np.where(h_c <= 0.5)[0]
    for _ in range(n_iter):
        if bc_fn():
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if i1 == i2 or j1 == j2 or i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
            continue
        save = [h_c[i1], h_c[i2], h_c[j1], h_c[j2]]
        h_c[i1] = 0.0; h_c[i2] = 0.0; h_c[j1] = 1.0; h_c[j2] = 1.0
        cn = _eval(h_c)
        if cn < best_c:
            best_c = cn
            best_h = h_c.copy()
            ones = np.where(h_c > 0.5)[0]
            zeros = np.where(h_c <= 0.5)[0]
        else:
            h_c[i1] = save[0]; h_c[i2] = save[1]
            h_c[j1] = save[2]; h_c[j2] = save[3]
    return best_h, best_c
