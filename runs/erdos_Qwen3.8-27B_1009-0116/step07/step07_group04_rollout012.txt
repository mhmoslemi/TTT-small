def adam_softmax(h, beta, t0, budget_s, lr=0.001, steps=3000):
    n = len(h)
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_f = eval_c5(h)
    for step in range(steps):
        if time.time() - t0 > budget_s - 10:
            break
        # Compute softmax-smoothed objective
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        s = beta * (c - c.max())
        s = np.clip(s, -500, 500)
        w = np.exp(s)
        w /= w.sum()
        f = c.max() + np.log(w.sum() + eps) / beta  # Actually f = logsumexp(beta*c)/beta
        
        # Gradient
        g = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:

                    g[i] += wk * (1.0 - h[i+sh])
                if sh > 0:

                    g[i] += wk * (1.0 - h[i-sp])

            best_f = f_new
        # else: keep h

    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = project_sum(h, target)
    # Fix any remaining tiny error
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def softmax_obj_grad(h, beta, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = beta * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    f = m + np.log(e.sum()) / beta
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return f, g

def adam_opt(h, betas, t0, budget_s, lr=0.0005, steps_per_beta=2000):
    n = len(h)
    dx = 2.0 / n
    for beta in betas:
        m = np.zeros(n)
        v = np.zeros(n)
        b1, b2, eps = 0.9, 0.999, 1e-8
        t = 0
        for step in range(steps_per_beta):
            elapsed = time.time() - t0
            if elapsed > budget_s - 15:
                break
            f, g = softmax_obj_grad(h, beta, dx)
            t += 1
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g * g
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_sum(h_new, n * 0.5)
            f_new = eval_c5(h_new)
            # Accept if better (using exact C5, not smoothed)
            f_cur = eval_c5(h)
            if f_new <= f_cur:
                h = h_new
        h = repair(h, n)
    return h

def lbfgs_with_projection(h0, beta, t0, budget_s, maxiter=500):
    n = len(h0)
    dx = 2.0 / n
    target = n * 0.5
    
    def obj(y):
        h = project_sum(y, target)
        f, _ = softmax_obj_grad(h, beta, dx)
        return f
    
    def grad(y):
        h = project_sum(y, target)
        _, g = softmax_obj_grad(h, beta, dx)
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        g[~mask] = 0.0
        return g

                       options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 50})

def local_search(h, n_tries, rng, t0, budget_s):
    n = len(h)

            if c < best_c:
                best_c = c
                best_h = h.copy()
            else:

                h[j] += d

```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def repair(h, n):
    h = project_sum(np.clip(h, 0.0, 1.0), n * 0.5)
    return h

def softmax_fg(h, beta):
    """Return (objective, gradient) of softmax-smoothed C5."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(beta * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    f = m + np.log(e.sum()) / beta
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return f, g
```

def adam_opt(h, beta, t0, t_end, lr, n_steps):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    f_cur = eval_c5(h)
    for _ in range(n_steps):
        if time.time() - t0 > t_end:
            break
        _, g = softmax_fg(h, beta)
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1**t)
        vhat = v / (1 - b2**t)
        h_new = project_sum(h - lr * mhat / (np.sqrt(vhat) + eps), n * 0.5)
        f_new = eval_c5(h_new)
        if f_new < f_cur:
            h = h_new
            f_cur = f_new
    return h

def lbfgs_opt(h0, beta, t0, t_end):
    n = len(h0)
    target = n * 0.5
    
    def obj(y):
        h = project_sum(y, target)
        f, _ = softmax_fg(h, beta)
        return f
    
    def grad(y):
        h = project_sum(y, target)
        _, g = softmax_fg(h, beta)
        g[(h <= 1e-10) | (h >= 1.0 - 1e-10)] = 0.0
        return g
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       options={'maxiter': 500, 'gtol': 1e-10, 'maxls': 30})
        h = project_sum(res.x, target)
        return repair(h, n)
    except:
        return repair(h0.copy(), n)

def local_search(h, rng, t0, t_end, scale):
    n = len(h)
    best_h = h.copy()
    best_c = eval_c5(h)
    while time.time() - t0 < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c = eval_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return best_h, best_c
