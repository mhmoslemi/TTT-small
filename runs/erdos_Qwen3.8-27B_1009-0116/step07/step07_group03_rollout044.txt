            if j < n - sp:

        h_try = best_h.copy()
        n_iter = min(int(t_remaining / 0.003), 50000)
        if n_iter > 50:
            h_try, c_try = _local_fine(h_try, n, n_iter, rng, t0 + budget_s - 2)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
