import numpy as np
from scipy.optimize import minimize
import time

_DEADLINE = 0.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                else:
                    a = max(diff, h[idx] - 1.0)
                h[idx] += a
                diff -= a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lags(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g

def _timeup(left=30):
    return time.time() > _DEADLINE - left

def _pg_topk(h, rounds, k_top, steps_init, left_sec=100):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if time.time() > _DEADLINE - left_sec:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top) * dx / k_top
        improved = False
        for si in range(len(steps)):
            hn = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5c - 1e-15:
                h = hn
                c5c = c5n
                improved = True
                steps[si] = min(steps[si] * 1.35, 0.5)
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _pg_softmax(h, rounds, temp, steps_init, left_sec=100):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if time.time() > _DEADLINE - left_sec:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        t = max(temp * np.exp(-_r / (rounds * 0.7)), 1e-5)
        cs = c - c.max()
        w = np.exp(cs / t)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            sh = (n - 1) - k
            wk = w[k]
            if sh >= 0:
                g[sh:] += wk * (1.0 - h[:n - sh])
                if sh > 0:
                    g[:n - sh] -= wk * h[sh:]
            else:
                sp = -sh
                g[:n - sp] += wk * (1.0 - h[sp:])
                g[sp:] -= wk * h[:n - sp]
        g *= dx
        improved = False
        for si in range(len(steps)):
            hn = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5c - 1e-15:

                c5c = c5n
                improved = True
                steps[si] = min(steps[si] * 1.3, 0.4)
                break

            steps = [s * 0.45 for s in steps]
            stall += 1
            if stall > 14:
                break
        else:
            stall = 0
    return h, c5c

def _push_binary(h, n):
    hb = np.where(h > 0.5, 1.0, 0.0)
    ones = hb.sum()
    target = n // 2
    if ones > target:
        order = np.argsort(h)[::-1]
        for i in range(int(ones - target)):
            hb[order[i]] = 0.0
    elif ones < target:
        order = np.argsort(h)[-1:]
        order_all = np.argsort(-h)
        for i in range(int(target - ones)):
            hb[order_all[i]] = 1.0
    hb = (hb + 0.3 * h - 0.15)
    return _repair(hb, n)

                    best_h = h.copy()
            else:

    return best_h, best_c

def _lsx_obj(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 0)
    return m + np.log(np.sum(np.exp(sc))) / a

def _lsx_grad(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 0)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k

        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))
