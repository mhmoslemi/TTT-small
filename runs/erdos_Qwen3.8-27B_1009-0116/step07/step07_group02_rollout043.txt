import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if abs(w[k]) < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _lse_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    return _grad_w(x, w)

def _hinge_obj(x, thresh):
    c = _cvals(x)
    ex = np.maximum(c - thresh, 0.0)
    return np.sum(ex * ex)

def _hinge_grad(x, thresh):
    c = _cvals(x)
    ex = np.maximum(c - thresh, 0.0)
    return _grad_w(x, 2.0 * ex)

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return float(np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    c = np.maximum(_cvals(x), 0.0)
    fp = float(np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    return _grad_w(x, w / ws)

def _run_lbfgs(h, obj, grad, arg, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 25:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, args=(arg,), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-16,
                                            'gtol': 1e-13, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _top_lag(h, m, max_rounds, deadline):
    n = len(h)
    c5c = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 25:
            break
        c = _cvals(h)
        top = np.argsort(c)[-m:]
        w = np.zeros(2 * n - 1)
        w[top] = 1.0 / m
        g = _grad_w(h, w)
        improved = False
        for st in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4]:
            h_new = _repair(h - st * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c, improved = h_new, c5n, True
                break
        if not improved:
            break
    return h, c5c

def _subgrad(h, steps, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):
        if time.time() > deadline - 20:
            break
        t = 0.015 / (1.0 + 0.01 * i)
        c = _cvals(h)
        m = c.max()
        act = np.where(c >= m - 1e-7)[0]
        w = np.zeros(2 * n - 1)
        w[act] = 1.0 / len(act)
        g = _grad_w(h, w)
        h = _repair(h - t * g, n)
    return h

def _local_block(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):

        i = rng.randint(n - 8)
        blen = rng.randint(3, 9)
        d = 0.01 * rng.randn(blen)
        d -= d.mean() * 0  # keep sum zero
        h2 = h.copy()
        h2[i:i + blen] += d
        h2 = _repair(h2, n)
        c5n = _eval(h2)
        if c5n < c5c - 1e-16:
            h = h2
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _local_pair_sweep(h, n_iter, rng, deadline):

    c5c = _eval(h)
    best_h = h.copy()

        ni, nj = h[i] + d, h[j] - d

    return best_h, best_c
