                diff -= a

                diff += a

    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _alpha_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _alpha_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh == 0:
                g[:] += wk * (1.0 - x)
            else:
                g[:n - sh] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _se(x, half):
    return x.sum() - half

def _se_j(x, half):
    return np.ones(len(x))

def _do_slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _se_j, 'args': (n * 0.5,)}]
    try:
        res = minimize(_alpha_obj, h, jac=_alpha_grad, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _adam(h, alpha, nsteps, lr, deadline):
    n = len(h)
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c = _eval(h)
    best = h.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    for _ in range(nsteps):
        if time.time() > deadline:
            break
        t += 1
        lr_t = lr * np.sqrt(1.0 - b2 ** t) / (1.0 - b1 ** t)
        g = _alpha_grad(h, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        h_new = h - lr_t * m / (np.sqrt(v) + eps)
        h_new = _proj(h_new, n * 0.5)
        c_new = _eval(h_new)
        h = h_new
        if c_new < best_c:
            best_c = c_new
            best = h.copy()
    return best, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    return np.repeat(h, f)

def _local_search(h, niter, rng, base_step, deadline):
    n = len(h)
    best = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 5 + rng.randint(0, 8)) % n
        d1 = base_step * rng.randn()
        d2 = base_step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c = _eval(h)
            if c < cur_c:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best, best_c

def _grad_local(h, niter, rng, base_step, deadline):
    n = len(h)
    best = h.copy()
    best_c = _eval(h)
    for _ in range(niter):
        if time.time() > deadline:
            break
        g = _alpha_grad(best, 50000)
        top_k = np.argsort(np.abs(g))[-6:]
        i = top_k[rng.randint(6)]
        j = top_k[rng.randint(6)]
        if i == j:
            continue
        d = base_step * rng.randn() * np.sign(g[i])
        if d > 0:
            d = -d
        ni, nj = best[i] + d, best[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best[i], best[j] = ni, nj
            c = _eval(best)
            if c < best_c:
                best_c = c
            else:
                best[i], best[j] = ni - d, nj + d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(h0)
    deadline = t0 + budget_s

    # Phase 1: Coarse-to-fine at n/4
    nc = max(64, n // 4)
    hc = _repair(_downsample(h0, n // nc), nc)
    for a in [200, 1000, 5000, 20000]:
        if time.time() > deadline - 600:
            break
        hc, cc = _adam(hc, a, 800, 0.05, deadline - 600)
    h_up = _repair(_upsample(hc, n // nc), n)
    for a in [5000, 50000]:
        if time.time() > deadline - 500:
            break
        h_up, cc = _adam(h_up, a, 600, 0.01, deadline - 500)
    if _eval(h_up) < best_c5:
        best_c5 = _eval(h_up)
        best_h = h_up.copy()

    # Phase 2: Adam alpha sweep from best
    for a in [200, 1000, 5000, 20000, 80000, 300000, 1000000]:
        if time.time() > deadline - 350:
            break
        h_n, c_n = _adam(best_h, a, 1500, 0.015, deadline - 350)
        if c_n < best_c5:
            best_c5 = c_n
            best_h = h_n.copy()

    # Phase 3: SLSQP refinement
    for a in [5000, 50000, 300000, 2000000]:
        if time.time() > deadline - 200:
            break
        h_n = _do_slsqp(best_h, a, 400, 1e-15)
        c_n = _eval(h_n)
        if c_n < best_c5:
            best_c5 = c_n
            best_h = h_n.copy()

    # Phase 4: Multi-kick escape
    for sigma in [0.08, 0.04, 0.02, 0.01]:
        if time.time() > deadline - 120:
            break
        h_kick = _repair(best_h + sigma * rng.randn(n), n)
        for a in [3000, 30000]:
            if time.time() > deadline - 120:
                break
            h_kick, c_n = _adam(h_kick, a, 800, 0.01, deadline - 120)
        h_kick = _do_slsqp(h_kick, 100000, 200, 1e-15)
        c_n = _eval(h_kick)
        if c_n < best_c5:
            best_c5 = c_n
            best_h = h_kick.copy()

    # Phase 5: Gradient-informed local search
    if time.time() < deadline - 60:
        ntr = int(min((deadline - 60 - time.time()) / 0.003, 300000))
        best_h, best_c5 = _grad_local(best_h, ntr, rng, 3e-5, deadline - 60)

    # Phase 6: Fine random local search
    if time.time() < deadline - 5:
        remaining = deadline - 5 - time.time()
        ntr = int(min(remaining / 0.0008, 400000))
        best_h, best_c5 = _local_search(best_h, ntr, rng, 4e-6, deadline - 5)

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 2:
        h_n = _do_slsqp(best_h, 5000000, 300, 1e-16)
        c_n = _eval(h_n)
        if c_n < best_c5:
            best_c5 = c_n
            best_h = h_n

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
