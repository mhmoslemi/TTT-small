import numpy as np

from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _vec_subgrad(h, m_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

            g += 1.0 - 2.0 * h

            g[d:n] -= h[:n-d]
        else:
            s = -d
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    return g * dx

def _subgrad_fast(h, rng, deadline, m_top=5):

    best_h, best_c = h.copy(), c5_cur
    lr = 0.03

            break

        g -= g.mean()

                c[k2] -= d_opt * d_opt

Let me focus on making the CD faster:
1. The main cost per CD iteration is the O(n) gradient computation + O(n) line search.
2. In the original, the gradient uses `_grad_at_pos` which creates two arrays of size 2n-1.

Actually, let me think about this differently. The gradient for position i is:
g_i[k] = (1-h[i+d])*I(0<=i+d<n) - h[i-d]*I(0<=i-d<n) where d=k-(n-1)

This can be written as:
g_i = (1-h) circularly shifted by i, minus h circularly shifted by -i, with appropriate truncation.

For the pair (i,j):
g_{ij} = g_i - g_j

The cost is O(n) to compute g_i and O(n) to compute g_j, plus O(n) for the line search. Total O(n) per iteration.

For n=560, each iteration involves about 560*3 = 1680 floating point ops (plus numpy overhead). In numpy, a single array operation on 1119 elements takes maybe 1-5 microseconds. So each CD iteration might take 10-50 microseconds. That means in 400 seconds, we can do 8-40 million iterations. That's a LOT.

The issue might be that after a few hundred thousand iterations, the CD gets stuck and no more improvement is found. The stable counter helps but isn't enough.

Let me add a "kick" mechanism: if stable > threshold, perturb h slightly by adding small random noise and re-project.

Also, I notice that the original code's CD doesn't try to be smart about which pairs to select. Let me add a gradient-informed selection:
- Compute the subgradient g (vector of size n) for the current h
- The best direction is -g projected to sum=0
- Pick i where -g[i] is most positive (wants h[i] increased) and j where -g[j] is most negative (wants h[j] decreased)
- Or: pick i,j proportional to |g[i]| and |g[j]|

Actually, a simpler heuristic: with probability 0.5, pick random pair. With probability 0.5, pick pair based on gradient (i from top half of |g|, j from bottom half).

Let me also add a "restart" mechanism: if progress stalls, keep the best solution but restart CD from a slightly perturbed version.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

def _vec_subgrad(h, m_top):
