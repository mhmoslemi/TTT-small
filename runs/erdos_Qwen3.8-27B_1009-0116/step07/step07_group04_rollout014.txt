def coord_descent_step(h, i, j, t0, budget_s, max_passes=3):
    """Do coordinate descent on h[i] and h[j] keeping h[i]+h[j] constant."""
    n = len(h)
    dx = 2.0/n
    s = h[i] + h[j]  # fixed sum
    # Compute base correlation (with h[i]=h[j]=0)
    # c_base[k] = full correlation of h with h[i]=h[j]=0
    # Then c[k] = c_base[k] + (α_k)*h[i] + (γ_k)*h[j]
    #          = c_base[k] + (α_k - γ_k)*h[i] + γ_k*s
    
    h[i] = 0
    h[j] = 0
    c0 = np.correlate(h, 1-h, mode='full')  # base with both zero
    
    # Now restore and compute coefficients
    h[i] = 0; h[j] = 1
    c1 = np.correlate(h, 1-h, mode='full')
    h[i] = 1; h[j] = 0
    c2 = np.correlate(h, 1-h, mode='full')
    
    # c(h[i], h[j]) = c0 + h[i]*(c2-c0) + h[j]*(c1-c0)
    # With h[i]+h[j]=s: c = c0 + h[i]*(c2-c0) + (s-h[i])*(c1-c0)
    #                  = c0 + s*(c1-c0) + h[i]*((c2-c0)-(c1-c0))
    #                  = c0 + s*(c1-c0) + h[i]*(c2-c1)
    
    a = c2 - c1  # coefficient of h[i]
    b = c0 + s*(c1-c0)  # intercept
    # c(h_i) = b + a * h_i
    
    # Minimize max_k (b[k] + a[k]*h_i) over h_i ∈ [max(0,s-1), min(1,s)]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    
    # The function f(x) = max_k(b[k] + a[k]*x) is convex piecewise linear
    # Minimum is at a breakpoint or boundary
    # Breakpoints are where b[i]+a[i]*x = b[j]+a[j]*x, i.e., x = (b[j]-b[i])/(a[i]-a[j])
    
    # For efficiency, we can use the fact that f is convex and do ternary search
    # Or just evaluate at a few key points
    
    # Actually, let's just do a simple approach: evaluate at boundaries and at 
    # the point where the two largest slopes intersect
    best_x = (lo + hi) / 2
    best_val = np.max(b + a * best_x)
    
    for x_try in [lo, hi, lo*0.3 + hi*0.7, lo*0.7 + hi*0.3]:
        val = np.max(b + a * x_try)
        if val < best_val:
            best_val = val
            best_x = x_try

    for frac in np.linspace(0, 1, 21):
        x_try = lo + frac * (hi - lo)
        val = np.max(b + a * x_try)
        if val < best_val:
            best_val = val
            best_x = x_try
    
    h[i] = best_x
    h[j] = s - best_x
    return h, best_val * dx

def solve_1d_convex(a, b, lo, hi, tol=1e-12):
    """Minimize max_k(b[k] + a[k]*x) over x in [lo, hi]."""
    # Binary search on the optimal value t
    t_lo = np.max(b + a * lo)  # lower bound on optimal value
    t_hi = np.max(b + a * hi)
    # Actually t_lo and t_hi aren't right for binary search direction
    # Let me think again...
    # f(x) = max_k(b[k] + a[k]*x) is convex
    # We want to find x* that minimizes f(x*)
    # By binary search on x won't work directly since f isn't monotone
    
    # Better: ternary search or find the breakpoint
    # Since f is convex piecewise linear, the minimum is at a breakpoint
    # A breakpoint is where two lines cross: x = (b[i]-b[j])/(a[j]-a[i])
    
    # Efficient approach: the subgradient at x* is 0 (or contains 0)
    # At x*, the active set A(x*) = {k: b[k]+a[k]*x* = f(x*)}
    # 0 ∈ conv{a[k] : k ∈ A(x*)}
    # So we need min_{k∈A} a[k] ≤ 0 ≤ max_{k∈A} a[k]
    
    # Let's just do a fine grid search followed by refinement
    # Since the function has at most 2n-1 breakpoints, a grid of 2n+1 points suffices
    # But that's 1121 points, each requiring O(n) evaluation = O(n^2) total ≈ 6M ops
    
    # Actually, even simpler: since f is convex, we can find the minimum by
    # finding where the subgradient changes sign
    # f(x) is non-increasing for x < x* and non-decreasing for x > x*
    # We can use bisection on the derivative (subgradient)
    
    # The left derivative at x is: a[k] where k achieves max and has the most negative a
    # The right derivative at x is: a[k] where k achieves max and has the most positive a
    
    # Bisection:
    for _ in range(60):  # enough for 1e-18 precision
        mid = (lo + hi) / 2
        vals = b + a * mid
        kmax = np.argmax(vals)
        v_max = vals[kmax]
        # Check if 0 is in subgradient at mid
        active = np.where(np.abs(vals - v_max) < 1e-10)[0]
        if len(active) == 0:
            active = np.array([kmax])
        a_min = a[active].min()
        a_max = a[active].max()
        if a_min <= 0 <= a_max:
            # mid is optimal (or very close)
            break
        elif a_min > 0:
            # All active slopes positive → move left
            hi = mid
        else:
            # All active slopes negative → move right
            lo = mid
    x_opt = (lo + hi) / 2
    return x_opt, np.max(b + a * x_opt)

    # [x2^2 x2 1] [b] = [cB]
    # [x3^2 x3 1] [d]   [cC]
    # For each k, solve this. But doing it element-wise on arrays is fine.
    
    # Actually, let me just use a simpler approach:
    # c(x) = cA * L1(x) + cB * L2(x) + cC * L3(x) where L are Lagrange basis
    # For evaluation at multiple x points, this is just:
    # c_eval = cA[:,None] * L1 + cB[:,None] * L2 + cC[:,None] * L3
    
    # Or even simpler: just evaluate the max at grid points using the quadratic formula
    # For a grid of x values, compute c_k(x) for all k and take max
    
    x_grid = np.linspace(lo, hi, n_grid)
    # c(x) for each k: use the quadratic formula

    # Using Newton's divided differences:
    f00 = (cB - cA) / (x2 - x1)
    f11 = (cC - cB) / (x3 - x2)
    f01 = (f11 - f00) / (x3 - x1)
    # c(x) = cA + f00*(x-x1) + f01*(x-x1)*(x-x2)
    
    best_x = x_grid[0]

            best_x = x
    
    h[i] = best_x
    h[j] = s - best_x
    return h, best_val * dx

import numpy as np
from scipy.optimize import minimize
import time
import functools

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_topk(h, k_top, steps, step0, t0, deadline):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    for _s in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, k_top)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 4.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_obj(x, alpha, n):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _cons_fun(x, target):
    return x.sum() - target

def _cons_jac(x, target):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    obj_f = functools.partial(_slsqp_obj, alpha=alpha, n=n)
    grad_f = functools.partial(_slsqp_grad, alpha=alpha, n=n)
    cf = functools.partial(_cons_fun, target=target)
    cj = functools.partial(_cons_jac, target=target)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord2(h, i, j):
    n = len(h)
    dx = 2.0 / n
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-14:
        return h, _eval(h)
    h[i] = lo; h[j] = s - lo
    cA = _corr(h)
    xm = (lo + hi) * 0.5
    h[i] = xm; h[j] = s - xm
    cB = _corr(h)
    h[i] = hi; h[j] = s - hi
    cC = _corr(h)
    f00 = (cB - cA) / (xm - lo)
    f01 = (cC - cA) / (hi - lo) - f00
    f01 /= (hi - xm)
    best_x = xm
    best_val = np.inf
    n_grid = 80
    for x in np.linspace(lo, hi, n_grid):
        ce = cA + f00 * (x - lo) + f01 * (x - lo) * (x - xm)
        cv = ce.max()
        if cv < best_val:
            best_val = cv
            best_x = x
    span = (hi - lo) / n_grid
    x_fine = np.linspace(max(lo, best_x - span), min(hi, best_x + span), 60)
    for x in x_fine:
        ce = cA + f00 * (x - lo) + f01 * (x - lo) * (x - xm)
        cv = ce.max()
        if cv < best_val:
            best_val = cv
            best_x = x
    h[i] = best_x
    h[j] = s - best_x
    return h, best_val * dx

def _coord2_pass(h, n_pairs, rng, t0, deadline, targeted=False):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_pairs):
        if time.time() > deadline:
            break
        if targeted and _it % 200 == 0:
            c = _corr(h)
            top_idx = np.argsort(c)[-30:]
            i = top_idx[rng.randint(30)]
            j = (i + 1 + rng.randint(0, 10)) % n
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 15)) % n
        h, c5n = _coord2(h, i, j)
        if c5n < c5c - 1e-15:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _local_search(h, n_tries, rng, t0, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    deadline = t0 + budget_s

    # Phase 1: Random 2-coord descent
    h1, c5v = _coord2_pass(h.copy(), 20000, rng, t0, t0 + 180)
    if c5v < best_c5:
        best_c5, best_h = c5v, h1.copy()

    # Phase 2: Targeted 2-coord descent
    h2, c5v = _coord2_pass(best_h.copy(), 20000, rng, t0, t0 + 360, targeted=True)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 3: Nesterov top-k
    if time.time() < deadline - 350:
        h3 = _nesterov_topk(best_h.copy(), 5, 5000, 0.0015, t0, t0 + 550)
        c5v = _eval(h3)
        if c5v < best_c5:
            best_c5, best_h = c5v, h3.copy()

    # Phase 4: SLSQP with log-sum-exp
    if time.time() < deadline - 250:
        h4 = _slsqp_opt(best_h.copy(), 20000, 1200, 1e-16)
        c5v = _eval(h4)
        if c5v < best_c5:
            best_c5, best_h = c5v, h4.copy()
        if time.time() < deadline - 200:
            h4 = _slsqp_opt(best_h.copy(), 100000, 1500, 1e-16)
            c5v = _eval(h4)
            if c5v < best_c5:
                best_c5, best_h = c5v, h4.copy()

    # Phase 5: Targeted 2-coord after SLSQP
    if time.time() < deadline - 120:
        h5, c5v = _coord2_pass(best_h.copy(), 15000, rng, t0, t0 + budget_s - 120, targeted=True)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5.copy()

    # Phase 6: Nesterov with higher k for polish
    if time.time() < deadline - 80:
        h6 = _nesterov_topk(best_h.copy(), 30, 2000, 0.0005, t0, t0 + budget_s - 70)
        c5v = _eval(h6)
        if c5v < best_c5:
            best_c5, best_h = c5v, h6.copy()

    # Phase 7: Final random 2-coord polish
    if time.time() < deadline - 40:
        h7, c5v = _coord2_pass(best_h.copy(), 10000, rng, t0, t0 + budget_s - 40)
        if c5v < best_c5:
            best_c5, best_h = c5v, h7.copy()

    # Phase 8: Fine local search
    if time.time() < deadline - 5:
        remaining = deadline - 5 - time.time()
        n_tries = int(min(remaining / 0.003, 30000))
        best_h, best_c5 = _local_search(best_h, n_tries, rng, t0, deadline - 2)

    # Final: one more targeted pass if time
    if time.time() < deadline - 2:
        rem = int((deadline - 2 - time.time()) / 0.002)
        rem = min(rem, 5000)
        if rem > 100:
            h8, c5v = _coord2_pass(best_h.copy(), rem, rng, t0, deadline - 1, targeted=True)
            if c5v < best_c5:
                best_c5, best_h = c5v, h8.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
