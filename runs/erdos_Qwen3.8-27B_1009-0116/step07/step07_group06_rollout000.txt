_LINE_FACTORS = tuple(2.0 ** (-k) for k in range(32))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if s <= 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
                for idx in order:
                    if diff <= 1e-13: break
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
            else:
                neg = -diff
                order = np.argsort(h)
                for idx in order:
                    if neg <= 1e-13: break
                    a = min(neg, h[idx])
                    h[idx] -= a
                    neg -= a
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval_c(c, n):
    return float(c.max() * 2.0 / n)

def _pair_batch(h, c, n, G, work, max_tries, max_accept):
    ncand = min(36, n)
    if ncand < 2: return 0
    order = np.argsort(G)
    lows = order[:ncand]
    highs = order[-ncand:]
    cmax = c.max()
    cand = []
    for aa in range(ncand):
        i = int(lows[aa]); gi = G[i]
        room_i = 1.0 - h[i]
        if room_i <= 1e-9:
            continue
        for bb in range(ncand):
            j = int(highs[bb])
            if j == i:
                continue
            diff = G[j] - gi
            if diff <= 1e-9:
                continue
            room = h[j]
            if room > room_i:
                room = room_i
            if room <= 1e-9:
                continue
            score = diff * room
            # add noise? no
            cand.append((score, i, j))
    if not cand: return 0
    cand.sort(reverse=True)
    accepted = 0
    lim = len(cand)
    if max_tries < lim: lim = max_tries
    for idx in range(lim):
        score, i, j = cand[idx]
        res = _line_trial(h, c, i, j, n, work)
        if res is None:
            continue
        t, g, k1, k2, phi = res
        t2 = t * t
        h[i] += t
        h[j] -= t
        c += t * g
        c[k1] += t2
        c[k2] += t2
        c[n - 1] -= 2.0 * t2
        accepted += 1
        if accepted >= max_accept:
            break
    return accepted

def _fw_move_from_G(h, c, n, G):
    target = 0.5*n
    order = np.argsort(G)
    full = int(target)
    frac = target - full
    z = np.zeros(n)
    if full > 0: z[order[:full]] = 1.0
    if frac > 1e-13 and full < n: z[order[full]] = frac
    diff = z - h
    if float(np.dot(diff, diff)) < 1e-16: return None
    cur = c.max()
    best_phi = cur
    best_h = None; best_c = None; best_a = 0.0
    hnew = np.empty(n)
    for fac in _FW_FACTORS:
        a = fac
        if a <= 0.0: break
        hnew[:] = h
        hnew += a * diff
        np.clip(hnew, 0.0, 1.0, out=hnew)
        if abs(hnew.sum() - target) > 1e-8:
            hnew = _repair(hnew, n)
        cn = _corr(hnew)
        mn = cn.max()
        if mn < best_phi:
            best_phi = mn
            best_a = a
            best_h = hnew.copy()
            best_c = cn.copy()
    if best_a > 0.0 and best_phi < cur - 1e-12:
        return best_h, best_c
    return None

import numpy as np
import time

_LINE_FACTORS = tuple(2.0 ** (-k) for k in range(32))
_FW_FACTORS = tuple(2.0 ** (-k) for k in range(28))


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if s <= 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if diff > 1e-12:
            for idx in np.argsort(1.0 - h):
                if diff <= 1e-13:
                    break
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
        elif diff < -1e-12:
            neg = -diff
            for idx in np.argsort(h):
                if neg <= 1e-13:
                    break
                a = min(neg, h[idx])
                h[idx] -= a
                neg -= a
    return h


def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _default_h(n, seed):
    rng = np.random.RandomState(seed)
    x = 2.0 * np.arange(n) / n
    h = 0.5 + 0.45 * np.sin(np.pi * x) + 0.05 * rng.randn(n)
    return _repair(np.clip(h, 0.0, 1.0), n)


def _make_sum_exact(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)

        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) > 1e-9:

    return h


def _grad_diff_into(h, i, j, n, L, g):
    g.fill(0.0)
    rev = 1.0 - h[::-1]
    g[i:i + n] += rev
    g[n - 1 - i:2 * n - 1 - i] -= h
    g[j:j + n] -= rev
    g[n - 1 - j:2 * n - 1 - j] += h


def _line_trial(h, c, i, j, n, work, scaled, gbuf):
    hi = min(1.0 - h[i], h[j])
    if hi <= 1e-13:

def _pair_batch(h, c, n, G, work, scaled, gbuf, ncand, max_tries, max_accept):
    if n < 2:

                room = room_i

                continue

        acc += 1
        if acc >= max_accept:
            break
    return acc


def _make_active(c, K):
    L = len(c)
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]
    vals = c[idx]
    cmax = vals.max()
    span = cmax - vals.min()
    T = max(0.25, 0.1 * span)
    w = np.exp((vals - cmax) / T)
    w /= w.sum()
    shifts = idx - (L // 2)
    mask = w > 1e-8
    if not mask.any():
        mask = np.ones(len(w), dtype=bool)
    return shifts[mask], w[mask]


def _grad_active(h, shifts, weights, n):
    G = np.zeros(n)
    for pos in range(len(shifts)):
        d = int(shifts[pos])
        w = float(weights[pos])
        if w <= 0.0:
            continue
        if d >= 0:
            l = n - d
            if l > 0:
                G[d:] += w * (1.0 - h[:l])
                G[:l] -= w * h[d:]
        else:
            e = -d
            l = n - e
            if l > 0:
                G[:l] += w * (1.0 - h[e:])
                G[e:] -= w * h[:l]
    return G


def _fw_move_from_G(h, c, n, G):
    target = 0.5 * n
    order = np.argsort(G)
    full = int(target)
    frac = target - full
    z = np.zeros(n)

    if float(np.dot(diff, diff)) < 1e-16:
        return None

            best_h = hnew.copy()
            best_c = cn.copy()

    if best_h is not None and best_phi < cur - 1e-12:

    return None
