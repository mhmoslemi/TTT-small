import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _lag_grad(h, k_idx, n):
    """Gradient of c[k_idx] with respect to h. Returns array of shape (n,)."""
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0:
        cnt = n - sh
        i1 = np.arange(cnt)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        cnt = n - sp
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(cnt)
        g[i2] -= h[i2 + sp]
    return g

def _softmax_grad(h, beta):
    """Gradient of softmax-smoothed max of correlations."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')  # raw, without dx scaling
    # For numerical stability
    cm = c.max()
    exp_c = np.exp(beta * (c - cm))
    w = exp_c / exp_c.sum()
    g = np.zeros(n)
    # Only process lags with significant weight
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]

        wk = w[k] / len(c)  # hmm, should I divide by len(c)?
        g += wk * _lag_grad(h, k, n)
    return g / beta  # chain rule for (1/beta)*log(sum exp(beta*c))

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lag_grad(h, k_idx, n):
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:cnt]
    else:
        sp = -sh
        cnt = n - sp
        g[sp:] += (1.0 - h[:cnt])
        g[:cnt] -= h[sp:]
    return g

def _topk_grad(h, topk, n):
    g = np.zeros(len(h))
    for k in topk:
        g += _lag_grad(h, k, n)
    return g / max(len(topk), 1)

def _softmax_grad(h, beta):
    n = len(h)
    c = _corrs_raw(h)
    cm = c.max()
    bc = beta * (c - cm)
    exp_bc = np.exp(bc - bc.max())
    w = exp_bc / exp_bc.sum()
    thresh = w.max() * 0.003
    active = np.where(w > thresh)[0]
    g = np.zeros(n)
    for k in active:
        if w[k] > 1e-10:
            g += w[k] * _lag_grad(h, k, n)
    return g

def _obj_pnorm(hv, p, penalty, n):
    dx = 2.0 / n
    c = np.correlate(hv, 1.0 - hv, mode='full') * dx
    val = (np.maximum(c, 1e-30) ** p).sum() ** (1.0 / p)
    return val + penalty * (hv.sum() - n * 0.5) ** 2

def _grad_pnorm(hv, p, penalty, n):
    dx = 2.0 / n
    c = np.correlate(hv, 1.0 - hv, mode='full') * dx
    cp = np.maximum(c, 1e-30) ** p
    s = cp.sum()
    factor = s ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if c[k] < 1e-12:
            continue
        wk = factor * c[k] ** (p - 1) * dx
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - hv[sh:])
            if sh > 0:
                g[sh:] -= wk * hv[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - hv[:cnt])
            g[:cnt] -= wk * hv[sp:]
    g += 2.0 * penalty * (hv.sum() - n * 0.5)
    return g

def _lbfgs_pnorm(h, p, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_pnorm, h.copy(), jac=_grad_pnorm,
                       args=(p, penalty, n), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 30})
        x = res.x
    except Exception:
        x = h
    return _proj_sum(x, n * 0.5)

        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 5: L-BFGS high-p refinement
    t_end = min(budget_s - 200, t0 + 650)
    if elapsed() < t_end:
        for p, pen in [(256, 1e7), (1024, 1e8)]:
            if elapsed() > t_end:
                break
            h_try = _lbfgs_pnorm(best_h.copy(), p, pen, 400)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5, best_h = c_try, h_try

    # Phase 6: Perturb and retry
    t_end = min(budget_s - 150, t0 + 800)
    if elapsed() < t_end:
        best_h, best_c5 = _perturb_retry(best_h, best_c5, t0, t_end, rng)

    # Phase 7: Targeted coordinate descent
    t_end = min(budget_s - 60, t0 + 1050)
    if elapsed() < t_end:
        n_iter = max(1000, int((t_end - elapsed()) / 0.004))
        best_h, best_c5 = _topk_coord(best_h, n_iter, t0, t_end, rng, 5)

    # Phase 8: Final L-BFGS polish
    if elapsed() < budget_s - 10:
        h_try = _lbfgs_pnorm(best_h.copy(), 512, 5e7, 200)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    best_h = _proj_sum(best_h, n / 2.0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
