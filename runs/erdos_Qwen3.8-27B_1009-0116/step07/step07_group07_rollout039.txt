def _corrs(h):

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

def _obj_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = np.sum(c**p)**(1.0/p)
    return val + penalty * (h.sum() - n * 0.5)**2

def _grad_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cp = np.maximum(c, 1e-30)**p
    s = cp.sum()
    factor = s**(1.0/p - 1.0)
    g = np.zeros(n)

            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:

                g[i2] -= wk * h[i2 - sh]
        else:

            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0

        else:

def _corrs(h):
    n = len(h)

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

    return h

def _obj_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.sum(c**p)**(1.0/p) + penalty * (h.sum() - n*0.5)**2

def _grad_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_safe = np.maximum(c, 1e-30)
    s = c_safe.sum()**p
    factor = s**(1.0/p - 1.0)
    g = np.zeros(n)
    for k in range(2*n-1):
