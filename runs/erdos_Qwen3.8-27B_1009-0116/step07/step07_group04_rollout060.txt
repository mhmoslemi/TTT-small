def _topk_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argpartition(c, -k_top)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:

                g[i2] -= wk * h[i2 - sh]

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]

        if time.time() > deadline:
            break
        g = _topk_grad(h, k_top)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1**(t+1))
        v_hat = v / (1 - b2**(t+1))

            f = f_new
            if f < best_f:

                best_h = h.copy()
        else:
            lr *= 0.9  # backtrack

def _repair(h, n):

    target = n / 2.0

def _proj_to_sum(h, target):

    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, k_top):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if k_top >= len(c):

    else:

import numpy as np
from scipy.optimize import minimize
import time

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    return _proj_to_sum(h, target)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if k_top >= len(c):
        top_k = np.arange(len(c))
    else:
        top_k = np.argpartition(c, -k_top)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _adam_opt(h, k_top, steps, lr0, lr1, t0, deadline):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    for t in range(steps):
        if time.time() > deadline:
            break
        lr = lr1 + 0.5 * (lr0 - lr1) * (1 + np.cos(np.pi * min(t, steps - 1) / max(steps - 1, 1)))
        g = _topk_grad(h, k_top)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** (t + 1))
        v_hat = v / (1 - b2 ** (t + 1))
        h_new = _proj_to_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
        elif t < 50:
            pass
        else:
            h = _repair(h, n)
            f = _eval(h)
    return _repair(best_h, n)

def _dc_dh_i(h, i, n):
    c = np.zeros(2 * n - 1)
    dx = 2.0 / n
    sh_lo = max(-(n - 1), i - (n - 1))
    sh_hi = min(n - 1, i)
    if sh_lo <= sh_hi:
        sr = np.arange(sh_lo, sh_hi + 1)
        cr = sr + (n - 1)
        c[cr] += (1.0 - h[i - sr])
    sh_lo2 = max(-(n - 1), -i)
    sh_hi2 = min(n - 1, n - 1 - i)
    if sh_lo2 <= sh_hi2:
        sr2 = np.arange(sh_lo2, sh_hi2 + 1)
        cr2 = sr2 + (n - 1)
        c[cr2] -= h[i + sr2]
    return c * dx

def _pair_descent(h, n_sweeps, rng, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    f = c.max()
    best_h = h.copy()
    best_f = f
    for sweep in range(n_sweeps):
        if time.time() > deadline:

        top_shifts = np.argsort(c)[-10:]
        active = set()
        for k in top_shifts:

                if sh > 0:

            else:
                active.update(range(-sh, n))
                active.update(range(n - sh))
        active_list = list(active)
        order = rng.permutation(n)
        active_set = set(active_list)
        for idx in range(0, n - 1, 2):

            if i in active_set and j in active
                pass
            if i == j:

            di = _dc_dh_i(h, i, n)
            dj = _dc_dh_i(h, j, n)
            d = di - dj

            hi_b = min(1.0 - h[i], h[j])
            if lo_b >= hi_b:

            m1 = lo_b + (hi_b - lo_b) / 3

def _slsqp_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp_run(h, alpha, maxiter, t0, deadline):
    n = len(h)
    if time.time() > deadline:
        return _repair(h, n)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    cons = {'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac, 'args': (target,)}
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cons_eq(x, target):
    return x.sum() - target

def _cons_eq_jac(x, target):
    return np.ones(len(x))

def _fine_search(h, n_iters, rng, t0, deadline):
    n = len(h)
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = 3e-6 * rng.randn()
        h[i] += scale
        h[j] -= scale
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            f_new = _eval(h)
            if f_new < f - 1e-20:
                f = f_new
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
            else:
                h[i] -= scale
                h[j] += scale
        else:
            h[i] -= scale
            h[j] += scale
    return best_h, best_f
