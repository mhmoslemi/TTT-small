import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, m, n):
    s = m - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2-s:hi2-s]
    return g

def _full_grad(h, n, top_idx, weights):
    """Compute weighted gradient of top lags."""
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        grad += weights[i] * _grad_lag(h, k, n)
    return grad

def _pair_cd(h, deadline, max_iter=100000, seed=0):
    """Gradient-guided pairwise coordinate descent."""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stagnant > 8000:
            break
        
        # Get gradient info for pair selection
        top_idx = np.argsort(c)[-4:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += _grad_lag(h, k, n)
        grad /= len(top_idx)
        grad -= grad.mean()
        
        # Select pair: i should have high grad (decreasing h[i] helps), j should have low grad
        # h[i] + d, h[j] - d: effect on objective ~ d*(grad[j] - grad[i])
        # We want grad[j] - grad[i] > 0, i.e., grad[j] > grad[i]
        # But we also want h[i] < 1 (can increase) and h[j] > 0 (can decrease)

            # Quick evaluation

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, m, n):
    """Corrected gradient of c[m] w.r.t. h where c=correlate(h,1-h,'full')."""
    s = m - (n - 1)
    g = np.zeros(n)
    # Term: -h[p+s] for 0<=p+s<n => p in [max(0,-s), min(n, n-s))
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if lo1 < hi1:
        g[lo1:hi1] -= h[lo1 + s : hi1 + s]
    # Term: +(1-h[p-s]) for 0<=p-s<n => p in [max(0,s), min(n, s+n))
    lo2 = max(0, s)
    hi2 = min(n, s + n)
    if lo2 < hi2:
        g[lo2:hi2] += (1.0 - h[lo2 - s : hi2 - s])
    return g

def _mom_grad(h, deadline, M=6, T0=5.0, max_iter=30000, seed=0):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    T = T0
    v = np.zeros(n)
    beta = 0.85
    lr = 0.02
    stagnant = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stagnant > 4000:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-M:]
        Tc = T * c[top_idx]
        w = np.exp(Tc - Tc.max())
        w /= w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            grad += w[i] * _grad_lag(h, k, n)
        grad *= dx
        grad -= grad.mean()
        v = beta * v + grad
        v -= v.mean()
        vn = np.linalg.norm(v)
        if vn < 1e-13:
            break
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_sum(h - f * lr * v / vn, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if c5_cur < best_c - 1e-14:
            best_c = c5_cur
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if not improved:
            v *= 0.5
            T = min(T * 1.01, 60.0)
        else:
            T = min(T * 1.0005, 60.0)
        if stagnant > 800 and stagnant % 800 == 0:
            h = _proj_sum(h + 0.02 * rng.randn(n), n * 0.5)
            c5_cur = _eval(h)
            v = np.zeros(n)
    return best_h, best_c

def _pair_cd(h, deadline, max_iter=100000, seed=0):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stagnant > 6000:
            break
        top_idx = np.argsort(c)[-4:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += _grad_lag(h, k, n)
        grad /= len(top_idx)
        grad -= grad.mean()
        mask_i = h < 0.9999
        mask_j = h > 0.0001
        if not mask_i.any() or not mask_j.any():
            break
        ri = rng.rand()
        rj = rng.rand()
        if ri < 0.65:
            thresh_i = np.percentile(grad[mask_i], 25)
            ci = np.where(mask_i & (grad < thresh_i))[0]
        else:
            ci = np.where(mask_i)[0]
        if rj < 0.65:
            thresh_j = np.percentile(grad[mask_j], 75)
            cj = np.where(mask_j & (grad > thresh_j))[0]
        else:
            cj = np.where(mask_j)[0]
        if len(ci) == 0:
            ci = np.where(mask_i)[0]
        if len(cj) == 0:
            cj = np.where(mask_j)[0]
        i = ci[rng.randint(len(ci))]
        j = cj[rng.randint(len(cj))]
        if i == j:

        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-13:
            continue
        best_d = 0.0
        best_c5_loc = c5_cur
        for frac in [0.5, 0.3, 0.7, 0.15, 0.9, 0.1]:
            d = d_lo + frac * (d_hi - d_lo)
            if abs(d) < 1e-12:

            h[i] += d
            h[j] -= d
            c5t = _eval(h)
            h[i] -= d
            h[j] += d
            if c5t < best_c5_loc:

                best_d = d
        if best_d == 0.0:

        h[i] += best_d
        h[j] -= best_d
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_cur = c.max()
        if c5_cur < best_c - 1e-14:

            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
    return best_h, best_c

def _fine_polish(h, deadline, seed=0):
    n = len(h)
    rng = np.random.RandomState(seed)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(400000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
