def _repair(h, n):
    t = n*0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t/s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s-t) < 1e-12: break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0-h[i]); h[i] += a; d -= a

import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(8):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo = float(np.min(v)) - 1.0
        hi = float(np.max(v)) + 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            s = np.clip(v - mid, 0.0, 1.0).sum()
            if s > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        convR = np.fft.irfft(H * Urev, Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * Ur, Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _fista_smooth(h0, alpha, max_time, L=10.0, accel=True):
    if max_time <= 0.0:
        h = h0.copy()
        return h, h.copy(), _c5_direct(h)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _smooth_value(h, alpha)
    fh = fy
    best_h = h.copy()
    best_c = _c5_direct(h)
    stale = 0
    Lcur = L
    for it in range(300000):
        if time.time() - t0 > max_time:
            break
        if accel and it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y = h
            fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(14):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h
            h = xn
            fh = fx
            stale = 0
            Lcur = max(Lcur * 0.85, 1.0)
            if it % 6 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t = 1.0
            fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 10:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _run_alphas(h0, alphas, total_time, deadline, L=10.0, accel=True):
    hcur = h0.copy()
    best_h = h0.copy()
    best_c = _c5_direct(best_h)
    total_time = min(total_time, max(0.0, deadline - time.time()))
    per = total_time / max(1, len(alphas))
    for a in alphas:
        if deadline - time.time() < 2.0 or per <= 0.0:
            break
        hf, bh, bc = _fista_smooth(hcur, a, per, L, accel)
        chf = _c5_direct(hf)
        chb = _c5_direct(bh)
        if bc < best_c:
            best_c = bc
            best_h = bh
        if chf < best_c:
            best_c = chf
            best_h = hf
        hcur = hf if chf <= chb else bh
    return best_h, best_c, hcur

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _anti_pal(h):
    n = len(h)
    return _proj(0.5 + 0.5 * (h - h[::-1]), n * 0.5)

def _pal_dev(h):
    n = len(h)
    return _proj(0.5 + 0.5 * (h + h[::-1] - 1.0), n * 0.5)

def _active_grad_C(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 8.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += int(pos.size)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += int(pos.size)
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv
    wN *= inv
    return _grad_from_weights(h, wP, wN)

def _local_search(h0, rng, deadline, start_step=2.0e-5):
    tl0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    P, Nv, _ = _pn_arrays(h)
    g = _active_grad_C(h, P, Nv)
    order = np.argsort(g)
    q = max(5, n // 8)
    low = order[:q]
    high = order[-q:]
    recompute = 2048
    it = 0
    while time.time() < deadline:
        if it >= recompute:
            c = _c5_direct(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            P, Nv, _ = _pn_arrays(h)
            g = _active_grad_C(h, P, Nv)
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            recompute = it + 2048
        frac = (time.time() - tl0) / max(1e-9, deadline - tl0)
        base = max(1e-10, start_step * (1e-5) ** frac)
        if rng.random() < 0.7:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i != j:
            if g[i] <= g[j]:
                s = 1
            elif g[j] < g[i]:
                s = -1
            else:
                s = 1 if rng.random() < 0.5 else -1
            if s > 0:
                avail = min(1.0 - h[i], h[j])
            else:
                avail = min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                mult = 0.4 + 1.4 * rng.random()
                if rng.random() < 0.03:
                    mult *= 8.0
                d = s * min(avail, base * mult)
                h[i] += d
                h[j] -= d
                cn = _c5_direct(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
        it += 1
    return best_h, best_c
