    alpha = alpha0

        if time.time() > deadline:
            break
        if it > 0 and it % 300 == 0:
            alpha = min(alpha * grow, 80000000.0)
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(hp - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _coord_sweep(h, deadline):
    """Coordinate-wise descent with sum constraint."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for sweep in range(200):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            old_i = h[i]
            # Try moving mass between i and a partner
            for j_off in [1, 2, 3]:
                j = (i + j_off) % n
                if j == i:
                    continue
                # Try small transfer
                for d in [1e-5, -1e-5, 5e-6, -5e-6]:
                    ni = h[i] + d
                    nj = h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < c5c - 1e-18:
                            c5c = c5n
                            if c5n < bc:
                                bc = c5n
                                bh = h.copy()
                            break
                        h[i], h[j] = ni - d, nj + d
                    if time.time() > deadline:
                        break
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_base = 1e-6
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = step_base * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step_base = max(step_base * 0.95, 1e-8)
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP refinement (0-180s)
    dl1 = min(deadline, t0 + 180)
    for alpha, tk, mi in [(3000, 50, 500), (15000, 30, 400), (60000, 15, 350)]:
        if time.time() > dl1:
            break
        h = _slsqp(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Adaptive Nesterov (180-480s)
    dl2 = min(deadline, t0 + 480)
    h_w = best_h.copy()
    for alpha0, grow, tk, st, ns in [
        (5000, 2.0, 35, 0.004, 3000),
        (50000, 1.5, 18, 0.002, 3000),
        (400000, 1.3, 10, 0.001, 3000),
        (2000000, 1.2, 6, 0.0006, 2000)
    ]:
        if time.time() > dl2:
            break
        h_w = _nesterov_adaptive(h_w, alpha0, grow, tk, ns, st, dl2)
        c5v = _eval(h_w)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_w.copy()

    # Phase 3: Block annealing with diverse blocks (480-720s)
    dl3 = min(deadline, t0 + 720)
    blocks = [1, 2, 3, 5, 7, 10, 15, 25, 40]
    h_a, c5_a = _block_anneal(best_h.copy(), 60000, rng, dl3, 0.005, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5a = c5_a
        best_h = h_a.copy()

    # Phase 4: Coordinate sweep (720-860s)
    dl4 = min(deadline, t0 + 860)
    h_cs, c5_cs = _coord_sweep(best_h.copy(), dl4)
    if c5_cs < best_c5:
        best_c5 = c5_cs
        best_h = h_cs.copy()

    # Phase 5: High-alpha Nesterov polish (860-950s)
    dl5 = min(deadline, t0 + 950)
    h_n2 = _nesterov_adaptive(best_h.copy(), 5000000, 1.1, 4, 2500, 0.0004, dl5)
    c5v = _eval(h_n2)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_n2.copy()

    # Phase 6: Fine polish (950-end)
    dl6 = deadline
    rem = max(0, dl6 - time.time())
    n_pol = min(int(rem / 0.0015), 60000)
    best_h, best_c5 = _fine_polish(best_h.copy(), n_pol, rng, dl6)

    # Final repair and eval
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
