from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_corr_k(h, k, n):
    """Gradient of correlation at lag k with respect to h"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += dx * (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= dx * h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += dx * (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= dx * h[i2 + sp]
    return g

def _lse_obj_and_grad(h, tau, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    w = exp_c / exp_c.sum()
    obj = c_max + tau * np.log(exp_c.sum()) + penalty * (h.sum() - n * 0.5) ** 2
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_corr_k(h, k, n)
    g += 2.0 * penalty * (h.sum() - n * 0.5)
    return obj, g

def _lbfgs_lse(h, tau, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _lse_obj_and_grad(x, tau, penalty)[0]
    
    def grad(x):
        return _lse_obj_and_grad(x, tau, penalty)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16,
                       'gtol': 1e-14, 'maxls': 30})
        out = np.clip(res.x, 0, 1)
        return _proj_sum(out, n * 0.5)
    except Exception:
        return _proj_sum(h, n * 0.5)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_corr_k(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += dx * (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= dx * h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += dx * (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= dx * h[i2 + sp]
    return g

def _lse_obj(x, tau, penalty):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    cm = c.max()
    ec = np.exp((c - cm) / tau)
    return cm + tau * np.log(ec.sum()) + penalty * (x.sum() - n * 0.5) ** 2

def _lse_grad(x, tau, penalty):

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    ec = np.exp((c - cm) / tau)
    w = ec / ec.sum()
    g = np.zeros(n)

            continue
        g += w[k] * _grad_corr_k(x, k, n)

        return _proj_sum(np.clip(res.x, 0, 1), n * 0.5)
