import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _c_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -350, 350)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -350, 350)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, kk in enumerate(idx):
        if w[ii] < 1e-16:
            continue
        sh = kk - (n - 1)
        j = np.arange(n)
        ip = (j + sh) % n
        im = (j - sh) % n
        valid_p = (j + sh >= 0) & (j + sh < n)
        valid_m = (j - sh >= 0) & (j - sh < n)
        g += w[ii] * dx * ((1.0 - x[ip]) * valid_p - x[im] * valid_m)
    return g

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      args=(alpha, top_k),
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _pgd_momentum(h, alpha, top_k, step0, niter, mu=0.75):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    for _ in range(niter):
        g = _smooth_grad(h, alpha, top_k)
        v = mu * v - step0 * g
        trial = _proj_sum(h + v, target)

            h = trial
            cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            v *= 0.5
    return best_h, best_c

def _anneal(h, niter, rng, T_hi=1e-5, T_lo=1e-9):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    cool = (T_lo / T_hi) ** (1.0 / max(niter, 1))

        new = np.clip(old + T * 12 * rng.randn(bs), 0, 1)
        new += (old.sum() - new.sum()) / bs
        new = np.clip(new, 0, 1)
        new += (old.sum() - new.sum()) / bs
        new = np.clip(new, 0, 1)
        h[i:i+bs] = new
        c_new = _eval(h)
        if c_new < cur_c or rng.rand() < np.exp(min(0, -(c_new - cur_c) / max(T, 1e-16))):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i:i+bs] = old
    return best_h, best_c

def _fine_local(h, niter, rng, step=3e-7):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for _ in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

            if c_new < cur_c:

                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _block_opt(h, bsz, maxit):
    n = len(h)
    nb = n // bsz
    h = h.copy()
    for _pass in range(3):
        for b in range(nb):
            sl = slice(b * bsz, (b+1) * bsz)
            x0 = h[sl].copy()
            bsum = x0.sum()
            def ob(x, _sl=sl, _h=h):
                ht = _h.copy()
                ht[_sl] = x
                return _eval(ht)
            def oc(x, _bs=bsum):
                return x.sum() - _bs
            def oj(x, _bs=bsum):
                return np.ones(bsz)
            try:
                cons = [{'type': 'eq', 'fun': oc, 'jac': oj, 'args': (bsum,)}]
                res = minimize(ob, x0, method='SLSQP',
                             bounds=[(0,1)]*bsz, constraints=cons,
                             options={'maxiter': maxit, 'ftol': 1e-16})
                h[sl] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)
