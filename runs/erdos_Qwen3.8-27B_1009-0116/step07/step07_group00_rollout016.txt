    # Phase 4: Block annealing
    dl4 = min(deadline, t0 + 750)
    blocks = [2, 3, 4, 5, 7, 10]
    h_a, c5_a = _anneal(best_h.copy(), 50000, rng, dl4, 0.004, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 5: High-alpha FISTA
    dl5 = min(deadline, t0 + 920)
    for a, k, L0, iters in [(50000000, 3, 500, 6000), (200000000, 2, 400, 6000)]:
        if time.time() > dl5:
            break
        h = _fista(best_h.copy(), a, k, iters, dl5, L0)
        h = _repair(h, n0)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Gradient-based polish
    dl6 = deadline
    rem = max(0.0, dl6 - time.time())
    n_gp = min(int(rem * 80), 50000)
    best_h, best_c5 = _grad_polish(best_h, n_gp, 5000000, 3, dl6)

    # Phase 7: Fine 2-point polish
    rem = max(0.0, dl6 - time.time())
    n_fp = min(int(rem * 200), 40000)
    best_h, best_c5 = _fine_polish(best_h, n_fp, rng, dl6)

    # Phase 8: 3-point search
    rem = max(0.0, dl6 - time.time())
    if rem > 2:
        n_3 = min(int(rem * 150), 20000)
        best_h, best_c5 = _3pt(best_h, n_3, rng, dl6)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
