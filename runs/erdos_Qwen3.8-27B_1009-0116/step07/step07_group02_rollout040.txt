        for step in [0.03, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-18:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _window_search(h, n_iter, rng, deadline, wmin=2, wmax=6):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        ws = rng.randint(wmin, wmax + 1)
        i = rng.randint(max(1, n - ws + 1))
        d = rng.randn(ws) * 4e-5
        d -= d.mean()
        ni = h[i:i + ws].copy() + d
        if np.all(ni >= -1e-12) and np.all(ni <= 1.0 + 1e-12):
            h[i:i + ws] = np.clip(ni, 0, 1)
            if abs(h.sum() - n * 0.5) < 1e-8:
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i:i + ws] = ni - d
            else:
                h[i:i + ws] = ni - d
    return best_h, best_c

def _point_search(h, n_iter, rng, deadline, scale=5e-6):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, max(1, n // 8))) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _targeted_attack(h, deadline):
    """Focus on reducing the single maximum correlation."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(2000):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        kmax = np.argmax(c)
        sh = (n - 1) - kmax
        g = np.zeros(n)
        if sh >= 0:
            np_ = n - sh
            g[:np_] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:np_]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:-sp] -= h[sp:]
        g *= dx
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-18:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick L-BFGS (2 alphas)
    h = h0.copy()
    for alpha in [5000, 50000]:
        if time.time() > deadline - 900:
            break
        h = _lbfgs(h, alpha, 2000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Top-k descent with momentum (multiple configs)
    h = best_h.copy()
    for (nl, st, sp0, tol) in [(5, 600, 0.015, 1e-9), (10, 500, 0.01, 1e-8),
                                (20, 400, 0.008, 1e-7), (50, 300, 0.005, 1e-6)]:
        if time.time() > deadline - 500:
            break
        h_tk, c5_tk = _topk_descent(h, nl, st, sp0, tol, deadline, mom=0.6)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()
        h = best_h.copy()

    # Phase 3: Line search top-k
    if time.time() < deadline - 400:
        for nl in [3, 5, 10]:
            if time.time() > deadline - 400:
                break
            h_ls, c5_ls = _line_search_topk(best_h.copy(), nl, deadline)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 4: Windowed local search (diverse window sizes)
    if time.time() < deadline - 250:
        h_ws, c5_ws = _window_search(best_h.copy(), 150000, rng, deadline, 2, 5)
        if c5_ws < best_c5:
            best_c5 = c5_ws
            best_h = h_ws.copy()
        h_ws, c5_ws = _window_search(best_h.copy(), 80000, rng, deadline, 3, 7)
        if c5_ws < best_c5:
            best_c5 = c5_ws
            best_h = h_ws.copy()

    # Phase 5: Targeted single-lag attack
    if time.time() < deadline - 200:
        h_ta, c5_ta = _targeted_attack(best_h.copy(), deadline)
        if c5_ta < best_c5:
            best_c5 = c5_ta
            best_h = h_ta.copy()

    # Phase 6: More top-k passes after local search
    if time.time() < deadline - 150:
        for (nl, st, sp0, tol) in [(5, 400, 0.01, 1e-10), (10, 300, 0.008, 1e-9)]:
            if time.time() > deadline - 150:
                break
            h_tk, c5_tk = _topk_descent(best_h.copy(), nl, st, sp0, tol, deadline, mom=0.5)
            if c5_tk < best_c5:
                best_c5 = c5_tk
                best_h = h_tk.copy()

    # Phase 7: Extended 2-point search with decreasing scale
    if time.time() < deadline - 60:
        for scale in [5e-6, 2e-6, 1e-6, 5e-7, 2e-7]:
            if time.time() > deadline - 40:
                break
            h_ps, c5_ps = _point_search(best_h.copy(), 60000, rng, deadline, scale)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()

    # Phase 8: Final targeted + window polish
    if time.time() < deadline - 20:
        h_ta, c5_ta = _targeted_attack(best_h.copy(), deadline)
        if c5_ta < best_c5:
            best_c5 = c5_ta
            best_h = h_ta.copy()
        if time.time() < deadline - 5:
            h_ws, c5_ws = _window_search(best_h.copy(), 40000, rng, deadline, 2, 4)
            if c5_ws < best_c5:
                best_c5 = c5_ws
                best_h = h_ws.copy()
            h_ps, c5_ps = _point_search(best_h.copy(), 50000, rng, deadline, 1e-7)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
