import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:sp] -= h[sp:]
    return g

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(x, k)
    return g * dx

def _topk_obj(x, top_k, alpha):
    c = _cvals(x)
    ct = c[top_k]

    s = np.clip(alpha * (ct - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

    s = np.clip(alpha * (ct - m), -400, 0)

        if wk < 1e-14:
            continue
        g += wk * _grad_lag(x, ki)
    return g * dx

    return (np.sum(c ** p)) ** (1.0 / p)

    n = len(x)

    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(x, k)
    return g * dx
