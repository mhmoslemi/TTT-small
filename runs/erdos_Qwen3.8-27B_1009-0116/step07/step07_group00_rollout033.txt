import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_and_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    smax = m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    w = np.exp(s - s.max())
    w /= w.sum()
    # Gradient
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return smax, g, m, idx, w

def _nesterov_fista(h, alpha, topk, steps, start_step, deadline):
    n = len(h)
    target = n * 0.5
    f, g, m, _, _ = _smax_and_grad(h, alpha, topk)
    v = np.zeros(n)
    step = start_step
    momentum = 0.85
    for it in range(steps):
        if time.time() > deadline:
            break
        momentum = 0.85 * 0.999 ** it
        v = momentum * v + (1.0 - momentum) * g
        hn = _proj(h - step * v, target)
        fn, gn, mn, _, _ = _smax_and_grad(hn, alpha, topk)
        if fn < f - 1e-18:
            h, g, f = hn, gn, fn
            step = min(step * 1.01, start_step * 10)
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f, g, _, _, _ = _smax_and_grad(h, alpha, topk)
    return _repair(h, n)

def _max_chase(h, n_steps, deadline):
    """Directly chase the maximum lag with directed perturbations."""
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = c.argmax()
        sh = (n - 1) - k_star
        # Gradient of the max lag
        gk = np.zeros(n)
        if sh >= 0:
            gk[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                gk[sh:] -= h[:-sh]
        else:
            sp = -sh
            gk[sp:] += (1.0 - h[:-sp])
            gk[:n-sp] -= h[sp:]
        # Normalized step
        gn = np.linalg.norm(gk)
        if gn < 1e-12:
            break
        gk /= gn
        # Try reducing this lag
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            hn = _proj(h - step * gk, target)
            cn = _eval(hn)
            if cn < best_c - 1e-15:
                h = hn
                best_c = cn
                best_h = h.copy()
                break
    return best_h, best_c

def _lbfgs_step(h, alpha, topk, maxiter, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        val, _, m, _, _ = _smax_and_grad(x, alpha, topk)
        # Penalty for constraint violation
        pen = 1000.0 * (x.sum() - n*0.5)**2
        return val + pen
    
    def grad(x):
        val, g, m, _, _ = _smax_and_grad(x, alpha, topk)
        g += 2000.0 * (x.sum() - n*0.5)
        return g
    
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)]*n,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return _repair(r.x, n)
    except:
        return h

def _restart_optimize(h, n, amplitude, alpha, topk, steps, deadline, rng):
    """Perturb, optimize, keep if better."""
    c_cur = _eval(h)
    h_pert = h + amplitude * rng.randn(n)
    h_pert = _proj(h_pert, n*0.5)
    h_opt = _nesterov_fista(h_pert, alpha, topk, steps, 0.003, deadline)
    c_new = _eval(h_opt)
    if c_new < c_cur:
        return h_opt, c_new
    return h, c_cur

def _coord_descent(h, niter, deadline):
    """Coordinate descent focusing on the max lag."""
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _eval(h)
    for it in range(niter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = c.argmax()
        sh = (n - 1) - k_star
        # Find the pair (i, i+sh) that contributes most
        if sh >= 0:
            contrib = h[:n-sh] * (1.0 - h[sh:])
        else:
            sp = -sh
            contrib = h[sp:] * (1.0 - h[:-sp])
        i_max = contrib.argmax()
        # Try adjusting h[i_max] and h[i_max+sh]
        for delta in [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]:
            hn = h.copy()
            if sh >= 0:
                j1, j2 = i_max, i_max + sh
            else:
                j1, j2 = i_max - sh, i_max
            v1 = hn[j1] + delta
            v2 = hn[j2] - delta
            if 0 <= v1 <= 1 and 0 <= v2 <= 1:
                hn[j1], hn[j2] = v1, v2
                cn = _eval(hn)
                if cn < best_c - 1e-15:
                    h = hn
                    best_c = cn
                    best_h = h.copy()
                    break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_obj(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    target = n * 0.5
    f = _smax_obj(h, alpha, topk)
    g = _smax_grad(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    momentum = 0.85
    for it in range(steps):
        if time.time() > deadline:
            break
        momentum = min(0.85, 0.4 + 0.45 * (0.999 ** it))
        v = momentum * v + (1.0 - momentum) * g
        hn = _proj(h - step * v, target)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-18:
            h, g, f = hn, _smax_grad(hn, alpha, topk), fn
            step = min(step * 1.015, step0 * 12.0)
        else:
            step *= 0.45
            h = _proj(h - step * v, target)
            f = _smax_obj(h, alpha, topk)
            g = _smax_grad(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj_fn(x):
        return _smax_obj(x, alpha, topk)
    def grad_fn(x):
        return _smax_grad(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        r = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _max_chase(h, niter, deadline):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(niter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = c.argmax()
        sh = (n - 1) - k_star
        gk = np.zeros(n)
        if sh >= 0:
            gk[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                gk[sh:] -= h[:-sh]
        else:
            sp = -sh
            gk[sp:] += (1.0 - h[:-sp])
            gk[:n - sp] -= h[sp:]
        gn = np.linalg.norm(gk)
        if gn < 1e-12:
            break
        gk /= gn
        accepted = False
        for step in [0.03, 0.015, 0.007, 0.003, 0.001]:
            hn = _proj(h - step * gk, target)
            cn = _eval(hn)
            if cn < best_c - 1e-15:
                h, best_c = hn, cn
                best_h = h.copy()
                accepted = True
                break
        if not accepted:
            # Try a small random perturbation blended with gradient
            rp = 0.004 * np.random.randn(n)
            for step in [0.01, 0.005, 0.002]:
                hn = _proj(h - step * (gk * 0.5 + rp * 0.5), target)
                cn = _eval(hn)
                if cn < best_c - 1e-15:
                    h, best_c = hn, cn
                    best_h = h.copy()
                    break
    return best_h, best_c

def _structured_restart(h, n, n_restarts, deadline, rng):
    best_h = h.copy()
    best_c = _eval(h)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Structured perturbation: sine at random frequency + noise
        freq = rng.randint(1, 8)
        phase = 2.0 * np.pi * rng.random()
        amp = 0.03 + 0.04 * rng.random()
        perturb = amp * np.sin(freq * np.pi * np.arange(n) / n + phase)
        perturb += 0.01 * rng.randn(n)
        hn = _proj(h + perturb, n * 0.5)
        hn = _nesterov(hn, 50000, 8, 1500, 0.003, deadline)
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < best_c:
            best_c = cn
            best_h = hn.copy()
    return best_h, best_c

def _coord_descent(h, niter, deadline):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(niter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = c.argmax()
        sh = (n - 1) - k_star
        if sh >= 0:
            contrib = h[:n - sh] * (1.0 - h[sh:])
            i_max = contrib.argmax()
        else:
            sp = -sh
            contrib = h[sp:] * (1.0 - h[:-sp])
            i_max = contrib.argmax()
        if sh >= 0:
            j1, j2 = i_max, i_max + sh
        else:
            j1, j2 = i_max + sh, i_max
        for delta in [0.015, 0.008, 0.003, 0.001, -0.015, -0.008, -0.003, -0.001]:
            v1, v2 = h[j1] + delta, h[j2] - delta
            if 0.0 <= v1 <= 1.0 and 0.0 <= v2 <= 1.0:
                h[j1], h[j2] = v1, v2
                cn = _eval(h)
                if cn < best_c - 1e-15:
                    best_c = cn
                    best_h = h.copy()
                    break
                else:
                    h[j1], h[j2] = v1 - delta, v2 + delta
    return best_h, best_c

def _fine_2pt(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = 1.5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
