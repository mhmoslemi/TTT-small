    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    mx = c.max()
    active = np.where(c >= mx - tol)[0]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _adam(h_init, alpha, topk, steps, lr, deadline):
    n = len(h)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    h = h_init.copy()
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        m_arr = beta1 * m_arr + (1 - beta1) * g
        v_arr = beta2 * v_arr + (1 - beta2) * g * g
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = _proj(h - step, n * 0.5)
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 5000
_g_topk = 30

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _obj_w(x):
    return _smax_obj(x, _g_alpha, _g_topk)

def _jac_w(x):
    return _smax_grad(x, _g_alpha, _g_topk)

def _constraint_sum(x):
    return x.sum() - len(x) * 0.5

def _jac_constraint(x):
    return np.ones(len(x))

def _lbfgs(h, alpha, topk, maxit, deadline):
    global _g_alpha, _g_topk
    n = len(h)
    _g_alpha = alpha
    _g_topk = topk
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': _constraint_sum, 'jac': _jac_constraint}]
    try:
        r = minimize(_obj_w, h, jac=_jac_w, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-14})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h_init, alpha, topk, steps, lr, deadline):
    n = len(h_init)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    h = h_init.copy()
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        mc = m_arr / (1 - b1 ** t)
        vc = v_arr / (1 - b2 ** t)
        h = _proj(h - lr * mc / (np.sqrt(vc) + eps), n * 0.5)
    return _repair(h, n)

def _anneal_struct(h, niter, rng, deadline, T0):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):

        elif move_type == 1:  # mirror a block

    n = len(h)
    m = n // factor
    chunks = [h[i*factor:(i+1)*factor] for i in range(m)]

    m = n * factor
    result = np.zeros(m)
    for i in range(n):

    return result

from scipy.optimize import minimize

_ga = 5000

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a
                    d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a
                    d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _sobj(h, a, tk):
    c = _cor(h)
    m = c.max()
    if tk < len(c):
        idx = np.argpartition(c, -tk)[-tk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(a * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / a

    n = len(h)
    c = _cor(h)
    m = c.max()

        else:

def _adam(hi, a, tk, st, lr, dl):

    v = np.zeros(n)

        h = _proj(h - lr * m / (1 - b1**t) / (np.sqrt(v / (1 - b2**t)) + eps), n * 0.5)
    return _repair(h, n)
