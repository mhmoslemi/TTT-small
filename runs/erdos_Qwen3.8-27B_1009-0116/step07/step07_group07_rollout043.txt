    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    shifted = tau * (c - c.max())  # for numerical stability
    logse = np.log(np.exp(shifted).sum()) / tau
    return logse + penalty * (h.sum() - n*0.5)**2

def _logsumexp_grad(h, tau, penalty):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    shifted = tau * (c - c.max())
    weights = np.exp(shifted) / np.exp(shifted).sum()  # softmax

        sh = (n-1) - k
        if sh >= 0:

            i2 = np.arange(n-sp)
            g[i2] -= wk * h[i2+sp]
    gp = 2.0 * penalty * (h.sum() - n*0.5)

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _lse_obj(h, tau, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = tau * (c - m)
    val = m + np.log(np.exp(sh).sum()) / tau
    return val + penalty * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, tau, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = tau * (c - m)
    e = np.exp(sh)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        s = (n - 1) - k
        if s >= 0:
            cnt = n - s
            g[:cnt] += wk * (1.0 - h[s:s + cnt])
            if s > 0:
                g[s:] -= wk * h[:cnt]
        else:
            sp = -s
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp + cnt]
    return g + 2.0 * penalty * (h.sum() - n * 0.5)

def _topk_grad(h, topk_idx):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in topk_idx:
        s = (n - 1) - k
        if s >= 0:
            cnt = n - s
            g[:cnt] += (1.0 - h[s:s + cnt])
            if s > 0:
                g[s:] -= h[:cnt]
        else:
            sp = -s
            cnt = n - sp
            g[sp:] += (1.0 - h[:cnt])
            g[:cnt] -= h[sp:sp + cnt]
    return g * dx / len(topk_idx)

def _lbfgs(h, fn, jac, args, maxiter, bnds):
    try:
        res = minimize(fn, h, jac=jac, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 40})
        return res.x
    except Exception:
        return h.copy()

def _adam_subgrad(h, n_iter, k_top, lr0, el, bc):
    n = len(h)
    h_c = h.copy().astype(float)
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_c = _eval(h_c)
    best_h = h_c.copy()
    lr = lr0
    for t in range(1, n_iter + 1):
        if bc():
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-k_top:]
        g = _topk_grad(h_c, topk)
        lr_t = lr0 * (0.999 ** t)
        m_s = b1 * m_s + (1 - b1) * g
        v_s = b2 * v_s + (1 - b2) * g * g
        mh = m_s / (1 - b1 ** t)
        vh = v_s / (1 - b2 ** t)
        h_try = _proj_sum(h_c - lr_t * mh / (np.sqrt(vh) + eps), n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h_c = h_try

def _local2(h, n_iter, scale0, rng, el, bc):

    h_c = h.copy().astype(float)
    c_best = _eval(h_c)
    best_h = h_c.copy()
    scale = scale0
    for i in range(n_iter):

            break
        a = rng.randint(n)

        na, nb = h_c[a] + d, h_c[b_] - d
        if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:

                best_h = h_c.copy()
            else:
                h_c[a], h_c[b_] = na - d, nb + d
        if i % 15000 == 14999:
            scale *= 0.65
    return best_h, c_best
