```python
import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    # Ensure 0<=h<=1 and sum(h)=n/2
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Project to correct sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _softmax_obj(h, alpha, n):
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _softmax_grad(h, alpha, n):
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g
```

```python
        j = (i + rng.randint(2, 8)) % n
        k = (i + rng.randint(9, 18)) % n
        d1, d2 = 2e-5 * rng.randn(), 2e-5 * rng.randn()
        a, b2, c2 = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= a <= 1.0 and 0.0 <= b2 <= 1.0 and 0.0 <= c2 <= 1.0:
            h[i], h[j], h[k] = a, b2, c2
            f = _eval(h)
            if f < bf:
                bf, bh = f, h.copy()
            else:
                h[i], h[j], h[k] = a - d1, b2 - d2, c2 + d1 + d2
    return bh, bf

def _local2(h, n_tries, rng, t0, tlim):
    n = len(h)
    bf, bh = _eval(h), h.copy()
    for _ in range(n_tries):
        if time.time() - t0 > tlim:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 6e-6 * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
            h[i], h[j] = a, b
            f = _eval(h)
            if f < bf:
                bf, bh = f, h.copy()
            else:
                h[i], h[j] = a - d, b + d
    return bh, bf

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=np.float64).copy(), n)
    best_h, best_c = h.copy(), _eval(h)

    # Phase 1: Adam alpha=200 wide smoothing
    h, c = _adam(h, 200, 8000, 0.012, t0, 90)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 2: Adam alpha=2000
    h, c = _adam(best_h.copy(), 2000, 5000, 0.006, t0, 200)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 3: Coordinate descent (exact 1D per coordinate)
    h, c = _coord(best_h.copy(), 20, t0, 450)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 4: Adam alpha=15000
    h, c = _adam(best_h.copy(), 15000, 3000, 0.003, t0, 580)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 5: Coordinate descent again
    h, c = _coord(best_h.copy(), 20, t0, 780)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 6: Adam alpha=80000 sharp
    h, c = _adam(best_h.copy(), 80000, 2000, 0.001, t0, 880)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 7: Random restarts with re-optimization
    for _r in range(4):
        if time.time() - t0 > budget_s - 150:
            break
        h_r = best_h + 0.015 * rng.randn(n)
        h_r = _repair(h_r, n)
        h_r, c_r = _adam(h_r, 5000, 1500, 0.005, t0, budget_s - 250)
        h_r, c_r = _coord(h_r, 5, t0, budget_s - 100)
        if c_r < best_c:
            best_h, best_c = h_r.copy(), c_r

    # Phase 8: Final coordinate descent
    h, c = _coord(best_h.copy(), 25, t0, budget_s - 40)
    if c < best_c:
        best_h, best_c = h.copy(), c

    # Phase 9: 3-point local search
    rem = budget_s - 3 - (time.time() - t0)
    nt3 = int(min(rem / 0.005, 30000))
    if nt3 > 200:
        best_h, best_c = _local3(best_h, nt3, rng, t0, budget_s - 3)

    # Phase 10: 2-point ultra-fine
    rem = budget_s - 1 - (time.time() - t0)
    nt2 = int(min(rem / 0.003, 20000))
    if nt2 > 100:
        best_h, best_c = _local2(best_h, nt2, rng, t0, budget_s - 0.5)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
```
