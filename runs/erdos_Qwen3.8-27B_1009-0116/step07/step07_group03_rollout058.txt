    c = np.correlate(h, 1.0 - h, mode='full') * dx

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a

    if time.time() < t_cd2_end:
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, t_cd2_end, rng, T0=20)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Fine local search (18% time)
    t_fs_end = t0 + budget_s * 0.90
    if time.time() < t_fs_end:
        h = best_h.copy()
        best_h, best_c5 = _fine_search(h, n, t_fs_end, rng)

    # Phase 5: Final coord polish (8% time)
    t_final = t0 + budget_s - 4
    if time.time() < t_final:
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, t_final, rng, T0=18)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Tiny final tweak
    if time.time() < t0 + budget_s - 1:
        h = best_h.copy()
        best_h, best_c5 = _fine_search(h, n, t0 + budget_s - 0.5, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
