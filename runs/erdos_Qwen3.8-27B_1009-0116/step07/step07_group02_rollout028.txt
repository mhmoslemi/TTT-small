def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Binary search for threshold
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _grad_lag(h, k):
    """Gradient of the k-th correlation value w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14 * w_sum:
            continue
        wk = w[k] / w_sum
        g += wk * _grad_lag(x, k)
    return g

    c = np.maximum(_cvals(x), 0.0)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _grad_k(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        ww = w[k] / wsum
        if ww < 1e-14:
            continue
        g += ww * _grad_k(x, k)
    return g

def _lbfgs(h, func, grad, args, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 30:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h, jac=grad, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except:
        return h

def _boundary_walk(h, max_rounds, deadline):
    """For near-binary step functions, walk boundaries to improve C5"""
    n = len(h)
    c5 = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 5:
            break
        # Find transition points (where value changes significantly)
        improved = False
        for i in range(n - 1):
            if time.time() > deadline - 5:
                break
            # Try swapping h[i] and h[i+1] (if they're very different)
            if abs(h[i] - h[i+1]) < 0.1:
                continue
            # Try: move boundary right (h[i] becomes like h[i+1])
            hi_save = h[i]
            hj_save = h[i+1]
            # Make h more binary at the boundary
            if h[i] > 0.5:
                h[i] = min(h[i] + 0.5, 1.0)
                h[i+1] = max(h[i+1] - 0.5, 0.0)
            else:
                h[i] = max(h[i] - 0.5, 0.0)
                h[i+1] = min(h[i+1] + 0.5, 1.0)
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                improved = True
            else:
                h[i] = hi_save
                h[i+1] = hj_save
                h = _repair(h, n)
        if not improved:
            break
    return h, c5

def _local2(h, n_iter, rng, deadline, scale):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _block_pattern(n, nblocks, rng):
    """Create a step pattern with nblocks 1-regions"""
    h = np.zeros(n)
    ones_total = n // 2
    per_block = ones_total // nblocks
    remainder = ones_total - per_block * nblocks
    # Randomly place blocks
    positions = sorted(rng.choice(n, nblocks, replace=False))
    for i, pos in enumerate(positions):
        w = per_block + (1 if i < remainder else 0)
        for j in range(w):
            h[(pos + j) % n] = 1.0
    return _repair(h, n)

    m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

    return np.repeat(h, factor).copy()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _grad_k(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        ww = w[k] / wsum
        if ww < 1e-14:
            continue
        g += ww * _grad_k(x, k)
    return g

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14 * wsum:
            continue
        g += (w[k] / wsum) * _grad_k(x, k)
    return g

def _lbfgs(h, func, grad, args, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 30:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h, jac=grad, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_subgrad(h, steps, step0, mom, tol, deadline):
    n = len(h)
    v = np.zeros(n)
    for i in range(steps):
        if time.time() > deadline - 20:
            break
        t = step0 / (1.0 + 0.04 * i)
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for k in active:
            g += _grad_k(h, k)
        g /= len(active)
        v = mom * v + (1.0 - mom) * g
        h = _repair(h - t * v, n)
    return h

def _boundary_search(h, max_rounds, deadline):
    n = len(h)
    c5 = _eval(h)

            break

            h[i+1] = max(h[i+1] - 0.3, 0.0)

                h[i+1] = sv[1]
                h = _repair(h, n)
            transitions = np.where(np.abs(np.diff(h)) > 0.05)[0]
        if not improved:
            break
    return h, c5

def _bit_swap(h, n_iter, rng, deadline):

    best_h = h.copy()
    best_c = c5c

    for _ in range(n_iter):

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c, ones, zeros

def _local2(h, n_iter, rng, deadline, scale):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _block_pattern(n, nblocks, rng):
    h = np.zeros(n)
    ones_total = n // 2
    per_block = ones_total // nblocks
    rem = ones_total - per_block * nblocks
    positions = sorted(rng.choice(n, nblocks, replace=False))
    for idx, pos in enumerate(positions):
        w = per_block + (1 if idx < rem else 0)
        for j in range(w):
            h[(pos + j) % n] = 1.0
    return _repair(h, n)

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor, n_target):
    out = np.repeat(h, factor)
    if len(out) >= n_target:
        return out[:n_target].copy()
    return np.pad(out, (0, n_target - len(out)), mode='edge').copy()
