def _grad_lag(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += dx * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx * h[:m]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:-sp])
        g[:-sp] -= dx * h[sp:]
    return g

def _grad_topk(h, k_list):
    n = len(h)
    g = np.zeros(n)
    for k in k_list:
        g += _grad_lag(h, k, n)
    g /= len(k_list)
    return g

def _grad_smooth(h, k_list, temp):
    """Weighted gradient with softmax over lags"""
    n = len(h)
    c = _cvals(h)
    w = np.exp(c[k_list] / temp)
    w /= w.sum()
    g = np.zeros(n)
    for j, k in enumerate(k_list):
        g += w[j] * _grad_lag(h, k, n)
    return g, c[k_list]

def _pd_h(h, k_list, n_steps, base_step, deadline, rng=None):
    """Projected gradient descent on top-k"""
    n = len(h)
    best_c5 = _eval(h)
    best = h.copy()
    for i in range(n_steps):
        if time.time() > deadline - 5:
            break
        step = base_step / (1.0 + 0.1 * np.sqrt(i))
        g = _grad_topk(h, k_list)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        # Tentative step
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _repair(h - mult * step * g / gnorm, n)
            c5_new = _eval(h_new)
            if c5_new < best_c5 - 1e-18:
                h = h_new
                best_c5 = c5_new
                best = h.copy()
                break
    return best, best_c5

def _ls_anneal(h, rng, n_iter, deadline, T0=1e-4, T1=1e-8):
    """Annealed local search"""
    n = len(h)
    c5_cur = _eval(h)
    best_c5 = c5_cur
    best_h = h.copy()
    
    for i in range(n_iter):
        if time.time() > deadline - 2:
            break
        T = T0 * (T1/T0) ** (i / max(n_iter-1, 1))
        # Get gradient for direction
        c = _cvals(h)
        top2 = np.argsort(c)[-2:]
        g = np.zeros(n)
        for k in top2:
            g += _grad_lag(h, k, n)
        g /= 2.0
        
        # Direction: negative gradient + noise
        scale = 1e-5
        d = -g * scale + T * rng.randn(n)
        
        h_new = _repair(h + d, n)
        c5_new = _eval(h_new)
        
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        elif rng.random() < np.exp(-(c5_new - c5_cur) / max(T * 10, 1e-15)):
            h = h_new
            c5_cur = c5_new
    
    return best_h, best_c5

def _pair_search(h, rng, n_iter, deadline):
    """2-point local search with gradient hint"""
    n = len(h)
    c5_cur = _eval(h)
    best_c5 = c5_cur
    best_h = h.copy()
    
    for i in range(n_iter):

    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
