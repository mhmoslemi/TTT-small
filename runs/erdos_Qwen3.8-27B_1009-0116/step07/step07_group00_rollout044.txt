def _repair(h, n):
    # Clamp and adjust sum
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    diff -= a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nest(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)

        fn = _smax(hn, alpha, topk)

            h = hn
            f = fn

                bh = h.copy()
