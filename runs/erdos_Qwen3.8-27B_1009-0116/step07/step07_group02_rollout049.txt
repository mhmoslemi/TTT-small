def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _lag_grad(h, k):
    """Grad of c_k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _lag_grad(x, k)
    return g

def _lbfgs(h, alpha, maxit, deadline):

    if time.time() > deadline - 30:

    bnds = [(0.0, 1.0)] * n

                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

    """Subgradient on top-k lags with backtracking."""
    n = len(h)
    best_c5 = _eval(h)

        if time.time() > deadline - 30:
            break
        t = step0 / (1.0 + 0.05 * i)
        c = _cvals(h)
        top_k = np.argsort(c)[-k_count:]
        m = c[top_k[0]]  # min of top k
        # Only use lags within tol of max
        active = top_k[c[top_k] >= m - 1e-6]
        g = np.zeros(n)
        for k in active:
            g += _lag_grad(h, k)
        g /= len(active)
        # Backtracking line search

            if c5_try < best_c5:

                best_c5 = c5_try
                break
        h = best_h
    return h, best_c5
