def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

def _grad_lags(h, lags):

def _slsqp_grad(x, alpha, n):

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

        return _slsqp_obj(x, alpha, n)

def _weighted_grad(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            # c[k] = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh]) * dx
            # d/dh[j]: dx*(1-h[j+sh]) for j in 0..n-1-sh
            #          -dx*h[j-sh] for j in sh..n-1
            cnt = n - sh
            g[:cnt] += w[k] * dx * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= w[k] * dx * h[:n-sh]
        else:
            sp = -sh
            # c[k] = sum_{i=sp}^{n-1} h[i]*(1-h[i-sp]) * dx
            cnt = n - sp
            g[sp:] += w[k] * dx * (1.0 - h[:cnt])
            g[:cnt] -= w[k] * dx * h[sp:n]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)

            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize
import time

_SL_ALPHA = 1.0
_SL_TARGET = 0.0
_SL_N = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _wgrad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * dx * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * dx * (1.0 - h[:cnt])
            g[:cnt] -= wk * dx * h[sp:]
    return g

def _softmax_w(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _nesterov_sm(h, alpha, budget_s, t0, max_steps):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.75
    step = 0.002
    best_f = f
    best_h = h.copy()
    for _s in range(max_steps):
        if time.time() - t0 > budget_s:
            break
        c = _corr(h)
        w = _softmax_w(c, alpha)
        g = _wgrad(h, w)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.01, 0.02)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.5
            if step < 1e-9:
                step = 0.001
                v[:] = 0
    return best_h, best_f

def _slsqp_obj(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_SL_ALPHA * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / _SL_ALPHA

def _slsqp_grad(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_SL_ALPHA * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _wgrad(x, w)

def _slsqp_cfun(x):
    return x.sum() - _SL_TARGET

def _slsqp_cjac(x):
    return np.ones(_SL_N)

def _slsqp(h, alpha, maxiter, t0, budget_s):
    global _SL_ALPHA, _SL_TARGET, _SL_N
    n = len(h)
    _SL_ALPHA = alpha
    _SL_TARGET = n * 0.5
    _SL_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() - t0 > budget_s:
            return _repair(h, n)
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, budget_s, t0, rng):
    n = len(h)

    best_h = h.copy()

    step = 0.003

            if h_work[idxs[ii]] < 0 or h_work[idxs[ii]] > 1:
                valid = False
                break

            if c < c5c - 1e-15:

                c5c = c
                step = min(step * 1.02, 0.02)
                if c5c < best_c:
                    best_c = c5c

            else:
                step *= 0.97
                if step < 1e-9:
                    step = 0.002
        else:
            step *= 0.95
    return best_h, best_c

    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

    return np.repeat(h, f).copy()

import numpy as np
from scipy.optimize import minimize
import time

_SL_ALPHA = 1.0
_SL_TARGET = 0.0
_SL_N = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _wgrad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * dx * (1.0 - h[sh:])
            g[sh:] -= wk * dx * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * dx * (1.0 - h[:cnt])
            g[:cnt] -= wk * dx * h[sp:]
    return g

def _softmax_w(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _nesterov_sm(h, alpha, budget_s, t0, max_steps):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.75
    step = 0.005
    best_f = f
    best_h = h.copy()
    for _s in range(max_steps):
        if time.time() - t0 > budget_s:
            break
        c = _corr(h)
        w = _softmax_w(c, alpha)
        g = _wgrad(h, w)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.01, 0.05)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.5
            if step < 1e-9:
                step = 0.003
                v[:] = 0.0
    return best_h, best_f

def _slsqp_obj(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_SL_ALPHA * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / _SL_ALPHA

def _slsqp_grad(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_SL_ALPHA * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _wgrad(x, w)

def _slsqp_cfun(x):
    return x.sum() - _SL_TARGET

def _slsqp_cjac(x):
    return np.ones(_SL_N)

def _slsqp(h, alpha, maxiter, t0, budget_s):
    global _SL_ALPHA, _SL_TARGET, _SL_N
    n = len(h)
    _SL_ALPHA = alpha
    _SL_TARGET = n * 0.5
    _SL_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() - t0 > budget_s:
            return _repair(h, n)
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, budget_s, t0, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    step = 0.004
    stall = 0
    for _it in range(2000000):
        if time.time() - t0 > budget_s:
            break
        npts = rng.randint(2, 8)
        idxs = rng.choice(n, npts, replace=False)
        deltas = step * rng.randn(npts)
        deltas -= deltas.mean()
        h_work = h.copy()
        ok = True
        for ii in range(npts):
            h_work[idxs[ii]] += deltas[ii]
            if h_work[idxs[ii]] < 0 or h_work[idxs[ii]] > 1:
                ok = False
                break
        if ok:
            c = _eval(h_work)
            if c < c5c - 1e-15:
                h = h_work
                c5c = c
                step = min(step * 1.015, 0.03)
                stall = 0
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
            else:
                stall += 1
                if stall > 500:
                    step *= 0.6
                    stall = 0
                    if step < 1e-9:
                        step = 0.003
        else:
            step *= 0.95
    return best_h, best_c

def _down(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _up(h, f):
    return np.repeat(h, f).copy()

def _try_start(h_start, n, budget_s, t0):
    h_s = _repair(h_start, n)
    h_r, c_r = _nesterov_sm(h_s, 200, budget_s, t0, 1500)
    return h_r, c_r
