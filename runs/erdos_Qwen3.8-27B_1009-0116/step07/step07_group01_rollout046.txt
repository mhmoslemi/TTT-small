import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha, top_k=50):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    smooth_val = m + np.log(wsum) / alpha
    w /= wsum
    # Get top k indices by weight
    if top_k < len(w):
        idx_top = np.argpartition(w, -top_k)[-top_k:]
        idx_top = idx_top[w[idx_top] > 1e-12]
    else:
        idx_top = np.where(w > 1e-12)[0]
    g = np.zeros(n)
    for k in idx_top:
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        wk_dx = wk * dx
        if shift >= 0:
            g[:n-shift] += wk_dx * (1.0 - h[shift:])
            if shift > 0:
                g[shift:] -= wk_dx * h[:n-shift]
        else:
            sp = -shift
            g[sp:] += wk_dx * (1.0 - h[:n-sp])
            g[:n-sp] -= wk_dx * h[sp:]
    return g, smooth_val

def _adam_step(h, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * (grad * grad)
    mhat = m / (1 - b1**t)
    vhat = v / (1 - b2**t)
    return h - lr * mhat / (np.sqrt(vhat) + eps), m, v

def _adam_opt(h, alpha, n_steps, lr, top_k=50):
    n = len(h)
    target = n / 2.0
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = _eval(h)
    for step in range(n_steps):
        t += 1
        grad, sv = _grad_smooth(h, alpha, top_k)
        h_new, m_adam, v_adam = _adam_step(h, grad, m_adam, v_adam, t, lr)
        h_new = _proj(h_new, target)
        h_new = np.clip(h_new, 0, 1)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return _repair(best_h, n), best_c

def _local_search(h, n_iter, rng, step_size):
    n = len(h)

    best_h = h.copy()
    best_c = c5
    for _ in range(n_iter):

def _local_search_2pt(h, n_iter, rng, step_size):

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_smooth(h, alpha, top_k=60):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    w /= wsum
    if top_k < len(w):
        idx = np.argpartition(w, -top_k)[-top_k:]
        idx = idx[w[idx] > 1e-13]
    else:
        idx = np.where(w > 1e-13)[0]
    g = np.zeros(n)
    for k in idx:
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            g[:n - shift] += wk * (1.0 - h[shift:])
            if shift > 0:
                g[shift:] -= wk * h[:n - shift]
        else:
            sp = -shift
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _adam_opt(h, alpha, n_steps, lr0, top_k=60, explore_every=200, explore_mag=0.02):
    n = len(h)
    target = n * 0.5
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    t = 0
    rng_local = np.random.RandomState(12345 + alpha % 10000)
    for step in range(n_steps):
        t += 1
        lr = lr0 / (1.0 + 0.005 * t)
        grad = _grad_smooth(h, alpha, top_k)
        if step > 0 and step % explore_every == 0:
            grad += explore_mag * rng_local.randn(n) / np.sqrt(n)
        m_a = 0.9 * m_a + 0.1 * grad
        v_a = 0.999 * v_a + 0.001 * (grad * grad)
        m_hat = m_a / (1.0 - 0.9**t)
        v_hat = v_a / (1.0 - 0.999**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = _proj(h_new, target)
        h_new = np.clip(h_new, 0, 1)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if step % 50 == 0 and c_new > best_c + 1e-6:
            h = 0.7 * best_h + 0.3 * h_new
            h = _proj(h, target)
    return _repair(best_h, n), best_c

def _nesterov_opt(h, alpha, n_steps, lr0, top_k=50):
    n = len(h)
    target = n * 0.5
    v_n = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    t = 0
    for step in range(n_steps):
        t += 1
        lr = lr0 / (1.0 + 0.01 * t)
        v_n = 0.95 * v_n - lr * _grad_smooth(h + 0.95 * v_n, alpha, top_k)
        h_new = _proj(h + v_n, target)
        h_new = np.clip(h_new, 0, 1)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return _repair(best_h, n), best_c

def _local3(h, n_iter, rng, step0):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c = c5
    step = step0
    stall = 0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 7 + rng.randint(0, 10)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
                stall = 0
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
                stall += 1
        if stall >= 2000:
            step *= 1.3
            stall = 0
        elif stall < 500 and step > step0:
            step *= 0.95
    return best_h, best_c

def _local2(h, n_iter, rng, step0):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c = c5
    step = step0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

            if c5n < c5:

                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
