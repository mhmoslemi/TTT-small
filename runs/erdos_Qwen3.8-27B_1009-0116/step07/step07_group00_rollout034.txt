def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_cube(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        sh = (n - 1) - idx[ii]
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov_opt(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    best_h = h.copy()
    best_f = f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj_cube(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.05, step0 * 10.0)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.5
            h = _proj_cube(hp - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(best_h, n)

def _directed_subgrad(h, nsteps, deadline, step0):
    """Subgradient on exact max with momentum - directed at worst lag"""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    v = np.zeros(n)
    beta = 0.88
    step = step0
    dx = 2.0 / n
    for it in range(nsteps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        k_star = np.argmax(c)
        sh = (n - 1) - k_star
        g = np.zeros(n)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        v = beta * v + (1.0 - beta) * g

                best_h = h.copy()
        else:

    if time.time() > deadline:

    except Exception:
