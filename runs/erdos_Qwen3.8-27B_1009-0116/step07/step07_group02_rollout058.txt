def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    s = (n - 1) - k
    if s >= 0:
        m = n - s
        g[:m] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:m]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _sm_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:

        g += w[k] * _grad_k(x, k, n)
    return g

        return _proj(r.x, len(h))
    except Exception:

def _adam(h, grad_fn, arg, steps, lr, dl):

    m1 = np.zeros(n)
    m2 = np.zeros(n)

            except Exception:
                h[st:en] = ib
    return h, bc

def _local(h, k, niter, rng, dl, sc):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(niter):
        if time.time() > dl - 1:
            break
        idx = rng.choice(n, size=k, replace=False)
        d = sc * rng.randn(k)
        d[-1] = -d[:-1].sum()
        hn = h.copy()
        ok = True
        for j, i in enumerate(idx):
            v = h[i] + d[j]
            if v < -1e-10 or v > 1.0 + 1e-10:
                ok = False
                break
            hn[i] = v
        if ok:
            c5n = _eval(hn)
            if c5n < c5c - 1e-18:
                h, c5c = hn, c5n
                if c5c < bc:
                    bc, bh = c5c, h.copy()
    return bh, bc

def _freq_perturb(h, rng, n):
    x = np.arange(n)
    hp = h.copy()
    for f in rng.choice([1, 2, 3, 5, 7, 11, 13], size=3):
        amp = 0.005 * rng.randn()
        hp += amp * np.cos(2.0 * np.pi * f * x / n + rng.rand() * 2 * np.pi)
    return _proj(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _proj(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _eval(h0)

    # Phase 1: L-BFGS log-sum-exp with increasing alpha
    h = h0.copy()
    for alpha in [200, 1000, 5000, 25000, 100000]:
        if time.time() > dl - 450:
            break
        h = _lbfgs(h, _sm_obj, _sm_grad, alpha, 2500, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 2: p-norm
    h = best_h.copy()
    for p in [20, 60, 200]:
        if time.time() > dl - 370:
            break
        h = _lbfgs(h, _pn_obj, _pn_grad, p, 2000, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 3: Adam optimizer with softmax
    if time.time() < dl - 280:
        h = _adam(best_h.copy(), _sm_grad, 50000, 5000, 0.02, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 4: Top-lag descent
    if time.time() < dl - 220:
        for nl in [2, 3, 4, 5]:
            if time.time() > dl - 220:
                break
            ht, ct = _toplag(best_h.copy(), nl, 500, dl)
            if ct < best_c5:
                best_c5, best_h = ct, ht.copy()

    # Phase 5: Multi-start with frequency perturbations
    if time.time() < dl - 160:
        for _ in range(4):
            if time.time() > dl - 160:
                break
            hp = _freq_perturb(best_h, rng, n)
            ho = _lbfgs(hp, _sm_obj, _sm_grad, 50000, 1500, dl)
            cv = _eval(ho)
            if cv < best_c5:
                best_c5, best_h = cv, ho.copy()

    # Phase 6: Block coordinate descent
    if time.time() < dl - 120:
        hb, cb = _block_cd(best_h.copy(), 4, 2, dl)
        if cb < best_c5:
            best_c5, best_h = cb, hb.copy()
        hb, cb = _block_cd(best_h.copy(), 3, 2, dl)
        if cb < best_c5:
            best_c5, best_h = cb, hb.copy()

    # Phase 7: Subgradient tight tolerance
    if time.time() < dl - 80:
        hs = _subgrad(best_h, 500, 1e-10, 0.008, dl)
        cv = _eval(hs)
        if cv < best_c5:
            best_c5, best_h = cv, hs.copy()

    # Phase 8: Final high-alpha L-BFGS
    if time.time() < dl - 50:
        h = _lbfgs(best_h, _sm_obj, _sm_grad, 500000, 2500, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 9: 3-point local search
    if time.time() < dl - 5:
        hl, cl = _local(best_h, 3, 20000, rng, dl, 4e-5)
        if cl < best_c5:
            best_c5, best_h = cl, hl.copy()

    # Phase 10: 2-point ultra-fine polish
    if time.time() < dl - 1:
        hl, cl = _local(best_h, 2, 50000, rng, dl, 1.5e-5)
        if cl < best_c5:
            best_c5, best_h = cl, hl.copy()

    best_h = _proj(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
