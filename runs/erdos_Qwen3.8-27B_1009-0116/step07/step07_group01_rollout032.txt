            v = h_trial - h
            h = h_trial
            step = min(step * 1.04, step0 * 6.0)
        else:
            v = 0.5 * v
            step *= 0.78
            if step < step0 * 0.04:
                h = best_h.copy()
                v = np.zeros(n)
                step = step0
    return best_h, best_c

def _slsqp_obj(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    mask = w > 1e-8
    for k in np.where(mask)[0]:
        g += w[k] * _grad_single(x, k)
    return g

def _sq_sum(x, n):
    return x.sum() - n / 2.0

def _sq_sum_jac(x, n):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sq_sum, 'jac': _sq_sum_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _local2(h, n_iter, rng, step_sz, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step_sz * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def _local3(h, n_iter, rng, step_sz, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        k = (i + 5 + rng.randint(0, 6)) % n
        d1 = step_sz * rng.randn()
        d2 = step_sz * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i] -= d1
                h[j] -= d2
                h[k] += d1 + d2
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out

def _lag_cycling(h, steps_per_lag, top_k, step0, deadline):
    n = len(h)
    target = n * 0.5
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c = _eval(h)
    c = _c_vals(h)
    active = np.argsort(c)[-top_k:]
    idx = 0
    step = step0
    for _ in range(steps_per_lag * top_k):
        if time.time() > deadline:
            break
        k = active[idx % top_k]
        g = _grad_single(h, k)
        h_trial = _proj_sum(h - step * g, target)
        c_trial = float(np.max(np.correlate(h_trial, 1.0 - h_trial, mode='full')) * 2.0 / n)
        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
            h = h_trial
            step = min(step * 1.05, step0 * 5.0)
        else:
            step *= 0.75
            if step < step0 * 0.1:
                step = step0
        c = _c_vals(h)
        active = np.argsort(c)[-top_k:]
        idx += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine (15%)
    dl1 = t0 + budget_s * 0.15
    for ratio in [4, 8, 16, 32]:
        nc = max(24, n // ratio)
        hc = _repair(_downsample(h, n // nc), nc)
        for alpha in [300, 3000, 30000]:
            if time.time() > dl1:
                break
            hc = _slsqp(hc, alpha, 200, dl1)
        h_up = _repair(_upsample(hc, n // nc), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Nesterov true subgradient (30%)
    dl2 = t0 + budget_s * 0.45
    for tk, sidestep in [(2, 0.008), (3, 0.005), (5, 0.003), (8, 0.002)]:
        if time.time() > dl2:
            break
        h_new, c_new = _nesterov(best_h, tk, 80000, sidestep, dl2)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 3: Lag cycling (15%)
    dl3 = t0 + budget_s * 0.60
    for tk, ss in [(4, 0.004), (6, 0.003), (10, 0.002)]:
        if time.time() > dl3:
            break
        h_new, c_new = _lag_cycling(best_h, 5000, tk, ss, dl3)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 4: SLSQP refinement (20%)
    dl4 = t0 + budget_s * 0.80
    for alpha in [500, 3000, 15000, 80000]:
        if time.time() > dl4 - 5:
            break
        h_new = _slsqp(best_h, alpha, 1200, dl4)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 5: 3-point local search (12%)
    dl5 = t0 + budget_s * 0.92
    h_new, c_new = _local3(best_h, 400000, rng, 4e-6, dl5)
    if c_new < best_c5:
        best_c5 = c_new
        best_h = h_new.copy()

    # Phase 6: Ultra-fine 2-point polish (8%)
    dl6 = deadline - 0.5
    h_new, c_new = _local2(best_h, 600000, rng, 1.5e-7, dl6)
    if c_new < best_c5:
        best_c5 = c_new
        best_h = h_new.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
