        for s_mult in [0.4, 0.2, 0.1, 0.05, 0.02, 0.01]:
            step = s_mult / (lam * gn + 1e-10)
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval_c5(h_new)
            if c5n < best_c - 1e-16:
                h = h_new
                best_c = c5n
                best_h = h.copy()
                stall = 0
                accepted = True
                break
        if not accepted:
            stall += 1
            if stall > 200:
                break
        lam = min(lam * 1.03, 800.0)
    return best_h, best_c

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    dx = 2.0 / n
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * dx

def _pair_cd(h, n_iter, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h).copy()
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    no_improve = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if no_improve > 10000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            no_improve += 1
            continue
        for _ in range(35):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-14:
            no_improve += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            no_improve = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            no_improve += 1
    return best_h, best_c

def _cyclical_cd(h, deadline):
    n = len(h)
    c = _correl(h).copy()
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    offsets = [1, 2, 3, 5, 8, 13, 21, 37, 64, 128]
    stall_sweeps = 0
    for sweep in range(300):
        if time.time() > deadline:
            break
        improved = False
        for off in offsets:
            if time.time() > deadline:
                break
            for i in range(0, n, 4):
                if time.time() > deadline:
                    break
                j = (i + off) % n
                if i == j:
                    continue
                d0 = j - i
                k1 = n - 1 + d0
                k2 = n - 1 - d0
                g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_lo >= d_hi - 1e-15:
                    continue
                for _ in range(30):
                    d_mid = 0.5 * (d_lo + d_hi)
                    vals = c + d_mid * g
                    if 0 <= k1 < 2 * n - 1:
                        vals[k1] += d_mid * d_mid
                    if 0 <= k2 < 2 * n - 1:
                        vals[k2] += d_mid * d_mid
                    kmax = np.argmax(vals)
                    sg = g[kmax]
                    if kmax == k1 or kmax == k2:
                        sg += 2.0 * d_mid
                    if sg > 1e-14:
                        d_hi = d_mid
                    elif sg < -1e-14:
                        d_lo = d_mid
                    else:
                        break
                d_opt = 0.5 * (d_lo + d_hi)
                if abs(d_opt) < 1e-14:
                    continue
                h[i] += d_opt
                h[j] -= d_opt
                c += d_opt * g
                if 0 <= k1 < 2 * n - 1:
                    c[k1] += d_opt * d_opt
                if 0 <= k2 < 2 * n - 1:
                    c[k2] += d_opt * d_opt
                c5_new = c.max()
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                    if c5_cur < best_c:
                        best_c = c5_cur
                        best_h = h.copy()
        if not improved:
            stall_sweeps += 1
            if stall_sweeps > 8:
                break
        else:
            stall_sweeps = 0
    return best_h, best_c

def _fine_polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval_c5(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(200000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval_c5(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval_c5(best_h)

    # Phase 0: Multi-start exploration
    for ps in [0.03, 0.08, 0.15, 0.25]:
        hp = h + ps * rng.randn(n0)
        hp = _repair(_proj_sum(hp, n0 * 0.5), n0)
        c5p = _eval_c5(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 1: Smooth subgradient descent (70s)
    dl1 = t0 + min(70, budget_s * 0.07)
    h1, c5_1 = _smooth_descent(best_h.copy(), rng, dl1, n_iter=2000)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Random pair CD - main (280s)
    dl2 = t0 + min(350, budget_s * 0.4)
    h2, c5_2 = _pair_cd(best_h.copy(), 120000, rng, dl2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Cyclical CD with structured offsets (180s)
    dl3 = t0 + min(530, budget_s * 0.62)
    h3, c5_3 = _cyclical_cd(best_h.copy(), dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Smooth descent round 2 (80s)
    dl4 = t0 + min(610, budget_s * 0.72)
    h4, c5_4 = _smooth_descent(best_h.copy(), rng, dl4, n_iter=1200)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Random pair CD round 2 (250s)
    dl5 = t0 + min(860, budget_s * 0.92)
    h5, c5_5 = _pair_cd(best_h.copy(), 100000, rng, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Fine polish (remaining)
    dl6 = t0 + (budget_s - 5)
    h6, c5_6 = _fine_polish(best_h.copy(), rng, dl6)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    # Final verification
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, len(best_h))
