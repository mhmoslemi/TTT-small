import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_cvals(h).max())

def _grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-16:
            continue
        d = n - 1 - k
        if d >= 0:
            m = n - d
            g[:m] += wk * (1.0 - h[d:d+m])
            if d > 0:
                g[d:d+m] -= wk * h[:m]
        else:
            sp = -d
            m = n - sp
            g[sp:] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[sp:]
    return g

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    w = np.exp(s)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(len(x))
    w /= w_sum
    return _grad_w(x, w)

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 5e-12, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _momentum_toplag(h, n_steps, n_lags, step0, beta, deadline):
    n = len(h)
    v = np.zeros(n)
    c5_cur = _eval(h)
    for i in range(n_steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.01 * i)
        c = _cvals(h)
        m = c.max()
        tol = 1e-10 * m
        active = np.where(c >= m - tol)[0]
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = _grad_w(h, w)
        v = beta * v - t * g
        h_new = _repair(h + v, n)
        c5n = _eval(h_new)
        if c5n < c5_cur + 1e-15:
            h = h_new
            if c5n < c5_cur:
                c5_cur = c5n
        else:
            v *= 0.5
    return h, c5_cur

def _anneal_polish(h, n_rounds, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for r in range(n_rounds):
        if time.time() > deadline - 10:
            break
        noise_amp = 0.003 * (1 - r / n_rounds)
        h_pert = h + noise_amp * rng.randn(n)
        h_pert = _repair(h_pert, n)
        h_pert = _lbfgs(h_pert, 100000, 2000, deadline)
        c5p = _eval(h_pert)
        if c5p < best_c - 1e-12:
            best_c = c5p
            best_h = h_pert.copy()
            h = h_pert.copy()
    return best_h, best_c

def _top_lag_descent(h, n_lags, max_rounds, deadline):
    n = len(h)
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 15:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / n_lags
        g = _grad_w(h, w)
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _local_2pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

    c5c = _eval(h)
