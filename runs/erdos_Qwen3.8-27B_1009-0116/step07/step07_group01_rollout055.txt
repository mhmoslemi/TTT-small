            else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(-(1.0 - h)) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, t):
    lo, hi = -3.0, 3.0
    for _ in range(65):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _downsample(h, m):
    n = len(h)
    f = n // m
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _grad_lse(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-13:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s >= 0:
            if s == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                cnt = n - s
                g[:cnt] += wk * (1.0 - h[s:s+cnt])
                g[s:s+cnt] -= wk * h[:cnt]
        else:
            sp = -s
            cnt = n - sp
            g[sp:sp+cnt] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp+cnt]
    return g

def _pgd(h, n, alpha, max_iter, step0, deadline):
    h = h.copy()
    c = _correl(h)
    best_c = c.max()
    step = step0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        m = c.max()
        sa = np.clip(alpha * (c - m), -400, 400)
        sa -= sa.max()
        w = np.exp(sa); w /= w.sum()
        g = _grad_lse(h, w, n)
        h2 = _proj(h - step * g, n * 0.5)
        c2 = _correl(h2)
        if c2.max() < best_c - 1e-15:
            h, c, best_c = h2, c2, c2.max()
            step = min(step * 1.06, step0 * 10)
        else:
            step *= 0.4
            if step < step0 * 5e-4:
                break
    return _repair(h, n), best_c

def _slsqp_smooth(h, alpha, maxiter, deadline):
    n = len(h)
    dx = 2.0 / n
    bsum = n * 0.5
    
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        return _grad_lse(x, w, n)
    
    def _cons_fun(x):
        return x.sum() - bsum
    
    def _cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() > deadline:
            return h
        return _repair(res.x, n)
    except Exception:
        return h

def _guided_local(h, rng, max_iter, base_step, deadline):
    n = len(h)
    h = h.copy()
    c = _correl(h)
    c5 = c.max()
    best_h = h.copy()
    best_c5 = c5
    top_count = 5
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        # Recompute top lags periodically
        top_idx = np.argsort(c)[-top_count:]
        shifts = np.abs(top_idx - (n - 1)).astype(int)
        
        # Pick a shift from top lags (weighted toward largest)
        s = shifts[rng.randint(top_count)]
        d = base_step * rng.randn()
        
        if s == 0:
            # Modify two random points to push away from 0.5
            i1 = rng.randint(n)
            i2 = (i1 + 1 + rng.randint(0, 6)) % n
            # Push i1 toward 1 if h[i1] > 0.5, toward 0 if h[i1] < 0.5
            push1 = d if h[i1] > 0.5 else -d
            push2 = -push1
            a, b = h[i1] + push1, h[i2] + push2
            if 0 <= a <= 1 and 0 <= b <= 1:
                h[i1], h[i2] = a, b
                c_new = _correl(h)
                c5n = c_new.max()
                if c5n < c5:
                    c5, c = c5n, c_new
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i1], h[i2] = a - push1, b - push2
        else:
            # Modify a pair at the chosen shift
            if s >= n - 1:
                i1 = rng.randint(n)
                i2 = (i1 + 1) % n
                s_use = 1
            else:
                i1 = rng.randint(n - s)
                i2 = i1 + s
                s_use = s
            a = h[i1] + d
            b = h[i2] - d
            if 0 <= a <= 1 and 0 <= b <= 1:
                h[i1], h[i2] = a, b
                c_new = _correl(h)
                c5n = c_new.max()
                if c5n < c5:
                    c5, c = c5n, c_new
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i1], h[i2] = h[i1] - d, h[i2] + d
    return _repair(best_h, n), best_c5
