```python
def _grad_softmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    C = np.correlate(h, b, mode='full') * dx
    # Softmax weights
    C_max = C.max()
    exp_C = np.exp(beta * (C - C_max))
    w = exp_C / exp_C.sum()
    
    # Gradient
    g = np.zeros(n)
    # For each shift s (corresponding to index k = n-1+s):
    # ∂C[k]/∂h[j] = (1-h[j+s])*I(0<=j+s<=n-1) - h[j-s]*I(0<=j-s<=n-1)
    for s in range(-(n-1), n):
        k = n - 1 + s
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        # Term 1: (1-h[j+s]) for valid j: max(0,-s) <= j <= min(n-1, n-1-s)
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo <= j_hi:
            g[j_lo:j_hi+1] += wk * (1.0 - h[j_lo+s:j_hi+s+1])
        # Term 2: -h[j-s] for valid j: max(0,s) <= j <= min(n-1, n-1+s)
        j_lo2 = max(0, s)
        j_hi2 = min(n-1, n-1+s)
        if j_lo2 <= j_hi2:
            g[j_lo2:j_hi2+1] -= wk * h[j_lo2-s:j_hi2-s+1]
    return g
```

```python
def _grad_max(h, beta):
    """Gradient of softmax approximation to max correlation."""
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    C_max = C.max()
    exp_C = np.exp(beta * (C - C_max))
    w = exp_C / exp_C.sum()
    
    g = np.zeros(n)
    b = 1.0 - h
    # For each lag index k in [0, 2n-2]:
    # s = k - (n-1) is the shift
    # ∂C[k]/∂h[j] = b[j+s]*I(valid) - h[j-s]*I(valid)

        j_hi = min(n - 1, n - 1 - s)

        j_hi2 = min(n - 1, n - 1 + s)
```

```python
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        if _ % 200 == 0:
            push = 0.03
            h2 = h.copy()
            h2[h2 < push] = 0.0
            h2[h2 > 1.0 - push] = 1.0
            h2 = _proj_sum(h2, n / 2.0)
            c2 = _eval(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                h = h2.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    def bc():
        return el() > budget_s - 15

    # Use initial_h_values if available
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam with ramping beta on initial (n=560)
    if el() < budget_s * 0.25:
        h_try, c_try = _adam_loop(best_h, 30, 0.005, 50000, el, bc, adapt_beta=True)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 2: Try different n values with various starts
    for n_test in [400, 720]:
        if el() > budget_s * 0.55:
            break
        for mode in [0, 1, 3]:
            if el() > budget_s * 0.55:
                break
            h_s = _make_start(n_test, mode, rng)
            h_try, c_try = _adam_loop(h_s, 20, 0.004, 20000, el, bc, adapt_beta=True)
            if c_try < best_c5:
                best_c5, best_h = c_try, h_try

    # Phase 3: Second Adam pass on best with higher starting beta
    if el() < budget_s * 0.70:
        h_try, c_try = _adam_loop(best_h, 100, 0.003, 40000, el, bc, adapt_beta=True)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 4: Nesterov top-k
    if el() < budget_s - 200:
        n_cur = len(best_h)
        h_try, c_try = _nesterov_topleast(best_h, 6000, 2, 50, 0.01, el, bc)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 5: Quantize and refine
    if el() < budget_s - 150:
        iref = min(int((budget_s - 150 - el()) / 0.003), 50000)
        h_try, c_try = _quantize_refine(best_h, iref, rng, el, bc)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 6: Final Adam polish
    if el() < budget_s - 100:
        h_try, c_try = _adam_loop(best_h, 500, 0.002, 20000, el, bc, adapt_beta=False)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 7: Fine 2-point local search
    if el() < budget_s - 20:
        n_it = min(int((budget_s - 20 - el()) / 0.002), 100000)
        h_try, c_try = _local_2pt(best_h, n_it, 5e-5, rng, el, bc)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Final repair
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
```
