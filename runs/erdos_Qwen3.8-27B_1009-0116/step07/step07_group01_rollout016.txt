For shift sh < 0 (sp = -sh):
c = sum_{i=sp}^{n-1} h[i-sp] * (1-h[i])
∂c/∂h[j]: j appears as i-sp (coef: 1-h[j+sp]) or as i (coef: -h[j-sp])
= (1-h[j+sp]) * [j+sp < n]  +  (-h[j-sp]) * [j-sp >= 0]

Same formula! With shift sh (positive or negative, using |sh|):
∂c/∂h[j] = (1-h[j+|sh|]) * [j+|sh| < n]  +  (-h[j-|sh|]) * [j-|sh| >= 0]

So the total gradient:
g[j] = dx * sum_{sh} w[sh] * [(1-h[j+|sh|]) * I(j+|sh|<n) - h[j-|sh|] * I(j-|sh|>=0)]

This can be written as:
g[j] = dx * [sum_{sh: j+|sh|<n} w[sh] - sum_{sh: j+|sh|<n} w[sh]*h[j+|sh|] - sum_{sh: j-|sh|>=0} w[sh]*h[j-|sh|]]

The first sum is just sum of w[sh] for valid sh.
The second and third sums are weighted correlations of w and h!

So:
g[j] = dx * [W(j) - (w * h)[j] - (w_rev * h)[j]]

where W(j) is the sum of w[sh] for sh such that j+|sh| < n or j-|sh| >= 0 (which is most of them), and (w*h) is a correlation.

Actually, this is getting complicated. Let me just use a direct but vectorized approach.

For each shift sh, the contribution to the gradient is:
- For j = 0 to n-1-sh: g[j] += w[sh] * (1-h[j+sh])
- For j = sh to n-1: g[j] -= w[sh] * h[j-sh] (if sh > 0)

For negative shifts (sp = -sh):
- For j = sp to n-1: g[j] += w[sh] * (1-h[j-sp])
- For j = 0 to n-1-sp: g[j] -= w[sh] * h[j+sp]

```python
def _grad_vec(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for sh in range(2*n-1):
        wi = w[sh]
        if wi < 1e-15:
            continue
        lag = (n-1) - sh  # positive shift
        if lag >= 0:
            # h[i] * (1-h[i+lag]) for i=0..n-1-lag
            g[:n-lag] += wi * (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= wi * h[:-lag]
        else:
            sp = -lag
            g[sp:] += wi * (1.0 - h[:n-sp])
            g[:n-sp] -= wi * h[sp:]
    return g * dx
```

One more idea: **coordinate descent with optimal step**. For each variable h[j], compute the optimal value (given all other variables fixed) that minimizes the max correlation. This is a 1D optimization problem that can be solved analytically or with bisection.

We want to minimize max_k (a_k * h[j] + b_k) over h[j] ∈ [0, 1] while maintaining the sum constraint (which means if we change h[j], we must change something else).

This is a linear programming problem in 1D, solvable by finding the intersection points of the lines.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    for _ in range(200):
        h *= t/s; h = np.clip(h, 0, 1); s = h.sum()
        if abs(s-t) < 1e-12: break
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0: a = min(d, 1.0-h[i]); h[i] += a; d -= a
            else: a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _proj(h, t):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        m = (lo+hi)*0.5
        if np.clip(h-m, 0, 1).sum() > t: lo = m
        else: hi = m
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _obj(x, a):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/len(x))
    m = c.max(); s = np.clip(a*(c-m), -400, 400); s -= s.max()
    return m + np.log(np.exp(s).sum())/a

def _grad_full(x, a, tk):
    n = len(x); c = np.correlate(x, 1.0-x, mode='full')*(2.0/n)
    m = c.max(); s = np.clip(a*(c-m), -400, 400); s -= s.max()
    w = np.exp(s); w /= w.sum()
    if tk and tk < len(w):
        idx = np.argsort(w)[-tk:]; w2 = np.zeros_like(w); w2[idx] = w[idx]
        w2 /= w2.sum(); w = w2
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1)-k
        if sh >= 0:
            g[:n-sh] += w[k]*(1.0-x[sh:])
            if sh > 0: g[sh:] -= w[k]*x[:n-sh]
        else:
            sp = -sh; g[sp:] += w[k]*(1.0-x[:n-sp]); g[:n-sp] -= w[k]*x[sp:]
    return g

def _gsingle(h, k, n):
    dx = 2.0/n; sh = (n-1)-k; g = np.zeros(n)
    if sh >= 0:

        if sh > 0: g[sh:] -= h[:n-sh]
    else:
        sp = -sh; g[sp:] += (1.0-h[:n-sp]); g[:n-sp] -= h[sp:]
    return g * dx

def _cj(x, n): return np.ones(n)

def _slsqp(h, a, mi, ft, tk=None):
