import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _nesterov_multi(h, n, duration_s, t_end, top_k, mu, step0):
    dx = 2.0 / n
    target = n * 0.5
    best_c5 = _eval(h)
    best_h = h.copy()
    v = np.zeros(n)
    step = step0
    st = time.time()
    while time.time() < min(st + duration_s, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(h, k, n, dx)
        g /= top_k
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, target)
        c_new = _eval(h_new)
        if c_new < m:
            h = h_new
            step = min(step * 1.04, step0 * 10.0)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            step *= 0.4
            if step < 1e-9:
                step = step0
                v *= 0.1
    return h, best_c5, best_h

def _pair_cd(h, n, duration_s, t_end, rng, top_T, spacing):
    dx = 2.0 / n
    target = n * 0.5
    best_c5 = _eval(h)
    best_h = h.copy()
    gr = (np.sqrt(5) - 1) / 2

        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        grads = np.array([_grad_lag(h, k, n, dx) for k in top_idx])
        improved_sweep = False
        for _ in range(n):
