from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _grad_top(h, top_k, weight_power=1.0):
    n = len(h)
    c = _corr_full(h)
    idx = np.argsort(c)[::-1][:top_k]
    vals = c[idx]
    if weight_power > 1:
        w = vals ** weight_power
    else:
        w = np.ones(top_k)
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += w[i] * _grad_lag(h, k)
    return g

def _gd_refine(h, check_fn, top_k, n_rounds, init_steps):
    n = len(h)
    c5c = _eval(h)
    steps = list(init_steps)
    stall = 0
    for r in range(n_rounds):
        if check_fn():
            break
        g = _grad_top(h, top_k, weight_power=2.0)
        improved = False
        for si in range(len(steps)):
            s = steps[si]
            h_new = _project_sum(h - s * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [s * 1.3] + [x for x in steps if x < s * 0.4]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _lsx_obj(x, n, a):
    c = _corr_full(x)
    m = c.max()
    sc = np.clip(a * (c - m), -40, 40)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad(x, n, a):
    c = _corr_full(x)
    m = c.max()
    sc = np.clip(a * (c - m), -40, 40)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _slsqp_polish(h, n, check_fn, temps):
    c5c = _eval(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn * 0.5,
             'jac': lambda x, nn=n: np.ones(nn), 'args': (n,)}]
    for a in temps:
        if check_fn():
            break
        try:
            res = minimize(_lsx_obj, h, args=(n, a), jac=_lsx_grad,
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-16})
            hr = _repair(res.x, n)
            c5r = _eval(hr)
            if c5r < c5c:
                h, c5c = hr, c5r
        except Exception:
            pass
    return h, c5c

def _gradient_local_search(h, n_combined, rng, check_fn):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    step = 0.005
    stagnant = 0
    while n_combined > 0 and not check_fn():

        gn = np.linalg.norm(g)

        g = g / gn
        found = False

            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:

                c5c = c5n
                step *= 1.15
                stagnant = 0
                found = True
                if c5n < best_c:

                    best_h = h.copy()
                n_combined -= 1
                break
            step *= 0.55
        if not found:
            stagnant += 1
            if stagnant > 15:

                step = 0.005
                stagnant = 0
            n_combined -= 1
    return best_h, best_c

def _random_2pt(h, ntry, rng, c5c, check_fn):
    n = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):

                    best_c = c5n
                    best_h = h.copy()
