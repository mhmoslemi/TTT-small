import numpy as np
import time
from scipy.optimize import minimize

_fc = {}

def _proj_sum(h, t):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    t = float(np.clip(t, 0.0, float(h.size)))
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.clip(h - m, 0.0, 1.0).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _fix_sum(h, t):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    if h.size == 0:
        return h
    t = float(t)
    for _ in range(4):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        i = int(np.argmax(h * (1.0 - h)))
        if 0.0 < h[i] < 1.0 and abs(d) < 0.25:
            h[i] += d
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return _proj_sum(h, t)

def _resample(a, no, nn):
    dxo = 2.0 / no
    dxn = 2.0 / nn
    y = np.linspace(0.0, 2.0, no + 1)
    f = np.empty(no + 1)
    f[0] = 0.0
    f[1:] = np.cumsum(a) * dxo
    f[-1] = 1.0
    yn = np.linspace(0.0, 2.0, nn + 1)
    fn = np.interp(yn, y, f)
    return _proj_sum(np.diff(fn) / dxn, nn / 2.0)

def _perturb(b, n, r, amp):
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    z = np.zeros(n)
    for f, a in ((1, 0.5), (2, 0.3), (3, 0.2), (4, 0.1)):
        z += a * np.sin(np.pi * f * t + r.uniform(0.0, 6.283185))
    x = 0.5 + amp * z if b is None else np.asarray(b, dtype=float) + amp * z
    return _fix_sum(_proj_sum(np.clip(x, 0.0, 1.0), n / 2.0), n / 2.0)

def _hc(x, n, s):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if s:
        m = n // 2
        h = np.empty(n, dtype=float)
        h[:m] = x
        h[m:] = x[::-1]
    else:
        h = x
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return h, np.maximum(c, 0.0)

def _ex(x, n, s):
    return float(np.max(_hc(x, n, s)[1]))

def _eh(h, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(np.maximum(c, 0.0)))

def _lume(c, b):
    m = float(np.max(c))
    if b <= 0:
        return float(np.mean(c))
    if m <= 0.0:
        return 0.0
    z = np.exp(b * (c - m))
    return m + float(np.log(z.sum())) / b

def _sw(c, b):
    m = float(np.max(c))
    z = np.exp(b * (c - m))
    s = z.sum()
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return z / s

def _of(n, f):
    k = (n, f)
    v = _fc.get(k)
    if v is None:
        v = np.fft.rfft(np.ones(n), f)
        _fc[k] = v
    return v

def _gw(h, n, w):
    dx = 2.0 / n
    L = (2 * n - 1) + n - 1
    f = 1 << (L - 1).bit_length()
    fh = np.fft.rfft(h, f)
    fo = _of(n, f) - fh
    fw = np.fft.rfft(w, f)
    fr = np.fft.rfft(w[::-1], f)
    A = np.fft.irfft(fw * fo, f)[n - 1:2 * n - 1]
    B = np.fft.irfft(fr * fh, f)[n - 1:2 * n - 1]
    return dx * (A - B)

def _projdesc(x, n, s, b, its, st, dl):
    tx = n / 4.0 if s else n / 2.0
    m = len(x)
    x = _proj_sum(np.asarray(x, dtype=float), tx)
    bx = x.copy()
    bf = _lume(_hc(x, n, s)[1], b)
    step = st
    stall = 0
    for _ in range(its):
        if time.time() >= dl:
            break
        h, c = _hc(x, n, s)
        w = _sw(c, b)
        g = _gw(h, n, w)
        if s:
            g = g[:m] + g[m:][::-1]
        gn = float(np.linalg.norm(g))
        if gn < 1e-13:
            break
        a = step / max(gn, 1e-13)
        ok = False
        for __ in range(10):
            y = _proj_sum(x - a * g, tx)
            fy = _lume(_hc(y, n, s)[1], b)
            if fy < bf - 1e-14:
                ok = True
                break
            a *= 0.5
            if a < 1e-10:
                break
        if ok:
            x = y
            bx = x.copy()
            bf = fy
            step = min(step * 1.03, 0.5)
            stall = 0
        else:
            step *= 0.5
            stall += 1
            if stall > 80 or step < 1e-9:
                break
    return bx

def _actdesc(x, n, s, tol, its, st, dl):
    tx = n / 4.0 if s else n / 2.0
    m = len(x)
    x = _proj_sum(np.asarray(x, dtype=float), tx)
    bx = x.copy()
    bc = _ex(x, n, s)
    step = st
    stall = 0
    for _ in range(its):
        if time.time() >= dl:
            break
        h, c = _hc(x, n, s)
        cx = float(c.max())
        if cx < bc:
            bc = cx
            bx = x.copy()
        act = np.flatnonzero(c >= cx - max(tol, 1e-12))
        if act.size == 0:
            act = np.array([int(np.argmax(c))])
        w = np.zeros(c.shape)
        w[act] = 1.0 / act.size
        g = _gw(h, n, w)
        if s:
            g = g[:m] + g[m:][::-1]
        gn = float(np.linalg.norm(g))
        if gn < 1e-13:
            break
        a = step / max(gn, 1e-13)
        ok = False
        for __ in range(12):
            y = _proj_sum(x - a * g, tx)
            cy = _ex(y, n, s)
            if cy < bc - 1e-13:
                ok = True
                break
            a *= 0.5
            if a < 1e-10:
                break
        if ok:
            x = y
            bx = x.copy()
            bc = cy
            step = min(step * 1.03, 0.3)
            stall = 0
        else:
            step *= 0.6
            stall += 1
            if stall > 40 or step < 1e-9:
                break
    return bx

def _sob(x, n, s, b, p):
    v = _lume(_hc(x, n, s)[1], b)
    if p > 0.0:
        tx = n / 4.0 if s else n / 2.0
        sc = 2.0 if s else 1.0
        q = (np.asarray(x, dtype=float).sum() - tx) * sc
        v += p * q * q
    return float(v)

def _sgd(x, n, s, b, p):
    x = np.asarray(x, dtype=float)
    h, c = _hc(x, n, s)
    w = _sw(c, b)
    g = _gw(h, n, w)
    m = len(x)
    if s:
        g = g[:m] + g[m:][::-1]
    if p > 0.0:
        tx = n / 4.0 if s else n / 2.0
        sc = 2.0 if s else 1.0
        g = g + (2.0 * p * sc * sc) * (x.sum() - tx)
    return g

def _lbf(x, n, s, b, p, mi, dl):
    if time.time() >= dl:
        return x
    try:
        r = minimize(_sob, np.asarray(x, dtype=float), jac=_sgd,
                     args=(n, s, b, p), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * len(x),
                     options={'maxiter': int(mi), 'ftol': 1e-15,
                              'gtol': 1e-10, 'maxls': 20})
        return _proj_sum(r.x, n / 4.0 if s else n / 2.0)
    except Exception:
        return x

def _rsrch(x, n, s, dl, sc, rng):
    tx = n / 4.0 if s else n / 2.0
    m = len(x)
    x = _proj_sum(np.asarray(x, dtype=float), tx)
    bc = _ex(x, n, s)
    bx = x.copy()
    for it in range(200000):
        if time.time() >= dl:
            break
        i = rng.randint(m)
        j = (i + 1 + rng.randint(0, 7)) % m
        if i == j:
            continue
        d = rng.randn() * sc
        ni = x[i] + d
        nj = x[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            x[i] = ni
            x[j] = nj
            c = _ex(x, n, s)
            if c < bc - 1e-15:
                bc = c
                bx = x.copy()
                sc = min(sc * 1.05, 5e-3)
            else:
                x[i] = ni - d
                x[j] = nj + d
        if it % 2000 == 0:
            sc = max(sc * 0.75, 1e-7)
    return bx, bc

def _refine(x, n, s, dl, rng):
    tx = n / 4.0 if s else n / 2.0
    x = _proj_sum(np.asarray(x, dtype=float), tx)
    bx = x.copy()
    bc = _ex(x, n, s)
    if time.time() >= dl:
        return bx
    for tol in (0.02, 0.008, 0.003):
        rem = dl - time.time()
        if rem <= 3.0:
            break
        loc = time.time() + min(20.0, max(2.0, rem))
        y = _actdesc(x, n, s, tol, 400, 0.03, loc)
        c = _ex(y, n, s)
        if c < bc - 1e-15:
            bx = y.copy()
            bc = c
        x = y if c < bc + 0.004 else bx
    for b in (5, 10, 20, 40, 80, 160, 320):
        rem = dl - time.time()
        if rem <= 4.0:
            break
        loc = time.time() + min(50.0, max(3.0, rem * 0.15))
        y = _projdesc(x, n, s, b, 3000, 0.05, loc)
        c = _ex(y, n, s)
        if c < bc - 1e-15:
            bx = y.copy()
            bc = c
        x = y if c < bc + 0.004 else bx
    for tol in (0.001, 3e-4, 1e-4, 3e-5, 1e-5, 3e-6, 1e-6):
        rem = dl - 30.0 - time.time()
        if rem <= 3.0:
            break
        loc = time.time() + min(15.0, max(2.0, rem))
        y = _actdesc(x, n, s, tol, 3000, 0.02, loc)
        c = _ex(y, n, s)
        if c < bc - 1e-15:
            bx = y.copy()
            bc = c
        x = y if c < bc + 0.002 else bx
    while time.time() < dl - 80.0:
        rem = dl - 80.0 - time.time()
        if rem <= 5.0:
            break
        loc = time.time() + min(45.0, rem)
        y = _lbf(x, n, s, 200, 1e4, 250, loc)
        c = _ex(y, n, s)
        if c < bc - 1e-15:
            bx = y.copy()
            bc = c
        x = y if c < bc + 0.002 else bx
    if time.time() < dl - 10.0:
        y, c = _rsrch(x, n, s, min(dl - 5.0, time.time() + 30.0), 2e-4, rng)
        if c < bc - 1e-15:
            bx = y.copy()
            bc = c
    return _proj_sum(bx, tx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 12.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = _perturb(None, 560, rng, 1.0)
    h0 = np.clip(h0, 0.0, 1.0)
    n0 = len(h0)
    if n0 == 0:
        h0 = _perturb(None, 560, rng, 1.0)
        n0 = 560
    if n0 % 2:
        h0 = _resample(h0, n0, n0 + 1)
        n0 += 1
    h0 = _fix_sum(h0, n0 / 2.0)
    ho = h0.copy()
    best_h = h0.copy()
    best_c = _eh(h0, n0)
    best_n = n0
    if float(budget_s) < 40.0:
        best_h = _fix_sum(best_h, best_n / 2.0)
        return best_h, float(_eh(best_h, best_n)), best_n
    reserve = min(180.0, max(60.0, 0.12 * float(budget_s)))
    md = hard - reserve
    q = min(md - 300.0, t0 + 70.0)
    if q > t0 + 15.0:
        hs = 0.5 * (h0 + h0[::-1])
        hs = _fix_sum(hs, n0 / 2.0)
        xs = hs[:n0 // 2]
        y = _refine(xs, n0, True, q, rng)
        hf = _fix_sum(_hc(y, n0, True)[0], n0 / 2.0)
        c = _eh(hf, n0)
        if c < best_c - 1e-15:
            best_h, best_c, best_n = hf, c, n0
    fq = min(md - 250.0, t0 + 140.0)
    if fq > time.time() + 15.0:
        y = _refine(ho.copy(), n0, False, fq, rng)
        hf = _fix_sum(y, n0 / 2.0)
        c = _eh(hf, n0)
        if c < best_c - 1e-15:
            best_h, best_c, best_n = hf, c, n0
    raw_main = None
    main_n = best_n
    if time.time() < md - 50.0:
        tn = int(round(1.8 * max(n0, best_n) / 4.0)) * 4
        tn = max(480, min(tn, 1120))
        if float(budget_s) < 600.0:
            tn = min(tn, 840)
        elif float(budget_s) < 900.0:
            tn = min(tn, 960)
        else:
            tn = min(tn, 1008)
        if tn <= best_n:
            tn = max(480, ((best_n // 4) + 1) * 4)
        if tn % 4:
            tn += 4 - (tn % 4)
        main_n = tn
        raw = _resample(best_h, best_n, main_n) if main_n > best_n else best_h.copy()
        raw = _fix_sum(raw, main_n / 2.0)
        hs = 0.5 * (raw + raw[::-1])
        x = _proj_sum(hs[:main_n // 2], main_n / 4.0)
        y = _refine(x, main_n, True, md, rng)
        hf = _fix_sum(_hc(y, main_n, True)[0], main_n / 2.0)
        c = _eh(hf, main_n)
        if c < best_c - 1e-15:
            best_h, best_c, best_n = hf, c, main_n
        raw_main = raw
    if raw_main is not None and time.time() < hard - 60.0:
        y = _refine(raw_main.copy(), main_n, False, min(hard - 20.0, time.time() + 70.0), rng)
        hf = _fix_sum(y, main_n / 2.0)
        c = _eh(hf, main_n)
        if c < best_c - 1e-15:
            best_h, best_c, best_n = hf, c, main_n
    if time.time() < hard - 70.0:
        rn = main_n
        for _ in range(2):
            if time.time() > hard - 40.0:
                break
            base = _resample(best_h, best_n, rn) if best_n != rn else best_h.copy()
            hr = _perturb(base, rn, rng, 0.08)
            hrs = 0.5 * (hr + hr[::-1])
            xr = _proj_sum(hrs[:rn // 2], rn / 4.0)
            y = _refine(xr, rn, True, min(hard - 10.0, time.time() + 50.0), rng)
            hf = _fix_sum(_hc(y, rn, True)[0], rn / 2.0)
            c = _eh(hf, rn)
            if c < best_c - 1e-15:
                best_h, best_c, best_n = hf, c, rn
    best_h = _fix_sum(best_h, best_n / 2.0)
    return best_h, float(_eh(best_h, best_n)), best_n
