                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, lags, dx):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g * dx

def _projected_grad_desc(h, n, iters, k_top, init_steps, check_fn):
    c5c = _eval(h)
    steps = list(init_steps)
    stall = 0
    for _ in range(iters):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top_k, 2.0 / n) / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _project_sum(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.35] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            stall += 1
            if stall > 14:
                break
        else:
            stall = 0
    return h, c5c

def _adam_opt(h, n, iters, lr0, k_top, temp, check_fn):
    c5c = _eval(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    for _ in range(iters):
        if check_fn():
            break
        t += 1
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        mx = c.max()
        sc = np.clip((c - mx) / temp, -400, 400)
        w = np.exp(sc - sc.max())
        w /= w.sum()
        sig = np.where(w > w.max() * 5e-7)[0]
        g = _grad_lags(h, sig, 2.0 / n)
        g *= w[sig][:, None] if len(sig) > 1 else w[sig[0]]
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        mc = m / (1 - beta1 ** t)
        vc = v / (1 - beta2 ** t)
        step = lr * mc / (np.sqrt(vc) + 1e-8)
        h_new = _project_sum(h - step, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            h = h_new
            c5c = c5n
        else:
            lr *= 0.9
    return h, c5c

def _slsqp_smooth(h, n, maxiter, a_val, check_fn):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(a_val * (c - m), -350, 350)
        return m + np.log(np.sum(np.exp(s - s.max()))) / a_val

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(a_val * (c - m), -350, 350)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            sh = (n - 1) - k
            wk = w[k] * (2.0 / n)
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g

    def sfun(x):
        return x.sum() - n * 0.5

    def sjac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sfun, 'jac': sjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_3pt(h, ntry, rng, c5c, check_fn, amp=5e-6):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        k = (i + 6 + rng.randint(0, 5)) % nc
        d1 = amp * rng.randn()
        d2 = amp * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < best_c - 1e-19:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
    return best_h, best_c

def _interp_to(h, n_new):
    n_old = len(h)
    x_o = np.linspace(0, 2, n_old, endpoint=False)
    x_n = np.linspace(0, 2, n_new, endpoint=False)
    return _repair(np.interp(x_n, x_o, h, period=2.0), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 25

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Projected gradient with increasing k_top
    for kt in [6, 10, 14, 20, 12, 8, 16, 10]:
        if _bc():
            break
        h2, c52 = _projected_grad_desc(best_h.copy(), n0, 700, kt,
                                        [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 4e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: ADAM with smooth max
    if not _bc():
        for lr_i, tmp in [(3e-3, 0.008), (1.5e-3, 0.005), (8e-4, 0.004)]:
            if _bc():
                break
            h2, c52 = _adam_opt(best_h.copy(), n0, 400, lr_i, 12, tmp, _bc)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2

    # Phase 3: SLSQP smooth max
    if _el() < budget_s - 350:
        for av in [40000, 80000, 150000]:
            if _bc():
                break
            hr = _slsqp_smooth(best_h.copy(), n0, 500, av, _bc)
            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr

    # Phase 4: Resolution exploration
    for n_try in [480, 640, 720]:
        if _bc():
            break
        h_try = _interp_to(best_h, n_try)
        h2, c52 = _projected_grad_desc(h_try, n_try, 400, 8,
                                        [0.04, 0.02, 0.01, 0.005, 0.002], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Perturbation restarts
    for pert in [0.004, 0.008, 0.015]:
        if _bc():
            break
        h_p = best_h + pert * rng.randn(len(best_h))
        h_p = _repair(h_p, len(best_h))
        h2, c52 = _projected_grad_desc(h_p, len(best_h), 500, 8,
                                        [0.03, 0.015, 0.008, 0.004, 0.002, 0.001], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Fine gradient polish (small k, small steps)
    for kt in [3, 5, 4, 6, 3]:
        if _bc():
            break
        h2, c52 = _projected_grad_desc(best_h.copy(), len(best_h), 700, kt,
                                        [0.012, 0.006, 0.003, 0.0015, 0.0008, 4e-4, 2e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: ADAM fine-tune
    if not _bc():
        h2, c52 = _adam_opt(best_h.copy(), len(best_h), 500, 5e-4, 6, 0.003, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 8: 3-point local search (coarse)
    if _el() < budget_s - 80:
        nt = min(int((budget_s - 80 - _el()) / 0.005), 120000)
        best_h, best_c5 = _local_3pt(best_h, nt, rng, best_c5, _bc, amp=4e-6)

    # Phase 9: 3-point local search (fine)
    if _el() < budget_s - 15:
        nt = min(int((budget_s - 15 - _el()) / 0.005), 100000)
        best_h, best_c5 = _local_3pt(best_h, nt, rng, best_c5, _bc, amp=1.5e-6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
