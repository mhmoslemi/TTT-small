def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:

                    h[idx] -= a
                    diff += a

        if time.time() - t0 > t_end:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 8)) % n
        k = (i + 10 + rng.randint(0, 20)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c = _eval(h)
            if c < c5 - 1e-18:
                c5 = c
                if c < best_c:
                    best_c = c
                    best = h.copy()
                step = min(step * 1.06, init_step * 4.0)
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
                if step > init_step * 0.005:
                    step *= 0.65
    return best, best_c

def _downsample(h, m):
    n = len(h)
    f = n // m
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Annealed softmax Nesterov (broad to sharp)
    alphas1 = [30, 100, 500, 2000, 8000, 30000]
    h, c5 = _nesterov(h, alphas1, 4000, 0.004, 0.80, t0, t0 + 250)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Very sharp alpha refinement
    alphas2 = [30000, 80000, 200000]
    h, c5 = _nesterov(best_h.copy(), alphas2, 3000, 0.0015, 0.85, t0, t0 + 450)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 3: Low-resolution escape (different basin)
    if time.time() - t0 < budget_s - 350:
        n_lo = n0 // 4
        m = n0 // n_lo
        h_lo = _downsample(best_h, n_lo)
        h_lo = _repair(h_lo, n_lo)
        alphas_lo = [50, 500, 3000, 10000]
        h_lo, _ = _nesterov(h_lo, alphas_lo, 3000, 0.01, 0.78, t0, t0 + budget_s - 400)
        h_hi = np.repeat(_repair(h_lo, n_lo), m).copy()
        h_hi = _repair(h_hi, n0)
        alphas_hi = [5000, 30000]
        h_hi, c5_hi = _nesterov(h_hi, alphas_hi, 2500, 0.002, 0.83, t0, t0 + budget_s - 250)
        if c5_hi < best_c5:
            best_c5, best_h = c5_hi, h_hi.copy()

    # Phase 4: Shake and re-optimize (multiple attempts)
    if time.time() - t0 < budget_s - 200:
        for trial in range(4):
            if time.time() - t0 > budget_s - 100:
                break
            noise_scale = 0.015 + 0.01 * trial
            h_sh = _repair(best_h + noise_scale * rng.randn(n0), n0)
            c5_sh_init = _eval(h_sh)
            if c5_sh_init < best_c5 + 0.008:
                alphas_sh = [5000, 30000, 100000]
                h_sh, c5_sh = _nesterov(h_sh, alphas_sh, 2000, 0.002, 0.84, t0, t0 + budget_s - 90)
                if c5_sh < best_c5:
                    best_c5, best_h = c5_sh, h_sh.copy()

    # Phase 5: Final sharp Nesterov polish
    if time.time() - t0 < budget_s - 120:
        alphas_fin = [100000, 500000, 1000000]
        h, c5 = _nesterov(best_h.copy(), alphas_fin, 2500, 0.0005, 0.90, t0, t0 + budget_s - 100)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: 3-point local search (adaptive step)
    if time.time() - t0 < budget_s - 40:
        elapsed = time.time() - t0
        n_tries = int(min((budget_s - 50 - elapsed) * 80, 20000))
        best_h, best_c5 = _local_3pt(best_h, n_tries, rng, t0, t0 + budget_s - 30, init_step=2e-4)

    # Phase 7: 2-point ultra-fine polish
    if time.time() - t0 < budget_s - 3:
        elapsed = time.time() - t0
        n_tries = int(min((budget_s - 4 - elapsed) * 100, 25000))
        best_h, best_c5 = _local_search_2pt(best_h, n_tries, rng, t0, t0 + budget_s - 2, init_step=8e-5)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)

def _local_search_2pt(h, n_tries, rng, t0, t_end, init_step):
    n = len(h)
    c5 = _eval(h)
    best = h.copy()
    best_c = c5
    step = init_step
    for _ in range(n_tries):
        if time.time() - t0 > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < c5 - 1e-18:
                c5 = c
                if c < best_c:
                    best_c = c
                    best = h.copy()
                step = min(step * 1.06, init_step * 3.0)
            else:
                h[i], h[j] = ni - d, nj + d
                if step > init_step * 0.005:
                    step *= 0.6
    return best, best_c
