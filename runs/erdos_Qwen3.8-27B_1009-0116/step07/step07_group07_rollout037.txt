import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n / 2.0) > 1e-10:
        h = _proj_sum(h, n / 2.0)
    return h.copy()

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corrs(h)))

def _soft_obj(h, beta, pen):
    c = _corrs(h)
    m = c.max()
    s = np.exp(beta * (c - m)).sum()
    return m + np.log(s) / beta + pen * (h.sum() - len(h) / 2.0) ** 2

def _soft_grad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h)
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * pen * (h.sum() - n / 2.0)
    return g

def _lbfgs(h, beta, pen, maxiter):
    n = len(h)
    try:
        r = minimize(_soft_obj, h, jac=_soft_grad, args=(beta, pen),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-20,
                              'gtol': 1e-14, 'maxls': 25})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_active(h, n_iter, k_min, k_max, step0, bc):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    beta_mom = 0.7
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_min
    dx = 2.0 / n
    for _ in range(n_iter):
        if bc():
            break
        c = _corrs(h_c)
        cm = c.max()
        top5 = np.argsort(c)[-5:]
        floor = c[top5].min()
        mask = c >= floor - 1e-9
        n_active = mask.sum()
        if n_active < k:
            idx = np.argsort(c)[-k:]
            mask[idx] = True
            n_active = k
        g = np.zeros(n)
        for kk in np.where(mask)[0]:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_c[sh:])
                g[sh:] -= h_c[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_c[:n - sp])
                g[:n - sp] -= h_c[sp:]
        g *= dx / n_active
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta_mom * v + (1.0 - beta_mom) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n / 2.0)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-18:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.02, 0.025)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-10:
                break
            if stall > 40:
                k = min(k + 2, k_max)
                stall = 0
    return best_h, best_c

def _pair_local(h, n_iter, rng, bc):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    for _ in range(n_iter):
        if bc():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = rng.randn() * 2e-5
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            cn = _eval(h_c)
            if cn < best_c:
                best_c = cn
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
    return best_h, best_c

def _ternary_pair(h, n_pairs, rng, n_steps, bc):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    for _ in range(n_pairs):
        if bc():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        lo_d = max(-h_c[i], h_c[j] - 1.0)
        hi_d = min(1.0 - h_c[i], -h_c[j])
        if hi_d - lo_d < 1e-12:
            continue
        a, b = lo_d, hi_d
        for _s in range(n_steps):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            h_c[i], h_c[j] = h_c[i] + m1, h_c[j] - m1
            c1 = _eval(h_c)
            h_c[i], h_c[j] = h_c[i] + m2, h_c[j] - m2
            c2 = _eval(h_c)
            if c1 < c2:
                b = m2
            else:
                a = m1
        d = 0.5 * (a + b)
        h_c[i], h_c[j] = h_c[i] + d, h_c[j] - d
        cn = _eval(h_c)
        if cn < best_c:
            best_c = cn
            best_h = h_c.copy()
            h_c = best_h.copy()
        else:
            h_c[i], h_c[j] = h_c[i] - d, h_c[j] + d
    return best_h, best_c

def _perturb_reopt(h, rng, base_beta, pen, bc):

    h_p = _proj_sum(h_p, n / 2.0)

    return h_r, ct
