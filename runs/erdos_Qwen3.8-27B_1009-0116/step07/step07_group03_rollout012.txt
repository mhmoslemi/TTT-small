def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _obj_smooth(h, alpha):
    c = _compute_c(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _compute_c(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    f = _obj_smooth(h, alpha)
    v = np.zeros(n)
    mu = 0.87
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_smooth(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 12.0)
        else:
            step *= 0.35
            if step < 1e-12:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smooth(h, alpha)
    return h

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _coord_descent(h, n, duration_s, top_T, rng, t_end, wide=False):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    nb = max(3, n // (15 if wide else 10))
    while time.time() < min(end_t, t_end - 5):
        c = _compute_c(h)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        improved = False
        for _ in range(n * 2):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, nb)) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(22):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 20:
            scale = min(1e-3, 2e-5 * np.sqrt(no_improve))
            h += scale * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _fine_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 1.5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _try_start(h0, n, rng, t0, budget_s):
    t_end_full = t0 + budget_s
    best_h = _repair(h0.copy(), n)
    best_c5 = _eval(best_h)
    for alpha in [1000, 5000, 20000, 80000, 300000]:
        te = min(time.time() + 12, t0 + budget_s * 0.12)
        h = _nesterov(best_h.copy(), alpha, 300, 0.004, te)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()
    for top_T in [4, 7, 10, 12]:
        if time.time() > t_end_full * 0.7:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, min(budget_s * 0.08, 80), top_T, rng, t_end_full)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()
    if time.time() < t_end_full * 0.85:
        for alpha in [500000, 2000000]:
            te = min(time.time() + 15, t_end_full * 0.84)
            h = _nesterov(best_h.copy(), alpha, 200, 0.002, te)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5, best_h = c5v, h.copy()
    if time.time() < t_end_full * 0.95:
        for top_T in [10, 15, 20]:
            if time.time() > t_end_full * 0.94:
                break
            h = best_h.copy()
            h, c5v = _coord_descent(h, n, min(budget_s * 0.03, 50), top_T, rng, t_end_full, wide=True)
            if c5v < best_c5:
                best_c5, best_h = c5v, h.copy()
    if time.time() < t_end_full - 3:
        rem = t_end_full - time.time() - 2
        ntr = min(int(rem / 0.004), 50000)
        if ntr > 200:
            h = best_h.copy()
            best_h, best_c5 = _fine_search(h, ntr, rng, t_end_full - 2)
    return best_h, best_c5
