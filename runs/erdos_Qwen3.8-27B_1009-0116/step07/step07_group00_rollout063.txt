import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax(c, alpha):
    m = c.max()
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def _smax_obj(h, alpha):
    return _smax(np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h)), alpha)

def _smax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    e = np.exp(np.clip(alpha * (c - m), -500, 0))
    w = e / e.sum()
    g = np.zeros(n)

        if wk < 1e-14:

        else:
            m2 = -mk
            g[:n - m2] += wk * oh[m2:]
            g[m2:] -= wk * h[:n - m2]
    return g

        j = (i + rng.randint(1, 5)) % nc
        d = 5e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-20:
                c5 = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 5
    try:
        n0 = len(initial_h_values)
        h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    except NameError:
        n0 = 560
        h = _proj(np.random.RandomState(seed).rand(n0), n0 * 0.5)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: smooth proximal (low alpha)
    dl1 = min(dl, t0 + 90)
    h = _prox_loop(h.copy(), 500, 5000, dl1, 0.05)
    if _eval(h) < best_c5:
        best_c5, best_h = _eval(h), h.copy()

    # Phase 2: medium-alpha proximal
    dl2 = min(dl, t0 + 250)
    for a, s0, ns in [(3000, 0.02, 6000), (10000, 0.01, 6000), (30000, 0.005, 5000)]:
        if time.time() > dl2:
            break
        h = _prox_loop(best_h.copy(), a, ns, dl2, s0)
        if _eval(h) < best_c5:
            best_c5, best_h = _eval(h), h.copy()

    # Phase 3: multi-start perturbation
    dl3 = min(dl, t0 + 400)
    for _ in range(6):
        if time.time() > dl3:
            break
        hp = _proj(best_h + 0.04 * rng.randn(len(best_h)), len(best_h) * 0.5)
        hp = _prox_loop(hp, 5000, 2000, dl3, 0.01)
        if _eval(hp) < best_c5:
            best_c5, best_h = _eval(hp), hp.copy()

    # Phase 4: aggressive large-block annealing
    dl4 = min(dl, t0 + 720)
    ha, ca = _anneal(best_h.copy(), 500000, rng, dl4, 0.03, 1e-10, 0.04)
    if ca < best_c5:
        best_c5, best_h = ca, ha.copy()

    # Phase 5: high-alpha proximal refinement
    dl5 = min(dl, t0 + 900)
    for a, s0, ns in [(50000, 0.004, 5000), (200000, 0.002, 4000)]:
        if time.time() > dl5:
            break
        h = _prox_loop(best_h.copy(), a, ns, dl5, s0)
        if _eval(h) < best_c5:
            best_c5, best_h = _eval(h), h.copy()

    # Phase 6: second gentler annealing round
    dl6 = min(dl, t0 + budget_s - 40)
    ha2, ca2 = _anneal(best_h.copy(), 200000, rng, dl6, 0.008, 1e-12, 0.015)
    if ca2 < best_c5:
        best_c5, best_h = ca2, ha2.copy()

    # Phase 7: final proximal + polish
    dl7 = min(dl, t0 + budget_s - 20)
    h = _prox_loop(best_h.copy(), 500000, 3000, dl7, 0.002)
    if _eval(h) < best_c5:
        best_c5, best_h = _eval(h), h.copy()
    rem = max(0, dl - time.time())
    nh, nc = _polish(best_h, min(int(rem / 0.0015), 100000), rng, dl)
    if nc < best_c5:
        best_c5, best_h = nc, nh

    best_h = _repair(best_h, len(best_h))
    return best_h, _eval(best_h), len(best_h)
