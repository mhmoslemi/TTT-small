            best_c5 = c5u
            best_h = hu.copy()

    # Phase 2: Full-res SLSQP sweep
    for alpha in [5000, 20000, 80000, 300000, 1000000]:
        if _el() > 300:
            break
        hn = _slsqp_opt(best_h.copy(), alpha, 1500, 1e-15)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = hn.copy()

    # Phase 3: Nesterov accelerated projected gradient
    if _el() < budget_s - 250:
        for a0, a1, steps, s0, mu in [(2000, 50000, 400, 0.003, 0.7),
                                       (5000, 200000, 350, 0.002, 0.75),
                                       (20000, 500000, 300, 0.0015, 0.8)]:
            if _el() > budget_s - 140:
                break
            hn, c5n = _nesterov_pg(best_h.copy(), steps, a0, a1, s0, mu)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn.copy()

    # Phase 4: Another SLSQP at very high alpha
    if _el() < budget_s - 120:
        for alpha in [3000000, 10000000]:
            if _el() > budget_s - 60:
                break
            hn = _slsqp_opt(best_h.copy(), alpha, 600, 1e-16)
            c5n = _eval(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn.copy()

    # Phase 5: Gradient-informed local search
    if _el() < budget_s - 30:
        ntries = min(int((budget_s - 30 - _el()) / 0.003), 500000)
        best_h, best_c5 = _grad_local_search(best_h.copy(), rng, ntries, 5e-6)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 3:
        npol = min(int((budget_s - 3 - _el()) / 0.001), 600000)
        best_h, best_c5 = _fine_2point(best_h.copy(), rng, npol, 2e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
