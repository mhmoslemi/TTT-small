    h = np.clip(h, 0.0, 1.0).copy()

        return h

Note that |A \ (A-k)| = |A| - |A ∩ (A-k)| = 1 - |A ∩ (A-k)|.

So we want to maximize min_k |A ∩ (A-k)|.

This is related to the "difference set" problem. For a set A of measure 1 in a circle of length 2, we want the autocorrelation A*k̂ to be as large as possible at every non-zero shift.

The lower bound on max_k |A\(A-k)| is related to additive combinatorics.

For the continuous case, the optimal might be something like a "fat Cantor set" or a specific structured pattern.

        g /= na
        h = _repair(h - t * g, n)
    return h

def _toplag(h, nl, rnds, dl):
    n = len(h); dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(rnds):
        if time.time() > dl - 20:
            break
        c = _cvals(h)
        tk = np.argsort(c)[-nl:]
        g = np.zeros(n)
        for k in tk:
            _grad_lag(g, h, n, k)
        g /= nl
        imp = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5, 1e-5]:
            hn = _repair(h - st * g, n)
            cn = _eval(hn)
            if cn < c5c - 1e-16:
                h = hn; c5c = cn; imp = True; break
        if not imp:
            break
    return h, c5c

def _deflate(h, iters, dl):
    n = len(h)
    for _ in range(iters):
        if time.time() > dl - 10:
            break
        c = _cvals(h)
        ks = c.argmax()
        g = np.zeros(n)
        _grad_lag(g, h, n, ks)
        c5c = c.max()
        imp = False
        for st in [0.03, 0.015, 0.007, 0.003, 0.001, 5e-4, 1e-4, 5e-5]:
            hn = _repair(h - st * g, n)
            cn = _eval(hn)
            if cn < c5c - 1e-15:
                h = hn; imp = True; break
        if not imp:
            break
    return h

def _local2(h, it, rng, dl):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(it):
        if time.time() > dl - 2:
            break
        i = rng.randint(n); j = (i + 1 + rng.randint(0, 5)) % n
        d = 4e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-19:
                c5c = cn
                if cn < bc:
                    bc = cn; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _local3(h, it, rng, dl):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(it):
        if time.time() > dl - 3:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 10)) % n
        k = (i + rng.randint(11, 25)) % n
        d1 = 2e-5 * rng.randn(); d2 = 2e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < c5c - 1e-19:
                c5c = cn
                if cn < bc:
                    bc = cn; bh = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return bh, bc

def _interp_up(h, f):
    n = len(h); m = n * f
    ox = np.arange(n); nx = np.arange(m) / f
    return np.clip(np.interp(nx, ox, h), 0, 1)

def _init_cos(n, fr, am, ph):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.ones(n) * 0.5
    for f, a in zip(fr, am):
        h += a * np.cos(2 * np.pi * f * x + ph)
    return np.clip(h, 0, 1)

def _init_blocks(n):
    h = np.zeros(n); bl = max(4, n // 16)
    for i in range(0, n, bl * 2):
        h[i:i + bl] = 1.0
    return h

def _init_smooth(n, sd):
    r = np.random.RandomState(sd)
    h = r.rand(n)
    k = np.array([1, 2, 3, 2, 1], dtype=float)
    k /= k.sum()
    for _ in range(4):
        h = np.convolve(h, k, mode='same')
    return np.clip(h, 0, 1)

def _init_bumps(n):
    x = np.linspace(0, 2, n, endpoint=False)
    s = 0.25
    h = (0.4 * np.exp(-0.5 * ((x - 0.4) / s) ** 2) +
         0.25 * np.exp(-0.5 * ((x - 1.1) / s) ** 2) +
         0.25 * np.exp(-0.5 * ((x - 1.7) / s) ** 2))
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); dl = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh = h0.copy(); bc5 = _eval(bh); bn = n0

    # Phase 1: Multi-start at low resolution
    nl = 64
    inits = [
        _init_cos(nl, [0.5, 1, 2], [0.15, 0.1, 0.05], 0),
        _init_cos(nl, [0.5, 1.5, 3], [0.12, 0.08, 0.04], 1.0),
        _init_cos(nl, [1, 2, 4], [0.1, 0.06, 0.03], 0.5),
        _init_cos(nl, [0.25, 0.75, 1.5], [0.2, 0.1, 0.05], 0.3),
        _init_blocks(nl),
        _init_smooth(nl, seed),
        _init_bumps(nl),
    ]
    cands = []
    for hi in inits:
        hi = _repair(hi, nl)
        ho = _lbfgs(hi, _pnorm_obj, _pnorm_grad2, (30,), 400, dl)
        ho, cv = _toplag(ho, 3, 80, dl)
        cands.append((ho, cv))

    ups = []
    for hc, cv in cands:
        hu = _repair(_interp_up(hc, n0 // nl), n0)
        ups.append((hu, _eval(hu)))
    all_c = [(h0, bc5)] + ups
    all_c.sort(key=lambda x: x[1])

    # Phase 2: Full-res p-norm refinement on top candidates
    for idx in range(min(4, len(all_c))):
        if time.time() > dl - 400:
            break
        hc = all_c[idx][0].copy()
        for p in [12, 40]:
            if time.time() > dl - 300:
                break
            hc = _lbfgs(hc, _pnorm_obj, _pnorm_grad2, (p,), 2500, dl)
            cv = _eval(hc)
            if cv < bc5:
                bc5 = cv; bh = hc.copy()

    # Phase 3: LSE with increasing sharpness
    for a in [2000, 20000, 100000, 500000]:
        if time.time() > dl - 150:
            break
        hc = _lbfgs(bh.copy(), _lse_obj, _lse_grad, (a,), 4000, dl)
        cv = _eval(hc)
        if cv < bc5:
            bc5 = cv; bh = hc.copy()

    # Phase 4: Subgradient + deflate
    if time.time() < dl - 200:
        h = _subgrad(bh.copy(), 600, 0.012, 1e-8, dl)
        cv = _eval(h)
        if cv < bc5:
            bc5 = cv; bh = h.copy()
    if time.time() < dl - 120:
        h = _deflate(bh.copy(), 300, dl)
        cv = _eval(h)
        if cv < bc5:
            bc5 = cv; bh = h.copy()

    # Phase 5: Top-lag with various widths
    if time.time() < dl - 100:
        for nl2 in [2, 3, 5, 8, 4]:
            if time.time() > dl - 100:
                break
            h, cv = _toplag(bh.copy(), nl2, 300, dl)
            if cv < bc5:
                bc5 = cv; bh = h.copy()

    # Phase 6: Very high alpha LSE + subgradient polish
    if time.time() < dl - 60:
        h = _lbfgs(bh.copy(), _lse_obj, _lse_grad, (2000000,), 5000, dl)
        cv = _eval(h)
        if cv < bc5:
            bc5 = cv; bh = h.copy()
    if time.time() < dl - 35:
        h = _subgrad(bh, 2000, 0.004, 1e-10, dl)
        cv = _eval(h)
        if cv < bc5:
            bc5 = cv; bh = h.copy()

    # Phase 7: Deflate again
    if time.time() < dl - 20:
        h = _deflate(bh, 500, dl)
        cv = _eval(h)
        if cv < bc5:
            bc5 = cv; bh = h.copy()

    # Phase 8: Local search
    if time.time() < dl - 5:
        h, cv = _local3(bh, 50000, rng, dl)
        if cv < bc5:
            bc5 = cv; bh = h.copy()
    if time.time() < dl - 1:
        h, cv = _local2(bh, 80000, rng, dl)
        if cv < bc5:
            bc5 = cv; bh = h.copy()

    bh = _repair(bh, bn)
    bc5 = _eval(bh)
    return (bh, bc5, bn)
