import time

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5

        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:

        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lse(h, beta, penalty):
    """Gradient of LSE smooth max of correlations + constraint penalty."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights (numerically stable)
    cm = np.max(c)
    exp_c = np.exp(beta * (c - cm))
    w = exp_c / exp_c.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    g *= dx
    if penalty > 0:
        g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

def _adam_lse(h0, beta_schedule, n_steps_per_beta, lr, penalty, bc_fn):
    """Adam optimizer with LSE smooth max, adaptive beta."""
    n = len(h0)
    h_c = h0.copy()
    best_c = _eval(h_c)
    best_h = h_c.copy()
    
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    for beta in beta_schedule:
        for _ in range(n_steps_per_beta):
            if bc_fn():

            g = _grad_lse(h_c, beta, penalty)

                best_h = h_new.copy()

        if bc_fn():

import numpy as np

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    return _proj_sum(np.clip(h, 0, 1), n * 0.5)

def _grad_lse(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    exp_c = np.exp(np.clip(beta * (c - cm), -80, 80))
    w = exp_c / exp_c.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    if penalty > 0:
        g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

def _adam_lse(h0, beta_list, steps_per_beta, lr_base, penalty, bc_fn):
    n = len(h0)
    h_c = h0.copy()
    best_c = _eval(h_c)
    best_h = h_c.copy()
    b1, b2, eps = 0.82, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for idx, beta in enumerate(beta_list):
        lr = lr_base / (1.0 + idx * 0.3)
        for _ in range(steps_per_beta):
            if bc_fn():
                return best_h, best_c
            t += 1
            g = _grad_lse(h_c, beta, penalty)
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g * g
            mhat = m / (1.0 - b1 ** t)
            vhat = v / (1.0 - b2 ** t)
            h_new = _proj_sum(h_c - lr * mhat / (np.sqrt(vhat) + eps), n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_c = h_new
    return best_h, best_c

def _lbfgs_pnorm(h, p, penalty, maxiter):
    n = len(h)
    npts = len(h)

    def obj(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full') * 2.0 / npts
        val = np.sum(np.maximum(c, 1e-30) ** p) ** (1.0 / p)
        return val + penalty * (hh.sum() - npts * 0.5) ** 2

    def grad(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full') * 2.0 / npts
        cc = np.maximum(c, 1e-30)
        s = np.sum(cc ** p)
        fac = s ** (1.0 / p - 1.0)
        g = np.zeros(npts)
        for k in range(2 * npts - 1):
            wk = fac * cc[k] ** (p - 1) * 2.0 / npts
            if wk < 1e-18:
                continue
            sh = (npts - 1) - k
            if sh >= 0:
                g[:npts - sh] += wk * (1.0 - hh[sh:])
                if sh > 0:
                    g[sh:] -= wk * hh[:npts - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - hh[:npts - sp])
                g[:npts - sp] -= wk * hh[sp:]
        g += 2.0 * penalty * (hh.sum() - npts * 0.5)
        return g

    bnds = [(0.0, 1.0)] * npts
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15, 'maxls': 40})
        return _repair(res.x, npts)
    except Exception:
        return _repair(h, npts)

def _local_search(h, n_iter, rng, el_fn, bc_fn, scale_init=2e-4, n_pts=3):

    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = scale_init

        else:
            ds[-1] = -(ds[:-1]).sum()
        trial = h_c.copy()

        c_new = _eval(h_c)
        if c_new < best_c:
            best_c = c_new
            best_h = h_c.copy()
        if it % 3000 == 0:

    return best_h, best_c

def _coord_descent(h, n_sweeps, bc_fn):

    h_c = h.copy()

        improved = False

                hi_j = 1.0
                c_lo = c_hi = 1e18
                h_save = h_c[j]
                h_c[j] = 0.0
                s_adj = h_c.sum() - n * 0.5
                if abs(s_adj) < 1e-15:

                s_adj = h_c.sum() - n * 0.5
                if abs(s_adj) < 1e-15:
                    c_hi = _eval(h_c)

                    h_c[j] = mid
                    h_c = _proj_sum(h_c, n * 0.5)
                    c_mid = _eval(h_c)
                    if c_mid < c_lo:
                        lo_j = mid
                        c_lo = c_mid
                    else:
                        hi_j = mid
                    h_c[j] = mid
