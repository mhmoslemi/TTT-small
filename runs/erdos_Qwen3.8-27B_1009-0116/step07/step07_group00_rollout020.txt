    bh = h.copy()
    bc = c5c
    sizes = [3, 5, 7, 10, 15, 20]
    for _ in range(niter):
        if time.time() > deadline:
            break
        B = sizes[rng.randint(len(sizes))]
        if B >= n // 2:
            continue
        i = rng.randint(0, n - B)
        j = rng.randint(0, n - B)
        if abs(i - j) < B:
            continue
        si = h[i:i+B].copy()
        sj = h[j:j+B].copy()
        h[i:i+B] = sj
        h[j:j+B] = si
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
        else:
            h[i:i+B] = si
            h[j:j+B] = sj
    return bh, bc

def _grad_polish(h, rng, niter, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 3e-4
    for _ in range(niter):
        if time.time() > deadline:
            break
        g = _subgrad(h, 2)
        ng = np.linalg.norm(g)
        if ng > 1e-14:
            d = -g / ng * step
        else:
            d = rng.randn(n) * step
        d += 8e-6 * rng.randn(n)
        d -= d.mean()
        hn = _repair(h + d, n)
        c5n = _eval(hn)
        if c5n < c5c:
            h, c5c = hn, c5n
            step = min(step * 1.003, 1.5e-3)
            if c5n < bc:
                bc = c5n
                bh = h.copy()
        else:
            step *= 0.65
            if step < 5e-7:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 4)) % n
                dd = 2e-5 * rng.randn()
                ni, nj = h[i] + dd, h[j] - dd
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c:
                        c5c = c5n
                        if c5n < bc:
                            bc = c5n
                            bh = h.copy()
                    else:
                        h[i], h[j] = ni - dd, nj + dd
                step = 1e-4
    return bh, bc

def _smax_obj(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -300, 300)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _slsqp_short(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (n,)}]
    try:
        r = minimize(_smax_obj, h, args=(alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _cons_fun(x, n):
    return x.sum() - n * 0.5

def _sa_block(h, rng, niter, deadline, T0, T1, sizes):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        B = sizes[rng.randint(len(sizes))]
        i = rng.randint(0, max(1, n - B))
        seg = h[i:i+B].copy()
        noise = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(B)
        noise -= noise.mean()
        seg += noise
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i:i+B] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP warm start
    dl1 = min(deadline, t0 + 80)
    h1 = _slsqp_short(h, 3000, 5, 200, dl1)
    c1 = _eval(h1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Nesterov exact subgradient (multiple configs)
    dl2 = min(deadline, t0 + 350)
    configs2 = [(2, 0.004, 4000, 0.82), (3, 0.003, 3500, 0.80),
                (1, 0.002, 4000, 0.85), (2, 0.001, 3500, 0.84)]
    for tk, st, ns, bt in configs2:
        if time.time() > dl2:
            break
        ht = _nesterov_ex(best_h.copy(), tk, ns, st, dl2, bt)
        c5 = _eval(ht)
        if c5 < best_c5:
            best_c5 = c5
            best_h = ht.copy()

    # Phase 3: Simulated annealing with block perturbations
    dl3 = min(deadline, t0 + 600)
    best_h, best_c5 = _sa_block(best_h.copy(), rng, 30000, dl3,
                                0.002, 1e-8, [3, 5, 7, 10, 15])

    # Phase 4: Block swap local search
    dl4 = min(deadline, t0 + 850)
    best_h, best_c5 = _block_swap(best_h.copy(), rng, 60000, dl4)

    # Phase 5: Nesterov refinement after local search
    dl5 = min(deadline, t0 + 1000)
    for tk, st, ns, bt in [(2, 0.001, 2500, 0.85), (1, 0.0005, 2500, 0.88)]:
        if time.time() > dl5:
            break
        ht = _nesterov_ex(best_h.copy(), tk, ns, st, dl5, bt)
        c5 = _eval(ht)
        if c5 < best_c5:
            best_c5 = c5
            best_h = ht.copy()

    # Phase 6: Gradient-guided fine polish
    dl6 = deadline
    best_h, best_c5 = _grad_polish(best_h.copy(), rng, 80000, dl6)

    # Final repair
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
