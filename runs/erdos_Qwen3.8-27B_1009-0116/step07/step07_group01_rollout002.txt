def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_grad(h):
    """Returns (c5_value, gradient array)"""
    n = len(h)
    dx = 2.0 / n
    bh = 1.0 - h
    c = np.correlate(h, bh, mode='full') * dx
    c5 = c.max()
    # Find active lags
    active = np.where(c > c5 - 1e-10)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Weight active lags uniformly (or by softmax)
    w = np.ones(len(active)) / len(active)
    g = np.zeros(n)
    for idx, k in enumerate(active):
        m = k - (n - 1)
        wk = w[idx] * dx
        # ∂c_k/∂h[j]:
        # j+m in [0,n-1]: contributes (1-h[j+m])
        # j-m in [0,n-1]: contributes -h[j-m]
        if m >= 0:
            # j+m in [0,n-1] => j in [0, n-1-m]
            j1 = np.arange(0, n - m)
            g[j1] += wk * (1.0 - h[j1 + m])
            if m > 0:

                g[j2] -= wk * h[j2 - m]

                j2 = np.arange(0, n - p)
                g[j2] -= wk * h[j2 + p]
    return c5, g

def _proj_sum_box(v, target):
    """Project v onto {x: sum(x)=target, 0≤x≤1}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0, 1)

def _nesterov_grad(h, alpha, n_iter, step0, top_k, t_end):

    n = len(h)
    h_x = h.copy()
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h_x)
    best_h = h_x.copy()
    best_c = cur_c
    t0 = time.time()
    
    for it in range(n_iter):
        if time.time() - t0 > t_end:
            break
        # Compute gradient
        c = np.correlate(h_x, 1.0 - h_x, mode='full') * (2.0 / n)
        m = c.max()

            c_a = c[sorted_idx]

            s -= s.max()
            w = np.exp(s)

        else:

def _slsqp_fast(h, alpha, maxiter, t_end):

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        dx_ = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            mm = k - (n - 1)
            wk = w[k] * dx_
            if mm >= 0:
                j1 = np.arange(0, n - mm)
                g[j1] += wk * (1.0 - x[j1 + mm])
                if mm > 0:
                    j2 = np.arange(mm, n)
                    g[j2] -= wk * x[j2 - mm]

                if p > 0:
                    j2 = np.arange(0, n - p)
                    g[j2] -= wk * x[j2 + p]
        return g

        if time.time() - t0 < t_end:

        pass
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_box_sum(v, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0, 1)

def _smoothed_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smoothed_grad(x, alpha, top_k):
    n = len(x)
    dx_ = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx_
    m = c.max()
    if top_k > 0 and top_k < 2 * n - 1:
        idx = np.argsort(c)[-top_k:]
        c_a = c[idx]
        s = np.clip(alpha * (c_a - c_a.max()), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
    else:
        idx = np.arange(2 * n - 1)
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-12:
            continue
        mm = k - (n - 1)
        wk = w[ii] * dx_
        if mm >= 0:
            j1 = np.arange(0, n - mm)
            g[j1] += wk * (1.0 - x[j1 + mm])
            if mm > 0:
                j2 = np.arange(mm, n)
                g[j2] -= wk * x[j2 - mm]
        else:
            p = -mm
            j1 = np.arange(p, n)
            g[j1] += wk * (1.0 - x[j1 - p])
            if p > 0:
                j2 = np.arange(0, n - p)
                g[j2] -= wk * x[j2 + p]
    return g

def _nesterov(h, alpha, n_iter, step0, top_k, t_limit):
    n = len(h)
    h_x = _repair(h.copy(), n)
    v = np.zeros(n)
    mu = 0.75
    step = step0
    cur_c = _eval(h_x)
    best_h = h_x.copy()
    best_c = cur_c
    t0 = time.time()
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > t_limit:
            break
        g = _smoothed_grad(h_x, alpha, top_k)
        v = mu * v + (1.0 - mu) * g
        accepted = False
        for _t in range(10):
            h_new = _proj_box_sum(h_x - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h_x = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 12.0)
                accepted = True
                break
            step *= 0.4
        if not accepted:
            step = min(step * 1.5, step0 * 12.0)
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best_h = h_x.copy()
            stall = 0
        else:
            stall += 1
            if stall > 800:
                h_x = _repair(h_x + 0.01 * np.random.randn(n) * 0.1, n)
                v = np.zeros(n)
                step = step0
                stall = 0
                cur_c = _eval(h_x)
    return _repair(best_h, n), best_c

def _slsqp_wrap(h, alpha, maxiter, t_limit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        return _smoothed_obj(x, alpha)

    def _obj_jac(x):
        return _smoothed_grad(x, alpha, 0)

    target_sum = n * 0.5

    def _cons_fun(x):
        return x.sum() - target_sum

    def _cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    t0 = time.time()
    try:
        res = minimize(_obj, h, jac=_obj_jac, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() - t0 <= t_limit:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step, t_limit):

    h_s = h.copy()
    c5cur = _eval(h_s)
    best_h = h_s.copy()

    t0 = time.time()

            break

        d1 = base_step * rng.randn()
        d2 = base_step * rng.randn()
        ni, nj, nk = h_s[i]+d1, h_s[j]+d2, h_s[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h_s[i], h_s[j], h_s[k] = ni, nj, nk
            c5n = _eval(h_s)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h_s.copy()
            else:
                h_s[i], h_s[j], h_s[k] = ni-d1, nj-d2, nk+d1+d2
        i2 = rng.randint(n)
        j2 = (i2 + 1 + rng.randint(0, 2)) % n

        ni2, nj2 = h_s[i2]+d, h_s[j2]-d
        if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:

            c5n = _eval(h_s)
            if c5n < c5cur:

                if c5n < best_c:
                    best_c = c5n
                    best_h = h_s.copy()
            else:
                h_s[i2], h_s[j2] = h_s[i2]-d, h_s[j2]+d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out
