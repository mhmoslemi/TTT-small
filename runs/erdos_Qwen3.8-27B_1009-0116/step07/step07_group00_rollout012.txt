from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):

def _proj(h, target):

        else:

        blocks = [1, 2, 3, 4, 6]
        niter = int((dl5 - time.time()) / 0.004)
        h, c5v = _block_anneal(best_h.copy(), niter, rng, dl5, blocks)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 6: Final coordinate descent polish
    dl6 = min(deadline, t0 + budget_s - 15)
    if time.time() < dl6:
        rem = max(1, int((dl6 - time.time()) / 0.06))
        h, c5v = _coord_descent(best_h.copy(), rem, 80, dl6, rng)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 7: Fine 2-point polish
    if time.time() < deadline:
        n_polish = int((deadline - time.time()) / 0.0012)
        h, c5v = _fine_polish(best_h.copy(), n_polish, rng, deadline)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
