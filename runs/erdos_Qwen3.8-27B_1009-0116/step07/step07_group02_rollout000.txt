def _block_search(h, n_iter, rng, deadline):
    """Block-based local search: perturb blocks of consecutive values."""
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):

        start = rng.randint(0, n - blen)
        block = h[start:start+blen].copy()
        # Random perturbation of the block
        pert = rng.randn(blen) * 0.005
        # Try shifting the block
        mode = rng.randint(3)
        if mode == 0:
            new_block = block + pert
        elif mode == 1:
            # Cyclic shift of block
            shift = rng.randint(1, blen)
            new_block = np.roll(block, shift)
        else:
            # Reverse block
            new_block = block[::-1]

                best_c = c5n
                best_h = h.copy()
        else:

            h = _repair(h, n)

                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
