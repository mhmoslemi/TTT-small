def _repair(h, n):

    if abs(s - target) > 1e-9:

        if abs(diff) > 1e-12:

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()) / np.sum(np.exp(s - s.max()))
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        sh = (n - 1) - idx[ii]
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov_opt(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_val(h, alpha, topk)
    best_f = f
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.85
    step = step0
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_val(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
            stall = 0
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.5
            hn = _proj(h - step * v, n * 0.5)
            f = _smax_val(hn, alpha, topk)
            h = hn
            stall += 1
            if stall > 200:
                break
    return _repair(h, n), _eval_full(_repair(h, n))

def _adam_opt(h, alpha, topk, steps, lr, deadline):
    n = len(h)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    f = _smax_val(h, alpha, topk)
    best_f = f
    best_h = h.copy()
    t = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        t += 1
        g = _grad_topk(h, alpha, topk)
        m_hat = b1 * m_hat + (1 - b1) * g
        v_hat = b2 * v_hat + (1 - b2) * g * g
        m_hat_c = m_hat / (1 - b1**t)
        v_hat_c = v_hat / (1 - b2**t)
        hn = h - lr * m_hat_c / (np.sqrt(v_hat_c) + eps)
        hn = _proj(hn, n * 0.5)
        fn = _smax_val(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            lr *= 0.8
    return _repair(h, n), _eval_full(_repair(h, n))

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval_full(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.001 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval_full(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_search(h, niter, rng, deadline):
    """Targeted fine search using gradient direction"""
    n = len(h)
    c5c = _eval_full(h)
    bh = h.copy()
    bc = c5c
    step = 5e-5
    for it in range(niter):
        if time.time() > deadline:
            break
        # Get gradient of top-5 correlations
        g = _grad_topk(h, 10000, 5)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        direction = -g / gn
        # Try a few step sizes
        improved = False
        for s in [step, step * 0.5, step * 0.1]:
            hn = _proj(h + s * direction, n * 0.5)
            c5n = _eval_full(hn)
            if c5n < c5c - 1e-18:
                h = hn
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
                step = min(step * 1.1, 1e-3)
                improved = True
                break
        if not improved:
            # Try random perturbations
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = step * 0.1 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval_full(h)
                if c5n < c5c - 1e-18:
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
            step *= 0.95
    return bh, bc

For the worst lag k*, the correlation is:

The "error" at each position i is h[i] * (1-h[i+k]). To reduce c_k*, we want to reduce the contribution of each term. The most effective changes are at positions where h[i] * (1-h[i+k]) is largest.

We could do: for the top-50 positions with largest contribution, try small perturbations that reduce the product h[i] * (1-h[i+k]).

def _directed_search(h, niter, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval_full(h)
    bh = h.copy()
    bc = c5c
    
    for it in range(niter):
        if time.time() > deadline:
            break
        # Find the worst lag
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = c.argmax()
        shift = k_star - (n - 1)  # actual shift
        
        # Find positions with largest contribution
        if shift >= 0:
            # c_shift = sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i]) * dx
            contrib = h[shift:] * (1.0 - h[:n-shift])
        else:
            p = -shift
            # c_{-p} = sum_{i=p}^{n-1} h[i] * (1-h[i-p]) * dx
            contrib = h[p:] * (1.0 - h[:n-p])
        
        # Get top 20 contributing positions
        top_idx = np.argsort(contrib)[-20:]
        
        improved = False
        for idx in top_idx:
            if shift >= 0:
                i = idx + shift  # position in h that's h[i+shift]
                j = idx  # position in h that's h[i]
            else:
                i = idx  # position in h that's h[i]
                j = idx + (-shift)  # position in h that's h[i-p]
            
            # Try to reduce h[i] * (1-h[j]) or h[j] * (1-h[i])
            # Reduce h[i]: move small amount from h[i] to h[j] (if h[j] < 1)
            d = 0.001 * rng.random()
            for s in [d, -d, d*0.5, -d*0.5]:
                hi, hj = h[i] + s, h[j] - s
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h[i], h[j] = hi, hj
                    c5_try = _eval_full(h)
                    if c5_try < c5c - 1e-18:
                        c5c = c5_try
                        if c5c < bc:
                            bc = c5c
                            bh = h.copy()
                        improved = True
                        break
                    else:
                        h[i], h[j] = hi - s, hj + s
            if improved:
                break
        if not improved and it % 100 == 0:
            # Occasionally try random moves
            i = rng.randint(n)
            j = rng.randint(n)
            d = 0.001 * rng.randn()
            hi, hj = h[i] + d, h[j] - d
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h[i], h[j] = hi, hj
                h = _repair(h, n)
                c5_try = _eval_full(h)
                if c5_try < c5c:
                    c5c = c5_try
                    if c5c < bc:
                        bc = c5c
                        bh = h.copy()
                else:
                    h[i], h[j] = hi - d, nj + d  # BUG: should be hj + d
                    h = _repair(h, n)
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval_full(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_val(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c.copy()
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        sh = (n - 1) - idx[ii]
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov_phase(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_val(h, alpha, topk)
    best_f = f
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.82
    step = step0
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_val(hn, alpha, topk)
        if fn < f - 1e-16:
            h = hn
            f = fn
            step = min(step * 1.04, step0 * 10.0)
            stall = 0
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.4
            hn = _proj(h - step * v, n * 0.5)
            f = _smax_val(hn, alpha, topk)
            h = hn
            stall += 1
            if stall > 300:
                v *= 0.1
                step = step0
                stall = 0
    return _repair(h, n), _eval_full(_repair(h, n))

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval_full(h)
    bh = h.copy()
    bc = c5c

        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))

        seg = h[i0:i0 + bs].copy()

            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc
