        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    best_h = h.copy()
    best_f = f
    stall = 0
    for i in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-15:
            h = hn
            f = fn
            if fn < best_f:
                best_f = fn
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            step = min(step * 1.015, step0 * 10.0)
        else:
            step *= 0.55
            if step < 1e-12:
                break
            stall += 1
        if stall > 500:
            break
    return _repair(best_h, n)

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def _obj(x):
        return _smax_obj(x, alpha, topk)
    def _jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_obj, h, jac=_jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, block_sizes):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = block_sizes[rng.randint(len(block_sizes))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, ndim):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        idx = rng.randint(nc, size=ndim)
        idx = np.unique(idx)
        if len(idx) < 2:
            continue
        scale = 8e-7 if ndim <= 2 else 5e-7
        d = scale * rng.randn(len(idx))
        d -= d.mean()
        old_vals = h[idx].copy()
        new_vals = h[idx] + d
        if np.all(new_vals >= 0) and np.all(new_vals <= 1):
            h[idx] = new_vals
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[idx] = old_vals
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n0 = len(initial_h_values)
    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP quick refinement on parent
    dl1 = min(deadline, t0 + 100)
    for alpha, tk, mi in [(5000, 50, 200), (20000, 30, 200)]:
        if time.time() > dl1:
            break
        h = _slsqp_opt(h_parent.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Multi-scale Nesterov (broad to sharp)
    dl2 = min(deadline, t0 + 420)
    for alpha, tk, st, ns in [(5000, 80, 0.006, 2500),
                               (20000, 40, 0.003, 2500),
                               (100000, 20, 0.001, 3000),
                               (500000, 10, 0.0005, 3000),
                               (2000000, 6, 0.0002, 3000)]:
        if time.time() > dl2:
            break
        h = _nesterov(best_h.copy(), alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Block annealing (diverse block sizes, long)
    dl3 = min(deadline, t0 + 720)
    blocks = [1, 2, 3, 4, 5, 7, 10, 14, 20]
    h_a, c5_a = _anneal(best_h.copy(), 80000, rng, dl3, 0.006, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5a
        best_h = h_a.copy()

    # Phase 4: Post-anneal Nesterov + SLSQP at very high alpha
    dl4 = min(deadline, t0 + 940)
    for alpha, tk, st, ns in [(10000000, 5, 0.0001, 4000),
                               (50000000, 3, 0.00005, 4000)]:
        if time.time() > dl4:
            break
        h = _nesterov(best_h.copy(), alpha, tk, ns, st, dl4)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    if time.time() < dl4:
        h = _slsqp_opt(best_h.copy(), 50000000, 4, 150, dl4)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Multi-dim polish (2-point then 3-point)
    dl5 = deadline
    remaining = max(0, dl5 - time.time())
    n_pol = min(int(remaining * 0.5 / 0.001), 40000)
    best_h, best_c5 = _polish(best_h.copy(), n_pol, rng, t0 + budget_s * 0.55, 2)

    # Phase 6: 3-point fine polish
    rem2 = max(0, t0 + budget_s - 5 - time.time())
    n_3 = min(int(rem2 / 0.0015), 30000)
    best_h, best_c5 = _polish(best_h.copy(), n_3, rng, t0 + budget_s - 5, 3)

    # Final: one more ultra-high alpha SLSQP if time permits
    if time.time() < t0 + budget_s - 4:
        h = _slsqp_opt(best_h.copy(), 100000000, 3, 100, t0 + budget_s - 3)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
