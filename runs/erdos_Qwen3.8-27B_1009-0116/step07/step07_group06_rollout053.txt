```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:

        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)

        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]

def _pair_cd(h, n_iter, rng, deadline, greedy_frac=0.3):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stable > 5000:
            break
        # Pair selection: mix of random and near-peak
        if rng.rand() < greedy_frac:
            # Pick i near a peak of c, j random
            idx_peak = np.argmax(c)
            shift_peak = idx_peak - (n - 1)
            # Coordinates involved in this peak are near each other with offset shift_peak
            # Pick i randomly, j = i + shift_peak or i - shift_peak
            i = rng.randint(n)
            if abs(shift_peak) < n:
                j = (i + shift_peak) % n if shift_peak >= 0 else (i - abs(shift_peak)) % n
                if j == i:
                    j = (j + 1) % n
            else:
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at(h, i, n) - _grad_at(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(50):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g

            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid

            else:

        d_opt = 0.5 * (d_lo + d_hi)

        h[i] += d_opt

        c += d_opt * g

        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()

        else:

                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:

        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = c.max()
        if c5 < best_c - 1e-15:
            best_c = c5
            best_h = h.copy()
        # Get active set
        thresh = c5 - 1e-6
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(c)[-n_active:]
        elif len(active) > n_active:
            active = np.argsort(c)[-n_active:]
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                # c[k] = sum_i h[i+s]*(1-h[i]), i=0..n-1-s
                # dg/dh[i] = -(h[i+s]) if i < n-s (from (1-h[i]) term)
                #           + (1-h[i-s]) if i>=s (from h[i+s] term, i+s = i's partner)
                # Actually: ∂c[k]/∂h[p] for s>=0:
                #   (1-h[p-s]) if s<=p <=n-1-s... 
                # Let me use the formula: ∂c[k]/∂h[p] = (1-h[p-s])*[p>=s] - h[p+s]*[p+s<n]
                for p in range(n):
                    val = 0.0
                    if s <= p:
                        val += (1.0 - h[p - s])
                    if p + s < n:
                        val -= h[p + s]
                    g[p] += val
            else:
                t = -s
                # c[k] = sum_i h[i]*(1-h[i+t]), i=0..n-1-t
                # ∂c[k]/∂h[p] = (1-h[p+t])*[p+t<n] - h[p-t]*[p>=t]
                for p in range(n):
                    val = 0.0
                    if p + t < n:
                        val += (1.0 - h[p + t])
                    if p >= t:
                        val -= h[p - t]
                    g[p] += val
        g *= dx / len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        # Projected gradient step with backtracking
        step = min(0.05, 1.0 / gn)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_to_sum(h - mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            h = np.clip(h + 0.01 * rng.randn(n), 0, 1)
            h = _proj_to_sum(h, n * 0.5)
    return best_h, best_c
```

Actually, for the subgradient, I can use the _grad_at function:

The gradient of c[k] with respect to h is what I need. Let me define:
grad_c_k[p] = ∂c[k]/∂h[p]

For shift s >= 0: grad_c_k[p] = (1-h[p-s])*[p>=s] - h[p+s]*[p+s<n]
For shift s < 0, t=-s: grad_c_k[p] = (1-h[p+t])*[p+t<n] - h[p-t]*[p>=t]

I can vectorize this:

```python
def _grad_c_k(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    dx = 2.0 / n
    if s >= 0:
        # (1-h[p-s]) for p=s..n-1
        g[s:] += (1.0 - h[:n-s])
        # -h[p+s] for p=0..n-1-s
        g[:n-s] -= h[s:]
    else:
        t = -s
        # (1-h[p+t]) for p=0..n-1-t
        g[:n-t] += (1.0 - h[t:])
        # -h[p-t] for p=t..n-1
        g[t:] -= h[:n-t]
    return g * dx
```

This is O(n) per call, vectorized. For n_active=5, total gradient computation is 5*O(n) = O(5n) ≈ 2800 operations. Very fast.

Let me also think about what "greedy pair selection" means more carefully. The idea is:
- The correlation peaks are at certain shifts s
- At shift s, the coordinates that interact are (i, i+s) for valid i
- So if shift s is a peak, changing h[i] and h[i+s] for various i would be most effective

For the greedy selection:
1. Find the top correlation peak index k_peak
2. shift = k_peak - (n-1)
3. Pick random i, set j = i + shift (mod n) or i - shift

This should focus the optimization on the most problematic lags.

Let me also add a "block flip" perturbation:
- Pick a random block of length L (5-30)
- Flip h[i] -> 1-h[i] for i in the block
- Check if C5 improves

And a "cyclic shift" perturbation:
- Shift the entire h array by 1-3 positions
- Check if C5 improves

These are structural perturbations that can jump to different basins.

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _grad_c_k(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    else:
        t = -s
        if t < n:
            g[:n - t] += (1.0 - h[t:])
            g[t:] -= h[:n - t]
    return g * (2.0 / n)

def _pair_cd(h, n_iter, rng, deadline, greedy_frac=0.35):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stable > 6000:
            break
        if rng.rand() < greedy_frac:
            top_idx = np.argpartition(c, -3)[-3:]
            idx_peak = top_idx[rng.randint(3)]
            shift_peak = idx_peak - (n - 1)
            i = rng.randint(n)
            if 1 <= abs(shift_peak) < n:
                j = (i + shift_peak) % n
                if j == i:
                    j = (j + 1) % n
            else:
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at(h, i, n) - _grad_at(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            stable += 1
            continue
        for _ in range(50):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _active_subgrad(h, n_iter, rng, deadline):
    n = len(h)
    m_top = 5
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    no_improve = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if no_improve > 300:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        c5 = c.max()
        top_k = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top_k:
            g += _grad_c_k(h, k, n)
        g /= m_top
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.03, 0.5 / gn)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj_to_sum(h - mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                no_improve = 0
                break
        if not improved:
            h = np.clip(h + 0.02 * rng.randn(n), 0, 1)
            h = _proj_to_sum(h, n * 0.5)
            no_improve += 1
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def _block_flip(h, rng, n):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(200):
        L = rng.randint(4, 25)
        start = rng.randint(0, n - L)
        h2 = h.copy()
        h2[start:start+L] = 1.0 - h2[start:start+L]
        h2 = _proj_to_sum(h2, n * 0.5)
        c2 = _eval(h2)
        if c2 < best_c - 1e-8:
            best_c = c2
            best_h = h2
            h = h2
    return best_h, best_c

def _coarse_sweep_cd(h, n_iter, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        h2 = h.copy()
        opts = []
        for d in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
            v = h[i] + d
            if 0.0 <= v <= 1.0:
                h2[i] = v
                c = _eval(h2)
                if c < c5_cur:
                    opts.append((c, d))
        if opts:
            opts.sort()
            best_d = opts[0][1]
            h[i] += best_d
            h = _proj_to_sum(h, n * 0.5)
            c5_cur = _eval(h)
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
    return best_h, best_c
```
