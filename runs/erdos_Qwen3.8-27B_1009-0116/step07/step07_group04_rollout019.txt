    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.7
        # Update weights
        tau = max(0.01, tau * 0.998)
        w = np.exp((c_new - c_new.max()) / tau)
        w /= w.sum()
    return _repair(best_h, n), best_c5

def _slsqp(h, alpha, maxiter, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max())) - 1.0) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        wt = np.exp(s - s.max())
        wt /= (wt.sum() - 1.0) if wt.sum() > 1 else 1.0
        # Actually, the softplus log-sum-exp gradient
        s_vals = alpha * (c - m)
        s_vals = np.clip(s_vals, -500, 500)
        m2 = s_vals.max()
        exp_s = np.exp(s_vals - m2)
        total = exp_s.sum()
        w_norm = exp_s / total
        g = np.zeros(n)
        for k in range(2*n - 1):
            if w_norm[k] < 1e-10:
                continue
            sh = (n-1) - k
            wk = w_norm[k] * alpha * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _local_search(h, n_tries, rng, t0, budget_s):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    for _ in range(n_tries):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 12)) % n
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    if c5 < best_c5:

    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    
    # Phase 3: SLSQP refinement (multiple alphas)
    for alpha in [3000, 10000, 50000]:
        if time.time() - t0 > budget_s - 150:

        h_c = _slsqp(best_h.copy(), alpha, 800, t0, budget_s)
        c5v = _eval(h_c)

        h, c5 = _coord_descent(best_h.copy(), 10000, rng, t0, budget_s - 80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    
    # Phase 5: High-alpha SLSQP
    if time.time() - t0 < budget_s - 60:
        h_c = _slsqp(best_h.copy(), 200000, 500, t0, budget_s - 40)

    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))

Wait, I have a problem with the SLSQP gradient. Let me fix the softplus formulation.

The objective is: min_h max_k c_k(h)

The log-sum-exp approximation: f(h) ≈ m + (1/α) * log(Σ exp(α(c_k - m)))  where m = max(c_k)

Gradient of log(Σ exp(α(c_k - m))) w.r.t. h:
= Σ_k [α(c_k - m)_k / Σ_j exp(α(c_j - m))] * ∇c_k / ∇h
= α * Σ_k w_k * ∇c_k  where w_k = exp(α(c_k - m)) / Σ exp(α(c_j - m))

But wait, m is also a function of h (it's the max). This makes the gradient more complex.

Actually, a simpler softplus: f(h) = (1/α) * log(Σ_k exp(α * c_k(h)))

∂f/∂h[i] = Σ_k [exp(α c_k) / Σ_j exp(α c_j)] * ∂c_k/∂h[i] / 1  (the 1/α and α cancel)
= Σ_k p_k * ∂c_k/∂h[i]  where p_k = exp(α c_k) / Σ exp(α c_j)

This is cleaner. Let me use this.

And f(h) = c_max + (1/α) * log(Σ exp(α(c_k - c_max)))

∂f/∂h = Σ_k p_k * ∂c_k/∂h (since the max term's gradient is absorbed)
