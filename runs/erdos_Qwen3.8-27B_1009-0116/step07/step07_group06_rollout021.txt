    """Compute subgradient of c5 = max(c) w.r.t. h."""
    active = np.where(c >= c5 - eps)[0]
    n_active = len(active)
    if n_active == 0:
        return np.zeros(n)
    g = np.zeros(n)
    for k in active:

        if s == 0:

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5 < best_c:
                    best_c, best_h = c5, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    n = n0
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)

    # Try structured perturbations for better start
    perturbs = []
    for ps in [0.02, 0.05, 0.10, 0.20]:
        hp = _proj_to_sum(h_init + ps * rng.randn(n), n * 0.5)
        perturbs.append(hp)
    # Block-swap perturbation
    hs = h_init.copy()
    b1, b2 = rng.randint(0, n // 2), rng.randint(n // 3, n - n // 3)
    blen = min(20, n // 10)
    hs[b1:b1 + blen], hs[b2:b2 + blen] = hs[b2:b2 + blen].copy(), hs[b1:b1 + blen].copy()
    perturbs.append(_proj_to_sum(hs, n * 0.5))

    cands = [h_init] + perturbs
    picks = []
    for hc in cands:
        hc_r = _repair(hc, n)
        c5r = _eval(hc_r)
        if c5r < best_c5:
            best_c5, best_h = c5r, hc_r
        picks.append((hc_r, c5r))
    picks.sort(key=lambda x: x[1])
    start_h = picks[0][0].copy()

    remaining = budget_s - (time.time() - t0)
    # Phase 1: Fast subgradient (15% of remaining)
    dl1 = time.time() + remaining * 0.12
    h1, c1 = _proj_subgrad(start_h, n, t0, dl1, n_max=1500)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Pairwise CD with SA (35%)
    dl2 = time.time() + remaining * 0.35
    h2, c2 = _pair_cd(best_h, n, rng, t0, dl2, use_sa=True, sa_t0_temp=0.003)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Subgradient refine (10%)
    dl3 = time.time() + remaining * 0.10
    h3, c3 = _proj_subgrad(best_h, n, t0, dl3, n_max=800)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Pairwise CD greedy (25%)
    dl4 = time.time() + remaining * 0.25
    h4, c4 = _pair_cd(best_h, n, rng, t0, dl4, use_sa=False, sa_t0_temp=0)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Try from second-best start
    if len(picks) > 1 and picks[1][1] < best_c5 + 0.005:
        dl5 = time.time() + remaining * 0.08
        h5, c5_5 = _pair_cd(picks[1][0], n, rng, t0, dl5, use_sa=False, sa_t0_temp=0)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5

    # Phase 6: Final CD + polish (remaining)
    dl6 = t0 + budget_s - 8
    h6, c6 = _pair_cd(best_h, n, rng, t0, dl6, use_sa=False, sa_t0_temp=0)
    if c6 < best_c5:
        best_c5, best_h = c6, h6

    # Fine polish
    dl7 = t0 + budget_s - 2
    h7, c7 = _fine_polish(best_h, rng, t0, dl7)
    if c7 < best_c5:
        best_c5, best_h = c7, h7

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
