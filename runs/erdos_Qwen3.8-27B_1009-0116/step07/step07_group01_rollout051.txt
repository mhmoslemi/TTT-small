def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n

        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                diff -= a

                h[idx] -= a
                diff += a

def _eval(h):

def _carr(h):

def _proj(h, target):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _softobj(c, alpha):

    s -= s.max()
    ln = np.log(np.exp(s).sum())
    return m + ln / alpha, np.exp(s) / np.exp(s).sum()
