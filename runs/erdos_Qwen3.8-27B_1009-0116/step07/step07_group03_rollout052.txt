        mid = (lo + hi) * 0.5

            no_imp += 1
        if no_imp > 60:
            h += 1e-4 * rng.randn(n)
            h = _proj_sum(h, n * 0.5)
            no_imp = 0
        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj_sum(h, n * 0.5)
    return h, best_c5

def _fine_search(h, n_tries, rng, t_deadline):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    scale = 3e-6
    for _ in range(n_tries):
        if time.time() > t_deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < best_c - 1e-19:
                best_c = cn
                best_h = h.copy()
                scale *= 1.01
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_end = t0 + budget_s
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov graduated non-convexity
    for a in [2000, 10000, 60000, 300000, 1200000]:
        tdl = min(time.time() + 18, t0 + budget_s * 0.18)
        if time.time() >= tdl:
            break
        h2, c5v = _nesterov(best_h.copy(), a, tdl, 0.02)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 2: Top-k subgradient (global movement)
    tdl = t0 + budget_s * 0.33
    for m in [5, 10, 20, 5]:
        if time.time() >= tdl:
            break
        h2, c5v = _topk_subgrad(best_h.copy(), m, tdl, rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 3: Coordinate descent (multiple T values)
    tdl = t0 + budget_s * 0.58
    for T in [3, 5, 8, 12, 6]:
        if time.time() >= tdl:
            break
        h2, c5v = _coord_descent(best_h.copy(), n, budget_s * 0.05, T, rng, t_end)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 4: High-alpha Nesterov for fine tuning
    tdl = t0 + budget_s * 0.72
    for a in [3000000, 8000000, 20000000]:
        if time.time() >= tdl:
            break
        h2, c5v = _nesterov(best_h.copy(), a, tdl, 0.008)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 5: Extended coordinate descent
    tdl = t0 + budget_s * 0.86
    for T in [4, 7, 10, 5]:
        if time.time() >= tdl:
            break
        h2, c5v = _coord_descent(best_h.copy(), n, budget_s * 0.035, T, rng, t_end)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 6: Fine local search
    if time.time() < t_end - 12:
        n_tries = min(int((t_end - 16 - time.time()) / 0.003), 60000)
        if n_tries > 200:
            h2, c5v = _fine_search(best_h.copy(), n_tries, rng, t_end - 10)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h2.copy()

    # Phase 7: Final coordinate polish
    if time.time() < t_end - 4:
        rem = t_end - time.time() - 5
        if rem > 2:
            h2, c5v = _coord_descent(best_h.copy(), n, rem, 5, rng, t_end)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h2.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
