        idx = np.argsort(c)[-tk:]
        ct = c[idx]; m = ct.max()
        s = np.clip(al * (ct - m), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        wf = np.zeros(2 * n - 1); wf[idx] = w
        g = _gw(h, wf); g -= g.mean()
        ok = False
        for _ in range(25):
            hn = _proj(h - step * g, n * 0.5)
            cn = _eval(hn)
            if cn < bestc:
                bestc = cn; best = hn.copy(); h = hn
                step = min(step * 1.08, s0 * 10.0)
                ok = True; break
            step *= 0.5
        if not ok:
            step = min(step * 1.2, s0 * 10.0)
    return best, bestc

def _local(h, niter, rng, s0, wide):
    n = len(h)
    c5 = _eval(h); bh = h.copy(); bc = c5
    step = s0; stag = 0
    for _ in range(niter):
        if c5 < bc - 1e-15:
            stag = 0
        else:
            stag += 1
        sp = step * (1.0 + 0.3 * min(stag // 3000, 8))
        r = rng.rand()
        if r < 0.45:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
            d = sp * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5:
                    c5 = c5n
                    if c5 < bc: bc = c5; bh = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        elif r < 0.85:
            if wide:
                idx = rng.choice(n, 3, replace=False)
            else:
                i = rng.randint(n)
                idx = [i, (i + 1 + rng.randint(0, 5)) % n, (i + 2 + rng.randint(0, 8)) % n]
            d1 = sp * rng.randn(); d2 = sp * rng.randn()
            ni = h[idx[0]] + d1; nj = h[idx[1]] + d2; nk = h[idx[2]] - d1 - d2
            if all(0.0 <= v <= 1.0 for v in (ni, nj, nk)):
                h[idx[0]], h[idx[1]], h[idx[2]] = ni, nj, nk
                c5n = _eval(h)
                if c5n < c5:
                    c5 = c5n
                    if c5 < bc: bc = c5; bh = h.copy()
                else:
                    h[idx[0]], h[idx[1]], h[idx[2]] = ni - d1, nj - d2, nk + d1 + d2
        else:
            i = rng.randint(n)
            j = (i + 1) % n
            tmp = h[i]; h[i] = h[j]; h[j] = tmp
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
                if c5 < bc: bc = c5; bh = h.copy()
            else:
                h[i] = tmp; h[j] = h[i] - tmp + tmp
                h[i] = tmp; h[j] = h[i]
                h[j] = tmp
                h[i] = bh[i] if c5n == c5 else h[i]
    return bh, bc

def _ds(h, f):
    n = len(h); m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(axis=1), m)

def _us(h, f):
    return np.repeat(h, f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    tl = lambda: time.time() - t0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP moderate alpha
    for alpha, miter in [(5000, 600), (50000, 800)]:
        if tl() > 70: break
        hn = _slsqp(best_h.copy(), alpha, miter, 1e-13)
        c = _eval(hn)
        if c < best_c5: best_c5, best_h = c, hn

    # Phase 2: Subgradient descent on top lags
    if tl() < budget_s - 400:
        for tk, ns, s0, al in [(4, 3000, 0.005, 500), (8, 2000, 0.003, 5000), (3, 2500, 0.006, 200)]:
            if tl() > budget_s - 180: break
            hn, c = _subgrad(best_h.copy(), tk, ns, s0, al)
            if c < best_c5: best_c5, best_h = c, hn

    # Phase 3: High-alpha SLSQP
    if tl() < budget_s - 250:
        for alpha, miter in [(300000, 800), (2000000, 600)]:
            if tl() > budget_s - 120: break
            hn = _slsqp(best_h.copy(), alpha, miter, 1e-14)
            c = _eval(hn)
            if c < best_c5: best_c5, best_h = c, hn

    # Phase 4: Multi-resolution blend
    if tl() < budget_s - 120:
        for ratio in [2, 3]:
            if tl() > budget_s - 60: break
            nc = n // ratio
            if nc < 60: continue
            hc = _ds(best_h, ratio)
            hc = _slsqp(hc, 100000, 400, 1e-12)
            hu = _repair(_us(hc, ratio), n)
            for mix in [0.3, 0.5, 0.7]:
                hbl = _repair(mix * hu + (1.0 - mix) * best_h, n)
                c = _eval(hbl)
                if c < best_c5: best_c5, best_h = c, hbl

    # Phase 5: Perturbation restarts
    if tl() < budget_s - 200:
        for _ in range(4):
            if tl() > budget_s - 100: break
            pert = best_h + 0.02 * rng.randn(n)
            pert = _repair(pert, n)
            hn = _slsqp(pert, 200000, 400, 1e-12)
            c = _eval(hn)
            if c < best_c5: best_c5, best_h = c, hn

    # Phase 6: Broad 3-point local search
    if tl() < budget_s - 40:
        nt = int((budget_s - 40 - tl()) * 500)
        nt = max(nt, 20000)
        best_h, best_c5 = _local(best_h.copy(), nt, rng, 2e-5, True)

    # Phase 7: Focused 3-point local search (narrow indices)
    if tl() < budget_s - 20:
        nt = int((budget_s - 20 - tl()) * 550)
        nt = max(nt, 10000)
        best_h, best_c5 = _local(best_h.copy(), nt, rng, 8e-6, False)

    # Phase 8: Ultra-fine 2-point polish
    if tl() < budget_s - 1.5:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = int((budget_s - 1.5 - tl()) * 1200)
        for _ in range(npol):
            if tl() > budget_s - 1.5: break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 4)) % nc2
            d = 4e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c5: best_c5, best_h = c5c, h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
