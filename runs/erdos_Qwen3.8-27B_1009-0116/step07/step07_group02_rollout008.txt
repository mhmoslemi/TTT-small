import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-10:
        return h
    # Bisection on threshold
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _smax_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-13:
            continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _proj(h, n):
    return _repair(h, n)

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:

                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12,
                                'factr': 1e8, 'pgtol': 1e-12})
        return _repair(res.x, n)
    except:

def _subgrad(h, steps, step0, tol, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):

        m = c.max()

            else:

        g *= dx / len(active)

    dx = 2.0 / n
    c = _cvals(h)  # shape (2n-1,)
    m = len(c)

    gs = np.zeros((m, n))
    for k in range(m):
        sh = (n-1) - k
        g = np.zeros(n)
        if sh >= 0:
            np_ = n - sh
            g[:np_] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:np_]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:-sp] -= h[sp:]
        gs[k] = g * dx
    
    # LP: min t
    # s.t. c_k + gs[k]·δ ≤ t  for all k
    #      ∑δ_i = 0
    #      -h_i ≤ δ_i ≤ 1-h_i
    #
    # Variables: [t, δ_0, ..., δ_{n-1}]  (n+1 variables)
    #
    # c_k + gs[k]·δ ≤ t  →  gs[k]·δ - t ≤ -c_k
    # In standard form for linprog: A_ub @ x ≤ b_ub
    #
    # x = [t, δ_0, ..., δ_{n-1}]
    
    # Objective: minimize t → c = [1, 0, 0, ..., 0]
    c_obj = np.zeros(n+1)
    c_obj[0] = 1.0
    
    # Constraints: gs[k]·δ - t ≤ -c_k
    # [-1, gs[k,0], gs[k,1], ..., gs[k,n-1]] · [t, δ] ≤ -c_k
    A_ub = np.zeros((m, n+1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = gs
    b_ub = -c
    
    # Equality: ∑δ_i = 0
    # A_eq @ x = b_eq
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: 0 ≤ t (let's say t is unbounded below), -h_i ≤ δ_i ≤ 1-h_i
    bounds = [(None, None)]  # t is free
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                c5n = _eval(h_new)
                if c5n < _eval(h):

    except:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _smax_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-13,
                                'factr': 1e8, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad(h, steps, step0, tol, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):

        t = step0 / (1.0 + 0.01 * i)
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]

            else:

                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / len(active)

    return h

    if time.time() < deadline - 180:
        for nl in [1, 2, 3, 5]:
            if time.time() > deadline - 180:
                break
            h_ml, c5_ml = _topdesc(best_h.copy(), nl, 800, deadline - 170)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()

    # Phase 5: Second LP pass (after other optimizations)
    if time.time() < deadline - 120:
        h_lp, c5_lp = _lp_descent(best_h.copy(), 300, deadline - 100)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 6: Local search (multi-scale)
    if time.time() < deadline - 60:
        h_ls, c5_ls = _local_search(best_h.copy(), 50000, rng, deadline - 50)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 7: Final L-BFGS polish
    if time.time() < deadline - 40:
        h = _lbfgs(best_h, 5000000, 3000, deadline - 30)
        cv = _eval(h)
        if cv < best_c5:
            best_c5 = cv
            best_h = h.copy()

    # Phase 8: Final LP polish
    if time.time() < deadline - 15:
        h_lp, c5_lp = _lp_descent(best_h.copy(), 100, deadline - 5)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 9: Ultra-fine local search
    if time.time() < deadline - 2:
        h_ls, c5_ls = _local_search(best_h.copy(), 30000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
