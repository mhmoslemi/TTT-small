def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

def _grad_k(h, k):
    n = len(h)

    g = np.zeros(n)

        nn = n - sh
        idx = np.arange(nn)
