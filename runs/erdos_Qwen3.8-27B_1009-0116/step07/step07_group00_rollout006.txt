        if abs(diff) > 1e-12:

                    diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smap_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _fista(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    t = 1.0
    f_y = _smap_obj(y, alpha, topk)
    best = _eval(x)
    bx = x.copy()
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(y, alpha, topk)
        step = step0 * (1.0 + 0.3 / np.sqrt(it + 1))
        for _ in range(10):
            yn = _proj(y - step * g, target)
            fn = _smap_obj(yn, alpha, topk)
            if fn <= f_y + 1e-12:
                break
            step *= 0.5
        else:
            t = 1.0
            y = x.copy()
            f_y = _smap_obj(y, alpha, topk)

        f_y = _smap_obj(y, alpha, topk)
        c5x = _eval(x)
        if c5x < best:
            best = c5x
            bx = x.copy()
    return bx, best

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smap_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_topk(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _directed_anneal(h, rng, deadline, niter):
    """Perturb indices that appear in the worst lags, biased to reduce them."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    topk = 8
    for it in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        idx_top = np.argpartition(c, -topk)[-topk:]
        # Find indices that appear in top lag computations
        hot = np.zeros(n, dtype=bool)
        for k in idx_top:
            sh = (n - 1) - k
            if sh >= 0:
                hot[:n-sh] = True
                if sh > 0:
                    hot[sh:] = True
            else:
                sp = -sh
                hot[sp:] = True
                hot[:n-sp] = True
        hot_idx = np.where(hot)[0]
        if len(hot_idx) < 2:
            continue
        # Pick 2-4 indices from hot set
        npick = rng.randint(2, 5)
        picks = rng.choice(hot_idx, size=min(npick, len(hot_idx)), replace=False)
        # Direction: negative gradient at these indices
        grad = _grad_topk(h, 50000, topk)
        d = 3e-5 * rng.randn(len(picks))
        # Bias: move against gradient
        d -= 0.5 * grad[picks]
        d -= d.mean()
        h_new = h.copy()
        h_new[picks] += d
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            h = h_new
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc
