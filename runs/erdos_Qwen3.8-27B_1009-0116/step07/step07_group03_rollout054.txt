import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lsmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(sh - sh.max()))) / alpha

def _lsmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-9)[0]

        wk = w[k] * dx

        else:
            sp = -shft
            cnt = n - sp
            g[sp:sp+cnt] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp+cnt]

            j = (i + 1 + rng.randint(0, n // 6)) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i] + 1e-12, h[j] - 1.0 + 1e-12)
            d_hi = min(1.0 - h[i] - 1e-12, h[j] - 1e-12)
            if d_lo >= d_hi - 1e-11:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(top_vals + b * x1)
            f2 = np.max(top_vals + b * x2)
            for _ in range(18):
                if f1 < f2:
                    d_hi = x1; x1 = d_hi - gr * (d_hi - d_lo); f1 = np.max(top_vals + b * x1)
                    x2 = x1 if f1 < f2 else x2
                    x2 = d_lo + gr * (d_hi - d_lo); f2 = np.max(top_vals + b * x2)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo); f2 = np.max(top_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-11:
                h[i] += d_opt; h[j] -= d_opt
                improved = True
        if improved:
            nc = _eval(h)
            if nc < best_c5 - 1e-13:
                best_c5 = nc; no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 35:
            h += 0.008 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        h = _repair(h, n)
    return h, best_c5

def _block_cd(h, n, bsz, duration, rng, t_end):
    dx = 2.0 / n
    end_t = min(time.time() + duration, t_end)
    best_c5 = _eval(h)
    c5 = best_c5
    no_imp = 0
    while time.time() < end_t:
        start = rng.randint(max(1, n - bsz))
        blk = h[start:start+bsz].copy()
        s_blk = blk.sum()
        delta = rng.randn(bsz) * 0.015
        delta -= delta.mean()
        nb = np.clip(blk + delta, 0, 1)
        diff = s_blk - nb.sum()
        for _ in range(30):
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                idx = np.argmax(1.0 - nb); a = min(diff, 1.0 - nb[idx]); nb[idx] += a; diff -= a
            else:
                idx = np.argmin(nb); a = min(-diff, nb[idx]); nb[idx] -= a; diff += a
        nb = np.clip(nb, 0, 1)
        h[start:start+bsz] = nb
        c5n = _eval(h)
        if c5n < c5 - 1e-13:
            c5 = c5n
            if c5n < best_c5:
                best_c5 = c5n
            no_imp = 0
        else:
            h[start:start+bsz] = blk
            no_imp += 1
        if no_imp > 150:
            h += 0.004 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_imp = 0
    return h, best_c5

def _binary_perturb(h, n, rng, scale=0.08):
    hp = h.copy()
    for i in range(n):
        if hp[i] > 0.5:
            hp[i] = min(1.0, hp[i] + scale * rng.rand())
        else:
            hp[i] = max(0.0, hp[i] - scale * rng.rand())
    return _proj_to_sum(hp, n * 0.5)

def _freq_perturb(h, n, rng, amp=0.025):
    hp = h.copy()
    freq = rng.randint(1, max(2, n // 5))
    phase = rng.rand() * 2.0 * np.pi
    hp += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return _proj_to_sum(hp, n * 0.5)

def _fine_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    scales = [2e-5, 8e-6, 3e-6, 1e-6]
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n // 15)) % n
        sc = scales[rng.randint(len(scales))]
        d = sc * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov with geometric alpha schedule
    alphas1 = [2000, 8000, 30000, 100000, 300000]
    t1_end = t0 + budget_s * 0.10
    h = _nesterov(h, alphas1, t1_end)
    c5 = _eval(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Active-set pair CD
    for top_T in [4, 6, 3, 5]:
        if time.time() > t0 + budget_s * 0.38:
            break
        h = best_h.copy()
        h, c5 = _pair_cd(h, n, top_T, budget_s * 0.07, rng, t0 + budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: Nesterov refinement (higher alpha)
    if time.time() < t0 + budget_s * 0.50:
        t3_end = t0 + budget_s * 0.49
        h = best_h.copy()
        h = _nesterov(h, [200000, 600000, 1500000], t3_end)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Block coordinate descent (new: multi-element updates)
    for bsz in [5, 8, 4]:
        if time.time() > t0 + budget_s * 0.62:
            break
        h = best_h.copy()
        h, c5 = _block_cd(h, n, bsz, budget_s * 0.05, rng, t0 + budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 5: Structured perturbations + re-optimization
    perturbations = []
    perturbations.append(_binary_perturb(best_h, n, np.random.RandomState(seed + 7)))
    perturbations.append(_freq_perturb(best_h, n, np.random.RandomState(seed + 13)))
    perturbations.append(_binary_perturb(best_h, n, np.random.RandomState(seed + 21), 0.04))
    perturbations.append(best_h + 0.015 * np.random.RandomState(seed + 31).randn(n))
    for idx, hp in enumerate(perturbations):
        if time.time() > t0 + budget_s * 0.78:
            break
        hp = _proj_to_sum(hp, n * 0.5)
        hp = _nesterov(hp, [100000, 400000], t0 + budget_s * 0.75)
        hp, c5 = _pair_cd(hp, n, 4, budget_s * 0.025, rng, t0 + budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Phase 6: Fine local search
    if time.time() < t0 + budget_s - 15:
        remaining = t0 + budget_s - 15 - time.time()
        n_iter = min(int(remaining * 150), 60000)
        if n_iter > 500:
            h = best_h.copy()
            best_h, best_c5 = _fine_search(h, n_iter, rng, t0 + budget_s - 10)

    # Phase 7: Final pair CD polish
    if time.time() < t0 + budget_s - 5:
        remaining = t0 + budget_s - 5 - time.time()
        h = best_h.copy()
        h, c5 = _pair_cd(h, n, 5, max(remaining, 1.0), rng, t0 + budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Final: extra fine search on best
    if time.time() < t0 + budget_s - 3:
        remaining = t0 + budget_s - 3 - time.time()
        n_iter = min(int(remaining * 200), 30000)
        if n_iter > 200:
            h = best_h.copy()
            best_h, best_c5 = _fine_search(h, n_iter, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
