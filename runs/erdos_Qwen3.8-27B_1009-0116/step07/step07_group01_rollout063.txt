```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    m = r.max()
    e = np.exp(np.clip(alpha * (r - m), -400, 0))
    es = e.sum()
    fval = m + np.log(es) / alpha
    w = e / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return fval, g

def _proj_grad(h0, alpha, iters, step0, beta, tl):
    h = h0.copy()
    n = len(h)
    v = np.zeros(n)
    cur = _eval(h)
    step = step0
    for it in range(iters):
        if time.time() > tl:
            break
        f, g = _smgrad(h, alpha)
        v = beta * v + (1.0 - beta) * g
        taken = False
        for _ in range(25):
            h_new = _proj(h - step * v, n * 0.5)
            c = _eval(h_new)
            if c < cur - 1e-15:
                h = h_new
                cur = c
                step = min(step * 1.08, step0 * 15.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 15.0)
    return h, cur

def _lbfgs_obj(x, alpha, lam):
    n = len(x)
    h = np.clip(x, 0, 1)
    f, _ = _smgrad(h, alpha)
    penalty = lam * (h.sum() - n * 0.5) ** 2
    return f + penalty

def _lbfgs_round(x, alpha, lam, n):
    f, g = _smgrad(np.clip(x, 0, 1), alpha)
    pen_g = 2 * lam * (np.clip(x, 0, 1).sum() - n * 0.5) * np.ones(n)
    return f + lam * (np.clip(x, 0, 1).sum() - n * 0.5) ** 2, f + pen_g
```

def _lbfgs_pen(h0, alpha, lam, maxiter, tl):
    n = len(h0)
    def obj(x):
        f, _ = _smgrad(x, alpha)
        return f + lam * (x.sum() - n * 0.5) ** 2
    def grad(x):
        f, g = _smgrad(x, alpha)
        g = g + 2 * lam * (x.sum() - n * 0.5)
        return g
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return _repair(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    m = r.max()
    e = np.exp(np.clip(alpha * (r - m), -400, 0))
    es = e.sum()
    fval = m + np.log(es) / alpha
    w = e / es
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return fval, g

        f, g = _smgrad(h, alpha)

                h = h_new
                cur = c
                step = min(step * 1.08, step0 * 15.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 15.0)
    return h, cur

def _lbfgs_pen(h0, alpha, lam, maxiter, t_end):
    n = len(h0)
    def _obj(x):
        f, _ = _smgrad(x, alpha)
        return f + lam * (x.sum() - n * 0.5) ** 2

        return g + 2.0 * lam * (x.sum() - n * 0.5)
