def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(-h if diff > 0 else h)
        for i in idx:
            if diff > 0:
                amt = min(diff, 1.0 - h[i]); h[i] += amt; diff -= amt
            else:
                amt = min(-diff, h[i]); h[i] -= amt; diff += amt
            if abs(diff) < 1e-15: break

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(-h if diff > 0 else h)
        for i in idx:
            if diff > 0:
                amt = min(diff, 1.0 - h[i]); h[i] += amt; diff -= amt
            else:
                amt = min(-diff, h[i]); h[i] -= amt; diff += amt
            if abs(diff) < 1e-15: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_k(h, k, n, dx):
    sh = k - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    v1 = (j + sh >= 0) & (j + sh < n)
    v2 = (j - sh >= 0) & (j - sh < n)
    g[v1] = dx * (1.0 - h[j[v1] + sh])
    g[v2] -= dx * h[j[v2] - sh]
    return g

def _subgrad_opt(h0, n, t_end, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    dx = 2.0 / n
    v = np.zeros(n)
    eta = 0.03
    mu = 0.85
    avg_buf = []
    for t in range(1, t_end + 1):
        if time.time() > t_end + 500: break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cm = c.max()
        thresh = cm - max(0.005, 0.01 * cm)
        active = np.where(c >= thresh)[0]
        if len(active) == 0: active = np.array([np.argmax(c)])
        if len(active) > 15: active = np.argsort(c)[-15:]
        g = np.zeros(n)
        for kk in active:
            g += _grad_k(h, kk, n, dx)
        g /= len(active)
        v = mu * v + (1.0 - mu) * g
        gv = np.sqrt(np.dot(v, v)) + 1e-30
        step = eta * (1.0 + 3.0 / (t + 10)) / gv
        accepted = False
        for _bt in range(8):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cm:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                step = min(step * 1.1, 5.0 / gv)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 1.3
        if t % 100 == 0:
            avg_buf.append(h.copy())
            if len(avg_buf) > 15: avg_buf.pop(0)

def _slsqp(h0, alpha, maxiter, n):
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(2 * n - 1):
            if w[kk] < 1e-14: continue
            g += w[kk] * _grad_k(x, kk, n, dx)
        return g
    def _cfun(x): return x.sum() - n * 0.5
    def _cjac(x): return np.ones(n)
    bnds = [(0.0, 1.0)] * n
