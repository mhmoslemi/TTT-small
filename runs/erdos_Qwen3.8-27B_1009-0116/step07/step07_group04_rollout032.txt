import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lsm_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-14:
            continue
        shift = (n-1) - k
        wk = w[k] * dx
        if shift >= 0:
            i1 = np.arange(n - shift)
            g[i1] += wk * (1.0 - h[i1 + shift])
            if shift > 0:
                i2 = np.arange(shift, n)
                g[i2] -= wk * h[i2 - shift]
        else:
            sp = -shift
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g, c, w

def _bb_gradient(h, alpha, n_iter, t0, budget):
    n = len(h)
    target = n * 0.5
    h = _proj(h.copy(), target)
    f_prev = _eval(h)
    x_prev = h.copy()
    g_prev = _lsm_grad(h, alpha)[0]
    
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        g = _lsm_grad(h, alpha)[0]
        
        # Barzilai-Borwein step
        s = h - x_prev
        y = g - g_prev
        sy = s @ y
        ss = s @ s
        yy = y @ y
        if yy < 1e-20:
            step = 0.01
        elif it % 2 == 0:
            step = ss / sy if abs(sy) > 1e-20 else 0.01
        else:
            step = sy / yy if abs(yy) > 1e-20 else 0.01
        step = np.clip(step, 1e-6, 1.0)
        
        h_new = _proj(h - step * g, target)
        f_new = _eval(h_new)
        
        if f_new < f_prev + 1e-15:
            x_prev, h = h.copy(), h_new
            g_prev, g = g.copy(), _lsm_grad(h, alpha)[0]
            f_prev = f_new

            f_new = _eval(h_new)
            x_prev, h = h.copy(), h_new
            g_prev = _lsm_grad(h, alpha)[0]
            f_prev = f_new
        
        x_prev, g_prev = h.copy(), g.copy()
    
    return _proj(h, target), f_prev

def _anneal_local_search(h, n_tries, rng, t0, budget, temp0=0.001, decay=0.999):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    temp = temp0
    for _ in range(n_tries):
        if time.time() - t0 > budget:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

            c_new = _eval(h)
            if c_new < cur_c or rng.rand() < np.exp((cur_c - c_new) / temp):

                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:

        temp *= decay

import numpy as np
from scipy.optimize import minimize
import time
import functools

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    target = n / 2.0
    h = _proj(h, target)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
            diff -= np.sign(diff) * a
            if abs(diff) < 1e-12:
                break
    return h

def _lsm_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _bb_optimize(h, alpha, n_iter, t0, deadline):
    n = len(h)
    target = n * 0.5
    h = _proj(h.copy(), target)
    f = _eval(h)
    x_old = h.copy()
    g_old = _lsm_grad(h, alpha)
    for it in range(n_iter):
        if time.time() - t0 > deadline:
            break
        g = _lsm_grad(h, alpha)
        s = h - x_old
        y = g - g_old
        sy = s.dot(y)
        ss = s.dot(s)
        yy = y.dot(y)
        if yy < 1e-20 or abs(sy) < 1e-20:
            step = 0.01
        elif it % 2 == 0:
            step = max(ss / sy, 1e-8)
        else:
            step = max(sy / yy, 1e-8)
        step = min(step, 1.0)
        for _bt in range(12):
            h_try = _proj(h - step * g, target)
            f_try = _eval(h_try)
            if f_try < f + 1e-14:
                break
            step *= 0.5
        else:
            break
        x_old = h.copy()
        g_old = g.copy()
        h = h_try
        f = f_try
    return _repair(h, n), f

def _nesterov_topk(h, k_top, n_steps, step0, t0, deadline):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(n_steps):
        if time.time() - t0 > deadline:
            break
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = dx / k_top
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, target)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 6.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj(h_prev - step * v, target)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _cons_eq(x, target):
    return x.sum() - target

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    target = n * 0.5
    sf = functools.partial(_slsqp_obj, alpha=alpha, n=n)
    sg = functools.partial(_slsqp_grad, alpha=alpha, n=n)
    cf = functools.partial(_cons_eq, target=target)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': _cons_jac}]
    try:
        res = minimize(sf, h, jac=sg, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _block_local_search(h, t0, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    step = 2e-5
    for _ in range(200000):
        if time.time() - t0 > deadline:
            break
        b = rng.randint(n - 4)
        idx = np.arange(b, b + 4)
        d = step * rng.randn(4)
        d -= d.mean()
        old = h[idx].copy()
        new = old + d
        if np.any(new < 0) or np.any(new > 1):
            continue
        h[idx] = new
        c_new = _eval(h)
        if c_new < cur_c - 1e-18:
            cur_c = c_new
            step = min(step * 1.05, 5e-4)
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            h[idx] = old
            if c_new > cur_c + 5e-6:
                step *= 0.8
    return best_h, best_c

def _fine_2pt(h, t0, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c

                    best_c, best_h = c_new, h.copy()
            else:

                step *= 0.95
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()
