                    best_h = h.copy()
                step = min(step * 1.15, smax)
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                step = max(step * 0.88, smin)
    return best_h, best_c

def _fine_local(h, ntry, rng, c5c, deadline):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    step = 2e-5
    for _ in range(ntry):
        if time.time() > deadline:
            break
        d = step * rng.randn()
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _triplet_local(h, ntry, rng, c5c, deadline):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    step = 3e-4
    for _ in range(ntry):
        if time.time() > deadline:
            break
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        k = (i + 9 + rng.randint(0, 8)) % nc
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                step = min(step * 1.1, 2e-3)
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
                step = max(step * 0.9, 1e-5)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Gradient descent with varying top-k
    for kt in [5, 3, 7, 4, 3, 6, 3, 5]:
        if time.time() > dl - 250:
            break
        h2, c52 = _mlgd(best_h.copy(), 500, kt,
                        [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4],
                        dl - 250)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Perturbation escapes
    for _p in range(4):
        if time.time() > dl - 220:
            break
        ns = 0.004 + 0.002 * rng.rand()
        hp = best_h.copy() + ns * rng.randn(n0)
        hp = _repair(hp, n0)
        h2, c52 = _mlgd(hp, 300, 4,
                        [0.04, 0.02, 0.01, 0.005, 0.002, 0.001],
                        dl - 220)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: SLSQP with multiple smoothing levels
    for a_val in [8000.0, 25000.0, 80000.0]:
        if time.time() > dl - 180:
            break
        bnds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n0,)}]
        try:
            res = minimize(_lsx_obj, best_h.copy(),
                           args=(n0, a_val), jac=_lsx_grad,
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-17})
            hr = _repair(res.x, n0)
            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr
        except Exception:
            pass

    # Phase 4: Fine gradient polish
    for kt in [3, 4, 3, 5, 3]:
        if time.time() > dl - 80:
            break
        h2, c52 = _mlgd(best_h.copy(), 300, kt,
                        [0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4],
                        dl - 80)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Adaptive 2-point local search (main improvement)
    if time.time() < dl - 35:
        ntry = int((dl - 35 - time.time()) / 0.003)
        best_h, best_c5 = _adaptive_local(best_h, ntry, rng, best_c5,
                                          dl - 35, 1e-3, 5e-6, 5e-3)

    # Phase 6: Triplet local search
    if time.time() < dl - 15:
        ntry = int((dl - 15 - time.time()) / 0.004)
        best_h, best_c5 = _triplet_local(best_h, ntry, rng, best_c5, dl - 15)

    # Phase 7: Ultra-fine 2-point polish
    if time.time() < dl - 2:
        ntry = int((dl - 2 - time.time()) / 0.002)
        best_h, best_c5 = _fine_local(best_h, ntry, rng, best_c5, dl - 2)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
