from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, k):
    """Compute gradient of sum of top-k correlations."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    topk_idx = np.argsort(c)[-k:]
    g = np.zeros(n)
    for ki in topk_idx:
        sh = (n - 1) - ki
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / k

def _pnorm_obj(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.sum(c**p)**(1.0/p) + penalty * (h.sum() - n*0.5)**2

def _pnorm_grad(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cp = np.maximum(c, 1e-30)**p

    factor = s**(1.0/p - 1.0)
    g = np.zeros(n)

def _lbfgs(h0, p, penalty, maxiter, t_limit):
    t0 = time.time()
    n = len(h0)
    bnds = [(0.0, 1.0)]*n
    def obj(x):
        if time.time() - t0 > t_limit:
            raise TimeoutError()
        return _pnorm_obj(x, p, penalty)
    def grad(x):
        if time.time() - t0 > t_limit:
            raise TimeoutError()
        return _pnorm_grad(x, p, penalty)
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)
