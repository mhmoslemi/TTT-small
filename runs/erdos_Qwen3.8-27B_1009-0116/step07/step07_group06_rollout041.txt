def _active_grad(h, n, top_m=5):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find top M lags
    idx = np.argsort(c)[-top_m:]
    # Compute gradient for each active lag
    g = np.zeros(n)
    for k in idx:
        s = k - (n - 1)  # shift
        for i in range(n):
            # h[i] as first factor
            if 0 <= i + s < n:
                g[i] += (1.0 - h[i + s])
            # h[i] as second factor (h[i] = h[j+s] where j = i-s)
            if 0 <= i - s < n:
                g[i] -= h[i - s]
    g *= dx / top_m
    return g, c.max(), c

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection"""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    """Compute subgradient from top m active lags"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        # For each i, compute contribution
        # h[i] as first factor: (1-h[i+s]) if 0<=i+s<n
        # h[i] as second factor: -h[i-s] if 0<=i-s<n
        for i in range(n):
            if 0 <= i + s < n:
                g[i] += (1.0 - h[i + s])
            if 0 <= i - s < n:
                g[i] -= h[i - s]
    g *= dx / m
    return g, c.max(), c

def _grad_descent(h, rng, t0, deadline, m=10):
    """Active-set subgradient descent with backtracking"""
    n = len(h)
    h = h.copy()
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    gamma = 0.01  # initial step
    for it in range(20000):
        if time.time() > deadline:
            break
        g, c5_cur, c = _active_grad(h, n, m)
        # Project gradient to null space of 1-vector
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        # Backtracking line search
        accepted = False
        step = gamma
        for _bt in range(10):
            h_new = _proj(h - step * g / gn, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                gamma = min(0.05, gamma * 1.1)
                accepted = True
                if c5_cur < best_c:
                    best_c = c5_cur
                    best_h = h.copy()
                break
            step *= 0.5
        if not accepted:
            gamma *= 0.7
            if gamma < 1e-6:
                break
    return best_h, best_c
