    if time.time() > deadline:
        return h
    ob = _Obj(alpha, topk)
    co = _Cons(n)
    try:
        r = minimize(ob.f, h, jac=ob.g, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints={'type': 'eq', 'fun': co.f, 'jac': co.g},
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgsb(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    ob = _Obj(alpha, topk)
    try:
        r = minimize(ob.f, h, jac=ob.g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-15, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.015, step0 * 6.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.015 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _coord_search(h, niter, rng, deadline, scale):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _restart_optimize(h0, alpha, topk, maxit, deadline):
    n = len(h0)
    ob = _Obj(alpha, topk)
    co = _Cons(n)
    try:
        r = minimize(ob.f, h0, jac=ob.g, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints={'type': 'eq', 'fun': co.f, 'jac': co.g},
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP + L-BFGS-B refinement
    dl1 = min(deadline, t0 + 160)
    for alpha, tk, mi in [(3000, 50, 250), (15000, 25, 250), (60000, 12, 300)]:
        if time.time() > dl1:
            break
        h = _slsqp(best_h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        h2 = _lbfgsb(best_h, alpha, tk, mi, dl1)
        c5v2 = _eval(h2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = h2.copy()

    # Phase 2: Nesterov with sweeping alpha
    dl2 = min(deadline, t0 + 450)
    for alpha, tk, st, ns in [(8000, 35, 0.006, 2500), (40000, 20, 0.003, 2500),
                               (150000, 10, 0.001, 2500), (800000, 6, 0.0005, 2000)]:
        if time.time() > dl2:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Block simulated annealing
    dl3 = min(deadline, t0 + 680)
    blocks = [1, 2, 3, 4, 5, 7, 11]
    h_a, c5_a = _block_anneal(best_h.copy(), 60000, rng, dl3, 0.006, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 4: Aggressive coordinate search (larger steps)
    dl4 = min(deadline, t0 + 870)
    remaining = max(0, dl4 - time.time())
    n_co = min(int(remaining / 0.004), 80000)
    best_h, best_c5 = _coord_search(best_h, n_co, rng, dl4, 0.008)

    # Phase 5: Restart from perturbed copies
    dl5 = min(deadline, t0 + budget_s - 30)
    for attempt in range(3):
        if time.time() > dl5:
            break
        pert = best_h + 0.02 * rng.randn(n0)
        pert = _repair(np.clip(pert, 0, 1), n0)
        h_r = _restart_optimize(pert, 50000, 12, 200, dl5)
        h_r = _nesterov(h_r, 200000, 8, 500, 0.001, dl5)
        c5r = _eval(h_r)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_r.copy()

    # Phase 6: Final fine polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.003), 60000)
    best_h, best_c5 = _coord_search(best_h, n_pol, rng, deadline, 0.002)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
