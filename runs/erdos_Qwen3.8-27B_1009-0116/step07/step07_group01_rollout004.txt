def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

        i = rng.randint(0, n - blen)
        j = rng.randint(0, n - blen)
        if abs(i - j) < blen:
            continue
        block1 = h[i:i+blen].copy()
        block2 = h[j:j+blen].copy()
        h[i:i+blen] = block2
        h[j:j+blen] = block1
        cn = _eval(h)
        if cn < cur_c:
            cur_c = cn
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            h[i:i+blen] = block1
            h[j:j+blen] = block2
    return best_h, best_c

def _thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        c = i
        p = 0
        while c:
            p ^= c & 1
            c >>= 1
        t[i] = p
    return (t >= 0.5).astype(float)

def _balanced_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.permutation(n)
    h[idx[:half]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _time_left():
        return budget_s - 3 - (time.time() - t0)

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)

    candidates = []
    candidates.append(_to_binary(h_init, n))
    candidates.append(_to_binary(_thue_morse(n), n))
    for s2 in range(3):
        candidates.append(_to_binary(_balanced_binary(n, np.random.RandomState(seed + 100 + s2)), n))

    # Phase 1: Quick SLSQP on initial (30s)
    end1 = t0 + 30
    for alpha in [30000, 200000]:
        if time.time() > end1:
            break
        for c_init in candidate[:1] if time.time() > end1 - 15 else candidates[:2]:
            pass
    h_s = _slsqp(h_init.copy(), 50000, 400, 1e-14)
    c_s = _eval(h_s)
    if c_s < best_c5:
        best_c5 = c_s
        best_h = h_s.copy()
        candidates.insert(0, _to_binary(h_s, n))

    # Phase 2: Swap search on each candidate
    n_cand = min(4, len(candidates))
    per_cand = max(40, int((budget_s - 120) / n_cand))
    for ci in range(n_cand):
        if _time_left() < per_cand * (n_cand - ci) - 60:
            break
        ht = candidates[ci].copy()
        end_phase = min(t0 + per_cand, t0 + budget_s - 80)
        ht, ct = _swap_search(ht, end_phase, rng, temp0=0.005, cool_period=4000, cool_rate=0.96)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()
        ht2, ct2 = _block_search(ht, min(time.time() + 10, t0 + budget_s - 70), rng)
        if ct2 < best_c5:
            best_c5 = ct2
            best_h = ht2.copy()

    # Phase 3: Multi-resolution refinement
    if _time_left() > 150:
        for ratio in [4, 7]:
            if _time_left() < 70:
                break
            nc = n // ratio
            hc = best_h[:nc * ratio].reshape(nc, ratio).mean(axis=1)
            hc = _repair(hc, nc)
            hc = _slsqp(hc, 100000, 500, 1e-14)
            h_up = np.repeat(hc, ratio)[:n]
            h_up = _repair(h_up, n)
            h_bin = _to_binary(h_up, n)
            h_bin, c_bin = _swap_search(h_bin, min(t0 + 60, t0 + _time_left() - 50), rng, 0.004)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
            h_bin2, c_bin2 = _block_search(h_bin, min(time.time() + 10, t0 + _time_left() - 40), rng)
            if c_bin2 < best_c5:
                best_c5 = c_bin2
                best_h = h_bin2.copy()

    # Phase 4: Intensive swap on best
    if _time_left() > 80:
        h_best_bin = _to_binary(best_h, n)
        h_best_bin, c_best = _swap_search(h_best_bin, t0 + budget_s - 50, rng, 0.003, 5000, 0.98)
        if c_best < best_c5:
            best_c5 = c_best
            best_h = h_best_bin.copy()

    # Phase 5: Continuous polish
    if _time_left() > 30:
        h_cont = best_h + 0.02 * rng.randn(n)
        h_cont = _repair(h_cont, n)
        h_cont = _slsqp(h_cont, 1000000, 400, 1e-16)
        cn = _eval(h_cont)
        if cn < best_c5:
            best_c5 = cn
            best_h = h_cont.copy()

    # Phase 6: Final swap
    if _time_left() > 15:
        h_fb = _to_binary(best_h, n)
        h_fb, cf = _swap_search(h_fb, t0 + budget_s - 1, rng, 0.001, 2000, 0.99)
        if cf < best_c5:
            best_c5 = cf
            best_h = h_fb.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
