    g = np.zeros(n)
    if s >= 0:
        # first index: i in [0, n-1-s], partner i+s
        g[:n-s] += (1 - h[s:])
        # second index: i+s in [s, n-1], partner i
        g[s:] -= h[:n-s]

        g[sp:] += (1 - h[:n-sp])
        # second index: i-sp in [0, n-1-sp], partner i
        g[:n-sp] -= h[sp:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

        gbuf[n-1-j : 2*n-1-j] -= omh
        gbuf[j : j+n] += hrev
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        d_mid = 0.5 * (d_lo + d_hi)
        d_opt = d_mid
        for _ in range(20):
            d_mid = 0.5 * (d_lo + d_hi)
            vbuf[:] = c
            vbuf += d_mid * gbuf
            vbuf[k1] += d_mid * d_mid
            vbuf[k2] += d_mid * d_mid
            kmax = np.argmax(vbuf)
            sg = gbuf[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-13:
                d_hi = d_mid
            elif sg < -1e-13:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            continue
        h[i] += d_opt; h[j] -= d_opt
        omh[i] -= d_opt; omh[j] += d_opt
        hrev[n-1-i] += d_opt; hrev[n-1-j] -= d_opt
        c += d_opt * gbuf
        c[k1] += d_opt * d_opt
        c[k2] += d_opt * d_opt
        c5_new = c.max()
        delta = c5_new - c5_cur
        if delta < -1e-14 or (delta > 0 and delta < T * abs(rng.randn())):
            c5_cur = c5_new
            if c5_cur < best_c:
                best_c = c5_cur; best_h = h.copy()
        else:
            h[i] -= d_opt; h[j] += d_opt
            omh[i] += d_opt; omh[j] -= d_opt
            hrev[n-1-i] -= d_opt; hrev[n-1-j] += d_opt
            c -= d_opt * gbuf
            c[k1] -= d_opt * d_opt
            c[k2] -= d_opt * d_opt
    return best_h, best_c

def _fine_polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(300000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _structured(n, rng, kind):
    x = np.linspace(0, 2, n, endpoint=False)
    if kind == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x + rng.rand())
    elif kind == 1:
        h = 0.5 + 0.25 * np.sin(4 * np.pi * x) + 0.1 * np.sin(8 * np.pi * x)
    elif kind == 2:
        h = 0.5 + 0.2 * np.cos(2*np.pi*x) + 0.15*np.sin(3*np.pi*x + 1)
    else:
        h = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
    return _proj_to_sum(h, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    # Multi-start
    for p in range(6):
        ht = _structured(n0, rng, p)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()
    for ps in [0.05, 0.1, 0.2, 0.4]:
        hp = _proj_to_sum(h + ps * rng.randn(n0), n0 * 0.5)
        cp = _eval(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()
    # Phase 1: subgradient
    dl1 = t0 + min(35, budget_s * 0.04)
    h1, c1 = _subgrad_fast(best_h.copy(), rng, t0, dl1, m_top=5)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    # Phase 2: main CD
    dl2 = t0 + budget_s * 0.40
    h2, c2 = _pair_cd(best_h.copy(), rng, t0, dl2, n_iter=500000)
    if c2 < best_c5:
        best_c5, best_h = c2, h2
    # Phase 3: anneal CD for exploration
    dl3 = t0 + budget_s * 0.55
    h3, c3 = _anneal_cd(best_h.copy(), rng, t0, dl3, n_iter=150000)
    if c3 < best_c5:
        best_c5, best_h = c3, h3
    # Phase 4: subgradient on improved
    dl4 = t0 + budget_s * 0.62
    h4, c4 = _subgrad_fast(best_h.copy(), rng, t0, dl4, m_top=4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4
    # Phase 5: more CD
    dl5 = t0 + budget_s * 0.85
    h5, c5 = _pair_cd(best_h.copy(), rng, t0, dl5, n_iter=500000)
    if c5 < best_c5:
        best_c5, best_h = c5, h5
    # Phase 6: final subgradient
    dl6 = t0 + budget_s * 0.90
    h6, c6 = _subgrad_fast(best_h.copy(), rng, t0, dl6, m_top=3)
    if c6 < best_c5:
        best_c5, best_h = c6, h6
    # Phase 7: fine polish
    dl7 = t0 + (budget_s - 5)
    h7, c7 = _fine_polish(best_h.copy(), rng, t0, dl7)
    if c7 < best_c5:
        best_c5, best_h = c7, h7
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
