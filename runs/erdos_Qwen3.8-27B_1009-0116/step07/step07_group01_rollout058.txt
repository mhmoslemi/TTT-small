    m = k - (n - 1)  # actual shift, ranges -(n-1) to (n-1)
    g = np.zeros(n)
    if m >= 0:

        jmax = n - 1 - m
        # h[j] contributes (1-h[j+m])*dx for j in [0, jmax]
        g[0:jmax+1] += dx * (1.0 - h[m:m+1+jmax])  # h[m] to h[m+jmax] = h[m] to h[n-1]
        # h[j+m] contributes -h[j]*dx for j in [0, jmax], so h[t] for t in [m, n-1]
        g[m:n] -= dx * h[0:jmax+1]  # h[0] to h[jmax]
    else:
        s = -m  # s > 0
        # c = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])*dx
        jmax = n - 1 - s
        # h[j+s] contributes (1-h[j])*dx for j in [0, jmax]
        # So h[t] for t in [s, n-1]
        g[s:n] += dx * (1.0 - h[0:jmax+1])  # h[0] to h[jmax]
        # h[j] contributes -h[j+s]*dx for j in [0, jmax]
        # So h[t] for t in [0, jmax]
        g[0:jmax+1] -= dx * h[s:s+1+jmax]  # h[s] to h[s+jmax] = h[s] to h[n-1]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        jmax = n - 1 - m
        if jmax >= 0:
            g[0:jmax+1] += dx * (1.0 - h[m:m+1+jmax])
            g[m:n] -= dx * h[0:jmax+1]
    else:
        s = -m
        jmax = n - 1 - s
        if jmax >= 0:
            g[s:n] += dx * (1.0 - h[0:jmax+1])
            g[0:jmax+1] -= dx * h[s:s+1+jmax]
    return g

def _grad_smooth(h, alpha, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    active = np.argsort(c)[-top_k:]
    ca = c[active]
    s = np.clip(alpha * (ca - ca.max()), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(active):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_lag(h, k)
    return g

def _psgd(h, alpha, steps, s0, top_k, mu):
    n = len(h)
    v = np.zeros(n)
    step = s0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _grad_smooth(h, alpha, top_k)
        v = mu * v + (1.0 - mu) * g
        gn = np.linalg.norm(v)
        if gn < 1e-18:
            break
        v = v / gn
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new; cur_c = c_new
                step = min(step * 1.04, s0 * 8.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, s0 * 8.0)
    return _repair(h, n), cur_c

def _subgrad(h, steps, s0):

    return _repair(h, n), cur_c

def _local3(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    stag = 0
    for _ in range(n_iter):
        step = base_step * min(1.0 + stag * 0.01, 5.0)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        k = (i + 8 + rng.randint(0, 12)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy(); stag = 0
                else:
                    stag += 1
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
                stag += 1
    return best_h, best_c

def _local2(h, n_iter, rng, base_step):

def _down(h, f):
    n = len(h); m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _up(h, f):
    return np.repeat(h, f)
