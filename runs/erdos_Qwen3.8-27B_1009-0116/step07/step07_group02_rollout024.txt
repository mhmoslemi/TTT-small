def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

        if _ % 500 == 0:
            c = _cvals(h)
            top_idx = np.argsort(c)[-5:]
    return best_h, best_c

def _perturb_reopt(h, n_pert, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for p in range(n_pert):
        if time.time() > deadline - 60:
            break
        h2 = h.copy()
        mask = rng.random(n) < 0.3
        h2[mask] += 0.02 * rng.randn(mask.sum())
        h2 = _repair(h2, n)
        h2 = _lbfgs(h2, 50000, 1500, deadline)
        c2 = _eval(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    return best_h, best_c

def _up(h, f):
    return np.repeat(h, f).copy()

def _down(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _ps2pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_iter):
        if time.time() > deadline - 1:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h0.copy(), _eval(h0), n0

    # Phase 1: L-BFGS log-sum-exp sweep
    h = h0.copy()
    for alpha in [3000, 15000, 60000, 300000]:
        if time.time() > deadline - 500:
            break
        h = _lbfgs(h, alpha, 5000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Tilted subgradient (exponential weighting of lags)
    if time.time() < deadline - 350:
        h2 = _tilted_subgrad(best_h.copy(), 1500, 0.015, 800.0, deadline)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2
        h2 = _tilted_subgrad(best_h.copy(), 1000, 0.008, 2000.0, deadline)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2

    # Phase 3: Perturb-and-reoptimize (escape local min)
    if time.time() < deadline - 250:
        h3, c3 = _perturb_reopt(best_h.copy(), 3, rng, deadline)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Multi-scale downsample/optimize/upsample
    if time.time() < deadline - 200:
        nh = n0 // 2
        hh = _repair(_down(best_h, 2), nh)
        for a in [10000, 100000]:
            if time.time() > deadline - 200:
                break
            hh = _lbfgs(hh, a, 4000, deadline)
        hu = _repair(_up(hh, 2), n0)
        hu = _lbfgs(hu, 200000, 4000, deadline)
        c5v = _eval(hu)
        if c5v < best_c5:
            best_c5, best_h = c5v, hu

    # Phase 5: Final L-BFGS high-alpha
    if time.time() < deadline - 120:
        h5 = _lbfgs(best_h, 500000, 5000, deadline)
        c5v = _eval(h5)
        if c5v < best_c5:
            best_c5, best_h = c5v, h5

    # Phase 6: Tilted subgradient fine pass
    if time.time() < deadline - 80:
        h6 = _tilted_subgrad(best_h.copy(), 3000, 0.003, 5000.0, deadline)
        c5v = _eval(h6)
        if c5v < best_c5:
            best_c5, best_h = c5v, h6

    # Phase 7: Nudge search on worst lags
    if time.time() < deadline - 40:
        h7, c7 = _nudge_search(best_h.copy(), 40000, rng, deadline)
        if c7 < best_c5:
            best_c5, best_h = c7, h7

    # Phase 8: Ultra-fine 2-point polish
    if time.time() < deadline - 2:
        h8, c8 = _ps2pt(best_h, 120000, rng, deadline)
        if c8 < best_c5:
            best_c5, best_h = c8, h8

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
