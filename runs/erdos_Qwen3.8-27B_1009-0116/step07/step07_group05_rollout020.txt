import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, lags, n):
    g = np.zeros(n)
    for d in lags:
        ad = abs(d)
        L = n - ad
        if L <= 0:
            continue
        if d >= 0:
            g[ad:n] += (1.0 - h[:L])
            g[:L] -= h[ad:ad+L]
        else:
            g[:L] += (1.0 - h[ad:ad+L])
            g[ad:n] -= h[:L]
    return g

def _smooth_obj(x, n, beta):

    m = c.max()
    sc = np.clip(beta * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / beta

    w /= w.sum()
    g = np.zeros(n)

        wk = w[n-1+d] * dx

        if wk2 < 1e-12:

        g[:L] += wk2 * (1.0 - x[d:d+L])
        g[d:n] -= wk2 * x[:L]

    dx = 2.0 / n
    h_k = h.copy().astype(float)
    h_prev = h_k.copy()
    z = h_k.copy()
    t_mom = 1.0
    lr = base_lr
    c5c = _eval(h_k)
    stall = 0

        g = _smooth_grad(z, n, beta)
        z_new = _project(z - lr * g, n * 0.5)
        c5n = _eval(z_new)
        if c5n <= c5c + 1e-10:

            z = np.clip(z, 0.0, 1.0)

            c5c = c5n

def _mlgd(h, n, rounds, k_top, steps_init, t0, budget):
