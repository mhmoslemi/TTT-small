    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl_vec(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _grad(h, alpha, topk):
    n = len(h)
    c = _correl_vec(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()

            g[l2:] += wk * (1.0 - h[:-l2])
            g[:n - l2] -= wk * h[l2:]

    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.9
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.01, step0 * 3.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _pair_descent(h, niter, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    c = _correl_vec(h)
    # Focus on top lags
    for it in range(niter):
        if time.time() > deadline:
            break
        i = it % n
        j = (i + 1 + (it // n) % 3) % n
        s = h[i] + h[j]
        # Try moving mass from i to j
        lo, hi = max(0, s - 1), min(1, s)
        # Current value
        t0 = h[i]
        best_t = t0
        best_c = c5c
        # Try a few values

            if c5n < best_c:

                best_t = t
        h[i], h[j] = best_t, s - best_t
        if best_c < c5c:
            c5c = best_c
            if c5c < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

By Lagrange duality: minimize c[a] + c[b] (if they're both active at the optimum).

The gradient of c[a] + c[b] is just the sum of the gradients of each. Much simpler!

Let me implement this:

```python
def _top2_obj(h, k1, k2):
    c = _correl_vec(h)
    return c[k1] + c[k2]

def _top2_grad(h, k1, k2):
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    nmid = n - 1
    for k in [k1, k2]:
        lag = k - nmid
        if lag >= 0:
            g[:n-lag] += dx * (1.0 - h[lag:])
            g[lag:] -= dx * h[:-lag]
        else:
            l2 = -lag
            g[l2:] += dx * (1.0 - h[:-l2])
            g[:n-l2] -= dx * h[l2:]
    return g
```

Then use Nesterov on this objective. When the top 2 lags change, update k1, k2.

This avoids the smoothmax bias entirely and gives exact gradients for the active constraints.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break

    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]

    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl_vec(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl_vec(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    nmid = n - 1
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        lag = k - nmid
        wk = w[ii] * dx
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:-lag]
        else:
            l2 = -lag
            g[l2:] += wk * (1.0 - h[:-l2])
            g[:n - l2] -= wk * h[l2:]
    return g

def _nesterov_smax(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.9
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 3.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _topk_grad(h, klist):
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    nmid = n - 1
    for k in klist:
        lag = k - nmid
        if lag >= 0:
            g[:n - lag] += dx * (1.0 - h[lag:])
            g[lag:] -= dx * h[:-lag]
        else:
            l2 = -lag
            g[l2:] += dx * (1.0 - h[:-l2])
            g[:n - l2] -= dx * h[l2:]
    return g

def _topk_nesterov(h, klist, steps, step0, deadline):
    n = len(h)
    c = _correl_vec(h)
    f = c[klist].sum()
    v = np.zeros(n)
    beta = 0.85
    step = step0
    kl = list(klist)
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _topk_grad(h, kl)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        cn = _correl_vec(hn)
        fn = cn[kl].sum()
        if fn < f:
            h, f, c = hn, fn, cn
            step = min(step * 1.005, step0 * 2.0)
        else:
            step *= 0.5
            h = _proj(h - step * v, n * 0.5)
            c = _correl_vec(h)
            f = c[kl].sum()
        if it % 150 == 0:
            kl = np.argpartition(c, -len(kl))[-len(kl):].tolist()
    return _repair(h, n)

def _pair_ternary(h, i, j, iters=18):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:
        return
    for _ in range(iters):
        m1 = lo + (hi - lo) * 0.333333
        m2 = hi - (hi - lo) * 0.333333
        h[i], h[j] = m1, s - m1
        f1 = _eval(h)
        h[i], h[j] = m2, s - m2
        f2 = _eval(h)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) * 0.5
    h[i], h[j] = t, s - t

def _pair_pass(h, offset, iters, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    pairs_tried = 0
    for _ in range(iters):
        if time.time() > deadline:
            break
        i = pairs_tried % n
        j = (i + 1 + (pairs_tried // n) % 7) % n
        old_i, old_j = h[i], h[j]
        _pair_ternary(h, i, j, 18)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            if c5c < bc:
                bc, bh = c5c, h.copy()
        else:
            h[i], h[j] = old_i, old_j
        pairs_tried += 1
    return bh, bc, c5c

def _fine_perturb(h, niter, deadline, base_scale):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    scale = base_scale
    rng = np.random.RandomState(12345)
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
                scale = min(scale * 1.05, 5e-4)
            else:
                h[i], h[j] = ni - d, nj + d
                scale *= 0.8
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Nesterov smoothmax (fast global optimization)
    dl1 = min(deadline, t0 + 120)
    for alpha, tk, st, ns in [(20000, 20, 0.004, 4000), (100000, 10, 0.0015, 4000), (400000, 5, 0.0005, 4000)]:
        if time.time() > dl1:
            break
        h = _nesterov_smax(best_h, alpha, tk, ns, st, dl1)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Exact top-K sum Nesterov (no smoothmax bias)
    dl2 = min(deadline, t0 + 300)
    c = _correl_vec(best_h)
    for tk in [4, 3, 2]:
        if time.time() > dl2:
            break
        kl = np.argpartition(c, -tk)[-tk:].tolist()
        h = _topk_nesterov(best_h, kl, 5000, 0.001, dl2)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        c = _correl_vec(best_h)

    # Phase 3: Pair ternary search (exact 1D optimization per pair)
    dl3 = min(deadline, t0 + 650)
    for rnd in range(5):
        if time.time() > dl3:
            break
        best_h, best_c5, _ = _pair_pass(best_h, rnd * 3, 20000, dl3)

    # Phase 4: Fine perturbation search
    dl4 = min(deadline, t0 + 900)
    best_h, best_c5 = _fine_perturb(best_h, 80000, dl4, 2e-5)

    # Phase 5: Another pair pass with remaining time
    if time.time() < deadline - 10:
        best_h, best_c5, _ = _pair_pass(best_h, 0, 15000, min(deadline, t0 + budget_s - 10))

    # Phase 6: Final fine polish
    if time.time() < deadline:
        best_h, best_c5 = _fine_perturb(best_h, 50000, deadline, 1e-6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
