            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

def _grad_weighted(h, alpha, max_lags):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    idx = np.argsort(c)[-max_lags:]
    s = alpha * (c[idx] - cm)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[i] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _fista(h_init, alpha, n_steps, step0, t0, time_limit, max_lags=100):

    h = h_init.copy()
    y = h.copy()
    t = 1.0
    step = step0
    f = _eval(h)
    for _s in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        g = _grad_weighted(y, alpha, max_lags)
        y_new = _proj_to_sum(y - step * g, n * 0.5)
        f_new = _eval(y_new)
        if f_new < f:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = y_new + ((t - 1.0) / t_new) * (y_new - y)
            h = y_new
            f = f_new
            t = t_new
            step = min(step * 1.01, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-14:
                break
    return h, f

        j = int(np.argmin(g))

def _slsqp_refine(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        return g

    def _cf(x):
        return x.sum() - target

    def _cj(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: FISTA low alpha (broad landscape)
    h1, f1 = _fista(h, 200, 4000, 0.008, t0, 50, max_lags=200)
    if f1 < best_c5:
        best_c5, best_h = f1, h1.copy()

    # Phase 2: FISTA medium alpha
    h2, f2 = _fista(best_h.copy(), 1000, 4000, 0.003, t0, 120, max_lags=100)
    if f2 < best_c5:
        best_c5, best_h = f2, h2.copy()

    # Phase 3: FISTA high alpha (focused on max lags)
    h3, f3 = _fista(best_h.copy(), 5000, 4000, 0.001, t0, 200, max_lags=40)
    if f3 < best_c5:
        best_c5, best_h = f3, h3.copy()

    # Phase 4: SLSQP refinement (moderate smoothing)
    if time.time() - t0 < budget_s - 300:
        h4 = _slsqp_refine(best_h.copy(), 3000, 2000, 1e-16)
        f4 = _eval(h4)
        if f4 < best_c5:
            best_c5, best_h = f4, h4.copy()

    # Phase 5: SLSQP high alpha
    if time.time() - t0 < budget_s - 200:
        h5 = _slsqp_refine(best_h.copy(), 15000, 2000, 1e-17)
        f5 = _eval(h5)
        if f5 < best_c5:
            best_c5, best_h = f5, h5.copy()

    # Phase 6: FISTA very high alpha (near-exact max gradient)
    if time.time() - t0 < budget_s - 120:
        h6, f6 = _fista(best_h.copy(), 50000, 3000, 0.0004, t0, budget_s - 100, max_lags=10)
        if f6 < best_c5:
            best_c5, best_h = f6, h6.copy()

    # Phase 7: Gradient-guided local search
    if time.time() - t0 < budget_s - 60:
        n_tries = min(int((budget_s - 60 - (time.time() - t0)) / 0.012), 60000)
        best_h, best_c5 = _grad_guided_search(best_h, n_tries, rng, t0, budget_s - 30)

    # Phase 8: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = _fine_1d_search(best_h, n, rng, t0, budget_s - 2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
