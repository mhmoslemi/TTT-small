def _repair(h, n):
    # Ensure 0<=h<=1 and sum(h) = n/2
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:

    return h

def _proj_to_sum(h, target):
    # Project h onto {x : sum(x) = target, 0 <= x <= 1}
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_single(h, k, n):
    # Gradient of c_k(h) = sum_i h[i]*(1-h[(i+k-(n-1)) % ...]) ...
    # Actually for np.correlate(h, 1-h, 'full'):
    # result[m] = sum_i h[i] * (1-h)[m - i + (n-1)] for appropriate range
    # = sum_i h[i] * (1 - h[m - i + (n-1)])
    # where m ranges from 0 to 2n-2
    # Let shift = m - (n-1), so shift ranges from -(n-1) to (n-1)
    # c_k = sum_i h[i] * (1 - h[i + shift]) where i+shift in [0, n-1]
    # dc_k/dh[j] = (1 - h[j+shift]) for j where j+shift in [0,n-1]  [from h[i] term, i=j]
    #            - h[j-shift] for j where j-shift in [0,n-1]  [from (1-h) term, i=j-shift]
    shift = k - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        # h[i] * (1-h[i+shift]): i from 0 to n-1-shift
        i_range = np.arange(0, n - shift)
        g[i_range] += (1.0 - h[i_range + shift])
        # If shift > 0: h[i-shift] term for i from shift to n-1
        if shift > 0:
            i_range2 = np.arange(shift, n)
            g[i_range2] -= h[i_range2 - shift]
    else:
        s2 = -shift
        # i from s2 to n-1: h[i] * (1-h[i-s2])
        i_range = np.arange(s2, n)
        g[i_range] += (1.0 - h[i_range - s2])
        # i from 0 to n-1-s2: h[i+s2] appears in (1-h) term
        i_range2 = np.arange(0, n - s2)
        g[i_range2] -= h[i_range2 + s2]
    return g * (2.0 / n)

    n = len(h)
    target = n / 2.0
    c = _corrs(h)
    f = c.max()
    v = np.zeros(n)
    mu = 0.9
    step = 0.005
    best_h = h.copy()
    best_f = f
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Compute top-K active gradients
        top_idx = np.argsort(c)[-top_K:]
        g = np.zeros(n)
        for k_idx in top_idx:
            g += _grad_single(h, k_idx, n)
        g /= len(top_idx)
        
        # Nesterov step
        v = mu * v + (1.0 - mu) * g
        h_try = _proj_to_sum(h - step * v, target)
        c_try = _corrs(h_try)
        f_try = c_try.max()
        
        if f_try < f - 1e-14:
            h = h_try
            f = f_try
            c = c_try
            step = min(step * 1.05, 0.05)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.6
            if step < 1e-8:
                break
    
    return best_h, best_f

import numpy as np
from scipy.optimize import linprog
import time

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n / 2.0) > 1e-12:
        h = _proj_to_sum(h, n / 2.0)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_k(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        r = n - s
        g[:r] += 1.0 - h[s:s + r]
        if s > 0:
            g[s:] -= h[:r]
    else:
        s2 = -s
        r = n - s2
        g[s2:] += 1.0 - h[:r]
        g[:r] -= h[s2:]
    return g * (2.0 / n)

def _nesterov_sm(h, alpha, t_end, max_iter=600):
    n = len(h)
    target = n / 2.0
    c = _corrs(h)
    f = c.max()
    v = np.zeros(n)
    mu = 0.88
    step = 0.004
    best_h, best_f = h.copy(), f
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        s = np.clip(alpha * (c - f), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k_idx in range(len(c)):
            if w[k_idx] < 1e-12:
                continue
            g += w[k_idx] * _grad_k(h, k_idx, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_try = _proj_to_sum(h - step * v, target)
        f_try = _eval(h_try)
        if f_try < f - 1e-15:
            h = h_try
            f = f_try
            c = _corrs(h)
            step = min(step * 1.04, 0.03)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.55
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, target)
            f = _eval(h)
            c = _corrs(h)
    return best_h, best_f

def _nesterov_exact(h, t_end, top_K=5, max_iter=400):
    n = len(h)
    target = n / 2.0
    c = _corrs(h)
    f = c.max()
    v = np.zeros(n)
    mu = 0.92
    step = 0.003
    best_h, best_f = h.copy(), f
    for it in range(max_iter):
        if time.time() > t_end:
            break
        top_idx = np.argsort(c)[-top_K:]
        g = np.zeros(n)
        for k_idx in top_idx:
            g += _grad_k(h, k_idx, n)
        g /= len(top_idx)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_try = _proj_to_sum(h - step * v, target)
        f_try = _eval(h_try)

def _lp_polish(h, t_end, max_iters=500):
    n = len(h)
    target = n / 2.0
    best_h, best_f = h.copy(), _eval(h)
    all_k = np.arange(2 * n - 1)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        c = _corrs(h)
        c_max = c.max()
        tol = max(0.003, 0.001 * (it + 1) / 20.0)
        active = c > c_max - tol
        if active.sum() < 3:
            idx = np.argsort(c)[-15:]
            active = np.zeros(len(c), dtype=bool)
            active[idx] = True
        aidx = np.where(active)[0]
        K = len(aidx)
        A = np.zeros((K, n + 1))
        b = np.zeros(K)
        for ii, k_idx in enumerate(aidx):
            A[ii, :n] = _grad_k(h, int(k_idx), n)
            A[ii, n] = -1.0
            b[ii] = -c[k_idx]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = list(zip(-h - 1e-14, 1.0 - h + 1e-14)) + [(0, None)]
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.status != 0:
            break
        delta = res.x[:n]
        improved = False
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.02]:
            h_try = np.clip(h + alpha * delta, 0, 1)
            h_try = _proj_to_sum(h_try, target)
            f_try = _eval(h_try)
            if f_try < best_f - 1e-13:
                h = h_try
                best_f = f_try
                best_h = h.copy()
                improved = True
                break
        if not improved and it > 5:
            break
    return best_h, best_f

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = min(time.time() + duration_s, t_end - 5)
    no_improve = 0
    while time.time() < end_t:
        c = _corrs(h)
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        b_mat = np.zeros((len(top_idx), n))
        for ii, k_idx in enumerate(top_idx):
            b_mat[ii] = _grad_k(h, int(k_idx), n)
        sweep_improved = False
        n_pairs = n * 3
        for _ in range(n_pairs):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(3, n // 8))) % n
            if i == j:
                continue
            dvec = b_mat[:, i] - b_mat[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = float(np.max(a_vals + dvec * x1))
            f2 = float(np.max(a_vals + dvec * x2))
            for _ in range(25):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = float(np.max(a_vals + dvec * x1))
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = float(np.max(a_vals + dvec * x2))
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            h = _proj_to_sum(h, n * 0.5)
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 30:
            h += 5e-6 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
    return h, best_c5

def _fine_local(h, n_iter, rng, t_end):

    target = n / 2.0
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c

        i = rng.randint(n)

        ni, nj = h[i] + d, h[j] - d
