For lag k (shift s = k - (n-1)):
- It involves h[i] and h[i+s] for all valid i
- So if s > 0, it couples h[0..n-s-1] with h[s..n-1]

If I perturb h[j], it affects all lags s where j-s or j+s is in range. Specifically, it affects lag k if j appears in the correlation sum for that k.

For a focused local search: identify the top few lags, find which h indices they involve, and perturb those indices preferentially.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topleague(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    top_idx = np.argsort(c)[-top_k:]
    c_top = c[top_idx]
    m_top = c_top.max()
    s = np.clip(alpha * (c_top - m_top), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(top_idx):
        wk = w[ii]
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        wkd = wk * dx
        if sh >= 0:
            g[:n - sh] += wkd * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wkd * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wkd * (1.0 - h[:n - sp])
            g[:n - sp] -= wkd * h[sp:]
    g -= g.mean()
    return g, m, top_idx

def _smoothed_gd(h, n_iter, alpha, step0, top_k, mom=0.82):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    cur_c = _eval(h)
    step = step0
    for _ in range(n_iter):
        g, m, _ = _grad_topleague(h, alpha, top_k)
        v = mom * v + (1.0 - mom) * g
        taken = False
        for _t in range(14):
            hn = _proj(h - step * v, target)
            cn = _eval(hn)
            if cn < cur_c:
                h = hn
                cur_c = cn
                step = min(step * 1.06, step0 * 12.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.4, step0 * 12.0)
    return _repair(h, n), cur_c

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n

    def _of(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s2 = np.clip(alpha * (c - m), -400, 400)
        s2 -= s2.max()
        return m + np.exp(s2).sum() / alpha

    def _cf(x):
        return x.sum() - 0.5 * n

    def _cj(x):
        return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_of, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _sa_local(h, n_iter, rng, base_step, t_start=0.003, t_end=1e-7):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        frac = it / max(1, n_iter)
        T = t_start * (t_end / t_start) ** frac
        step = base_step * (1.0 + 300.0 * T)
        np_ = rng.randint(2, 5)
        idxs = rng.choice(n, np_, replace=False)
        ds = rng.randn(np_) * step
        ds -= ds.mean()
        h_new = h.copy()
        for i in range(np_):
            h_new[idxs[i]] += ds[i]
        if np.any(h_new < -0.05) or np.any(h_new > 1.05):
            continue
        h_new = _repair(h_new, n)
        cn = _eval(h_new)
        if cn < c5c or rng.rand() < np.exp(-(cn - c5c) / max(T, 1e-10)):
            c5c = cn
            h = h_new
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
    return best_h, best_c

def _polish_2pt(h, n_iter, rng, step):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
            continue
        h[i], h[j] = max(0.0, min(1.0, ni)), max(0.0, min(1.0, nj))
        cn = _eval(h)
        if cn < c5c:
            c5c = cn
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            h[i], h[j] = ni, nj
    return best_h, best_c

def _multi_start_gd(h, n, rng, alpha, top_k, step0, n_iter, t0, budget_left_fn):
    best_h = h.copy()
    best_c = _eval(best_h)
    h_cur = h.copy()
    for attempt in range(5):
        if budget_left_fn() < 30:
            break
        if attempt > 0:
            h_cur = h.copy()
            npert = n // 20
            ids = rng.choice(n, npert, replace=False)
            h_cur[ids] += rng.randn(npert) * 0.01
            h_cur = _repair(h_cur, n)
        h_new, c5n = _smoothed_gd(h_cur, n_iter, alpha, top_k, step0)
        if c5n < best_c:
            best_c = c5n
            best_h = h_new.copy()
    return best_h, best_c
