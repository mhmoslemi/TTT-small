import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_k(h, lag):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - lag
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _grad_topk(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        g += _grad_k(h, k)
    return g * (2.0 / n) / len(lags)

def _grad_soft(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    w = np.exp((c - m) / temp)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:

        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

    b1 = 0.9
    for i in range(n_iter):

            break
        if mode == 'soft':

            g = _grad_soft(h, t)

        h *= target / s

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_k(h, idx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - idx
    if sh >= 0:
        g[sh:] += 1.0 - h[:n - sh]
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += 1.0 - h[sp:]
        g[sp:] -= h[:n - sp]
    return g

def _g_topk(h, idxs):
    n = len(h)
    g = np.zeros(n)
    for k in idxs:
        g += _grad_k(h, k)
    return g * (2.0 / n) / len(idxs)

def _g_soft(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    w = np.exp((c - m) / temp)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g
