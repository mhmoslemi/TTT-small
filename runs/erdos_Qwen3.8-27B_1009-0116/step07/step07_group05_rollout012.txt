    configs = [(5, [0.02, 0.01, 0.005, 0.002], 350),
               (10, [0.01, 0.005, 0.002, 0.001], 350),
               (15, [0.008, 0.004, 0.002, 0.001], 300),
               (25, [0.005, 0.002, 0.001, 0.0005], 300),
               (40, [0.003, 0.001, 0.0005, 0.0002], 300),
               (60, [0.002, 0.001, 0.0005, 0.0002, 0.0001], 250)]
    for kt, sn, nd in configs:
        if time.time() - t0 > budget_s - 180:
            break
        h2, c2 = _mlgd(best_h.copy(), nd, kt, sn, t0, budget_s)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Structured sinusoidal perturbations
    for freq in [1, 2, 3, 4, 6, 8, 2, 3]:
        if time.time() - t0 > budget_s - 150:
            break
        hs = best_h.copy()
        phase = rng.uniform(0, 2 * np.pi)
        hs += 0.004 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        hs = _repair(hs, n)
        h2, c2 = _mlgd(hs, 200, 10, [0.008, 0.004, 0.002, 0.001], t0, budget_s)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 4: Random perturbations and re-optimization
    if time.time() - t0 < budget_s - 130:
        h2, c2 = _perturb_try(best_h, best_c5, rng, t0, budget_s, 12)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 5: Final fine multi-lag polish
    for kt, sn in [(5, [0.003, 0.001, 0.0005]), (8, [0.002, 0.001, 0.0005]),
                   (12, [0.001, 0.0005, 0.0002]), (20, [0.0008, 0.0004, 0.0002])]:
        if time.time() - t0 > budget_s - 100:
            break
        h2, c2 = _mlgd(best_h.copy(), 250, kt, sn, t0, budget_s)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 6: 3-point local search
    el = time.time() - t0
    nt3 = max(0, min(int((budget_s - 30 - el) / 0.0035), 150000))
    if nt3 > 50:
        best_h, best_c5 = _fine3(best_h, nt3, rng, best_c5, t0, budget_s)

    # Phase 7: 2-point ultra-fine
    el = time.time() - t0
    nt2 = max(0, min(int((budget_s - 8 - el) / 0.0025), 150000))
    if nt2 > 50:
        best_h, best_c5 = _fine2(best_h, nt2, rng, best_c5, t0, budget_s)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
