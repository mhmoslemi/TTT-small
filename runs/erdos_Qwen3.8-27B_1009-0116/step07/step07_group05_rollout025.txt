        for si in range(len(steps)):

            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                ns = steps[si] * 1.4
                steps = [ns] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _sum_eq(x):
    return x.sum() - x.size * 0.5

def _sum_eq_jac(x):
    return np.ones(x.size)

def _sm_obj(x, a):
    n = x.size
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _sm_grad(x, a):
    n = x.size
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _upsample(h, factor):
    n = len(h)
    nn = n * factor
    ox = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    nx = np.linspace(0.0, 2.0, nn, endpoint=False) + 1.0 / nn
    return np.interp(nx, ox, h, period=2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Gradient cycling at base resolution
    for kt in [3, 5, 4, 3, 5, 4, 3]:
        if time.time() - t0 > budget_s - 350:
            break
        h2, c52 = _mulmg(t0, budget_s, best_h.copy(), 300, kt,
                         [0.06, 0.03, 0.015, 0.008, 0.004, 0.002])
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Multi-resolution upsample to 2x
    if time.time() - t0 < budget_s - 400 and n0 <= 700:
        n1 = n0 * 2
        h1 = _upsample(best_h, 2)
        h1 = _repair(h1, n1)
        c51 = _eval(h1)
        for kt in [3, 5, 4, 3, 5]:
            if time.time() - t0 > budget_s - 200:
                break
            h2, c52 = _mulmg(t0, budget_s, h1.copy(), 200, kt,
                             [0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
            if c52 < c51:
                c51 = c52
                h1 = h2
        # Keep best of upsolved and downsampled
        hb = h1.reshape(n0, 2).mean(axis=1)
        hb = _repair(hb, n0)
        cb = _eval(hb)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1

    # Phase 3: L-BFGS-B with smooth max
    if time.time() - t0 < budget_s - 200 and len(best_h) <= 900:
        n = len(best_h)
        a_s = 50000.0
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
        for _ in range(2):
            if time.time() - t0 > budget_s - 100:
                break
            try:
                res = minimize(_sm_obj, best_h, args=(a_s,),
                               jac=_sm_grad, method='L-BFGS-B',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-16, 'gtol': 1e-13})
                hr = _repair(res.x, n)
                cr = _eval(hr)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr
            except Exception:
                pass

    # Phase 4: Perturbation + gradient (SA-like escape)
    if time.time() - t0 < budget_s - 150:
        n = len(best_h)
        for T in [0.008, 0.004, 0.002, 0.001]:
            if time.time() - t0 > budget_s - 40:
                break
            for _ in range(3):
                if time.time() - t0 > budget_s - 30:
                    break
                hp = best_h + T * rng.randn(n)
                hp = _repair(hp, n)
                h2, c52 = _mulmg(t0, budget_s, hp, 80, 3,
                                 [0.03, 0.015, 0.008, 0.004, 0.002])
                if c52 < best_c5:
                    best_c5 = c52
                    best_h = h2

    # Phase 5: Adaptive-k gradient polish
    for kt in [3, 4, 5, 3, 4, 3, 5, 3]:
        if time.time() - t0 > budget_s - 35:
            break
        h2, c52 = _mulmg(t0, budget_s, best_h.copy(), 200, kt,
                         [0.008, 0.004, 0.002, 0.001, 5e-4, 2e-4])
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Multi-scale 2-point local search
    if time.time() - t0 < budget_s - 10:
        n = len(best_h)
        hw = best_h.copy()
        cw = best_c5
        elapsed = time.time() - t0
        ntry = min(int((budget_s - 12 - elapsed) / 0.004), 300000)
        for it in range(ntry):
            if time.time() - t0 > budget_s - 5:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            scale = 3e-6 if (it % 5 < 2) else 1e-5
            d = scale * rng.randn()
            ni, nj = hw[i] + d, hw[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                hw[i], hw[j] = ni, nj
                cn = _eval(hw)
                if cn < cw - 1e-19:
                    cw = cn
                else:
                    hw[i] -= d
                    hw[j] += d
        if cw < best_c5:
            best_c5 = cw
            best_h = hw

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
