```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval_h(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        wk = w[k] * dx
        if d >= 0:
            i1 = np.arange(n - d)
            g[i1] += wk * (1.0 - h[i1 + d])
            if d > 0:
                i2 = np.arange(d, n)
                g[i2] -= wk * h[i2 - d]
        else:
            s2 = -d
            i1 = np.arange(s2, n)
            g[i1] += wk * (1.0 - h[i1 - s2])
            i2 = np.arange(n - s2)
            g[i2] -= wk * h[i2 + s2]
    return g

def _nesterov_run(h, alpha, steps, step0, t_end):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for it in range(steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_sum(h_prev - step * v, n * 0.5)
            f = _smooth_obj(h, alpha)
    return h

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_c5 = _eval_h(h)
    end_t = time.time() + duration_s
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        # Precompute gradients for top lags
        grads = np.zeros((len(top_idx), n))
        for t, k in enumerate(top_idx):
            d = k - (n - 1)
            if d >= 0:
                i1 = np.arange(n - d)
                grads[t, i1] += (1.0 - h[i1 + d])
                if d > 0:
                    i2 = np.arange(d, n)
                    grads[t, i2] -= h[i2 - d]
            else:
                s2 = -d
                i1 = np.arange(s2, n)
                grads[t, i1] += (1.0 - h[i1 - s2])
                i2 = np.arange(n - s2)
                grads[t, i2] -= h[i2 + s2]
        grads *= dx
        improved = False
        n_sweeps = min(n, 200)
        for _ in range(n_sweeps):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(20):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            new_c5 = _eval_h(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 30:
            h += 1e-5 * rng.randn(n)
            h = _proj_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_sum(h, n * 0.5)
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval_h(h)
    best_h = h.copy()
    best_c = c5c
    step_sizes = [4e-6, 2e-6, 1e-6, 5e-6]
    si = 0
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = step_sizes[si] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval_h(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                stall = 0
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                si = min(si + 1, len(step_sizes) - 1)
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                stall += 1
                if stall > 200:
                    si = max(si - 1, 0)
                    stall = 0
    return best_h, best_c

def _binary_start(h, n):
    """Round to binary, fix sum to n/2"""
    hb = (h >= 0.5).astype(float)
    deficit = n // 2 - int(hb.sum())
    if deficit > 0:
        # Need more ones: set the largest fractional values to 1
        frac = np.argsort(h)[::-1]
        for idx in frac[:deficit]:
            hb[idx] = 1.0
    elif deficit < 0:
        # Need fewer ones: set the smallest values to 0
        deficit = -deficit
        frac = np.argsort(h)[:deficit]
        for idx in frac[:deficit]:
            hb[idx] = 0.0
    return hb
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval_h(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sobj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        d = k - (n - 1)
        wk = w[k] * dx
        if d >= 0:
            i1 = np.arange(n - d)
            g[i1] += wk * (1.0 - h[i1 + d])
            if d > 0:
                g[d:] -= wk * h[:n-d]
        else:
            s2 = -d
            i1 = np.arange(s2, n)
            g[i1] += wk * (1.0 - h[i1 - s2])
            g[:n-s2] -= wk * h[s2:]
    return g

def _nest(h, alpha, steps, step0, mu, t_end):
    n = len(h)
    f = _sobj(h, alpha)
    v = np.zeros(n)
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _sgrad(h, alpha)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _sobj(h_new, alpha)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj_sum(hp - step * v, n * 0.5)
            f = _sobj(h, alpha)
    return h

def _cd(h, n, dur, topT, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best = _eval_h(h)
    end = time.time() + dur
    stall = 0
    while time.time() < min(end, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[-topT:]
        av = c[top]
        grads = np.zeros((len(top), n))
        for t, k in enumerate(top):
            d = k - (n - 1)
            if d >= 0:
                grads[t, :n-d] += (1.0 - h[d:])
                if d > 0:
                    grads[t, d:] -= h[:n-d]
            else:
                s2 = -d
                grads[t, s2:] += (1.0 - h[:n-s2])
                grads[t, :n-s2] -= h[s2:]
        grads *= dx
        imp = False
        for _ in range(min(n, 300)):
            if time.time() > min(end, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j: continue
            b = grads[:, i] - grads[:, j]
            lo = max(-h[i], h[j] - 1.0) + 1e-14
            hi = min(1.0 - h[i], h[j]) - 1e-14
            if lo >= hi: continue
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = np.max(av + b * x1)
            f2 = np.max(av + b * x2)
            for _ in range(20):
                if f1 < f2:
                    hi, x2, f2 = x2, x1, f1
                    x1 = hi - gr * (hi - lo)
                    f1 = np.max(av + b * x1)
                else:
                    lo, x1, f1 = x1, x2, f2
                    x2 = lo + gr * (hi - lo)
                    f2 = np.max(av + b * x2)
            dopt = (lo + hi) / 2.0
            if abs(dopt) > 1e-14:
                h[i] += dopt
                h[j] -= dopt
                imp = True
        if imp:
            nc = _eval_h(h)
            if nc < best - 1e-15:
                best = nc
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 25:
            h += 1e-5 * rng.randn(n)
            h = _proj_sum(h, n * 0.5)
            stall = 0
        if abs(h.sum() - n*0.5) > 1e-7:
            h = _proj_sum(h, n * 0.5)
    return h, best

def _lsearch(h, niter, rng, t_end):
    n = len(h)
    c5c = _eval_h(h)
    best_h, best_c = h.copy(), c5c
    sizes = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
    si = 0
    stall = 0
    for _ in range(niter):
        if time.time() > t_end: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = sizes[si] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval_h(h)
            if cn < c5c - 1e-18:
                c5c = cn
                stall = 0
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
                si = min(si + 1, len(sizes) - 1)
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                stall += 1
                if stall > 300:
                    si = max(si - 1, 0)
                    stall = 0
    return best_h, best_c

def _bin_start(h, n):
    hb = (h >= 0.5).astype(float)
    n1 = int(hb.sum())
    n0 = n // 2
    if n1 > n0:
        idx = np.argsort(h)[:n1 - n0]
        hb[idx] = 0.0
    elif n1 < n0:
        idx = np.argsort(h)[::-1][:n0 - n1]
        hb[idx] = 1.0
    return hb
```

```python
                    a = min(-diff, h[idx])
```

            x2 = lo + gr * (hi - lo)
            f1 = np.max(av + b * x1)
            f2 = np.max(av + b * x2)
            for _ in range(20):
                if f1 < f2:
                    hi, x2, f2 = x2, x1, f1
                    x1 = hi - gr * (hi - lo)
                    f1 = np.max(av + b * x1)
                else:
                    lo, x1, f1 = x1, x2, f2
                    x2 = lo + gr * (hi - lo)
                    f2 = np.max(av + b * x2)
            dopt = (lo + hi) * 0.5
            if abs(dopt) > 1e-14:
                h[i] += dopt
                h[j] -= dopt
                imp = True
        if imp:
            nc = _eval_h(h)
            if nc < best - 1e-15:
                best = nc
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 25:
            h += 1e-5 * rng.randn(n)
            h = _proj_sum(h, n * 0.5)
            stall = 0
        if abs(h.sum() - n * 0.5) > 1e-7:
            h = _proj_sum(h, n * 0.5)
    return h, best

def _lsearch(h, niter, rng, t_end):
    n = len(h)
    c5c = _eval_h(h)
    best_h, best_c = h.copy(), c5c
    sizes = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
    si = 0
    stall = 0
    for _ in range(niter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = sizes[si] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval_h(h)
            if cn < c5c - 1e-18:
                c5c = cn
                stall = 0
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
                si = min(si + 1, len(sizes) - 1)
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                stall += 1
                if stall > 300:
                    si = max(si - 1, 0)
                    stall = 0
    return best_h, best_c

def _bin_start(h, n):
    hb = (h >= 0.5).astype(float)
    n1 = int(hb.sum())
    tgt = n // 2
    if n1 > tgt:
        idx = np.argsort(h)[:n1 - tgt]
        hb[idx] = 0.0
    elif n1 < tgt:
        idx = np.argsort(h)[::-1][:tgt - n1]
        hb[idx] = 1.0
    return hb
