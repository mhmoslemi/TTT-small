def _subgrad_step(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top:
        sh = k - (n - 1)
        if sh >= 0:
            idx_lo = max(0, n - sh)
            for p in range(idx_lo):
                g[p] += (1.0 - h[p + sh])
                if sh > 0:
                    g[p + sh] -= 0  # handled below
            for p in range(n - sh):
                g[p + sh] -= h[p]
        else:
            sp = -sh
            for p in range(n - sp):
                g[p] += (1.0 - h[p + sp])
                g[p + sp] -= h[p]
    g *= dx / top_k
    return g

def _subgrad_descent(h, rng, deadline, max_iter=800):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _subgrad_step(h, top_k=4)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        step = min(0.02, 0.3 / gn)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1]:
            h_new = _proj_sum(h - mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c
