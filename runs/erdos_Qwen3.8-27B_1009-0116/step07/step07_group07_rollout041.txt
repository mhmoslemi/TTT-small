    best_h = h_c.copy()
    stall = 0
    k = k_start
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        if stall > 50:
            k = min(k + 2, k_max)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        dx = 2.0 / n
        for kk in topk:
            wk = dx / len(topk)
            sh = (n-1) - kk
            if sh >= 0:
                g[:n-sh] += wk*(1.0-h_c[sh:])
                g[sh:] -= wk*h_c[:n-sh]
            else:
                sp=-sh
                g[sp:] += wk*(1.0-h_c[:n-sp])
                g[:n-sp] -= wk*h_c[sp:]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta*v + (1.0-beta)*(g/gn)
        h_try = _proj_sum(h_c - step*v, n*0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step*1.04, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _lsexp_descent(h, n_steps, T_start, T_end, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    T = T_start
    step = 0.008
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    for _ in range(n_steps):
        if bc_fn():
            break
        T = T_start * (T_end/T_start)**(_/max(n_steps,1))
        g = _lsexp_grad(h_c, T, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        v = momentum*v + (1-momentum)*g
        h_try = _proj_sum(h_c - step*v, n*0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step*1.03, 0.06)
        else:
            step *= 0.45
            if step < 1e-10:
                break
            stall += 1
            if stall > 80:
                v = np.zeros(n)
                step = 0.005
                stall = 0
    return best_h, best_c

def _local_search(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 8e-5
    c_vals = _corrs(h_c)
    top_idx = np.argsort(c_vals)[-25:]
    for it in range(n_iter):
        if bc_fn():
            break
        if it % 2000 == 0 and it > 0:
            scale *= 0.65
            c_vals = _corrs(h_c)
            top_idx = np.argsort(c_vals)[-25:]
        if it % 3 == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 8)) % n
        else:
            i = top_idx[rng.randint(len(top_idx))]
            j = rng.randint(n)
        d = rng.randn() * scale
        ni, nj = h_c[i]+d, h_c[j]-d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            c_new = _eval(h_c)
            if c_new < best_c:
                best_c = c_new
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni-d, nj+d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def el():
        return time.time() - t0
    def bc():
        return el() > budget_s - 15
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    # Phase 1: L-BFGS-B p-norm warmup
    for p in [32, 128, 512]:
        if el() > 80: break
        h_try = _lbfgs_pnorm(best_h.copy(), p, 1e6, 400, n0)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    # Phase 2: Log-sum-exp gradient descent (adaptive T)
    if el() < budget_s - 400:
        h_try, c_try = _lsexp_descent(best_h, 4000, 0.03, 0.0005, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    # Phase 3: L-BFGS-B high p polish
    if el() < budget_s - 350:
        for p in [1024, 2048, 4096]:
            if el() > budget_s - 350: break
            h_try = _lbfgs_pnorm(best_h.copy(), p, 1e7, 300, n0)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try
    # Phase 4: Nesterov top-k
    if el() < budget_s - 250:
        h_try, c_try = _nesterov_topk(best_h, 10000, 4, 60, 0.012, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    # Phase 5: L-BFGS-B ultra-high p
    if el() < budget_s - 200:
        for p in [4096, 8192]:
            if el() > budget_s - 200: break
            h_try = _lbfgs_pnorm(best_h.copy(), p, 1e8, 300, n0)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try
    # Phase 6: Nesterov finer
    if el() < budget_s - 150:
        h_try, c_try = _nesterov_topk(best_h, 8000, 2, 25, 0.004, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    # Phase 7: Log-sum-exp very low T
    if el() < budget_s - 100:
        h_try, c_try = _lsexp_descent(best_h, 3000, 0.002, 0.0002, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    # Phase 8: Local search
    if el() < budget_s - 20:
        nt = min(int((budget_s - 20 - el()) / 0.003), 150000)
        best_h, best_c5 = _local_search(best_h, nt, rng, el, bc)
    # Phase 9: Final L-BFGS polish
    if el() < budget_s - 5:
        h_try = _lbfgs_pnorm(best_h.copy(), 4096, 1e8, 150, n0)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
