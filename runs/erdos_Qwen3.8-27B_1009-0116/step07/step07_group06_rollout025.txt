import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = _proj_to_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _grad_lag(h, j, n):
    g = np.zeros(n)
    if j >= 0 and j < n:
        g[:n-j] += (1.0 - h[j:])
        g[j:] -= h[:n-j]
    elif j < 0 and -j < n:
        ja = -j
        g[ja:] += (1.0 - h[:n-ja])
        g[:n-ja] -= h[ja:]
    return g

def _smooth_momentum(h, rng, deadline):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _eval(best_h)
    momentum = np.zeros(n)
    for beta in [5, 30, 100, 400]:
        lr = min(0.03, 4.0 / (beta + 1))
        mom_decay = 0.82
        for it in range(3000):
            if time.time() > deadline:
                return best_h
            h = np.clip(h, 0, 1)
            c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
            top_k = np.argsort(c)[-min(12, 2*n-1):]
            c_top = c[top_k]
            c_max = c_top.max()
            shifted = np.clip(beta * (c_top - c_max), -300, 0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w /= w_sum
            grad = np.zeros(n)
            for idx, wi in zip(top_k, w):
                if wi < 1e-10:
                    continue
                j = idx - (n - 1)
                grad += wi * _grad_lag(h, j, n)
            grad *= beta * (2.0 / n)
            momentum = mom_decay * momentum - lr * grad
            h_new = h + momentum
            h_new = _proj_to_sum(h_new, target)
            c5_new = _eval(h_new)
            if c5_new < best_c - 1e-14:
                best_c = c5_new
                best_h = h_new.copy()
            h = h_new
            lr *= 0.9995
    return best_h

def _pair_cd(h, n_iter, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 3000 == 0 and stable > 8000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(50):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _fine_polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(500000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _try_coarser(h_init, rng, t0, deadline):
    """Try a coarser discretization to escape local optima, then upsample."""
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c = _eval(best_h)
    for n_coarse in [n_init // 2, n_init // 4]:
        if n_coarse < 64:
            continue
        if time.time() > deadline:
            break
        h_c = np.interp(np.linspace(0, n_init, n_coarse, endpoint=False),
                        np.arange(n_init), h_init).clip(0, 1)
        h_c = _proj_to_sum(h_c, n_coarse / 2.0)
        h_c = _repair(h_c, n_coarse)
        c_c = _eval(h_c)
        if c_c < 0.39:
            h_c, c_cd = _pair_cd(h_c, 20000, rng, min(time.time() + 60, deadline))
            if c_cd < 0.39:
                h_up = np.interp(np.linspace(0, n_coarse, n_init, endpoint=False),
                                 np.arange(n_coarse), h_c).clip(0, 1)
                h_up = _proj_to_sum(h_up, n_init / 2.0)
                h_up = _repair(h_up, n_init)
                c_up = _eval(h_up)
                if c_up < best_c:
                    best_c = c_up
                    best_h = h_up.copy()
    return best_h, best_c
