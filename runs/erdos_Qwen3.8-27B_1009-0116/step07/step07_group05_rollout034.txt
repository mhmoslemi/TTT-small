Let me think about the subgradient computation more carefully.

For lag k (shift s in correlation array, where s = (n-1) + k_offset):
c_k = dx * sum_{valid i} h[i] * (1 - h[i+s_offset])

The gradient of c_k w.r.t. h[j]:
- For each term in the sum where h[j] appears:
  - If j is the first factor: (1 - h[j+s_offset]) * dx
  - If j is the second factor: -h[j-s_offset] * dx

So gradient of c_k w.r.t. h is:
For shift s (where s can be negative, 0, or positive):
- If s >= 0: c_s = dx * sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
  d(c_s)/d(h[j]) = dx * (1-h[j+s]) for j in [0, n-1-s] (from first factor)
                + (-dx) * h[j-s] for j in [s, n-1] (from second factor, where j-s is in [0, n-1-s])
  
- If s < 0, let p = -s: c_{-p} = dx * sum_{i=p}^{n-1} h[i] * (1-h[i-p])
  d(c_{-p})/d(h[j]) = dx * (1-h[j-p]) for j in [p, n-1] (first factor)
                   + (-dx) * h[j+p] for j in [0, n-1-p] (second factor)

This is what the _grad_lags function in the previous code computes.

For the max, the subgradient is the gradient of the active lag (or a convex combination if multiple lags are tied).

For a smoothed max (log-sum-exp), the gradient is a weighted average of all lag gradients, weighted by softmax.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_at_shift(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[:m] += (1.0 - h[s:]) * dx
        if s > 0:
            g[s:] -= h[:m] * dx
    else:
        p = -s
        g[p:] += (1.0 - h[:-p]) * dx
        g[:n-p] -= h[p:] * dx
    return g

def _mom_opt(h, n_iter, margin, a0, beta, check_fn):
    n = len(h)
    c5c = _eval(h)
    v = np.zeros(n)
    alpha = a0
    stall = 0
    for _ in range(n_iter):
        if check_fn():
            break
        c = _corrvals(h)
        mx = c.max()
        mask = c > mx - margin
        idx = np.where(mask)[0]
        if len(idx) == 0:
            idx = np.array([np.argmax(c)])
        g = np.zeros(n)
        for si in idx:
            g += _grad_at_shift(h, si - (n - 1))
        g /= len(idx)
        v = beta * v - alpha * g
        hn = _repair(h + v, n)
        cn = _eval(hn)
        if cn < c5c - 1e-14:
            h = hn
            c5c = cn
            alpha = min(alpha * 1.05, 0.5)
            stall = 0
        else:
            alpha *= 0.6
            v *= 0.3
            stall += 1
            if stall > 20:
                v[:] = 0
                alpha = max(a0 * 0.4, 1e-6)
                stall = 0
                if alpha < 1e-7:
                    break
    return h, c5c

def _anneal(h, n_steps, T0, T1, check_fn, rng):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    for t in range(n_steps):
        if check_fn():
            break
        frac = t / max(n_steps, 1)
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 0.005 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            delta = cn - c5c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                c5c = cn
                if cn < best_c:
                    best_c = cn
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, best_c

def _lsx_smooth(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad_smooth(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_at_shift(x, k - (n - 1))
    return g

def _slsqp(h, n, a, maxiter, check_fn):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _lsx_smooth(x, n, a), h,
                       jac=lambda x: _lsx_grad_smooth(x, n, a),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hr = _repair(res.x, n)
        return hr, _eval(hr)
    except Exception:
        return h.copy(), _eval(h)

def _local_multi(h, n_try, rng, c5c, check_fn, npoints):
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_at_shift(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s]) * dx
        g[:n - s] -= h[s:] * dx
    else:
        p = -s
        g[:n - p] += (1.0 - h[p:]) * dx
        g[p:] -= h[:n - p] * dx
    return g

def _mom_opt(h, n_iter, margin, a0, beta, check_fn):
    n = len(h)
    c5c = _eval(h)
    v = np.zeros(n)
    alpha = a0
    stall = 0
    for _ in range(n_iter):
        if check_fn():
            break
        c = _corrvals(h)
        mx = c.max()
        idx = np.where(c > mx - margin)[0]
        if len(idx) == 0:
            idx = np.array([np.argmax(c)])
        g = np.zeros(n)
        for si in idx:
            g += _grad_at_shift(h, si - (n - 1))
        g /= len(idx)
        v = beta * v - alpha * g
        hn = _repair(h + v, n)
        cn = _eval(hn)
        if cn < c5c - 1e-14:
            h = hn
            c5c = cn
            alpha = min(alpha * 1.03, 0.3)
            stall = 0
        else:
            alpha *= 0.65
            v *= 0.4
            stall += 1
            if stall > 18:
                v[:] = 0.0
                alpha = max(a0 * 0.5, 1e-6)
                stall = 0
                if alpha < 1e-7:
                    break
    return h, c5c

def _lsx_obj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _grad_at_shift(x, k - (n - 1))
    return g

            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        k = (i + 10 + rng.randint(0, 9)) % n
        d1 = 5e-6 * rng.randn()
        d2 = 5e-6 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < bc - 1e-18:
                bc = cn
                best = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bu():
        return _el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Momentum subgradient with various margins
    margins = [0.015, 0.008, 0.004, 0.002, 0.001]
    for mg in margins:
        if _bu():
            break
        h2, c2 = _mom_opt(best_h.copy(), 8000, mg, 0.06, 0.85, _bu)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 2: Simulated annealing
    if _el() < budget_s - 180:
        nt = min(int((budget_s - 180 - _el()) / 0.00012), 2500000)
        h2, c2 = _anneal(best_h.copy(), nt, 0.015, 1e-8, _bu, rng)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: SLSQP with two temperatures
    if _el() < budget_s - 100:
        for a_vals in [40000.0, 120000.0]:
            if _bu():
                break
            h2, c2 = _slsqp_run(best_h, a_vals, 600)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 4: Fine momentum optimization
    for mg in [0.003, 0.001, 0.0005]:
        if _bu():
            break
        h2, c2 = _mom_opt(best_h.copy(), 10000, mg, 0.025, 0.9, _bu)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 5: Block perturbation + re-optimize
    for _ in range(4):
        if _bu():
            break
        bp = best_h.copy()
        bstart = rng.randint(0, n0 - 40)
        bsize = 30 + rng.randint(0, 30)
        bp[bstart:bstart + bsize] += 0.03 * rng.randn(bsize)
        bp = _repair(bp, n0)
        h2, c2 = _mom_opt(bp, 4000, 0.004, 0.04, 0.85, _bu)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 6: Shift + noise restart
    for sh in [1, 2, 3]:
        if _bu():
            break
        hp = np.roll(best_h, sh).copy()
        hp += 0.002 * rng.randn(n0)
        hp = _repair(hp, n0)
        h2, c2 = _mom_opt(hp, 3000, 0.003, 0.03, 0.85, _bu)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 7: 2-point local search
    if _el() < budget_s - 50:
        nt2 = min(int((budget_s - 50 - _el()) / 0.0001), 1500000)
        best_h, best_c5 = _local_2pt(best_h, nt2, rng, best_c5, _bu)

    # Phase 8: 3-point local search
    if _el() < budget_s - 15:
        nt3 = min(int((budget_s - 15 - _el()) / 0.00012), 1000000)
        best_h, best_c5 = _local_3pt(best_h, nt3, rng, best_c5, _bu)

    # Final: ultra-fine 2-point
    if _el() < budget_s - 3:
        nt4 = min(int((budget_s - 3 - _el()) / 0.00008), 500000)
        best_h, best_c5 = _local_2pt(best_h, nt4, rng, best_c5, _bu)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
```
