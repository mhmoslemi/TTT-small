def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _topm_grad(h, mlags):
    n = len(h)
    dx = 2.0 / n
    c = _cvals(h)
    top_k = np.argsort(c)[-mlags:]
    g = np.zeros(n)
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            np_ = n - sh
            g[:np_] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:np_]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:-sp] -= h[sp:]
    return g * dx / mlags

def _hns(h, n_iter, step_sizes, rng, deadline):
    """Targeted local search: perturb pairs related to top lags"""
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    bestc = c5c
    for ss in step_sizes:
        for _ in range(n_iter):
            if time.time() > deadline - 5:
                return best, bestc
            c = _cvals(h)
            top_k = np.argsort(c)[-3:]
            k = top_k[rng.randint(len(top_k))]
            sh = (n - 1) - k
            if abs(sh) >= n:
                i = rng.randint(n)
            else:
                if sh >= 0:
                    i = rng.randint(0, max(n - sh, 1))
                else:
                    i = rng.randint(-sh, n)
            j = i + sh if sh != 0 else (i + 1) % n
            d = ss * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bestc:
                        bestc = c5n
                        best = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return best, bestc

def _random_pair_ls(h, n_iter, step, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    bestc = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bestc:
                    bestc = c5n
                    best = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best, bestc

def _subgrad_adv(h, steps, step0, deadline):
    """Adaptive subgradient: focus on worst lags, Polyak-like step"""
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    best = h.copy()
    bestc = _eval(h)
    for t in range(steps):
        if time.time() > deadline - 10:
            break
        step = step0 / (1.0 + 0.01 * t)
        c = _cvals(x)
        m = c.max()
        tol = 1e-7 + 0.5 * m * (1.0 - t / steps) * 0.1
        active = np.where(c >= m - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= x[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - x[:-sp])
                g[:-sp] -= x[sp:]
        g *= dx / len(active)
        x = _proj(x - step * g, n)
        c5 = _eval(x)
        if c5 < bestc:
            bestc = c5
            best = x.copy()
    return best, bestc

def _pm_pass(h, mlags, max_rounds, deadline):
    """Top-m projected gradient with backtracking"""
    n = len(h)
    c5c = _eval(h)
    x = h.copy()
    for r in range(max_rounds):
        if time.time() > deadline - 10:
            break
        step_base = 0.04 / (1.0 + 0.05 * r)
        g = _topm_grad(x, mlags)
        improved = False
        for step in [step_base, step_base*0.5, step_base*0.25, step_base*0.1, 1e-4, 1e-5]:
            x_new = _proj(x - step * g, n)
            c5n = _eval(x_new)
            if c5n < c5c - 1e-16:
                x = x_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return x, c5c

def _gen_start(n, rng, idx):
    x = np.linspace(0, 2, n, endpoint=False)
    if idx == 0:
        h = 0.5 + 0.45 * np.sin(2*np.pi*x*0.75) + 0.1*np.sin(4*np.pi*x)
    elif idx == 1:
        h = 0.5 + 0.4 * np.sin(2*np.pi*x*1.5+0.3) + 0.15*np.cos(6*np.pi*x)
    elif idx == 2:
        h = 0.5 + 0.35*np.cos(2*np.pi*x*2.0) + 0.2*np.sin(5*np.pi*x+1.0)
    elif idx == 3:
        h = 0.5 + 0.4*np.sin(2*np.pi*x*3.0+0.7)
    elif idx == 4:
        h = (x<0.4).astype(float)*0.9 + (x>1.1).astype(float)*0.7 + 0.2*np.sin(4*np.pi*x)
    else:
        h = rng.rand(n)*0.5+0.25
    return _proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _proj(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Diverse low-res multi-start (n=140)
    n_low = 140
    candidates = []
    for idx in range(6):
        hl = _gen_start(n_low, rng, idx)
        for alpha in [800, 5000]:
            if time.time() > deadline - 700:
                break
            bnds = [(0.0, 1.0)] * n_low
            try:
                res = minimize(_smax_obj, hl, jac=_smax_grad, args=(alpha,),
                               method='L-BFGS-B', bounds=bnds,
                               options={'maxiter': 1500, 'ftol': 1e-14})
                hl = _proj(res.x, n_low)
            except Exception:
                pass
        cl = _eval(hl)
        candidates.append((cl, hl))
        if time.time() > deadline - 600:
            break
    candidates.sort(key=lambda x: x[0])

    # Phase 2: Upsample best low-res to n0, also keep h0
    upsampled = []
    for cl, hl in candidates[:3]:
        if time.time() > deadline - 500:
            break
        # Linear interp from n_low to n0
        xp = np.linspace(0, 2, n_low, endpoint=False)
        xq = np.linspace(0, 2, n0, endpoint=False)
        hu = np.interp(xq, xp, hl, period=2.0)
        hu = _proj(hu, n0)
        upsampled.append(hu)
    all_starts = [h0] + upsampled

    # Phase 3: L-BFGS refinement on all starts with increasing alpha
    for start_i, hs in enumerate(all_starts):
        if time.time() > deadline - 300:
            break
        h = hs.copy()
        for alpha in [2000, 10000, 50000, 200000, 800000]:
            if time.time() > deadline - 300:
                break
            bnds = [(0.0, 1.0)] * n0
            try:
                res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                               method='L-BFGS-B', bounds=bnds,
                               options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-11})
                h = _proj(res.x, n0)
            except Exception:
                pass
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
                break

    # Phase 4: Top-m projected gradient passes
    if time.time() < deadline - 200:
        for ml in [2, 3, 5, 4]:
            if time.time() > deadline - 200:
                break
            hp, cp = _pm_pass(best_h.copy(), ml, 800, deadline)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()

    # Phase 5: Adaptive subgradient
    if time.time() < deadline - 150:
        hg, cg = _subgrad_adv(best_h, 3000, 0.015, deadline)
        if cg < best_c5:
            best_c5 = cg
            best_h = hg.copy()

    # Phase 6: Very high alpha L-BFGS
    if time.time() < deadline - 100:
        bnds = [(0.0, 1.0)] * n0
        try:
            res = minimize(_smax_obj, best_h, jac=_smax_grad, args=(2000000,),
                           method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12})
            h = _proj(res.x, n0)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
        except Exception:
            pass

    # Phase 7: Targeted local search (pairs at top-lag shifts)
    if time.time() < deadline - 60:
        hs, cs = _hns(best_h, 500, [0.05, 0.02, 0.008, 0.003, 0.001, 3e-4, 1e-4, 3e-5, 1e-5, 3e-6, 1e-6], rng, deadline)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()

    # Phase 8: Random pair local search (fine)
    if time.time() < deadline - 20:
        for ss in [1e-4, 3e-5, 1e-5, 3e-6, 1e-6, 3e-7]:
            if time.time() > deadline - 20:
                break
            hs, cs = _random_pair_ls(best_h, 20000, ss, rng, deadline)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()

    # Final: one more top-m pass + tiny local search
    if time.time() < deadline - 5:
        hp, cp = _pm_pass(best_h.copy(), 3, 200, deadline)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
        hs, cs = _random_pair_ls(best_h, 50000, 1e-7, rng, deadline)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()

    best_h = _proj(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
