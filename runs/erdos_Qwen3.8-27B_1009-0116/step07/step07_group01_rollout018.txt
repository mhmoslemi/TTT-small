    bnds = [(0.0, 1.0)] * N
    try:
        res = minimize(_obj, h0.copy(), _jac, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 60})
        return _repair(res.x, N)
    except Exception:
        return _repair(h0, N)

def _grad_descent(h, n_steps, step0, top_k, temp, momentum):
    N = len(h)
    target = N / 2.0
    v = np.zeros(N)
    step = step0
    cur_c = _eval(h)
    for _ in range(n_steps):
        g = _topk_grad(h, top_k, temp)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, target)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 12.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.5, step0 * 12.0)
    return h, cur_c

def _slsqp_smooth(h, alpha, maxiter, ftol):
    N = len(h)
    dx = 2.0 / N
    target = N / 2.0

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(N)
        for K in range(2 * N - 1):
            if w[K] < 1e-14:
                continue
            g += w[K] * dx * _grad_lag(x, K)
        return g

    def _cfun(x):
        return x.sum() - target

    def _cjac(x):
        return np.ones(N)

    bnds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_obj, h.copy(), _jac, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, N)
    except Exception:
        return _repair(h, N)

def _binary_relax(h, thresh):
    N = len(h)
    h = h.copy()
    h[h < thresh] = 0.0
    h[h > 1.0 - thresh] = 1.0
    return _repair(h, N)

def _downsample(h, nc):
    n = len(h)
    f = n // nc
    if f < 1:
        return h.copy()
    return h[:nc * f].reshape(nc, f).mean(axis=1).copy()

def _upsample(h, n):
    return np.repeat(h, n // len(h)).copy()

def _local_search(h, n_trials, rng, step_size):
    N = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_h = h.copy()
    cur_c = best_c
    for _ in range(n_trials):
        i = rng.randint(N)
        j = (i + 1 + rng.randint(0, 4)) % N
        d = step_size * rng.randn()
        ni, nj = cur_h[i] + d, cur_h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            cur_h[i], cur_h[j] = ni, nj
            c = _eval(cur_h)
            if c < cur_c:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = cur_h.copy()
            else:
                cur_h[i], cur_h[j] = ni - d, nj + d
    return best_h, best_c

def _gradient_local(h, n_steps, step, top_k, temp):
    N = len(h)
    target = N / 2.0
    cur_c = _eval(h)
    for _ in range(n_steps):
        g = _topk_grad(h, top_k, temp)
        h_new = _proj(h - step * g, target)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
        else:
            h_new2 = _proj(h - step * 0.5 * g, target)
            c_new2 = _eval(h_new2)
            if c_new2 < cur_c:
                h = h_new2
                cur_c = c_new2
    return h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    N = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-scale L-BFGS-B
    for ratio in [4, 2]:
        if _el() > 100:
            break
        nc = max(32, N // ratio)
        hc = _repair(_downsample(best_h, nc), nc)
        for alpha in [2000, 20000, 200000]:
            if _el() > 130:
                break
            hc = _lbfgs_smooth(hc, alpha, 400)
        h_up = _repair(_upsample(hc, N), N)
        h_gd, c_gd = _grad_descent(h_up, 200, 0.01, 6, 3000, 0.75)
        c_new = _eval(h_gd)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_gd.copy()

    # Phase 2: Full-res L-BFGS-B alpha sweep
    for alpha in [5000, 50000, 500000, 5000000]:
        if _el() > 350:
            break
        h_new = _lbfgs_smooth(best_h.copy(), alpha, 400)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 3: SLSQP at moderate alpha
    if _el() < budget_s - 350:
        for alpha in [20000, 200000]:
            if _el() > budget_s - 250:
                break
            h_new = _slsqp_smooth(best_h.copy(), alpha, 200, 1e-15)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 4: Gradient descent with various params
    if _el() < budget_s - 250:
        for (steps, s0, tk, temp, mom) in [
            (400, 0.005, 5, 2000, 0.8),
            (300, 0.003, 8, 8000, 0.85),
            (300, 0.002, 3, 500, 0.9),
            (400, 0.004, 12, 20000, 0.7),
        ]:
            if _el() > budget_s - 120:
                break
            h_gd, _ = _grad_descent(best_h.copy(), steps, s0, tk, temp, mom)
            c_new = _eval(h_gd)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_gd.copy()

    # Phase 5: Binary relaxation + re-optimize
    if _el() < budget_s - 200:
        for thresh in [0.05, 0.02]:
            if _el() > budget_s - 120:
                break
            h_bin = _binary_relax(best_h.copy(), thresh)
            c_bin = _eval(h_bin)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
            h_re = _lbfgs_smooth(best_h.copy(), 100000, 300)
            c_re = _eval(h_re)
            if c_re < best_c5:
                best_c5 = c_re
                best_h = h_re.copy()

    # Phase 6: Very fine gradient steps (top 1-2 lags)
    if _el() < budget_s - 100:
        h_gd, c_gd = _gradient_local(best_h.copy(), 5000, 0.001, 2, 500)
        c_new = _eval(h_gd)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_gd.copy()

    # Phase 7: Fine local search
    if _el() < budget_s - 15:
        n_trials = min(int((budget_s - 15 - _el()) / 0.0012), 600000)
        best_h, best_c5 = _local_search(best_h, n_trials, rng, 4e-5)

    # Phase 8: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 2 - _el()) / 0.0009), 500000)
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 3)) % nc2
            d = 1.5e-6 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
