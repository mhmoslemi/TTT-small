```python
def _subgrad_descent(h, rng, deadline, max_iter=3000):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    m_top = 3
    lr = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            # Vectorized gradient for this shift
            if sh >= 0:
                idx1 = np.arange(n - sh)
                g[idx1] += (1.0 - h[idx1 + sh])
                idx2 = np.arange(sh, n)
                g[idx2] -= h[idx2 - sh]
            else:
                sp = -sh
                idx1 = np.arange(sp, n)
                g[idx1] += (1.0 - h[idx1 - sp])
                idx2 = np.arange(n - sp)
                g[idx2] -= h[idx2 + sp]
        g *= dx / m_top
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        # Backtracking line search
        step = lr / gn
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_to_sum(h - mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            lr *= 0.95
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
        # Adaptive learning rate
        if improved:
            lr = min(lr * 1.05, 0.05)
    return best_h, best_c
```

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)

        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
