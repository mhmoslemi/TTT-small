def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.83
    step = step0
    best_f = f
    best_h = h.copy()
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.45
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
            if f < best_f:
                best_f = f
                best_h = h.copy()
    return _repair(best_h, n)

def _subgrad_phase(h, niter, deadline, lr0=0.003):
    n = len(h)
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    for it in range(niter):
        if time.time() > deadline:
            break
        lr = lr0 / (1.0 + it * 0.005)
        c = _correl(h)
        k_star = np.argmax(c)
        g = np.zeros(n)
        dx = 2.0 / n
        sh = (n - 1) - k_star
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:-sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:-sp])
            g[:n - sp] -= dx * h[sp:]
        hn = _proj(h - lr * g, n * 0.5)
        fn = _eval(hn)
        if fn < f:
            h = hn
            f = fn
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            lr *= 0.5
            hn = _proj(h - lr * g, n * 0.5)
            fn = _eval(hn)
            if fn < f:
                h = hn
                f = fn
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
    return _repair(best_h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or (T > 1e-8 and rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12))):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline, step0=1e-5, step_min=1e-8):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = step0
    for it in range(niter):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            step *= 0.6
        if step < step_min:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc - 1e-15:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _3pt_polish(h, niter, rng, deadline, step0=5e-6):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = step0
    for it in range(niter):
        if time.time() > deadline:
            break
        if it % 5000 == 0 and it > 0:
            step *= 0.55
        if step < 1e-9:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        k = (i + 1 + rng.randint(5, 12)) % nc
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0<=ni<=1 and 0<=nj<=1 and 0<=nk<=1:
            h[i],h[j],h[k] = ni,nj,nk
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc - 1e-15:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i],h[j],h[k] = ni-d1, nj-d2, nk+d1+d2
    return bh, bc

def _slsqp_refine(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_topk(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with moderate alpha
    dl1 = min(deadline, t0 + 120)
    for alpha, tk, mi in [(5000, 40, 200), (30000, 20, 200)]:
        if time.time() > dl1:
            break
        h_tmp = _slsqp_refine(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h_tmp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_tmp.copy()

    # Phase 2: Nesterov with increasing alpha
    dl2 = min(deadline, t0 + 350)
    nested_configs = [
        (10000, 40, 0.005, 2500),
        (50000, 25, 0.003, 3000),
        (200000, 15, 0.0015, 3000),
        (800000, 10, 0.0008, 3000),
        (3000000, 7, 0.0004, 3000),
    ]
    for alpha, tk, st, ns in nested_configs:
        if time.time() > dl2:
            break
        h_tmp = _nesterov(best_h.copy(), alpha, tk, ns, st, dl2)
        c5v = _eval(h_tmp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_tmp.copy()

    # Phase 3: Subgradient on true max
    dl3 = min(deadline, t0 + 420)
    if time.time() < dl3:
        h_tmp = _subgrad_phase(best_h.copy(), 5000, dl3, lr0=0.004)
        c5v = _eval(h_tmp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_tmp.copy()

    # Phase 4: Block annealing
    dl4 = min(deadline, t0 + 650)
    blocks = [1, 2, 3, 4, 5, 7, 10, 15]
    h_a, c5_a = _block_anneal(best_h.copy(), 80000, rng, dl4, 0.004, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 5: Second Nesterov round from annealed solution
    dl5 = min(deadline, t0 + 820)
    for alpha, tk, st, ns in [(500000, 8, 0.0005, 2500), (2000000, 5, 0.0002, 2500)]:
        if time.time() > dl5:
            break
        h_tmp = _nesterov(best_h.copy(), alpha, tk, ns, st, dl5)
        c5v = _eval(h_tmp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_tmp.copy()

    # Phase 6: Subgradient again
    dl6 = min(deadline, t0 + 880)
    if time.time() < dl6:
        h_tmp = _subgrad_phase(best_h.copy(), 3000, dl6, lr0=0.002)
        c5v = _eval(h_tmp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_tmp.copy()

    # Phase 7: Extensive fine polish (geometric step decay)
    dl7 = min(deadline, t0 + budget_s - 60)
    best_h, best_c5 = _fine_polish(best_h.copy(), 500000, rng, dl7, step0=2e-5, step_min=1e-9)

    # Phase 8: 3-point polish
    dl8 = t0 + budget_s - 5
    best_h, best_c5 = _3pt_polish(best_h.copy(), 300000, rng, dl8, step0=3e-6)

    # Final: one more fine polish at very small steps
    if time.time() < t0 + budget_s - 2:
        best_h, best_c5 = _fine_polish(best_h.copy(), 100000, rng, t0 + budget_s - 2,
                                       step0=1e-7, step_min=1e-10)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
