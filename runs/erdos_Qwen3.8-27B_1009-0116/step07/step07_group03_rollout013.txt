    # Phase 1: Brief Nesterov to polish initial
    for alpha in [8000, 50000, 200000]:
        t_end = min(t0 + 15, t0 + budget_s * 0.06)
        h = _nesterov(h, alpha, 300, 0.003, t_end)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Aggressive coordinate descent with larger neighborhood
    elapsed = time.time() - t0
    cd_budget = budget_s * 0.30 - elapsed
    if cd_budget > 20:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, cd_budget, 6, rng, t0 + budget_s, neighborhood=n // 4)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.08, 10, rng, t0 + budget_s, neighborhood=n // 3)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Simulated annealing to escape local minima
    elapsed = time.time() - t0
    sa_budget = budget_s * 0.15 - elapsed
    if sa_budget > 15:
        h = best_h.copy()
        h, c5v = _anneal_local(h, n, sa_budget, rng, t0 + budget_s, step_init=2e-4)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Coordinate descent after annealing
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.75:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.08, 5, rng, t0 + budget_s, neighborhood=n // 5)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.06, 8, rng, t0 + budget_s, neighborhood=n // 6)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Peak shaving
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.90:
        remaining = (t0 + budget_s * 0.92) - time.time()
        if remaining > 5:
            n_iter = int(min(remaining * 200, 20000))
            h = best_h.copy()
            h, c5v = _peak_shave(h, n, n_iter, rng, t0 + budget_s * 0.93)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 6: Final micro annealing for last bit of polish
    elapsed = time.time() - t0
    if elapsed < budget_s - 10:
        remaining = (t0 + budget_s - 5) - time.time()
        if remaining > 5:
            h = best_h.copy()
            h, c5v = _anneal_local(h, n, remaining * 0.6, rng, t0 + budget_s - 3, step_init=1e-5)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
            remaining2 = (t0 + budget_s - 5) - time.time()
            if remaining2 > 3:
                h = best_h.copy()
                h, c5v = _peak_shave(h, n, int(min(remaining2 * 300, 5000)), rng, t0 + budget_s - 2)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h.copy()

    # Phase 7: Final coordinate polish
    elapsed = time.time() - t0
    if elapsed < budget_s - 3:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, max(1, (t0 + budget_s) - time.time() - 3), 5, rng, t0 + budget_s, neighborhood=n // 7)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
