import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cyc_cd(h, n, n_sweeps, deadline, three_n=40):
    best = h.copy()
    best_c = _eval(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            # Compute coefficient vector s
            s = np.zeros(2*n-1)
            p1 = np.arange(i, i+n)
            j1 = n-1 - np.arange(n)
            s[p1] += (1.0 - h[j1])
            p2 = np.arange(n-1-i, n-1+n)
            s[p2] -= h[np.arange(n)]
            
            # 1D subproblem: minimize max_p(corr[p] + (t-h[i])*s[p]) over t in [0,1]
            # = min max_p(s[p]*t + corr[p] - h[i]*s[p])
            b = corr - h[i] * s
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(three_n):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(s * m1 + b) < np.max(s * m2 + b):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            # Update
            delta = t_opt - h[i]
            if abs(delta) > 1e-12:
                h[i] = t_opt
                corr += delta * s
        h = _repair(h, n)
        # Recompute corr after repair
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = _eval(h)
        if c < best_c:
            best_c = c
            best = h.copy()
    return best, best_c
