import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_val(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _fista(h, alpha, topk, max_steps, step0, deadline):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    t = 1.0
    step = step0
    f = _smax_val(y, alpha, topk)
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        g = _smax_grad(y, alpha, topk)
        x_new = _proj(y - step * g, target)
        f_new = _smax_val(x_new, alpha, topk)
        for _ in range(10):
            if f_new <= f + 1e-10:
                break
            step *= 0.4
            x_new = _proj(y - step * g, target)
            f_new = _smax_val(x_new, alpha, topk)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        x = x_new
        f = f_new
        step = min(step * 1.08, step0 * 80.0)
        t = t_new
    return _repair(x, n)

def _slsqp_quick(h, alpha, topk, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        obj_fn = lambda x: _smax_val(x, alpha, topk)
        jac_fn = lambda x: _smax_grad(x, alpha, topk)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                 'jac': lambda x: np.ones(n)}]
        r = minimize(obj_fn, h, jac=jac_fn, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': 200, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _targeted_search(h, niter, rng, deadline):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    step = 0.01
    for it in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        threshold = m - 0.001
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = [c.argmax()]
        else:
            active = active[rng.randint(len(active))]
        k = int(active) if np.isscalar(active) else int(active[0])
        sh = (n - 1) - k
        # Gradient direction for this lag
        dx = 2.0 / n
        g = np.zeros(n)
        if sh >= 0:
            d = (1.0 - h[sh:]) * dx
            g[:n - sh] += d
            if sh > 0:
                g[sh:] -= h[:-sh] * dx
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp]) * dx
            g[:n - sp] -= h[sp:] * dx
        # Move along -g, but only for a few random coordinates
        n_move = rng.randint(1, 6)
        coords = rng.choice(n, n_move, replace=False)
        direction = -g[coords]
        norm = np.sqrt(np.sum(direction**2)) + 1e-15
        direction = direction / norm
        # Try step
        accepted = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            trial = h.copy()
            trial[coords] += step * frac * direction
            trial = _repair(trial, n)
            c5n = _eval(trial)
            if c5n < c5 - 1e-18:
                h = trial
                c5 = c5n
                step = min(step * 1.3, 0.1)
                accepted = True
                if c5 < bc:
                    bc = c5
                    bh = h.copy()
                break
        if not accepted:
            step *= 0.5
    return bh, bc

def _pair_flip(h, niter, rng, deadline, scale):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 < ni < 1.0 and 0.0 < nj < 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _fine_polish(h, niter, rng, deadline, scale=2e-6):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-19:
                c5 = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc
