import time

    n = len(h)

def _nesterov_topk(h, n_iter, k0, k1, s0, el_fn, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.75
    step = s0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    k = k0
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(hc)
        if stall > 40:
            k = min(k + 1, k1)
            stall = 0
        topk = np.argsort(c)[-k:]
        g = _topk_grad(hc, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1 - beta) * (g / gn)
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 1e-16:
            hc = ht
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.04, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0., 1.).copy()
    if abs(h.sum() - n * 0.5) > 1e-10:
        h = _proj_sum(h, n * 0.5)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1. - h, 'full')) * 2. / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1. - h, 'full') * 2. / n

def _lse_obj(h, T, pen):
    c = _corrs(h)
    cm = c.max()
    n = len(h)
    return (cm + np.log(np.sum(np.exp(T * (c - cm))))) / T + pen * (h.sum() - 0.5 * n) ** 2

def _lse_grad(h, T, pen):
    n = len(h)
    dx = 2. / n
    c = _corrs(h)
    cm = c.max()
    e = np.exp(T * (c - cm))
    w = T * e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1. - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1. - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2. * pen * (h.sum() - 0.5 * n)
    return g

def _pnorm_obj(h, p, pen):
    n = len(h)
    c = np.maximum(_corrs(h), 1e-30)
    return (np.sum(c ** p)) ** (1. / p) + pen * (h.sum() - 0.5 * n) ** 2

def _pnorm_grad(h, p, pen):
    n = len(h)
    dx = 2. / n
    c = np.maximum(_corrs(h), 1e-30)
    cp = c ** p
    s = cp.sum()
    factor = s ** (1. / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = factor * c[k] ** (p - 1) * dx
        if wk < 1e-14:
            continue
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1. - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1. - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2. * pen * (h.sum() - 0.5 * n)
    return g

def _lbfgs(h, obj_fn, grad_fn, args, maxiter):
    n = len(h)
    bnds = [(0., 1.)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16,
                       'gtol': 1e-12, 'maxls': 40, 'maxfun': maxiter * 3})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _topk_grad(h, idx):
    n = len(h)
    dx = 2. / n
    g = np.zeros(n)
    for k in idx:
        wk = dx
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1. - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1. - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g / len(idx)

def _nesterov(h, n_iter, k0, k1, s0, el_fn, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.75
    step = s0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    k = k0
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(hc)
        if stall > 50:
            k = min(k + 2, k1)
            stall = 0
            step = max(step, 0.005)
        topk = np.argsort(c)[-k:]
        g = _topk_grad(hc, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1. - beta) * (g / gn)
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 1e-15:
            hc = ht
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.05, 0.06)
        else:
            step *= 0.45
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _gauss_perturb(h, n, sigma, rng):
    p = rng.randn(n)
    ks = max(5, n // 40)
    ker = np.exp(-0.5 * (np.arange(-ks, ks + 1) / max(2, ks // 4)) ** 2)
    ker /= ker.sum()
    p = np.convolve(p, ker, mode='same')
    return _repair(h + sigma * p, n)

def _fine_search(h, n_iter, rng, el_fn, bc_fn, mode=2):
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    scale = 2e-4
    for _ in range(n_iter):
        if bc_fn():
            break
        if mode == 2:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 8)) % n
            d = rng.randn() * scale
            ni, nj = hc[i] + d, hc[j] - d
            if 0. <= ni <= 1. and 0. <= nj <= 1.:
                hc[i], hc[j] = ni, nj
                cn = _eval(hc)
                if cn < best_c:
                    best_c = cn
                    best_h = hc.copy()
                else:
                    hc[i], hc[j] = ni - d, nj + d
        else:
            i = rng.randint(n)
            j = (i + 2 + rng.randint(0, 10)) % n
            m = (i + 12 + rng.randint(0, 20)) % n
            d1 = 1.5e-4 * rng.randn()
            d2 = 1.5e-4 * rng.randn()
            ni, nj, nm = hc[i] + d1, hc[j] + d2, hc[m] - d1 - d2
            if 0. <= ni <= 1. and 0. <= nj <= 1. and 0. <= nm <= 1.:
                hc[i], hc[j], hc[m] = ni, nj, nm
                cn = _eval(hc)
                if cn < best_c:
                    best_c = cn
                    best_h = hc.copy()
                else:
                    hc[i], hc[j], hc[m] = ni - d1, nj - d2, nm + d1 + d2
        if _ % 4000 == 0:
            scale *= 0.72
    return best_h, best_c
