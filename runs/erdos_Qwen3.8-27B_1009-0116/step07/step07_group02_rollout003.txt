import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _lse_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _topk_grad(h, k_list):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_list:
        sh = (n - 1) - k
        if sh >= 0:
            np_ = n - sh
            g[:np_] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:np_]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:-sp] -= h[sp:]
    g *= dx / len(k_list)
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-11,
                                'maxls': 30, 'maxfun': maxit * 5})
        return _repair(res.x, n)
    except Exception:
        return h

def _topk_anneal(h, deadline, total_steps=6000):
    n = len(h)
    c5_cur = _eval(h)
    k_max = 30
    for phase in range(6):
        k_lo = max(1, k_max // (phase + 1))
        k_hi = max(k_lo, k_max // max(1, phase))
        steps = total_steps // 6
        for i in range(steps):
            if time.time() > deadline - 25:
                return h, _eval(h)
            c = _cvals(h)
            top = np.argsort(c)[-k_hi:]
            g = _topk_grad(h, top)
            t = 0.04 / (1.0 + 0.05 * (phase * steps + i))
            h_new = _repair(h - t * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-18:
                h = h_new
                c5_cur = c5n
    return h, c5_cur

def _block_local(h, n_iter, rng, deadline, block_sizes=(2, 3, 4, 5)):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 5:
            break
        bsz = block_sizes[rng.randint(len(block_sizes))]
        i = rng.randint(0, n - bsz)
        delta = 2e-5 * rng.randn(bsz)
        delta -= delta.sum() * 0.1
        h[:n] = h  # ensure contiguous
        new_block = h[i:i+bsz] + delta
        if np.any(new_block < 0) or np.any(new_block > 1):
            delta *= 0.3
            new_block = h[i:i+bsz] + delta
            if np.any(new_block < 0) or np.any(new_block > 1):
                continue
        h[i:i+bsz] = new_block
        c5n = _eval(h)
        if c5n < c5c - 1e-19:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i:i+bsz] = h[i:i+bsz] - delta
    return best_h, best_c

def _coord_descent(h, n_passes, deadline):
    n = len(h)
    c5c = _eval(h)
    for p in range(n_passes):
        if time.time() > deadline - 10:
            break
        for i in range(n):
            if time.time() > deadline - 10:
                return h, _eval(h)
            # Try small adjustments to h[i], compensated by neighbors
            for d in [5e-5, -5e-5, 1e-4, -1e-4]:
                j = (i + 1) % n
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    c5n = _eval(h)
                    if c5n < c5c - 1e-19:
                        c5c = c5n
                        break
                h[i] -= d
                h[j] += d
    return h, c5c

def _perturb_restart(h, rng, n_pert):

    h_p = h.copy()

    # Phase 5: Block local search
    if time.time() < deadline - 100:
        h, c5v = _block_local(best_h.copy(), 60000, rng, deadline)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Coordinate descent
    if time.time() < deadline - 40:
        h, c5v = _coord_descent(best_h.copy(), 3, deadline)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 7: Perturbation restart + quick L-BFGS
    if time.time() < deadline - 60:
        for _ in range(3):
            if time.time() > deadline - 60:
                break
            h_p = _perturb_restart(best_h, rng, 3)
            h_p = _lbfgs(h_p, 500000, 1500, deadline)
            c5v = _eval(h_p)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_p.copy()

    # Phase 8: Final top-k polish
    if time.time() < deadline - 30:
        h, c5v = _topk_anneal(best_h.copy(), deadline, total_steps=4000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 9: Ultra-fine block polish
    if time.time() < deadline - 10:
        h, c5v = _block_local(best_h.copy(), 40000, rng, deadline, block_sizes=(2, 2, 3, 3))
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 10: Final micro coordinate descent
    if time.time() < deadline - 3:
        h, c5v = _coord_descent(best_h.copy(), 2, deadline)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
