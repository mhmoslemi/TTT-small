import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _alt_minmax(h, n_iters, deadline, eta=2.0):

    best_h = h.copy()

    alpha = 1.0

            break

        lam = c_pos ** eta

                sp = -d

        h_bb[order[:half]] = 1.0
        if n % 2 == 1:

        alpha *= 0.995
        h = (1.0 - alpha) * h + alpha * h_bb

        c5_new = c.max()
        if c5_new < best_c - 1e-10:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        g[lo1:hi1+1] += 1.0 - h[np.arange(lo1, hi1+1) + p - (n-1)]
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[(n-1) + p - np.arange(lo2, hi2+1)]
    return g * (2.0 / n)

def _alt_minmax(h, n_iters, deadline, eta=2.0):
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    best_h = h.copy()
    best_c = _eval(h)
    alpha = 1.0
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_pos = np.maximum(c, 1e-10)
        lam = c_pos ** eta
        lam /= lam.sum()
        y = 1.0 - h
        w = np.zeros(n)
        for k in range(n_corr):
            if lam[k] < 1e-7:
                continue
            d = k - (n - 1)
            if d >= 0:
                w[:n-d] += lam[k] * y[d:]
            else:
                sp = -d
                w[sp:] += lam[k] * y[:n-sp]
        order = np.argsort(w)
        h_bb = np.zeros(n)
        half = n // 2
        h_bb[order[:half]] = 1.0
        if n % 2 == 1:
            h_bb[order[half]] = 2.0 * (n / 2.0 - half)
        alpha *= 0.993
        h = (1.0 - alpha) * h + alpha * h_bb
        h = _repair(h, n)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_new = c.max()
        if c5_new < best_c - 1e-10:
            best_c = c5_new
            best_h = h.copy()
    return best_h, best_c

def _pair_cd(h, n_iter, rng, t0, deadline, n_bisect=20):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    nc = 2 * n - 1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it > 500 and stable > 3000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(n_bisect):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < nc:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < nc:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < nc:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < nc:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < nc:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < nc:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _subgrad_descent(h, rng, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    m_top = 3
    for it in range(2000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            if sh >= 0:
                if sh < n:
                    g[:n-sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n-sh]
            else:
                sp = -sh
                if sp < n:
                    g[sp:] += (1.0 - h[:n-sp])
                if sp > 0:
                    g[:n-sp] -= h[sp:]
        g *= dx / m_top
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.01, 0.5 / gn)
        improved = False
        for s_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_to_sum(h - s_mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            continue
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def _fine_polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(80000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-16:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
