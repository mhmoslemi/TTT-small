import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, top_k, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    m = c[idx].max()
    s = (c[idx] - m) * temp
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        s_abs = abs(k - (n - 1))
        if s_abs == 0:
            g += 2.0 * w[ii] * dx * (0.5 - h)
        else:
            g[:n - s_abs] -= w[ii] * dx * h[s_abs:]
            g[s_abs:] += w[ii] * dx * (1.0 - h[:n - s_abs])
    return g, c

def _gd(h, n_iter, step0, mom, top_k, temp, t0, deadline):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur = _eval(h)
    stall = 0
    for it in range(n_iter):

            break
        g, _ = _grad_topk(h, top_k, temp)
        v = mom * v + (1.0 - mom) * g
        taken = False
        for _ in range(10):
            h_new = _proj_sum(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur:
                h = h_new
                cur = c_new
                stall = 0
                step = min(step * 1.03, step0 * 12.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            stall += 1
            step = min(step * 1.5, step0 * 12.0)
            if stall > 50:

                d = np.concatenate([[step * 0.3 * np.random.randn(4)], np.zeros(n - 4)])
                h = _proj_sum(h + d, n * 0.5)
                cur = _eval(h)
                stall = 0
    return h, cur

def _blk_obj(x, h_full, start, end):
    ht = h_full.copy()
    ht[start:end] = x
    return _eval(ht)

def _blk_con(x, bsum):
    return x.sum() - bsum

def _blk_jac(x, bsum):
    return np.ones(len(x))

def _block_opt(h, bsz, maxit, t0, deadline):
    n = len(h)
    nb = n // bsz
    for _pass in range(3):
        if time.time() > deadline:
            break
        for b in range(nb):
            sl0 = b * bsz
            sl1 = sl0 + bsz
            x0 = h[sl0:sl1].copy()
            bsum = x0.sum()
            cons = [{'type': 'eq', 'fun': _blk_con, 'jac': _blk_jac, 'args': (bsum,)}]
            try:
                res = minimize(_blk_obj, x0, args=(h, sl0, sl1), method='SLSQP',
                               bounds=[(0, 1)] * bsz, constraints=cons,
                               options={'maxiter': maxit, 'ftol': 1e-15})
                h[sl0:sl1] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _local_informed(h, n_iter, rng, step, t0, deadline):
    n = len(h)
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    for _ in range(n_iter):
        if time.time() > deadline:

        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top = np.argsort(c)[-5:]
        k = top[rng.randint(5)]
        sh = k - (n - 1)
        s_abs = abs(sh)
        if s_abs >= n:

        i_max = n - s_abs - 1
        if i_max <= 0:

        i = rng.randint(0, i_max + 1)
        if sh >= 0:
            j = i + s_abs
        else:
            j = i - s_abs
        d = step * rng.randn()
        h[i] += d
        h[j] -= d
        h = _proj_sum(h, n * 0.5)
        c_new = _eval(h)
        if c_new < cur:

            if c_new < best_c:

                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
            h = _proj_sum(h, n * 0.5)
    return best_h, best_c

def _local_random(h, n_iter, rng, step, t0, deadline):
    n = len(h)
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    for _ in range(n_iter):
        if time.time() > deadline:

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur:
                cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
