        return time.time() - t0

    def _bc():
        return _el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam with annealed temperature (broad exploration then focus)
    h1, c1 = _adam_run(best_h.copy(), n0, 3000, 0.008, 0.001, 0.15, _bc)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Focused top-k gradient descent
    for kt in [3, 5, 4, 6, 3, 5]:
        if _bc():
            break
        h2, c2 = _topk_desc(best_h.copy(), n0, 400, kt,
                            [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001], _bc)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: More Adam refinement with lower LR
    h3, c3 = _adam_run(best_h.copy(), n0, 2000, 0.003, 0.0005, 0.06, _bc)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: SLSQP polish
    if _el() < budget_s - 120:
        bnds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jacobian, 'args': (n0,)}]
        for _att in range(2):
            if _bc():
                break
            try:
                res = minimize(_slsqp_obj, best_h, args=(n0,), jac=_slsqp_grad,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16})
                hr = _repair(res.x, n0)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5, best_h = c5r, hr
            except Exception:
                pass

    # Phase 5: Perturbation + re-optimize (escape local minima)
    for _r in range(4):
        if _bc():
            break
        h_sh = best_h.copy()
        h_sh += 0.003 * rng.randn(n0)
        h_sh = _repair(h_sh, n0)
        h4, c4 = _adam_run(h_sh, n0, 800, 0.004, 0.001, 0.08, _bc)
        if c4 < best_c5:
            best_c5, best_h = c4, h4
        h5, c5 = _topk_desc(best_h.copy(), n0, 200, 4,
                            [0.03, 0.015, 0.008, 0.004, 0.002], _bc)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 6: Final top-k polish
    for kt in [3, 5, 4, 3]:
        if _bc():
            break
        h6, c6 = _topk_desc(best_h.copy(), n0, 300, kt,
                            [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4], _bc)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Phase 7: Gradient-informed local search
    if _el() < budget_s - 5:
        remaining = budget_s - 5 - _el()
        ntry = int(min(remaining / 0.004, 150000))
        best_h, best_c5 = _grad_local(best_h, n0, ntry, rng, best_c5, _bc)

    # Final tiny Adam
    if _el() < budget_s - 3:
        h7, c7 = _adam_run(best_h.copy(), n0, 500, 0.0005, 0.0002, 0.02, _bc)
        if c7 < best_c5:
            best_c5, best_h = c7, h7

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
