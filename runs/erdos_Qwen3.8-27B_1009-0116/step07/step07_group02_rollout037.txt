import numpy as np
import time
from scipy.optimize import minimize


def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size == 0:
            break
        h[idx] += diff / float(idx.size)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if idx.size:
            p = int(idx[0])
            h[p] += diff
            if h[p] < 0.0:
                h[p] = 0.0
            elif h[p] > 1.0:
                h[p] = 1.0
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _resample_to(h, m):
    n = len(h)
    if m == n:
        return _repair(h, n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.linspace(0.0, 2.0, m, endpoint=False)
    v = np.interp(y, x, h)
    return _repair(v, m)


def _mirror(h):
    n = len(h)
    return _repair(0.5 * (h + 1.0 - h[::-1]), n)


def _antiper(h):
    n = len(h)
    m = n // 2
    if m <= 0:
        return _repair(h, n)
    a = 0.5 * (h[:m] + 1.0 - h[m:2 * m])
    h2 = np.empty(n, dtype=float)
    h2[:m] = a
    h2[m:2 * m] = 1.0 - a
    if n > 2 * m:
        h2[2 * m:] = h[2 * m:]
    return _repair(h2, n)


def _lowpass(h, frac):
    n = len(h)
    f = np.fft.rfft(h)
    cut = max(1, int(len(f) * frac))
    f = f.copy()
    f[cut:] = 0.0
    y = np.real(np.fft.irfft(f, n))
    return _repair(y, n)


def _top_indices(c, m):
    m = min(m, c.size)
    if m <= 0:
        return np.zeros(0, dtype=int)
    if c.size > m:
        return np.argpartition(c, -m)[-m:]
    return np.argsort(c)[-m:]


def _add_coeff(h, p, sgn, D, n):
    z = n - 1
    D[z] += sgn * (1.0 - 2.0 * h[p])
    if p > 0:
        q = np.arange(p, dtype=int)
        D[n - 1 - p + q] += -sgn * h[q]
        D[n - 1 + p - q] += sgn * (1.0 - h[q])
    if p < n - 1:
        d = np.arange(1, n - p, dtype=int)
        D[n - 1 - d] += sgn * (1.0 - h[p + d])
        D[n - 1 + d] += -sgn * h[p + d]


def _pair_coeff(h, i, j, n):
    D = np.zeros(2 * n - 1, dtype=float)
    _add_coeff(h, i, 1.0, D, n)
    _add_coeff(h, j, -1.0, D, n)
    D *= (2.0 / float(n))
    return D


def _pair_delta_max(c, D, d, z, pos, neg, dx):
    base = c + d * D
    val = float(np.max(base))
    q = d * d * dx
    v = float(base[z] - 2.0 * q)
    if v > val:
        val = v
    v = float(base[pos] + q)
    if v > val:
        val = v
    v = float(base[neg] + q)
    if v > val:
        val = v
    return val

        else:

        old_i = float(h[i])

            noimp += 1

def _make_weights(c, top_m, tau):
    idx = _top_indices(c, top_m)
    mx = float(c[idx].max())
    z = (c[idx] - mx) / tau
    z -= float(z.max())
    w = np.exp(z)
    w /= w.sum()
    full = np.zeros(c.size, dtype=float)
    full[idx] = w
    return full


def _weight_obj(x, w, lam):
    c = _cvals(x)
    err = float(np.sum(x) - 0.5 * len(x))
    return float(np.dot(w, c) + lam * err * err)


def _weight_grad(x, w, lam):
    n = len(x)
    L = 2 * n - 1
    g = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    for k in range(L):
        wk = float(w[k])
        if wk <= 1e-18:
            continue
        sh = n - 1 - k
        f = wk * dx
        if sh >= 0:
            m = n - sh
            g[:m] += f * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= f * x[:m]
        else:
            sp = -sh
            g[sp:] += f * (1.0 - x[:-sp])
            g[:-sp] -= f * x[sp:]
    if lam != 0.0:
        err = float(np.sum(x) - 0.5 * n)
        g += 2.0 * lam * err
    return g


def _weighted_lbfgs(h, w, lam, maxiter, deadline):
    if time.time() > deadline - 10:
        return h
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _weight_obj,
            h,
            jac=_weight_grad,
            args=(w, lam),
            method="L-BFGS-B",
            bounds=bnds,
            options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-10},
        )
        return _repair(res.x, n)
    except Exception:
        return h


def _multi_weighted(h, deadline, rng):
    n = len(h)
    h = _repair(h, n).copy()
    bestc = float(_eval(h))
    besth = h.copy()

    configs = (
        (5, 1e-4, 320),
        (10, 1e-3, 260),
        (20, 1e-2, 180),
        (40, 5e-2, 120),
    )
    for top_m, tau, maxit in configs:
        if time.time() > deadline - 20:
            break
        curh = besth.copy()
        curc = bestc
        for _ in range(8):
            if time.time() > deadline - 15:

            w = _make_weights(_cvals(curh), top_m, tau)
            h2 = _weighted_lbfgs(curh, w, 5.0, maxit, deadline)
            c2 = float(_eval(h2))
            if c2 < curc + 1e-12:
                curh, curc = h2, c2
            else:

            if curc < bestc:
                bestc = curc
                besth = curh.copy()

    besth = _repair(besth, n)
    bestc = _eval(besth)
    return besth, bestc
