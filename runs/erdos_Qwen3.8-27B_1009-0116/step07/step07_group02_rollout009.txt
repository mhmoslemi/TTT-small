Actually, standard Nesterov accelerated gradient:
- y_k = x_k + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})
- x_{k+1} = proj(y_k - (1/L) * ∇f(y_k))
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2))/2

The step size 1/L where L is the Lipschitz constant of the gradient. For our problem, L is hard to compute exactly, but we can use backtracking.

from scipy.optimize import minimize
import time

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _grad_correct(h, w):
    """Correct gradient of sum_k w[k]*c_k."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        np_ = n - abs(sh)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[sh:]
        else:
            sp = -sh
            g[:np_] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:np_]
    return g

def _smooth_w(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w /= ws
    return w

def _topk_w(h, k):
    c = _corr(h)
    idx = np.argsort(c)[-k:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    return w

    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

    w = _smooth_w(x, alpha)
    return _grad_correct(x, w)

def _nesterov_pg(h, alpha, n_iter, deadline):
    """Nesterov accelerated projected gradient."""
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    step = 0.3
    
    for i in range(n_iter):
        if time.time() > deadline - 20:
            break
        
        # Nesterov momentum
        mu = (t - 1) / (t + 1)
        y = h_curr + mu * (h_curr - h_prev)
        y = _proj(y, n)
        
        w = _smooth_w(y, alpha)
        g = _grad_correct(y, w)
        
        h_new = _proj(h_curr - step * g, n)
        
        h_prev = h_curr.copy()
        h_curr = h_new
        t += 1
        step = 0.3 / (1 + 0.02 * i)
    
    return h_curr

def _lbfgs_opt(h, alpha, max_iter, deadline):

def _topk_descent(h, k, n_iter, deadline):
    """Descent using only top-k lags."""
    n = len(h)
    step = 0.05
    c5_cur = _eval(h)
    
    for i in range(n_iter):
        if time.time() > deadline - 10:
            break
        w = _topk_w(h, k)
        g = _grad_correct(h, w)
        
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_try = _proj(h - s * g, n)
            c5_try = _eval(h_try)
            if c5_try < c5_cur - 1e-16:
                h = h_try
                c5_cur = c5_try
                improved = True
                break
        
        if not improved:
            step *= 0.7
            if step < 1e-8:
                break
        else:
            step = min(0.1, step * 1.02)
    
    return h

def _local_search(h, rng, deadline, n_iter=50000):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    
    for i in range(n_iter):
        if time.time() > deadline - 2:
            break
        # Random 2-point swap
        a = rng.randint(n)
        b = (a + 1 + rng.randint(0, 5)) % n
        d = 1e-5 * rng.randn()
        na, nb = h[a] + d, h[b] - d
        if 0 <= na <= 1 and 0 <= nb <= 1:

            c5n = _eval(h)
            if c5n < c5c - 1e-19:

                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[a], h[b] = na - d, nb + d
    
    return best_h, best_c
