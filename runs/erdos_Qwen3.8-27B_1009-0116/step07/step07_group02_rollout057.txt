    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_c(h, k, n):
    """Gradient of the k-th correlation value w.r.t. h"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:

        g[:np_] += (1.0 - h[sh:])

        g[sp:] += (1.0 - h[:-sp])

    return g * dx

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_one(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_one(x, k, n)
    return g

def _pbsobj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pbsgrad(x, p):
    n = len(x)
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_one(x, k, n)
    return g

def _lbfgs(h, obj, jac, argr, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, args=argr, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxit, 'ftol': 1e-16,
                                             'gtol': 1e-12, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _active_grad_descent(h, n_active, n_steps, deadline):
    n = len(h)
    c5 = _eval(h)
    lr0 = 0.03
    for i in range(n_steps):
        if time.time() > deadline - 8:
            break
        cx = _cvals(h)
        idx = np.argsort(cx)[-n_active:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_one(h, k, n)
        g /= len(idx)
        lr = lr0 / (1.0 + 0.05 * i)
        h_new = _repair(h - lr * g, n)
        c5n = _eval(h_new)
        if c5n > c5 + 1e-13:
            lr *= 0.3
            h_new = _repair(h - lr * g, n)
            c5n = _eval(h_new)
        h = h_new
        c5 = c5n
    return h, c5

def _local_search(h, n_iter, deadline, seed=7):
    n = len(h)
    rng = np.random.RandomState(seed)

                best = h.copy()
                scale = min(scale * 1.3, 2e-3)

    n = len(h)
    m = n * f
    xo = np.arange(n)
    xn = np.linspace(0, n - 1, m)

    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()
