import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0, 1)
    return _proj(h, 0.5 * n)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(n - s)
        g[i1] += 1.0 - h[i1 + s]
        if s > 0:
            i2 = np.arange(s, n)
            g[i2] -= h[i2 - s]
    else:
        p = -s
        i1 = np.arange(p, n)
        g[i1] += 1.0 - h[i1 - p]
        i2 = np.arange(n - p)
        g[i2] -= h[i2 + p]
    return g * (2.0 / n)

def _subgrad_topk(h, k):
    c = _corr(h)
    n = len(h)
    top = np.argsort(c)[-k:]
    g = np.zeros(n)
    for kk in top:
        g += _grad_shift(h, int(kk) - (n - 1))
    return g / k

def _nesterov(h, k_top, n_iter, step_init, t0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step_init
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        g = _subgrad_topk(h, k_top)
        v = mu * v + (1.0 - mu) * g
        h_old = h.copy()
        improved = False
        st = step
        for _bt in range(12):
            h_new = _proj(h_old - st * v, 0.5 * n)
            f_new = _eval(h_new)
            if f_new <= f + 1e-18:
                h = h_new
                f = min(f, f_new)
                improved = True
                break
            st *= 0.55
        if improved:
            step = min(step * 1.06, step_init * 30.0)
        else:
            step *= 0.35
            if step < 1e-14:
                break
    return _repair(h, n), _eval(h)

def _lse_val(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max())) + 1e-300) / alpha

def _lse_grad(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    nx = len(x)
    g = np.zeros(nx)
    for kk in range(2 * nx - 1):
        if w[kk] > 1e-14:
            g += w[kk] * _grad_shift(x, int(kk) - (nx - 1))
    return g

def _cons_eq(x):
    return x.sum() - 0.5 * len(x)

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxit, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}
    try:
        res = minimize(_lse_val, h, args=(alpha, n), jac=_lse_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_coord(h, n_steps, rng, t0, t_end):
    n = len(h)
    best = h.copy()
    best_c = _eval(h)
    cur = h.copy()
    cur_c = best_c
    g_cached = None
    for it in range(n_steps):
        if time.time() > t_end:
            break
        if g_cached is None or it % 20 == 0:
            c_arr = _corr(cur)
            top_k = int(np.argmax(c_arr))
            s = top_k - (n - 1)
            g_cached = _grad_shift(cur, s) - np.mean(_grad_shift(cur, s))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 8e-5 * (1.0 if g_cached[i] < 0 else -1.0)
        cur[i] += d
        cur[j] -= d

                if c_new < best_c:
                    best_c = c_new
                    best = cur.copy()
            else:
                cur[i] -= d
                cur[j] += d
        else:
            cur[i] -= d
            cur[j] += d
    return best, best_c
