import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_mlgd(h, lags, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g * dx / len(lags)

def _grad_mgd(h, rounds, k_top, steps_init, momentum, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)

            break

        top_k = np.argsort(c)[-k_top:]

        improved = False
        for si in range(len(steps)):
            v = momentum * v + (1.0 - momentum) * g
            h_new = _project(h - steps[si] * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 8:
                    steps = steps[:8]
                break
            v = 0.5 * v
        if not improved:
            v *= 0.5
            steps = [s * 0.35 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _nesterov_ph(h, rounds, k_top, step0, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    y = h.copy()

        step = step0 / (1.0 + 0.03 * _r)
        y_new = _project(y - step * g, n * 0.5)

        h_new = y_new + (t_new - 1.0) / t_new * (y_new - h)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)

            stall = 0
        else:

            stall += 1
            if stall > 40:

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _obj_lse(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = beta * (c - m)
    sc = np.clip(sc, -40, 40)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / beta

def _grad_lse(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = beta * (c - m)
    sc = np.clip(sc, -40, 40)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        wk = w[k] * (2.0 / n)

        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:

        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g
