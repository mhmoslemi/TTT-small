import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_comp(h, j, s_vals, n):
    js = j + s_vals
    jms = j - s_vals
    m1 = (js >= 0) & (js < n)
    m2 = (jms >= 0) & (jms < n)
    g = np.zeros(2 * n - 1)
    g[m1] = 1.0 - h[js[m1]]
    g[m2] -= h[jms[m2]]
    return g

def _coord_1d(c, D, hj, hl):
    lo = max(-hj, hl - 1.0)
    hi = min(1.0 - hj, hl)
    if lo >= hi - 1e-15:
        return 0.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        val = c + mid * D
        k = np.argmax(val)
        g = D[k]
        if g > 1e-14:
            hi = mid
        elif g < -1e-14:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def _coord_descent(h, c, s_vals, nsteps, rng, deadline):
    n = len(h)
    best_c = c.max() * 2.0 / n
    best_h = h.copy()
    for step in range(nsteps):
        if time.time() > deadline:
            break
        j = rng.randint(n)
        l = (j + 1 + rng.randint(0, n - 2)) % n
        gj = _grad_comp(h, j, s_vals, n)
        gl = _grad_comp(h, l, s_vals, n)
        D = gj - gl
        delta = _coord_1d(c, D, h[j], h[l])
        if abs(delta) < 1e-15:
            continue
        c += delta * D
        h[j] += delta
        h[l] -= delta
        cm = c.max() * 2.0 / n
        if cm < best_c - 1e-8:
            best_c = cm
            best_h = h.copy()
    return h, c, best_h

def _smax_obj(h, alpha, topk):
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]

        sh = s_vals[idx[ii]]
        wk = w[ii] * dx
        # dc/dh[j] = (1-h[j+sh])*[0<=j+sh<n] - h[j-sh]*[0<=j-sh<n]
        for j in range(n):
            val = 0.0
            if 0 <= j + sh < n:
                val += (1.0 - h[j + sh])
            if 0 <= j - sh < n:
                val -= h[j - sh]
            g[j] += wk * val

def _grad_topk_vec(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    s_vals = np.arange(-(n-1), n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        sh = s_vals[idx[ii]]
        wk = w[ii] * dx
        js = np.arange(n)
        m1 = (js + sh >= 0) & (js + sh < n)
        m2 = (js - sh >= 0) & (js - sh < n)
        tmp = np.zeros(n)
        tmp[m1] = 1.0 - h[js[m1] + sh]
        tmp[m2] -= h[js[m2] - sh]
        g += wk * tmp
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk_vec(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 3)) % nc
        d = 1e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
