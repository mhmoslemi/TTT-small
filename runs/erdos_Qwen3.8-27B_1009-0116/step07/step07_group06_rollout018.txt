    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-m_top:]
    
    g = np.zeros(n)
    for k in top_k:
        sh = k - (n - 1)
        if sh >= 0:
            # (1-h[p+sh]) valid for p in [0, n-1-sh]
            # h[p-sh] valid for p in [sh, n-1] (if sh>0; if sh=0, it's h[p])
            if sh > 0:
                g[:n-sh] += (1.0 - h[sh:])
                g[sh:] -= h[:n-sh]
            else:  # sh == 0
                g += (1.0 - h) - h  # = 1 - 2h
                # Actually: (1-h[p]) - h[p] = 1-2h[p]
        else:
            sp = -sh
            # (1-h[p-sp]) valid for p in [sp, n-1]
            # h[p+sp] valid for p in [0, n-1-sp]
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    
    g *= dx / m_top
    h_new = h - alpha * g
    h_new = _proj_to_sum(h_new, n * 0.5)
    return h_new

            idx_f = i + k - (n - 1)
            if 0 <= idx_f < n:
                g_k[ki] += (1.0 - h[idx_f])
            idx_b = i - k + (n - 1)
            if 0 <= idx_b < n:
                g_k[ki] -= h[idx_b]
        
        # Gradient from h[j] decreasing by 1 (negate)
        for k in range(lo_j, hi_j + 1):
            kj = k - lo
            idx_f = j + k - (n - 1)
            if 0 <= idx_f < n:
                g_k[kj] -= (1.0 - h[idx_f])
            idx_b = j - k + (n - 1)
            if 0 <= idx_b < n:
                g_k[kj] += h[idx_b]
        
        # Quadratic terms
        kn1 = n - 1 - lo
        if 0 <= kn1 < len(f_k):
            f_k[kn1] -= 2.0
        if 0 <= k1 < 2 * n - 1:
            kk1 = k1 - lo
            if 0 <= kk1 < len(f_k):
                f_k[kk1] += 1.0
        if 0 <= k2 < 2 * n - 1 and k2 != k1:
            kk2 = k2 - lo
            if 0 <= kk2 < len(f_k):
                f_k[kk2] += 1.0
        
        # Bisection
        for _ in range(bisect_iters):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c_sub + d_mid * g_k + d_mid * d_mid * f_k
            c5_new = max(vals.max(), c5_cur)
            kmax = np.argmax(vals)
            if vals[kmax] < c5_cur - 1e-15:
                # Max is outside k_range, no useful gradient
                sg = 0.0
            else:
                sg = g_k[kmax] + 2.0 * d_mid * f_k[kmax]
            
            if sg > 1e-12:
                d_hi = d_mid
            elif sg < -1e-12:
                d_lo = d_mid
            else:
                break
        
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-12:
            stable += 1
            continue
        
        h[i] += d_opt
        h[j] -= d_opt
        c[k_range] += d_opt * g_k + d_opt * d_opt * f_k
        
        c5_new = c.max()
        if c5_new < c5_cur - 1e-13:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c[k_range] -= d_opt * g_k + d_opt * d_opt * f_k
            stable += 1
    
    return best_h, best_c
