import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nbfgs(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'gtol': 1e-12})
        return _repair(r.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.83
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.4
            if step < step0 * 0.01:
                step = step0
            h = _proj(h - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _active_equalize(h, n, deadline, nrounds=200):
    """Target the worst lags specifically."""
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for rnd in range(nrounds):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        # Find top active lags
        n_act = min(8, len(c))
        idx = np.argsort(c)[-n_act:]
        active_max = c[idx]
        # Compute gradient for each active lag and average
        g = np.zeros(n)
        dx = 2.0 / n
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += dx * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= dx * h[:-sh]
            else:
                sp = -sh
                g[sp:] += dx * (1.0 - h[:-sp])
                g[:n - sp] -= dx * h[sp:]
        g /= len(idx)
        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            continue
        g /= gn
        # Try several step sizes
        best_step = 0
        best_c = c5c
        for ss in [1e-5, 5e-6, 1e-6, 5e-7, 1e-7, 5e-8]:
            hn = _proj(h - ss * g, n * 0.5)
            cn = _eval(hn)
            if cn < best_c:
                best_c = cn
                best_step = ss
        if best_step > 0:
            h = _proj(h - best_step * g, n * 0.5)
            c5c = _eval(h)
            if c5c < bc:
                bc = c5c
                bh = h.copy()
        else:
            # No improvement, try a random perturbation to escape
            d = 0.001 * np.random.randn(n)
            d -= d.mean()
            hn = _repair(h + d, n)
            cn = _eval(hn)
            if cn < c5c * 1.001:  # allow slight worsening
                h = hn
                c5c = cn
    return bh, bc

def _local_hill(h, n, rng, deadline, niter):
    """Multi-scale local hill climbing."""
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        scale = 1e-5 * (1 + 0.5 * np.sin(it * 0.01))
        # Pick 2-4 random indices
        ns = rng.randint(2, 5)
        idxs = rng.choice(n, ns, replace=False)
        d = scale * rng.randn(ns)
        d -= d.mean()
        hn = h.copy()
        ok = True
        for ii, ix in enumerate(idxs):
            val = hn[ix] + d[ii]
            if val < 0 or val > 1:
                ok = False
                break
            hn[ix] = val
        if not ok:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < c5c - 1e-18:
            h = hn
            c5c = cn
            if cn < bc:
                bc = cn
                bh = h.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nbfgs(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'gtol': 1e-14, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.83
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.4
            if step < step0 * 0.01:
                step = step0
            h = _proj(h - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _active_equalize(h, n, deadline, nrounds):
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    dx = 2.0 / n
    for rnd in range(nrounds):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        n_act = min(10, len(c))
        idx = np.argsort(c)[-n_act:]
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += dx * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= dx * h[:-sh]
            else:
                sp = -sh
                g[sp:] += dx * (1.0 - h[:-sp])
                g[:n - sp] -= dx * h[sp:]
        g /= len(idx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            continue
        g /= gn
        best_step = 0
        best_c = c5c
        for ss in [2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7]:
            hn = _proj(h - ss * g, n * 0.5)
            cn = _eval(hn)
            if cn < best_c:
                best_c = cn
                best_step = ss
        if best_step > 0:
            h = _proj(h - best_step * g, n * 0.5)
            c5c = _eval(h)
            if c5c < bc:
                bc = c5c
                bh = h.copy()
        else:
            d = 5e-4 * np.random.randn(n)
            d -= d.mean()
            hn = _repair(h + d, n)
            if _eval(hn) < c5c + 1e-6:
                h = hn
                c5c = _eval(h)
    return bh, bc

def _local_hill(h, n, rng, deadline, niter):
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        scale = 3e-6 * (1.0 + 0.5 * np.sin(it * 0.007))
        ns = rng.randint(2, 5)
        idxs = rng.choice(n, ns, replace=False)
        d = scale * rng.randn(ns)
        d -= d.mean()
        hn = h.copy()
        ok = True
        for ii in range(ns):
            val = hn[idxs[ii]] + d[ii]
            if val < 0.0 or val > 1.0:
                ok = False
                break
            hn[idxs[ii]] = val
        if not ok:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < c5c - 1e-18:
            h = hn
            c5c = cn
            if cn < bc:
                bc = cn
                bh = h.copy()
    return bh, bc

def _perturb_restart(h, n, rng, alpha, topk, steps, deadline):
    c5c = _eval(h)
    for _ in range(3):
        if time.time() > deadline:
            break
        d = 0.003 * rng.randn(n)
        d -= d.mean()
        hp = _repair(h + d, n)
        cn = _eval(hp)
        if cn < c5c + 5e-4:
            hp = _nesterov(hp, alpha, topk, steps, 0.001, deadline)
            cn = _eval(hp)
            if cn < c5c:
                h = hp
                c5c = cn
    return h, c5c
