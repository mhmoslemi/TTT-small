    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})

def _topk_descent(h, k_top, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for r in range(max_rounds):
        if time.time() > deadline - 10:

        c = _cvals(h)
        top = np.argsort(c)[-k_top:]

                np_ = n - sh

    # Project h - g onto {0≤x≤1, sum(x)=n/2}
    x = h - g
    return _repair(x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 25:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _topk_descent(h, k_top, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for r in range(max_rounds):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        top = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for kk in top:
            sh = (n - 1) - kk
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / k_top
        improved = False
        for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _sa_search(h, n_iter, t_start, t_end, deadline, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for i in range(n_iter):
        if time.time() > deadline - 3:
            break
        T = t_start * (t_end / t_start) ** (float(i) / max(n_iter, 1))
        npts = rng.randint(2, 5)
        idx = rng.choice(n, size=npts, replace=False)
        d = rng.randn(npts)
        d -= d.mean()
        scale = rng.uniform(0.002, 0.06)
        h_new = h.copy()
        h_new[idx] += scale * d
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)

        if delta < 0:

            c5c = c5n

            h = h_new
            c5c = c5n
    return best_h, best_c
