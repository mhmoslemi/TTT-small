    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lag_grad_dir(h, k):
    n = len(h)
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 30:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxcor': 50, 'maxls': 80})
        return _repair(res.x, n)
    except Exception:
        return h

def _face_search(h, n_active, max_rounds, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 20:
            break
        c = _cvals(best_h)
        top_idx = np.argsort(c)[-n_active:]
        g = np.zeros(n)
        for k in top_idx:
            g += _lag_grad_dir(best_h, k)
        g /= n_active
        g -= g.mean()
        improved = False
        for step in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 5e-4, 1e-4, 5e-5, 1e-5]:
            h_new = _repair(best_h - step * g, n)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5n
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def _nesterov(h, alpha, lr, mu, n_iters, deadline):
    n = len(h)
    x = h.copy()
    xp = h.copy()
    best_x = h.copy()
    best_c5 = _eval(h)
    for i in range(n_iters):
        if time.time() > deadline - 15:
            break
        y = _repair(x + mu * (x - xp), n)
        g = _smooth_grad(y, alpha)
        lr_i = lr / (1.0 + 0.03 * i)
        x_new = _repair(x - lr_i * g, n)
        c5n = _eval(x_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_x = x_new.copy()
        xp = x
        x = x_new
    return best_x, best_c5

def _sys_local(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    step = 1e-4
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        if step < 1e-9:
            break
        i = rng.randint(n)
        if rng.rand() < 0.5:
            j = (i + 1 + rng.randint(0, 5)) % n
        else:
            j = rng.randint(n)
        if j == i:
            continue
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                step = min(step * 1.05, 5e-4)
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                step *= 0.75
    return best_h, best_c

def _tri_local(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    step = 5e-5
    for _ in range(n_iter):
        if time.time() > deadline - 3:

        if step < 1e-9:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 10)) % n
        k = (i + rng.randint(10, 30)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        dk = -(d1 + d2)
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] + dk

                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk - dk

def _perturb_restart(h, rng, n_pert, deadline):
