from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, pen, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - n * 0.5) ** 2

def _smooth_grad(x, alpha, pen, n, top_k=30):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m_top = c_top.max()
    s = np.clip(alpha * (c_top - m_top), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g

def _slsqp_pen(h, alpha, pen, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, pen, n), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': ftol,
                                              'gtol': 1e-12, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _adam_proj(h, alpha, n_steps, lr, top_k):
    n = len(h)
    target = n * 0.5
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    cur_c = _eval(h)
    for t in range(1, n_steps + 1):
        g = _smooth_grad(h, alpha, 0.0, n, top_k)
        g -= g.mean()
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj(h - step, target)
        c_new = _eval(h_new)
        if c_new < cur_c:
            cur_c = c_new
        else:
            h_new = h
    return _repair(h, n), cur_c

def _local_search(h, n_iter, rng, t0, t_end, base_step, use_sa):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    step = base_step
    T = c5cur * 0.005
    stagnation = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 500 == 0:
            T = max(c5cur * 0.001, 1e-7)
        mode = rng.randint(3)
        if mode == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = step * rng.randn()
            if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                h[i] += d
                h[j] -= d
                c5n = _eval(h)
                if c5n < c5cur:
                    c5cur = c5n
                    step = min(step * 1.03, base_step * 8)
                    stagnation = 0
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
                    if use_sa and rng.rand() < np.exp(-(c5n - c5cur) / T):
                        h[i] += d
                        h[j] -= d
                        c5cur = c5n
                    stagnation += 1
        elif mode == 1:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 8)) % n
            k = (i + 9 + rng.randint(0, 12)) % n
            d1 = step * rng.randn()
            d2 = step * rng.randn()
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h[i], h[j], h[k] = ni, nj, nk
                c5n = _eval(h)
                if c5n < c5cur:
                    c5cur = c5n
                    step = min(step * 1.03, base_step * 8)
                    stagnation = 0
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
                    if use_sa and rng.rand() < np.exp(-(c5n - c5cur) / T):
                        h[i], h[j], h[k] = ni, nj, nk
                        c5cur = c5n
                    stagnation += 1
        else:
            blen = rng.randint(3, 9)
            i0 = rng.randint(n - blen)
            block = h[i0:i0 + blen].copy()
            pert = rng.randn(blen)
            pert -= pert.mean()
            pert *= step * 0.5
            new_block = block + pert
            if np.all(new_block >= 0) and np.all(new_block <= 1):
                h[i0:i0 + blen] = new_block
                c5n = _eval(h)
                if c5n < c5cur:
                    c5cur = c5n
                    step = min(step * 1.05, base_step * 10)
                    stagnation = 0
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i0:i0 + blen] = block
                    if use_sa and rng.rand() < np.exp(-(c5n - c5cur) / T):
                        h[i0:i0 + blen] = new_block
                        c5cur = c5n
                    stagnation += 1
        if stagnation > 20000:
            step = max(step * 0.7, base_step * 0.1)
            stagnation = 0
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    out = np.empty(len(h) * factor)
    for i in range(len(h)):
        out[i * factor:(i + 1) * factor] = h[i]
    return out
