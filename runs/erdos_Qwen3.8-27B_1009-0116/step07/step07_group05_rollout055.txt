Combined: max(-h[i], h[j]-1) ≤ δ ≤ min(1-h[i], h[j])

And we want to minimize max_k (corr[k_idx] + dx*δ*b[k_idx]) over this range.

This is a 1D convex problem. I'll use binary search:
- The subgradient at δ is: dx * b[k*] where k* is an index achieving the max
- Actually the subgradient is a set: {dx * Σ α_k * b[k] : α is a distribution on argmax indices}
- 0 is in the subgradient iff min_{k in argmax} b[k] ≤ 0 ≤ max_{k in argmax} b[k]

Binary search: at each δ, find max_k c_k(δ) and the set of k's achieving it. If all b[k] for those k are > 0, increase δ. If all < 0, decrease δ.

Simpler: just do ternary search (the function is convex, so ternary works):
Evaluate at 1/3 and 2/3 points, discard the worse side, repeat.

Or even simpler: since the function is piecewise linear and convex, the minimum is at a "kink" point. We can find it by checking all pairs of lines... but that's O(n²).

```python
def _bisect_pair(corr, b, dx, delta_lo, delta_hi, n_iter=40):
    """Find δ that minimizes max_k (corr[k] + dx*δ*b[k])"""
    lo = delta_lo
    hi = delta_hi
    for _ in range(n_iter):
        mid = (lo + hi) / 2
        vals = corr + dx * mid * b
        max_val = vals.max()
        # Subgradient: b[k] for k where vals[k] ≈ max_val
        active = np.where(vals >= max_val - 1e-12)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        sg_min = b[active].min()
        sg_max = b[active].max()
        if sg_min <= 0 and sg_max >= 0:
            break
        if sg_min > 0:
            hi = mid  # all active slopes positive, need to decrease δ
        else:
            lo = mid  # all active slopes negative, need to increase δ
    return (lo + hi) / 2
```

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, a):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sc = np.clip(a * (c - m), -50, 0)
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wkv = w[k]
        if sh >= 0:
            if sh == 0:
                g += wkv * (1.0 - 2.0 * h)
            else:
                g[sh:] += wkv * (1.0 - h[:n - sh])
                g[:n - sh] -= wkv * h[sh:]
        else:
            p = -sh
            g[:n - p] += wkv * (1.0 - h[p:])
            g[p:] -= wkv * h[:n - p]
    return g * dx

def _fista(h0, n, a, n_iter, check_fn, L):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_c = _eval(x)
    for _k in range(n_iter):
        if check_fn():
            break
        g = _grad_smooth(y, a)
        x_new = _proj(y - g / L, n * 0.5)
        c5 = _eval(x_new)
        if c5 < best_c:
            best_c = c5
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = _proj(y_new, n * 0.5)
        x = x_new
        t = t_new
    return x, best_c

def _dc_dh_vec(h, i, n):
    """Compute dc_sh/dh[i] for all shifts sh in [-(n-1), n-1], returned as array of length 2n-1"""
    b = np.zeros(2 * n - 1)
    for k_idx in range(2 * n - 1):
        sh = (n - 1) - k_idx
        if sh >= 0:
            if sh <= i < n:
                b[k_idx] += (1.0 - h[i - sh])
            if 0 <= i <= n - 1 - sh:
                b[k_idx] -= h[i + sh]
        else:
            p = -sh
            if 0 <= i <= n - 1 - p:
                b[k_idx] += (1.0 - h[i + p])
            if p <= i < n:
                b[k_idx] -= h[i - p]
    return b

def _bisect_delta(corr, b, dx, lo, hi):
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = corr + dx * mid * b
        mv = vals.max()
        active = vals >= (mv - 1e-14)
        if not active.any():
            active[np.argmax(vals)] = True
        sg_lo = b[active].min()
        sg_hi = b[active].max()
        if sg_lo <= 0 and sg_hi >= 0:
            return mid
        if sg_lo > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def _pair_cd(h0, n, n_pairs, rng, check_fn):
    h = h0.copy()
    corr = _corr(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    dx = 2.0 / n
    for _p in range(n_pairs):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        if j == i:
            j = (i + 1) % n
        bi = _dc_dh_vec(h, i, n)
        bj = _dc_dh_vec(h, j, n)
        b = bi - bj
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dhi - dlo < 1e-12:
            continue
        d_opt = _bisect_delta(corr, b, dx, dlo, dhi)
        if abs(d_opt) < 1e-14:
            continue
        h[i] += d_opt
        h[j] -= d_opt
        corr += dx * d_opt * b
        c5_new = float(corr.max())
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def _fine_search(h0, n, n_try, rng, check_fn):
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_try):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < best_c - 1e-18:
                best_c = c
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 20

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: FISTA with moderate smoothing
    for a, L, iters in [(50, 10.0, 400), (150, 6.0, 300), (500, 3.0, 200), (2000, 1.5, 150)]:
        if _bc():
            break
        h2, c52 = _fista(best_h, n, a, iters, _bc, L)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Paired coordinate descent
    if _el() < budget_s - 60:
        nt = min(20000, int((budget_s - 60 - _el()) / 0.005))
        best_h, best_c5 = _pair_cd(best_h, n, nt, rng, _bc)

    # Phase 3: FISTA again (sharper)
    if _el() < budget_s - 80:
        for a, L, iters in [(1000, 2.0, 200), (5000, 1.0, 100)]:
            if _bc():
                break
            h2, c52 = _fista(best_h, n, a, iters, _bc, L)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2

    # Phase 4: More paired CD
    if _el() < budget_s - 40:
        nt = min(30000, int((budget_s - 40 - _el()) / 0.003))
        best_h, best_c5 = _pair_cd(best_h, n, nt, rng, _bc)

    # Phase 5: Try shifted/noised variants with quick FISTA
    for shift in [1, 2, -1]:
        if _bc():
            break
        h_sh = np.roll(best_h, shift).copy()
        h_sh += 0.001 * rng.randn(n)
        h_sh = _repair(h_sh, n)
        h2, c52 = _fista(h_sh, n, 500, 100, _bc, 3.0)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
            best_h, best_c5 = _pair_cd(best_h, n, 5000, rng, _bc)

    # Phase 6: Final paired CD
    if _el() < budget_s - 30:
        nt = min(20000, int((budget_s - 30 - _el()) / 0.003))
        best_h, best_c5 = _pair_cd(best_h, n, nt, rng, _bc)

    # Phase 7: Fine random search
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.002), 200000)
        best_h, best_c5 = _fine_search(best_h, n, nt, rng, _bc)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
