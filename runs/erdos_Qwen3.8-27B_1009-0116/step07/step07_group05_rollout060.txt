import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _project_sum(h, target):
    """Project h onto {x : 0<=x<=1, sum(x)=target}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_single_lag(h, k):
    """Gradient of c_k = sum_i h[i+k]*(1-h[i]) (or shifted) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    dx = 2.0 / n
    if sh >= 0:
        # c_k = sum_{i=0}^{n-1-sh} h[i+sh] * (1-h[i])
        # d(c_k)/d(h[i]) = -h[i+sh] for i < n-sh (appears in 1-h[i])
        # d(c_k)/d(h[i]) = (1-h[i-sh]) for i >= sh (appears in h[i])
        g[sh:] += (1.0 - h[:n - sh])  # h appears as first arg
        g[:n - sh] -= h[sh:]         # h appears as second arg (in 1-h)
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])  # from 1-h part
        g[sp:] -= h[:n - sp]          # from h part
    return g * dx

def _grad_topk(h, k_top):
    n = len(h)
    c = _correl(h)
    top_idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in top_idx:
        g += _grad_single_lag(h, k)
    return g / k_top, top_idx

def _pgd_step(h, g, step):
    """Projected gradient step with sum constraint"""
    n = len(h)
    h_new = h - step * g
    h_new = _project_sum(h_new, n * 0.5)
    return h_new

def _run_pgd(h, budget_fn, max_iter, k_top, step0, step_decay):
    """Projected gradient descent with top-k lag focus"""
    best_h = h.copy()
    best_c = _eval(best_h)
    n = len(h)
    step = step0
    stall = 0
    for it in range(max_iter):
        if budget_fn():
            break
        g, top_idx = _grad_topk(h, k_top)
        # Backtracking line search
        improved = False
        for inner in range(20):
            h_new = _pgd_step(h, g, step * (0.5 ** inner))
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
                h = h_new
                improved = True
                step = min(step * 1.3, 0.5)
                break
        if not improved:
            step *= 0.5
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return best_h, best_c

def _exchange_eval(h, i, j, delta):
    """Evaluate after exchange h[i]+=delta, h[j]-=delta"""
    val_i = h[i] + delta
    val_j = h[j] - delta
    if val_i < 0 or val_i > 1 or val_j < 0 or val_j > 1:
        return 1e10
    h[i] = val_i
    h[j] = val_j
    c = _eval(h)
    h[i] -= delta
    h[j] += delta
    return c

def _optimize_pair(h, i, j, c5c, rng):
    """Find optimal delta for exchange between i and j"""
    # Bisection on delta
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return h, c5c
    best_delta = 0.0
    best_c = c5c
    for _ in range(30):
        mid = (lo + hi) * 0.5
        h[i] += mid
        h[j] -= mid
        c = _eval(h)
        h[i] -= mid
        h[j] += mid
        if c < best_c:
            best_c = c
            best_delta = mid
        # Subgradient direction: if c increased, we need to check
        # Since max of linear functions, use secant-like approach
        # Simplified: just check midpoint
        lo2 = (lo + hi) * 0.25
        hi2 = (lo + hi) * 0.75
        h[i] += lo2
        h[j] -= lo2
        c_lo = _eval(h)
        h[i] -= lo2
        h[j] += lo2
        h[i] += hi2
        h[j] -= hi2
        c_hi = _eval(h)
        h[i] -= hi2
        h[j] += hi2
        if c_lo <= c_hi:
            hi = mid
        else:
            lo = mid
    if abs(best_delta) > 1e-15:
        h[i] += best_delta
        h[j] -= best_delta
    return h, best_c

def _block_perturb(h, rng, block_size, c5c, budget_fn, n_tries):
    """Perturb a random block and keep if improved"""
    n = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        if budget_fn():
            break
        i = rng.randint(0, n - block_size)
        orig_block = h[i:i+block_size].copy()
        # Perturb block
        h[i:i+block_size] += 0.01 * rng.randn(block_size)
        h[i:i+block_size] = _project_sum(h[i:i+block_size], orig_block.sum())
        c = _eval(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        else:
            h[i:i+block_size] = orig_block
    return best_h, best_c

def _grad_k(h, k):
    """Gradient of output[k] = sum_j h[j]*(1-h[k-j]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    # k-m must be in [0, n-1], so m is in [k-(n-1), k] clipped to [0, n-1]
    m_lo = max(0, k - n + 1)
    m_hi = min(n - 1, k)
    if m_lo > m_hi:

    m = np.arange(m_lo, m_hi + 1)
    j = k - m  # k-m values, all in [0, n-1] by construction
    g[m] = (1.0 - 2.0 * h[j]) * (2.0 / n)  # multiply by dx
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(np.max(_correl(h)))

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k):

    g = np.zeros(n)
    m_lo = max(0, k - n + 1)

    j = k - m
    dx = 2.0 / n
    g[m] = (1.0 - 2.0 * h[j]) * dx
