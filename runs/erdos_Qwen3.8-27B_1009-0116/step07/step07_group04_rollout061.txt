def _coord_descent(h, n_sweeps, t0, deadline):

        for i in order:
            if time.time() > deadline:
                break

            k = np.arange(2 * n - 1)

            a_idx = i + k - (n - 1)

            d[mask_v] -= h[a_idx[mask_v]]

            # d[n-1] currently = (1-h[i]) - h[i] = 1 - 2*h[i]
            # The actual c[n-1] change is: t*(1-t)*dx - old_h_i*(1-old_h_i)*dx + t*(d_linear_other)

            # Solve: min_t max_k (c_curr[k] + t*d[k]) for t in [0,1]

            lo, hi = 0.0, 1.0

                    lo = mid
                else:

    c5_final = _eval(h)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h.copy()
    return h_best, c5_best

                d[m1] += (1.0 - h[j_a[m1]])

                d[m2] -= h[a_idx[m2]]
                old_t = h[idx]
                lo, hi = 0.0, 1.0
                for _ in range(40):
                    mid = (lo + hi) * 0.5
                    kmax = np.argmax(c_curr + (mid - old_t) * d)
                    sl = d[kmax]
                    if sl > 1e-14:
                        hi = mid
                    elif sl < -1e-14:
                        lo = mid
                    else:
                        break
                t_new = (lo + hi) * 0.5
                h[idx] = t_new
                c_curr += (t_new - old_t) * d
    c5_final = _eval(h)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    deadline = t0 + budget_s

    # Phase 1: Nesterov warm-up (moderate alpha)
    h1, c1 = _nesterov(best_h.copy(), 500, 200, 0.008, t0 + 10)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()
    h2, c2 = _nesterov(best_h.copy(), 5000, 300, 0.003, t0 + 25)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 2: Coordinate descent (primary improvement)
    h3, c3 = _coord_descent(best_h.copy(), 40, t0 + 250)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 3: SLSQP with smooth max
    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline - 120:
            break
        h_c = _slsqp_opt(best_h.copy(), alpha, 600, deadline - 80)
        c_c = _eval(h_c)
        if c_c < best_c5:
            best_c5, best_h = c_c, h_c.copy()

    # Phase 4: More coordinate descent after SLSQP
    if time.time() < deadline - 60:
        h4, c4 = _coord_descent(best_h.copy(), 40, deadline - 30)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Block coordinate descent (adjacent pairs)
    if time.time() < deadline - 40:
        h5, c5 = _block_coord_descent(best_h.copy(), 30, deadline - 15)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Final coordinate polish
    if time.time() < deadline - 20:
        h6, c6 = _coord_descent(best_h.copy(), 20, deadline - 8)
        if c6 < best_c5:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Ultra-fine local search
    if time.time() < deadline - 3:
        n_iter = min(int((deadline - 3 - time.time()) / 0.002), 60000)
        h7, c7 = _local_search(best_h.copy(), n_iter, rng, deadline - 0.5)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
