        h2, c52 = _momentum_mlgd(best_h.copy(), n0, 350, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: SLSQP with soft-max at multiple temperatures
    if _el() < budget_s - 200:
        n = n0
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        for alpha in [0.08, 0.04, 0.02]:
            if _bc():
                break
            try:
                res = minimize(
                    lambda x: _soft_obj(x, n, alpha),
                    best_h,
                    jac=lambda x: _soft_grad(x, n, alpha),
                    method='SLSQP', bounds=bnds, constraints=cons,
                    options={'maxiter': 400, 'ftol': 1e-15})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 5: Block local search
    if _el() < budget_s - 50:
        for bs in [4, 6, 3]:
            if _bc():
                break
            nt = min(int((budget_s - 50 - _el()) / 0.004), 40000)
            best_h, best_c5 = _block_search(best_h, nt, rng, best_c5, bs, _bc)

    # Phase 6: Final momentum polish
    for _ in range(3):
        if _bc():
            break
        h2, c52 = _momentum_mlgd(best_h.copy(), n0, 250, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: Ultra-fine 2-point search
    if _el() < budget_s - 5:
        nc = n0
        for _ in range(min(int((budget_s - 5 - _el()) / 0.0008), 400000)):
            if _el() > budget_s - 5:
                break
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 5)) % nc
            d = 1.5e-6 * rng.randn()
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                c5n = _eval(best_h)
                if c5n < best_c5:
                    best_c5 = c5n
                else:
                    best_h[i], best_h[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
