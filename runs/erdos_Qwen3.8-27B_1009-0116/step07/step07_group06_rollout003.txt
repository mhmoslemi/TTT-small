import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad_topk(h, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-topk:]
    g = np.zeros(n)
    for k in top_idx:
        d = k - (n - 1)
        if d == 0:
            g += (1.0 - 2.0 * h) * dx
        elif d > 0:
            g[:n-d] += (1.0 - h[d:]) * dx
            g[d:] -= h[:-d] * dx
        else:
            dp = -d
            g[dp:] += (1.0 - h[:-dp]) * dx
            g[:n-dp] -= h[dp:] * dx
    g /= topk
    return g, c

def _subgrad_descent(h, rng, deadline, topk_start=3):
    n = len(h)
    c5_cur = _eval(h)
    best_h, best_c = h.copy(), c5_cur
    topk = topk_start
    stall = 0
    for it in range(30000):
        if time.time() > deadline:
            break
        g, c = _subgrad_topk(h, topk)
        c5_cur = float(c.max())
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = min(0.08, 1.5 / gn)
        accepted = False
        for mult in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.02]:
            h_new = _proj_sum(h - mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                accepted = True
                break
        if accepted:
            stall = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            stall += 1
            if stall % 200 == 0:
                topk = min(topk + 2, 30)
            if stall > 2000:
                break
        if it % 3000 == 2999:
            kick = 0.02 * rng.randn(n)
            hk = _proj_sum(h + kick, n * 0.5)
            ck = _eval(hk)
            if ck < c5_cur:
                h = hk
                c5_cur = ck
                stall = 0

def _grad_diff(h, i, j, n):
    dx = 2.0 / n
    m = 2 * n - 1
    g = np.zeros(m)
    for pos in (i, j):
        lo = max(0, n - 1 - (pos))
        for d in range(-(n-1), n):
            k = n - 1 + d
            contrib = 0.0
            if 0 <= pos + d < n:
                contrib += 1.0 - h[pos + d]
            if d != 0 and 0 <= pos - d < n:
                contrib -= h[pos - d]
            if pos == i:
                g[k] += contrib * dx
            else:
                g[k] -= contrib * dx
    return g

def _interpolate_to_n(h, new_n):

        return h.copy()
    x_old = np.linspace(0, 2.0, n)
