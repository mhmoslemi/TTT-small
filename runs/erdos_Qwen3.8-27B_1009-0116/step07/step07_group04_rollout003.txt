def _subgrad_topm(h, m, dx):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * dx
    # Get top m shifts
    idx = np.argsort(c)[-m:]
    vals = c[idx]
    # Weight by strength (exp or just equal)
    w = np.ones(m) / m
    g = np.zeros(n)
    for t in range(m):
        k = idx[t]
        s = k - (n-1)
        if s >= 0:
            g[0:n-s] += w[t] * (1-h[s:n]) * dx
            g[s:n] -= w[t] * h[0:n-s] * dx
        else:
            p = -s
            g[p:n] += w[t] * (1-h[0:n-p]) * dx
            g[0:n-p] -= w[t] * h[p:n] * dx
    return g

            break
        c = np.correlate(h, 1.0-h, mode='full') * dx
        c_max = c.max()
        if c_max < c_star:
            c_star = c_max
            t0_iter = 1.0  # reset step size when improving
        
        # Active set: shifts within eps of max
        active = np.where(c > c_max - eps)[0]
        if len(active) == 0:
            active = np.argsort(c)[-3:]
        if len(active) > 50:
            active = np.argsort(c)[-50:]
        
        # Subgradient: average of gradients at active shifts
        g = np.zeros(n)
        for k in active:
            s = k - (n-1)
            if s >= 0:
                g[0:n-s] += (1-h[s:n]) * dx
                g[s:n] -= h[0:n-s] * dx
            else:
                p = -s
                g[p:n] += (1-h[0:n-p]) * dx
                g[0:n-p] -= h[p:n] * dx
        g /= len(active)
        
        # Polyak step
        norm_sq = max(g @ g, 1e-12)
        step = (c_star - c_max) / norm_sq

    s = h.sum()
    if abs(s - target) > 1e-9:

        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

        for k in active:
            s = k - (n - 1)
            g += _grad_at_shift(h, s, dx)
        g /= len(active)

        norm_sq = max(float(g @ g), 1e-15)
        step = (c_star - c_cur_max) / norm_sq

        step = max(step, 1e-9)
        
        h_new = _proj_sum(h - step * g, target)
        c_new = _eval(h_new)
        
        if c_new < c_cur_max:
            h = h_new
            c_max = c_new
        else:
            # Backtracking
            for _ in range(10):
                step *= 0.5
                h_new = _proj_sum(h - step * g, target)
                c_new = _eval(h_new)
                if c_new < c_cur_max:
                    h = h_new
                    c_max = c_new
                    break
            else:
                t_counter *= 0.98
        
        t_counter *= 0.99995
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = h * target / s if s > 1e-12 else np.full(n, 0.5)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = h.min() - 1.0, 1.0 - h.max()
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(h, s, dx):
    n = len(h)
    g = np.zeros(n)
    if 0 <= s < n:
        g[0:n-s] += (1.0 - h[s:n]) * dx
        g[s:n] -= h[0:n-s] * dx
    elif -n < s < 0:
        p = -s
        g[p:n] += (1.0 - h[0:n-p]) * dx
        g[0:n-p] -= h[p:n] * dx
    return g

def _active_subgrad(h, t0, budget):
    n = len(h); dx = 2.0/n; target = n*0.5
    c_star = _eval(h) + 0.003
    t_mom = 1.0; v = np.zeros(n)
    eps = 0.002; step0 = 0.01
    for it in range(4000000):
        if time.time() - t0 > budget: break
        c = np.correlate(h, 1.0-h, mode='full') * dx
        cmax = float(c.max())
        if cmax < c_star:
            c_star = cmax; eps = max(eps*0.97, 1e-5)
        thr = cmax - eps
        active = np.where(c >= thr)[0]
        if len(active) == 0: active = np.array([np.argmax(c)])
        if len(active) > 25: active = np.argsort(c)[-25:]
        g = np.zeros(n)
        for k in active:
            g += _grad_shift(h, k-(n-1), dx)
        g /= len(active)
        t_new = (1.0 + np.sqrt(1.0+4.0*t_mom*t_mom)) * 0.5
        theta = (t_mom - 1.0) / t_new
        v = theta*v + g; t_mom = t_new
        step = step0 / np.sqrt(1.0 + 0.5*it*0.001)
        for _bt in range(15):
            h_new = _proj_sum(h - step*v, target)
            cn = _eval(h_new)
            if cn < cmax: break
            step *= 0.5
        h = h_new
    return _repair(h, n)

def _slsqp_obj(x, alpha, n):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s-s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        sh = (n-1) - k; wk = w[k]*2.0/n
        g += wk * _grad_shift(x, sh, 1.0)
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h); bnds = [(0.0,1.0)]*n; target = n*0.5
    import functools
    obj_f = functools.partial(_slsqp_obj, alpha=alpha, n=n)
    grad_f = functools.partial(_slsqp_grad, alpha=alpha, n=n)
    def _cfun(x): return x.sum() - target
    def _cjac(x): return np.ones(n)
    cons = [{'type':'eq','fun':_cfun,'jac':_cjac}]
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except: return _repair(h, n)

def _coord_descent(h, n_sweeps, rng, t0, budget):
    n = len(h); dx = 2.0/n; target = n*0.5
    best_h = h.copy(); best_c = _eval(h)
    for _sw in range(n_sweeps):
        if time.time() - t0 > budget: break
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 > budget: break
            c = np.correlate(h, 1.0-h, mode='full') * dx
            cmax = float(c.max())
            active = np.argsort(c)[-5:]
            g_j = 0.0
            for k in active:
                s = k - (n-1)
                gs = _grad_shift(h, s, dx)
                g_j += gs[j]
            g_j /= len(active)
            if g_j > 1e-10:
                new_v = h[j] - 0.01 * g_j
            elif g_j < -1e-10:
                new_v = h[j] - 0.01 * g_j
            else:
                continue
            new_v = np.clip(new_v, 0.0, 1.0)
            if new_v == h[j]: continue
            delta = new_v - h[j]
            h[j] = new_v
            cn = _eval(h)
            if cn > cmax:
                h[j] = new_v - delta
    return best_h if best_c <= _eval(h) else h, min(best_c, _eval(h))

def _local_2pt(h, n_tries, rng, t0, budget):
    n = len(h); c5c = _eval(h); best_h = h.copy(); best_c = c5c
    for _ in range(n_tries):
        if time.time() - t0 > budget: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-19:
                c5c = cn
                if cn < best_c: best_c = cn; best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c
