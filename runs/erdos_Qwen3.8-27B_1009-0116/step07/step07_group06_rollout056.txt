import numpy as np
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), c

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = _proj(h.copy(), n * 0.5)
    d = n * 0.5 - h.sum()
    if abs(d) > 1e-10:
        idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _subgrad(h, rng, t0, deadline):
    n = len(h)
    target = n * 0.5
    c5, c = _eval(h)
    best_h, best_c = h.copy(), c5
    step_scale = 0.04
    stagnant = 0
    while time.time() < deadline:
        M = min(3 + stagnant // 25, 25)
        top = np.argsort(c)[-M:]
        w = np.exp((c[top] - c.max()) * 4.0)
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top):
            s = ki - (n - 1)
            if s >= 0:
                ms = np.arange(n - s)
                g[ms] += wi * (1.0 - h[ms + s])
                g[ms + s] -= wi * h[ms]
            else:
                ss = -s
                ms = np.arange(ss, n)
                g[ms] += wi * (1.0 - h[ms - ss])
                g[ms - ss] -= wi * h[ms]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = step_scale / (gn + 1e-10)
        improved = False
        for m in [1.0, 0.5, 0.25, 0.1, 0.05]:
            hn = _proj(h - m * step * g, target)
            c5n, cn = _eval(hn)
            if c5n < c5 - 1e-13:
                h, c, c5 = hn, cn, c5n
                improved = True
                break
        if improved:
            stagnant = 0
            if c5 < best_c: best_c, best_h = c5, h.copy()
            step_scale = min(step_scale * 1.015, 0.08)
        else:
            stagnant += 1
            step_scale *= 0.72
            if stagnant > 250 or step_scale < 1e-10:
                h = _proj(h + 0.012 * rng.randn(n), target)
                c5, c = _eval(h)
                stagnant = 0
                step_scale = 0.03
                if c5 < best_c: best_c, best_h = c5, h.copy()
    return best_h, best_c
