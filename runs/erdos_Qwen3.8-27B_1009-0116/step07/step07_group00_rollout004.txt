import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:]) * dx
        if sh > 0:
            g[sh:] -= h[:-sh] * dx
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp]) * dx
        g[:n - sp] -= h[sp:] * dx
    return g

def _smax_obj_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _cvec(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 0)
    e = np.exp(s)
    e_sum = e.sum()
    w = e / e_sum
    obj = m + np.log(e_sum) / alpha
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, int(idx[ii]), dx)
    return obj, g

def _nest_bb(h0, alpha, topk, steps, step0, deadline):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.88
    step = step0
    f, g = _smax_obj_grad(h, alpha, topk)
    h_s, g_s = h.copy(), g.copy()
    for it in range(steps):
        if time.time() > deadline:
            break
        ds = h - h_s
        ys = g - g_s
        a_val = ds @ ds
        b_val = ds @ ys
        if b_val > 1e-14:
            step = max(step0 * 0.05, min(a_val / b_val, step0 * 25.0))
        h_s, g_s = h.copy(), g.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn, gn = _smax_obj_grad(hn, alpha, topk)
        if fn > f + 1e-14:
            step *= 0.4
            hn = _proj(h - step * v, n * 0.5)
            fn, gn = _smax_obj_grad(hn, alpha, topk)
        h = hn
        f = fn
        g = gn
    return _repair(h, n)

def _slsqp_solve(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        o, _ = _smax_obj_grad(x, alpha, topk)
        return o
    def jac(x):
        _, g = _smax_obj_grad(x, alpha, topk)
        return g
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _fine_polish(h, niter, rng, deadline, step_scale):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = step_scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _3pt_polish(h, niter, rng, deadline, step_scale):

    c5c = _eval(h)
    bh = h.copy()

        d2 = step_scale * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return bh, bc
