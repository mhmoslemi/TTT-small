```python
def _softmax_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    # Numerical stability: subtract max
    e = np.exp((c - c_max) / T)
    s = e.sum()
    val = T * (np.log(s) + c_max / T)  # = T * log(s) + c_max
    # Wait: T * log(sum(exp(c/T))) = T * log(exp(c_max/T) * sum(exp((c-c_max)/T)))
    # = T * (c_max/T + log(sum(exp((c-c_max)/T))))
    # = c_max + T * log(s)
    return val + penalty * (h.sum() - n * 0.5)**2

def _softmax_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    e = np.exp((c - c_max) / T)
    w = e / e.sum()  # softmax weights, sums to 1
    # grad_f = sum_k w[k] * dc[k]/dh[i]
    # (since d/dh_i [c_max + T*log(s)] = T * (1/s) * sum e_k * dc_k/dh_i = sum w_k * dc_k/dh_i)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    gp = 2.0 * penalty * (h.sum() - n * 0.5)
    return g + gp
```

```python
    dx = 2.0 / n
    h_c = h.copy().astype(float)
    c = np.correlate(h_c, 1.0 - h_c, mode='full') * dx
    best_c5 = c.max()
    best_h = h_c.copy()
    nu = np.arange(2 * n - 1) - (n - 1)  # lags
    
    scale = 5e-5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _eval(h):

def _slab(c, h):

        g[i1] += (1.0 - h[i1 - sp])
```
