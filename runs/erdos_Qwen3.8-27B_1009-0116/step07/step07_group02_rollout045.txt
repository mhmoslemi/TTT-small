import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Bisection for the right shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lag_grad(h, k):
    """Gradient of c_k w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    s = w.sum()
    if s < 1e-30:
        return np.zeros(n)
    w /= s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:

            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs_lp(h, alpha, maxit, deadline):
    if time.time() > deadline - 20:
        return h
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:

                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12, 'maxcor': 30})
        return _repair(res.x, n)
    except:
        return h

def _lbfgs_pnorm(h, p, maxit, deadline):

def _top_lag_descent(h, n_lags, max_rounds, deadline):
    n = len(h)
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 5:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += _lag_grad(h, k)
        g /= n_lags
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        # Line search
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5, 1e-5]:

def _local_search(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    scales = [1e-5, 5e-6, 1e-6, 5e-7, 1e-7]
    for it in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        scale = scales[rng.randint(len(scales))]
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def _gradient_local(h, n_iter, rng, deadline):
    """Gradient-informed local search"""

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lag_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        np_ = n - sh
        g[:np_] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:np_]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])

    return g

    return (np.sum(c ** p)) ** (1.0 / p)

    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)

            h = _repair(h - 1e-6 * g, n)
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    for alpha in [3000, 15000, 80000, 400000, 2000000]:
        if time.time() > deadline - 350:
            break
        h = _lbfgs_lp(h, alpha, 5000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    for p in [30, 100, 500]:
        if time.time() > deadline - 250:
            break
        h = _lbfgs_pnorm(h, p, 3000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    if time.time() < deadline - 200:
        h_sg, c5_sg = _subgrad_line(best_h.copy(), 1000, deadline)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()

    if time.time() < deadline - 150:
        h_sl, c5_sl = _single_lag_descent(best_h.copy(), 800, deadline)
        if c5_sl < best_c5:
            best_c5, best_h = c5_sl, h_sl.copy()

    for nl in [5, 3, 2, 1]:
        if time.time() > deadline - 100:
            break
        h_tl, c5_tl = _top_lag_descent(best_h.copy(), nl, 600, deadline)
        if c5_tl < best_c5:
            best_c5, best_h = c5_tl, h_tl.copy()

    if time.time() < deadline - 80:
        h_gl, c5_gl = _gradient_local(best_h, 40000, rng, deadline)
        if c5_gl < best_c5:
            best_c5, best_h = c5_gl, h_gl.copy()

    if time.time() < deadline - 30:
        h_sl, c5_sl = _single_lag_descent(best_h.copy(), 500, deadline)
        if c5_sl < best_c5:
            best_c5, best_h = c5_sl, h_sl.copy()

    if time.time() < deadline - 5:
        h_ls, c5_ls = _random_local(best_h, 150000, rng, deadline)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    if time.time() < deadline - 2:
        h_gl, c5_gl = _gradient_local(best_h, 40000, rng, deadline)
        if c5_gl < best_c5:
            best_c5, best_h = c5_gl, h_gl.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
