import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _softmax_val_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c * dx - m * dx)
    s = np.clip(s, -200, 200)
    s -= s.max()
    w = np.exp(s)
    sw = w.sum()
    w /= sw
    val = m * dx + np.log(sw) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        shift = k - (n - 1)
        wk = w[k] * dx
        if shift >= 0:
            g[shift:] += wk * (1.0 - h[:n - shift])
            g[:n - shift] -= wk * h[shift:]
        else:
            s2 = -shift
            g[:n - s2] += wk * (1.0 - h[s2:])
            g[s2:] -= wk * h[:n - s2]
    return val, g

def _proj_grad_opt(h, t0, budget_s, alpha_start, alpha_end, n_iter):
    n = len(h)
    target = n / 2.0
    h = _repair(h.copy(), n)
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    step = 0.01
    n_active = 0
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 8:
            break
        frac = it / max(n_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        val, g = _softmax_val_grad(h, alpha)
        h_new = _proj_sum(h - step * g, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h = h_new
            f = f_new
            n_active = 0
            step = min(step * 1.05, 0.05)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.6
            if step < 1e-10:
                break
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f:
                h = h_new
                f = f_new
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
    return _repair(best_h, n)

def _coord_descent(h, n_tries, rng, t0, budget_s):
    n = len(h)
    target = n / 2.0
    f_cur = _eval(h)
    best_h = h.copy()
    best_f = f_cur
    h = h.copy()
    for _ in range(n_tries):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        if i == j:
            j = (j + 1) % n
        d = 1e-4 * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            f_new = _eval(h)
            if f_new < f_cur - 1e-17:
                f_cur = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return _repair(best_h, n)

def _coord_descent_wide(h, n_tries, rng, t0, budget_s):
    n = len(h)
    f_cur = _eval(h)
    best_h = h.copy()
    best_f = f_cur
    h = h.copy()
    for _ in range(n_tries):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 20)) % n
        k = (i + 22 + rng.randint(0, 40)) % n
        d1 = 1e-4 * rng.randn()
        d2 = 1e-4 * rng.randn()
        hi, hj, hk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= hi <= 1 and 0 <= hj <= 1 and 0 <= hk <= 1:
            h[i], h[j], h[k] = hi, hj, hk
            f_new = _eval(h)
            if f_new < f_cur - 1e-17:
                f_cur = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = hi - d1, hj - d2, hk + d1 + d2
    return _repair(best_h, n)

def _topk_nesterov(h, k_top, steps, step0, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = _repair(h.copy(), n)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _ in range(steps):
        if time.time() - t0 > budget_s - 5:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = dx / k_top
        for k in top_k:
            shift = k - (n - 1)
            if shift >= 0:
                g[shift:] += wk * (1.0 - h[:n - shift])
                g[:n - shift] -= wk * h[shift:]
            else:
                s2 = -shift
                g[:n - s2] += wk * (1.0 - h[s2:])
                g[s2:] -= wk * h[:n - s2]
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-17:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 5.0)
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.5
            if step < 1e-11:
                break
    return _repair(best_h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()
