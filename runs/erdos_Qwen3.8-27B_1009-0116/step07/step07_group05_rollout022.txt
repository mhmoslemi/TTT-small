                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_soft(c, h, n, T):
    sc = np.clip(T * (c - c.max()), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _grad_mlag(h, lags, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g * dx / len(lags)

def _adam_opt(h, n, schedule, check_fn):
    best = h.copy()
    best_c = _eval(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    for T, lr, niter in schedule:
        for _ in range(niter):
            if check_fn():
                return best, best_c
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            g = _grad_soft(c, h, n, T)
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g * g
            t += 1
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            h = _project_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
            c5 = _eval(h)
            if c5 < best_c:
                best_c = c5
                best = h.copy()
    return best, best_c

def _mlgd(h, rounds, k_top, steps_init, check_fn):
    n = len(h)
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-k_top:]
        g = _grad_mlag(h, top_k, n)
        improved = False
        for si in range(len(steps)):
            h_new = _project_sum(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 12:
                    steps = steps[:12]
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    n_new = n * factor
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new * 0.5)

def _downsample(h, factor):
    n = len(h)
    n_new = n // factor
    h_c = h[:n_new * factor].reshape(n_new, factor).mean(axis=1)
    return _project_sum(h_c, n_new * 0.5)

def _block_search(h, rng, ntry, c5c, check_fn, bsize=3):
    n = len(h)
    best = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if check_fn():
            break
        start = rng.randint(0, max(1, n - bsize))
        idx = np.arange(start, min(start + bsize, n))
        old = h[idx].copy()
        d = 0.008 * rng.randn(len(idx))
        nv = np.clip(old + d, 0, 1)
        diff = old.sum() - nv.sum()
        if abs(diff) < 0.08:
            nv = np.clip(nv + diff / len(nv), 0, 1)
            h[idx] = nv
            c5n = _eval(h)
            if c5n < best_c - 1e-15:
                best_c = c5n
                best = h.copy()
            else:
                h[idx] = old
    return best, best_c

def _local_2pt(h, ntry, rng, c5c, check_fn):
    n = len(h)
    best = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 2.5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c - 1e-18:
                best_c = c5n
                best = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 15

    # Phase 1: Adam with annealed soft-max (coarse to fine T)
    if budget_s > 120:
        sched1 = [(5.0, 0.03, 150), (10.0, 0.02, 200), (20.0, 0.015, 250),
                  (40.0, 0.01, 300), (80.0, 0.008, 300), (160.0, 0.005, 350)]
        h2, c52 = _adam_opt(best_h.copy(), n0, sched1, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Multi-lag gradient cycling
    k_cycle = [3, 5, 7, 4, 6, 3, 5, 4, 3]
    for ki in range(len(k_cycle)):
        if _bc():
            break
        kt = k_cycle[ki]
        h2, c52 = _mlgd(best_h.copy(), 500, kt, [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: Coarse-to-fine multi-resolution
    if _el() < budget_s - 250:
        h_coarse = _downsample(best_h, 2)
        nc = len(h_coarse)
        h_coarse, _ = _mlgd(h_coarse, 400, 5, [0.06, 0.03, 0.015, 0.008, 0.004, 0.002], _bc)
        h_up = _upsample(h_coarse, 2)
        sched_cf = [(30.0, 0.01, 200), (80.0, 0.006, 250), (160.0, 0.004, 250)]
        h_up, c5_up = _adam_opt(h_up, n0, sched_cf, _bc)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up

    # Phase 4: More Adam at high T (fine polish)
    if _el() < budget_s - 200:
        sched2 = [(200.0, 0.003, 400), (400.0, 0.002, 400), (800.0, 0.001, 500),
                  (1600.0, 0.0006, 500), (3200.0, 0.0003, 500)]
        h2, c52 = _adam_opt(best_h.copy(), n0, sched2, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: MLGD high-precision
    for kt in [3, 5, 4, 3, 5, 4]:
        if _bc():
            break
        h2, c52 = _mlgd(best_h.copy(), 350, kt, [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4], _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Block coordinate search
    if _el() < budget_s - 100:
        h2, c52 = _block_search(best_h.copy(), rng, 5000, best_c5, _bc, bsize=3)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
        h2, c52 = _block_search(best_h.copy(), rng, 4000, best_c5, _bc, bsize=4)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: Final Adam ultra-fine
    if _el() < budget_s - 50:
        sched3 = [(4096.0, 0.0002, 400), (8192.0, 0.0001, 400)]
        h2, c52 = _adam_opt(best_h.copy(), n0, sched3, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 8: 2-point local polish
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.002), 200000)
        best_h, best_c5 = _local_2pt(best_h, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
