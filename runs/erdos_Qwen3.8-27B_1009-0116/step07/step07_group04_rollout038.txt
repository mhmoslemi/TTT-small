def _acf_softmax(h, alpha):
    """Compute soft-max of correlation values with temperature 1/alpha"""
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _acf_softmax_grad(h, alpha, num_active=200):
    """Gradient of soft-max correlation"""
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    # Only keep top num_active
    if num_active < len(w):
        top_idx = np.argsort(w)[-num_active:]
        w[:num_active] = 0  # zero out non-top
        w[top_idx] = w[top_idx] / w[top_idx].sum()  # renormalize (approx)
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        sh = (n-1) - k
        dk = w[k] * 2.0/n
        if sh >= 0:
            g[:n-sh] += dk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= dk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += dk * (1.0 - h[:n-sp])
            g[:n-sp] -= dk * h[sp:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

        else:
            hi = mid

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

    return _lse_grad(x, alpha, n, topk=500)

def _lbfgs_opt(h, alpha, n, maxiter, t_end):
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    # Use penalty for sum constraint
    def obj_pen(x):
        base = _lbfgs_obj(x, alpha, n)
        pen = 1e4 * (x.sum() - target) ** 2
        return base + pen

        g += 2e4 * (x.sum() - target) * np.ones(n)
        return g

                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(35):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _shift_grad(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:n])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _topm_grad(h, n, m):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)

    return g / m

    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    ws = e.sum()
    if ws < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / ws
    if topk < len(w):
        idx = np.argsort(w)[-topk:]
        w2 = np.zeros(len(w))
        w2[idx] = w[idx]

    g = np.zeros(n)

        else:
            sp = -sh

    return g

def _lbfgs_obj(x, alpha, n, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    base = m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    pen = lam * (x.sum() - n * 0.5) ** 2
    return base + pen

def _lbfgs_grad(x, alpha, n, lam):
    g = _lse_grad(x, alpha, n, topk=400)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

    try:
        res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad,
                       args=(alpha, n, lam), method='L-BFGS-B',
                       bounds=bnds,
