def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

            best_n_c5, best_n_h = ca, ha

        # Phase B: Fast pair CD (50% of n's time)
        dl_b = min(time.time() + time_per_n * 0.5, n_dl - time_per_n * 0.3)
        hb, cb = _fast_cd(best_n_h.copy(), rng, dl_b)
        if cb < best_n_c5:
            best_n_c5, best_n_h = cb, hb

        # Phase C: Binary push + CD
        if time.time() < n_dl - time_per_n * 0.15:
            hp2 = _binary_push(best_n_h, n, rng, strength=0.4)
            c5p2 = _eval(hp2)
            if c5p2 < best_n_c5:
                best_n_c5, best_n_h = c5p2, hp2
            dl_c = n_dl - time_per_n * 0.1
            hc, cc = _fast_cd(best_n_h.copy(), rng, dl_c)
            if cc < best_n_c5:
                best_n_c5, best_n_h = cc, hc

        # Phase D: Final subgradient + CD
        if time.time() < n_dl - 5:
            hd, cd = _subgrad_refine(best_n_h.copy(), rng, n_dl - 10)
            if cd < best_n_c5:
                best_n_c5, best_n_h = cd, hd
            if time.time() < n_dl - 3:
                he, ce = _fast_cd(best_n_h.copy(), rng, n_dl - 2)
                if ce < best_n_c5:
                    best_n_c5, best_n_h = ce, he

        evaluatedNs.append((n, best_n_c5, best_n_h.copy()))
        if best_n_c5 < best_overall_c5:
            best_overall_c5 = best_n_c5
            best_overall_h = best_n_h.copy()

    # Final round: take the best n and do more CD with remaining time
    if best_overall_h is not None and time.time() < t0 + budget_s - 10:
        n_best = len(best_overall_h)
        h_final = best_overall_h.copy()
        c5_final = _eval(h_final)
        # One more intensive CD pass
        hf, cf = _fast_cd(h_final, rng, t0 + budget_s - 5)
        if cf < c5_final:
            c5_final, h_final = cf, hf
        # Binary push + CD
        if time.time() < t0 + budget_s - 5:
            hp3 = _binary_push(h_final, n_best, rng, strength=0.2)
            c5p3 = _eval(hp3)
            if c5p3 < c5_final:
                c5_final, h_final = c5p3, hp3
            if time.time() < t0 + budget_s - 3:
                hf2, cf2 = _fast_cd(h_final, rng, t0 + budget_s - 2)
                if cf2 < c5_final:
                    c5_final, h_final = cf2, hf2
        best_overall_h = h_final
        best_overall_c5 = c5_final

    best_overall_h = _repair(best_overall_h, len(best_overall_h))
    best_overall_c5 = _eval(best_overall_h)
    return (best_overall_h, best_overall_c5, len(best_overall_h))
