        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def nesterov(h, k_top, alpha, n_steps, lr0, deadline):
    n = len(h)
    target = n * 0.5
    f = eval_c5(h)
    best_f, best_h = f, h.copy()
    v = np.zeros(n)
    mu = 0.7
    lr = lr0
    stall = 0
    for i in range(n_steps):
        if time.time() > deadline:
            break
        g = grad_fn(h, k_top, alpha)
        v = mu * v + (1.0 - mu) * g
        h_new = repair(h - lr * v, target)
        f_new = eval_c5(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            lr = min(lr * 1.02, lr0 * 15.0)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
            h_new = repair(h - lr * v, target)
            f_new = eval_c5(h_new)
            if f_new < f:
                h = h_new
                f = f_new
            stall += 1
        if stall > 300:
            v *= 0.01
            lr = lr0
            stall = 0
    return best_h, best_f

def local_search(h, n_tries, rng, deadline):
    n = len(h)
    f = eval_c5(h)
    best_h, best_f = h.copy(), f
    step = 5e-5
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n-1)) % n
        d = step * rng.randn()
        hi, hj = h[i]+d, h[j]-d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            fn = eval_c5(h)
            if fn < f - 1e-16:
                f = fn
                if f < best_f:
                    best_f, best_h = f, h.copy()
                step = min(step * 1.05, 0.01)
            else:
                h[i], h[j] = hi-d, hj+d
                step *= 0.95
                if step < 1e-8:
                    step = 5e-5
    return best_h, best_f

def repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    # Simple approach: shift and clip, then fix residual
    shift = diff / len(h)
    h2 = np.clip(h + shift, 0, 1)
    s2 = h2.sum()
    diff2 = target - s2
    if abs(diff2) < 1e-12:
        return h2
    # Fix residual by adjusting non-boundary elements
    mask = (h2 > 0) & (h2 < 1)
    n_free = mask.sum()
    if n_free > 0:
        adj = diff2 / n_free
        h2[mask] = np.clip(h2[mask] + adj, 0, 1)
        s3 = h2.sum()
        if abs(s3 - target) > 1e-12:
            # Fallback to bisection

                    lo = mid
                else:
                    hi = mid
            h2 = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h2

Let me also think about whether there's a fundamentally better approach than Nesterov:

**Coordinate descent**: For each coordinate i, find the optimal h[i] given all others. Since the objective is max of quadratics (in h[i]), this is a 1D minimax problem.

For coordinate i, the correlations involve h[i] in the following way:
- For lag sh (where the correlation at index n-1+sh involves h[i] and h[i±sh]):
  - g_{n-1+sh} = sum_j h[j]*(1-h[j+sh])
  - h[i] appears as: h[i]*(1-h[i+sh]) + h[i-sh]*(1-h[i]) = h[i] - h[i]*h[i+sh] + h[i-sh] - h[i-sh]*h[i]
  - = h[i] - h[i]*(h[i+sh] + h[i-sh]) + h[i-sh]
  - This is linear in h[i]! (since h[i+sh] and h[i-sh] are fixed)
  
  Wait, that's only true if i±sh are different from i. If sh = 0, then h[i]*(1-h[i]) = h[i] - h[i]², which IS quadratic.

So for sh ≠ 0: each g_k is linear in h[i]
For sh = 0: g_0 is quadratic (concave) in h[i]

The max of several linear functions and one concave quadratic is a convex function of h[i] (max of affine and concave... wait, that's not necessarily convex).

Actually, the max of affine functions is convex. The max of a concave function and affine functions is not necessarily convex or concave. But in 1D, we can find the minimum by checking:
1. The minimum of the concave quadratic (at its vertex)
2. The points where the concave quadratic crosses each affine function
3. The minimum of the max at the boundary [0, 1]

This is solvable in O(k_top) time per coordinate. So a full coordinate descent pass is O(n * k_top) time, which is very fast!

def coord_descent(h, k_top, n_passes, deadline):

    target = n * 0.5

        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Find optimal h[i] given all others
            # Evaluate g_k for a range of x values
            x_old = h[i]
            # Try candidates: 0, 1, x_old, and values where the top correlations are equalized
            candidates = [0.0, 1.0, x_old]
            # Sample around x_old
            for dx in [1e-3, -1e-3, 1e-2, -1e-2, 5e-3, -5e-3, 0.01, -0.01, 0.05, -0.05]:
                x = x_old + dx

            best_x = x_old

            for x in candidates:
                h[i] = x

                # For now, just evaluate (sum constraint violation is small)

        if not improved:
