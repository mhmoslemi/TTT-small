            break
        frac = it / max(niter, 1)
        T = T0 * (T1 / T0) ** frac
        x = np.arange(n) / n
        amp = 0.04 * max(T, 1e-12) ** 0.5
        freq = 1.0 + 8.0 * rng.randint(1, 5)
        phase = 2.0 * np.pi * rng.rand()
        pert = amp * np.sin(2.0 * np.pi * freq * x + phase)
        pert += amp * 0.5 * rng.randn(n)
        pert -= pert.mean()
        hn = _repair(h + pert, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _restart_search(h, rng, deadline, nrestarts):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for _ in range(nrestarts):
        if time.time() > deadline:
            break
        h2 = best_h.copy()
        frac = 0.2 + 0.3 * rng.rand()
        mask = rng.rand(n) < frac
        h2[mask] += 0.08 * rng.randn(int(mask.sum()))
        h2 = _repair(h2, n)
        alpha = 50000 + 200000 * rng.rand()
        tk = 8 + int(12 * rng.rand())
        h2 = _nesterov(h2, alpha, tk, 800, 0.004, deadline)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()
    return best_h, best_c5

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = 1.5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _3pt(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 3)) % nc
        k = (i + 4 + rng.randint(0, 6)) % nc
        d1 = 1.2e-6 * rng.randn()
        d2 = 1.2e-6 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Initial Nesterov refinement (80s)
    dl1 = min(deadline, t0 + 80)
    for alpha, tk, st, ns in [(5000, 40, 0.005, 4000), (50000, 20, 0.003, 4000)]:
        if time.time() > dl1:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coarse-to-fine multi-scale (120s)
    dl2 = min(deadline, t0 + 200)
    for G in [4, 8]:
        if time.time() > dl2:
            break
        hu = _coarse_fine(best_h, G, 30000, 12, 3000, dl2)
        c5v = _eval(hu)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hu.copy()

    # Phase 3: Multi-alpha Nesterov sweep (200s)
    dl3 = min(deadline, t0 + 400)
    for alpha, tk, st, ns in [(30000, 20, 0.004, 5000),
                               (150000, 12, 0.002, 5000),
                               (600000, 8, 0.001, 4000)]:
        if time.time() > dl3:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl3)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Large-scale block annealing (250s) - larger T than before
    dl4 = min(deadline, t0 + 650)
    bs_list = [1, 2, 3, 4, 5, 7, 10, 14, 20]
    best_h, best_c5 = _anneal(best_h.copy(), 80000, rng, dl4, 0.08, 1e-9, bs_list, 0.035)

    # Phase 5: Structured (sinusoidal) annealing (100s)
    dl5 = min(deadline, t0 + 750)
    best_h, best_c5 = _structured_anneal(best_h.copy(), 15000, rng, dl5, 0.03, 1e-9)

    # Phase 6: Restart-based global search (120s)
    dl6 = min(deadline, t0 + 870)
    best_h, best_c5 = _restart_search(best_h, rng, dl6, 25)

    # Phase 7: Final Nesterov (80s)
    dl7 = min(deadline, t0 + budget_s - 70)
    for alpha, tk, st, ns in [(200000, 10, 0.002, 5000),
                               (2000000, 5, 0.0005, 4000)]:
        if time.time() > dl7:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl7)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 8: Fine polish + 3-point
    rem1 = max(0, deadline - time.time())
    n_pol = min(int(rem1 * 0.6 / 0.0012), 60000)
    dl8a = t0 + budget_s - 3 - max(0, rem1 * 0.4 / 0.0018)
    best_h, best_c5 = _fine_polish(best_h, n_pol, rng, dl8a)

    if time.time() < t0 + budget_s - 3:
        rem2 = max(0, t0 + budget_s - 3 - time.time())
        n_3 = min(int(rem2 / 0.0018), 40000)
        best_h, best_c5 = _3pt(best_h, n_3, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
