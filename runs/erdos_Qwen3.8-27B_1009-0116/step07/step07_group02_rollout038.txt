import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_cvals(h).max())

def _ls_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _ls_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _fourier_noise(h, rng, n_modes=10, scale=0.05):
    """Add smooth noise in Fourier space."""
    n = len(h)
    x = np.arange(n)
    noise = np.zeros(n)
    for _ in range(min(n_modes, n // 2)):
        k = rng.randint(1, n // 2)
        amp = scale / (k + 1) * rng.randn()
        phase = rng.uniform(0, 2 * np.pi)
        noise += amp * np.cos(2 * np.pi * k * x / n + phase)
    return _repair(h + noise, n)

def _perturb(h, rng, scale=0.02):
    """Random perturbation with integral preservation."""
    n = len(h)
    return _repair(h + scale * rng.randn(n), n)

def _lbfgs(h, alpha, maxiter, deadline):
    n = len(h)
    if time.time() > deadline - 10:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha,),
                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                               'maxls': 30, 'maxfun': maxiter * 3})
        return _repair(res.x, n)
    except Exception:
        return h

def _topk_descent(h, n_top, max_rounds, deadline):
    """Descent on the top-k most active shifts."""
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_top
        ng = np.linalg.norm(g)
        if ng < 1e-15:
            break
        # Backtracking line search
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 1e-4, 1e-5]:
            h_new = _repair(h - step * g / (ng + 1e-15), n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-17:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5

def _local_search(h, n_iter, rng, deadline):
    """Fine 2-point local search."""
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 1e-6 * rng.randn()
        nh, jh = h[i] + d, h[j] - d
        if 0 <= nh <= 1 and 0 <= jh <= 1:
            h[i], h[j] = nh, jh
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[i], h[j] = nh - d, jh + d
    return h, c5c

**Idea: Use a different parameterization.**

Instead of optimizing h directly, optimize a "logit" parameter:
h[i] = sigmoid(logit[i])

This automatically enforces 0 < h < 1. The integral constraint becomes:
Σ sigmoid(logit[i]) * dx = 1

This is a non-convex constraint in logit space, but we can use a penalty.

The advantage: the optimizer can push logits to ±∞ to get exact 0s and 1s without hitting the boundary.

def _sigmoid(t):
    return 1.0 / (1.0 + np.exp(-np.clip(t, -50, 50)))

def _ls_obj_logit(x, alpha):
    h = _sigmoid(x)
    # Penalty for integral constraint
    integral = h.sum() * (2.0 / len(h))
    penalty = 1000.0 * (integral - 1.0) ** 2
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _ls_grad_logit(x, alpha):
    h = _sigmoid(x)
    dh = h * (1 - h)  # derivative of sigmoid
    n = len(h)
    dx = 2.0 / n
    # Integral penalty gradient
    integral = h.sum() * dx
    penalty_g = 2 * 1000.0 * (integral - 1.0) * dx * dh
    # Log-sum-exp gradient
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return penalty_g
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g * dh + penalty_g

This is interesting because it removes the bound constraints (0≤h≤1) from the optimizer's perspective. The sigmoid naturally keeps h in (0,1). The integral constraint is handled by the penalty.

But the downside: when h is close to 0 or 1, the sigmoid gradient dh = h(1-h) becomes very small, which means the optimizer makes tiny progress. This is the "saturation" problem.

To mitigate: use a "soft" sigmoid with a temperature parameter, or just stick with the bounded formulation.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_cvals(h).max())

def _ls_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _ls_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxiter, deadline):
    n = len(h)
    if time.time() > deadline - 10:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha,),
                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                               'maxls': 20, 'maxfun': maxiter * 3})
        return _repair(res.x, n)
    except Exception:
        return h

def _swap_search(h, n_iter, rng, deadline):
    """Try swapping pairs of values (preserves integral exactly)."""
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        # Try partial swap: move δ from i to j
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 2)) % n
        if i == j:
            continue
        d = rng.uniform(-0.5, 0.5) * min(h[i], 1 - h[j], 0.3)
        if abs(d) < 1e-9:
            continue
        h[i] += d
        h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return h, c5c

def _topk_descent(h, n_top, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_top
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 1e-4]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-17:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5
