import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        h *= target / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_h(h, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    act = np.argsort(c)[-topk:]
    g = np.zeros(n)
    for k in act:
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += dx * (1.0 - h[:n - sh])
            g[:n - sh] -= dx * h[sh:]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return g, c, m

def _proj_grad(h, n, max_iter, t0, t_end):
    mu = 0.7
    v = np.zeros(n)
    step = 5e-5
    cur_c = _eval(h)
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g, c, m = _grad_h(h, 15)
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g /= gn
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(12):
            h_new = _project(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new; cur_c = c_new
                step = min(step * 1.08, 2e-4)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.4, 2e-4)
    return _repair(h, n), cur_c

def _local_search(h, end_time, rng):

    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    s2, s3, s4 = 8e-5, 6e-5, 4e-5
    stag = 0
    while time.time() < end_time:
        r = rng.random()
        if r < 0.35:
            pts, step = 2, s2
        elif r < 0.65:
            pts, step = 3, s3
        else:
            pts, step = 4, s4
        i0 = rng.randint(n)
        idx = [i0]

            d *= 1 if rng.random() < 0.5 else -1

        if len(set(idx)) < pts:
            continue
        dvec = step * rng.randn(pts)
        dvec -= dvec.mean()
        new_h = h.copy()
        ok = True

        if not ok:

        c_new = _eval(new_h)

def _slsqp_sub(h, n, group_size, maxit, t_end):
    ng = n // group_size
    for _ in range(6):
        if time.time() > t_end:
            break
        for b in range(ng):
            if time.time() > t_end:
                break
            sl = np.arange(b * group_size, b * group_size + group_size)
            if sl[-1] >= n:
                sl = sl[:n - b * group_size]
            x0 = h[sl].copy()
            bsum = x0.sum()
            def _ob(x, _sl=sl, _x0=x0):
                ht = h.copy()
                ht[_sl] = x
                return _eval(ht)
            def _oc(x):
                return x.sum() - bsum
            def _oj(x):
                return np.ones(len(x))
            try:
                cons = [{'type': 'eq', 'fun': _oc, 'jac': _oj}]
                res = minimize(_ob, x0, method='SLSQP',
                               bounds=[(0, 1)] * len(sl), constraints=cons,
                               options={'maxiter': maxit, 'ftol': 1e-14})
                h[sl] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)
