import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    if abs(h.sum() - t) > 1e-9:
        h = _proj_sum(h, t)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_subgrad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / len(idx)

            break

            stall = 0

        else:
            step *= dec
            stall += 1
            if stall > 60 or step < 1e-10:

    d = h - 0.5
    h2 = np.clip(h + s * d * np.abs(d), 0.0, 1.0)
    return _proj_sum(h2, n * 0.5)

def _lse_obj(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return float(m + np.log(np.sum(np.exp(c - m))) / beta + pen * (h.sum() - n * 0.5) ** 2)

def _lse_grad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    w = np.exp(c - m)
    sw = w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] / sw * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - k

            i1 = np.arange(cnt)
            g[i1] += wk * (1.0 - h[i1 + sh])

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g + 2.0 * pen * (h.sum() - n * 0.5)
