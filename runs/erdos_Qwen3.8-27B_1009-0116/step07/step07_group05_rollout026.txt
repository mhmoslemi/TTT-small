import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, n, lags, weights, dx):
    g = np.zeros(n)
    n1 = n - 1
    for idx, wt in zip(lags, weights):
        if wt < 1e-14:
            continue
        offset = idx - n1
        w = wt * dx
        if offset >= 0:
            if n - offset > 0:
                g[:n-offset] += w * (1.0 - h[offset:])
            if offset > 0:
                g[offset:] -= w * h[:n-offset]
        else:
            sp = -offset
            if n - sp > 0:
                g[sp:] += w * (1.0 - h[:n-sp])
            if sp > 0:
                g[:n-sp] -= w * h[sp:]
    return g

def _nesterov_pm(h, n, p, k_top, step0, rounds, check_fn):
    dx = 2.0 / n
    tgt = n * 0.5
    h = h.copy()
    vm = h.copy()
    t = 1.0
    step = step0
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[-k_top:]
        top = c[idx]
        mp = np.mean(top ** p)
        w = top ** (p - 1) / (k_top * mp)
        g = _grad_lags(h, n, idx, w, 1.0)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (1 - t) / t_new
        vm2 = h + beta * (vm - h)
        h_try = _project_sum(vm2 - step * g, tgt)
        c5_try = _eval(h_try)
        c5_cur = _eval(h)
        if c5_try < c5_cur - 1e-16:
            vm = vm2
            h = h_try
            t = t_new
            step = min(step * 1.15, 0.2)
            stall = 0
        else:
            step *= 0.5
            stall += 1
            if stall > 8:
                h_try2 = _project_sum(h - step * 2 * g, tgt)
                if _eval(h_try2) < c5_cur:
                    h = h_try2
                    vm = h
                    step *= 1.3
                    stall = 0
                else:
                    break
            t = t_new
    return h, _eval(h)

def _mlgd(h, rounds, k_top, steps_init, check_fn):
    n = len(h)

        else:
