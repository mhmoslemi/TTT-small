import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            no_improve += 1
        if no_improve > 18:
            scale = 1.5e-5 + 4e-7 * no_improve
            h += scale * rng.randn(n)
            h = _proj(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-7:
            h = _proj(h, n * 0.5)
    return h, best_c5

def _local_refine(h, n, duration_s, rng, t_end):
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    end_t = time.time() + duration_s
    while time.time() < min(end_t, t_end - 2):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Brief Nesterov to refine starting point
    for alpha in [20000, 200000, 1000000]:
        t_end = min(t0 + 12, t0 + budget_s * 0.04)
        h = _nesterov(h, alpha, 150, 0.003, t_end)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Multi-scale hybrid coordinate descent
    sched = [
        (3, 0.09), (5, 0.10), (8, 0.11), (12, 0.11),
        (20, 0.12), (15, 0.09), (10, 0.08), (6, 0.07), (4, 0.06),
    ]
    for top_k, frac in sched:
        if time.time() > t0 + budget_s * 0.72:
            break
        h = best_h.copy()
        h, c5v = _cd_hybrid(h, n, budget_s * frac, top_k, rng, t0 + budget_s, n_gs=20)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Nesterov polish with very high alpha
    if time.time() < t0 + budget_s * 0.80:
        h = best_h.copy()
        for alpha in [5000000, 20000000]:
            t_end = min(time.time() + 12, t0 + budget_s * 0.78)
            h = _nesterov(h, alpha, 150, 0.0008, t_end)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 4: Final ultra-fine CD targeting the top lag
    if time.time() < t0 + budget_s - 30:
        rem = t0 + budget_s - 30 - time.time()
        h = best_h.copy()
        h, c5v = _cd_hybrid(h, n, rem * 0.6, 3, rng, t0 + budget_s - 5, n_gs=25, frac_struct=0.7)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        h = best_h.copy()
        h, c5v = _cd_hybrid(h, n, rem * 0.4, 5, rng, t0 + budget_s - 5, n_gs=22, frac_struct=0.6)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Micro local search
    if time.time() < t0 + budget_s - 3:
        rem = t0 + budget_s - 3 - time.time()
        h = best_h.copy()
        best_h, best_c5 = _local_refine(h, n, rem, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
