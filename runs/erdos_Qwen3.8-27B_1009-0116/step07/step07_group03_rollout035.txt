                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            nc = _eval(h)
            if nc < best_c5:
                best_c5 = nc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 20:
            h = _proj_to_sum(h + 3e-5 * rng.randn(n), n * 0.5)
            no_improve = 0
    return h, best_c5

def _block_cd(h, n, duration_s, rng, t_end):
    """Block coordinate descent: shift blocks of consecutive points"""
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    block_sizes = [3, 5, 8]
    while time.time() < min(end_t, t_end - 3):
        c = _corrs(h)
        top_T = 8
        top_idx = np.sort(np.argsort(c)[-top_T:])
        a_vals = c[top_idx]
        grads = _lag_grads(h, top_idx, n)
        improved = False
        for _ in range(n * 3):
            if time.time() > min(end_t, t_end - 3):
                break
            bs = block_sizes[rng.randint(len(block_sizes))]
            i = rng.randint(n - bs)
            # Block move: add delta to h[i:i+bs], subtract evenly from random other positions
            j_base = rng.randint(n - bs)
            if j_base <= i + bs:
                j_base = (j_base + bs + 1) % max(1, n - bs)
            if j_base <= i + bs:
                continue
            b_sum = grads[:, i:i + bs].sum(axis=1) - grads[:, j_base:j_base + bs].sum(axis=1)
            d_max = min(0.99 - h[i:i + bs].max(), h[j_base:j_base + bs].max()) - 1e-13
            d_min = max(-0.99 + h[i:i + bs].min(), 1.01 - h[j_base:j_base + bs].min()) + 1e-13
            if d_max <= d_min:
                continue
            # Bisection for optimal block shift
            lo, hi = d_min, d_max
            for _ in range(25):
                mid = (lo + hi) * 0.5
                f_mid = np.max(a_vals + b_sum * mid)
                lo2 = lo + (hi - lo) * 0.3
                hi2 = hi - (hi - lo) * 0.3
                f_lo = np.max(a_vals + b_sum * lo2)
                f_hi = np.max(a_vals + b_sum * hi2)
                if f_mid <= f_lo and f_mid <= f_hi:
                    break
                if f_lo < f_hi:
                    hi = hi2
                else:
                    lo = lo2
            d_opt = (lo + hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i:i + bs] += d_opt
                h[j_base:j_base + bs] -= d_opt
                improved = True
        if improved:
            nc = _eval(h)
            if nc < best_c5:
                best_c5 = nc
        s = h.sum()
        if abs(s - n * 0.5) > 1e-8:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 1.5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov with graduated alphas
    for alpha in [1500, 10000, 60000, 300000, 1200000]:
        if time.time() > t0 + budget_s * 0.08:
            break
        dur = min(12, t0 + budget_s * 0.08 - time.time())
        if dur < 2:
            break
        h = _nesterov(h, alpha, 400, 0.002, t0 + budget_s * 0.08)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coordinate descent (multiple configs)
    for top_T, frac in [(5, 0.06), (10, 0.06), (15, 0.05)]:
        if time.time() > t0 + budget_s * 0.30:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * frac, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Block coordinate descent (different geometry)
    if time.time() < t0 + budget_s * 0.45:
        h = best_h.copy()
        h, c5v = _block_cd(h, n, budget_s * 0.10, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: High-alpha Nesterov refinement
    if time.time() < t0 + budget_s * 0.58:
        h = best_h.copy()
        for alpha in [5000000, 20000000]:
            t_end = min(time.time() + 8, t0 + budget_s * 0.56)
            h = _nesterov(h, alpha, 200, 0.0008, t_end)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 5: Perturbation restarts + Nesterov
    if time.time() < t0 + budget_s * 0.72:
        for pert in [5e-5, 2e-4, 8e-5]:
            if time.time() > t0 + budget_s * 0.70:
                break
            h_p = _proj_to_sum(best_h + pert * rng.randn(n), n * 0.5)
            h_p = _nesterov(h_p, 50000, 150, 0.001, t0 + budget_s * 0.70)
            h_p = _nesterov(h_p, 500000, 150, 0.0005, t0 + budget_s * 0.70)
            c5v = _eval(h_p)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_p.copy()

    # Phase 6: Final coordinate descent
    if time.time() < t0 + budget_s * 0.85:
        for top_T in [8, 12]:
            if time.time() > t0 + budget_s * 0.84:
                break
            h = best_h.copy()
            dur = min(budget_s * 0.03, t0 + budget_s * 0.84 - time.time())
            if dur > 2:
                h, c5v = _coord_descent(h, n, dur, top_T, rng, t0 + budget_s)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h.copy()

    # Phase 7: Block CD polish
    if time.time() < t0 + budget_s * 0.92:
        h = best_h.copy()
        dur = min(budget_s * 0.03, t0 + budget_s * 0.92 - time.time())
        if dur > 3:
            h, c5v = _block_cd(h, n, dur, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 8: Ultra-fine local search
    if time.time() < t0 + budget_s - 4:
        remaining = t0 + budget_s - time.time() - 2
        n_tries = min(int(remaining / 0.002), 100000)
        if n_tries > 1000:
            h = best_h.copy()
            best_h, best_c5 = _local_search(h, n_tries, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
