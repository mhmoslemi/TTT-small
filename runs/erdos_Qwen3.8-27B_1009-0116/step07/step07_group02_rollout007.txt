import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _coord_opt(h, n, deadline):
    """Coordinate descent: for each i, optimize h[i] to minimize max c_k"""
    c = _cvals(h)
    c5c = c.max()
    for i in range(n):
        if time.time() > deadline - 10:
            break
        # Compute B_k for this i
        B = np.zeros(2*n-1)
        for k in range(2*n-1):
            s = k - (n-1)
            if s >= 0:
                b = 0.0
                if i + s <= n-1:
                    b += 1.0 - h[i+s]
                if i >= s and i <= n-1-s:
                    b -= h[i-s]
                B[k] = b
            else:
                t = -s
                b = 0.0
                if i >= t:
                    b += 1.0 - h[i-t]
                if i + t <= n-1:
                    b -= h[i+t]
                B[k] = b
        # c_k(v) = c_k + (v - h[i]) * B_k
        # Find v in [0,1] minimizing max_k(c_k + (v-h[i])*B_k)
        # Bisection on the subgradient
        lo_v, hi_v = 0.0, 1.0
        for _ in range(30):
            mid_v = (lo_v + hi_v) * 0.5
            delta = mid_v - h[i]
            c_new = c + delta * B
            m = c_new.max()
            active = c_new >= m - 1e-15
            g = B[active].mean()  # subgradient (average of active B's)
            if g < 0:
                hi_v = mid_v  # should decrease v
            else:
                lo_v = mid_v  # should increase v
        v_opt = (lo_v + hi_v) * 0.5
        h[i] = v_opt
    return h, _eval(h)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _compute_B(h, n):
    B = np.zeros((n, 2 * n - 1))
    o = 1.0 - h
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                B[:, k] = o - h
            else:
                B[:n - s, k] += o[s:]
                B[s:, k] -= h[:n - s]
        else:
            t = -s
            B[t:, k] += o[:n - t]
            B[:n - t, k] -= h[t:]
    return B

def _coord_pass(h, n, order, deadline):
    c = _cvals(h)
    B = _compute_B(h, n)
    for i in order:
        if time.time() > deadline - 5:

        B_i = B[i]

        lo_v, hi_v = 0.0, 1.0

            delta = mid_v - cur
            c_t = c + delta * B_i
            m = c_t.max()
            active = c_t >= m - 1e-13
            g = B_i[active].mean()
            if g < -1e-13:
                hi_v = mid_v
            elif g > 1e-13:
                lo_v = mid_v
            else:
                break
        v_opt = (lo_v + hi_v) * 0.5
        if abs(v_opt - cur) > 1e-15:

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx_v = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx_v
        if sh >= 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g
