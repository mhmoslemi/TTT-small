        h_c = _repair(h_c, n_c)
        h_c, c5c = _mom_opt(h_c, n_c, 500, 3, [0.12, 0.06, 0.03, 0.015], t0, budget_s)
        h_f = _upsample(h_c, factor)
        if len(h_f) < n0:
            h_f = np.pad(h_f, (0, n0 - len(h_f)), mode='edge')
        h_f = _repair(h_f, n0)
        h_f, c5f = _mom_opt(h_f, n0, 300, 5, [0.03, 0.015, 0.008], t0, budget_s)
        if c5f < best_c:
            best_c = c5f
            best_h = h_f
    return best_h, best_c

def _local_3pt(h, ntry, rng, c5c, t0, budget_s):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if time.time() - t0 > budget_s - 10:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        k = (i + 7 + rng.randint(0, 5)) % nc
        d1 = 5e-6 * rng.randn()
        d2 = 5e-6 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 < ni < 1 and 0 < nj < 1 and 0 < nk < 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < best_c - 1e-19:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
    return best_h, best_c

def _local_2pt(h, ntry, rng, c5c, t0, budget_s):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if time.time() - t0 > budget_s - 4:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 3)) % nc
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 < ni < 1 and 0 < nj < 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c - 1e-19:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _softmom_phase(h, n, temp, rounds, t0, budget_s):
    v = np.zeros(n)
    c5c = _eval(h)
    steps = [0.02, 0.01, 0.005, 0.002]
    stall = 0
    for _r in range(rounds):
        if time.time() - t0 > budget_s - 12:
            break
        g, _ = _softmax_grad(h, temp)
        improved = False
        for si in range(len(steps)):
            v = 0.85 * v + 0.15 * (-steps[si] * g)
            h_new = _project(h + v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s > steps[si] * 2.0]
                if len(steps) < 3:
                    steps.append(steps[0] * 0.3)
                break
        if not improved:
            v *= 0.5
            steps = [s * 0.5 for s in steps]
            if len(steps) < 3:
                steps.append(steps[0] * 0.3)
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine with momentum
    h1, c1 = _coarse_fine(best_h, n0, t0, budget_s)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: Multi-lag gradient descent with momentum
    for kt in [3, 5, 4, 6, 3, 5, 4]:
        if time.time() - t0 > budget_s - 60:
            break
        h2, c52 = _mom_opt(best_h.copy(), n0, 400, kt, [0.04, 0.02, 0.01, 0.005, 0.002], t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: Softmax gradient (global lag awareness)
    for temp in [0.01, 0.005, 0.003]:
        if time.time() - t0 > budget_s - 50:
            break
        h2, c52 = _softmom_phase(best_h.copy(), n0, temp, 250, t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: Structured perturbations (sinusoidal + shift)
    for s in [0, 1, 2, 3, 1, 2]:
        if time.time() - t0 > budget_s - 80:
            break
        h_sh = best_h.copy()
        if s > 0:
            h_sh = np.roll(h_sh, s)
        phase = rng.uniform(0, 2 * 3.14159)
        h_sh += 0.004 * np.sin(2 * 3.14159 * 2.0 * np.arange(n0) / n0 + phase)
        h_sh += 0.002 * rng.randn(n0)
        h_sh = _repair(h_sh, n0)
        h2, c52 = _mom_opt(h_sh, n0, 350, 3, [0.05, 0.02, 0.01, 0.005], t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Fine polish with small steps
    for kt in [3, 5, 4, 3]:
        if time.time() - t0 > budget_s - 30:
            break
        h2, c52 = _mom_opt(best_h.copy(), n0, 300, kt, [0.008, 0.004, 0.002, 0.001, 5e-4], t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: 3-point local search
    if time.time() - t0 < budget_s - 20:
        nt = min(int((budget_s - 20 - (time.time() - t0)) / 0.004), 80000)
        best_h, best_c5 = _local_3pt(best_h, nt, rng, best_c5, t0, budget_s)

    # Phase 7: 2-point ultra-fine
    if time.time() - t0 < budget_s - 3:
        nt = min(int((budget_s - 3 - (time.time() - t0)) / 0.002), 80000)
        best_h, best_c5 = _local_2pt(best_h, nt, rng, best_c5, t0, budget_s)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
