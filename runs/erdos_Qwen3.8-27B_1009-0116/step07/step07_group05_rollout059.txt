import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _active_grad(h, lags, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += dx * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= dx * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += dx * (1.0 - h[sp:])
            g[sp:] -= dx * h[:n - sp]
    return g / len(lags)

def _gd_active(h, n, max_iters, check_fn):

    stall = 0

        active = np.where(c > thresh)[0]
        if len(active) < 3:
            active = np.argsort(c)[-5:]
        g = _active_grad(h, active, n)
        improved = False
        for _ in range(8):

            if c5n < c5c - 1e-15:

                c5c = c5n
                step = min(step * 1.25, 0.3)
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.4
            stall += 1
            if stall > 20 or step < 1e-9:
                break
        else:
            stall = 0
    return h, c5c

def _lsx_obj(x, n, temp):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip((c - m) / temp, -40, 0)
    return m + temp * np.log(np.sum(np.exp(sc)))

def _lsx_grad(x, n, temp):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip((c - m) / temp, -40, 0)
    w = np.exp(sc)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:

        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g
