from scipy.optimize import linprog

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - t) > 1e-10:
        h *= t / s; h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                a = min(abs(d), h[i]) if d < 0 else min(d, 1.0 - h[i])
                h[i] += a if d > 0 else -a
                d += a if d < 0 else -a
                if abs(d) < 1e-12: break
    return h

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(45):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t: lo = m
        else: hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k):
    n = len(h); dx = 2.0 / n; s = (n - 1) - k
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[:m] = 1.0 - h[s:s+m]
        g[s:] -= h[:m]
    else:
        p = -s; m = n - p
        g[p:] = 1.0 - h[:m]
        g[:m] -= h[p:]
    return g * dx

    w /= ws

    return g
