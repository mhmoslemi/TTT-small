def _solve_pair_1d(c_top, b, d_lo, d_hi):
    T = len(c_top)
    # All pairwise intersections
    i_idx, j_idx = np.triu_indices(T, 1)
    denom = b[i_idx] - b[j_idx]
    valid = np.abs(denom) > 1e-15
    candidates = np.array([d_lo, d_hi])
    if np.any(valid):
        tc = (c_top[j_idx[valid]] - c_top[i_idx[valid]]) / denom[valid]
        in_range = (tc > d_lo) & (tc < d_hi)
        candidates = np.concatenate([candidates, tc[in_range]])
    # Evaluate max at each candidate
    # Shape: (n_cands, T)

    obj = vals.max(axis=1)
    best = np.argmin(obj)
    return candidates[best]

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_and_top(h, n, dx, top_T):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-top_T:]
    return c, top_idx

def _grad_lag(h, sh, n, dx):
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        if m > 0:
            g[:m] += (1.0 - h[sh:sh+m]) * dx
        if sh > 0:
            g[sh:] -= h[:n-sh] * dx
    else:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m]) * dx
        g[:m] -= h[sp:sp+m] * dx
    return g

def _solve_1d(c_top, b, d_lo, d_hi):
    T = len(c_top)
    best_t = 0.0
    best_v = float(np.max(c_top))
    for tc in (d_lo, d_hi):
        v = float(np.max(c_top + b * tc))
        if v < best_v:
            best_v = v
            best_t = tc
    i_idx, j_idx = np.triu_indices(T, 1)
    denom = b[i_idx] - b[j_idx]
    mask = np.abs(denom) > 1e-15
    if np.any(mask):
        tc = (c_top[j_idx[mask]] - c_top[i_idx[mask]]) / denom[mask]
        inr = (tc > d_lo + 1e-14) & (tc < d_hi - 1e-14)
        if np.any(inr):
            cands = tc[inr]
            vals = c_top[None, :] + b[None, :] * cands[:, None]
            obj = vals.max(axis=1)
            bi = int(np.argmin(obj))
            if obj[bi] < best_v:
                best_v = float(obj[bi])
                best_t = float(cands[bi])
    return best_t

def _deterministic_sweep(h, n, dx, top_T, max_dist, t_end, n_sweeps=None):
    c, top_idx = _corr_and_top(h, n, dx, top_T)
    c_top = c[top_idx]
    grads = np.array([_grad_lag(h, k - (n-1), n, dx) for k in top_idx])
    best_c5 = float(c.max())
    sweep_count = 0
    no_imp = 0
    while time.time() < t_end and (n_sweeps is None or sweep_count < n_sweeps):
        sweep_count += 1
        improved = False
        for dist in range(1, max_dist + 1):
            if time.time() > t_end:
                break
            for i in range(n - dist):
                j = i + dist
                d_lo = max(-h[i], h[j] - 1.0) + 1e-14
                d_hi = min(1.0 - h[i], h[j]) - 1e-14
                if d_lo >= d_hi:
                    continue
                b = grads[:, i] - grads[:, j]
                t_opt = _solve_1d(c_top, b, d_lo, d_hi)
                if abs(t_opt) > 1e-13:

def _nesterov_step(h, n, dx, alpha, step, t_end):
    def _smooth_obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

    def _smooth_grad(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = k - (n - 1)
            wk = w[k] * dx
            if sh >= 0:
                m2 = n - sh
                g[:m2] += wk * (1.0 - h[sh:sh+m2])
                if sh > 0:
                    g[sh:] -= wk * h[:n-sh]
            else:
                sp = -sh
                m2 = n - sp
                g[sp:] += wk * (1.0 - h[:m2])
                g[:m2] -= wk * h[sp:sp+m2]
        return g

    f = _smooth_obj(h)
    v = np.zeros(n)
    mu = 0.85
    while time.time() < t_end:
        g = _smooth_grad(h)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new)
        if f_new < f - 1e-18:
            h = h_new
            f = f_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_to_sum(h - step * v, n * 0.5)
            f = _smooth_obj(h)
    return h

def _subgrad_step(h, n, dx, top_T, step, t_end, n_steps=100):
    for _ in range(n_steps):

        c, top_idx = _corr_and_top(h, n, dx, top_T)

        grads = np.array([_grad_lag(h, k - (n-1), n, dx) for k in top_idx])

        if c5_new < c5:
            h = h_new
        else:
            step *= 0.7
            if step < 1e-10:
                break
    return h
