import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_topk(h, alpha, K):
    c = _corr(h)
    n = len(h)
    m = c.max()
    if K >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argpartition(c, -K)[-K:]
    top_c = c[idx]
    s = np.clip(alpha * (top_c - m), -500, 500)
    ps = s - s.max()
    return m + np.log(np.sum(np.exp(ps))) / alpha

def _grad_topk(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    if K >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argpartition(c, -K)[-K:]
    top_c = c[idx]
    s = np.clip(alpha * (top_c - m), -500, 500)
    ps = s - s.max()
    w = np.exp(ps)
    w /= w.sum()
    g = np.zeros(n)
    for jj in range(len(idx)):
        k = idx[jj]
        sh = k - (n - 1)
        wk = w[jj] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_topk(h, alpha, K, steps, step0, t_end):
    n = len(h)
    f = _obj_topk(h, alpha, K)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_topk(h, alpha, K)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_topk(h_new, alpha, K)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.05, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_topk(h, alpha, K)
    return h

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = min(time.time() + duration_s, t_end - 5)
    no_improve = 0
    while time.time() < end_t:
        c = _corr(h)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, int(k), n, dx) for k in top_idx])
        a_vals = c[top_idx]
        sweep_improved = False
        n_sweeps = n
        for _ in range(n_sweeps):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j:

            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:

            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(30):

                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:

    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c

            c5n = float(c.max())

                    best_c = c5n
                    best_h = h.copy()
            else:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_topk(h, alpha, K):
    c = _corr(h)
    m = c.max()
    nl = 2 * len(h) - 1
    if K >= nl:
        idx = np.arange(nl)
    else:
        idx = np.argpartition(c, -K)[-K:]
    top_c = c[idx]
    s = np.clip(alpha * (top_c - m), -500, 500)
    ps = s - s.max()
    return m + np.log(np.sum(np.exp(ps))) / alpha

def _grad_topk(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    nl = 2 * n - 1
    if K >= nl:
        idx = np.arange(nl)
    else:
        idx = np.argpartition(c, -K)[-K:]
    top_c = c[idx]
    s = np.clip(alpha * (top_c - m), -500, 500)
    ps = s - s.max()
    w = np.exp(ps)
    w /= w.sum()
    g = np.zeros(n)
    for jj in range(len(idx)):
        k = idx[jj]
        sh = k - (n - 1)
        wk = w[jj] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_topk(h, alpha, K, steps, step0, t_end):
    n = len(h)
    f = _obj_topk(h, alpha, K)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_topk(h, alpha, K)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_topk(h_new, alpha, K)
        if f_new + 1e-15 < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 12.0)
        else:

            if step < 1e-14:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_topk(h, alpha, K)
    return h

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx

        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g
