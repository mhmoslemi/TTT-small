            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top2 = np.argpartition(c, -2)[-2:]
        g = np.zeros(n)
        dx = 2.0 / n
        for kk in top2:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += dx * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= dx * h[:-sh]
            else:
                sp = -sh
                g[sp:] += dx * (1.0 - h[:-sp])
                g[:n - sp] -= dx * h[sp:]
        g *= 0.5
        improved = False
        for fac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            hn = _proj(h - step * fac * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5 - 1e-18:
                h = hn
                c5 = c5n
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            step *= 0.75
            if step < 1e-11:
                step = step0 * 0.08
    return best_h, best_c5

def _block_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        ws = rng.choice([2, 3, 5, 7, 10, 15])
        i0 = rng.randint(0, n - ws)
        seg = h[i0:i0+ws].copy()
        t = np.linspace(-1, 1, ws)
        bump = np.exp(-1.5*t**2)
        d = 0.004 * max(T, 1e-9)**0.45 * rng.randn() * bump
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+ws] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 4e-6
    stall = 0
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                stall = 0
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
                step = min(step * 1.08, 6e-5)
            else:
                h[i], h[j] = ni - d, nj + d
                stall += 1
                if stall > 50:
                    step *= 0.7
                    stall = 0
                if step < 5e-9:
                    step = 4e-6
    return bh, bc

def _interpolate(h, new_n):
    n = len(h)
    old_x = np.linspace(0, 2, n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: p-norm gradient with moderate p (broad smoothing)
    dl1 = min(deadline, t0 + 120)
    for p, tk, st in [(4, 60, 0.012), (8, 35, 0.008), (16, 20, 0.006)]:
        if time.time() > dl1:
            break
        hg = _proj_grad(best_h, p, tk, 1500, st, dl1)
        cv = _eval(hg)
        if cv < best_c5:
            best_c5 = cv
            best_h = hg.copy()

    # Phase 2: Worst-2-lag pursuit (targeted)
    dl2 = min(deadline, t0 + 320)
    hpl, c5pl = _worst2_pursuit(best_h.copy(), 25000, dl2, 0.004)
    if c5pl < best_c5:
        best_c5 = c5pl
        best_h = hpl.copy()

    # Phase 3: High p-norm (near-exact max)
    dl3 = min(deadline, t0 + 520)
    for p, tk, st in [(40, 10, 0.003), (100, 5, 0.002), (200, 3, 0.001)]:
        if time.time() > dl3:
            break
        hg = _proj_grad(best_h, p, tk, 2500, st, dl3)
        cv = _eval(hg)
        if cv < best_c5:
            best_c5 = cv
            best_h = hg.copy()

    # Phase 4: Worst-2 pursuit with smaller steps
    dl4 = min(deadline, t0 + 700)
    hpl, c5pl = _worst2_pursuit(best_h.copy(), 30000, dl4, 0.001)
    if c5pl < best_c5:
        best_c5 = c5pl
        best_h = hpl.copy()

    # Phase 5: Try interpolation to nearby n for diversity
    if time.time() < t0 + budget_s - 80:
        for try_n in [n0 - 8, n0 + 8]:
            if try_n < 100 or try_n >= 1000:
                continue
            h_try = _interpolate(best_h, try_n)
            h_try = _repair(h_try, try_n)
            hg = _proj_grad(h_try, 20, 10, 800, 0.005, min(deadline, t0 + budget_s - 40))
            hg = _repair(hg, try_n)
            cv = _eval(hg)
            if cv < best_c5:
                # Interpolate back to original n for consistency
                if try_n != n0:
                    hg = _interpolate(hg, n0)
                    hg = _repair(hg, n0)
                    cv = _eval(hg)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = hg.copy()
            if time.time() > deadline:
                break

    # Phase 6: Structured annealing
    dl5 = min(deadline, t0 + budget_s - 40)
    ha, c5a = _block_anneal(best_h.copy(), 25000, rng, dl5, 0.003, 1e-10)
    if c5a < best_c5:
        best_c5 = c5a
        best_h = ha.copy()

    # Phase 7: Final fine polish with adaptive steps
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0012), 60000)
    best_h, best_c5 = _fine_polish(best_h, n_pol, rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
