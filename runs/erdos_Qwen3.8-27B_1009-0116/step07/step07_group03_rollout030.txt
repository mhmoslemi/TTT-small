                if d_lo < d_cross < d_hi:

    for d in candidates:
        val = np.max(c_top + d * gs)

def _grad_lag(h, sh, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if sh == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif sh > 0:

        g[:n - sp] -= dx * h[sp:]

def _pair_opt(h, i, j, top_k, c, n, T):
