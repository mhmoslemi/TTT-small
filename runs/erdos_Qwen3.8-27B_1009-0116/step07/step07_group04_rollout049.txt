    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > limit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = (rng.rand() * 2.0 - 1.0) * 5e-6
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _block_swap(h, n_tries, rng, t0, limit):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > limit:
            break
        blk = rng.randint(2, 6)
        ia = rng.randint(0, n - blk)
        ib = rng.randint(0, n - blk)
        if abs(ia - ib) < blk:
            continue
        scale = rng.rand() * 0.005
        da = scale * rng.randn(blk)
        da -= da.mean() * 0.5
        na = h[ia:ia+blk] + da
        nb = h[ib:ib+blk] - da
        if np.all(na >= 0) and np.all(na <= 1) and np.all(nb >= 0) and np.all(nb <= 1):
            h[ia:ia+blk] = na
            h[ib:ib+blk] = nb
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[ia:ia+blk] = na - da
                h[ib:ib+blk] = nb + da
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Nesterov k=8
    h, c5v = _nesterov(h.copy(), 8, 6000, 0.0015, t0, 80)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 2: Nesterov k=20
    h, c5v = _nesterov(best_h.copy(), 20, 4000, 0.0008, t0, 180)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 3: SLSQP with increasing alpha
    for alpha in [8000, 60000, 300000]:
        if time.time() - t0 > 450:
            break
        hc = _slsqp_opt(best_h.copy(), alpha, 2000, t0, 450)
        c5v = _eval(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc.copy()

    # Phase 4: Multi-scale escape
    if time.time() - t0 < budget_s - 200:
        n_lo = max(n0 // 4, 64)
        f = n0 // n_lo
        h_lo = h[:n_lo * f].reshape(n_lo, f).mean(axis=1).copy()
        h_lo = _repair(h_lo, n_lo)
        for kt in [5, 15, 30]:
            h_lo, _ = _nesterov(h_lo, kt, 2000, 0.003, t0, budget_s - 100)
        h_hi = np.repeat(h_lo, f)[:n0]
        h_hi = _repair(h_hi, n0)
        h_hi, c5v2 = _nesterov(h_hi, 15, 3000, 0.001, t0, budget_s - 60)
        c5v = _eval(h_hi)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_hi.copy()

    # Phase 5: High-alpha SLSQP
    if time.time() - t0 < budget_s - 120:
        hc = _slsqp_opt(best_h.copy(), 800000, 1500, t0, budget_s - 100)
        c5v = _eval(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc.copy()

    # Phase 6: Final Nesterov polish
    if time.time() - t0 < budget_s - 60:
        h, c5v = _nesterov(best_h.copy(), 40, 2500, 0.0003, t0, budget_s - 40)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 7: Targeted medium local search
    if time.time() - t0 < budget_s - 30:
        ntries = int((budget_s - 30 - (time.time() - t0)) * 250)
        best_h, best_c5 = _targeted_2pt(best_h, min(ntries, 50000), rng, t0, budget_s - 15)

    # Phase 8: Block swap search
    if time.time() - t0 < budget_s - 15:
        ntries = int((budget_s - 15 - (time.time() - t0)) * 200)
        best_h, best_c5 = _block_swap(best_h, min(ntries, 20000), rng, t0, budget_s - 8)

    # Phase 9: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 3:
        ntries = int((budget_s - 3 - (time.time() - t0)) * 400)
        best_h, best_c5 = _fine_2pt(best_h, min(ntries, 40000), rng, t0, budget_s - 1)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
