                cn = c.max()
                if cn < best_c5:
                    best_c5 = cn; best_d = d_opt
                else:
                    h[i] -= d_opt; h[j] += d_opt
                    c -= d_opt*g
                    if 0 <= k1 < 2*n-1: c[k1] -= d_opt*d_opt
                    if 0 <= k2 < 2*n-1: c[k2] -= d_opt*d_opt
            if best_d != 0.0:
                improved = True
        c5_cur = c.max()
        if c5_cur < best_c:
            best_c = c5_cur; best_h = h.copy()
        if not improved and sw > 5: break
    return best_h, best_c

def _polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy(); best_c = c5_cur
    step_sz = 5e-5
    for it in range(1000000):
        if time.time() > deadline: break
        step_sz *= (1.0 - 1e-9)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = step_sz * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    # Try perturbations
    for ps in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
        hp = _proj_to_sum(np.clip(h + ps*rng.randn(n0), 0, 1), n0/2.0)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()
    # Phase 1: Frank-Wolfe (25%)
    dl1 = t0 + budget_s * 0.25
    h1, c1 = _fw_opt(best_h.copy(), rng, t0, dl1, n_act=5)
    if c1 < best_c5: best_c5, best_h = c1, h1
    # Phase 2: FW with more active lags (15%)
    dl2 = t0 + budget_s * 0.40
    h2, c2 = _fw_opt(best_h.copy(), rng, t0, dl2, n_act=10)
    if c2 < best_c5: best_c5, best_h = c2, h2
    # Phase 3: Directed pair CD (30%)
    dl3 = t0 + budget_s * 0.70
    h3, c3 = _pair_cd(best_h.copy(), rng, t0, dl3, 200000)
    if c3 < best_c5: best_c5, best_h = c3, h3
    # Phase 4: Sweep CD (10%)
    dl4 = t0 + budget_s * 0.80
    h4, c4 = _sweep_cd(best_h.copy(), rng, t0, dl4, 100)
    if c4 < best_c5: best_c5, best_h = c4, h4
    # Phase 5: Pair CD again (10%)
    dl5 = t0 + budget_s * 0.90
    h5, c5v = _pair_cd(best_h.copy(), rng, t0, dl5, 100000)
    if c5v < best_c5: best_c5, best_h = c5v, h5
    # Phase 6: Final polish (last 5%)
    dl6 = t0 + budget_s - 5
    h6, c6 = _polish(best_h.copy(), rng, t0, dl6)
    if c6 < best_c5: best_c5, best_h = c6, h6
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
