def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_i(h, i, n):
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - i)
    hi1 = min(2 * n - 2, 2 * n - 2 - i)

    if lo2 <= hi2:
        idx2 = (n - 1) + i - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * dx

            break
        if it > 10000 and stable > 8000:
            break

        j = rng.randint(n)

        k1 = n - 1 + d0
        k2 = n - 1 - d0
        
        g = _grad_i(h, i, n) - _grad_i(h, j, n)

        d_hi = min(1.0 - h[i], h[j])
