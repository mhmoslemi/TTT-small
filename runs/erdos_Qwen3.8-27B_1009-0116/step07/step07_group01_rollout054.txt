import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} via bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_single(h, s):
    """Gradient of correlation at shift s (0 <= s <= n-1)."""
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    return g

def _topk_grad(h, k, alpha):
    """Top-k smoothed-max gradient."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Convert to shifts
    shifts = np.zeros(2*n-1, dtype=int)
    for kk in range(n):
        shifts[kk] = kk
    for kk in range(n, 2*n-1):
        shifts[kk] = kk - n + 1
    # Top-k indices
    top_idx = np.argsort(c)[-k:]
    # Softmax weights
    ct = c[top_idx]
    s = alpha * (ct - ct.max())
    w = np.exp(s); w /= w.sum()
    # Gradient
    dx = 2.0 / n
    g = np.zeros(n)
    for ii, kk in enumerate(top_idx):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_single(h, shifts[kk])
    return dx * g, c.max() * dx

def _grad_descent(h, alpha, n_iter, k, base_step, elapsed_fn):
    """Projected gradient descent with top-k smooth max."""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    v = np.zeros(n)
    mu = 0.7
    step = base_step
    for it in range(n_iter):
        if elapsed_fn() > 0:
            break
        g, cur_c = _topk_grad(h, k, alpha)
        v = mu * v + (1 - mu) * g
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        # Line search
        taken = False
        for _ in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                step = min(step * 1.1, base_step * 10)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = max(step, base_step * 0.1)
            step = min(step * 1.3, base_step * 10)
    return best_h, best_c
