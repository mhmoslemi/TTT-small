def lse_gradient(h, beta):
    """Gradient of log-sum-exp objective."""
    n = len(h)
    c = get_corr(h)
    cm = c.max()
    weights = np.exp(beta * (c - cm))
    W = weights.sum()
    dx = 2.0 / n
    
    g = np.zeros(n)
    # For each lag k, add (dw_k/dh[i]) to g[i]
    # dw_k/dh[i] = weights[k] * (2/n) * [contribution patterns]
    
    # Vectorized: for each shift value
    # This is complex. Let me use the per-position approach.
    for k in range(2 * n - 1):
        w = weights[k] / W
        if w < 1e-14:
            continue

        if shift >= 0:
            L = n - shift
            if L <= 0:
                continue
            # d c_k / d h[i] = (2/n) * [(1-h[i+shift]) - h[i-shift]] (with valid range checks)
            # i in [0, L-1]: + (1-h[i+shift])
            # i in [shift, n-1]: - h[i-shift]
            idx1 = np.arange(L)
            g[idx1] += w * dx * (1.0 - h[idx1 + shift])
            idx2 = np.arange(shift, n)
            g[idx2] -= w * dx * h[idx2 - shift]
        else:
            s = -shift
            L = n - s
            if L <= 0:
                continue
            idx1 = np.arange(s, n)
            g[idx1] += w * dx * (1.0 - h[idx1 - s])
            idx2 = np.arange(L)
            g[idx2] -= w * dx * h[idx2 + s]
    
    return g

    # Phase 2: Focused CD - main optimization (~500s)
    dl2 = t0 + min(550, budget_s * 0.50)
    h2, c5_2 = _focused_cd(best_h.copy(), 5000000, rng, dl2, top_m=3)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Standard CD with more iterations (~200s)
    dl3 = t0 + min(780, budget_s * 0.72)
    h3, c5_3 = _pair_cd(best_h.copy(), 5000000, rng, dl3, top_m=4)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Another focused CD round with different top_m (~150s)
    dl4 = t0 + min(960, budget_s * 0.87)
    h4, c5_4 = _focused_cd(best_h.copy(), 5000000, rng, dl4, top_m=5)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Fine polish (~130s)
    dl5 = deadline - 3
    h5, c5_5 = _fine_polish(best_h.copy(), rng, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Final repair and verify
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, len(best_h))
