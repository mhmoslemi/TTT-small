            break
        t += 1
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        tk = np.argsort(c)[-kt:]
        lags = [int(kk) for kk in tk]
        g = _grad_lags(h, lags, n) * (2.0 / n) / len(lags)
        lrt = lr / (1.0 + 0.005 * t)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        for sc in [1.0, 0.4, 0.1, 0.01]:
            hn = _project(h - sc * lrt * mh / (np.sqrt(vh) + 1e-8), n * 0.5)
            c5n = _eval(hn)
            if c5n < c5c - 1e-16:
                c5c = c5n
                h = hn
                break
    return h, c5c

def _sa(h, n, steps, t_init, t_fin, kt, t0, budget_s, rng):
    c5c = _eval(h)
    tc = t_init
    cool = (t_fin / t_init) ** (1.0 / max(steps, 1))
    for _ in range(steps):
        if time.time() - t0 > budget_s - 30:
            break
        tc *= cool
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        tk = np.argsort(c)[-kt:]
        g = _grad_lags(h, [int(kk) for kk in tk], n) * (2.0 / n) / kt
        hn = _project(h - 0.03 * g + 0.015 * rng.randn(n), n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(tc, 1e-10)):
            c5c = c5n
            h = hn
    return h, c5c

def _obj_smax(x, n, temp):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip((c - m) / temp, -50, 50)
    return m + temp * np.log(np.sum(np.exp(sc)))

def _grad_smax(x, n, temp):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip((c - m) / temp, -50, 50)
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        wk = w[k] * (2.0 / n)
        if s >= 0:
            g[s:] += wk * (1.0 - x[:n - s])
            g[:n - s] -= wk * x[s:]
        else:
            t = -s
            g[:n - t] += wk * (1.0 - x[t:])
            g[t:] -= wk * x[:n - t]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _slsqp(h, n, temp, maxiter, t0, budget_s):
    if time.time() - t0 > budget_s - 60:
        return h
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_smax, h, args=(n, temp), jac=_grad_smax,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _sharpen(h, n, alpha):
    h2 = h.copy()
    pos = h2 > 0.5
    h2[pos] += alpha * (1.0 - h2[pos])
    h2[~pos] -= alpha * h2[~pos]
    return _repair(h2, n)

def _fine3(h, n, ntry, rng, c5c, t0, budget_s):
    best = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if time.time() - t0 > budget_s - 15:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 5)) % n
        d1 = 4e-5 * rng.randn()
        d2 = 4e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < best_c:
                best_c = c5n
                best = h.copy()
            else:
                h[i], h[j], h[k] = best[i], best[j], best[k]
    return best, best_c

def _fine2(h, n, ntry, rng, c5c, t0, budget_s):
    best = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if time.time() - t0 > budget_s - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c:
                best_c = c5n
                best = h.copy()
            else:
                h[i], h[j] = best[i], best[j]
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam with diverse hyperparameters
    configs = [(0.06, 0.8, 0.99, 3), (0.04, 0.9, 0.999, 5),
               (0.025, 0.85, 0.99, 7), (0.015, 0.9, 0.999, 4)]
    for lr, b1, b2, kt in configs:
        h2, c52 = _adam(best_h.copy(), n0, 700, lr, b1, b2, kt, t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: SLSQP with smooth max (multiple temperatures)
    for temp in [0.015, 0.008, 0.025]:
        hr = _slsqp(best_h.copy(), n0, temp, 400, t0, budget_s)
        c5r = _eval(hr)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hr

    # Phase 3: Simulated annealing for exploration
    h2, c52 = _sa(best_h.copy(), n0, 2000, 0.04, 0.002, 5, t0, budget_s, rng)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2

    # Phase 4: Post-anneling Adam polish
    for kt in [3, 5, 4, 6]:
        h2, c52 = _adam(best_h.copy(), n0, 500, 0.01, 0.9, 0.999, kt, t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Sharpen toward binary + re-optimize
    for alpha in [0.08, 0.15, 0.05]:
        h_sh = _sharpen(best_h, n0, alpha)
        h2, c52 = _adam(h_sh, n0, 400, 0.015, 0.85, 0.99, 3, t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Shifted + noised restarts
    for shift in [1, 2, 3, 7, 13, 29, 50]:
        h_sh = np.roll(best_h, shift).copy()
        h_sh += 0.003 * rng.randn(n0)
        h_sh = _repair(h_sh, n0)
        h2, c52 = _adam(h_sh, n0, 350, 0.025, 0.85, 0.99, 3, t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: Final SLSQP
    hr = _slsqp(best_h.copy(), n0, 0.01, 350, t0, budget_s)
    c5r = _eval(hr)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hr

    # Phase 8: Fine 3-point search
    nt = min(int((budget_s - 20 - (time.time() - t0)) / 0.004), 60000)
    if nt > 100:
        best_h, best_c5 = _fine3(best_h, n0, nt, rng, best_c5, t0, budget_s)

    # Phase 9: Fine 2-point ultra polish
    nt = min(int((budget_s - 3 - (time.time() - t0)) / 0.003), 60000)
    if nt > 100:
        best_h, best_c5 = _fine2(best_h, n0, nt, rng, best_c5, t0, budget_s)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
