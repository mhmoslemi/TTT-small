    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-10:
        h = _proj_sum(h, n * 0.5)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_p(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return (np.sum(c ** p)) ** (1.0 / p) + pen * (h.sum() - n * 0.5) ** 2

def _grad_p(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_safe = np.maximum(c, 1e-30)
    cp = c_safe ** (p - 1)
    tot = cp.sum()
    wf = tot ** (1.0 / p - 1.0) * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = wf * cp[k]
        if wk < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
        else:
            t = -s
            g[t:] += wk * (1.0 - h[:n - t])
            g[:n - t] -= wk * h[t:]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, p, pen, iters):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_p, h, jac=_grad_p, args=(p, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': iters, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _micro_search(h, max_iters, rng, el, bc):
    n = len(h)
    h_c = h.copy()
    c_best = _eval(h_c)
    h_best = h_c.copy()
    scale = 1e-4
    for i in range(max_iters):
        if bc():
            break
        i1 = rng.randint(n)
        i2 = (i1 + 1 + rng.randint(0, 9)) % n
        d = rng.randn() * scale
        v1 = h_c[i1] + d
        v2 = h_c[i2] - d
        if v1 < 0 or v1 > 1 or v2 < 0 or v2 > 1:
            continue
        h_c[i1] = v1
        h_c[i2] = v2
        c_new = _eval(h_c)
        if c_new < c_best:
            c_best = c_new
            h_best = h_c.copy()
            scale = min(scale * 1.01, 3e-3)
        else:
            h_c[i1] = v1 - d
            h_c[i2] = v2 + d
            if i % 5000 == 0:
                scale *= 0.82
    return h_best, c_best

def _3pt_search(h, max_iters, rng, el, bc):
    n = len(h)
    h_c = h.copy()
    c_best = _eval(h_c)
    h_best = h_c.copy()
    scale = 4e-5
    for i in range(max_iters):
        if bc():
            break
        i1 = rng.randint(n)
        i2 = (i1 + 1 + rng.randint(0, 5)) % n
        i3 = (i2 + 2 + rng.randint(0, 10)) % n
        d1 = rng.randn() * scale
        d2 = rng.randn() * scale
        v1 = h_c[i1] + d1
        v2 = h_c[i2] + d2
        v3 = h_c[i3] - d1 - d2
        if 0 <= v1 <= 1 and 0 <= v2 <= 1 and 0 <= v3 <= 1:
            h_c[i1], h_c[i2], h_c[i3] = v1, v2, v3
            c_new = _eval(h_c)
            if c_new < c_best:
                c_best = c_new
                h_best = h_c.copy()
            else:
                h_c[i1] = v1 - d1
                h_c[i2] = v2 - d2
                h_c[i3] = v3 + d1 + d2
        if i % 8000 == 0:
            scale *= 0.75
    return h_best, c_best

def _nesterov_topk(h, n_iter, k, step0, el, bc):
    n = len(h)
    h_c = h.copy()
    v = np.zeros(n)
    beta = 0.6
    step = step0
    c_best = _eval(h_c)
    h_best = h_c.copy()
    stall = 0
    for _ in range(n_iter):
        if bc():
            break
        c = np.correlate(h_c, 1.0 - h_c, mode='full') * (2.0 / n)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        for ki in topk:
            s = ki - (n - 1)
            if s >= 0:
                g[:n - s] += (1.0 - h_c[s:])
                g[s:] -= h_c[:n - s]
            else:
                t = -s
                g[t:] += (1.0 - h_c[:n - t])
                g[:n - t] -= h_c[t:]
        g *= (2.0 / n) / len(topk)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = beta * v + (1 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < c_best - 1e-16:
            h_c = h_try
            c_best = c_try
            h_best = h_c.copy()
            stall = 0
            step = min(step * 1.03, 0.03)
        else:
            step *= 0.55
            stall += 1
            if step < 1e-9:
                break
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0
    def bc():
        return time.time() - t0 > budget_s - 8

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS moderate p
    for p in [16, 64, 256]:
        if el() > 100:
            break
        ht = _lbfgs(best_h.copy(), p, 1e5, 350)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 2: L-BFGS high p
    for p in [1024, 4096]:
        if el() > 280:
            break
        ht = _lbfgs(best_h.copy(), p, 1e7, 250)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 3: Nesterov top-k
    if el() < budget_s - 350:
        ht, ct = _nesterov_topk(best_h, 6000, 5, 0.01, el, bc)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 4: 3-point search
    if el() < budget_s - 300:
        best_h, best_c5 = _3pt_search(best_h, 150000, rng, el, bc)

    # Phase 5: Very fine 2-point search (main workhorse)
    if el() < budget_s - 40:
        best_h, best_c5 = _micro_search(best_h, 20000000, rng, el, bc)

    # Phase 6: Final L-BFGS polish
    if el() < budget_s - 5:
        ht = _lbfgs(best_h.copy(), 2048, 1e7, 120)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
