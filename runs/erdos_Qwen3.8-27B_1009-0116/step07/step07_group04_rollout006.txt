            g[i1] += wk * (1.0 - x[i1-sp])
            i2 = np.arange(n-sp)
            g[i2] -= wk * x[i2+sp]

Option 1: Use the equality constraint with SLSQP (slow but exact)
Option 2: Use L-BFGS-B with a penalty (fast but approximate)
Option 3: Parameterize to avoid the constraint (e.g., optimize n-1 variables, derive the last)
Option 4: Project after each L-BFGS-B step

Option 4 is best: run L-BFGS-B with only bounds, then project after. But L-BFGS-B doesn't support callback-based projection mid-optimization.

Actually, Option 2 with a large penalty should work well enough. The sum constraint is just one scalar, so a strong penalty will keep it close, and then we repair at the end.

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_softmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.exp(np.clip(beta * (c - m), -500, 0))
    w = s / s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_soft(h, t0, deadline, beta0=100.0, steps=15000):
    n = len(h)
    target = n * 0.5
    f = eval_c5(h)
    best_h = h.copy()
    best_f = f
    v = np.zeros(n)
    mu = 0.88
    step = 0.003
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        prog = min(1.0, it / (steps * 0.7))
        beta = beta0 * (80.0 ** prog)
        g = _grad_softmax(h, beta)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = proj_sum(h - step * v, target)
        f_new = eval_c5(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.015, 0.01)
            stall = 0
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            step *= 0.6
            if step < 1e-10:
                break
            stall += 1
            if stall > 100:
                break
    return best_h, best_f

def _lbfgs_smooth(h, beta, maxiter, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    pen = 5e4

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.exp(np.clip(beta * (c - m), -500, 0))
        base = m + np.log(s.sum()) / beta
        return base + pen * (x.sum() - target) ** 2

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.exp(np.clip(beta * (c - m), -500, 0))
        w = s / s.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        g += 2.0 * pen * (x.sum() - target)
        return g

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-9})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def _focused_descend(h, n_iter, deadline):
    n = len(h)
    c5c = eval_c5(h)
    best_h = h.copy()
    best_c = c5c
    c_all = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_k = np.argsort(c_all)[-8:]
    for it in range(n_iter):
        if time.time() > deadline:
            break
        k = top_k[it % len(top_k)]
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            d = (1.0 - h[i1 + sh]) - h[i1]
            direction = np.zeros(n)
            direction[i1] += d
            if sh > 0:
                direction[np.arange(sh, n)] -= h[np.arange(sh, n) - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            direction = np.zeros(n)
            direction[i1] += (1.0 - h[i1 - sp])
            direction[np.arange(n - sp)] += -h[np.arange(n - sp) + sp]
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            continue
        direction /= norm
        for sgn in [1, -1]:
            for scale in [5e-5, 2e-5, 1e-5]:
                h_try = proj_sum(h + sgn * scale * direction, n * 0.5)
                c5n = eval_c5(h_try)
                if c5n < c5c:
                    h = h_try
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                    break
            else:
                continue
            break
        if it % 200 == 0:
            c_all = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top_k = np.argsort(c_all)[-8:]
    return best_h, best_c

def _micro_search(h, n_tries, rng, t0, deadline):
    n = len(h)
    c5c = eval_c5(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = eval_c5(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
