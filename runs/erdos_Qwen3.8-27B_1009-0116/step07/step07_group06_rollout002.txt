        hp = np.clip(h + ps * rng.randn(n0), 0, 1)
        hp = _proj_to_sum(hp, n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Also try interpolated n=480
    h480 = _interp_to_n(best_h, 480)
    c480 = _eval(h480)
    if c480 < best_c5:
        best_c5 = c480
        best_h = h480.copy()

    # Phase 1: Pair CD (40% budget)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _pair_cd(best_h.copy(), 300000, rng, dl1, lag_bias=0.45)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Top-gradient descent (10% budget)
    dl2 = t0 + budget_s * 0.40
    h2, c5_2 = _topgrad_cd(best_h.copy(), rng, dl2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Pair CD again (20% budget)
    dl3 = t0 + budget_s * 0.60
    h3, c5_3 = _pair_cd(best_h.copy(), 250000, rng, dl3, lag_bias=0.55)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Perturbation + re-optimize (20% budget)
    n_per = len(best_h)
    sub_dl = dl3 + (t0 + budget_s * 0.80 - dl3) / 3.0
    for pert_scale in [0.04, 0.08, 0.15]:
        if time.time() > t0 + budget_s * 0.80:
            break
        sub_end = min(t0 + budget_s * 0.80, time.time() + (t0 + budget_s * 0.80 - time.time()) / 3.0)
        hp, cp = _perturb_optimize(best_h.copy(), rng, sub_end, pert_scale)
        if cp < best_c5:
            best_c5, best_h = cp, hp

    # Phase 5: Final pair CD polish (10%)
    dl5 = t0 + budget_s * 0.92
    h5, c5_5 = _pair_cd(best_h.copy(), 150000, rng, dl5, lag_bias=0.3)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Fine polish
    dl6 = dead_end - 5
    h6, c5_6 = _fine_polish(best_h.copy(), rng, dl6)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
