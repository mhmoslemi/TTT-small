    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax(x, alpha, topk), h,
                     jac=lambda x: _grad(x, alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        mode = rng.random()
        hn = h.copy()
        if mode < 0.35:
            bs = rng.randint(2, 12)
            i0 = rng.randint(0, max(1, n - bs))
            d = 0.008 * max(T, 1e-9)**0.4 * rng.randn(bs)
            d -= d.mean()
            hn[i0:i0+bs] = np.clip(hn[i0:i0+bs] + d, 0, 1)
        elif mode < 0.65:
            bs = rng.randint(10, 50)
            i0 = rng.randint(0, max(1, n - bs))
            d = 0.02 * max(T, 1e-9)**0.3 * rng.randn(bs)
            d -= d.mean()
            hn[i0:i0+bs] = np.clip(hn[i0:i0+bs] + d, 0, 1)
        elif mode < 0.85:
            i = rng.randint(n)
            hn[i] = np.clip(hn[i] + 0.04 * rng.randn(), 0, 1)
        else:
            i = rng.randint(n)
            bs = rng.randint(1, 5)
            for j in range(bs):
                jj = (i + rng.randint(0, 6)) % n
                hn[jj] = np.clip(hn[jj] + 0.02 * rng.randn(), 0, 1)
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _fine_cd(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_size = 1e-4
    for it in range(niter):
        if time.time() > deadline:
            break
        # Adaptive step: shrink if no progress in last 500 iters
        if it > 0 and it % 500 == 0:
            step_size *= 0.6
        if rng.random() < 0.4:
            i = rng.randint(n)
            d = step_size * rng.randn()
            ni = h[i] + d
            if 0 <= ni <= 1:
                h[i] = ni
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                else:
                    h[i] = ni - d
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            d = step_size * 0.6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _multi_start_polish(h, rng, deadline):
    """Try small random restarts from current best and re-optimize briefly"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    n_restarts = 5
    t_per = max(5, (deadline - time.time()) / (n_restarts + 1))
    for ri in range(n_restarts):
        if time.time() > deadline:
            break
        tl = time.time() + t_per
        h_p = h + 0.005 * rng.randn(n)
        h_p = _repair(h_p, n)
        h_p = _slsqp_do(h_p, 200000, 10, 50, tl)
        c5v = _eval(h_p)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_p.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP sweep (180s)
    dl1 = min(deadline, t0 + 180)
    for alpha, tk, mi in [(5000, 30, 200), (25000, 20, 250), (100000, 12, 250)]:
        if time.time() > dl1:
            break
        h = _slsqp_do(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Nesterov projected gradient (250s)
    dl2 = min(deadline, t0 + 430)
    for alpha, tk, st, ns in [(25000, 25, 0.004, 2000), (100000, 12, 0.002, 2000),
                               (500000, 8, 0.001, 2000), (2000000, 6, 0.0005, 1500)]:
        if time.time() > dl2:
            break
        h = _nest_step(best_h.copy(), alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Annealing (250s)
    dl3 = min(deadline, t0 + 680)
    h_a, c5a = _anneal(best_h.copy(), 35000, rng, dl3, 0.008, 1e-10)
    if c5a < best_c5:
        best_c5 = c5a
        best_h = h_a.copy()

    # Phase 4: Second SLSQP + restarts (150s)
    dl4 = min(deadline, t0 + 830)
    for alpha, tk, mi in [(300000, 8, 150), (1000000, 5, 150)]:
        if time.time() > dl4:
            break
        h = _slsqp_do(best_h.copy(), alpha, tk, mi, dl4)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    if time.time() < dl4 - 10:
        best_h, best_c5 = _multi_start_polish(best_h, rng, dl4)

    # Phase 5: Fine coordinate descent (remaining)
    dl5 = deadline - 2
    remaining = max(0, dl5 - time.time())
    n_cd = min(int(remaining / 0.0015), 60000)
    best_h, best_c5 = _fine_cd(best_h.copy(), n_cd, rng, dl5)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
