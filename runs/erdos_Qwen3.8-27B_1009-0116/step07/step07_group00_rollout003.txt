        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)
        ct = c[idx[len(c)-topk:]]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -300, 300)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        pidx = np.argpartition(c, -topk)
        idx = pidx[len(c)-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 6.0)
        else:
            step *= 0.5
            if step < 1e-9:
                break
            h = _proj(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _slsqp_local(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax(x, alpha, topk), h,
                     jac=lambda x: _grad_smax(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _de_phase(h_init, n, popsize, ngen, F, CR, deadline, rng):
    target = n * 0.5
    pop = np.zeros((popsize, n))
    pop[0] = _repair(h_init, n)
    for i in range(1, popsize):
        strat = rng.randint(4)
        if strat == 0:
            pop[i] = _repair(h_init + 0.08 * rng.randn(n), n)
        elif strat == 1:
            p = h_init.copy()
            for _ in range(4):
                blk = rng.randint(2, 25)
                i0 = rng.randint(0, max(1, n - blk))
                p[i0:i0+blk] = rng.uniform(0, 1, blk)
            pop[i] = _repair(p, n)
        elif strat == 2:
            p = np.where(rng.rand(n) < 0.5 + 0.1 * rng.randn(), 0.0, 1.0)
            p += 0.08 * rng.randn(n)
            pop[i] = _repair(p, n)
        else:
            p = h_init.copy()
            p[rng.choice(n, size=n//4, replace=False)] = rng.uniform(0, 1, n//4)
            pop[i] = _repair(p, n)
    costs = np.array([_eval(pop[i]) for i in range(popsize)])
    bi = costs.argmin()
    best_h, best_c = pop[bi].copy(), costs[bi]
    for gen in range(ngen):
        if time.time() > deadline:
            break
        order = np.argsort(costs)
        ps = pop[order]
        cs = costs[order]
        for i in range(1, popsize):
            a, b, c = rng.choice(popsize, 3, replace=False)
            v = ps[a].copy()
            mask = rng.rand(n) < CR
            v[mask] = ps[a][mask] + F * (ps[b][mask] - ps[c][mask])
            v = _repair(np.clip(v, 0, 1), n)
            cv = _eval(v)
            if cv <= cs[i]:
                ps[i] = v
                cs[i] = cv
        pop = ps
        costs = cs
        bi = costs.argmin()
        if costs[bi] < best_c:
            best_c = costs[bi]
            best_h = pop[bi].copy()
    return best_h, best_c

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc, bh = c5c, h.copy()
    return bh, bc

def _coord_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    steps = [2e-5, 8e-6, 3e-6, 1e-6, -2e-5, -8e-6, -3e-6, -1e-6]
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n // 8)) % n
        if i == j:
            continue
        improved = False
        for d in steps:
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                    improved = True
                    break
                h[i], h[j] = ni - d, nj + d
        if improved and c5c < bc:
            bc, bh = c5c, h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP warm-up
    dl1 = min(deadline, t0 + 60)
    h = _slsqp_local(h, 30000, 20, 400, dl1)
    if _eval(h) < best_c5:
        best_c5, best_h = _eval(h), h.copy()

    # Phase 2: DE exploration
    dl2 = min(deadline, t0 + 280)
    de_h, de_c = _de_phase(best_h, n0, 16, 100, 0.55, 0.85, dl2, rng)
    if de_c < best_c5:
        best_c5, best_h = de_c, de_h.copy()

    # Phase 3: Nesterov high-alpha sweep
    dl3 = min(deadline, t0 + 550)
    for alpha, tk, st, ns in [(80000, 10, 0.003, 2500), (400000, 6, 0.0015, 2500),
                              (2000000, 4, 0.0007, 2500), (10000000, 3, 0.0003, 2000)]:
        if time.time() > dl3:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl3)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 4: SLSQP polish
    dl4 = min(deadline, t0 + 660)
    for alpha, tk, mi in [(100000, 8, 300), (500000, 5, 200)]:
        if time.time() > dl4:
            break
        h = _slsqp_local(best_h, alpha, tk, mi, dl4)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 5: Block annealing
    dl5 = min(deadline, t0 + 820)
    blocks = [1, 2, 3, 4, 5, 7, 10, 14]
    best_h, best_c5 = _block_anneal(best_h.copy(), 60000, rng, dl5, 0.004, 1e-11, blocks)

    # Phase 6: Final Nesterov
    dl6 = min(deadline, t0 + 920)
    for alpha, tk, st, ns in [(8000000, 3, 0.0002, 2500), (40000000, 2, 0.0001, 2000)]:
        if time.time() > dl6:
            break
        h = _nesterov(best_h, alpha, tk, ns, st, dl6)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 7: Coordinate descent
    rem = max(0, deadline - time.time())
    n_cd = min(int(rem / 0.006), 80000)
    best_h, best_c5 = _coord_polish(best_h, n_cd, rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
