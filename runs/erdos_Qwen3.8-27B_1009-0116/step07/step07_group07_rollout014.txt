    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: LogSumExp with decreasing tau (smooth optimization)
    for tau in [0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001]:
        if el() > budget_s * 0.18:
            break
        htry = _lbfgs(best_h.copy(), _lse_obj, _lse_grad, (tau, 1e5), 350)
        ct = _eval(htry)
        if ct < best_c5:
            best_c5 = ct
            best_h = htry

    # Phase 2: p-norm L-BFGS
    for p in [16, 64, 128, 256, 512]:
        if el() > budget_s * 0.35:
            break
        htry = _lbfgs(best_h.copy(), _pnorm_obj, _pnorm_grad, (p, 1e6), 300)
        ct = _eval(htry)
        if ct < best_c5:
            best_c5 = ct
            best_h = htry

    # Phase 3: Nesterov top-k (coarse)
    if el() < budget_s - 160:
        htry, ct = _nesterov(best_h, 12000, 3, 50, 0.012, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = htry

    # Phase 4: Block moves
    if el() < budget_s - 120:
        nt = min(int((budget_s - 120 - el()) * 60), 40000)
        htry, ct = _block_search(best_h, nt, rng, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = htry

    # Phase 5: Nesterov (fine)
    if el() < budget_s - 80:
        htry, ct = _nesterov(best_h, 10000, 2, 30, 0.006, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = htry

    # Phase 6: Point search
    if el() < budget_s - 25:
        nt = min(int((budget_s - 25 - el()) * 180), 140000)
        best_h, best_c5 = _pt_search(best_h, nt, rng, el, bc)

    # Phase 7: Final Nesterov polish
    if el() < budget_s - 5:
        htry, ct = _nesterov(best_h, 4000, 2, 15, 0.003, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = htry

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
