import numpy as np
import time

def _repair(h, n):
    # Keep h in [0,1] and sum(h) = n/2
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) > 1e-12:
        # Distribute diff
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 0:
                h += (diff / total_room) * room
        else:
            total_h = h.sum()
            if total_h > 0:
                h += (diff / total_h) * h
        h = np.clip(h, 0.0, 1.0)
        # Final exact fix
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff2 > 0:
                    amt = min(diff2, 1.0 - h[idx])
                    h[idx] += amt
                    diff2 -= amt
                else:
                    amt = min(-diff2, h[idx])
                    h[idx] -= amt
                    diff2 += amt
                if abs(diff2) < 1e-13:
                    break
    return h

    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    n_lag = 2 * n - 1
    improved = False
    for i in range(n):
        if check_fn and check_fn():
            break

        b = np.zeros(n_lag)
        sl, sh = -i, n - 1 - i
        if sh >= sl:

        sl2, sh2 = i-(n-1), i
        if sh2 >= sl2:
            b[n-1+sl2 : n-1+sh2+1] -= h[i+sl2 : i+sh2+1]
        
        old = h[i]
        a = corr - b * old

        f0 = a.max()
        f1 = (a + b).max()
        bx, bf = 0.0, f0
        if f1 < bf:

        lo, hi = 0.0, 1.0
        for _ in range(30):

            v2 = (a + b * (hi - d)).max()
            if v1 <= v2:

                if v1 < bf: bf, bx = v1, lo + d
            else:
                lo += d
                if v2 < bf: bf, bx = v2, hi - d
            if hi - lo < 1e-11:

            h[i] = bx
            corr += b * (bx - old)
            improved = True
    return h, corr, improved

- Let x = h[i], y = c - x
- Objective: max_k (a_k' + d_k * x) where a_k' = a_k + b_k^j * c, d_k = b_k^i - b_k^j
- This is 1D convex, solved by ternary search over x ∈ [max(0, c-1), min(1, c)]

This is elegant! I can do pair coordinate descent with the sum constraint built in.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h = np.clip(h * (target / s), 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    amt = min(diff, 1.0 - h[idx])
                    h[idx] += amt
                    diff -= amt
                else:
                    amt = min(-diff, h[idx])
                    h[idx] -= amt
                    diff += amt
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cd_pass(h, n, corr, t0, budget_s):
    elapsed = time.time() - t0
    if elapsed > budget_s - 25:
        return h, corr, False
    n_lag = 2 * n - 1
    improved = False
    for i in range(n):
        if time.time() - t0 > budget_s - 25:
            return h, corr, improved
        b = np.zeros(n_lag)
        sl, sh = -i, n - 1 - i
        if sh >= sl:
            b[n - 1 + sl : n - 1 + sh + 1] += (1.0 - h[i + sl : i + sh + 1])
        sl2, sh2 = i - (n - 1), i
        if sh2 >= sl2:
            b[n - 1 + sl2 : n - 1 + sh2 + 1] -= h[i + sl2 : i + sh2 + 1]
        old = h[i]
        a = corr - b * old
        f0 = a.max()
        f1 = (a + b).max()
        bx, bf = 0.0, f0
        if f1 < bf:
            bx, bf = 1.0, f1
        lo, hi = 0.0, 1.0
        for _ in range(32):
            d = (hi - lo) / 3.0
            v1 = (a + b * (lo + d)).max()
            v2 = (a + b * (hi - d)).max()
            if v1 <= v2:
                hi -= d
                if v1 < bf:
                    bf, bx = v1, lo + d
            else:
                lo += d
                if v2 < bf:
                    bf, bx = v2, hi - d
            if hi - lo < 1e-12:
                break
        if abs(bx - old) > 1e-12:
            h[i] = bx
            corr += b * (bx - old)
            improved = True
    return h, corr, improved

def _cd_optimize(h, n, max_passes, t0, budget_s):
    c5c = _eval(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    for _ in range(max_passes):
        if time.time() - t0 > budget_s - 25:
            break
        h, corr, imp = _cd_pass(h, n, corr, t0, budget_s)
        nc = _eval(h)
        if nc < c5c:
            c5c = nc
        else:
            h = _repair(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            nc2 = _eval(h)
            if nc2 < c5c:
                c5c = nc2
            if not imp:
                break
    h = _repair(h, n)
    c5c = _eval(h)
    return h, c5c

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_polish(h, n, k_top, rounds, t0, budget_s):
    dx = 2.0 / n
    c5c = _eval(h)
    steps = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4]
    stall = 0
    for _ in range(rounds):
        if time.time() - t0 > budget_s - 25:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(corr)[-k_top:]
        g = np.zeros(n)
        for k in top:
            s = k - (n - 1)
            if s >= 0:
                g[s:] += (1.0 - h[:n - s])
                if s > 0:
                    g[:n - s] -= h[s:]
            else:
                sp = -s
                g[:n - sp] += (1.0 - h[sp:])
                g[sp:] -= h[:n - sp]
        g *= dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _proj(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.6] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 12:
                    steps = steps[:12]
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _resample(h, n_old, n_new):
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]), period=1.0)
    return _repair(h_new, n_new)

def _2pt_local(h, n, ntry, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c - 1e-19:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
