```
    n=len(h)
    if c is None: c=_correl(h)
    m=c.max()

       ct=c[idx]
    else: idx=np.arange(len(c)); ct=c
    s=alpha*(ct-m)
    s-=s.max()
```

```
def _add_lag(g, q, wk, h):

       g[q:] += wk
       g[q:] -= wk*h[:-q]
```

```
def _pair_polish(h, deadline, rng):
    n = len(h)
    h = _repair(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = _eval(h)
    c5 = best_c
    g = np.zeros(n)
    mags = (1e-5, 5e-5, 2e-4, 1e-3, 5e-3)
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 1000 == 0:
            _, g = _smax_fg(h, 30000, 8)
        i = int(rng.randint(n))
        j = (i + int(rng.randint(1, n))) % n
        if i == j:
            continue
        diff = float(g[i] - g[j])
        if abs(diff) > 1e-12:
            sgn = 1.0 if diff > 0.0 else -1.0
        else:
            sgn = 1.0 if rng.random() < 0.5 else -1.0
        base = mags[int(rng.randint(len(mags)))]
        cands = (sgn * base, sgn * base * 0.25, -sgn * base * 0.25, -sgn * base * 0.08)
        bd = 0.0
        bc = c5
        for d in cands:
            a = h[i] + d
            bval = h[j] - d
            if 0.0 <= a <= 1.0 and 0.0 <= bval <= 1.0:
                h[i] = a
                h[j] = bval
                cc = _eval(h)
                if cc < bc - 1e-15:
                    bc = cc
                    bd = d
                h[i] = a - d
                h[j] = bval + d
        if bd != 0.0:
            h[i] += bd
            h[j] -= bd
            c5 = bc
            if c5 < best_c - 1e-15:
                best_c = c5
                best_h = h.copy()
    return _repair(best_h, n), best_c
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog, Bounds


def _proj(x, target):
    x = np.asarray(x, dtype=float)
    if abs(x.sum() - target) < 1e-12 and x.min() >= -1e-12 and x.max() <= 1.0 + 1e-12:
        return np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def _repair(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    h = _proj(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0.0 else np.argsort(h)
        rem = diff
        for idx in order:
            if rem > 0.0:
                a = min(rem, 1.0 - h[idx])
                h[idx] += a
                rem -= a
            else:
                a = min(-rem, h[idx])
                h[idx] -= a
                rem += a
            if abs(rem) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)


def _correl(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))


def _eval(h):
    return float(np.max(_correl(h)))


def _lag_grad(p, h):
    n = len(h)
    dx = 2.0 / n
    q = p - (n - 1)
    g = np.zeros(n)
    if q == 0:
        return dx * (1.0 - 2.0 * h)
    if q > 0:
        g[q:] += dx
        g[q:] -= dx * h[:-q]
        g[:n - q] -= dx * h[q:]
    else:
        r = -q
        g[:n - r] += dx
        g[:n - r] -= dx * h[r:]
        g[r:] -= dx * h[:-r]
    return g


def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = float(c.max())
    L = c.size
    tk = int(topk)
    if tk < L:
        idx = np.argpartition(c, -tk)[-tk:]
        ct = c[idx]
    else:
        ct = c
    s = alpha * (ct - m)
    sm = float(s.max())
    e = np.exp(s - sm)
    return float(m + (sm + np.log(e.sum())) / alpha)


def _smax_fg(h, alpha, topk):
    c = _correl(h)
    m = float(c.max())
    L = c.size
    tk = int(topk)
    if tk < L:
        idx = np.argpartition(c, -tk)[-tk:]
        ct = c[idx]
    else:
        idx = np.arange(L)
        ct = c
    s = alpha * (ct - m)
    sm = float(s.max())
    e = np.exp(s - sm)
    w = e / e.sum()
    f = float(m + (sm + np.log(e.sum())) / alpha)
    g = np.zeros(len(h))
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _lag_grad(int(idx[ii]), h)
    return f, g


def _smax_grad(h, alpha, topk):
    return _smax_fg(h, alpha, topk)[1]


def _pen_obj(x, alpha, topk, mu):
    return _smax_obj(x, alpha, topk) + 0.5 * mu * (x.sum() - 0.5 * len(x)) ** 2


def _pen_grad(x, alpha, topk, mu):
    return _smax_fg(x, alpha, topk)[1] + mu * (x.sum() - 0.5 * len(x)) * np.ones_like(x)


def _nesterov(h, alpha, topk, maxit, step0, deadline):
    n = len(h)
    target = 0.5 * n
    h = _repair(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = _eval(h)
    f, _ = _smax_fg(h, alpha, topk)
    v = np.zeros(n)
    step = float(step0)
    beta = 0.7
    stall = 0
    for _ in range(int(maxit)):
        if time.time() > deadline:
            break
        f, g = _smax_fg(h, alpha, topk)
        d = g - g.mean()
        ng = float(np.dot(d, d))
        if ng < 1e-24:
            break
        y = beta * v + (1.0 - beta) * d
        st = step
        acc = False
        for _ in range(10):
            hn = _proj(h - st * y, target)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f - 1e-15:
                h = hn
                f = fn
                acc = True
                break
            st *= 0.5
        if acc:
            v = y
            step = min(step * 1.03, max(step0 * 30.0, 0.01))
            stall = 0
            cc = _eval(h)
            if cc < best_c - 1e-14:
                best_c = cc
                best_h = h.copy()
        else:
            v = 0.5 * v
            step *= 0.5
            stall += 1
            if stall > 50:
                break
    return _repair(best_h, n), best_c


def _lp_dr(h, deadline, topm=8, maxac=28):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    best = _repair(h, n)
    best_c = _eval(best)
    rs = (0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.4)
    ri = 0
    stuck = 0
    while time.time() < deadline:
        if ri >= len(rs):
            ri = 0
            stuck += 1
            if stuck > 2:
                break
        r = rs[ri]
        c = _correl(best)
        m = float(c.max())
        near = max(1e-7, 0.4 * r * r)
        idxn = np.where(c >= m - near)[0]
        tm = min(int(topm), len(c))
        idx = np.unique(np.concatenate((idxn, np.argpartition(c, -tm)[-tm:])))
        if idx.size > maxac:
            idx = idx[np.argsort(c[idx])[-maxac:]]
        if idx.size == 0:
            break
        A = []
        b = []
        h0 = best
        for p in idx:
            g = _lag_grad(int(p), h0)
            A.append(np.concatenate((g, np.array([-1.0]))))
            b.append(c[p] - float(np.dot(g, h0)))
        A = np.array(A)
        b = np.array(b)
        obj = np.zeros(n + 1)
        obj[-1] = 1.0
        lo = np.maximum(0.0, best - r)
        hi = np.minimum(1.0, best + r)
        lo = np.concatenate((lo, np.array([-2.0])))
        hi = np.concatenate((hi, np.array([2.0])))
        Aeq = np.zeros((1, n + 1))
        Aeq[0, :n] = 1.0
        bq = np.array([target])
        res = None
        try:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=bq,
                          bounds=Bounds(lo, hi), method='highs')
        except Exception:
            try:
                res = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=bq,
                              bounds=Bounds(lo, hi))
            except Exception:
                res = None
        if res is None or not getattr(res, 'success', False):
            ri += 1
            continue
        hlp = _proj(np.asarray(res.x[:n], dtype=float), target)
        h0 = best
        local_best = h0
        local_bc = best_c
        for th in (1.0, 0.85, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05):
            ht = h0 + th * (hlp - h0)
            ht = _repair(ht, n)
            ct = _eval(ht)
            if ct < local_bc - 1e-13:
                local_bc = ct
                local_best = ht
        if local_bc < best_c - 1e-13:
            best = local_best
            best_c = local_bc
            stuck = 0
            if ri + 1 < len(rs):
                ri += 1
            else:
                ri = 0
        else:
            ri += 1
    return _repair(best, n), _eval(best)


def _binary_round(h):
    n = len(h)
    k = int(round(0.5 * n))
    q = np.zeros(n)
    q[np.argsort(h)[::-1][:k]] = 1.0
    return _repair(q, n)


def _perturb(h, rng, amp, nblocks=1):
    n = len(h)
    h = h.copy()
    for _ in range(nblocks):
        bl = int(rng.choice((4, 8, 16, 32, 64)))
        i = int(rng.randint(0, max(1, n - bl + 1)))
        d = amp * rng.randn(bl)
        d -= d.mean()
        h[i:i + bl] += d
    return _repair(h, n)


def _down(h, g):
    n = len(h)
    m = n // g
    if m <= 0:
        return h.copy()
    return h[:m * g].reshape(m, g).mean(axis=1)


def _up(h, n):
    if len(h) == n:
        return h.copy()
    idx = np.linspace(0, len(h) - 1, n).round().astype(int)
    return h[idx]


def _downup_start(h, rng, amp):
    g = int(rng.choice((2, 4, 8)))
    low = _down(h, g)
    low = low + amp * rng.randn(low.size)
    low = _repair(low, len(low))
    return _repair(_up(low, len(h)), len(h))


def _default_h(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = (0.5
         + 0.25 * np.sin(3.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
         + 0.10 * np.cos(5.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi)))
    h = np.clip(h, 0.0, 1.0)
    return _repair(h, n)


def _pair_polish(h, deadline, rng):
    n = len(h)
    if n < 2:
        return _repair(h, n), _eval(h)
    h = _repair(np.asarray(h, dtype=float), n)
    best_h = h.copy()
    best_c = _eval(h)
    c5 = best_c
    g = np.zeros(n)
    mags = (1e-5, 5e-5, 2e-4, 1e-3, 5e-3)
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 1000 == 0:
            _, g = _smax_fg(h, 30000, 8)
        i = int(rng.randint(n))
        j = (i + int(rng.randint(1, n))) % n
        if i == j:
            continue
        diff = float(g[i] - g[j])
        if abs(diff) > 1e-12:
            sgn = 1.0 if diff > 0.0 else -1.0
        else:
            sgn = 1.0 if rng.random() < 0.5 else -1.0
        base = mags[int(rng.randint(len(mags)))]
        cands = (sgn * base, sgn * base * 0.25, -sgn * base * 0.25, -sgn * base * 0.08)
        bd = 0.0
        bc = c5
        for d in cands:
            a = h[i] + d
            bval = h[j] - d
            if 0.0 <= a <= 1.0 and 0.0 <= bval <= 1.0:
                h[i] = a
                h[j] = bval
                cc = _eval(h)
                if cc < bc - 1e-15:
                    bc = cc
                    bd = d
                h[i] = a - d
                h[j] = bval + d
        if bd != 0.0:
            h[i] += bd
            h[j] -= bd
            c5 = bc
            if c5 < best_c - 1e-15:
                best_c = c5
                best_h = h.copy()
    return _repair(best_h, n), best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)
    try:
        base = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if base.size < 8:
            raise ValueError
    except Exception:
        base = _default_h(320, rng)
    n = int(len(base))
    best_h = _repair(base, n)
    best_c = _eval(best_h)

    dl = min(t_end - 360.0, t0 + 180.0)
    if dl > time.time() + 5.0:
        dh, dc = _lp_dr(best_h, dl, 8, 24)
        if dc < best_c:
            best_h, best_c = dh, dc

    if time.time() < t_end - 340.0:
        lb_args = ((200, 250, 10.0, 35), (2000, 80, 100.0, 35), (15000, 20, 1000.0, 30))
        for alpha, tk, mu, mi in lb_args:
            if time.time() > t_end - 340.0:
                break
            try:
                res = minimize(_pen_obj, best_h, args=(alpha, tk, mu), jac=_pen_grad,
                               method='L-BFGS-B', bounds=(0.0, 1.0),
                               options={'maxiter': int(mi), 'ftol': 1e-15, 'gtol': 1e-13})
                hx = getattr(res, 'x', None)
                if hx is not None and np.all(np.isfinite(hx)):
                    hx = np.asarray(hx, dtype=float)
                    if hx.size == n:
                        hh = _repair(hx, n)
                        cc = _eval(hh)
                        if cc < best_c:
                            best_h, best_c = hh, cc
            except Exception:
                pass

    if n > 192 and time.time() < t_end - 320.0:
        for f in (2, 3):
            n2 = n // f
            if n2 < 96 or time.time() > t_end - 300.0:
                continue
            start = _repair(_up(_down(best_h, f), n2), n2)
            start = _repair(start + 0.015 * rng.randn(n2), n2)
            dl2 = min(t_end - 300.0, time.time() + 55.0)
            dh, dc = _lp_dr(start, dl2, 8, 20)
            if time.time() + 20.0 < dl2:
                dh, dc = _nesterov(dh, 4000, 20, 4000, 0.004, dl2)
            if dc < best_c - 1e-13:
                best_h, best_c, n = dh, dc, n2
                if n <= 128:
                    break

    amp = 0.02
    noimp = 0
    while time.time() < t_end - 230.0:
        if time.time() + 75.0 > t_end - 230.0:
            break
        r = rng.random()
        if r < 0.12:
            start = _binary_round(best_h)
        elif r < 0.32:
            start = _downup_start(best_h, rng, min(0.16, amp * 0.7))
        else:
            start = _perturb(best_h, rng, amp, 1 if rng.random() < 0.85 else 3)
        dl = min(t_end - 230.0, time.time() + 70.0)
        hs, cs = _lp_dr(start, min(dl, time.time() + 20.0), 6, 16)
        if time.time() + 25.0 < dl:
            hs, cs = _nesterov(hs, 6000, 18, 2000, 0.004, dl)
        if cs < best_c - 1e-13:
            best_h, best_c = hs, cs
            noimp = 0
            if amp < 0.08:
                amp = min(0.09, amp * 1.05)
        else:
            noimp += 1
            amp = min(0.28, amp * 1.2)
            if noimp > 12:
                amp = 0.05
                noimp = 0

    if time.time() < t_end - 110.0:
        dl = min(t_end - 90.0, time.time() + 80.0)
        dh, dc = _lp_dr(best_h, dl, 10, 32)
        if dc < best_c:
            best_h, best_c = dh, dc

    if time.time() < t_end - 70.0:
        dl = min(t_end - 50.0, time.time() + 45.0)
        dh, dc = _nesterov(best_h, 30000, 10, 5000, 0.002, dl)
        if dc < best_c:
            best_h, best_c = dh, dc

    if time.time() < t_end - 45.0:
        try:
            res = minimize(_pen_obj, best_h, args=(20000, 10, 1000.0), jac=_pen_grad,
                           method='L-BFGS-B', bounds=(0.0, 1.0),
                           options={'maxiter': 25, 'ftol': 1e-15, 'gtol': 1e-13})
            hx = getattr(res, 'x', None)
            if hx is not None and np.all(np.isfinite(hx)):
                hx = np.asarray(hx, dtype=float)
                if hx.size == n:
                    hh = _repair(hx, n)
                    cc = _eval(hh)
                    if cc < best_c:
                        best_h, best_c = hh, cc
        except Exception:
            pass

    if time.time() < t_end - 2.0:
        best_h, best_c = _pair_polish(best_h, t_end - 2.0, rng)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, float(best_c), n)
```
