import numpy as np
import time


def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = float(h.sum())
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = np.clip(h * (target / s), 0.0, 1.0)
        diff = target - float(h.sum())
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _proj(v, target):
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    if hi <= lo:
        return np.clip(v, 0.0, 1.0)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _eval(h):
    c = _corr(h)
    return float(c.max() * 2.0 / len(h))


def _grad_weighted(h, idx, w):
    n = len(h)
    g = np.zeros(n, dtype=float)
    for t, wt in zip(idx, w):
        if wt < 1e-13:
            continue
        d = int(t) - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                g[d:] += wt * (1.0 - h[:L])
                g[:L] -= wt * h[d:]
        else:
            s = -d
            if s < n:
                g[:n - s] += wt * (1.0 - h[s:])
                g[s:] -= wt * h[:n - s]
    return g


def _proj_opt(h, stop_time, alpha, K, steps_init):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = 0.5 * n
    c = _corr(h)
    best_u = float(c.max())
    best_h = h.copy()
    steps = list(steps_init)
    stall = 0
    m = 2 * n - 1
    K = max(1, min(K, m))

    while time.time() < stop_time:
        c = _corr(h)
        cur_u = float(c.max())

        idx = np.argpartition(c, -K)[-K:]
        cm = c[idx]
        sc = alpha * (cm - cm.max())
        w = np.exp(sc)
        sw = float(w.sum())
        if sw <= 1e-300 or not np.isfinite(sw):
            w = np.full(K, 1.0 / K)
        else:
            w /= sw

        g = _grad_weighted(h, idx, w)
        gmax = float(np.max(np.abs(g)))
        if (not np.isfinite(gmax)) or gmax < 1e-15:
            break
        g /= gmax

        improved = False
        for si in range(len(steps)):
            if time.time() >= stop_time:
                break
            hnew = _proj(h - steps[si] * g, target)
            u = float(_corr(hnew).max())
            if u < cur_u - 1e-12:
                h = hnew
                cur_u = u
                improved = True
                stall = 0
                base = steps[si]
                steps = [base * 1.4, base * 0.4] + [s for s in steps if s < base * 0.5]

def _get_top(c, K, n):
    K = max(1, min(K, 2 * n - 1))
    idx = np.argpartition(c, -K)[-K:]
    L = idx - (n - 1)
    return idx, L, c[idx]


def _active_shifts(Ltop, ctop, max_shifts=12):
    if len(Ltop) == 0:
        return [1, 2, 3]
    order = np.argsort(ctop)[::-1]
    shifts = []
    for j in order:
        s = int(abs(Ltop[j]))
        if s > 0 and s not in shifts:
            shifts.append(s)
        if len(shifts) >= max_shifts:
            break
    if not shifts:
        shifts = [1, 2, 3]
    return shifts


def _line_pair(h, n, ctop, Ltop, p, q, M):
    A = np.zeros(len(Ltop), dtype=float)

    pos = p - Ltop
    v = (pos >= 0) & (pos < n)
    A[v] += 1.0 - h[pos[v]]

    pos = q - Ltop
    v = (pos >= 0) & (pos < n)
    A[v] -= 1.0 - h[pos[v]]

    pos = p + Ltop
    v = (pos >= 0) & (pos < n)
    A[v] -= h[pos[v]]

    pos = q + Ltop
    v = (pos >= 0) & (pos < n)
    A[v] += h[pos[v]]

    B = np.zeros(len(Ltop), dtype=float)
    B[Ltop == 0] -= 2.0
    dq = q - p
    B[Ltop == dq] += 1.0
    dp = p - q
    B[Ltop == dp] += 1.0

    hp = float(h[p])
    hq = float(h[q])
    lo = max(-hp, hq - 1.0)
    hi = min(1.0 - hp, hq)
    if hi - lo < 1e-8:
        return 0.0, float(ctop.max())

    ds = np.linspace(lo, hi, M)
    vals = ctop[None, :] + A[None, :] * ds[:, None] + B[None, :] * ds[:, None] * ds[:, None]
    rowmax = vals.max(axis=1)
    bi = int(np.argmin(rowmax))
    bd = float(ds[bi])
    bv = float(rowmax[bi])

    if 0 < bi < M - 1:
        d0 = float(ds[bi - 1])
        d1 = float(ds[bi + 1])
        hs = np.linspace(d0, d1, 13)
        v2 = ctop[None, :] + A[None, :] * hs[:, None] + B[None, :] * hs[:, None] * hs[:, None]
        r2 = v2.max(axis=1)
        bi2 = int(np.argmin(r2))
        if r2[bi2] < bv:
            bd = float(hs[bi2])
            bv = float(r2[bi2])

    return bd, bv

                cur_u = float(c.max())
                idx, Ltop, ctop = _get_top(c, K, n)
                top_u = float(ctop.max())
                shifts = _active_shifts(Ltop, ctop, 12)
                it = 0
            elif noimp % 2 == 0:
                idx, Ltop, ctop = _get_top(c, K, n)
                top_u = float(ctop.max())
                shifts = _active_shifts(Ltop, ctop, 12)

        if cur_u > best_u + max(0.15, 20.0 * anneal):
            h = best_h.copy()
            c = _corr(h)
            cur_u = float(c.max())
            idx, Ltop, ctop = _get_top(c, K, n)
            top_u = float(ctop.max())
            shifts = _active_shifts(Ltop, ctop, 12)

    best_h = _repair(best_h, n)
    c = _corr(best_h)
    return best_h, float(c.max() * 2.0 / n)


def _random3_local(h, stop_time, rng):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    c = _corr(h)
    best_h = h.copy()
    best_u = float(c.max())

    while time.time() < stop_time:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        k = int(rng.randint(n))
        if i == j or i == k or j == k:
            continue
        a = 0.018 * rng.randn()
        b = 0.018 * rng.randn()
        t = 0.2 + 0.8 * rng.rand()
        da = t * a
        db = t * b
        dc = -(da + db)
        ni = h[i] + da
        nj = h[j] + db
        nk = h[k] + dc
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i] = ni
            h[j] = nj
            h[k] = nk
            u2 = float(_corr(h).max())
            if u2 < best_u - 1e-10:
                best_u = u2
                best_h = h.copy()
            else:
                h[i] -= da
                h[j] -= db
                h[k] -= dc

    best_h = _repair(best_h, n)
    c = _corr(best_h)
    return best_h, float(c.max() * 2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    iv = globals().get('initial_h_values')
    if iv is None:
        n = 560
        x = 2.0 * np.arange(n) / n
        start = _repair(0.5 + 0.5 * np.sin(np.pi * x), n)
    else:
        n = len(iv)
        start = _repair(np.asarray(iv, dtype=float).copy(), n)

    best_h = start.copy()
    best_c5 = _eval(best_h)
    steps = [1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.0125]
    reserve = max(5.0, 0.025 * budget_s)

    smooth_end = min(deadline - reserve - 240.0, time.time() + 170.0)
    if time.time() < smooth_end:
        h, c5 = _multi_smooth(best_h, smooth_end, steps)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    pair_end = min(deadline - reserve - 150.0, time.time() + 140.0)
    if time.time() < pair_end:
        h, c5 = _pair_polish(best_h, pair_end, rng, K=96, M=11)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    alts = []
    for sh in [1, 2, 3, 5]:
        alts.append(np.roll(best_h, sh))
    for sig in [0.012, 0.03, 0.07]:
        alts.append(best_h + sig * rng.randn(n))
    if n >= 8:
        x = 2.0 * np.arange(n) / n
        for f in [2, 3, 5]:
            alts.append(0.5 + 0.5 * np.sin(np.pi * f * x + 0.37 * f))
    alts = alts[:8]

    final_reserve = max(18.0, 0.13 * budget_s)
    remaining = deadline - time.time() - final_reserve
    if remaining > 60.0 and len(alts) > 0:
        per = remaining / float(len(alts))
        for arr in alts:
            end = time.time() + min(per, remaining - 20.0)
            if time.time() >= end or end <= time.time() + 2.0:
                continue
            arr = _repair(np.clip(arr, 0.0, 1.0), n)
            stop1 = time.time() + 0.45 * (end - time.time())
            arr, _ = _multi_smooth(arr, stop1, steps)
            if time.time() < end:
                arr, c5a = _pair_polish(arr, end, rng, K=80, M=9)
            else:
                c5a = _eval(arr)
            if c5a < best_c5:
                best_c5 = c5a
                best_h = arr

    if time.time() < deadline - reserve:
        stop_s = min(deadline - reserve - 85.0, time.time() + 90.0)
        if time.time() < stop_s:
            h, c5s = _multi_smooth(best_h, stop_s, steps)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h

        stop_p = min(deadline - 35.0, time.time() + 170.0)
        if time.time() < stop_p:
            h, c5p = _pair_polish(best_h, stop_p, rng, K=192, M=13)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = h

    stop_r = deadline - 8.0
    if time.time() < stop_r:
        h, c5r = _random3_local(best_h, stop_r, rng)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h

    best_h = _repair(best_h, n)
    c5 = _eval(best_h)
    return (best_h, c5, n)
