import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(corr)), corr

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _nesterov_topk(h, n, t_end, top_k, lr0, mu):
    dx = 2.0 / n
    steps = 0
    v = np.zeros(n)
    step = lr0
    h = h.copy()
    while time.time() < t_end:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        # Get top_k indices
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        # Compute weighted gradient (uniform weights for top_k)
        g = np.zeros(n)
        for k in top_idx:
            g += _lag_grad(h, k, n, dx)
        g /= top_k
        
        # Nesterov step
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < m:  # m is old value
            h = h_new
            step = min(step * 1.02, lr0 * 10)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
        steps += 1
    return h, _eval(h)

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        sweep_improved = False
        n_pairs = n * 2
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            # Golden section search
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(40):  # More iterations for precision
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 60:
            h += 1e-4 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _nesterov_topk(h, n, t_end, top_k, lr0, mu, polyak_f=False):
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    step = lr0

    while time.time() < t_end:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += _lag_grad(h, k, n, dx)
        g /= top_k
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        if polyak_f and f_cur > f_best:
            step = min(0.5 * (f_cur - f_best) / (gnorm * gnorm), lr0 * 20)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f_cur:
            h = h_new
            f_cur = f_new
            if f_cur < f_best:
                f_best = f_cur
            step = min(step * 1.03, lr0 * 12.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f_cur = _eval(h)
    return h, f_cur
