import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-13: break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        if m.sum() == 0: break
        h[m] += d / m.sum()
        np.clip(h, 0, 1, out=h)
    return h

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad(h, alpha, n, tk=0):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if tk > 0:
        idx = np.argsort(c)[-tk:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / tk
    else:

        w = np.exp(s)
        w /= w.sum()

        if a < 1e-14: continue

            g[:n-sp] -= a * h[sp:]
