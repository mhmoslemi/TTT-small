            g[m1] += wk * (1.0 - h[m1 + t])
            m2 = np.arange(t, n)
            g[m2] -= wk * h[m2 - t]

    s = k - (n - 1)

        g[m1] += dx * (1.0 - h[m1 - s])
        m2 = np.arange(n - s)
        g[m2] -= dx * h[m2 + s]
    else:
        t = -s
        m1 = np.arange(n - t)
        g[m1] += dx * (1.0 - h[m1 + t])
        m2 = np.arange(t, n)
        g[m2] -= dx * h[m2 - t]
    return g

    n = len(h)
    g = np.zeros(n)
    for k in topk:
        g += _grad_single_lag(h, k)
    return g / len(topk)

    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _lse_obj(h, T, pen):
    c = _cvals(h)
    cm = float(c.max())
    val = cm + float(np.log(np.exp(T * (c - cm)).sum())) / T
    return val + pen * (h.sum() - len(h) * 0.5) ** 2

def _lse_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = _cvals(h)
    cm = float(c.max())
    w = np.exp(T * (c - cm))
    w /= w.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for k in active:
        wk = w[k] * dx
        s = k - (n - 1)
        if s >= 0:
            m1 = np.arange(s, n)
            g[m1] += wk * (1.0 - h[m1 - s])
            m2 = np.arange(n - s)
            g[m2] -= wk * h[m2 + s]
        else:
            t = -s
            m1 = np.arange(n - t)
            g[m1] += wk * (1.0 - h[m1 + t])
            m2 = np.arange(t, n)
            g[m2] -= wk * h[m2 - t]
    return g + 2.0 * pen * (h.sum() - n * 0.5)

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        m1 = np.arange(s, n)
        g[m1] += dx * (1.0 - h[m1 - s])
        m2 = np.arange(n - s)
        g[m2] -= dx * h[m2 + s]
    else:
        t = -s
        m1 = np.arange(n - t)
        g[m1] += dx * (1.0 - h[m1 + t])
        m2 = np.arange(t, n)
        g[m2] -= dx * h[m2 - t]
    return g

def _topk_gradient(h, topk):
    n = len(h)
    g = np.zeros(n)
    for k in topk:
        g += _grad_lag(h, k)
    return g / len(topk)

def _pgd_topk(h, n_iter, k, step0, beta, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    step = step0
    stall = 0
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _cvals(h_c)
        topk = np.argsort(c)[-k:]
        g = _topk_gradient(h_c, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.02, 0.06)
        else:
            step *= 0.55
            stall += 1
            if step < 1e-10 or stall > 60:
                break
    return best_h, best_c

def _coord_search(h, n_iter, rng, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 8e-5
    for it in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = rng.randn() * scale
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            c_new = _eval(h_c)
            if c_new < best_c:
                best_c = c_new
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
        if it % 4000 == 0:
            scale *= 0.82
    return best_h, best_c

def _lbfgs_lse(h, T, pen, maxiter, bnds):
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(T, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-17,
                                'gtol': 1e-13, 'maxls': 30})
        return _repair(res.x, len(res.x))
    except Exception:
        return _repair(h, len(h))
