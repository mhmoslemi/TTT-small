        if time.time() > tlimit:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 8)) % n
        k = (i + 10 + rng.randint(0, 15)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = cur_h[i] + d1, cur_h[j] + d2, cur_h[k] - d1 - d2
        if all(0.0 <= v <= 1.0 for v in (ni, nj, nk)):
            cur_h[i], cur_h[j], cur_h[k] = ni, nj, nk
            c = _eval(cur_h)
            if c < cur_c:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = cur_h.copy()
            else:
                cur_h[i], cur_h[j], cur_h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _down_up(h, factor, n_target):
    n = len(h)
    m = n // (n // n_target) if n >= n_target else n
    f = n // m
    small = h[:m * f].reshape(m, f).mean(axis=1).copy()
    big = np.repeat(small, f).copy()
    if len(big) != n_target:
        big = np.resize(big, n_target)
    return _repair(big, n_target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    t_slsqp_end = t0 + budget_s * 0.55
    t_local_end = t0 + budget_s * 0.85

    # Phase 1: Quick Nesterov
    tl = min(t0 + 80, t_slsqp_end - 60)
    h1 = _nesterov(h, 3000, 6000, 0.004, tl)
    c1 = _eval(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: SLSQP sweep with increasing alpha
    alphas = [20000, 100000, 500000, 2000000, 5000000]
    for al in alphas:
        if time.time() > t_slsqp_end - 5:
            break
        hc = _run_slsqp(best_h.copy(), al, 2500, 1e-16)
        cc = _eval(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()

    # Phase 3: Nesterov with high alpha between SLSQPs
    if time.time() < t_slsqp_end + 80:
        h3 = _nesterov(best_h.copy(), 2000000, 4000, 0.0008, t_slsqp_end + 80)
        c3 = _eval(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Perturbation escape + re-optimize
    for trial in range(3):
        if time.time() > t_local_end - 30:
            break
        h_pert = best_h + 0.01 * rng.randn(n)
        h_pert = _proj_to_sum(h_pert, n / 2.0)
        h_pert = _repair(h_pert, n)
        hc = _run_slsqp(h_pert, 2000000, 1500, 1e-16)
        cc = _eval(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()

    # Phase 5: Multi-point local search (medium)
    if time.time() < t_local_end - 10:
        n_iter = min(50000, int((t_local_end - 10 - time.time()) / 0.004))
        best_h, best_c5 = _multi_point_search(best_h, max(n_iter, 1000), 2e-4, rng, t_local_end - 10)

    # Phase 6: 2-point local search (fine)
    t_final = t0 + budget_s - 5
    if time.time() < t_final - 20:
        n_iter = min(80000, int((t_final - 20 - time.time()) / 0.003))
        best_h, best_c5 = _local_search(best_h, max(n_iter, 5000), 1e-4, rng, t_final - 20)

    # Phase 7: Ultra-fine 2-point polish
    if time.time() < t_final - 2:
        n_iter = min(60000, int((t_final - 2 - time.time()) / 0.0025))
        best_h, best_c5 = _local_search(best_h, max(n_iter, 3000), 3e-5, rng, t_final - 2)

    # Phase 8: Tiny step final polish
    if time.time() < t_final - 0.5:
        n_iter = min(40000, int((t_final - 0.5 - time.time()) / 0.002))
        best_h, best_c5 = _local_search(best_h, max(n_iter, 2000), 1e-5, rng, t_final)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
