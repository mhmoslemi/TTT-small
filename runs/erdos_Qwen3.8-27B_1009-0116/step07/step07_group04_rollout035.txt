import numpy as np
import time
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for lag in range(2 * n - 1):
        if w[lag] < 1e-13:
            continue
        sh = (n - 1) - lag
        wt = w[lag] * dx
        if sh == 0:
            g += wt * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wt * (1.0 - h[sh:])
            g[sh:] -= wt * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wt * (1.0 - h[:n - sp])
            g[:n - sp] -= wt * h[sp:]
    return m, g

def _project(h, tgt):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0, 1)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _fista(h_init, n_it, a0, a1, step, t0, t_end):
    n = len(h_init)
    tgt = n * 0.5
    x = h_init.copy(); z = h_init.copy(); t = 1.0
    best = _eval(x); bx = x.copy(); f_x = best
    for k in range(n_it):
        if time.time() > t_end:
            break
        a = a0 * (a1 / a0) ** (k / max(n_it, 1))
        _, g = _grad_smooth(z, a)
        y = x + ((t - 1) / (t + 1)) * (x - z)
        st = step / (1.0 + 0.005 * a)
        z_new = _project(y - st * g, tgt)
        f_new = _eval(z_new)
        if f_new < f_x:
            x = z_new; f_x = f_new
            if f_new < best:
                best = f_new; bx = x.copy()
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
    return _repair(bx, n), best

def _slsqp(h_init, alpha, maxit, t_end):
    n = len(h_init)
    tgt = n * 0.5
    def obj(v):
        c = np.correlate(v, 1.0 - v, mode='full') * 2.0 / n
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def grad(v):
        _, g = _grad_smooth(v, alpha)
        return g
    def cfun(v):
        return v.sum() - tgt
    def cjac(v):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h_init, n)

def _lbfgsb(h_init, alpha, maxit, t_end):

    bc = _eval(h); bh = h.copy()
    h = h.copy()
    for _ in range(ntries):

            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 10)) % n
        k = (i + rng.randint(11, 25)) % n
        d1 = 5e-5 * rng.randn(); d2 = 5e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c = _eval(h)
            if c < bc - 1e-18:
                bc = c; bh = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return bh, bc

def _local2(h, ntries, rng, t_end):

    bc = _eval(h); bh = h.copy()
    h = h.copy()
    for _ in range(ntries):

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n

        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < bc - 1e-18:
                bc = c; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _block_perturb(h, rng, n_scales, t_end):
    n = len(h)
    best_h = h.copy(); best_c = _eval(h)
    for _ in range(n_scales):
        if time.time() > t_end:
            break
        trial = h.copy()
        for _ in range(rng.randint(3, 8)):
            start = rng.randint(n - 30)
            length = rng.randint(5, 30)
            amp = rng.uniform(0.01, 0.05)
            blk = trial[start:start + length]
            blk += amp * rng.randn(length)
            trial[start:start + length] = np.clip(blk, 0, 1)
        trial = _repair(trial, n)
        c = _eval(trial)
        if c < best_c:
            best_c = c; best_h = trial.copy(); h = trial.copy()
    return best_h, best_c
