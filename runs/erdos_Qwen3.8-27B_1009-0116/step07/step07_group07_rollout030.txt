    """Given h and positions i, j, find optimal redistribution of h[i]+h[j]"""
    s = h[i] + h[j]  # total mass to redistribute
    # We want to find t that minimizes the objective with h[i]=t, h[j]=s-t

    # But the objective is expensive to evaluate.

    best_h = h_c.copy()

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode='full') * 2.0/n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n/2) > 1e-9:
        h = _proj_sum(h, n/2)
    return h

def _obj_p(h, p, pen):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * 2.0/n
    return float(np.sum(c**p)**(1.0/p)) + pen*(h.sum()-n*0.5)**2

def _grad_p(h, p, pen):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')*dx
    cp = np.maximum(c, 1e-30)**p
    S = cp.sum()
    fac = S**(1.0/p-1.0)
    w = fac * np.maximum(c, 0.0)**(p-1) * dx
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-25: continue
        sh = (n-1)-k
        if sh >= 0:
            g[:n-sh] += w[k]*(1.0-h[sh:])
            if sh > 0: g[sh:] -= w[k]*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k]*(1.0-h[:n-sp])
            g[:n-sp] -= w[k]*h[sp:]
    return g + 2.0*pen*(h.sum()-n*0.5)

    n = len(h)

                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)

        return _repair(h, n)

    cp = np.maximum(c, 1e-30)**p
    S = cp.sum()
    fac = S**(1.0/p-1.0)
    w = fac*np.maximum(c, 0.0)**(p-1)*dx
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-25: continue
        sh = (n-1)-k
        if sh >= 0:
            g[:n-sh] += w[k]*(1.0-h[sh:])
            if sh > 0: g[sh:] -= w[k]*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k]*(1.0-h[:n-sp])
            g[:n-sp] -= w[k]*h[sp:]
    return g + 2.0*pen*(h.sum()-n*0.5)

def _lbfgs(h, p, pen, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    try:
        res = minimize(_obj_p, h, jac=_grad_p, args=(p, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _perturb_c(c, h, idx, d, n):
    """In-place update of c for h[idx] += d (O(n) vectorized)"""
    k1e = min(2*n-2, idx+n-1)+1
    cnt = k1e-idx
    if cnt > 0:
        c[idx:k1e] += d*(1.0-h[n-1-np.arange(cnt)])
    k2s = n-1-idx
    c[k2s:k2s+n] -= d*h

def _fast_search(h, n_iter, rng, t0, budget, scale0=4e-4):
    n = len(h)
    h_c = h.copy().astype(float)
    c = _corrs(h_c).copy()
    best_c = c.max()
    best_h = h_c.copy()
    scale = scale0
    stall = 0
    for it in range(n_iter):
        if time.time()-t0 > budget: break
        if stall > 80:
            scale *= 0.55
            if scale < 1e-11: break
            stall = 0
            for _ in range(5):
                i=rng.randint(n); j=(i+1+rng.randint(0,25))%n
                d=rng.randn()*scale*10
                if 0<=h_c[i]+d<=1 and 0<=h_c[j]-d<=1:
                    h_c[i]+=d; h_c[j]-=d
                    _perturb_c(c,h_c,i,d,n); _perturb_c(c,h_c,j,-d,n)
            fm = c.max()
            if fm < best_c: best_c=fm; best_h=h_c.copy()
            continue
        # Target hot shifts
        cmax = c.max()
        hot = np.where(c > cmax-0.002)[0]
        if len(hot)==0: hot = np.argsort(c)[-3:]
        k = hot[rng.randint(len(hot))]
        sh = (n-1)-k
        s = abs(sh)
        if s==0 or s>=n:
            i=rng.randint(n); j=(i+1+rng.randint(0,15))%n
        elif sh>0:
            i=rng.randint(n-s); j=i+s
        else:
            i=rng.randint(n-s); j=i-s
        d = rng.randn()*scale
        ni = h_c[i]+d; nj = h_c[j]-d
        if 0<=ni<=1 and 0<=nj<=1:
            h_c[i]=ni; h_c[j]=nj
            _perturb_c(c,h_c,i,d,n); _perturb_c(c,h_c,j,-d,n)
            nm = c.max()
            if nm < best_c-1e-15:
                best_c=nm; best_h=h_c.copy(); stall=0
            else:
                h_c[i]=ni-d; h_c[j]=nj+d
                _perturb_c(c,h_c,i,-d,n); _perturb_c(c,h_c,j,d,n)
                stall+=1
        else:
            stall+=1
    return best_h, best_c

def _batch_search(h, n_iter, rng, t0, budget):
    """Try 2-pair batch moves for coordinated improvements"""
    n = len(h)
    h_c = h.copy().astype(float)
    c = _corrs(h_c).copy()
    best_c = c.max()
    best_h = h_c.copy()
    scale = 2e-4
    stall = 0
    for it in range(n_iter):
        if time.time()-t0 > budget: break
        if stall > 120:
            scale *= 0.6
            if scale < 1e-11: break
            stall = 0
            continue
        cmax = c.max()
        top5 = np.argsort(c)[-5:]
        # Pick 2 shifts and one pair each
        k1, k2 = top5[rng.randint(5)], top5[rng.randint(5)]
        if k1 == k2: continue
        pairs = []
        for k in (k1, k2):
            sh = abs((n-1)-k)
            if 0 < sh < n:
                i = rng.randint(n-sh)
                if (n-1)-k > 0:
                    pairs.append((i, i+sh))
                else:
                    pairs.append((i, i-sh))
            else:
                i = rng.randint(n)
                pairs.append((i, (i+1+rng.randint(0,10))%n))
        (i1,j1),(i2,j2) = pairs
        d1 = rng.randn()*scale; d2 = rng.randn()*scale
        ok = True
        for i,j,d in ((i1,j1,d1),(i2,j2,d2)):
            if not (0<=h_c[i]+d<=1 and 0<=h_c[j]-d<=1): ok=False; break
        if not ok:
            stall+=1; continue
        h_c[i1]+=d1; h_c[j1]-=d1; h_c[i2]+=d2; h_c[j2]-=d2
        _perturb_c(c,h_c,i1,d1,n); _perturb_c(c,h_c,j1,-d1,n)
        _perturb_c(c,h_c,i2,d2,n); _perturb_c(c,h_c,j2,-d2,n)
        nm = c.max()
        if nm < best_c-1e-15:
            best_c=nm; best_h=h_c.copy(); stall=0
        else:
            h_c[i1]-=d1; h_c[j1]+=d1; h_c[i2]-=d2; h_c[j2]+=d2
            _perturb_c(c,h_c,i1,-d1,n); _perturb_c(c,h_c,j1,d1,n)
            _perturb_c(c,h_c,i2,-d2,n); _perturb_c(c,h_c,j2,d2,n)
            stall+=1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def bc(): return time.time()-t0 > budget_s-25
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    # Phase 1: L-BFGS warm-up
    for p, pen, mi in [(16,1e5,300),(64,1e6,300),(256,1e7,250)]:
        if bc(): break
        ht=_lbfgs(best_h.copy(),p,pen,mi)
        ct=_eval(ht)
        if ct<best_c5: best_c5=ct; best_h=ht
    # Phase 2: Very high p L-BFGS
    for p in [1024,4096]:
        if bc(): break
        ht=_lbfgs(best_h.copy(),p,1e8,300)
        ct=_eval(ht)
        if ct<best_c5: best_c5=ct; best_h=ht
    # Phase 3: Aggressive fast local search
    if not bc():
        best_h,best_c5=_fast_search(best_h,2000000,rng,t0,budget_s-120,scale0=5e-4)
    # Phase 4: L-BFGS polish
    if not bc():
        for p in [2048,8192]:
            if bc(): break
            ht=_lbfgs(best_h.copy(),p,1e9,200)
            ct=_eval(ht)
            if ct<best_c5: best_c5=ct; best_h=ht
    # Phase 5: Batch 2-pair search
    if not bc():
        best_h,best_c5=_batch_search(best_h,1000000,rng,t0,budget_s-60)
    # Phase 6: Fine single-pair search
    if not bc():
        best_h,best_c5=_fast_search(best_h,2000000,rng,t0,budget_s-30,scale0=1e-4)
    # Phase 7: Final L-BFGS
    if not bc():
        ht=_lbfgs(best_h.copy(),4096,1e9,150)
        ct=_eval(ht)
        if ct<best_c5: best_c5=ct; best_h=ht
    # Phase 8: Ultra-fine search
    if not bc():
        best_h,best_c5=_fast_search(best_h,1000000,rng,t0,budget_s-5,scale0=3e-5)
    best_h=_repair(best_h,len(best_h))
    best_c5=_eval(best_h)
    return (best_h, best_c5, len(best_h))
