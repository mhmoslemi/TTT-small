                    diff -= a
                else:

        g *= dx
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, n * 0.5)
        c5_new = _eval(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < bc:
                bc = c5
                bh = h.copy()
            step = min(step * 1.015, step0 * 5.0)
        else:
            step *= 0.7
    return _repair(bh, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.0015 * max(T, 1e-10)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n <= c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _pair_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        topidx = np.argpartition(c, -4)[-4:]
        k = topidx[rng.randint(4)]
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = rng.randn() * 4e-6
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Warm-up Nesterov
    dl1 = min(deadline, t0 + 120)
    for alpha, tk, st, ns in [(3000, 50, 0.006, 1500), (10000, 30, 0.004, 1500)]:
        if time.time() > dl1:
            break
        h = _nesterov_ar(best_h, alpha, tk, ns, st, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Main Nesterov with adaptive restarts
    dl2 = min(deadline, t0 + 500)
    for alpha, tk, st, ns in [(50000, 20, 0.003, 2500), (300000, 12, 0.0015, 2500)]:
        if time.time() > dl2:
            break
        h = _nesterov_ar(best_h, alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Cyclical subgradient on worst lag
    dl3 = min(deadline, t0 + 700)
    h_c = _cyc_subgrad(best_h, 20000, 0.004, dl3)
    c5v = _eval(h_c)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_c.copy()

    # Phase 4: Block annealing
    dl4 = min(deadline, t0 + 880)
    blocks = [1, 2, 3, 4, 6, 8]
    h_a, c5_a = _anneal(best_h.copy(), 25000, rng, dl4, 0.003, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 5: High-alpha Nesterov for fine tuning
    dl5 = min(deadline, t0 + budget_s - 80)
    for alpha, tk, st, ns in [(2000000, 6, 0.0005, 3000), (10000000, 4, 0.0003, 3000)]:
        if time.time() > dl5:
            break
        h = _nesterov_ar(best_h, alpha, tk, ns, st, dl5)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Targeted pair polish on top lags
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0025), 50000)
    best_h, best_c5 = _pair_polish(best_h, n_pol, rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
