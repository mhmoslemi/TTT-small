import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_lag(h, l):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if l >= 0:
        if l < n:
            g[l:] += (1.0 - h[:n - l])
            g[:n - l] += -h[l:]
    else:
        m = -l
        if m < n:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] += -h[m:]
    g *= dx
    return g

def _grad_topk(h, topk, sharp):
    n = len(h)
    c = _correl(h)
    full_len = 2 * n - 1
    if topk >= full_len:
        idx = np.arange(full_len)
        w = np.ones(full_len) / full_len
    else:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = sharp * (ct - ct.max())
        w = np.exp(s)
        w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-12:
            g += w[ii] * _grad_lag(h, int(idx[ii]) - (n - 1))
    return g

def _nesterov(h, topk, sharp, steps, step0, mom, deadline):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, topk, sharp)
        hp = h.copy()
        v = mom * v + (1.0 - mom) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _eval(hn)
        if fn < f - 1e-19:
            h = hn
            f = fn
            step = min(step * 1.012, step0 * 30.0)
        else:
            step *= 0.35
            h = _proj(hp - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _binary_push(h, n, rng, strength):
    target = n / 2.0
    h2 = h.copy()
    mask_hi = h2 > 0.5
    mask_lo = ~mask_hi
    h2[mask_hi] = np.clip(h2[mask_hi] + strength * rng.uniform(0.5, 1.0, mask_hi.sum()), 0, 1)
    h2[mask_lo] = np.clip(h2[mask_lo] - strength * rng.uniform(0.5, 1.0, mask_lo.sum()), 0, 1)
    return _repair(h2, n)

def _slsqp_obj(x, alpha, topk):
    cx = _correl(x)
    m2 = cx.max()
    if topk < len(cx):
        idx2 = np.argpartition(cx, -topk)[-topk:]
        ct2 = cx[idx2]
    else:
        ct2 = cx
    s2 = np.clip(alpha * (ct2 - m2), -400, 400)
    return m2 + (s2.max() + np.log(np.sum(np.exp(s2 - s2.max())))) / alpha

def _slsqp(x, alpha, topk, n, maxit, deadline):
    c0 = x.sum()
    def _sumcust(x2):
        return x2.sum() - n * 0.5
    try:
        res = minimize(lambda x2: _slsqp_obj(x2, alpha, topk), x,
                       jac=lambda x2: _grad_topk(x2, topk, alpha),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sumcust}],
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(x, n)
