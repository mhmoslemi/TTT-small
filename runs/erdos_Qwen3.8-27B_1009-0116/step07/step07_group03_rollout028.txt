import numpy as np
from scipy.optimize import minimize, linprog
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = _proj_to_sum(h, target)
    return h

def _fourier_basis(n, M):
    # Returns matrix V of shape (n, 2M) where h = 0.5 + V @ coeffs
    t = np.arange(n)
    V = np.zeros((n, 2 * M))
    for m in range(1, M + 1):
        V[:, 2*(m-1)] = np.cos(2 * np.pi * m * t / n)
        V[:, 2*(m-1) + 1] = np.sin(2 * np.pi * m * t / n)
    return V

def _spectral_opt(n, M, t_end, rng):
    V = _fourier_basis(n, M)
    c0 = np.zeros(2 * M)
    
    def obj(coeffs):
        h = 0.5 + V @ coeffs
        h = np.clip(h, 0, 1)
        h = _proj_to_sum(h, n * 0.5)
        return _eval(h)
    
    best_c = obj(c0)
    best_coeffs = c0.copy()
    
    # Nelder-Mead on the spectral coefficients
    for attempt in range(20):
        if time.time() > t_end - 5:
            break
        scale = 0.3 * (0.7 ** attempt)
        x0 = rng.randn(2 * M) * scale
        res = minimize(lambda c: _eval(np.clip(0.5 + V @ c, 0, 1) if abs((0.5 + V @ c).sum() - n/2) < 1e-6 else _proj_to_sum(np.clip(0.5 + V @ c, 0, 1), n/2)),
                       x0, method='Nelder-Mead', options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
        h_cand = np.clip(0.5 + V @ res.x, 0, 1)
        h_cand = _proj_to_sum(h_cand, n * 0.5)
        c_val = _eval(h_cand)
        if c_val < best_c:
            best_c = c_val
            best_coeffs = res.x.copy()
    
    h_best = np.clip(0.5 + V @ best_coeffs, 0, 1)
    h_best = _proj_to_sum(h_best, n * 0.5)
    return h_best, best_c

def _smooth_eval(h, alpha):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

    w = np.exp(s)
    w /= w.sum()

def _grad_from_weights(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:n-lag]
        else:
            l2 = -lag
            g[l2:] += wk * (1.0 - h[:n-l2])
            g[:n-l2] -= wk * h[l2:]
    return g

def _nesterov_opt(h, alpha, steps, t_end, step0=0.003):
    n = len(h)
    target = n * 0.5
    f = _smooth_eval(h, alpha)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for it in range(steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        g -= g.mean()  # project to zero-mean
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, target)
        f_new = _smooth_eval(h_new, alpha)
        if f_new <= f + 1e-18:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return h

def _subgrad_step(h, K, t_end, n_steps, step_max):
    """Subgradient steps with bounded line search."""
    n = len(h)
    target = n * 0.5
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-K:]
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / K
        g = _grad_from_weights(h, w, 2.0 / n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        g = g / gnorm
        # Bounded line search
        # Find max step for box constraints
        pos = g > 1e-12
        neg = g < -1e-12
        t_max = step_max
        if pos.any():
            t_max = min(t_max, (1.0 - h[pos]).min() / g[pos].max())
        if neg.any():
            t_max = min(t_max, h[neg].min() / (-g[neg].max()))
        t_max = max(t_max, 1e-12)
        # Golden section search on [0, t_max]
        gr = (np.sqrt(5) - 1) / 2
        a, b = 0.0, t_max
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        fa = _eval(_proj_to_sum(np.clip(h + c1 * g, 0, 1), target))
        fb = _eval(_proj_to_sum(np.clip(h + c2 * g, 0, 1), target))
        for _ in range(20):
            if fa < fb:
                b = c2; c2 = c1; fb = fa; c1 = b - gr * (b - a)
                fa = _eval(_proj_to_sum(np.clip(h + c1 * g, 0, 1), target))
            else:
                a = c1; c1 = c2; fa = fb; c2 = a + gr * (b - a)
                fb = _eval(_proj_to_sum(np.clip(h + c2 * g, 0, 1), target))
        t_opt = (a + b) / 2.0
        h = _proj_to_sum(np.clip(h + t_opt * g, 0, 1), target)
    return h

def _coord_descent(h, n, duration_s, top_T, rng, t_end):

    target = n * 0.5
    best_c5 = _eval(h)
    end_t = min(time.time() + duration_s, t_end - 5)
    no_improve = 0
    while time.time() < end_t:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        # Precompute gradients for top lags
        grads = np.zeros((top_T, n))
        for t_i, k in enumerate(top_idx):
            lag = k - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                g[:n-lag] = (1.0 - h[lag:])
                g[lag:] -= h[:n-lag]
            else:
                l2 = -lag
                g[l2:] = (1.0 - h[:n-l2])
                g[:n-l2] -= h[l2:]
            grads[t_i] = g * dx
        a_vals = c[top_idx]
        improved = False
        n_pairs = n * 3
        for _ in range(n_pairs):
            if time.time() > end_t:

            i = rng.randint(n)

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_to_sum(h, target)
    return h

def _smooth_eval(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_weighted(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:n - lag]
        else:
            l2 = -lag
            g[l2:] += wk * (1.0 - h[:n - l2])
            g[:n - l2] -= wk * h[l2:]
    return g

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    return _grad_weighted(h, w, dx)

def _nesterov(h, alpha, steps, t_end, step0):
    n = len(h)
    target = n * 0.5
    f = _smooth_eval(h, alpha)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    stall = 0
    for it in range(steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        g -= g.mean()
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, target)
        f_new = _smooth_eval(h_new, alpha)
        if f_new <= f + 1e-18:
            h = h_new
            f = f_new
            step = min(step * 1.015, step0 * 12.0)
            stall = 0
        else:
            stall += 1
            step *= 0.6
            if step < 1e-12:
                break
            if stall > 30:
                v *= 0.3
                stall = 0
    return h

def _subgrad_line(h, K, steps, t_end, tmax):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-K:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / K
        g = _grad_weighted(h, w, dx)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = -g / gn
        t_top = tmax
        pos = d > 1e-13
        neg = d < -1e-13
        if pos.any():
            t_top = min(t_top, (1.0 - h[pos]).min() / d[pos].max() * 0.99)
        if neg.any():
            t_top = min(t_top, h[neg].min() / (-d[neg].max()) * 0.99)
        t_top = max(t_top, 1e-10)
        gr = (np.sqrt(5) - 1) / 2
        a, b = 0.0, t_top
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        h1 = _proj_to_sum(np.clip(h + c1 * d, 0, 1), target)
        h2 = _proj_to_sum(np.clip(h + c2 * d, 0, 1), target)
        f1, f2 = _eval(h1), _eval(h2)
        for _ in range(18):
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1; c1 = b - gr * (b - a)
                h1 = _proj_to_sum(np.clip(h + c1 * d, 0, 1), target)
                f1 = _eval(h1)
            else:
                a = c1; c1 = c2; f1 = f2; c2 = a + gr * (b - a)
                h2 = _proj_to_sum(np.clip(h + c2 * d, 0, 1), target)
                f2 = _eval(h2)
        t_opt = (a + b) * 0.5
        h_new = _proj_to_sum(np.clip(h + t_opt * d, 0, 1), target)
        if _eval(h_new) < _eval(h):
            h = h_new
    return h

def _coord_descent(h, n, dur, top_T, rng, t_end):
    dx = 2.0 / n
    target = n * 0.5
    best_c5 = _eval(h)
    end_t = min(time.time() + dur, t_end - 4)
    no_imp = 0
    gr = (np.sqrt(5) - 1) / 2
    while time.time() < end_t:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        grads = np.zeros((top_T, n))
        for t_i, k in enumerate(top_idx):
            lag = k - (n - 1)
            gg = np.zeros(n)
            if lag >= 0:
                gg[:n - lag] = 1.0 - h[lag:]
                gg[lag:] -= h[:n - lag]
            else:
                l2 = -lag
                gg[l2:] = 1.0 - h[:n - l2]
                gg[:n - l2] -= h[l2:]
            grads[t_i] = gg * dx
        improved = False
        for _ in range(n * 3):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 10))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(22):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            nc = _eval(h)
            if nc < best_c5:
                best_c5 = nc
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > 35:
            h += 5e-6 * rng.randn(n)
            h = _proj_to_sum(h, target)
            no_imp = 0
        if abs(h.sum() - target) > 1e-7:
            h = _proj_to_sum(h, target)
    return h, best_c5

def _micro(h, n_iter, rng, t_end):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 2.5e-6 * rng.randn()
        h[i] += d
        h[j] -= d
        h[i] = min(max(h[i], 0.0), 1.0)
        h[j] = min(max(h[j], 0.0), 1.0)
        cn = _eval(h)
        if cn < c5c - 1e-19:
            c5c = cn
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c
