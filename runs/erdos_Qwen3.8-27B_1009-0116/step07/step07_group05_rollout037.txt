import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def softmax_corr(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sc = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    return c, m, w

def softmax_grad(h, alpha):
    c, m, w = softmax_corr(h, alpha)
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g, m

def nesterov_phase(h, alphas, iters_per_alpha, check_fn):
    n = len(h)
    best_h = h.copy()
    best_c = eval_c5(h)
    x = h.copy()
    v = np.zeros(n)
    mu = 0.85
    
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if check_fn():
                break
            y = x + mu * v
            g, _ = softmax_grad(y, alpha)
            step = 0.05 / (1.0 + 0.01 * alpha)
            new_v = mu * 0.5 * (y - x) - step * g
            x_new = project_sum(x + new_v, n * 0.5)
            c_new = eval_c5(x_new)
            if c_new < best_c:
                best_c = c_new
                best_h = x_new.copy()
            x = x_new
            v = new_v
    return best_h, best_c

def topk_phase(h, k_top, step_list, n_iter, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = eval_c5(h)
    best_h = h.copy()
    best_c = c5c
    steps = list(step_list)
    
    for _ in range(n_iter):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in top_idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[sh:] += (1.0 - h[:n - sh])
                if sh > 0:
                    g[:n - sh] -= h[sh:]
            else:
                sp = -sh
                g[:n - sp] += (1.0 - h[sp:])
                g[sp:] -= h[:n - sp]
        g *= dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = project_sum(h - steps[si] * g, n * 0.5)
            c5n = eval_c5(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            steps = [s * 0.4 for s in steps]
    
    return best_h, best_c

def fast_2pt(h, ntry, rng, check_fn):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = eval_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
    a_all = np.arange(-(n - 1), n)
    
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = 5e-6 * rng.randn()
        ni = best_h[i] + d
        nj = best_h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        # Compute delta corr
        delta = np.zeros(2 * n - 1)
        # i-a valid: 0<=i-a<n
        ta = i - a_all
        m1 = (ta >= 0) & (ta < n)
        if m1.any():
            delta[m1] += d * (1.0 - best_h[ta[m1]])
        # i+a valid: 0<=i+a<n
        ta = i + a_all
        m2 = (ta >= 0) & (ta < n)
        if m2.any():
            delta[m2] -= d * best_h[ta[m2]]
        # j-a valid
        ta = j - a_all
        m3 = (ta >= 0) & (ta < n)
        if m3.any():
            delta[m3] -= d * (1.0 - best_h[ta[m3]])
        # j+a valid
        ta = j + a_all
        m4 = (ta >= 0) & (ta < n)
        if m4.any():
            delta[m4] += d * best_h[ta[m4]]
        
        new_max = (corr + delta).max()
        if new_max < best_c - 1e-18:
            best_h[i] = ni
            best_h[j] = nj
            corr += delta
            best_c = new_max
    
    return best_h, best_c

def slsqp_phase(h, check_fn):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        a = 50000.0
        sc = np.clip(a * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(sc - sc.max()))) / a
    def grad(x):
        g, _ = softmax_grad(x, 50000.0)
        return g
    def sumf(x):
        return x.sum() - n * 0.5
    def sumj(x):
        return np.ones(len(x))
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sumf, 'jac': sumj}]
    best_h = h.copy()
    best_c = eval_c5(h)
    for _ in range(3):
        if check_fn():
            break

                         bounds=bnds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-17})
            hr = repair(res.x, n)
            c5r = eval_c5(hr)
            if c5r < best_c:
                best_c = c5r
                best_h = hr
        except Exception:
            pass
    return best_h, best_c
