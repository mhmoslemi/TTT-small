    m = c.max()
    idx = np.argsort(c)[-K:]
    sc = alpha * c[idx]
    sc -= sc.max()
    e = np.exp(sc)
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for i, k in enumerate(idx):
        sh = (n - 1) - k
        wk = w[i]
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return obj, g * (2.0 / n)

def _grad_desc(h, n, alpha, K, step_sizes, rounds, check_fn):
    """Gradient descent with line search on softmax-smoothed max."""
    best_h = h.copy()
    best_c = _eval(h)
    steps = list(step_sizes)
    stall = 0
    for _ in range(rounds):
        if check_fn():
            break
        _, g = _softmax_obj_grad(h, n, alpha, K)
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < best_c - 1e-16:
                h = h_new
                best_c = c5n
                best_h = h.copy()
                improved = True
                steps[si] = min(steps[si] * 1.4, 1.0)
                break
        if not improved:
            steps = [s * 0.4 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return best_h, best_c

def _fista(h, n, alpha, K, L, iters, check_fn):
    """FISTA (accelerated gradient) for smoothed max."""
    x = h.copy()
    y = h.copy()
    tk = 1.0
    best_h = x.copy()
    best_c = _eval(x)
    for k in range(iters):
        if check_fn():
            break
        _, g = _softmax_obj_grad(y, n, alpha, K)
        step = 1.0 / L
        x_new = _project(y - step * g, n * 0.5)
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) * 0.5
        y = x_new + ((tk - 1.0) / tk_new) * (x_new - x)
        x = x_new
        tk = tk_new
        c5n = _eval(x)
        if c5n < best_c:
            best_c = c5n
            best_h = x.copy()
    return best_h, best_c

def _local_search(h, n, rng, c5c, ntry, scale, check_fn):
    """Point-pair local search."""
    best_h = h.copy()
    best_c = c5c
    h_work = h.copy()
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h_work[i] + d, h_work[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_work[i], h_work[j] = ni, nj
            c5n = _eval(h_work)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h_work.copy()
            else:
                h_work[i], h_work[j] = h_work[i] - d, h_work[j] + d
    return best_h, best_c

def _block_perturb(h, n, rng, c5c, ntry, block_size, scale, check_fn):
    """Perturb contiguous blocks of h."""

    best_c = c5c

            break

            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h_try.copy()
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0. 0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _softmax_grad(h, n, alpha, K):
    c = _corr(h)
    m = c.max()
    idx = np.argsort(c)[-K:]
    sc = alpha * c[idx]
    sc -= sc.max()
    e = np.exp(sc)
    w = e / e.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        sh = (n - 1) - k
        wk = w[i]
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g * (2.0 / n)

def _gd(h, n, alpha, K, steps_init, rounds, check_fn):
    best_h, best_c = h.copy(), _eval(h)
    st = list(steps_init)
    stall = 0
    for _ in range(rounds):
        if check_fn():
            break
        g = _softmax_grad(h, n, alpha, K)
        ok = False
        for si in range(len(st)):
            hn = _project(h - st[si] * g, n * 0.5)
            cn = _eval(hn)
            if cn < best_c - 1e-16:
                h, best_c, best_h = hn, cn, hn.copy()
                ok = True
                st[si] = min(st[si] * 1.35, 2.0)
                break
        if ok:
            stall = 0
        else:
            st = [s * 0.4 for s in st]
            stall += 1
            if stall > 12:
                break
    return best_h, best_c

def _fista(h, n, alpha, K, step, iters, check_fn):
    x, y, tk = h.copy(), h.copy(), 1.0
    best_h, best_c = x.copy(), _eval(x)
    for k in range(iters):
        if check_fn():
            break
        g = _softmax_grad(y, n, alpha, K)
        xn = _project(y - step * g, n * 0.5)
        tkn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        y = xn + ((tk - 1.0) / tkn) * (xn - x)
        x = xn
        tk = tkn
        cn = _eval(x)
        if cn < best_c:
            best_c, best_h = cn, x.copy()
    return best_h, best_c

def _local(h, n, rng, c5c, ntry, scale, check_fn):
    best_h, best_c = h.copy(), c5c
    hw = h.copy()
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = scale * rng.randn()
        ni, nj = hw[i] + d, hw[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hw[i], hw[j] = ni, nj
            cn = _eval(hw)
            if cn < c5c - 1e-18:
                c5c = cn
                if cn < best_c:
                    best_c, best_h = cn, hw.copy()
            else:
                hw[i], hw[j] = hw[i] - d, hw[j] + d
    return best_h, best_c

def _local3(h, n, rng, c5c, ntry, scale, check_fn):

        d2 = scale * rng.randn()
        ni, nj, nk = hw[i] + d1, hw[j] + d2, hw[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:

                    best_c, best_h = cn, hw.copy()
            else:

    return best_h, best_c

def _block_explore(h, n, rng, c5c, ntry, bs, scale, check_fn):

    for _ in range(ntry):
        if check_fn():

        s = rng.randint(0, n - bs)

        cn = _eval(ht)
        if cn < c5c - 1e-14:
            h, c5c = ht, cn
            if cn < best_c:

    return best_h, best_c

def _shift_explore(h, n, rng, c5c, ntry, check_fn):
