def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

def _grad_active(h, n, lags):
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g / len(lags)

    v = np.zeros(n)
    steps = list(steps_init)
    stall = 0
    for r in range(rounds):
        if check_fn():
            break
        frac = r / max(rounds - 1, 1)
        temp = temp_init * (temp_final / temp_init) ** frac
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        active = np.where(c > cmax - temp)[0]
        if len(active) < 3:
            active = np.argsort(c)[-5:]
        g = _grad_active(h, n, active)
        v = beta * v + (1.0 - beta) * g
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.5]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.4 for s in steps]
            stall += 1
            if stall > 10:
                v *= 0.2
                stall = 0
                if steps[0] < 5e-7:

        else:
            stall = 0

        h_new = _project(best_h + pert, nc * 0.5)

        if c5n < best_c - 1e-16:
            best_c = c5n
            best_h = h_new
    return best_h, best_c

def _lsx_obj(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 30000.0
    sc = np.clip(a * (c - m), -250, 250)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 30000.0
    sc = np.clip(a * (c - m), -250, 250)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    two_over_n = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * two_over_n
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)

    def _bc():
        return time.time() - t0 > budget_s - 15

    # Phase 1: Multi-start gradient descent (fast, ~30s)
    starts = [h_init.copy(), np.roll(h_init, 2), np.roll(h_init, -3)]
    for sv in starts:
        if _bc():
            break
        sv = _repair(sv.copy(), n0)
        sv += 0.002 * rng.randn(n0)
        sv = _repair(sv, n0)
        h2, c52 = _descent(sv, n0, 500, _bc,
                           [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4],
                           0.85, 0.025, 0.002)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Focused gradient with Nesterov
    for tp in [(0.012, 0.001), (0.006, 0.0004), (0.003, 0.0002)]:
        if _bc():
            break
        h2, c52 = _descent(best_h.copy(), n0, 400, _bc,
                           [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4],
                           0.9, tp[0], tp[1])
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: SLSQP polish
    if time.time() - t0 < budget_s - 300:
        bnds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n0,)}]
        for _ in range(2):
            if _bc():
                break
            try:
                res = minimize(_lsx_obj, best_h, args=(n0,), jac=_lsx_grad,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16})
                hr = _repair(res.x, n0)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 4: Sine kicks
    if time.time() - t0 < budget_s - 200:
        ns = min(3000, int((budget_s - 200 - (time.time() - t0)) / 0.004))
        h2, c52 = _sine_kick(best_h, ns, rng, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
        # More gradient after sine kicks
        if time.time() - t0 < budget_s - 100:
            h2, c52 = _descent(best_h.copy(), n0, 200, _bc,
                               [0.008, 0.004, 0.002, 0.001, 5e-4],
                               0.9, 0.005, 0.0005)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2

    # Phase 5: Extended 3-point local search
    if time.time() - t0 < budget_s - 60:
        rem = budget_s - 60 - (time.time() - t0)
        ntr = int(rem / 0.0025)
        h2, c52 = _local3(best_h, ntr, rng, best_c5, _bc, 1.5e-4)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: 2-point fine polish
    if time.time() - t0 < budget_s - 5:
        rem = budget_s - 5 - (time.time() - t0)
        ntr = int(rem / 0.002)
        h2, c52 = _local2(best_h, ntr, rng, best_c5, _bc, 8e-5)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
