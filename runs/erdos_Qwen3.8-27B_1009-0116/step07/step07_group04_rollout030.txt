```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Use binary search for exact projection
        lo, hi = -1.5, 1.5
        for _ in range(80):
            mid = (lo + hi) * 0.5
            s2 = np.clip(h - mid, 0, 1).sum()
            if s2 > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h

def _proj(h, target):
    """Project h onto {0≤x≤1, sum(x)=target}"""
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlations(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_active(h, active_k):
    """Gradient of (1/|A|)*sum_{k in A} c_k evaluated at h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active_k:
        # k is offset in np.correlate output, map to shift
        sh = (n - 1) - k
        # ∂c_k/∂h[j] = dx*(1-h[(j+sh)%n]) - dx*h[(j-sh)%n]
        for j in range(n):
            g[j] += dx * (1.0 - h[(j + sh) % n]) - dx * h[(j - sh) % n]
    g /= len(active_k)
    return g
```

```python
def _cvxpy_cutting_plane(h, n_active, n_iters, t0, budget_s):
    import cvxpy as cp
    n = len(h)
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 15:
            break
        c = _correlations(h)
        top_idx = np.argsort(c)[-n_active:]
        # Linearize at current h
        t = cp.Variable()
        h_var = cp.Variable(n)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        for k_idx in top_idx:
            sh = (n-1) - k_idx
            g_k = (1.0 - np.roll(h, -sh)) - np.roll(h, sh)  # gradient of c_k at h (times dx)
            # c_k ≈ c[k_idx] + dx * g_k · (h_var - h)
            c_k_linear = c[k_idx] + (2.0/n) * (g_k @ (h_var - h))
            constraints.append(c_k_linear <= t)
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.ECOS)
        if h_var.value is not None:
            h_new = np.array(h_var.value).flatten()
            h_new = np.clip(h_new, 0, 1)
            h_new = _proj(h_new, n/2.0)
            if _eval(h_new) < _eval(h):
                h = h_new
    return h
```

```python
def _proj(h, target):
    lo, hi = -2.0, 2.0

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
```

The previous solution got 0.380891. The target is 0.38085. The gap is 0.00004. This is extremely small and requires very precise optimization.

Key insight: at the optimum, the gradient of f(h) = max_k c_k(h) should be "balanced" - the gradient contributions from all active k's should sum to zero (projected onto the feasible set). If I can find the exact active set and solve the KKT conditions, I'll be at the optimum.

For a small active set (say 5 shifts), the KKT conditions are:
- There exist λ_k ≥ 0, ∑λ_k = 1 such that for each i:
  - If 0 < h[i] < 1: ∑_k λ_k · ∂c_k/∂h[i] + ν = 0 (where ν is the Lagrange multiplier for the sum constraint)
  - If h[i] = 0: ∑_k λ_k · ∂c_k/∂h[i] + ν ≤ 0
  - If h[i] = 1: ∑_k λ_k · ∂c_k/∂h[i] + ν ≥ 0

This is a system of equations that I could potentially solve iteratively.

**New idea: Dual ascent on λ**

1. Start with h from initial_h_values.
2. Compute c[k], find active set A (top-k shifts).
3. Initialize λ_k = 1/|A| for k ∈ A.
4. Iterate:
   a. Compute g = ∑_{k∈A} λ_k · ∇c_k at current h.
   b. Update h: h_new = proj(h - α·g, [0,1]^n, sum=n/2).
   c. Update λ: λ_k ← λ_k · exp(β·c[k]) / Z (softmax on active set).
   d. Optionally add/remove shifts from active set.
5. Decrease α over time.

This is essentially a primal-dual method for the min-max problem. It might converge faster than the previous approaches because the dual variables (λ) directly encode which constraints are active.

```python
import numpy as np
import time
import cvxpy as cp

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) > 1e-10:
        h = _proj(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_single(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _grad_weighted(h, k_indices, weights):
    n = len(h)
    g = np.zeros(n)
    for idx, k in enumerate(k_indices):
        g += weights[idx] * _grad_single(h, k)
    return g

def _nesterov_adapt(h, k0, k1, steps, step0, t0, time_limit):
    n = len(h)
    v = np.zeros(n)
    step = step0
    mu = 0.82
    for s in range(steps):
        if time.time() > time_limit:
            break
        k_cur = k0 + int((k1 - k0) * s / max(steps, 1))
        c = _corr_full(h)
        top_k = np.argsort(c)[-k_cur:]
        g = _grad_weighted(h, top_k, np.ones(k_cur) / k_cur)
        v = mu * v + (1 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < c.max():
            h = h_new
            step = min(step * 1.04, step0 * 30)
        else:
            step *= 0.45
            if step < 1e-12:
                break
            h_new = _proj(h - step * v, n * 0.5)
            if _eval(h_new) < _eval(h):
                h = h_new
    return _repair(h, n)

def _primal_dual(h, k_start, k_end, steps, step0, t0, time_limit):
    n = len(h)
    c = _corr_full(h)
    top_idx = list(np.argsort(c)[-k_start:])
    lam = np.ones(len(top_idx)) / len(top_idx)
    step = step0
    beta = 100.0
    for s in range(steps):
        if time.time() > time_limit:
            break
        k_cur = k_start + (k_end - k_start) * s // max(steps, 1)
        g = _grad_weighted(h, top_idx, lam)
        h = _proj(h - step * g, n * 0.5)
        c = _corr_full(h)
        new_top = list(np.argsort(c)[-k_cur:])
        if new_top != top_idx:
            top_idx = new_top
            lam = np.ones(len(top_idx)) / len(top_idx)
        c_act = np.array([c[k] for k in top_idx])
        log_w = beta * c_act
        log_w -= log_w.max()
        w = np.exp(log_w)
        w /= w.sum()
        lam = 0.6 * lam + 0.4 * w
        step *= 0.9995
        beta *= 1.002
    return _repair(h, n)

def _cvxpy_refine(h, k_top, n_iters, t0, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_iters):
        if time.time() > time_limit:
            break
        c = _corr_full(best_h)
        top_k = list(np.argsort(c)[-k_top:])
        t_var = cp.Variable()
        h_var = cp.Variable(n)
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n * 0.5]
        for k in top_k:
            g_k = _grad_single(best_h, k)
            cons.append(c[k] + g_k @ (h_var - best_h) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, max_iter=8000, tol=1e-9)
        except Exception:
            continue
        if h_var.value is None:
            continue
        h_lp = np.clip(np.array(h_var.value).flatten(), 0, 1)
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1]:
            h_try = _proj(best_h + alpha * (h_lp - best_h), n * 0.5)
            c_try = _eval(h_try)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                break
    return best_h, best_c

def _sinusoidal_search(h, n, rng, t0, time_limit, n_tries=300):

    for _ in range(n_tries):

        pert = best_h + sig * eps * np.sin(2 * np.pi * m * np.arange(n) / n + phi)
        h_try = _proj(pert, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()

                if c_try2 < best_c - 1e-15:
                    best_c = c_try2
                    best_h = h_try2.copy()
                    h_try = h_try2
                else:
                    break
    return best_h, best_c

def _block_local(h, n, rng, t0, time_limit, n_tries=8000):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):
        if time.time() > time_limit:
            break
        i = rng.randint(n - 4)
        blk = rng.randint(2, 5)
        j = (i + rng.randint(5, n // 2)) % (n - blk)
        d = 4e-5 * rng.randn()
        for b in range(blk):
            ni = h[i + b] + d
            nj = h[j + b] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i + b] = ni
                h[j + b] = nj
            else:
                for b2 in range(b):
                    h[i + b2] -= d
                    h[j + b2] += d
                break
        else:
            c_try = _eval(h)
            if c_try < best_c - 1e-18:
                best_c = c_try
                best_h = h.copy()
            else:
                for b in range(blk):
                    h[i + b] -= d
                    h[j + b] += d
    return best_h, best_c

def _fine_2pt(h, n, rng, t0, time_limit, n_tries=30000):

    for _ in range(n_tries):

                best_c = c_try
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()
```
