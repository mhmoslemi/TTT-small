            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_c.copy()

    # Phase 6: Final Nesterov polish (k=1, very focused)
    t_end6 = min(t0 + 50, t0 + budget_s - 40)
    h6, f6 = _nesterov_topk(best_h.copy(), 1, 5000, 0.0003, t_end6)
    if f6 < best_c5:
        best_c5, best_h = f6, h6.copy()

    # Phase 7: 3-point local search
    t_end7 = min(t0 + 80, t0 + budget_s - 20)
    best_h, best_c5 = _local_3pt(best_h, 60000, rng, t_end7, 5e-5)

    # Phase 8: 2-point local search (finer)
    t_end8 = min(t0 + 100, t0 + budget_s - 3)
    best_h, best_c5 = _local_2pt(best_h, 80000, rng, t_end8, 1e-5)

    # Phase 9: Ultra-fine 2-point
    t_end9 = t0 + budget_s - 1
    best_h, best_c5 = _local_2pt(best_h, 50000, rng, t_end9, 3e-6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
