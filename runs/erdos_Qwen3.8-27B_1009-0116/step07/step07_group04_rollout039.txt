import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_k(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        end = n - sh
        g[:end] = 1.0 - h[sh:]
        if sh > 0: g[sh:] -= h[:end]
    else:
        sp = -sh
        g[sp:] = 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g * dx

def _nesterov(h, t0, budget, phases, base_step):
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n)
    mu, step = 0.80, base_step
    for k_val, n_it in phases:
        for _ in range(n_it):
            if time.time() - t0 > budget - 8: return best_h, best_f
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top = np.argpartition(c, -k_val)[-k_val:]
            g = np.zeros(n)
            for kk in top:
                g += _grad_k(h, kk, n)
            g /= k_val
            hp = h.copy()
            v = mu * v + (1.0 - mu) * g
            hn = _proj_sum(h - step * v, n * 0.5)
            fn = _eval(hn)
            if fn < f:
                h, f = hn, fn
                if f < best_f: best_f, best_h = f, h.copy()
                step = min(step * 1.015, base_step * 8.0)
                mu = min(mu + 0.001, 0.93)
            else:
                step *= 0.6
                if step < 1e-10: break
                h = _proj_sum(hp - step * v, n * 0.5)
                f = _eval(h)
    return best_h, best_f

def _slsqp_solve(h_in, alpha, maxiter, ftol):
    n = len(h_in)
    dx = 2.0 / n
    target = n * 0.5
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        ax = np.clip(alpha * (c - m), -500.0, 500.0)
        return m + np.log(np.sum(np.exp(ax - ax.max()))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        ax = np.clip(alpha * (c - m), -500.0, 500.0)
        w = np.exp(ax - ax.max())
        ws = w.sum()
        if ws < 1e-300: return np.zeros(n)
        w /= ws
        g = np.zeros(n)
        thr = 1e-12
        for k in range(2 * n - 1):
            if w[k] < thr: continue
            g += w[k] * _grad_k(x, k, n)
        return g
    def _cons_fun(x):
        return x.sum() - target
    def _cons_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h_in.copy(), jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h_in, n)

def _local_search(h_in, n_tries, rng, t0, budget):
    n = len(h_in)
    h = h_in.copy()
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    step = 5e-5
    stag = 0

                stag += 1
                if stag > 200:

                    stag = 0
                    if step < 1e-9: break
