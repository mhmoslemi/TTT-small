import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smoothed_grad(h, n, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp((c - m) / temp)
    w = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s >= 0:
            g[:n - s] += wk * (1.0 - h[s:])
            if s > 0:
                g[s:] -= wk * h[:n - s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _topk_grad(h, n, top_k):
    c = _eval_full(h)
    idx = np.argsort(c)[-top_k:]
    w = np.exp(c[idx] - c[idx].max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, k in zip(w, idx):
        s = k - (n - 1)
        wk2 = wk * dx
        if s >= 0:
            g[:n - s] += wk2 * (1.0 - h[s:])
            if s > 0:
                g[s:] -= wk2 * h[:n - s]
        else:
            sp = -s
            g[sp:] += wk2 * (1.0 - h[:n - sp])
            g[:n - sp] -= wk2 * h[sp:]
    return g

def _pgd_hb(h, n_iters, t_start, t_end, mu, step0, check_fn):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    prev = h.copy()
    step = step0
    stall = 0
    for it in range(n_iters):
        if check_fn():
            break
        frac = it / max(n_iters - 1, 1)
        temp = t_start * (t_end / t_start) ** frac
        c = _eval_full(h)
        cur_max = c.max()
        g = _smoothed_grad(h, n, temp)
        y = h + mu * (h - prev)
        h_new = _project(y - step * g, n * 0.5)
        c_new = _eval_full(h_new)
        if c_new.max() < cur_max - 1e-16:
            prev = h.copy()
            h = h_new
            step = min(step * 1.03, 0.6)
            stall = 0
        else:
            step *= 0.65
            prev = h
            stall += 1
            if stall > 20:
                step = max(step, 0.005)
                mu = 0.5
                stall = 0
        if c_new.max() < best_c - 1e-15:
            best_c = c_new.max()
            best_h = h_new.copy()
    return best_h, best_c

def _topk_pgd(h, n_iters, top_k, step0, check_fn):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    prev = h.copy()
    step = step0
    stall = 0
    for it in range(n_iters):
        if check_fn():
            break
        c = _eval_full(h)
        cur_max = c.max()
        g = _topk_grad(h, n, top_k)
        y = h + 0.7 * (h - prev)
        h_new = _project(y - step * g, n * 0.5)
        c_new = _eval_full(h_new)
        if c_new.max() < cur_max - 1e-16:
            prev = h.copy()
            h = h_new
            step = min(step * 1.02, 0.3)
            stall = 0
        else:
            step *= 0.7
            prev = h
            stall += 1
            if stall > 25:
                step = max(step, 0.003)
                stall = 0
        if c_new.max() < best_c - 1e-15:
            best_c = c_new.max()
            best_h = h_new.copy()
    return best_h, best_c

def _sa_local(h, ntry, rng, t0, check_fn):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for it in range(ntry):
        if check_fn():
            break
        T = t0 * (0.3 / t0) ** (it / max(ntry - 1, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        if rng.rand() < 0.3:
            k = (j + 1 + rng.randint(0, 8)) % n
            d1 = 0.02 * rng.randn()
            d2 = 0.02 * rng.randn()
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h[i], h[j], h[k] = ni, nj, nk
                c_new = _eval(h)
                dC = c_new - cur_c
                if dC < 0 or rng.rand() < np.exp(-dC / max(T, 1e-10)):
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
        else:
            d = 0.015 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                dC = c_new - cur_c
                if dC < 0 or rng.rand() < np.exp(-dC / max(T, 1e-10)):
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _fine_polish(h, ntry, rng, check_fn):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for it in range(ntry):
        if check_fn():

        i = rng.randint(n)

            else:
