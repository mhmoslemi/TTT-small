import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').copy() * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_pos(p, h, n):
    g = np.zeros(2 * n - 1)
    s1_lo = max(-(n - 1), p - n + 1)
    s1_hi = min(n - 1, p)
    if s1_lo <= s1_hi:
        s1 = np.arange(s1_lo, s1_hi + 1)
        g[n - 1 + s1] += (1.0 - h[p - s1])
    s2_lo = max(-(n - 1), -p)
    s2_hi = min(n - 1, n - 1 - p)
    if s2_lo <= s2_hi:
        s2 = np.arange(s2_lo, s2_hi + 1)
        g[n - 1 + s2] -= h[p + s2]
    return g * (2.0 / n)

def _grad_lag(h, s, n):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += (1.0 - h[:-s])
        g[:n - s] -= h[s:]
    else:
        sp = -s
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _cd(h, n_iter, rng, t0, deadline, topk=3):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    c5 = float(c.max())
    best_c = c5
    best_h = h.copy()
    stable = 0
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        if stable > 10000 and it % 500 == 0:
            break
        # Select pair: gradient-guided on top-k lags
        if it % 4 != 3:
            top_idx = np.argsort(c)[-topk:]
            s_list = top_idx - (n - 1)
            grad_sum = np.zeros(n)
            for si in s_list:
                grad_sum += _grad_lag(h, int(si), n)
            grad_sum /= len(top_idx)
            i = int(np.argmin(grad_sum))
            j = int(np.argmax(grad_sum))
            if i == j:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        g = _grad_pos(i, h, n) - _grad_pos(j, h, n)
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])

            elif sg < -1e-13:

def _subgrad(h, n_iter, rng, t0, deadline):

    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    m_top = 5
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = int(k) - (n - 1)
            g += _grad_lag(h, sh, n)
        g *= dx / m_top
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.02, 0.5 / gn)
        for s_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_sum(h - s_mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def _polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(800000):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        scale = max(1e-6, 0.02 * max(0.01, c5_cur - 0.3808))
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5_cur - 1e-18:
            c5_cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Try structured perturbations for diverse starts
    for ps in [0.02, 0.05, 0.1, 0.2, 0.4]:
        hp = np.clip(h + ps * rng.randn(n0), 0, 1)
        hp = _proj_sum(hp, n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 1: Quick subgradient (10% budget)
    dl1 = t0 + budget_s * 0.08
    h1, c5_1 = _subgrad(best_h.copy(), 2000, rng, t0, dl1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Aggressive CD with correct quadratic (40% budget)
    dl2 = t0 + budget_s * 0.48
    h2, c5_2 = _cd(best_h.copy(), 200000, rng, t0, dl2, topk=3)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Subgradient refinement (10% budget)
    dl3 = t0 + budget_s * 0.58
    h3, c5_3 = _subgrad(best_h.copy(), 2000, rng, t0, dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Second CD round with different topk (20% budget)
    dl4 = t0 + budget_s * 0.78
    h4, c5_4 = _cd(best_h.copy(), 200000, rng, t0, dl4, topk=5)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Fine polish (20% budget)
    dl5 = t0 + (budget_s - 5)
    h5, c5_5 = _polish(best_h.copy(), rng, t0, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Final: one more short CD
    if time.time() < t0 + budget_s - 2:
        h6, c5_6 = _cd(best_h.copy(), 50000, rng, t0, t0 + budget_s - 2, topk=2)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
