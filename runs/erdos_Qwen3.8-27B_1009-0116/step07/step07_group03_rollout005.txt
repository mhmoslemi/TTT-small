```python
    t5 = t0 + budget_s - 4
    if time.time() < t5:
        h2, fv = _fine_search(best_h.copy(), t5, rng)
        if fv < best_c5:
            best_c5 = fv
            best_h = h2.copy()

    # Final repair
    best_h = _proj_to_sum(best_h, n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
