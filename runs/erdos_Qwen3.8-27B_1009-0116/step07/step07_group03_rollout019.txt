        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _pair_cd(h, n, duration_s, top_T, rng, t_end):

    best_c5 = _eval(h)
    end_t = min(time.time() + duration_s, t_end - 3)
    prev_best = best_c5
    stuck = 0
    while time.time() < end_t:
        c = _c_vals(h, n)
        top_k = np.argsort(c)[-top_T:]
        n_sweeps = n  # one full sweep
        for _ in range(n_sweeps):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n // 8)) % n
            if i == j:
                continue
            # For top constraints, compute the effect of moving mass from i to j
            # c_k changes by: delta * (grad_k[i] - grad_k[j])
            b = np.zeros(top_T)
            for ti, k in enumerate(top_k):
                sh = k - (n - 1)
                if sh >= 0:
                    if i < n - sh:

                    if sh > 0 and i >= sh:

                        b[ti] += dx * h[j - sh]

                    if i + sp < n:

                    if j >= sp:
                        b[ti] -= dx * (1.0 - h[j - sp])
                    if j + sp < n:
                        b[ti] += dx * h[j + sp]
            
            # We want to minimize max(c[top_k] + d*b)
            # where d is the amount moved from i to j
            # h[i] += d, h[j] -= d
            d_lo = max(-h[i], h[j] - 1.0) + 1e-12
            d_hi = min(1.0 - h[i], h[j]) - 1e-12

            b_g = d_hi

            d_g = a + gr * (b_g - a)

                h[i] += d_opt
                h[j] -= d_opt

            stuck = 0

                stuck = 0

def _nesterov_phase(h, alpha, n_steps, step0, t_end):

    c = _c_vals(h, n)
    m = c.max()

    f = m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    v = np.zeros(n)

        c = _c_vals(h, n)
        m = c.max()
        s_clip = np.clip(alpha * (c - m), -400, 400)

        for k in range(2 * n - 1):

        if time.time() > t0 + budget_s * 0.09:
            break
        h = _nesterov(h, alpha, ns, 0.003, min(time.time() + 32, t0 + budget_s * 0.09))
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 2: Aggressive pair CD
    for top_T in [5, 10, 15, 20]:
        if time.time() > t0 + budget_s * 0.38:
            break
        h = best_h.copy()
        h, c5v = _pair_cd(h, n, budget_s * 0.07, top_T, rng, t_go)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 3: High-alpha Nesterov
    if time.time() < t0 + budget_s * 0.52:
        for alpha, ns in [(200000, 350), (800000, 250)]:
            if time.time() > t0 + budget_s * 0.52:
                break
            h = best_h.copy()
            h = _nesterov(h, alpha, ns, 0.001, min(time.time() + 38, t0 + budget_s * 0.52))
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()

    # Phase 4: Triple + pair CD
    if time.time() < t0 + budget_s * 0.72:
        h = best_h.copy()
        h, c5v = _triple_cd(h, n, budget_s * 0.06, 15, rng, t_go)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()
        for top_T in [10, 20, 30]:
            if time.time() > t0 + budget_s * 0.72:
                break
            h = best_h.copy()
            h, c5v = _pair_cd(h, n, budget_s * 0.035, top_T, rng, t_go)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()

    # Phase 5: Ultra-fine local search
    if time.time() < t0 + budget_s - 15:
        h = best_h.copy()
        c5c = _eval(h)
        remaining = t0 + budget_s - 13 - time.time()
        n_tries = int(min(remaining / 0.004, 60000))
        for _ in range(n_tries):
            if time.time() > t0 + budget_s - 13:
                break
            i = rng.randint(n)
            j = (i + rng.randint(1, 20)) % n
            if i == j:
                continue
            d = 1.5e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cc = _eval(h)
                if cc < c5c:
                    c5c = cc
                    if cc < best_c5:
                        best_c5 = cc; best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 6: Final pair CD polish
    if time.time() < t0 + budget_s - 4:
        rem = t0 + budget_s - 5 - time.time()
        if rem > 3:
            h = best_h.copy()
            h, c5v = _pair_cd(h, n, rem, 10, rng, t_go)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
