Actually, I realize the log-sum-exp subgradient is a good idea but computing the gradient correctly requires care. The log-sum-exp of c_k = (1/beta) * log(Σ_k exp(beta * c_k)) has gradient:
∂/∂h_i [LSE] = (1/beta) * Σ_k exp(beta * c_k) * ∂c_k/∂h_i / Σ_k exp(beta * c_k)
= Σ_k w_k * ∂c_k/∂h_i

where w_k = exp(beta * c_k) / Σ exp(beta * c_j).

This is a weighted average of the gradients of individual c_k, weighted by their "importance" under the LSE.

To compute this efficiently:
- Compute c = correlate(h, 1-h, 'full') * dx
- Compute weights w = softmax(beta * c) (with softmax for numerical stability)
- The gradient is Σ_k w_k * ∂c_k/∂h

But ∂c_k/∂h is a vector of length n for each k, and there are 2n-1 values of k. Computing this naively is O(n²) which is too slow.

Alternative: use the fact that ∂c_k/∂h_i has a specific structure. 

c_k (shift sh) = Σ_i h[i+sh] * (1-h[i]) (for sh ≥ 0, appropriate ranges)

∂c_k/∂h[j] = (1-h[j-sh]) for j ≥ sh (from the h[i+sh] term)
             - h[j+sh] for j ≤ n-1-sh (from the (1-h[i]) term)

So the gradient of LSE w.r.t. h[j] is:
g[j] = Σ_k w_k * [(1-h[j-sh_k]) * 1_{j≥sh_k} - h[j+sh_k] * 1_{j≤n-1-sh_k}]

This is still O(n * (number of significant w_k)). If beta is large, only top few k have significant weight, making it O(n * m_top).

For beta small (temperature high), many k contribute and it's O(n²). But we can use the fact that for small beta, LSE ≈ max + O(1/beta) and the gradient is approximately the subgradient of max.

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_mat(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _lse_grad(h, n, m, beta):

        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _fine_polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    it = 0
    while time.time() < deadline:
        it += 1
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if -1e-12 <= ni <= 1 + 1e-12 and -1e-12 <= nj <= 1 + 1e-12:
            h[i], h[j] = min(1, max(0, ni)), min(1, max(0, nj))
            c5n = _eval(h)
            if c5n < c5_cur - 1e-17:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d + h[i] - h[i], nj + d + h[j] - h[j]
                h[i] = ni - d
                h[j] = nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Try diverse perturbations
    for ps in [0.02, 0.05, 0.1, 0.2, 0.4]:
        hp = np.clip(h + ps * rng.randn(n0), 0, 1)
        hp = _proj(hp, n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 1: LSE subgradient (50s)
    dl1 = t0 + 50
    h1, c1 = _lse_phase(best_h.copy(), rng, t0, dl1, beta0=5.0, beta1=40.0, m=5)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Main pair CD (480s)
    dl2 = t0 + 530
    h2, c2 = _pair_cd(best_h.copy(), 150000, rng, dl2, bisect_iters=35)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Focused CD targeting top lag (150s)
    dl3 = t0 + 680
    h3, c3 = _focused_cd(best_h.copy(), 80000, rng, dl3)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: LSE again with higher beta (80s)
    dl4 = t0 + 760
    h4, c4 = _lse_phase(best_h.copy(), rng, t0, dl4, beta0=20.0, beta1=100.0, m=4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Another CD pass (150s)
    dl5 = t0 + 910
    h5, c5 = _pair_cd(best_h.copy(), 80000, rng, dl5, bisect_iters=35)
    if c5 < best_c5:
        best_c5, best_h = c5, h5

    # Phase 6: Focused CD (90s)
    dl6 = t0 + 1000
    h6, c6 = _focused_cd(best_h.copy(), 50000, rng, dl6)
    if c6 < best_c5:
        best_c5, best_h = c6, h6

    # Phase 7: Fine polish (80s)
    dl7 = t0 + budget_s - 3
    h7, c7 = _fine_polish(best_h.copy(), rng, dl7)
    if c7 < best_c5:
        best_c5, best_h = c7, h7

    # Final repair and verify
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
