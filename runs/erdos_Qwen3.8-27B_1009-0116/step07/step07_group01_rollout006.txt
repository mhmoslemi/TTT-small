        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_smooth(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _active_grad(h, top_k, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _all_lags(h)
    active = np.argsort(c)[-top_k:]
    c_a = c[active]
    m_a = c_a.max()
    s = np.clip(alpha * (c_a - m_a), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(active):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_pg(h, alpha, steps, step0, top_k):
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    beta = 0.9
    step = step0
    cur_c = _eval(h)
    for i in range(steps):
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        g = _active_grad(y, top_k, alpha)
        h_new = _proj(y - step * (g + beta * v), n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            m = (beta_t - 1.0) / (1.0 + beta)
            v = m * (h_new - y)
            y = h_new
            cur_c = c_new
            step = min(step * 1.05, step0 * 10.0)
        else:
            step *= 0.5
            y = h_new
    return _repair(h_new, n), cur_c

def _block_opt(h, bsz, maxit):
    n = len(h)
    nb = n // bsz
    for _pass in range(3):
        for b in range(nb):
            sl = slice(b * bsz, (b + 1) * bsz)
            x0 = h[sl].copy()
            bsum = x0.sum()
            def _ob(x, _sl=sl, _h=h):
                ht = _h.copy()
                ht[_sl] = x
                return _eval(ht)
            def _oc(x, _bs=bsum):
                return x.sum() - _bs
            def _oj(x, _bs=bsum):
                return np.ones(len(x))
            try:
                cons = [{'type': 'eq', 'fun': _oc, 'jac': _oj, 'args': (bsum,)}]
                res = minimize(_ob, x0, method='SLSQP',
                               bounds=[(0, 1)] * bsz, constraints=cons,
                               options={'maxiter': maxit, 'ftol': 1e-15})
                h[sl] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _local3(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 6 + rng.randint(0, 10)) % n
        d1 = base_step * rng.randn()
        d2 = base_step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution coarse-to-fine
    for ratio in [4, 8, 16]:
        nc = max(32, n // ratio)
        hc = _repair(_downsample(h, n // nc), nc)
        for alpha in [500, 3000, 15000]:
            if _el() > 50:
                break
            hc = _slsqp_smooth(hc, alpha, 300, 1e-14)
        h_up = _repair(_upsample(hc, n // nc), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: SLSQP with increasing alpha
    for alpha in [2000, 8000, 30000, 100000, 300000, 1000000]:
        if _el() > 280:
            break
        h_new = _slsqp_smooth(best_h, alpha, 1500, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Nesterov projected gradient with tight active sets
    if _el() < budget_s - 380:
        for alpha, steps, s0, tk in [(5000, 600, 0.004, 3),
                                      (20000, 500, 0.003, 3),
                                      (80000, 400, 0.002, 4),
                                      (300000, 350, 0.0015, 3),
                                      (800000, 300, 0.001, 5)]:
            if _el() > budget_s - 280:
                break
            h_new, c5n = _nesterov_pg(best_h.copy(), alpha, steps, s0, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Block coordinate descent
    if _el() < budget_s - 220:
        for bsz in [12, 20, 28]:
            if _el() > budget_s - 140:
                break
            h_new = _block_opt(best_h.copy(), bsz, 60)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Very high alpha SLSQP
    if _el() < budget_s - 120:
        for alpha in [3000000, 10000000]:
            if _el() > budget_s - 60:
                break
            h_new = _slsqp_smooth(best_h, alpha, 800, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 6: Restart with perturbation
    if _el() < budget_s - 40:
        h_pert = best_h + 0.002 * rng.randn(n)
        h_pert = _repair(h_pert, n)
        h_pert = _nesterov_pg(h_pert, 50000, 200, 0.002, 3)[0]
        c5p = _eval(h_pert)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = h_pert.copy()

    # Phase 7: 3-point local search
    if _el() < budget_s - 10:
        remaining = budget_s - 10 - _el()
        ntries = min(int(remaining / 0.002), 200000)
        best_h, best_c5 = _local3(best_h, ntries, rng, 5e-6)

    # Phase 8: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 2 - _el()) / 0.001), 250000)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 5)) % nc2
            d = 1.5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
