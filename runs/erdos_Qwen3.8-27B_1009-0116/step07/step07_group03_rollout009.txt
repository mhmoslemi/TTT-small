def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        sweep_improved = False
        n_pairs = n
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            dist = 1 + rng.randint(0, max(2, n // 6))
            j = (i + dist) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(28):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 45:
            h += rng.randn(n) * (5e-6 + 1e-5 * rng.rand())
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _single_lag_attack(h, n, n_steps, t_end):
    """Directly target the single worst lag with projected gradient steps."""
    dx = 2.0 / n
    best_c5 = _eval(h)
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = int(np.argmax(c))
        g = _lag_grad(h, k_max, n, dx)
        # Projected gradient descent step
        for lr in [5e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
            h_new = _proj_to_sum(h - lr * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c5:
                h = h_new
                best_c5 = c_new
                break
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        if time.time() > t_end:
            break
        # Adaptive step size: shrink over time
        frac = it / max(1, n_iter)
        ss = 3e-6 * (1.0 - 0.6 * frac) + 1e-7
        i = rng.randint(n)
        dist = 1 + rng.randint(0, 10)
        j = (i + dist) % n
        d = ss * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _sin_perturb(h, n, rng):
    freq = rng.uniform(1.5, n / 8.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.008, 0.04)
    x = np.arange(n)
    pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
    return _proj_to_sum(h + pert, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov warm-up (short, multiple alphas)
    for alpha in [2000, 8000, 40000, 150000]:
        t_end = min(t0 + 18, t0 + budget_s * 0.07)
        h = _nesterov(h, alpha, 300, 0.003, t_end)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coordinate descent (main workhorse)
    for top_T in [3, 5, 7, 4]:
        if time.time() > t0 + budget_s * 0.32:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.07, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Single-lag direct attack
    if time.time() < t0 + budget_s * 0.45:
        h = best_h.copy()
        n_att = max(100, int((t0 + budget_s * 0.45 - time.time()) / 0.01))
        h, c5v = _single_lag_attack(h, n, n_att, t0 + budget_s * 0.45)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Structured perturbation + re-optimize
    if time.time() < t0 + budget_s * 0.60:
        for _ in range(5):
            if time.time() > t0 + budget_s * 0.57:
                break
            h = _sin_perturb(best_h, n, rng)
            h = _nesterov(h, 80000, 200, 0.002, min(time.time() + 12, t0 + budget_s * 0.55))
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()
            h2 = best_h.copy()
            h2, c5v2 = _coord_descent(h2, n, 12, 5, rng, t0 + budget_s * 0.58)
            if c5v2 < best_c5:
                best_c5 = c5v2
                best_h = h2.copy()

    # Phase 5: High-alpha Nesterov polish
    if time.time() < t0 + budget_s * 0.72:
        for alpha in [400000, 1500000]:
            if time.time() > t0 + budget_s * 0.70:
                break
            h = best_h.copy()
            h = _nesterov(h, alpha, 300, 0.001, min(time.time() + 18, t0 + budget_s * 0.70))
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 6: Extended coordinate descent
    if time.time() < t0 + budget_s * 0.84:
        for top_T in [3, 5, 4, 6]:
            if time.time() > t0 + budget_s * 0.82:
                break
            h = best_h.copy()
            h, c5v = _coord_descent(h, n, budget_s * 0.03, top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 7: Ultra-fine local search
    if time.time() < t0 + budget_s - 8:
        remaining = (t0 + budget_s) - time.time() - 5
        n_tries = min(int(remaining / 0.0025), 80000)
        if n_tries > 500:
            h = best_h.copy()
            best_h, best_c5 = _local_search(h, n_tries, rng, t0 + budget_s - 3)

    # Phase 8: Final coordinate polish
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, max(1, budget_s - (time.time() - t0) - 2), 5, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
