    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_corr_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:n])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _topk_grad(h, topk):
    n = len(h)
    g = np.zeros(n)
    for k in topk:
        g += _grad_corr_k(h, k, n)
    return g / len(topk)

def _obj_pnorm(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, mode='full') * dx, 1e-30)
    return np.sum(c**p)**(1.0/p) + pen * (h.sum() - n*0.5)**2

def _grad_pnorm(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, mode='full') * dx, 1e-30)
    cp = c**p
    s = cp.sum()
    if s < 1e-30:
        return 2.0*pen*(h.sum()-n*0.5)*np.ones(n)
    factor = s**(1.0/p - 1.0)
    g = np.zeros(n)
    for k in range(2*n - 1):
        w = factor * c[k]**(p-1)
        if w > 1e-15:
            g += w * _grad_corr_k(h, k, n)
    return g + 2.0*pen*(h.sum()-n*0.5)*np.ones(n)

def _lbfgs(h, p, pen, maxiter):
    n = len(h)
    try:
        res = minimize(_obj_pnorm, h, jac=_grad_pnorm, args=(p, pen),
                       method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-13, 'maxls': 40})
        return _proj_sum(res.x, n*0.5)
    except Exception:
        return _proj_sum(h, n*0.5)

def _subgrad_descent(h, n_iter, k_init, k_max, step0, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_init
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-k:]
        g = _topk_grad(h_c, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break
        v /= vn
        improved = False
        for s in [step, step*0.5, step*0.25, step*1.5]:
            h_try = _proj_sum(h_c - s * v, n*0.5)
            c_try = _eval(h_try)
            if c_try < best_c - 1e-16:
                h_c = h_try
                best_c = c_try
                best_h = h_c.copy()
                improved = True
                step = min(s * 1.08, 0.06)
                stall = 0
                break
        if not improved:
            step *= 0.5
            stall += 1
            if stall > 60:
                if stall > 200:
                    break
                v[:] = 0.0
                step = step0 * 0.7
                stall = 0
                k = min(k + 2, k_max)
    return best_h, best_c

def _fine_local(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 5e-5
    for it in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = rng.randn() * scale
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            c_new = _eval(h_c)
            if c_new < best_c:
                best_c = c_new
                best_h = h_c.copy()
                scale = min(scale * 1.01, 8e-5)
            else:
                h_c[i], h_c[j] = ni - d, nj + d
        if it % 8000 == 7999:
            scale *= 0.75
            if scale < 1e-7:
                scale = 5e-5
    return best_h, best_c

def _3pt_local(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 3e-5
    for it in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        m = (i + 9 + rng.randint(0, 20)) % n
        d1 = rng.randn() * scale
        d2 = rng.randn() * scale
        ni, nj, nm = h_c[i]+d1, h_c[j]+d2, h_c[m]-d1-d2
        if 0<=ni<=1 and 0<=nj<=1 and 0<=nm<=1:
            h_c[i], h_c[j], h_c[m] = ni, nj, nm
            c_new = _eval(h_c)
            if c_new < best_c:
                best_c = c_new
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j], h_c[m] = ni-d1, nj-d2, nm+d1+d2
        if it % 6000 == 5999:
            scale *= 0.8
    return best_h, best_c

def _anneal_perturb(h, rng, amplitude, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    for _ in range(200):
        if bc_fn():
            break
        idx = rng.choice(n, size=max(n//10, 50), replace=False)
        h_try = h_c + rng.randn(len(idx)) * amplitude
        h_try = _proj_sum(h_try, n*0.5)
        c_try = _eval(h_try)
        if c_try < best_c:
            h_c = h_try
            best_c = c_try
    return h_c, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    def bc():
        return el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _proj_sum(np.array(initial_h_values, dtype=float), n0 * 0.5)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick L-BFGS with moderate p
    for p in [8, 16, 32, 64]:
        if bc():
            break
        h_try = _lbfgs(best_h.copy(), p, 1e5, 200)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 2: Subgradient descent (main optimizer)
    if el() < budget_s - 350:
        best_h, best_c5 = _subgrad_descent(best_h, 40000, 2, 20, 0.02, el, bc)

    # Phase 3: L-BFGS high-p polish
    if el() < budget_s - 250:
        for p in [128, 256, 512]:
            if bc():
                break
            h_try = _lbfgs(best_h.copy(), p, 1e6, 250)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5, best_h = c_try, h_try

    # Phase 4: Subgradient again, finer
    if el() < budget_s - 180:
        best_h, best_c5 = _subgrad_descent(best_h, 30000, 1, 10, 0.008, el, bc)

    # Phase 5: Annealing perturbation to escape basin
    if el() < budget_s - 150:
        for amp in [0.02, 0.01, 0.005]:
            if bc():
                break
            h_try, c_try = _anneal_perturb(best_h, rng, amp, el, bc)
            if c_try < best_c5:
                best_c5, best_h = c_try, h_try

    # Phase 6: Subgradient fine
    if el() < budget_s - 100:
        best_h, best_c5 = _subgrad_descent(best_h, 20000, 2, 8, 0.005, el, bc)

    # Phase 7: 3-point local search
    if el() < budget_s - 60:
        nt = min(int((budget_s - 60 - el()) * 800), 50000)
        best_h, best_c5 = _3pt_local(best_h, max(nt, 5000), rng, el, bc)

    # Phase 8: 2-point fine local search
    if el() < budget_s - 25:
        nt = min(int((budget_s - 25 - el()) * 2000), 100000)
        best_h, best_c5 = _fine_local(best_h, max(nt, 10000), rng, el, bc)

    # Phase 9: Final L-BFGS
    if el() < budget_s - 5:
        h_try = _lbfgs(best_h.copy(), 512, 1e6, 100)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    best_h = _proj_sum(best_h, len(best_h) * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
