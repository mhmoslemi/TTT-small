    v_t = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    
    for t in range(1, n_steps + 1):

        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        m = c.max()

        s = np.clip(s, -500, 500)
        w = np.exp(s)
        w /= w.sum()
        
        # Gradient

        dx_ = 2.0 / n

            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k

                    g[j] += wk * (1.0 - h[j - sp])
                for j in range(n - sp):
                    g[j] -= wk * h[j + sp]

    dx_ = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx_

    # We need: for sh >= 0: g[:n-sh] += w[k]*dx*(1-h[sh:]) and g[sh:] -= w[k]*dx*h[:n-sh]
    # For sh < 0, sp = -sh: g[sp:] += w[k]*dx*(1-h[:n-sp]) and g[:n-sp] -= w[k]*dx*h[sp:]

        k = n - 1 - sh
        wk = w[k] * dx_
        if wk < 1e-14:

        g[:n-sh] += wk * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= wk * h[:n-sh]

        k = n - 1 + sp  # this is > n-1, so k goes up to 2n-2
        wk = w[k] * dx_
        if wk < 1e-14:
            continue
        g[sp:] += wk * (1.0 - h[:n-sp])
        g[:n-sp] -= wk * h[sp:]
    
    return g

        if time.time() - t0 > budget_s - 2:
            break
        i = rng.randint(m)
        j = (i + 1 + rng.randint(0, 4)) % m
        d = step * rng.randn()
        gi, gj = g[i] + d, g[j] - d
        if 0 <= gi <= 1 and 0 <= gj <= 1:
            g[i], g[j] = gi, gj
            h[:m] = g
            h[m:] = 1.0 - g[::-1]
            c = _eval(h)
            if c < cur_c:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_g = g.copy()
            else:
                g[i], g[j] = gi - d, gj + d
    return best_g, best_c

def _local2_free(h, n_iter, rng, step, t0, budget_s):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    cur_c = best_c
    for _ in range(n_iter):
        if time.time() - t0 > budget_s - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cur_c:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _proj_grad_free(h, alpha, n_steps, lr, t0, budget_s):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    cur_c = best_c
    step = lr
    for _ in range(n_steps):
        if time.time() - t0 > budget_s - 5:
            break
        g = _grad_lse(h, alpha)
        h_new = _proj(h - step * g, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            step = min(step * 1.05, lr * 8)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            step *= 0.5
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h0 = _repair(h0, n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    m = n // 2
    g = (h0[:m] + 1.0 - h0[m:][::-1]) / 2.0
    g = np.clip(g, 0.0, 1.0)
    hs = np.concatenate([g, 1.0 - g[::-1]])
    cs = _eval(hs)
    if cs < best_c5:
        best_c5 = cs
        best_h = hs.copy()

    # Phase 1: Adam symmetric, moderate alpha
    for alpha, steps, lr in [(500, 30000, 0.02), (2000, 20000, 0.01), (8000, 15000, 0.005)]:
        if time.time() - t0 > 250:
            break
        g, c_new = _adam_sym(g, alpha, steps, lr, t0, budget_s)
        hs = np.concatenate([g, 1.0 - g[::-1]])
        if c_new < best_c5:
            best_c5 = c_new
            best_h = hs.copy()

    # Phase 2: Local search symmetric
    for step, niter in [(0.02, 80000), (0.005, 150000), (0.001, 150000)]:
        if time.time() - t0 > budget_s - 350:
            break
        g, c_new = _local3_sym(g, niter, rng, step, t0, budget_s)
        hs = np.concatenate([g, 1.0 - g[::-1]])
        if c_new < best_c5:
            best_c5 = c_new
            best_h = hs.copy()

    # Phase 3: High-alpha Adam symmetric
    for alpha, steps, lr in [(30000, 10000, 0.002), (100000, 8000, 0.001), (400000, 5000, 0.0004)]:
        if time.time() - t0 > budget_s - 200:
            break
        g, c_new = _adam_sym(g, alpha, steps, lr, t0, budget_s)
        hs = np.concatenate([g, 1.0 - g[::-1]])
        if c_new < best_c5:
            best_c5 = c_new
            best_h = hs.copy()

    # Phase 4: Fine 2-point symmetric local search
    for step, niter in [(0.0005, 200000), (0.0001, 200000)]:
        if time.time() - t0 > budget_s - 150:
            break
        g, c_new = _local2_sym(g, niter, rng, step, t0, budget_s)
        hs = np.concatenate([g, 1.0 - g[::-1]])
        if c_new < best_c5:
            best_c5 = c_new
            best_h = hs.copy()

    # Phase 5: Non-symmetric projected gradient (break symmetry)
    if time.time() - t0 < budget_s - 100:
        h_ns = best_h.copy()
        for alpha, steps, lr in [(5000, 3000, 0.01), (20000, 2000, 0.005)]:
            if time.time() - t0 > budget_s - 60:
                break
            h_ns, c_new = _proj_grad_free(h_ns, alpha, steps, lr, t0, budget_s)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_ns.copy()

    # Phase 6: Ultra-fine non-symmetric 2-point polish
    if time.time() - t0 < budget_s - 5:
        rem = budget_s - 5 - (time.time() - t0)
        niter = min(int(rem / 0.0012), 400000)
        for step, frac in [(8e-5, 0.6), (2e-5, 0.4)]:
            ni2 = int(niter * frac)
            best_h, best_c5 = _local2_free(best_h, ni2, rng, step, t0, budget_s)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
