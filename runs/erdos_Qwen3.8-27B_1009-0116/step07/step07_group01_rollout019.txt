def _repair(h, n):
    # Ensure sum = n/2, 0 <= h <= 1
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h *= target / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Fix remaining by adjusting individual entries
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _active_gradient(h, n, elapsed_fn, time_budget):
    dx = 2.0 / n
    half_n = n * 0.5
    c5cur = _eval(h)
    top_k = 8  # number of active lags
    alpha = 100.0  # softmax temperature for weights
    
    for iteration in range(10000):
        if elapsed_fn() > time_budget:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = corr.max()
        
        # Get top lags
        top_idx = np.argsort(corr)[-top_k:]
        c_top = corr[top_idx]
        
        # Soft weights
        s = alpha * (c_top - m)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for ii, k in enumerate(top_idx):
            if w[ii] < 1e-15:
                continue
            # k is index in full correlation (0 to 2n-2)
            shift = k - (n - 1)  # actual shift
            # ∂c/∂h[j] for this lag:
            # + (1-h[j-shift]) for valid j-shift
            # - h[j+shift] for valid j+shift
            # Use vectorized ops
            if shift >= 0:

                i1 = np.arange(shift, n)
                g[i1] += w[ii] * (1.0 - h[i1 - shift])
                # h[j] as second factor: j+shift valid when j+shift < n, i.e., j < n-shift
                if shift > 0:
                    i2 = np.arange(n - shift)
                    g[i2] -= w[ii] * h[i2 + shift]

                # h[j] as first factor: j+sp valid when j < n-sp
                i1 = np.arange(n - sp)

                i2 = np.arange(sp, n)
                g[i2] -= w[ii] * h[i2 - sp]

            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _logsumexp_obj(x, alpha, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _logsumexp_grad(x, alpha, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    g *= dx
    return g

def _slsqp(h0, alpha, maxiter):
    n = len(h0)
    dx = 2.0 / n
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_logsumexp_obj, h0, jac=_logsumexp_grad,
                      args=(alpha, n, dx), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h0

def _subgrad_step(h, top_k, alpha_w, use_topk):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    
    if use_topk:
        # Soft weights over top-k lags
        idx = np.argsort(c)[-top_k:]
        c_t = c[idx]
        s = alpha_w * (c_t - m)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        # Build full weight vector
        wfull = np.zeros(2 * n - 1)
        for i, k in enumerate(idx):
            wfull[k] = w[i]
    else:
        # Full softmax
        s = alpha_w * (c - m)
        s -= s.max()
        wfull = np.exp(s)
        wfull /= wfull.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if wfull[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = wfull[k]
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    g *= dx
    # Project onto tangent space (sum = 0)
    g -= g.mean()
    return g, c, m

def _subgrad_opt(h, n, elapsed_fn, t_end, n_iter, top_k, alpha_w, base_step):
    target = n * 0.5
    c5cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5cur
    v = np.zeros(n)
    mu = 0.7
    step = base_step
    
    for it in range(n_iter):
        if elapsed_fn() > t_end:
            break
        
        g, c, m = _subgrad_step(h, top_k, alpha_w, top_k > 0)
        v = mu * v + (1.0 - mu) * g
        
        # Line search
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, target)
            c5n = _eval(h_new)
            if c5n < c5cur - 1e-16:
                h = h_new
                c5cur = c5n
                if c5cur < best_c5:
                    best_c5 = c5cur
                    best_h = h.copy()
                step = min(step * 1.15, base_step * 8)
                taken = True
                break
            step *= 0.5
        
        if not taken:
            step = min(step * 1.3, base_step * 8)
            # Perturb to escape
            if it % 200 == 199:
                h += 0.0005 * np.random.randn(n)
                h = _repair(h, n)
                c5cur = _eval(h)
    
    return best_h, best_c5

def _guided_local(h, n, rng, elapsed_fn, t_end, base_step):
    target = n * 0.5
    c5cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5cur
    stagnant = 0
    
    while elapsed_fn() < t_end:
        # Compute gradient guidance
        g, c, m = _subgrad_step(h, 8, 50.0, True)
        
        # Choose 3 points biased by |g|
        # Sample from distribution proportional to |g| + epsilon
        weights = np.abs(g) + 0.01
        weights /= weights.sum()
