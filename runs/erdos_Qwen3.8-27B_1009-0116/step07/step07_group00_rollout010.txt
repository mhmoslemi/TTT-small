Actually, for a fixed k:
Σ_i h[i](1-h[(i+k)%n]) = Σ_{i≠i0} h[i](1-h[(i+k)%n]) + h[i0](1-h[(i0+k)%n]) + h[(i0-k)%n](1-h[i0])

= const + h[i0](1-h[(i0+k)%n]) + h[(i0-k)%n] - h[(i0-k)%n]*h[i0]
= const + h[i0](1 - h[(i0+k)%n] - h[(i0-k)%n]) + h[(i0-k)%n]

So it's linear in h[i0]! The coefficient of h[i0] is (1 - h[(i0+k)%n] - h[(i0-k)%n]).

So for each i, f_i(v) = max_k [a_k * v + b_k] where v = h[i], and a_k, b_k are known.

The minimum of max of linear functions over v ∈ [0,1] is found where the active constraints cross.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c.copy()
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-14:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov_adaptive(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        gn = np.linalg.norm(v)
        if gn < 1e-14:
            break
        hn = _proj(h - step * v / gn * step0, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-16:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
            if step > step0 * 2.0:
                beta = min(beta + 0.005, 0.93)
        else:
            step *= 0.55
            beta = max(0.5, beta - 0.03)
            h = _proj(h - step * v / max(gn, 1e-14) * step0, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_topk(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _grad_search(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 0.004
    stall = 0
    for it in range(niter):
        if time.time() > deadline:
            break
        g = _grad_topk(h, 50000, 15)
        pos = np.where(g > 1e-12)[0]
        neg = np.where(g < -1e-12)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        pw = g[pos] ** 1.5
        nw = (-g[neg]) ** 1.5
        pws, nws = pw.sum(), nw.sum()
        if pws < 1e-20 or nws < 1e-20:
            break
        i = pos[rng.choice(len(pos), p=pw / pws)]
        j = neg[rng.choice(len(neg), p=nw / nws)]
        s = min(step, h[i], 1.0 - h[j])
        if s < 1e-13:
            stall += 1
            if stall > 2000:
                break
            continue
        h[i] -= s
        h[j] += s
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            stall = 0
            if c5n < bc:
                bc = c5n
                bh = h.copy()
            step = min(step * 1.015, 0.015)
        else:
            h[i] += s
            h[j] -= s
            step *= 0.75
            stall += 1
            if step < 1e-9:
                step = 5e-6
    return bh, bc
