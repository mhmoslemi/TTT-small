import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _lag_gradient(h, p):
    """Gradient of correlation at lag p (absolute value of lag) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if p == 0:
        return 1.0 - 2.0 * h
    if p >= n:
        return g
    # For lag ±p: grad[m] = (1-h[m-p]) for m>=p, minus h[m+p] for m<p... 
    # Actually: first term at positions p..n-1, second term at positions 0..n-1-p
    g[p:] += 1.0 - h[:n-p]
    g[:n-p] -= h[p:]
    return g

def _topk_gradient(h, k):
    """Weighted gradient focusing on top-k lags"""
    n = len(h)
    c = _correl(h)
    cmax = c.max()
    # Find indices of top-k
    if k < len(c):
        idx = np.argpartition(c, -k)[-k:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    
    # Weight by how close to max (soft selection)
    # Use softmax with high temperature for near-hard selection
    temp = max((cmax - ct).max(), 1e-12)
    w = np.exp(-(cmax - ct) / (0.1 * temp))
    w /= w.sum()
    
    # Compute weighted gradient
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-10:
            continue
        j = idx[ii]
        p = abs(j - (n - 1))
        g += w[ii] * _lag_gradient(h, p)
    return g, cmax

def _project_sum(h, target):
    """Project h onto {sum = target, 0 <= h <= 1}"""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_descent(h, topk, steps, step0, deadline):
    """Projected gradient descent with momentum, targeting top-k lags"""
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    beta = 0.9
    step = step0
    best_h = h.copy()
    best_f = f
    
    for it in range(steps):
        if time.time() > deadline:
            break
        g, cmax = _topk_gradient(h, topk)
        v = beta * v + (1 - beta) * g
        # Adaptive step
        hn = _project_sum(h - step * v, n * 0.5)
        fn = _eval(hn)
        if fn < f:
            h = hn
            f = fn
            if f < best_f:
                best_f = f
                best_h = h.copy()
            step = min(step * 1.05, step0 * 10)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _project_sum(h - step * v, n * 0.5)

    return best_h, best_f

def _structured_anneal(h, niter, rng, deadline, T0, T1):
    """Annealing with structured moves: window shifts, pattern inversions"""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:

        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))

        i0 = rng.randint(0, n)
        hn = h.copy()
        
        if move_type == 0:
            # Window bump: add gaussian bump
            bw = rng.randint(5, 30)
            amp = 0.05 * T ** 0.5
            bw = min(bw, n - i0) if i0 + bw > n else bw
            i0 = min(i0, n - 2)
            x = np.arange(bw)
            bump = amp * np.exp(-((x - bw/2)**2) / (2*(bw/4)**2))
            # Zero-mean constraint

            hn[i0:i0+bw] += bump
        elif move_type == 1:
            # Swap segments

            j0 = min(i0 + L + rng.randint(0, L), n - L)
            if j0 > i0:
                seg1 = hn[i0:i0+L].copy()
                seg2 = hn[j0:j0+L].copy()
                scale = 0.1 * T ** 0.5
                hn[i0:i0+L] = seg1 + scale * (seg2 - seg1)
                hn[j0:j0+L] = seg2 + scale * (seg1 - seg2)
        elif move_type == 2:

            i0 = min(i0, n - L)

        else:

        hn = _project_sum(hn, n * 0.5)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lag_grad(h, p):
    n = len(h)
    g = np.zeros(n)
    if p == 0:
        return 1.0 - 2.0 * h
    if p >= n:
        return g
    g[p:] += 1.0 - h[:n - p]
    g[:n - p] -= h[p:]
    return g

def _topk_grad(h, k):
