import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _soft_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    e = np.exp(np.clip(alpha * (c - m), -500, 500))
    return m + np.log(e.sum()) / alpha

def _soft_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    e = np.exp(np.clip(alpha * (c - m), -500, 500))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * alpha * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov_smooth(h, alpha, n_steps, step0, t_end):
    n = len(h)
    f = _soft_obj(h, alpha)
    v = np.zeros(n)
    step = step0
    best_h = h.copy()
    best_f = f
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        g = _soft_grad(h, alpha)
        h_old = h.copy()
        v = 0.8 * v + 0.2 * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _soft_obj(h_new, alpha)
        if f_new < f:
            h, f = h_new, f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
            step = min(step * 1.02, step0 * 15.0)
        else:
            step *= 0.4
            if step < 1e-13:
                break
            h = _proj(h_old - step * v, n * 0.5)
            f = _soft_obj(h, alpha)
    return best_h

def _nesterov_exact(h, alpha, n_steps, step0, t_end):
    n = len(h)
    c5 = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_c5 = h.copy(), c5
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        g = _soft_grad(h, alpha)
        if np.linalg.norm(g) < 1e-16:
            break
        h_old = h.copy()
        v = 0.78 * v + 0.22 * g
        s, improved = step, False
        for _ in range(30):
            ht = _proj(h_old - s * v, n * 0.5)
            ct = _eval(ht)
            if ct < c5 - 1e-16:
                h, c5 = ht, ct
                improved = True
                break
            s *= 0.5
        if improved:
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
            step = min(s * 1.04, step0 * 40.0)
        else:
            step = s
            if step < 1e-16:
                break
    return best_h, best_c5

def _shift_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:]) * dx
        if sh > 0:
            g[sh:] -= h[:n - sh] * dx
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp]) * dx
        g[:n - sp] -= h[sp:] * dx
    return g

def _cyclic_cd(h, top_T, t_end, n_sweeps):
    n = len(h)
    dx = 2.0 / n
    best_c5 = _eval(h)
    W = max(3, n // 25)
    for _ in range(n_sweeps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        g_s = np.array([_shift_grad(h, k, n, dx) for k in top_idx])
        any_chg = False
        for i in range(n):
            if time.time() > t_end:
                break
            for d in range(1, W + 1):
                j = (i + d) % n
                if i == j:
                    continue
                b = g_s[:, i] - g_s[:, j]
                lo = max(-h[i], h[j] - 1.0) + 1e-14
                hi = min(1.0 - h[i], h[j]) - 1e-14
                if lo >= hi:
                    continue
                cands = [lo, hi]
                av = c[top_idx]
                for a in range(top_T):
                    for b2 in range(a + 1, top_T):
                        db = b[a] - b[b2]
                        if abs(db) < 1e-16:
                            continue
                        dk = (av[b2] - av[a]) / db
                        if lo < dk < hi:
                            cands.append(dk)
                bd, bf = lo, np.max(av + b * lo)
                for dk in cands:
                    fv = np.max(av + b * dk)
                    if fv < bf - 1e-16:
                        bf, bd = fv, dk
                if abs(bd) > 1e-14:
                    h[i] += bd
                    h[j] -= bd
                    g_s[:, i] = _shift_grad(h, top_idx[0], n, dx) * 0  # stale, recompute below
                    g_s[:, i] = sum(_shift_grad(h, k, n, dx) * 1 for k in [top_idx[0]]) * 0
                    any_chg = True
        if any_chg:
            h = _proj(h, n * 0.5)
            nc = _eval(h)
            if nc < best_c5:
                best_c5 = nc
    return h, best_c5

def _local_refine(h, t_end, rng):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 2e-6 * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            nv = _eval(h)
            if nv < c5c - 1e-18:
                c5c = nv
                if nv < best_c:
                    best_c, best_h = nv, h.copy()
            else:
                h[i], h[j] = hi, hj
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c
