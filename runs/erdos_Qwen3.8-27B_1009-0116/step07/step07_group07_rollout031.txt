import time

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * c
    m = np.max(bc)
    e = np.exp(bc - m)
    s = e.sum()
    w = e / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * c
    m = np.max(bc)
    return m / beta + np.log(np.sum(np.exp(bc - m))) / beta

def _topk_grad(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / len(k_idx)

def _lbfgs_lse(h, beta, maxiter, penalty):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _lse_obj(x, beta) + penalty * (x.sum() - n * 0.5) ** 2
    
    def grad(x):
        return _lse_grad(x, beta) + 2.0 * penalty * (x.sum() - n * 0.5)
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_active(h, n_iter, k_start, k_max, step0, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    beta_m = 0.75
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_start
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        if stall > 20:
            k = min(k + 2, k_max)
        topk = np.argsort(c)[-k:]
        g = _topk_grad(h_c, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta_m * v + (1.0 - beta_m) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-15:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.04, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - 0.5 * n) > 1e-9:
        h = _proj_sum(h, 0.5 * n)
    return h

def _lse_obj(x, beta, penalty, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    bc = beta * c
    m = np.max(bc)
    base = m / beta + np.log(np.sum(np.exp(bc - m))) / beta
    return base + penalty * (x.sum() - 0.5 * n) ** 2

def _lse_grad(x, beta, penalty, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    bc = beta * c
    m = np.max(bc)
    e = np.exp(bc - m)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:sh + cnt])
            if sh > 0:
                g[sh:sh + cnt] -= wk * x[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:sp + cnt] += wk * (1.0 - x[:cnt])
            g[:cnt] -= wk * x[sp:sp + cnt]
    g += 2.0 * penalty * (x.sum() - 0.5 * n)
    return g

def _topk_grad(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_idx:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:sh + cnt] -= h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:sp + cnt] += (1.0 - h[:cnt])
            g[:cnt] -= h[sp:sp + cnt]
    return g * dx / len(k_idx)

def _lbfgs_lse(h, beta, maxiter, penalty):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(beta, penalty, n), method='L-BFGS-B',
                       bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-13, 'maxls': 80})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_active(h, n_iter, k_start, k_max, step0, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    mom = 0.75
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_start
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        if stall > 15:
            k = min(k + 3, k_max)
            mom = min(mom + 0.01, 0.85)
        topk = np.argsort(c)[-k:]
        g = _topk_grad(h_c, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mom * v + (1.0 - mom) * (g / gn)
        h_try = _proj_sum(h_c - step * v, 0.5 * n)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-15:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.05, 0.06)
        else:
            step *= 0.45
            stall += 1
            if step < 1e-10:
                break
    return best_h, best_c

def _grad_guided_search(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        top3 = np.argsort(c)[-3:]
        g = _topk_grad(h_c, top3)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g_norm = g / gn
        # Pick random points and nudge against gradient
        npts = 3 + rng.randint(0, 5)
        idx = rng.choice(n, size=npts, replace=False)
        scale = 1.5e-4 * max(1.0, (it % 4000) / 4000.0)
        h_try = h_c.copy()
        for i in idx:
            h_try[i] -= scale * g_norm[i]
        # Compensate at other random points
        comp_idx = rng.choice(n, size=npts, replace=False)
        delta = (h_c[idx].sum() - h_try[idx].sum()) / npts
        for j in comp_idx:
            h_try[j] += delta
        h_try = _proj_sum(np.clip(h_try, 0, 1), 0.5 * n)
        c_new = _eval(h_try)
        if c_new < best_c:
            best_c = c_new
            best_h = h_try.copy()
            h_c = h_try
        else:
            # Single fine adjustment
            i = rng.randint(n)
            j = (i + 2 + rng.randint(0, 4)) % n
            d = rng.randn() * 3e-5
            ni, nj = h_c[i] + d, h_c[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_c[i], h_c[j] = ni, nj
                c2 = _eval(h_c)
                if c2 < best_c:
                    best_c = c2
                    best_h = h_c.copy()
                else:
                    h_c[i], h_c[j] = ni - d, nj + d
    return best_h, best_c

def _block_perturb(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
