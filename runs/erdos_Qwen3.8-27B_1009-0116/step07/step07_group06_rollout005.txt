    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

One idea: **Use the dual problem**. The min-max can be written as:
min_h max_k c_k(h) = min_h min_{p≥0, Σp=1} Σ_k p_k c_k(h)  (by minimax theorem, if we had convexity)

But we don't have convexity, so this doesn't directly apply. However, we can still use it as a heuristic:
- Maintain a distribution p over lags
- Optimize h to minimize Σ_k p_k c_k (a weighted sum, which is easier since it's still quadratic)
- Update p to put more weight on lags where c_k is large
- Iterate

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, p)
    hi1 = min(2 * n - 2, p + n - 1)
    if lo1 <= hi1:
        m_arr = np.arange(lo1, hi1 + 1)
        g[lo1:hi1+1] += 1.0 - h[p + n - 1 - m_arr]
    lo2 = max(0, n - 1 - p)
    hi2 = min(2 * n - 2, n - 1 + p)
    if lo2 <= hi2:
        m_arr = np.arange(lo2, hi2 + 1)
        g[lo2:hi2+1] -= h[p + m_arr - (n - 1)]
    return g * (2.0 / n)

def _precompute_grads(h, n):
    return [_grad_pos(h, i, n) for i in range(n)]

        it += 1

            break

            h_candidate = _repair(h + temp * rng.randn(n), n)
            c5t = _eval(h_candidate)
            if c5t < c5:
                h = h_candidate
                c = _corr(h)
                c5 = c5t
                no_improve = 0
                temp *= 0.7
            else:
                temp *= 0.5
                if temp < 1e-5:
                    break

        j = rng.randint(n)
        if i == j:
            continue
        gi = _grad_pos(h, i, n)
        gj = _grad_pos(h, j, n)
        g = gi - gj
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        k_ij = i + n - 1 - j
        k_ji = j + n - 1 - i
        for _ in range(42):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k_ij < ns:
                vals[k_ij] += d_mid * d_mid
            if 0 <= k_ji < ns:
                vals[k_ji] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k_ij or kmax == k_ji:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            no_improve += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k_ij < ns:
            c[k_ij] += d_opt * d_opt
        if 0 <= k_ji < ns:
            c[k_ji] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            no_improve = 0
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k_ij < ns:
                c[k_ij] -= d_opt * d_opt
            if 0 <= k_ji < ns:
                c[k_ji] -= d_opt * d_opt
            no_improve += 1
    return best_h, best_c

def _sweep_cd(h, rng, deadline):
    """Sweep through indices sequentially with neighbor pairs."""
    n = len(h)
    ns = 2 * n - 1
    c = _corr(h)
    c5 = c.max()
    best_h = h.copy()
    best_c = c5
    no_improve = 0
    while time.time() < deadline:
        no_improve_reset = False
        for i in range(n):
            for dj in [1, 2, -1, -2, rng.randint(1, n//4)]:
                j = (i + dj) % n
                if i == j:
                    continue
                gi = _grad_pos(h, i, n)
                gj = _grad_pos(h, j, n)
                g = gi - gj
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_lo >= d_hi - 1e-15:
                    continue
                k_ij = i + n - 1 - j
                k_ji = j + n - 1 - i
                for _ in range(35):
                    d_mid = 0.5 * (d_lo + d_hi)
                    vals = c + d_mid * g
                    if 0 <= k_ij < ns:
                        vals[k_ij] += d_mid * d_mid
                    if 0 <= k_ji < ns:
                        vals[k_ji] += d_mid * d_mid
                    kmax = np.argmax(vals)
                    sg = g[kmax]
                    if kmax == k_ij or kmax == k_ji:
                        sg += 2.0 * d_mid
                    if sg > 1e-14:
                        d_hi = d_mid
                    elif sg < -1e-14:
                        d_lo = d_mid
                    else:
                        break
                d_opt = 0.5 * (d_lo + d_hi)
                if abs(d_opt) < 1e-12:
                    continue
                h[i] += d_opt
                h[j] -= d_opt
                c += d_opt * g
                if 0 <= k_ij < ns:
                    c[k_ij] += d_opt * d_opt
                if 0 <= k_ji < ns:
                    c[k_ji] += d_opt * d_opt
                c5_new = c.max()
                if c5_new < c5 - 1e-14:
                    c5 = c5_new
                    no_improve_reset = True
                    if c5 < best_c:
                        best_c = c5
                        best_h = h.copy()
                else:
                    h[i] -= d_opt
                    h[j] += d_opt
                    c -= d_opt * g
                    if 0 <= k_ij < ns:
                        c[k_ij] -= d_opt * d_opt
                    if 0 <= k_ji < ns:
                        c[k_ji] -= d_opt * d_opt
            if time.time() > deadline:
                break
        if not no_improve_reset:
            break
    return best_h, best_c

def _multiscale_refine(h, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for ratio in [2, 3]:
        n_coarse = n // ratio
        if n_coarse < 50:
            continue
        dl_sub = min(time.time() + 60, deadline)
        while time.time() < dl_sub:
            idx = np.arange(n_coarse) * ratio
            h_coarse = h[idx]
            h_coarse = _repair(h_coarse, n_coarse)
            h_opt, c_opt = _pair_cd(h_coarse, rng, time.time() + 30, escape_every=5000, temp=0.003)
            h_fine = np.repeat(h_opt, ratio)
            if len(h_fine) > n:
                h_fine = h_fine[:n]
            elif len(h_fine) < n:
                pad = np.full(n - len(h_fine), 0.5)
                h_fine = np.concatenate([h_fine, pad])
            h_fine = _repair(h_fine, n)
            c_fine = _eval(h_fine)
            if c_fine < best_c:
                best_c = c_fine
                best_h = h_fine.copy()
            if time.time() >= dl_sub:
                break
    return best_h, best_c

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad(h, rng, deadline, step_scale=1.0):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(3000):
        if time.time() > deadline:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-4:]
        g = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            for p in range(n):
                val = 0.0
                if sh >= 0:
                    if p + sh < n:
                        val += (1.0 - h[p + sh])
                    if sh > 0 and p - sh >= 0:
                        val -= h[p - sh]
                else:
                    sp = -sh
                    if p - sp >= 0:
                        val += (1.0 - h[p - sp])
                    if p + sp < n:
                        val -= h[p + sp]
                g[p] += val
        g *= dx / len(top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.05, 0.3 / gn) * step_scale
        accepted = False
        for s_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_sum(h - s_mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                accepted = True
                break
        if not accepted:
            step_scale *= 0.9
            if step_scale < 0.01:
                break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Multi-start: try a few perturbations
    for ps in [0.05, 0.1, 0.2, 0.35]:
        hp = _repair(np.clip(h + ps * rng.randn(n0), 0, 1), n0)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Block pattern start
    x = np.linspace(0, 2, n0, endpoint=False)
    h_blk = 0.5 + 0.3 * np.sin(2 * np.pi * 7 * x / 2.0) + 0.15 * np.sin(2 * np.pi * 13 * x / 2.0)
    h_blk = _repair(h_blk, n0)
    c5b = _eval(h_blk)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = h_blk.copy()

    # Phase 1: Subgradient (25s)
    dl1 = t0 + min(25, budget_s * 0.04)
    h1, c5_1 = _subgrad(best_h.copy(), rng, dl1, step_scale=1.5)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Random pair CD (300s)
    dl2 = t0 + min(325, budget_s * 0.35)
    h2, c5_2 = _pair_cd(best_h.copy(), rng, dl2, escape_every=10000, temp=0.008)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Sweep CD (250s)
    dl3 = t0 + min(575, budget_s * 0.60)
    h3, c5_3 = _sweep_cd(best_h.copy(), rng, dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Another random pair CD round (200s)
    dl4 = t0 + min(775, budget_s * 0.80)
    h4, c5_4 = _pair_cd(best_h.copy(), rng, dl4, escape_every=12000, temp=0.004)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Multi-scale refinement (150s)
    dl5 = t0 + min(925, budget_s * 0.92)
    h5, c5_5 = _multiscale_refine(best_h.copy(), rng, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Final CD polish (remaining time)
    dl6 = t0 + budget_s - 5
    h6, c5_6 = _pair_cd(best_h.copy(), rng, dl6, escape_every=20000, temp=0.001)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    # Final repair and verify
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
