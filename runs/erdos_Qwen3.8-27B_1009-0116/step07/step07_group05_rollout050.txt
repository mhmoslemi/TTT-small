def _pair_sweep(h, n, window, check_fn):
    """One sweep of pair swaps."""
    c5 = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    improved = 0
    for i in range(n):
        if check_fn():
            break
        for dj in range(1, window+1):
            j = (i + dj) % n  # circular for variety
            d_min = max(-h[i], h[j] - 1.0)
            d_max = min(1.0 - h[i], h[j])
            if d_max - d_min < 1e-12:
                continue
            g = _pair_grad_diff(h, i, j)
            d_opt = _optimal_d(c, g, d_min, d_max)
            if abs(d_opt) < 1e-12:
                continue
            h[i] += d_opt
            h[j] -= d_opt
            # Re-evaluate
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            c5_new = c.max()
            if c5_new < c5:
                c5 = c5_new
                improved += 1
            else:
                h[i] -= d_opt
                h[j] += d_opt
                c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    return h, c5, improved

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mlgd(h, rounds, k_top, steps_init, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        # Compute subgradient
        g = np.zeros(n)
        for k_idx in top_k:
            k = k_idx - (n-1)
            sh = (n-1) - k  # shift in correlate terms
            if sh >= 0:
                g[sh:] += (1.0 - h[:n-sh])
                if sh > 0:
                    g[:n-sh] -= h[sh:]
            else:
                sp = -sh
                g[:n-sp] += (1.0 - h[sp:])
                g[sp:] -= h[:n-sp]
        g *= dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si]*0.5]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.3 for s in steps]
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    return h, c5c

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            space = 1.0 - h
            ts = space.sum()
            if ts > 1e-12:
                h += diff * space / ts
            else:
                h += diff / n
        else:
            space = h.copy()
            ts = space.sum()
            if ts > 1e-12:
                h += diff * space / ts
            else:
                h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            order = np.argsort(1.0 - h) if diff2 > 0 else np.argsort(h)
            for idx in order:
                if diff2 > 0:
                    a = min(diff2, 1.0 - h[idx])
                    h[idx] += a; diff2 -= a
                else:
                    a = min(-diff2, h[idx])
                    h[idx] -= a; diff2 += a
                if abs(diff2) < 1e-12:
                    break
    return h

def _pair_grad(h, i, j):
    n = len(h)
    g = np.zeros(2 * n - 1)
    ki = np.arange(i, i + n)
    kj = np.arange(j, j + n)
    g[ki] += 1.0 - 2.0 * h[kj - j + (ki - i) - (ki - i)]
    # Correct: for k in [i, i+n-1], k-i in [0,n-1], g[k] += 1-2h[k-i]
    g[ki] += 1.0 - 2.0 * h[ki - i]
    # For k in [j, j+n-1], k-j in [0,n-1], g[k] -= 1-2h[k-j]
    g[kj] -= 1.0 - 2.0 * h[kj - j]
    return g

def _optimal_d(c, g, d_lo, d_hi, top_m=40):
    if d_hi - d_lo < 1e-14:
        return 0.0
    cmax = c.max()
    # Competitive lags: those near max or with large gradient
    thr = cmax - 0.02
    active = np.where(c > thr)[0]
    if len(active) < top_m:
        extra = np.argsort(np.abs(g))[-top_m:]
        active = np.unique(np.concatenate([active, extra]))
    if len(active) > top_m:
        active = active[np.argsort(c[active])[-top_m:]]
    ca = c[active]
    ga = g[active]
    # Candidate d values
    cands = np.array([d_lo, d_hi, 0.0])
    n_act = len(active)
    # Pairwise intersections of top lags
    for a in range(n_act):
        for b in range(a + 1, n_act):
            dg = ga[a] - ga[b]
            if abs(dg) < 1e-14:
                continue
            ds = (ca[b] - ca[a]) / dg
            if d_lo < ds < d_hi:
                cands = np.append(cands, ds)
    # Evaluate
    cands = np.clip(cands, d_lo, d_hi)
    # Vectorized: shape (D, 2n-1)
    vals = np.max(ca[None, :] + cands[:, None] * ga[None, :], axis=1)
    best_idx = np.argmin(vals)
    return float(cands[best_idx])

def _pair_sweep(h, n, window, check_fn):
    c5 = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    improved = 0
    for i in range(n):
        if check_fn():
            break
        for dj in range(1, window + 1):
            j = (i + dj) % n
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            g = _pair_grad(h, i, j)
            d_opt = _optimal_d(c, g, d_lo, d_hi)
            if abs(d_opt) < 1e-13:
                continue
            h[i] += d_opt
            h[j] -= d_opt
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            c5_new = c.max()
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                improved += 1
            else:
                h[i] -= d_opt
                h[j] += d_opt
    return h, c5, improved

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mlgd(h, rounds, k_top, steps_init, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k_idx in top_k:
            k = k_idx  # correlate index
            # h[i] appears in c[k] via h[i]*(1-h[k-i]) and h[k-i]*(1-h[i])
            # grad contribution to h[i]: (1-2h[k-i]) for valid k-i
            lo_i = max(0, k - n + 1)
            hi_i = min(n - 1, k)
            for ii in range(lo_i, hi_i + 1):
                g[ii] += 1.0 - 2.0 * h[k - ii]
        g *= dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.5]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.3 for s in steps]
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    return h, c5c

def _local_2pt(h, ntry, rng, c5c, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c
