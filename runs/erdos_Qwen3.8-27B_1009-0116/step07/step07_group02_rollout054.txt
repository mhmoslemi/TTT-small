            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _pobj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pgrad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, obj, jac, args, iters, dl):
    n = len(h)
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, jac=jac, args=args, method='L-BFGS-B', bounds=b,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-11, 'maxfun': iters * 2})
        return _repair(r.x, n)
    except Exception:
        return h

def _topk_descent(h, nk, rounds, dl):
    n = len(h)
    c5c = _eval(h)
    for _ in range(rounds):
        if time.time() > dl - 15:
            break
        c = _cvals(h)
        tk = np.argsort(c)[-nk:]
        g = np.zeros(n)
        dx = 2.0 / n
        for k in tk:
            sh = (n - 1) - k
            if sh >= 0:
                m = n - sh
                g[:m] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:m]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / nk
        imp = False
        for st in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
            hn = _repair(h - st * g, n)
            cn = _eval(hn)
            if cn < c5c - 1e-15:
                h, c5c, imp = hn, cn, True
                break
        if not imp:
            break
    return h, c5c

def _coarse_fine(best_h, dl):
    n = len(best_h)
    n2 = n // 2
    h2 = best_h[:n2 * 2].reshape(n2, 2).mean(axis=1)
    h2 = _repair(h2, n2)
    for a in [2000, 20000, 200000]:
        if time.time() > dl - 60:
            break
        h2 = _lbfgs(h2, _smooth_obj, _smooth_grad, (a,), 2000, dl)
    h2, _ = _topk_descent(h2, 4, 300, dl)
    hu = np.repeat(h2, 2)
    hu = _repair(hu, n)
    for a in [50000, 500000]:
        if time.time() > dl - 40:
            break
        hu = _lbfgs(hu, _smooth_obj, _smooth_grad, (a,), 2000, dl)
    hu, cu = _topk_descent(hu, 3, 200, dl)
    return hu, cu

def _coord_desc(h, niter, rng, dl):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    deltas = [1e-5, -1e-5, 5e-6, -5e-6, 2e-6, -2e-6, 1e-6, -1e-6]
    for _ in range(niter):
        if time.time() > dl - 1:
            break
        i = rng.randint(n)
        done = False
        for d in deltas:
            ni = h[i] + d
            if 0.0 <= ni <= 1.0:
                h[i] = ni
                cn = _eval(h)
                if cn < c5c - 1e-18:
                    c5c = cn
                    if cn < bc:
                        bc = cn
                        bh = h.copy()
                else:
                    h[i] = ni - d
                    done = True
                    break
            else:
                done = True
                break
        if done:
            pass
    return bh, bc

def _two_pt(h, niter, rng, dl):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > dl - 1:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-18:
                c5c = cn
                if cn < bc:
                    bc = cn
                    bh = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _sine_init(n, f1, f2, a1, a2, p1, p2):
    x = np.arange(n) / n * 2.0
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h0.copy(), _eval(h0)

    # Phase 1: Coarse optimization at n/2
    n2 = n0 // 2
    h2 = h0[:n2 * 2].reshape(n2, 2).mean(axis=1)
    h2 = _repair(h2, n2)
    for a in [1000, 5000, 50000]:
        if time.time() > dl - 600:
            break
        h2 = _lbfgs(h2, _smooth_obj, _smooth_grad, (a,), 1500, dl)
    h2, _ = _topk_descent(h2, 4, 200, dl)
    hu = _repair(np.repeat(h2, 2), n0)
    cu = _eval(hu)
    if cu < best_c5:
        best_c5, best_h = cu, hu.copy()

    # Phase 2: Full-res L-BFGS (logsumexp)
    h = best_h.copy()
    for a in [5000, 50000, 500000]:
        if time.time() > dl - 450:
            break
        h = _lbfgs(h, _smooth_obj, _smooth_grad, (a,), 3000, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 3: P-norm variation
    h = best_h.copy()
    for p in [20, 60, 200]:
        if time.time() > dl - 350:
            break
        h = _lbfgs(h, _pobj, _pgrad, (p,), 2000, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 4: Top-k descent
    for nk in [2, 3, 5, 8]:
        if time.time() > dl - 280:
            break
        ht, ct = _topk_descent(best_h.copy(), nk, 500, dl)
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()

    # Phase 5: Perturbation escapes
    for scale in [0.04, 0.02, 0.06, 0.015, 0.08]:
        if time.time() > dl - 180:
            break
        hp = _repair(best_h + scale * rng.randn(n0), n0)
        hp = _lbfgs(hp, _smooth_obj, _smooth_grad, (100000,), 1500, dl)
        hp, cp = _topk_descent(hp, 5, 200, dl)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    # Phase 6: Alternative starts
    if time.time() < dl - 200:
        inits = [_sine_init(n0, 1.0, 2.5, 0.3, 0.12, 0.1, 0.8),
                 _sine_init(n0, 0.5, 1.5, 0.35, 0.1, 0.4, 1.2),
                 _repair(0.5 + 0.25 * np.sin(np.arange(n0) * 0.1 + 0.3) +
                         0.15 * np.sin(np.arange(n0) * 0.37), n0)]
        for ini in inits:
            if time.time() > dl - 200:
                break
            hr = _lbfgs(ini, _smooth_obj, _smooth_grad, (5000,), 2000, dl)
            hr = _lbfgs(hr, _smooth_obj, _smooth_grad, (100000,), 1500, dl)
            hr, cr = _topk_descent(hr, 5, 150, dl)
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()

    # Phase 7: Second coarse-fine
    if time.time() < dl - 130:
        hc, cc = _coarse_fine(best_h.copy(), dl)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()

    # Phase 8: High-alpha L-BFGS
    if time.time() < dl - 70:
        h = _lbfgs(best_h, _smooth_obj, _smooth_grad, (2000000,), 4000, dl)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 9: Final top-k
    if time.time() < dl - 20:
        ht, ct = _topk_descent(best_h.copy(), 3, 300, dl)
        if ct < best_c5:
            best_c5, best_h = ct, ht.copy()

    # Phase 10: Coordinate descent
    if time.time() < dl - 5:
        hc, cc = _coord_desc(best_h, 40000, rng, dl)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()

    # Phase 11: Two-point micro
    if time.time() < dl - 1:
        h2p, c2p = _two_pt(best_h, 60000, rng, dl)
        if c2p < best_c5:
            best_c5, best_h = c2p, h2p.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
