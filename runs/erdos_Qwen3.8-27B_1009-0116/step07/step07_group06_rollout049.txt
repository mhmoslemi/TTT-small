def _repair(h, n):

    target = n / 2.0

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

        nt = max(2, int(2 + 6 * frac))
        g, c_cur = _lse_grad(h, tau, nt)
        accepted = False
        for t in range(10):
            h_new = _proj(h - lr * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                accepted = True
                break
            lr *= 0.6
        if not accepted:
            stal += 1
            if stal > 800:
                break
            lr = max(lr, 0.005)
        else:
            stal = 0
        if it % 1000 == 0:
            lr = 0.03 * (1.0 - 0.5 * frac)
    return best_h, best_c

def _gadp(h, p, n):
    """dc_k/dh[p] for all k (length 2n-1), scaled by 2/n."""
    g = np.full(2 * n - 1, 2.0 / n)
    lo = max(0, n - 1 - p)
    hi = min(2 * n - 2, 2 * n - 2 - p)
    if lo <= hi:
        idx = np.arange(lo, hi + 1) - (n - 1) + p
        g[lo:hi + 1] -= (2.0 / n) * h[idx]
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, n - 1 + p)
    if lo2 <= hi2:
        idx2 = (p + n - 1) - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= (2.0 / n) * h[idx2]
    return g

def _pair_cd(h, n_iters, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5 = c.max()
    best_h = h.copy()
    best_c = c5
    stal = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it > 3000 and stal > 8000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _gadp(h, i, n) - _gadp(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(42):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-14:
            stal += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stal = 0
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stal += 1
    return best_h, best_c

def _block_cd(h, n_iters, rng, deadline, blk=2):
    """Block coordinate descent: update blk coordinates at once."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5 = c.max()
    best_h = h.copy()
    best_c = c5
    stal = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it > 2000 and stal > 5000:
            break
        i = rng.randint(n)
        js = rng.randint(n, size=blk - 1)
        js = js[js != i]
        if len(js) < blk - 1:
            js = np.array([i + 1, i + 2])[:blk - 1] % n
        idxs = np.concatenate([[i], js])
        d_max = min(1.0 - h[i], h[js].min())
        d_min = max(h[i] - 1.0, -(h[js].min()))
        d_lo, d_hi = max(d_min, -h[i]), min(d_max, h[js].min())
        if d_lo >= d_hi - 1e-15:
            continue
        g = np.zeros(2 * n - 1)
        for p in idxs:
            sign = 1.0 if p == i else -1.0
            g += sign * _gadp(h, p, n)
        extra_pairs = []
        for a in range(len(idxs)):
            for b in range(a + 1, len(idxs)):
                da, db = idxs[a], idxs[b]
                sa, sb = (1.0 if da == i else -1.0), (1.0 if db == i else -1.0)
                off = db - da
                kk1 = n - 1 + off
                kk2 = n - 1 - off
                coeff = sa * sb
                if coeff != 0:
                    extra_pairs.append((kk1, coeff))
                    if kk1 != kk2:
                        extra_pairs.append((kk2, coeff))
        for _ in range(35):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            for kk, cc in extra_pairs:
                if 0 <= kk < 2 * n - 1:
                    vals[kk] += cc * d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            for kk, cc in extra_pairs:
                if kk == kmax:
                    sg += cc * 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-14:
            stal += 1
            continue
        h[i] += d_opt
        for p in js:
            h[p] -= d_opt
        c += d_opt * g
        for kk, cc in extra_pairs:
            if 0 <= kk < 2 * n - 1:
                c[kk] += cc * d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stal = 0
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        else:
            h[i] -= d_opt
            for p in js:
                h[p] += d_opt
            c -= d_opt * g
            for kk, cc in extra_pairs:
                if 0 <= kk < 2 * n - 1:
                    c[kk] -= cc * d_opt * d_opt
            stal += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Quick LSE to polish initial (30s)
    dl = t0 + 30
    _, c1 = _lse_descent(h.copy(), rng, dl, max_iter=10000)
    if c1 < best_c5:
        best_c5, best_h = c1, _

    # Phase 2: Pairwise CD (250s)
    dl = t0 + 280
    h2, c2 = _pair_cd(best_h.copy(), 120000, rng, dl)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: LSE to escape (150s)
    dl = t0 + 430
    h3, c3 = _lse_descent(best_h.copy(), rng, dl, max_iter=30000)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Block CD (200s)
    dl = t0 + 630
    h4, c4 = _block_cd(best_h.copy(), 60000, rng, dl, blk=2)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: More pairwise CD (200s)
    dl = t0 + 830
    h5, c5_ = _pair_cd(best_h.copy(), 100000, rng, dl)
    if c5_ < best_c5:
        best_c5, best_h = c5_, h5

    # Phase 6: Perturbation + re-optimize (150s)
    dl = t0 + 980
    for ps in [0.05, 0.12, 0.2]:
        if time.time() > dl:
            break
        hp = np.clip(best_h + ps * rng.randn(n0), 0, 1)
        hp = _proj(hp, n0 * 0.5)
        c_norm = _eval(hp)
        if c_norm < best_c5:
            best_c5, best_h = c_norm, hp
        # Quick CD on perturbed
        dl_sub = min(t0 + 980, time.time() + 40)
        hp2, c_hp2 = _pair_cd(hp, 30000, rng, dl_sub)
        if c_hp2 < best_c5:
            best_c5, best_h = c_hp2, hp2

    # Phase 7: Final LSE + CD (70s)
    dl = t0 + budget_s - 10
    h7, c7 = _lse_descent(best_h.copy(), rng, min(dl, t0 + budget_s - 30), max_iter=10000)
    if c7 < best_c5:
        best_c5, best_h = c7, h7
    h8, c8 = _pair_cd(best_h.copy(), 40000, rng, min(dl, t0 + budget_s - 5))
    if c8 < best_c5:
        best_c5, best_h = c8, h8

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
