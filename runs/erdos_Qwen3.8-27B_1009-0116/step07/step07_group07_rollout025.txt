        if np.clip(h - mid, 0, 1).sum() > target:

def _lp_obj(h, p, penalty):

    c = _corr(h)
    cm = np.maximum(c, 0.001)
    return np.sum(cm**p)**(1.0/p) + penalty * (h.sum() - n*0.5)**2

def _lp_grad(h, p, penalty):

    dx = 2.0 / n
    c = _corr(h)
    cp = np.maximum(c, 1e-30)**p
    s = cp.sum()
    factor = s**(1.0/p - 1.0)
    g = np.zeros(n)
    for k in range(2*n-1):
        if c[k] < 1e-14:
            continue
        wk = factor * c[k]**(p-1) * dx
        sh = (n-1) - k

            i2 = np.arange(n-sp)
            g[i2] -= wk * h[i2+sp]

    cmax = np.max(c)

def _lbfgs(h0, obj, grad, maxiter, args=()):
    n = len(h0)
    bnds = [(0.0, 1.0)]*n
    try:
        res = minimize(obj, h0, jac=grad, args=args, method='L-BFGS-B',

def _nesterov_topk(h0, n_iter, k_start, k_max, step0, el_fn, bc_fn):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.8
    step = step0
    best_c = _eval(h)
    best_h = h.copy()
    stall = 0
    k = k_start
    dx = 2.0/n
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corr(h)
        if stall > 50:
            k = min(k+1, k_max)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        for ki in topk:

                i1 = np.arange(n-sh)
                g[i1] += (1.0 - h[i1+sh])

                i2 = np.arange(n-sp)

    return best_h, best_c

def _block_search(h0, n_iter, rng, el_fn, bc_fn):
    n = len(h0)
    h = h0.copy()
    best_c = _eval(h)
    best_h = h.copy()
    scale = 5e-5
    for it in range(n_iter):
        if bc_fn():
            break
        # Block move: pick a contiguous block and add a smooth perturbation
        blen = rng.randint(2, 15)
        start = rng.randint(0, n - blen + 1)
        weights = np.exp(-0.5*(np.arange(blen)-blen/2)**2/(blen/4+1))
        weights /= weights.sum()

        h_new = h.copy()
        h_new[start:start+blen] += d * weights
        # Also perturb one other point to maintain sum
        j = rng.randint(0, n)
        h_new[j] -= d * weights.sum()
        h_new = np.clip(h_new, 0, 1)
        h_new = _proj_sum(h_new, n*0.5)

            best_h = h_new.copy()
            h = h_new.copy()
        if it % 3000 == 0:
            scale *= 0.75
    return best_h, best_c

def _finelocal(h0, n_iter, rng, el_fn, bc_fn):
    n = len(h0)
    h = h0.copy()

def _pair_ls(h0, ni, rng, t0, budge):
    n = len(h0)
    h = h0.copy()
    best_c = _eval(h)
    best_h = h.copy()
    for it in range(ni):
        if time.time() > budge - 15: break
        i = rng.randint(n)
        j = (i+1+rng.randint(0,12)) % n
        if i == j: continue
        s = h[i]+h[j]
        lo = max(0.0, s-1.0)
        hi = min(1.0, s)
        if hi-lo < 1e-14: continue
        bm = h[i]
        bv = np.inf
        ts = np.linspace(lo, hi, 13)
        for t in ts:
            h[i]=t; h[j]=s-t
            v = float(np.max(_corr(h)))
            if v < bv: bv=v; bm=t
        dl=(hi-lo)/13
        lo2,hi2 = max(lo,bm-dl), min(hi,bm+dl)
        ts2 = np.linspace(lo2,hi2,11)
        for t in ts2:
            h[i]=t; h[j]=s-t
            v = float(np.max(_corr(h)))
            if v < bv: bv=v; bm=t
        h[i]=bm; h[j]=s-bm
        if bv < best_c:
            best_c=bv; best_h=h.copy()
    return best_h, best_c

def _fineloc(h0, ni, rng, t0, budge):
    n = len(h0)
    h = h0.copy()
    best_c = _eval(h)
    best_h = h.copy()
    sc = 3e-6
    for it in range(ni):
        if time.time() > budge - 15: break
        i = rng.randint(n)
        j = (i+1+rng.randint(0,8)) % n
        d = rng.randn()*sc
        ni_,nj_ = h[i]+d, h[j]-d
        if 0<=ni_<=1 and 0<=nj_<=1:
            h[i],h[j]=ni_,nj_
            c=_eval(h)
            if c<best_c: best_c=c; best_h=h.copy()
            else: h[i],h[j]=ni_-d,nj_+d
        if it%10000==0: sc*=0.7
    return best_h, best_c

def _perturb(h, n, rng, amp, seed2):
    r = np.random.RandomState(seed2)
    freq = r.uniform(1,4)
    phase = r.uniform(0,2*np.pi)
    h_new = h + amp*r.randn(n)*np.sin(2*np.pi*freq*np.arange(n)/n+phase)
    return _repair(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS p-norm
    for p, pen in [(16,1e4),(32,1e5),(64,1e6),(128,1e7)]:
        if time.time()-t0 > 100: break
        ht = _lbfgs(best_h.copy(), _pobj, _pgrad, 400, (p,pen))
        ct = _eval(ht)
        if ct < best_c5: best_c5=ct; best_h=ht

    # Phase 2: Nesterov top-k (main)
    if time.time()-t0 < budget_s-450:
        ht, ct = _nest_topk(best_h, 15000, 3, 70, 0.02, t0, budget_s)
        if ct < best_c5: best_c5=ct; best_h=ht

    # Phase 3: Pair-optimized local search
    if time.time()-t0 < budget_s-250:
        nt = min(int((budget_s-250-(time.time()-t0))/0.004), 80000)
        best_h, best_c5 = _pair_ls(best_h, nt, rng, t0, budget_s)

    # Phase 4: Perturbation + quick re-opt
    for ps in [7, 13, 29]:
        if time.time()-t0 > budget_s-200: break
        hp = _perturb(best_h, n, rng, 0.04, seed+ps)
        ht = _lbfgs(hp, _pobj, _pgrad, 200, (64,1e6))
        ct = _eval(ht)
        if ct < best_c5: best_c5=ct; best_h=ht
        ht2, ct2 = _nest_topk(ht, 4000, 2, 30, 0.01, t0, budget_s)
        if ct2 < best_c5: best_c5=ct2; best_h=ht2

    # Phase 5: Nesterov round 2
    if time.time()-t0 < budget_s-150:
        ht, ct = _nest_topk(best_h, 10000, 2, 50, 0.012, t0, budget_s)
        if ct < best_c5: best_c5=ct; best_h=ht

    # Phase 6: More pair search
    if time.time()-t0 < budget_s-80:
        nt = min(int((budget_s-80-(time.time()-t0))/0.004), 60000)
        best_h, best_c5 = _pair_ls(best_h, nt, rng, t0, budget_s)

    # Phase 7: Fine local search
    if time.time()-t0 < budget_s-20:
        nt = min(int((budget_s-20-(time.time()-t0))/0.002), 100000)
        best_h, best_c5 = _fineloc(best_h, nt, rng, t0, budget_s)

    # Phase 8: Final L-BFGS polish
    if time.time()-t0 < budget_s-5:
        ht = _lbfgs(best_h.copy(), _pobj, _pgrad, 150, (256,1e8))
        ct = _eval(ht)
        if ct < best_c5: best_c5=ct; best_h=ht

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
