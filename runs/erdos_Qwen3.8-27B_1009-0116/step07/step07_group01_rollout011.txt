def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine (45s)
    for ratio in [8, 4, 2]:
        nc = max(24, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        for alpha in [2000, 20000]:
            if _el() > 45:
                break
            hc = _slsqp_focus(hc, alpha, 200, 0)
        h_up = _repair(_upsample(hc, n // nc), n)
        if _eval(h_up) < best_c5:
            best_c5 = _eval(h_up)
            best_h = h_up.copy()

    # Phase 2: SLSQP focused on top lags (135s)
    slsqp_configs = [
        (5000, 0, 600),
        (20000, 8, 400),
        (80000, 5, 350),
        (300000, 3, 250),
        (1000000, 2, 200),
        (3000000, 2, 150),
    ]
    for alpha, tk, mit in slsqp_configs:
        if _el() > 180:
            break
        h_new = _slsqp_focus(best_h, alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Projected gradient (250s)
    pg_configs = [
        (3000, 500, 0.006, 8, 0.7),
        (10000, 500, 0.004, 6, 0.75),
        (40000, 500, 0.003, 5, 0.8),
        (100000, 500, 0.002, 4, 0.8),
        (300000, 400, 0.001, 3, 0.85),
        (1000000, 400, 0.0005, 2, 0.85),
        (3000000, 300, 0.0003, 2, 0.9),
    ]
    for alpha, steps, s0, tk, mu in pg_configs:
        if _el() > 430:
            break
        h_new = _proj_grad_focus(best_h.copy(), alpha, steps, s0, tk, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: Second SLSQP round with very high alpha (50s)
    if _el() < budget_s - 250:
        for alpha, tk, mit in [(5000000, 2, 200), (10000000, 1, 150)]:
            if _el() > budget_s - 200:
                break
            h_new = _slsqp_focus(best_h, alpha, mit, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Informed local search (uses worst lag)
    if _el() < budget_s - 80:
        time_left = budget_s - 80 - _el()
        _t = _el()
        for _ in range(50):
            _eval_worst(best_h)
        eps = max((_el() - _t) / 50.0, 1e-6)
        evals_per_s = 1.0 / eps
        ntries = min(int(time_left * evals_per_s * 0.65), 3000000)
        best_h, best_c5 = _local_informed(best_h, ntries, rng, 5e-6)

    # Phase 6: 3-point random local search
    if _el() < budget_s - 40:
        time_left = budget_s - 40 - _el()
        _t = _el()
        for _ in range(50):
            _eval(best_h)
        eps = max((_el() - _t) / 50.0, 1e-6)
        evals_per_s = 1.0 / eps
        ntries = min(int(time_left * evals_per_s * 0.65), 2000000)
        best_h, best_c5 = _local_3pt(best_h, ntries, rng, 3e-6)

    # Phase 7: 2-point ultra-fine polish
    if _el() < budget_s - 2:
        time_left = budget_s - 2 - _el()
        _t = _el()
        for _ in range(50):
            _eval(best_h)
        eps = max((_el() - _t) / 50.0, 1e-6)
        evals_per_s = 1.0 / eps
        npol = min(int(time_left * evals_per_s * 0.75), 10000000)
        best_h, best_c5 = _local_2pt(best_h, npol, rng, 8e-8)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
