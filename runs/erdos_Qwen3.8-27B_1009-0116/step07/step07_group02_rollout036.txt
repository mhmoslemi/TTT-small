def _fourier_basis(x, M):
    """Compute Fourier basis matrix. x: (n,), M: int -> (n, 2M+1)"""
    n = len(x)
    B = np.ones((n, 1))
    for k in range(1, M+1):
        B = np.column_stack([B, np.cos(k*np.pi*x), np.sin(k*np.pi*x)])
    return B

def _fourier_to_h(theta, x, M):
    """Convert Fourier coefficients to h values."""
    n = len(x)
    h = np.full(n, 0.5)
    idx = 0
    for k in range(1, M+1):
        h += theta[idx] * np.cos(k*np.pi*x)
        h += theta[idx+1] * np.sin(k*np.pi*x)
        idx += 2
    return h

def _fit_fourier(h, x, M):
    """Least-squares fit of h to Fourier series."""
    B = _fourier_basis(x, M)
    # B has shape (n, 2M+1), first column is ones (mean=0.5)

    h_centered = h - 0.5

    return theta

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode='full') * (2.0/n)

def _project_h(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        else:
            hi = mid

    """Compute gradient of weighted sum of correlations."""
    g = np.zeros(n)
    for k in range(2*n-1):
        if c_w[k] < 1e-14:

            sp = -sh
            g[sp:] += wk*(1.0-h[:-sp])

    return g

    """Fit Fourier coefficients to h (excluding DC)."""
    n = len(x)
    # B matrix: columns are cos(k*pi*x), sin(k*pi*x) for k=1..M

        B[:, 2*i+1] = np.sin(ki*np.pi*x)
    theta, _, _, _ = np.linalg.lstsq(B, h - 0.5, rcond=None)
    return theta

    """Evaluate Fourier series at points x."""
    h = np.full(len(x), 0.5)
    idx = 0
    for k in range(1, M+1):
        h += theta[idx]*np.cos(k*np.pi*x)
        h += theta[idx+1]*np.sin(k*np.pi*x)
        idx += 2
    return h

def _fourier_obj_grad(theta, x, M, alpha, lam, n_grid):
    """Objective and gradient for Fourier optimization."""
    h = _fourier_eval(theta, x, M)
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -400, 0)
    obj = m + np.log(np.sum(np.exp(s)))/alpha
    w = np.exp(s)
    w /= w.sum()
    # Penalty
    viol_lo = np.maximum(0, -h)
    viol_hi = np.maximum(0, h-1)
    obj += lam*(np.sum(viol_lo**2) + np.sum(viol_hi**2))
    # Gradient w.r.t. h
    dx = 2.0/n_grid
    g_h = _grad_corr(h, w, dx, n_grid)
    g_h += lam*(2*viol_lo - 2*viol_hi)
    # Gradient w.r.t. theta
    # dtheta_j = sum_i B_ij * g_h_i
    B = np.zeros((len(x), 2*M))

        B[:, 2*i+1] = np.sin((i+1)*np.pi*x)
    g_theta = B.T @ g_h
    return obj, g_theta

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_corr(h, w, dx, n):
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _smooth_obj(h, alpha):
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    return _grad_corr(h, w, dx, n)

def _fourier_fit(h, x, M):
    n = len(x)
    B = np.zeros((n, 2 * M))
    for i in range(M):
        B[:, 2 * i] = np.cos((i + 1) * np.pi * x)
        B[:, 2 * i + 1] = np.sin((i + 1) * np.pi * x)
    theta, _, _, _ = np.linalg.lstsq(B, h - 0.5, rcond=None)
    return theta

def _fourier_eval(theta, x, M):
    h = np.full(len(x), 0.5)
    for i in range(M):
        h += theta[2 * i] * np.cos((i + 1) * np.pi * x)
        h += theta[2 * i + 1] * np.sin((i + 1) * np.pi * x)
    return h

def _fobj(t, x, M, alpha, lam, n_g):
    h = _fourier_eval(t, x, M)
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    obj = m + np.log(np.sum(np.exp(s))) / alpha
    v1 = np.maximum(0, -h)
    v2 = np.maximum(0, h - 1)
    obj += lam * (np.sum(v1 ** 2) + np.sum(v2 ** 2))
    return obj

def _fgrad(t, x, M, alpha, lam, n_g):
    h = _fourier_eval(t, x, M)
    c = _cvals(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n_g
    g_h = _grad_corr(h, w, dx, n_g)
    g_h += lam * (2 * np.maximum(0, -h) - 2 * np.maximum(0, h - 1))
    B = np.zeros((len(x), 2 * M))
    for i in range(M):
        B[:, 2 * i] = np.cos((i + 1) * np.pi * x)
        B[:, 2 * i + 1] = np.sin((i + 1) * np.pi * x)
    return B.T @ g_h

def _optimize_full(h,
                   alpha, maxit, deadline):
    if time.time() > deadline - 10:
        return h
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-12})
        return _project_h(res.x, n)
    except Exception:
        return h

def _optimize_fourier(theta0, x, M, alpha, lam, maxit, deadline):
    if time.time() > deadline - 10:
        return theta0
    n_g = len(x)
    try:
        res = minimize(_fobj, theta0, jac=_fgrad,
                       args=(x, M, alpha, lam, n_g),
                       method='L-BFGS-B',
                       options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-11})
        return res.x
    except Exception:
        return theta0

def _top_lag(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 15:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:cnt]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_lags
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5]:
            h_new = _project_h(h - step * g, n)
            if _eval(h_new) < c5_cur - 1e-16:
                c5_cur = _eval(h_new)
                h = h_new
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _subgrad(h, steps, step0, tol, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.05 * i)
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:cnt]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / len(active)
        h = _project_h(h - t * g, n)
    return h

def _local_search(h, n_iter, rng, deadline, mode):

    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):

                h[i] = ni

                if c5n < c5c - 1e-19:
                    c5c = c5n

                        best_h = h.copy()

                    h[i] = ni - d

            j = (i + 1 + rng.randint(0, 4)) % n

        else:

                        best_h = h.copy()
