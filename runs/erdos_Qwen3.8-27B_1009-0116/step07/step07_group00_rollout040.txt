    except Exception:
        return _repair(h, n)

def _anneal_adaptive(h, niter, rng, deadline, T0=0.008, T1=1e-12):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    accept_rate = 0.5
    blocks = [2, 3, 4, 6, 8, 12, 16, 24]
    stall = 0
    for it in range(niter):
        if time.time() > deadline:
            break
        T = max(T0 * (T1 / T0) ** (it / max(niter, 1)), T1)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        scale = 0.02 * np.sqrt(bs) * (1.0 + 2.0 * (1.0 - accept_rate))
        d = scale * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _proj_sum(hn, target)
        f_new = _eval(hn)
        acc = f_new < f or rng.random() < np.exp(-(f_new - f) / max(T, 1e-15))
        accept_rate = 0.95 * accept_rate + 0.05 * (1.0 if acc else 0.0)
        if acc:
            h = hn
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 3000:
                stall = 0
                jitter = 0.005 * rng.randn(n)
                jitter -= jitter.mean()
                h = _proj_sum(h + jitter, target)
                f = _eval(h)
        else:
            stall += 1
    return best_h, best_f

def _coord_polish(h, niter, rng, deadline):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        step_idx = min(int(it / 3000), len(step_sizes) - 1)
        delta = step_sizes[step_idx] * rng.choice([1.0, -1.0])
        v = h[i] + delta
        if 0.0 <= v <= 1.0:
            hn = h.copy()
            hn[i] = v
            hn = _proj_sum(hn, target)
            f_new = _eval(hn)
            if f_new < f - 1e-17:
                h = hn
                f = f_new
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
    return best_h, best_f

def _line_search_polish(h, niter, rng, deadline):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h = h.copy()
    best_f = f
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        v0 = h[i]
        for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
            v = v0 + delta
            if 0.0 <= v <= 1.0:
                hn = h.copy()
                hn[i] = v
                hn = _proj_sum(hn, target)
                f_new = _eval(hn)
                if f_new < f - 1e-17:
                    h = hn
                    f = f_new
                    if f < best_f:
                        best_f = f
                        best_h = h.copy()
                    v0 = v
                else:
                    break
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    total_dl = t0 + budget_s - 3
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP refinement
    dl1 = min(total_dl, t0 + 120)
    h = _slsqp_refine(h, 20000, 20, 250, dl1)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 2: Nesterov with multiple alpha settings
    dl2 = min(total_dl, t0 + 380)
    configs = [(30000, 12, 0.003, 2500), (100000, 8, 0.0015, 2500),
               (500000, 5, 0.0008, 2500), (2000000, 3, 0.0004, 2000)]
    for alpha, tk, st, ns in configs:
        if time.time() > dl2:
            break
        hc, fc = _nesterov_phase(best_h.copy(), alpha, tk, ns, st, dl2)
        if fc < best_c5:
            best_c5 = fc
            best_h = hc.copy()

    # Phase 3: Adaptive annealing
    dl3 = min(total_dl, t0 + 700)
    ha, fa = _anneal_adaptive(best_h.copy(), 60000, rng, dl3)
    if fa < best_c5:
        best_c5 = fa
        best_h = ha.copy()

    # Phase 4: Second SLSQP on best
    dl4 = min(total_dl, t0 + 870)
    h = _slsqp_refine(best_h.copy(), 80000, 8, 200, dl4)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 5: Coordinate + line-search polish
    dl5 = total_dl
    rem = max(0, dl5 - time.time())
    if rem > 10:
        h_cd, f_cd = _coord_polish(best_h.copy(), min(int(rem * 0.4 / 0.002), 60000), rng, dl5 - rem * 0.5)
        if f_cd < best_c5:
            best_c5 = f_cd
            best_h = h_cd.copy()
        rem2 = max(0, dl5 - time.time())
        if rem2 > 5:
            h_ls, f_ls = _line_search_polish(best_h.copy(), min(int(rem2 / 0.002), 40000), rng, dl5)
            if f_ls < best_c5:
                best_c5 = f_ls
                best_h = h_ls.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
