    ex = np.maximum(0, c - tau)
    return ex.sum()**2 + pen * (h.sum() - len(h)*0.5)**2

def _hinge_g(h, tau, pen):
    c = _corrs(h)
    ex = np.maximum(0, c - tau)
    n = len(h)
    idxs = np.where(ex > 0)[0]
    g = np.zeros(n)
    if len(idxs) == 0:
        g[:] = 2.0 * pen * (h.sum() - n*0.5)
        return g
    dx = 2.0 / n
    total_ex = ex.sum()
    for k in idxs:
        wk = 2.0 * ex[k] * total_ex * dx

            g[:m] += wk * (1.0 - h[sh:sh+m])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = _proj_sum(h, target)
    return h

def _eval(h):
    return float(np.max(_corrs(h)))

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_w(h, idxs, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k, wk in zip(idxs, weights):
        sh = (n - 1) - int(k)
        if sh >= 0:
            m = n - sh
            g[:m] += wk * (1.0 - h[sh:sh+m])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g * dx

def _lsep_obj(h, t, pen):
    c = _corrs(h)
    cm = c.max()
    e = np.exp((c - cm) / max(t, 1e-15))
    return cm + t * np.log(e.sum()) + pen * (h.sum() - len(h)*0.5)**2

def _lsep_grad(h, t, pen):
    c = _corrs(h)
    cm = c.max()
    t = max(t, 1e-15)
    e = np.exp((c - cm) / t)
    w = e / e.sum()
    act = w > 1e-10
    idxs = np.where(act)[0]
    n = len(h)
    g = _grad_w(h, idxs, w[act])
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _hinge_obj(h, tau, pen):
    c = _corrs(h)
    ex = np.maximum(0, c - tau)
    return ex.sum()**2 + pen * (h.sum() - len(h)*0.5)**2

def _hinge_grad(h, tau, pen):
    c = _corrs(h)
    ex = np.maximum(0, c - tau)
    n = len(h)
    idxs = np.where(ex > 0)[0]
    g = np.zeros(n)
    if len(idxs) > 0:
        total = ex.sum()
        wts = 2.0 * ex[idxs] * total
        g = _grad_w(h, idxs, wts)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, ofn, gfn, args, nit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(ofn, h, jac=gfn, args=args, method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': nit, 'ftol': 1e-17,
                                           'gtol': 1e-13, 'maxls': 60})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _topk_descent(h, n_it, k0, step0, tl):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.72
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    k = k0
    for _ in range(n_it):
        if time.time() > tl:
            break
        c = _corrs(hc)
        if stall > 15:
            k = min(k + 2, n // 4)
        topk = np.argsort(c)[-k:]
        g = _grad_w(hc, topk, np.ones(k))
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1 - beta) * (g / gn)
        h_try = _proj_sum(hc - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-15:
            hc = h_try
            best_c = c_try
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.025, 0.06)
        else:
            step *= 0.52
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _mp_search(h, n_it, np_, scale, tl, seed=7):
    n = len(h)
    hc = h.copy().astype(float)
    bc = _eval(hc)
    bh = hc.copy()
    rng = np.random.RandomState(seed)
    sc = scale
    for it in range(n_it):
        if time.time() > tl:
            break
        idx = rng.choice(n, np_, replace=False)
        d = rng.randn(np_) * sc
        d -= d.mean()
        hn = hc.copy()
        hn[idx] += d
        ok = np.all(hn >= -0.01) and np.all(hn <= 1.01)
        if ok:
            hn = _proj_sum(hn, n * 0.5)
            cn = _eval(hn)
            if cn < bc:
                bc = cn
                bh = hn.copy()
        if it > 0 and it % 15000 == 0:
            sc *= 0.7
    return bh, bc
