from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / len(h))

def _cvals(h):
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / len(h))

def _grad_w(w, x, dx):
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = n - 1 - k
        wk = w[k] * dx
        if sh >= 0:
            m = n - sh
            g[:m] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:m]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _lse_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    return _grad_w(w, x, 2.0 / n)

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    return _grad_w(w, x, 2.0 / n)

    n = len(h)
    if time.time() > deadline - 10:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

    n = len(h)

                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})

import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = 0.5 * n
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / len(h))

def _cvals(h):
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / len(h))

def _gw(w, x, dx):
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = n - 1 - k
        wk = w[k] * dx
        if sh >= 0:
            m = n - sh
            g[:m] += wk * (1.0 - x[sh:])
            if sh:
                g[sh:] -= wk * x[:m]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _lo(x, a):
    c = _cvals(x)
    m = c.max()
    s = np.clip(a * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / a

def _lg(x, a):
    n = len(x)
    c = _cvals(x)
    s = np.clip(a * (c - c.max()), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    return _gw(w, x, 2.0 / n)

def _po(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c**p))**(1.0/p)

def _pg(x, p):
    n = len(x)
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c**p))**(1.0/p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c/fp)**(p-1)
    if w.sum() < 1e-30:
        return np.zeros(n)
    w /= w.sum()
    return _gw(w, x, 2.0/n)

def _LB(h, a, mi, dl):
    if time.time() > dl - 10: return h
    try:
        r = minimize(_lo, h, jac=_lg, args=(a,), method='L-BFGS-B',
                     bounds=[(0.,1.)]*len(h), options={'maxiter':mi,'ftol':1e-15,'gtol':1e-10})
        return _proj(r.x, len(h))
    except: return h

def _LP(h, p, mi, dl):
    if time.time() > dl - 10: return h
    try:
        r = minimize(_po, h, jac=_pg, args=(p,), method='L-BFGS-B',
                     bounds=[(0.,1.)]*len(h), options={'maxiter':mi,'ftol':1e-15,'gtol':1e-10})
        return _proj(r.x, len(h))
    except: return h

def _SL(h, st, tol, dl):
    n = len(h); dx = 2.0/n
    for i in range(st):
        if time.time() > dl - 10: break

        h = _proj(h - t*_gw(w,h,dx), n)
    return h

def _TL(h, nl, nr, dl):

def _BC(h, bs, nr, dl):

    n = len(x)

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g
