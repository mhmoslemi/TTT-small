def _repair(h, n):

    target = n / 2.0

def _subgrad_at(h, n, top_m_indices):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in top_m_indices:
        k = m - (n - 1)
        q = np.arange(n)
        qk = q + k
        mk = q - k
        mask1 = (qk >= 0) & (qk < n)
        mask2 = (mk >= 0) & (mk < n)
        g[mask1] += 1.0 - h[qk[mask1]]
        g[mask2] -= h[mk[mask2]]
    g *= dx / len(top_m_indices)
    return g

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        diff = tgt - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                else:
                    a = max(diff, -h[idx])
                h[idx] += a
                diff -= a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_pos(h, p, n, ks):
    g = np.zeros(2 * n - 1)
    pk = p + ks
    m1 = (pk >= 0) & (pk < n)
    g[m1] += 1.0 - h[pk[m1]]
    mk = p - ks
    m2 = (mk >= 0) & (mk < n)
    g[m2] -= h[mk[m2]]
    return g

def _cd_opt(h, n, rng, deadline, bisect=20):
    dx = 2.0 / n
    N2 = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5 = c.max()
    best_h = h.copy()
    best_c5 = c5
    stable = 0
    epoch = 0
    
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-14:
            continue
        g = (_grad_pos(h, i, n, ks) - _grad_pos(h, j, n, ks)) * dx
        k1 = n - 1 + (j - i)
        k2 = n - 1 + (i - j)
        for _ in range(bisect):
            d = 0.5 * (d_lo + d_hi)
            vals = c + d * g
            if 0 <= k1 < N2:
                vals[k1] -= d * d * dx
            if 0 <= k2 < N2:
                vals[k2] -= d * d * dx
            km = np.argmax(vals)
            sg = g[km]
            if km == k1 or km == k2:
                sg -= 2.0 * d * dx
            if sg > 1e-13:
                d_hi = d
            elif sg < -1e-13:
                d_lo = d
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-12:
            stable += 1
            if stable > 20000:
                # Plateau: try random restart
                scale = 0.1 * (1.0 + epoch * 0.05)
                h2 = _proj_to_sum(h + scale * rng.randn(n), n * 0.5)
                c2 = np.correlate(h2, 1.0 - h2, mode='full') * dx
                if c2.max() < c5 * 1.001:
                    h = h2
                    c = c2.copy()
                    c5 = c.max()
                    stable = 0
                    epoch += 1
                else:
                    break
            continue
        # Apply move
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < N2:
            c[k1] -= d_opt * d_opt * dx
        if 0 <= k2 < N2:
            c[k2] -= d_opt * d_opt * dx
        c5_new = c.max()
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            stable = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Reject
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < N2:
                c[k1] += d_opt * d_opt * dx
            if 0 <= k2 < N2:
                c[k2] += d_opt * d_opt * dx
            stable += 1
    return best_h, best_c5
