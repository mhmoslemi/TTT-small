import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _soft_grad(h, n, beta):
    c = _corr(h, n)
    w = np.exp(beta * (c - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _soft_descend(h, n, max_iter, beta0, beta1, check_fn):
    h = h.copy()
    best_c = _eval(h)
    stall = 0
    for it in range(max_iter):
        if check_fn():
            break
        beta = beta0 * (max(1.0, beta1 / beta0)) ** (it / max(1, max_iter - 1))
        g = _soft_grad(h, n, beta)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        step = 0.02
        improved = False
        for _ in range(45):
            h_try = _project_sum(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < best_c - 1e-15:
                best_c = c_try
                h = h_try
                improved = True
                break
            step *= 0.72
        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0
            step *= 1.3
            if step > 0.1:
                step = 0.1
    return h, best_c

def _bump_search(h, n, rng, ntry, check_fn):
    best_h = h.copy()
    best_c = _eval(best_h)
    for _ in range(ntry):
        if check_fn():
            break
        center = rng.randint(n)
        width = 2 + rng.randint(0, 10)
        amp = 0.003 * rng.randn()
        idx = np.arange(center - width // 2, center - width // 2 + width + 1) % n
        t = (np.arange(width + 1) - width / 2.0) / (width / 2.0 + 1.0)
        bump = amp * np.exp(-1.8 * t * t)
        h_try = best_h.copy()
        h_try[list(idx)] += bump
        if h_try.min() < -0.05 or h_try.max() > 1.05:
            continue
        h_try = _repair(h_try, n)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    return best_h, best_c

def _3pt_search(h, n, rng, ntry, scale, check_fn):

    for _ in range(ntry):

                best_h = h.copy()
