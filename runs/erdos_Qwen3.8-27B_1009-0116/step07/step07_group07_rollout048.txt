```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj_sum(h, n * 0.5)

def _grad_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cp = np.maximum(c, 1e-30) ** p
    s = cp.sum()
    factor = s ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if c[k] < 1e-14:
            continue
        wk = factor * c[k] ** (p - 1) * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    gp = 2.0 * penalty * (h.sum() - n * 0.5)
    return g + gp

def _obj_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = np.sum(c ** p) ** (1.0 / p)
    return val + penalty * (h.sum() - n * 0.5) ** 2

def _lbfgs(h, p, pen, iters):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj_pnorm, h, jac=_grad_pnorm,
                     args=(p, pen), method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': iters,
                     'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
        return _repair(r.x)
    except Exception:
        return _repair(h)

def _nesterov_topk(h, n_iter, k0, kmax, step0, elapsed_fn, budget_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.75
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    k = k0
    for _ in range(n_iter):
        if budget_fn():
            break
        c = _corrs(hc)
        if stall > 20:
            k = min(k + 1, kmax)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        dx = 2.0 / n
        for kk in topk:
            sh = (n - 1) - kk
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - hc[i1 + sh])
                if sh > 0:
                    g[sh:] -= hc[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - hc[:n - sp])
                g[:n - sp] -= hc[sp:]
        g *= dx / len(topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 1e-16:
            hc = ht
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _lsexp_nesterov(h, n_iter, T0, T1, step0, elapsed_fn, budget_fn):
    # Log-sum-exp annealing with Nesterov
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.7
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    
    for i in range(n_iter):
        if budget_fn():
            break
        T = T0 * (T1 / T0) ** (i / max(n_iter - 1, 1))
        c = _corrs(hc)
        # softmax weights
        c_shift = c - c.max()
        w = np.exp(c_shift / T)
        w /= w.sum()
        # Gradient of log-sum-exp
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):

            else:
                sp = -sh
                g[sp:] += w[k] * (1.0 - hc[:n - sp])
                g[:n - sp] -= w[k] * hc[sp:]
        g *= dx / T
        gn = np.linalg.norm(g)
        if gn < 1e-15:

def _grad_local_search(h, n_iter, rng, elapsed_fn, budget_fn):
    # Gradient-informed local search
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    scale = 5e-5
    for i in range(n_iter):
        if budget_fn():
            break
        c = _corrs(hc)
        top3 = np.argsort(c)[-3:]
        # Compute gradient of max (use top-3 average)
        dx = 2.0 / n
        g = np.zeros(n)
        for k in top3:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += (1.0 - hc[sh:])
                if sh > 0:
                    g[sh:] -= hc[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - hc[:n-sp])
                g[:n-sp] -= hc[sp:]
        g *= dx / 3.0
        # Random coordinate pair along gradient
        idx = np.argsort(np.abs(g))[-8:]
        i_idx = rng.choice(idx)
        j_idx = (i_idx + 1 + rng.randint(0, 4)) % n
        d = scale * rng.randn()
        # Move in gradient direction for i, opposite for j (to preserve sum)
        ni = hc[i_idx] - d * np.sign(g[i_idx]) if g[i_idx] != 0 else hc[i_idx] + d
        nj = hc[j_idx] + d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            hc[i_idx], hc[j_idx] = ni, nj
            ct = _eval(hc)
            if ct < best_c:
                best_c = ct
                best_h = hc.copy()
                scale = min(scale * 1.1, 2e-4)
            else:
                hc[i_idx], hc[j_idx] = ni - (ni - hc[i_idx]), nj - (nj - hc[j_idx])
                # revert

                hc[j_idx] = hc[j_idx]
        if i % 3000 == 0:
            scale *= 0.75
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj_sum(h, n * 0.5)

def _obj_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = np.sum(c ** p) ** (1.0 / p)
    return val + penalty * (h.sum() - n * 0.5) ** 2

def _grad_pnorm(h, p, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cp = np.maximum(c, 1e-30) ** p
    s = cp.sum()
    factor = s ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if c[k] < 1e-14:
            continue
        wk = factor * c[k] ** (p - 1) * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    gp = 2.0 * penalty * (h.sum() - n * 0.5)
    return g + gp

def _lbfgs(h, p, pen, iters):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj_pnorm, h, jac=_grad_pnorm,
                     args=(p, pen), method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': iters,
                     'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
        return _repair(r.x)
    except Exception:
        return _repair(h)

def _lsexp_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    val = cm + T * np.log(np.sum(np.exp((c - cm) / T)))
    return val + penalty * (h.sum() - n * 0.5) ** 2

def _lsexp_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    w = np.exp((c - cm) / T)
    w /= w.sum()
    gp = 2.0 * penalty * (h.sum() - n * 0.5)
    g = np.full(n, gp)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov_lsexp(h, n_iter, T0, T1, step0, elapsed_fn, budget_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.72
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    for i in range(n_iter):
        if budget_fn():
            break
        T = T0 * (T1 / T0) ** (i / max(n_iter - 1, 1))
        g = _lsexp_grad(hc, T, 0.0)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 1e-17:
            hc = ht
            best_c = ct
            best_h = hc.copy()
            step = min(step * 1.015, 0.06)
        else:
            step *= 0.55
            if step < 1e-10:
                break
    return best_h, best_c

def _nesterov_topk(h, n_iter, k0, kmax, step0, elapsed_fn, budget_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.75
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    k = k0
    dx = 2.0 / n
    for _ in range(n_iter):
        if budget_fn():
            break
        c = _corrs(hc)
        if stall > 25:
            k = min(k + 1, kmax)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        for kk in topk:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n-sh] += (1.0 - hc[sh:])
                if sh > 0:
                    g[sh:] -= hc[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - hc[:n-sp])
                g[:n-sp] -= hc[sp:]
        g *= dx / len(topk)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 1e-17:
            hc = ht
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.02, 0.04)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-10:
                break
    return best_h, best_c

def _guided_local(h, n_iter, rng, elapsed_fn, budget_fn):
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    dx = 2.0 / n
    scale = 8e-5
    for i in range(n_iter):
        if budget_fn():
            break
        c = _corrs(hc)
        top4 = np.argsort(c)[-4:]
        g = np.zeros(n)
        for kk in top4:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n-sh] += (1.0 - hc[sh:])
                if sh > 0:
                    g[sh:] -= hc[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - hc[:n-sp])
                g[:n-sp] -= hc[sp:]
        g *= dx / 4.0
        mag = np.abs(g)
        top_idx = np.argsort(mag)[-12:]
        ii = top_idx[rng.randint(len(top_idx))]
        jj = (ii + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        sg = np.sign(g[ii])
        if sg == 0:
            sg = 1.0
        ni = hc[ii] - sg * d
        nj = hc[jj] + d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[ii], hc[jj] = ni, nj
            ct = _eval(hc)
            if ct < best_c:
                best_c = ct
                best_h = hc.copy()
                scale = min(scale * 1.02, 3e-4)
            else:
                hc[ii] += sg * d
                hc[jj] -= d
        else:
            scale *= 0.9
        if i % 4000 == 0 and i > 0:
            scale *= 0.7
    return best_h, best_c

def _random_local(h, n_iter, rng, scale0, elapsed_fn, budget_fn):
```
