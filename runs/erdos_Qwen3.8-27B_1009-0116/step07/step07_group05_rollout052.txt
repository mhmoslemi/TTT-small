            for dv in [0.005, 0.002, 0.001, -0.005, -0.002, -0.001, 0.0005, -0.0005]:
                nv = h[idx] + dv
                if 0 <= nv <= 1:
                    h[idx] = nv
                    h = _repair(h, n)
                    cn = _eval(h)
                    if cn < best_c - 1e-16:
                        best_c = cn; best_v = nv
                    else:
                        h[idx] = best_v
                        h = _repair(h, n)
            h[idx] = best_v
            h = _repair(h, n)
            c5 = _eval(h)
    return h, c5

def _micro2(h0, deadline, rng, step=5e-6):
    n = len(h0)
    h = h0.copy()
    c5 = _eval(h)
    while time.time() < deadline:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5 - 1e-19:
                c5 = cn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5

def _micro3(h0, deadline, rng, step=7e-6):
    n = len(h0)
    h = h0.copy()
    c5 = _eval(h)
    while time.time() < deadline:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        k = (i + 10 + rng.randint(0, 9)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < c5 - 1e-19:
                c5 = cn
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-lag gradient descent (~40s)
    for kt in [3, 5, 4, 3, 5, 4]:
        if time.time() > deadline - 700:
            break
        h2, c52 = _mlag_descend(best_h, 250, kt,
            [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001], deadline - 550)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2

    # Phase 2: Smooth max + momentum (~250s)
    dl2 = min(deadline - 300, t0 + 250)
    h2, c52 = _smooth_momentum(best_h, dl2, temp0=0.008, temp1=0.002)
    if c52 < best_c5:
        best_c5 = c52; best_h = h2

    # Phase 3: Multi-lag with finer steps (~40s)
    for kt in [3, 5, 4, 3]:
        if time.time() > deadline - 350:
            break
        h2, c52 = _mlag_descend(best_h, 200, kt,
            [0.015, 0.008, 0.004, 0.002, 0.001, 5e-4, 2e-4], deadline - 280)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2

    # Phase 4: Coordinate-wise optimization (~120s)
    dl4 = min(deadline - 100, t0 + 650)
    h2, c52 = _coord_opt(best_h, dl4, n_rounds=60, n_coords=60)
    if c52 < best_c5:
        best_c5 = c52; best_h = h2

    # Phase 5: Restart with structured perturbation (~60s)
    if time.time() < deadline - 120:
        h_sh = best_h.copy()
        freq = rng.choice([2, 3, 5, 7])
        phase = rng.uniform(0, 2 * np.pi)
        h_sh += 0.005 * np.sin(2 * np.pi * freq * np.arange(n0) / n0 + phase)
        h_sh = _repair(h_sh, n0)
        h2, c52 = _mlag_descend(h_sh, 150, 3,
            [0.02, 0.01, 0.005, 0.002, 0.001], deadline - 80)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2

    # Phase 6: Smooth momentum round 2 (~80s)
    if time.time() < deadline - 60:
        dl6 = min(deadline - 40, t0 + 800)
        h2, c52 = _smooth_momentum(best_h, dl6, temp0=0.004, temp1=0.001)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2

    # Phase 7: Final multi-lag polish (~30s)
    if time.time() < deadline - 30:
        for kt in [3, 5]:
            h2, c52 = _mlag_descend(best_h, 100, kt,
                [0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4], deadline - 20)
            if c52 < best_c5:
                best_c5 = c52; best_h = h2

    # Phase 8: 3-point micro search
    if time.time() < deadline - 25:
        best_h, best_c5 = _micro3(best_h, deadline - 22, rng, step=6e-6)

    # Phase 9: 2-point micro search (ultra-fine)
    if time.time() < deadline - 3:
        best_h, best_c5 = _micro2(best_h, deadline - 3, rng, step=4e-6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
