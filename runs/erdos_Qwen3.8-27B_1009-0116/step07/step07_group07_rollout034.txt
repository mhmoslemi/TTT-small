        p <<= 1
    return p

def _fft_corr(h):
    n = len(h)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    cc = np.fft.irfft(np.conj(A) * B, N)
    idx = (np.arange(2 * n - 1) - (n - 1)) % N
    return cc[idx]

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_fft_corr(h) * dx))

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-8:
        h = _proj_sum(h, n * 0.5)
    return h

def _pnorm_obj(h, p, pen):
    n = len(h)
    c = _fft_corr(h) * (2.0 / n)
    return float(np.sum(np.maximum(c, 1e-30) ** p) ** (1.0 / p) + pen * (h.sum() - n * 0.5) ** 2)

def _pnorm_grad(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = _fft_corr(h) * dx
    cp = np.maximum(c, 1e-30) ** p
    s = cp.sum()
    w = (1.0 / p) * s ** (1.0 / p - 1.0) * cp * dx
    thresh = np.max(w) * 1e-5
    active = np.where(w > thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(w)])
    g = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        wc = w[k]
        lo1, hi1 = max(0, m), min(n, n + m)
        if lo1 < hi1:
            g[lo1:hi1] += wc * (1.0 - h[lo1 - m:hi1 - m])
        lo2, hi2 = max(0, -m), min(n, n - m)
        if lo2 < hi2:
            g[lo2:hi2] -= wc * h[lo2 + m:hi2 + m]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, p, pen, maxiter, deadline):
    n = len(h)
    if time.time() > deadline:
        return _repair(h, n)
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_nesterov(h, n_iter, k_start, k_max, step0, rng, deadline):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_start
    step = step0
    beta = 0.7
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        c = _fft_corr(h_c) * (2.0 / n)
        if stall > 40:
            k = min(k + 1, k_max)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        for kk in topk:
            m = kk - (n - 1)
            lo1, hi1 = max(0, m), min(n, n + m)
            if lo1 < hi1:
                g[lo1:hi1] += (1.0 - h_c[lo1 - m:hi1 - m])
            lo2, hi2 = max(0, -m), min(n, n - m)
            if lo2 < hi2:
                g[lo2:hi2] -= h_c[lo2 + m:hi2 + m]
        g *= (2.0 / n) / len(topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _local_search(h, n_iter, rng, deadline):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 2e-4
    for i in range(n_iter):
        if time.time() > deadline:
            break
        a = rng.randint(n)
        b = (a + 1 + rng.randint(0, 8)) % n
        d = rng.randn() * scale
        na, nb = h_c[a] + d, h_c[b] - d
        if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:
            h_c[a], h_c[b] = na, nb
            c = _eval(h_c)
            if c < best_c:
                best_c = c
                best_h = h_c.copy()
            else:
                h_c[a], h_c[b] = na - d, nb + d
        if i > 0 and i % 12000 == 0:
            scale *= 0.55
    return best_h, best_c

def _block_search(h, n_iter, rng, deadline):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 2e-4
    for i in range(n_iter):
        if time.time() > deadline:
            break
        blk = 2 + rng.randint(0, 2)
        idx = rng.choice(n, size=blk, replace=False)
        ds = rng.randn(blk) * scale
        ds[-1] = -ds[:-1].sum()
        nv = h_c[idx] + ds
        if np.all(nv >= 0) and np.all(nv <= 1):
            h_c[idx] = nv
            c = _eval(h_c)
            if c < best_c:
                best_c = c
                best_h = h_c.copy()
            else:
                h_c[idx] -= ds
        if i > 0 and i % 10000 == 0:
            scale *= 0.6
    return best_h, best_c

def _simulated_annealing(h, n_iter, rng, deadline):
    n = len(h)
    h_c = h.copy().astype(float)
    c_cur = _eval(h_c)
    best_c = c_cur
    best_h = h_c.copy()
    T = 8e-4
    T_min = 1e-8
    cool = 0.9997
    scale = 4e-4
    for i in range(n_iter):
        if time.time() > deadline:
            break
        blk = 2 + rng.randint(0, 2)
        idx = rng.choice(n, size=blk, replace=False)
        ds = rng.randn(blk) * scale
        ds[-1] = -ds[:-1].sum()
        nv = h_c[idx] + ds
        if np.all(nv >= 0) and np.all(nv <= 1):
            h_c[idx] = nv
            c_new = _eval(h_c)
            if c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / T):
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_c.copy()
            else:
                h_c[idx] -= ds
        T = max(T * cool, T_min)
        if i > 0 and i % 25000 == 0:
            scale *= 0.7
    return best_h, best_c

def _make_sinusoidal(n, freq, rng):
    x = np.arange(n)
    h = np.zeros(n)
    for _ in range(6):
        f = freq * (0.5 + rng.random())
        phase = rng.uniform(0, 2 * np.pi)
        h += 0.15 * np.sin(2 * np.pi * f * x / n + phase)
    h = 0.5 + h
    h = np.clip(h, 0, 1)
    return _proj_sum(h, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Try sinusoidal alternative
    h_sin = _make_sinusoidal(n, 3.0, rng)
    h_sin = _lbfgs(h_sin, 16, 1e4, 200, min(t0 + 15, deadline))
    c_sin = _eval(h_sin)
    if c_sin < best_c5:
        best_c5 = c_sin
        best_h = h_sin
        h0 = best_h.copy()

    # Phase 1: L-BFGS p-norm sweep
    p_list = [4, 8, 16, 32, 64, 128, 256, 512]
    for p in p_list:
        if time.time() > t0 + 90:
            break
        pen = 1e4 * (2.0 ** (p // 16))
        h_try = _lbfgs(best_h.copy(), p, pen, 600, min(t0 + 95, deadline))
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 2: High-p L-BFGS
    for p in [1024, 2048, 4096]:
        if time.time() > t0 + 220:
            break
        h_try = _lbfgs(best_h.copy(), p, 1e7, 350, min(t0 + 225, deadline))
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 3: Top-k Nesterov descent
    if time.time() < deadline - 350:
        h_try, c_try = _topk_nesterov(best_h, 10000, 3, 100, 0.02, rng, min(t0 + 380, deadline))
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 4: Simulated annealing
    if time.time() < deadline - 250:
        h_try, c_try = _simulated_annealing(best_h, 60000, rng, min(t0 + 600, deadline))
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 5: L-BFGS after SA
    if time.time() < deadline - 120:
        for p in [2048, 4096]:
            if time.time() > deadline - 120:
                break
            h_try = _lbfgs(best_h.copy(), p, 1e8, 250, min(t0 + 700, deadline))
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 6: Block search
    if time.time() < deadline - 100:
        h_try, c_try = _block_search(best_h, 120000, rng, min(t0 + 950, deadline))
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 7: Fine 2-point local search
    if time.time() < deadline - 10:
        h_try, c_try = _local_search(best_h, 500000, rng, deadline)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 8: Final polish
    if time.time() < deadline:
        h_try = _lbfgs(best_h.copy(), 2048, 1e8, 200, deadline)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
