_g_lam = 0.0
_g_tgt = 35.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(500):
        if s > 1e-15:
            h = h * (tgt / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-12:
            break
        d = tgt - s
        if d > 0:
            for i in np.argsort(1.0 - h):
                if d < 1e-12:
                    break
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
        else:
            for i in np.argsort(h):
                if d > -1e-12:
                    break
                a = min(-d, h[i])
                h[i] -= a
                d += a
    return h

def _obj_func(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_g_alpha * (c - m), -400, 400)
    v = m + np.log(np.sum(np.exp(s))) / _g_alpha
    v += _g_lam * (x.sum() - _g_tgt) ** 2
    return v

def _con_func(x):
    return x.sum() - _g_tgt

def _hot_indices(h, n, top_k):
    c = _corr(h)
    top_lags = np.argsort(c)[::-1][:top_k]
    hot = set()
    for tl in top_lags:
        if tl < 0 or tl >= 2 * n - 1:
            continue
        lag = tl - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for i in range(lo, hi):
            hot.add(i)
            if lag != 0:
                hot.add(i + abs(lag))
    return sorted(hot)

def _coord(h, n, t_end, rng, n_rounds):

    for r in range(n_rounds):

        else:
