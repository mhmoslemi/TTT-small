        j = np.arange(n)
        p, q = j + s, j - s
        vp = (p >= 0) & (p < n)
        vq = (q >= 0) & (q < n)
        grad[vp] += 1.0 - h[p[vp]]
        grad[vq] -= h[q[vq]]
    grad /= top_k
    return grad

def _backtrack(h, n, grad, dx, c5_cur, max_lr):
    for lr in [max_lr, max_lr*0.5, max_lr*0.25, max_lr*0.1, max_lr*0.05, max_lr*0.01]:
        h_new = _repair(h - lr * grad, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            return h_new, c5_new, lr
    return None, None, 0.0

def _coord_refine(h, n, dx, n_iter, rng):
    h = h.copy()
    c5_cur = _c5(h, dx)
    deltas = [0.01, 0.02, 0.04, 0.08, -0.01, -0.02, -0.04, -0.08]
    for _ in range(n_iter):
        improved = False
        for _ in range(15):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in deltas:
                ni, nj = hi + d, hj - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c5_cur:
                        c5_cur = c
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5_cur

def _block_swap(h, n, dx, rng):
    h = h.copy()
    c5_cur = _c5(h, dx)
    for _ in range(50):
        b = rng.randint(1, max(2, n//8))
        i = rng.randint(0, n - b)
        j = rng.randint(0, n - b)
        if abs(i - j) <= b:
            continue
        seg_i = h[i:i+b].copy()
        seg_j = h[j:j+b].copy()
        h[i:i+b] = seg_j
        h[j:j+b] = seg_i
        h = _repair(h, n)
        c = _c5(h, dx)
        if c < c5_cur:
            c5_cur = c
        else:
            h[i:i+b] = seg_i
            h[j:j+b] = seg_j
    return h, c5_cur

def _make_random(n, rng):
    h = np.zeros(n)
    ones = n // 2
    pos = rng.choice(n, size=ones, replace=False)
    h[pos] = 1.0
    h += 0.15 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, np.inf, 70

    def _upd(h, n):
        nonlocal best_h, best_c5, best_n
        h = _repair(h, n)
        c = _c5(h, 2.0/n)
        if c < best_c5:
            best_c5, best_h, best_n = c, h.copy(), n

    n_init = 70
    try:
        h_init = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
        n_init = len(h_init)
    except NameError:
        h_init = _repair(np.full(70, 0.5), 70)
        n_init = 70

    _upd(h_init, n_init)

    n_list = sorted(set([n_init, 60, 80, 90, 100]))
    time_per_n = (budget_s - 120) / len(n_list)

    for n in n_list:
        t_n = time.time() - t0
        if t_n > budget_s - 80:
            break
        dx = 2.0 / n
        deadline = min(t0 + budget_s - 50, t0 + t_n + time_per_n * (1.5 if n == n_init else 1.0))

        if n == n_init:
            h = h_init.copy()
        else:
            h = _make_random(n, rng)

        _upd(h, n)
        c5_cur = _c5(h, dx)

        # Phase A: Subgradient with top-1
        lr = 0.05
        for it in range(8000):
            if time.time() > deadline:
                break
            grad = _subgrad_dir(h, n, 1)
            h_new, c_new, lr_used = _backtrack(h, n, grad, dx, c5_cur, lr)
            if h_new is not None:
                h, c5_cur, lr = h_new, c_new, max(lr_used * 1.1, 0.001)
                _upd(h, n)
            else:
                lr *= 0.7
                if lr < 0.0001:
                    break

        # Phase B: Coordinate refinement
        h, c5_cur = _coord_refine(h, n, dx, 300, rng)
        _upd(h, n)

        # Phase C: Subgradient top-3
        lr = 0.03
        for it in range(4000):
            if time.time() > deadline:
                break
            grad = _subgrad_dir(h, n, 3)
            h_new, c_new, lr_used = _backtrack(h, n, grad, dx, c5_cur, lr)
            if h_new is not None:
                h, c5_cur, lr = h_new, c_new, max(lr_used * 1.05, 0.001)
                _upd(h, n)
            else:
                lr *= 0.8
                if lr < 0.0001:
                    break

        # Phase D: More coord + block swaps
        h, c5_cur = _coord_refine(h, n, dx, 200, rng)
        _upd(h, n)
        h, c5_cur = _block_swap(h, n, dx, rng)
        _upd(h, n)

        # Phase E: Fine subgradient
        lr = 0.01
        for top_k in [1, 2, 5]:
            for it in range(2000):
                if time.time() > deadline:
                    break
                grad = _subgrad_dir(h, n, top_k)
                h_new, c_new, lr_used = _backtrack(h, n, grad, dx, c5_cur, lr)
                if h_new is not None:
                    h, c5_cur, lr = h_new, c_new, max(lr_used * 1.02, 0.0005)
                    _upd(h, n)
                else:
                    lr *= 0.85
                    if lr < 5e-5:
                        break

    # Final polish on global best
    if best_h is not None:
        dx_b = 2.0 / best_n
        h = best_h.copy()
        c5_cur = best_c5
        lr = 0.005
        for it in range(5000):
            if time.time() > t0 + budget_s - 10:
                break
            grad = _subgrad_dir(h, best_n, 1)
            h_new, c_new, lr_used = _backtrack(h, best_n, grad, dx_b, c5_cur, lr)
            if h_new is not None:
                h, c5_cur, lr = h_new, c_new, max(lr_used * 1.01, 0.0003)
                _upd(h, best_n)
            else:
                lr *= 0.8
                if lr < 1e-5:
                    break
        h, c5_cur = _coord_refine(h, best_n, dx_b, 150, rng)
        _upd(h, best_n)

    if best_h is None:
        best_h = _repair(np.full(70, 0.5), 70)
        best_c5 = _c5(best_h, 2.0/70)
        best_n = 70

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0/best_n)
    return (best_h, best_c5, best_n)
