    deltas_coarse = [0.01, 0.03, 0.05, 0.08]
    plateau = 0
    for it in range(max_iter):
        if time.time() > budget_end:
            break
        corr = _corr(h)
        m = np.max(corr)
        top_n = min(12, n_corr)
        top_idx = np.argsort(corr)[-top_n:]
        improved = False
        use_coarse = (it % 20 < 5)
        deltas = deltas_coarse if use_coarse else deltas_fine
        for kc in top_idx:
            offset = kc - (n - 1)
            if offset >= 0:
                i_vals = list(range(offset, n))
            else:
                i_vals = list(range(0, n + offset))
            if not i_vals:
                continue
            contribs = np.array([h[i] * (1.0 - h[i - offset]) for i in i_vals])
            top_pairs = np.argsort(contribs)[-4:]
            for pi in top_pairs:
                i = i_vals[pi]
                j = i - offset
                if i == j:
                    continue
                for d in deltas:
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        oi, oj = h[i], h[j]
                        h[i] -= d
                        h[j] += d
                        c = _c5(h, dx)
                        if c < best:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = oi, oj
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        oi, oj = h[i], h[j]
                        h[i] += d
                        h[j] -= d
                        c = _c5(h, dx)
                        if c < best:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = oi, oj
        # Random pair swaps for diversity
        if it % 3 == 0:
            for _ in range(4):
                i, j = rng.choice(n, 2, replace=False)
                for d in [0.005, 0.01, 0.02, 0.04]:
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        oi, oj = h[i], h[j]
                        h[i] += d
                        h[j] -= d
                        c = _c5(h, dx)
                        if c < best:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = oi, oj
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        oi, oj = h[i], h[j]
                        h[i] -= d
                        h[j] += d
                        c = _c5(h, dx)
                        if c < best:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = oi, oj
        if improved:
            plateau = 0
        else:
            plateau += 1
        if plateau > 200 and it > 500:
            # Restart with small random perturbation
            perturb = 0.02 * rng.randn(n)
            h_new = _repair(h + perturb, n)
            c_new = _c5(h_new, dx)
            if c_new < best + 0.001:
                h = h_new
                best = c_new
            plateau = 0
            if it > 4000:
                break
    return h, best

def _lbfgs_pen(h, n, alpha, pen, maxiter):
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        val = m + np.log(np.sum(np.exp(s))) / alpha
        val += pen * (np.sum(x) - n / 2.0) ** 2
        return val
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    def try_update(h, n):
        nonlocal best_h, best_c5, best_n
        dx = 2.0 / n
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Get initial construction
    n_init = 70
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = np.full(n_init, 0.5)
    h_init = _repair(h_init, n_init)
    try_update(h_init, n_init)

    # === Phase 1: SLSQP initial refinement on n_init ===
    h_work = h_init.copy()
    for alpha in [200, 500, 1000]:
        if time.time() - t0 > budget_s * 0.3:
            break
        h_new = _slsqp_refine(h_work, n_init, alpha, 200)
        c_new = _c5(h_new, 2.0 / n_init)
        if c_new <= _c5(h_work, 2.0 / n_init) + 1e-8:
            h_work = h_new
    try_update(h_work, n_init)

    # === Phase 2: L-BFGS-B with penalty for constraint tightening ===
    for pen in [500, 5000, 50000]:
        if time.time() - t0 > budget_s * 0.4:
            break
        h_new = _lbfgs_pen(h_work, n_init, 500, pen, 300)
        c_new = _c5(h_new, 2.0 / n_init)
        if c_new <= _c5(h_work, 2.0 / n_init) + 1e-8:
            h_work = h_new
    try_update(h_work, n_init)

    # === Phase 3: Targeted coordinate descent (main workhorse) ===
    dx = 2.0 / n_init
    h_work, c_local = _targeted_cd(h_work, n_init, dx, t0 + budget_s * 0.75, rng, max_iter=6000)
    try_update(h_work, n_init)

    # === Phase 4: Final SLSQP polish ===
    for alpha in [1000, 3000, 5000]:
        if time.time() - t0 > budget_s - 60:
            break
        h_new = _slsqp_refine(h_work, n_init, alpha, 300)
        c_new = _c5(h_new, 2.0 / n_init)
        if c_new <= _c5(h_work, 2.0 / n_init) + 1e-10:
            h_work = h_new
    try_update(h_work, n_init)

    # === Phase 5: Brief targeted CD on the polished result ===
    h_work, _ = _targeted_cd(h_work, n_init, 2.0 / n_init, t0 + budget_s - 80, rng, max_iter=3000)
    try_update(h_work, n_init)

    # === Phase 6: Try other n values ===
    remaining = budget_s - (time.time() - t0) - 20
    if remaining > 30:
        other_ns = [64, 60, 72, 80, 90, 100]
        for trial_n in other_ns:
            if time.time() - t0 > budget_s - 25:
                break
            # Start from interpolated version of best
            h_t = np.interp(np.linspace(0, n_init - 1, trial_n),
                            np.arange(n_init), best_h).copy()
            h_t = _repair(h_t, trial_n)
            # Quick SLSQP
            h_t = _slsqp_refine(h_t, trial_n, 500, 150)
            try_update(h_t, trial_n)
            # Brief targeted CD
            h_t, _ = _targeted_cd(h_t, trial_n, 2.0 / trial_n,
                                  t0 + budget_s - 30, rng, max_iter=800)
            try_update(h_t, trial_n)

    # === Final validation ===
    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = _c5(best_h, 2.0 / 70)
        best_n = 70

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
