    idx = np.argsort(-h)

def _swap_eval(h, i, j, dx, n):
    """Evaluate C5 after swapping h[i](1->0) and h[j](0->1). Returns (c5, changed)."""
    h[i] = 0.0
    h[j] = 1.0
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    h[i] = 1.0
    h[j] = 0.0

    bounds = [(0.0, 1.0)] * n

        return float(np.sum(x) - target)

                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq_con}],

                diff -= add
            else:

                hc[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
