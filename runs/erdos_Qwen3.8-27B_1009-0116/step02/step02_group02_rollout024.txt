For each pair (i, j), the max correlation as a function of δ (h[i]+δ, h[j]-δ) is a piecewise function. We can approximate it by trying a few δ values and picking the best.

- Or increase h[m+L] for m where h[m] is high (reducing the "1-h" part)

This suggests a targeted perturbation: for the worst lag, find the pairs that contribute most to c_L and adjust them.

Let me also add the subgradient polish. The idea: at the current point, the max is achieved at one or a few lags. The subgradient of max_k c_k(h) is the convex hull of {∇c_k(h) : c_k(h) = max}. For a single active lag, it's just ∇c_k(h). We can take a step in the negative subgradient direction with a shrinking step size.

But ∇c_k(h) for a specific k is easy to compute (just the one-lag version of _sgrad). Let me implement:

```python
def _subgrad_polish(h, target, dx, iters):
    n = len(h)
    hc = h.copy()
    bc = _c5(hc, dx)
    c = _corr(hc)
    max_val = c.max()
    active = np.where(np.abs(c - max_val) < max_val * 0.01 + 1e-10)[0]
    step_size = 0.1
    for it in range(iters):
        # Recompute active set
        c = _corr(hc)
        max_val = c.max()
        active = np.where(c > max_val - max(0.01, max_val * 0.05))[0]
        # Subgradient = average of gradients at active lags
        g = np.zeros(n)
        for k in active:
            L = k - (n - 1)
            if L >= 0:
                nm = n - L
                g[:nm] += (1.0 - hc[L:])
                g[L:] -= hc[:nm]
            else:
                Lp = -L
                nm = n - Lp
                g[Lp:] += (1.0 - hc[:nm])
                g[:nm] -= hc[Lp:]
        g /= len(active)
        # Step
        y = hc - step_size * g
        hc_new = _proj(y, target)
        nc = _c5(hc_new, dx)
        if nc < bc:
            hc = hc_new
            bc = nc
        step_size *= 0.999
    return hc, bc
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        order = np.argsort(x) if diff < 0 else np.argsort(1.0 - x)
        for k in order:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a; diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = _corr(x)
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            nm = n - L
            g[:nm] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:nm]
        else:
            Lp = -L
            nm = n - Lp
            g[Lp:] += wk * (1.0 - x[:nm])
            g[:nm] -= wk * x[Lp:]
    return g

def _fista(h0, alpha, target, iters, dx):
    n = len(h0)
    x = h0.copy(); z = h0.copy(); t = 1.0
    bx = x.copy(); bc = _c5(x, dx)
    step = min(1.0 / (4.0 * alpha * n + 4.0), 0.4)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 5:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = xn + ((t - 1.0) / tn) * (xn - x)
        z = _proj(z, target)
        x = xn; t = tn
        if it % 200 == 0 or it == iters - 1:
            c = _c5(x, dx)
            if c < bc:
                bc = c; bx = x.copy()
    return bx, bc

def _anneal(h, dx, n, iters, T0, T1, rng):
    hc = h.copy(); curr_c = _c5(hc, dx)
    best_h = hc.copy(); best_c = curr_c
    for it in range(iters):
        T = T0 * (T1 / T0) ** (it / max(1, iters - 1))
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.04, 0.04)
        ni, nj = hc[i] + d, hc[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        nc = _c5(hc, dx)
        dc = nc - curr_c
        if dc < 0 or (T > 1e-8 and rng.rand() < np.exp(-dc / T)):
            curr_c = nc
            if nc < best_c:
                best_c = nc; best_h = hc.copy()
        else:
            hc[i], hc[j] = hc[i] - d,
