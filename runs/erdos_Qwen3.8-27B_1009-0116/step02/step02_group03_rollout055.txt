import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_onto_box_sum(y, target_sum):
    """Project y onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(y)
    # Find t such that sum(clip(y - t, 0, 1)) = target_sum
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(y - t, 0.0, 1.0)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_vals = np.exp(alpha * shifted)
    p = exp_vals / exp_vals.sum()
    obj = max_corr + np.log(exp_vals.sum()) / alpha
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        m = k - (n - 1)
        pk = p[k]
        if pk < 1e-14:
            continue
        if m >= 0:
            if m < n:

                grad[js] += pk * (1.0 - h[js - m])
            if m > 0:
                js2 = np.arange(0, n - m)

                grad[js2] += pk * (-h[js2 - r])
    return obj, grad

def projected_gradient(h, target_sum, n_iters, alpha_schedule):
    """Projected gradient descent with Adam."""
    n = len(h)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    lr = 0.01
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_obj = np.inf
    best_h = h.copy()
    
    for it in range(n_iters):
        if it % max(1, len(alpha_schedule)) == 0:
            alpha = alpha_schedule[min(it // max(1, len(alpha_schedule)) // 5, len(alpha_schedule)-1)]
        # Actually let me just use a single alpha passed in
        
        obj, grad = smooth_obj_and_grad(h, alpha)
        t += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad ** 2
        m_hat = m_adam / (1 - beta1 ** t)
        v_hat = v_adam / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        y = h - step
        h_new = project_onto_box_sum(y, target_sum)
        
        obj_new = smooth_obj_and_grad(h_new, alpha)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_obj

        improved = False

        order = rng.permutation(n)
        for i in order:

                ni = hi + d
                if ni < 0 or ni > 1:

                # Prefer the coordinate with most room
                h[i] = ni

                # Adjust the coordinate with most room

                    # Need to add
                    candidates = np.argsort(1.0 - h)[::-1]
                else:
                    # Need to subtract
                    candidates = np.argsort(h)
                for j in candidates:
                    if j == i:
                        continue

                    if diff > 0:

                        if add <= 0:

            h[j] -= d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        best_n = len(h0)
        target = best_n / 2.0
        h0 = project_box_sum(h0, target)
    except NameError:
        best_n = 70
        target = best_n / 2.0
        h0 = project_box_sum(0.5 * np.ones(best_n) + 0.1 * rng.randn(best_n), target)

    dx = 2.0 / best_n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Projected gradient with increasing alpha
    alphas_phase1 = [30, 60, 100, 200, 400, 800, 1500, 3000]
    for a in alphas_phase1:
        if time.time() - t0 > budget_s - 200:
            break
        h_new, c5_new = proj_grad(best_h, target, a, 0.008, 1500, 0.85)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cd, c5_cd = pair_coord(best_h, target, dx, 50, rng, t0 + budget_s - 180)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Very high alpha refinement
    for a in [5000, 8000, 15000]:
        if time.time() - t0 > budget_s - 100:
            break
        h_new, c5_new = proj_grad(best_h, target, a, 0.003, 2000, 0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cd, c5_cd = pair_coord(best_h, target, dx, 100, rng, t0 + budget_s - 80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: L-BFGS-B polish (uses exact gradient via lse)
    if time.time() - t0 < budget_s - 50:
        for a in [5000, 10000]:
            if time.time() - t0 > budget_s - 30:
                break
            def _obj(x):
                o, _ = lse_obj_grad(x, a)
                return o
            def _grad(x):
                _, g = lse_obj_grad(x, a)
                return g
            bounds = [(0.0, 1.0)] * best_n
            try:
                res = minimize(_obj, best_h, jac=_grad, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 500, 'ftol': 1e-18})
                h_res = project_box_sum(res.x, target)
                c5_res = compute_c5(h_res, dx)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
            except Exception:
                pass
        h_cd, c5_cd = pair_coord(best_h, target, dx, 150, rng, t0 + budget_s - 10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Binarization attempts
    if time.time() - t0 < budget_s - 30:
        h_bin, c5_bin = brighten(best_h.copy(), target, dx, 2000, rng, t0 + budget_s - 20)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_cd, c5_cd = pair_coord(best_h, target, dx, 80, rng, t0 + budget_s - 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Alternative starts (if budget remains)
    if time.time() - t0 < budget_s - 120:
        for kind in range(4):
            if time.time() - t0 > budget_s - 60:
                break
            h_alt = make_start(best_n, seed + kind * 101 + best_n * 7, kind)
            for a in [100, 500, 2000]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_new, c5_new = proj_grad(h_alt, target, a, 0.008, 800, 0.85)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            h_cd, c5_cd = pair_coord(best_h, target, dx, 60, rng, t0 + budget_s - 10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final verification
    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, best_n)
