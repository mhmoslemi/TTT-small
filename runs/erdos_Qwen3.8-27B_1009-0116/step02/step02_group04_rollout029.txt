def _compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _repair(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target/n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        if hp.sum() > target:

        g[:n-k] += -wk_neg * h[k:]

def _compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _repair(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _project(h, target, n):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    m = corr.max()
    s = alpha * (corr - m)
    es = np.exp(np.clip(s, -400, 0))
    w = es / es.sum()
    g = w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[n - 1 + k]
        wn = w[n - 1 - k]

        g[k:] += wn * (1.0 - h[:n - k])
        g[:n - k] += -wn * h[k:]

            idx = rng.choice(n, size=moves, replace=False)
            d = scale * (2.0 * rng.random(moves) - 1.0)
            d -= d.mean()
            new_vals = h[idx] + d
            if np.any(new_vals < -1e-12) or np.any(new_vals > 1.0 + 1e-12):
                continue
            h[idx] = new_vals
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                stall = 0
            else:
                h[idx] -= d
                stall += 1
                if stall > 3000:
                    break
    return h, best_c5

def _lbfgs_refine(h, n, target, dx, alpha, deadline, maxiter=300):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    def obj(x):
        _, fv, _ = _smooth_grad(x, alpha, n)
        return fv
    def grad(x):
        gv, _, _ = _smooth_grad(x, alpha, n)
        return gv
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        hc = _project(res.x, target, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _slsqp_refine(h, n, target, dx, alpha, deadline, maxiter=200):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    def obj(x):
        _, fv, _ = _smooth_grad(x, alpha, n)
        return fv
    def grad(x):
        gv, _, _ = _smooth_grad(x, alpha, n)
        return gv
    eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
          'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=[eq],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n, target)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    def rem():
        return budget_s - (time.time() - t_start)

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n, target)
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)

    # Phase 1: Projected gradient, increasing alpha (280s)
    t_end = t_start + 280
    for alpha in [40, 100, 250, 600, 1500, 400]:
        if time.time() > t_end:
            break
        h_try, c5_try = _proj_grad(h, n, target, dx, alpha, 3000, t_end, lr0=0.3, momentum=0.85)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h = best_h.copy()

    # Phase 2: L-BFGS-B refinement (150s)
    t_end = t_start + 430
    for alpha in [200, 800, 2000, 5000]:
        if time.time() > t_end or rem() < 400:
            break
        h_try, c5_try = _lbfgs_refine(best_h, n, target, dx, alpha, t_end, 200)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 3: Active-set subgradient (120s)
    t_end = t_start + 550
    for m_top in [2, 3, 5]:
        if time.time() > t_end or rem() < 300:
            break
        h_try, c5_try = _subgrad_descent(best_h, n, target, dx, m_top, 5000, t_end, lr0=0.02)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Multi-restart (200s)
    t_end = t_start + 750
    for r in range(15):
        if time.time() > t_end or rem() < 200:
            break
        pert = 0.04 * rng.randn(n)
        h_try = _repair(best_h + pert, n, target)
        h_try, c5_try = _proj_grad(h_try, n, target, dx, 800, 800, t_end, lr0=0.1, momentum=0.8)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h_try, c5_try = _lbfgs_refine(h_try, n, target, dx, 1500, t_end, 100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: SLSQP with equality constraint (150s)
    if rem() > 100:
        t_sl = time.time() + min(150, rem() - 60)
        for alpha in [300, 1000]:
            if time.time() > t_sl:
                break
            h_try, c5_try = _slsqp_refine(best_h, n, target, dx, alpha, t_sl, 150)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 6: Intensive local search (remaining - 20s)
    if rem() > 40:
        ls_dl = time.time() + max(20, rem() - 20)
        best_h, best_c5 = _local_search(best_h, n, dx, target, rng, ls_dl, max_iter=200000)

    # Final polish
    if rem() > 25:
        h_try, c5_try = _lbfgs_refine(best_h, n, target, dx, 8000, time.time() + min(15, rem() - 5), 100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    if rem() > 10:
        best_h, best_c5 = _local_search(best_h, n, dx, target, rng, time.time() + min(8, rem() - 2), max_iter=50000)

    best_h = _repair(best_h, n, target)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
