import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-14:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

    c = _corr(h)
    cm = np.max(c)
    p = np.exp(alpha * (c - cm))
    p /= p.sum()
    return cm + np.log(p.sum()) / alpha  # wait, that's just cm + log(n_full)/alpha

    # = cm + (1/alpha)*log(sum p_unnorm) where p_unnorm = exp(alpha*(c-cm))
    # And sum p_unnorm = sum p * sum_unnorm / p... no.

    sum_unnorm = np.exp(alpha * (c - cm)).sum()

        Lo = max(0, -k)

        diff = target - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _smooth_val(x, n, alpha):
    c = _corr(x)
    cm = np.max(c)
    ec = np.exp(alpha * (c - cm))
    return cm + np.log(ec.sum()) / alpha

def _smooth_grad(x, n, alpha):
    c = _corr(x)
    cm = np.max(c)
    ec = np.exp(alpha * (c - cm))
    p = ec / ec.sum()
    n_full = 2 * n - 1
    g = np.zeros(n)
    for m in range(n_full):
        pm = p[m]
        if pm < 1e-16:
            continue
        k = m - (n - 1)
        Lo = max(0, -k)
        Hi = min(n - 1, n - 1 - k)
        j1_lo = max(0, Lo + k)
        j1_hi = min(n - 1, Hi + k)
        if j1_lo <= j1_hi:
            g[j1_lo:j1_hi + 1] += pm * (1.0 - x[j1_lo - k:j1_hi - k + 1])
        if Lo <= Hi:
            g[Lo:Hi + 1] -= pm * x[Lo + k:Hi + k + 1]
    return g

def _obj(x, n, alpha, target, pen):
    return _smooth_val(x, n, alpha) + pen * (x.sum() - target) ** 2

def _obj_grad(x, n, alpha, target, pen):
    g = _smooth_grad(x, n, alpha)
    g += 2.0 * pen * (x.sum() - target)
    return g

def _worst_lag_grad(x, n, m):
    k = m - (n - 1)
    Lo = max(0, -k)
    Hi = min(n - 1, n - 1 - k)
    g = np.zeros(n)
    j1_lo = max(0, Lo + k)
    j1_hi = min(n - 1, Hi + k)
    if j1_lo <= j1_hi:
        g[j1_lo:j1_hi + 1] += (1.0 - x[j1_lo - k:j1_hi - k + 1])
    if Lo <= Hi:
        g[Lo:Hi + 1] -= x[Lo + k:Hi + k + 1]
    return g

def _proj(h, target, n):
    h = np.clip(h, 0.0, 1.0)
    return _repair(h, n, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n, target)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: L-BFGS-B with smooth max, increasing alpha and penalty
    for alpha in [50, 200, 800, 3000, 12000, 50000]:
        for _r in range(2):
            if elapsed() > budget_s * 0.35:
                break
            pen = 5000.0
            try:
                res = minimize(
                    lambda x: _obj(x, n, alpha, target, pen),
                    h, jac=lambda x: _obj_grad(x, n, alpha, target, pen),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-14})
                hc = _proj(res.x, target, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
                h = hc
            except Exception:
                pass

    # Phase 2: Projected gradient on worst lag (alternating)
    if elapsed() < budget_s * 0.6:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        for si, step in enumerate(steps):
            iters = 400 if si < 2 else 200
            for _ in range(iters):
                if elapsed() > budget_s * 0.6:
                    break
                c = _corr(h)
                m = int(np.argmax(c))
                g = _worst_lag_grad(h, n, m)
                h_new = _proj(h - step * g, target, n)
                c5_new = _c5(h_new, dx)
                if c5_new < c5_cur:
                    h, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h = h_new  # stay projected but don't count as improvement

    # Phase 3: Pair-move local search (all pairs, small steps)
    if elapsed() < budget_s * 0.8:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = [0.001, 0.002, 0.004, 0.008, 0.015]
        for _pass in range(200):
            if elapsed() > budget_s * 0.8:
                break
            any_imp = False
            for i in range(n):
                if elapsed() > budget_s * 0.8:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        for sgn in (1.0, -1.0):
                            ni = hi + sgn * d
                            nj = hj - sgn * d
                            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                                continue
                            h[i], h[j] = ni, nj
                            h = _proj(h, target, n)
                            c = _c5(h, dx)
                            if c < c5_cur - 1e-15:
                                c5_cur = c
                                any_imp = True
                            else:
                                h = _proj(h + (h[i] - ni) * np.eye(n)[i] + (h[j] - nj) * np.eye(n)[j], target, n)
                                # restore
                                h = _repair(np.clip(h, 0, 1), n, target)
                                h[i], h[j] = ni, nj
                                h = _proj(h, target, n)
                    h[i], h[j] = hi, hj
                    h = _proj(h, target, n)
                h[i] = hi
            if not any_imp:
                break
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    # Phase 4: SLSQP polish on smooth max
    if elapsed() < budget_s * 0.92:
        h = best_h.copy()
        for alpha in [30000, 100000]:
            if elapsed() > budget_s * 0.92:
                break
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                     'jac': lambda x: np.ones(n)}]
            try:
                res = minimize(
                    lambda x: _smooth_val(x, n, alpha),
                    h, jac=lambda x: _smooth_grad(x, n, alpha),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 300, 'ftol': 1e-16})
                hc = _proj(res.x, target, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
                    h = hc
            except Exception:
                pass

    # Phase 5: Ultra-fine coordinate descent (random)
    if elapsed() < budget_s - 10:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = [0.0001, 0.0003, 0.0005, -0.0001, -0.0003, -0.0005]
        max_iters = 200000
        it = 0
        while elapsed() < budget_s - 10 and it < max_iters:
            it += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                h = _proj(h, target, n)
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h = _proj(h - (h[i] - ni) * np.eye(n)[:, i] - (h[j] - nj) * np.eye(n)[:, j], target, n)
                    h[i], h[j] = ni, nj
                    h = _proj(h, target, n)

    best_h = _repair(best_h, n, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
