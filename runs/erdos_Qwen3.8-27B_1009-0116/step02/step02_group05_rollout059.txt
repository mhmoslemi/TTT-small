Actually, let me try a completely different approach: **use the exact max (not smoothed) with a subgradient method.**

The subgradient of max(f_1, ..., f_m) at a point where f_k is the maximum is ∇f_k. So:
1. Compute all lag values
2. Find the argmax lag k*
3. Compute gradient of that specific lag with respect to h
4. Take a step in the -gradient direction
5. Project to feasible set
6. Repeat

This is "Frank-Wolfe" or "subgradient" method for the max constraint.

Let me implement this:

```python
def _subgrad_step(h, n, dx, lr=0.01, deadline=None, max_iter=5000):
    target = n / 2.0
    best = h.copy()
    best_c = _c5(best, dx)
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        corr = _corr(best)
        # Find top lags
        top_k = np.argpartition(corr, -3)[-3:]
        # Average gradient of top lags
        grad = np.zeros(n)
        for d_idx in top_k:
            d = d_idx - (n - 1)  # actual lag
            grad += _grad_lag(best, d, n)
        grad /= len(top_k)
        # Penalize constraint violation
        grad += 0.1 * (np.sum(best) - target) / n  # small penalty
        # Take step
        new_h = best - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = _repair(new_h, n)
        c = _c5(new_h, dx)
        if c < best_c:
            best = new_h
            best_c = c
            # Don't decrease lr on improvement
        else:
            lr *= 0.95
```

For the gradient of lag d:
corr[n-1+d] = sum_i h[i]*(1-h[i+d]) where 0≤i<n, 0≤i+d<n

∂/∂h[j] of corr[n-1+d]:
- If j appears as i (first factor): i=j, valid if 0≤j<n and 0≤j+d<n → contributes (1-h[j+d])
- If j appears as i+d (second factor): i=j-d, valid if 0≤j-d<n and 0≤j<n → contributes -h[j-d]

So:
∂corr[n-1+d]/∂h[j] = (1-h[j+d])*I(0≤j+d<n) - h[j-d]*I(0≤j-d<n)

where I(·) is the indicator.

```python
def _grad_lag(h, d, n):
    grad = np.zeros(n)
    if d >= 0:
        # i ranges 0 to n-1-d
        # First factor: h[i], second: h[i+d]
        i_range = np.arange(n - d)
        grad[i_range] += (1.0 - h[i_range + d])
        # For second factor: h[i+d], i+d ranges d to n-1
        grad[d:] -= h[:n-d]
    else:
        dd = -d
        # i ranges dd to n-1
        # First factor: h[i], second: h[i-dd]
        i_range = np.arange(dd, n)
        grad[i_range] += (1.0 - h[i_range - dd])
        # For second factor: h[i-dd], i-dd ranges 0 to n-1-dd
        grad[:n-dd] -= h[dd:]
    return grad
```

```python
    if s < 1e-14:

    if d >= 0:
        nr = n - d
        grad[:nr] += (1.0 - h[d:])
        grad[d:] -= h[:nr]
    else:
        dd = -d
        grad[dd:] += (1.0 - h[:n-dd])
        grad[:n-dd] -= h[dd:]
    return grad

                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = _repair(res.x, n)
        if _c5(hx, dx) < _c5(h, dx):
            return hx
    except:
        pass
    return h.copy()

    lr = lr_init

            break
        corr = _corr(best)
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
```

```python
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                best[i], best[j] = oi, oj
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return best, best_c

def _sa(h, n, dx, deadline, T_init=0.025, T_min=0.002, max_iter=20000):
    best = h.copy()
    best_c = _c5(best, dx)
    cur_c = best_c
    T = T_init
    rng = np.random.RandomState(42)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        delta = (rng.rand() - 0.5) * 0.15
        ni, nj = best[i] + delta, best[j] - delta
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        oi, oj = best[i], best[j]
        best[i], best[j] = ni, nj
        nc = _c5(best, dx)
        if nc < cur_c or rng.rand() < np.exp(-(nc - cur_c) / T):
            cur_c = nc
            if nc < best_c:
                best_c = nc
        else:
            best[i], best[j] = oi, oj
        T = max(T_min, T * 0.9997)
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c = _c5(h0, dx)

    def _ok(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: Subgradient descent (targeted at the max)
    if _ok(350):
        for tk, lr0 in [(3, 0.12), (5, 0.08)]:
            if not _ok(300):
                break
            dl = t0 + budget_s - 350
            h_new, c_new = _subgrad(best_h, n, dx, dl, 60000, tk, lr0)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Phase 2: SLSQP polish
    for alpha in [60, 150, 400, 1200]:
        if not _ok(250):
            break
        h_new = _slsqp(best_h, n, dx, alpha, 350, 1e-15)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 3: Directed local search
    if _ok(200):
        dl = t0 + budget_s - 200
        best_h, best_c = _directed_ls(best_h, n, dx, dl, 600)

    # Phase 4: Simulated annealing exploration
    if _ok(180):
        dl = t0 + budget_s - 180
        h_sa, c_sa = _sa(best_h, n, dx, dl, 0.03, 0.004, 25000)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa
        if _ok(150):
            h_new = _slsqp(best_h, n, dx, 300, 250, 1e-15)
            c_new = _c5(h_new, dx)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
            if _ok(130):
                dl2 = t0 + budget_s - 130
                best_h, best_c = _directed_ls(best_h, n, dx, dl2, 400)

    # Phase 5: Second subgradient pass (different params)
    if _ok(130):
        dl = t0 + budget_s - 130
        h_new, c_new = _subgrad(best_h, n, dx, dl, 40000, 7, 0.04)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        if _ok(100):
            dl2 = t0 + budget_s - 100
            best_h, best_c = _directed_ls(best_h, n, dx, dl2, 350)

    # Phase 6: Restart with perturbations
    if _ok(100):
        for sigma in [0.12, 0.2, 0.06, 0.3]:
            if not _ok(60):
                break
            hp = _repair(best_h + sigma * rng.randn(n), n)
            dl = t0 + budget_s - 60
            h_new, c_new = _subgrad(hp, n, dx, dl, 15000, 3, 0.09)
            if c_new < best_c - 0.0003 and _ok(50):
                h_new = _slsqp(h_new, n, dx, 400, 200, 1e-15)
                c_new = _c5(h_new, dx)
                if _ok(40):
                    dl2 = t0 + budget_s - 40
                    h_new, c_new = _directed_ls(h_new, n, dx, dl2, 250)
                if c_new < best_c:
                    best_h, best_c = h_new, c_new

    # Phase 7: Final polish
    if _ok(25):
        h_new = _slsqp(best_h, n, dx, 3000, 300, 1e-16)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    if _ok(8):
        dl = t0 + budget_s - 3
        best_h, best_c = _directed_ls(best_h, n, dx, dl, 300)

    best_h = _repair(best_h, n)
    best_c = _c5(best_h, dx)
    return (best_h, best_c, n)
```
