import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth(h, a):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _grad_smooth(h, a):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s)
    w /= np.sum(w)
    g = np.zeros(n)
    for t in range(n):
        gt = 0.0
        for k in range(2 * n - 1):
            mk = k - (n - 1)
            if abs(w[k]) < 1e-30:
                continue
            if 0 <= t + mk < n:
                gt += w[k] * (1.0 - h[t + mk])
            if 0 <= t - mk < n:
                gt -= w[k] * h[t - mk]
        g[t] = gt
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, tgt / n)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _pgd(h, n, dx, deadline, steps, a_init=100, lr_init=0.02):
    """Projected gradient with adaptive alpha and step size"""
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    a = a_init
    lr = lr_init
    for s in range(steps):
        if time.time() >= deadline:
            break
        g = _grad_smooth(h, a)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new *= tgt / max(h_new.sum(), 1e-12)
        h_new = np.clip(h_new, 0.0, 1.0)
        d = tgt - h_new.sum()
        if abs(d) > 1e-10:
            i = int(np.argmax(1.0 - h_new)) if d > 0 else int(np.argmax(h_new))
            h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
        else:
            # Reduce step size
            lr *= 0.9
            if lr < 1e-6:
                lr = lr_init * 0.5
                a = max(a * 0.7, 5)
        if s % 100 == 0 and s > 0:
            a = max(a * 0.8, 5)
    return h, best

def _anneal_local(h, n, dx, rng, deadline, T0=0.01, T1=0.0001):
    """Simulated annealing with pair swaps"""
    best = _c5(h, dx)
    h = h.copy()
    cur_c = best
    n_steps = 0
    T = T0
    while time.time() < deadline:
        batch = 500
        for _ in range(batch):
            if time.time() >= deadline:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = 0.01 * rng.uniform(-1, 1)
            hi, hj = h[i] + d, h[j] - d
            if hi < -1e-9 or hi > 1 + 1e-9 or hj < -1e-9 or hj > 1 + 1e-9:
                continue
            h[i], h[j] = hi, hj
            c_new = _c5(h, dx)
            n_steps += 1
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best:
                    best = c_new
            else:
                # Maybe accept
                if rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-8)):
                    cur_c = c_new
                else:
                    h[i], h[j] = hi - d, hj + d
        T = max(T1, T * 0.97)
    return h, best

def _slsqp(h, n, dx, a, maxit, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    tgt = n / 2.0
    try:
        res = minimize(
            lambda x, aa=a: _smooth(x, aa), h,
            method='SLSQP', bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
            options={'maxiter': maxit, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth_fn(x, a):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _grad_smooth(h, a):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s)
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    g = np.zeros(n)
    for t in range(n):
        gt = 0.0
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-25:
                continue
            mk = k - (n - 1)
            if 0 <= t + mk < n:
                gt += wk * (1.0 - h[t + mk])
            if 0 <= t - mk < n:
                gt -= wk * h[t - mk]
        g[t] = gt
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, tgt / n)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h
