def _sgrad(x, a, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx1 = np.arange(n - lag)
            g[idx1] += wk * (1.0 - x[idx1 + lag])
            g[idx1 + lag] -= wk * x[idx1]
        else:
            L = -lag
            idx1 = np.arange(L, n)
            g[idx1] += wk * (1.0 - x[idx1 - L])
            g[idx1 - L] -= wk * x[idx1]
    return g

def _fista(h0, alpha, target, n, dx, iters):
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx, bc = h0.copy(), _c5(h0, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for _ in range(iters):
        gz = _sgrad(z, alpha, n)
        y = z - step * gz
        xn = _proj(y, target)
        fn = _sobj(xn, alpha)
        quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
        cnt = 0
        while fn > quad + 1e-10 and cnt < 6:
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_pass(h, dx, n, c_vals, order):
    """One pass of exact 1D coordinate descent with golden-section refinement."""
    target = n / 2.0
    n2 = 2 * n - 1
    for j in order:
        hj = h[j]
        # Compute slopes: for lag m, if 0<=m-j<n and m!=2j: slope=(1-2*h[m-j])*dx
        slopes = np.zeros(n2)
        for m in range(n2):
            k2 = m - j
            if 0 <= k2 < n and m != 2 * j:
                slopes[m] = (1.0 - 2.0 * h[k2]) * dx
        quad_m = 2 * j
        if quad_m >= n2:
            quad_m = -1
        # Grid eval
        ts = np.linspace(0.0, 1.0, 161)
        lin = c_vals + slopes * ts[:, None]
        if quad_m >= 0:
            lin[:, quad_m] += (ts * (1.0 - ts) - hj * (1.0 - hj)) * dx
        fv = lin.max(axis=1)
        bi = int(np.argmin(fv))
        lo = float(ts[max(0, bi - 1)])
        hi = float(ts[min(160, bi + 1)])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(35):
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            v1 = c_vals + slopes * m1
            if quad_m >= 0:
                v1 = v1.copy()
                v1[quad_m] += (m1 * (1.0 - m1) - hj * (1.0 - hj)) * dx
            f1 = float(v1.max())
            v2 = c_vals + slopes * m2
            if quad_m >= 0:
                v2 = v2.copy()
                v2[quad_m] += (m2 * (1.0 - m2) - hj * (1.0 - hj)) * dx
            f2 = float(v2.max())
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        best_t = (lo + hi) * 0.5
        h[j] = best_t
        # Update c_vals
        c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    # Project at end of pass
    h = _proj(h, target)
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return h, c_vals, float(c_vals.max())

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)

    h = _proj(np.array(initial_h_values, dtype=float), tgt)
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    bh = h.copy()
    bc = float(c_vals.max())

    # Phase 1: FISTA with moderate alpha
    for a in [100, 300, 800]:
        if time.time() - t0 > budget_s - 120:
            break
        h, c = _fista(bh, a, tgt, n, dx, 1500)
        if c < bc:
            bc = c
            bh = h.copy()

    h = bh.copy()
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    order = np.arange(n)

    # Phase 2: Intensive CD (main workhorse)
    passes = 0
    while time.time() - t0 < budget_s - 100:
        h, c_vals, c = _cd_pass(h, dx, n, c_vals, order)
        passes += 1
        if c < bc:
            bc = c
            bh = h.copy()
        # Occasionally shuffle order
        if passes % 50 == 0:
            order = rng.permutation(n)

    # Phase 3: FISTA with high alpha polish
    if time.time() - t0 < budget_s - 60:
        h, c = _fista(bh, 5000, tgt, n, dx, 800)
        if c < bc:
            bc = c
            bh = h.copy()
        h = bh.copy()
        c_vals = np.correlate(h, 1.0 - h, mode='full') * dx

    # Phase 4: More CD after FISTA polish
    while time.time() - t0 < budget_s - 40:
        h, c_vals, c = _cd_pass(h, dx, n, c_vals, order)
        if c < bc:
            bc = c
            bh = h.copy()
        if (passes % 30) == 0:
            order = rng.permutation(n)
        passes += 1

    # Phase 5: Perturbation + CD (escape local optima)
    for s in [0.05, 0.03, 0.08, 0.04]:
        if time.time() - t0 > budget_s - 50:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        hv = hp.copy()
        cv = np.correlate(hv, 1.0 - hv, mode='full') * dx
        for _ in range(30):
            hv, cv, cv_max = _cd_pass(hv, dx, n, cv, order)
            if cv_max < bc - 1e-10:
                bc = cv_max
                bh = hv.copy()
        h = bh.copy()
        c_vals = np.correlate(h, 1.0 - h, mode='full') * dx

    # Phase 6: Final high-alpha FISTA + CD
    if time.time() - t0 < budget_s - 20:
        h, c = _fista(bh, 20000, tgt, n, dx, 500)
        if c < bc:
            bc = c
            bh = h.copy()

    if time.time() - t0 < budget_s - 8:
        h = bh.copy()
        c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        for _ in range(20):
            h, c_vals, c = _cd_pass(h, dx, n, c_vals, order)
            if c < bc:
                bc = c
                bh = h.copy()

    bh = _proj(bh, tgt)
    c_final = _c5(bh, dx)
    if c_final < bc:
        bc = c_final
    return (bh, bc, n)
