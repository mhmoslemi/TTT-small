def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min > d_max - 1e-12:
                continue
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:

                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):

    try:

                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

This is exactly what _g_vec computes for a single component! _g_vec(h, p, n) gives the gradient of c with respect to h[p] for ALL lags. But I want the gradient of c at a specific lag with respect to ALL h values. These are transposes of each other.

Let me just compute it directly for each active lag:

```python
def _grad_c_lag(h, lag, n):
    """Gradient of c[lag+n-1] w.r.t. h"""
    g = np.zeros(n)
    # (1 - h[q+lag]) for 0 <= q+lag < n, i.e., -lag <= q < n-lag
    lo1 = max(0, -lag)
    hi1 = min(n, n - lag)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1+lag:hi1+lag])
    # -h[q-lag] for 0 <= q-lag < n, i.e., lag <= q < n+lag
    lo2 = max(0, lag)
    hi2 = min(n, n + lag)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2-lag:hi2-lag]
    return g
```

This is O(n) per lag, and we sum over active lags. If there are |A| active lags, total is O(n*|A|). Very fast.

        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    d_best = ds2[np.argmin(c5s2)]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _eq_con(x, target):
    return np.sum(x) - target

def _obj_smax(x, alpha):
    return _smooth_max_val(_corr(x), alpha)

                      options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:

def _first_order(h, n, dx, target, alpha, lr, n_iters, mom):
    h = h.copy()
    v = np.zeros(n)
    cur = _c5(h, dx)
    for _ in range(n_iters):
        corr = _corr(h)
        g = _grad_smooth(h, corr, alpha, n)
        v = mom * v + (1.0 - mom) * g
        for sc in [1.0, 0.5, 0.25, 0.125, 0.0625]:

            c5t = _c5(ht, dx)
            if c5t < cur - 1e-15:
                h = ht
                cur = c5t
                break
        else:
            break
    return h

def _adam_step(h, n, dx, target, alpha, lr, n_iters):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur = _c5(h, dx)
    for t in range(1, n_iters + 1):
        corr = _corr(h)
        g = _grad_smooth(h, corr, alpha, n)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        for sc in [1.0, 0.5, 0.25, 0.1]:
            ht = _repair(h - sc * step, n)
            c5t = _c5(ht, dx)
            if c5t < cur - 1e-15:
                h = ht
                cur = c5t
                break
        else:
            break
    return h
