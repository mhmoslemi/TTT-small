import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _coord_step(h, j, n):
    """Solve 1D subproblem: min_t max_k (A_k + B_k*t) for t in [0,1]."""
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    k_vals = np.arange(2 * n - 1) - (n - 1)
    jk1 = j - k_vals
    jk2 = j + k_vals
    mask1 = (jk1 >= 0) & (jk1 < n)
    mask2 = (jk2 >= 0) & (jk2 < n)
    B = np.zeros(2 * n - 1)
    B[mask1] += (1.0 - h[jk1[mask1]])
    B[mask2] -= h[jk2[mask2]]
    A = corr_full - B * h[j]
    # Evaluate at 41 points for accuracy
    t_samples = np.linspace(0.0, 1.0, 41)
    vals = np.max(A[None, :] + B[None, :] * t_samples[:, None], axis=1)
    best_t = t_samples[np.argmin(vals)]
    return best_t

def _subgradient_step(h, c5, corr, n, lr):
    """One subgradient step targeting the active max constraint."""
    k_star = np.argmax(corr)
    idx = k_star
    k = idx - (n - 1)
    j_arr = np.arange(n)
    jk1 = j_arr - k
    jk2 = j_arr + k
    m1 = (jk1 >= 0) & (jk1 < n)
    m2 = (jk2 >= 0) & (jk2 < n)
    grad = np.zeros(n)
    grad[m1] += (1.0 - h[jk1[m1]])
    grad[m2] -= h[jk2[m2]]
    grad -= np.mean(grad)  # project onto sum=const direction
    new_h = h - lr * grad
    return _repair(new_h, n)

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _local_search_block(h, n, dx, rng, c5_cur, best_c5, best_h, deadline):
    """Intensive local search with block and pairwise moves."""
    h = h.copy()
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for scale in scales:
        if time.time() >= deadline:
            break
        stalled = 0
        while stalled < 4 and time.time() < deadline:
            improved = False
            for _ in range(3000):
                if time.time() >= deadline:
                    break
                r = rng.random()
                if r < 0.4:
                    # Pairwise swap
                    i, j = rng.randint(n, size=2)
                    if i == j:
                        continue
                    d = scale * (2.0 * rng.random() - 1.0)
                    ni, nj = h[i] + d, h[j] - d
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                elif r < 0.75:
                    # Block move (2-4 adjacent)
                    bs = rng.randint(2, 5)
                    if bs >= n:
                        bs = 2
                    s = rng.randint(0, n - bs)
                    idxs = np.arange(s, s + bs)
                    d = (rng.random(bs) - 0.5) * 2.0 * scale
                    d -= d.mean()
                    nv = h[idxs] + d
                    if np.any(nv < -1e-10) or np.any(nv > 1.0 + 1e-10):
                        continue
                    old = h[idxs].copy()
                    h[idxs] = nv
                else:
                    # Single coordinate nudge
                    j = rng.randint(n)
                    d = scale * (2.0 * rng.random() - 1.0)
                    nv = h[j] + d
                    if nv < -1e-10 or nv > 1.0 + 1e-10:
                        continue
                    h[j] = nv
                    old = np.array([h[j]])
                    idxs = np.array([j])
                else:
                    pass
                c5_new, _ = _compute_c5(h, dx)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                else:
                    if r < 0.4:
                        h[i], h[j] = h[j] + d, h[i] - d
                    else:
                        h[idxs] = old
            if not improved:
                stalled += 1
    return h, c5_cur, best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5, _ = _compute_c5(h, dx)

    deadline = t_start + budget_s

    # Phase 1: Coordinate descent (fast, exact 1D subproblems)
    t_end = min(t_start + 250, deadline - 400)
    for sweep in range(200):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            new_t = _coord_step(h, j, n)
            if abs(new_t - h[j]) > 1e-9:
                h[j] = new_t
                improved = True
        h = _repair(h, n)
        c5, _ = _compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break

    # Phase 2: Subgradient + coordinate descent interleaved
    t_end = min(t_start + 400, deadline - 250)
    h = best_h.copy()
    c5_cur, corr = _compute_c5(h, dx)
    lr = 0.05
    for it in range(2000):
        if time.time() > t_end:
            break
        if it % 50 == 0:
            # Coordinate descent sweep
            for j in range(n):
                h[j] = _coord_step(h, j, n)
            h = _repair(h, n)
            c5_cur, corr = _compute_c5(h, dx)
            if c5_cur < best_c5 - 1e-13:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            # Subgradient step
            h = _subgradient_step(h, c5_cur, corr, n, lr)
            c5_cur, corr = _compute_c5(h, dx)
            if c5_cur < best_c5 - 1e-13:
                best_c5 = c5_cur
                best_h = h.copy()
        lr *= 0.999  # Slow decay

    # Phase 3: L-BFGS-B with smooth max (multiple alpha levels)
    t_end = min(t_start + 650, deadline - 200)
    h = best_h.copy()
    for alpha in [80, 200, 500, 1200, 3000]:
        if time.time() > t_end:
            break
        try:
            def obj_lbfgs(x, a=alpha):
                corr_x = np.correlate(x, 1.0 - x, mode='full')
                sm = _smooth_max(corr_x, a)
                pen = 500.0 * (np.sum(x) - target_sum) ** 2
                return sm + pen
            res = minimize(obj_lbfgs, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12})
            hc = _repair(res.x, n)
            c5, _ = _compute_c5(hc, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = hc.copy()
                h = hc.copy()
        except Exception:
            pass
    # Also try SLSQP with constraint
    if time.time() < t_end - 60:
        try:
            def obj_slsqp(x, a=2000):
                corr_x = np.correlate(x, 1.0 - x, mode='full')
                return _smooth_max(corr_x, a)
            eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds,
                          constraints=[eq], options={'maxiter': 600, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5, _ = _compute_c5(hc, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = hc.copy()
                h = hc.copy()
        except Exception:
            pass

    # Phase 4: Intensive block local search
    t_end = min(t_start + 950, deadline - 100)
    h, c5_cur, best_c5, best_h = _local_search_block(
        best_h, n, dx, rng, best_c5, best_c5, best_h, t_end)

    # Phase 5: Final coordinate descent + micro-polish
    t_end = deadline - 20
    h = best_h.copy()
    for sweep in range(100):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            new_t = _coord_step(h, j, n)
            if abs(new_t - h[j]) > 1e-11:
                h[j] = new_t
                improved = True
        h = _repair(h, n)
        c5, _ = _compute_c5(h, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break

    # Final SLSQP micro-polish
    if time.time() < deadline - 10:
        try:
            def obj_final(x, a=5000):
                corr_x = np.correlate(x, 1.0 - x, mode='full')
                return _smooth_max(corr_x, a)
            eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds,
                          constraints=[eq], options={'maxiter': 300, 'ftol': 1e-17})
            hc = _repair(res.x, n)
            c5, _ = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Final local search micro-polish
    if time.time() < deadline - 5:
        h2, c5_2, best_c5, best_h = _local_search_block(
            best_h, n, dx, rng, best_c5, best_c5, best_h, deadline - 1)

    best_h = _repair(best_h, n)
    best_c5, _ = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
