import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    corr = np.correlate(h, 1.0 - h, mode='full')
    fvals = corr * dx
    sm = _smooth_max(fvals, alpha)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    val = sm + penalty
    
    # gradient
    m = np.max(fvals)
    s = np.clip(alpha * (fvals - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    # sm gradient w.r.t. corr: w (same shape as corr)
    # corr[k] for k=0..2n-2, shift = k-(n-1)
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for ki, sh in enumerate(shifts):

        if wk < 1e-12:
            continue
        # d corr[ki]/dh[j] = (1-h[j+sh]) if j+sh valid, plus -h[j-sh] if j-sh valid

Valid i range: i ∈ [max(0, -s), min(n, n-s))

- If j is a valid i (i.e., j+s is valid): contributes (1 - h[j+s])
- If j-s is a valid i (i.e., j = i+s for valid i, so i=j-s is valid): contributes -h[j-s]
- These are the same when j+s and j-s are both valid.

So: ∂corr[k]/∂h[j] = [j+s valid] * (1-h[j+s]) - [j-s valid and j-s in valid_i_range] * h[j-s]

Wait I need to be more careful. Let's say s is the shift.

corr[k] = ∑_{i ∈ S} h[i] * (1 - h[i+s]) where S = {i : 0≤i<n, 0≤i+s<n}

Expand: corr[k] = ∑_{i∈S} h[i] - ∑_{i∈S} h[i]*h[i+s]

∂/∂h[j] of ∑_{i∈S} h[i] = 1 if j∈S, else 0

∂/∂h[j] of ∑_{i∈S} h[i]*h[i+s]:
- j appears as h[i] when i=j: contributes h[j+s] if j∈S
- j appears as h[i+s] when i+s=j, i=j-s: contributes h[j-s] if j-s∈S

Where j∈S means max(0,-s) ≤ j < min(n, n-s)

- First term applies for j ∈ [lo, hi)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_max_val(fvals, alpha):
    m = np.max(fvals)
    s = np.clip(alpha * (fvals - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softmax_weights(fvals, alpha):
    m = np.max(fvals)
    s = np.clip(alpha * (fvals - m), -500, 500)
    e = np.exp(s)
    return e / np.sum(e)

def _corr_grad(h, n):
    """Compute d(corr)/dh as a (2n-1, n) matrix of partial derivatives."""
    # For shift s, valid i in [lo, hi)
    # d_corr[shift_idx]/dh[j] = (lo<=j<hi)*(1-h[j+s]) - (lo+s<=j<hi+s)*h[j-s]
    result = np.zeros((2 * n - 1, n))
    for ki in range(2 * n - 1):
        sh = ki - (n - 1)
        lo = max(0, -sh)
        hi = min(n, n - sh)
        if lo >= hi:
            continue
        # First term: j in [lo, hi), contributes (1-h[j+sh])
        jj = np.arange(lo, hi)
        result[ki, jj] += (1.0 - h[jj + sh])
        # Second term: j in [lo+sh, hi+sh) intersect [0,n), contributes -h[j-sh]
        j2lo = max(0, lo + sh)
        j2hi = min(n, hi + sh)
        if j2lo < j2hi:
            jj2 = np.arange(j2lo, j2hi)
            result[ki, jj2] -= h[jj2 - sh]
    return result

def _fun(x, alpha, dx, n, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    fvals = corr * dx
    sm = _smooth_max_val(fvals, alpha)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return sm + penalty

def _jac(x, alpha, dx, n, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    fvals = corr * dx
    w = _softmax_weights(fvals, alpha)
    cg = _corr_grad(h, n)
    grad = (w @ cg).ravel() * dx
    grad += 2.0 * lam * (np.sum(x) - n / 2.0)
    return grad

def _lbfgs_opt(h, n, dx, alpha, lam, maxiter, timeout):
    t_start = time.time()
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_fun, x0, jac=_jac, args=(alpha, dx, n, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e16})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _sweep_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    no_improve = 0
    step_scales = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, -0.01, -0.02, -0.05, -0.1, -0.2, -0.5]
    while time.time() < deadline and no_improve < 15:
        improved = False
        n_pairs = min(n * (n - 1) // 2, 500)
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for sc in step_scales:
                delta = sc
                ni, nj = best[i] - delta, best[j] + delta
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = best[i] + delta, best[j] - delta
                if improved:
                    break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return best, best_c5

def _grad_descent(h, n, dx, deadline, lr0=0.01):
    best = h.copy()
    best_c5 = _c5(best, dx)
    lr = lr0
    stall = 0
    while time.time() < deadline and stall < 60:
        h = np.clip(best, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        fvals = corr * dx
        w = _softmax_weights(fvals, 200)
        cg = _corr_grad(h, n)
        grad = (w @ cg).ravel() * dx
        # Project to sum-zero direction
        grad -= np.mean(grad)
        # Clip direction to respect bounds
        pos_mask = grad < 0
        neg_mask = grad > 0
        max_up = (1.0 - h) * (lr / max(1, np.abs(grad).max()))
        max_dn = h * (lr / max(1, np.abs(grad).max()))
        step = np.where(pos_mask, np.minimum(-grad, max_up).clip(0, None),
                        np.where(neg_mask, np.maximum(-grad, -max_dn).clip(None, 0), 0))
        step -= np.mean(step)
        new_h = best + step
        new_h = np.clip(new_h, 0, 1)
        new_h = _repair(new_h, n)
        nc = _c5(new_h, dx)
        if nc < best_c5 - 1e-14:
            best = new_h
            best_c5 = nc
            stall = 0
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.7
            stall += 1
    return best, best_c5

def _structured_init(n, freq, phase, noise, rng):
    t = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + noise * rng.randn(n)
    h += 0.25 * np.sin(2 * np.pi * * t + phase)

def _binary_init(n, rng):
