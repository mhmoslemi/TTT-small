import time

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -400, 400)))) / alpha

def _pair_delta(i, j, hi_v, hj_v, s, h, n):
    ncorr = 2 * n - 1
    A = np.zeros(ncorr); B = np.zeros(ncorr); C = np.zeros(ncorr)
    for k in range(ncorr):
        m = k - (n - 1)
        # p=i term: valid if 0<=i+m<n
        if 0 <= i + m < n:
            im = i + m
            if im == j:
                A[k] += 1 - s; B[k] += 1; C[k] -= hi_v * (1 - hj_v)
            elif im == i:
                pass  # m==0, handled with p=j diagonal
            else:
                A[k] += (1 - h[im]); C[k] -= hi_v * (1 - h[im])
        # p=j term: valid if 0<=j+m<n
        if 0 <= j + m < n:
            jm = j + m
            if jm == i:
                A[k] += -(s + 1); B[k] += 1; C[k] += hi_v * (1 + hj_v)
            elif jm == j:
                if m == 0:
                    # m==0: combined diagonal handled below
                    pass
                else:
                    A[k] += -(1 - h[jm]); C[k] += hi_v * (1 - h[jm])
            else:
                A[k] += -(1 - h[jm]); C[k] += hi_v * (1 - h[jm])
        # m==0 special: t(1-t)+(s-t)(1-s+t) - hi(1-hi)-hj(1-hj) = -2t^2+2st-2hi*hj
        if m == 0:
            A[k] += 2 * s; B[k] += -2; C[k] += -2 * hi_v * hj_v
        # p=i-m term: p=i-m valid, p!=i, p!=j
        p = i - m
        if 0 <= p < n and p != i and p != j:
            A[k] += -h[p]; C[k] += h[p] * hi_v
        # p=j-m term: p=j-m valid, p!=i, p!=j
        p = j - m
        if 0 <= p < n and p != i and p != j:
            A[k] += h[p]; C[k] += -h[p] * hi_v
    return A, B, C

def _eval_max(A, B, C, base, t, dx):
    # result[k](t) = base[k] + A[k]*t + B[k]*t^2 + C[k]
    vals = base + A * t + B * t * t + C
    return float(np.max(vals) * dx)

def _cd_pass(h, n, dx, n_evals=9):
    ncorr = 2 * n - 1
    best_c5 = _c5(h, dx)
    improved_any = False
    pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for (i, j) in pair_list:
        hi_v = h[i]; hj_v = h[j]; s = hi_v + hj_v
        lo = max(0.0, s - 1.0); hi = min(1.0, s)
        if hi - lo < 1e-12:
            continue
        base = np.correlate(h, 1.0 - h, mode='full')
        A, B, C = _pair_delta(i, j, hi_v, hj_v, s, h, n)
        # Evaluate at n_evals points
        best_t = hi_v; best_c = np.inf
        if n_evals <= 1:
            ts = np.array([lo, hi])
        else:
            ts = np.linspace(lo, hi, n_evals)
        for t in ts:
            c = _eval_max(A, B, C, base, t, dx)
            if c < best_c:
                best_c = c; best_t = t
        # Refine with golden-section if interior
        if best_t > lo + 1e-10 and best_t < hi - 1e-10:
            a, b = lo, hi
            for _ in range(8):

                m2 = b - 0.382 * (b - a)
                c1 = _eval_max(A, B, C, base, m1, dx)
                c2 = _eval_max(A, B, C, base, m2, dx)
                if c1 < c2:
                    b = m2
                else:
                    a = m1
            best_t = 0.5 * (a + b)

            h[i] = best_t; h[j] = s - best_t
            best_c5 = best_c
            improved_any = True
        else:
            h[i] = hi_v; h[j] = hj_v
    return h, best_c5, improved_any

def _cd_loop(h, n, dx, t_start, t_limit, n_passes=2000, n_evals=9):
    for _ in range(n_passes):
        if time.time() - t_start > t_limit:
            break
        h, c, imp = _cd_pass(h, n, dx, n_evals)
        if not imp:
            break
    return h, _c5(h, dx)

def _slsqp(h, n, bounds, cons, alpha, maxiter, ftol=1e-14):
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

            if c < best_c5:
                best_c5, best_h = c, h.copy()
                h2, c2 = _cd_loop(h, n, dx, t0, min(budget_s - 50, elapsed() + 40), n_passes=1000, n_evals=11)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    # Phase 7: Final fine CD
    if elapsed() < budget_s - 15:
        h, c = _cd_loop(best_h, n, dx, t0, budget_s - 18, n_passes=3000, n_evals=21)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 8: Final SLSQP micro-polish
    if elapsed() < budget_s - 5:
        h = _slsqp(best_h, n, bounds, cons, 2500, 400, 1e-16)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
        # One more CD pass
        if elapsed() < budget_s - 2:
            h, c = _cd_loop(best_h, n, dx, t0, budget_s - 3, n_passes=500, n_evals=21)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
