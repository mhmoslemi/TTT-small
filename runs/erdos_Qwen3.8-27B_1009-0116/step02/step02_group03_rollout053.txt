def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _soft_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    # Repair h to satisfy 0<=h<=1 and sum(h)=n/2
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        if s == 0:
            h = np.full(n, 0.5)
            break
        h = h * (target / s)
        h = np.clip(h, 0, 1)
    # Final correction
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                a = min(diff, 1 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

                else:
                    h[ia] += delta
                    h[ib] -= delta
        if not improved:
            prob = 0.05 + 0.1 * (it / float(max_iters))
            if rng.rand() < prob:
                i1, i2 = rng.choice(n, 2, replace=False)
                d = rng.uniform(0.02, 0.08)
                if rng.rand() < 0.5:
                    h[i1] -= d
                    h[i2] += d
                else:
                    h[i1] += d
                    h[i2] -= d
                if h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
                    h[i1] += d if h[i1] < h[i1] else -d
                    h = _repair(h, n)
                else:
                    c = _c5_val(h, n)
                    if c < best_c:
                        best_c = c
    return _repair(h, n), best_c

def _coarse_to_fine(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 1, n_coarse, endpoint=False)
    x_f = np.linspace(0, 1, n_fine, endpoint=False)
    h_f = np.interp(x_f, x_c, h_coarse, period=1.0)
    return _repair(h_f, n_fine)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70
    deadline = t0 + budget_s - 15

    def _try(h, n):
        nonlocal best_h, best_c5, best_n
        h = _repair(h, n)
        c = _c5_val(h, n)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
            best_n = n

    init_h = None
    init_n = 0
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    n_list = [70, 60, 80, 50, 90, 100]
    if init_n > 0:
        if init_n in n_list:
            n_list.remove(init_n)
        n_list = [init_n] + n_list

    for n_idx, n in enumerate(n_list):
        if time.time() > deadline:
            break
        starts = []
        if init_h is not None and init_n == n:
            starts.append(_repair(init_h.copy(), n))
        for kind in range(6):
            starts.append(_structured_init(n, seed + kind * 131 + n * 7, kind))
        starts.append(_fourier_init(n, seed + 999 + n, 4))
        starts.append(_fourier_init(n, seed + 2000 + n, 6))
        starts.append(_repair(0.5 + 0.3 * rng.randn(n), n))

        remaining_n = len(n_list) - n_idx
        time_per_n = (deadline - t0) / max(1, remaining_n)
        max_starts = max(3, min(len(starts), int(time_per_n / 18)))
        starts = starts[:max_starts]

        for h0 in starts:
            if time.time() > deadline:
                break
            h = h0.copy()
            _try(h, n)
            cur_c = _c5_val(h, n)

            for alpha in [50, 150, 500, 2000]:
                if time.time() > deadline:
                    break
                h_new = _lbfgs_opt(h, n, alpha, 150)
                c_new = _c5_val(h_new, n)
                if c_new < cur_c - 1e-9:
                    h = h_new
                    cur_c = c_new
                _try(h, n)

            if time.time() < deadline - 10:
                h_new = _slsqp_opt(h, n, 3000, 150)
                c_new = _c5_val(h_new, n)
                if c_new < cur_c - 1e-9:
                    h = h_new
                    cur_c = c_new
                _try(h, n)

            if time.time() < deadline - 5:
                h_ls, c_ls = _targeted_ls(h, n, 300, deadline, rng)
                _try(h_ls, n)
                if c_ls < cur_c:
                    h = h_ls
                    cur_c = c_ls
                _try(h, n)

    if best_h is not None and time.time() < deadline:
        for alpha in [5000, 10000]:
            if time.time() > deadline - 5:
                break
            h_new = _slsqp_opt(best_h, best_n, alpha, 100)
            _try(h_new, best_n)
        h_fin, c_fin = _targeted_ls(best_h, best_n, 500, deadline, rng)
        _try(h_fin, best_n)

    if best_h is None:
        n = 70
        best_h = _repair(np.full(n, 0.5), n)
        best_c5 = _c5_val(best_h, n)
        best_n = n

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
