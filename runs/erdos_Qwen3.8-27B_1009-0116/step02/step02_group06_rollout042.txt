                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _compute_all_g(h, n):
    """Compute g vectors for all indices. Returns (n, 2n-1) matrix."""
    m = 2 * n - 1
    G = np.zeros((n, m))
    for p in range(n):
        k = np.arange(-(n - 1), n)
        idx1 = p - k
        mask1 = (idx1 >= 0) & (idx1 < n)
        G[p, mask1] += (1.0 - h[idx1[mask1]])
        idx2 = p + k
        mask2 = (idx2 >= 0) & (idx2 < n)
        G[p, mask2] -= h[idx2[mask2]]
    return G

    m = 2 * n - 1
    corr = _corr(h)
    for _ in range(max_passes):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        G = _compute_all_g(h, n)
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                d_min = max(-h[i], h[j] - 1.0)
                d_max = min(1.0 - h[i], h[j])
                if d_max - d_min < 1e-14:
                    continue
                delta = G[i] - G[j]
                # Find d that minimizes max_k(corr[k] + d*delta[k])
                # Use coarse + fine grid
                ds = np.linspace(d_min, d_max, 21)
                vals = np.max(corr[None, :] + ds[:, None] * delta[None, :], axis=1)
                best_idx = np.argmin(vals)
                if vals[best_idx] < np.max(corr) - 1e-14:
                    d_coarse = ds[best_idx]
                    # Fine search
                    lo = max(d_min, d_coarse - (d_max - d_min) / 21)
                    hi = min(d_max, d_coarse + (d_max - d_min) / 21)
                    ds_fine = np.linspace(lo, hi, 21)
                    vals_fine = np.max(corr[None, :] + ds_fine[:, None] * delta[None, :], axis=1)
                    best_fine = np.argmin(vals_fine)
                    d_best = ds_fine[best_fine]
                    h[i] += d_best
                    h[j] -= d_best
                    corr += d_best * delta
                    improved = True
        if not improved:
            break
    return h

    m = 2 * n - 1

        d = np.random.uniform(d_min, d_max)
        # Compute new corr
        # Need G for this... expensive. Let me just recompute.
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d

            corr = corr_new
            cur_c5 = new_c5
            if cur_c5 < best_val:
                best_val = cur_c5
                best_h = h.copy()
            T *= 0.9999  # slow cooling
    return best_h, best_h  # return best

                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dhi - dlo < 1e-14:
                    continue
