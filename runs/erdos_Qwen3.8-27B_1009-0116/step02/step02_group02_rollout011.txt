        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - h[i + lag])
                g[i + lag] -= wk * h[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - h[i - L])
                g[i - L] -= wk * h[i]
    return g

def _fista(h0, alpha, target, iters):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 8:
            fn = _smax(_corr(xn), alpha)
            quad = _smax(_corr(z), alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_single(h, dx, n, passes, rng, steps):
    bc = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        imp = False
        for i in range(n):
            hi0 = h[i]
            for d in steps:
                ni = hi0 + d
                if ni < 0.0 or ni > 1.0:
                    continue
                h[i] = ni
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    hi0 = ni
                    imp = True
                    break
            h[i] = hi0
        if not imp:
            break
    return h, bc

def _cd_pair(h, dx, n, passes, rng):
    bc = _c5(h, dx)
    h = h.copy()
    steps = [0.003, 0.007, 0.015, 0.03, 0.06, 0.1, 0.18,
             -0.003, -0.007, -0.015, -0.03, -0.06, -0.1, -0.18]
    for p in range(passes):
        imp = False
        for _ in range(n * 5):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in steps:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
            if not (imp and h[i] != hi0 and h[j] != hj0):
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _cd_swap(h, dx, n, passes, rng):
    bc = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        imp = False
        for _ in range(n * 4):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c = _c5(h, dx)
            if c <= bc + 1e-14:
                bc = min(bc, c)
                imp = True
            else:
                h[i], h[j] = h[j], h[i]
        if not imp:
            break
    return h, bc

def _subgrad(h, target, lr, iters, n_active=5):
    n = len(h)
    h = h.copy()
    for _ in range(iters):
        c = _corr(h)
        cm = float(c.max())
        thresh = cm - 0.05
        active = np.where(c >= thresh)[0]
        if len(active) > n_active:
            active = active[np.argsort(c[active])[::-1][:n_active]]
        g = np.zeros(n)
        for k in active:
            wt = 1.0 / len(active)
            lag = k - (n - 1)
            if lag >= 0:
                for i in range(n - lag):
                    g[i] += wt * (1.0 - h[i + lag])
                    g[i + lag] -= wt * h[i]
            else:
                L = -lag
                for i in range(L, n):
                    g[i] += wt * (1.0 - h[i - L])
                    g[i - L] -= wt * h[i]
        h = _proj(h - lr * g, target)
    return h

def _slsqp_obj(x, a):
    return _smax(_corr(x), a)

def _slsqp_con(x, t):
    return float(np.sum(x)) - t

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)

    h0 = _proj(np.array(initial_h_values, dtype=float).copy(), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: FISTA with increasing alpha
    for a in [40, 80, 150, 300, 600, 1200, 2500, 5000]:
        if time.time() - t0 > budget_s - 80:
            break
        h, c = _fista(bh, a, tgt, iters=2500)
        if c < bc:
            bc = c
            bh = h

    # Phase 2: Fine single-coordinate descent
    if time.time() - t0 < budget_s - 30:
        ss = [0.003, 0.007, 0.015, 0.03, 0.06, -0.003, -0.007, -0.015, -0.03, -0.06]
        h, c = _cd_single(bh, dx, n, 300, rng, ss)
        if c < bc:
            bc = c
            bh = h

    # Phase 3: Pairwise CD + swap CD
    if time.time() - t0 < budget_s - 20:
        h, c = _cd_pair(bh, dx, n, 400, rng)
        if c < bc:
            bc = c
            bh = h
        h, c = _cd_swap(bh, dx, n, 200, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 4: Subgradient refinement
    if time.time() - t0 < budget_s - 40:
        for lr in [0.003, 0.001, 0.0004, 0.00015]:
            if time.time() - t0 > budget_s - 20:
                break
            h = _subgrad(bh, tgt, lr, iters=400)
            c = _c5(h, dx)
            if c < bc:
                bc = c
                bh = h
        ss2 = [0.001, 0.003, 0.007, 0.015, -0.001, -0.003, -0.007, -0.015]
        h, c = _cd_single(bh, dx, n, 200, rng, ss2)
        if c < bc:
            bc = c
            bh = h

    # Phase 5: Diverse random restarts
    sigmas = [0.04, 0.07, 0.1, 0.15, 0.25, 0.4, 0.6, 0.06, 0.12, 0.2,
              0.3, 0.5, 0.8, 0.08, 0.18, 0.35, 0.55, 0.7, 0.1, 0.22,
              0.45, 0.05, 0.13, 0.3, 0.5, 0.07, 0.6, 0.15, 0.38, 0.9]
    for s in sigmas:
        if time.time() - t0 > budget_s - 70:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        h, c = _fista(hp, 800, tgt, iters=1500)
        if c < bc:
            bc = c
            bh = h
            if time.time() - t0 < budget_s - 35:
                h2, c2 = _cd_single(bh, dx, n, 150, rng, ss2)
                if c2 < bc:
                    bc = c2
                    bh = h2
                h2, c2 = _cd_pair(bh, dx, n, 100, rng)
                if c2 < bc:
                    bc = c2
                    bh = h2

    # Phase 6: Structured restarts (quasi-periodic with irrational structure)
    if time.time() - t0 < budget_s - 80:
        idx = np.arange(n, dtype=float)
        for freq in [1, 2, 3, 5, 7, 1, 2, 3]:
            if time.time() - t0 > budget_s - 40:
                break
            for amp in [0.15, 0.3, 0.4, 0.55]:
                if time.time() - t0 > budget_s - 30:
                    break
                ph = rng.rand() * 2.0 * np.pi
                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * idx / n + ph)
                hs += 0.3 * amp * np.sin(2.0 * np.pi * (freq * 2 + 1) * idx / n + 2.0 * ph)
                hs = _proj(hs, tgt)
                h, c = _fista(hs, 500, tgt, iters=1500)
                if c < bc:
                    bc = c
                    bh = h
                    h2, c2 = _cd_single(bh, dx, n, 100, rng, ss2)
                    if c2 < bc:
                        bc = c2
                        bh = h2

    # Phase 7: SLSQP polish at multiple alphas
    bnds = [(0.0, 1.0)] * n
    for a_polish in [5000, 12000, 25000]:
        if time.time() - t0 > budget_s - 25:
            break
        try:
            r = minimize(_slsqp_obj, bh, args=(a_polish,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 600, 'ftol': 1e-16})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    # Phase 8: Ultra-fine final CD
    if time.time() - t0 < budget_s - 12:
        ss_fin = [0.001, 0.002, 0.005, 0.01, -0.001, -0.002, -0.005, -0.01]
        h, c = _cd_single(bh, dx, n, 500, rng, ss_fin)
        if c < bc:
            bc = c
            bh = h
        h, c = _cd_pair(bh, dx, n, 300, rng)
        if c < bc:
            bc = c
            bh = h
        h, c = _cd_swap(bh, dx, n, 300, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 9: Final subgradient + CD cycle
    if time.time() - t0 < budget_s - 10:
        for lr in [0.0005, 0.0002]:
            if time.time() - t0 > budget_s - 5:
                break
            h = _subgrad(bh, tgt, lr, iters=300)
            c = _c5(h, dx)
            if c < bc:
                bc = c
                bh = h
            h, c = _cd_single(bh, dx, n, 200, rng, ss_fin)
            if c < bc:
                bc = c
                bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
