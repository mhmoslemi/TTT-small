        for _ in range(ntry):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in [0.005, 0.01, 0.02, 0.04, 0.06, -0.005, -0.01, -0.02, -0.04, -0.06]:
                ni, nj = hi + d, hj - d
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best:
                    best = c
                    improved = True
                    break
                else:
                    h[i], h[j] = hi, hj
        if not improved and it > 80:
            break
    return h, best

def _make_start(n, seed, kind):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = 0.5 + 0.25 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.12 * rng.randn(n)
    elif kind == 2:
        period = rng.randint(3, 10)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < (period + 1) // 2 else 0.0
        h += 0.12 * rng.randn(n)
    elif kind == 3:
        freq = rng.randint(2, 9)
        ph = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        h += 0.06 * rng.randn(n)
    elif kind == 4:
        nb = rng.randint(3, 8)
        h = np.zeros(n)
        bs = max(2, n // (2 * nb))
        for bi in range(nb):
            start = bi * 2 * bs
            if start + bs <= n:
                h[start:start + bs] = 1.0
        h += 0.1 * rng.randn(n)
    else:
        h = rng.uniform(0.1, 0.9, n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    def upd(h, n):
        nonlocal best_h, best_c5, best_n
        h = _repair(h, n)
        c = _c5(h, 2.0 / n)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            best_n = n
        return c

    init_h = None
    init_n = 70
    try:
        init_h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
        init_n = len(init_h)
    except NameError:
        init_h = _repair(np.full(70, 0.5), 70)
        init_n = 70

    n_list = [init_n]
    for nn in [50, 60, 64, 68, 72, 76, 80, 84, 90, 100, 120]:
        if nn != init_n and nn not in n_list:
            n_list.append(nn)

    time_alloc = (budget_s - 120) / len(n_list)

    for ni, n in enumerate(n_list):
        if time.time() - t0 > budget_s - 60:
            break
        dx = 2.0 / n
        n_starts = 4
        if n == init_n:
            n_starts = 5
        starts = []
        if n == init_n:
            starts.append(init_h.copy())
        for kind in range(n_starts):
            starts.append(_make_start(n, seed + kind * 104729 + n * 37, kind))

        for si, h0 in enumerate(starts):
            if time.time() - t0 > budget_s - 50:
                break
            h = h0.copy()
            upd(h, n)

            # Phase 1: Coarse LBFGS
            for alpha, ktop, lam in [(30, 5, 0.0), (80, 10, 0.01), (200, 10, 0.03)]:
                if time.time() - t0 > budget_s - 50:
                    break
                h2 = _lbfgs(h, n, alpha, ktop, lam, 200)
                if _c5(h2, dx) < _c5(h, dx):
                    h = h2
                    upd(h, n)

            # Phase 2: Fine LBFGS
            for alpha, ktop, lam in [(400, 15, 0.05), (600, 20, 0.02), (800, 10, 0.1)]:
                if time.time() - t0 > budget_s - 50:
                    break
                h2 = _lbfgs(h, n, alpha, ktop, lam, 150)
                if _c5(h2, dx) < _c5(h, dx):
                    h = h2
                    upd(h, n)

            # Phase 3: SLSQP
            for alpha, ktop, lam in [(300, 10, 0.0), (800, 15, 0.05), (1500, 20, 0.02)]:
                if time.time() - t0 > budget_s - 40:
                    break
                h2 = _slsqp(h, n, alpha, ktop, lam, 200)
                if _c5(h2, dx) < _c5(h, dx):
                    h = h2
                    upd(h, n)

            # Phase 4: Pair coordinate descent
            cd_iters = max(40, int((time_alloc - (time.time() - t0)) / 0.05))
            h, c = _pair_coord(h, n, dx, min(cd_iters, 200), t0 + budget_s - 30, rng)
            upd(h, n)

            # Phase 5: Single coordinate
            h, c = _coord_target(h, n, dx, min(cd_iters, 150), t0 + budget_s - 20, rng)
            upd(h, n)

    # Final polish on global best
    if best_h is not None:
        dx_b = 2.0 / best_n
        for alpha, ktop, lam in [(2000, 20, 0.0), (3000, 15, 0.05)]:
            if time.time() - t0 > budget_s - 15:
                break
            h2 = _slsqp(best_h, best_n, alpha, ktop, lam, 300)
            upd(h2, best_n)
        if time.time() - t0 < budget_s - 10:
            h2 = _lbfgs(best_h, best_n, 5000, 20, 0.03, 300)
            upd(h2, best_n)
        h_f, c_f = _pair_coord(best_h, best_n, dx_b, 300, t0 + budget_s - 3, rng)
        upd(h_f, best_n)
        h_f2, c_f2 = _coord_target(best_h, best_n, dx_b, 200, t0 + budget_s - 1, rng)
        upd(h_f2, best_n)

    if best_h is None:
        best_h = init_h
        best_c5 = _c5(init_h, 2.0 / init_n)
        best_n = init_n

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
