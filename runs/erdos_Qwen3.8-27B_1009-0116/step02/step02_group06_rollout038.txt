    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _pair_ternary(h, i, j, corr, n, iters=35):
    hi, hj = h[i], h[j]
    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_min > d_max - 1e-14:
        return h, corr, False
    gd = _g_vec(h, i, n) - _g_vec(h, j, n)
    cur_max = float(np.max(corr))
    a, b = d_min, d_max
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = float(np.max(corr + m1 * gd))
        f2 = float(np.max(corr + m2 * gd))
        if f1 < f2:
            b = m2
        else:
            a = m1
    d_best = (a + b) / 2.0
    f_best = float(np.max(corr + d_best * gd))
    if f_best < cur_max - 1e-13:
        h[i] = hi + d_best
        h[j] = hj - d_best
        corr = corr + d_best * gd
        return h, corr, True
    return h, corr, False

def _full_sweep(h, n, dx, rng, iters=35, shuffle=True):
    corr = _corr(h)
    improved = False
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    if shuffle:
        rng.shuffle(pairs)
    for (i, j) in pairs:
        h, corr, imp = _pair_ternary(h, i, j, corr, n, iters)
        if imp:
            improved = True
    return h, corr, improved

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
