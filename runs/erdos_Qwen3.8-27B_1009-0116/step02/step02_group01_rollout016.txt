    term_mm[mask_mm] = -h[mm[mask_mm]]
    
    s = (term_jp + term_jm) - (term_mp + term_mm)
    return s

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _pair_slopes(h, j, m):
    n = len(h)
    K = 2 * n - 1
    ks = np.arange(K)
    jp = j + ks
    mj = (jp >= 0) & (jp < n)
    a1 = np.zeros(K)
    a1[mj] = 1.0 - h[jp[mj]]
    jm2 = j - ks
    mj2 = (jm2 >= 0) & (jm2 < n)
    a2 = np.zeros(K)
    a2[mj2] = -h[jm2[mj2]]
    mp = m + ks
    mm = (mp >= 0) & (mp < n)
    b1 = np.zeros(K)
    b1[mm] = 1.0 - h[mp[mm]]
    mm2 = m - ks
    mm2v = (mm2 >= 0) & (mm2 < n)
    b2 = np.zeros(K)
    b2[mm2v] = -h[mm2[mm2v]]
    return (a1 + a2) - (b1 + b2)


def _optimal_delta(c0, s, lo, hi):
    if hi - lo < 1e-14:
        return 0.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = c0 + mid * s
        amax = np.max(vals)
        active = (vals >= amax - 1e-9)
        smin = np.min(s[active])
        smax = np.max(s[active])
        if smin <= 0.0 <= smax:
            return mid
        elif smin > 0.0:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)


def _pair_descent(h, max_passes=30, time_limit=None, t0=None):
    n = len(h)
    c = _corr(h)
    for p in range(max_passes):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and t0 is not None:
                if time.time() - t0 > time_limit:
                    return h
            for m in range(j + 1, n):
                s = _pair_slopes(h, j, m)
                lo = max(-h[j], h[m] - 1.0)
                hi = min(1.0 - h[j], h[m])
                if lo >= hi:
                    continue
                delta = _optimal_delta(c, s, lo, hi)
                if abs(delta) > 1e-13:
                    h[j] += delta
                    h[m] -= delta
                    c += delta * s
                    improved = True
        if not improved:
            break
    return h


def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha
