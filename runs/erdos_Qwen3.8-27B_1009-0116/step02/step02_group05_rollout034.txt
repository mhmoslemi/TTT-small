def _corr_and_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    val = m + np.log(np.sum(e)) / alpha
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k - 1)
        for j in range(j_lo, j_hi + 1):
            # First term: (1-h[j+n-1-k])
            idx2 = j + n - 1 - k
            grad[j] += wk * (1.0 - h[idx2])
            # Second term: -h[j+k-n+1] if j+k-n+1 in J_k
            idx1 = j + k - n + 1
            if j_lo <= idx1 <= j_hi:
                grad[j] -= wk * h[idx1]
    return val, grad
