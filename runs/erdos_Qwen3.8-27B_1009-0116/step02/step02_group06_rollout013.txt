import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _smooth_max_grad(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    return w  # weights for each correlation term

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid, max_rounds=30):
    improved = False
    for i in range(n):
        for j in range(i+1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min > d_max - 1e-12:
                continue
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _proj_grad_step(h, n, dx, alpha, lr, n_inner=50):
    """Projected gradient with backtracking."""
    h = h.copy()
    for _ in range(n_inner):
        c = _corr(h)
        w = _smooth_max_grad(c, alpha)
        # gradient: sum_k w_k * dc_k/dh
        # dc_k/dh_j = (1-h[j+k])*1[0<=j+k<n] - h[j-k]*1[0<=j-k<n]
        grad = np.zeros(n)
        for k in range(-(n-1), n):
            wk = w[k + n - 1]
            if wk < 1e-10:
                continue
            # term 1: (1-h[j+k]) for j in [0,n) and j+k in [0,n)
            for j in range(n):
                if 0 <= j+k < n:
                    grad[j] += wk * (1.0 - h[j+k])
                if 0 <= j-k < n:
                    grad[j] -= wk * h[j-k]
        # backtracking line search
        best_h = h.copy()
        best_val = _smooth_max(c, alpha)
        step = lr
        for _bt in range(20):
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _repair(h_new, n)
            val = _smooth_max(_corr(h_new), alpha)
            if val < best_val - 1e-15:
                best_h = h_new
                best_val = val
                break
            step *= 0.5
        h = best_h
    return h

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _grad_h(h, n, alpha):
    """Gradient of smooth_max(corr(h), alpha) with respect to h."""
    c = _corr(h)
    w = _smooth_max_grad(c, alpha)
    grad = np.zeros(n)
    for ki in range(len(w)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        k = ki - (n - 1)
        # (1-h[j+k]) for j where 0<=j<n and 0<=j+k<n
        jlo1 = max(0, -k)
        jhi1 = min(n, n - k)
        if jlo1 < jhi1:
            grad[jlo1:jhi1] += wk * (1.0 - h[jlo1+k:jhi1+k])
        # -h[j-k] for j where 0<=j<n and 0<=j-k<n
        jlo2 = max(0, k)
        jhi2 = min(n, n + k)
        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= wk * h[jlo2-k:jhi2-k]
    return grad

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min > d_max - 1e-12 or d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    d_best = ds2[np.argmin(c5s2)]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _slsqp(h, n, target, bounds, alpha, maxiter, t0, budget_s):
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def eq_con(x):
        return np.sum(x) - target
    def eq_con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _lfgsb(h, n, target, bounds, alpha, penalty, maxiter, t0, budget_s):
    def obj(x):
        return _smooth_max(_corr(x), alpha) + penalty * (np.sum(x) - target) ** 2
    def obj_grad(x):
        g = _grad_h(x, n, alpha)
        g += 2 * penalty * (np.sum(x) - target)
        return g
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _proj_grad(h, n, alpha, n_steps, lr0, t0, budget_s):
    h = h.copy()
    lr = lr0
    for _ in range(n_steps):
        if time.time() - t0 > budget_s:
            break
        c = _corr(h)
        cur_val = _smooth_max(c, alpha)
        g = _grad_h(h, n, alpha)
        step = lr
        accepted = False
        for _bt in range(30):
            h_new = h - step * g
            h_new = _repair(h_new, n)
            val = _smooth_max(_corr(h_new), alpha)
            if val < cur_val - 1e-16:
                h = h_new
                lr = min(lr * 1.5, 0.5)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr *= 0.7
            if lr < 1e-8:
                break
    return h

def _anneal(h, n, dx, n_iters, T0, T1, t0, budget_s, rng):
    h = h.copy()
    corr = _corr(h)
    cur_c5 = float(np.max(corr) * dx)
    for it in range(n_iters):
        if time.time() - t0 > budget_s:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iters))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_min > d_max - 1e-12:
            continue
        d = (d_max - d_min) * rng.rand() + d_min
        if abs(d) < 1e-12:
            continue
        gd = _g_vec(h, i, n) - _g_vec(h, j, n)
        new_corr = corr + d * gd
        new_c5 = float(np.max(new_corr) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            h[i] = hi + d
            h[j] = hj - d
            corr = new_corr
            cur_c5 = new_c5
    return h, cur_c5

    for it in range(n_iters):
        if time.time() > t_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iters))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_min > d_max - 1e-12:
            continue
        d = (d_max - d_min) * rng.rand() + d_min
        if abs(d) < 1e-12:
            continue
        gd = _g_vec(h, i, n) - _g_vec(h, j, n)
        new_corr = corr + d * gd
        new_c5 = float(np.max(new_corr) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-14)):
            h[i] = hi + d
            h[j] = hj - d
            corr = new_corr
            cur_c5 = new_c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Quick coordinate sweeps to polish start
    h = best_h.copy()
    corr = _corr(h)
    for _ in range(5):
        if time.time() - t0 > budget_s * 0.06:
            break
        h, corr, imp = _sweep(h, n, dx, corr, 15, 0, 0)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        if not imp:
            break

    # Phase 2: L-BFGS-B with penalty (fast, different from SLSQP)
    for alpha, pen in [(80, 5), (200, 50), (600, 200)]:
        if time.time() - t0 > budget_s * 0.2:
            break
        hc = _lfgsb(best_h, n, target, bounds, alpha, pen, 400)
        hc = _repair(hc, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 3: SLSQP with explicit equality constraint
    for alpha in [300, 800, 2000]:
        if time.time() - t0 > budget_s * 0.38:
            break
        hc = _slsqp(best_h, n, target, bounds, alpha, 800)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 4: Projected gradient (first-order, different character)
    if time.time() - t0 < budget_s * 0.5:
        hc = _proj_grad(best_h, n, 400, 500, 0.03, budget_s * 0.55 + t0 - t0)
        t_lim = t0 + budget_s * 0.55
        hc = _proj_grad(best_h, n, 400, 400, 0.03, t_lim)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 5: Coordinate sweeps coarse then fine
    h = best_h.copy()
    corr = _corr(h)
    for _ in range(30):
        if time.time() - t0 > budget_s * 0.62:
            break
        h, corr, imp = _sweep(h, n, dx, corr, 15, 0, 0)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        if not imp:
            break

    t_lim5 = t0 + budget_s * 0.72
    h = best_h.copy()
    corr = _corr(h)
    for _ in range(40):
        if time.time() > t_lim5:
            break
        h, corr, imp = _sweep(h, n, dx, corr, 21, 41, 0.015)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        if not imp:
            break

    # Phase 6: Simulated annealing on coordinate pairs
    t_lim6 = t0 + budget_s * 0.82
    h_a, c5_a = _anneal(best_h, n, dx, 80000, 0.003, 0.00005, t_lim6, rng)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a
    else:
        # Polish from anneal result anyway
        ha2 = _repair(h_a, n)
        ca2 = _corr(ha2)
        h_a, ca2, _ = _sweep(h_a, n, dx, ca2, 15, 21, 0.02)
        c5_a2 = float(np.max(ca2) * dx)
        if c5_a2 < best_c5:
            best_c5, best_h = c5_a2, h_a

    # Phase 7: Diverse random restarts (binary-inspired)
    for sr in range(5):
        t_lim7 = t0 + budget_s * 0.92
        if time.time() > t_lim7:
            break
        # Binary-like: pick random subset, blend with noise
        h_r = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h_r[idx] = 1.0
        h_r = 0.4 * h_r + 0.6 * rng.rand(n)
        h_r = _repair(h_r, n)
        hc = _slsqp(h_r, n, target, bounds, 500, 400)
        corr_r = _corr(hc)
        hc, corr_r, _ = _sweep(hc, n, dx, corr_r, 15, 21, 0.02)
        c5 = float(np.max(corr_r) * dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 8: Final ultra-fine sweep
    t_lim8 = t0 + budget_s * 0.97
    h = best_h.copy()
    corr = _corr(h)
    h, corr, _ = _sweep(h, n, dx, corr, 25, 51, 0.004)
    c5 = float(np.max(corr) * dx)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 9: Final SLSQP polish
    if time.time() < t0 + budget_s - 3:
        try:
            hc = _slsqp(best_h, n, target, bounds, 8000, 1000)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
