import numpy as np
from scipy.optimize import minimize
import time


def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    rem = abs(diff)
    for idx in order:
        if rem < 1e-14:
            break
        if diff > 0:
            a = min(rem, 1.0 - h[idx])
            h[idx] += a
        else:
            a = min(rem, h[idx])
            h[idx] -= a
        rem = abs(target - h.sum())
    return h


def _obj_lse(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    return (m + float(np.sum(np.exp(np.clip(alpha * (corr - m), -30, 30)))) / alpha) * dx


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _targeted_descent(h, dx, n, t_end, base_steps):
    """Pair descent prioritizing worst-shift reducing moves."""
    best = _c5(h, dx)
    scale = 1.0
    stall = 0
    while time.time() < t_end:
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_shifts = np.argsort(-corr)[:4]
        steps = base_steps * scale
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in steps:
                ni = vi + d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                ni = np.clip(ni, 0.0, 1.0)
                need = -(ni - vi)
                # Build candidate j list: prioritize j's in top shifts relative to i
                j_list = list(range(n))
                j_list.remove(i)
                # Score j by involvement in worst shifts
                scores = np.zeros(n)
                for ws in worst_shifts:
                    lag = ws - (n - 1)
                    if 0 <= i + lag < n:
                        scores[i + lag] += corr[ws]
                    if 0 <= i - lag < n:
                        scores[i - lag] += corr[ws]
                j_list.sort(key=lambda j: -scores[j])
                for j in j_list:
                    nj = h[j] + need
                    if nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    nj = np.clip(nj, 0.0, 1.0)
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-15:
                        best = c
                        improved = True
                        break
                    h[i], h[j] = vi, nj - need
                if improved:
                    break
        if not improved:
            stall += 1
            scale *= 0.6
            if scale < 0.01:
                break
            if stall > 6:
                break
        else:
            stall = 0
    return h, best


def _sa_continuous(h, dx, n, rng, t_end, T0=0.0015, Tend=5e-6):
    """SA with continuous pair moves."""
    cur_c5 = _c5(h, dx)
    best_c5 = cur_c5
    best_h = h.copy()
    it = 0
    while time.time() < t_end:
        it += 1
        T = T0 * (Tend / T0) ** min(it / 300000, 1.0)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.04, 0.04)
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        new_c5 = _c5(h, dx)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if rng.rand() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-12)):
                cur_c5 = new_c5
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5, cur_c5


def _binary_project(h, n):
    """Project to nearest binary with correct sum, using h as ranking."""
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def _bstart(n, rng):
    t = n // 2
    b = np.zeros(n)
    b[rng.choice(n, t, replace=False)] = 1.0
    return b
