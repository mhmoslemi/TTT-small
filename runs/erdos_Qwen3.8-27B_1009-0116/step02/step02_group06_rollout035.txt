import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_corr(h, k_star, n):
    """Gradient of c_{k_star} with respect to h"""
    lag = k_star - (n - 1)
    g = np.zeros(n)
    j = np.arange(n)
    jm = j - lag
    jp = j + lag

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    i1 = p - k
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = p + k
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        fr = (h > 1e-10) & (h < 1 - 1e-10)
        if fr.sum() > 0:
            h[fr] = np.clip(h[fr] + d / fr.sum(), 0, 1)
    return h

def _smx(c, a):
    m = float(np.max(c))
    return m + float(np.log(np.sum(np.exp(np.clip(a * (c - m), -500, 500))))) / a

def _obj(x, a):
    return _smx(_corr(x), a)

def _eqc(x, t):
    return float(np.sum(x)) - t

def _slsqp(h, n, t, bds, a, mi):
    try:
        r = minimize(_obj, h, args=(a,), method="SLSQP", bounds=bds,
                     constraints=[{"type": "eq", "fun": _eqc, "args": (t,)}],
                     options={"maxiter": mi, "ftol": 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

def _sweep(h, n, dx, cr, nc, nf, fw):
    imp = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            dl = max(-hi, hj - 1.0)
            dh = min(1.0 - hi, hj)
            if dl >= dh - 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dl, dh, nc)
            nv = cr[None, :] + ds[:, None] * gd[None, :]
            cv = np.max(nv, axis=1) * dx
            bi = int(np.argmin(cv))
            if cv[bi] < float(np.max(cr)) * dx - 1e-15:
                db = ds[bi]
                if nf > 2 and fw > 0:
                    lo2 = max(dl, db - fw)
                    hi2 = min(dh, db + fw)
                    ds2 = np.linspace(lo2, hi2, nf)
                    nv2 = cr[None, :] + ds2[:, None] * gd[None, :]
                    cv2 = np.max(nv2, axis=1) * dx
                    db = ds2[int(np.argmin(cv2))]
                h[i] = hi + db
                h[j] = hj - db
                cr = cr + db * gd
                imp = True
    return h, cr, imp

    x = 2.0 * (np.arange(n) + 0.5) / n

        h += rng.uniform(-0.3, 0.3) * np.sin(np.pi * m * x)
