Wait, I need to be more careful with the gradient. Let me re-derive.

c(k) for the full correlation np.correlate(h, 1-h, 'full'):
The k-th element (0-indexed) of the full output is:
c_full[k] = sum_i h[i + k - (n-1)] * (1-h)[i] for valid i

Actually, numpy's correlate documentation says:
np.correlate(a, v) = sum_n a[n+k] * conj(v[n])

For mode='full', k ranges from -(len(v)-1) to len(a)-1.
The output has length len(a)+len(v)-1 = 2n-1.
Output index k_out corresponds to lag = k_out - (len(v)-1) = k_out - (n-1).

So c_full[k_out] = sum_i a[i + k_out - (n-1)] * v[i] = sum_i h[i + lag] * (1-h)[i]
where lag = k_out - (n-1), and we sum over valid i (those where 0 ≤ i < n and 0 ≤ i+lag < n).

∂c_full[k_out]/∂h[j]:
= sum_i δ(i+lag = j) * (1-h)[i] + sum_i h[i] * ∂(1-h[i])/∂h[j]
= (1-h)[j-lag] * 1[0≤j-lag<n] - h[j-lag] * 1[0≤j-lag<n]

Wait, let me be more careful:
c = sum_i h[i+lag] * (1-h[i])

∂c/∂h[j] = sum_i [δ_{i+lag,j} * (1-h[i]) + h[i+lag] * (-δ_{i,j})]
= (1-h[j-lag]) * 1[i=j-lag is valid] - h[j-lag] * 1[i=j-lag is valid]

Wait: the first term: i+lag = j → i = j-lag. Valid if 0 ≤ j-lag < n. Value: (1-h[j-lag]).
The second term: i = j. Valid if 0 ≤ j < n (always) and 0 ≤ j+lag < n. Value: -h[j+lag].

So ∂c/∂h[j] = (1-h[j-lag]) * 1[0≤j-lag<n] - h[j+lag] * 1[0≤j+lag<n]

For the weighted sum:
g[j] = sum_{k_out} w[k_out] * [(1-h[j-lag_k]) * 1[0≤j-lag_k<n] - h[j+lag_k] * 1[0≤j+lag_k<n]]

where lag_k = k_out - (n-1).

Let me split into positive and negative lags:
For lag ≥ 0 (k_out ≥ n-1):
  g[j] += w[k_out] * [(1-h[j-lag]) * 1[0≤j-lag<n] - h[j+lag] * 1[0≤j+lag<n]]
  For j-lag ≥ 0: contributes (1-h[j-lag])
  For j+lag < n: contributes -h[j+lag]

For lag < 0 (k_out < n-1), let L = -lag > 0:
  g[j] += w[k_out] * [(1-h[j+L]) * 1[0≤j+L<n] - h[j-L] * 1[0≤j-L<n]]
  For j+L < n: contributes (1-h[j+L])
  For j-L ≥ 0: contributes -h[j-L]

For lag = 0:
  g[j] += w[n-1] * [(1-h[j]) - h[j]] = w[n-1] * (1-2h[j])

def _cd_pair(h, dx, target, n_passes, rng):

    for p in range(n_passes):

        for _ in range(n * 5):  # 5 passes over all pairs
            i = rng.randint(n)
            j = rng.randint(n)

            if lo >= hi:
                continue

            # Actually we want to minimize, so find d that minimizes c5
            best_d = 0.0
            best_c = bc
            # Coarse grid first
            for frac in [0.25, 0.5, 0.75]:

                h[j] -= d

                a = max(a, lo)
                b = min(b, hi)
                for _ in range(30):

                    h[i] -= m1
                    h[j] += m1
                    h[i] += m2
                    h[j] -= m2
                    c2 = _c5(h, dx)

                    h[j] += m2

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * x)
    for lag in range(1, n):
        wl = w[n-1+lag]
        wr = w[n-1-lag]
        if wl > 1e-14:
            g[lag:] += wl * (1.0 - x[:n-lag])
            g[:n-lag] -= wl * x[lag:]
        if wr > 1e-14:
            g[:n-lag] += wr * (1.0 - x[lag:])
            g[lag:] -= wr * x[:n-lag]
    return g

def _fista(h0, alpha, target, iters, t0, budget):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        if time.time() - t0 > budget - 10:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        # Backtracking
        for _ in range(5):
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_ternary(h, dx, n_passes, rng, t0, budget):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    for p in range(n_passes):
        if time.time() - t0 > budget - 15:
            break
        improved = 0
        for _ in range(n * 6):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-6:
                continue
            # Ternary search
            a, b = lo, hi
            for _ in range(25):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[i] += m1; h[j] -= m1
                c1 = _c5(h, dx)
                h[i] -= m1; h[j] += m1
                h[i] += m2; h[j] -= m2
                c2 = _c5(h, dx)
                h[i] -= m2; h[j] += m2
                if c1 < c2:
                    b = m2
                else:
                    a = m1
            d_opt = 0.5 * (a + b)
            if d_opt > lo + 1e-8 and d_opt < hi - 1e-8:
                h[i] += d_opt
                h[j] -= d_opt
                c_new = _c5(h, dx)
                if c_new < bc - 1e-10:
                    bc = c_new
                    improved += 1
            elif d_opt <= lo + 1e-8 or d_opt >= hi - 1e-8:
                # Check boundary
                d_try = max(lo + 1e-9, min(hi - 1e-9, d_opt))
                h[i] += d_try; h[j] -= d_try
                c_new = _c5(h, dx)
                h[i] -= d_try; h[j] += d_try
                if c_new < bc - 1e-10:
                    h[i] += d_try; h[j] -= d_try
                    bc = c_new
                    improved += 1
        if improved == 0:
            break
    return h, bc

def _subgrad_step(h, target, dx, step, t0, budget):
    """One subgradient step on the actual max."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    # Active set: lags within 0.001 of max
    active = np.where(c >= cmax - 0.002)[0]
    if len(active) == 0:
        return h, cmax
    # Compute subgradient (average gradient of active lags)
    w = np.zeros(2*n-1)
    w[active] = 1.0 / len(active)
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * h)
    for lag in range(1, n):
        wl = w[n-1+lag]
        wr = w[n-1-lag]
        if wl != 0:
            g[lag:] += wl * (1.0 - h[:n-lag])
            g[:n-lag] -= wl * h[lag:]
        if wr != 0:
            g[:n-lag] += wr * (1.0 - h[lag:])
            g[lag:] -= wr * h[:n-lag]
    # Take step
    h_new = h - step * g
    h_new = _proj(h_new, target)
    c_new = _c5(h_new, dx)
    if c_new < cmax:
        return h_new, c_new
    # Try smaller step
    h_new = h - 0.5 * step * g
    h_new = _proj(h_new, target)
    c_new = _c5(h_new, dx)
    if c_new < cmax:
        return h_new, c_new
    return h, cmax

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * x)
    for lag in range(1, n):
        wl = w[n - 1 + lag]
        wr = w[n - 1 - lag]
        if wl > 1e-14:
            g[lag:] += wl * (1.0 - x[:n - lag])
            g[:n - lag] -= wl * x[lag:]
        if wr > 1e-14:
            g[:n - lag] += wr * (1.0 - x[lag:])
            g[lag:] -= wr * x[:n - lag]
    return g

def _fista(h0, alpha, target, iters, t0, budget):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        if time.time() - t0 > budget - 8:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        for _ in range(5):
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_ternary(h, dx, n_passes, rng, t0, budget):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    for p in range(n_passes):
        if time.time() - t0 > budget - 12:
            break
        improved = 0
        for _ in range(n * 6):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-7:
                continue
            a, b = lo, hi
            for _ in range(28):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[i] += m1
                h[j] -= m1
                c1 = _c5(h, dx)
                h[i] -= m1
                h[j] += m1
                h[i] += m2
                h[j] -= m2
                c2 = _c5(h, dx)
                h[i] -= m2
                h[j] += m2
                if c1 < c2:
                    b = m2
                else:
                    a = m1
            d_opt = 0.5 * (a + b)
            d_opt = max(lo + 1e-9, min(hi - 1e-9, d_opt))
            h[i] += d_opt
            h[j] -= d_opt
            c_new = _c5(h, dx)
            if c_new < bc - 1e-11:
                bc = c_new
                improved += 1
            else:
                h[i] -= d_opt
                h[j] += d_opt
                c_check = _c5(h, dx)
                if c_check < bc - 1e-11:
                    bc = c_check
                    improved += 1
        if improved == 0:
            break
    return h, bc

def _subgrad_step(h, target, dx, step_sz):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    active = np.where(c >= cmax - 0.003)[0]
    if len(active) == 0:
        return h, cmax
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * h)
    for lag in range(1, n):
        wl = w[n - 1 + lag]
        wr = w[n - 1 - lag]
        if wl != 0.0:
            g[lag:] += wl * (1.0 - h[:n - lag])
            g[:n - lag] -= wl * h[lag:]
        if wr != 0.0:
            g[:n - lag] += wr * (1.0 - h[lag:])
            g[lag:] -= wr * h[:n - lag]
    for frac in [1.0, 0.5, 0.25, 0.1]:
        h_new = _proj(h - frac * step_sz * g, target)
        c_new = _c5(h_new, dx)
        if c_new < cmax:
            return h_new, c_new
    return h, cmax

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target
