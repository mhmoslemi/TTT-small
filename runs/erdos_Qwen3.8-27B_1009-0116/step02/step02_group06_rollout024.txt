def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _optimal_d(c, g, d_lo, d_hi):
    if d_hi - d_lo < 1e-15:
        return 0.0
    f0 = c.max()
    act0 = c >= f0 - 1e-12
    if act0.any() and g[act0].min() <= 0 <= g[act0].max():
        return 0.0
    lo, hi = d_lo, d_hi
    for _ in range(48):
        if hi - lo < 1e-13:
            break
        mid = 0.5 * (lo + hi)
        vals = c + mid * g
        fm = vals.max()
        act = vals >= fm - 1e-11
        if not act.any():
            act = (vals == fm)
        gmin = g[act].min()
        gmax = g[act].max()
        if gmin <= 0 <= gmax:
            return mid
        elif gmin > 0:
            hi = mid
        else:
            lo = mid
    best_d, best_f = d_lo, np.max(c + d_lo * g)
    for d in (d_hi, lo, hi, 0.5 * (lo + hi)):
        fd = np.max(c + d * g)
        if fd < best_f:
            best_f, best_d = fd, d
    return best_d

def _sweep(h, n, dx, corr, t_end, max_passes=400):
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        pair_imp = False
        for i in range(n):
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-13:
                    continue
                gd = _g_vec(h, i, n) - _g_vec(h, j, n)
                d_opt = _optimal_d(corr, gd, d_lo, d_hi)
                if abs(d_opt) < 1e-13:
                    continue
                new_max = np.max(corr + d_opt * gd)
                cur_max = np.max(corr)
                if new_max < cur_max - 1e-14:
                    h[i] = np.clip(hi + d_opt, 0.0, 1.0)
                    h[j] = np.clip(hj - d_opt, 0.0, 1.0)
                    corr = corr + d_opt * gd
                    hi = h[i]
                    pair_imp = True
        if not pair_imp:
            break
    return h, corr

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _pnorm_obj(x, p):
    c = _corr(x)
    return float(np.sum(c ** p) ** (1.0 / p))

def _pnorm_slsq(h, n, target, bounds, p, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_pnorm_obj, h, args=(p,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up
    for alpha in [20, 60, 150, 300, 600]:
        if time.time() - t0 > budget_s - 600:
            break
        hc = _slsqp(h, n, target, bounds, alpha, 400)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
            h = hc

    # Phase 2: p-norm SLSQP
    for p in [40, 80, 160]:
        if time.time() - t0 > budget_s - 500:
            break
        hc = _pnorm_slsq(h, n, target, bounds, p, 300)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
            h = hc

    # Phase 3: Long coordinate sweep (main work)
    t_sweep_end = t0 + budget_s - 200
    h_s = best_h.copy()
    corr_s = _corr(h_s)
    _sweep(h_s, n, dx, corr_s, t_sweep_end, max_passes=600)
    c5 = _c5(h_s, dx)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_s.copy()

    # Phase 4: Random perturbations + local optimize
    for trial in range(10):
        if time.time() - t0 > budget_s - 100:
            break
        amp = 0.04 + 0.02 * trial
        h_p = best_h + amp * rng.randn(n)
        h_p = _repair(h_p, n)
        if trial % 2 == 0:
            hc = _slsqp(h_p, n, target, bounds, 150, 200)
        else:
            hc = _pnorm_slsq(h_p, n, target, bounds, 60, 200)
        corr_p = _corr(hc)
        t_end_p = min(t0 + budget_s - 80, time.time() + 30)
        hc, corr_p = _sweep(hc, n, dx, corr_p, t_end_p, max_passes=100)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 5: Final polish sweep
    if time.time() - t0 < budget_s - 8:
        h_f = best_h.copy()
        corr_f = _corr(h_f)
        _sweep(h_f, n, dx, corr_f, t0 + budget_s - 5, max_passes=100)
        c5 = _c5(h_f, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 4:
        try:
            hc = _slsqp(best_h, n, target, bounds, 800, 300)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
