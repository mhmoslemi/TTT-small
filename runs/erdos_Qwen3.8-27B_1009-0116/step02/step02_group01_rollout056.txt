def repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return c

    return np.max(compute_c(h, len(h))) * dx

def grad_c_k(h, n, k):

        if 0 <= jm < n:

        jp = j + m
        if 0 <= jp < n:
            g[j] -= h[jp]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_val(h, dx):
    return np.max(compute_c(h, len(h))) * dx


def grad_c_k(h, n, k):
    m = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        jm = j - m
        if 0 <= jm < n:
            g[j] += (1.0 - h[jm])
        jp = j + m
        if 0 <= jp < n:
            g[j] -= h[jp]
    return g


def grad_smooth_max(h, n, alpha):
    c = compute_c(h, n)
    cm = np.max(c)
    s = alpha * (c - cm)
    s = np.clip(s, -500, 500)
    p = np.exp(s)
    p /= p.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w > 1e-14:
            g += w * grad_c_k(h, n, k)
    return g, cm + np.log(p.sum()) / alpha


def lhs_sum_constraint(x):
    return np.sum(x) - len(x) / 2.0


def smooth_obj_fn(x, alpha, n):
    c = compute_c(x, n)
    cm = np.max(c)
    s = alpha * (c - cm)
    s = np.clip(s, -500, 500)
    return cm + np.log(np.sum(np.exp(s))) / alpha


def solve_lp_relax(n, max_time=45):
    t_start = time.time()
    h_var = cp.Variable(n)
    Y = cp.Variable((n, n))
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for i in range(n):
        for j in range(n):
            cons.append(Y[i, j] <= h_var[i])
            cons.append(Y[i, j] <= h_var[j])
            cons.append(Y[i, j] >= h_var[i] + h_var[j] - 1)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        vi = [i for i in range(n) if 0 <= i + m < n]
        terms_h = [h_var[i + m] for i in vi]
        terms_y = [Y[i + m, i] for i in vi]
        cons.append(cp.sum(terms_h) - cp.sum(terms_y) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solvers = [cp.GLOP, cp.CBC, cp.SCIP, cp.GLPK]
    h_sol = None
    for solver in solvers:
        if time.time() - t_start > max_time:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if h_var.value is not None:
                h_sol = np.array(h_var.value, dtype=float)
                break
        except Exception:
            continue
    if h_sol is None:
        return None, None
    return h_sol, prob.value
