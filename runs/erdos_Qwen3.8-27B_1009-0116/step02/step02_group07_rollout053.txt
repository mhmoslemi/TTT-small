        room = 1.0 - h
        add = target - s
        scale = add / room.sum() if room.sum() > 0 else 0

        diff = target - h_new.sum()
        if abs(diff) > 1e-10:

def _project_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1}."""
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    # clip(h-tau, 0, 1) = h-tau for h-tau in (0,1), = 0 for h-tau<=0, = 1 for h-tau>=1
    # sum = sum_{h_i>tau} (h_i - tau) + count_{h_i > tau+1} * 1 + ...
    # Actually: for each coordinate:
    #   if h_i - tau <= 0: contributes 0
    #   if h_i - tau >= 1: contributes 1
    #   if 0 < h_i - tau < 1: contributes h_i - tau
    # Let lo = min(0, h.min()-1) and hi = max(1, h.max())...

        if s > target:

def _smooth_grad(x, dx, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ex = np.exp(np.clip(alpha * (c - m), -500, 500))
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        p_lo1 = max(0, s)
        p_hi1 = min(n - 1, n - 1 + s)
        if p_lo1 <= p_hi1:
            idx = np.arange(p_lo1, p_hi1 + 1)
            grad[idx] += wk * (1.0 - x[idx - s])
        p_lo2 = max(0, -s)
        p_hi2 = min(n - 1, n - 1 - s)
        if p_lo2 <= p_hi2:
            idx = np.arange(p_lo2, p_hi2 + 1)
            grad[idx] -= wk * x[idx + s]
    return dx * grad

def _smooth_fg(x, dx, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ex = np.exp(np.clip(alpha * (c - m), -500, 500))
    w = ex / ex.sum()
    f = m + np.log(ex.sum()) / alpha
    n = len(x)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        p_lo1 = max(0, s)
        p_hi1 = min(n - 1, n - 1 + s)
        if p_lo1 <= p_hi1:
            idx = np.arange(p_lo1, p_hi1 + 1)
            grad[idx] += wk * (1.0 - x[idx - s])
        p_lo2 = max(0, -s)
        p_hi2 = min(n - 1, n - 1 - s)
        if p_lo2 <= p_hi2:
            idx = np.arange(p_lo2, p_hi2 + 1)
            grad[idx] -= wk * x[idx + s]
    grad *= dx
    return f, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _cd_pass_fast(h, n, dx, n_grid):
    hi_c5 = _c5(h, dx)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            orig_i = h[i]
            orig_j = h[j]
            best_t = orig_i
            best_c = hi_c5
            ts = np.linspace(lo, hi, n_grid)
            for t in ts:
                h[i] = t
                h[j] = hsum - t
                c = _c5(h, dx)
                if c < best_c:
                    best_c = c
                    best_t = t
            if best_c < hi_c5 - 1e-12:
                h[i] = best_t
                h[j] = hsum - best_t
                hi_c5 = best_c
                improved = True
            else:
                h[i] = orig_i
                h[j] = orig_j
    return h, hi_c5, improved

def _sum_eq(x, target):
    return x.sum() - target
