    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 1.0 / 2.0)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _pair_opt(h, i, j, n, dx, n_eval=15):
    """Optimize pair (i,j) given all else fixed."""
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h[i], _c5(h, dx)
    
    best_t = h[i]
    best_f = _c5(h, dx)
    # Evaluate at multiple points
    for t in np.linspace(lo, hi, n_eval):
        if abs(t - best_t) < 1e-12:
            continue
        h[i] = t
        h[j] = s - t
        f = _c5(h, dx)
        if f < best_f:
            best_f = f
            best_t = t
    h[i] = best_t
    h[j] = s - best_t
    return best_t, best_f

                if time.time() >= deadline:
                    return h, best
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                best_t = h[i]
                best_f = best
                for t in np.linspace(lo, hi, n_eval):
                    if abs(t - best_t) < 1e-10:
                        continue
                    h[i] = t
                    h[j] = s - t
                    f = _c5(h, dx)
                    if f < best_f:
                        best_f = f
                        best_t = t
                h[i] = best_t
                h[j] = s - best_t
                if best_f < best - 1e-13:
                    best = best_f
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 2:
                break
        else:
            no_improve = 0
    return h, best

def _local_search(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for scale in scales:
        stalled = 0
        while time.time() < deadline and stalled < 8000:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                stalled += 1
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best - 1e-14:
                best = c5
                stalled = 0
            else:
                h[i] -= d
                h[j] += d
                stalled += 1
    return h, best

def _triple_sweep(h, n, dx, deadline, n_eval=5):
    """Optimize triples of variables for critical interactions."""
    best = _c5(h, dx)
    h = h.copy()
    for i in range(0, n, 3):
        if time.time() >= deadline:
            break
        for j in range(i + 1, min(i + 10, n)):
            for k in range(j + 1, min(j + 10, n)):
                if time.time() >= deadline:
                    return h, best
                s = h[i] + h[j] + h[k]
                # Grid over (h[i], h[j]) with h[k] = s - h[i] - h[j]
                best_f = best
                for ti in np.linspace(max(0, s - 2), min(1, s), n_eval):
                    for tj in np.linspace(max(0, s - ti - 1), min(1, s - ti), n_eval):
                        tk = s - ti - tj
                        if tk < -1e-10 or tk > 1 + 1e-10:
                            continue
                        if abs(ti - h[i]) < 1e-10 and abs(tj - h[j]) < 1e-10:
                            continue
                        h[i], h[j], h[k] = ti, tj, tk
                        f = _c5(h, dx)
                        if f < best_f:
                            best_f = f
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    end_time = t_start + budget_s

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)

    best_overall_h = None
    best_overall_c5 = 1e9
    best_overall_n = len(h_init)

    # === Main optimization at n=70 ===
    n = len(h_init)
    dx = 2.0 / n
    h = _repair(h_init.copy(), n)
    c5 = _c5(h, dx)

    # Phase 1: SLSQP warmup (40s)
    t_end = t_start + 40
    for alpha in [30, 80, 200, 500]:
        if time.time() >= t_end:
            break
        h, c5 = _slsqp_opt(h, n, dx, alpha, 500, t_end)

    # Phase 2: Pair sweep (220s)
    t_end = t_start + 260
    h, c5 = _pair_sweep(h, n, dx, t_end, n_eval=13)

    # Phase 3: SLSQP refinement (80s)
    t_end = t_start + 340
    for alpha in [200, 500, 1000, 2000, 4000]:
        if time.time() >= t_end:
            break
        h, c5 = _slsqp_opt(h, n, dx, alpha, 700, t_end)

    # Phase 4: Detailed pair sweep (180s)
    t_end = t_start + 520
    h, c5 = _pair_sweep(h, n, dx, t_end, n_eval=17)

    # Phase 5: Local search (100s)
    t_end = t_start + 620
    h, c5 = _local_search(h, n, dx, rng, t_end)

    # Phase 6: SLSQP + triple + pair (120s)
    t_end = t_start + 740
    for alpha in [1000, 3000, 6000]:
        if time.time() >= t_end:
            break
        h, c5 = _slsqp_opt(h, n, dx, alpha, 1000, t_end)
    if time.time() < t_end:
        h, c5 = _pair_sweep(h, n, dx, min(time.time() + 40, t_end), n_eval=21)

    # Phase 7: Final intensive local search + pair (100s)
    t_end = min(t_start + 840, end_time - 120)
    h, c5 = _local_search(h, n, dx, rng, t_end)
    if time.time() < end_time - 30:
        h, c5 = _pair_sweep(h, n, dx, min(time.time() + 30, end_time - 20), n_eval=19)

    best_overall_h = _repair(h, n)
    best_overall_c5 = _c5(best_overall_h, dx)
    best_overall_n = n

    # === Try n=80 if time permits ===
    if time.time() < end_time - 130:
        n2 = 80
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h2 = np.interp(x_new, x_old, h_init)
        h2 = _repair(h2, n2)
        c52 = _c5(h2, dx2)
        t_end2 = min(end_time - 20, time.time() + 110)
        for alpha in [100, 400, 1000]:
            if time.time() >= t_end2:
                break
            h2, c52 = _slsqp_opt(h2, n2, dx2, alpha, 600, t_end2)
        if time.time() < t_end2:
            h2, c52 = _pair_sweep(h2, n2, dx2, t_end2, n_eval=11)
        if c52 < best_overall_c5:
            best_overall_c5 = c52
            best_overall_h = _repair(h2, n2)
            best_overall_n = n2

    # === Try n=60 as alternative ===
    if time.time() < end_time - 80:
        n3 = 60
        dx3 = 2.0 / n3
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n3, endpoint=False) + 1.0 / n3
        h3 = np.interp(x_new, x_old, h_init)
        h3 = _repair(h3, n3)
        c53 = _c5(h3, dx3)
        t_end3 = min(end_time - 10, time.time() + 60)
        for alpha in [200, 800]:
            if time.time() >= t_end3:
                break
            h3, c53 = _slsqp_opt(h3, n3, dx3, alpha, 500, t_end3)
        if time.time() < t_end3:
            h3, c53 = _pair_sweep(h3, n3, dx3, t_end3, n_eval=13)
        if c53 < best_overall_c5:
            best_overall_c5 = c53
            best_overall_h = _repair(h3, n3)
            best_overall_n = n3

    best_overall_h = _repair(best_overall_h, best_overall_n)
    best_overall_c5 = _c5(best_overall_h, 2.0 / best_overall_n)
    return (best_overall_h, best_overall_c5, best_overall_n)
