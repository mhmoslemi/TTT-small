def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _obj(x, n):
    return float(np.max(_corr(x))) * (2.0 / n)

def _obj_lse(x, alpha, n):
    c = _corr(x)
    m = c.max()
    return float((m + np.exp(np.clip(alpha*(c-m), -50, 50)).sum()/alpha) * (2.0/n))

def _obj_topk(x, k, n):
    c = _corr(x)
    idx = np.argpartition(c, -k)[-k:]
    return float(c[idx].sum() * (2.0/n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1-h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
