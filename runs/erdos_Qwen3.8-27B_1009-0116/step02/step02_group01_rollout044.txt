import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _proj(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h


def _subgrad_topm(h, n, m=3):
    c = _corr(h)
    idx = np.argsort(c)[::-1][:m]
    Lc = n - 1
    g = np.zeros(n)
    for ki in idx:
        L = ki - Lc
        if L >= 0:
            for j in range(L, n):
                g[j] += (1.0 - h[j - L])
            for j in range(0, n - L):
                g[j] -= h[j + L]
        else:
            Lp = -L
            for j in range(0, n - Lp):
                g[j] += (1.0 - h[j + Lp])
            for j in range(Lp, n):
                g[j] -= h[j - Lp]
    return g / m, c


def _line_search(h, g, n, dx, c5_cur):
    best_h = h.copy()
    best_val = c5_cur
    step = 0.05
    for _ in range(25):
        h_try = _proj(h - step * g, n)
        v = _c5(h_try, dx)
        if v < best_val - 1e-14:
            best_val = v
            best_h = h_try
            step = min(step * 1.6, 2.0)
        else:
            step *= 0.45
            if step < 1e-7:
                break
    return best_h, best_val


def _smooth_obj(x, n, dx, alpha):
    c = _corr(x) * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _slsqp(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return _smooth_obj(x, n, dx, alpha)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _proj(res.x, n)
    except Exception:
        return h.copy()


def _resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)
