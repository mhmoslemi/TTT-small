    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()

        return h

    rem = diff

            res = minimize(_lse_func, best_h, args=(alpha, n), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Subgradient projection (200s)
    p2_end = t0 + 260
    if time.time() < p2_end:
        h = best_h.copy()
        p2_best = best_c5
        p2_best_h = best_h.copy()
        n_iter = 80000
        for it in range(n_iter):
            if time.time() > p2_end:
                break
            frac = it / n_iter
            step = 0.015 * (1.0 - 0.8 * frac)
            tol = 1.0 - 0.5 * frac
            grad = _grad_smooth(h, n, tol)
            h = h - step * grad
            h = h + (target - h.sum()) / n
            h = np.clip(h, 0.0, 1.0)
            h = _repair(h, n)
            c5_try = _c5(h, dx)
            if c5_try < p2_best - 1e-13:
                p2_best = c5_try
                p2_best_h = h.copy()
        if p2_best < best_c5:
            best_c5, best_h = p2_best, p2_best_h

    # Phase 3: Binary structure search (200s)
    p3_end = t0 + 460
    if time.time() < p3_end:
        h_bin = _round_binary(best_h, n)
        for _ in range(30):
            if time.time() > p3_end - 5:
                break
            h_bin, _ = _binary_sweep(h_bin, n, max_pairs=200)
        # Interpolate: blend binary with continuous
        for w in [1.0, 0.8, 0.6]:
            if time.time() > p3_end - 3:
                break
            h_blend = w * h_bin + (1.0 - w) * best_h
            h_blend = _repair(h_blend, n)
            c5_try = _c5(h_blend, dx)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_blend

    # Phase 4: Coordinate descent (350s)
    p4_end = t0 + 810
    if time.time() < p4_end:
        h = best_h.copy()
        loc_best = best_c5
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, -0.01, -0.005, -0.002, -0.001, -0.0005]
        sweeps = 0
        stagnate = 0
        while time.time() < p4_end:
            sweeps += 1
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time.time() > p4_end:
                    break
                vi = h[i]
                for d in deltas:
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    if need > 0:
                        oji = np.argsort(h)[:25]
                    else:
                        oji = np.argsort(-h)[:25]
                    h[i] = ni
                    found = False
                    for j in oji:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0 or nj > 1:
                            continue
                        h[j] = nj
                        c5_try = _c5(h, dx)
                        if c5_try < loc_best - 1e-13:
                            loc_best = c5_try
                            improved = True
                            found = True
                            break
                        h[j] -= need
                    if not found:
                        h[i] = vi
            if improved:
                stagnate = 0
            else:
                stagnate += 1
                if stagnate >= 4:
                    break
        if loc_best < best_c5:
            best_c5, best_h = loc_best, h

    # Phase 5: Fine polish + final SLSQP
    p5_end = t0 + budget_s - 10
    if time.time() < p5_end:
        h = best_h.copy()
        loc_best = best_c5
        fd = [0.0003, 0.0006, 0.001, -0.0003, -0.0006, -0.001]
        for _ in range(30):
            if time.time() > p5_end - 15:
                break
            imp = False
            for i in rng.permutation(n):
                if time.time() > p5_end - 15:
                    break
                vi = h[i]
                for d in fd:
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    oji = np.argsort(h)[:12] if need > 0 else np.argsort(-h)[:12]
                    h[i] = ni
                    for j in oji:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0 or nj > 1:
                            continue
                        h[j] = nj
                        c5_try = _c5(h, dx)
                        if c5_try < loc_best - 1e-14:
                            loc_best = c5_try
                            imp = True
                            break
                        h[j] -= need
                    else:
                        h[i] = vi
                    if imp:
                        break
            if not imp:
                break
        if loc_best < best_c5:
            best_c5, best_h = loc_best, h

    if time.time() < t0 + budget_s - 5:
        try:
            res = minimize(_lse_func, best_h, args=(2000, n), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
