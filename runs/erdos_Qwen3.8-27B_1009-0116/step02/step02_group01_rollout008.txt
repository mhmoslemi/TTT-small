    def _repair(h, n):

        target = n / 2.0
        s = h.sum()
        if abs(s - target) < 1e-14:
            return h
        if s < 1e-14:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-14:
                    break
        return h

    def _corr_c(h, dx):
        return np.correlate(h, 1.0 - h, mode='full') * dx

    def _c5(h, dx):
        return float(np.max(_corr_c(h, dx)))

    def _smooth_max_obj(h, dx, alpha):
        c = _corr_c(h, dx)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -500, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def _smooth_max_grad(h, dx, alpha):
        n = len(h)
        c = _corr_c(h, dx)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -500, 0)
        e = np.exp(s)
        e_sum = e.sum()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr_c(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, dx):
    return float(np.max(_corr_c(h, dx)))

def _smooth_obj(h, dx, alpha):
    c = _corr_c(h, dx)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, dx, alpha):
    n = len(h)
    c = _corr_c(h, dx)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 0)
    e = np.exp(s)
    e_sum = e.sum()
    if e_sum < 1e-300:
        return np.zeros(n)
    w = e / e_sum
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-12:
            continue
        lag = p - (n - 1)
        wc = w[p] * dx
        if lag >= 0:
            grad[:n - lag] += wc * (1.0 - h[lag:])
            grad[lag:] -= wc * h[:n - lag]
        else:
            L = -lag
            grad[L:] += wc * (1.0 - h[:n - L])
            grad[:n - L] -= wc * h[L:]
    return grad

def _lag_grad(h, dx, p):
    n = len(h)
    lag = p - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += dx * (1.0 - h[lag:])
        grad[lag:] -= dx * h[:n - lag]
    else:
        L = -lag
        grad[L:] += dx * (1.0 - h[:n - L])
        grad[:n - L] -= dx * h[L:]
    return grad

def _proj_grad_step(h, dx, alpha, step, n):
    g = _smooth_grad(h, dx, alpha)
    g_centered = g - g.mean()
    h_new = h - step * g_centered
    h_new = _repair(h_new, n)
    return h_new

        return np.sum(x) - target

                           options={'maxiter': 400, 'ftol': 1e-16})
            hc = _repair(res.x, n)
