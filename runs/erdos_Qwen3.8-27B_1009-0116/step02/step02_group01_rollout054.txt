        else:

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    tgt = 0.5 * n
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _smax(c, a):
    m = c.max()
    z = np.clip(a * (c - m), -600, 600)
    e = np.exp(z)
    return m + np.log(e.sum()) / a, e / e.sum()

def _grad(h, q, n, dx):
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = q[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        for j in range(n):
            p = j - L
            if 0 <= p < n:
                g[j] += w * (1.0 - h[p])
            p2 = j + L
            if 0 <= p2 < n:
                g[j] -= w * h[p2]
    return g * dx

def _pgd(h0, n, dx, alpha, n_iter, lr_init, mu, t0, t_end):
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best = _c5(h, dx)
    bh = h.copy()
    c = _corr(h)
    fm_old, _ = _smax(c, alpha)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        _, q = _smax(c, alpha)
        g = _grad(h, q, n, dx)
        v = mu * v + (1.0 - mu) * g
        accepted = False
        for _bs in range(10):
            if time.time() > t_end:
                return bh, best
            h_new = _repair(h - lr * v, n)
            c_new = _corr(h_new)
            fm_new, _ = _smax(c_new, alpha)
            if fm_new <= fm_old * 1.001 + 1e-15:
                h = h_new
                c = c_new
                fm_old = fm_new
                lr = min(lr * 1.1, lr_init * 3.0)
                accepted = True
                break
            lr *= 0.6
        if not accepted:
            break
        c5c = _c5(h, dx)
        if c5c < best - 1e-12:
            best = c5c
            bh = h.copy()
    return bh, best

def _try_binary(h, n, dx):
    hb = (h > 0.5).astype(float)
    hb = _repair(hb, n)
    return hb, _c5(hb, dx)

def _make_h(n, rng, mode):
    if mode == 0:
        x = np.linspace(0, 1, n)
        h = 1.0 / (1.0 + np.exp(10.0 * (x - 0.5)))
    elif mode == 1:
        h = 0.5 + 0.45 * np.sin(2 * np.pi * 2.0 * np.arange(n, dtype=float) / n)
    elif mode == 2:
        h = np.zeros(n)
        h[:n // 3] = 1.0
        h[2 * n // 3:] = 1.0
    elif mode == 3:
        h = np.zeros(n)
        h[::2] = 1.0
    else:
        h = rng.rand(n)
    return _repair(h, n)

def _slsqp_polish(h0, n, dx, t0, t_end):
    tgt = 0.5 * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        z = np.clip(5000.0 * (c - m), -600, 600)
        return (m + np.log(np.exp(z).sum()) / 5000.0) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt}]
    bounds = [(0.0, 1.0)] *
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
    return _repair(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_end = t0 + budget_s - 15

    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Determine n options
    n_start = None
    hv = None
    try:
        hv = np.array(initial_h_values, dtype=float)
        n_start = len(hv)
    except Exception:
        pass

    n_opts = []
    if n_start and 20 <= n_start <= 200:
        n_opts.append(n_start)
    for nn in [70, 90, 100, 50, 120, 140]:
        if nn not in n_opts and 20 <= nn <= 200:
            n_opts.append(nn)

    for n in n_opts:
        if time.time() > t_end:
            break
        dx = 2.0 / n
        rem = t_end - time.time()
        if rem < 20:
            break

        # Initial point
        if n == n_start and hv is not None:
            h0 = _repair(hv.copy(), n)
        else:
            h0 = _make_h(n, rng, rng.randint(5))

        c5_init = _c5(h0, dx)
        h_cur = h0.copy()

        # Multi-alpha Nesterov
        alphas = [30, 80, 200, 500, 1500, 5000, 20000]
        for alpha in alphas:
            if time.time() > t_end - 20:
                break
            rem2 = t_end - time.time()
            n_iter = max(100, int(rem2 * 300))
            lr_init = 0.5 / (1.0 + alpha * dx * 0.1)
            h_cur, c5c = _pgd(h_cur, n, dx, alpha, n_iter, lr_init, 0.92, t0, t_end)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_cur.copy()
                best_n = n
            # Binary rounding attempt
            hb, c5b = _try_binary(h_cur, n, dx)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = hb
                best_n = n
                h_cur = hb

        # SLSQP polish
        if time.time() < t_end - 25:
            h_ref = h_cur.copy() if best_n != n else best_h.copy()
            try:
                hc = _slsqp_polish(h_ref, n, dx, t0, t_end)
                c5s = _c5(hc, dx)
                if c5s < best_c5:
                    best_c5 = c5s
                    best_h = hc
                    best_n = n
            except Exception:
                pass

    # Final: a bit more Nesterov on the overall best
    if best_h is not None:
        n_f = best_n
        dx_f = 2.0 / n_f
        if time.time() < t_end - 5:
            best_h, _ = _pgd(best_h, n_f, dx_f, 50000, 200, 0.005, 0.95, t0, t_end)
        best_h = _repair(best_h, n_f)
        best_c5 = _c5(best_h, dx_f)

    if best_h is None:
        n_f = 70 if n_start is None else n_start
        dx_f = 2.0 / n_f
        if hv is not None and n_start == n_f:
            best_h = _repair(hv.copy(), n_f)
        else:
            best_h = _repair(np.full(n_f, 0.5), n_f)
        best_c5 = _c5(best_h, dx_f)

    return (best_h, best_c5, len(best_h))
