                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break
    return best, best_c5

def _sim_anneal(h, n, dx, rng, deadline, T0, Tc):
    best = h.copy()
    cur = h.copy()
    cur_c5 = _c5(cur, dx)
    best_c5 = cur_c5
    T = T0
    cool = 0.998
    steps = np.array([0.02, 0.04, 0.06, 0.01, 0.08, -0.02, -0.04, -0.06, -0.01, -0.08])
    while time.time() < deadline:
        if T < Tc:
            T = T0
            cur = best.copy()
            cur_c5 = best_c5
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            T *= cool
            continue
        d = steps[int(rng.randint(len(steps)))]
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            T *= cool
            continue
        cur[i], cur[j] = ni, nj
        nc = _c5(cur, dx)
        delta = nc - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[i] - d, cur[j] + d
        T *= cool
    return best, best_c5

def _targeted_peak(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    small_steps = [0.01, 0.02, 0.03, -0.01, -0.02, -0.03]
    while time.time() < deadline:
        corr = _corr(best)
        max_idx = np.argmax(corr)
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for d in small_steps:
                ni = best[i] + d
                if ni < 0 or ni > 1:
                    continue
                j = int(rng.randint(n))
                if j == i:
                    continue
                nj = best[j] - d
                if nj < 0 or nj > 1:
                    continue
                old_i, old_j = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                    break
                else:
                    best[i], best[j] = old_i, old_j
            if improved:
                break
        if not improved:
            k = int(rng.randint(n))
            best[k] = min(1.0, max(0.0, best[k] + rng.uniform(-0.15, 0.15)))
            best = _repair(best, n)
    return best, best_c5

def _diverse_init(n, rng, mode):
    t = np.linspace(0, 2.0, n, endpoint=False)
    if mode == 0:
        h = np.full(n, 0.5) + 0.3 * rng.randn(n)
    elif mode == 1:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * t * 3) + 0.1 * rng.randn(n)
    elif mode == 2:
        h = np.where(np.arange(n) % 2 == 0, 0.7, 0.3) + 0.1 * rng.randn(n)
    elif mode == 3:
        h = 0.5 + 0.2 * np.cos(4 * np.pi * t) + 0.15 * np.sin(10 * np.pi * t) + 0.05 * rng.randn(n)
    else:
        h = rng.rand(n) * 0.6 + 0.2
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)

    # Phase 1: SLSQP from initial with alpha sweep
    for a in [40, 80, 160, 320, 640, 1280]:
        if time.time() > t0 + budget_s - 250:
            break
        hc = _slsqp2(best_h, n, dx, a, 400, 1e-15)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: Greedy local search
    if time.time() < t0 + budget_s - 180:
        deadline_ls = t0 + budget_s - 180
        best_h, best_c5 = _greedy_ls(best_h, n, dx, rng, deadline_ls)

    # Phase 3: Simulated annealing
    if time.time() < t0 + budget_s - 130:
        deadline_sa = t0 + budget_s - 130
        best_h, best_c5 = _sim_anneal(best_h, n, dx, rng, deadline_sa, T0=0.04, Tc=0.001)

    # Phase 4: SLSQP restarts from diverse perturbations
    sigmas = [0.15, 0.25, 0.35, 0.1, 0.2, 0.4, 0.12, 0.3, 0.18, 0.28]
    it = 0
    while time.time() < t0 + budget_s - 100:
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp2(hp, n, dx, 250, 250, 1e-14)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
        elif c5 < best_c5 + 0.004 and time.time() < t0 + budget_s - 70:
            dl = t0 + budget_s - 70
            hl, cl = _sim_anneal(hc, n, dx, rng, dl, T0=0.02, Tc=0.001)
            if cl < best_c5:
                best_c5, best_h = cl, hl
        it += 1

    # Phase 5: Targeted peak reduction
    if time.time() < t0 + budget_s - 50:
        best_h, best_c5 = _targeted_peak(best_h, n, dx, rng, t0 + budget_s - 50)

    # Phase 6: SLSQP high-alpha polish
    for a in [800, 1600, 3200, 6400]:
        if time.time() > t0 + budget_s - 20:
            break
        hc = _slsqp2(best_h, n, dx, a, 500, 1e-15)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 7: Final greedy + SA micro-tuning
    if time.time() < t0 + budget_s - 8:
        best_h, best_c5 = _greedy_ls(best_h, n, dx, rng, t0 + budget_s - 8)
    if time.time() < t0 + budget_s - 3:
        best_h, best_c5 = _sim_anneal(best_h, n, dx, rng, t0 + budget_s - 2, T0=0.008, Tc=0.0003)

    # Phase 8: Try diverse initializations if time remains
    if time.time() < t0 + budget_s - 40:
        for mode in range(4):
            if time.time() > t0 + budget_s - 40:
                break
            hd = _diverse_init(n, rng, mode)
            hc = _slsqp2(hd, n, dx, 200, 300, 1e-14)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
            elif c5 < best_c5 + 0.005:
                dl = t0 + budget_s - 25
                hl, cl = _greedy_ls(hc, n, dx, rng, dl)
                if cl < best_c5:
                    best_c5, best_h = cl, hl

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
