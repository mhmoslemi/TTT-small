def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _smooth_obj_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    e = np.exp(s)
    e_sum = e.sum()
    w = alpha * e / e_sum
    obj = m + np.log(e_sum) / alpha
    g = np.zeros(n)
    for j in range(len(c)):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            g[lo1:hi1] += w[j] * (1.0 - x[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            g[lo2:hi2] -= w[j] * x[lo2 + lag:hi2 + lag]
    return obj, g

def _worst_lag_grad(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    j = int(np.argmax(c))
    val = c[j]
    g = np.zeros(n)
    lag = j - (n - 1)
    lo1 = max(0, lag)
    hi1 = min(n, n + lag)
    if hi1 > lo1:
        g[lo1:hi1] += (1.0 - x[lo1 - lag:hi1 - lag])
    lo2 = max(0, -lag)
    hi2 = min(n, n - lag)
    if hi2 > lo2:
        g[lo2:hi2] -= x[lo2 + lag:hi2 + lag]
    return g, float(val), j

            g[a1:b1] += w[j] * (1.0 - x[a1-lag:b1-lag])
        a2, b2 = max(0, -lag), min(n, n - lag)
        if b2 > a2:
            g[a2:b2] -= w[j] * x[a2+lag:b2+lag]
    return obj, g

def _worst_grad(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    j = int(c.argmax())
    g = np.zeros(n)
    lag = j - (n - 1)
    a1, b1 = max(0, lag), min(n, n + lag)
    if b1 > a1:
        g[a1:b1] += 1.0 - x[a1-lag:b1-lag]

    if b2 > a2:
        g[a2:b2] -= x[a2+lag:b2+lag]
    return g, float(c[j])

    """Projected gradient with backtracking. Returns updated h, c5, best_h, best_c5."""
    n = len(h)

                    break
                step *= 0.5

                if step < 1e-10:
                    break
    return h, best_c5, best_h

def _proj(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
