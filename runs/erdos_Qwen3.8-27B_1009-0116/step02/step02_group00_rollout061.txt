def _subgrad(h, margin=1e-8):

    corr = _corr(h)
    m = np.max(corr)
    n = len(h)
    active = np.where(corr >= m - margin)[0]
    g = np.zeros(n)
    for k in active:
        # grad[k] for h[j] is (1-2*h[k-j]) if 0 <= k-j < n

def _repair_sum(h, target):
    """Ensure sum(h) = target while keeping h in [0,1]"""
    h = np.clip(h, 0.0, 1.0)

        return h
    if diff > 0:
        # Need to add: add to entries with room
        room = 1.0 - h
        # Water filling
        order = np.argsort(-room)  # most room first
        for i in order:
            if diff <= 1e-12:

            add = min(diff, room[i])

import numpy as np
from scipy.interpolate import interp1d
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_val(h, dx):
    return float(np.max(_corr(h))) * dx


def _subgrad_h(h, margin):
    corr = _corr(h)
    m = np.max(corr)
    n = len(h)
    active = np.where(corr >= m - margin)[0]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            g[js:je + 1] += (1.0 - 2.0 * h[k - js:k - je + 1])
    g /= len(active)
    return g, len(active)


def _repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(h.sum())
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff < 1e-13:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(-h)
        for i in order:
            if diff > -1e-13:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def _interp_to_n(h_src, n_new):
    n_old = len(h_src)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h_src, kind='cubic', fill_value='extrapolate', bounds_error=False)
    return np.clip(f(x_new), 0.0, 1.0)


def _subgrad_descent(h, dx, target, n, t_end, rng):
    c5 = _c5_val(h, dx)
    best_h = h.copy()
    best_c5 = c5
    lr = 0.04
    mu = 0.6
    mom = np.zeros(n)
    margin = 1e-8
    stall = 0
    while time.time() < t_end - 0.5:
        g, n_active = _subgrad_h(h, margin)
        mom = mu * mom - lr * g
        h_new = np.clip(h + mom, 0.0, 1.0)
        h_new = _repair_sum(h_new, target)
        new_c5 = _c5_val(h_new, dx)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.92
                margin *= 2.0
                if margin > 5.0:
                    margin = 5.0
                stall = 0
        h = h_new
    return best_h, best_c5


def _sa_search(h, dx, target, n, t_end, rng):
    best_h = h.copy()
    best_c5 = _c5_val(h, dx)
    T0, Tend = 0.006, 0.00002
    max_it = 2000000
    for it in range(max_it):
        if time.time() > t_end - 0.3:
            break
        frac = it / float(max_it)
        T = T0 * (Tend / T0) ** frac
        dmax = 0.05 * (1.0 - 0.7 * frac)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (rng.rand() - 0.5) * 2.0 * dmax
        ni = h[i] + delta
        nj = h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        new_c5 = _c5_val(h, dx)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
        elif new_c5 <= best_c5 + T:
            pass
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c5


def _targeted_polish(h, dx, target, n, t_end):
    best_h = h.copy()
    best_c5 = _c5_val(h, dx)
    deltas = [0.003, 0.001, 0.005, 0.0005, 0.008, -0.003, -0.001, -0.005, -0.0005, -0.008]
    no_imp = 0
    while time.time() < t_end - 0.5:
        improved = False
        for i in range(n):
            if time.time() > t_end - 0.3:
                break
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = vi - ni
                if need > 1e-12:
                    cands = np.argsort(h)[:8]
                elif need < -1e-12:
                    cands = np.argsort(-h)[:8]
                else:
                    continue
                found = False
                for j in cands:
                    if j == i:
                        continue
                    nj = h[j] - need
                    if nj < 0.0 or nj > 1.0:

                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5_val(h, dx)
                    if c5_try < best_c5 - 1e-14:

                    else:

                    break
        if not improved:

            if no_imp >= 4:
                break
        else:
            no_imp = 0
    return best_h, best_c5

    time_per = max(5.0, (t_end - time.time()) / n_restarts - 1.0)
