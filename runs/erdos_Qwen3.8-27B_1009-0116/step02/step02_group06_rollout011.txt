import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order.tolist():
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, corr, topk, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        hi = h[i]
        for j in range(n):
            if i == j:
                continue
            hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            top_idx = np.argsort(corr)[-topk:]
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = corr[top_idx][None, :] + ds[:, None] * gd[top_idx][None, :]
            c5s = np.max(vals, axis=1)
            bi = np.argmin(c5s)

                d_best = ds[bi]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    vals2 = corr[top_idx][None, :] + ds2[:, None] * gd[top_idx][None, :]

                h[i] += d_best
                h[j] -= d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slsqp_eq(x, target):
    return np.sum(x) - target

def _slsqp_step(h, n, target, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'args': (target,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_step(h, n, corr, eta, topk):
    top_idx = np.argsort(corr)[-topk:]
    v = np.zeros(n)
    for k_idx in top_idx:
        v += _g_vec(h, 0, n) * 0  # placeholder, we need per-lag gradient
    # Compute gradient of each active lag wrt h
    gmat = np.zeros((n, len(top_idx)))
    nc = 2 * n - 1
    offset = n - 1
    for li, k_abs in enumerate(top_idx):
        lag = int(k_abs) - offset
        for m in range(n):
            grad = 0.0
            m1 = m + lag
            if 0 <= m1 < n:
                grad += (1.0 - h[m1])
            m2 = m - lag
            if 0 <= m2 < n:
                grad -= h[m2]
            gmat[m, li] = grad

    w = np.exp(20.0 * (vals - vals.max()))
    w /= w.sum()
    v = gmat @ w
    v -= v.mean()  # preserve sum constraint
    h_new = h - eta * v
    h_new = _repair(h_new, n)
    corr_new = _corr(h_new)
    return h_new, corr_new
