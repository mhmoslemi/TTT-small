import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _shifted(arr, shift, n):
    result = np.zeros(n)
    if shift >= 0:
        if shift < n:
            result[:n - shift] = arr[shift:]
    else:
        s = -shift
        if s < n:
            result[s:] = arr[:n - s]
    return result


def _grad_lse(h, alpha, n):
    corr = _corr(h)
    c = corr * (2.0 / n)
    m = np.max(c)
    exponents = np.clip(alpha * (c - m), -50, 50)
    exps = np.exp(exponents)
    total = np.sum(exps)
    w = exps / total
    grad = np.zeros(n)
    # Only compute for lags with significant weight
    active = w > 1e-10
    for idx in np.where(active)[0]:
        lag = idx - (n - 1)
        g = _shifted(1.0 - h, lag, n) - _shifted(h, -lag, n)
        grad += w[idx] * g
    grad *= (2.0 / n)
    val = (m + np.log(total) / alpha) * (2.0 / n)
    return grad, val


def _proj_integral(h, target, n):
    """Project h to [0,1]^n with sum = target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.clip(h[mask] + add, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            sub = -diff / cnt
            h[mask] = np.clip(h[mask] - sub, 0.0, 1.0)
    # Final exact fix
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _lse_gradient_descent(h, n, max_iter, alpha_init, alpha_end, beta):
    """LSE gradient descent with integral constraint penalty."""
    h = h.copy()
    best_h = h.copy()
    best_val = _c5(h, 2.0 / n)
    dx = 2.0 / n
    target = n / 2.0
    n_lag = len(_corr(h))
    for it in range(max_iter):
        alpha = alpha_init * (alpha_end / alpha_init) ** (it / max_iter)
        grad, lse_val = _grad_lse(h, alpha, n)
        # Add integral constraint gradient
        sum_diff = float(np.sum(h)) - target
        grad += 2.0 * beta * sum_diff * np.ones(n)
        # Line search
        step = 0.05
        for _ in range(30):
            h_trial = _proj_integral(h - step * grad, target, n)
            c5_trial = _c5(h_trial, dx)
            if c5_trial < lse_val + 0.01 * step * np.linalg.norm(grad):
                h = h_trial
                break
            step *= 0.5
        else:
            h = _proj_integral(h - step * grad, target, n)
        c5_cur = _c5(h, dx)
        if c5_cur < best_val:
            best_val = c5_cur
            best_h = h.copy()
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _shifted(arr, shift, n):
    result = np.zeros(n)
    if shift >= 0:
        if shift < n:
            result[:n - shift] = arr[shift:]
    else:
        s = -shift
        if s < n:
            result[s:] = arr[:n - s]
    return result


def _grad_lag(h, lag, n):
    return _shifted(1.0 - h, lag, n) - _shifted(h, -lag, n)


def _grad_lse(h, alpha, n):
    corr = _corr(h)
    c = corr * (2.0 / n)
    m = np.max(c)
    exponents = np.clip(alpha * (c - m), -50, 50)
    exps = np.exp(exponents)
    total = np.sum(exps)
    w = exps / total
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        lag = idx - (n - 1)
        grad += w[idx] * _grad_lag(h, lag, n)
    grad *= (2.0 / n)
    val = (m + np.log(total) / alpha) * (2.0 / n)
    return grad, val


def _proj(h, target, n):

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h
