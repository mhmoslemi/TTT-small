        h = _repair(h, n)
        cur_c5 = _c5(h, dx)
        if prev_c5 - cur_c5 < 1e-13:
            break
        prev_c5 = cur_c5
    return h, _c5(h, dx)

def _slsqp_opt(h, n, dx, alpha, lam, maxiter=250):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_smooth, h, args=(alpha, dx, lam), method='SLSQP',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _nm_opt(h, n, dx, lam, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_smooth, h, args=(80.0, dx, lam), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _tent_init(n, power=1.0):
    t = np.linspace(0, 1, n)
    h = np.cos(np.pi * t) ** 2
    return _repair(0.5 + 0.4 * (2 * h - 1), n)

def _block_init(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    for _ in range(n // 4):
        i = rng.randint(n - 1)
        h[i] = rng.uniform(0, 1)
        h[i + 1] = rng.uniform(0, 1)
    return _repair(h, n)

def _sin_init(n, k, rng):
    t = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * k * t / 2.0) + 0.1 * np.sin(2 * np.pi * (k + 1) * t / 2.0 + rng.uniform(0, 6.28))
    return _repair(h, n)

def _local_search2d(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    for it in range(30000):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        s_other = best.sum() - best[i] - best[j]
        target = n / 2.0
        corr = np.zeros(2 * n - 1)
        off = n - 1
        for c in range(2 * n - 1):
            vals = []
            ii = c - off
            if 0 <= ii < n and ii != i and ii != j:
                vals.append(best[ii])
            jj = c - off - i
            if 0 <= jj < n and jj != i and jj != j and jj != ii:
                pass
        # Simpler: just try a few 2D moves
        for di in [-0.05, -0.02, 0.0, 0.02, 0.05]:
            for dj in [-0.05, -0.02, 0.0, 0.02, 0.05]:
                ni = best[i] + di
                nj = best[j] + dj
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                if ni + nj + s_other < 0 or ni + nj + s_other > n:
                    continue
                best[i], best[j] = ni, nj
                best = _repair(best, n)
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                else:
                    best[i], best[j] = ni, nj
                    best = _repair(best, n)
        if best_c5 < _c5(best, dx) - 1e-9:
            best = _repair(best, n)
    return best, best_c5

def _fine_local(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = np.array([0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02])
    for it in range(20000):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        for d in steps:
            ni, nj = best[i] + d, best[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-12:
                best_c5 = nc
            else:
                best[i], best[j] = ni, nj
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx0)
    best_n = n0

    def rem(margin=20):
        return t0 + budget_s - margin - time.time()

    # Phase 1: Coordinate descent from initial (different from SLSQP-first approach)
    if rem(400) > 0:
        h_cd, c5_cd = _coord_descent(h0.copy(), n0, lam=200.0, max_sweeps=40, deadline=t0 + budget_s - 400)
        if c5_cd < best_c5:
            best_c5, best_h, best_n = c5_cd, h_cd, n0

    # Phase 2: SLSQP polish with various alphas
    if rem(250) > 0:
        for alpha in [30, 60, 120, 250]:
            if rem(250) <= 0:
                break
            hp = _slsqp_opt(best_h.copy(), best_n, 2.0 / best_n, alpha, 50.0, 200)
            c5p = _c5(hp, 2.0 / best_n)
            if c5p < best_c5 - 1e-6:
                best_c5, best_h, best_n = c5p, hp, best_n

    # Phase 3: Try different n values with diverse inits
    n_trials = [n0, max(30, n0 - 15), min(120, n0 + 20), n0 // 2 if n0 >= 40 else n0]
    seen_ns = set()
    for n in n_trials:
        if n in seen_ns or n < 20 or n > 200:
            continue
        seen_ns.add(n)
        if rem(120) <= 0:
            break
        dx = 2.0 / n
        inits = [
            _tent_init(n, 1.0),
            _block_init(n, rng),
            _sin_init(n, 2, rng),
            _sin_init(n, 3, rng),
            _sin_init(n, 4, rng),
        ]
        for init in inits:
            if rem(120) <= 0:
                break
            h_t = init.copy()
            # Quick SLSQP
            h_t = _slsqp_opt(h_t, n, dx, 80, 200.0, 150)
            # Coordinate descent
            h_t, c5_t = _coord_descent(h_t, n, lam=500.0, max_sweeps=15, deadline=t0 + budget_s - 120)
            if c5_t < best_c5 - 1e-6:
                best_c5, best_h, best_n = c5_t, h_t.copy(), n

    # Phase 4: Aggressive coordinate descent on best (high lam for tight sum)
    if rem(100) > 0:
        for lam in [500, 2000, 5000]:
            if rem(100) <= 0:
                break
            h_cd, c5_cd = _coord_descent(best_h.copy(), best_n, lam=lam, max_sweeps=30, deadline=t0 + budget_s - 100)
            if c5_cd < best_c5:
                best_c5, best_h, best_n = c5_cd, h_cd, best_n

    # Phase 5: Fine local search (pair moves, small steps)
    if rem(60) > 0:
        deadline_ls = t0 + budget_s - 30
        best_h, best_c5 = _fine_local(best_h, best_n, 2.0 / best_n, rng, deadline_ls)

    # Phase 6: Final SLSQP + coord descent polish
    if rem(15) > 0:
        for alpha in [300, 600, 1000]:
            if rem(15) <= 5:
                break
            hp = _slsqp_opt(best_h.copy(), best_n, 2.0 / best_n, alpha, 0.0, 150)
            c5p = _c5(hp, 2.0 / best_n)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p
        if rem(8) > 0:
            best_h, best_c5 = _coord_descent(best_h.copy(), best_n, lam=1000.0, max_sweeps=20, deadline=t0 + budget_s - 5)

    # Phase 7: Ultra-fine local search
    if rem(8) > 0:
        deadline_fine = t0 + budget_s - 2
        best_h, best_c5 = _fine_local(best_h, best_n, 2.0 / best_n, rng, deadline_fine)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
