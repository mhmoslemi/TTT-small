                diff -= adj

                diff += adj

    best_h = h.copy()
    best_c5 = _c5_val(h, dx)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    def elapsed():
        return time.time() - t0

    # Phase 1: Intensive exact pair descent
    lim1 = t0 + min(budget_s * 0.30, 300)
    h, c = _descent(h, n, pairs, rng, lim1, t0, max_sweeps=800)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 2: Multi-restart with structured perturbations
    lim2 = t0 + min(budget_s * 0.65, 700)
    sigmas = [0.04, 0.08, 0.12, 0.2, 0.3, 0.06, 0.15, 0.25, 0.4, 0.03, 0.1, 0.35, 0.5, 0.07]
    for si, sig in enumerate(sigmas):
        if time.time() > lim2 - 20:
            break
        h_p = best_h + sig * rng.randn(n)
        h_p = _repair(h_p, n)
        sub_lim = min(time.time() + 50, lim2 - 10)
        h_p, c_p = _descent(h_p, n, pairs, rng, sub_lim, t0, max_sweeps=200)
        if c_p < best_c5:
            best_c5, best_h = c_p, h_p.copy()

    # Phase 3: Smooth-mode restart (Fourier-based perturbation)
    if time.time() < lim2 - 30:
        for trial in range(4):
            if time.time() > lim2 - 15:
                break
            freqs = rng.choice([1, 2, 3, 4, 5, 6], size=3, replace=False)
            x_grid = np.linspace(0, 2, n)
            pert = np.zeros(n)
            for f in freqs:
                pert += rng.randn() * np.cos(np.pi * f * x_grid) + rng.randn() * np.sin(np.pi * f * x_grid)
            h_p = best_h + 0.15 * pert
            h_p = _repair(h_p, n)
            h_p, c_p = _descent(h_p, n, pairs, rng, min(time.time() + 40, lim2 - 10), t0, max_sweeps=150)
            if c_p < best_c5:
                best_c5, best_h = c_p, h_p.copy()

    # Phase 4: Fine descent on best solution
    lim3 = t0 + budget_s - 25
    h = best_h.copy()
    h, c = _descent(h, n, pairs, rng, lim3, t0, max_sweeps=600)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 5: Final structured sweeps (no randomization, multiple passes)
    if time.time() < t0 + budget_s - 8:
        h = best_h.copy()
        for _ in range(30):
            if time.time() > t0 + budget_s - 8:
                break
            changed = False
            for i in range(n):
                for j in range(i + 1, n):
                    if _pair_update(h, i, j):
                        changed = True
            if not changed:
                break
            c = _c5_val(_repair(h, n), dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
