    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha):
    corr = _corr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    if time.time() >= deadline:
        return h, _c5(h, dx)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:

This is the "softmax" subgradient. As α→∞, it concentrates on the active constraint.

The projection onto Ω (box + hyperplane) can be done efficiently.

For the 1D line search on pair (i,j):

- δ[k] = change in c[k] per unit transfer from j to i
- Find d that minimizes max_k (c[k] + d*δ[k]) over d ∈ [d_min, d_max]

The change in c[k] when h[i]→h[i]+d and h[j]→h[j]-d:
For each correlation term c[p] = Σ h[valid]*(1-h[valid+s]):
- If i appears as the first factor: contribution changes by d*(1-h[i+s])
- If i appears as the second factor (i.e., i = a+s for some a): contribution changes by -d*h[i-s]
- Similarly for j with opposite sign

So δ[p] = ∂c_p/∂h[i] - ∂c_p/∂h[j] (computed once)

And the new c[p] ≈ c[p] + d*δ[p] (first order). For small d, this is accurate.

To find the optimal d: minimize max_p (c[p] + d*δ[p]) over d ∈ [d_lo, d_hi].

This is a 1D problem. The function is piecewise linear (upper envelope of lines). The minimum of the upper envelope is at a point where two lines intersect at the top, or at a boundary.

Algorithm:
1. Compute all lines: y_p(d) = c[p] + d*δ[p]
2. The minimum of max_p y_p(d) occurs where:
   - Two lines y_a(d) = y_b(d) and both are on the upper envelope
   - Or at d_lo or d_hi
3. For each pair (a,b), compute d_ab = (c[b]-c[a])/(δ[a]-δ[b]) if δ[a]≠δ[b]
4. Check if d_ab is in [d_lo, d_hi] and if at d_ab, both a and b are on the upper envelope
5. Pick the d that gives the smallest max

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _project(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except:
        return h

def _pgd(h, n, alpha, n_iter, eta, deadline):
    for it in range(n_iter):
        if time.time() >= deadline:
            break
        c = _corr(h)
        m = np.max(c)
        w = np.exp(np.clip(alpha * (c - m), -500, 500))
        wsum = w.sum()
        if wsum < 1e-30:

        w /= wsum
        # Compute gradient
        grad = np.zeros(n)
        for p in range(len(c)):
            if w[p] < 1e-12:

            # ∂c_p/∂h[j] = (1-h[j+s])*[valid] - h[j-s]*[valid]

            second = np.zeros(n)

                first[:n-s] = 1.0 - h[s:]
                second[s:] = h[:n-s]
            else:
                idx = -s
                first[idx:] = 1.0 - h[:n-idx]
                second[:n-idx] = h[idx:]
            grad += w[p] * (first - second)
        # Proximal gradient step with backtracking
        step = eta
        for _bt in range(20):

            if _c5(h_new, 2.0/n) < _c5(h, 2.0/n) + 1e-15:
                h = h_new
                break
            step *= 0.5

    h = h.copy()
    scales_decreasing = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for scale in scales_decreasing:

                i = rng.randint(n)
                j = rng.randint(n)

                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:

                else:

    return h, best

def _try_n(h70, n_new, rng, deadline):
