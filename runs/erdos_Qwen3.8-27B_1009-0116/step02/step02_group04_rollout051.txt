import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -50, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, dx, deadline, alpha, maxiter):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, n, alpha)
    def grad(x):
        # Numerical gradient (expensive but n=70 is small)
        g = np.zeros(n)
        eps = 1e-6
        f0 = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - f0) / eps
            x[i] -= eps
        return g
    def con_fun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:

        best_h, best_c5 = _greedy_local(best_h, n, dx, rng, td)

    # Phase 5: Sweep-based local (fine scales)
    if rem() > 40:
        best_h, best_c5 = _sweep_local(best_h, n, dx, time.time()+min(35, rem()-10))

    # Phase 6: Final polish
    if rem() > 20:
        best_h, best_c5 = _slsqp_run(best_h, n, dx, time.time()+min(15, rem()-8), 4000, 800)
    if rem() > 8:
        best_h, best_c5 = _greedy_local(best_h, n, dx, rng, time.time()+min(5, rem()-3))

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
