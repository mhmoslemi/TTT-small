```python
def compute_grad(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1-h, 'full')
    e = np.exp(alpha * (c - c.max()))
    w = e / e.sum()
    # w has length 2n-1, index idx corresponds to k = idx-(n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        for idx in range(2*n-1):
            k = idx - (n-1)
            term = 0
            if 0 <= i-k < n:
                term += (1 - h[i-k])
            if 0 <= i+k < n:
                term -= h[i+k]
            grad[i] += w[idx] * term
    return dx * grad
```

```python
def _slsqp_analytic(h, n, dx, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    def fun(x):
        val, _ = _obj_grad(x, alpha, dx, for_grad=False)
        return val
    
    def grad(x):
        _, g = _obj_grad(x, alpha, dx, for_grad=True)
        return g
    
    try:
        res = minimize(fun, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
```

```python
    best_val = _c5(best, dx)

            break
        val, g = _obj_grad(best, alpha, dx)
        # Project gradient onto constraint: remove mean
        d = g - np.mean(g)
        # Normalize
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            break
        d = d / dn
        # Line search
        improved = False
        for trial in range(10):
            h_new = _repair(best - step * dn * d, n)
            v_new = _c5(h_new, dx)
            if v_new < best_val - 1e-12:
                best = h_new
                best_val = v_new
                step = min(step * 1.5, 0.1)
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-5:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _obj_grad(x, alpha, dx):
    n = len(x)
    c = _corr(x)
    cm = c.max()
    e = np.exp(np.clip(alpha * (c - cm), -500, 500))
    se = e.sum()
    val = (cm + np.log(se) / alpha) * dx
    w = e / se
    term1 = np.convolve(w, 1.0 - x, 'full')[n - 1:2 * n - 1]
    corr_wh = np.correlate(w, x, 'full')
    term2 = corr_wh[n - 1:2 * n][::-1]
    grad = dx * (term1 - term2)
    return val, grad

def _slsqp_analytic(h, n, dx, alpha, maxiter, ftol=1e-15):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    def f(x):
        v, _ = _obj_grad(x, alpha, dx)
        return v

    def g(x):
        _, gr = _obj_grad(x, alpha, dx)
        return gr

    try:
        res = minimize(f, h, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_descent(h, n, dx, alpha, n_steps, deadline):
    best = h.copy().astype(float)
    best_c5 = _c5(best, dx)
    step = 0.02
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        _, g = _obj_grad(best, alpha, dx)
        d = g - np.mean(g)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn
        improved = False
        for _t in range(12):
            h_new = _repair(best - step * dn * d, n)
            v = _c5(h_new, dx)
            if v < best_c5 - 1e-13:
                best, best_c5, step = h_new, v, min(step * 1.4, 0.15)
                improved = True
                break
            step *= 0.45
        if not improved:
            step *= 0.4
            if step < 1e-6:
                break
    return best, best_c5

def _pair_search(h, n, dx, rng, deadline):
    best = h.copy().astype(float)
    best_c5 = _c5(best, dx)
    steps = np.array([0.003, 0.007, 0.015, 0.03, 0.06,
                      -0.003, -0.007, -0.015, -0.03, -0.06])
    stall = 0
    while time.time() < deadline:
        improved = False
        for _ in range(80):
            if time.time() > deadline:
                break
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                vc = _c5(best, dx)
                if vc < best_c5 - 1e-13:
                    best_c5 = vc
                    improved = True
                else:
                    best[i], best[j] = oi, oj
                if improved:
                    break
            if improved:
                break
        if improved:
            stall = 0
            steps *= 1.15
            steps = np.clip(steps, -0.3, 0.3)
        else:
            stall += 1
            if stall > 25:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Multi-alpha SLSQP with analytic gradient
    for a in [12, 30, 60, 120, 250, 500, 1000]:
        if rem() < 180:
            break
        hc = _slsqp_analytic(best_h, n, dx, a, 400)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-14:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: Gradient descent with moderate alpha
    if rem() > 120:
        dl = t0 + budget_s - 120
        for a in [80, 200]:
            if time.time() > dl:
                break
            hg, cg = _grad_descent(best_h, n, dx, a, 600, dl)
            if cg < best_c5:
                best_c5, best_h = cg, hg

    # Phase 3: SLSQP restarts
    sigmas = [0.3, 0.15, 0.5, 0.2, 0.4, 0.1, 0.6, 0.25, 0.35, 0.08]
    for it in range(20):
        if rem() < 60:
            break
        s = sigmas[it % len(sigmas)]
        hp = _repair(best_h + s * rng.randn(n), n)
        hc = _slsqp_analytic(hp, n, dx, 300, 250)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        elif c5 < best_c5 + 0.003 and rem() > 80:
            hl, cl = _pair_search(hc, n, dx, rng, t0 + budget_s - 50)
            if cl < best_c5:
                best_c5, best_h = cl, hl

    # Phase 4: High-alpha SLSQP polish
    for a in [800, 1500, 3000]:
        if rem() < 15:
            break
        hc = _slsqp_analytic(best_h, n, dx, a, 800, 1e-16)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 5: Final pair search
    if rem() > 8:
        best_h, best_c5 = _pair_search(best_h, n, dx, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
