import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, n_coarse):
    corr = _corr(h)
    cur_max = np.max(corr) * dx
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gi = _g_vec(h, i, n)
            gj = _g_vec(h, j, n)
            gd = gi - gj
            ds = np.linspace(d_min, d_max, n_coarse)
            nc = corr[None, :] + ds[:, None] * gd[None, :]
            c5s = np.max(nc, axis=1) * dx
            bi = np.argmin(c5s)
            if c5s[bi] < cur_max - 1e-14:
                d_best = ds[bi]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                cur_max = c5s[bi]
                improved = True
    return h, improved

def _sweep_fine(h, n, dx, n_coarse, fine_n, fine_wid):
    corr = _corr(h)
    cur_max = np.max(corr) * dx
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gi = _g_vec(h, i, n)
            gj = _g_vec(h, j, n)
            gd = gi - gj
            ds = np.linspace(d_min, d_max, n_coarse)
            nc = corr[None, :] + ds[:, None] * gd[None, :]
            c5s = np.max(nc, axis=1) * dx
            bi = np.argmin(c5s)
            if c5s[bi] < cur_max - 1e-14:
                d_best = ds[bi]
                lo = max(d_min, d_best - fine_wid)
                hi2 = min(d_max, d_best + fine_wid)
                if hi2 - lo > 1e-12:
                    ds2 = np.linspace(lo, hi2, fine_n)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                    cur_max = c5s2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, improved

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _kick(h, n, rng, strength, n_kick):
    idx = rng.choice(n, size=min(n_kick, n), replace=False)
    h[idx] += strength * rng.randn(len(idx))
    return _repair(h, n)

def _struct_kick(h, n, rng):
    idx = rng.choice(n, size=min(3, n), replace=False)
    for a in range(len(idx)):
        for b in range(a + 1, len(idx)):
            delta = 0.03 * rng.randn()
            h[idx[a]] += delta
            h[idx[b]] -= delta
    return _repair(h, n)
