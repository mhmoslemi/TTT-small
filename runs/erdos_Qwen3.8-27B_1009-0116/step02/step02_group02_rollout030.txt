                         options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-15})
            hf = _proj(r.x, tgt); cf = _c5(hf, dx)
            if cf < bc:
                bc = cf; bh = hf
        except Exception:
            pass

    # Phase 10: Final CD (both modes)
    if _rem(t0, budget_s) > 15:
        h, c = _cd_corrected(bh, dx, n, 200, rng)
        if c < bc:
            bc = c; bh = h
    if _rem(t0, budget_s) > 12:
        h, c = _cd_corrected(bh, dx, n, 20, rng, deterministic=True)
        if c < bc:
            bc = c; bh = h

    # Phase 11: Final SLSQP
    if _rem(t0, budget_s) > 15:
        try:
            r = minimize(_slsqp_obj, bh, args=(30000,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 800, 'ftol': 1e-18})
            hf = _proj(r.x, tgt); cf = _c5(hf, dx)
            if cf < bc:
                bc = cf; bh = hf
        except Exception:
            pass

    # Final squeeze
    if _rem(t0, budget_s) > 8:
        h, c = _cd_corrected(bh, dx, n, 200, rng)
        if c < bc:
            bc = c; bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
