def _fast_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    one = 1.0 - h
    for k in range(2*n - 1):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        lag = k - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:

                # c[k] = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = float(y.min()) - 2.0, float(y.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    th = 0.5 * (lo + hi)
    x = np.clip(y - th, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)[::-1]
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                amt = min(diff, 1.0 - x[k])
                x[k] += amt
                diff -= amt
            else:
                amt = min(-diff, x[k])
                x[k] -= amt
                diff += amt
            if abs(diff) < 1e-13:
                break
    return x

def _corr(x):
    return np.correlate(x, 1.0 - x, mode='full')

def _softmax_weights(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    return e / float(e.sum())

def _sobj(x, a):
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + float(np.log(np.exp(s).sum())) / a

def _grad_smooth(x, w):
    n = len(x)
    g = np.zeros(n)
    one = 1.0 - x
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-15:

        lag = k - n1
        L = abs(lag)
        if L == 0:
            g += wk * (1.0 - 2.0 * x)

            g[L:] += wk * one[:n - L]
        else:

def _fista(h0, alpha, target, t0, tmax, max_iter=10000):

    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * n)
    
    for it in range(max_iter):
        if time.time() - t0 > tmax - 5:
            break
        c = _corr(z)
        w = _softmax_weights(c, alpha)
        gz = _grad_smooth(z, w)
        
        # Backtracking
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 10:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(gz @ (xn - z)) + 0.5 / step * float((xn - z) @ (xn - z))
            if fn <= quad + 1e-9:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z_new = xn + (t - 1.0) / tn * (xn - x)
        z = _proj(z_new, target)
        x = xn
        t = tn
        
        cval = _c5(x, dx)
        if cval < bc - 1e-12:
            bc = cval
            bx = x.copy()
    
    return bx, bc

def _subgrad_step(h, max_iter=2000, t0=None, tmax=None):

    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    bx = x.copy()
    bc = _c5(x, dx)
    lr = 1.0 / (2.0 * n)

            break
        c = _corr(x)
        k_star = int(c.argmax())
        n1 = n - 1
        lag = k_star - n1
        L = abs(lag)
        g = np.zeros(n)
        one = 1.0 - x
        if L == 0:
            g = 1.0 - 2.0 * x
        elif lag > 0:
            g[:n - L] -= x[L:]
            g[L:] += one[:n - L]
        else:
            g[:n - L] += one[L:]
            g[L:] -= x[:n - L]
        
        x_new = x - lr * g
        x = _proj(x_new, n / 2.0)
        
        cval = _c5(x, dx)
        if cval < bc - 1e-12:
            bc = cval
            bx = x.copy()
        lr *= 0.9999  # slight decay
    return bx, bc

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(x):
    return np.correlate(x, 1.0 - x, mode='full')

def _proj(y, target):
    n = len(y)
    lo, hi = float(y.min()) - 2.0, float(y.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    th = 0.5 * (lo + hi)
    x = np.clip(y - th, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x)[::-1] if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                amt = min(diff, 1.0 - x[k])
                x[k] += amt
                diff -= amt
            else:
                amt = min(-diff, x[k])
                x[k] -= amt
                diff += amt
            if abs(diff) < 1e-13:
                break
    return x

def _smw(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    return e / float(e.sum())

def _sobj(x, a):
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + float(np.log(np.exp(s).sum())) / a

def _grad_smooth(x, w):
    n = len(x)
    g = np.zeros(n)
    one = 1.0 - x
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        lag = k - n1
        L = abs(lag)
        if L == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - L] -= wk * x[L:]
            g[L:] += wk * one[:n - L]
        else:
            g[:n - L] += wk * one[L:]
            g[L:] -= wk * x[:n - L]
    return g

def _fista(h0, alpha, target, t0, tmax, max_iter=10000):
    n = len(h0)
    dx = 2.0 / n
    x, z = h0.copy(), h0.copy()
    t = 1.0
    bx, bc = x.copy(), _c5(x, dx)
    step = 1.0 / (3.0 * n)
    for it in range(max_iter):
        if time.time() > tmax - 3:
            break
        c = _corr(z)
        w = _smw(c, alpha)
        gz = _grad_smooth(z, w)
        y = z - step * gz
        xn = _proj(y, target)
        for _ in range(8):
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(gz @ (xn - z)) + 0.5 / step * float((xn - z) @ (xn - z))
            if fn <= quad + 1e-9:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        cval = _c5(x, dx)
        if cval < bc - 1e-13:
            bc, bx = cval, x.copy()
    return bx, bc

def _cd_random(h, dx, target, rng, t0, tmax, npasses=200):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    dvals = np.concatenate([np.array([0.001, 0.003, 0.008, 0.015, 0.03, 0.06, 0.12, 0.2]),
                            -np.array([0.001, 0.003, 0.008, 0.015, 0.03, 0.06, 0.12, 0.2])])
    for p in range(npasses):
        if time.time() > tmax - 5:
            break
        imp = False
        for _ in range(n * 15):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
            else:
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _cd_sweep(h, dx, target, t0, tmax, nsweeps=20):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    dvals = [0.003, 0.01, 0.025, 0.05, 0.1, 0.2, -0.003, -0.01, -0.025, -0.05, -0.1, -0.2]
    for _ in range(nsweeps):
        if time.time() > tmax - 3:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                hi0, hj0 = h[i], h[j]
                for d in dvals:
                    ni, nj = hi0 + d, hj0 - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < bc - 1e-13:
                        bc = c
                        imp = True
                        break
                    h[i], h[j] = hi0, hj0
                else:
                    h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _subgrad(h, target, t0, tmax, max_iter=5000):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    bx, bc = x.copy(), _c5(x, dx)
    lr = 0.02
    for it in range(max_iter):
        if time.time() > tmax - 3:
            break
        c = _corr(x)
        k_star = int(c.argmax())
        n1 = n - 1
        lag = k_star - n1
        L = abs(lag)
        g = np.zeros(n)
        one = 1.0 - x
        if L == 0:
            g = 1.0 - 2.0 * x
        elif lag > 0:
            g[:n - L] -= x[L:]
            g[L:] += one[:n - L]
        else:
            g[:n - L] += one[L:]
            g[L:] -= x[:n - L]
        x = _proj(x - lr * g, target)
        cval = _c5(x, dx)
        if cval < bc - 1e-13:
            bc, bx = cval, x.copy()
        lr = max(lr * 0.9998, 0.002)
    return bx, bc

def run(seed=42, budget_s=1100, **kwargs):
