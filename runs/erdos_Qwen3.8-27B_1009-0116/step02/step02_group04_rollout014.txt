```python
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr_arr(h)) * dx)

    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha, eps=1e-6):
    """Numerical gradient of smoothed objective."""
    n = len(x)
    g = np.zeros(n)
    f0 = _smooth_obj(x, alpha)
    for i in range(n):
        xp = x.copy()
        xp[i] += eps
        g[i] = (_smooth_obj(xp, alpha) - f0) / eps
    return g

def _project_sum(h, target, n):
    """Project h to [0,1]^n with sum = target."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on shift
    lo, hi = h.min() - 1.0, h.max() - 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix remaining
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - hp)
        else:
            order = np.argsort(hp)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - hp[idx])
                hp[idx] += add
                diff -= add
            else:
                sub = min(-diff, hp[idx])
                hp[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return hp

    return _project_sum(h, n/2.0, n)

    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    cur_obj = _smooth_obj(h, alpha_init)
    eta = 0.01
    prev_g = None
    prev_h = None
    
    while time.time() < deadline:
        alpha = min(alpha_init * 2, alpha_init * (1 + 0.001))  # slowly increase
        # Actually let's keep alpha fixed for gradient stability
        g = _smooth_grad(h, alpha)
        
        # Barzilai-Borwein step
        if prev_g is not None:
            s = h - prev_h
            y = g - prev_g
            sy = s @ y
            yy = y @ y
            if yy > 1e-12:
                if sy > 1e-12:
                    eta = (s @ s) / sy
                else:
                    eta = (s @ s) / (yy * (s @ s / max(sy, 1e-12)))
                eta = max(eta * 0.5, 1e-8)
                eta = min(eta, 0.1)
        
        h_new = _project_sum(h - eta * g, n/2.0, n)
        new_obj = _smooth_obj(h_new, alpha)
        
        # Backtracking if needed

            h_new = _project_sum(h - eta * g, n/2.0, n)
```

```python
def _coord_cycling(h, n, dx, deadline):
    """Coordinate cycling with 1D line search on pairs."""
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    h = h.copy()

    # where d changes h[i] by d and h[j] by -d
    for d in range(20):  # number of passes
        if time.time() >= deadline:
            break
        for i in range(n):
            if time.time() >= deadline:
                break
            for j in range(i+1, n):
                # Binary search for optimal d in [-hi, 1-hj] ∩ [-hj, 1-hi]
                d_lo = max(-h[i], -h[j])
                d_hi = min(1-h[i], 1-h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                # Golden section
                gr = 0.618
                a, b = d_lo, d_hi
                c = b - gr*(b-a)
                d_val = a + gr*(b-a)
                # Evaluate at c and d
                h[i] += c; h[j] -= c
                fc = _compute_c5(h, dx)
                h[i] -= c; h[j] += c
                h[i] += d_val; h[j] -= d_val
                fd = _compute_c5(h, dx)
                h[i] -= d_val; h[j] += d_val
                # 6 more iterations of golden section
                for _ in range(6):

                        b = d_val; d_val = c; fd = fc
                        c = b - gr*(b-a)
                        h[i] += c; h[j] -= c
                        fc = _compute_c5(h, dx)
                        h[i] -= c; h[j] += c
                    else:
```

```python
        if c < bc: bc = c

    # Phase 3: SSA (200s)
    te = t0 + 420
    if time.time() < te:
        bh, c = _ssa(bh, n, dx, rng, te, T0=0.003, sc=0.025)
        if c < bc: bc = c

    # Phase 4: Fine SLSQP (150s)
    te = t0 + 570
    for a in [400, 800, 1500, 3000]:
        if time.time() > te: break
        h = _slq(bh, n, a, 800)
        c = _c5(h, dx)
        if c < bc: bc, bh = c, h.copy()

    # Phase 5: Coordinate cycling (150s)
    te = t0 + 720
    if time.time() < te:
        bh, c = _cyc(bh, n, dx, te)
        if c < bc: bc = c

    # Phase 6: Hill climbing (150s)
    te = t0 + 870
    if time.time() < te:
        bh, c = _hc(bh, n, dx, rng, te)
        if c < bc: bc = c

    # Phase 7: Final SSA + SLSQP (130s)
    te = t0 + budget_s - 10
    if time.time() < te - 50:
        bh, c = _ssa(bh, n, dx, rng, te - 50, T0=0.0004, sc=0.008)
        if c < bc: bc = c
    if time.time() < te:
        h = _slq(bh, n, 5000, 1200)
        c = _c5(h, dx)
        if c < bc: bc, bh = c, h.copy()

    bh = _rep(bh, n)
    bc = _c5(bh, dx)
    return (bh, bc, n)
```
