        r = minimize(obj, h0, method='L-BFGS-B', bounds=bnds,
                     constraints=cons, options={'maxiter': 2000, 'ftol': 1e-18,
                                                 'gtol': 1e-14, 'maxfun': 50000})
        hf = _proj(r.x, target)
        cf = _c5(hf, 2.0 / n)
        return hf, cf
    except Exception:
        return h0, _c5(h0, 2.0 / n)

def _active_refine(h, dx, n, target, t0, budget):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(corr.max())
    thresh = m - 0.002
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([n - 1])
    n_act = len(active)
    if n_act > 8:
        top = np.argsort(corr)[-8:]
        active = top
    a = 500.0

    def obj_act(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        ca = c[active]
        ma = float(ca.max())
        s = np.clip(a * (ca - ma), -500.0, 500.0)
        return ma + np.log(float(np.exp(s).sum())) / a

    def grad_act(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        ca = c[active]
        ma = float(ca.max())
        s = np.clip(a * (ca - ma), -500.0, 500.0)
        e = np.exp(s)
        w = e / float(e.sum())
        g = np.zeros(n)
        for idx, k in enumerate(active):
            wk = w[idx]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            if lag == 0:
                g += wk * (1.0 - 2.0 * v)
            elif lag > 0:
                g[lag:] += wk * (1.0 - v[:n-lag])
                g[:n-lag] -= wk * v[lag:]
            else:
                L = -lag
                g[:n-L] -= wk * v[L:]
                g[L:] += wk * (1.0 - v[:n-L])
        return g

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: float(np.sum(v)) - target,
             'jac': lambda v: np.ones(n)}]
    try:
        r = minimize(obj_act, h, jac=grad_act, method='L-BFGS-B',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 80000})
        hf = _proj(r.x, target)
        cf = _c5(hf, dx)
        return hf, cf
    except Exception:
        return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)

    h0 = _proj(np.array(initial_h_values, dtype=np.float64), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: FISTA ladder with progressive alpha
    for a in [30, 60, 120, 250, 500, 1000, 2000, 5000]:
        if time.time() - t0 > budget_s - 60:
            break
        h, c = _fista(bh, a, tgt, 6000, t0, budget_s)
        if c < bc:
            bc = c
            bh = h
        if time.time() - t0 > budget_s - 40:
            break
        h2, c2 = _cd(bh, dx, n, rng, 2000, t0, budget_s)
        if c2 < bc:
            bc = c2
            bh = h2

    # Phase 2: Triple-move CD for more flexibility
    if time.time() - t0 < budget_s - 30:
        h, c = _cd3(bh, dx, n, rng, 3000, t0, budget_s)
        if c < bc:
            bc = c
            bh = h

    # Phase 3: Diverse restarts
    restarts = [
        (0.15, 'gauss'), (0.25, 'gauss'), (0.1, 'gauss'), (0.3, 'gauss'),
        (0.2, 'shift'), (0.1, 'shift'), (0.15, 'gauss'), (0.08, 'gauss'),
        (0.25, 'gauss'), (0.35, 'gauss'), (0.12, 'gauss'), (0.18, 'gauss'),
    ]
    for sigma, kind in restarts:
        if time.time() - t0 > budget_s - 80:
            break
        if kind == 'gauss':
            hp = _proj(bh + sigma * rng.randn(n), tgt)
        else:
            shift = rng.randint(1, 5)
            hp = _proj(np.roll(bh, shift), tgt)
        for a in [300, 1000]:
            h, c = _fista(hp, a, tgt, 3000, t0, budget_s)
            if c < bc:
                bc = c
                bh = h
                if time.time() - t0 < budget_s - 20:
                    h2, c2 = _cd(bh, dx, n, rng, 1000, t0, budget_s)
                    if c2 < bc:
                        bc = c2
                        bh = h2
                        break

    # Phase 4: Structured initializations
    if time.time() - t0 < budget_s - 150:
        freqs = [1, 2, 3, 4, 5, 6]
        for freq in freqs:
            if time.time() - t0 > budget_s - 60:
                break
            for amp in [0.2, 0.35, 0.5]:
                for ph in [0.0, np.pi / 2, np.pi]:
                    if time.time() - t0 > budget_s - 50:
                        break
                    xg = np.linspace(0, 2, n, endpoint=False)
                    hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * xg / 2.0 + ph)
                    hs = _proj(hs, tgt)
                    h, c = _fista(hs, 500, tgt, 3000, t0, budget_s)
                    if c < bc:
                        bc = c
                        bh = h
                        if time.time() - t0 < budget_s - 30:
                            h2, c2 = _cd(bh, dx, n, rng, 800, t0, budget_s)
                            if c2 < bc:
                                bc = c2
                                bh = h2

    # Phase 5: L-BFGS-B polish
    if time.time() - t0 < budget_s - 30:
        hf, cf = _lbfgs(bh, tgt, t0, budget_s)
        if cf < bc:
            bc = cf
            bh = hf
            h2, c2 = _cd(bh, dx, n, rng, 1500, t0, budget_s)
            if c2 < bc:
                bc = c2
                bh = h2

    # Phase 6: Active-set refinement
    if time.time() - t0 < budget_s - 25:
        hf, cf = _active_refine(bh, dx, n, tgt, t0, budget_s)
        if cf < bc:
            bc = cf
            bh = hf
            h2, c2 = _cd(bh, dx, n, rng, 1000, t0, budget_s)
            if c2 < bc:
                bc = c2
                bh = h2

    # Phase 7: Final intensive CD
    if time.time() - t0 < budget_s - 10:
        h, c = _cd(bh, dx, n, rng, 5000, t0, budget_s)
        if c < bc:
            bc = c
            bh = h

    # Phase 8: Final triple CD
    if time.time() - t0 < budget_s - 8:
        h, c = _cd3(bh, dx, n, rng, 2000, t0, budget_s)
        if c < bc:
            bc = c
            bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
