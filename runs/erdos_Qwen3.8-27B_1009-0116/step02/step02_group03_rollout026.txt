def _corr_max(h, n):

def _compute_c5(h, dx):

    return float(corr.max() * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Iterative repair
    for _ in range(100):
        if s > 0:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 1e-12:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff < 1e-12: break
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
        elif diff < -1e-12:
            order = np.argsort(h)
            for idx in order:

                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _c5_of(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _c5_scaled(h, n):
    return _c5_of(h) * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-14:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-10:
        diff = target - s
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def _grad_corr(h, n, alpha):
    """Analytical gradient of log-sum-exp smoothed max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (corr - corr.max())
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p /= p.sum()
    g = np.zeros(n)
    for m in range(2*n-1):
        if p[m] < 1e-14:
            continue
        k = m - (n-1)
        if k >= 0:
            g[:n-k] += p[m] * (1.0 - h[k:])
            g[k:] -= p[m] * h[:n-k]
        else:
            kk = -k
            g[kk:] += p[m] * (1.0 - h[:n-kk])
            g[:n-kk] -= p[m] * h[kk:]
    g -= g.mean()
    return g

def _smooth_obj(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter):
    target = n / 2.0
    def obj(x):
        return _smooth_obj(x, n, alpha)
    def grad(x):
        return _grad_corr(x, n, alpha)
    def con(x):
        return x.sum() - target
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5_raw(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _c5_scaled(h, n):
    return _c5_raw(h) * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-14:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
    return h

def _grad_corr(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sh = np.clip(alpha * (corr - corr.max()), -500, 500)
    p = np.exp(sh); p /= p.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += p[m] * (1.0 - h[k:])
            g[k:] -= p[m] * h[:n-k]
        else:
            kk = -k
            g[kk:] += p[m] * (1.0 - h[:n-kk])
            g[:n-kk] -= p[m] * h[kk:]
    g -= g.mean()
    return g

def _smooth_obj(x, n, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter):
    target = n / 2.0
    def obj(x):
        return _smooth_obj(x, n, alpha)
    def grd(x):
        return _grad_corr(x, n, alpha)
    def con(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, jac=grd, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_sweep(h, n, max_sweeps, budget_end, rng):
    h = h.copy()
    best = _c5_raw(h)
    sizes_coarse = [0.2, 0.1, 0.05, -0.2, -0.1, -0.05]
    sizes_fine = [0.02, 0.01, -0.02, -0.01]
    sizes_ultra = [0.005, 0.002, -0.005, -0.002]

    for sweep in range(max_sweeps):
        if time.time() > budget_end:
            break
        improved = False
        for sz in sizes_coarse + sizes_fine + sizes_ultra:
            for i in range(n):
                hi = h[i]
                if hi < 0.01 and sz > 0:
                    continue
                if hi > 0.99 and sz < 0:
                    continue
                for j in range(i + 1, n):
                    hj = h[j]
                    if hj < 0.01 and sz < 0:
                        continue
                    if hj > 0.99 and sz > 0:
                        continue
                    ni, nj = hi + sz, hj - sz
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    nm = _c5_raw(h)
                    if nm < best - 1e-14:
                        best = nm
                        improved = True
                        hi, hj = ni, nj
                    else:
                        h[i], h[j] = hi, hj
        if not improved:
            # Random fine perturbations
            for _ in range(200):
                i = rng.randint(n); j = rng.randint(n)
                if i == j:
                    continue
                d = rng.uniform(-0.003, 0.003)
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                nm = _c5_raw(h)
                if nm < best - 1e-14:
                    best = nm
                    improved = True
                else:
                    h[i], h[j] = ni - d, nj + d
            if not improved:
                break
    return h, best * 2.0 / n

    return starts
