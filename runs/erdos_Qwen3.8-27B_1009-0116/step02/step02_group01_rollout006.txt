            h[free] = 1.0
            diff -= room.sum()
            # Still need more, but we're at max capacity
            return h

        h[free] += scale * room
        h[free] = np.clip(h[free], 0.0, 1.0)

        else:
            h = h_new  # still move, momentum helps
            if c5_cur > _c5(best_h, dx) + 0.002:
                h = best_h.copy()
                c5_cur = best_c5
                v *= 0.5
        if elapsed_fn() > time_limit:
            break
    return best_h, best_c5

def _slsqp_phase(h, n, dx, alpha, elapsed_fn, time_limit, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = _corr(x)
        m = c.max()
        s = alpha * (c - m)
        s = np.clip(s, -50, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def jac(x):
        c = _corr(x)
        m = c.max()
        shifted = np.clip(alpha * (c - m), -50, 0)
        exps = np.exp(shifted)
        p = exps / exps.sum()
        g = np.zeros(n)
        for idx in range(len(c)):
            if p[idx] > 1e-10:
                g += p[idx] * _grad_one_lag(x, n, idx)
        return g

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project(res.x, n)
        c5 = _c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    dx = 2.0 / n

    h = _project(h_init, n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient with momentum
    if elapsed() < budget_s - 300:
        best_h, best_c5 = _subgradient_phase(best_h, n, dx, elapsed, budget_s - 300, n_iters=20000)

    # Phase 2: SLSQP polish with alpha=2000
    if elapsed() < budget_s - 200:
        hc, c5 = _slsqp_phase(best_h, n, dx, 2000, elapsed, budget_s - 200, maxiter=400)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 3: Local search (simulated annealing pairs)
    if elapsed() < budget_s - 150:
        best_h, best_c5 = _local_search(best_h, n, dx, rng,
                                        lambda: elapsed() > budget_s - 150, n_iters=8000)

    # Phase 4: Subgradient again (refined)
    if elapsed() < budget_s - 120:
        best_h, best_c5 = _subgradient_phase(best_h, n, dx, elapsed, budget_s - 120, n_iters=15000)

    # Phase 5: SLSQP with higher alpha
    if elapsed() < budget_s - 80:
        hc, c5 = _slsqp_phase(best_h, n, dx, 8000, lambda: time.time() - t0, budget_s - 80, maxiter=300)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 6: Coordinate descent
    if elapsed() < budget_s - 40:
        best_h, best_c5 = _coord_descent(best_h, n, dx,
                                         lambda: elapsed() > budget_s - 40, n_rounds=15)

    # Phase 7: Fine local search
    if elapsed() < budget_s - 10:
        best_h, best_c5 = _local_search(best_h, n, dx, rng,
                                        lambda: elapsed() > budget_s - 10, n_iters=6000)

    # Phase 8: Final SLSQP
    if elapsed() < budget_s - 5:
        hc, c5 = _slsqp_phase(best_h, n, dx, 20000, lambda: time.time() - t0, budget_s - 5, maxiter=200)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    best_h = _project(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
