from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(i+1, n):  # only i < j to avoid duplicates
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min > d_max - 1e-12:
                continue
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _slsqp(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj, h, args=(alpha,), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _binary_local_search(h, n, dx, max_iter=5000):
    """Round to binary, then flip pairs to reduce C5."""
    # Round to binary with sum = n/2
    target = n / 2.0
    bits = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    if bits.sum() > target:
        # Remove some ones (those closest to 0.5 from above)
        ones = np.where(bits == 1)[0]
        order = np.argsort(h[ones])  # smallest h among ones → flip first
        for idx in order:
            if bits.sum() == target:
                break
            bits[idx] = 0
    elif bits.sum() < target:
        zeros = np.where(bits == 0)[0]
        order = np.argsort(-h[zeros])  # largest h among zeros → flip first
        for idx in order:
            if bits.sum() == target:
                break
            bits[idx] = 1
    
    corr = _corr(bits)
    best_c5 = float(np.max(corr) * dx)
    
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if bits[i] != bits[j]:
                    # Flip: bits[i] -> 1-bits[i], bits[j] -> 1-bits[j]
                    # This maintains sum
                    new_bits = bits.copy()
                    new_bits[i] = 1.0 - new_bits[i]
                    new_bits[j] = 1.0 - new_bits[j]
                    nc = _corr(new_bits)
                    nc5 = float(np.max(nc) * dx)
                    if nc5 < best_c5 - 1e-14:
                        bits = new_bits
                        corr = nc
                        best_c5 = nc5
                        improved = True
        if not improved:
            break
    return bits, best_c5

                best_c5 = c5
                best_h = hc
            h, corr, _ = _sweep(best_h.copy(), n, dx, 25, 61, 0.01)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    # Phase 5: Subgradient steps
    if _elapsed() < budget_s - 100:
        hs = _subgrad(best_h.copy(), n, dx, 2000, 0.05)
        c5 = _c5(hs, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hs
        h, corr, _ = _sweep(best_h.copy(), n, dx, 25, 61, 0.01)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 6: Random restarts
    for sr in range(8):
        if _elapsed() > budget_s - 60:
            break
        h_p = _repair(best_h + 0.04 * rng.randn(n), n)
        hc = _slsqp(h_p, n, dx, target, bounds, 500, 300)
        h, corr, _ = _sweep(hc, n, dx, 25, 61, 0.015)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 7: Ultra-fine final sweep
    if _elapsed() < budget_s - 30:
        h, corr, _ = _sweep(best_h.copy(), n, dx, 30, 101, 0.004)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        h, corr, _ = _sweep(best_h.copy(), n, dx, 40, 131, 0.002)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 8: Final SLSQP
    if _elapsed() < budget_s - 5:
        try:
            hc = _slsqp(best_h, n, dx, target, bounds, 8000, 800)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
