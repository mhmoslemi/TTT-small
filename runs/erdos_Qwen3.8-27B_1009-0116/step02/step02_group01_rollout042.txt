import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lp_linearize(h, n):
    """Solve LP: min t s.t. A@h <= t, sum(h)=n/2, 0<=h<=1 using linearized c[k]."""
    target = n / 2.0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        for i in range(lo, hi + 1):
            j = i + k
            if 0 <= j < n and 0 <= i < n:
                A[k, j] += (1.0 - h[i])
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return _repair(res.x[:n], n)
    except Exception:
        pass
    return h

def _slsqp_optimize(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_descent(h, n, dx, max_passes):
    h = h.copy()
    c5_cur = _c5(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02])
    for _ in range(max_passes):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-14:
                            c5_cur = c
                            imp = True
                        else:
                            h[i], h[j] = ni - d, nj + d
        if not imp:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    dx = 2.0 / n
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Successive LP linearization
    h_cur = best_h.copy()
    for it in range(300):
        if elapsed() > budget_s - 400:
            break
        h_new = _lp_linearize(h_cur, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new.copy()
        # Add tiny noise to avoid cycling
        h_cur = _repair(h_cur + 1e-6 * rng.randn(n), n)

    # Phase 2: SLSQP polish with increasing alpha
    for alpha in [1000, 5000, 20000, 50000]:
        if elapsed() > budget_s - 300:
            break
        h_try = _slsqp_optimize(best_h, n, alpha, 400, 1e-15)
        c5_try = _c5(h_try, dx)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 3: Basin hopping (restart + LP + SLSQP)
    if elapsed() < budget_s - 200:
        for bh_it in range(20):
            if elapsed() > budget_s - 200:
                break
            scale = 0.08 * (1.0 + 0.5 * np.exp(-bh_it / 5.0))
            h_pert = _repair(best_h + scale * rng.randn(n), n)
            # Quick LP iterations
            for _ in range(20):
                if elapsed() > budget_s - 200:
                    break
                h_pert = _lp_linearize(h_pert, n)
            # SLSQP polish
            h_pert = _slsqp_optimize(h_pert, n, 5000, 200, 1e-14)
            c5_pert = _c5(h_pert, dx)
            if c5_pert < best_c5:
                best_c5, best_h = c5_pert, h_pert

    # Phase 4: More LP iterations from best
    if elapsed() < budget_s - 150:
        h_cur = best_h.copy()
        for it in range(500):
            if elapsed() > budget_s - 150:
                break
            h_new = _lp_linearize(h_cur, n)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = _repair(h_new + 1e-7 * rng.randn(n), n)

    # Phase 5: Coordinate descent
    if elapsed() < budget_s - 60:
        h_cd, c5_cd = _coord_descent(best_h, n, dx, 50)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Phase 6: Final SLSQP + LP alternating
    if elapsed() < budget_s - 20:
        for _ in range(30):
            if elapsed() > budget_s - 20:
                break
            best_h = _slsqp_optimize(best_h, n, 30000, 200, 1e-16)
            c5_s = _c5(best_h, dx)
            if c5_s < best_c5:
                best_c5 = c5_s
            best_h = _lp_linearize(best_h, n)
            c5_l = _c5(best_h, dx)
            if c5_l < best_c5:
                best_c5 = c5_l

    # Phase 7: Ultra-fine coordinate descent
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([0.0005, 0.001, 0.002, -0.0005, -0.001, -0.002])
        count = 0
        while elapsed() < budget_s - 3 and count < 20000:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                count += 1
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
            count += 1

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
