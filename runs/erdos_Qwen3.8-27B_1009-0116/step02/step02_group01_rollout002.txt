    return np.max(_corr(h)) * dx

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    es = np.exp(s)
    return m + np.log(np.sum(es)) / alpha + penalty * (np.sum(x) - target) ** 2

def _smooth_obj_grad(x, alpha, penalty, target):

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)

    w = es / np.sum(es)
    f = m + np.log(np.sum(es)) / alpha + penalty * (np.sum(x) - target) ** 2
    g = 2.0 * penalty * (np.sum(x) - target) * np.ones(n)

        if wk < 1e-12:

        lag = k - (n - 1)

            if i_min <= i1 <= i_max and 0 <= j < n:

            i2 = j
            if i_min <= i2 <= i_max and 0 <= j + lag < n:
                g[j] -= wk * x[j + lag]

    if s < 1e-12:

        if abs(diff) > 1e-12:

                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_obj(x, alpha, pen, target):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    es = np.exp(s)
    return m + np.log(np.sum(es)) / alpha + pen * (np.sum(x) - target) ** 2


def _smooth_grad(x, alpha, pen, target):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    es = np.exp(s)
    w = es / np.sum(es)
    f = m + np.log(np.sum(es)) / alpha + pen * (np.sum(x) - target) ** 2
    dp = 2.0 * pen * (np.sum(x) - target)
    g = dp * np.ones(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            i_min, i_max = 0, n - 1 - lag
        else:
            i_min, i_max = -lag, n - 1
        for j in range(n):
            i1 = j - lag
            if i_min <= i1 <= i_max:
                g[j] += wk * (1.0 - x[i1])
            i2 = j
            if i_min <= i2 <= i_max and 0 <= j + lag < n:
                g[j] -= wk * x[j + lag]
    return f, g


def _worst_lag_grad(h, n, k_worst):
    lag = k_worst - (n - 1)
    if lag >= 0:
        i_min, i_max = 0, n - 1 - lag
    else:
        i_min, i_max = -lag, n - 1
    g = np.zeros(n)
    for j in range(n):
        i1 = j - lag
        if i_min <= i1 <= i_max:
            g[j] += (1.0 - h[i1])
        i2 = j
        if i_min <= i2 <= i_max and 0 <= j + lag < n:
            g[j] -= h[j + lag]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h
