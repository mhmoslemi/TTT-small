def _correlation_gradient(p, h, n):
    """Compute gradient of weighted sum of correlations w.r.t. h.
    p: weights for each correlation lag (length 2n-1)
    h: the function values (length n)
    returns: gradient (length n)
    """
    grad = np.zeros(n)
    for m in range(2*n - 1):
        w = p[m]
        if w < 1e-15:
            continue
        # For lag m: corr[m] = sum_i h[i] * (1-h[i-m+n-1])
        # d(corr[m])/d(h[j]) = (1-h[j-m+n-1]) - h[j+m-n+1] (where valid)

            idx1 = j - m + n - 1  # where h[j] is the first factor
            idx2 = j + m - n + 1  # where h[j] is in the second factor (1-h)
            g = 0.0
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[j] += w * g
    return grad

```python
def _compute_grad(p, h, N):
    """Compute gradient of smoothed max correlation w.r.t. h.
    p: softmax weights over the 2N-1 correlation values
    h: function values (length N)
    """
    grad = np.zeros(N)
    for m in range(2*N - 1):
        w = p[m]
        if w < 1e-16:
            continue
        shift = m - N + 1
        # First part: (1-h[j-shift]) for valid j-shift
        # j-shift ∈ [0, N-1] → j ∈ [shift, shift+N-1] ∩ [0, N-1]
        j_lo = max(0, shift)
        j_hi = min(N-1, shift + N - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1) - shift
            grad[j_lo:j_hi+1] += w * (1.0 - h[idx])
        # Second part: -h[j+shift] for valid j+shift
        # j+shift ∈ [0, N-1] → j ∈ [-shift, N-1-shift] ∩ [0, N-1]
        j_lo2 = max(0, -shift)
        j_hi2 = min(N-1, N - 1 - shift)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1) + shift
            grad[j_lo2:j_hi2+1] -= w * h[idx2]
    return grad
```

This is O(N * (2N-1)) = O(N²) which for N=70 is about 10000 operations per gradient evaluation. Very fast.

    h = h.copy()
    topK = 5

        corr = _corr(h)
        c5_val = corr.max() * dx
        if c5_val < best - 1e-12:
            best = c5_val
        # Find top lags
        top_idx = np.argsort(corr)[-topK:]
        improved = False
        for _ in range(iters_per_lag):
            if time.time() > deadline:
                return h, best
            # Pick a random lag from top
            m = top_idx[rng.randint(topK)]
            shift = m - N + 1
            # Pick a random valid pair for this lag
            # Pairs: (i, i+shift) for valid i

            if h[i] > h[j]:

                h[i] -= d
                h[j] += d
            else:
                d = 0.001 * rng.random()
                h[j] -= d
                h[i] += d

                    h[j] -= d
