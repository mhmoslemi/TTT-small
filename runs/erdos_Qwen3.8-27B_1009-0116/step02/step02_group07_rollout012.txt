def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    """Compute D[m] for pair (i,j). D[m] = ci[m] - cj[m]."""
    D = np.zeros(2*N - 1)
    for m in range(2*N - 1):

        p2 = i + m - (N-1)
        ci = 0.0
        if 0 <= p1 < N:
            ci += 1.0 - h[p1]
        if 0 <= p2 < N:
            ci -= h[p2]
        p3 = j - m + (N-1)
        p4 = j + m - (N-1)
        cj = 0.0
        if 0 <= p3 < N:
            cj += 1.0 - h[p3]
        if 0 <= p4 < N:
            cj -= h[p4]
        D[m] = ci - cj
    return D

    lo = u_lo
    hi = u_hi
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        vals = r + mid * D
        m_idx = np.argmax(vals)
        # Subgradient at mid: D[m_idx] (if unique max) or interval
        # Check for ties
        max_val = vals[m_idx]
        ties = np.where(np.isclose(vals, max_val, atol=1e-12))[0]

            slope_hi = D[ties].max()
            if slope_lo <= 0 and slope_hi >= 0:
                # At minimum
                break
            elif slope_lo > 0:
                hi = mid
            else:
                lo = mid
        else:
            if D[m_idx] > 0:
                hi = mid
            elif D[m_idx] < 0:
                lo = mid
            else:
                break

    candidates = [lo, hi, (lo+hi)/2.0]
    best_u = lo
    best_val = np.inf
    for u in candidates:
        v = np.max(r + u * D)
        if v < best_val:
            best_val = v
            best_u = u
    return best_u, best_val

                r = r + best_u * D
                current_max = r.max()
                improved = True

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            a = min(abs(diff), room)
            if diff > 0:
                h[idx] += a
                diff -= a
            else:
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _compute_D(h, i, j, N):
    D = np.zeros(2 * N - 1)
    Nm1 = N - 1
    for m in range(2 * N - 1):
        p1 = i - m + Nm1
        p2 = i + m - Nm1
        ci = 0.0
        if 0 <= p1 < N:
            ci += 1.0 - h[p1]
        if 0 <= p2 < N:
            ci -= h[p2]
        p3 = j - m + Nm1
        p4 = j + m - Nm1
        cj = 0.0
        if 0 <= p3 < N:
            cj += 1.0 - h[p3]
        if 0 <= p4 < N:
            cj -= h[p4]
        D[m] = ci - cj
    return D

def _fast_cd(h, n, dx, n_grid, t_limit, t_start, max_passes=5000):
    N = n
    best_c = _c5_val(h, dx)
    r = _corr(h)
    cur_max = r.max()
    for _ in range(max_passes):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(N):
            for j in range(i + 1, N):
                s = h[i] + h[j]
                lo_t = max(0.0, s - 1.0)
                hi_t = min(1.0, s)
                if hi_t - lo_t < 1e-10:
                    continue
                D = _compute_D(h, i, j, N)
                if np.max(np.abs(D)) < 1e-15:
                    continue
                u_lo = lo_t - h[i]
                u_hi = hi_t - h[i]
                u_vals = np.linspace(u_lo, u_hi, n_grid)
                # Vectorized evaluation
                new_maxes = np.max(r[np.newaxis, :] + u_vals[:, np.newaxis] * D[np.newaxis, :], axis=1)
                best_k = np.argmin(new_maxes)
                if new_maxes[best_k] < cur_max - 1e-13:
                    u = u_vals[best_k]
                    h[i] += u
                    h[j] -= u
                    r += u * D
                    cur_max = r.max()
                    improved = True
        if not improved:
            break
        if cur_max * dx < best_c:
            best_c = cur_max * dx
    return h, best_c

    best_h = h.copy()
    best_c = _c5_val(h, dx)
    cur_h = h.copy()
    r = _corr(cur_h)
    cur_max = r.max()
    cur_c = cur_max * dx
    T = 0.05

        i = rng.randint(N)
