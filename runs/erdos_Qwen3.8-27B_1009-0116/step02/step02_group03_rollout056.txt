```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > 1e-12:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _smooth_max_grad(corr, T):
    """Compute smooth max and its gradient weights."""
    s = T * (corr - np.max(corr))
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    val = np.max(corr) + np.log(e.sum()) / T
    return val, w

def _adam_step(h, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad**2
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps), m, v

def _adam_optimize(h0, n, budget_end, T_init=50, T_end=2000, n_iter=5000, lr=0.1):
    h = h0.copy()
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() > budget_end:
            break
        # Schedule T
        frac = it / max(1, n_iter - 1)
        T = T_init + (T_end - T_init) * frac
        # Schedule lr
        lr_t = lr / (1 + 0.01 * it)
        
        corr = _corr(h)
        val, w = _smooth_max_grad(corr, T)
        
        # Compute gradient: sum_k w_k * dcorr[k]/dh[j]
        grad = np.zeros(n)
        ncorr = len(corr)  # 2n-1
        for k in range(ncorr):
            if w[k] < 1e-10:
                continue
            shift = k - (n - 1)
            # First term: (1-h[j-s]) for valid j
            # valid j: max(0, s) <= j < min(n, n+s) if s>=0; 0 <= j < n+s if s<0
            j_lo = max(0, shift)
            j_hi = min(n, n + shift)
            if j_lo < j_hi:
                grad[j_lo:j_hi] += w[k] * (1.0 - h[j_lo - shift:j_hi - shift])
            # Second term: -h[j+s] for valid j
            # valid j: max(0, -s) <= j < min(n, n-s)
            j_lo2 = max(0, -shift)
            j_hi2 = min(n, n - shift)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= w[k] * h[j_lo2 + shift:j_hi2 + shift]
        
        h, m, v = _adam_step(h, grad, m, v, it + 1, lr_t)
        h = _repair(h, n)
        
        c5_now = _c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
        if time_left < 20:
            break
        dx = 2.0 / n

        starts = []
        if init_h is not None and init_n == n:
            starts.append(init_h.copy())
        starts.extend(_make_starts(n, seed, rng))

        time_per_start = time_left / (len(starts) + 1) * 0.85

        for si, h0 in enumerate(starts):
            t_phase = time.time()
            t_left = total_time - (t_phase - t0) - 20
            if t_left < 15:
                break

            h = h0.copy()
            _try_update(h, n)

            # Adam phase 1: coarse
            t_adam1 = min(t_phase + time_per_start * 0.45, t0 + total_time - 25)
            h, _ = _adam_opt(h, n, t_adam1, n_iters=4000, T0=30, T1=800, lr0=0.2)
            _try_update(h, n)

            # Coord descent
            t_cd_end = min(t_adam1 + 30, t0 + total_time - 25)
            h, _ = _coord_descent(h, n, dx, t_cd_end, n_rounds=150)
            _try_update(h, n)

            # Adam phase 2: fine
            t_adam2 = min(t_phase + time_per_start * 0.75, t0 + total_time - 20)
            t_left2 = total_time - (time.time() - t0) - 20
            if t_left2 > 15:
                h, _ = _adam_opt(h, n, min(t_adam2, time.time() + t_left2 * 0.5), n_iters=3000, T0=500, T1=4000, lr0=0.06)
                _try_update(h, n)

            # Pair descent
            t_pd_end = min(time.time() + 20, t0 + total_time - 18)
            if time.time() < t_pd_end:
                h, _ = _pair_descent(h, n, dx, t_pd_end, n_rounds=60, rng=rng)
                _try_update(h, n)

            # Final coord on best
            h, _ = _coord_descent(h, n, dx, min(time.time() + 15, t0 + total_time - 15), n_rounds=100)
            _try_update(h, n)

    # Final polish on global best
    if _g_best_h is not None:
        t_fin = time.time()
        dx_b = 2.0 / _g_best_n
        t_left_fin = total_time - (t_fin - t0) - 5
        if t_left_fin > 20:
            h, _ = _adam_opt(_g_best_h, _g_best_n, min(t_fin + t_left_fin * 0.4, t0 + total_time - 10),
                             n_iters=2000, T0=2000, T1=8000, lr0=0.03)
            _try_update(h, _g_best_n)
            t_left_fin2 = total_time - (time.time() - t0) - 3
            if t_left_fin2 > 10:
                h, _ = _coord_descent(_g_best_h, _g_best_n, dx_b, t0 + total_time - 3, n_rounds=200)
                _try_update(h, _g_best_n)
                h, _ = _pair_descent(_g_best_h, _g_best_n, dx_b, t0 + total_time - 2, n_rounds=50, rng=rng)
                _try_update(h, _g_best_n)

    if _g_best_h is None:
        nn = 70
        _g_best_h = _repair(np.full(nn, 0.5), nn)
        _g_best_c5 = _c5(_g_best_h, 2.0 / nn)
        _g_best_n = nn

    _g_best_h = _repair(_g_best_h, _g_best_n)
    _g_best_c5 = _c5(_g_best_h, 2.0 / _g_best_n)
    return (_g_best_h, _g_best_c5, _g_best_n)
```
