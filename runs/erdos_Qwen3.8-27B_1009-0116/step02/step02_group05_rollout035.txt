import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx]); h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx]); h[idx] -= sub; diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _compute_a(h):
    n = len(h)
    a = np.zeros((n, 2 * n - 1))
    for si in range(2 * n - 1):
        k = si - (n - 1)
        if k >= 0:
            a[:n-k, si] = 1.0 - h[k:]
            a[k:, si] -= h[:n-k]
        else:
            ak = -k
            a[ak:, si] = 1.0 - h[:n-ak]
            a[:n-ak, si] -= h[ak:]
    return a

def _pair_cd_sweep(h, n, dx, rng, deadline, max_sweeps=200):
    c = _corr(h)
    best = h.copy()
    best_c5 = float(np.max(c) * dx)
    sweep = 0
    stall = 0
    while sweep < max_sweeps and time.time() < deadline:
        a = _compute_a(h)
        c = _corr(h)
        improved_sweep = False
        indices = np.arange(n)
        if sweep % 3 == 1:
            indices = indices[::-1]
        elif sweep % 3 == 2:
            indices = rng.permutation(n)
        for i in indices:
            for j in indices:
                if i == j:
                    continue
                da = a[i] - a[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                lo, hi = d_lo, d_hi
                f_lo = float(np.max(c + da * lo))
                f_hi = float(np.max(c + da * hi))
                for _ in range(45):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = float(np.max(c + da * m1))
                    f2 = float(np.max(c + da * m2))
                    if f1 <= f2:
                        hi, f_hi = m2, f2
                    else:
                        lo, f_lo = m1, f1
                delta = (lo + hi) / 2.0
                f_new = float(np.max(c + da * delta))
                f_old = float(np.max(c))
                if f_new < f_old - 1e-13:
                    h[i] += delta
                    h[j] -= delta
                    c = _corr(h)
                    improved_sweep = True
        if improved_sweep:
            nc = float(np.max(c) * dx)
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                best = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
        sweep += 1
    return best, best_c5

def _annealed_local_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = float(np.max(_corr(h)) * dx)
    cur = h.copy()
    cur_c5 = best_c5
    T = 0.003
    T_min = 0.0001
    steps = np.array([0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2,
                      -0.01, -0.02, -0.03, -0.05, -0.08, -0.12, -0.2])
    stall = 0
    trial = 0
    while time.time() < deadline:
        trial += 1
        if trial % 500 == 0:
            T = max(T_min, T * 0.95)
            if T <= T_min * 1.01:
                T = T_min
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = float(rng.choice(steps))
        ni, nj = cur[i] + d, cur[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        cur[i], cur[j] = ni, nj
        nc = float(np.max(_corr(cur)) * dx)
        delta_c = nc - cur_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            cur[i], cur[j] = cur[i] - d, cur[j] + d
            stall += 1
        if stall > 2000:
            cur = best.copy()
            cur_c5 = best_c5
            T = max(T_min, T * 1.5)
            stall = 0
    return best, best_c5

def _slsqp_polish(h, n, dx, alpha, maxiter):
    def _smax(x):
        m = np.max(x)
        s = np.clip(alpha * (x - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _obj(z):
        return float(_smax(_corr(z)) * dx)
    def _obj_grad(z):
        c = _corr(z)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s) / np.sum(np.exp(s))
        g = np.zeros(n)
        for si in range(2 * n - 1):
            k = si - (n - 1)
            wt = w[si] * dx
            if abs(wt) < 1e-15:
                continue
            if k >= 0:
                g[:n-k] += wt * (1.0 - z[k:])
                g[k:] -= wt * z[:n-k]
            else:
                ak = -k
                g[ak:] += wt * (1.0 - z[:n-ak])
                g[:n-ak] -= wt * z[ak:]
        return g
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda z: z.sum() - target,
             'jac': lambda z: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n), _c5(res.x, dx)
    except Exception:
        return h.copy(), _c5(h, dx)

def _fine_local_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = float(np.max(_corr(h)) * dx)
    steps = np.array([0.002, 0.004, 0.008, 0.015, 0.025, 0.04,
                      -0.002, -0.004, -0.008, -0.015, -0.025, -0.04])
    stall = 0
    while time.time() < deadline:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        for d in steps:
            ni, nj = best[i] + d, best[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = float(np.max(_corr(best)) * dx)
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                stall = 0
            else:
                best[i], best[j] = oi, oj
            if stall == 0:
                break
        else:
            stall += 1
            if stall > 100:
                stall = 0
                for d in steps:
                    ni, nj = best[i] + d, best[j] - d
                    if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                        continue
                    oi, oj = best[i], best[j]
                    best[i], best[j] = ni, nj
                    nc = float(np.max(_corr(best)) * dx)
                    if nc < best_c5 - 1e-14:
                        best_c5 = nc
                        stall = 0
                    else:
                        best[i], best[j] = oi, oj
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    n = len(initial_h_values)
