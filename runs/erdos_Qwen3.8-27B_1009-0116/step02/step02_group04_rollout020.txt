def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _grad_smax(h, alpha, top_k=12):
    n = len(h)
    c = _corr(h)
    nc = len(c)
    if top_k < nc:
        top_idx = np.argpartition(c, nc - top_k)[nc - top_k:]
        ref = c[top_idx[0]]
        w = np.zeros(nc)
        for i in top_idx:
            w[i] = np.exp(alpha * (c[i] - ref))
    else:
        ref = np.max(c)
        w = np.exp(np.clip(alpha * (c - ref), -500, 500))
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    offset = n - 1
    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - offset
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
        else:
            sp = -s
            if sp < n:
                grad[:n - sp] += wk * (1.0 - h[sp:])
                grad[sp:] -= wk * h[:n - sp]
    return grad

def _proj(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    return _repair(h, n)

def _grad_descent(h, n, dx, deadline, alpha=50, top_k=12, lr0=0.01):
    """Backtracking line search gradient descent on smax."""
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    lr = lr0
    it = 0
    while time.time() < deadline:
        it += 1
        g = _grad_smax(h, alpha, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break

                h = h_try
                c_new = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h.copy()
                break
            lr *= 0.7
        else:
            lr *= 0.3
        # Adapt alpha
        if it % 200 == 0:
            alpha *= 1.5
            if alpha > 500:
                alpha = 500
    return best_h, best_c5

        return h, _c5(h, dx)
    target = n / 2.0

        c = _corr(x)
        m = np.max(c)
        s = a * (c - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / a
    
    def obj_grad(x, a=alpha):
        g = _grad_smax(x, a)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        if c5 < _c5(h, dx):
            return hc, c5
    except Exception:
        pass
    return h, _c5(h, dx)

    h_alt = _repair(h + 0.06 * rng.randn(n), n)
    c5_alt = _c5(h_alt, dx)

    # Phase 1: Gradient descent on initial (200s)
    dl = t0 + min(200, budget_s - 500)
    for alpha in [30, 70, 150]:
        if time.time() > dl or rem() < 400:
            break
        h, c5 = _grad_descent(h, n, dx, dl, alpha=alpha, top_k=12, lr0=0.03)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Multi-resolution (200s)
    if rem() > 350:
        dl = min(t0 + budget_s - 350, time.time() + 200)
        h, c5 = _multi_resolution(best_h, n, dx, rng, dl)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Gradient descent from best (200s)
    if rem() > 200:
        dl = min(t0 + budget_s - 200, time.time() + 200)
        for alpha in [100, 200, 400]:
            if time.time() > dl or rem() < 150:
                break
            h, c5 = _grad_descent(best_h, n, dx, dl, alpha=alpha,
                                   top_k=10, lr0=0.01)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Targeted local search (250s)
    if rem() > 100:
        dl = min(t0 + budget_s - 80, time.time() + 250)
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl,
                                         top_k=5, scale0=0.03)

    # Phase 5: SLSQP polish (80s)
    if rem() > 70:
        dl = time.time() + min(70, rem() - 30)
        for alpha in [200, 500]:
            if time.time() > dl:
                break
            best_h, best_c5 = _slsqp_refine(best_h, n, dx, alpha, 400, dl)

    # Phase 6: Final micro local search (50s)
    if rem() > 30:
        dl = time.time() + min(40, rem() - 10)
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl,
                                         top_k=4, scale0=0.01)

    # Phase 7: Tiny SLSQP finish
    if rem() > 15:
        best_h, best_c5 = _slsqp_refine(best_h, n, dx, 800, 200,
                                         time.time() + min(10, rem() - 5))

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
