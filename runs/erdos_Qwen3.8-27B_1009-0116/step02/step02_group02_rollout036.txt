```python
def _bottleneck_grad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    # Use the top 10% of lags
    threshold = np.percentile(c, 90)
    active = c > threshold
    w = c[active] - c[active].min() + 1.0
    w /= w.sum()
    g = np.zeros(n)
    active_indices = np.where(active)[0]
    for ai in active_indices:
        wi = w[ai]
        lag = ai - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += wi * (1.0 - h[idx + lag])
            g[idx + lag] -= wi * h[idx]
        else:
            L = -lag
            idx = np.arange(L, n)
            g[idx] += wi * (1.0 - h[idx - L])
            g[idx - L] -= wi * h[idx]
    return g
```

```python
    n = len(y)
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0-x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0-x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
            if abs(diff) < 1e-13: break
    return x
```

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = float(y.min())-1.0, float(y.max())+1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if float(np.clip(y-mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y-0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0-x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0-x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
            if abs(diff) < 1e-13: break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0-x, mode='full')
    m = float(c.max())
    s = np.clip(a*(c-m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum()))/a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0-x, mode='full')
    m = float(c.max())
    w = np.exp(np.clip(a*(c-m), -500.0, 500.0))
    w /= float(w.sum())
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-13: continue
        lag = k-(n-1)
        if lag >= 0:
            idx = np.arange(n-lag)
            g[idx] += w[k]*(1.0-x[idx+lag])
            g[idx+lag] -= w[k]*x[idx]
        else:
            L = -lag
            idx = np.arange(L, n)
            g[idx] += w[k]*(1.0-x[idx-L])
            g[idx-L] -= w[k]*x[idx]
    return g

def _fista(h0, alpha, tgt, iters=3000):
    n, dx = len(h0), 2.0/len(h0)
    x = h0.copy(); z = h0.copy(); t = 1.0
    bx, bc = x.copy(), _c5(x, dx)
    step = 1.0/(4.0*alpha*n+4.0)
    for _ in range(iters):
        gz = _sgrad(z, alpha)
        xn = _proj(z-step*gz, tgt)
        for _ in range(8):
            fn = _sobj(xn, alpha)
            qd = _sobj(z, alpha)+float(gz.dot(xn-z))+0.5/step*float((xn-z).dot(xn-z))
            if fn <= qd+1e-10: break
            step *= 0.5; xn = _proj(z-step*gz, tgt)
        tn = 0.5*(1.0+np.sqrt(1.0+4.0*t*t))
        z = _proj(xn+(t-1.0)/tn*(xn-x), tgt)
        x = xn; t = tn
        c = _c5(x, dx)
        if c < bc: bc, bx = c, x.copy()
    return bx, bc

def _lbfgs_obj(x, a, pen, tgt):
    return _sobj(x, a) + pen*(float(x.sum())-tgt)**2

def _lbfgs_grad(x, a, pen, tgt):
    return _sgrad(x, a) + 2.0*pen*(float(x.sum())-tgt)

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, tgt):
    return float(x.sum()) - tgt

def _lbfgs(h, tgt, a, pen=500.0, maxit=800):
    bnds = [(0.0, 1.0)]*len(h)
    try:
        r = minimize(_lbfgs_obj, h, args=(a, pen, tgt), jac=_lbfgs_grad,
                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-12})
        return _proj(r.x, tgt)
    except Exception:
        return h

def _slsqp(h, tgt, a, maxit=200):

    try:
        r = minimize(_slsqp_obj, h, args=(a,), method='SLSQP', bounds=bnds,

                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _proj(r.x, tgt)
    except Exception:
        return h
