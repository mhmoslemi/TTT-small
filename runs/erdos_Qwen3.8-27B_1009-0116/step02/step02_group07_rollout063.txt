def _min_pl(beta, alpha):
    """Minimize max_m(beta[m]+alpha[m]*t) over t in [0,1]."""
    if np.all(alpha >= 0):
        return 0.0
    if np.all(alpha <= 0):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = beta + alpha * mid
        mval = vals.max()
        active = vals >= mval - 1e-12
        amin = alpha[active].min()
        amax = alpha[active].max()
        if amin >= -1e-14:
            hi = mid
        elif amax <= 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

    return x.sum() - target

def _smooth_obj(x, alpha, n):

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter, ftol):

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha, n), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_pass(h, n, dx, order):
    hi_c5 = _c5(h, dx)
    improved = False
    for i in order:
        orig = h[i]
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = orig
        t_opt = _min_pl(c0, c1 - c0)
        if abs(t_opt - orig) > 1e-10:

            if nc < hi_c5 - 1e-13:
                hi_c5 = nc
                improved = True
            else:
                h[i] = orig
    return h, hi_c5, improved
