        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lse_obj_grad(h, alpha):
    """Objective and gradient for (1/alpha)*log(sum exp(alpha*c_k))."""
    n = len(h)
    c = _corr(h)
    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)
    sum_exp = exp_ac.sum()
    obj = (ac_max + np.log(sum_exp)) / alpha
    p = exp_ac / sum_exp  # softmax weights
    
    # Gradient: sum_k p_k * dc_k/dh_j
    # dc_k/dh_j = (1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)
    # where lag = k - (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        for j in range(n):
            g = 0.0
            jp = j + lag
            if 0 <= jp < n:
                g += 1.0 - h[jp]
            jm = j - lag
            if 0 <= jm < n:
                g -= h[jm]
            grad[j] += pk * g
    return obj, grad

def _slsqp_solve(h0, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj_fun(x):
        return _lse_obj_grad(x, alpha)[0]
    
    def obj_grad(x):
        return _lse_obj_grad(x, alpha)[1]
    
    res = minimize(obj_grad, h0, method='SLSQP', jac=True,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def _proj_grad(h, n, alpha, lr, n_iter, lam=0.0):
    """Projected gradient descent with backtracking line search."""
    target = n / 2.0
    h = h.copy()
    best = h.copy()
    best_obj = _c5(h, 2.0/n)
    for it in range(n_iter):
        obj, grad = _lse_obj_grad(h, alpha)
        if lam > 0:
            grad = grad + 2.0 * lam * (h.sum() - target)
        # Projected gradient step
        h_new = h - lr * grad
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, 2.0/n)
        if c5_new < best_obj - 1e-15:
            h = h_new
            best = h.copy()
            best_obj = c5_new
    return best, best_obj

def _coord_descent(h, n, deltas, n_passes):
    """Coordinate descent with pair moves."""
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _c5(h, dx)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                h[i] = hi
        if not improved:
            break
    return h, c5_cur

def _sim_anneal(h, n, rng, t_start, t_end, n_iter):
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _c5(h, dx)
    best = h.copy()
    best_c5 = c5_cur
    T = t_start
    cool = (t_end / t_start) ** (1.0 / max(n_iter, 1))
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.05, 0.05)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_new = _c5(h, dx)
        if c5_new < c5_cur or rng.rand() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        else:
            h[i], h[j] = hi, hj
        T *= cool
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_ALPHA = 1.0
_SUM_TARGET = 0.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _lse_obj_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    ac = alpha * c
    acm = ac.max()
    eac = np.exp(ac - acm)
    se = eac.sum()
    obj = (acm + np.log(se)) / alpha
    p = eac / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        for j in range(n):
            g = 0.0
            jp = j + lag
            if 0 <= jp < n:
                g += 1.0 - h[jp]
            jm = j - lag
            if 0 <= jm < n:
                g -= h[jm]
            grad[j] += pk * g
    return obj, grad

def _obj_cb(x):
    return _lse_obj_grad(x, _ALPHA)[0]

def _grad_cb(x):
    return _lse_obj_grad(x, _ALPHA)[1]

def _sum_con(x):
    return np.sum(x) - _SUM_TARGET

def _sum_con_jac(x):
    return np.ones(len(x))

def _slsqp_round(h0, n, alpha, maxiter, ftol, bounds, cons):
    global _ALPHA
    _ALPHA = alpha
    res = minimize(_obj_cb, h0, jac=_grad_cb, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def _proj_grad_phase(h, n, dx, target, alpha, lr, n_iter, t0, tl):
    global _ALPHA
    _ALPHA = alpha
    h = h.copy()
    best = h.copy()
    best_c5 = _c5(h, dx)
    for _ in range(n_iter):
        if time.time() > tl:
            break
        _, grad = _lse_obj_grad(h, alpha)
        grad += 2.0 * 200.0 * (h.sum() - target)
        h2 = _repair(h - lr * grad, n)
        c52 = _c5(h2, dx)
        if c52 < _c5(h, dx) - 1e-15:
            h = h2
            if c52 < best_c5:
                best_c5 = c52
                best = h.copy()
    return best, best_c5

def _coord_search(h, n, dx, deltas, n_passes, t0, tl):
    h = h.copy()
    c5_cur = _c5(h, dx)
    for _pass in range(n_passes):
        if time.time() > tl:
            break
        imp = False
        for i in range(n):
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            hi, hj = ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                h[i] = hi
        if not imp:
            break
    return h, c5_cur
