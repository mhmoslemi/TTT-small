    # Phase 6: Nelder-Mead on top-impact coordinates
    if elapsed() < budget_s - 40:
        c_arr = _corr(best_h) * dx
        m = c_arr.max()
        active_idx = np.where(c_arr > m - m * 0.02)[0]
        # Find h indices that appear most in active correlations
        impact = np.zeros(n)
        for k in active_idx:
            sh = k - n + 1
            lo = max(0, -sh)
            hi = min(n, n - sh)
            impact[lo:hi] += 1
            lo2 = max(0, sh)
            hi2 = min(n, n + sh)
            impact[lo2:hi2] += 1
        top_idx = np.argsort(impact)[-15:]
        if len(top_idx) >= 4:
            h = _nm_subset(best_h, n, dx, top_idx, t0, budget_s - 10)
            c = _c5_val(h, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Phase 7: Multi-restart (targeted)
    sigmas = [0.04, 0.08, 0.12, 0.03, 0.06, 0.1, 0.15]
    for si, sig in enumerate(sigmas):
        if elapsed() > budget_s - 30:
            break
        h_p = best_h + sig * rng.randn(n)
        h_p = _repair(h_p, n)
        h_p = _slsqp(h_p, n, dx, 400, 300)
        c = _c5_val(h_p, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_p.copy()
            h2, c2 = _cd_loop(h_p, n, dx, min(budget_s - 25, elapsed() + 50), t0, 200)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

    # Phase 8: Final polish
    if elapsed() < budget_s - 10:
        h = _slsqp(best_h, n, dx, 5000, 400)
        c = _c5_val(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        h, c = _cd_loop(best_h, n, dx, budget_s - 5, t0, 200)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        h = _subgrad_polish(best_h, n, dx, t0, budget_s - 2, 2000)
        c = _c5_val(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
