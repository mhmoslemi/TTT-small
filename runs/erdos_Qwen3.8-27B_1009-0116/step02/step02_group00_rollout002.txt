    for it in range(500000):
        if time.time() > t_end:
            break
        T = max(T * cool_rate, T_min)
        move_type = rng.randint(0, 3)
        
        if move_type == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = (rng.rand() - 0.5) * 2.0 * max(T * 50, 0.005)
            old_i, old_j = h[i], h[j]
            h[i] = old_i + delta
            h[j] = old_j - delta
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i], h[j] = old_i, old_j
                continue
        elif move_type == 1:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            k = rng.randint(n)
            if k == i or k == j:
                continue
            old_i = h[i]
            delta = (rng.rand() - 0.5) * 0.04
            h[i] = old_i + delta
            h[j] += -delta * 0.6
            h[k] += -delta * 0.4
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1 or h[k] < 0 or h[k] > 1:
                h[i] = old_i
                h[j] += delta * 0.6
                h[k] += delta * 0.4
                continue
        
        new_c5 = _c5(h, dx)
        if new_c5 <= old_c5:
            old_c5 = new_c5
            stall = 0
        elif rng.rand() < np.exp(-(new_c5 - old_c5) / (T * dx)):
            old_c5 = new_c5
            stall += 1
        else:
            if move_type == 0:
                h[i], h[j] = old_i, old_j
            elif move_type == 1:
                pass  # already swapped back logically
            else:
                h[j] += delta * 0.6
                h[k] += delta * 0.4
            stall += 1
        
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
        
        if stall > 20000:
            h = best_h.copy()
            old_c5 = best_c5
            T = max(T * 3.0, 0.003)
            stall = 0
    
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    # Phase 1: Multi-start SLSQP (18% budget)
    p1_end = t_start + budget_s * 0.18
    for trial in range(6):
        if time.time() > p1_end:
            break
        if trial == 0:
            h_try = h0.copy()
        else:
            h_try = _rand_init(n, rng, base=h0)
        h_ref, c5_ref = _slsqp_step(h_try, n, bounds, cons, p1_end)
        if c5_ref < best_c5:
            best_c5, best_h = c5_ref, h_ref

    # Phase 2: SA with restarts (42% budget)
    p2_end = t_start + budget_s * 0.60
    n_restarts = 3
    sa_chunk = (p2_end - time.time()) / max(n_restarts, 1)
    for r in range(n_restarts):
        if time.time() > p2_end:
            break
        chunk_end = t_start + budget_s * 0.60 - (n_restarts - 1 - r) * sa_chunk
        chunk_end = min(chunk_end, p2_end)
        h_seed = best_h + rng.randn(n) * 0.03
        h_seed = _repair(np.clip(h_seed, 0, 1), n)
        h_sa, c5_sa = _sa_phase(h_seed, n, dx, chunk_end, rng)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa
        # Quick SLSQP from SA result
        if time.time() < p2_end - 5:
            h_r, c5_r = _slsqp_step(best_h, n, bounds, cons, p2_end - 2)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    # Phase 3: Coordinate descent (22% budget)
    p3_end = t_start + budget_s * 0.82
    if time.time() < p3_end:
        h = best_h.copy()
        local_best = best_c5
        step_scale = 1.0
        no_imp = 0
        while time.time() < p3_end and no_imp < 8:
            improved = False
            base = 0.008 * step_scale
            steps = np.array([base*0.2, base*0.5, base, base*2, base*4,
                             -base*0.2, -base*0.5, -base, -base*2, -base*4])
            for i in range(n):
                if time.time() > p3_end:
                    break
                vi = h[i]
                for d in steps:
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    if need > 0:
                        cand = np.argsort(h)[:12]
                    else:
                        cand = np.argsort(-h)[:12]
                    found = False
                    for j in cand:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c5_try = _c5(h, dx)
                        if c5_try < local_best - 1e-15:
                            local_best = c5_try
                            improved = True
                            found = True
                            break
                        h[i], h[j] = vi, nj - need
                    if found:
                        break
            if not improved:
                no_imp += 1
                step_scale *= 0.5
            else:
                no_imp = 0
                step_scale = max(step_scale * 0.8, 0.1)
        if local_best < best_c5:
            best_c5, best_h = local_best, h

    # Phase 4: Final polish
    if time.time() < t_start + budget_s - 3:
        h_f, c5_f = _slsqp_step(best_h, n, bounds, cons, t_start + budget_s - 2)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        # Ultra-fine coord
        h = best_h.copy()
        local_best = best_c5
        fd = np.array([0.0003, 0.0008, 0.0015, -0.0003, -0.0008, -0.0015])
        for i in range(n):
            if time.time() > t_start + budget_s - 2:
                break
            vi = h[i]
            for d in fd:
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                need = -d
                cand = np.argsort(h)[:6] if need > 0 else np.argsort(-h)[:6]
                for j in cand:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-15:
                        local_best = c5_try
                        break
                    h[i], h[j] = vi, nj - need
                else:
                    continue
                break
        if local_best < best_c5:
            best_c5, best_h = local_best, h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
