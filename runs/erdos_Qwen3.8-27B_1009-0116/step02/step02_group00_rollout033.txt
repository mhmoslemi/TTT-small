def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _project(h, n):
    target = n / 2.0
    def _sum_at(tau):
        return float(np.sum(np.clip(h - tau, 0.0, 1.0)))
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if _sum_at(mid) > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

    corr = _corr(x)
    m = np.max(corr)
    sh = np.clip(alpha * (corr - m), -500.0, 500.0)
    return (m + np.log(np.exp(sh).sum()) / alpha) * (2.0 / n)

    return float(np.sum(x) - target)

    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _project(h, n):
    target = n / 2.0
    def _sa(tau):
        return float(np.sum(np.clip(h - tau, 0.0, 1.0)))
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if _sa(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _obj_alpha(x, alpha, n):
    corr = _corr(x)
    m = np.max(corr)
    sh = np.clip(alpha * (corr - m), -500.0, 500.0)
    return (m + np.log(np.exp(sh).sum()) / alpha) * (2.0 / n)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _grad_smooth(h, alpha, n):
    corr = _corr(h)
    m = np.max(corr)
    sh = np.clip(alpha * (corr - m), -500.0, 500.0)
    s = np.exp(sh)
    w = s / s.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        d = k - (n - 1)
        wk = w[k]
        if d >= 0:
            t1 = np.zeros(n)
            t1[:n - d] = omh[d:]
            t2 = np.zeros(n)
            t2[d:] = h[:n - d]
        else:
            t1 = np.zeros(n)
            t1[-d:] = omh[:n + d]
            t2 = np.zeros(n)
            t2[:n + d] = h[-d:]
        grad += wk * (t1 - t2)
    return grad * dx

def _interp_parent(n_par, n):
    if n_par == n:
        return np.array(initial_h_values, dtype=float)
    xp = np.linspace(0.0, 1.0, n_par, endpoint=False) + 0.5 / n_par
    fp = initial_h_values
    xm = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
    return np.interp(xm, xp, fp)

def _binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _slsqp_optimize(h0, n, alpha, t_end, maxiter=400):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_alpha, h0, args=(alpha, n), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h0

def _pgrad_refine(h, n, t_end, rng):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    alpha = 2000
    step = 0.5 / alpha
    stalled = 0
    for it in range(50000):
        if time.time() > t_end:
            break
        grad = _grad_smooth(h, alpha, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        gdir = -grad / gnorm
        f_now = _obj_alpha(h, alpha, n)
        s = step
        moved = False
        for ls in range(25):
            h_try = _project(h + s * gdir, n)
            f_try = _obj_alpha(h_try, alpha, n)
            if f_try < f_now - 1e-6 * s * gnorm:
                h = h_try
                moved = True
                break
            s *= 0.5
        if not moved:
            stalled += 1
            if stalled > 30:
                h = _repair(h + 0.02 * rng.randn(n) * (1.0 - 2.0 * np.abs(h - 0.5)), n)
                stalled = 0
                alpha = max(500, alpha // 2)
                continue
        else:
            stalled = 0
            step = max(s, 0.1 / alpha)
        c5_now = _c5(h, dx)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def _fine_descent(h, n, t_end):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5(h, dx)
    no_improve = 0
    while time.time() < t_end and no_improve < 50:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in [0.0002, -0.0002, 0.0005, -0.0005, 0.001, -0.001]:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    cands = np.argsort(h)[:12]
                else:
                    cands = np.argsort(-h)[:12]
                for j in cands:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        improved = True
                        break
                    h[i] = vi
                    h[j] = nj - need
                if improved:
                    break
            if not improved:
                h[i] = vi
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            h = _repair(h, n)
    return h, best_c5

def _sa_block(h, n, t_end, rng):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    T = 0.003
    Tend = 0.00002
    max_it = 800000
    for it in range(max_it):
        if time.time() > t_end:
            break
        frac = it / max_it
        cur_T = T * (Tend / T) ** frac
        blen = rng.randint(2, 6)
        i0 = rng.randint(0, n - blen + 1)
        block = slice(i0, i0 + blen)
        delta = (rng.rand() - 0.5) * 0.08 * (1.0 - 0.6 * frac)
        h_new = h[block] + delta
        if np.any(h_new < -0.01) or np.any(h_new > 1.01):
            continue
        h_new = np.clip(h_new, 0.0, 1.0)
        h_try = h.copy()
        h_try[block] = h_new
        h_try = _repair(h_try, n)
        c5_try = _c5(h_try, dx)
        if c5_try <= best_c5 + cur_T:
            h = h_try
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5
