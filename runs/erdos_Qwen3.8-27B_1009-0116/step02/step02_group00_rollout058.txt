def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

For the local search, I'll do:
- For each coordinate i, try small perturbations
- For each perturbation, try to compensate with another coordinate j
- Evaluate full C₅ for each candidate

The key optimization is to be smart about which j to try. Instead of trying all j, try:
- j = (i+1) % n (neighbor)
- j that maximizes "room" (for positive delta) or "excess" (for negative delta)
- A few random j

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-11:
                break
    return h


def _grad_lse(h, alpha, n):
    C = np.correlate(h, 1.0 - h, mode='full')
    maxC = float(np.max(C))
    expC = np.exp(alpha * (C - maxC))
    w = expC / expC.sum()
    dx = 2.0 / n
    m = np.arange(2 * n - 1) - (n - 1)
    K = np.arange(n)
    d1 = K[None, :] - m[:, None]
    v1 = (d1 >= 0) & (d1 < n)
    h1 = np.where(v1, h[np.clip(d1, 0, n - 1)], 0.0)
    t1 = np.where(v1, 1.0 - h1, 0.0)
    d2 = K[None, :] + m[:, None]
    v2 = (d2 >= 0) & (d2 < n)
    h2 = np.where(v2, h[np.clip(d2, 0, n - 1)], 0.0)
    t2 = np.where(v2, h2, 0.0)
    return dx * (w @ (t1 - t2))


def _obj_lse(h, alpha, n):
    C = np.correlate(h, 1.0 - h, mode='full')
    maxC = float(np.max(C))
    s = np.exp(alpha * (C - maxC)).sum()
    return (maxC + np.log(s) / alpha) * (2.0 / n)


def _eq_fun(h, target):
    return float(h.sum() - target)


def _eq_jac(h, target):
    return np.ones_like(h)


def _slsqp_lse(h, alpha, n, bounds, target, maxiter=300):
    try:
        res = minimize(_obj_lse, h, args=(alpha, n),
                       jac=lambda x, a=alpha, nn=n: _grad_lse(x, a, nn),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()


def _local_search(h, n, dx, target, time_limit, rng):
    best = _c5(h, dx)
    h = h.copy()
    deltas = [0.003, 0.006, 0.012, 0.025, 0.05, -0.003, -0.006, -0.012, -0.025, -0.05]
    no_improve = 0
    order = list(range(n))
    t_start = time.time()
    while time.time() - t_start < time_limit:
        improved = False
        for i in order:
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                need = -d
                if need > 0:
                    cands = np.argsort(h)[:12]
                else:
                    cands = np.argsort(-h)[:12]
                found = False
                for j in cands:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c_try = _c5(h, dx)
                    if c_try < best - 1e-14:
                        best = c_try
                        improved = True
                        found = True
                        no_improve = 0
                        break
                    else:
                        h[i], h[j] = vi, h[j] - need
                if found:
                    break
            if time.time() - t_start > time_limit:
                break
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
            rng.shuffle(order)
    return h, best


def _sa_search(h, n, dx, rng, time_limit):
    best = _c5(h, dx)
    h_cur = h.copy()
    best_h = h.copy()
    t_start = time.time()
    T0 = 0.003
    tend = 0.00002
    it = 0
    max_it = 5000000
    while time.time() - t_start < time_limit:
        it += 1
        frac = min(it / max_it, 1.0)
        T = T0 * (tend / T0) ** frac
        dm = 0.05 * (1.0 - 0.7 * frac)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 2.0 * dm
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c_try = _c5(h_cur, dx)
        if c_try <= best + T * 0.1:
            if c_try < best - 1e-15:
                best = c_try
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
    return best_h, best
