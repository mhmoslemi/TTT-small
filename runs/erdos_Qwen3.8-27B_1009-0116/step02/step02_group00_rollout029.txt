            for _ in range(20):
                h_new = _project(h - step * v, target)
                f1 = _c5(h_new, dx)
                if f1 < best_c5 - 1e-15:
                    h = h_new
                    best_c5 = f1
                    best_h = h.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break

    return best_h, best_c5

def _fine_tune(h, target, t_limit):
    """Very fine coordinate adjustments."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5(h, dx)
    t_end = time.time() + t_limit
    deltas = [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]
    for i in range(n):
        if time.time() > t_end:
            break
        for d in deltas:
            if time.time() > t_end:
                break
            ni = h[i] + d
            if ni < 0.0 or ni > 1.0:
                continue
            need = -d
            if need > 0:
                cands = np.argsort(h)
            else:
                cands = np.argsort(-h)
            for j in cands[:20]:
                if j == i:
                    continue
                nj = h[j] + need
                if nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                c5_try = _c5(h, dx)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    break
                else:
                    h[i] -= d
                    h[j] -= need
    return h, best_c5

def _run_phase(h, target, t_limit, rng):
    n = len(h)
    best_h, best_c5 = _optimize(h, target, t_limit * 0.8)
    if time.time() < time.time() - t_limit * 0.8 + t_limit - 3:
        t_rem = t_limit - (time.time() - (time.time() - t_limit))
        h2, c5_2 = _fine_tune(best_h, target, min(t_rem * 0.5, 60))
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    # One more gradient pass if time
    if time.time() < time.time() + 1:
        t_rem2 = max(5, t_limit - (best_c5 - best_c5 + 0) - 0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    h0 = _project(h0, n0 / 2.0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx0)
    best_n = n0

    def _elapsed():
        return time.time() - t_start

    def _remaining():
        return budget_s - _elapsed()

    # Phase 1: Optimize n=70 with gradient method
    if _remaining() > 60:
        t_lim = min(_remaining() * 0.3, 250)
        h_new, c5 = _optimize(h0, n0 / 2.0, t_lim)
        h_new, c5 = _fine_tune(h_new, n0 / 2.0, min(60, _remaining() * 0.1))
        if c5 < best_c5:
            best_h, best_c5, best_n = h_new, c5, n0

    # Phase 2: Try different resolutions
    for n_try in [80, 90, 100, 60]:
        if _remaining() < 40:
            break
        x0 = np.linspace(0, 2, best_n)
        x1 = np.linspace(0, 2, n_try)
        h_interp = np.interp(x1, x0, best_h)
        h_interp = _project(h_interp, n_try / 2.0)
        t_lim = min(_remaining() * 0.15, 120)
        h_new, c5 = _optimize(h_interp, n_try / 2.0, t_lim)
        if c5 < best_c5:
            best_h, best_c5, best_n = h_new, c5, n_try
            h_new, c5 = _fine_tune(best_h, best_n / 2.0, min(40, _remaining() * 0.05))
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 3: Intensive re-optimization of best solution
    if _remaining() > 80:
        n = best_n
        target = n / 2.0
        t_lim = min(_remaining() * 0.3, 250)
        h_new, c5 = _optimize(best_h, target, t_lim)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
        h_new, c5 = _fine_tune(best_h, target, min(80, _remaining() * 0.15))
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 4: Perturbation restarts
    n = best_n
    target = n / 2.0
    dx = 2.0 / n
    if _remaining() > 30:
        max_restarts = int(min(_remaining() // 20, 40))
        for r in range(max_restarts):
            if _remaining() < 25:
                break
            pert_scale = 0.01 + 0.03 * (r / max(1, max_restarts - 1))
            h_pert = best_h + rng.randn(n) * pert_scale
            h_pert = _project(h_pert, target)
            t_lim = min(_remaining() - 25, 40)
            h_new, c5 = _optimize(h_pert, target, t_lim)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Final fine-tune
    if _remaining() > 10:
        best_h, best_c5 = _fine_tune(best_h, best_n / 2.0, min(_remaining() - 5, 60))

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
