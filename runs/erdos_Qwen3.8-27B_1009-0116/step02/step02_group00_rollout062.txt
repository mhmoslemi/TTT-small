import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lp_value(h, p, dx):
    corr = _corr(h) * dx
    return float(np.sum(corr**p)) ** (1.0/p)

def _lp_grad(h, p, dx):
    n = len(h)
    corr = _corr(h) * dx
    F = float(np.sum(corr**p))
    if F < 1e-30:
        return np.zeros(n)
    w = corr**(p-1)  # shape (2n-1,)
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        wk = w[k]
        if abs(wk) < 1e-30:
            continue
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += wk * (1.0 - h[:n-lag])
                grad[:n-lag] -= wk * h[lag:]
        else:
            al = -lag
            if n - al > 0:
                grad[:n-al] += wk * (1.0 - h[al:])
                grad[al:] -= wk * h[:n-al]
    return grad * (F**(1.0/p - 1.0) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

def _proj_grad(h, grad_n, lr, n):

    h_new = h - lr * grad_n
    h_new = np.clip(h_new, 0.0, 1.0)
    # project to sum = target
    s = float(h_new.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h_new = h_new + (target - s) / n
    h_new = np.clip(h_new, 0.0, 1.0)
    # final repair
    h_new = _repair(h_new, n)
    return h_new

def _coord_descent(h, n, dx, t_end, max_rounds=100):

    best_h = h.copy()
    target = n / 2.0
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for step in steps:
        if time.time() > t_end:
            break
        for rnd in range(max_rounds):
            if time.time() > t_end:
                break
            improved = False
            for i in range(n):
                if time.time() > t_end:
                    break
                vi = best_h[i]
                for d in [step, -step]:
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    # find j to compensate
                    if need > 0:
                        order = np.argsort(best_h)
                    else:
                        order = np.argsort(-best_h)
                    for j in order:
                        if j == i:
                            continue
                        nj = best_h[j] + need
                        if nj < 0 or nj > 1:

                        c5_try = _c5(best_h, dx)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            improved = True
                            break
                        else:
                            best_h[i] = vi
                            best_h[j] = nj - need
            if not improved:
                break

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lp_grad(h, p, dx):
    n = len(h)
    raw = _corr(h)
    s = raw * dx
    F = float(np.sum(s**p))
    if F < 1e-30:
        return np.zeros(n)
    w = s**(p-1) * dx
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        wk = w[k]
        if abs(wk) < 1e-30:
            continue
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[lag:] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[lag:]
        else:
            al = -lag
            m = n - al
            if m > 0:
                grad[:m] += wk * (1.0 - h[al:])
                grad[al:] -= wk * h[:m]
    return grad * F**(1.0/p - 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b
                diff += b
            if abs(diff) < 1e-13:
                break
    return h

def _cd(h, n, dx, t_end, rounds=50):

    best = h.copy()
    target = n / 2.0
    steps = [0.01, 0.004, 0.001, 0.0004, 0.0001]
    for step in steps:
        if time.time() > t_end:
            break
        for rnd in range(rounds):
            if time.time() > t_end:
                break
            imp = False
            for i in range(n):
                if time.time() > t_end:
                    break
                vi = best[i]
                for d in (step, -step):
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    if need > 0:
                        order = np.argsort(best)
                    else:
                        order = np.argsort(-best)
                    for j in order:
                        if j == i:
                            continue
                        nj = best[j] + need
                        if nj < 0 or nj > 1:
                            continue
                        best[i] = ni
                        best[j] = nj
                        c = _c5(best, dx)
                        if c < best_c5 - 1e-14:
                            best_c5 = c
                            imp = True
                            break
                        else:
                            best[i] = vi
                            best[j] = nj - need
            if not imp:
                break
    return best, best_c5

def _lse_obj(x, alpha, n):
    corr = _corr(x) * (2.0/n)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha*(corr-m), -50, 50))
    return m + float(np.sum(s))/alpha

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lp_val(h, p, dx):
    c = _corr(h) * dx
    return float(np.sum(c**p))**(1.0/p)

def _lp_grad(h, p, dx):
    n = len(h)
    raw = _corr(h)
    s = raw * dx
    F = float(np.sum(s**p))
    if F < 1e-30:
        return np.zeros(n)
    w = s**(p-1) * dx
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n-1)
        wk = w[k]
        if abs(wk) < 1e-30:
            continue
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[lag:] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[lag:]
        else:
            al = -lag
            m = n - al
            if m > 0:
                grad[:m] += wk * (1.0 - h[al:])
                grad[al:] -= wk * h[:m]
    return grad * F**(1.0/p - 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b; diff += b
            if abs(diff) < 1e-13:
                break
    return h

            break

                break
            imp = False
            for i in range(n):
                if time.time() > t_end:
                    break
                vi = best[i]
                for d in (step, -step):
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    if need > 0:
                        order = np.argsort(best)
                    else:
                        order = np.argsort(-best)
                    if top_j is not None:
                        order = order[:top_j]
                    for j in order:
                        if j == i:

                        nj = best[j] + need
                        if nj < 0 or nj > 1:

                        best[i] = ni; best[j] = nj
                        c = _c5(best, dx)
                        if c < best_c5 - 1e-14:
                            best_c5 = c; imp = True; break
                        else:
                            best[i] = vi; best[j] = nj - need
            if not imp:
                break
    return best, best_c5

    corr = _corr(x) * (2.0/n)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha*(corr-m), -50, 50))
    return m + float(np.sum(s))/alpha
