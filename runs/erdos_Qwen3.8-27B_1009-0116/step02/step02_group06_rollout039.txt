import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_corr(h, n):
    """g[m, l] = d correlate(h,1-h)[m] / d h[l]"""
    nc = 2*n - 1
    g = np.zeros((nc, n))
    for m in range(nc):
        s = m - (n-1)
        i_start = max(0, -s)
        i_end = min(n, n - s)
        for i in range(i_start, i_end):
            j = i + s
            g[m, i] += (1.0 - h[j])
            g[m, j] -= h[i]
    return g

def _solve_lp(c, g, h0, n, target):
    """Solve: min t s.t. c_k + g_k·(h-h0) ≤ t, 0≤h≤1, sum(h)=target
    Variables: [t, h_0, ..., h_{n-1}] (n+1 variables)
    """
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[0] = 1.0
    
    # Inequality constraints: c_k + g_k·(h-h0) - t ≤ 0 for each k
    # => -t + g_k[0]*h[0] + ... + g_k[n-1]*h[n-1] ≤ -c_k + g_k·h0
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    for k in range(2*n-1):
        A_ub[k, 0] = -1.0
        A_ub[k, 1:] = g[k]
        b_ub[k] = -c[k] + g[k] @ h0
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, None)] + [(0.0, 1.0)]*n  # t ≥ 0, 0 ≤ h_i ≤ 1
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]  # h part
    return None

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min > d_max - 1e-12:
                continue
            if d_max - d_min < 1e-12:
                continue
            # Compute gradient of corr w.r.t. (h[i] += d, h[j] -= d)
            # corr_new[m] = corr[m] + d * (gc[i,m] - gc[j,m])
            # where gc[l,m] = d correlate[m] / d h[l]
            gd = _gc_single(h, i, n) - _gc_single(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = np.argmin(c5s1)
            cur_max = np.max(corr) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _gc_single(h, l, n):
    """Compute d correlate(h, 1-h) / d h[l] as a vector of length 2n-1."""
    nc = 2*n - 1

            j = i + s
            if i == l:
                gc[m] += (1.0 - h[j])
            if j == l:
                gc[m] -= h[i]
    return gc

            gc[i, m] += (1.0 - h[j])
            gc[j, m] -= h[i]
    return gc

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lp_step(h, n, target):
    """One LP linearization step."""
    corr = _corr(h)
    gc = _full_gc(h, n)
    
    # LP: min t s.t. corr[m] + gc[h][m]·δ ≤ t, |δ|_sum constraint
    # Variables: [t, h_0, ..., h_{n-1}], n+1 vars
    # Constraints: 
    #   -t + sum_l gc[l,m]*h_l ≤ -corr[m] + sum_l gc[l,m]*h_l (old) for each m
    #   Wait, the linearization is: corr_m(new_h) ≈ corr_m(h) + sum_l gc[l,m]*(h_new[l] - h[l])
    #   So: corr_m(h) + sum_l gc[l,m]*(h_new[l]-h[l]) ≤ t
    #   => -t + sum_l gc[l,m]*h_new[l] ≤ sum_l gc[l,m]*h[l] - corr_m(h)
    
    nc = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.zeros((nc, n + 1))
    b_ub = np.zeros(nc)
    for m in range(nc):
        A_ub[m, 0] = -1.0
        A_ub[m, 1:] = gc[:, m]  # gc[l,m] for l=0..n-1
        b_ub[m] = gc[:, m] @ h - corr[m]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, None)] + [(0.0, 1.0)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None

def _sweep(h, n, dx):
    """One pass of coordinate pair sweep."""
    corr = _corr(h)
    cur_max = np.max(corr)
    improved = False
    for _pass in range(3):  # multiple passes within one sweep
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_min > d_max - 1e-12 or d_max - d_min < 1e-12:
                    continue
                # Compute the change in corr for h[i]+d, h[j]-d
                # gc for endpoint i and j
                # corr_new[m] = corr[m] + d * (gc_i[m] - gc_j[m]) + (d^2 if m corresponds to shift j-i)
                # For simplicity, use the quadratic correction
                s = j - i
                m_idx = s + (n - 1)  # the shift where both i and j are in the same pair
                # Linear part:
                # For all m, change = d * (gc[i,m] - gc[j,m])
                # Quadratic correction for m_idx:
                # The pair (i,j) contributes h[i]*(1-h[j]) in c_{m_idx}
                # After change: (h[i]+d)*(1-h[j]+d) = h[i]*(1-h[j]) + d*(1-h[j]) + d*h[i] + d^2
                # Linear part already accounts for d*(1-h[j]) + d*h[i] (through gc)
                # So we need to add d^2
                # Actually let me just compute gc properly
                
                # Compute gc[i] and gc[j] on the fly
                gc_i = np.zeros(2*n-1)
                gc_j = np.zeros(2*n-1)
                for m in range(2*n-1):
                    s2 = m - (n-1)
                    i_start = max(0, -s2)
                    i_end = min(n, n - s2)
                    for ii in range(i_start, i_end):
                        jj = ii + s2
                        if ii == i:
                            gc_i[m] += (1.0 - h[jj])
                        if jj == i:
                            gc_i[m] -= h[ii]
                        if ii == j:
                            gc_j[m] += (1.0 - h[jj])
                        if jj == j:
                            gc_j[m] -= h[ii]
                gd = gc_i - gc_j
                # Quadratic term for the specific shift
                quad = np.zeros(2*n-1)
                quad[m_idx] = 1.0  # d^2 coefficient
                
                # 1D optimization: minimize max_m (corr[m] + d*gd[m] + d^2*quad[m])
                # over d in [d_min, d_max]
                # This is a convex optimization in d (max of convex functions)
                # Use golden section or fine grid
                ds = np.linspace(d_min, d_max, 30)
                obj = corr[None, :] + ds[:, None] * gd[None, :] + ds[:, None]**2 * quad[None, :]
                vals = np.max(obj, axis=1)
                bi = np.argmin(vals)
                if vals[bi] < cur_max - 1e-10:
                    # Fine search
                    lo = max(d_min, ds[bi] - 0.05)
                    hi2 = min(d_max, ds[bi] + 0.05)
                    ds2 = np.linspace(lo, hi2, 40)
                    obj2 = corr[None, :] + ds2[:, None] * gd[None, :] + ds2[:, None]**2 * quad[None, :]
                    vals2 = np.max(obj2, axis=1)
                    d_best = ds2[np.argmin(vals2)]
                    h[i] = hi + d_best
                    h[j] = hj - d_best
                    corr = _corr(h)  # recompute exactly
                    cur_max = np.max(corr)
                    improved = True
    return h, improved

The problem is: min t s.t. c_k(h) ≤ t/dx for all k, 0≤h≤1, sum(h)=n/2.

The subgradient of max_k c_k(h) with respect to h is any convex combination of the active gradients. We can use this for a projected subgradient method:

1. Find active set A = {k : c_k(h) = max_k c_k(h)} (or near-active)
2. Compute g = (1/|A|) * sum_{k in A} gc[:,k] (average gradient of active constraints)
3. Take a step: h_new = h - step_size * g
4. Project onto the feasible set: {h : 0≤h≤1, sum(h)=n/2}
5. Update step size

The projection step is: clip h to [0,1], then adjust to satisfy sum(h)=n/2.

This is simple and might work well, especially with adaptive step sizes.

Actually, one more consideration. The `_sweep` function is the most important part for fine-tuning. Let me make sure it's correct and efficient.

The key insight in the sweep: for a fixed pair (i,j), changing h[i] by +d and h[j] by -d changes the correlation by:
Δc_m = d * (gc[i,m] - gc[j,m]) + d² (if m = j-i+n-1)

This is because:
- The bilinear term h[i]*(1-h[j]) appears in c_{j-i+n-1} (when i≠j)
- Changing both introduces a d² term

For m ≠ j-i+n-1, the change is purely linear in d.

So the 1D problem is: minimize max_m (corr[m] + d*gd[m] + d²*δ_{m,mi}) over d ∈ [d_lo, d_hi].

This is a convex optimization in d (max of convex functions, each is either linear or quadratic with positive coefficient). I could solve it exactly using bisection or by finding the active constraints. But a fine grid search (25 coarse + 40 fine points) should be accurate enough.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(_cor(h).max() * dx)

def _gc(h, n):
    nc = 2 * n - 1
    g = np.zeros((n, nc))
    for m in range(nc):
        s = m - (n - 1)
        a = max(0, -s)
        b = min(n, n - s)
        for i in range(a, b):
            g[i, m] += 1.0 - h[i + s]
            g[i + s, m] -= h[i]
    return g

def _rep(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                x = min(d, 1.0 - h[i]); h[i] += x; d -= x
            else:
                x = min(-d, h[i]); h[i] -= x; d += x
            if abs(d) < 1e-14: break
    return h

def _lp(h, n, t):
    c = _cor(h)
    g = _gc(h, n)
    nc = 2 * n - 1
    co = np.zeros(n + 1); co[0] = 1.0
    Au = np.zeros((nc, n + 1))
    bu = np.zeros(nc)
    for m in range(nc):
        Au[m, 0] = -1.0; Au[m, 1:] = g[:, m]
        bu[m] = g[:, m].dot(h) - c[m]
    Ae = np.zeros((1, n + 1)); Ae[0, 1:] = 1.0
    be = np.array([t])
    try:
        r = linprog(co, A_ub=Au, b_ub=bu, A_eq=Ae, b_eq=be,
                    bounds=[(0.0, None)] + [(0.0, 1.0)] * n, method='highs')
        if r.success: return r.x[1:]
    except Exception: pass
    return None

def _lpd(h, n, t, dx):
    hl = _lp(h, n, t)
    if hl is None: return h
    d = hl - h
    bt, bc_ = 0.0, _c5(h, dx)
    for tt in np.linspace(0.05, 1.0, 15):
        ht = _rep(h + tt * d, n)
        c = _c5(ht, dx)
        if c < bc_: bc_, bt = c, tt
    return _rep(h + bt * d, n) if bt > 0 else h

def _sw(h, n, dx):
    g = _gc(h, n)
    c = _cor(h)
    km = c.max()
    imp = False
    for i in range(n):
        for j in range(n):
            if i == j: continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12: continue
            gd = g[i] - g[j]
            mi = j - i + n - 1
            ds = np.linspace(lo, hi, 25)
            ob = c[None, :] + ds[:, None] * gd[None, :]
            ob[:, mi] += ds ** 2
            v = ob.max(1)
            bi = v.argmin()
            if v[bi] < km - 1e-10:
                l2 = max(lo, ds[bi] - 0.1)
                h2 = min(hi, ds[bi] + 0.1)
                d2 = np.linspace(l2, h2, 40)
                o2 = c[None, :] + d2[:, None] * gd[None, :]
                o2[:, mi] += d2 ** 2
                db = d2[o2.max(1).argmin()]
                h[i] += db; h[j] -= db
                c = _cor(h); km = c.max()
                imp = True
    return h, imp

def _sm(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + float(np.log(s.sum())) / a

def _fm(x, a):
    return _sm(_cor(x), a)

def _fj(x, a):
    n = len(x)
    c = _cor(x)
    m = c.max()
    w = np.exp(np.clip(a * (c - m), -500, 500))
    w /= w.sum()
    return _gc(x, n) @ w

    return x.sum() - t

    return np.ones_like(x)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(_cor(h).max() * dx)

def _gc(h, n):
    nc = 2 * n - 1
    g = np.zeros((n, nc))
    for m in range(nc):
        s = m - (n - 1)
        a = max(0, -s)
        b = min(n, n - s)
        for i in range(a, b):
            g[i, m] += 1.0 - h[i + s]
            g[i + s, m] -= h[i]
    return g

def _rep(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                x = min(d, 1.0 - h[i]); h[i] += x; d -= x
            else:
                x = min(-d, h[i]); h[i] -= x; d += x
            if abs(d) < 1e-14:
                break
    return h

def _lp(h, n, t):
    c = _cor(h)
    g = _gc(h, n)
    nc = 2 * n - 1
    co = np.zeros(n + 1); co[0] = 1.0
    Au = np.zeros((nc, n + 1))
    bu = np.zeros(nc)
    for m in range(nc):
        Au[m, 0] = -1.0
        Au[m, 1:] = g[:, m]
        bu[m] = g[:, m].dot(h) - c[m]
    Ae = np.zeros((1, n + 1)); Ae[0, 1:] = 1.0
    be = np.array([t])
    try:
        r = linprog(co, A_ub=Au, b_ub=bu, A_eq=Ae, b_eq=be,
                    bounds=[(0.0, None)] + [(0.0, 1.0)] * n, method='highs')
        if r.success:
            return r.x[1:]
    except Exception:
        pass
    return None

    hl = _lp(h, n, t)
    if hl is None:
        return h
    d = hl - h
    bt, bc_ = 0.0, _c5(h, dx)
    for tt in np.linspace(0.05, 1.0, 15):

        c = _c5(ht, dx)
        if c < bc_:
            bc_, bt = c, tt
    if bt > 0:
        return _rep(h + bt * d, n)
    return h

def _sw(h, n, dx):
    g = _gc(h, n)
    c = _cor(h)
    km = c.max()
    imp = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12:
                continue
            gd = g[i] - g[j]
            mi = j - i + n - 1
            ds = np.linspace(lo, hi, 25)
            ob = c[None, :] + ds[:, None] * gd[None, :]
            ob[:, mi] += ds ** 2
            v = ob.max(1)
            bi = v.argmin()
            if v[bi] < km - 1e-10:
                l2 = max(lo, ds[bi] - 0.1)
                h2 = min(hi, ds[bi] + 0.1)
                d2 = np.linspace(l2, h2, 40)
                o2 = c[None, :] + d2[:, None] * gd[None, :]
                o2[:, mi] += d2 ** 2
                db = d2[o2.max(1).argmin()]
                h[i] += db
                h[j] -= db
                c = _cor(h)
                km = c.max()
                imp = True
    return h, imp

def _sm(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / a

def _fm(x, a):
    return _sm(_cor(x), a)
