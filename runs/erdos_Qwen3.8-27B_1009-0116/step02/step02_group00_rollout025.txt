                diff -= add
            else:

                diff += sub

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _obj_lse(x, alpha, n):
    c = _corr(x)
    m = c.max()
    return (m + np.exp(np.clip(alpha*(c - m), -30, 30)).sum() / alpha) * (2.0/n)

def _eq(x, t):
    return float(x.sum() - t)

def _subgrad(h, n, eta, topk=5):
    c = _corr(h)
    r = len(c)
    k = min(topk, r)
    idx = np.argpartition(c, -k)[-k:]
    # Compute gradient of top-k correlations
    g = np.zeros(n)
    for ki in idx:
        lag = int(ki) - (n - 1)
        for i in range(n):
            g[i] = g[i] + 1.0  # from sum h[m] term
            if 0 <= i - lag < n:
                g[i] -= h[i - lag]
            if 0 <= i + lag < n:
                g[i] -= h[i + lag]
    g /= k
    h_new = np.clip(h - eta * g, 0.0, 1.0)
    return _repair(h_new, n)
