    target = n / 2.0

    lo, hi = -float(np.max(h)) - 1, float(np.max(1 - h)) + 1

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _obj_grad(h, alpha, dx):
    n = len(h)
    c = _corr(h) * dx
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -50, 500)
    e = np.exp(s)
    se = e.sum()
    w = e / se
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:-lag])
            g[:-lag] -= wk * h[lag:]
        else:
            ln = -lag
            g[:n - ln] += wk * (1.0 - h[ln:])
            g[ln:] -= wk * h[:n - ln]
    g *= dx
    obj = m + np.log(se) / alpha
    return obj, g

def _project(h, n):
    target = n * 0.5
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    out = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - out)
        else:
            idx = np.argsort(out)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - out[i])
                out[i] += a
                diff -= a
            else:
                s = min(-diff, out[i])
                out[i] -= s
                diff += s
            if abs(diff) < 1e-14:
                break
    return out

def _mgd(h, n, dx, alpha, steps, lr, mu, deadline):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best = _c5(h, dx)
    stall = 0
    for _ in range(steps):
        if time.time() > deadline:
            break
        _, g = _obj_grad(h, alpha, dx)
        v = mu * v - lr * g
        trial = _project(h + v, n)
        c = _c5(trial, dx)
        if c < best - 1e-16:
            h = trial
            best = c
            stall = 0
        else:
            stall += 1
            v *= 0.5
            if stall > 80:
                v = np.zeros(n)
                trial = _project(h - 0.002 * g, n)
                c2 = _c5(trial, dx)
                if c2 < best:
                    h = trial
                    best = c2
                lr *= 0.6
                stall = 0
                if lr < 1e-9:
                    break
    return h, best

def _top_lag_grad(h, n, top_k=5):
    c = _corr(h)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        c[k] / c[idx].max()  # weight
    w = c[idx] / c[idx].sum()
    for ki, kk in enumerate(idx):
        wk = w[ki] * 0.1
        lag = kk - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:-lag])
            g[:-lag] -= wk * h[lag:]
        else:
            ln = -lag
            g[:n - ln] += wk * (1.0 - h[ln:])
            g[ln:] -= wk * h[:n - ln]
    return g
