        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.tile(np.r_[1.0, 0.0], (n + 1) // 2)[:n].astype(float) + 0.08 * rng.randn(n)
    elif kind == 2:
        h = np.zeros(n)
        h[np.mod(np.arange(n), 3) < 2] = 1.0
        h += 0.08 * rng.randn(n)
    elif kind == 3:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n) + 0.05 * rng.randn(n)
    else:
        h = 0.5 + 0.15 * rng.randn(n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, np.inf, 0
    n_list = [70, 50, 100, 36, 140]
    has_init = False
    init_n = 0
    try:
        init_n = len(initial_h_values)
        has_init = True
        if init_n not in n_list:
            n_list.insert(0, init_n)
        else:
            n_list.remove(init_n)
            n_list.insert(0, init_n)
    except NameError:
        pass
    tlim = budget_s - 30
    wts = []
    for idx, ni in enumerate(n_list):
        if has_init and ni == init_n:
            wts.append(3.5 if idx == 0 else 1.5)
        else:
            wts.append(1.0)
    wsum = sum(wts)
    talloc = [tlim * w / wsum for w in wts]

    for ni, tl in zip(n_list, talloc):
        t_end_g = time.time() + tl
        if time.time() >= t_end_g:
            continue
        dx = 2.0 / ni
        starts = []
        if has_init and ni == init_n:
            starts.append(repair(np.array(initial_h_values, dtype=float).copy(), ni))
        ns = max(3, min(6, int(tl / 35)))
        for k in range(ns):
            starts.append(mkinit(ni, seed + k * 7919 + ni * 13, k % 5))
        for h0 in starts:
            if time.time() >= t_end_g:
                break
            h = h0.copy()
            c5 = c5v(h, dx)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), ni
            for al in [100, 500, 2000]:
                if time.time() >= t_end_g:
                    break
                hn = slsqp(h, ni, al, 200)
                c5n = c5v(hn, dx)
                if c5n < best_c5:
                    best_c5, best_h, best_n = c5n, hn.copy(), ni
                if c5n < c5v(h, dx):
                    h = hn
            h, c5ls = lsearch(h, ni, dx, 300, t_end_g)
            if c5ls < best_c5:
                best_c5, best_h, best_n = c5ls, h.copy(), ni
            if time.time() < t_end_g - 15:
                hb, c5b = binls(h, ni, dx, 80, t_end_g - 10)
                if c5b < best_c5:
                    best_c5, best_h, best_n = c5b, hb.copy(), ni
                if c5b < c5v(h, dx):
                    h = hb
                    for al in [500, 2000]:
                        if time.time() >= t_end_g:
                            break
                        hn = slsqp(h, ni, al, 150)
                        hn, c5n = lsearch(hn, ni, dx, 100, t_end_g)
                        if c5n < best_c5:
                            best_c5, best_h, best_n = c5n, hn.copy(), ni
            ha, c5a = anneal(h, ni, dx, t_end_g, rng)
            if c5a < best_c5:
                best_c5, best_h, best_n = c5a, ha.copy(), ni
            if c5a < c5v(h, dx) and time.time() < t_end_g:
                ha2 = slsqp(ha, ni, 1000, 150)
                ha2, c5a2 = lsearch(ha2, ni, dx, 100, t_end_g)
                if c5a2 < best_c5:
                    best_c5, best_h, best_n = c5a2, ha2.copy(), ni

    if best_h is not None:
        dx_b = 2.0 / best_n
        t_fin = t0 + budget_s - 5
        hf, cf = lsearch(best_h, best_n, dx_b, 300, t_fin)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()
        if time.time() < t0 + budget_s - 15:
            hf2 = slsqp(best_h, best_n, 5000, 150)
            hf2, cf2 = lsearch(hf2, best_n, dx_b, 150, t0 + budget_s - 5)
            if cf2 < best_c5:
                best_c5, best_h = cf2, hf2.copy()

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_n = 70
        best_c5 = c5v(best_h, 2.0 / 70)
    best_h = repair(best_h, best_n)
    best_c5 = c5v(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
