        h_new[active] = np.clip(res.x, 0, 1)
        return _repair(h_new, n)
    except:
        return h

def _lp_phase(h, n, t_end, rng):
    target = n / 2.0
    dx = 2.0 / n
    for alpha in [200, 800, 3000, 8000]:
        if time.time() > t_end:
            break
        def obj(x):
            return _obj_pen(x, alpha, target, 100.0)
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                         bounds=[(0, 1)] * n,
                         options={'maxiter': 600, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = _repair(res.x, n)
            if _c5(h_new, dx) < _c5(h, dx):
                h = h_new
        except:
            pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    def _try(h, n):
        nonlocal best_h, best_c5, best_n
        h = _repair(h, n)
        c5 = _c5(h, 2.0 / n)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    init_h = None
    init_n = 70
    try:
        init_n = len(initial_h_values)
        init_h = _repair(np.array(initial_h_values, dtype=float).copy(), init_n)
    except NameError:
        init_h = np.full(70, 0.5)
        init_n = 70

    _try(init_h, init_n)
    n = init_n
    dx = 2.0 / n
    h = init_h.copy()
    t_end = t0 + budget_s - 15

    # Phase 1: L-BFGS-B smoothing
    h = _lp_phase(h, n, min(t_end, t0 + 120), rng)
    _try(h, n)

    # Phase 2: Intensive targeted local search (main driver)
    for rnd in range(6):
        t_left = t_end - time.time()
        if t_left < 10:
            break
        h, c = _targeted_local(h, n, dx, int(min(2000, t_left * 80)), t_end, rng)
        _try(h, n)

    # Phase 3: NM on active variables
    if time.time() < t_end:
        h_new = _nm_subset(h, n, dx, t_end, rng)
        _try(h_new, n)
        if _c5(h_new, dx) < _c5(h, dx):
            h = h_new

    # Phase 4: More targeted local
    for rnd in range(4):
        if time.time() > t_end:
            break
        h, c = _targeted_local(h, n, dx, 1500, t_end, rng)
        _try(h, n)

    # Phase 5: Try other resolutions
    for n2 in [60, 80, 90, 100, 120]:
        if time.time() > t_end - 30:
            break
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = _repair(h2, n2)
        _try(h2, n2)
        h2 = _lp_phase(h2, n2, min(t_end, time.time() + 40), rng)
        _try(h2, n2)
        h2, c = _targeted_local(h2, n2, dx2, 600, t_end, rng)
        _try(h2, n2)

    # Final polish on global best
    if best_h is not None and time.time() < t_end:
        dx_b = 2.0 / best_n
        hf = best_h.copy()
        for _ in range(5):
            if time.time() > t_end:
                break
            hf, c = _targeted_local(hf, best_n, dx_b, 1000, t_end, rng)
            _try(hf, best_n)
        if time.time() < t_end - 5:
            hf2 = _nm_subset(hf, best_n, dx_b, t_end, rng)
            _try(hf2, best_n)

    if best_h is None:
        best_h = _repair(np.full(70, 0.5), 70)
        best_c5 = _c5(best_h, 2.0 / 70)
        best_n = 70

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
