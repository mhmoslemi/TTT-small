import numpy as np
from scipy.optimize import minimize
import time

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_vec(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        h[:] = 0.5
        return h
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_sm(x, alpha, dx):
    return _smooth_max(_corr_vec(x), alpha) * dx

def _sum_fn(x, target):
    return np.sum(x) - target

def _slsqp_opt(h, n, dx, alpha, maxiter=300, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_fn, 'args': (target,)}]
    try:
        res = minimize(_obj_sm, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if hasattr(res, 'x') and res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()
