For the paired update (h[i] += δ, h[j] -= δ), the change in r[m] is:

5. Update coefs (they depend on h!)

The issue is step 5: after updating h, the coefficients change. Specifically:
- coef_i[m] contains terms (1-h[...]) and -h[...], which change when h changes
- But only coef_i and coef_j change (and they change in a predictable way)

Actually, when h[i] increases by δ:
- For coef_k[m] where the "1-h[i]" term appears: changes by -δ
- For coef_k[m] where the "-h[i]" term appears: changes by -δ (since it's -h[i] and h[i] increased)

- coef_i[m] changes because h[i] might appear in the h[...] terms

So the coefficient matrix changes globally when any h changes. This makes the sweep more complex.

```python
import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _compute_coef_vec(h, i, n):
    """Coefficient of h[i] in each r[m], m=0..2n-2"""
    nc = 2 * n - 1
    coef = np.zeros(nc)
    # First part: (1 - h[i+m-n+1]) for valid indices
    for m in range(nc):
        j = i + m - n + 1
        if 0 <= j < n:
            coef[m] += (1.0 - h[j])
    # Second part: -h[i-m+n-1] for valid indices
    for m in range(nc):
        j = i - m + n - 1
        if 0 <= j < n:
            coef[m] -= h[j]
    return coef

def _min_max_linear(r, d, delta_lo, delta_hi, n_eval=64):
    """Minimize max_m (r[m] + delta*d[m]) over delta in [delta_lo, delta_hi]"""
    if delta_lo >= delta_hi:
        return 0.0
    deltas = np.linspace(delta_lo, delta_hi, n_eval)
    # Vectorized: shape (n_eval, nc)
    vals = r[None, :] + deltas[:, None] * d[None, :]
    maxes = np.max(vals, axis=1)
    best_idx = np.argmin(maxes)
    best_delta = deltas[best_idx]
    # Refine with local search
    if 0 < best_idx < n_eval - 1:
        lo = deltas[best_idx - 1]
        hi = deltas[best_idx + 1]
        for _ in range(10):
            mid = (lo + hi) / 2
            v_lo = np.max(r + lo * d)
            v_hi = np.max(r + hi * d)
            v_mid = np.max(r + mid * d)
            if v_mid <= v_lo:
                lo = mid
            else:
                hi = mid
        best_delta = (lo + hi) / 2
    return best_delta

def _sweep(h, n, dx, n_eval=64, n_passes=200):
    """Coordinate-pair sweep"""
    target = n / 2.0
    r = _corr(h)
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            best_delta = 0.0
            best_j = -1
            best_obj = float(np.max(r))
            # Compute coef for i
            ci = _compute_coef_vec(h, i, n)
            for j in range(n):
                if j == i:
                    continue
                hj = h[j]
                hi = h[i]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_lo >= d_hi - 1e-12:
                    continue
                cj = _compute_coef_vec(h, j, n)
                d = ci - cj
                delta = _min_max_linear(r, d, d_lo, d_hi, n_eval)
                obj = float(np.max(r + delta * d))
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_delta = delta
                    best_j = j
            if best_j >= 0:
                h[i] += best_delta
                h[best_j] -= best_delta
                r = r + best_delta * (ci - _compute_coef_vec(h, best_j, n))
                # Recompute r for safety
                r = _corr(h)
                improved = True
        if not improved:
            break
    return h
```

Actually, one more thought: the _min_max_linear function. Let me make sure it's correct and efficient.

Given r (current correlation, shape (nc,)) and d (delta coefficient, shape (nc,)):
Minimize max_m (r[m] + δ*d[m]) over δ ∈ [δ_lo, δ_hi]

This is a 1D convex optimization. The function g(δ) = max_m (r[m] + δ*d[m]) is convex.

Approach: 
1. Coarse grid: 100 points
2. Find best
3. Refine with golden section or bisection on subgradient

For bisection on subgradient:
- At point δ, the subgradient set is {d[m] : m ∈ argmax(r + δ*d)}
- If min(subgrad) > 0: g is increasing, move left
- If max(subgrad) < 0: g is decreasing, move right
- If 0 ∈ [min(subgrad), max(subgrad)]: δ is optimal (or at a kink)

```python
def _min_max_linear(r, d, lo, hi, n_coarse=50, n_fine=20):
    if lo >= hi - 1e-15:
        return 0.0
    # Coarse search
    deltas = np.linspace(lo, hi, n_coarse)
    # Evaluate max at each delta
    vals = r[None, :] + deltas[:, None] * d[None, :]
    maxes = vals.max(axis=1)
    idx = maxes.argmin()
    
    if idx == 0:
        return lo
    if idx == n_coarse - 1:
        return hi
    
    # Fine search around best
    f_lo = deltas[idx-1]
    f_hi = deltas[idx+1]
    
    # Bisection on subgradient
    for _ in range(n_fine):
        mid = (f_lo + f_hi) / 2
        r_mid = r + mid * d
        max_val = r_mid.max()
        # Subgradient: d values at argmax positions
        active = np.abs(r_mid - max_val) < 1e-10
        sg_min = d[active].min()
        sg_max = d[active].max()
        if sg_min > 0:  # function increasing at mid
            f_hi = mid
        elif sg_max < 0:  # function decreasing
            f_lo = mid
        else:
            # 0 is in subgradient, mid is optimal
            break
    
    return (f_lo + f_hi) / 2
```

```python
import time

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)

        for idx in order:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _coef_vec(h, i, n):
    nc = 2 * n - 1
    coef = np.zeros(nc)
    for m in range(nc):
        j = i + m - n + 1
        if 0 <= j < n:
            coef[m] += 1.0 - h[j]
        j = i - m + n - 1
        if 0 <= j < n:
            coef[m] -= h[j]
    return coef

def _min_max_lin(r, d, lo, hi):
    if lo >= hi - 1e-15:
        return 0.0
    deltas = np.linspace(lo, hi, 64)
    vals = r[None, :] + deltas[:, None] * d[None, :]
    maxes = vals.max(axis=1)
    idx = int(maxes.argmin())
    if idx == 0:
        return lo
    if idx == len(deltas) - 1:
        return hi
    f_lo, f_hi = deltas[idx-1], deltas[idx+1]
    for _ in range(20):
        mid = 0.5 * (f_lo + f_hi)
        rm = r + mid * d
        mv = rm.max()
        active = rm >= mv - 1e-12
        if d[active].min() > 0:
            f_hi = mid
        elif d[active].max() < 0:
            f_lo = mid
        else:
            break
    return 0.5 * (f_lo + f_hi)

def _pair_sweep(h, n, dx, n_passes=300):
    r = _corr(h)
    for _ in range(n_passes):
        any_change = False
        for i in range(n):
            ci = _coef_vec(h, i, n)
            hi = h[i]
            best_obj = float(r.max())
            best_d, best_j = 0.0, -1
            for j in range(n):
                if j == i:
                    continue
                hj = h[j]
                dlo = max(-hi, hj - 1.0)
                dhi = min(1.0 - hi, hj)
                if dlo >= dhi - 1e-13:
                    continue
                cj = _coef_vec(h, j, n)
                dd = ci - cj
                delta = _min_max_lin(r, dd, dlo, dhi)
                obj = float(np.max(r + delta * dd))
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_d, best_j = delta, j
            if best_j >= 0:
                h[i] += best_d
                h[best_j] -= best_d
                r = _corr(h)
                any_change = True
        if not any_change:
            break
    return h

    def obj(x):
        c = _corr(x)
        m = c.max()
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return np.sum(x) - n / 2.0
```

```python
def _all_coefs(h, n):
    nc = 2 * n - 1
    coefs = np.zeros((n, nc))
    one_minus_h = 1.0 - h
    for i in range(n):
        # First part: (1-h[i+m-n+1]) for valid m
        # j = i + m - n + 1, valid when 0 <= j < n → n-1-i <= m <= 2n-2-i
        m_start = max(0, n - 1 - i)
        m_end = min(2 * n - 2, 2 * n - 2 - i)
        if m_start <= m_end:
            j_start = i + m_start - n + 1  # = max(0, ...)
            coefs[i, m_start:m_end+1] += one_minus_h[j_start:j_start+(m_end-m_start+1)]
        # Second part: -h[i-m+n-1] for valid m
        # j = i - m + n - 1, valid when 0 <= j < n → i <= m <= i+n-1
        m_start2 = max(0, i)

        if m_start2 <= m_end2:

            # j decreases as m increases
            coefs[i, m_start2:m_end2+1] -= h[j_end2:j_start2+1][::-1]
    return coefs
```

```python
                any_change = True
        if not any_change:

    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _all_coefs(h, n):
    nc = 2 * n - 1
    coefs = np.zeros((n, nc))
    omh = 1.0 - h
    for i in range(n):
        ms = max(0, n - 1 - i)
        me = min(2 * n - 2, 2 * n - 2 - i)
        if ms <= me:
            js = max(0, i + ms - n + 1)
            coefs[i, ms:me + 1] += omh[js:js + (me - ms + 1)]
        ms2 = max(0, i)
        me2 = min(2 * n - 2, i + n - 1)
        if ms2 <= me2:
            je = i - me2 + n - 1
            js2 = i - ms2 + n - 1
            coefs[i, ms2:me2 + 1] -= h[je:js2 + 1][::-1]
    return coefs

def _min_max_lin(r, d, lo, hi):
    if lo >= hi - 1e-14:
        return 0.0
    deltas = np.linspace(lo, hi, 48)
    vals = r[None, :] + deltas[:, None] * d[None, :]
    maxes = vals.max(axis=1)
    idx = int(maxes.argmin())
    if idx == 0:
        return lo
    if idx == len(deltas) - 1:
        return hi
    f_lo, f_hi = deltas[idx - 1], deltas[idx + 1]
    for _ in range(25):
        mid = 0.5 * (f_lo + f_hi)
        rm = r + mid * d
        mv = float(rm.max())
        active = rm >= mv - 1e-13
        sg_lo = float(d[active].min())
        sg_hi = float(d[active].max())
        if sg_lo > 0:
            f_hi = mid
        elif sg_hi < 0:
            f_lo = mid
        else:
            break
    return 0.5 * (f_lo + f_hi)

def _pair_sweep(h, n, dx, max_passes=400):
    r = _corr(h)
    cur_max = float(r.max())
    for _ in range(max_passes):
        any_chg = False
        coefs = _all_coefs(h, n)
        for i in range(n):
            ci = coefs[i]
            hi_v = h[i]
            best_obj = cur_max + 1e-9
            best_d, best_j = 0.0, -1
            for j in range(n):
                if j == i:
                    continue
                hj_v = h[j]
                dlo = max(-hi_v, hj_v - 1.0)
                dhi = min(1.0 - hi_v, hj_v)
                if dlo >= dhi - 1e-13:
                    continue
                dd = ci - coefs[j]
                delta = _min_max_lin(r, dd, dlo, dhi)
                obj = float(np.max(r + delta * dd))
                if obj < best_obj:
                    best_obj = obj
                    best_d, best_j = delta, j
            if best_j >= 0:
                h[i] += best_d
                h[best_j] -= best_d
                r = _corr(h)
                cur_max = float(r.max())
                any_chg = True
        if not any_chg:
            break
    return h

def _slsqp_sm(h, n, alpha, maxiter):
    def obj(x):
        c = _corr(x)
        m = float(c.max())
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return float(np.sum(x) - n / 2.0)
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _perturb_reopt(h, n, rng, noise_scale, max_passes):
    hp = h + noise_scale * rng.randn(n)
    hp = _repair(hp, n)
    hp = _pair_sweep(hp, n, 0.0, max_passes)
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Intensive pair sweep from initial
    if rem() > 350:
        h = _pair_sweep(best_h.copy(), n, dx, 400)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: SLSQP polish
    if rem() > 280:
        h2 = _slsqp_sm(best_h.copy(), n, 150, 600)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 3: Pair sweep again
    if rem() > 200:
        h = _pair_sweep(best_h.copy(), n, dx, 400)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Perturbation + re-optimization (escape local optima)
    for attempt in range(15):
        if rem() < 80:
            break
        scale = 0.04 if attempt % 3 == 0 else (0.02 if attempt % 3 == 1 else 0.08)
        mp = 80 if rem() < 150 else 150
        hp = _perturb_reopt(best_h, n, rng, scale, mp)
        c5 = _c5(hp, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Phase 5: SLSQP with different alpha
    if rem() > 50:
        h2 = _slsqp_sm(best_h.copy(), n, 400, 800)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 6: Final pair sweep
    if rem() > 20:
        h = _pair_sweep(best_h.copy(), n, dx, 300)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 7: Tiny SLSQP polish
    if rem() > 8:
        try:
            h2 = _slsqp_sm(best_h.copy(), n, 1000, 500)
            c5 = _c5(h2, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h2
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
