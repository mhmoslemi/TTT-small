import numpy as np
from scipy.optimize import minimize
import time

_GN = 70
_GA = 100.0
_GLAM = 0.0

def _sum_con(x):
    return float(np.sum(x)) - _GN * 0.5

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_arr(h))) * dx

def _smooth_obj(x):
    c = _corr_arr(x)
    m = float(np.max(c))
    s = np.clip(_GA * (c - m), -300, 300)
    val = m + float(np.log(np.sum(np.exp(s)))) / _GA
    if _GLAM > 0:
        val += _GLAM * (float(np.sum(x)) - _GN * 0.5) ** 2
    return val

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(600):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < 1e-13:
            h.fill(0.5)
            break
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 5e-14:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 5e-14:
                break
            if diff > 0:
                amt = min(diff, 1.0 - h[idx])
                h[idx] += amt
                diff -= amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt
                diff += amt
    return h

def _slsqp_run(h, n, alpha, maxit, t_end):
    global _GN, _GA, _GLAM
    _GN, _GA, _GLAM = n, alpha, 0.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_run(h, n, alpha, pen, maxit, t_end):
    global _GN, _GA, _GLAM
    _GN, _GA, _GLAM = n, alpha, pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-9})
        return _repair(res.x, n)

    h = h.copy()
