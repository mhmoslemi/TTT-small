import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def _lse_obj(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    return m * dx + np.log(np.sum(np.exp(s))) * dx / alpha

def _topk_lse(x, alpha, k, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m * dx + np.log(np.sum(np.exp(s))) * dx / alpha

def _sumsq_topk(x, k, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx]
    return np.sum(c * c) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        h[:] = 0.5
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

    best_c5, _ = _compute_c5(h, dx)

    t_local = time.time()
    t_dur = deadline - t_local
    if t_dur <= 0:

    for si, scale in enumerate(scales):
        stall = 0
        while time.time() < deadline and stall < 8:
            stall += 1
            for _ in range(4000):
                if time.time() >= deadline:
                    return h, best_c5
                i = rng.randint(n)

                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                hi = h[i] + d

                if hi < -1e-9 or hi > 1.0 + 1e-9 or hj < -1e-9 or hj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = hi, hj
                c5, _ = _compute_c5(h, dx)
                accept = c5 < best_c5 - 1e-11
                if use_sa:
                    frac = (t_local + (time.time() - t_local)) / max(t_dur, 1e-9)
                    temp = 0.002 * (1.0 - frac) * (1.0 - si / len(scales))

                            accept = True
                if accept:
                    best_c5 = c5
                    stall = 0
                else:
                    h[i], h[j] = hi - d, hj + d
