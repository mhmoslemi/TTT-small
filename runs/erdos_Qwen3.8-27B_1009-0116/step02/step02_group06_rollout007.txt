def _optimize_coord_exact(h, i, c, n, dx):
    offset = n - 1
    coeff = np.zeros(2 * n - 1)
    for k in range(max(-(n-1), i-(n-1)), min(n-1, i)+1):
        j = i - k
        coeff[k + offset] += (1.0 - h[j])

        j = i + k
        coeff[k + offset] -= h[j]
    
    lo = -h[i]
    hi = 1.0 - h[i]

    vals = {lo: np.max(c + lo * coeff), 0: np.max(c), hi: np.max(c + hi * coeff)}
    best_d = min(vals, key=vals.get)

    # For lines a and b: c[a] + d*coeff[a] = c[b] + d*coeff[b]

    nz = np.nonzero(coeff)[0]
    m = len(nz)

    offset = n - 1
    coeff = np.zeros(2 * n - 1)
    # Vectorized coefficient computation
    # Term 1: for k in [max(-(n-1), i-(n-1)), min(n-1, i)]: coeff[k+offset] += 1-h[i-k]
    k_range1 = np.arange(max(-(n-1), i-(n-1)), min(n-1, i)+1)
    j_range1 = i - k_range1
    idx1 = k_range1 + offset
    coeff[idx1] += 1.0 - h[j_range1]
    # Term 2: for k in [max(-(n-1), -i), min(n-1, n-1-i)]: coeff[k+offset] -= h[i+k]
    k_range2 = np.arange(max(-(n-1), -i), min(n-1, n-1-i)+1)
    j_range2 = i + k_range2
    idx2 = k_range2 + offset
    coeff[idx2] -= h[j_range2]
    
    lo = -h[i]
    hi = 1.0 - h[i]
    
    # Evaluate at boundaries
    v_lo = np.max(c + lo * coeff)
    v_0 = np.max(c)
    v_hi = np.max(c + hi * coeff)
    best_val = min(v_lo, v_0, v_hi)
    if best_val == v_lo: best_d = lo
    elif best_val == v_0: best_d = 0.0
    else: best_d = hi
    
    # Vectorized pairwise intersections
    nz = np.nonzero(coeff)[0]
    m = len(nz)
    if m > 1:
        ca = c[nz]  # (m,)
        co = coeff[nz]  # (m,)
        # All pairwise intersections
        # d_ab = (ca[b] - ca[a]) / (co[a] - co[b])
        ca_ab = ca[None, :] - ca[:, None]  # (m,m) -- wait, need to be careful
        co_ab = co[:, None] - co[None, :]  # (m,m)
        # d[a,b] = (ca[b] - ca[a]) / (co[a] - co[b]) = -ca_ab / co_ab
        valid = np.abs(co_ab) > 1e-15
        d_mat = np.zeros((m, m))
        d_mat[valid] = -ca_ab[valid] / co_ab[valid]
        # Only upper triangle
        d_mat = np.triu(d_mat, 1)
        # Filter to (lo, hi)
        valid2 = (d_mat > lo + 1e-14) & (d_mat < hi - 1e-14) & (d_mat != 0)
        if np.any(valid2):

            evals = (c[None, :] + d_cands[:, None] * coeff[None, :]).max(axis=1)
            bi = np.argmin(evals)
            if evals[bi] < best_val - 1e-15:
                best_val = evals[bi]
                best_d = d_cands[bi]

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _get_coeff(h, i, n):
    """Get the coefficient vector for changing h[i]."""
    coeff = np.zeros(2 * n - 1)
    offset = n - 1
    k1 = np.arange(max(-(n-1), i-(n-1)), min(n-1, i)+1)
    j1 = i - k1
    coeff[k1 + offset] += 1.0 - h[j1]
    k2 = np.arange(max(-(n-1), -i), min(n-1, n-1-i)+1)
    j2 = i + k2
    coeff[k2 + offset] -= h[j2]
    return coeff

def _opt1d(c, coeff, h_i, n):
    """Find best delta to add to h_i. Returns (delta, new_max_val)."""
    lo = -h_i
    hi = 1.0 - h_i
    if hi - lo < 1e-15:
        return 0.0, 0.0
    
    # Evaluate at boundaries
    v_lo = float(np.max(c + lo * coeff))
    v_hi = float(np.max(c + hi * coeff))
    v_0 = float(np.max(c))
    
    best_val = min(v_lo, v_0, v_hi)
    if best_val == v_lo:
        best_d = lo
    elif best_val == v_hi:
        best_d = hi
    else:

        co = coeff[nz]
        ca_diff = ca[None, :] - ca[:, None]  # ca[b] - ca[a] for row a, col b
        co_diff = co[:, None] - co[None, :]  # co[a] - co[b]
        valid = np.abs(co_diff) > 1e-14
        d_mat = np.full((m, m), np.nan)
        d_mat[valid] = -ca_diff[valid] / co_diff[valid]
        d_mat = np.triu(d_mat, 1)
        mask = ~np.isnan(d_mat) & (d_mat > lo + 1e-13) & (d_mat < hi - 1e-13)
        if np.any(mask):
            d_cands = d_mat[mask]
            evals = (c[None, :] + d_cands[:, None] * coeff[None, :]).max(axis=1)
            bi = int(np.argmin(evals))
            if evals[bi] < best_val - 1e-14:
                best_val = float(evals[bi])
                best_d = float(d_cands[bi])
    
    return best_d, best_val

def _full_sweep(h, n, dx, max_passes=200):

    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(c) * dx)
    h = h.copy()
    
    for p in range(max_passes):
        improved = False
        for i in range(n):

            d, new_v = _opt1d(c, coeff, h[i], n)
            if abs(d) > 1e-12:

                c = c + d * coeff
                if new_v * dx < best_c5 - 1e-14:
                    best_c5 = new_v * dx

def _pair_sweep(h, n, dx, pairs, inner_iter=8):

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _dcf(h, i, n):
    coeff = np.zeros(2 * n - 1)
    offset = n - 1
    k_lo1 = max(-(n - 1), i - (n - 1))
    k_hi1 = min(n - 1, i)
    k1 = np.arange(k_lo1, k_hi1 + 1)
    coeff[k1 + offset] += 1.0 - h[i - k1]
    k_lo2 = max(-(n - 1), -i)
    k_hi2 = min(n - 1, n - 1 - i)
    k2 = np.arange(k_lo2, k_hi2 + 1)
    coeff[k2 + offset] -= h[i + k2]
    return coeff

def _opt1d(c, coeff, h_i):
    lo = -h_i
    hi = 1.0 - h_i
    if hi - lo < 1e-15:
        return 0.0
    v0 = float(np.max(c))
    vlo = float(np.max(c + lo * coeff))
    vhi = float(np.max(c + hi * coeff))
    best_val = v0
    best_d = 0.0
    if vlo < best_val:
        best_val, best_d = vlo, lo
    if vhi < best_val:
        best_val, best_d = vhi, hi
    nz = np.nonzero(coeff)[0]
    m = len(nz)
    if m > 1:
        ca = c[nz]
        co = coeff[nz]
        cd = ca[None, :] - ca[:, None]
        co_d = co[:, None] - co[None, :]
        valid = np.abs(co_d) > 1e-14
        dmat = np.full((m, m), np.nan)
        dmat[valid] = -cd[valid] / co_d[valid]
        dmat = np.triu(dmat, 1)
        mask = ~np.isnan(dmat) & (dmat > lo + 1e-13) & (dmat < hi - 1e-13)
        if np.any(mask):
            dc = dmat[mask]
            ev = (c[None, :] + dc[:, None] * coeff[None, :]).max(axis=1)
            bi = int(np.argmin(ev))
            if ev[bi] < best_val - 1e-14:
                best_val = float(ev[bi])
                best_d = float(dc[bi])
    return best_d

def _sweep(h, n, dx, max_passes=300):
    c = np.correlate(h, 1.0 - h, mode='full')
    h = h.copy()
    for _ in range(max_passes):
        moved = False
        for i in range(n):
            cf = _dcf(h, i, n)
            d = _opt1d(c, cf, h[i])
            if abs(d) > 1e-13:
                h[i] += d
                c += d * cf
                moved = True
        if not moved:
            break
    return h, float(np.max(c) * dx)

def _pair_opt(h, n, dx, pairs, iters=6):
    c = np.correlate(h, 1.0 - h, mode='full')
    h = h.copy()
    for (i, j) in pairs:
        for _ in range(iters):
            cfi = _dcf(h, i, n)
            di = _opt1d(c, cfi, h[i])
            if abs(di) > 1e-13:
                h[i] += di
                c += di * cfi
            cfj = _dcf(h, j, n)
            dj = _opt1d(c, cfj, h[j])
            if abs(dj) > 1e-13:
                h[j] += dj
                c += dj * cfj
    return h, float(np.max(c) * dx)

def _top_pairs(h, n, topk=15):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(c)
    active = np.where(c > 0.85 * mx)[0]
    offset = n - 1
    scores = np.zeros(n)
    for lag_idx in active:
        lag = lag_idx - offset
        for i in range(n):
            if 0 <= i + lag < n:
                scores[i] += 1
    top = np.argsort(scores)[-topk:]
    pairs = []
    for a in range(len(top)):
        for b in range(a + 1, len(top)):
            pairs.append((int(top[a]), int(top[b])))
    return pairs

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Exact coordinate descent (many passes)
    h = best_h.copy()
    for rnd in range(8):
        if time.time() - t0 > budget_s - 400:
            break
        h, c5 = _sweep(h, n, dx, max_passes=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Active-pair optimization
    if time.time() - t0 < budget_s - 250:
        for pp in range(8):
            if time.time() - t0 > budget_s - 250:
                break
            pairs = _top_pairs(best_h, n, topk=15)
            h = best_h.copy()
            h, c5 = _pair_opt(h, n, dx, pairs, iters=8)
            h, c5 = _sweep(h, n, dx, max_passes=200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: All-pair sweep in chunks
    if time.time() - t0 < budget_s - 120:
        all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(all_pairs)
        h = best_h.copy()
        for st in range(0, len(all_pairs), 300):
            if time.time() - t0 > budget_s - 120:
                break
            chunk = all_pairs[st:st + 300]
            h, c5 = _pair_opt(h, n, dx, chunk, iters=4)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final coordinate polish
    if time.time() - t0 < budget_s - 20:
        h = best_h.copy()
        h, c5 = _sweep(h, n, dx, max_passes=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Tiny perturbations + local opt
    if time.time() - t0 < budget_s - 10:
        for scale in [0.005, 0.01, 0.02, 0.005]:
            if time.time() - t0 > budget_s - 10:
                break
            h_p = best_h + scale * rng.randn(n)
            h_p = _repair(h_p, n)
            h_p, c5 = _sweep(h_p, n, dx, max_passes=300)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_p.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
