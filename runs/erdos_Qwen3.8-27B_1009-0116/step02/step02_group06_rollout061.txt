def _project(y, target):
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _smo_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smo_obj(x, alpha, dx):
    return _smo_max(_corr(x), alpha) * dx

def _smo_grad(x, alpha, dx, n):
    c = _corr(x)
    m = np.max(c)
    logw = np.clip(alpha * (c - m), -300, 0)
    lwm = np.max(logw)
    w = np.exp(logw - lwm)
    ws = np.sum(w)
    if ws < 1e-300:

    else:
        w /= ws

        if wi < 1e-18:
            continue
        a_lo, a_hi = max(0, s), min(n, n+s)

        b_lo, b_hi = max(0, -s), min(n, n-s)

    return grad * dx

                      constraints=[{'type':'eq','fun':eq}],

        return _project(res.x, target)
    except Exception:
        return h

def _grad_mat(h, n):
    G = np.zeros((n, 2*n-1))
    for i in range(n):
        for so in range(2*n-1):
            s = so - (n-1)
            v = 0.0
            a = i - s
            if 0 <= a < n:
                v += 1.0 - h[a]
            b = i + s
            if 0 <= b < n:
                v -= h[b]
            G[i, so] = v
    return G

    best = float(np.max(corr) * dx)
    for _ in range(mr):

            break
        G = _grad_mat(h, n)
        imp = False
        for i in range(n):

                d_hi = min(1.0-h[i], h[j])

                gc = G[i] - G[j]
                ds = np.linspace(d_lo, d_hi, nc)
                vals = np.max(corr[None,:] + ds[:,None]*gc[None,:], axis=1)

                        imp = True
        c5v = float(np.max(corr) * dx)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(y, target):
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _smo_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _eq_fun(x, target):
    return float(np.sum(x) - target)

def _eq_jac(x, target):
    return np.ones(len(x))

def _obj_wrap(x, alpha, dx):
    return _smo_max(_corr(x), alpha) * dx

def _obj_jac(x, alpha, dx):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    logw = np.clip(alpha * (c - m), -300, 0)
    lwm = np.max(logw)
    w = np.exp(logw - lwm)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= ws
    grad = np.zeros(n)
    for so in range(2 * n - 1):
        s = so - (n - 1)
        wi = w[so]
        if wi < 1e-18:
            continue
        a_lo, a_hi = max(0, s), min(n, n + s)
        if a_lo < a_hi:
            grad[a_lo:a_hi] += wi * (1.0 - x[np.arange(a_lo, a_hi) - s])
        b_lo, b_hi = max(0, -s), min(n, n - s)
        if b_lo < b_hi:
            grad[b_lo:b_hi] -= wi * x[np.arange(b_lo, b_hi) + s]
    return grad * dx

def _slsqp(h, n, dx, target, alpha, maxiter):
    try:
        res = minimize(_obj_wrap, h, args=(alpha, dx), method='SLSQP',
                       jac=_obj_jac, bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _eq_fun,
                                     'jac': _eq_jac, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(res.x, target)
    except Exception:
        return h

def _grad_mat(h, n):
    G = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for so in range(2 * n - 1):
            s = so - (n - 1)
            v = 0.0
            a = i - s
            if 0 <= a < n:
                v += 1.0 - h[a]
            b = i + s
            if 0 <= b < n:
                v -= h[b]
            G[i, so] = v
    return G

def _sweep(h, n, dx, nc, nf, fw, mr, t_end):
    corr = _corr(h).copy()
    best = float(np.max(corr) * dx)
    for _ in range(mr):
        if time.time() >= t_end:
            break
        G = _grad_mat(h, n)
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                gc = G[i] - G[j]
                ds = np.linspace(d_lo, d_hi, nc)
                vals = np.max(corr[None, :] + ds[:, None] * gc[None, :], axis=1)
                bi = int(np.argmin(vals))
                if vals[bi] < np.max(corr) - 1e-15:
                    db = ds[bi]
                    if nf > 2 and fw > 0:
                        lo2 = max(d_lo, db - fw)
                        hi2 = min(d_hi, db + fw)
                        ds2 = np.linspace(lo2, hi2, nf)
                        vals2 = np.max(corr[None, :] + ds2[:, None] * gc[None, :], axis=1)
                        db = ds2[int(np.argmin(vals2))]
                    if abs(db) > 1e-14:
                        h[i] += db
                        h[j] -= db
                        corr += db * gc
                        imp = True
        c5v = float(np.max(corr) * dx)
        if c5v < best:
            best = c5v
        if not imp:
            break
    return h, best

def _pgrad_step(h, n, dx, target, alpha, n_iter):
    y = h.copy()
    lr = 0.02
    mom = 0.85
    for it in range(n_iter):
        grad = _obj_jac(y, alpha, dx)
        h_new = _project(y - lr * grad, target)
        y = h_new + mom * (h_new - y)
        h = h_new
        lr *= 0.998
    return h
