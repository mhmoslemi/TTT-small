    if val_hi < best_val:
        best_val, best_t = val_hi, t_hi
    
    # Bisection
    lo, hi = t_lo, t_hi
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = C + mid * g
        f_mid = vals.max()

                lo = mid
            else:

    ks = np.arange(2 * n - 1)
    # index into h for h[i+k-n+1]

    b = i - ks + n - 1
    c = j + ks - n + 1
    d = j - ks + n - 1
    
    def safe_idx(idx):
        return np.where((idx >= 0) & (idx < n), h[np.clip(idx, 0, n-1)], 0.0)
    
    g = (1.0 - safe_idx(a)) - safe_idx(b) - (1.0 - safe_idx(c)) + safe_idx(d)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _c5val(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _compute_g(h, n, i, j):
    ks = np.arange(2 * n - 1)
    a = i + ks - n + 1
    b = i - ks + n - 1
    c = j + ks - n + 1
    d = j - ks + n - 1
    def _sg(idx):
        m = (idx >= 0) & (idx < n)
        return np.where(m, h[np.clip(idx, 0, n - 1)], 0.0)
    return (1.0 - _sg(a)) - _sg(b) - (1.0 - _sg(c)) + _sg(d)

def _solve_1d(C, g, t_lo, t_hi):
    best_t = 0.0
    best_f = float(C.max())
    for tc in [t_lo, 0.0, t_hi]:
        fv = float(np.max(C + tc * g))
        if fv < best_f:
            best_f, best_t = fv, tc
    lo, hi = t_lo, t_hi
    for _ in range(45):
        if hi - lo < 1e-16:
            break
        mid = (lo + hi) * 0.5
        vals = C + mid * g
        fmid = float(vals.max())
        if fmid < best_f:
            best_f, best_t = fmid, mid
        active = vals >= fmid - 1e-13
        if not active.any():
            break
        ga = g[active]
        if ga.min() >= -1e-14:
            hi = mid
        elif ga.max() <= 1e-14:
            lo = mid
        else:
            break
    return best_t, best_f

def _cd_pass(h, n, C_cur, pairs):
    imps = 0
    for (i, j) in pairs:
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-14:
            continue
        g = _compute_g(h, n, i, j)
        t_opt, f_opt = _solve_1d(C_cur, g, t_lo, t_hi)
        if abs(t_opt) > 1e-11 and f_opt < float(C_cur.max()) - 1e-13:
            h[i] = np.clip(h[i] + t_opt, 0.0, 1.0)
            h[j] = np.clip(h[j] - t_opt, 0.0, 1.0)
            C_cur = C_cur + t_opt * g
            imps += 1
    C_cur = np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha
