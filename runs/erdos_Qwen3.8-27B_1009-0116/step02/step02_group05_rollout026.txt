def _corr(h):

def _c5(h, dx):

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_obj(h, alpha, dx):
    x = _corr(h)
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h.copy()

**Alternative idea: Use the KKT conditions**

At the optimum, the KKT conditions say:
- For the active constraints (those k where c_k = max), the gradient of c_k should be a combination of gradients of inactive c_k's plus the constraint gradients.
- Specifically, if the max is achieved at k*, then ∇c_{k*} should be "balanced" by the other constraints.

This suggests an algorithm:
1. Find the current maximizer k*
2. Try to decrease c_{k*} while not increasing any other c_k above the current max
3. This is a linear constraint optimization (approximately, if we linearize c_k around the current point)

Linearized: c_k + ∇c_k · Δh ≤ current_max for all k, and c_{k*} + ∇c_{k*} · Δh ≤ current_max - ε

This is a linear program! We can solve it quickly.

def _linearize_step(h, n, dx, kstar, eps):
    # Compute gradient of each c_k w.r.t. h
    # c_k = sum_j h[j+off(k)] * (1-h[j])
    # dc_k/dh[i] = contribution from h[i] appearing as first factor minus as (1-h) factor
    
    corr = _corr(h)
    current_max = np.max(corr)
    
    # Compute gradients numerically (cheap for n=70)
    grad = np.zeros((2*n-1, n))
    for i in range(n):
        h2 = h.copy()
        h2[i] += 1e-7
        c2 = _corr(h2)
        grad[:, i] = (c2 - corr) / 1e-7
        h2[i] -= 2e-7
        c2 = _corr(h2)
        grad[:, i] += (corr - c2) / 1e-7
        h2[i] += 1e-7
    
    # Solve LP: minimize grad[kstar] · Δh (to decrease c_kstar)
    # s.t. grad[k] · Δh <= current_max - corr[k] - eps for all k != kstar
    #      grad[kstar] · Δh <= -eps  (decrease by at least eps)
    #      |Δh[i]| <= step_size
    #      sum(Δh) = 0 (to maintain sum constraint)
    
    # Actually, let's just use scipy to solve this LP
    from scipy.optimize import linprog
    
    step_size = 0.05
    # Variables: Δh[0], ..., Δh[n-1]
    # Minimize: grad[kstar] · Δh
    c_vec = grad[kstar]
    
    A_ub = []
    b_ub = []
    for k in range(2*n-1):
        if k == kstar:
            continue
        A_ub.append(grad[k].tolist())
        b_ub.append((current_max - corr[k]) - eps)
    # Must decrease c_kstar
    A_ub.append(-grad[kstar].tolist())
    b_ub.append(eps)
    
    A_eq = [[1.0]*n]
    b_eq = [0.0]
    
    bounds = [(-step_size, step_size)] * n
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = h + res.x
        h_new = _repair(h_new, n)

    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_obj(x, alpha, dx):
    cp = _corr(x)
    m = np.max(cp)
    s = np.clip(alpha * (cp - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _sum_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_con, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _numerical_grad(h, n, alpha):
    eps = 1e-7
    grad = np.zeros(n)
    for i in range(n):
        h[i] += eps
        gp = _smooth_obj(h, alpha, 1.0)
        h[i] -= 2.0 * eps
        gm = _smooth_obj(h, alpha, 1.0)
        h[i] += eps
        grad[i] = (gp - gm) / (2.0 * eps)
    return grad

def _local_search(h, n, dx, rng, deadline, alpha_g):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    step_scale = 1.0
    rnd = 0
    base_steps = [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.2]
    while time.time() < deadline:
        rnd += 1
        if rnd % 150 == 0 and time.time() < deadline - 80:
            pol = _slsqp(best, n, dx, alpha_g, 120, 1e-13)
            pc = _c5(pol, dx)
            if pc < best_c5:
                best = pol
                best_c5 = pc
        steps = np.array(base_steps) * step_scale
        all_steps = np.concatenate([steps, -steps])
        improved = False
        ntrials = 350
        for _ in range(ntrials):
            if rnd % 8 == 0:
                grad = _numerical_grad(best, n, alpha_g)
                i = int(np.argmin(grad))
                j = int(np.argmax(grad))
                if i == j:
                    i = int(rng.randint(n))
                    while j == i:
                        j = int(rng.randint(n))
                d = float(rng.choice(steps))
            else:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                d = float(rng.choice(all_steps))
            if i == j:
                continue
            ni = best[i] + d
            nj = best[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-13:
                best_c5 = nc
                improved = True
                step_scale = min(step_scale * 1.05, 1.0)
            else:
                best[i], best[j] = oi, oj
            if improved:
                break
        if improved:
            stall = 0
        else:

                best_c5 = _c5(best, dx)
                step_scale = 1.0
            if stall > 100:
