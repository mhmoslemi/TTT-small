from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lse_obj(x, alpha, n):
    corr = _corr(x)
    m = float(corr.max())
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + s.sum() / alpha) * (2.0 / n)

def _lse_grad(x, alpha, n):
    corr = _corr(x)
    m = float(corr.max())
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    dv = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dv
        if wi < 1e-14:

        lag = idx - (n - 1)

                if j <= n - 1 - lag:
                    grad[j] -= wi * x[j + lag]

                if j <= n - 1 - ma:
                    grad[j] += wi * (1.0 - x[j + ma])
                if j >= ma:

            break
        improved = False
        # Compute gradient with high alpha
        corr = _corr(h)
        m = corr.max()
        alpha = 5000
        w = np.exp(alpha * (corr - m))
        w /= w.sum()
        dv = 2.0 / n
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            wi = w[idx] * dv
            if wi < 1e-14:
                continue
            lag = idx - (n - 1)
            if lag >= 0:
                for j in range(n):
                    if lag <= j:
                        grad[j] += wi * (1.0 - h[j - lag])
                    if j <= n - 1 - lag:
                        grad[j] -= wi * h[j + lag]
            else:
                ma = -lag
                for j in range(n):
                    if j <= n - 1 - ma:
                        grad[j] += wi * (1.0 - h[j + ma])
                    if j >= ma:
                        grad[j] -= wi * h[j - ma]
        
        for i in range(n):
            if time.time() > budget_end:
                break
            g_i = grad[i]
            if abs(g_i) < 1e-8:
                continue
            direction = -1.0 if g_i > 0 else 1.0
            for step in [0.002, 0.001, 0.0005, 0.0002]:
                d = direction * step
                ni = h[i] + d
                if ni < 0 or ni > 1:
                    continue
                need = -d
                # Find best j
                if need > 0:
                    cands = np.argsort(h)[:12]
                else:
                    cands = np.argsort(-h)[:12]
                for j in cands:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5_try = _c5(h, dx)
                    if c5_try < best - 1e-15:
                        best = c5_try
                        improved = True
                        break
                    else:
                        h[i], h[j] = h[i] - d, h[j] - need
                if improved:
                    break
        if not improved:
            break
    return h, best
