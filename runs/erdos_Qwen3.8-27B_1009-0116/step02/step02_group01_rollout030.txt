    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_c5_corr(h)) * dx

def _grad_corr(h, k):
    """Gradient of correlation at index k w.r.t. h"""
    n = len(h)
    d = k - n + 1  # lag
    g = np.zeros(n)
    for j in range(n):
        if 0 <= j + d < n:
            g[j] += (1.0 - h[j + d])
        if 0 <= j - d < n:
            g[j] -= h[j - d]
    return g

def _grad_max(h, alpha=50):

    c = _c5_corr(h)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    weights = np.exp(s) / np.sum(np.exp(s))
    # Gradient is weighted sum of individual gradients
    g = np.zeros(len(h))
    for idx, w in enumerate(weights):

            g += w * _grad_corr(h, idx)
    return g

                diff -= add

                diff += sub

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, -d)
    hi1 = min(n, n - d)
    if hi1 > lo1:
        g[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
    lo2 = max(0, d)
    hi2 = min(n, n + d)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 - d:hi2 - d]
    return g

def _grad_lse(h, alpha):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] > 1e-8:
            d = k - n + 1
            g += p[k] * _grad_lag(h, d)
    return g, m

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha

    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        v, _ = _smooth_obj_and_grad(x, alpha)
        return v
    
    def grad(x):
        _, g = _smooth_obj_and_grad(x, alpha)
        return g
    
    def cons_fun(x):
        return np.sum(x) - target
    
    cons = [cons_fun]  # SLSQP handles eq constraints as 0
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                        bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_fun}],
                        options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() - t0 > t_limit:
            return None
        return _repair(res.x, n)
    except Exception:
        return None

def _projected_grad(h, alpha, step_scale, n_iters, dx, t_limit):
    h = h.copy()
    c5_cur = _c5(h, dx)
    t0 = time.time()
    step = step_scale
    for it in range(n_iters):

        g, _ = _grad_lse(h, alpha)
        # Projected gradient descent with backtracking
        lr = step

            c5_new = _c5(h_new, dx)
            obj_new = _smooth_obj_and_grad(h_new, alpha)[0]
            obj_cur = _smooth_obj_and_grad(h, alpha)[0]

            lr *= 0.5
        if c5_new < c5_cur:

            c5_cur = c5_new
        step *= 0.99

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    lo1, hi1 = max(0, -d), min(n, n - d)
    if hi1 > lo1:
        g[lo1:hi1] += 1.0 - h[lo1 + d:hi1 + d]
    lo2, hi2 = max(0, d), min(n, n + d)
    if hi2 > lo2:
        g[lo2:hi2] -= h[lo2 - d:hi2 - d]
    return g

def _lsed(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    g = np.zeros(len(h))
    for k in np.where(p > 1e-10)[0]:
        g += p[k] * _grad_lag(h, k - len(h) + 1)
    return m + np.log(e.sum()) / alpha, g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _slsqp(h0, n, alpha, mx, ft, tm):
    tgt = n / 2.0
    def f(x):
        return _lsed(x, alpha)[0]
    def g(x):
        return _lsed(x, alpha)[1]
    def cf(x):
        return x.sum() - tgt
    try:
        r = minimize(f, h0, jac=g, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': cf}],
                     options={'maxiter': mx, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return None

def _projgrad(h, alpha, steps, dx, tm):
    h = h.copy()
    t0 = time.time()
    for it in range(steps):
        if time.time() - t0 > tm:
            break
        val, g = _lsed(h, alpha)
        lr = 0.02 * (0.97 ** (it // 80))
        improved = False
        for _ in range(25):
            hn = _repair(h - lr * g, len(h))
            vn, _ = _lsed(hn, alpha)
            cn = _c5(hn, dx)
            co = _c5(h, dx)
            if vn < val or cn < co:
                h = hn
                val, _ = _lsed(h, alpha)
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
    return h

def _make_h(n, rng, stype):
    if stype == 0:
        h = rng.rand(n) * 0.4 + 0.3
    elif stype == 1:

    else:
        h = rng.rand(n) * 0.5 + 0.25
    return _repair(h, n)
