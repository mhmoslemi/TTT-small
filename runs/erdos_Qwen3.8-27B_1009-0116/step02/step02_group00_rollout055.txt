    shift = k - (n - 1)
    grad = np.zeros(n)

    b = min(n, n - shift)

    b2 = min(n, n + shift)
    grad[a2:b2] -= h[a2-shift:b2-shift]

    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing LSE sharpness
    p1_end = t_start + min(budget_s * 0.10, 100)
    for alpha in [200, 600, 1500, 3000]:
        if time.time() > p1_end:
            break
        def obj_p1(x, a=alpha):
            return _lse_obj(_corr(x), a, n) + 2000.0 * (np.sum(x) - target) ** 2
        try:
            res = minimize(obj_p1, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-12})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Projected gradient with adaptive LSE and peak targeting
    p2_end = t_start + min(budget_s * 0.52, 520)
    h = best_h.copy()
    pg_best = best_c5
    alpha_cur = 100.0
    step_cur = 0.02
    stall = 0

    while time.time() < p2_end:
        c5_val = _c5(h, dx)
        if c5_val < pg_best - 1e-14:
            pg_best = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 400:
            scale = 0.015 + 0.02 * min(stall / 400.0, 2.0)
            h = _repair(h + rng.randn(n) * scale, n)
            pg_best = _c5(h, dx)
            stall = 0
            alpha_cur = max(50.0, alpha_cur * 0.6)
            step_cur = min(0.04, step_cur * 1.4)
            continue

        g = _grad_weighted(h, n, alpha_cur)
        se = float(np.sum(h)) - target
        g += 4.0 * se
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        accepted = False
        for _ in range(18):
            h_new = _repair(h - step_cur * g, n)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_val - 1e-16:
                h = h_new
                accepted = True
                break
            step_cur *= 0.65
        if not accepted:
            step_cur = max(step_cur * 0.5, 0.0008)

        if stall % 80 == 79:
            alpha_cur = min(alpha_cur * 1.25, 12000.0)

    if pg_best < best_c5:
        best_c5, best_h = pg_best, h

    # Phase 3: Multi-restart coordinate descent
    p3_end = t_start + min(budget_s * 0.85, 880)
    if time.time() < p3_end:
        h = best_h.copy()
        cd_best = best_c5
        dpos = np.array([0.002, 0.005, 0.01, 0.015, 0.025, 0.04, 0.06])
        deltas = np.concatenate([dpos, -dpos])
        stall = 0
        while time.time() < p3_end and stall < 6:
            improved = False
            for i in range(n):
                if time.time() > p3_end:
                    break
                vi = h[i]
                for d in deltas:
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    cands = np.argsort(h)[:10] if need > 0 else np.argsort(-h)[:10]
                    for j in cands:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c5_try = _c5(h, dx)
                        if c5_try < cd_best - 1e-14:
                            cd_best = c5_try
                            improved = True
                            break
                        h[i], h[j] = vi, h[j] - need
                    else:
                        continue
                    break
            if improved:
                stall = 0
            else:
                stall += 1
        if cd_best < best_c5:
            best_c5, best_h = cd_best, h

    # Phase 4: Fine gradient + final L-BFGS polish
    if time.time() < t_start + budget_s - 30:
        h2 = best_h.copy()
        fine_best = best_c5
        s2 = 0.005
        f4_end = t_start + budget_s - 30
        while time.time() < f4_end:
            c5_v = _c5(h2, dx)
            if c5_v < fine_best - 1e-14:
                fine_best = c5_v
                best_h = h2.copy()
            g = _grad_weighted(h2, n, 8000.0)
            se = float(np.sum(h2)) - target
            g += 4.0 * se
            gn = np.linalg.norm(g)
            if gn < 1e-13:
                break
            h2 = _repair(h2 - s2 * g, n)
            if _c5(h2, dx) > c5_v + 0.001:
                s2 *= 0.7
        if fine_best < best_c5:
            best_c5, best_h = fine_best, h2

    if time.time() < t_start + budget_s - 3:
        def obj_fin(x):
            return _lse_obj(_corr(x), 10000, n) + 8000.0 * (np.sum(x) - target) ** 2
        try:
            res = minimize(obj_fin, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-14})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
