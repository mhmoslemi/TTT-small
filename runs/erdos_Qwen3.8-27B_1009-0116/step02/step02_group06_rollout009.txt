import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sweep_1d(corr, gd, d_min, d_max):
    """Find d that minimizes max(corr + d*gd) over [d_min, d_max]"""
    if d_max - d_min < 1e-14:
        return 0.0, float(np.max(corr))
    
    def f(d):
        return float(np.max(corr + d * gd))
    
    res = minimize_scalar(f, bounds=(d_min, d_max), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 200})
    if res.success:
        return res.x, res.fun
    # Fallback: check endpoints and midpoint
    best_d, best_f = 0.0, f(0.0)
    for d in [d_min, d_max, (d_min+d_max)/2]:
        fv = f(d)
        if fv < best_f:
            best_f, best_d = fv, d
    return best_d, best_f

def _sweep_pairs(h, n, dx, corr, max_passes=50, time_limit=None, t0=None):
    """Iterate over all pairs, doing exact 1D optimization"""
    improved_total = False
    for _pass in range(max_passes):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        for i in range(n):
            for j in range(i+1, n):  # Only i < j to avoid double counting
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-12:
                    continue
                gd = _g_vec(h, i, n) - _g_vec(h, j, n)
                d_best, f_best = _sweep_1d(corr, gd, d_min, d_max)
                cur_max = float(np.max(corr))
                if f_best < cur_max - 1e-12:
                    h[i] = hi + d_best
                    h[j] = hj - d_best
                    corr = corr + d_best * gd
                    improved = True
                    improved_total = True
        if not improved:
            break
    return h, corr, improved_total

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Exact 1D coordinate sweeps (many passes)
    h = best_h.copy()
    corr = _corr(h)
    h, corr, _ = _sweep_pairs(h, n, dx, corr, 50, t0, budget_s - 450)
    corr = _corr(h)
    c5 = float(np.max(corr) * dx)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: SLSQP with smooth max (increasing alpha for sharper max)
    for alpha in [80, 150, 300, 600, 1200, 2500, 5000]:
        if time.time() - t0 > budget_s - 300:
            break
        hc = _slsqp(best_h, n, target, bounds, alpha, 800)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 3: p-norm SLSQP (alternative smoothing)
    for p in [40, 80, 160]:
        if time.time() - t0 > budget_s - 220:
            break
        hc = _pnorm_slsqp(best_h, n, target, bounds, p, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 4: More exact sweeps after optimization
    h = best_h.copy()
    corr = _corr(h)
    h, corr, _ = _sweep_pairs(h, n, dx, corr, 40, t0, budget_s - 150)
    corr = _corr(h)
    c5 = float(np.max(corr) * dx)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 5: Random restarts with multi-start
    for sr in range(10):
        if time.time() - t0 > budget_s - 60:
            break
        scale = 0.06 + 0.04 * (sr % 3)
        h_p = best_h + scale * rng.randn(n)
        h_p = _repair(h_p, n)
        hc = _slsqp(h_p, n, target, bounds, 300, 400)
        corr_r = _corr(hc)
        hc, corr_r, _ = _sweep_pairs(hc, n, dx, corr_r, 20, t0, budget_s - 60)
        c5 = float(np.max(_corr(hc)) * dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 6: Final ultra-precise polish
    if time.time() - t0 < budget_s - 20:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep_pairs(h, n, dx, corr, 30, t0, budget_s - 20)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if time.time() - t0 < budget_s - 10:
        hc = _slsqp(best_h, n, target, bounds, 8000, 1200)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
            corr = _corr(best_h)
            best_h, corr, _ = _sweep_pairs(best_h, n, dx, corr, 15, t0, budget_s - 10)
            best_c5 = float(np.max(corr) * dx)

    if time.time() - t0 < budget_s - 3:
        hc = _slsqp(best_h, n, target, bounds, 500, 600)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h = hc
            best_c5 = c5
            corr = _corr(best_h)
            best_h, corr, _ = _sweep_pairs(best_h, n, dx, corr, 10, t0, budget_s - 3)
            best_c5 = float(np.max(corr) * dx)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
