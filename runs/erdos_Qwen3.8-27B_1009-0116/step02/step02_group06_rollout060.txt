```python
                       bounds=bounds, constraints=cons,
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _g_pos(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_once(h, n, dx, corr, nc, nf, fw):
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min >= d_max - 1e-14:
                continue
            gd = _g_pos(h, i, n) - _g_pos(h, j, n)
            ds = np.linspace(d_min, d_max, nc)
            cm = corr[None, :] + ds[:, None] * gd[None, :]
            c5s = np.max(cm, axis=1) * dx
            bi = int(np.argmin(c5s))
            cur = np.max(corr) * dx
            if c5s[bi] < cur - 1e-14:
                db = ds[bi]
                if nf > 2 and fw > 0:
                    lo = max(d_min, db - fw)
                    hi2 = min(d_max, db + fw)
                    ds2 = np.linspace(lo, hi2, nf)
                    cm2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c52 = np.max(cm2, axis=1) * dx
                    db = ds2[int(np.argmin(c52))]
                h[i] += db
                h[j] -= db
                corr = corr + db * gd
                improved = True
    return h, corr, improved

def _slsqp(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _resample(h_old, n_old, n_new):
    xo = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    hn = np.interp(xn, xo, np.concatenate([h_old, [h_old[0]]]))
    return _repair(hn, n_new)

    if time.time() - t0 < budget - 5:
        hc = _slsqp(h, n, target, 3000, 500)
        if _c5(hc, dx) < _c5(h, dx):
            h = hc
    return h

    return (best_h, best_c5, best_n)
```
