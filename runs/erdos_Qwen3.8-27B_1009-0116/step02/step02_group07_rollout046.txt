import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sum_con_fun(x, target):
    return x.sum() - target

def _opt_slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'args': (n / 2.0,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

def _opt_lbfgs(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'args': (n / 2.0,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 30})
        if res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

def _golden_pair(h, i, j, hs, lo, hi, dx, n_iter=22):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    h[i], h[j] = x1, hs - x1
    f1 = _c5(h, dx)
    h[i], h[j] = x2, hs - x2
    f2 = _c5(h, dx)
    for _ in range(n_iter):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, x2, f2 = x2, x1, f1
            x1 = hi - gr * (hi - lo)
            h[i], h[j] = x1, hs - x1
            f1 = _c5(h, dx)
        else:
            lo, x1, f1 = x1, x2, f2
            x2 = lo + gr * (hi - lo)
            h[i], h[j] = x2, hs - x2
            f2 = _c5(h, dx)
    mid = (lo + hi) * 0.5
    h[i], h[j] = mid, hs - mid
    return mid, _c5(h, dx)

def _pair_cd(h, n, dx, t_limit, t_start, max_passes, grid_n):
    hi_c5 = _c5(h, dx)
    for _p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        any_imp = False
        for i in range(n):
            if time.time() - t_start > t_limit:
                break
            for j in range(n - 1, i, -1):
                hs = h[i] + h[j]
                lo = max(0.0, hs - 1.0)
                hi = min(1.0, hs)
                if hi - lo < 1e-10:
                    continue
                oi, oj = h[i], h[j]
                bt, bc = oi, hi_c5
                step = (hi - lo) / grid_n
                for k in range(grid_n):

                    h[i], h[j] = t, hs - t
                    cv = _c5(h, dx)
                    if cv < bc:

                    hi_c5 = bc
                    any_imp = True
                else:
