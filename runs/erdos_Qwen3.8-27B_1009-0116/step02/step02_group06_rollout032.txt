def _pair_line_search(corr, g_diff, d_min, d_max, n_eval=32):
    """Find d in [d_min, d_max] minimizing max(corr + d*g_diff)"""
    if d_max - d_min < 1e-14:
        return 0.0
    # Grid search
    ds = np.linspace(d_min, d_max, n_eval)
    vals = np.max(corr[None, :] + ds[:, None] * g_diff[None, :], axis=1)
    bi = np.argmin(vals)
    # Refine around best
    if bi > 0 and bi < n_eval - 1:
        lo, hi = ds[bi-1], ds[bi+1]
    else:
        lo, hi = d_min, d_max
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(30):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1 = np.max(corr + m1 * g_diff)
        f2 = np.max(corr + m2 * g_diff)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-14:
            break
    return (lo + hi) / 2

            if 0 <= i - p < n:
                val += 1.0 - h[i - p]
            if 0 <= i + p < n:
                val -= h[i + p]
            G[m, i] = val
    return G

def _line_search_convex(corr, g, d_min, d_max):
    if d_max - d_min < 1e-14:
        return 0.0
    n_ev = 48
    ds = np.linspace(d_min, d_max, n_ev)
    vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
    bi = int(np.argmin(vals))
    if bi > 0 and bi < n_ev - 1:
        lo, hi = ds[bi - 1], ds[bi + 1]
    elif bi == 0:
        lo, hi = d_min, ds[1]
    else:
        lo, hi = ds[-2], d_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(40):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1 = float(np.max(corr + m1 * g))
        f2 = float(np.max(corr + m2 * g))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-15:
            break
    return (lo + hi) / 2.0

def _pair_sweep(h, n, dx, n_eval=48, max_passes=100):
    target = n / 2.0
    improved_total = 0
    for _ in range(max_passes):
        corr = _corr(h)
        G = _grad_cols(h, n)
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                d_min = max(-h[i], h[j] - 1.0)
                d_max = min(1.0 - h[i], h[j])
                if d_max - d_min < 1e-14:
                    continue
                g = G[:, i] - G[:, j]
                d = _line_search_convex(corr, g, d_min, d_max)
                if abs(d) < 1e-14:
                    continue
                new_corr = corr + d * g
                new_max = float(np.max(new_corr)) * dx
                cur_max = float(np.max(corr)) * dx
                if new_max < cur_max - 1e-15:
                    h[i] += d
                    h[j] -= d
                    corr = new_corr
                    improved = True
        if improved:
            improved_total += 1
        else:
            break
    h = _repair(h, n)
    return h

def _slsqp_opt(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h.copy(), args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _rand_dir_search(h, n, dx, rng, n_tries):
    target = n / 2.0
    corr = _corr(h)
    G = _grad_cols(h, n)
    for _ in range(n_tries):
        d = rng.randn(n)
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            continue
        d /= dn
        g = G.T @ d  # (2n-1,)
        if np.abs(g).max() < 1e-14:
            continue
        d_min, d_max = -1.0, 1.0
        for k in range(n):
            if d[k] > 0:
                d_min = max(d_min, -h[k] / d[k])
            elif d[k] < 0:
                d_max = min(d_max, (1.0 - h[k]) / (-d[k]))
        if d_max - d_min < 1e-14:
            continue
        t = _line_search_convex(corr, g, d_min, d_max)
        if abs(t) < 1e-12:
            continue
        new_corr = corr + t * g
        if float(np.max(new_corr)) < float(np.max(corr)) - 1e-15:
            h = h + t * d
            corr = new_corr
            h = _repair(h, n)
    return h

def _block_perturb(h, n, dx, rng, n_tries, strength):
    target = n / 2.0
    for _ in range(n_tries):
        blen = rng.randint(3, min(12, n))
        start = rng.randint(0, n - blen)
        h2 = h.copy()
        for k in range(blen):
            idx = start + k
            h2[idx] = h[idx] + strength * rng.randn()
        h2 = _repair(h2, n)
        if _c5(h2, dx) < _c5(h, dx):
            h = h2
    return h

            hc = _rand_dir_search(best_h, n, dx, rng, 50)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
            hc = _block_perturb(best_h, n, dx, rng, 30, 0.05)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
            hc = _pair_sweep(best_h, n, dx, n_eval=48, max_passes=30)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc

    # Phase 5: Fine coordinate sweeps
    if _remaining() > 120:
        best_h = _pair_sweep(best_h, n, dx, n_eval=64, max_passes=80)
        best_c5 = _c5(best_h, dx)

    # Phase 6: Random restarts
    for sr in range(5):
        if _remaining() < 100:
            break
        h_pert = best_h + 0.06 * rng.randn(n)
        h_pert = _repair(h_pert, n)
        hc = _slsqp_opt(h_pert, n, dx, target, bounds, 500, 500)
        hc = _pair_sweep(hc, n, dx, n_eval=48, max_passes=40)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 7: Final polish
    if _remaining() > 30:
        best_h = _pair_sweep(best_h, n, dx, n_eval=64, max_passes=50)
        best_c5 = _c5(best_h, dx)

    if _remaining() > 10:
        hc = _slsqp_opt(best_h, n, dx, target, bounds, 15000, 1200)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    if _remaining() > 5:
        best_h = _pair_sweep(best_h, n, dx, n_eval=64, max_passes=30)
        best_c5 = _c5(best_h, dx)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
