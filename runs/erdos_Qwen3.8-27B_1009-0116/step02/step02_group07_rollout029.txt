            delta = (lo + hi) * 0.5
            fval = corr + delta * dg
            new_max = float(fval.max())
            if new_max < hi_c5 - 1e-13:
                h[i] += delta
                h[j] -= delta
                corr = fval
                hi_c5 = new_max
                improved = True
    return h, hi_c5, improved

def _cd_loop(h, n, dx, t_limit, t_start, n_bisect=55, max_passes=5000):
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        h, c5, imp = _cd_pass_fast(h, n, dx, n_bisect)
        if not imp:
            break
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP with increasing alpha
    for a in [60, 120, 250, 500, 1000, 2000]:
        if elapsed() > budget_s - 400:
            break
        h = _slsqp(h, n, a, 500, 1e-14)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Fast coordinate descent (many passes)
    if elapsed() < budget_s - 300:
        h, c = _cd_loop(best_h, n, dx, budget_s - 320, t0, n_bisect=55, max_passes=3000)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 3: SLSQP polish with higher alpha
    if elapsed() < budget_s - 150:
        h = _slsqp(best_h, n, 3000, 600, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
            h2, c2 = _cd_loop(h, n, dx, min(budget_s - 160, elapsed() + 60), t0, n_bisect=55, max_passes=500)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 4: More CD
    if elapsed() < budget_s - 100:
        h, c = _cd_loop(best_h, n, dx, budget_s - 120, t0, n_bisect=55, max_passes=3000)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 5: Multi-restart with diverse perturbations
    sigmas = [0.06, 0.1, 0.15, 0.04, 0.2, 0.08, 0.12, 0.05]
    for si in range(len(sigmas)):
        if elapsed() > budget_s - 50:
            break
        h_p = best_h + sigmas[si] * rng.randn(n)
        h_p = _repair(h_p, n)
        a = 400 + 300 * si
        h = _slsqp(h_p, n, a, 350, 1e-13)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
            h2, c2 = _cd_loop(h, n, dx, min(budget_s - 55, elapsed() + 50), t0, n_bisect=55, max_passes=400)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 5b: Antipodal-structured restart
    if elapsed() < budget_s - 80:
        h_ant = np.zeros(n)
        for i in range(n):
            partner = (i + n // 2) % n
            if i < partner:
                v = rng.uniform(0.2, 0.8)
                h_ant[i] = v
                h_ant[partner] = 1.0 - v
        h_ant = _repair(h_ant, n)
        h = _slsqp(h_ant, n, 500, 400, 1e-13)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
        h2, c2 = _cd_loop(h, n, dx, min(budget_s - 85, elapsed() + 40), t0, n_bisect=55, max_passes=300)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 6: Final intensive CD
    if elapsed() < budget_s - 20:
        h, c = _cd_loop(best_h, n, dx, budget_s - 25, t0, n_bisect=60, max_passes=5000)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 7: Final SLSQP
    if elapsed() < budget_s - 5:
        h = _slsqp(best_h, n, 5000, 400, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
