from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = abs(k - (n - 1))
        if L == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            for i in range(n - L):
                g[i] += wk * (1.0 - x[i + L])
                g[i + L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters=5000, t0=0, budget=1100):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (2.0 * alpha * n + 1.0)
    for it in range(iters):
        if t0 > 0 and (time.time() - t0) > budget - 10:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + np.dot(gz, xn - z) + 0.5 / step * np.dot(xn - z, xn - z)
            if fn <= quad + 1e-9:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd(h, dx, n, passes, rng):
    dvals = [0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.05,
             -0.001, -0.002, -0.003, -0.005, -0.008, -0.01, -0.015, -0.02, -0.03, -0.05]
    bc = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        imp = False
        for _ in range(n * 12):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bc - 1e-12:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _bin_anneal(h0, dx, n, target, rng, iters=3000):
    # Project to binary
    h = (h0 > 0.5).astype(float)
    # Fix to exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    tgt = n // 2
    if len(ones) > tgt:
        excess = len(ones) - tgt
        idx = np.argsort(h0[ones])[:excess]
        h[ones[idx]] = 0.0
    elif len(ones) < tgt:
        need = tgt - len(ones)
        idx = np.argsort(-h0[zeros])[:need]
        h[zeros[idx]] = 1.0
    bh = h.copy()
    bc = _c5(bh, dx)
    for it in range(iters):
        T = 0.5 * (0.01 / 0.5) ** (it / max(iters - 1, 1))
        om = np.where(bh == 1.0)[0]
        zm = np.where(bh == 0.0)[0]
        if len(om) == 0 or len(zm) == 0:
            break
        i = om[rng.randint(len(om))]
        j = zm[rng.randint(len(zm))]
        bh[i], bh[j] = 0.0, 1.0
        c = _c5(bh, dx)
        if c < bc or rng.rand() < np.exp(-(c - bc) / max(T, 1e-10)):
            bc = c
        else:
            bh[i], bh[j] = 1.0, 0.0
    return bh, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    tgt0 = n0 / 2.0
    h_init = _proj(np.array(initial_h_values, dtype=float), tgt0)
    
    best_h = h_init.copy()
    best_c = _c5(best_h, dx0)
    best_n = n0
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: FISTA on n=70 with increasing alpha
    bh = h_init.copy()
    for a in [50, 100, 200, 400, 800, 1500, 3000]:
        if rem() < 30:
            break
        h, c = _fista(bh, a, tgt0, iters=4000, t0=t0, budget=budget_s)
        if c < best_c:
            best_c = c
            best_h = h
            best_n = n0
        bh = h if c <= _c5(bh, dx0) else bh
    
    # Phase 2: Binary annealing
    if rem() > 60:
        hb, cb = _bin_anneal(bh, dx0, n0, tgt0, rng, iters=3000)
        # Refine binary with FISTA
        if rem() > 30:
            hr, cr = _fista(hb, 500, tgt0, iters=3000, t0=t0, budget=budget_s)
            if cr < best_c:
                best_c = cr
                best_h = hr
                best_n = n0
            bh = hr if cr < _c5(bh, dx0) else bh
    
    # Phase 3: CD refinement
    if rem() > 20:
        bh2, bc2 = _cd(bh, dx0, n0, 200, rng)
        if bc2 < best_c:
            best_c = bc2
            best_h = bh2
            best_n = n0
        bh = bh2 if bc2 < _c5(bh, dx0) else bh
    
    # Phase 4: Try n=140
    if rem() > 120:
        n1 = 2 * n0
        dx1 = 2.0 / n1
        tgt1 = n1 / 2.0
        x70 = np.linspace(0, 2, n0, endpoint=False)
        x140 = np.linspace(0, 2, n1, endpoint=False)
        h140 = np.interp(x140, x70, best_h)
        h140 = _proj(h140, tgt1)
        for a in [200, 500, 1000, 2000]:
            if rem() < 30:
                break
            h, c = _fista(h140, a, tgt1, iters=3000, t0=t0, budget=budget_s)
            h140 = h
        if rem() > 20:
            h140, c140 = _cd(h140, dx1, n1, 100, rng)
        if c140 < best_c:
            best_c = c140
            best_h = h140
            best_n = n1
    
    # Phase 5: Random restarts on best_n
    for s in [0.05, 0.1, 0.15, 0.2]:
        if rem() < 40:
            break
        hp = _proj(bh + s * rng.randn(n0), tgt0)
        h, c = _fista(hp, 500, tgt0, iters=2000, t0=t0, budget=budget_s)
        if rem() > 10:
            h, c = _cd(h, dx0, n0, 50, rng)
        if c < best_c:
            best_c = c
            best_h = h
            best_n = n0
        bh = h if c < _c5(bh, dx0) else bh
    
    # Phase 6: SLSQP polish
    if rem() > 25:
        nn = best_n
        dxn = 2.0 / nn
        tgt = nn / 2.0
        bnds = [(0.0, 1.0)] * nn
        try:
            r = minimize(_slsqp_obj, best_h, args=(5000,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 500, 'ftol': 1e-14})
            hf = _proj(np.clip(r.x, 0, 1), tgt)
            cf = _c5(hf, dxn)
            if cf < best_c:
                best_c = cf
                best_h = hf
        except:
            pass
    
    # Phase 7: Final CD
    if rem() > 10:
        nn = best_n
        dxn = 2.0 / nn
        h, c = _cd(best_h, dxn, nn, 200, rng)
        if c < best_c:
            best_c = c
            best_h = h
    
    best_h = _proj(best_h, best_n / 2.0)
    best_c = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c, best_n)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = abs(k - (n - 1))
        if L == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            for i in range(n - L):
                g[i] += wk * (1.0 - x[i + L])
                g[i + L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters=5000, t0=0.0, budget=1100.0):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (2.0 * alpha * n + 1.0)
    stale = 0
    for it in range(iters):
        if t0 > 0 and (time.time() - t0) > budget - 15.0:
            break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 8:
            fn = _sobj(xn, alpha)
            qd = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= qd + 1e-9:
                break
            step *= 0.5
            if step < 1e-15:
                break
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
            if step < 1e-15:
                break
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
            stale = 0
        else:
            stale += 1
        if stale > 800:
            break
    return bx, bc

def _cd(h, dx, n, passes, rng, t0=0.0, budget=1100.0):
    dvals = [0.001, 0.002, 0.003, 0.005, 0.007, 0.01, 0.015, 0.02, 0.03, 0.04, 0.06,
                -0.001, -0.002, -0.003, -0.005, -0.007, -0.01, -0.015, -0.02, -0.03, -0.04, -0.06]
    bc = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        if (time.time() - t0) > budget - 15.0:
            break
        imp = False
        for _ in range(n * 12):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _bin_greedy(h0, dx, n, target, rng, t0=0.0, budget=1100.0):
    h = (h0 > 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    tgt = n // 2
    if len(ones) != tgt:
        if len(ones) > tgt:
            idx = np.argsort(h0[ones])[:len(ones) - tgt]
            h[ones[idx]] = 0.0
        else:
            zeros = np.where(h == 0.0)[0]
            idx = np.argsort(-h0[zeros])[:tgt - len(ones)]
            h[zeros[idx]] = 1.0
    bc = _c5(h, dx)
    for _ in range(20):
        if (time.time() - t0) > budget - 15.0:
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for m in order:
            if (time.time() - t0) > budget - 15.0:
                break
            old = h[m]
            h[m] = 1.0 - old
            c = _c5(h, dx)
            if c < bc - 1e-13:
                bc = c
                improved = True
            else:
                h[m] = old
        if not improved:
            break
    return h, bc

def _bin_anneal(h0, dx, n, target, rng, iters=2000, t0=0.0, budget=1100.0):
    hb, _ = _bin_greedy(h0, dx, n, target, rng, t0, budget)
    bc = _c5(hb, dx)
    hb = hb.copy()
    for it in range(iters):
        if (time.time() - t0) > budget - 15.0:
            break
        T = 0.3 * (0.005 / 0.3) ** (it / max(iters - 1, 1))
        om = np.where(hb == 1.0)[0]
        zm = np.where(hb == 0.0)[0]
        if len(om) == 0 or len(zm) == 0:
            break
        i = om[rng.randint(len(om))]
        j = zm[rng.randint(len(zm))]
        hb[i], hb[j] = 0.0, 1.0
        c = _c5(hb, dx)
        if c < bc or (c - bc) / max(T, 1e-12) < -np.log(max(rng.rand(), 1e-300)):
            bc = c
        else:
            hb[i], hb[j] = 1.0, 0.0
    return hb, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

def _rem(t0, budget_s):
