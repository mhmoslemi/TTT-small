        h += 0.1 * np.cos(4 * t)
    return _project(h, target)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _project(np.array(initial_h_values, dtype=float).copy(), target)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def deadline(frac=1.0):
        return t0 + frac * budget_s

    # Phase 1: Active set gradient descent (main workhorse)
    if time.time() < deadline(0.35):
        h, c5 = _active_descent(h0, target, n, dx, 300, deadline(0.35))
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 2: Nelder-Mead polish on free variables
    if time.time() < deadline(0.55):
        h_nm = best_h.copy()
        free_mask = (h_nm > 5e-4) & (h_nm < 1.0 - 5e-4)
        free_idx = np.where(free_mask)[0]
        if len(free_idx) >= 3:
            fix_idx = free_idx[-1]
            vary_idx = free_idx[:-1]
            x0 = h_nm[vary_idx].copy()
            obj_wrap = lambda x: _nm_obj(x, h_nm, vary_idx, fix_idx, target, dx)
            try:
                res = minimize(obj_wrap, x0, method='Nelder-Mead',
                               options={'maxiter': 8000, 'xatol': 1e-11,
                                        'fatol': 1e-14, 'adaptive': True})
                h_nm[vary_idx] = res.x
                h_nm[fix_idx] = target - (h_nm.sum() - h_nm[fix_idx])
                h_nm = _project(h_nm, target)
                c5 = _c5(h_nm, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_nm.copy()
            except Exception:
                pass

    # Phase 3: Pair local search
    if time.time() < deadline(0.80):
        h, c5 = _pair_search(best_h, target, n, dx, deadline(0.80), rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 4: Another active descent round
    if time.time() < deadline(0.92):
        h, c5 = _active_descent(best_h, target, n, dx, 200, deadline(0.92))
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 5: Multi-start from structured inits
    if time.time() < deadline(0.97):
        for mode in range(3):
            if time.time() > deadline(0.97):
                break
            h_init = _fourier_init(n, target, rng, mode)
            h_init += 0.05 * rng.randn(n)
            h_init = _project(h_init, target)
            h2, c5_2 = _active_descent(h_init, target, n, dx, 80, deadline(0.97))
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2

    # Phase 6: Final fine pair polish
    if time.time() < t0 + budget_s - 3:
        h, c5 = _pair_search(best_h, target, n, dx, t0 + budget_s - 3, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    best_h = _project(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
