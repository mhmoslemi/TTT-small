    s = h.sum()

                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, v):
    return np.correlate(h, v, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h, 1.0 - h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x, 1.0 - x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = _repair(res.x, n)
        if _c5(hx, dx) < _c5(h, dx):
            return hx
    except Exception:
        pass
    return h.copy()

def _cd2(h, n, dx, t_limit, t_start, n_grid, max_passes=100000, rng=None):
    best_c5 = _c5(h, dx)
    v = 1.0 - h
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        if rng is not None:
            rng.shuffle(pairs)
        improved = 0
        for (i, j) in pairs:
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-12:
                continue
            oi, oj = h[i], h[j]
            vi, vj = v[i], v[j]
            bt, bc = oi, best_c5
            ts = np.linspace(lo + 1e-10, hi - 1e-10, n_grid)
            for t in ts:
                h[i] = t; v[i] = 1.0 - t
                h[j] = hsum - t; v[j] = 1.0 - (hsum - t)
                c = float(np.max(_corr(h, v)) * dx)
                if c < bc:
                    bc = c; bt = t
            if bt != oi and bc < best_c5 - 1e-13:
                span = hi - lo
                d = span / max(n_grid - 1, 1) * 0.5
                lo2 = max(lo, bt - d)
                hi2 = min(hi, bt + d)
                for t in np.linspace(lo2, hi2, 5):
                    h[i] = t; v[i] = 1.0 - t
                    h[j] = hsum - t; v[j] = 1.0 - (hsum - t)
                    c = float(np.max(_corr(h, v)) * dx)
                    if c < bc:
                        bc = c; bt = t
                h[i] = bt; v[i] = 1.0 - bt
                h[j] = hsum - bt; v[j] = 1.0 - (hsum - bt)
                best_c5 = bc
                improved += 1
            else:
                h[i] = oi; v[i] = vi
                h[j] = oj; v[j] = vj
        if improved == 0:
            break
    return h, _c5(h, dx)

def _cd3(h, n, dx, t_limit, t_start, n_grid, max_passes=50000, rng=None):
    """3-block CD: vary h[i],h[i+1],h[i+2] keeping sum fixed."""
    best_c5 = _c5(h, dx)
    v = 1.0 - h
    for p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        start = rng.randint(n) if rng is not None else 0
        for s in range(n):
            i = (start + s) % n
            j = (i + 1) % n
            k = (i + 2) % n
            if i == j or j == k or i == k:
                continue
            h3 = h[i] + h[j] + h[k]
            if h3 < 1e-10 or h3 > 3.0 - 1e-10:
                continue
            oi, oj, ok = h[i], h[j], h[k]
            vi, vj, vk = v[i], v[j], v[k]
            best_vals = (oi, oj, ok)
            best_c = best_c5
            # Sample the triangle: a+b+c=h3, 0<=a,b,c<=1
            # Parameterize: a = lo_a + f1*(hi_a-lo_a), b = lo_b + f2*(hi_b-lo_b)
            # Simplify: use a 2D grid over (a,b) with c=h3-a-b
            na, nb = n_grid, n_grid
            for fa in range(na):
                a_lo = max(0.0, h3 - 2.0)
                a_hi = min(1.0, h3)
                if a_hi - a_lo < 1e-10:
                    continue
                a = a_lo + (a_hi - a_lo) * fa / max(na - 1, 1)
                for fb in range(nb):
                    b_lo = max(0.0, h3 - 1.0 - a)
                    b_hi = min(1.0, h3 - a)
                    if b_hi - b_lo < 1e-10:
                        continue
                    b = b_lo + (b_hi - b_lo) * fb / max(nb - 1, 1)
                    c = h3 - a - b
                    if c < -1e-10 or c > 1.0 + 1e-10:
                        continue
                    h[i] = a; h[j] = b; h[k] = c
                    v[i] = 1.0 - a; v[j] = 1.0 - b; v[k] = 1.0 - c
                    cv = float(np.max(_corr(h, v)) * dx)
                    if cv < best_c:
                        best_c = cv
                        best_vals = (a, b, c)
            if best_vals != (oi, oj, ok) and best_c < best_c5 - 1e-13:
                h[i], h[j], h[k] = best_vals
                v[i] = 1.0 - h[i]; v[j] = 1.0 - h[j]; v[k] = 1.0 - h[k]
                best_c5 = best_c
                improved = True
            else:
                h[i], h[j], h[k] = oi, oj, ok
                v[i], v[j], v[k] = vi, vj, vk
        if not improved:
            break
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
