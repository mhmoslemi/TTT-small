def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - x[idx]); x[idx] += a; diff -= a
            else:
                a = min(-diff, x[idx]); x[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):

            g[d:] += wk * (1.0 - x[:n-d])
            g[:n-d] -= wk * x[d:]
