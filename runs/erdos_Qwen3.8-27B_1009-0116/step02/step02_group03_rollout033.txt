def _repair(h, n):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            # Need to reduce: reduce the largest values
            order = np.argsort(h)[::-1]
            diff = target - s
            for idx in order:
                if diff >= 0:
                    break
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
        else:
            order = np.argsort(1.0 - h)
            diff = target - s
            for idx in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
                h[idx] += d
            else:
                d = min(-diff, h[idx])
                h[idx] -= d
            diff = target - h.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w = w / w.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            if lag < n:
                idx = np.arange(lag, n)
                grad[idx] += wj * (1.0 - h[idx - lag])
            if lag < n - 1:
                idx = np.arange(0, n - lag)
                grad[idx] -= wj * h[idx + lag]
        else:
            ell = -lag
            if ell < n:
                idx = np.arange(0, n - ell)
                grad[idx] += wj * (1.0 - h[idx + ell])
                idx = np.arange(ell, n)
                grad[idx] -= wj * h[idx - ell]
    return grad

def _subgrad(h, n, alpha, steps):
    h = h.copy()
    for it in range(steps):
        g = _grad_smooth(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.02 / (1.0 + 0.05 * it)
        h = _repair(h - step * g / gn, n)
    return h

def _pair_local(h, n, n_sweeps, rng):
    h = h.copy()
    best = _c5(h, n)
    deltas = np.array([0.003, 0.007, 0.015, 0.03, 0.06,
                       -0.003, -0.007, -0.015, -0.03, -0.06])
    for _sweep in range(n_sweeps):
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            for j in rng.permutation(n):
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = _c5(h, n)
                    if c5 < best:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
        if not improved and _sweep > 5:
            break
    return h, best

def _init_binary(n, rng, pt):
    h = np.zeros(n)
    if pt == 0:
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    elif pt == 1:
        blk = max(2, n // 14)
        for i in range(0, n, 2 * blk):
            h[i:i + blk] = 1.0
    elif pt == 2:
        ph = rng.uniform(0, 2 * np.pi)
        t = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * np.arange(n) / n + ph)
        h = (t > 0.5).astype(float)
        diff = n // 2 - int(h.sum())
        if diff > 0:
            idx = np.where(h == 0)[0]
            if len(idx) >= diff:
                h[idx[:diff]] = 1.0
        elif diff < 0:
            idx = np.where(h == 1)[0]
            if len(idx) >= -diff:
                h[idx[:diff]] = 0.0
    elif pt == 3:
        h[:n // 2] = 1.0
        for _ in range(max(1, n // 8)):
            a, b = rng.randint(0, n // 2), rng.randint(n // 2, n)
            h[a], h[b] = h[b], h[a]
    else:
        h = (np.arange(n) % 2 == 0).astype(float)
        while h.sum() > n // 2:
            h[rng.randint(n)] = 0
        while h.sum() < n // 2:
            idx = np.where(h == 0)[0]
            if len(idx) == 0:
                break
            h[rng.choice(idx)] = 1
    return h

def _lbfgs(h, n, alpha, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _slsqp(h, n, alpha, maxit):
    bounds = [(0.0, 1.0)] * n
    def _obj(x):
        return _smooth_obj(x, alpha)
    def _con(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con}],
                       options={'maxiter': maxit, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h
