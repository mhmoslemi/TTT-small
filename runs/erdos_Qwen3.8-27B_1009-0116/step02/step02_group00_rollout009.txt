import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            step = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-14:
                break
    return h


def _eq_con(x, target):
    return x.sum() - target


def _obj_c5(x, dx):
    return float(np.max(np.correlate(x, 1.0 - x, mode='full'))) * dx


def _obj_lse(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + s.sum() / alpha) * dx


def _subgrad(h, n, lag_idx):
    offset = lag_idx - n + 1
    grad = np.zeros(n)
    for i in range(n):
        j1 = i + offset
        if 0 <= j1 < n:
            grad[i] += (1.0 - h[j1])
        j2 = i - offset
        if 0 <= j2 < n:
            grad[i] -= h[j2]
    return grad


def _fine_local(h, dx, n, t_limit, rng):
    local_best = _c5(h, dx)
    steps = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.03]
    no_improve = 0
    while time.time() < t_limit and no_improve < 6:
        improved = False
        for i in range(n):
            if time.time() > t_limit:
                break
            hi = h[i]
            for j in range(n):
                if j == i:
                    continue
                hj = h[j]
                for s in steps:
                    for sign in (1, -1):
                        ni = hi + sign * s
                        nj = hj - sign * s
                        if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < local_best - 1e-15:
                            local_best = c
                            improved = True
                            hi = ni
                            break
                        else:
                            h[i], h[j] = hi, hj
                    if improved:
                        break
                if improved:
                    break
            if time.time() > t_limit:
                break
        no_improve = 0 if improved else no_improve + 1
    return h, local_best


def _multi_point_escape(h, dx, n, rng, t_limit):
    best_h = h.copy()
    best_c = _c5(h, dx)
    it = 0
    while time.time() < t_limit and it < 50000:
        it += 1
        h_try = h.copy()
        n_pts = rng.randint(2, 5)
        idxs = rng.choice(n, n_pts, replace=False)
        deltas = rng.randn(n_pts) * 0.02
        deltas -= deltas.mean()
        for idx, d in zip(idxs, deltas):
            h_try[idx] += d
        h_try = _repair(h_try, n)
        c = _c5(h_try, dx)
        if c < best_c:
            best_c, best_h = c, h_try.copy()
            h = h_try.copy()
    return best_h, best_c
