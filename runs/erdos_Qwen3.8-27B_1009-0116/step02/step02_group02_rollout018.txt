    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())

            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

        else:
            idx = np.argsort(x)

                diff -= a
            else:
                a = min(-diff, x[k])

                diff += a

                dhi = min(1.0-h[i], h[j])       # h[i]+d <= 1, h[j]-d <= 1
                dlo = max(dlo, -h[i])
                dhi = min(dhi, h[j])
                dlo = max(dlo, 0.0) if False else dlo  # keep negative d possible
                if dhi <= dlo:
                    continue
                # Try a few log-spaced values
                for d in [0.01, 0.03, 0.1, 0.3, 0.5, 0.8] + [-0.01, -0.03, -0.1, -0.3, -0.5, -0.8]:
                    if d < dlo or d > dhi:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c = c
                        best_d = d
                    h[i] -= d
                    h[j] += d
                if best_d != 0:
                    h[i] += best_d
                    h[j] -= best_d
                    if best_c < bc - 1e-14:
                        bc = best_c
                        imp = True
        if not imp:
            break
    return h, bc

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters=2000):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy(); z = h0.copy(); t = 1.0
    bx = x.copy(); bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        fn = _sobj(xn, alpha)
        quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5/step*float(np.dot(xn-z, xn-z))
        tries = 0
        while fn > quad + 1e-10 and tries < 6:
            step *= 0.5; y = z - step * gz; xn = _proj(y, target)
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn-z)) + 0.5/step*float(np.dot(xn-z,xn-z))
            tries += 1
        tn = 0.5*(1.0+np.sqrt(1.0+4.0*t*t))
        z = _proj(xn+(t-1.0)/tn*(xn-x), target)
        x = xn; t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c; bx = x.copy()
    return bx, bc

def _cd(h, dx, n, n_passes, rng, fine=True):
    bc = _c5(h, dx)
    h = h.copy()
    if fine:
        dvals = [0.005,0.01,0.02,0.04,0.07,0.1,0.15,0.25,0.4,0.7,
                 -0.005,-0.01,-0.02,-0.04,-0.07,-0.1,-0.15,-0.25,-0.4,-0.7]
    else:
        dvals = [0.02,0.05,0.1,0.2,0.5,-0.02,-0.05,-0.1,-0.2,-0.5]
    for p in range(n_passes):
        imp = False
        order = rng.permutation(n)
        for i in order:
            for j in order:
                if i == j:
                    continue
                hi0, hj0 = h[i], h[j]
                for d in dvals:
                    ni, nj = hi0+d, hj0-d
                    if ni < -1e-9 or ni > 1.0+1e-9 or nj < -1e-9 or nj > 1.0+1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < bc - 1e-13:
                        bc = c; imp = True; break
                    h[i], h[j] = hi0, hj0
                else:
                    h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _sa(h, dx, n, n_trials, rng, T0=0.003):
    h = h.copy()
    cur = bc = _c5(h, dx)
    for t in range(n_trials):
        Tm = T0 * max(0.01, 1.0 - t/float(n_trials))
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = (rng.rand()-0.5)*0.5
        ni, nj = h[i]+d, h[j]-d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c = _c5(h, dx)
        if c < cur or rng.rand() < np.exp(-(c-cur)/Tm):
            cur = c
            if c < bc: bc = c
        else:
            h[i], h[j] = ni-d, nj+d
    return h, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

def _rem(t0, budget_s):
    return budget_s - (time.time() - t0)

    # Phase 1: FISTA from parent
