            break
        diff = target - s
        idxs = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in idxs:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _pen_obj(x, alpha, pen):
    v = _smooth_obj(x, alpha)
    if pen > 0.0:
        v += pen * (np.sum(x) - _g_target) ** 2
    return v

def _con_eq(x):
    return np.sum(x) - _g_target

def _slsqp(h, n, alpha, maxiter):
    global _g_target
    _g_target = n * 0.5
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _con_eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, n, alpha, pen, maxiter):
    global _g_target
    _g_target = n * 0.5
    try:
        res = minimize(_pen_obj, h, args=(alpha, pen), method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_descent(h, n, dx, t_end, max_passes=400):
    best = _c5(h, dx)
    base_deltas = np.array([0.003, 0.007, 0.015, 0.03, 0.06, 0.12])
    stagnant = 0
    for p in range(max_passes):
        if time.time() > t_end:
            return h, best
        improved = False
        scale = 1.0 / (1.0 + 0.04 * p)
        if scale < 0.1:
            scale = 0.1
        deltas = np.concatenate([base_deltas * scale, -base_deltas * scale])
        nd = len(deltas)
        for i in range(n):
            if time.time() > t_end:
                return h, best
            hi_orig = h[i]
            for j in range(i + 1, n):
                hi = h[i]
                hj = h[j]
                for di in range(nd):
                    d = deltas[di]
                    ni = hi + d
                    nj = hj - d
                    if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c = _c5(h, dx)
                    if c < best:
                        best = c
                        improved = True
                        hi = ni
                    else:
                        h[i] = hi
                        h[j] = hj
        if not improved:
            stagnant += 1
            if stagnant >= 12:
                break
        else:
            stagnant = 0
    return h, best

def _make_diverse_starts(n, seed, num):
    rng = np.random.RandomState(seed)
    starts = []
    starts.append(_repair(np.full(n, 0.5) + 0.05 * rng.randn(n), n))
    for freq in [2, 3, 4, 6, 8]:
        ph = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        starts.append(_repair(h, n))
        if len(starts) >= num:
            break
    while len(starts) < num:
        h = (rng.rand(n) > 0.5).astype(float) * 0.7 + 0.15 * rng.rand(n)
        starts.append(_repair(h, n))
    return starts[:num]

def _perturb_reshuffle(h, n, rng, sigma):
    hp = h + sigma * rng.randn(n)
    hp = np.clip(hp, 0.0, 1.0)
    return _repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_target
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 70

    n_list = [50, 60, 70, 80, 100]
    has_init = False
    init_n = 0
    try:
        init_n = len(initial_h_values)
        has_init = True
        if init_n not in n_list:
            n_list.append(init_n)
        n_list.remove(init_n)
        n_list = [init_n] + n_list
    except NameError:
        pass

    time_per_n = (budget_s - 100) / float(len(n_list))

    for ni, n in enumerate(n_list):
        n_t_end = t0 + (ni + 1.0) * time_per_n
        if n_t_end > t_end - 50:
            n_t_end = t_end - 50
        if time.time() > n_t_end:
            break
        dx = 2.0 / n

        starts = []
        if has_init and n == init_n:
            starts.append(_repair(np.array(initial_h_values, dtype=float).copy(), n))
        starts.extend(_make_diverse_starts(n, seed + ni * 1009, 7))

        candidates = []
        for h0 in starts:
            if time.time() > n_t_end - 15:
                break
            h = h0.copy()
            h = _lbfgs(h, n, 80.0, 200.0, 200)
            h = _lbfgs(h, n, 250.0, 2000.0, 200)
            h = _slsqp(h, n, 500, 300)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
            candidates.append(h)

        if not candidates:
            continue

        n_cand = len(candidates)
        scores = np.zeros(n_cand)
        for ci in range(n_cand):
            scores[ci] = _c5(candidates[ci], dx)
        order = np.argsort(scores)

        top_k = min(3, n_cand)
        for rank in range(top_k):
            if time.time() > n_t_end - 20:
                break
            idx = int(order[rank])
            h_cd = candidates[idx].copy()
            h_cd, c5_cd = _coord_descent(h_cd, n, dx, n_t_end, 350)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

        # SLSQP polish with high alpha after CD
        if time.time() < n_t_end - 15:
            h_pol = _slsqp(h_cd, n, 3000, 400)
            c5_pol = _c5(h_pol, dx)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n
            h_cd2, c5_cd2 = _coord_descent(h_pol, n, dx, n_t_end, 200)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n

        # Perturbation escape: try a few random perturbations of best
        if time.time() < n_t_end - 10:
            for _ in range(3):
                esc = _perturb_reshuffle(h_cd, n, rng, 0.03)
                esc = _slsqp(esc, n, 500, 200)
                esc2, c5_esc = _coord_descent(esc, n, dx, n_t_end, 80)
                if c5_esc < best_c5:
                    best_c5 = c5_esc
                    best_h = esc2.copy()
                    best_n = n

    # Final global polish
    if best_h is not None and time.time() < t_end - 10:
        dx_b = 2.0 / best_n
        h_fin, c5_fin = _coord_descent(best_h, best_n, dx_b, t_end - 3, 300)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if time.time() < t_end - 5:
            h_fin2 = _slsqp(best_h, best_n, 5000, 300)
            c5_fin2 = _c5(h_fin2, dx_b)
            if c5_fin2 < best_c5:
                best_c5 = c5_fin2
                best_h = h_fin2.copy()
                h_fin3, c5_fin3 = _coord_descent(h_fin2, best_n, dx_b, t_end - 1, 100)
                if c5_fin3 < best_c5:
                    best_c5 = c5_fin3
                    best_h = h_fin3.copy()

    if best_h is None:
        n = 70
        best_h = _repair(np.full(n, 0.5), n)
        best_c5 = _c5(best_h, 2.0 / n)
        best_n = n

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
