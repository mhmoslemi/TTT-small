def _full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_full_corr(h))) * dx

def _compute_all_alpha(h, n):
    n_lags = 2*n - 1
    alpha = np.zeros((n, n_lags))
    for p in range(n):
        k_start = max(-(n-1), p-(n-1))
        k_end = min(n-1, p)
        if k_start <= k_end:
            ks = np.arange(k_start, k_end+1)
            alpha[p, ks+(n-1)] -= (1.0 - h[p-ks])
        k2_start = max(-(n-1), -p)
        k2_end = min(n-1, n-1-p)
        if k2_start <= k2_end:
            ks2 = np.arange(k2_start, k2_end+1)
            alpha[p, ks2+(n-1)] += h[p+ks2]
    return alpha

def _make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h >= 0.5).astype(float)
    n_target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])  # flip smallest ones first
        for i in order:
            if n_ones == n_target: break
            h_bin[ones_idx[i]] = 0; n_ones -= 1
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])  # flip largest zeros first
        for i in order:
            if n_ones == n_target: break
            h_bin[zeros_idx[i]] = 1; n_ones += 1
    return h_bin

def _paley_sequence(v, skip=0):
    """Generate Paley difference set of size v (v prime, v≡3 mod 4)."""
    # Quadratic residues mod v
    qr = set()
    for x in range(1, v):
        qr.add((x*x) % v)
    # Create sequence
    seq = np.zeros(v)
    for i in range(v):
        if i in qr:
            seq[i] = 1.0
    return seq

def _binary_local_search(h_bin, n, dx, t0, budget_s, reserve=100, max_sweeps=500):
    corr = _full_corr(h_bin)
    best_c5 = float(np.max(corr)) * dx
    best_h = h_bin.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget_s - reserve:
            break
        
        alpha = _compute_all_alpha(h_bin, n)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        a1 = alpha[ones]
        a0 = alpha[zeros]
        diffs = a1[:, None, :] - a0[None, :, :]
        new_maxs = np.max(corr[None, None, :] + diffs, axis=2)
        
        cur_max = float(np.max(corr))
        min_nm = float(np.min(new_maxs))
        
        if min_nm >= cur_max - 1e-14:
            break
        
        idx = np.unravel_index(np.argmin(new_maxs), new_maxs.shape)
        p = ones[idx[0]]
        q = zeros[idx[1]]
        
        h_bin[p] = 0.0
        h_bin[q] = 1.0
        corr = corr + alpha[p] - alpha[q]
        
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_bin.copy()
    
    return best_h, best_c5

def _binary_anneal(h_bin, n, dx, t0, budget_s, reserve=100, rng=None, max_iters=5000):
    """Simulated annealing on binary solutions."""
    corr = _full_corr(h_bin)
    best_c5 = float(np.max(corr)) * dx
    best_h = h_bin.copy()
    cur_c5 = best_c5
    
    T_start = 0.1
    T_end = 0.001
    
    for it in range(max_iters):
        if time.time() - t0 > budget_s - reserve:
            break
        
        T = T_start * (T_end/T_start) ** (it / max_iters)
        
        alpha = _compute_all_alpha(h_bin, n)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        # Random swap
        pi = rng.randint(len(ones))
        qi = rng.randint(len(zeros))
        p = ones[pi]
        q = zeros[qi]
        
        new_corr = corr + alpha[p] - alpha[q]
        new_max = float(np.max(new_corr)) * dx
        
        delta = new_max - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h_bin[p] = 0.0
            h_bin[q] = 1.0
            corr = new_corr
            cur_c5 = new_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
    
    return best_h, best_c5

def _continuous_polish(h, n, dx, t0, budget_s, reserve=50):
    """SLSQP polish in continuous space."""
    from scipy.optimize import minimize
    
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        c = _full_corr(x)
        # Smooth max
        alpha = 500
        m = np.max(c)
        return float(m + np.log(np.sum(np.exp(alpha*(c-m))))/alpha) * dx
    
    def constraint(x):
        return np.sum(x) - target
    
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint}],
                   options={'maxiter': 300, 'ftol': 1e-12})
    
    if res.success or hasattr(res, 'x'):
        h_new = np.clip(res.x, 0, 1)
        # Repair integral
        target = n/2
        h_new = h_new * (target / h_new.sum())
        h_new = np.clip(h_new, 0, 1)
        # Fix integral exactly
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                order = np.argsort(1 - h_new)
            else:
                order = np.argsort(h_new)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1-h_new[idx])
                    h_new[idx] += add; diff -= add
                else:
                    sub = min(-diff, h_new[idx])
                    h_new[idx] -= sub; diff += sub
                if abs(diff) < 1e-12: break
        return h_new
    return h
