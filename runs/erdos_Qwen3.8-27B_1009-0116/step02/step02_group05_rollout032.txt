def _subgrad_step(h, n, lr, top_k=3):
    c = _corr(h)
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        j = k - (n - 1)
        if j >= 0:
            # ∂c/∂h[m] = 1-2h[n-1+j-m] for m in [j, n-1]
            m_start, m_end = j, n - 1
            idx = n - 1 + j - np.arange(m_start, m_end + 1)
            grad[m_start:m_end+1] += (1.0 - 2.0 * h[idx])
        else:
            j_abs = -j
            # ∂c/∂h[m] = 1-2h[n-1-j_abs-m] for m in [0, n-1-j_abs]
            m_start, m_end = 0, n - 1 - j_abs
            idx = n - 1 - j_abs - np.arange(m_start, m_end + 1)
            grad[m_start:m_end+1] += (1.0 - 2.0 * h[idx])
    grad /= len(top_idx)
    # Subgradient step: we want to DECREASE max corr, so go opposite to gradient
    h_new = h - lr * grad
    return _repair(h_new, n)
```

This is O(n * top_k) per step! Very fast. I can do millions of steps.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_fn(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _cons_fn(x, target):
    return np.sum(x) - target

def _run_slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    try:
        res = minimize(_obj_fn, h, args=(alpha, dx), method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _cons_fn, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _subgrad_dir(h, n, top_k=3):
    """Compute normalized subgradient direction for the max lag"""
    c = _corr(h)
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        j = k - (n - 1)
        if j >= 0:
            m_range = np.arange(j, n)
            idx = n - 1 + j - m_range
            grad[m_range] += (1.0 - 2.0 * h[idx])
        else:
            ja = -j
            m_range = np.arange(0, n - ja)
            idx = n - 1 - ja - m_range
            grad[m_range] += (1.0 - 2.0 * h[idx])
    grad /= len(top_idx)
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return np.zeros(n)
    return grad / norm

def _subgrad_descent(h, n, dx, lr0, n_steps, top_k, deadline):
    """Subgradient descent with backtracking line search"""
    best = h.copy()
    best_c5 = _c5(h, dx)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        d = _subgrad_dir(h, n, top_k)
        if np.all(d == 0):
            break
        # Backtracking line search
        lr = lr0
        improved = False
        for bt in range(10):
            h_try = h - lr * d
            h_try = _repair(h_try, n)
            c5_try = _c5(h_try, dx)
            if c5_try < best_c5 * (1 + 1e-12):
                h = h_try
                improved = True
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best = h.copy()
                break
            lr *= 0.5
        if not improved:
            # Try a smaller lr
            lr = lr0 * 0.01
            h_try = h - lr * d
            h_try = _repair(h_try, n)
            c5_try = _c5(h_try, dx)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
                best = h.copy()
    return best, best_c5

def _sa_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    cur = h.copy()
    cur_c5 = best_c5
    T0 = 0.008
    T_end = 1e-7
    total_steps = 15000
    idx = 0
    steps_pos = np.array([0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18])
    
    while time.time() < deadline:
        frac = min(idx / total_steps, 1.0)
        T = T0 * (T_end / T0) ** frac
        idx += 1
        ni = int(rng.randint(n))
        nj = int(rng.randint(n))
        if ni == nj:
            continue
        step = rng.choice(steps_pos) * rng.choice([-1.0, 1.0])
        old_i, old_j = cur[ni], cur[nj]
        new_i, new_j = old_i + step, old_j - step
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        cur[ni], cur[nj] = new_i, new_j
        nc = _c5(cur, dx)
        delta = nc - cur_c5
        if delta < 0 or (T > 1e-9 and rng.rand() < np.exp(-delta / T)):
            cur_c5 = nc
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[ni], cur[nj] = old_i, old_j
    return best, best_c5

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
