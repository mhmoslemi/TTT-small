    """Compute Erdős minimum overlap constant"""
    dx = 2.0 / n
    target_sum = n / 2
    # Enforce clipping to bounds
    x_clipped = np.clip(x, 0, 1)
    h_array = build_h_function(x_clipped)
    # Force integral to 35
    if abs(sum(h_array) - target_sum) > 1e-4:
        diff = sum(h_array) - target_sum
        # Clip the fractional part least significant elements
        if diff > 0:
            h_array[np.argsort(x)[1:][:-int(diff/(dx/5))]] = 0
        else:
            h_array[np.argsort(-x)[1:][:abs(int(diff/(dx/5))]] = 1
    corr = np.correlate(h_array, 1.0 - h_array, mode='full')
    return np.max(corr) * dx
