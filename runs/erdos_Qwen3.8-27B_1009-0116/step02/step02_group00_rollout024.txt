import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _soft_max(corr, alpha):
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return m + float(np.sum(s)) / alpha


def _obj_soft(x, n, alpha):
    corr = _corr(x)
    return _soft_max(corr, alpha) * (2.0 / n)


def _obj_topk(x, n, k):
    corr = _corr(x)
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum()) * (2.0 / n)


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _topk_grad(h, n, top_k):
    corr = _corr(h)
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)
    for m in top_idx:
        d = m - n + 1
        if d >= 0:
            g[:n - d] += (1.0 - h[d:])
            g[d:] -= h[:n - d]
        else:
            dd = -d
            g[dd:] += (1.0 - h[:n - dd])
            g[:n - dd] -= h[dd:]
    g /= float(len(top_idx))
    g -= np.mean(g)
    return g


def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def _targeted_sa(h, n, dx, rng, t_end, init_T=0.003, end_T=0.00005):
    h = h.copy()
    c5 = _c5(h, dx)
    it_max = 2000000
    no_improve = 0
    patience = 80000
    for it in range(it_max):
        if time.time() > t_end:
            break
        frac = min(it / it_max, 1.0)
        T = init_T * (end_T / init_T) ** frac
        lr = 0.03 * (1.0 - 0.6 * frac)
        noise_std = 0.04 * (1.0 - 0.5 * frac)
        g = _topk_grad(h, n, 3)
        noise = rng.randn(n) * noise_std
        h_try = _repair(h - lr * (g + noise), n)
        c5_try = _c5(h_try, dx)
        if c5_try < c5 or rng.rand() < np.exp(-(c5_try - c5) / max(T, 1e-12)):
            h = h_try
            if c5_try < c5 - 1e-15:
                c5 = c5_try
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > patience:
            h = h + rng.randn(n) * 0.08
            h = _repair(h, n)
            c5 = _c5(h, dx)
            no_improve = 0
    return h, c5


def _coord_descent(h, n, dx, t_end):
    h = h.copy()
    c5 = _c5(h, dx)
    deltas = np.array([0.001, 0.002, 0.005, 0.01, 0.02,
                       -0.001, -0.002, -0.005, -0.01, -0.02])
    no_improve = 0
    while time.time() < t_end and no_improve < 4:
        improved = False
        for i in range(n):
            if time.time() > t_end:

            for d in deltas:

                if ni < 0.0 or ni > 1.0:

                need = -d
                if need > 0:
                    candidates = np.argsort(h)[:12]

                for j in candidates:

                    nj = h[j] + need

                    h[i] = ni

                    else:

            hc = _repair(res.x, n)

        except Exception:
            pass
        best_h, best_c5 = _coord_descent(best_h, best_n, 2.0 / best_n,
                                         time.time() + max(5, budget_s - (time.time() - t_start) - 5))

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
