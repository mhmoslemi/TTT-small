            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _targeted_local(h, n, dx, t_end, rng):
    h = h.copy()
    best = _c5(h, dx)
    while time.time() < t_end:
        corr = _corr(h)
        top_k = 4
        top_idx = np.argsort(corr)[-top_k:]
        improved = False
        for li in top_idx:
            shift = li - (n - 1)
            if shift == 0:
                continue
            s = abs(shift)
            if s >= n:
                continue
            if shift > 0:
                pa = np.arange(s, n)
                pb = np.arange(0, n - s)
            else:
                pa = np.arange(0, n - s)
                pb = np.arange(s, n)
            contrib = h[pa] * (1.0 - h[pb])
            for mp in np.argsort(contrib)[::-1][:3]:
                a, b = pa[mp], pb[mp]
                for frac in [0.2, 0.4, 0.6, 0.9]:
                    amt = frac * min(h[a], 1.0 - h[b])
                    if amt < 1e-7:
                        continue
                    h[a] -= amt
                    h[b] += amt
                    nc = _c5(h, dx)
                    if nc < best - 1e-12:
                        best = nc
                        improved = True
                        break
                    h[a] += amt
                    h[b] -= amt
                if improved:
                    break
            if improved:
                break
        if not improved:
            n_p = rng.randint(1, 4)
            idxs = rng.choice(n, n_p, replace=False)
            for idx in idxs:
                d = (rng.rand() - 0.5) * 0.04
                nv = h[idx] + d
                if 0 <= nv <= 1:
                    h[idx] = nv
            h = _repair(h, n)
            if _c5(h, dx) > best + 0.003:
                break
    h = _repair(h, n)
    return h, best


def _fine_descend(h, n, dx, t_end, rng):
    h = h.copy()
    best = _c5(h, dx)
    steps = np.array([0.001, 0.003, 0.008, 0.02, 0.05])
    all_d = np.concatenate([steps, -steps])
    no_imp = 0
    while time.time() < t_end and no_imp < 4:
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.time() > t_end:
                break
            vi = h[i]
            for d in all_d:
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                found = False
                for j in order[:12]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    h[i] = ni
                    h[j] = nj
                    nc = _c5(h, dx)
                    if nc < best - 1e-14:
                        best = nc
                        improved = True
                        found = True
                        break
                    h[i] = vi
                    h[j] += -need
                if found:
                    break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
    return _repair(h, n), best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Progressive SLSQP
    p1_end = t0 + budget_s * 0.10
    for alpha in [200, 600, 1500, 4000, 8000, 15000]:
        if time.time() > p1_end:
            break
        try:
            res = minimize(_obj_lse, best_h, args=(alpha, n),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Targeted local search
    p2_end = t0 + budget_s * 0.35
    h, c5 = _targeted_local(best_h, n, dx, p2_end, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 3: SA with multi-point moves
    p3_end = t0 + budget_s * 0.65
    hsa = best_h.copy()
    sab = best_c5
    sabh = best_h.copy()
    T0, Tend = 0.0025, 0.00001
    max_it = 8000000
    cnt = 0
    while time.time() < p3_end and cnt < max_it:
        cnt += 1
        fr = cnt / max_it
        T = T0 * (Tend / T0) ** fr
        np_ = rng.randint(2, 5)
        idxs = rng.choice(n, np_, replace=False)
        deltas = (rng.rand(np_) - 0.5) * 0.06 * (1.0 - 0.6 * fr)
        deltas -= deltas.mean()
        valid = True
        applied = []
        for idx, d in zip(idxs, deltas):
            nv = hsa[idx] + d
            if nv < 0 or nv > 1:
                valid = False
                break
            hsa[idx] = nv
            applied.append(idx)
        if not valid:
            for idx in applied:
                hsa[idx] -= deltas[list(idxs).index(idx)]
            continue
        nc = _c5(hsa, dx)
        if nc <= sab + T:
            if nc < sab - 1e-14:
                sab = nc
                sabh = hsa.copy()
        else:
            for k in range(len(applied)):
                hsa[idxs[k]] -= deltas[k]
    if sab < best_c5:
        best_c5, best_h = sab, sabh

    # Phase 4: Fine coordinate descent
    p4_end = t0 + budget_s * 0.85
    h, c5 = _fine_descend(best_h, n, dx, p4_end, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 5: Final SLSQP + targeted polish
    p5_end = t0 + budget_s - 5
    if time.time() < p5_end:
        for alpha in [8000, 20000, 50000]:
            if time.time() > p5_end - 20:
                break
            try:
                res = minimize(_obj_lse, best_h, args=(alpha, n),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-16})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass
    if time.time() < t0 + budget_s - 2:
        h, c5 = _targeted_local(best_h, n, dx, t0 + budget_s - 2, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
