def coord_descent_step(h, n, dx, i):
    # Find optimal h[i] given all other coordinates
    # For each lag k, C(k) = A_k * h[i] + B_k
    # where A_k = 1 - h[(i+k) % ... ] - h[(i-k) % ...] (but careful with boundaries)
    
    # Actually, for linear correlation:
    # corr = np.correlate(h, 1-h, 'full')
    # corr has length 2n-1, indexed 0 to 2n-2
    # corr[k] = sum_j h[j] * (1-h[j+k-(n-1)]) ... 
    # Actually np.correlate(a, v, 'full')[k] = sum_j a[j+k] * v[j] for valid indices
    # So corr[k] = sum_j h[j+k-n+1... ] 
    
    # Let me just compute A and B directly
    h_copy = h.copy()
    h_copy[i] = 0  # Set to 0 to get the B terms
    corr_0 = np.correlate(h_copy, 1 - h_copy, 'full')
    
    h_copy[i] = 1  # Set to 1 to get A+B terms
    corr_1 = np.correlate(h_copy, 1 - h_copy, 'full')
    
    A = corr_1 - corr_0  # slope for each lag
    B = corr_0  # intercept for each lag
    
    # Now minimize max_k (A_k * x + B_k) for x in [0,1]
    # Find breakpoints
    # At breakpoint between k and l: x = (B_l - B_k) / (A_k - A_l)
    
    best_x = 0.0
    best_val = np.max(A * 0.0 + B)  # value at x=0
    # Check x=1
    val_1 = np.max(A * 1.0 + B)
    if val_1 < best_val:
        best_val = val_1
        best_x = 1.0
    
    # Find all breakpoints in (0,1)
    for k in range(len(A)):
        for l in range(k+1, len(A)):
            if A[k] != A[l]:
                x_bp = (B[l] - B[k]) / (A[k] - A[l])
                if 0.0 < x_bp < 1.0:
                    val = np.max(A * x_bp + B)
                    if val < best_val:
                        best_val = val
                        best_x = x_bp
    
    return best_x

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)

                diff -= add

                diff += sub

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _coord_opt(h, i):
    h0 = h.copy()
    h0[i] = 0.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    h1 = h.copy()
    h1[i] = 1.0
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    A = c1 - c0
    B = c0
    xs = np.linspace(0.0, 1.0, 180)
    vals = np.max(A[None, :] * xs[:, None] + B[None, :], axis=1)
    bi = int(np.argmin(vals))
    lo = xs[max(0, bi - 1)]
    hi = xs[min(len(xs) - 1, bi + 1)]
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(A * m1 + B) < np.max(A * m2 + B):
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

def _coord_descent(h, n, max_sweeps, deadline):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = _c5(h, dx)
    for s in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            nv = _coord_opt(h, i)
            if abs(nv - h[i]) > 1e-9:
                h[i] = nv
                improved = True
        h = _repair(h, n)
        cc = _c5(h, dx)
        if cc < best_c5:
            best_c5 = cc
        elif not improved:
            break
    return h, best_c5

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_obj(x, alpha, dx):
    return _smooth_max(np.correlate(x, 1.0 - x, mode='full'), alpha) * dx

def _sum_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_con, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _perturb_restart(h, n, sigma, rng):
    return _repair(h + sigma * rng.randn(n), n)

def _two_point_search(h, n, dx, rng, iters, deadline):
    best = h.copy()
    bc = _c5(best, dx)
    for _ in range(iters):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = rng.randn() * 0.04
        ni, nj = best[i] + d, best[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < bc - 1e-13:
                bc = nc
            else:
                best[i], best[j] = oi, oj
    return best, bc

def _snap_binary(h, n, tol):
    h = h.copy()
    changed = False
    for i in range(n):
        if h[i] < tol:
            h[i] = 0.0
            changed = True
        elif h[i] > 1.0 - tol:
            h[i] = 1.0
            changed = True
    if changed:
        h = _repair(h, n)
    return h
