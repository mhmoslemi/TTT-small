def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smax_val(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_obj(x, alpha):
    return _smax_val(_corr(x), alpha)

def _smax_grad(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    e = np.exp(s)

    n = len(x)

            if 0 <= j - tau < n:

    return grad
