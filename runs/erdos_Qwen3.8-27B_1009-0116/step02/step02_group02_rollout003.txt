from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = _corr(x)
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue

        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += wk * (1.0 - x[idx + lag])
            g[idx + lag] -= wk * x[idx]
        else:
            L = -lag
            idx = np.arange(L, n)
            g[idx] += wk * (1.0 - x[idx - L])
            g[idx - L] -= wk * x[idx]
    return g

def _lbfgs(h0, alpha, target, maxiter=3000):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _sobj(x, alpha)
    def grad(x):
        return _sgrad(x, alpha)
    try:
        r = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                     bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return _proj(r.x, target)
    except Exception:
        return h0.copy()

import numpy as np
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = _corr(x)
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            g[idx] += wk * (1.0 - x[idx + lag])
            g[idx + lag] -= wk * x[idx]
        else:
            L = -lag
            idx = np.arange(L, n)
            g[idx] += wk * (1.0 - x[idx - L])
            g[idx - L] -= wk * x[idx]
    return g
