**Key new idea: Use the structure of the correlation**

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _all_c(h, dx):
    return np.asarray(_corr(h), dtype=float) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _subgrad_c(h, n, lag):
    g = np.zeros(n)
    for i in range(n):
        ip, im = i + lag, i - lag
        if 0 <= ip < n:
            g[i] += 1.0 - h[ip]
        if 0 <= im < n:
            g[i] -= h[im]
    return g

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _sum_con(x):
    return x.sum() - 35.0

def _slq(x, n, alpha, maxiter, bounds, cons):
    try:
        r = minimize(_smooth_obj, x, args=(alpha,), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return x.copy()

def _cd_pass(h, n, dx, n_grid):
    hi = _c5(h, dx)
    imp = False
    for i in range(n):
        for j in range(i + 1, n):
            hs = h[i] + h[j]
            lo, hib = max(0.0, hs - 1.0), min(1.0, hs)
            if hib - lo < 1e-10:
                continue
            oi, oj = h[i], h[j]
            bt, bc = oi, hi
            for t in np.linspace(lo + 1e-9, hib - 1e-9, n_grid):
                h[i], h[j] = t, hs - t
                c = _c5(h, dx)
                if c < bc:
                    bc, bt = c, t
            if bc < hi - 1e-13:
                h[i], h[j] = bt, hs - bt
                hi = bc
                imp = True
            else:
                h[i], h[j] = oi, oj
    return h, hi, imp

def _cd_loop(h, n, dx, t0, tl, ng, mp=5000):
    for _ in range(mp):
        if time.time() - t0 > tl:
            break
        h, c5, im = _cd_pass(h, n, dx, ng)
        if not im:
            break
    return h, _c5(h, dx)

def _sgd_phase(h, n, dx, t0, tl, rng):
    base = h.copy()
    bc5 = _c5(h, dx)
    h = h.copy()
    al, bet, st = 0.12, 0.82, 0
    mom = np.zeros(n)
    for _ in range(30000):
        if time.time() - t0 > tl:
            break
        ca = _all_c(h, dx)
        ks = int(np.argmax(ca))
        lag = ks - (n - 1)
        g = _subgrad_c(h, n, lag)
        idx2 = np.argsort(ca)[::-1]
        if len(idx2) > 1 and ca[idx2[1]] > ca[idx2[0]] - 0.008:
            lag2 = int(idx2[1]) - (n - 1)
            g = 0.5 * (g + _subgrad_c(h, n, lag2))
        mom = bet * mom + g
        h_new = _repair(h - al * mom, n)
        cn = _c5(h_new, dx)
        if cn < bc5 - 1e-14:
            base, bc5, h, st = h_new.copy(), cn, h_new, 0
        else:
            st += 1
            al *= 0.985
            if st > 600:
                h = _repair(base + 0.05 * rng.randn(n), n)
                al, st, mom = 0.12, 0, np.zeros(n)
    return base, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con}]

    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh, bc5 = h.copy(), _c5(h, dx)

    # Phase 1: SLSQP warm-up
    for a in [80, 200, 500]:
        if elapsed() > budget_s - 550:
            break
        h = _slq(h, n, a, 600, bounds, cons)
        c = _c5(h, dx)
        if c < bc5:
            bh, bc5 = h.copy(), c

    # Phase 2: Subgradient descent
    if elapsed() < budget_s - 450:
        h, c = _sgd_phase(bh, n, dx, t0, budget_s - 450, rng)
        if c < bc5:
            bh, bc5 = h.copy(), c

    # Phase 3: CD coarse
    if elapsed() < budget_s - 350:
        h, c = _cd_loop(bh, n, dx, t0, budget_s - 350, 13, 400)
        if c < bc5:
            bh, bc5 = h.copy(), c

    # Phase 4: SLSQP mid
    if elapsed() < budget_s - 300:
        h = _slq(bh, n, 800, 800, bounds, cons)
        c = _c5(h, dx)
        if c < bc5:
            bh, bc5 = h.copy(), c

    # Phase 5: SGD + CD cycles
    for cyc in range(4):
        if elapsed() > budget_s - 180:
            break
        h, c = _sgd_phase(bh, n, dx, t0, min(budget_s - 200, elapsed() + 70), rng)
        if c < bc5:
            bh, bc5 = h.copy(), c
        h, c = _cd_loop(bh, n, dx, t0, min(budget_s - 180, elapsed() + 55), 17, 300)
        if c < bc5:
            bh, bc5 = h.copy(), c

    # Phase 6: SLSQP high
    if elapsed() < budget_s - 150:
        h = _slq(bh, n, 1500, 700, bounds, cons)
        c = _c5(h, dx)
        if c < bc5:
            bh, bc5 = h.copy(), c

    # Phase 7: Multi-restart
    sigmas = [0.08, 0.15, 0.05, 0.2, 0.12]
    for si, sg in enumerate(sigmas):
        if elapsed() > budget_s - 70:
            break
        hp = _repair(bh + sg * rng.randn(n), n)
        hp = _slq(hp, n, 400 + 200 * si, 400, bounds, cons)
        hp, c = _sgd_phase(hp, n, dx, t0, min(budget_s - 90, elapsed() + 40), rng)
        hp, c = _cd_loop(hp, n, dx, t0, min(budget_s - 70, elapsed() + 35), 15, 200)
        c = _c5(hp, dx)
        if c < bc5:
            bh, bc5 = hp.copy(), c

    # Phase 8: Fine CD

        h, c = _cd_loop(bh, n, dx, t0, budget_s - 25, 25, 600)
        if c < bc5:
            bh, bc5 = h.copy(), c

        if c < bc5:
            bh, bc5 = h.copy(), c

    bh = _repair(bh, n)
    bc5 = _c5(bh, dx)
    return (bh, bc5, n)
