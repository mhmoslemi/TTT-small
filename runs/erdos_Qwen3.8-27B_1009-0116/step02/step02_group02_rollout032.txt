    # Compute all lags
    corr = np.correlate(h, 1-h, mode='full') * dx
    # Find the max (active lag)
    m = corr.max()
    # Subgradient: gradient of the active term
    # corr[k] = sum_i h[i] * (1-h[i+lag]) * dx for valid i
    # d/dh[j] corr[k] = dx * (sum of terms involving h[j])
    lag = int(np.argmax(corr)) - (n-1)
    # For positive lag: corr = sum_{i=0}^{n-lag-1} h[i]*(1-h[i+lag])*dx
    # d/dh[i] for i < n-lag: (1-h[i+lag])*dx

def _cutting_plane(h0, dx, n, max_iters=1000):
    target = n/2.0
    h = h0.copy().astype(float)
    best_c = _c5(h, dx)
    best_h = h.copy()
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0-h, mode='full')
        # Active lags: those with corr close to max
        max_c = corr.max()
        active = np.where(corr >= max_c - 0.01)[0]  # 1% tolerance
        
        # Build LP:
        # min t
        # s.t. sum_{i} h[i]*(1-h[i+k]) <= t for k in active (linearized at h)
        #      sum h[i] = target
        #      0 <= h[i] <= 1
        #      t >= 0
        
        # Linearization: c_k(h) ≈ c_k(h0) + grad_k(h0)·(h - h0)
        # = c_k(h0) - grad_k(h0)·h0 + grad_k(h0)·h
        # So constraint: grad_k·h <= t - c_k(h0) + grad_k·h0
        # i.e., grad_k·h - t <= c_k(h0) - grad_k·h0
        
        # c_k(h0) = corr[k]
        # grad_k(h0) = _grad_lag(h0, n, lag_k)
        # c_k(h0) - grad_k(h0)·h0 = corr[k] - dot(grad_k, h0)
        
        n_vars = n + 1  # h[0..n-1], t
        A_ub = []
        b_ub = []
        
        for k in active:
            lag = k - (n-1)
            gk = _grad_lag(h, n, lag)  # gradient w.r.t. h only
            # Constraint: gk·h - t <= corr[k] - gk·h
            # Vector: [gk, -1]
            row = np.zeros(n_vars)
            row[:n] = gk
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(corr[k] - np.dot(gk, h))
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Equality: sum(h) = target, t free (well, t >= 0)
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds: 0 <= h[i] <= 1, t >= 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        # Objective: minimize t
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            h_new = _proj(h_new, target)
            c_new = _c5(h_new, dx)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        else:
            break
    
    return best_h, best_c
