def _optimal_d(corr, gd, d_min, d_max):
    """Find d in [d_min, d_max] that minimizes max(corr + d*gd) using ternary search"""
    lo, hi = d_min, d_max
    for _ in range(50):  # ~50 iterations for machine precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(corr + m1 * gd)
        f2 = np.max(corr + m2 * gd)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d = (lo + hi) / 2
    return d, np.max(corr + d * gd)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _optimal_d(corr, gd, d_min, d_max):
    lo, hi = d_min, d_max
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(corr + m1 * gd))
        f2 = float(np.max(corr + m2 * gd))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d = (lo + hi) * 0.5
    return d

def _pair_sweep(h, n, dx, refine=True):
    corr = _corr(h).copy()
    improved = False
    for i in range(n):
        hi = h[i]
        if hi <= 1e-14 or hi >= 1.0 - 1e-14:
            continue
        for j in range(n):
            if i == j:
                continue
            hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            d = _optimal_d(corr, gd, d_min, d_max)
            new_max = float(np.max(corr + d * gd))
            cur_max = float(np.max(corr))
            if new_max < cur_max - 1e-13:
                h[i] = hi + d
                h[j] = hj - d
                corr = corr + d * gd
                hi = h[i]
                improved = True
    return h, improved

def _subgrad(h, n, alpha, n_iter=25):
    for it in range(n_iter):
        c = _corr(h)
        m = float(np.max(c))
        w = np.exp(np.clip(alpha * (c - m), -500, 500))
        w /= w.sum()
        g = np.zeros(n)
        for ki in range(2 * n - 1):
            if w[ki] < 1e-10:
                continue
            t = ki - (n - 1)
            wk = w[ki]
            if t >= 0:
                v1 = np.arange(0, n - t)
                g[v1] += wk * (1.0 - h[v1 + t])
                v2 = np.arange(t, n)
                g[v2] -= wk * h[v2 - t]
            else:
                v1 = np.arange(-t, n)
                g[v1] += wk * (1.0 - h[v1 + t])
                v2 = np.arange(0, -t)
                g[v2] -= wk * h[v2 - t]
        step = 0.04 / (1.0 + 0.02 * it)
        h = _repair(h - step * g, n)
    return h

def _slsqp(h, n, target, bounds, alpha, maxiter=400):
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def eqc(x):
        return float(np.sum(x) - target)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eqc}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
