    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-15:
        return np.full(n, 0.5)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, dx, alpha):
    c = _corr(x) * dx
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _lag_grad(h, L, n):
    g = np.zeros(n)
    if L == 0:
        return 1.0 - 2.0 * h
    if L > 0:
        g[:n - L] -= h[L:]
        g[L:] += (1.0 - h[:n - L])
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _chase_max(h, dx, n, max_iters, time_limit, t0, rng):
    c5_cur = _c5(h, dx)
    step = 0.008
    momentum = np.zeros(n)
    mom_coeff = 0.3
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        c = _corr(h)
        worst_k = int(np.argmax(c))
        worst_L = worst_k - (n - 1)
        g = _lag_grad(h, worst_L, n)
        g_proj = g - np.mean(g)
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * g_proj
        direction = momentum
        accepted = False
        for s_try in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h_try = h - s_try * direction
            h_try = _repair(h_try, n)
            c5_try = _c5(h_try, dx)
            if c5_try < c5_cur - 1e-15:
                h = h_try
                c5_cur = c5_try
                accepted = True
                step = min(step * 1.05, 0.05)
                break
        if not accepted:
            step *= 0.7
            if step < 1e-9:
                break
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
    return h, c5_cur

def _local_search(h, dx, n, time_limit, t0):
    c5_cur = _c5(h, dx)
    deltas = np.array([0.0005, 0.001, 0.002, 0.005, 0.01, 0.02])
    for _pass in range(100):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            hi = h[i]
            for j in range(i + 1, n):
                if time.time() - t0 > time_limit:
                    break
                hj = h[j]
                for d in deltas:
                    for sgn in (1, -1):
                        ni = hi + sgn * d
                        nj = hj - sgn * d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-14:
                            c5_cur = c
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
            h[i] = hi
        if not improved:
            break
    return h, c5_cur

def _anneal(h, dx, n, rng, time_limit, t0):
    c5_cur = _c5(h, dx)
    T = 0.0004
    min_T = 1e-9
    steps_arr = np.array([0.001, 0.002, 0.005, 0.01, 0.03])

            c5_cur = c5_new
        else:
            h[i], h[j] = hi, hj
        T = max(T * 0.9995, min_T)
    return h, c5_cur
