    # Phase 7: Fine coordinate descent + random tweaks
    dl7 = dl_end - 3
    bh, bc5 = _cd(bh, n, dl7)

    # Phase 8: Ultra-fine random local search
    hc = bh.copy()
    c5c = bc5
    tiny = [0.0003, 0.0001, 0.00005, -0.0003, -0.0001, -0.00005]
    while time.time() < dl_end - 2:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        oi, oj = hc[i], hc[j]
        ni, nj = oi + d, oj - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[i], hc[j] = ni, nj
            c = _c5(hc)
            if c < c5c:
                c5c = c
                if c < bc5:
                    bc5, bh = c, hc.copy()
            else:
                hc[i], hc[j] = oi, oj

    # Final: try a couple of small perturbations + refine
    if time.time() < dl_end - 1:
        for _ in range(3):
            if time.time() > dl_end - 1:
                break
            hp = bh + 0.005 * rng.randn(n)
            hp = _fix(hp, n, tgt)
            hp, c5r = _cd(hp, n, min(time.time() + 30, dl_end - 0.5))
            if c5r < bc5:
                bc5, bh = c5r, hp

    bh = _fix(bh, n, tgt)
    bc5 = _c5(bh)
    return (bh, bc5, n)
