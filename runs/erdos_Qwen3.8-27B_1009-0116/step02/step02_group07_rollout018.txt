import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _soft_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lbfgs(h, n, alpha, maxiter=250):
    def obj(x):
        return _soft_max(_corr(x), alpha)
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    return _repair(res.x, n)

def _focused_cd(h, n, dx, t_limit, t_start, top_k=6, n_refine=5):

            break

        thresh = m * (1.0 - 0.08)
        active = np.where(c >= thresh)[0]
        if len(active) > top_k:
            active = active[np.argsort(c[active])[::-1][:top_k]]

                    continue
                orig_i = h[i]
                best_t, best_c = orig_i, hi_c5
                for t in np.linspace(lo + 1e-10, hi - 1e-10, n_refine):
                    h[i] = t; h[j] = hsum - t
                    cv = _c5(h, dx)

                if best_c < hi_c5 - 1e-13:
                    h[i] = best_t; h[j] = hsum - best_t
                    hi_c5 = best_c; imp = True
                else:
                    h[i] = orig_i
        if not imp:
            break
    return h, hi_c5

    hi_c5 = _c5(h, dx)

        if shuffled:

        imp = False

            lo = max(0.0, hsum - 1.0)

            h[s:min(s + bw, n)] = 1.0
        return _repair(h, n)
