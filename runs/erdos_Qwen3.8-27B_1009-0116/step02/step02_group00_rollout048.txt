def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _obj_lse(x, alpha, dx, n):
    corr = _corr(x)
    m = float(np.max(corr))
    return (m + float(np.log(np.sum(np.exp(alpha * (corr - m))))) / alpha) * dx


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = float(h.sum())

                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:

    return h
