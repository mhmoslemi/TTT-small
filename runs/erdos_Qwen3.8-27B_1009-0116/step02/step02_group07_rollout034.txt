import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sensitivity(h, n):
    """Compute D[m, i] = dc[m]/dh[i] (linear approximation)"""
    m = np.arange(2*n-1)[:, None]
    i = np.arange(n)[None, :]
    idx1 = i - m + (n-1)  # j - m + n - 1 when j=i
    idx2 = i + m - (n-1)  # k + m - n + 1 when k=i
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    hclip = h[np.clip(idx1, 0, n-1)]
    hclip2 = h[np.clip(idx2, 0, n-1)]
    D = np.where(mask1, 1.0 - hclip, 0.0) - np.where(mask2, hclip2, 0.0)
    return D  # shape (2n-1, n)

def _solve_1d_minimax(base, d, h_i, lo, hi):
    """Find t in [lo,hi] minimizing max_m (base[m] + (t-h_i)*d[m])"""
    # Ternary search (convex function)
    a, b = lo, hi
    for _ in range(40):
        m1 = a + (b-a)/3.0
        m2 = b - (b-a)/3.0
        f1 = np.max(base + (m1-h_i)*d)
        f2 = np.max(base + (m2-h_i)*d)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a+b)/2.0

def _cd_fast(h, n, D, corr, n_passes, t_limit, t_start):
    """Fast coordinate descent using precomputed sensitivity"""
    dx = 2.0/n
    best_c5 = np.max(corr) * dx
    for _pass in range(n_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                d = D[:, i] - D[:, j]
                t_new = _solve_1d_minimax(corr, d, h[i], lo, hi)
                if abs(t_new - h[i]) < 1e-12:
                    continue
                # Check improvement
                delta = t_new - h[i]
                new_corr_max = np.max(corr + delta * d)
                if new_corr_max < np.max(corr) - 1e-14:
                    h[i] = t_new
                    h[j] = s - t_new
                    corr = corr + delta * d
                    improved = True
        # Recompute D at end of pass (h changed)
        D = _sensitivity(h, n)
        # Verify correlation
        corr = np.correlate(h, 1.0-h, mode='full')
        c5_val = np.max(corr) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
    return h, best_c5, improved

    # Phase 3: SLSQP softmax polish
    if el() < budget_s - 250:
        for al in [200, 500, 1000]:
            if el() > budget_s - 250:
                break
            h = _slsqp(best_h, n, _obj_soft, (al,), 400, 1e-15)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    # Phase 4: More CD
    h, c = _cd(best_h, n, budget_s - 200, t0, mp=50000)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 5: Multi-restart with perturbations
    if el() < budget_s - 100:
        sigmas = [0.05, 0.1, 0.15, 0.08, 0.12, 0.2, 0.06, 0.18]
        for si in range(len(sigmas)):
            if el() > budget_s - 80:
                break
            hp = _perturb(best_h, n, sigmas[si], rng)
            h = _slsqp(hp, n, _obj_topk, (6,), 300, 1e-14)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
                h2, c2 = _cd(h, n, min(budget_s - 60, el() + 100), t0, mp=10000)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    # Phase 6: Sine perturbation restarts
    if el() < budget_s - 80:
        for k in [2, 3, 5, 1, 4]:
            if el() > budget_s - 60:
                break
            hp = _sine_h(n, k, 0.3, rng.uniform(0, 6.28), rng)
            hp = 0.5 * best_h + 0.5 * hp
            hp = _repair(hp, n)
            h = _slsqp(hp, n, _obj_topk, (8,), 250, 1e-14)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
                h2, c2 = _cd(h, n, min(budget_s - 50, el() + 80), t0, mp=5000)
                if c2 < best_c5:
                    best_c5, best_h = c2, h2.copy()

    # Phase 7: Final CD polish
    if el() < budget_s - 30:
        h, c = _cd(best_h, n, budget_s - 20, t0, mp=50000)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 8: Final SLSQP
    if el() < budget_s - 5:
        h = _slsqp(best_h, n, _obj_soft, (3000,), 300, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
