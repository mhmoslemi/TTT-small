def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_corr(h, n):
    ells = (np.arange(2 * n - 1) - (n - 1)).astype(float)
    js = np.arange(n)
    jm = js[:, None].astype(float) - ells[None, :]
    jp = js[:, None].astype(float) + ells[None, :]
    mask_m = (jm >= 0) & (jm < n)
    mask_p = (jp >= 0) & (jp < n)
    ci = np.clip(jm.astype(int), 0, n - 1)
    cj = np.clip(jp.astype(int), 0, n - 1)
    vals_m = np.where(mask_m, 1.0 - h[ci[0]], 0.0)
    vals_p = np.where(mask_p, h[cj[0]], 0.0)
    return vals_m.T - vals_p.T, mask_m.T, mask_p.T

def _solve_lp(h_star, n, dx, delta):
    corr_vals = _corr(h_star)
    grad = np.zeros((2 * n - 1, n))
    ells = np.arange(2 * n - 1) - (n - 1)
    js = np.arange(n)
    for k in range(2 * n - 1):
        ell = ells[k]
        jm = js - ell
        jp = js + ell
        vm = ((jm >= 0) & (jm < n)).astype(float)
        vp = ((jp >= 0) & (jp < n)).astype(float)
        jim = np.clip(jm, 0, n - 1)
        jpj = np.clip(jp, 0, n - 1)
        grad[k] = vm * (1.0 - h_star[jim]) - vp * h_star[jpj]

    m = 2 * n - 1
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    A_ub[:m, :n] = grad * dx
    A_ub[:m, n] = -1.0
    b_ub[:m] = grad @ (h_star * dx) - corr_vals * dx
    A_ub[m:, :n] = np.diag(np.concatenate([np.ones(n), -np.ones(n)]))
    b_ub[m:] = np.concatenate([h_star + delta, h_star - delta])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return h_star.copy(), False
    return res.x[:n], True

def _cd_pass(h, n, dx, n_grid=25):
    hi_c5 = _c5(h, dx)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            orig_i, orig_j = h[i], h[j]
            best_t, best_c = orig_i, hi_c5
            for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                h[i] = t
                h[j] = hsum - t
                c = _c5(h, dx)
                if c < best_c:
                    best_c, best_t = c, t
            if best_t != orig_i:
                span = (hi - lo) / n_grid
                lo2 = max(lo, best_t - span)
                hi2 = min(hi, best_t + span)
                for t in np.linspace(lo2, hi2, 13):
                    h[i] = t
                    h[j] = hsum - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c, best_t = c, t
                if best_c < hi_c5 - 1e-13:
                    h[i], h[j] = best_t, hsum - best_t
                    hi_c5 = best_c
                    improved = True
                else:
                    h[i], h[j] = orig_i, orig_j
            else:
                h[i], h[j] = orig_i, orig_j
    return h, hi_c5, improved

def _slqp_smooth(h, n, alpha, maxiter=500):
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

        return time.time() - t0

    # Phase 1: SLP with adaptive trust region

        h_new, ok = _solve_lp(h, n, dx, delta)

            delta *= 0.7
            if delta < 0.005:

            continue
        h_new = _repair(h_new, n)
        c_new = _c5(h_new, dx)
        if c_new < best_c5 - 1e-15:

    # Phase 2: Coordinate descent

    delta = 0.05
