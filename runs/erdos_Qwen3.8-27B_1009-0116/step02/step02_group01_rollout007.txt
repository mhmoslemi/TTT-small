    return np.max(_corr(h)) * dx

def _proj(h, target):
    """Project h onto {0≤h≤1, sum(h)=target}"""
    h = np.array(h, dtype=float)
    # Use bisection on the offset
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        clipped = np.clip(h + mid, 0, 1)
        s = clipped.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)/2, 0, 1)

    """Compute gradient of the worst-lag correlation component."""
    c = _corr(h)
    k = np.argmax(c)
    lag = k - (n - 1)
    lo = max(0, -lag)
    hi = min(n-1, n-1-lag)
    g = np.zeros(n)

            g[j] += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            g[j] -= h[j - lag]
    return g, k, c[k]

    m = np.max(c)

            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    c = _corr(h)

    d = k - (n - 1)
    g = np.zeros(n)

    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    for m in range(lo2, hi2 + 1):

    return g, k, c
