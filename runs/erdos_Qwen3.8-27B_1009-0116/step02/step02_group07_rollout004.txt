        h = _opt(starts[si], n, a, 600, 1e-14)
        c = _c5(h, dx)
        if c < bc5:
            bc5, bh = c, h.copy()
        if time.time() - t0 < budget_s - 70 and per > 20:
            h2, c2 = _cd_loop(h, n, dx, min(time.time() + per * 0.4, budget_s - 70),
                              t0, 19, 14, 1000)
            if c2 < bc5:
                bc5, bh = c2, h2.copy()

    # Phase 4: Fine CD (higher resolution)
    if time.time() - t0 < budget_s - 40:
        h, c = _cd_loop(bh, n, dx, budget_s - 45, t0, 25, 20, 8000)
        if c < bc5:
            bc5, bh = c, h.copy()

    # Phase 5: Final high-alpha SLSQP
    if time.time() - t0 < budget_s - 15:
        h = _opt(bh, n, 6000, 800, 1e-16)
        c = _c5(h, dx)
        if c < bc5:
            bc5, bh = c, h.copy()

    # Phase 6: Final fine CD
    if time.time() - t0 < budget_s - 3:
        h, c = _cd_loop(bh, n, dx, budget_s - 4, t0, 31, 20, 3000)
        if c < bc5:
            bc5, bh = c, h.copy()

    bh = _repair(bh, n)
    bc5 = _c5(bh, dx)
    return (bh, bc5, n)
