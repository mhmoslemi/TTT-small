            if time.time() >= deadline:
                return h, best_c5
            elapsed = time.time() - t0
            temp = 0.003 * max(0.01, 1.0 - elapsed / total_time)
            improved = False
            for _ in range(600):
                if time.time() >= deadline:
                    return h, best_c5
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c5 = _c5_val(h, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                else:
                    accept = c5 - best_c5 < temp and rng.random() < np.exp(-(c5 - best_c5) / max(temp, 1e-15))
                    if accept and rng.random() < 0.1:
                        improved = improved or c5 < best_c5
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d
            if improved:
                stalled = 0
            else:
                stalled += 1
                if stalled >= 4:
                    break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0

    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = _c5_val(h, dx)

    # Phase 1: Full SLSQP warmstart (40s)
    if budget_s - (time.time() - t0) > 700:
        for a in [80, 200]:
            end = min(t0 + 40, time.time() + 20)
            h, c5 = _full_slsqp(h, n, dx, a, 400, end)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
            h = best_h

    # Phase 2: Block coordinate descent (main, ~550s)
    bs = 10
    nblk = n // bs
    if nblk * bs < n:
        bs = n // (n // bs) if n // bs > 0 else n
        nblk = n // bs
    blocks_base = [list(range(b * bs, min((b + 1) * bs, n))) for b in range(max(1, nblk))]
    deadline2 = t0 + min(600, budget_s - 350)

    rounds = 0
    while time.time() < deadline2:
        rounds += 1
        alpha = 100 + 20 * min(rounds, 50)
        order = blocks_base[:]
        rng.shuffle(order)
        for blk in order:
            if time.time() >= deadline2:
                break
            h2 = _block_opt(best_h, n, blk, alpha, tgt, deadline2)
            c5 = _c5_val(h2, dx)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, h2

    # Phase 3: Full SLSQP refinement with increasing alpha (200s)
    deadline3 = min(t0 + 820, time.time() + 200)
    for a in [200, 500, 1000, 2000, 5000]:
        if time.time() >= deadline3 or budget_s - (time.time() - t0) < 120:
            break
        h, c5 = _full_slsqp(best_h, n, dx, a, 600, deadline3)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        # Small perturbation restart
        if time.time() < deadline3 - 30:
            hp = _repair(best_h + 0.02 * rng.randn(n), n)
            h2, c52 = _full_slsqp(hp, n, dx, a, 300, deadline3)
            if c52 < best_c5:
                best_c5, best_h = c52, h2.copy()

    # Phase 4: Annealed local search (120s)
    if budget_s - (time.time() - t0) > 50:
        dl = min(time.time() + 120, t0 + budget_s - 30)
        best_h, best_c5 = _anneal_local(best_h, n, dx, rng, dl, t0)

    # Phase 5: Final SLSQP + local polish (30s)
    if budget_s - (time.time() - t0) > 15:
        h, c5 = _full_slsqp(best_h, n, dx, 10000, 800, time.time() + 15)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    if budget_s - (time.time() - t0) > 10:
        dl = time.time() + min(10, budget_s - (time.time() - t0) - 2)
        best_h, best_c5 = _anneal_local(best_h, n, dx, rng, dl, t0)

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
