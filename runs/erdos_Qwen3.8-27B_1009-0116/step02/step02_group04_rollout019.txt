import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_arr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, -h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-9:
                break
    return h

def _smooth_obj(x, n, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    try:
        res = minimize(
            lambda x: _smooth_obj(x, n, alpha), h,
            method='SLSQP', bounds=bounds, constraints=[eq_con],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

_eq_target = 0.0

def _eq_fun(x):
    return np.sum(x) - _eq_target

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_arr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = max(diff, -h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-9:
                break
    return h

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, dx, bounds, alpha, maxiter, deadline):
    global _eq_target
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    _eq_target = n / 2.0
    try:
        res = minimize(
            lambda x: _smooth_obj(x, alpha), h,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': _eq_fun}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _coord_step(h, m, n, dx):
    inv_n = 1.0 / n
    f = 1.0 - inv_n
    dmax_m = (1.0 - h[m]) / f
    dmin_m = -h[m] / f
    oth = np.delete(h, m)
    dmax_o = (oth.min() / inv_n) if oth.min() > 1e-15 else 0.0
    dmin_o = (oth.max() - 1.0) * n
    lo = max(dmin_m, dmin_o)
    hi = min(dmax_m, dmax_o)
    if hi - lo < 1e-14:
        return h.copy()
    a, b = lo, hi
    best_d = 0.0
    best_v = _compute_c5(h, dx)
    idx_arr = np.arange(n)
    mask = idx_arr != m
    for _ in range(30):
        t1 = a + (b - a) / 3.0
        t2 = b - (b - a) / 3.0
        h1 = h.copy()
        h1[m] += t1 * f
        h1[mask] -= t1 * inv_n
        h1 = np.clip(h1, 0.0, 1.0)
        h2 = h.copy()
        h2[m] += t2 * f
        h2[mask] -= t2 * inv_n
        h2 = np.clip(h2, 0.0, 1.0)
        v1 = _compute_c5(h1, dx)
        v2 = _compute_c5(h2, dx)
        if v1 < best_v:
            best_v, best_d = v1, t1
        if v2 < best_v:
            best_v, best_d = v2, t2
        if v1 < v2:
            b = t2
        else:
            a = t1
    if best_d != 0.0:
        h_new = h.copy()
        h_new[m] += best_d * f
        h_new[mask] -= best_d * inv_n
        h_new = np.clip(h_new, 0.0, 1.0)
        if _compute_c5(h_new, dx) <= best_v + 1e-13:
            return h_new
    return h.copy()

def _coord_sweep(h, n, dx, deadline, max_sweeps=5000):
    h = h.copy()
    best = _compute_c5(h, dx)
    for _sw in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        for m in range(n):
            if time.time() >= deadline:
                return h, best
            cur = _compute_c5(h, dx)
            h_new = _coord_step(h, m, n, dx)
            c5 = _compute_c5(h_new, dx)
            if c5 < cur - 1e-12:
                h = h_new
                improved = True
                if c5 < best:
                    best = c5
        if not improved:
            break
    return h, best

def _pair_swap(h, i, j, d, n, dx):
    h[i] += d
    h[j] -= d

def _targeted_swaps(h, n, dx, rng, deadline, n_tries=30000):
    h = h.copy()
    best = _compute_c5(h, dx)
    cnt = 0
    while time.time() < deadline and cnt < n_tries:
        cnt += 1
        corr = _corr_arr(h)
        ki = int(np.argmax(corr))
        tau = ki - (n - 1)
        if tau >= 0:
            js = np.arange(n - tau)
            contribs = h[js + tau] * (1.0 - h[js])
        else:
            s = -tau
            js = np.arange(n - s)
            contribs = h[js] * (1.0 - h[js + s])
        if contribs.size == 0:
            continue
        order = np.argsort(contribs)[::-1]
        for jj in order[:6]:
            i = int(js[jj])
            if tau >= 0:
                a, b = i + tau, i
            else:
                a, b = i, i + (-tau)
            if a >= n or b >= n or a == b:
                continue
            d = 0.005 + 0.03 * rng.random()
            if h[a] < d + 1e-9 or h[b] > 1.0 - d - 1e-9:
                continue
            h[a] -= d
            h[b] += d
            c5 = _compute_c5(h, dx)
            if c5 < best - 1e-12:
                best = c5
            else:
                h[a] += d
                h[b] -= d
    return h, best

def _random_swaps(h, n, dx, rng, deadline, scale=0.02):
    h = h.copy()
    best = _compute_c5(h, dx)
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d
        h[j] -= d
        c5 = _compute_c5(h, dx)
        if c5 < best - 1e-12:
            best = c5
        else:
            h[i] -= d
            h[j] += d
    return h, best

def _make_structured(n, kind, rng):
    if kind == 0:
        h = np.zeros(n)
        for i in range(n // 2):
            h[i] = 1.0
    elif kind == 1:
        period = max(2, n // 5)
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * np.arange(n) / period)
    elif kind == 2:
        h = 0.5 + 0.4 * np.cos(4.0 * np.pi * np.arange(n) / n) + 0.1 * rng.randn(n)
    else:
        h = 0.5 + 0.35 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
