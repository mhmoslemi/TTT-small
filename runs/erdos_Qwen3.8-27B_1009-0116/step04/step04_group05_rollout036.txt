def project_feasible(h, n):
    """Project onto {h in [0,1]^n : sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Water-filling
    if diff > 0:
        # Need to add mass, prioritize elements with most room
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in order:
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
            if remaining < 1e-12:
                break
    else:
        # Need to remove mass, prioritize elements with most mass
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14: break
            a = min(d, 1.0-h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * dx

def grad_corr(h, ki, n):
    g = np.zeros(n)
    L = ki - (n-1)
    if L == 0:
        g[:] = 1.0 - 2.0*h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

    n = _SM['n']

    m = np.max(c)
    s = np.clip(_SM['a']*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s)))/_SM['a'] + _SM['l']*(x.sum()-n*0.5)**2

    n = _SM['n']

    m = np.max(c)
    s = np.clip(_SM['a']*(c-m), -500, 500)

    bh, bc = h.copy(), c5v(h, dx)
    h = bh.copy()
    gamma = 0.05
    momentum = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_corr(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

_SM = {'a': 100.0, 'l': 50.0, 'n': 70}

def sm_obj(x):
    n = _SM['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(_SM['a'] * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _SM['a'] + _SM['l'] * (x.sum() - n * 0.5) ** 2

def sm_jac(x):
    n = _SM['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(_SM['a'] * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-14:
            g += pw[k] * grad_corr(x, k, n)
    g /= _SM['a']
    g += 2.0 * _SM['l'] * (x.sum() - n * 0.5)
    return g

def lbfgs_refine(h, n, dx, t0, lim):
    _SM['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    for a, lm in [(50, 80), (200, 60), (1000, 40), (5000, 20), (20000, 10), (100000, 5), (400000, 2)]:
        if time.time() - t0 > lim:
            break
        _SM['a'], _SM['l'] = float(a), float(lm)
        try:
            r = minimize(sm_obj, bh.copy(), jac=sm_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-18})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def subgrad_method(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    h = bh.copy()
    gamma = 0.05
    mom = np.zeros(n)
    best_f = bc
    for it in range(8000):
        if time.time() - t0 > lim:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = c.max()
        fval = cm * dx
        thr = cm - 0.004 * max(c.std(), 1e-10)
        idx = np.where(c >= thr)[0]
        if len(idx) > 5:
            idx = np.argsort(c)[::-1][:5]
        s = np.zeros(n)
        for ki in idx:
            s += grad_corr(h, ki, n)
        s /= len(idx)
        s -= s.mean()
        sn = max(np.dot(s, s), 1e-10)
        if fval < best_f:
            gamma = min(gamma * 1.02, (best_f - fval + 0.005) / sn)
        else:
            gamma = max(gamma * 0.97, 0.0005)
        mom = 0.5 * mom + 0.5 * s
        hn = repair(h - gamma * mom, n)
        cn = c5v(hn, dx)
        if cn < fval:
            h = hn
            if cn < bc:
                bc, bh = cn, h.copy()
            best_f = min(best_f, cn)
        else:
            gamma *= 0.8
            hn2 = repair(h - 0.1 * gamma * mom, n)
            cn2 = c5v(hn2, dx)
            if cn2 < fval:
                h = hn2
                if cn2 < bc:
                    bc, bh = cn2, h.copy()
    return bh, bc

def active_set(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [1, 2, 3, 5, 8, 13, 21, 34, 55]:
        h = bh.copy()
        step = 0.1
        stalled = 0
        for _ in range(1200):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.001 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += grad_corr(h, ki, n)
            g /= len(idx)
            g -= g.mean()
            ok = False
            for _ in range(50):
                hn = repair(h - step * g, n)
                cn = c5v(hn, dx)
                if cn < c5v(h, dx) - 1e-18:
                    h = hn
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.1, 2.0)
                    stalled = 0
                    ok = True
                    break
                step *= 0.7
            if not ok:
                stalled += 1
                if stalled > 15:
                    break
    return bh, bc

def coord_descent(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    while time.time() - t0 < lim:
        imp = False
        for _ in range(4000):
            if time.time() - t0 > lim:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in (0.02, 0.005, 0.001, 0.0003, -0.02, -0.005, -0.001, -0.0003):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 5:
            break
    return bh, bc

def spectral_smooth(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.1, 0.2, 0.35, 0.5, 0.35, 0.2, 0.1, 0.05]:
        h = bh.copy()
        for _ in range(400):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_refine(bh, n, dx, t0, lim - 180)

        d = (rng.rand() - 0.5) * 0.001

                if cc < bc:

            else:

def de_explore(h_best, n, dx, seed, t0, lim):
    best_h, best_c = h_best.copy(), c5v(h_best, dx)
    bds = [(0.0, 1.0)] * n
    def obj(x):
        if time.time() - t0 > lim:
            return best_c + 1.0
        return c5v(repair(x, n), dx)
    try:
        mi = max(5, int((lim - (time.time() - t0)) * 0.7 / 3.0))
        res = differential_evolution(obj, bds, strategy='best1bin',
                                     maxiter=mi, popsize=12,
                                     mutation=(0.3, 0.8), recombination=0.8,
                                     seed=seed, polish=False, tol=1e-8)
        if hasattr(res, 'x') and res.x is not None:
            hc = repair(res.x, n)
            c = c5v(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
    except Exception:
        pass
    return best_h, best_c

def make_starts(n, seed, h_ref):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    starts = []
    for amp in [0.05, 0.1, 0.2, 0.35]:
        starts.append(repair(h_ref + amp * rng.randn(n), n))
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.45]:
            for ph in [0, np.pi / 2, np.pi]:
                starts.append(repair(0.5 + amp * np.sin(np.pi * freq * x + ph), n))
    for _ in range(4):
        starts.append(repair(rng.uniform(0, 1, n), n))
    return starts
