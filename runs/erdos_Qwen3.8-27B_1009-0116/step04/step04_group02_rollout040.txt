def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')

    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

            best_h, best_c5 = hh, cc

    # Phase 3: Multi-restart optimization (150s)
    t_end = time.time() + min(150, budget_s - (t_end - t_start) - 80)
    for r in range(20):
        if time.time() >= t_end:
            break
        hp = perturb_restart(best_h, n, rng, r % 6)
        hh, cc = lbfgs_opt(hp, n, 800, target, 3000, 150)
        hh, cc = slsqp_opt(hh, n, 5000, target, 200)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 4: Block local search (60s)
    t_end = time.time() + min(60, budget_s - (time.time() - t_start) - 60)
    if time.time() < t_end:
        best_h, best_c5 = block_local(best_h, n, dx, rng, t_end, 10000)

    # Phase 5: Fine local search (120s)
    t_end = time.time() + min(120, budget_s - (time.time() - t_start) - 30)
    if time.time() < t_end:
        best_h, best_c5 = fine_local_search(best_h, n, dx, rng, t_end, n_trials=250000)

    # Phase 6: SLSQP + L-BFGS polish (30s)
    t_end = time.time() + min(30, budget_s - (time.time() - t_start) - 10)
    if time.time() < t_end:
        hh, cc = slsqp_opt(best_h, n, 20000, target, 300)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
        hh, cc = lbfgs_opt(best_h, n, 50000, target, 5000, 300)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 7: Ultra-fine local search (remaining)
    t_end = time.time() + max(5, min(20, budget_s - (time.time() - t_start) - 2))
    if time.time() < t_end:
        best_h, best_c5 = fine_local_search(best_h, n, dx, rng, t_end, n_trials=80000)

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
