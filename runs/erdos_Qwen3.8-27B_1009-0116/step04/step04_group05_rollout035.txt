            bh, bc = r, c
        rem = budget_s - time.time()

    # Phase 6: Second smooth pass with higher temp (100s)
    if rem > 100:
        r, c = smooth_opt(bh, bn, 2.0/bn, [10000, 50000, 200000, 800000], 5.0,
                          t0 + 610, 250)
        if c < bc:
            bh, bc = r, c
        rem = budget_s - time.time()

    # Phase 7: Active + spec combo (60s)
    if rem > 70:
        r, c = active_opt(bh, bn, 2.0/bn, t0 + 660)
        r, c = spec_opt(r, bn, 2.0/bn, t0 + 690)
        if c < bc:
            bh, bc = r, c
        rem = budget_s - time.time()

    # Phase 8: Upsample to 2n (200s)
    n2 = bn * 2
    if n2 < 1000 and rem > 180:
        xo = np.linspace(0, bn, bn, endpoint=False)
        xn = np.linspace(0, n2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=bn), n2)
        d2 = 2.0 / n2
        r, c = smooth_opt(h2, n2, d2, [3000, 15000, 60000, 200000], 5.0, t0 + 850, 250)
        r, c = active_opt(r, n2, d2, t0 + 920)
        r, c = spec_opt(r, n2, d2, t0 + 950)
        if c < bc:
            bh, bc, bn = r, c, n2
        rem = budget_s - time.time()

    # Phase 9: Polish (60s)
    if rem > 50:
        rng = np.random.RandomState(seed + 777)
        r, c = cd_polish(bh, bn, 2.0/bn, rng, t0 + budget_s - 15, 50000)
        if c < bc:
            bh, bc = r, c
        rem = budget_s - time.time()

    # Phase 10: Perturbation restart (30s)
    if rem > 35:
        rng2 = np.random.RandomState(seed + 1234)
        hp = repair(bh + 0.02 * rng2.randn(bn), bn)
        r, c = smooth_opt(hp, bn, 2.0/bn, [500, 5000, 30000], 10.0, t0 + budget_s - 5, 120)
        r, c = active_opt(r, bn, 2.0/bn, t0 + budget_s - 3)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5val(bh, 2.0 / bn)
    return (bh, bc, bn)
