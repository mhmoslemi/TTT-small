from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _pnorm_obj(h, p, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(c, 1e-30)
    K = len(c)
    return (np.sum(c**p) / K)**(1.0/p) * dx

def _pnorm_grad(h, p, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(c, 1e-30)
    K = len(c)
    sp = np.sum(c**p)
    w = c**(p-1) * (sp/K)**(1.0/p - 1.0) / K * dx
    g = np.zeros(n)
    for k in range(K):
        wk = w[k]
        if wk < 1e-20:
            continue
        sh = (n-1) - k
        if sh == 0:
            g += wk * (1.0 - 2.0*h)
        elif sh > 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        else:
            sp2 = -sh
            g[sp2:] += wk * (1.0 - h[:n-sp2])
            g[:n-sp2] -= wk * h[sp2:]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lbfgs_pnorm(h, p, dx, maxiter, beta):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _pnorm_obj(x, p, dx) + beta * (x.sum() - target)**2
    
    def grad(x):
        return _pnorm_grad(x, p, dx) + 2*beta*(x.sum() - target)
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    n = len(h)

        k_w = c.argmax()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _pnorm_obj(x, p, dx, beta, target):
    c = np.correlate(x, 1.0 - x, mode='full')
    c = np.maximum(c, 1e-30)
    K = len(c)
    return (np.sum(c**p) / K)**(1.0/p) * dx + beta * (x.sum() - target)**2

def _pnorm_grad(x, p, dx, beta, target):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    c = np.maximum(c, 1e-30)
    K = len(c)
    sp = np.sum(c**p)
    w = c**(p-1) * (sp/K)**(1.0/p - 1.0) / K * dx
    g = np.zeros(n)
    for k in range(K):
        wk = w[k]
        if wk < 1e-20:
            continue
        sh = (n-1) - k
        if sh == 0:
            g += wk * (1.0 - 2.0*x)
        elif sh > 0:
            g[sh:] += wk * (1.0 - x[:n-sh])
            g[:n-sh] -= wk * x[sh:]
        else:
            j = -sh
            g[:n-j] += wk * (1.0 - x[j:])
            g[j:] -= wk * x[:n-j]
    g += 2.0 * beta * (x.sum() - target)
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lbfgs(h, p, dx, beta, target, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, method='L-BFGS-B',
                       args=(p, dx, beta, target), bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _peak_shave(h, nsteps):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(nsteps):
        c = np.correlate(h, 1.0 - h, mode='full')
        k_w = c.argmax()
        sh = (n-1) - k_w
        g = np.zeros(n)
        if sh == 0:
            g = 1.0 - 2.0*h
        elif sh > 0:
            g[sh:] += (1.0 - h[:n-sh])
            g[:n-sh] -= h[sh:]
        else:
            j = -sh
            g[:n-j] += (1.0 - h[j:])
            g[j:] -= h[:n-j]
        g *= dx
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.0005]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                break
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)
