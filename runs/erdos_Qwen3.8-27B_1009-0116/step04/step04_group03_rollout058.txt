import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        # Term 1: j in [max(0,-lag), min(n, n-lag)), contributes wk*(1-h[j+lag])
        t1s = max(0, -lag)
        t1e = min(n, n - lag)
        if t1s < t1e:
            g[t1s:t1e] += wk * (1.0 - h[t1s + lag:t1e + lag])
        # Term 2: j in [max(0,lag), min(n, n+lag)), contributes -wk*h[j-lag]
        t2s = max(0, lag)
        t2e = min(n, n + lag)
        if t2s < t2e:
            g[t2s:t2e] -= wk * h[t2s - lag:t2e - lag]
    return g

def _cfunc(h, n):
    return np.sum(h) - n / 2.0

def _cfunc_jac(h, n):
    return np.ones(n)

def _pen_obj(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    s -= s.max()
    val = m + np.log(np.sum(np.exp(s))) / alpha
    penalty = 500.0 * (np.sum(h) - n / 2.0) ** 2
    return val + penalty

def _pen_obj_jac(h, n, alpha):
    g = _gs(h, alpha)
    g += 1000.0 * (np.sum(h) - n / 2.0)
    return g

def _slsqp_opt(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfunc, 'jac': _cfunc_jac, 'args': (n,)}]
    try:
        res = minimize(_sf, h0, jac=_gs, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _lbfgs_opt(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pen_obj, h0, jac=_pen_obj_jac, args=(n, alpha),
                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _local_search(h, n, rng, t_end):
    c5c = _eval(h)
    scale = 0.006
    no_imp = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.03, 0.015)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 300:
                scale *= 0.82
                no_imp = 0
                if scale < 5e-7:
                    break
                if rng.rand() < 0.05:
                    scale = max(scale, 0.003)
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        t1s = max(0, -lag)
        t1e = min(n, n - lag)
        if t1s < t1e:
            g[t1s:t1e] += wk * (1.0 - h[t1s + lag:t1e + lag])
        t2s = max(0, lag)
        t2e = min(n, n + lag)
        if t2s < t2e:
            g[t2s:t2e] -= wk * h[t2s - lag:t2e - lag]
    return g

    return np.sum(h) - n / 2.0

def _cfunc_jac(h, n):
    return np.ones(n)

def _pen_obj(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    s -= s.max()
    val = m + np.log(np.sum(np.exp(s))) / alpha
    return val + 800.0 * (np.sum(h) - n / 2.0) ** 2

def _pen_jac(h, n, alpha):
    g = _gs(h, alpha)
    g += 1600.0 * (np.sum(h) - n / 2.0)
    return g

def _slsqp_opt(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfunc, 'jac': _cfunc_jac, 'args': (n,)}]
    try:
        res = minimize(_sf, h0, jac=_gs, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _lbfgs_opt(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pen_obj, h0, jac=_pen_jac, args=(n, alpha),
                      method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)
