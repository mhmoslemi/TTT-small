        T = t_hi * (t_lo / t_hi) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_h, new_c5 = exact_pair(h, i, j, n, dx, corr)
        delta = new_c5 - curr_c5
        if delta < -1e-15 or (T > t_lo * 0.1 and np.exp(-delta / T) > rng.random()):
            h = new_h
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5

def grad_exact_pairs(h, n, dx, rng, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        top5 = np.argsort(c_vals)[::-1][:5]
        best_new_h = h
        best_new_c5 = best_c5
        for mi in top5:
            k = mi - (n - 1)
            ks = k
            iz = np.arange(n)
            v1 = (iz + ks >= 0) & (iz + ks < n)
            v2 = (iz - ks >= 0) & (iz - ks < n)
            g = np.zeros(n)
            g[v1] += (1.0 - h[iz[v1] + ks])
            g[v2] += -h[iz[v2] - ks]
            g *= dx
            g -= np.mean(g)
            inc = np.argsort(g)[:6]
            dec = np.argsort(g)[::-1][:6]
            for a in inc:
                for b in dec:
                    if a == b:
                        continue
                    nh, nc = exact_pair(h, a, b, n, dx, corr)
                    if nc < best_new_c5 - 1e-15:
                        best_new_c5 = nc
                        best_new_h = nh
        if best_new_c5 >= best_c5 - 1e-14:
            r = rng.random()
            if r < 0.3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            nh, nc = exact_pair(h, i, j, n, dx, corr)
            if nc < best_c5 + 0.0005:
                best_new_h, best_new_c5 = nh, nc
        h = best_new_h
        if best_new_c5 < best_c5:
            best_c5 = best_new_c5
            best_h = h.copy()
    return best_h, best_c5

def lbfgs_logsumexp(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
        return repair(res.x, n), compute_c5(repair(res.x, n), dx)
    except Exception:
        return h, compute_c5(h, dx)

def fine_coord(h, n, dx, rng, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        corr = np.correlate(h, 1.0 - h, mode='full')
        nh, nc = exact_pair(h, i, j, n, dx, corr)
        if nc < best_c5 - 1e-15:
            h = nh
            best_c5 = nc
            best_h = h.copy()
        elif nc < best_c5 + 1e-12:
            h = nh
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remain():
        return budget_s - (time.time() - t_start)

    h70 = np.array(initial_h_values, dtype=float)
    n0 = len(h70)
    dx0 = 2.0 / n0
    h0 = repair(h70.copy(), n0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx0)
    best_n = n0

    # Phase 1: L-BFGS warm start (40s)
    t_end = t_start + 40
    hc = h0.copy()
    cc = best_c5
    for alpha in [500, 3000, 15000]:
        if time.time() >= t_end:
            break
        hh, c_val = lbfgs_logsumexp(hc, n0, alpha, n0 / 2.0, 5000, 150)
        if c_val < cc:
            hc, cc = hh, c_val
    if cc < best_c5:
        best_h, best_c5, best_n = hc.copy(), cc, n0

    # Phase 2: Exact-pair SA (280s) - main exploration
    if remain() > 400:
        t_end = time.time() + min(280, remain() - 350)
        h2, c2 = sa_exact_pairs(hc, n0, dx0, rng, t_end, t_hi=0.004, t_lo=0.00001)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n0

    # Phase 3: Gradient-guided exact pairs (250s)
    if remain() > 200:
        t_end = time.time() + min(250, remain() - 160)
        h3, c3 = grad_exact_pairs(best_h, n0, dx0, rng, t_end)
        if c3 < best_c5:
            best_h, best_c5, best_n = h3, c3, n0

    # Phase 4: L-BFGS polish (120s)
    if remain() > 100:
        t_end = time.time() + min(120, remain() - 80)
        for alpha in [5000, 30000, 100000]:
            if time.time() >= t_end:
                break
            hh, c_val = lbfgs_logsumexp(best_h, n0, alpha, n0 / 2.0, 10000, 300)
            if c_val < best_c5:
                best_h, best_c5 = hh, c_val

    # Phase 5: Fine coordinate search (100s)
    if remain() > 80:
        t_end = time.time() + min(100, remain() - 50)
        h5, c5 = fine_coord(best_h, n0, dx0, rng, t_end)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Final SA polish (40s)
    if remain() > 30:
        t_end = time.time() + min(40, remain() - 20)
        h6, c6 = sa_exact_pairs(best_h, n0, dx0, rng, t_end, t_hi=0.001, t_lo=0.000001)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    # Phase 7: Ultra-fine coordinate (20s)
    if remain() > 15:
        t_end = time.time() + min(18, remain() - 8)
        h7, c7 = fine_coord(best_h, n0, dx0, rng, t_end)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
