def soft_max_obj(v, n, p):
    """Power-sum soft max approximation"""
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    cmax = np.max(corr)
    # Only consider values near the max
    mask = corr > cmax - 0.5
    if np.sum(mask) == 0:
        return cmax
    shifted = corr[mask] - cmax
    return cmax + np.log(np.sum(np.exp(p * shifted))) / p

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(20000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def eval_lags(h, n, dx, shifts):
    vals = []
    for s in shifts:
        if s == 0:
            v = np.sum(h * (1.0 - h))
        elif s > 0:
            v = np.sum(h[:-s] * (1.0 - h[s:]))
        else:
            v = np.sum(h[-s:] * (1.0 - h[:-s]))
        vals.append(v * dx)
    return max(vals)

def slsqp_opt(h, n, p, maxiter=500):
    dx = 2.0 / n
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        cmax = np.max(corr)
        shifted = np.clip(corr - cmax, -500.0, 0.0)
        return cmax + np.log(np.sum(np.exp(p * shifted))) / p
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def targeted_descent(h, n, t_end, rng, top_k=6):
    h = h.copy()
    dx = 2.0 / n
    best = compute_c5(h, n)
    corr = compute_corr(h, n)
    top_idx = np.argsort(corr)[-top_k:]
    shifts = sorted(set(top_idx - (n - 1)))
    shifts = [s for s in shifts if s != 0]
    if not shifts:
        shifts = [1]
    steps = [0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    stall = 0
    for it in range(2000000):
        if time.time() > t_end:
            break
        if it % 3000 == 0:
            corr = compute_corr(h, n)
            top_idx = np.argsort(corr)[-top_k:]
            new_shifts = sorted(set(top_idx - (n - 1)))
            shifts = [s for s in new_shifts if s != 0] or [1]
            c5_full = compute_c5(h, n)
            if c5_full < best:
                best = c5_full
        s = shifts[rng.randint(len(shifts))]
        if s > 0:
            lo, hi = 0, n - s
        else:
            lo, hi = -s, n
        i = rng.randint(lo, max(lo + 1, hi))
        j = i + s
        hid, hjd = h[i], h[j]
        found = False
        for d in steps:
            for sign in (1, -1):
                delta = sign * d
                ni, nj = hid + delta, hjd - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5_new = eval_lags(h, n, dx, shifts)
                if c5_new < best - 1e-11:
                    best = c5_new
                    stall = 0
                    found = True
                    break
                h[i], h[j] = hid, hjd
            if found:
                break
        if not found:
            stall += 1
        if stall > 50000:
            break
    return h, best

def binary_nudge(h, n, t_end, rng):
    dx = 2.0 / n
    h = h.copy()
    best = compute_c5(h, n)
    corr = compute_corr(h, n)
    top_idx = np.argsort(corr)[-6:]
    shifts = [s for s in (top_idx - (n - 1)) if s != 0] or [1]
    for it in range(200000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        v = h[i]
        target = 1.0 if v > 0.5 else 0.0
        step = 0.01
        h[i] = np.clip(v + (target - v) * step, 0.0, 1.0)
        c5_new = eval_lags(h, n, dx, shifts)
        full_check = (it % 500 == 0)
        if full_check:
            c5_new = compute_c5(h, n)
        if c5_new < best - 1e-12:
            best = c5_new
            h = repair(h, n)
        else:
            h[i] = v
    return h, best
