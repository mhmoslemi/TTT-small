    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, target, n_iter=300):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(n_iter):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

    c = _corr(h)
    m = c.max()
    soft = m + np.log(np.exp(alpha * (c - m)).sum()) / alpha
    return soft + mu * (h.sum() - target) ** 2

    weights = np.zeros(len(c))

    top_idx = np.argsort(c)[-top_n:]
    mx = c.max()
    w = np.exp(-0.5 * (mx - c[top_idx]))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        grad += wi * _grad_lag(h, int(ki) - (n - 1))
    return _repair(h - lr * grad, target)

def _subgrad_loop(h, dx, target, t_end, top_n=5, lr0=0.012):
    best = _c5(h, dx)
    h = h.copy()
    lr = lr0
    stall = 0
    while time.time() < t_end:
        h_new = _subgrad_step(h, lr, target, top_n)
        c_new = _c5(h_new, dx)
        if c_new < best - 1e-13:
            best = c_new
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.92
                stall = 0
            if _c5(h_new, dx) < _c5(h, dx):
                h = h_new
        if lr < 5e-7:
            break
    return h, best

def _slsqp(h0, dx, alpha, maxiter, target):
    n = len(h0)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    def obj(x):
        c = _corr(x)
        m = c.max()
        return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-17})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _local_search(h, dx, rng, t_end, target, n_iter=400000):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.006, 0.0025, 0.001, 0.0004, 0.00015, 0.00006]
    per = max(400, n_iter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return h, best
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                c = _c5(h, dx)
                if c < best:
                    best = c
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _targeted_local(h, dx, rng, t_end, target, n_iters=80000):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_lags = np.argsort(c)[-4:]
        for ki in top_lags:
            k = int(ki) - (n - 1)
            if k == 0:
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
            elif k > 0 and n - k > 0:
                i = rng.randint(n - k); j = i + k
            elif k < 0 and n + k > 0:
                j = rng.randint(n + k); i = j - k
            else:
                continue
            d = rng.uniform(-0.012, 0.012)
            h[i] += d; h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                c_new = _c5(h, dx)
                if c_new < best:
                    best = c_new
                    h = _repair(h, target)
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _anneal(h, dx, rng, t_end, target, t_max=0.008):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    T = t_max
    while time.time() < t_end and T > 1e-5:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.04, 0.04)
        h[i] += d; h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c_new = _c5(h, dx)
            if c_new < best or rng.random() < np.exp((best - c_new) / (T * dx)):
                best = min(best, c_new)
                h = _repair(h, target)
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
        T *= 0.995
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    tgt0 = n0 / 2.0
    best_h = h0.copy()
    best_c5 = _c5(h0, dx0)
    best_n = n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at n0
    t_end = t0 + 55
    for a in [200, 1000, 5000]:
        if time.time() > t_end:
            break
        h0, c = _slsqp(h0, dx0, a, 400, tgt0)
        if c < best_c5:
            best_h, best_c5 = h0.copy(), c

    # Phase 2: Subgradient at n0
    t_end = t0 + 220
    best_h, best_c5 = _subgrad_loop(best_h, 2.0 / best_n, best_n / 2.0, t_end, 5, 0.012)
    best_n = n0

    # Phase 3: Multi-restart
    t_end = t0 + 370
    for r in range(6):
        if time.time() > t_end or rem() < 350:
            break
        amp = 0.06 * (1.0 - r / 10.0)
        hp = best_h + amp * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, tgt0)
        hp, c = _slsqp(hp, dx0, 3000, 250, tgt0)
        hp, c = _subgrad_loop(hp, dx0, tgt0, min(time.time() + 15, t_end), 3, 0.008)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 4: Double resolution
    n1 = 2 * n0
    dx1 = 2.0 / n1
    tgt1 = n1 / 2.0
    h1 = _repair(_interp(best_h, n1), tgt1)
    c1 = _c5(h1, dx1)
    t_end = min(t0 + 620, t0 + budget_s - 380)
    for a in [500, 3000]:
        if time.time() > t_end:
            break
        h1, c1 = _slsqp(h1, dx1, a, 500, tgt1)
    t_end = min(t0 + 760, t0 + budget_s - 250)
    h1, c1 = _subgrad_loop(h1, dx1, tgt1, t_end, 5, 0.008)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, n1

    # Phase 5: Targeted local on best
    dxb = 2.0 / best_n
    tgtb = best_n / 2.0
    t_end = min(t0 + 900, t0 + budget_s - 130)
    best_h, best_c5 = _targeted_local(best_h, dxb, rng, t_end, tgtb, 60000)

    # Phase 6: Anneal + fine search
    t_end = t0 + budget_s - 80
    best_h, best_c5 = _anneal(best_h, dxb, rng, t_end, tgtb)
    best_h = _repair(best_h, tgtb)
    t_end = t0 + budget_s - 25
    best_h, best_c5 = _local_search(best_h, dxb, rng, t_end, tgtb, 600000)

    # Phase 7: Final subgradient + tiny search
    if rem() > 25:
        best_h, best_c5 = _subgrad_loop(best_h, dxb, tgtb, t0 + budget_s - 10, 3, 0.004)
    if rem() > 8:
        best_h, best_c5 = _local_search(best_h, dxb, rng, t0 + budget_s - 2, tgtb, 80000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
