import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_c_k(h, k, n):
    s = abs((n - 1) - k)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def _smoothed_grad(h, n, alpha, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    w_raw = np.exp(alpha * 0.1 * (c[idx] - c[idx].max()))
    w = w_raw / w_raw.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, k in zip(w, idx):
        g += wi * _grad_c_k(h, k, n)
    return g * dx

def _slsqp_opt(h, n, alpha, maxiter):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        return _smoothed_grad(x, len(x), alpha, 15)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, n, nsteps, alpha, step0, decay):
    h = h.copy()
    f = _eval(h)
    step = step0
    top_m = 8
    for it in range(nsteps):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-top_m:]
        w_raw = np.exp(0.05 * alpha * (c[idx] - c[idx].max()))
        w = w_raw / w_raw.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for wi, k in zip(w, idx):
            g += wi * _grad_c_k(h, k, n)
        g *= dx
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            h_new = _proj(h - step * mult * g, n)
            fn = _eval(h_new)
            if fn < f - 1e-14:
                h = h_new
                f = fn
                improved = True
                break
        if not improved:
            step *= 0.9
        else:
            step *= decay
    return h

def _upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_hi = np.zeros(n_new)
    for i in range(n):
        for j in range(factor):
            idx = i * factor + j
            if j == 0:
                t = 0.0
            else:
                t = j / factor
            if i + 1 < n:
                h_hi[idx] = h[i] * (1 - t) + h[i + 1] * t
            else:
                h_hi[idx] = h[i]
    return h_hi

def _pairwise_refine(h, n, rng, niter, scales):
    h = h.copy()
    c5c = _eval(h)
    for scale in scales:
        for _ in range(niter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return _repair(h, n), c5c
