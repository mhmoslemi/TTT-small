import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def topk_obj(x, target, pen, k=5, lam=10.0):
    """Objective: penalize top-k correlations heavily, others lightly"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    c5 = np.max(corr)
    # Get top k correlations
    top_idx = np.argsort(corr)[-k:]
    top_sum = np.sum(corr[top_idx])
    # Light penalty on all
    all_sum = np.sum(corr) / len(corr)
    return c5 + lam * top_sum / k + 0.1 * all_sum + pen * (np.sum(x) - target) ** 2

def lbfgs_opt(h, n, obj_func, args, maxiter, bounds):
    dx = 2.0 / n
    try:
        res = minimize(obj_func, h, args=args, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)
