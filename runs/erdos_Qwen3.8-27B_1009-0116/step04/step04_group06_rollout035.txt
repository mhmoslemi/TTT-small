    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add

                h[idx] -= sub
                diff += sub

    for _ in range(60):
        m = (lo + hi) * 0.5
        if np.sum(np.clip(h - m, 0.0, 1.0)) > t:
            lo = m
        else:
            hi = m
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = t - hp.sum()
    if abs(d) > 1e-14:
        mk = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if mk.any():
            hp[mk] += d / mk.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _grad_c5(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            l2 = -L
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _fista(h0, n, alpha, tend, lr, ni):
    xk = h0.copy()
    xk1 = h0.copy()
    tk = 1.0
    fk = _c5_val(xk, n)
    for it in range(ni):
        if time.time() > tend:
            break
        tk1 = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = xk + (tk - 1.0) / tk1 * (xk - xk1)
        y = np.clip(y, 0.0, 1.0)
        g = _grad_c5(y, n, alpha)
        x_new = _proj(y - lr * g, n)
        f_new = _c5_val(x_new, n)
        if f_new <= fk + 1e-15:
            xk1 = xk
            xk = x_new
            tk = tk1
            fk = f_new
        else:
            tk = 1.0
            xk1 = xk.copy()
            lr *= 0.85
        if it % 2000 == 1999:
            lr *= 0.92
    return xk

def _obj_s(x):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / _S_N)
    m = np.max(c)
    s = np.clip(_S_ALPHA * (c - m), -400, 400)
    v = m + np.log(np.sum(np.exp(s))) / _S_ALPHA
    if _S_BW > 0:
        v += _S_BW * np.sum(x * (1.0 - x))
    return v

def _grad_s(x):
    g = _grad_c5(x, _S_N, _S_ALPHA)
    if _S_BW > 0:
        g += 2.0 * _S_BW * (x - 0.5)
    return g

def _cf(x):
    return float(np.sum(x) - _S_N / 2.0)

def _cj(x):
    return np.ones(_S_N)

def _slsqp(hi, n, alpha, mi, bw):
    global _S_N, _S_ALPHA, _S_BW
    _S_N, _S_ALPHA, _S_BW = n, alpha, bw
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    r = minimize(_obj_s, hi, jac=_grad_s, method='SLSQP',
                 bounds=b, constraints=c, options={'maxiter': mi, 'ftol': 1e-18})
    return r.x

def _local(h, n, tend, rng, npass):
    c5 = _c5_val(h, n)
    dl = np.array([0.01, 0.04, 0.08, -0.01, -0.04, -0.08])
    for _ in range(npass):
        if time.time() > tend:
            break
        imp = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for i in idx:
            if time.time() > tend:
                break
            for j in idx:
                if j == i:
                    continue
                hi, hj = h[i], h[j]
                for d in dl:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5_val(h, n)
                    if c < c5 - 1e-15:
                        c5 = c; imp = True; hi, hj = ni, nj
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not imp:
            break
    return h, c5

def _interp(h, nf, nt):
    xo = np.linspace(0, 2.0, nf, endpoint=False)
    xn = np.linspace(0, 2.0, nt, endpoint=False)
    return _repair(np.interp(xn, xo, h, period=2.0), nt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    bh = h0.copy()
    bn = len(h0)
    bc5 = _c5_val(bh, bn)

    # Phase 1: FISTA with increasing alpha
    for al in [3000, 10000, 50000, 200000, 500000]:
        if time.time() - t0 > budget_s * 0.28:
            break
        hw = _fista(bh, bn, al, t0 + budget_s * 0.28, 0.03, 10000)
        cw = _c5_val(hw, bn)
        if cw < bc5:
            bc5 = cw; bh = hw.copy()

    # Phase 2: SLSQP polishing with binary push
    for al in [50000, 200000, 500000]:
        if time.time() - t0 > budget_s * 0.48:
            break
        for r in range(2):
            if time.time() - t0 > budget_s * 0.48:
                break
            hi = bh.copy() if r == 0 else _repair(bh + 0.015 * rng.randn(bn), bn)
            try:
                hx = _slsqp(hi, bn, al, 500, 0.002)
                hc = _repair(hx, bn)
                c5 = _c5_val(hc, bn)
                if c5 < bc5:
                    bc5 = c5; bh = hc
            except Exception:
                pass

    # Phase 3: Multi-resolution exploration
    for nt in [100, 120, 160, 200, 280]:
        if time.time() - t0 > budget_s * 0.72:
            break
        if nt == bn:
            continue
        hn = _interp(bh, bn, nt)
        hw = _fista(hn, nt, 100000, t0 + budget_s * 0.72, 0.015, 4000)
        cw = _c5_val(hw, nt)
        if cw < bc5 - 1e-7:
            bc5 = cw; bh = hw.copy(); bn = nt
        elif cw < bc5 + 0.002:
            try:
                hx = _slsqp(hw, nt, 200000, 400, 0.001)
                hc = _repair(hx, nt)
                c5 = _c5_val(hc, nt)
                if c5 < bc5:
                    bc5 = c5; bh = hc; bn = nt
            except Exception:
                pass

    # Re-optimize at best resolution with fresh FISTA
    if time.time() - t0 < budget_s * 0.82:
        hw = _fista(bh, bn, 300000, t0 + budget_s * 0.82, 0.01, 5000)
        cw = _c5_val(hw, bn)
        if cw < bc5:
            bc5 = cw; bh = hw.copy()
        try:
            hx = _slsqp(bh, bn, 500000, 400, 0.001)
            hc = _repair(hx, bn)
            c5 = _c5_val(hc, bn)
            if c5 < bc5:
                bc5 = c5; bh = hc
        except Exception:
            pass

    # Phase 4: Structured local search
    if time.time() - t0 < budget_s * 0.95:
        bh, bc5 = _local(bh, bn, t0 + budget_s * 0.95, rng, 50)

    # Phase 5: Ultra-fine random moves
    if time.time() - t0 < budget_s - 5:
        h = bh.copy()
        c5 = bc5
        et = t0 + budget_s - 3
        tv = np.array([0.0002, 0.0005, 0.001, 0.003, -0.0002, -0.0005, -0.001, -0.003])
        while time.time() < et:
            i = rng.randint(bn)
            j = rng.randint(bn)
            if i == j:
                continue
            d = tv[rng.randint(len(tv))]
            ni = h[i] + d
            nj = h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5_val(h, bn)
                if c < c5:
                    c5 = c; bc5 = c; bh = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    bh = _repair(bh, bn)
    bc5 = _c5_val(bh, bn)
    return (bh, bc5, bn)
