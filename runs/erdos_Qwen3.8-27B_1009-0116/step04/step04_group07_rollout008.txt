from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            space = 1.0 - h
            ss = space.sum()
            if ss < 1e-14: break
            h += d * space / ss
        else:
            space = h
            ss = space.sum()
            if ss < 1e-14: break
            h += d * space / ss
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    s = h.sum()
    d = target - s
    if abs(d) > 1e-13:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(d) < 1e-14: break
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
    return h

def smooth_grad(h, n, alpha):
    """Gradient of (1/alpha)*log(sum exp(alpha*corr_k)) w.r.t. h"""
    corr = get_corr(h, n)
    # Softmax weights
    ac = alpha * corr
    ac -= ac.max()
    w = np.exp(ac)
    w /= w.sum()
    # Gradient: sum_k w[k] * d(corr_k)/dh
    # corr[k] for lag = k-(n-1):
    # d(corr)/dh[i] = (1-h[i-L]) for i in [L, n-1] - h[i+L] for i in [0, n-1-L]
    # where L = |lag|
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        L = abs(lag)
        if L >= n:
            continue
        grad[L:] += w[k] * (1.0 - h[:n-L])
        grad[:n-L] -= w[k] * h[L:]
    return grad, corr

def nesterov_smooth(h, n, alpha_schedule, n_iter, t_end, rng):
    """Nesterov accelerated projected gradient on smooth objective"""
    h = h.copy()
    y = h.copy()
    mom = 0.9
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        alpha = alpha_schedule[min(it, len(alpha_schedule)-1)]
        grad_g, corr = smooth_grad(y, n, alpha)
        # Nesterov: step from y, then update y
        lr = 1.0 / (alpha + 1)
        h_new = h - lr * grad_g
        h_new = np.clip(h_new, 0, 1)
        h_new = repair(h_new, n)
        # Momentum update
        t_val = (1 + np.sqrt(1 + 4*mom*mom)) / 2
        y = h_new + mom * (h_new - h)
        y = np.clip(y, 0, 1)
        h = h_new
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def topk_subgrad(h, n, K, lr0, n_iter, t_end, rng):

    h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = lr0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = get_corr(h, n)
        top_k = np.argpartition(corr, -K)[-K:]
        # Weighted subgradient
        weights = np.zeros(len(corr))
        for i, k in enumerate(top_k):
            weights[k] = 1.0 / (i + 1)
        weights /= weights.sum()

        for k in top_k:

            L = abs(lag)
            if L >= n: continue
            grad[L:] += weights[k] * (1.0 - h[:n-L])
            grad[:n-L] -= weights[k] * h[L:]

        grad[:n-L] -= w[k] * h[L:]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            sp = 1.0 - h
            ss = sp.sum()
            if ss < 1e-14:
                break
            h += d * sp / ss
        else:
            ss = h.sum()
            if ss < 1e-14:
                break
            h += d * h / ss
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    d = target - s
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                amt = min(d, 1.0 - h[idx])
                h[idx] += amt
                d -= amt
            else:
                amt = min(-d, h[idx])
                h[idx] -= amt
                d += amt
    return h

def a_vector(h, n, i, j):
    nout = 2 * n - 1
    a = np.zeros(nout)
    for m in range(nout):
        k = m - (n - 1)
        v = 0.0
        ni = i - k
        if 0 <= ni < n:
            v += 1.0 - h[ni]
        ni2 = i + k
        if 0 <= ni2 < n:
            v -= h[ni2]
        nj = j - k
        if 0 <= nj < n:
            v -= 1.0 - h[nj]
        nj2 = j + k
        if 0 <= nj2 < n:
            v += h[nj2]
        a[m] = v
    return a

def bisection_line_search(h, n, i, j, corr):
    dlo = max(-h[i], h[j] - 1.0) + 1e-12
    dhi = min(1.0 - h[i], h[j]) - 1e-12
    if dhi - dlo < 1e-12:
        return 0.0
    av = a_vector(h, n, i, j)
    for _ in range(50):
        dm = 0.5 * (dlo + dhi)
        val = corr + dm * av
        am = av[np.argmax(val)]
        if am > 1e-12:
            dhi = dm
        elif am < -1e-12:
            dlo = dm
        else:
            break
    return 0.5 * (dlo + dhi)

def smooth_grad(h, n, alpha):
    corr = get_corr(h)
    ac = alpha * corr
    ac -= ac.max()
    exp_ac = np.exp(ac)
    Z = exp_ac.sum()
    w = exp_ac / Z
    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        L = abs(k)
        if L >= n:
            continue
        grad[L:] += w[m] * (1.0 - h[:n - L])
        grad[:n - L] -= w[m] * h[L:]
    return grad

def nesterov_smooth(h, n, t_end, rng):
    h = h.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    n_iter = 0
    t_start = time.time()
    while time.time() < t_end:
        n_iter += 1
        frac = min(1.0, n_iter / 3000.0)
        alpha = 50.0 + 5000.0 * frac
        lr = 1.0 / (alpha ** 0.5 + 1.0)
        grad = smooth_grad(h, n, alpha)
        v = 0.75 * v - lr * grad
        h_new = repair(h + v, n)
        h = h_new
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def topk_subgrad(h, n, K, t_end, rng):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = 0.05
    n_iter = 0
    while time.time() < t_end:
        n_iter += 1
        corr = get_corr(h)
        topk = np.argpartition(corr, -K)[-K:]
        weights = np.zeros(len(corr))
        for rank, idx in enumerate(topk):
            weights[idx] = 1.0 / (rank + 1)
        weights /= weights.sum()
        grad = np.zeros(n)
        for m in topk:
            k = m - (n - 1)
            L = abs(k)
            if L >= n:
                continue
            grad[L:] += weights[m] * (1.0 - h[:n - L])
            grad[:n - L] -= weights[m] * h[L:]
        h_new = repair(h - lr * grad, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.65
            if lr < 1e-8:
                break
    return h, best_c5

def coord_bisect(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    n_j_per_i = 4
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            js = [i] + list(rng.choice(np.delete(np.arange(n), i), size=n_j_per_i - 1, replace=False))
            for j in js:
                if j == i:
                    continue
                corr = get_corr(h)
                dm = bisection_line_search(h, n, i, j, corr)
                if abs(dm) < 1e-10:
                    continue
                ni = h[i] + dm
                nj = h[j] - dm
                if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                h = repair(h, n)
                c5_new = compute_c5(h, n)
                if c5_new < best - 1e-14:
                    best = c5_new
                    improved = True
                else:
                    h[i], h[j] = ni - dm, nj + dm
                    h = repair(h, n)
        if not improved:
            break
    return h, best

def try_other_n(best_h, best_n, best_c5, t_end, rng):
    for n_alt in [66, 68, 70, 72, 80]:
        if time.time() > t_end:
            break
        x_old = np.linspace(0.0, 1.0, best_n)
        x_new = np.linspace(0.0, 1.0, n_alt)
        h_seed = np.interp(x_new, x_old, best_h)
        h_seed = repair(h_seed, n_alt)
        h_a, c5_a = nesterov_smooth(h_seed, n_alt, t_end, rng)
        t_rem = min(t_end, time.time() + (t_end - time.time()) * 0.3)
        h_b, c5_b = coord_bisect(h_a, n_alt, min(t_rem, t_end), rng)
        if c5_b < best_c5:
            best_h, best_c5, best_n = h_b, c5_b, n_alt
    return best_h, best_c5, best_n

    t0 = time.time()
