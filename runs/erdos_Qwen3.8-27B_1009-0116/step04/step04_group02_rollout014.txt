def compute_c5(h, dx):

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def grad_active(h, n, active_k):
    """Compute subgradient of max over active lags w.r.t. h."""
    g = 1.0 - h
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    ii = np.arange(n)
    for k in active_k:
        # contrib to grad[i] from lag k:
        # (1-h[i+k-n+1]) - h[i-k+n-1] (valid indices)
        ig = ii + k - n + 1

        ih = ii - k + n - 1
        valid_h = (ih >= 0) & (ih < n)
        grad[valid_g] += 1.0 - h[ig[valid_g]]
        grad[valid_h] -= h[ih[valid_h]]
    return grad
