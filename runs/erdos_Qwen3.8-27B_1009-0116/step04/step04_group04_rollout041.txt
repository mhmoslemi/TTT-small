import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _grad_weighted(h, top_m, alpha):
    n = len(h)
    c = _corr_full(h)
    cshift = c - (n - 1)
    order = np.argsort(cshift)[::-1][:top_m]
    peak = cshift[order[0]]
    wts = np.exp(alpha * (cshift[order] - peak))
    wts /= wts.sum()
    grad = np.zeros(n)
    for w, li in zip(wts, order):
        k = int(li)
        if k > 0:
            grad[k:] += w * (1.0 - h[:-k])
            grad[:-k] += w * (-h[k:])
        elif k < 0:
            kk = -k
            grad[:-kk] += w * (1.0 - h[kk:])
            grad[kk:] += w * (-h[:-kk])
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad, float(peak) * (2.0 / n)

def _obj_and_grad(h, top_m, alpha, pen):
    n = len(h)
    tgt = n / 2.0
    grad_c, cval = _grad_weighted(h, top_m, alpha)
    f = cval + pen * (h.sum() - tgt) ** 2
    g = grad_c + 2.0 * pen * (h.sum() - tgt)
    return f, g

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

            def fpen(x):
                gc, cv = _grad_weighted(x, top_m, alpha)
                s = x.sum() - tgt
                return cv + p*s*s, gc + 2.0*p*s
            r = minimize(fpen, best_h, jac=True, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
            hc = _repair(r.x, tgt)
            c = _c5(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
        except Exception:

    return best_h, best_c

def _directed_local(h, dx, rng, t_end, n_iter):
    n = len(h)

    best = _c5(h, dx)
    h = h.copy()

    per_scale = max(50, n_iter // len(scales))

        for _ in range(per_scale):

            c = _corr_full(h)

            elif k < 0:
                pairs = np.arange(0, n + k)
            else:
                pairs = np.arange(n)
            if len(pairs) == 0:

            i = int(pairs[rng.randint(len(pairs))])
            if k >= 0:

            else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _grad_weighted(h, top_m, alpha):
    n = len(h)
    c = _corr_full(h)
    cshift = c
    peak_idx = int(np.argmax(cshift))
    peak = cshift[peak_idx]
    order = np.argsort(cshift)[::-1][:top_m]
    wts = np.exp(alpha * (cshift[order] - peak))
    wts /= wts.sum()
    grad = np.zeros(n)
    for w, li in zip(wts, order):
        k = int(li) - (n - 1)
        if k > 0:
            grad[k:] += w * (1.0 - h[:-k])
            grad[:-k] += w * (-h[k:])
        elif k < 0:
            kk = -k
            grad[:-kk] += w * (1.0 - h[kk:])
            grad[kk:] += w * (-h[:-kk])
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad, float(peak) * (2.0 / n)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _lbfgs_analytic(h, dx, top_m, alpha, pen_list, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best_h, best_c = h.copy(), _c5(h, dx)
    for pen in pen_list:
        def fun(x):
            gc, cv = _grad_weighted(x, top_m, alpha)
            s = x.sum() - tgt
            return cv + pen * s * s, gc + 2.0 * pen * s
        try:
            r = minimize(fun, best_h, jac=True, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            hc = _repair(r.x, tgt)
            c = _c5(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
        except Exception:
            pass
    return best_h, best_c

def _slsqp_refine(h, dx, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(lambda x: _c5(x, dx), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _directed_local(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    per_scale = max(20, n_iter // len(scales))
    for sc in scales:
        for _ in range(per_scale):
            if time.time() > t_end:
                return h, best
            c = _corr_full(h)
            cshift = c - (n - 1)
            abs_c = np.abs(cshift[:n])
            top = np.argsort(abs_c)[::-1][:12]
            k = int(top[rng.randint(len(top))])
            k = abs(k)
            if k == 0:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
            else:
                if k >= n:
                    k = n - 1
                i = rng.randint(n - k)
                j = i + k
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d
            h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                c_new = _c5(h, dx)
                if c_new < best:
                    best = c_new
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
        h = _repair(h, tgt)
    return h, best

def _pair_search(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        cshift = c - (n - 1)
        peak = float(cshift.max()) * (2.0 / n)
        if peak < best - 1e-12:
            break
        top = np.argsort(cshift)[::-1][:6]
        k = int(top[rng.randint(len(top))]) - (n - 1)
        k = abs(k)
        if k == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        else:
            k = min(k, n - 1)
            i = rng.randint(n - k)
            j = i + k
        magnitude = h[i] * (1.0 - h[j]) if h[i] > h[j] else h[j] * (1.0 - h[i])
        sc = 0.005 * min(1.0, magnitude + 0.1)
        d = sc * (2.0 * rng.random() - 1.0)
        h[i] += d
        h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c_new = _c5(h, dx)
            if c_new < best:
                best = c_new
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
        if it % 500 == 0:
            h = _repair(h, tgt)
    h = _repair(h, tgt)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
