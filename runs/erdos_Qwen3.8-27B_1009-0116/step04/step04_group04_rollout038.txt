from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = np.maximum(0, 1.0 - h)
        else:
            room = np.maximum(0, h)
        if room.sum() < 1e-15:
            break
        h += (d / room.sum()) * room
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _subgrad(h, top_k=15):
    n = len(h)
    c = _corr_full(h)
    cmax = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    # Exponential weighting
    w = np.exp(100.0 * (c[idx] - cmax) / max(cmax, 1e-10))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        s = ki - (n - 1)
        if s >= 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] += wi * (-h[s:])
        else:
            ss = -s
            grad[:n - ss] += wi * (1.0 - h[ss:])
            grad[ss:] += wi * (-h[:n - ss])
    return grad, cmax

def _sgd(h, dx, t_end, n_iter_max=5000):
    n = len(h)
    tgt = n / 2.0
    best_c = _c5_val(h, dx)
    h = h.copy()
    cmax = best_c / dx
    lr = 0.01
    for it in range(n_iter_max):
        if time.time() > t_end:
            break
        grad, cmax = _subgrad(h, 15)
        h_new = h - lr * grad
        h_new = _repair(h_new, tgt)
        c_new = _c5_val(h_new, dx)
        if c_new < best_c:

            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.9
        if lr < 1e-6:

    lo, hi = -max(h.max(), 1) - 1, max(h.min(), 0) + 1
    for _ in range(100):

        p = np.clip(h - mid, 0, 1)
        s = p.sum()
        if s > target:
            lo = mid
        else:

        if -1e-9 <= h[i] <= 1.0 + 1e-9 and -1e-9 <= h[j] <= 1.0 + 1e-9:
            h = _repair(h, tgt)
            c_new = _c5(h, dx)
            if c_new < best - 1e-15:
                best = c_new
                c = _corr(h)
                active = np.argsort(c)[::-1][:40]
                stall = 0
            else:
                h[i] -= d
                h[j] += d
                h = _repair(h, tgt)
                stall += 1
        else:
            h[i] -= d
            h[j] += d
        if stall > 800:
            si = min(si + 1, len(scales) - 1)
            stall = 0
            if si >= len(scales) - 1:
                break
    return h, best

def _interp(h, n_new):
    if n_new == len(h):
        return h.copy()
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = _c5(h0, dx0)
    best_n = n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient at n0 (100s)
    t1 = t0 + 100
    h1, c1 = _sgd_opt(best_h, dx0, t1, 15000)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Top-k SLSQP at n0 (120s)
    t2 = t0 + 220
    for k in [5, 8, 15, 25]:
        if time.time() > t2 or rem() < 350:
            break
        hp, cp = _slsqp_topk(best_h, dx0, k, 400)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 3: LSE SLSQP sweep at n0 (120s)
    t3 = t0 + 340
    for a in [500, 2000, 10000, 50000, 200000]:
        if time.time() > t3 or rem() < 300:
            break
        hp, cp = _slsqp_lse(best_h, dx0, a, 450)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 4: Local search at n0 (100s)
    t4 = t0 + 440
    best_h, best_c5 = _local_search(best_h, dx0, rng, t4, 150000)
    best_n = n0

    # Phase 5: Multi-start perturbation at n0 (100s)
    t5 = t0 + 540
    for r in range(6):
        if time.time() > t5 or rem() < 300:
            break
        hp = best_h + 0.04 * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, n0 / 2.0)
        hp, cp = _slsqp_lse(hp, dx0, 5000, 300)
        hp, cp = _slsqp_topk(hp, dx0, 8, 300)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 6: Higher resolution 2*n0 (200s)
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h_r = _repair(_interp(best_h, n1), n1 / 2.0)
    t6 = min(t0 + 750, t0 + budget_s - 200)
    if rem() > 200:
        t6a = min(t6, time.time() + 60)
        h_r, c_r = _sgd_opt(h_r, dx1, t6a, 6000)
        for k in [5, 10]:
            if time.time() > t6 or rem() < 120:
                break
            h_r, c_r = _slsqp_topk(h_r, dx1, k, 300)
        for a in [5000, 30000, 100000]:
            if time.time() > t6 or rem() < 120:
                break
            h_r, c_r = _slsqp_lse(h_r, dx1, a, 350)
        h_r, c_r = _local_search(h_r, dx1, rng, min(t6, time.time() + 40), 60000)
        if c_r < best_c5:
            best_h, best_c5, best_n = h_r, c_r, n1

    # Phase 7: Final SLSQP at best_n (120s)
    dxb = 2.0 / best_n
    if best_n != n0 and best_n != n1:
        best_h = _repair(_interp(best_h, best_n), best_n / 2.0)
    t7 = t0 + budget_s - 80
    for a in [100000, 300000, 800000]:
        if time.time() > t7:
            break
        hp, cp = _slsqp_lse(best_h, dxb, a, 500)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 8: Final local search (60s)
    t8 = t0 + budget_s - 10
    best_h, best_c5 = _local_search(best_h, dxb, rng, t8, 250000)

    # Phase 9: Tiny final polish
    if rem() > 12:
        best_h, best_c5 = _slsqp_topk(best_h, dxb, 5, 200)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
