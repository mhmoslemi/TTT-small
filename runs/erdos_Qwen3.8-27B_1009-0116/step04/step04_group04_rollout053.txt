def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

    return 0

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    n = len(h)
    # Quick fix
    d = target - h.sum()
    if abs(d) > 1e-15:
        # Distribute to non-saturated elements
        if d > 0:
            room = 1.0 - h
            order = np.argsort(-room)  # most room first
            for i in order:
                a = min(d, room[i])
                h[i] += a
                d -= a
                if abs(d) < 1e-14:

        else:
            room = h
            order = np.argsort(-room)  # most to remove first
            for i in order:

def _interp(h, n_new):

    """Compute soft-max approximation to max(c)"""
    cmax = c.max()
    shifted = np.clip(alpha * (c - cmax), -50, 50)
    return cmax + np.log(np.exp(shifted).sum()) / alpha

        ht = _repair(ht, target)
        c5 = _c5(ht, dx)
        if c5 < bc:

    except Exception:
        pass
    return h, bc
