def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)

    w = np.exp(s - s.max())

    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - int(k)
        wk = w[k] * dx
        if sh >= 0:

    return f, g

                h, f = h_new, f_new

            step *= 0.5
        else:

        h_prev = h
        step = min(step * 1.4, 0.05)
        mu = min(mu * 1.01, 0.9)
    return h, f

def _upsample_rep(h, factor):
    return np.repeat(h, factor)

def _active_lag_descent(h, n_rounds=200):
    n = len(h)
    c5 = _eval(h)
    dx = 2.0 / n
    for _ in range(n_rounds):
        c = np.correlate(h, 1.0 - h, mode='full')
        k_w = int(c.argmax())
        sh = (n - 1) - k_w
        g = np.zeros(n)
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += (1.0 - h[sh:sh+cnt])

                c5 = c5n

    return h, c5

    rng = np.random.RandomState(999)

            if c5n < c5 - 1e-15:

            else:

    return h, c5
