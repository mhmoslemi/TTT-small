        if 0 <= j + l < n:
            g += 1.0 - h[j + l]
        if 0 <= j - l < n:
            g -= h[j - l]
        grad[j] = g
    return grad

def directed_search(h, n, dx, rng, deadline, top_m=5):
    """Directed local search using subgradient of top lags"""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    stuck = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Get top M lags
        top_lags = np.argsort(corr)[-top_m:]  # indices in numpy correlation
        for lag_idx in top_lags[np.random.permutation(top_m)]:
            if time.time() >= deadline:
                break
            lag = lag_idx - (n - 1)  # actual lag
            if lag == 0:
                continue
            grad = grad_for_lag(h, n, lag)
            # Find best source (positive grad) and best target (negative grad)
            pos = np.where(grad > 0.05)[0]
            neg = np.where(grad < -0.05)[0]
            if len(pos) == 0 or len(neg) == 0:
                continue
            # Pick from top few
            pos_top = pos[np.argsort(grad[pos])[-min(5, len(pos)):]]
            neg_top = neg[np.argsort(grad[neg])[:min(5, len(neg))]]
            for i in pos_top:
                for j in neg_top:
                    if i == j:
                        continue
                    d = 0.003 * rng.uniform(0.5, 1.5)
                    for sign in [1, -1]:
                        ni, nj = h[i] + sign*d, h[j] - sign*d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c_new = compute_c5(h, dx)
                            if c_new < c5 - 1e-15:
                                c5 = c_new
                                break
                            h[i], h[j] = ni - sign*d, nj + sign*d  # revert
        # Occasionally do random moves
        if stuck > 50 and rng.random() < 0.3:
            i, j = rng.randint(n, 2)
            if i != j:
                d = 0.002 * (2*rng.random()-1)
                ni, nj = h[i]+d, h[j]-d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    if compute_c5(h, dx) < best_c5:
                        s = 50
    return best_h, best_c5

def fine_local(h, n, dx, rng, deadline):
    """Ultra-fine random pair search"""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()

                best_h = h.copy()
            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def slsqp_f(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_g(x, target):
    return np.sum(x) - target

def lbfgs(h, n, alpha, pen, maxiter=300):
    dx = 2.0 / n
    try:
        r = minimize(lse_obj, h, args=(alpha, n/2.0, pen), method='L-BFGS-B',
                    bounds=[(0.0,1.0)]*n, options={'maxiter':maxiter,'ftol':1e-16,'gtol':1e-8})
        hc = repair(r.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    try:
        r = minimize(slsqp_f, h, args=(alpha,), method='SLSQP',
                    bounds=[(0.0,1.0)]*n,
                    constraints=[{'type':'eq','fun':slsqp_g,'args':(n/2.0,)}],
                    options={'maxiter':maxiter,'ftol':1e-15})
        hc = repair(r.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def grad_lag(h, n, lag_idx):
    l = lag_idx - (n - 1)
    if l == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    if l > 0:
        grad[:n-l] += 1.0 - h[l:]
        grad[l:] -= h[:n-l]
    else:
        l = -l
        grad[l:] += 1.0 - h[:-l]
        grad[:n-l] -= h[l:]
    return grad

def directed_search(h, n, dx, rng, deadline, top_m=4):

    h = h.copy()

            l = li - (n-1)
