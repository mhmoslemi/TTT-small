import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                adj = min(d, 1.0 - h[i])
            else:
                adj = max(d, -h[i])
            h[i] += adj
            d -= adj
    return h

def c5val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def lag_grad(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

_G = {}

def sm_obj(x):
    n = _G['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    a = _G['a']
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + _G['lam'] * (x.sum() - n * 0.5) ** 2

def sm_jac(x):
    n = _G['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    a = _G['a']
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-15:
            g += pw[k] * lag_grad(x, k, n)
    g /= a
    g += 2.0 * _G['lam'] * (x.sum() - n * 0.5)
    return g

def softmax_phase(h, n, dx, t0, lim):
    _G['n'] = n
    bds = [(0.0, 1.0)] * n
    best, bestc = h.copy(), c5val(h, dx)
    seq = [(15, 200, 120), (60, 500, 100), (300, 1000, 80), (1500, 2000, 70),
           (6000, 4000, 55), (25000, 8000, 45), (100000, 20000, 40),
           (400000, 50000, 30), (1500000, 100000, 25), (5000000, 200000, 20)]
    for a, lam, mi in seq:
        if time.time() - t0 > lim:
            break
        _G['a'], _G['lam'] = float(a), float(lam)
        try:
            r = minimize(sm_obj, best.copy(), jac=sm_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': mi, 'ftol': 1e-19, 'gtol': 1e-16})
            hc = repair(r.x, n)
            c = c5val(hc, dx)
            if c < bestc:
                bestc, best = c, hc
        except Exception:
            pass
    return best, bestc

def active_phase(h, n, dx, t0, lim):
    best, bestc = h.copy(), c5val(h, dx)
    for mt in [2, 3, 5, 8, 12, 18, 25, 40]:
        hx = best.copy()
        v = np.zeros(n)
        step = 0.06
        stalled = 0
        for it in range(1200):
            if time.time() - t0 > lim:
                break
            c = np.correlate(hx, 1.0 - hx, mode='full')
            cm = c.max()
            thr = cm - 0.003 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) == 0:
                idx = np.array([np.argmax(c)])
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += lag_grad(hx, ki, n)
            g /= len(idx)
            g -= g.mean()
            v = 0.75 * v + g
            ok = False
            cur_c = c5val(hx, dx)
            for _ in range(35):
                hn = repair(hx - step * v, n)
                cn = c5val(hn, dx)
                if cn < cur_c - 1e-18:
                    hx = hn
                    cur_c = cn
                    if cn < bestc:
                        bestc, best = cn, hx.copy()
                    step = min(step * 1.13, 2.0)
                    stalled = 0
                    ok = True
                    break
                step *= 0.72
            if not ok:
                v *= 0.5
                stalled += 1
                if stalled > 12:
                    break
    return best, bestc

def coord_descent(h, n, dx, rng, t0, lim):
    best, bestc = h.copy(), c5val(h, dx)
    cc = bestc
    for outer in range(400):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(3000):
            if time.time() - t0 > lim:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = 0.008 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = c5val(h, dx)
                if c < cc - 1e-16:
                    h = repair(h, n)
                    cc = c5val(h, dx)
                    imp = True
                    if cc < bestc:
                        bestc, best = cc, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        if not imp and outer > 30:
            break
    return best, bestc

def fine_polish(h, n, dx, rng, t0, lim):
    best, bestc = h.copy(), c5val(h, dx)
    cc = bestc
    # Deterministic coordinate sweep
    for sweep in range(30):
        if time.time() - t0 > lim:
            break
        for i in range(n):
            if time.time() - t0 > lim:
                break
            orig = h[i]
            improved = False
            for d in [3e-4, -3e-4, 1e-3, -1e-3, 3e-3, -3e-3, 8e-3, -8e-3]:
                h[i] = orig + d
                if 0 <= h[i] <= 1:
                    c = c5val(h, dx)
                    if c < cc - 1e-16:
                        cc = c
                        if cc < bestc:
                            bestc, best = cc, h.copy()
                        improved = True
                        break
                h[i] = orig
            if not improved:
                h[i] = orig
    # Random pair polish
    for _ in range(80000):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.0015 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5val(h, dx)
            if c < cc - 1e-16:
                h = repair(h, n)
                cc = c5val(h, dx)
                if cc < bestc:
                    bestc, best = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, bestc

def pipeline(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    best, bestc = h.copy(), c5val(h, dx)
    frac = max((lim - (time.time() - t0)) / max(lim, 1), 0.1)
    t1 = t0 + (lim - t0) * 0.30
    r, c = softmax_phase(best, n, dx, t0, t1)
    if c < bestc:
        best, bestc = r, c
    t2 = t0 + (lim - t0) * 0.52
    r, c = active_phase(best, n, dx, t0, t2)
    if c < bestc:
        best, bestc = r, c
    t3 = t0 + (lim - t0) * 0.70
    r, c = softmax_phase(best, n, dx, t0, t3)
    if c < bestc:
        best, bestc = r, c
    t4 = t0 + (lim - t0) * 0.82
    r, c = active_phase(best, n, dx, t0, t4)
    if c < bestc:

    t5 = t0 + (lim - t0) * 0.92
    r, c = coord_descent(best, n, dx, rng, t0, t5)
    if c < bestc:
        best, bestc = r, c
    r, c = fine_polish(best, n, dx, rng, t0, lim - 1)
    if c < bestc:
        best, bestc = r, c
