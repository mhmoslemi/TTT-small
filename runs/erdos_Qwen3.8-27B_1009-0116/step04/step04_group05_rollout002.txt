def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0

    r, c = opt_h(h0, n0, dx0, seed, t0, budget_s - 300)
    if c < bc: bh, bc, bn = r, c, n0

    perturb_scales = [0.04, 0.08, 0.12, 0.06]
    for pi, sc in enumerate(perturb_scales):
        if time.time() - t0 > budget_s - 180:
            break
        rng = np.random.RandomState(seed + 100 + pi * 7)
        hp = repair(bh + sc * rng.randn(bn), bn)
        rem = budget_s - (time.time() - t0) - 80
        r, c = opt_h(hp, bn, 2.0 / bn, seed + 50 + pi, t0, budget_s - 80)
        if c < bc:
            bh, bc, bn = r, c, bn

    for mult in [2, 3]:
        n_new = bn * mult
        if n_new >= 1000 or time.time() - t0 > budget_s - 60:
            break
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, n_new, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n_new)
        d2 = 2.0 / n_new
        r, c = opt_h(h2, n_new, d2, seed + 200 + mult, t0, budget_s - 20)
        if c < bc:
            bh, bc, bn = r, c, n_new

    if time.time() - t0 < budget_s - 15:
        rng2 = np.random.RandomState(seed + 999)
        hp = repair(bh + 0.02 * rng2.randn(bn), bn)
        r, c = opt_h(hp, bn, 2.0 / bn, seed + 950, t0, budget_s - 3)
        if c < bc:
            bh, bc, bn = r, c, bn

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
