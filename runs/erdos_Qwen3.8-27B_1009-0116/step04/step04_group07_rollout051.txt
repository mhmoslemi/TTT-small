    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

    corr = np.correlate(v, 1.0 - v, mode='full') * (2.0 / n)
    idx = np.argpartition(corr, -K)[-K:]
    return np.sum(corr[idx])

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return repair(res.x, n)

                        improved = True
                        break
            if not improved:
                break
    return h, best_c5

def random_walk_polish(h, n, t_end, rng, n_moves=5000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    T = 0.0005
    for _ in range(n_moves):
        if time.time() > t_end:
            break
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        delta = (rng.rand() * 2 - 1) * 0.008
        hi, hj = h[i], h[j]
        ni, nj = hi + delta, hj - delta
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_new = compute_c5(h, n)
        if c5_new - c5 < 0 or rng.rand() < np.exp(-(c5_new - c5) / T):
            pass
        else:
            h[i], h[j] = hi, hj
        T *= 0.9998
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init)
    except NameError:
        init = np.full(64, 0.5)
        init_n = 64
    init = repair(init, init_n)
    best_h = init.copy()
    best_c5 = compute_c5(init, init_n)
    best_n = init_n

    # Phase 1: L-BFGS-B with escalating alpha (35% budget)
    t1 = t0 + budget_s * 0.35
    h_work = best_h.copy()
    for alpha in [3000, 10000, 50000, 200000, 1000000]:
        if time.time() > t1:
            break
        h_new = optimize_lbfgs(h_work, best_n, alpha, 10000, 1500, t1)
        c5 = compute_c5(h_new, best_n)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5 = h_new.copy(), c5
        if c5 < compute_c5(h_work, best_n) - 1e-14:
            h_work = h_new

    # Phase 2: SLSQP refinement (10% budget)
    t2 = t0 + budget_s * 0.45
    if time.time() < t2:
        for alpha in [100000, 500000]:
            if time.time() > t2:
                break
            h_new = optimize_slsqp(best_h.copy(), best_n, alpha, 400, t2)
            c5 = compute_c5(h_new, best_n)
            if c5 < best_c5 - 1e-14:
                best_h, best_c5 = h_new.copy(), c5

    # Phase 3: Systematic coordinate descent (25% budget)
    t3 = t0 + budget_s * 0.70
    h_cd, c5_cd = coord_descent_systematic(best_h.copy(), best_n, t3, rng, n_passes=30)
    if c5_cd < best_c5 - 1e-14:
        best_h, best_c5 = h_cd.copy(), c5_cd

    # Phase 4: Random walk polish (10% budget)
    t4 = t0 + budget_s * 0.80
    h_rw, c5_rw = random_walk_polish(best_h.copy(), best_n, t4, rng, n_moves=20000)
    if c5_rw < best_c5 - 1e-14:
        best_h, best_c5 = h_rw.copy(), c5_rw

    # Phase 5: Try other n values (12% budget)
    t5 = t0 + budget_s * 0.92
    if time.time() < t5:
        for n_alt in [60, 68, 70, 76, 80]:
            if time.time() > t5:
                break
            if n_alt == best_n:
                continue
            x_old = np.linspace(0.0, 1.0, best_n)
            x_new = np.linspace(0.0, 1.0, n_alt)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_alt = optimize_lbfgs(h_seed, n_alt, 200000, 10000, 1000, t5)
            c5_alt = compute_c5(h_alt, n_alt)
            h_alt2, c5_alt2 = coord_descent_systematic(h_alt, n_alt, t5, rng, n_passes=10)
            if c5_alt2 < c5_alt:
                h_alt, c5_alt = h_alt2, c5_alt2
            if c5_alt < best_c5 - 1e-14:
                best_h, best_c5, best_n = h_alt.copy(), c5_alt, n_alt

    # Phase 6: Final ultra-fine polish (3% budget)
    t6 = t0 + budget_s * 0.985
    if time.time() < t6:
        h_fin, c5_fin = coord_descent_systematic(best_h.copy(), best_n, t6, rng, n_passes=15)
        if c5_fin < best_c5 - 1e-14:
            best_h, best_c5 = h_fin.copy(), c5_fin
        h_fin2, c5_fin2 = random_walk_polish(best_h.copy(), best_n, t6, rng, n_moves=10000)
        if c5_fin2 < best_c5 - 1e-14:
            best_h, best_c5 = h_fin2.copy(), c5_fin2

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
