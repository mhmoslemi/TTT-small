    h = h.copy()

    best_c5 = compute_c5(h, n)
    step = 0.1
    cooldown = 0

        c5 = np.max(crr)

        grad = np.zeros(n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:

            if d > 0:

                h[idx] += a

    for alpha in [3000, 15000, 80000, 400000]:
        if time.time() > t1:
            break
        h_new = slsqp_pass(h.copy(), n, alpha, 800)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        elif c5_new < compute_c5(h, n) - 1e-15:
            h = h_new

    # Phase 2: Subgradient descent with various K
    t2 = t0 + budget_s * 0.32
    for K in [1, 3, 5, 8, 12]:
        if time.time() > t2:
            break
        h_new, c5_new = subgrad_descent(h, n, K, 50000, t2)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new

    # Phase 3: SLSQP again
    t3 = t0 + budget_s * 0.42
    if time.time() < t3:
        h_new = slsqp_pass(h.copy(), n, 2000000, 3000)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new

    # Phase 4: Fine coordinate descent (long)
    t4 = t0 + budget_s * 0.75
    h_cd, c5_cd = fine_cd(h, n, t4, 500000, rng)
    if c5_cd < best_c5 - 1e-15:
        best_h = h_cd.copy()
        best_c5 = c5_cd
        h = h_cd

    # Phase 5: Block moves
    t5 = t0 + budget_s * 0.85
    h_blk, c5_blk = block_cd(h, n, t5, rng)
    if c5_blk < best_c5 - 1e-15:
        best_h = h_blk.copy()
        best_c5 = c5_blk
        h = h_blk

    # Phase 6: Subgradient again on best
    if time.time() < t0 + budget_s * 0.90:
        h_new, c5_new = subgrad_descent(best_h, best_n, 3, 30000, t0 + budget_s * 0.90)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 7: Try other resolutions
    t7 = t0 + budget_s * 0.97
    for n_alt in [best_n - 4, best_n + 4, best_n + 8, best_n - 8, best_n + 16, best_n - 16]:
        if time.time() > t7:
            break
        if n_alt < 24 or n_alt > 200:
            continue
        x_old = np.linspace(0.0, 1.0, best_n)
        x_new = np.linspace(0.0, 1.0, n_alt)
        h_seed = np.interp(x_new, x_old, best_h)
        h_seed = repair(h_seed, n_alt)
        h_alt = slsqp_pass(h_seed, n_alt, 200000, 600)
        c5_alt = compute_c5(h_alt, n_alt)
        h_alt, c5_alt2 = fine_cd(h_alt, n_alt, t7, 100000, rng)
        c5_alt = min(c5_alt, c5_alt2)
        if c5_alt < best_c5 - 1e-15:
            best_h = h_alt.copy()
            best_c5 = c5_alt
            best_n = n_alt

    # Phase 8: Final ultra-fine polish
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = fine_cd(best_h, best_n, t0 + budget_s * 0.995, 200000, rng)
        if c5_fin < best_c5 - 1e-15:
            best_h = h_fin.copy()
            best_c5 = c5_fin

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
