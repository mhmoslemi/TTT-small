def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

        g[:n - M] += 1.0

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _grad_lag(h, n, L):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0
        g[L:] -= h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += 1.0
        g[:n - M] -= h[M:]
        g[M:] -= h[:n - M]
    return g

def _smooth_obj(h, n, alpha):
    c = _corr(h) * (2.0 / n)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -400, 400)))) / alpha

def _smooth_grad(h, n, alpha):
    dxv = 2.0 / n
    c = _corr(h) * dxv
    m = np.max(c)
    w = np.exp(np.clip(alpha * (c - m), -400, 400))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(h, n, k - (n - 1))
    return g * dxv

def _pgd_smooth(h, n, alpha, t_end, lr):
    h = h.copy()
    c5 = _c5(h, n)
    for _ in range(30000):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, n, alpha)
        for _bs in range(25):
            h_new = _project(h - lr * g, n)
            c5_new = _c5(h_new, n)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                lr *= 1.01
                break
            lr *= 0.65
    return h, c5

def _active_cycle(h, n, t_end, top_k):
    h = h.copy()
    c5 = _c5(h, n)
    lr = 0.03
    stall = 0
    while time.time() < t_end and stall < 80:
        c = _corr(h) * (2.0 / n)
        idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, n, k - (n - 1))
        g /= len(idx)
        improved = False
        for _bs in range(30):
            h_new = _project(h - lr * g, n)
            c5_new = _c5(h_new, n)
            if c5_new < c5 - 1e-18:
                h = h_new
                c5 = c5_new
                lr *= 1.03
                improved = True
                stall = 0
                break
            lr *= 0.7
        if not improved:
            stall += 1
            lr = 0.03
    return h, c5

def _slsqp_run(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(x.sum() - n / 2.0),
             'jac': lambda x: np.ones(n)}]
    return minimize(_smooth_obj, h_init, args=(n, alpha),
                    jac=_smooth_grad, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-16})

def _pair_search(h, n, t_end, rng):
    h = h.copy()
    c5 = _c5(h, n)
    deltas = [0.0005, 0.001, 0.003, 0.01, 0.03, -0.0005, -0.001, -0.003, -0.01, -0.03]
    idx_all = np.arange(n)
    no_imp = 0
    while time.time() < t_end and no_imp < 20:
        no_imp += 1
        idx = idx_all.copy()
        rng.shuffle(idx)
        improved = False
        for i in idx:
            if time.time() > t_end:
                break
            hi = h[i]
            for j in idx:
                if j == i:
                    continue
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5 - 1e-16:
                        c5 = c
                        hi, hj = ni, nj
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5

def _fine_search(h, n, t_end, rng):
    h = h.copy()
    c5 = _c5(h, n)
    tiny = [0.0001, 0.0002, 0.0005, 0.001, -0.0001, -0.0002, -0.0005, -0.001]
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5:
                c5 = c
            else:
                h[i], h[j] = ni, nj
    return h, c5

def _try_resolution(h_best, n_best, c5_best, n_try, t_end, rng):
    x_old = np.linspace(0, 2.0, n_best, endpoint=False)
    x_new = np.linspace(0, 2.0, n_try, endpoint=False)
    h_new = np.interp(x_new, x_old, h_best, period=2.0)
    h_new = _repair(h_new, n_try)
    h_new, c5_w = _active_cycle(h_new, n_try, t_end, 5)
    if c5_w < c5_best:
        return h_new, c5_w, n_try
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t1 = t0 + budget_s * 0.12
    t2 = t0 + budget_s * 0.30
    t3 = t0 + budget_s * 0.52
    t4 = t0 + budget_s * 0.82
    t5 = t0 + budget_s * 0.95

    n_init = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n_init)
    best_h = h0.copy()
    best_n = n_init
    best_c5 = _c5(best_h, best_n)

    # Phase 1: PGD smooth max
    for alpha in [2000, 8000, 30000, 100000]:
        if time.time() > t1:
            break
        tw = min(time.time() + 0.05, t1)
        h_w, c5_w = _pgd_smooth(best_h, best_n, alpha, tw, 0.04)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w

    # Phase 2: Active-set cycling
    for tk in [3, 5, 8]:
        if time.time() > t2:
            break
        tw = min(time.time() + 0.05, t2)
        h_w, c5_w = _active_cycle(best_h, best_n, tw, tk)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w

    # Phase 3: SLSQP polishing with restarts
    if time.time() < t3:
        for alpha in [50000, 150000]:
            for r in range(4):
                if time.time() > t3:
                    break
                if r == 0:
                    h_init = best_h.copy()
                else:
                    pert = 0.015 * rng.randn(best_n)
                    h_init = _repair(best_h + pert, best_n)
                try:
                    res = _slsqp_run(h_init, best_n, alpha, 500)
                    hc = _repair(res.x, best_n)
                    c5 = _c5(hc, best_n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc
                except Exception:
                    pass
            # Active cycle after SLSQP
            if time.time() < t3 - 5:
                h_w, c5_w = _active_cycle(best_h, best_n, t3, 3)
                if c5_w < best_c5:
                    best_c5 = c5_w
                    best_h = h_w

    # Phase 4: Try different resolutions
    if time.time() < t4 - 60:
        for n_try in [100, 160, 200]:
            if time.time() > t4 - 60:
                break
            if n_try == best_n:

            t_try = min(time.time() + budget_s * 0.05, t4 - 60)
            res = _try_resolution(best_h, best_n, best_c5, n_try, t_try, rng)
            if res is not None and res[1] < best_c5 - 1e-7:

    # Phase 5: Pair search
    if time.time() < t4:
        best_h, best_c5 = _pair_search(best_h, best_n, t4, rng)

    # Phase 6: Fine search
    if time.time() < t5:

    # Final active cycle squeeze

        h_w, c5_w = _active_cycle(best_h, best_n, t0 + budget_s - 5, 3)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
