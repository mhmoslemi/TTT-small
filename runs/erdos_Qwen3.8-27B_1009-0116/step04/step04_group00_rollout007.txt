    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n-L] += pk * (1.0 - h[L:])
            g[L:] -= pk * h[:n-L]
        else:
            M = -L
            g[M:] += pk * (1.0 - h[:n-M])
            g[:n-M] -= pk * h[M:]
    g *= 1.0 / (alpha * n)
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def _obj_soft(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _proj_momentum(h, n, dx, iters, time_fn, frac=0.25):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    v = np.zeros(n)
    step = 0.02
    mu = 0.9
    for it in range(iters):
        if time_fn() < 20:
            break
        g, cmat = _grad_max(h, n, frac)
        g_c = g - g.mean()
        vn = mu * v + 0.1 / np.sqrt(1.0 + 0.9 * (1.0 - mu) * it) * g_c
        vg = np.linalg.norm(vn)
        if vg < 1e-14:
            break
        vn /= vg
        for s_try in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_new = _repair(h - s_try * vn, n)
            c_new = _c5(h_new, dx)
            if c_new < bc5 - 1e-18:
                h = h_new
                v = vn
                bc5 = c_new
                bh = h.copy()
                break
            elif c_new <= bc5 + 1e-14:
                h = h_new
                v = vn
                break
        else:
            step *= 0.7
            v *= 0.5
            if step < 1e-8:
                break
        if it > 20:
            step = min(step * 1.02, 0.05)
    return bh, bc5

def _pair_descent(h, n, dx, iters, time_fn):
    bh = h.copy()
    c5c = _c5(bh, dx)
    cs = _corr(bh)
    cm = cm_cast = np.max(cs)
    # Focus on indices with high gradient contribution
    dlist = np.array([0.001, 0.003, 0.008, 0.015, 0.03,
                      -0.001, -0.003, -0.008, -0.015, -0.03])
    for it in range(iters):
        if time_fn() < 15:
            break
        cs = _corr(h)
        g, _ = _grad_max(h, n, 0.3)
        order = np.argsort(np.abs(g))[::-1][:min(60, n)]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 15, len(order))):
                j = order[jj]
                best_d = 0.0
                best_c = c5c
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < cm_cast:
                        cm_cast = c5c
                        bh = h.copy()
                    break
            if improved and (it % 5 == 0):
                improved = False
        if not improved:
            break
    return bh, cm_cast

def _micro_refine(h, n, dx, iters, time_fn, rng):
    bh = h.copy()
    c5c = _c5(bh, dx)
    for _ in range(iters):
        if time_fn() < 5:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.003, 0.003)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, c5c

def _optimize_routine(h0, n, dx, time_fn, phase_budget):
    bh = h0.copy()
    bc5 = _c5(bh, dx)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    # L-BFGS-B with soft max
    for a in [5000, 50000, 500000]:
        if time_fn() < phase_budget * 0.3:
            break
        try:
            res = minimize(
                lambda x, al=a: _obj_soft(x, al, 100.0), bh.copy(),
                jac=lambda x, al=a: _grad_soft(x, al, 100.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass

    # Momentum projected gradient
    if time_fn() > phase_budget * 0.4:
        bh, bc5 = _proj_momentum(bh, n, dx, 500, time_fn, frac=0.2)
    if time_fn() > phase_budget * 0.3:
        bh, bc5 = _proj_momentum(bh, n, dx, 300, time_fn, frac=0.4)

    # Pair descent
    if time_fn() > phase_budget * 0.2:
        bh, bc5 = _pair_descent(bh, n, dx, 300, time_fn)

    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left(frac=0.0):
        return budget_s * (1.0 - frac) - elapsed()

    # Start from parent
    n_parent = len(initial_h_values)
    dx_p = 2.0 / n_parent
    h_p = _repair(np.array(initial_h_values, dtype=float), n_parent)
    best_h = h_p.copy()
    best_c5 = _c5(best_h, dx_p)

    # Phase 1: Optimize parent resolution (n=140)
    if time_left(0.5) > 60:
        phase_end = elapsed() + min((budget_s * 0.35), time_left(0.5) - 30)
        def tf1():
            return phase_end - elapsed()
        h1, c1 = _optimize_routine(h_p, n_parent, dx_p, tf1, budget_s * 0.35)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Try n=100 from interpolation
    if time_left(0.55) > 80:
        n2 = 100
        dx2 = 2.0 / n2
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n_parent, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        phase_end2 = elapsed() + min((budget_s * 0.2), time_left(0.55) - 20)
        def tf2():
            return phase_end2 - elapsed()
        h2r, c2 = _optimize_routine(h2, n2, dx2, tf2, budget_s * 0.2)
        if c2 < best_c5:
            best_h, best_c5, n_parent, dx_p = h2r, c2, n2, dx2

    # Phase 3: Try n=120
    if time_left(0.65) > 60:
        n3 = 120
        dx3 = 2.0 / n3
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n_parent, endpoint=False), best_h)
        h3 = _repair(h3, n3)
        phase_end3 = elapsed() + min((budget_s * 0.15), time_left(0.65) - 15)
        def tf3():
            return phase_end3 - elapsed()
        h3r, c3 = _optimize_routine(h3, n3, dx3, tf3, budget_s * 0.15)
        if c3 < best_c5:
            best_h, best_c5, n_parent, dx_p = h3r, c3, n3, dx3

    # Phase 4: Aggressive local search on best
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    if time_left(0.75) > 40:
        phase_end4 = elapsed() + min((budget_s * 0.15), time_left(0.75) - 10)
        def tf4():
            return phase_end4 - elapsed()
        h4, c4 = _pair_descent(best_h.copy(), n_best, dx_best, 500, tf4)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Micro refinement with random search
    if time_left(0.88) > 20:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        phase_end5 = elapsed() + (budget_s * 0.12 - 5)
        def tf5():
            return phase_end5 - elapsed()
        n_micro = min(500000, int(tf5() * 5000))
        best_h, best_c5 = _micro_refine(best_h.copy(), n_best, dx_best,
                                         n_micro, tf5, rng)

    # Phase 6: Final tiny adjustments
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    if time_left(0.97) > 10:
        h_f = best_h.copy()
        c5_f = _c5(h_f, dx_final)
        tiny_steps = np.array([0.0003, 0.0007, 0.0015, -0.0003, -0.0007, -0.0015])
        n_fine = min(200000, int((budget_s * 0.03 - 2) * 4000))
        for _ in range(n_fine):
            if elapsed() > budget_s - 2:
                break
            i, j = rng.randint(n_final), rng.randint(n_final)
            if i == j:
                continue
            d = tiny_steps[rng.randint(len(tiny_steps))]
            ni, nj = h_f[i] + d, h_f[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_f[i], h_f[j] = ni, nj
                c = _c5(h_f, dx_final)
                if c < c5_f:
                    c5_f = c
                    best_h, best_c5 = h_f.copy(), c
                else:
                    h_f[i], h_f[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
