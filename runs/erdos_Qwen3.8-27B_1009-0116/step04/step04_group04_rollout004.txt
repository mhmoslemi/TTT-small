        else:
            ii = np.arange(n)
        i = int(ii[rng.randint(len(ii))])
        j = rng.randint(n)
        if i == j:
            continue
        for d in steps:
            h[i] += d; h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                nc = _c5(h, dx)
                if nc < best:
                    best = nc
                    c = _corr(h) * dx
                    break
            h[i] -= d; h[j] += d
    return h, best

def _anneal_pair(h, dx, tgt, rng, n_iter, t_end, T0=0.003):
    n = len(h)
    h = h.copy()
    best = _c5(h, dx)
    T = T0
    decay = (0.01) ** (1.0 / max(1, n_iter))
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        T *= decay
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.012, 0.012)
        h[i] += d; h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            nc = _c5(h, dx)
            if nc < best or rng.random() < np.exp(-(nc - best) / max(T, 1e-10)):
                best = min(best, nc)
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return h, best

def _interp(h, nnew):
    return np.interp(np.linspace(0, 1, nnew), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0.copy(), _c5(h0, dx0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP sweep with varying objectives at n0
    t_end = t0 + 200
    for kf, gm in [(0.30, 0.3), (0.20, 0.5), (0.12, 0.6), (0.08, 0.75), (0.05, 0.9)]:
        if time.time() > t_end or rem() < 400:
            break
        h0 = _slsqp(h0, dx0, kf, gm, 700)
        c = _c5(h0, dx0)
        if c < best_c5:
            best_h, best_c5 = h0.copy(), c

    # Phase 2: L-BFGS at n0
    t_end = min(t0 + 340, t0 + budget_s - 450)
    for kf, gm in [(0.15, 0.5), (0.10, 0.7), (0.06, 0.9)]:
        if time.time() > t_end:
            break
        h0, c = _lbfgs(h0, dx0, kf, gm, 400)
        if c < best_c5:
            best_h, best_c5 = h0.copy(), c

    # Phase 3: Coordinated local search at n0
    t_end = min(t0 + 480, t0 + budget_s - 350)
    best_h, best_c5 = _coord_desc(best_h, dpi(best_n), best_n / 2.0, rng, 300000, t_end)

    # Phase 4: Annealing at n0
    t_end = min(t0 + 600, t0 + budget_s - 280)
    best_h, best_c5 = _anneal_pair(best_h, 2.0 / best_n, best_n / 2.0, rng, 150000, t_end)

    # Phase 5: Higher resolution
    if rem() > 280:
        n1 = n0 * 2
        dx1 = 2.0 / n1
        h1 = _repair(_interp(best_h, n1), n1 / 2.0)
        c1 = _c5(h1, dx1)
        t_end = min(t0 + budget_s - 180, t0 + 800)
        for kf, gm in [(0.15, 0.5), (0.10, 0.7)]:
            if time.time() > t_end:
                break
            h1, c1 = _lbfgs(h1, dx1, kf, gm, 350)
            if c1 < best_c5:
                best_h, best_c5, best_n = h1.copy(), c1, n1
        t_end = min(t0 + budget_s - 120, t0 + 900)
        for kf, gm in [(0.10, 0.7), (0.06, 0.9)]:
            if time.time() > t_end:
                break
            h1 = _slsqp(h1, dx1, kf, gm, 700)
            c1 = _c5(h1, dx1)
            if c1 < best_c5:
                best_h, best_c5, best_n = h1.copy(), c1, n1
        if rem() > 80:
            t_end = t0 + budget_s - 60
            best_h, best_c5 = _coord_desc(best_h, 2.0 / best_n, best_n / 2.0, rng, 200000, t_end)

    # Phase 6: Final polish
    if rem() > 50:
        t_end = t0 + budget_s - 25
        htmp = _slsqp(best_h, 2.0 / best_n, 0.08, 0.8, 500)
        ct = _c5(htmp, 2.0 / best_n)
        if ct < best_c5:
            best_h, best_c5 = htmp, ct
    if rem() > 20:
        best_h, best_c5 = _coord_desc(best_h, 2.0 / best_n, best_n / 2.0, rng, 80000, t0 + budget_s - 2)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)

def dpi(n):
    return 2.0 / n
