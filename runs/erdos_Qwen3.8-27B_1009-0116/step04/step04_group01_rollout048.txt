def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

def _eval(h):

    return x.sum() - _sum_target[0]

def _sum_con_jac(x):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _ls_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    return m + np.log(np.sum(np.exp(s))) / alpha

    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]

        else:

        hp = best_h.copy()
        t = np.arange(nc)
        for f in rng.randint(1, nc // 2 + 1, size=5):
            hp += 0.012 * rng.randn() * np.sin(2 * np.pi * f * t / nc + rng.rand() * 6.28)
        starts.append(_repair(hp, nc))
        # Binary-ish start
        hb = (np.random.RandomState(seed + 7).rand(nc) < 0.5).astype(float)
        hb[:nc // 2] = 1.0
        hb[nc // 2:] = 0.0
        np.random.RandomState(seed + 7).shuffle(hb)
        starts.append(_repair(hb + 0.3 * (rng.rand(nc) - 0.33), nc))

        for hp in starts:
            if _el() > budget_s * 0.92:
                break
            h_new = _slsqp_opt(hp, 1000000, 2500, 1e-18)
            h_new, c5 = _lag_descent(h_new, 100, 2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = nc

    # Phase 5: Fine pairwise coordinate search
    if _el() < budget_s * 0.96:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.004, 0.001, 0.0003, 0.00008]:
            if _el() > budget_s * 0.98:
                break
            n_iter = 4000
            for _it in range(n_iter):
                if _el() > budget_s * 0.98:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                        best_n = nc
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 6: Final SLSQP + lag polish
    if _el() < budget_s - 5:
        h_new = _slsqp_opt(best_h, 10000000, 5000, 1e-19)
        h_new, c5 = _lag_descent(h_new, 150, 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
