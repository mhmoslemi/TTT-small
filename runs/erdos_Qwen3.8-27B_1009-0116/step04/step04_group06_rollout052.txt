                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-20})
    if time.time() > t_end:

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def c5_eval(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def smooth_obj(x, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -350, 350)
    obj = m + np.log(np.sum(np.exp(s))) / alpha
    v = x.sum() - n / 2.0
    return obj + pen * v * v

def smooth_grad(x, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -350, 350)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    g *= dx
    if pen > 0:
        g += 2.0 * pen * (x.sum() - n / 2.0)
    return g

def lbfgs_step(h, n, alpha, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h, args=(n, alpha, pen),
                   jac=smooth_grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-18})
    return project(np.clip(res.x, 0, 1), n)

def pgd_step(h, n, alpha, t_end, lr, maxit):
    h = h.copy()
    best = c5_eval(h, n)
    for _ in range(maxit):
        if time.time() > t_end:

        g = smooth_grad(h, n, alpha, 0.0)

            best = c_new

                best_c5 = c_p
                best_h = h_pert
            cands.append(h_pert)
        # Average of top-3 and re-optimize
        sorted_c = sorted(cands, key=lambda x: c5_eval(x, n))
        h_avg = project(np.mean(sorted_c[:3], axis=0), n)
        h_avg = lbfgs_step(h_avg, n, 200000, 50000, 200)
        c_avg = c5_eval(h_avg, n)
        if c_avg < best_c5:
            best_c5 = c_avg
            best_h = h_avg

    # Phase 5: Pair polish
    t5 = t0 + budget_s * 0.90
    best_h, best_c5 = pair_polish(best_h, n, t5, rng)

    # Phase 6: Final ultra-fine PGD
    if el() < budget_s - 20:
        best_h, best_c5 = pgd_step(best_h, n, 1000000, t0 + budget_s - 10, 0.05, 5000)

    best_h = repair(best_h, n)
    best_c5 = c5_eval(best_h, n)
    return (best_h, best_c5, n)
