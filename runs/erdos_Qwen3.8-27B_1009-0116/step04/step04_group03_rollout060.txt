import numpy as np
import time

def _project(h, n):
    target = n / 2.0
    if abs(h.sum() - target) < 1e-12:
        return np.clip(h, 0, 1)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.ones(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g -= 2.0 * wk * h
        elif lag > 0:
            g[lag:] -= wk * h[:n - lag]
            g[:n - lag] -= wk * h[lag:]
        else:
            a = -lag
            g[:n - a] -= wk * h[a:]
            g[a:] -= wk * h[:n - a]
    return g * (2.0 / n)

def _smobj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -300, 300)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _pgd_momentum(h, n, alpha, lr, beta, n_iters, t_end):
    h = h.copy()
    v = np.zeros(n)
    c5 = _eval(h)
    obj = _smobj(h, alpha)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        g = _smgrad(h, alpha)
        v = beta * v + (1.0 - beta) * g
        h_new = _project(h - lr * v, n)
        obj_new = _smobj(h_new, alpha)
        if obj_new < obj + 1e-14:
            h = h_new
            obj = obj_new
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
        else:
            v = 0.5 * v
            h_new = _project(h - 0.5 * lr * v, n)
            if _smobj(h_new, alpha) < obj + 1e-14:
                h = h_new
                obj = _smobj(h, alpha)
                c5 = _eval(h)
    return h, c5

def _subgrad_avg(h, n, lr0, n_iters, t_end):
    h = h.copy()
    c5 = _eval(h)
    h_avg = h.copy()
    for it in range(n_iters):
        if time.time() > t_end:
            break
        lr = lr0 / (1.0 + it * 0.005)
        c = np.correlate(h, 1.0 - h, mode='full')
        k = c.argmax()
        lag = k - (n - 1)
        g = np.ones(n)
        if lag == 0:
            g -= 2.0 * h
        elif lag > 0:
            g[lag:] -= h[:n - lag]
            g[:n - lag] -= h[lag:]
        else:
            a = -lag
            g[:n - a] -= h[a:]
            g[a:] -= h[:n - a]
        g *= 2.0 / n
        h_new = _project(h - lr * g, n)
        c5n = _eval(h_new)
        if c5n < c5:
            c5 = c5n
            h = h_new
        h_avg = h_avg + (h - h_avg) / (it + 1)
    c5_avg = _eval(_project(h_avg, n))
    if c5_avg < c5:
        return _project(h_avg, n), c5_avg
    return h, c5

def _local_search(h, n, rng, t_end):
    h = h.copy()
    c5 = _eval(h)
    scale = 0.004
    no_imp = 0
    for it in range(300000):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
            no_imp = 0
            scale = min(scale * 1.02, 0.015)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 150:
                scale *= 0.88
                no_imp = 0
                if scale < 1e-8:
                    break
    return h, c5

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _make_structured(n, freq, amp, phase=0.0):
    x = np.arange(n, dtype=float)
    return np.clip(0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase), 0, 1)

def _run_phase(h, n, t_end, rng):
    h = h.copy()
    c5 = _eval(h)
    alphas = [300, 1000, 4000, 15000, 60000, 200000]
    for alpha in alphas:
        if time.time() > t_end:
            break
        lr = 0.08 / max(1.0, alpha / 1000.0)
        h, c5 = _pgd_momentum(h, n, alpha, lr, 0.85, 3000, t_end)
    if time.time() < t_end - 10:
        h, c5 = _subgrad_avg(h, n, 0.008, 4000, t_end)
    return h, c5
