import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_of(h, dx):
    return float(_corr_full(h).max() * dx)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _subgrad(h, c, top_k=5):
    n = len(h)
    idx = np.argsort(c)[-top_k:]
    w = np.ones(len(idx)) / len(idx)
    grad = np.zeros(n)
    for aw, k_idx in zip(w, idx):
        m = int(k_idx) - (n - 1)
        if m == 0:
            grad += aw * (1.0 - 2.0 * h)
        elif m > 0:
            if m < n:
                grad[:n - m] += aw * (1.0 - h[m:])
            grad[m:] -= aw * h[:n - m]
        else:
            p = -m
            if p < n:
                grad[p:] += aw * (1.0 - h[:n - p])
            grad[:n - p] -= aw * h[p:]
    return grad

def _soft_obj(x, alpha, pen):
    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.exp(s).sum())) / alpha + pen * (x.sum() - len(x) / 2.0) ** 2

def _sg_momentum(h, dx, target, t_end, n_iter=100000):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_of(h, dx)
    c = _corr_full(h)
    v = np.zeros(n)
    lr = 0.015
    mu = 0.9
    stall = 0
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        g = _subgrad(h, c, 5)
        v = mu * v - lr * g
        h_new = _project(h + v, target)
        c_new = _corr_full(h_new)
        c_val = float(c_new.max() * dx)
        if c_val < best_c - 1e-15:
            best_c = c_val
            best_h = h_new.copy()
            h, c = h_new, c_new
            stall = 0
            lr = min(lr * 1.03, 0.06)
        else:
            stall += 1
            if stall > 40:
                lr *= 0.75
                stall = 0
            if lr < 1e-8:
                break
    return best_h, best_c

def _local_search(h, dx, target, rng, t_end, n_iter=200000):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_of(h, dx)
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    per = max(300, n_iter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return _project(best_h, target), best_c
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c = _c5_of(h, dx)
                if c < best_c:
                    best_c = c; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return _project(best_h, target), best_c

def _directed_search(h, dx, target, rng, t_end, n_iter=50000):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_of(h, dx)
    c = _corr_full(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        top_idx = np.argsort(c)[-8:]
        k_idx = int(top_idx[rng.randint(8)])
        m = k_idx - (n - 1)
        if m > 0:
            ni = n - m
        elif m < 0:
            ni = n + m
        else:
            ni = n
        if ni < 2:
            continue
        i1 = rng.randint(ni)
        j1 = i1 + m
        if m == 0:
            j2 = (i1 + 1 + rng.randint(ni - 1))
        else:
            j2 = j1
        d = rng.uniform(-0.008, 0.008)
        h[i1] += d; h[j2] -= d
        if 0 <= h[i1] <= 1 and 0 <= h[j2] <= 1:
            c_new = _c5_of(h, dx)
            if c_new < best_c:
                best_c = c_new; best_h = h.copy(); c = _corr_full(h)
            else:
                c = _corr_full(h)
        else:
            h[i1] -= d; h[j2] += d
    return _project(best_h, target), best_c

def _slsqp_polish(h, dx, target, t_end, alpha=5000):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, 0.0), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': 400, 'ftol': 1e-16})
        hc = _project(r.x, target)
        return hc, _c5_of(hc, dx)
    except Exception:
        return h, _c5_of(h, dx)

def _binary_push(h, target, eps):
    n = len(h)
    h2 = h.copy()
    mask_low = h2 < 0.5
    mask_high = ~mask_low
    h2[mask_low] = np.maximum(0.0, h2[mask_low] - eps)
    h2[mask_high] = np.minimum(1.0, h2[mask_high] + eps)
    return _project(h2, target)

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)
