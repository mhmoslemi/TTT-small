def _softgrad(x, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

            best_c5 = c5_3
            best_h = h3
            best_n = n3

    # Phase 4: Try n=200
    if remaining() > 200:
        n4 = 200
        dx4 = 2.0 / n4
        h4 = _interpolate(best_h, best_n, n4)
        h4 = _repair(h4, n4)
        h4r, c5_4 = _coord_descent(h4, n4, 1500, lambda: remaining() - 80)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4r
            best_n = n4

    # Phase 5: Try n=100
    if remaining() > 120:
        n5 = 100
        dx5 = 2.0 / n5
        h5 = _interpolate(best_h, best_n, n5)
        h5 = _repair(h5, n5)
        h5r, c5_5 = _coord_descent(h5, n5, 2000, lambda: remaining() - 40)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5r
            best_n = n5

    # Phase 6: Pairwise local search
    if remaining() > 15:
        dx_best = 2.0 / best_n
        h6, c5_6 = _pairwise_search(best_h.copy(), best_n, dx_best,
                                     lambda: remaining() - 3, rng)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    # Phase 7: Final coordinate descent pass
    if remaining() > 10:
        h7, c5_7 = _coord_descent(best_h.copy(), best_n, 500, lambda: remaining() - 2)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
