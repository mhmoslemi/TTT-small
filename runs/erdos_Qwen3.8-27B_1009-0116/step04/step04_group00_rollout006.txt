def binary_local_search(h, n, dx, time_limit, rng):
    # h is binary (0 or 1)
    best_c = max_corr(h) * dx
    best_h = h.copy()
    T = 0.01  # initial temperature
    
    while time.time() < time_limit:
        # Pick a random 1-position and 0-position to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i] = 0; h[j] = 1
        new_c = max_corr(h) * dx
        
        if new_c < best_c or rng.rand() < np.exp(-(new_c - best_c) / T):
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0  # undo
        
        T *= 0.999  # cool down
    
    return best_h, best_c
