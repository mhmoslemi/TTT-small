from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _gold_pair(h, n, i, j, iters=30):
    """Gold-section search on pair (i,j). Returns improved h and c5."""
    hi0, hj0 = h[i], h[j]
    d_lo = max(-hi0, hj0 - 1.0)
    d_hi = min(1.0 - hi0, hj0)
    if d_hi - d_lo < 1e-10:
        return h, _c5(h, n)
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = d_lo, d_hi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    h[i], h[j] = hi0 + c, hj0 - c
    fc = _c5(h, n)
    h[i], h[j] = hi0 + d, hj0 - d
    fd = _c5(h, n)
    for _ in range(iters):
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - invphi * (b - a)
            h[i], h[j] = hi0 + c, hj0 - c
            fc = _c5(h, n)
        else:
            a, c, fc = c, d, fd
            d = a + invphi * (b - a)
            h[i], h[j] = hi0 + d, hj0 - d
            fd = _c5(h, n)
    h[i], h[j] = hi0 + c, hj0 - c
    fc = _c5(h, n)
    if fc > fd:
        h[i], h[j] = hi0 + d, hj0 - d
        fc = fd
    return h, fc

def _pgd_step(h, n, alpha, lr):
    """One PGD step on smooth approximation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:

        lag = k - (n - 1)

            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]

def _cd_sweep(h, n, rng, budget_end, max_pairs=None, iters=25):

    c5_cur = _c5(h, n)
    idx = np.arange(n)

            pairs_done += 1
            if c5_new < c5_cur - 1e-14:

            d = a + ip * (b - a)
            h[i], h[j] = hi0 + d, hj0 - d
            fd = _c5(h, n)
    h[i], h[j] = hi0 + c, hj0 - c
    fc = _c5(h, n)
    if fc > fd:
        h[i], h[j] = hi0 + d, hj0 - d
        fc = fd
    return h, fc

def _cd(h, n, rng, t_end, iters):
    c5 = _c5(h, n)
    idx = np.arange(n)
    sweeps = 0
    while time.time() < t_end:
        rng.shuffle(idx)
        improved = 0
        for i in idx:
            if time.time() > t_end:
                return h, c5
            for j in idx:
                if j == i:
                    continue
                h, c5n = _gold(h, n, i, j, iters)
                if c5n < c5 - 1e-14:
                    c5 = c5n
                    improved += 1
        sweeps += 1
        if improved == 0 and sweeps > 3:
            # Try with different iteration count or break
            if sweeps > 10:
                break
            iters = max(15, iters - 2)
    return h, c5

def _block_cd(h, n, rng, t_end, iters):
    c5 = _c5(h, n)
    block_sizes = [2, 3, 4, 5]
    while time.time() < t_end:
        bs = block_sizes[rng.randint(len(block_sizes))]
        i = rng.randint(n - bs)
        j = (i + rng.randint(1, n - bs) + bs) % n
        if j <= i + bs:
            j = (i + bs + rng.randint(1, n - 2 * bs) + bs) % (n - 1) if n > 2 * bs else (i + bs + 1) % n
        hi = h[i:i+bs].copy()
        hj = h[j:j+bs].copy()
        if np.any(hi <= 0) or np.any(hi >= 1) or np.any(hj <= 0) or np.any(hj >= 1):
            continue
        d_lo = max(-hi.min(), hj.max() - 1.0)
        d_hi = min(1.0 - hi.max(), hj.min())
        if d_hi - d_lo < 1e-10:
            continue
        ip = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = d_lo, d_hi
        c = b - ip * (b - a)
        d = a + ip * (b - a)
        h[i:i+bs] = hi + c; h[j:j+bs] = hj - c
        fc = _c5(h, n)
        h[i:i+bs] = hi + d; h[j:j+bs] = hj - d
        fd = _c5(h, n)
        for _ in range(iters):
            if b - a < 1e-5:
                break
            if fc < fd:
                b, d, fd = d, c, fc
                c = b - ip * (b - a)
                h[i:i+bs] = hi + c; h[j:j+bs] = hj - c
                fc = _c5(h, n)
            else:
                a, c, fc = c, d, fd
                d = a + ip * (b - a)
                h[i:i+bs] = hi + d; h[j:j+bs] = hj - d
                fd = _c5(h, n)
        h[i:i+bs] = hi + c; h[j:j+bs] = hj - c
        fc = _c5(h, n)
        if fc > fd:
            h[i:i+bs] = hi + d; h[j:j+bs] = hj - d
            fc = fd
        if fc < c5:
            c5 = fc
    return h, c5

def _slsqp_con_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _slsqp_con_jac(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: PGD warm-up (0-70s)
    t1 = t0 + 70
    for alpha in [3000, 15000, 60000]:
        if time.time() > t1:
            break
        tw = min(time.time() + 22, t1)
        hw, cw = _pgd(best_h, n, alpha, tw, 0.05, 3000)
        cw = _c5(hw, n)
        if cw < best_c5:
            best_h, best_c5 = hw, cw

    # Phase 2: Multi-start (70-160s)
    t2 = t0 + 160
    for r in range(12):
        if time.time() > t2:
            break
        scale = 0.03 + 0.04 * rng.rand()
        h_init = _repair(best_h + scale * rng.randn(n), n)
        for alpha in [8000, 40000]:
            if time.time() > t2:
                break
            tw = min(time.time() + 6, t2)
            hw, cw = _pgd(h_init, n, alpha, tw, 0.03, 400)
            cw = _c5(hw, n)
            if cw < best_c5:
                best_h, best_c5 = hw, cw

    # Phase 3: Main coordinate descent with gold-section (160-980s)
    t3 = t0 + 980
    best_h, best_c5 = _cd(best_h, n, rng, t3, 22)

    # Phase 4: Block coordinate moves (980-1050s)
    t4 = t0 + 1050
    best_h, best_c5 = _block_cd(best_h, n, rng, t4, 18)

    # Phase 5: Final fine CD (1050-1085s)
    t5 = t0 + budget_s - 15
    best_h, best_c5 = _cd(best_h, n, rng, t5, 15)

    # Phase 6: SLSQP polish on smooth approx (last 10s)
    if time.time() < t0 + budget_s - 3:
        try:
            global _slsqp_n
            _slsqp_n = n
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: _slsqp_con_fun(x, n),
                     'jac': lambda x: _slsqp_con_jac(x, n)}]
            res = minimize(_smooth_obj, best_h, args=(n, 200000),
                           jac=lambda x: _smooth_grad(x, n, 200000),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5v = _c5(hc, n)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
