import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

    """Analytical gradient of log-sum-exp smoothed max correlation."""
    n = len(h)
    c = _corr(h)
    m = c.max()
    ws = np.exp(np.clip(alpha * (c - m), -500, 500))

    g = np.zeros(n)

        if w < 1e-12: continue
        s = k - (n - 1)

            g[:n-s] -= w * h[s:]
        else:
            a = -s
            g[:n-a] += w * (1.0 - h[a:])
