Actually, let me try a different strategy: **barrier method with projected gradient**.

The idea:
1. Add a log barrier for the bound constraints (0 ≤ h ≤ 1)
2. Add a quadratic penalty for the equality constraint (sum h = n/2)
3. Use projected gradient descent with backtracking line search
4. The "projection" is onto {h : 0 ≤ h ≤ 1, sum h = n/2}

The projection onto this set:
- Start with h
- Clip to [0,1]
- Adjust to meet sum = n/2 (by adding/subtracting a constant, then re-clipping, iterating)

This is actually the same as the repair function!

So the algorithm is:
1. Compute gradient of smooth max
2. Take step: h_new = h - step * grad
3. Project (repair): h_new = repair(h_new, n)
4. Check if objective decreased; if not, halve step
5. Repeat

With momentum for acceleration.

Let me also think about **Nesterov's accelerated gradient method**. For non-smooth objectives (max of smooth functions), proximal gradient methods work well. Here the "proximity operator" is the projection onto the feasible set, which is our repair function.

def nesterov_smooth_max(h, n, dx, t0, limit, alpha_init=50, alpha_final=20000):
    bh = h.copy()
    bc = c5val(h, dx)
    x = bh.copy()
    y = bh.copy()  # Nesterov momentum variable
    t_k = 1.0
    alpha = alpha_init
    
    for it in range(500):
        if time.time() - t0 > limit:
            break
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        beta = (t_k - 1) / t_new
        y = x + beta * (x - y)
        y = np.clip(y, 0, 1)
        
        # Gradient at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = corr.max()
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        ew = np.exp(s)
        pw = ew / ew.sum()
        
        g = np.zeros(n)
        for k in range(2*n - 1):
            if pw[k] > 1e-12:
                g += pw[k] * lag_gradient(y, k, n)
        g /= alpha
        g -= g.mean()  # preserve sum
        
        # Step size (Lipschitz constant estimate)
        step = 1.0 / (alpha * 4.0 + 1e-6)  # rough Lipschitz bound
        # Actually the Lipschitz constant of grad is bounded by... let me think
        # For smooth max, L ≤ alpha * max_k ||A_k||_2 where A_k is the quadratic form matrix
        # For our problem, ||A_k||_2 ≤ 2n (roughly), so L ≈ 2*alpha*n
        # But that's too conservative. Let's use a safer estimate.
        step = 0.5 / (alpha * 2.0 + 1)
        
        # Take step and project
        x_new = x - step * g
        x_new = repair(x_new, n)
        
        # Backtracking
        f_new = c5val(x_new, dx)
        f_old = c5val(x, dx)
        trials = 0
        while f_new > f_old - 1e-15 and trials < 20:
            step *= 0.5
            x_new = x - step * g
            x_new = repair(x_new, n)
            f_new = c5val(x_new, dx)
            trials += 1
        
        if f_new < f_old:
            x = x_new
            if f_new < bc:
                bc, bh = f_new, x.copy()
        
        t_k = t_new
        # Gradually increase alpha
        alpha *= 1.02
        if alpha > alpha_final:
            alpha = alpha_final
    
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            amt = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += amt
            d -= amt
    return h

def c5val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def lag_grad(h, k, n):
    g = np.zeros(n)
    L = k - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def sm_obj(x, n, alpha, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (x.sum() - n * 0.5) ** 2

def sm_jac(x, n, alpha, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(len(x))
    for k in range(2 * n - 1):
        if pw[k] > 1e-12:
            g += pw[k] * lag_grad(x, k, n)
    g /= alpha
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def opt_smooth(h, n, dx, t0, limit):
    bh, bc = h.copy(), c5val(h, dx)
    bds = [(0.0, 1.0)] * n
    for alpha, lam in [(40, 50), (150, 200), (500, 500), (2000, 1500),
                       (8000, 5000), (30000, 15000), (100000, 40000), (300000, 100000)]:
        if time.time() - t0 > limit:
            break
        try:
            r = minimize(sm_obj, bh.copy(), args=(n, alpha, lam),
                         jac=lambda x: sm_jac(x, n, alpha, lam),
                         method='L-BFGS-B', bounds=bds,
                         options={'maxiter': 250, 'ftol': 1e-19, 'gtol': 1e-16})
            hc = repair(r.x, n)
            c = c5val(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def prox_grad(h, n, dx, t0, limit):
    bh, bc = h.copy(), c5val(h, dx)
    x = bh.copy()
    v = np.zeros(n)
    t_k = 1.0
    alpha = 30.0
    for it in range(800):
        if time.time() - t0 > limit:
            break
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        beta = (t_k - 1) / t_new
        y = x + beta * v
        y = np.clip(y, 0, 1)
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = corr.max()
        s = np.clip(alpha * (corr - m), -500, 500)
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if pw[k] > 1e-12:
                g += pw[k] * lag_grad(y, k, n)
        g /= alpha
        g -= g.mean()
        step = 0.8 / (alpha * 1.5 + 1.0)
        x_new = x - step * g
        x_new = repair(x_new, n)
        f_new = c5val(x_new, dx)
        f_old = c5val(x, dx)
        tries = 0
        while f_new >= f_old - 1e-16 and tries < 25:
            step *= 0.5
            x_new = x - step * g
            x_new = repair(x_new, n)
            f_new = c5val(x_new, dx)
            tries += 1
        v = x_new - x
        x = x_new
        if f_new < bc:
            bc, bh = f_new, x.copy()
        t_k = t_new
        if it % 40 == 39:
            alpha = min(alpha * 1.15, 80000.0)
    return bh, bc

def active_opt(h, n, dx, t0, limit):
    bh, bc = h.copy(), c5val(h, dx)
    for topk in [3, 5, 8, 12, 20, 30]:
        if time.time() - t0 > limit:
            break
        x = bh.copy()
        step = 0.06
        for _ in range(200):
            if time.time() - t0 > limit:
                break
            corr = np.correlate(x, 1.0 - x, mode='full')
            thr = corr.max() - 0.004 * max(corr.std(), 1e-10)
            idx = np.where(corr >= thr)[0]
            if len(idx) > topk:
                idx = np.argsort(corr)[::-1][:topk]
            g = np.zeros(n)
            for k in idx:
                g += lag_grad(x, k, n)
            g /= len(idx)
            g -= g.mean()
            ok = False
            for _ in range(35):
                xn = repair(x - step * g, n)
                cn = c5val(xn, dx)
                if cn < c5val(x, dx) - 1e-18:
                    x = xn
                    ok = True
                    if cn < bc:
                        bc, bh = cn, x.copy()
                    step = min(step * 1.1, 1.5)
                    break
                step *= 0.7
            if not ok:
                step *= 0.6
                if step < 1e-6:
                    break
    return bh, bc

def fourier_de(n, t0, limit, seed, init_h):
    rng = np.random.RandomState(seed)
    K = 10
    np2 = 2 * K
    bds = [(-0.45, 0.45)] * np2
    idx = np.arange(n)

    def eval_params(p):
        x = np.full(n, 0.5)
        for k in range(K):
            x += p[2 * k] * np.cos(2 * np.pi * (k + 1) * idx / n)
            x += p[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * idx / n)
        x = repair(x, n)
        return c5val(x, 2.0 / n)

    H = np.fft.rfft(init_h)
    p0 = np.zeros(np2)
    for k in range(min(K, len(H) - 1)):
        p0[2 * k] = 2.0 * np.real(H[k + 1]) / n
        p0[2 * k + 1] = 2.0 * np.imag(H[k + 1]) / n
    p0 = np.clip(p0, -0.45, 0.45)

    pop = [p0.copy()]
    for i in range(29):
        p = rng.uniform(-0.35, 0.35, np2)
        pop.append(p)
    p_best = p0.copy()
    f_best = eval_params(p0)
    h_best = repair(np.full(n, 0.5) + np.zeros(n), n)
    x_tmp = np.full(n, 0.5)
    for k in range(K):
        x_tmp += p_best[2 * k] * np.cos(2 * np.pi * (k + 1) * idx / n)
        x_tmp += p_best[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * idx / n)
    h_best = repair(x_tmp, n)

    for gen in range(40):
        if time.time() - t0 > limit:
            break
        for i in range(30):
            if time.time() - t0 > limit:
                break
            others = [j for j in range(30) if j != i]
            a, b, c = rng.choice(others, 3, replace=False)
            F = 0.7 + 0.3 * rng.rand()
            CR = 0.7 + 0.2 * rng.rand()
            v = p_best + F * (pop[a] - pop[b]) + 0.3 * F * (pop[b] - pop[c])
            jrand = rng.randint(np2)
            trial = np.where(rng.rand(np2) < CR, v, pop[i])
            trial[jrand] = v[jrand]
            trial = np.clip(trial, -0.45, 0.45)
            f_tr = eval_params(trial)
            x_t = np.full(n, 0.5)
            for k in range(K):
                x_t += trial[2 * k] * np.cos(2 * np.pi * (k + 1) * idx / n)
                x_t += trial[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * idx / n)
            x_t = repair(x_t, n)
            f_cur = eval_params(pop[i])
            if f_tr < f_cur:
                pop[i] = trial
                if f_tr < f_best:
                    f_best = f_tr
                    p_best = trial.copy()
                    h_best = x_t.copy()
    return h_best, f_best

def pair_cd(h, n, dx, seed, t0, limit):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5val(h, dx)
    cc = bc
    stall = 0
    while time.time() - t0 < limit and stall < 100:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        imp = False
        for d in [0.006, 0.003, 0.0015, -0.006, -0.003, -0.0015]:
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = c5val(h, dx)
                if c < cc - 1e-15:
                    h = repair(h, n)
                    cc = c5val(h, dx)
                    stall = 0
                    imp = True
                    if cc < bc:
                        bc, bh = cc, h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        if not imp:
            stall += 1
    return bh, bc
