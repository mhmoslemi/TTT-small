            if _el() > budget_s - 150:
                break
            h2 = _slsqp_h(h2, alpha, 700, 1e-18)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h2.copy(), n2

    # Phase 4: Active-set gradient descent at best resolution
    if _el() < budget_s - 120:
        h = best_h.copy()
        nc = len(h)
        for m in [3, 5, 8, 15, 30]:
            if _el() > budget_s - 120:
                break
            h = _proj_grad_active(h, m, 300, 0.015)
            h = _repair(h, nc)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), nc

    # Phase 5: Multi-start exploration
    if _el() < budget_s - 120:
        nc = best_n
        for _r in range(25):
            if _el() > budget_s - 120:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                for _b in range(rng.randint(1, 3)):
                    bs = rng.randint(4, max(5, nc // 7))
                    st = rng.randint(0, nc - bs)
                    hp[st:st+bs] += 0.06 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 2:
                period = rng.choice([3, 4, 5, 6, 7, 8])
                hp = np.array([1.0 if (i % period) < period // 2 else 0.0
                               for i in range(nc)])
                hp += 0.05 * rng.randn(nc)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = np.concatenate([np.ones(nc // 2), np.zeros(nc - nc // 2)])
                hp = np.roll(hp, rng.randint(nc)) + 0.03 * rng.randn(nc)
                hp = _repair(hp, nc)
            elif strat == 4:
                hp = best_h.copy()
                for _f in range(rng.randint(2, 5)):
                    f = rng.randint(1, nc // 3 + 1)
                    hp += 0.008 * rng.randn() * np.sin(
                        2 * np.pi * f * np.arange(nc) / nc)
                hp = _repair(hp, nc)
            else:
                # Voronoi-like: random blocks
                hp = np.zeros(nc)
                splits = sorted(rng.choice(range(1, nc), rng.randint(dc := 4, 10), replace=False))
                splits = [0] + list(splits) + [nc]
                for b in range(len(splits) - 1):
                    hp[splits[b]:splits[b+1]] = rng.uniform(0, 1)
                hp = _repair(hp, nc)
            h_new = _slsqp_h(hp, 100000, 600, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase 6: Simulated-annealing pairwise search
    if _el() < budget_s - 50:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        T = 0.002
        for _it in range(80000):
            if _el() > budget_s - 50:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 0.004 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n - c5c < 0 or rng.rand() < np.exp(-(c5n - c5c) / T):
                c5c = c5n
                if c5c < best_c5:
                    best_c5, best_h = c5c, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
            T *= 0.9998

    # Phase 7: Final SLSQP polish
    if _el() < budget_s - 15:
        h = _slsqp_h(best_h, 600000, 1500, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 8: Ultra-fine pairwise
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(40000):
            if _el() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
