def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        d = target - s
        if abs(d) < 1e-12:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a

def soft_total_and_grad(h, n, alpha, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    exps = np.exp(s)
    total = np.sum(exps)
    w = exps / total
    obj = dx * (m + np.log(total) / alpha)
    
    grad = np.zeros(n)
    lags = np.arange(len(corr)) - (n - 1)
    for li, lag in enumerate(lags):
        if w[li] < 1e-14 or abs(lag) >= n:
            continue
        wl = w[li] * dx
        mp = np.arange(n) + lag
        mm = np.arange(n) - lag
        vp = (mp >= 0) & (mp < n)
        vm = (mm >= 0) & (mm < n)
        g = np.zeros(n)
        g[vp] = 1.0 - h[mp[vp]]
        g[vm] -= h[mm[vm]]
        grad += wl * g
    
    sf = h.sum()
    obj += pen * (sf - n / 2.0) ** 2
    grad += 2.0 * pen * (sf - n / 2.0)
    return obj, grad

    def fun_and_grad(x):
        return soft_total_and_grad(x, n, alpha, pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun_and_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                       'gtol': 1e-12, 'maxls': 50})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    no_improve = 0
    while time.time() < t_end:
        improved = False
        for _ in range(12 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            for d in steps:
                for sgn in (1, -1):
                    delta = sgn * d
                    ni, nj = hi + delta, hj - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue

                    c5 = compute_c5(h, n)

                        improved = True

                if improved:

        if improved:

        else:

corr[k] = sum_i h[i+k] * (1-h[i]) (valid i)

d/dh[m]:
