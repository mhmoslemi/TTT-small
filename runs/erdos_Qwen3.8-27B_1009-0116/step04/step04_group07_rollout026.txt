    else:
        best_h = repair(np.full(init_n, 0.5), init_n)
        best_c5 = compute_c5(best_h, best_n)
    n = best_n
    h_work = best_h.copy()

    def elapsed():
        return time.time() - t0

    # Phase 1: Projected subgradient (new method)
    for lr0 in [0.005, 0.003, 0.002]:
        if elapsed() > budget_s * 0.15:
            break
        h_sg, c5_sg = proj_subgrad(h_work, n, lr0, 8000, 4, t0 + budget_s * 0.15, rng)
        if c5_sg < best_c5 - 1e-14:
            best_h = h_sg.copy()
            best_c5 = c5_sg
            h_work = h_sg

    # Phase 2: SLSQP with top-k focused objective
    for alpha, kt in [(5000, 5), (30000, 4), (150000, 3)]:
        if elapsed() > budget_s * 0.30:
            break
        h_new = slsqp_topk(h_work.copy(), n, alpha, kt, 500)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new
        elif c5_new < compute_c5(h_work, n) - 1e-14:
            h_work = h_new

    # Phase 3: Targeted coordinate descent (long)
    if elapsed() < budget_s * 0.60:
        h_cd, c5_cd = targeted_cd(h_work, n, t0 + budget_s * 0.60, rng, 400000)
        if c5_cd < best_c5 - 1e-14:
            best_h = h_cd.copy()
            best_c5 = c5_cd
            h_work = h_cd

    # Phase 4: Multi-resolution
    if elapsed() < budget_s * 0.78:
        for n_alt in [48, 80, 96, 128, 160]:
            if elapsed() > budget_s * 0.78:
                break
            if n_alt == best_n:
                continue
            x_old = np.linspace(0.0, 1.0, best_n)
            x_new = np.linspace(0.0, 1.0, n_alt)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_alt = slsqp_topk(h_seed, n_alt, 30000, 3, 350)
            c5_alt = compute_c5(h_alt, n_alt)
            h_alt2, c5_alt2 = targeted_cd(h_alt, n_alt, t0 + budget_s * 0.78, rng, 25000)
            if c5_alt2 < c5_alt:
                h_alt, c5_alt = h_alt2, c5_alt2
            if c5_alt < best_c5 - 1e-14:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    # Phase 5: Simulated annealing
    if elapsed() < budget_s * 0.90:
        h_sa, c5_sa = simulated_anneal(best_h, best_n, 0.03, t0 + budget_s * 0.90, 60000, rng)
        if c5_sa < best_c5 - 1e-14:
            best_h = h_sa.copy()
            best_c5 = c5_sa

    # Phase 6: Final subgradient + SLSQP polish
    if elapsed() < budget_s * 0.97:
        h_fin, c5_fin = proj_subgrad(best_h, best_n, 0.002, 5000, 3, t0 + budget_s * 0.94, rng)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_fin2 = slsqp_topk(best_h.copy(), best_n, 800000, 3, 500)
        c5_fin2 = compute_c5(h_fin2, best_n)
        if c5_fin2 < best_c5 - 1e-14:
            best_h = h_fin2.copy()
            best_c5 = c5_fin2

    # Phase 7: Final targeted CD
    if elapsed() < budget_s * 0.995:
        h_fin3, c5_fin3 = targeted_cd(best_h, best_n, t0 + budget_s * 0.998, rng, 80000)
        if c5_fin3 < best_c5 - 1e-14:
            best_h = h_fin3.copy()
            best_c5 = c5_fin3

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
