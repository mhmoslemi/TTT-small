    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5

        if s > target:

        else:

    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)

                a = min(d, 1.0-h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
    return h

def _project(h, n):
    tgt = n/2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(h-mid, 0, 1.0)) > tgt: lo = mid
        else: hi = mid
    hp = np.clip(h - 0.5*(lo+hi), 0, 1.0)
    d = tgt - hp.sum()
    if abs(d) > 1e-14:
        m = (hp > 1e-10) & (hp < 1.0-1e-10)
        if m.sum() > 0:
            hp[m] += d / m.sum()
            hp = np.clip(hp, 0, 1.0)
    return hp
