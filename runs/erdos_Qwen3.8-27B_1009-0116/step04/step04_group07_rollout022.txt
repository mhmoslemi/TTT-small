                h[idx] += a
                d -= a
            else:

                h[idx] -= a
                d += a

            h[idx] += d
        else:
            idx = np.argmax(h)
            h[idx] += d
    return h

    except Exception:
        pass
    return h0

def slsqp_opt(h0, n, alpha, maxiter=400):
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        dx = 2.0 / n
        return dx * (m + np.log(np.sum(np.exp(s))) / alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if not np.any(np.isnan(res.x)):
            return repair(res.x, n)
    except Exception:
        pass
    return h0

def sim_anneal(h0, n, t_end, rng, sigma_base=0.04):
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    target = n / 2.0
    it = 0
    while time.time() < t_end:
        it += 1
        # Geometric cooling based on iteration count
        T = max(1e-9, 0.004 * (0.999997 ** it))
        # Propose pair move
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        sigma = sigma_base * (1.0 + 0.5 * np.exp(-it / 500000.0))
        delta = rng.uniform(-sigma, sigma)
        hi2 = h[i] + delta
        hj2 = h[j] - delta
        if hi2 < 0 or hi2 > 1 or hj2 < 0 or hj2 > 1:
            continue
        h[i], h[j] = hi2, hj2
        c5_new = compute_c5(h, n)
        d = c5_new - cur_c5
        if d <= 0 or rng.random() < np.exp(-d / T):
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = hi2 - delta, hj2 + delta  # revert
    return best_h, best_c5

def coord_descent_full(h0, n, t_end, rng):
    h = h0.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    steps_coarse = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    step_idx = 0
    stall = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                for step in steps_coarse[step_idx:step_idx + 3]:
                    for sign in (1, -1):
                        d = sign * step
                        ni, nj = hi + d, hj - d
                        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                            continue
                        h[i], h[j] = ni, nj
                        c5_new = compute_c5(h, n)
                        if c5_new < best_c5 - 1e-15:
                            best_c5 = c5_new
                            best_h = h.copy()
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
        # Three-point moves
        for _ in range(n):
            if time.time() > t_end:
                break
            i = rng.randint(n - 2)
            a, b, c = i, i + 1, i + 2
            for step in steps_coarse[step_idx:step_idx + 2]:
                for pat in [(step, -step, 0), (0, step, -step), (step, 0, -step),
                            (-step, step, 0), (0, -step, step), (-step, 0, step)]:
                    na, nb, nc = h[a] + pat[0], h[b] + pat[1], h[c] + pat[2]
                    if na < 0 or na > 1 or nb < 0 or nb > 1 or nc < 0 or nc > 1:
                        continue
                    h[a], h[b], h[c] = na, nb, nc
                    c5_new = compute_c5(h, n)
                    if c5_new < best_c5 - 1e-15:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                    else:
                        h[a], h[b], h[c] = na - pat[0], nb - pat[1], nc - pat[2]
        if improved:
            stall = 0
            if step_idx < len(steps_coarse) - 1 and stall > 50:
                step_idx = 0
        else:
            stall += 1
            if step_idx < len(steps_coarse) - 1:
                step_idx += 1
            elif stall > 20:
                break
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    # Get initial
    init_n = 64
    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass
    if init_h is None:
        x = np.linspace(0, 2, init_n)
        init_h = 0.5 + 0.45 * np.sin(np.pi * x)
    init_h = repair(init_h, init_n)
    best_h = init_h.copy()
    best_c5 = compute_c5(init_h, init_n)
    best_n = init_n

    # Phase 1: Quick L-BFGS-B warm start (first 8%)
    t1 = t0 + budget_s * 0.08
    h_work = best_h.copy()
    for alpha in [10000, 50000]:
        if time.time() > t1:
            break
        h_new = lbfgs_opt(h_work, best_n, alpha, 5000, 2000)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5, h_work = h_new.copy(), c5_new, h_new

    # Phase 2: Simulated Annealing (next 55%)
    t2 = t0 + budget_s * 0.63
    h_sa, c5_sa = sim_anneal(best_h.copy(), best_n, t2, rng)
    if c5_sa < best_c5 - 1e-14:
        best_h, best_c5 = h_sa.copy(), c5_sa

    # Phase 3: Full coordinate descent (next 25%)
    t3 = t0 + budget_s * 0.88
    h_cd, c5_cd = coord_descent_full(best_h.copy(), best_n, t3, rng)
    if c5_cd < best_c5 - 1e-14:
        best_h, best_c5 = h_cd.copy(), c5_cd

    # Phase 4: Try alternate n values (next 8%)
    t4 = t0 + budget_s * 0.96
    if time.time() < t4:
        x_old = np.linspace(0.0, 1.0, best_n, endpoint=False)
        for n_alt in [72, 80, 96]:
            if time.time() > t4:
                break
            x_new = np.linspace(0.0, 1.0, n_alt, endpoint=False)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_seed = lbfgs_opt(h_seed, n_alt, 30000, 2000, 1500)
            c5_alt = compute_c5(h_seed, n_alt)
            # Quick SA on this
            t_sub = min(t4, time.time() + 30)
            h_alt, c5_alt2 = sim_anneal(h_seed, n_alt, t_sub, rng)
            if c5_alt2 < c5_alt:
                h_seed, c5_alt = h_alt, c5_alt2
            h_alt2, c5_alt3 = coord_descent_full(h_seed, n_alt, min(t4, time.time() + 20), rng)
            if c5_alt3 < c5_alt:
                h_seed, c5_alt = h_alt2, c5_alt3
            if c5_alt < best_c5 - 1e-14:
                best_h, best_c5, best_n = h_seed.copy(), c5_alt, n_alt

    # Phase 5: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.995 and best_n <= 96:
        h_fin = slsqp_opt(best_h.copy(), best_n, 200000, 500)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h, best_c5 = h_fin.copy(), c5_fin
        # Tiny coord descent polish
        h_fin2, c5_fin2 = coord_descent_full(best_h, best_n, t0 + budget_s * 0.995, rng)
        if c5_fin2 < best_c5 - 1e-14:
            best_h, best_c5 = h_fin2.copy(), c5_fin2

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
