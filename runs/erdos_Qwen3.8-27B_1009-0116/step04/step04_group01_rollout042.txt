def _binary_round(h, n):
    """Round to binary with exactly n/2 ones, using the values as a priority."""
    target = n // 2
    order = np.argsort(-h)  # descending
    out = np.zeros(n)
    out[order[:target]] = 1.0
    return out

def _local_search_binary(h, rng, n_iters):
    """Local search on binary h: move a 1 to an adjacent or random 0 position."""
    n = len(h)
    c5 = _eval(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_iters):

        # Choose a target for the 1

            if j < 0 or j >= n or h[j] != 0:
                continue

            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = 1, 0
