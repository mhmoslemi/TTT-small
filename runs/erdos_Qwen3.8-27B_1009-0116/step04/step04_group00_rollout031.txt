    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

            best_x = x.copy()
            no_improve = 0
            step = min(step * 1.02, 0.03)
        else:
            no_improve += 1
            if no_improve > 100:
                step *= 0.7
                no_improve = 0
    return best_x, best_c

def _coord_descend(h, n, dx, n_sweeps, seed, deadline):
    rng = np.random.RandomState(seed)
    c5c = _c5(h, dx)
    dlist = np.array([0.0002, 0.0005, 0.001, 0.0025, 0.006,
                     -0.0002, -0.0005, -0.001, -0.0025, -0.006])
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        g = _subgrad(h, n)
        gc = g - g.mean()
        order = np.argsort(np.abs(gc))[::-1][:40]
        improved = False
        for ii in range(len(order)):
            if time.time() > deadline:
                break
            i = int(order[ii])
            for jj in range(ii + 1, min(ii + 12, len(order))):
                j = int(order[jj])
                bd, bc = 0.0, c5c
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < bc:
                            bc, bd = c, d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    break
        if not improved:
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            d = float(rng.choice(dlist))
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                else:
                    h[i], h[j] = ni - d, nj + d
    return h, c5c

def _sim_anneal(h, n, dx, n_iter, seed, deadline):
    rng = np.random.RandomState(seed)
    c5c = _c5(h, dx)
    h = h.copy()
    dlist = np.array([0.0003, 0.001, 0.003, 0.008,
                     -0.0003, -0.001, -0.003, -0.008])
    for it in range(n_iter):
        if time.time() > deadline:
            break
        temp = 0.0002 * (1.0 - it / float(n_iter)) + 0.00001
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        d = float(rng.choice(dlist))
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c or rng.rand() < np.exp(-(c - c5c) / temp):
                c5c = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: L-BFGS-B passes
    h_curr = h0.copy()
    for alpha, lam in [(2000, 60.0), (15000, 100.0), (80000, 150.0)]:
        dl = budget_s - 250
        if time.time() > dl:
            break
        h_curr, c5v = _lbfgsb(h_curr, n, alpha, lam, 500, dl)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_curr.copy()

    # Phase 2: FISTA accelerated projected subgradient
    dl = budget_s - 150
    if time.time() < dl:
        h_curr, c5v = _fista(best_h, n, 5000, 0.012, dl)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_curr.copy()

    # Phase 3: Coordinate descent
    dl = budget_s - 60
    if time.time() < dl:
        h_curr = best_h.copy()
        h_curr, c5v = _coord_descend(h_curr, n, dx, 2500, seed, dl)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_curr.copy()

    # Phase 4: Simulated anneal for escape
    dl = budget_s - 30
    if time.time() < dl:
        h_curr = best_h.copy()
        h_curr, c5v = _sim_anneal(h_curr, n, dx, 30000, seed + 100, dl)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_curr.copy()

    # Phase 5: Final L-BFGS-B + coord descent
    dl = budget_s - 10
    if time.time() < dl:
        h_curr = best_h.copy()
        h_curr, c5v = _lbfgsb(h_curr, n, 300000, 300.0, 600, dl)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_curr.copy()
        h_curr, c5v = _coord_descend(best_h.copy(), n, dx, 2000, seed + 200, budget_s - 5)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_curr.copy()

    # Phase 6: Try n=100 grid
    if time.time() < budget_s - 100:
        n2 = 100
        dx2 = 2.0 / n2
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        c5v2 = _c5(h2, dx2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = h2.copy()
            n, dx = n2, dx2
        dl2 = budget_s - 50
        h2, c5v2 = _lbfgsb(h2, n2, 10000, 80.0, 400, dl2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = h2.copy()
            n, dx = n2, dx2
        h2, c5v2 = _coord_descend(h2, n, dx, 2000, seed + 300, budget_s - 10)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = h2.copy()
            n, dx = n2, dx2

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
