import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_cx(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), c

def _soft_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            g[lag:] += wk * (-h[:n - lag])
        else:
            ll = -lag
            g[ll:] += wk * (1.0 - h[:n - ll])
            g[:n - ll] += wk * (-h[ll:])
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cfj(x, n):
    return np.ones(n)

def _opt(h, n, alpha, maxiter, method='SLSQP'):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cfj, 'args': (n,)}]
    try:
        if method == 'SLSQP':
            res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
        else:
            res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                           method='L-BFGS-B', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _gen_inits(n, rng, h_ref):

    cands.append(_repair(_interp(h_ref, len(h_ref), n), n))
    for freq in [1, 2, 3, 4, 5, 6, 7]:
        for amp in [0.3, 0.5, 0.65, 0.8]:

            cands.append(_repair(h, n))
    # Thue-Morse-like binary

        t = [0]
        for _ in range(int(np.ceil(np.log2(max(N, 2))))):
            t += [1 - x for x in t]
        return np.array(t[:N], dtype=float)
    tm = thue_morse(n)
    cands.append(_repair(tm * 1.0, n))

            best_h = hc.copy()

        h[i], h[j] = ni, nj
        c5n = _eval(h)
