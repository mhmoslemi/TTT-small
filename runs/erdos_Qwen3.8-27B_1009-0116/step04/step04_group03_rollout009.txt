import time

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_top(h, topm=3):
    """Return (max_val, top_lags)."""
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)
    m = c.max()
    idx = np.argsort(c)[-topm:]
    return m, idx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a

                    break
    return h

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _sobj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * h[i2 + lag:j2 + lag]
    return g

def _slsqp(h, n, alpha, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n,)}]
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cfun(x, n):
    return np.sum(x) - n / 2.0

def _cjac(x, n):
    return np.ones(n)

def _proj_grad(h, n, alpha, lr, t_end):
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    no_imp = 0
    while time.time() < t_end:
        g = _sgrad(h, alpha)
        h_new = _repair(h - lr * g, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-14:
            h = h_new; c5c = c5n; no_imp = 0; lr *= 0.99
            if c5n < best_c5:
                best_c5 = c5n; best_h = h.copy()
        else:
            no_imp += 1
            if no_imp > 30:
                lr *= 0.4; no_imp = 0
                if lr < 1e-12:
                    break
    return best_h, best_c5

def _local_search(h, n, rng, t_end, scale0=0.008):
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    scale = scale0
    no_imp = 0
    while time.time() < t_end:
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n; no_imp = 0
            scale = min(scale * 1.04, 0.025)
            if c5n < best_c5:

        else:

            no_imp += 1
            if no_imp > 250:
                scale *= 0.7; no_imp = 0
                if scale < 2e-7:
                    break
    return best_h, best_c5

    best_h, best_c5 = h.copy(), c5c
    while time.time() < t_end:

                    best_c5 = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)
