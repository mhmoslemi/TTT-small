def _repair(h, n):

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        i1 = max(0, -lag)
        i2 = min(n, n - lag)
        if i1 < i2:
            g[i1+lag:i2+lag] += w[k] * (1.0 - h[i1:i2])
            g[i1:i2] -= w[k] * h[i1+lag:i2+lag]
    return g

def _cf(h, n):
    return h.sum() - n / 2.0

def _cj(h, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _active_set_opt(h, n, t_end, n_rounds=5):
    """Optimize for the top few active lags using weighted sum."""
    c5c = _eval(h)
    h_cur = h.copy()
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        c = _corr(h_cur)
        m = c.max()
        # Find top lags within 2% of max
        threshold = m * 0.98
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Weighted sum objective
        w = (c[active] - threshold) + 0.01
        w = w / w.sum()
        # Gradient of weighted sum
        lag_list = active - (n - 1)
        g = np.zeros(n)
        for k, wt in zip(active, w):
            lag = k - (n - 1)
            i1 = max(0, -lag)
            i2 = min(n, n - lag)
            if i1 < i2:
                g[i1+lag:i2+lag] += wt * (1.0 - h_cur[i1:i2])
                g[i1:i2] -= wt * h_cur[i1+lag:i2+lag]
        # Projected gradient step with backtracking
        lr = 0.05
        for _ in range(15):
            h_try = _repair(h_cur - lr * g, n)
            if _eval(h_try) < c5c + 1e-12:
                break
            lr *= 0.5
        h_new = _repair(h_cur - lr * g, n)
        v = _eval(h_new)
        if v < c5c - 1e-15:
            c5c = v
            h_cur = h_new
        else:
            # Small random kick
            idx = np.random.randint(n, size=min(5, n))
            h_cur[idx] += np.random.randn(len(idx)) * 0.005
            h_cur = _repair(h_cur, n)
    return h_cur, c5c

def _sa_search(h, n, rng, t_end):
    """Simulated annealing with adaptive temperature."""
    h_cur = h.copy()
    c5c = _eval(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5c
    t_start = time.time()
    dur = max(1.0, t_end - t_start)
    step = 0
    T = 0.003
    while time.time() < t_end:
        step += 1
        frac = min(1.0, step / dur)
        T = 0.003 * (1e-3) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.randn() * 0.012
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c or (c5n - c5c) < 0.01 and rng.rand() < np.exp(-(c5n - c5c) / (T + 1e-10)):
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
    return best_h, best_c5

def _fine_search(h, n, rng, t_end):
    """Deterministic tiny-step local search."""
    h_c = h.copy()
    c5c = _eval(h_c)
    scale = 0.004
    no_imp = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h_c[i] + d, h_c[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_c[i], h_c[j] = ni, nj
        c5n = _eval(h_c)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.03, 0.015)
        else:
            h_c[i], h_c[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 800:
                scale *= 0.7
                no_imp = 0
                if scale < 2e-7:
                    break
    return h_c, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    T_DEAD = t0 + budget_s - 3

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(h0)
    best_n = n0

    # Phase 1: SLSQP at n0 with increasing alpha
    for alpha in [500, 3000, 20000, 80000, 300000, 800000]:
        if time.time() > T_DEAD - 350:
            break
        h_new = _slsqp(best_h, best_n, alpha, maxiter=350)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Multi-resolution exploration
    for n_try in [140, 120, 100, 80]:
        if time.time() > T_DEAD - 250:
            break
        h_int = _repair(_interp(best_h, best_n, n_try), n_try)
        for alpha in [20000, 200000]:
            if time.time() > T_DEAD - 200:
                break
            h_new = _slsqp(h_int, n_try, alpha, maxiter=250)
            c5_new = _eval(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try

    # Phase 3: Random restarts from perturbed parent
    for restart in range(4):
        if time.time() > T_DEAD - 200:
            break
        pert = best_h.copy() + rng.randn(best_n) * 0.03
        pert = _repair(pert, best_n)
        c5_p = _eval(pert)
        if c5_p < best_c5 + 0.02:
            for alpha in [100000, 400000]:
                if time.time() > T_DEAD - 150:
                    break
                h_new = _slsqp(pert, best_n, alpha, maxiter=200)
                c5_new = _eval(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 4: Active-set optimization
    if time.time() < T_DEAD - 80:
        t_as = time.time() + min(60, T_DEAD - time.time() - 40)
        h_as, c5_as = _active_set_opt(best_h.copy(), best_n, t_as, n_rounds=8)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()

    # Phase 5: Simulated annealing
    if time.time() < T_DEAD - 40:
        t_sa = time.time() + min(100, T_DEAD - time.time() - 20)
        h_sa, c5_sa = _sa_search(best_h.copy(), best_n, rng, t_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Fine local search
    if time.time() < T_DEAD - 8:
        t_fs = time.time() + min(50, T_DEAD - time.time() - 4)
        h_f, c5_f = _fine_search(best_h.copy(), best_n, rng, t_fs)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: Final polish
    if time.time() < T_DEAD - 2:
        h_pol = _slsqp(best_h.copy(), best_n, 2000000, maxiter=200)
        c5_pol = _eval(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
