Actually, I want to add one more idea: **dual ascent**. Since C5 = max_k g_k(h), we can use a dual approach:
- Maintain weights w_k on the shifts
- The objective becomes sum_k w_k * g_k(h) (a quadratic form)
- Minimize this quadratic form over h in the polytope
- Update w_k based on which shifts are "hot"

A simpler version: **weighted sum optimization**. Instead of the smooth max, maintain a running average of the top shifts and minimize the weighted sum. This guides the optimizer more directly.

Let me try a different approach: **iterative reweighting**.

```
1. Compute c5_vals = dx * correlate(h, 1-h, 'full')
2. Find top M shifts (say M=5)
3. Form a quadratic objective: sum_{k in top M} w_k * g_k(h)
4. Minimize this quadratic (with constraints) using SLSQP or L-BFGS-B
5. Repeat
```

Since g_k(h) is quadratic in h, the weighted sum is also quadratic. We can minimize a quadratic over a polytope efficiently with QP solvers. But the quadratic is not necessarily convex (it's -h^T A h + linear, where A is the shift matrix), so it's a QP with a non-convex objective. SLSQP would handle it.

Actually, this is essentially what the smooth max already does, just with explicit weights instead of the softmax. The smooth max with high alpha approximates the max, which is equivalent to putting all weight on the single worst shift. The reweighting approach might help because it considers multiple near-worst shifts simultaneously.

def hitchhiking_cd(h, n, t_end, max_rounds=100):
    h = h.copy()
    best = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            # Partner: the coordinate farthest from i (to minimize interference)
            j = (i + n // 2) % n
            hi, hj = h[i], h[j]
            for d in deltas:
                for sgn in (1, -1):
                    ni, nj = hi + sgn * d, hj - sgn * d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = compute_c5(h, n)
                        if c < best - 1e-15:
                            best = c
                            improved = True
                            break
                        h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved:
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            tr = r.sum()
            if tr < 1e-15:
                break
            h += d * r / tr
        else:
            th = h.sum()
            if th < 1e-15:
                break
            h += d * h / th
        h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
    return h


def smax(corr, a):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(a * (corr - m), -500, 500)))) / a


def _obj_sm(v, a):
    return smax(np.correlate(v, 1.0 - v, mode='full'), a)


def _obj_pen(v, a, p, n):
    return smax(np.correlate(v, 1.0 - v, mode='full'), a) + p * (v.sum() - n / 2.0) ** 2


def _con(v, n):
    return v.sum() - n / 2.0


def opt_slsqp(h, n, a, mi=300):
    try:
        r = minimize(partial(_obj_sm, a=a), h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': partial(_con, n=n)}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return h


def opt_lbfgs(h, n, a, p, mi=2000):
    try:
        r = minimize(partial(_obj_pen, a=a, p=p, n=n), h,
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mi, 'ftol': 1e-17, 'gtol': 1e-11})
        return repair(r.x, n)
    except Exception:
        return h


def cd_random(h, n, te, rng, mr=1500):

    b = compute_c5(h, n)
    st = [0.05, 0.02, 0.008, 0.003, 0.001, 3e-4]
    sl = 0
    for r in range(mr):
        if time.time() > te:
            break
        imp = False
        for _ in range(5 * n):
            if time.time() > te:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            for d in st:
                dn = False
                for s in (1, -1):
                    ni, nj = hi + s * d, hj - s * d
                    if 0 <= ni <= 1.0 and 0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c = compute_c5(h, n)
                        if c < b - 1e-15:
                            b, imp, dn = c, True, True
                            break
                        h[i], h[j] = hi, hj
                if dn:
                    break
        if imp:
            sl = 0
        else:
            sl += 1
            if sl > 25:
                break
            st = [s * 0.5 for s in st]
    return h, b
