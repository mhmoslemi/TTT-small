import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, n):
    L = k - (n - 1)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def smooth_phase(h, n, t0, lim):
    bh, bc = h.copy(), c5val(h)
    bds = [(0.0, 1.0)] * n
    for alpha in [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]:
        if time.time() - t0 > lim:
            break
        def _obj(x):
            c = _corr(x)
            m = c.max()
            s = np.clip(alpha * (c - m), -500, 500)
            return m + np.log(np.sum(np.exp(s))) / alpha + 500.0 * (x.sum() - n * 0.5) ** 2
        def _jac(x):
            c = _corr(x)
            m = c.max()
            s = np.clip(alpha * (c - m), -500, 500)
            ew = np.exp(s)
            pw = ew / ew.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if pw[k] > 1e-15:
                    g += pw[k] * _grad_lag(x, k, n)
            g /= alpha
            g += 1000.0 * (x.sum() - n * 0.5)
            return g
        try:
            r = minimize(_obj, bh.copy(), jac=_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': 600, 'ftol': 1e-20, 'gtol': 1e-18})
            hc = repair(r.x, n)
            c = c5val(hc)
            if c < bc - 1e-16:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def top_phase(h, n, t0, lim):
    bh, bc = h.copy(), c5val(h)
    for mt in [2, 3, 5, 7, 10, 15]:
        if time.time() - t0 > lim:
            break
        h = bh.copy()
        step = 0.06
        stalled = 0
        for _ in range(400):
            if time.time() - t0 > lim:
                break
            c = _corr(h)
            cm = c.max()
            thr = cm - 0.004 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for k in idx:
                g += _grad_lag(h, k, n)
            g /= len(idx)
            g -= g.mean()
            ok = False
            for _b in range(35):
                hn = repair(h - step * g, n)
                cn = c5val(hn)
                if cn < c5val(h) - 1e-18:
                    h = hn
                    ok = True
                    if cn < bc - 1e-16:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.1, 1.2)
                    stalled = 0
                    break
                step *= 0.7
            if not ok:
                stalled += 1
                if stalled > 10:
                    break
    return bh, bc

def coord_phase(h, n, rng, t0, lim):
    bh, bc = h.copy(), c5val(h)
    cc = bc
    halts = 1
    while halts < 5 and time.time() - t0 < lim:
        halts = 0
        improved = False
        for _ in range(800):
            if time.time() - t0 > lim:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if rng.rand() < 0.5:
                d = 0.008
            else:
                d = -0.008
            if rng.rand() < 0.3:
                d = 0.02 * (rng.rand() - 0.5)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = c5val(h)
                if c < cc - 1e-15:
                    h = repair(h, n)
                    cc = c5val(h)
                    improved = True
                    if cc < bc - 1e-16:
                        bc, bh = cc, h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        halts += 0 if improved else 1
    return bh, bc

def subgrad_phase(h, n, t0, lim):
    bh, bc = h.copy(), c5val(h)
    h = bh.copy()
    avg = h.copy()
    eta = 0.3
    for it in range(3000):
        if time.time() - t0 > lim:
            break
        c = _corr(h)
        km = int(c.argmax())
        g = _grad_lag(h, km, n)
        h = np.clip(h - eta * g, 0.0, 1.0)
        h = repair(h, n)
        avg += h
        eta *= 0.999
        if it % 50 == 0:
            ha = repair(avg / (it // 50 + 1), n)
            ca = c5val(ha)
            if ca < bc - 1e-16:
                bc, bh = ca, ha
    return bh, bc

def multi_start(h, n, seed, t0, lim):
    bh, bc = h.copy(), c5val(h)
    rng = np.random.RandomState(seed)
    for scale in [0.03, 0.06, 0.1, 0.04]:
        if time.time() - t0 > lim:
            break
        hp = repair(h + scale * rng.randn(n), n)
        r, c = smooth_phase(hp, n, t0, lim - 60)
        if c < bc - 1e-16:
            bc, bh = c, r
    return bh, bc
