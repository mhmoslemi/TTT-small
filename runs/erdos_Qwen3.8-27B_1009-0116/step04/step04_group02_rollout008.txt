import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lsq_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def raw_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lsq_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_step(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(raw_obj, h, args=(alpha,), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def exact_pair(h, i, j, dx, best_c5, n_it=22):
    d_lo = max(0.0, -h[i])
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-14:
        return False
    si, sj = h[i], h[j]
    lo, hi = d_lo, d_hi
    for _ in range(n_it):
        if hi - lo < 1e-15:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h[i], h[j] = si + m1, sj - m1
        c1 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
        h[i], h[j] = si + m2, sj - m2
        c2 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
        if c1 < c2:
            hi = m2
        else:
            lo = m1
        if abs(c1 - c2) < 1e-16:
            break
    d_star = (lo + hi) * 0.5
    h[i], h[j] = si + d_star, sj - d_star
    c_s = compute_c5(h, dx)
    if c_s < best_c5 - 1e-15:
        return True
    h[i], h[j] = si, sj
    return False

def structured_pairs(h, n, dx, rng, deadline, passes, n_it):
    best_c5 = compute_c5(h, dx)
    for p in range(passes):
        if time.time() >= deadline:
            break
        gaps = [1, 2, 3, 4, 6, 9, 14, 22, 35, 55, 70, 90, 110, 130]
        pairs = []

            for i in range(n - g):
                pairs.append((i, i + g))
        rng.shuffle(pairs)
        for (i, j) in pairs:
            if time.time() >= deadline:
                break
            if exact_pair(h, i, j, dx, best_c5, n_it):
                best_c5 = compute_c5(h, dx)
    return h, best_c5

def random_pairs(h, n, dx, rng, deadline, n_trials, n_it):
    best_c5 = compute_c5(h, dx)
    for _ in range(n_trials):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if exact_pair(h, i, j, dx, best_c5, n_it):
            best_c5 = compute_c5(h, dx)
    return h, best_c5

def block_search(h, n, dx, rng, deadline, n_trials):

            break

            continue
        e_max = min(np.min(h[i:i+bs]), np.min(1.0 - h[j:j+bs]))
        if e_max < 1e-11:
            continue
        sv_i, sv_j = h[i:i+bs].copy(), h[j:j+bs].copy()
        lo, hi = 0.0, e_max

                break

            h[j:j+bs] = sv_j - m1
            c1 = compute_c5(h, dx)
            h[i:i+bs] = sv_i + m2
            h[j:j+bs] = sv_j - m2
            c2 = compute_c5(h, dx)
            if c1 < c2:
                hi = m2
            else:
                lo = m1
        d_s = (lo + hi) * 0.5
        h[i:i+bs] = sv_i + d_s
        h[j:j+bs] = sv_j - d_s
        c_s = compute_c5(h, dx)

        else:
            h[i:i+bs] = sv_i
            h[j:j+bs] = sv_j
    return h, best_c5
