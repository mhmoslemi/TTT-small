    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

```python
def _precompute_effects(h, n):
    """For each index i, compute f_i[k] = effect of unit increase in h[i] on corr[k]."""
    # corr[n-1+lag] = sum_m h[m] * (1-h[m-lag]) for valid m, m-lag
    # Effect of h[i] += 1:
    #   As first factor (m=i): corr[n-1+lag] += (1-h[i-lag]) for 0 ≤ i-lag < n
    #   As second factor (m-lag=i, i.e., m=i+lag): corr[n-1+lag] -= h[i+lag] for 0 ≤ i+lag < n
    
    F = np.zeros((n, 2*n-1))  # F[i, k] = f_i[k]
    for i in range(n):
        # First factor: lag from i-n+1 to i (but clamped to [-(n-1), n-1])
        lo = max(i - n + 1, -(n-1))
        hi = min(i, n-1)
        lags = np.arange(lo, hi+1)
        idxs = i - lags  # these are valid indices in [0, n-1]
        F[i, n-1+lags] += (1.0 - h[idxs])
        
        # Second factor: lag from -i to n-1-i (clamped)
        lo = max(-i, -(n-1))
        hi = min(n-1-i, n-1)
        lags = np.arange(lo, hi+1)
        idxs = i + lags
        F[i, n-1+lags] -= h[idxs]
    
    return F
```

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _project(h, n):

        else:

                             bounds=bds, constraints=[{'type':'eq','fun':_scfun,'jac':_scjac}],

                hc = _repair(res.x, n); c = _c5(hc, n)
                if c < bc: bc = c; bh = hc.copy()
            except: pass
    return bh, bc

                # also try half-step
