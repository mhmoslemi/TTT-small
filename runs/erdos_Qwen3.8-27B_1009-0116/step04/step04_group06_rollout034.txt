The key innovation I want to add is: **subgradient method with the exact max** (not smoothed). At each step:
1. Find which lag k* achieves the max

4. Project onto the constraint set

This avoids the approximation error of the smooth max.

```python
def _exact_subgrad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    # Find active lags (within tolerance)
    tol = 1e-6 * m
    active = c >= m - tol
    # Average gradient over active lags
    g = np.zeros(n)
    count = 0
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        count += 1
        lag = k - (n - 1)
        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            l2 = -lag
            g[l2:] += (1.0 - h[:n - l2])
            g[:n - l2] -= h[l2:]
    g /= count
    return g * (2.0 / n)
```

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _grad_lag(h, n, active_mask):
    dx = 2.0 / n
    g = np.zeros(n)
    counts = 0
    for k in range(2 * n - 1):
        if not active_mask[k]:
            continue
        counts += 1
        lag = k - (n - 1)
        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            l2 = -lag
            g[l2:] += (1.0 - h[:n - l2])
            g[:n - l2] -= h[l2:]
    if counts > 0:
        g /= counts
    return g * dx

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - lag] += wk * (1.0 - x[lag:])
            g[lag:] -= wk * x[:n - lag]
        else:
            l2 = -lag
            g[l2:] += wk * (1.0 - x[:n - l2])
            g[:n - l2] -= wk * x[l2:]
    return g * dx

def _nest_pgd(h, n, alpha, end_t, lr=0.02):
    h = h.copy()
    y = h.copy()
    t = 1.0
    c5_cur = _c5(h, n)
    stalled = 0
    for it in range(100000):
        if time.time() > end_t:
            break
        g = _smooth_grad(y, n, alpha)
        y_new = _project(h - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = y_new + (t / t_new - 1.0) * (y_new - h)
        h = y_new
        t = t_new
        if it % 200 == 199:
            c5_new = _c5(h, n)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                stalled = 0
                lr = min(lr * 1.05, 0.08)
            else:
                stalled += 1
                lr *= 0.92
            if stalled > 15:
                break
    return h, _c5(h, n)

def _subgrad_pgd(h, n, end_t, lr=0.01):
    h = h.copy()
    c5_cur = _c5(h, n)
    stalled = 0
    for it in range(100000):
        if time.time() > end_t:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        active = c >= m - 1e-8 * max(m, 1e-10)
        g = _grad_lag(h, n, active)
        h = _project(h - lr * g, n)
        if it % 100 == 99:
            c5_new = _c5(h, n)
            if c5_new < c5_cur - 1e-13:
                c5_cur = c5_new
                stalled = 0
                lr = min(lr * 1.03, 0.05)
            else:
                stalled += 1
                lr *= 0.9
            if stalled > 25:
                break
    return h, _c5(h, n)

def _coord_desc(h, n, end_t):
    h = h.copy()
    c5_cur = _c5(h, n)
    deltas = np.array([0.005, 0.01, 0.02, 0.05, 0.1, -0.005, -0.01, -0.02, -0.05, -0.1])
    for _pass in range(500):
        if time.time() > end_t:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        active = c >= m - 0.08 * m
        g = _grad_lag(h, n, active)
        order = np.argsort(g)
        improved = False
        for i in order:
            if time.time() > end_t:
                break
            hi = h[i]
            for d in deltas:
                ni = hi + d
                if ni < 0 or ni > 1:
                    continue
                h[i] = ni
                c5_new = _c5(h, n)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    hi = ni
                    improved = True
                    break
                else:
                    h[i] = hi
        if not improved:
            break
    return h, c5_cur

def _slsqp_run(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def _cf(x):
        return float(np.sum(x) - target)
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    res = minimize(
        _smooth_obj, h_init, args=(n, alpha),
        jac=lambda x: _smooth_grad(x, n, alpha),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-20})
    return res.x

def _binary_init(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h

def _resample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return _repair(h_new, n_new)
