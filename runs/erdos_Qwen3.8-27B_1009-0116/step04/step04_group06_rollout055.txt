def _optimal_pair_step(h, i, j, n):
    """Find optimal d for transfer h[i]+=d, h[j]-=d that minimizes max_k(c_k + d*g_k)"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Compute g_k = dc_k/dh[i] - dc_k/dh[j]
    g = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        lag = k - (n - 1)
        # dc/dh[i] for this lag
        di = 0.0
        dj = 0.0
        if lag == 0:
            di = 1.0 - 2.0 * h[i]
            dj = 1.0 - 2.0 * h[j]
        elif lag > 0:
            if i < n - lag:
                di += 1.0 - h[i + lag]
            if i >= lag:
                di -= h[i - lag]
            if j < n - lag:
                dj += 1.0 - h[j + lag]
            if j >= lag:
                dj -= h[j - lag]
        else:
            l2 = -lag
            if i >= l2:
                di += 1.0 - h[i - l2]
            if i < n - l2:
                di -= h[i + l2]
            if j >= l2:
                dj += 1.0 - h[j - l2]
            if j < n - l2:
                dj -= h[j + l2]
        g[k] = di - dj
    # Now minimize max_k(c[k] + d*g[k]) over valid d range
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    if d_min >= d_max:
        return 0.0
    # Ternary search on [d_min, d_max]
    def f(d):
        return np.max(c + d * g)
    a, b = d_min, d_max
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    for _ in range(50):
        m1 = b - gr * (b - a)
        m2 = a + gr * (b - a)
        f1, f2 = f(m1), f(m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return 0.5 * (a + b)

def _c5_val(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _corr_vals(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

    g = np.zeros(n)

Wait, there's an issue with updating c incrementally. When I do h[i] += d, the new c is NOT simply c + d * gk because gk was computed at the OLD h. The gradient changes when h changes. So I need to recompute c after each move.

Actually, the issue is more subtle. Even the ternary search is approximate because c changes after each step. But for small steps, the linear approximation c + d*gk is good enough. And if the step is large, we recompute after.

Let me restructure to be cleaner and more efficient:

For the pair move, instead of ternary search (which requires many c evaluations), I can use the fact that for a pair move, c_k(d) = c_k(0) + d * gk[k] is EXACT (linear in d) because we're only changing two variables. Wait, is that true?

c_k = sum_i h[i+lag] * (1-h[i])

If I change h[i1] by +d and h[i2] by -d:
- h[i1] appears in c_k in at most 2 terms (as the "first" and "second" factor)
- h[i2] appears in c_k in at most 2 terms

So c_k(d) = c_k(0) + d * (contributions from h[i1]) - d * (contributions from h[i2])

The "contributions from h[i1]" to c_k is:
- If i1 is in the "first factor" position: (1 - h[i1 + lag])
- If i1 is in the "second factor" position: -h[i1 - lag] (with sign)

These are evaluated at the CURRENT h (with h[i1] at its current value, and h[i2] also at its current value). But wait, when we change h[i1] by d, does it affect the "contributions from h[i2]"? Only if i1 and i2 are "linked" through the same c_k term. 

Specifically, in c_k = sum_i h[i+lag](1-h[i]), the term h[i1](1-h[i1+lag]) involves both h[i1] and h[i1+lag]. If i2 = i1+lag, then changing both h[i1] and h[i2] means the term becomes (h[i1]+d)(1-(h[i2]-d)) = (h[i1]+d)(1-h[i2]+d) = h[i1](1-h[i2]) + d(1-h[i2]) + d*h[i1] + d².

So there's a d² term! The change is NOT linear in d in general!

This happens when i2 = i1 + lag or i2 = i1 - lag for some lag corresponding to some k.

For a given pair (i, j), the non-linearity occurs when |i - j| corresponds to some lag that appears in the correlation. Since lags range from -(n-1) to (n-1), and |i-j| < n (since both are in [0, n-1]), the non-linearity occurs for the specific k where lag = j - i or lag = i - j.

So for most k, the change IS linear. Only for k = (j-i) + (n-1) and k = (i-j) + (n-1) is there a quadratic term.

This means the ternary search is still valid (the function max_k(c_k + d*gk[k] + d²*quadratic_term) is still convex in d), but my linear approximation is slightly off for 2 specific k values.

For the purposes of finding a good step, the linear approximation should be close enough. And we verify by recomputing the actual c after the move.

def _sys_pairs(h, n, end_t, rng):

def _slsqp(h0, n, alpha, maxit):
    tgt = n*0.5
    def f(x):
        c=_corr(x,n); m=c.max()
        s=np.clip(alpha*(c-m),-400,400)
        return m+np.log(np.sum(np.exp(s)))/alpha
    def jf(x):
        c=_corr(x,n); m=c.max()
        s=np.clip(alpha*(c-m),-400,400)
        e=np.exp(s); w=e/e.sum()
        g=np.zeros(n); nx=2*n-1; dxv=2.0/n
        for k in range(nx):
            wk=w[k]
            if wk<1e-14: continue
            lag=k-(n-1)
            if lag==0: g+=wk*(1.0-2.0*x)
            elif lag>0:
                g[lag:]+=wk*(1.0-x[:n-lag]); g[:n-lag]-=wk*x[lag:]
            else:
                l2=-lag; g[:n-l2]+=wk*(1.0-x[l2:]); g[l2:]-=wk*x[:n-l2]
        return g*dxv
    res = minimize(f, h0, jac=jf, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':lambda x: x.sum()-tgt,'jac':lambda x: np.ones(n)}],
                   options={'maxiter':maxit,'ftol':1e-18})
    return _repair(res.x, n)
