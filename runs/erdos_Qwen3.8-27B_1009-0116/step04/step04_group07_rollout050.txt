                cn += h[xm] * (1.0 - h[x])
            if cn > best_cn:
                best_cn = cn
                best_x = x
        if best_cn < 1e-10 or best_x < 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
        else:
            s_val = s
            if 0 <= best_x + s_val < n:
                i, j = best_x, best_x + s_val
            elif 0 <= best_x - s_val < n:
                i, j = best_x - s_val, best_x
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(n - 1)) % n
        d = exact_2var_delta(h, n, i, j, shifts, corr)
        if abs(d) < 1e-13:
            no_imp += 1
        else:
            hi_b, hj_b = h[i], h[j]
            h[i] = np.clip(hi_b + d, 0.0, 1.0)
            h[j] = np.clip(hj_b - d, 0.0, 1.0)
            c5 = compute_c5(h, n)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                no_imp = 0
            else:
                h[i], h[j] = hi_b, hj_b
                no_imp += 1
        if no_imp > 80000:
            break
    return best_h, best_c5

def resample(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return repair(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 64
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(init_h)
        best_h = repair(init_h, best_n)
        best_c5 = compute_c5(best_h, best_n)
    except NameError:
        best_n = 64
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    # Phase 1: Exact local search (35% budget)
    t1 = t0 + budget_s * 0.30
    h1, c5_1 = local_search(best_h, best_n, rng, t1)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Targeted search on worst lag (15% budget)
    t2 = t0 + budget_s * 0.45
    rng2 = np.random.RandomState(seed + 1000)
    h2, c5_2 = targeted_search(best_h, best_n, rng2, t2)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: More local search with different seed (15% budget)
    t3 = t0 + budget_s * 0.60
    rng3 = np.random.RandomState(seed + 9999)
    h3, c5_3 = local_search(best_h, best_n, rng3, t3, kick_interval=20000, kick_sigma=0.03)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Try alternative n values (20% budget)
    t4 = t0 + budget_s * 0.80
    for n_alt in [50, 56, 60, 72, 80, 96, 100, 128]:
        if time.time() > t4:
            break
        if n_alt == best_n:
            continue
        time_per_n = 45.0
        h_seed = resample(best_h, best_n, n_alt)
        c5_seed = compute_c5(h_seed, n_alt)
        rng_alt = np.random.RandomState(seed + n_alt * 31)
        t_alt = min(t4, time.time() + time_per_n)
        h_opt, c5_opt = local_search(h_seed, n_alt, rng_alt, t_alt, kick_interval=15000)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_alt

    # Phase 5: Final polish (remaining budget)
    t5 = t0 + budget_s * 0.995
    rng5 = np.random.RandomState(seed + 424242)
    h5, c5_5 = local_search(best_h, best_n, rng5, t5, kick_interval=40000, kick_sigma=0.02)
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5

    # Final validation and repair
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
