            grad[:n - lag] -= wl * h[lag:]
        else:
            ll = -lag
            grad[:n - ll] += wl * (1.0 - h[ll:])
            grad[ll:] -= wl * h[:n - ll]
    return grad

def _pgd(h, target, alpha_list, step, iters_per_alpha, rng, t_end):
    best = _c5(h)
    best_h = h.copy()
    h = h.copy()
    n = len(h)
    for alpha in alpha_list:
        for it in range(iters_per_alpha):
            if time.time() > t_end:
                return best_h, best
            g = _grad_softmax(h, alpha)
            lr = step
            improved = False
            for _ in range(20):
                h_try = _proj_sum(h - lr * g, target)
                c_try = _c5(h_try)
                if c_try < best or c_try < _c5(h):
                    h = h_try
                    if c_try < best:
                        best = c_try
                        best_h = h.copy()
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                h = _proj_sum(h + 0.02 * (2.0 * rng.random(n) - 1.0), target)
    return best_h, best

def _coord_descent(h, target, rng, t_end, n_iter):
    best = _c5(h)
    best_h = h.copy()
    n = len(h)
    for it in range(n_iter):
        if time.time() > t_end:
            return best_h, best
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.05 * (2.0 * rng.random() - 1.0)
        h[i] += d
        h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c = _c5(h)
            if c < best:
                best = c
                best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return best_h, best

def _lag_focused(h, target, rng, t_end, n_iter):
    best = _c5(h)
    best_h = h.copy()
    n = len(h)
    for it in range(n_iter):
        if time.time() > t_end:
            return best_h, best
        c = np.correlate(h, 1.0 - h, mode='full')
        top_lag = np.argmax(c)
        lag = top_lag - (n - 1)
        if lag == 0:
            i = rng.randint(n)
            j = rng.randint(n)
        else:
            if lag > 0:
                pool = list(range(n - lag))
            else:
                pool = list(range(-lag, n))
            if not pool:
                i = rng.randint(n)
                j = rng.randint(n)
            else:
                i = pool[rng.randint(len(pool))]
                j = i + lag
        d = 0.03 * (2.0 * rng.random() - 1.0)
        h[i] += d
        h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c_new = _c5(_proj_sum(h, target))
            if c_new < best:
                best = c_new
                best_h = _proj_sum(h, target)
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return best_h, best

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    tgt0 = n0 / 2.0
    best_h, best_c = h0.copy(), _c5(h0)
    best_n = n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Projected gradient at base res with annealing (200s)
    t_end = t0 + 180
    alphas = [100, 300, 1000, 3000, 10000, 30000]
    for a in alphas:
        if rem() < 40:
            break
        h0, c = _pgd(h0, tgt0, [a], max(0.0005, 1.0 / a * 0.1),
                     max(200, 15000 // a), rng, t_end)
        if c < best_c:
            best_h, best_c = h0, c

    # Phase 2: Fine PGD with high alpha (150s)
    t_end = min(t0 + 330, t0 + budget_s - 500)
    for a in [50000, 100000, 200000, 500000]:
        if rem() < 30:
            break
        best_h, best_c = _pgd(best_h, tgt0, [a], 0.5 / a, 2000, rng, t_end)

    # Phase 3: Multi-restart with perturbations (200s)
    t_end = min(t0 + 530, t0 + budget_s - 300)
    for r in range(15):
        if rem() < 80:
            break
        hp = best_h.copy()
        amp = 0.12 * (1.0 - r / 18.0)
        f = 1.0 + r * 0.5
        x = np.linspace(0, 2, n0)
        hp += amp * np.sin(2.0 * np.pi * f * x / 2.0)
        hp = _repair(hp, tgt0)
        hc, cc = _pgd(hp, tgt0, [5000, 20000], 0.0002, 1500, rng, t_end)
        if cc < best_c:
            best_h, best_c = hc, cc

    # Phase 4: Interpolate to 2n and optimize
    if rem() > 300:
        n1 = 2 * n0
        tgt1 = n1 / 2.0
        h1 = _repair(_interp1d(best_h, n1), tgt1)
        c1 = _c5(h1)
        t_end = min(t0 + 830, t0 + budget_s - 200)
        for a in [2000, 10000, 50000, 200000]:
            if rem() < 20:
                break
            h1, c1 = _pgd(h1, tgt1, [a], 0.5 / a, 1500, rng, t_end)
        h1, c1 = _coord_descent(h1, tgt1, rng, min(t_end, t0 + budget_s - 100), 300000)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, n1

    # Phase 5: Lag-focused search on best
    dxb = 2.0 / best_n
    tgt_b = best_n / 2.0
    t_end = min(t0 + budget_s - 60, t0 + budget_s - 60)
    best_h, best_c = _lag_focused(best_h, tgt_b, rng, t_end, 400000)

    # Phase 6: Final coord descent + tiny PGD
    t_end = t0 + budget_s - 15
    best_h, best_c = _coord_descent(best_h, tgt_b, rng, t_end, 800000)
    if rem() > 30:
        best_h, best_c = _pgd(best_h, tgt_b, [500000], 0.000002, 5000, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, best_n / 2.0)
    best_c = _c5(best_h)
    return (best_h, best_c, best_n)
