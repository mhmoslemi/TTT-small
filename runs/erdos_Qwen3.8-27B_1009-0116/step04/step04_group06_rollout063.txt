import numpy as np
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _corr_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                s = min(-diff, h[idx])
                h[idx] -= s
                diff += s
    return h

def _compute_S(h, j, n):
    S = np.zeros(2 * n - 1)
    # First term: m' in [j, j+n-1], S[m'] += (1 - h[j+n-1-m'])
    for mp in range(j, j + n):
        S[mp] += (1.0 - h[j + n - 1 - mp])
    # Second term: m' in [n-1-j, 2n-j-2], S[m'] -= h[j-n+1+mp]
    for mp in range(n - 1 - j, 2 * n - j - 1):
        S[mp] -= h[j - n + 1 + mp]
    return S

def _opt_coord(h, j, n, out_vals):
    """Find optimal h[j] given others fixed. out_vals = current correlation (before dx)."""
    hj = h[j]
    S = _compute_S(h, j, n)
    dx = 2.0 / n
    
    # Ternary search on t in [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (out_vals + (m1 - hj) * S).max()
        f2 = (out_vals + (m2 - hj) * S).max()
        if f1 > f2:
            lo = m1
        else:
            hi = m2
    t = (lo + hi) / 2.0
    return t

def _coord_descent(h, n, max_passes=5000, tol=1e-12):
    for _pass in range(max_passes):
        out_vals = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for j in range(n):
            t_opt = _opt_coord(h, j, n, out_vals)
            if abs(t_opt - h[j]) > 1e-10:
                h[j] = t_opt
                improved = True
        if not improved:
            break
    return h

def _pair_opt(h, i, j, n, out_vals):
    """Optimize h[i] and h[j] jointly, keeping h[i]+h[j] constant."""
    si = h[i] + h[j]
    # Parametrize: h[i]=t, h[j]=si-t, t in [max(0,si-1), min(1,si)]
    lo = max(0.0, si - 1.0)
    hi = min(1.0, si)
    if lo >= hi - 1e-15:
        return h[i], h[j]
    
    # Change in out_vals when h[i]->t, h[j]->si-t:
    # Delta[m'] = (t-h[i])*S_i[m'] + ((si-t)-h[j])*S_j[m']
    #           = (t-h[i])*S_i[m'] + (h[i]-t)*S_j[m']

import numpy as np
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                s = min(-diff, h[idx])
                h[idx] -= s
                diff += s
    return h

def _S(h, j, n):
    S = np.zeros(2 * n - 1)
    S[j:j + n] += 1.0 - h[::-1]
    S[n - 1 - j:2 * n - j - 1] -= h
    return S

def _ternary_1d(out_v, S, hj, n_it=40):
    lo, hi = 0.0, 1.0
    for _ in range(n_it):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if (out_v + (m1 - hj) * S).max() > (out_v + (m2 - hj) * S).max():
            lo = m1
        else:
            hi = m2
    return (lo + hi) * 0.5

def _coord_descent(h, n, max_passes=10000, time_limit=None, order=None):
    n_range = list(range(n))
    for _pass in range(max_passes):
        if time_limit is not None and time.time() > time_limit:
            break
        out_v = _corr_raw(h).astype(float)
        any_ch = False
        for j in n_range:
            S = _S(h, j, n)
            hj = h[j]
            t = _ternary_1d(out_v, S, hj)
            if abs(t - hj) > 1e-11:
                out_v = out_v + (t - hj) * S
                h[j] = t
                any_ch = True
        if not any_ch:
            break

    c5_cur = _c5(h, n)
    step = 0
