def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

    return float(np.sum(x) - _GN / 2.0)

    return np.ones(_GN)

    global _GN
    _GN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:

                       jac=_smooth_grad, method='SLSQP',

    c5_cur = _c5_val(h, n)
    for it in range(max_iters):

    c5_cur = _c5_val(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T0
    n_total = 2 * n - 1

    it = 0

        it += 1
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_min = max(-h[i], h[j] - 1.0)
        d_max = min(1.0 - h[i], h[j])
        if d_max - d_min < 1e-10:

import numpy as np
from scipy.optimize import minimize
import time

def _c5_val(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

_GN = 140

def _con_fun(x):
    return float(np.sum(x) - _GN / 2.0)

def _con_jac(x):
    return np.ones(_GN)

def _slsqp(h_init, n, alpha, maxiter):
    global _GN
    _GN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smooth_obj, h_init, args=(n, alpha),
                       jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h_init

def _pgd(h, n, alpha, time_end, lr_init, n_iters):
    h = h.copy()
    lr = lr_init
    c5 = _c5_val(h, n)
    for it in range(n_iters):
        if time.time() > time_end:
            break
        g = _smooth_grad(h, n, alpha)
        h_new = _project(h - lr * g, n)
        c5_new = _c5_val(h_new, n)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            lr *= 0.7
        if it % 400 == 399:
            lr *= 0.75
    return h, c5

def _pair_pass(h, n, rng, time_end, n_frac=1.0):
    c5 = _c5_val(h, n)
    improved = True
    total_pairs = n * (n - 1) // 2
    max_pairs_per_pass = int(total_pairs * n_frac)
    fracs = np.array([0.0, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9,
                      0.98, 1.0, -0.02, -0.05, -0.1, -0.2, -0.35, -0.5, -0.65,
                      -0.8, -0.9, -0.98, -1.0])
    while improved and time.time() < time_end:
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        count = 0
        for ii in range(n):
            if time.time() > time_end or count >= max_pairs_per_pass:
                break
            i = idx[ii]
            for jj in range(ii + 1, n):
                if count >= max_pairs_per_pass:
                    break
                j = idx[jj]
                count += 1
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                span = d_hi - d_lo
                if span < 1e-10:
                    continue
                best_d = 0.0
                best_c = c5
                for f in fracs:
                    d = d_lo + f * span
                    h[i] += d
                    h[j] -= d
                    c = _c5_val(h, n)
                    if c < best_c:
                        best_c = c
                        best_d = d
                    h[i] -= d
                    h[j] += d
                if best_d != 0.0 and best_c < c5 - 1e-15:
                    # Refine around best_d
                    span2 = span * 0.05
                    bestd_ref = best_d
                    bestc_ref = best_c
                    for f2 in np.linspace(-1, 1, 7):
                        d2 = best_d + f2 * span2
                        if d2 < d_lo or d2 > d_hi:
                            continue
                        h[i] += d2
                        h[j] -= d2
                        c2 = _c5_val(h, n)
                        if c2 < bestc_ref:
                            bestc_ref = c2
                            bestd_ref = d2
                        h[i] -= d2
                        h[j] += d2
                    if bestc_ref < c5 - 1e-15:
                        h[i] += bestd_ref
                        h[j] -= bestd_ref
                        c5 = bestc_ref
                        improved = True
                    else:
                        h[i] += best_d
                        h[j] -= best_d
                        c5 = best_c
                        improved = True
    return h, c5

def _sa_optimize(h, n, rng, time_end, T0=0.02, T_min=0.001):
    h = h.copy()

    best_h = h.copy()
