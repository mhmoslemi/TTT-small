**This is the key insight!** The minimax decomposition allows us to solve a sequence of quadratic problems.

Algorithm:
1. Initialize λ (e.g., uniform or based on current h)
2. Inner: minimize Q(h) = Σ_k λ_k f_k(h) subject to constraints
   - This is a QP! Use SLSQP or projected gradient
3. Outer: update λ based on which k achieves the maximum
   - λ_k = exp(α * f_k(h)) / Σ_j exp(α * f_j(h)) (softmax)
   - Or: λ_k = 1 if k = argmax, 0 otherwise (hard)
4. Repeat

This is essentially **Fenchel dual / mirror descent in the dual**, and it's a standard approach for min-max problems.

```python
def inner_qp(h_init, lam, n, maxiter):
    """Minimize sum_k lam[k] * f_k(h) subject to constraints"""
    target = n / 2.0
    dx = 2.0 / n
    
    def qp_obj(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        return np.dot(lam, c) * dx
    
    def qp_grad(h):
        # grad = dx * (1 - sum_k lam[k] * (shifted h + shifted h))
        c = np.correlate(h, 1.0 - h, mode='full')
        # Actually, let's compute the gradient directly
        # f_k(h) = sum_i h[i] * (1 - h[(i + sh_k) mod stuff])
        # df_k/dh[i] = (1 - h[i+sh]) - h[i-sh] (with boundary)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if lam[k] < 1e-15:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += lam[k] * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= lam[k] * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += lam[k] * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= lam[k] * h[i2 + sp]
        g *= dx
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(qp_obj, h_init, jac=qp_grad,
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return repair(res.x, n)
```

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                    diff -= a
