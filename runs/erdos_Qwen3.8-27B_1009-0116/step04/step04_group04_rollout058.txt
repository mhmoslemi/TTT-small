import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            return h
        if d > 0:
            m = np.where(h < 1.0 - 1e-12)[0]
        else:
            m = np.where(h > 1e-12)[0]
        if len(m) == 0:
            return h
        h[m] = np.clip(h[m] + d / len(m), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            if abs(h.sum() - target) < 1e-13:
                break
    return h

def _soft_obj(x, alpha, pen, tgt):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - tgt) ** 2

def _slsqp(h, alpha, maxit):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, 0, tgt), h, method='SLSQP',
                     bounds=bds, constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxit, 'ftol': 1e-15})
        return _repair(r.x, tgt)
    except Exception:
        return h

def _lbfgs(h, alpha, maxit):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best = h.copy()
    bc = _c5(best, 2.0 / n)
    for pen in [5, 50, 500, 5000, 50000]:
        try:
            r = minimize(lambda x: _soft_obj(x, alpha, pen, tgt), best, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-8})
            hc = _repair(r.x, tgt)
            c = _c5(hc, 2.0 / n)
            if c < bc:
                best, bc = hc, c
        except Exception:
            pass
    return best, bc

def _subgrad(h, niter, t_end, rng):
    n = len(h)
    tgt = n / 2.0
    dx = 2.0 / n
    best = _c5(h, dx)
    h = h.copy()
    step = 0.012
    prev_best = best
    stall = 0
    for it in range(niter):
        if time.time() > t_end:
            break
        c = _corr(h)
        m = c.max()
        thr = np.percentile(c, 82)
        act = np.where(c >= thr)[0]
        g = np.zeros(n)
        for k in act:
            sh = k - (n - 1)
            if sh == 0:
                g[:] += 1.0 - 2.0 * h
            elif sh > 0:
                g[:n - sh] += (1.0 - h[sh:])
                g[sh:] += (-h[:n - sh])
            else:
                s = -sh
                g[s:] += (1.0 - h[:n - s])
                g[:n - s] += (-h[s:])
        g /= max(1, len(act))
        h_new = _repair(h - step * g, tgt)
        c2 = _c5(h_new, dx)
        if c2 < best - 1e-12:
            best = c2
            h = h_new
            step = min(step * 1.06, 0.10)
            stall = 0
        else:
            step *= 0.94
            stall += 1
            if stall > 80:
                step = 0.03
                stall = 0
                h = _repair(h + 0.01 * (2.0 * rng.random(n) - 1.0), tgt)
                prev_best = best
    return h, best

def _struct_local(h, dx, rng, t_end, niter):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002]
    per = max(15, niter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return h, best
            c = _corr(h)
            kmax = int(c.argmax())
            sh = kmax - (n - 1)
            if sh > 0:
                pool = np.arange(n - sh)
            elif sh < 0:
                pool = np.arange(-sh, n)
            else:
                pool = np.arange(n)
            i = int(pool[rng.randint(len(pool))])
            if sh != 0:
                j = i + abs(sh)
                if j >= n:
                    j = int(rng.randint(n))
            else:
                j = int(rng.randint(n))
            if j == i:
                j = (j + 1) % n
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d
            h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                c2 = _c5(h, dx)
                if c2 < best:
                    best = c2
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best

def _sim_anneal(h, dx, rng, t_end, niter):
    n = len(h)
    best = _c5(h, dx)
    cur = best
    h = h.copy()
    T = 0.004
    for it in range(niter):
        if time.time() > t_end:
            break
        T = max(T * 0.985, 1e-6)
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        d = rng.uniform(-0.015, 0.015)
        h[i] += d
        h[j] -= d
        if not (0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0):
            h[i] -= d
            h[j] += d
            continue
        c2 = _c5(h, dx)
        delta = c2 - cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur = c2
            if cur < best:
                best = cur
        else:
            h[i] -= d
            h[j] += d
    return h, best
