import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a_ = min(d, 1.0 - h[i])
            else:
                a_ = max(d, -h[i])
            h[i] += a_
            d -= a_
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def smo(x, n, a):
    c = get_corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / a

def smj(x, n, a):
    c = get_corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -400, 400)
    ew = np.exp(s - s.max())
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-15:
            g += pw[k] * grad_corr(x, k, n)
    return g / a

def lbfgs_go(h, n, a, maxit, t0, lim):
    bds = [(0.0, 1.0)] * n
    def f(x):
        return smo(x, n, a) + 10.0 * (x.sum() - n * 0.5) ** 2
    def jf(x):
        return smj(x, n, a) + 20.0 * (x.sum() - n * 0.5)
    try:
        r = minimize(f, h.copy(), jac=jf, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14})
        hc = repair(r.x, n)
        return hc
    except Exception:
        return h

def active_sg(h, n, dx, t0, lim):

    for n_act in [2, 3, 5, 8, 12, 20]:

        step = 0.12
        stall = 0
        for _ in range(400):
            if time.time() - t0 > lim:
                break
            c = get_corr(h)
            cm = c.max()
            thr = cm - 0.004 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]

                idx = np.argsort(c)[::-1][:n_act]

                g += grad_corr(h, ki, n)

def coord_cd(h, n, dx, rng, t0, lim, niter):

            c = c5v(hc, dx)
            if c < bc:
