        grad[:n-lag] += (1.0 - h[lag:n])  # from the "h[i]" factor

            h_curr = h_trial
            # Decrease lr slightly
            lr *= 0.999
        else:
            # Halve learning rate
            lr *= 0.7
            if lr < 1e-8:
                break
    return h_curr, best

def eval_c5(h, n):

    return float(np.max(corr) * (2.0/n))

def pair_line_search(h, i, j, n):
    """Find optimal δ for moving mass from j to i (or vice versa)."""
    hi, hj = h[i], h[j]
    # δ range: h[i]+δ ∈ [0,1], h[j]-δ ∈ [0,1]
    lo = max(-hi, -(1-hj))  # h[i]+δ >= 0, h[j]-δ <= 1
    hi_bound = min(1-hi, hj)  # h[i]+δ <= 1, h[j]-δ >= 0
    if hi_bound - lo < 1e-12:
        return 0.0
    # Bisection on the "slope" of C5(δ)
    # C5(δ) is max of linear functions, so convex in δ
    # Find where it changes from decreasing to increasing
    best_d = 0.0
    best_c5 = eval_c5(h, n)
    # Try bisection
    a, b = lo, hi_bound
    for _ in range(20):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        h[i], h[j] = hi + m1, hj - m1
        c1 = eval_c5(h, n)
        h[i], h[j] = hi + m2, hj - m2
        c2 = eval_c5(h, n)
        h[i], h[j] = hi, hj
        if c1 < c2:

        else:
            a = m1
        if b - a < 1e-9:

    # Ternary search for the minimum
    delta = (a + b) / 2
    h[i], h[j] = hi + delta, hj - delta
    c_new = eval_c5(h, n)
    h[i], h[j] = hi, hj
    if c_new < best_c5 - 1e-14:
        return delta
    return 0.0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
    return h

def soft_max_obj(v, n, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -300, 300)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp(h, n, alpha, maxiter=500):
    def obj(v):
        return soft_max_obj(v, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent_aggressive(h, n, t_end, rng):
    """Coordinate descent with adaptive step sizes and multiple scales."""
    h = h.copy()
    best = compute_c5(h, n)
    # Multi-scale step sizes
    for scale in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() > t_end:
            break
        improved = True
        while improved and time.time() < t_end:
            improved = False
            n_evals = 0
            for _ in range(3 * n):
                if time.time() > t_end or not improved:
                    break
                n_evals += 1
                i = rng.randint(n)
                j = (i + 1 + rng.randint(n - 1)) % n
                hid, hjd = h[i], h[j]
                # Try several delta values
                for d in [scale, -scale, scale*0.5, -scale*0.5, scale*2, -scale*2]:
                    ni, nj = hid + d, hjd - d
                    if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
    return h, best

def top_lag_balance(h, n, t_end, rng):

    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(2000):

        top_idx = np.argsort(corr)[-5:]  # top 5 lags
        # Compute average of top 5
        top_avg = np.mean(corr[top_idx])
        current_max = corr[np.argmax(corr)]
        if current_max - top_avg < 1e-8:
            continue  # already balanced
        # Gradient direction: decrease at top lag, increase at slightly below
        k_top = np.argmax(corr)
        lag = k_top - (n - 1)
        grad = np.zeros(n)
        if lag >= 0 and n - lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        elif lag < 0:
            lag2 = -lag
            grad[lag2:] += (1.0 - h[:n-lag2])
            grad[:n-lag2] -= h[lag2:]
        # Move in negative gradient direction (but paired to preserve sum)
        # Find i where grad is most negative (decreasing h[i] helps)
        # and j where grad is most positive (increasing h[j] hurts, so decrease)
        # Actually: we want to decrease corr[k_top]
        # d(corr[k_top])/d(h[i]) = grad[i]
        # To decrease corr: decrease h where grad > 0, increase h where grad < 0
        # Paired: increase h at argmin(grad), decrease h at argmax(grad)
        i_add = np.argmin(grad)
        i_sub = np.argmax(grad)
        if i_add == i_sub:

                if c5 < best - 1e-14:
                    best = c5
                else:
                    # revert
                    h[i_add] -= step
                    h[i_sub] += step
                    h = repair(h, n)
                break
    return h, best

    for n_alt in [n-4, n+4, n-8, n+8]:
        if n_alt < 32 or n_alt > 200:

        h_seed = repair(h_seed, n_alt)
