def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        while abs(diff) > 1e-12:
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                idx = int(np.argmax(h))
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)

def _grad_sm(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    p = np.exp(s - s.max())

    g = np.zeros(n)

        lag = k - (n - 1)

        else:
