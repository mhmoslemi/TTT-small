    c = _corr(h)
    m = c.max()
    ec = np.exp(alpha * (c - m))
    w = ec / ec.sum()
    order = np.argsort(w)[::-1]
    keep = set(order[:top_k].tolist())
    grad = np.zeros(n)
    for k in range(len(c)):
        if k not in keep:
            continue
        wk = w[k]
        sh = k - (n - 1)
        if sh == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            ml = n - sh
            grad[:ml] += wk * (1.0 - h[sh:])
            grad[sh:] -= wk * h[:ml]
        else:
            s = -sh
            ml = n - s
            grad[:ml] -= wk * h[s:]
            grad[s:] += wk * (1.0 - h[:ml])
    return grad

def _pgd(h, target, t_end, alpha, beta, lr, max_it=80000):
    n = len(h)
    dx = 2.0 / n
    best = _c5val(h, dx)
    h = h.copy()
    stall = 0
    for _ in range(max_it):
        if time.time() > t_end:
            break
        c = _c5val(h, dx)
        if c < best - 1e-10:
            best = c
            lr = min(lr * 1.015, 0.025)
            stall = 0
        elif c > best + 1e-8:
            stall += 1
            if stall > 15:
                lr = max(lr * 0.9, 1e-5)
        g = _grad_top(h, alpha, 12)
        if beta > 0:
            g += beta * (1.0 - 2.0 * h)
        h = _proj(h - lr * g, target)
    return h, best

def _anneal_search(h, target, t_end, rng, n_pass=40):
    n = len(h)
    dx = 2.0 / n
    best = _c5val(h, dx)
    h = h.copy()
    T = 0.003
    for p in range(n_pass):
        if time.time() > t_end:
            break
        T *= 0.97
        c = _corr(h)
        cmax = c.max()
        active = np.where(c > cmax - 1.0)[0]
        improved = False
        for k in active:
            if time.time() > t_end:
                break
            sh = k - (n - 1)
            if sh == 0:
                continue
            if sh > 0:
                ii = np.arange(n - sh)
                jj = ii + sh
            else:
                jj = np.arange(n + sh)
                ii = jj - sh
            sel = rng.choice(len(ii), min(40, len(ii)), replace=False)
            for idx in sel:
                i, j = int(ii[idx]), int(jj[idx])
                d = 0.004 * (2.0 * rng.random() - 1.0)
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cnew = _corr(h)
                    c5n = cnew.max() * dx
                    delta = c5n - best
                    if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
                        best = min(best, c5n)
                        h = _proj(h, target)
                        improved = True
                        break
                h[i] -= d
                h[j] += d
        if not improved:
            for _ in range(60):
                if time.time() > t_end:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = 0.003 * (2.0 * rng.random() - 1.0)
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    c5n = _c5val(h, dx)
                    delta = c5n - best
                    if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
                        best = min(best, c5n)
                        h = _proj(h, target)
                        improved = True
                        break
                h[i] -= d
                h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0 / 2.0)
    dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0.copy(), _c5val(h0, dx0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: PGD alpha sweep at n0
    t_end = t0 + 180
    for alpha in [200, 500, 1200, 3000]:
        if rem() < 500:
            break
        h0, c = _pgd(h0, n0 / 2.0, min(t_end, t0 + 45 - (time.time() - t0)), alpha, 0.0, 0.006)
        if c < best_c5:
            best_h, best_c5, best_n = h0.copy(), c, n0

    # Phase 2: Binarization pressure
    t_end = t0 + 320
    for beta in [0.05, 0.12, 0.25]:
        if rem() < 400:
            break
        best_h, c = _pgd(best_h, n0 / 2.0, min(t_end, t0 + 45 - (time.time() - t0)), 1500, beta, 0.004)
        if c < best_c5:
            best_c5 = c

    # Phase 3: Annealing local search at n0
    t_end = min(t0 + 600, t0 + budget_s - 350)
    best_h, best_c5 = _anneal_search(best_h, n0 / 2.0, t_end, rng, 35)

    # Phase 4: Multi-restart PGD
    t_end = min(t0 + 780, t0 + budget_s - 200)
    for r in range(8):
        if rem() < 220:
            break
        amp = 0.06 * (1.0 - r / 10.0)
        hp = best_h + amp * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, n0 / 2.0)
        hp, c = _pgd(hp, n0 / 2.0, min(t_end, t0 + 25 - (time.time() - t0)), 1000, 0.0, 0.005)
        if c < best_c5:
            best_h, best_c5, best_n = hp.copy(), c, n0

    # Phase 5: Upsample to 2*n0 and refine
    n1 = 2 * n0
    h1 = np.interp(np.linspace(0, 1, n1), np.linspace(0, 1, n0), best_h)
    h1 = _repair(h1, n1 / 2.0)
    dx1 = 2.0 / n1
    t_end = min(t0 + 950, t0 + budget_s - 120)
    for alpha in [500, 2000]:
        if rem() < 120:
            break
        h1, c = _pgd(h1, n1 / 2.0, min(t_end, t0 + 70 - (time.time() - t0)), alpha, 0.0, 0.002)
        if c < best_c5:
            best_h, best_c5, best_n = h1.copy(), c, n1

    # Phase 6: Final PGD polish + anneal
    tb = best_n
    t_end = min(t0 + budget_s - 40, t0 + 1050)
    for alpha in [4000, 10000]:
        if rem() < 40:
            break
        best_h, best_c5 = _pgd(best_h, tb / 2.0, min(t_end, t0 + 25 - (time.time() - t0)), alpha, 0.0, 0.001)
    if rem() > 30:
        best_h, best_c5 = _anneal_search(best_h, tb / 2.0, t0 + budget_s - 5, rng, 15)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5val(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
