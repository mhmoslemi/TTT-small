    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_objective(x, alpha, target, penalty, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    obj = m + np.log(np.sum(np.exp(s))) / alpha
    obj += penalty * (np.sum(x) - target) ** 2
    # Add L2 regularization towards target value to prevent degenerate solutions
    obj += lam * np.sum((x - 0.5) ** 2)
    return obj

    """Projected subgradient step on the exact max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_top = np.argsort(corr)[-top_k:]
    
    # Compute subgradient: average of gradients of active c(k)
    grad = np.zeros(n)
    for k in idx_top:

        shift = k - (n - 1)
        # d/dh[j] of sum_i h[i]*(1-h[i+shift])
        # = sum over i where i = j: (1-h[j+shift]) + sum over i where i+shift = j: -h[i]

            lo2, hi2 = 0, s2 - 1
            if s2 > 0:
                grad[lo2:hi2+1] += (-h[lo2+s2:hi2+1+s2]) * dx
    
    grad /= len(idx_top)
    
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, n)

def active_subgrad(h, n, dx, lr, n_iter, top_k=3):
    h = h.copy()
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        
        grad = np.zeros(n)
        for k in top_idx:
            shift = k - (n - 1)
            # gradient of c(k) w.r.t. h
            if shift >= 0:
                valid_i = np.arange(0, n - shift)
                g = np.zeros(n)
                g[valid_i] += (1.0 - h[valid_i + shift])
                g[valid_i + shift] += (-h[valid_i])
            else:
                s2 = -shift
                valid_i = np.arange(s2, n)
                g = np.zeros(n)
                g[valid_i] += (1.0 - h[valid_i - s2])
                g[valid_i - s2] += (-h[valid_i])
            grad += g
        grad *= dx / top_k
        
        # Projected gradient descent: h -= lr * grad, but maintain sum
        # Zero out the mean component to preserve sum
        grad_centered = grad - grad.mean()
        h_new = h - lr * grad_centered
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        
        if compute_c5(h_new, dx) < compute_c5(h, dx):
            h = h_new
        else:
            lr *= 0.5
    return h

                vi = np.arange(0, n - shift)
                grad[vi] += (1.0 - h[vi + shift])
                grad[vi + shift] += (-h[vi])

                s2 = -shift
                vi = np.arange(s2, n)
                grad[vi] += (1.0 - h[vi - s2])
                grad[vi - s2] += (-h[vi])

        c5n = compute_c5(h_new, dx)

            best_c5 = c5n
        else:

    h = h.copy()

    per_scale = n_trials // len(scales)
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d

                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not improved and scale < 0.005:
            break
    return h, best_c5

def active_subgrad(h, n, dx, lr_start, max_iters, top_k, deadline):
    h = h.copy()
    lr = lr_start
    best_c5 = compute_c5(h, dx)
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            shift = k - (n - 1)
            if shift >= 0:
                nv = n - shift
                grad[:nv] += (1.0 - h[shift:shift + nv])
                grad[shift:] += (-h[:nv])
            else:
                s2 = -shift
                nv = n - s2
                grad[s2:] += (1.0 - h[:nv])
                grad[:nv] += (-h[s2:])
        grad *= dx / top_k
        grad -= grad.mean()
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = repair(h_new, n)
        c5n = compute_c5(h_new, dx)
        if c5n < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5n
        else:
            lr *= 0.93
            if lr < 1e-9:
                lr = lr_start * 0.05
    return h, best_c5

def pair_search(h, n, dx, rng, n_trials, deadline):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for scale in [0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]:
        ns = n_trials // 6
        improved = False
        for _ in range(ns):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not improved and scale <= 0.005:
            break
    return h, best_c5

def perturb_h(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n // 5, 4 * n // 5))
    elif kind == 1:
        f = rng.randint(1, 6)
        hp += rng.uniform(0.03, 0.12) * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
    elif kind == 2:
        hp += rng.normal(0, 0.04, n)
    else:
        hp *= rng.uniform(0.8, 1.2)
    return repair(hp, n)

def upscale(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

            if cc < best_c5:

    # Phase 7: Final subgradient polish (50s)
    if time.time() < dfl - 40:
        te = min(time.time() + 40, dfl - 30)
        best_h, best_c5 = active_subgrad(best_h, best_n, 2.0 / best_n,
                                          0.002, 1000000, 2, te)

    # Phase 8: Ultra-fine pair search (25s)
    if time.time() < dfl - 8:
        te = min(time.time() + 20, dfl - 5)
        best_h, best_c5 = pair_search(best_h, best_n, 2.0 / best_n, rng, 300000, te)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
