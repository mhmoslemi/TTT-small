def _grad_step_lag(h, dx, kl, step_size, target):
    """Take a gradient step to reduce correlation at specific lag kl."""
    n = len(h)
    shift = int(kl) - (n-1)
    if shift == 0:
        return h, False
    # Correlation at this lag: sum_i h[i]*(1-h[i+shift]) (for shift>0)
    # d/dh[i] of this = (1-h[i+shift]) for i in [0, n-1-shift]
    #                    -h[i-shift] for i in [shift, n-1] (if shift>0)
    grad = np.zeros(n)
    if shift > 0:
        # i from 0 to n-1-shift: contribution h[i]*(1-h[i+shift])
        grad[:n-shift] += (1-h[shift:])
        # i from shift to n-1: contribution h[i-shift]*(1-h[i])
        grad[shift:] -= h[:-shift]
    else:
        s = -shift
        grad[s:] += (1-h[:-s])
        grad[:n-s] -= h[s:]
    # Project: subtract mean to maintain sum constraint
    grad -= grad.mean()
    # Clip for box constraint (approximate)
    grad *= step_size
    h_new = h + grad
    # Check bounds

        h_new = _repair(h_new, target)

    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - target) ** 2

    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14: break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()

            else:

            c = _c5(hc, dx)

            step *= 0.5
        step = max(step, 0.0005)
    return h, best

def _targeted_search(h, dx, rng, t_end, n_iters):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    c = _corr_full(h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        thresh = np.percentile(c, 85)
        active = np.where(c >= thresh)[0]
        kl = int(rng.choice(active))
        shift = kl - (n - 1)
        if shift == 0:
            continue
        if shift > 0:
            ii = np.arange(n - shift); jj = ii + shift
        else:
            jj = np.arange(n + shift); ii = jj - shift
        idx = int(rng.randint(len(ii)))
        i, j = int(ii[idx]), int(jj[idx])
        s = h[i] + h[j]
        scale = 0.001 + 0.015 * rng.random()
        if s < 1.0:
            d = scale
        elif s > 1.0:
            d = -scale
        else:
            d = scale * (2 * rng.random() - 1)
        h[i] -= d; h[j] += d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c_new = _c5(h, dx)
            if c_new < best:
                best = c_new
                c = _corr_full(h)
        else:
            h[i] += d; h[j] -= d
    return h, best

def _random_local(h, dx, rng, t_end, n_iters):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    per = max(50, n_iters // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return h, best
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2 * rng.random() - 1)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c = _c5(h, dx)
                if c < best:
                    best = c
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb_reopt(h, dx, rng, t_end, n_trials):
    n = len(h)
    tgt = n / 2.0
    best = h.copy()
    best_c = _c5(h, dx)
    for t in range(n_trials):
        if time.time() > t_end:
            break
        hp = best + 0.05 * (2 * rng.random(n) - 1)
        hp = _repair(hp, tgt)
        hp, c = _lbfgs_pass(hp, dx, [5000], 100, t_end)
        hp, c = _slsqp_pass(hp, dx, [50000], 150, t_end)
        if c < best_c:
            best, best_c = hp, c
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0.copy(), _c5(h0, dx0), n0

    # Phase 1: Quick polish at n0 (35s)
    te = t0 + 35
    h0, c = _lbfgs_pass(h0, dx0, [3000, 15000], 120, te)
    if c < best_c5:
        best_h, best_c5 = h0.copy(), c
    h0, c = _slsqp_pass(h0, dx0, [50000], 200, te)
    if c < best_c5:
        best_h, best_c5 = h0.copy(), c

    # Phase 2: Interpolate to n=2n0, thorough optimization (420s)
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h1 = _repair(_interp(best_h, n1), n1 / 2.0)
    te = t0 + 460
    h1, c1 = _lbfgs_pass(h1, dx1, [2000, 8000, 30000], 200, te)
    h1, c1 = _slsqp_pass(h1, dx1, [80000, 300000], 500, te)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, n1

    # Phase 3: Try n=4n0 only if budget allows (150s)
    rem1 = budget_s - (time.time() - t0)
    if rem1 > 500:
        n2 = min(4 * n0, 900)
        dx2 = 2.0 / n2
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        te = t0 + 620
        h2, c2 = _lbfgs_pass(h2, dx2, [5000, 20000], 150, te)
        h2, c2 = _slsqp_pass(h2, dx2, [200000], 300, te)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2.copy(), c2, n2

    # Phase 4: Perturb + reoptimize at best resolution (120s)
    nb = best_n
    dxb = 2.0 / nb
    te = t0 + budget_s - 350
    best_h, best_c5 = _perturb_reopt(best_h, dxb, rng, te, 4)

    # Phase 5: Gradient-based lag search (200s)
    te = t0 + budget_s - 150
    n_gsearch = 800000 if nb <= 280 else 400000
    best_h, best_c5 = _grad_lag_search(best_h, dxb, rng, te, n_gsearch)

    # Phase 6: Targeted local search (120s)
    te = t0 + budget_s - 30
    n_tsearch = 2000000 if nb <= 280 else 800000
    best_h, best_c5 = _targeted_search(best_h, dxb, rng, te, n_tsearch)

    # Phase 7: Final SLSQP + tiny local moves
    if budget_s - (time.time() - t0) > 25:
        best_h, best_c5 = _slsqp_pass(best_h, dxb, [400000], 200, t0 + budget_s - 10)
    if budget_s - (time.time() - t0) > 8:
        best_h, best_c5 = _random_local(best_h, dxb, rng, t0 + budget_s - 2, 80000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
