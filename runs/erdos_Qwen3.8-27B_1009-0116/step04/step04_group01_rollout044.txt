import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _correlate_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sum_con_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_con_jac(x):
    return np.ones(len(x))

def _smooth_obj(h, alpha, topk):
    n = len(h)
    c = _correlate_vec(h)
    dx = 2.0 / n
    topk = min(topk, len(c))
    idx = np.argsort(c)[::-1][:topk]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    s -= s.max()
    val = m + np.log(np.sum(np.exp(s))) / alpha
    return val * dx

def _smooth_grad(h, alpha, topk):
    n = len(h)
    c = _correlate_vec(h)
    dx = 2.0 / n
    topk = min(topk, len(c))
    idx = np.argsort(c)[::-1][:topk]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(topk):
        k = idx[ii]
        wk = w[ii] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp_wf(h, alpha, topk, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, topk, nsteps, step0):
    n = len(h)
    f = _smooth_obj(h, alpha, topk)
    step = step0
    for _ in range(nsteps):
        g = _smooth_grad(h, alpha, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for _bt in range(50):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            fn = _smooth_obj(h_new, alpha, topk)
            if fn <= f - 1e-6 * step * np.dot(g, h_new - h):
                h = h_new
                f = fn
                break
            step *= 0.5
        else:
            break
        step = min(step * 1.1, step0)
    return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.repeat(h, factor)
    return h_hi

def _pair_search(h, n_iter, scale, rng, best_c5):
    nc = len(h)
    c5c = _eval(h)
    best_h_local = h.copy()
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i] = ni
        h[j] = nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            best_h_local = h.copy()
            if c5c < best_c5:
                best_c5 = c5c
        else:
            h[i] = ni - d
            h[j] = nj + d
    return best_h_local, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _sum_con_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_con_jac(x):
    return np.ones(len(x))

def _smooth_obj(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    topk = min(topk, len(c))
    idx = np.argsort(c)[::-1][:topk]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    s -= s.max()
    val = m + np.log(np.sum(np.exp(s))) / alpha
    return val * dx

def _smooth_grad(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    topk = min(topk, len(c))
    idx = np.argsort(c)[::-1][:topk]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)

        return _repair(h, n)
