from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()

    s = h.sum()

        return np.full(n, 0.5)

def build_fourier(x, m):
    W = np.zeros((len(x), 2 * m))
    for k in range(1, m + 1):
        W[:, 2 * (k - 1)] = np.cos(k * np.pi * x)
        W[:, 2 * (k - 1) + 1] = np.sin(k * np.pi * x)
    return W

def fourier_start(x, m, rng):
    W = build_fourier(x, m)
    scale = 0.4 / np.sqrt(m)
    coeffs = scale * rng.randn(2 * m)
    h = 0.5 + W @ coeffs
    np.clip(h, 0, 1, out=h)
    return repair(h, len(h))

def slsqp_obj(v, alpha):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    vals = corr * (2.0 / n)
    mx = np.max(vals)
    s = np.clip(alpha * (vals - mx), -500, 500)
    return mx + np.log(np.sum(np.exp(s))) / alpha

def sum_con_fun(v):
    return v.sum() - v.shape[0] / 2.0

def sum_con_jac(v):
    return np.ones(v.shape[0])

def slsqp_opt(h, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    try:
        res = minimize(slsqp_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, rng, max_rounds=8000):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]
    stalled = 0
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        n_trials = 6 * n
        for _ in range(n_trials):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hi, hj = h[i], h[j]
            for d in steps:
                hit = False
                for sgn in (1, -1):
                    ni = hi + sgn * d
                    nj = hj - sgn * d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        hit = True
                        break
                    h[i], h[j] = hi, hj
                if hit:
                    break
        stalled = stalled + 1 if not improved else 0
        if stalled > 50:
            break
    return h, best

def multi_point_jump(h, n, rng, k_pts):
    h2 = h.copy()
    idx = rng.choice(n, size=k_pts, replace=False)
    src = idx[:k_pts // 2]

        h2[s_i] -= a
        h2[d_i] += a
    return repair(h2, n)

    h_t = repair(h_t, n)
