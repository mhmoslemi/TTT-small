    s = np.clip(alpha * (vals - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(h, lags, alpha):
    c = _corr(h)
    vals = c[lags]
    m = vals.max()
    s = np.clip(alpha * (vals - m), -500, 500)
    s -= s.max()
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(len(h))
    n = len(h)
    for idx, L in enumerate(lags):
        w = pw[idx]
        if w < 1e-15:
            continue
        if L == 0:
            g += w * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += w * (1.0 - h[L:])
            g[L:] -= w * h[:n - L]
        else:
            M = -L
            g[M:] += w * (1.0 - h[:n - M])
            g[:n - M] -= w * h[M:]
    return g

def _active_lags(h, margin=0.02, max_k=60):
    c = _corr(h)
    cm = c.max()
    active = np.where(c >= cm - margin)[0]
    if len(active) > max_k:
        active = np.argsort(c)[-max_k:]
    return active

def _proj_grad_opt(h, n, dx, target, alpha_list, iters_per_alpha, time_left_fn):
    h = h.copy()
    c5_cur = _c5(h, dx)
    for alpha in alpha_list:
        if time_left_fn() < 25:
            break
        active = _active_lags(h, 0.02, 60)
        lr = 0.03
        for _ in range(iters_per_alpha):
            if time_left_fn() < 25:
                break
            g = _softgrad(h, active, alpha)
            g_c = g - g.mean()
            ng = np.linalg.norm(g_c)
            if ng < 1e-14:
                break
            g_c /= ng
            h_new = _project_h(h - lr * g_c, target)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur:
                h = h_new
                c5_cur = c5_new
                lr = min(lr * 1.04, 0.05)
            else:
                lr *= 0.55
                if lr < 1e-9:
                    break
        active = _active_lags(h, 0.015, 50)
    return h, c5_cur

def _lbfgs_phase(h, n, dx, target, time_left_fn):
    h = h.copy()
    c5_cur = _c5(h, dx)
    best_local = h.copy()
    bc5_local = c5_cur
    for alpha in [5000, 20000, 80000, 300000]:
        if time_left_fn() < 20:
            break
        active = _active_lags(h, 0.012, 45)
        def obj(x):
            v = _corr(x)[active]
            m = v.max()
            s = np.clip(alpha * (v - m), -500, 500)
            s -= s.max()
            pen = 500.0 * (x.sum() - target) ** 2
            return m + np.log(np.sum(np.exp(s))) / alpha + pen
        def objg(x):
            g = _softgrad(x, active, alpha)
            g += 1000.0 * (x.sum() - target)
            return g
        try:
            res = minimize(obj, h.copy(), jac=objg, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-15})
            hc = _project_h(res.x, target)
            cv = _c5(hc, dx)
            if cv < c5_cur:
                h = hc
                c5_cur = cv
                if cv < bc5_local:
                    bc5_local = cv
                    best_local = h.copy()
        except Exception:
            pass
    return best_local, bc5_local

def _coord_descend(h, n, dx, target, time_left_fn, max_sweeps=150):
    h = h.copy()
    c5_cur = _c5(h, dx)
    dlist = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                      -0.001, -0.003, -0.008, -0.02, -0.05])
    for _ in range(max_sweeps):
        if time_left_fn() < 15:
            break
        active = _active_lags(h, 0.01, 50)
        g = _grad_lags(h, active)
        order = np.argsort(np.abs(g))[::-1][:45]
        improved = False
        for ii in range(len(order)):
            i = int(order[ii])
            for jj in range(ii + 1, min(ii + 18, len(order))):
                j = int(order[jj])
                bd, bc = 0.0, c5_cur
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        cv = _c5(h, dx)
                        if cv < bc:
                            bc, bd = cv, d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    h = _project_h(h, target)
                    c5_cur = _c5(h, dx)
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def _random_local(h, n, dx, target, rng, time_left_fn, n_iter=15000):
    h = h.copy()
    c5_cur = _c5(h, dx)
    for _ in range(n_iter):
        if time_left_fn() < 10:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.004, 0.004)
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cv = _c5(h, dx)
            if cv < c5_cur:
                c5_cur = cv
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    def tl():
        return budget_s - (time.time() - t0)

    # Start from initial
    h0 = _project_h(np.array(initial_h_values, dtype=float), target)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # Phase 1: Projected gradient with soft-max
    h1, c1 = _proj_grad_opt(h0, n, dx, target,
                            [80, 400, 2000, 10000, 50000, 200000, 800000], 600,
                            lambda: tl() - 120)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: L-BFGS-B polish
    h2, c2 = _lbfgs_phase(best_h, n, dx, target, lambda: tl() - 80)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: Subgradient on max (active set)
    h = best_h.copy()
    c5_cur = _c5(h, dx)
    step = 0.012
    for _ in range(3000):
        if tl() < 60:
            break
        active = _active_lags(h, 0.015, 50)
        g = _grad_lags(h, active)
        g_c = g - g.mean()
        ng = np.linalg.norm(g_c)
        if ng < 1e-13:
            break
        g_c /= ng
        h_new = _project_h(h - step * g_c, target)
        cv = _c5(h_new, dx)
        if cv < c5_cur - 1e-16:
            h, c5_cur = h_new, cv
            if cv < best_c5:
                best_h, best_c5 = h.copy(), cv
            step = min(step * 1.04, 0.035)
        else:
            step *= 0.45
            if step < 1e-9:
                break

    # Phase 4: Coordinate descent
    h4, c4 = _coord_descend(best_h, n, dx, target, lambda: tl() - 40, 180)
    if c4 < best_c5:
        best_h, best_c5 = h4, c4

    # Phase 5: Random local search
    h5, c5 = _random_local(best_h, n, dx, target, rng,
                           lambda: tl() - 20,
                           int(min(20000, max(0, (tl() - 20) * 150))))
    if c5 < best_c5:
        best_h, best_c5 = h5, c5

    # Phase 6: Try coarser n for global structure, then project up
    if tl() > 80:
        n_coarse = n // 2
        dx_c = 2.0 / n_coarse
        tgt_c = n_coarse / 2.0
        x_fine = np.linspace(0, 2, n, endpoint=False)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
        hc = np.interp(x_coarse, x_fine, best_h)
        hc = _project_h(hc, tgt_c)
        hc, cc = _proj_grad_opt(hc, n_coarse, dx_c, tgt_c,
                                [500, 5000, 50000, 500000], 400,
                                lambda: tl() - 40)
        # Interpolate back
        hb = np.interp(x_fine, x_coarse, hc)
        hb = _project_h(hb, target)
        cb = _c5(hb, dx)
        if cb < best_c5:
            best_h, best_c5 = hb, cb
            # Quick local fix
            h6, c6 = _coord_descend(best_h, n, dx, target, lambda: tl() - 20, 60)
            if c6 < best_c5:
                best_h, best_c5 = h6, c6

    # Phase 7: Final micro-polish
    if tl() > 8:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        nd = min(30000, int((tl() - 6) * 250))
        for _ in range(nd):
            if tl() < 6:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.002, 0.002)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cv = _c5(h, dx)
                if cv < c5_cur:
                    c5_cur = cv
                    if cv < best_c5:
                        best_h, best_c5 = h.copy(), cv
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _project_h(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
