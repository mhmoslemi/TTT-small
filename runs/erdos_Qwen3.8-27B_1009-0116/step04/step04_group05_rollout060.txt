import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _sm_obj(x, n, A, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(A * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / A + lam * (x.sum() - n * 0.5) ** 2

def _sm_jac(x, n, A, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(A * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-12:
            L = k - (n - 1)
            if L == 0:
                g[:] += pw[k] * (1.0 - 2.0 * x)
            elif L > 0:
                g[L:] += pw[k] * (1.0 - x[:n-L])
                g[:n-L] -= pw[k] * x[L:]
            else:
                mm = -L
                g[:n-mm] += pw[k] * (1.0 - x[mm:])
                g[mm:] -= pw[k] * x[:n-mm]
    g /= A
    if lam > 0:
        g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

from scipy.optimize import minimize
import time

_G = {}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            step = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += step
            d -= step
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def smooth_obj(x):
    n, A, lam = _G['n'], _G['A'], _G['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(A * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / A + lam * (x.sum() - n*0.5)**2

def smooth_jac(x):
    n, A, lam = _G['n'], _G['A'], _G['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(A * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2*n - 1):
        if pw[k] < 1e-12:
            continue
        L = k - (n - 1)
        if L == 0:
            g += pw[k] * (1.0 - 2.0*x)
        elif L > 0:
            g[L:] += pw[k] * (1.0 - x[:n-L])
            g[:n-L] -= pw[k] * x[L:]
        else:
            m2 = -L
            g[:n-m2] += pw[k] * (1.0 - x[m2:])
            g[m2:] -= pw[k] * x[:n-m2]
    g /= A
    if lam > 0:
        g += 2.0 * lam * (x.sum() - n*0.5)
    return g

    return np.ones(_G['n'])

            hc = repair(r.x, n)
            c = c5v(hc, dx)

                prev_eps = 0.0
            else:

        except Exception:
            break
    return bh, bc

    step = 0.05
    stall = 0
    for _ in range(500):

        else:
            m2 = -L
            g[:n-m2] += 1.0 - h[m2:]

def spectral_ph(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for alpha in [0.2, 0.35, 0.5, 0.35, 0.2, 0.1, 0.05]:
        h = bh.copy()
        for _ in range(200):

            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)

            nm = (1 - alpha) * mag + alpha * tgt
