def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break

                h[idx] += add
                diff -= add

                diff += sub
    return h

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on threshold."""
    target = n / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def _corr_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, -10.0, 10.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _adam_opt(h_init, n, alpha, end_t, lr=0.01):
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    c5c = _corr_max(h, n)
    while time.time() < end_t:
        t += 1
        g = _smooth_grad(h, n, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        step = lr * mh / (np.sqrt(vh) + eps)
        h_new = _project(h - step, n)
        c5n = _corr_max(h_new, n)
        if c5n < c5c:
            h, c5c = h_new, c5n
        else:
            lr *= 0.85
    return h, c5c

def _slsqp(h_init, n, alpha, maxiter):
    def obj(x):
        return _smooth_obj(x, n, alpha)
    def jac(x):
        return _smooth_grad(x, n, alpha)
    def con(x):
        return float(np.sum(x) - n / 2.0)
    try:
        res = minimize(obj, h_init, jac=jac, method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h_init, n)

    return h, c5c

def _single(h, n, end_t, rng):
    c5c = _corr_max(h, n)
    dl = np.array([1e-4, 3e-4, 1e-3, -1e-4, -3e-4, -1e-3])

            if c < c5c:

            else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, -10.0, 10.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _adam_opt(h_init, n, alpha, end_t, lr=0.01):
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    c5c = _corr_max(h, n)
    while time.time() < end_t:
        t += 1
        g = _smooth_grad(h, n, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        h_new = _project(h - step, n)
        c5n = _corr_max(h_new, n)
        if c5n < c5c:
            h, c5c = h_new, c5n
        else:
            lr *= 0.85
    return h, c5c

def _slsqp(h_init, n, alpha, maxiter):
    def obj(x):
        return _smooth_obj(x, n, alpha)
    def jac(x):
        return _smooth_grad(x, n, alpha)
    def con(x):
        return float(np.sum(x) - n / 2.0)
    try:
        res = minimize(obj, h_init, jac=jac, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h_init, n)

def _fast_pair(h, n, end_t, rng):
    dx = 2.0 / n
    c5c = _corr_max(h, n)
    deltas = np.array([1e-4, 3e-4, 1e-3, 3e-3, 1e-2,
                       -1e-4, -3e-4, -1e-3, -3e-3, -1e-2])
    lags = np.arange(2 * n - 1) - (n - 1)
    nc = 2 * n - 1
    for _pass in range(2000):
        if time.time() > end_t:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5c = float(np.max(c))
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        contribs = np.zeros((n, nc))
        for p in range(n):
            mp = (p + lags >= 0) & (p + lags < n)
            mm = (p - lags >= 0) & (p - lags < n)
            contribs[p, mp] += h[p + lags[mp]]
            contribs[p, mm] += h[p - lags[mm]]
        for ii in range(n):
            if time.time() > end_t + 20:
                break
            i = idx[ii]
            hi = h[i]
            ci = contribs[i]
            for jj in range(n):
                j = idx[jj]
                if j == i:
                    continue
                delta = ci - contribs[j]
                best_d, best_v = 0.0, c5c
                for d in deltas:
                    if hi + d < 0 or hi + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                        continue
                    v = float(np.max(c - d * delta))
                    if v < best_v - 1e-15:
                        best_v = v
                        best_d = d
                if best_d != 0.0:
                    h[i] = hi + best_d
                    h[j] -= best_d
                    c -= best_d * delta
                    c5c = best_v
                    improved = True
        if not improved and _pass > 5:
            break
    return h, c5c

def _single(h, n, end_t, rng):
    c5c = _corr_max(h, n)
    dl = np.array([1e-4, 3e-4, 1e-3, -1e-4, -3e-4, -1e-3])
    while time.time() < end_t:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = dl[rng.randint(len(dl))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _corr_max(h, n)
            if c < c5c:
                c5c = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _interp(h, n1, n2):
    x1 = np.linspace(0, 2.0, n1, endpoint=False)
    x2 = np.linspace(0, 2.0, n2, endpoint=False)
    return _repair(np.interp(x2, x1, h, period=2.0), n2)
