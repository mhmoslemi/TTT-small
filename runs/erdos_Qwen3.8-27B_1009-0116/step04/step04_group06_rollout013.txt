                    h[idx_i], h[idx_j] = ni, nj
                    if _c5(h, n) >= c5_cur:
                        h[idx_i], h[idx_j] = ni - d, nj + d
            if stumbles > 200:
                break
        else:
            stumbles = 0
        if c5_cur < best_c5:
            best_c5 = c5_cur; best_h = h.copy()

    # Final verification
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
