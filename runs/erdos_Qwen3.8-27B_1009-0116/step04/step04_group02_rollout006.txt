    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        # Compute smooth gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        s = alpha_smooth * (corr - m)
        s = np.clip(s, -300, 300)
        weights = np.exp(s)
        weights /= weights.sum()

        # dc_k/dh[m] = (1-h[m+n-1-k])*I[valid] - h[m-n+1+k]*I[valid]

        nfull = 2 * n - 1

                continue
            # h[j] * (1 - h[j + n - 1 - k])

            offset = n - 1 - k

            lo1 = max(0, -offset)
            hi1 = min(n, n - offset)
            for m in range(lo1, hi1):
                grad[m] += w * (1 - h[m + offset])
            # Second term: m+offset2 = j where j is the "second" index
            # h[j]*(1-h[j+offset]), d/dh[m] = -h[j] if j+offset = m, i.e., j = m-offset
            offset2 = n - 1 - k  # same as offset
            lo2 = max(0, -offset)
            hi2 = min(n, n - offset)
            for m in range(lo2, hi2):
                grad[m] -= w * h[m - offset]

        # Positions with high grad should decrease, low grad should increase
        # But we need sum constraint: move from i to j where grad[i] > grad[j]

        idx_sorted = np.argsort(grad)
        q = max(1, n // 4)
        i_pool = idx_sorted[-q:]  # high grad (want to decrease)
        j_pool = idx_sorted[:q]   # low grad (want to increase)
        
        i = i_pool[rng.randint(len(i_pool))]
        j = j_pool[rng.randint(len(j_pool))]
        if i == j:
            continue

        diff_grad = grad[i] - grad[j]
        if diff_grad <= 0:
            i, j = j, i
            diff_grad = -diff_grad
        scale = min(0.01, 0.001 * diff_grad + 0.0005)
        d = scale * (2.0 * rng.random() - 0.5)
        
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:

        h[i] = ni
        h[j] = nj
        c5n = compute_c5(h, dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i] -= d
            h[j] += d
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = (target / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def do_lbfgs(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        r = minimize(lse_obj, h, args=(alpha, target, pen), method='L-BFGS-B',
                    bounds=[(0,1)]*n, options={'maxiter':maxiter,'ftol':1e-16,'gtol':1e-8})
        return repair(r.x, n), compute_c5(repair(r.x, n), dx)
    except:
        return h.copy(), compute_c5(h, dx)

def fine_search(h, n, dx, rng, deadline, iters_per_scale, scales):
    h = h.copy()
    best = compute_c5(h, dx)
    full = np.correlate(h, 1.0-h, mode='full')
    for sc in scales:
        improved = False
        for _ in range(iters_per_scale):
            if time.time() >= deadline:
                return h, best
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0*rng.random() - 1.0)
            ni, nj = h[i]+d, h[j]-d
            if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:
                continue
            h[i], h[j] = ni, nj
            c = np.correlate(h, 1.0-h, mode='full')
            c5n = np.max(c) * dx
            if c5n < best - 1e-15:
                best = c5n
                improved = True
                full = c
            else:
                h[i], h[j] = ni-d, nj+d
        if not improved and sc < 0.003:
            break
    return h, best

def guided_search(h, n, dx, rng, deadline, n_iters):

    h = h.copy()
    best = compute_c5(h, dx)
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        if it % 50 == 0:
            full = np.correlate(h, 1.0-h, mode='full')
            # Identify top contributing positions for max lag
            kmax = np.argmax(full)
            t = n - 1 - kmax
            lo = max(0, -t)
            hi = min(n, n - t)
            if hi > lo:
                contrib = h[lo:hi] * (1.0 - h[lo+t:hi+t])
                top_pos = lo + np.argmax(contrib)
            else:
                top_pos = rng.randint(n)
        
        # 70% targeted, 30% random
        if rng.random() < 0.7:
            i = top_pos if rng.random() < 0.5 else (top_pos + t) % n
            j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:

        scale = 0.002 * (1.0 + 0.001 * (it % 1000))
        d = scale * (2.0*rng.random() - 1.0)
        ni, nj = h[i]+d, h[j]-d
        if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:

        c5n = np.max(np.correlate(h, 1.0-h, mode='full')) * dx
        if c5n < best - 1e-15:
            best = c5n
        else:
            h[i], h[j] = ni-d, nj+d
    return h, best

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n//4, 3*n//4))
    elif kind == 1:

    else:
        hp *= rng.uniform(0.8, 1.2)
    return repair(hp, n)
