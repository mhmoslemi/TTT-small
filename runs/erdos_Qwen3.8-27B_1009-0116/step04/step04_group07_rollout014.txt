            d = idx - (n - 1)
            weight = corr[idx] - np.median(corr)  # how bad is this shift
            weight = max(0, weight)
            # For this shift, push h[i] and h[i+d] toward agreement
            if d >= 0:
                for i in range(n - d):
                    target = (h[i] + h[i+d]) / 2  # midpoint
                    delta_h[i] += weight * (target - h[i]) * 0.5
                    delta_h[i+d] += weight * (target - h[i+d]) * 0.5
            else:
                dd = -d
                for i in range(n - dd):
                    target = (h[i] + h[i+dd]) / 2
                    delta_h[i] += weight * (target - h[i]) * 0.5
                    delta_h[i+dd] += weight * (target - h[i+dd]) * 0.5
        
        h_new = h + eta * delta_h
        h_new = repair(h_new, n)
        c5_new = compute_c5_full(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
        else:
            eta *= 0.9
            if eta < 0.0001:
                eta = 0.01  # reset
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def repair(h, n):
    return project_simplex(np.clip(h, 0.0, 1.0), n / 2.0)

def subgrad_step(h, n, lr, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top:
        d = idx - (n - 1)
        if d >= 0:
            grad[:n-d] += (1.0 - h[d:])
            grad[d:] -= h[:n-d]
        else:
            dd = -d
            grad[dd:] += (1.0 - h[:n-dd])
            grad[:n-dd] -= h[dd:]
    grad *= dx / len(top)
    return h - lr * grad, compute_c5(h, n)

def subgrad_opt(h, n, t_end, top_k=5, lr_init=0.08):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stall = 0
    for t in range(200000):
        if time.time() > t_end:
            break
        h_new, _ = subgrad_step(h, n, lr, top_k)
        v = 0.9 * v + (h_new - h)
        h_new = repair(h + v, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.15)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            v *= 0.5
            if stall > 500:
                lr = lr_init
                stall = 0
                v = np.zeros(n)
    return best_h, best_c5

def coord_descent(h, n, t_end, rng, max_rounds=5000):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.0008, 0.0003]
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        a = rng.randint(n)
        b = (a + 1 + rng.randint(n - 1)) % n
        ha, hb = h[a], h[b]
        for d in steps:
            for s in (1, -1):
                na, nb = ha + s * d, hb - s * d
                if na < 0 or na > 1 or nb < 0 or nb > 1:
                    continue
                h[a], h[b] = na, nb
                c5 = compute_c5(h, n)
                if c5 < best - 1e-15:
                    best = c5
                    ha, hb = na, nb
                else:
                    h[a], h[b] = ha, hb
    return h, best

def sa_opt(h, n, t_end, rng, T0=0.008, Tend=0.0005):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    n_est = max(1, int((t_end - time.time()) / 8e-6))
    for step in range(n_est):
        if time.time() > t_end:
            break
        T = T0 * (Tend / T0) ** (step / n_est)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        delta = (rng.random() - 0.5) * 0.08
        hi, hj = h[i], h[j]
        na, nb = hi + delta, hj - delta
        if na < 0 or na > 1 or nb < 0 or nb > 1:
            continue
        h[i], h[j] = na, nb
        c5_new = compute_c5(h, n)
        dc = c5_new - cur_c5
        if dc < 0 or rng.random() < np.exp(-dc / T):
            cur_c5 = c5_new
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj
    return best_h, best_c5

def slsqp_refine(h, n, alpha, maxiter=500):
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def try_alt_n(h_best, n_best, n_alt, t_end, rng):
    x_old = np.linspace(0.0, 1.0, n_best)
    x_new = np.linspace(0.0, 1.0, n_alt)
    h_seed = np.interp(x_new, x_old, h_best)
    h_seed = repair(h_seed, n_alt)
    t1 = min(time.time() + 0.4 * (t_end - time.time()), t_end)
    h1, c1 = subgrad_opt(h_seed, n_alt, t1, top_k=4, lr_init=0.06)
    t2 = min(time.time() + 0.3 * (t_end - time.time()), t_end)
    h2, c2 = coord_descent(h1, n_alt, t2, rng, 800)
    return h2, c2, n_alt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7777)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        init_h = None
        init_n = 64

    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n
    else:
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    n = best_n
    h_work = best_h.copy()

    # Phase 1: Subgradient optimization
    t1 = t0 + budget_s * 0.35
    h_work, c1 = subgrad_opt(h_work, n, t1, top_k=5, lr_init=0.08)
    if c1 < best_c5:
        best_h, best_c5 = h_work.copy(), c1
    h_work2, c1b = subgrad_opt(best_h.copy(), n, min(t1 + budget_s * 0.05, t0 + budget_s * 0.40), top_k=3, lr_init=0.05)
    if c1b < best_c5:
        best_h, best_c5 = h_work2.copy(), c1b
    h_work = best_h.copy()

    # Phase 2: Coordinate descent
    t2 = t0 + budget_s * 0.65
    h_work, c2 = coord_descent(h_work, n, t2, rng, 8000)
    if c2 < best_c5:
        best_h, best_c5 = h_work.copy(), c2
    h_work = best_h.copy()

    # Phase 3: Simulated annealing
    t3 = t0 + budget_s * 0.82
    h_sa, c3 = sa_opt(h_work, n, t3, rng, T0=0.006, Tend=0.0003)
    if c3 < best_c5:
        best_h, best_c5 = h_sa.copy(), c3
    h_work = best_h.copy()

    # Phase 4: SLSQP refinement
    t4 = t0 + budget_s * 0.92
    if time.time() < t4:
        for alpha in [50000, 200000]:
            if time.time() > t4:
                break
            h_new = slsqp_refine(h_work.copy(), n, alpha, 600)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5 - 1e-15:
                best_h, best_c5 = h_new.copy(), c5_new
                h_work = h_new
        h_work, c4 = coord_descent(h_work, n, min(time.time() + 3, t4), rng, 2000)
        if c4 < best_c5:
            best_h, best_c5 = h_work.copy(), c4

    # Phase 5: Try alternative n
    t5 = t0 + budget_s * 0.97
    if time.time() < t5:
        for n_alt in [60, 72, 80, 96, 128]:
            if time.time() > t5:
                break
            if n_alt == best_n or n_alt > 200:

        if cf < best_c5:
            best_h, best_c5 = hf, cf

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
