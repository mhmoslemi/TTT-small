def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        if abs(d) < 1e-15:

        h = np.clip(h + d / n, 0.0, 1.0)

        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:

        best_h, best_c5 = _targeted_local(best_h, dxb, rng, t0 + budget_s - 2, 100000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
