from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _to_binary(h, n_ones):
    """Convert continuous h to binary with exactly n_ones ones."""
    n = len(h)
    # Sort by h value, take top n_ones as 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n_ones]] = 1.0
    return b

def _bin_search(b, dx, rng, t_end, sa=False):
    """Binary local search with optional simulated annealing."""
    n = len(b)
    b = b.copy()
    best_c = _c5(b, dx)
    ones = np.where(b > 0.5)[0].tolist()
    zeros = np.where(b < 0.5)[0].tolist()
    T = 0.005  # initial temperature for SA
    T_min = 0.0001
    
    for it in range(50000000):
        if time.time() > t_end:
            break
        if sa:
            T = max(T_min, T * 0.9999)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        b[i] = 0.0
        b[j] = 1.0
        c = _c5(b, dx)
        
        if c < best_c:
            best_c = c
            ones.remove(i); ones.append(j)
            zeros.append(i); zeros.remove(j)
        elif sa and rng.random() < np.exp(-(c - best_c) / T):
            # Accept worse move (SA)
            ones.remove(i); ones.append(j)
            zeros.append(i); zeros.remove(j)
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return b, best_c

def _targeted_bin_search(b, dx, rng, t_end):

    n = len(b)
    b = b.copy()
    best_c = _c5(b, dx)

    zeros = set(np.where(b < 0.5)[0])

            if m == 0:

            else:

                partner = i + m
            else:

            candidates_zero = [z for z in zeros if z != partner]
            if candidates_zero:

                b[i] = 0.0; b[j] = 1.0
                c = _c5(b, dx)
                if c < best_c:
                    best_c = c
                    ones.remove(i); ones.add(j)
                    zeros.add(i); zeros.remove(j)
                    improvement = True
                    break
                else:
                    b[i] = 1.0; b[j] = 0.0
            
            # Option 2: swap b[partner] with a random one (other than i)
            candidates_one = [o for o in ones if o != i]
            if candidates_one:
                j = candidates_one[rng.randint(len(candidates_one))]
                b[partner] = 1.0; b[j] = 0.0
                c = _c5(b, dx)
                if c < best_c:
                    best_c = c
                    ones.remove(j); ones.add(partner)
                    zeros.add(j); zeros.remove(partner)
                    improvement = True
                    break
                else:
                    b[partner] = 0.0; b[j] = 1.0
        
        if not improvement:

            i = list(ones)[rng.randint(len(ones))]
            j = list(zeros)[rng.randint(len(zeros))]
            b[i] = 0.0; b[j] = 1.0

                best_c = c
                ones.remove(i); ones.add(j)
                zeros.add(i); zeros.remove(j)
            else:

    return b, best_c

def _slsqq_polish(h, dx, maxiter):
    """SLSQP polish on continuous h."""
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        # Add small regularization to push toward binary
        return m + 0.01 * np.sum(x * (1 - x))
    def obj_grad(x):
        c = _corr(x)
        m = c.max()
        # Subgradient of max
        active = np.where(c >= m - 1e-6)[0]
        grad = np.zeros(n)
        for k in active:
            ll = k - (n - 1)
            if ll >= 0:
                idx = np.arange(n - ll)
                grad[idx] += (1.0 - x[idx + ll])
                grad[idx + ll] -= x[idx]
            else:
                ll2 = -ll
                idx = np.arange(ll2, n)
                grad[idx] += (1.0 - x[idx - ll2])
                grad[idx - ll2] -= x[idx]
        grad /= len(active)
        grad += 0.02 * (1 - 2 * x)  # regularization gradient
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                     bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new, endpoint=False),
                     np.linspace(0, 1, len(h), endpoint=False), h)

def _c5(h, dx):

def _repair(h, target):
