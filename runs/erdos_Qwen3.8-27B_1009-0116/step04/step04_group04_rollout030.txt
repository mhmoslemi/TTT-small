import numpy as np

import time

# Helper functions: _corr, _c5, _repair, _smooth_obj, _grad_smooth

# Gradient computation for smooth max:
def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')  # length 2n-1
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()  # length 2n-1, p[k] for shift k-(n-1)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for i in range(n):
        # sum_k p[k] * ((1-h[i+shifts[k]])_valid - h[i-shifts[k]]_valid)
        v1 = (i + shifts)
        mask1 = (v1 >= 0) & (v1 < n)
        v2 = (i - shifts)
        mask2 = (v2 >= 0) & (v2 < n)
        t1 = np.dot(p[mask1], 1.0 - h[v1[mask1]])
        t2 = np.dot(p[mask2], h[v2[mask2]])
        grad[i] = t1 - t2
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-11:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-12:
                break
    return h

def _soft_obj(x, alpha, target, pen):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + 0.5 * pen * (x.sum() - target)**2

def _grad_soft(h, alpha, pen):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        v1 = i + shifts
        m1 = (v1 >= 0) & (v1 < n)
        v2 = i - shifts
        m2 = (v2 >= 0) & (v2 < n)
        grad[i] = np.dot(p[m1], 1.0 - h[v1[m1]]) - np.dot(p[m2], h[v2[m2]])
    grad += pen * (h.sum() - n / 2.0)
    return grad

def _proj_hessian(h, target):
    """Project onto {sum h = target, 0<=h<=1}"""
    return _repair(h, target)

def _lbfgs(h, dx, alpha, maxiter, pen=1000):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, tgt, pen), h,
                     jac=lambda x: _grad_soft(x, alpha, pen),
                     method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6})
        return _repair(r.x, tgt)
    except Exception:
        return h

def _gradient_local_search(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    alphas = [200, 500, 1000, 2000, 5000]
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    accepted = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        alpha = alphas[min(it // (n_iter // len(alphas)), len(alphas)-1)]
        g = _grad_soft(h, alpha, 0.0)  # gradient of smooth max only
        # Project: subtract mean to maintain sum
        g_proj = g - np.mean(g)
        
        # Try backtracking line search
        improved = False
        for ss in step_sizes:
            h_try = h - ss * g_proj
            h_try = _repair(h_try, tgt)
            c = _c5(h_try, dx)
            if c < best - 1e-15:
                h = h_try
                best = c
                accepted += 1
                improved = True
                break
        if not improved:
            # Try random pair move
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = 0.005 * (2.0 * rng.random() - 1.0)
                h2 = h.copy()
                h2[i] += d
                h2[j] -= d
                if -1e-9 <= h2[i] <= 1.0 + 1e-9 and -1e-9 <= h2[j] <= 1.0 + 1e-9:
                    h2 = _repair(h2, tgt)
                    c2 = _c5(h2, dx)
                    if c2 < best:
                        h, best = h2, c2
                        accepted += 1
    return h, best

def _pair_local_search(h, dx, rng, t_end, n_iter):
    """Focused pair moves guided by top lags"""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        # Top 3 lags
        top_k = np.argsort(c)[-3:]
        # For each top lag, compute "pressure" at each position
        pressure = np.zeros(n)
        for k in top_k:
            s = k - (n - 1)
            if s > 0:
                for i in range(n - s):
                    # Moving mass from i to i+s decreases c[k] if h[i]=1, h[i+s]=0
                    # We want to increase overlap: make h[i] and h[i+s] both 1 or both 0
                    pressure[i] += h[i] * (1 - h[i + s])
                    pressure[i + s] += -h[i] * (1 - h[i + s])
            elif s < 0:
                ss = -s
                for i in range(ss, n):
                    pressure[i] += h[i] * (1 - h[i - ss])
                    pressure[i - ss] += -h[i] * (1 - h[i - ss])
        # Find i with most positive pressure (should increase) and j with most negative (should decrease)
        i = int(np.argmax(pressure))
        j = int(np.argmin(pressure))
        if pressure[i] - pressure[j] < 1e-8:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
        for d in [0.02, 0.01, 0.005, 0.002]:
            h2 = h.copy()
            h2[i] += d
            h2[j] -= d
            if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                h2 = _repair(h2, tgt)
                c2 = _c5(h2, dx)
                if c2 < best:
                    h, best = h2, c2
                    break
    return h, best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)
