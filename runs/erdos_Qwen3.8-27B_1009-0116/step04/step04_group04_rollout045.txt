    best_h, best_c5, best_n = h0, _c5(h0, dx0), n0

    # Phase 1: Nesterov gradient at n0 (250s)
    t_end = min(t0 + 250, t0 + budget_s - 500)
    best_h, best_c5 = _nesterov(best_h, dx0, t_end, lr0=0.08, top_m=3)

    # Phase 2: Pair refine at n0 (150s)
    t_end = min(t0 + 400, t0 + budget_s - 350)
    best_h, best_c5 = _pair_refine(best_h, dx0, rng, t_end, 40000)

    # Phase 3: SLSQP polish at n0 (100s)
    t_end = min(t0 + 500, t0 + budget_s - 250)
    best_h, best_c5 = _slsqp_polish(best_h, dx0, 600)

    # Phase 4: Interpolate to 2*n0, optimize (300s)
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h1 = _repair(_interp1d(best_h, n1), n1 / 2.0)
    t_end = min(t0 + 700, t0 + budget_s - 200)
    h1, c1 = _nesterov(h1, dx1, t_end, lr0=0.04, top_m=4)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n1

    # Phase 5: Pair + block refine at n1 (200s)
    t_end = min(t0 + 880, t0 + budget_s - 120)
    dxb = 2.0 / best_n
    best_h, best_c5 = _pair_refine(best_h, dxb, rng, t_end, 60000)
    if time.time() < t0 + budget_s - 60:
        best_h, best_c5 = _block_refine(best_h, dxb, rng, t0 + budget_s - 60, 15000)

    # Phase 6: Final SLSQP
    if time.time() < t0 + budget_s - 30:
        best_h, best_c5 = _slsqp_polish(best_h, dxb, 500)

    # Phase 7: Tiny local search
    if time.time() < t0 + budget_s - 5:
        best_h, best_c5 = _pair_refine(best_h, dxb, rng, t0 + budget_s - 2, 30000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
