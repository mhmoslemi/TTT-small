def _project_1d(z, target):
    """Project z onto {0<=h<=1, sum(h)=target} via bisection on lambda."""
    n = len(z)
    lo, hi = -z.max() - target/n - 1, -z.min() + (n-target)/n + 1
    # Actually: h_i = clip(z_i + lam, 0, 1), find lam s.t. sum = target
    lo, hi = -z.max() - 1.0, -z.min() + 1.0
    for _ in range(80):

        h = np.clip(z + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(z + (lo+hi)/2, 0.0, 1.0)
    # Fix tiny numerical error
    d = target - h.sum()
    if abs(d) > 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(z, target):
    n = len(z)
    lo, hi = -z.max() - 1.0, -z.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _subgrad_grad(h, n_active):
    n = len(h)
    c = _corr(h)
    top_idx = np.argsort(c)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:])
            grad[s:] += -h[:m]
        else:
            t = -s
            m = n - t
            grad[t:] += (1.0 - h[:m])
            grad[:m] += -h[t:]
    return grad, c.max()

def _subgrad_optimize(h, n_active, lr0, n_iters, t_end, momentum=0.9):
    n = len(h)
    dx = 2.0 / n
    best = _c5(h, dx)
    h = h.copy()
    v = np.zeros(n)
    lr = lr0
    patience = 0
    for it in range(n_iters):
        if time.time() > t_end:
            break
        grad, _ = _subgrad_grad(h, n_active)
        gnorm = np.max(np.abs(grad)) + 1e-12
        v = momentum * v + (1 - momentum) * grad / gnorm
        step = lr / (np.max(np.abs(v)) + 1e-12)
        h_new = _project(h - step * v, n / 2.0)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                lr *= 0.75
                patience = 0
        if it % 200 == 199:
            lr = max(lr * 0.98, lr0 * 0.1)
    return h, best

def _slsqp_soft(h, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    dx = 2.0 / n
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.exp(s).sum()) / alpha
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs_soft(h, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    dx = 2.0 / n
    bds = [(0.0, 1.0)] * n
    best_h, best_c = h, _c5(h, dx)
    for pen in [50, 500, 5000]:
        def obj(x):
            c = _corr(x)
            m = c.max()
            s = np.clip(alpha * (c - m), -400, 400)
            return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - tgt) ** 2
        try:
            r = minimize(obj, best_h, method='L-BFGS-B', bounds=bds,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-6})
            hc = _project(r.x, tgt)
            c = _c5(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
        except Exception:
            pass
    return best_h, best_c

def _coord_search(h, rng, t_end, n_iter):
    n = len(h)
    dx = 2.0 / n
    best = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_frac = max(8, int(0.06 * len(c)))
        top_k = np.argsort(c)[-top_frac:]
        k = top_k[rng.randint(len(top_k))]
        s = k - (n - 1)
        if abs(s) < 1:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
        elif s > 0:
            i = rng.randint(n - s)
            if it % 3 == 0:
                j = i + s
            else:
                j = rng.randint(n)
                if j == i:
                    continue
        else:
            i = rng.randint(n + s)
            if it % 3 == 0:
                j = i - s
            else:
                j = rng.randint(n)
                if j == i:
                    continue
        d = rng.uniform(-0.02, 0.02)
        h[i] += d; h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            if _c5(h, dx) < best:
                best = _c5(h, dx)
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return h, best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)
