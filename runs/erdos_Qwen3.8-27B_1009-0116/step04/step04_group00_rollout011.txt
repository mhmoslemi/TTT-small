import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(x, target):
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - y) if diff > 0 else np.argsort(y)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - y[idx]); y[idx] += a; diff -= a
            else:
                a = min(-diff, y[idx]); y[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return y

def _softobj(x, alpha, n, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha*(c-m), -400, 400)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s)))/alpha + lam*(x.sum()-n/2.0)**2

def _softgrad(x, alpha, n, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha*(c-m), -400, 400)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        L = k-(n-1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk*(1.0-2.0*x)
        elif L > 0:
            g[L:] += pk*(1.0-x[:n-L]); g[:n-L] -= pk*x[L:]
        else:
            M = -L
            g[:n-M] += pk*(1.0-x[M:]); g[M:] -= pk*x[:n-M]
    g /= alpha
    g += 2.0*lam*(x.sum()-n/2.0)
    return g

def _active_subgrad(h, frac):
    c = _corr(h)
    n = len(h)
    cm = np.max(c)
    spread = cm - np.min(c)
    thresh = cm - max(frac*spread, 1e-10)
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k-(n-1)
        if L == 0:
            g += (1.0-2.0*h)
        elif L > 0:
            g[L:] += (1.0-h[:n-L]); g[:n-L] -= h[L:]
        else:
            M = -L
            g[:n-M] += (1.0-h[M:]); g[M:] -= h[:n-M]
    return g/len(active)

def _proj_grad(h, budget, frac=0.01):
    n = len(h)
    target = n/2.0
    dx = 2.0/n
    h = _project(h.copy(), target)
    c5c = _c5(h, dx)
    best_h, best_c5 = h.copy(), c5c
    step = 0.025
    t_end = time.time()+budget
    for _ in range(20000):
        if time.time() > t_end:
            break
        g = _active_subgrad(h, frac)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        g /= ng
        found = False
        for st in [step, step*0.5, step*0.25, step*0.1]:
            hn = _project(h - st*g, target)
            cn = _c5(hn, dx)
            if cn < c5c - 1e-16:
                h, c5c = hn, cn
                found = True
                if c5c < best_c5:
                    best_c5, best_h = c5c, h.copy()
                break
        if not found:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            step = min(step*1.08, 0.06)
    return best_h, best_c5

def _local_search(h, budget, n):
    dx = 2.0/n
    target = n/2.0
    c5c = _c5(h, dx)
    best_h, best_c5 = h.copy(), c5c
    t_end = time.time()+budget
    dlist = np.array([0.001,0.003,0.01,0.03,0.08,-0.001,-0.003,-0.01,-0.03,-0.08])
    # Use gradient to find hot indices
    for alpha in [100000, 500000, 2000000]:
        g = _softgrad(h, alpha, n, 0.0)
        order = np.argsort(np.abs(g))[::-1][:60]
        improved = True
        while improved and time.time() < t_end:
            improved = False
            for ii in range(len(order)):
                if time.time() > t_end:
                    break
                for jj in range(ii+1, min(ii+35, len(order))):
                    i, j = int(order[ii]), int(order[jj])
                    for d in dlist:
                        ni, nj = h[i]+d, h[j]-d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            h = _project(h, target)
                            c = _c5(h, dx)

                                break

                                h = _project(h, target)
            g = _softgrad(h, 500000, n, 0.0)
            order = np.argsort(np.abs(g))[::-1][:60]
    return best_h, best_c5

def _soft_optimize(h, n, bounds, budget):
    dx = 2.0/n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    t_end = time.time()+budget
    for a in [3000, 20000, 100000, 500000, 2000000, 10000000]:
        if time.time() > t_end - 3:
            break

            res = minimize(
                lambda x: _softobj(x, a, n, 80.0), best_h.copy(),

                options={'maxiter': 600, 'ftol': 1e-20, 'gtol': 1e-15})
            hc = _project(res.x, n/2.0)
            cv = _c5(hc, dx)
            if cv < best_c5:
                best_c5, best_h = cv, hc
        except Exception:
            pass
    return best_h, best_c5
