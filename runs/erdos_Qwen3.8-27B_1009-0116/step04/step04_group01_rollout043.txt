        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]

            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        g *= dx
        
        # Projected gradient step
        h_new = _proj(h - step * g, n / 2)
        h = h_new
        step *= 0.999  # slow decay
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]
    return g

def _obj_smooth(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s - s.max()))) / alpha) * (2.0 / len(h))

def _slsqp(h, alpha, maxiter=2000, ftol=1e-18):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, nsteps, step0, alpha_smooth):
    """Nesterov accelerated projected gradient."""
    n = len(h)
    h_t = h.copy().astype(float)
    y = h_t.copy()
    t = 1.0
    step = step0
    f_prev = _eval(h_t)
    for k in range(nsteps):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        g = _grad_smooth(y, alpha_smooth)
        y_new = y - step * g
        h_new = _proj(y_new, n / 2)
        # Momentum update
        y = h_new + ((t - 1.0) / t_new) * (h_new - h_t)
        h_t = h_new
        t = t_new
        # Adaptive step
        f_new = _eval(h_t)
        if f_new < f_prev:
            step = min(step * 1.02, step0 * 2)
            f_prev = f_new
        else:
            step *= 0.8
    return h_t

def _coord_descent(h, n_iters, rng, scales):
    """Coordinate descent with random pairs."""
    n = len(h)
    h = h.copy()
    best_f = _eval(h)
    for it in range(n_iters):
        scale = scales[it % len(scales)]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        f_new = _eval(h)
        if f_new < best_f - 1e-16:
            best_f = f_new
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, best_f

def _subgrad_topk(h, nsteps, step0, topk=8):
    """Subgradient method targeting top-k lags."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    f = _eval(h)
    step = step0
    for k in range(nsteps):
        c = np.correlate(h, 1.0 - h, mode='full')
        # Weight top-k lags
        top_idx = np.argsort(c)[-topk:]
        w_raw = c[top_idx] - c[top_idx[0]] + 1.0
        w = w_raw / w_raw.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            sh = (n - 1) - ki
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wi * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wi * h[:n - sh]
            else:
                sp = -sh
                cnt = n - sp
                g[sp:] += wi * (1.0 - h[:cnt])
                g[:cnt] -= wi * h[sp:]
        g *= dx
        # Line search
        best_h = h
        best_f = f
        for bt in range(40):
            h_try = _proj(h - step * g, n / 2)
            f_try = _eval(h_try)
            if f_try < best_f - 1e-10:
                best_h = h_try
                best_f = f_try
                break
            step *= 0.6
        h = best_h
        f = best_f
        step = min(step * 1.05, step0)
    return h

    def _el():

    if _el() < budget_s - 400:

        h_try, c5 = _coord_descent(best_h, n_iters, rng, scales)
        h_try = _repair(h_try, n0)
        c5 = _eval(h_try)
        if c5 < best_c5:
            best_c5 = c5

        n_iters = min(200000, remaining * 8000)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
