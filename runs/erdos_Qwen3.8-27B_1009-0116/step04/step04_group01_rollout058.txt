    s = h.sum()
    if s < 1e-12:

                    diff -= a

                    h[idx] -= a
                    diff += a

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_to_sum(h, target):

    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]

    except Exception:

def _coord_sweep(h, n_sweeps=5):
    """Coordinate descent: optimize each h[i] in turn."""
    n = len(h)
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        for i in range(n):
            # Compute base correlation with h[i] = 0
            h_save = h[i]
            if h_save < 1e-15:
                continue
            h[i] = 0.0
            base = np.correlate(h, 1.0 - h, mode='full')
            
            # Coefficient of t in c[k] when h[i] = t
            # First factor: h[i] appears as h[j+k] when j = i-k
            #   contribution: t * (1 - h[j]) = t * (1 - h[i-k])
            # Second factor: h[i] appears in (1-h[j]) when j = i
            #   contribution: h[j+k] * (1-t) vs h[j+k] * 1 → change: -t * h[j+k]
            
            coeff = np.zeros(2 * n - 1)
            # First factor: for each valid j, k = i - j, index = k + n - 1
            j_range = np.arange(max(0, i - (n-1)), min(n, i + 1))
            # Actually: j is valid (0≤j<n) and k=i-j is the lag
            # The full index is k + n - 1 = i - j + n - 1
            for j in range(n):
                k = i - j
                idx = k + n - 1
                if 0 <= idx < 2 * n - 1:
                    coeff[idx] += (1.0 - h[j])
            # Second factor: j = i, k = j' - i for j' in range(n)
            # h[j'] * (1-t) in true, h[j'] * 1 in base → diff = -t*h[j']
            for j in range(n):
                k = j - i
                idx = k + n - 1
                if 0 <= idx < 2 * n - 1:
                    coeff[idx] -= h[j]
            
            # Minimize max_k (base[k] + t*coeff[k]) over t in [0,1]
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = np.max(base + m1 * coeff)
                v2 = np.max(base + m2 * coeff)
                if v1 <= v2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            h[i] = t_opt
            h[i] = np.clip(h[i], 0.0, 1.0)
    
    return h

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample_linear(h, factor):
    n = len(h)
    n_new = n * factor
    x_old = np.linspace(0.0, 2.0, n)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def _downsample(h, factor):
    n = len(h)
    n_new = n // factor
    idx = np.arange(n_new) * factor
    return h[idx]

def _coord_sweep(h, n_sweeps, t0, deadline):
    """Coordinate descent: optimize each h[i] exactly (1D subproblem)."""
    n = len(h)
    dx = 2.0 / n
    t_grid = np.linspace(0.0, 1.0, 300)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            hv = h[i]
            h[i] = 0.0
            c_base = np.correlate(h, 1.0 - h, mode='full')
            h[i] = hv
            coeff = np.zeros(2 * n - 1)
            for j in range(n):
                idx1 = i - j + n - 1
                if 0 <= idx1 < 2 * n - 1:
                    coeff[idx1] += (1.0 - h[j])
                idx2 = j - i + n - 1
                if 0 <= idx2 < 2 * n - 1:
                    coeff[idx2] -= h[j]
            # Vectorized evaluation over t grid
            c_t = c_base[None, :] + t_grid[:, None] * coeff[None, :]
            c_t[:, n - 1] = c_base[n - 1] + t_grid - t_grid * t_grid
            F = c_t.max(axis=1)
            t_opt = t_grid[np.argmin(F)]
            if abs(t_opt - hv) > 1e-9:
                h[i] = t_opt
                improved = True
        if not improved:
            break
    return h

def _pairwise_refine(h, n_iters, scale, rng, t0, deadline):
    n = len(h)
    c5c = _eval(h)

        j = rng.randint(n)

        d = scale * rng.randn()

        else:
