  n=0: a0*v1
  n=1: a1*v2
  So c[-1] = a0*v1 + a1*v2

For k = 0: c[0] = a0*v0 + a1*v1 + a2*v2

For k = 1: c[1] = sum_n a[n] * v[n-1]
  n=1: a1*v0
  n=2: a2*v1
  So c[1] = a1*v0 + a2*v1

For k = 2: c[2] = a2*v0

In 'full' mode, the output is ordered from k=-(M-1) to k=N-1:
output = [a0*v2, a0*v1+a1*v2, a0*v0+a1*v1+a2*v2, a1*v0+a2*v1, a2*v0]

So output[p] corresponds to k = p - (M-1) = p - 2 (for M=3).

Now in our problem, a = h, v = 1-h.
output[p] = sum_n h[n] * (1-h[n-k]) where k = p - (n-1)

Let me denote lag = k = p - (n-1).

output[p] = sum_n h[n] * (1-h[n-lag])

For lag >= 0:
output[p] = sum_{n=lag}^{n-1} h[n] * (1-h[n-lag])

For lag < 0 (let ll = -lag):
output[p] = sum_{n=0}^{n-1-ll} h[n] * (1-h[n+ll])

Now the gradient dc/dh[i]:

For lag >= 0:
c = sum_{n=lag}^{n-1} h[n] * (1-h[n-lag])

dc/dh[i]:
- From h[n] term: i = n, so n = i, valid if lag <= i <= n-1. Contribution: (1-h[i-lag])
- From (1-h[n-lag]) term: n-lag = i, so n = i+lag, valid if lag <= i+lag <= n-1, i.e., 0 <= i <= n-1-lag. Contribution: -h[i+lag]

For lag < 0 (ll = -lag):
c = sum_{n=0}^{n-1-ll} h[n] * (1-h[n+ll])

dc/dh[i]:
- From h[n] term: n = i, valid if 0 <= i <= n-1-ll. Contribution: (1-h[i+ll])
- From (1-h[n+ll]) term: n+ll = i, so n = i-ll, valid if 0 <= i-ll <= n-1-ll, i.e., ll <= i <= n-1. Contribution: -h[i-ll]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-13:
                    break
                if diff > 0:
                    h[idx] = min(1.0, h[idx] + diff)
                else:
                    h[idx] = max(0.0, h[idx] + diff)
                diff = target - h.sum()
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for p in range(2 * n - 1):
        wk = w[p]
        if wk < 1e-15:
            continue
        lag = p - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - x[ll:])
            g[ll:] -= wk * x[:n - ll]
    return g

def _ones(n):
    return np.ones(n)

def _sumx(x, n):
    return np.sum(x) - n / 2.0

def _slsqp(h, n, alpha, maxit):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sumx, 'jac': _ones, 'args': (n,)}]
    try:
        r = minimize(_sf, h.copy(), jac=_gs, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _interp(hs, ns, nd):
    xs = np.linspace(0, 2, ns, endpoint=False)
    xd = np.linspace(0, 2, nd, endpoint=False)
    return np.interp(xd, xs, hs)

def _subgrad(h, n, t_end, max_it=50000):
    """Subgradient method with backtracking on the exact max."""
    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    step = 0.005
    for it in range(max_it):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        kp = c.argmax()
        lag = kp - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            ll = -lag
            g[:n - ll] += (1.0 - h[ll:])
            g[ll:] -= h[:n - ll]
        # Project gradient onto feasible direction (zero-mean to maintain sum)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        accepted = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            ht = h - s * g
            ht = _repair(ht, n)
            ct = _eval(ht)
            if ct < best_c - 1e-15:
                h = ht
                best_h = h.copy()
                best_c = ct
                step = min(step * 1.1, 0.02)
                accepted = True
                break
        if not accepted:
            step *= 0.85
            if step < 1e-7:
                break
    return best_h, best_c

def _local(h, n, rng, t_end):
    """Adaptive pair-move local search."""
    h = h.copy()
    c0 = _eval(h)
    scale = 0.01
    no_imp = 0
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        a, b = h[i] + d, h[j] - d
        if not (0 <= a <= 1 and 0 <= b <= 1):
            continue
        h[i], h[j] = a, b
        c1 = _eval(h)
        if c1 < c0 - 1e-15:
            c0 = c1
            no_imp = 0
            scale = min(scale * 1.03, 0.03)
        else:
            h[i], h[j] = a - d, b + d
            no_imp += 1
            if no_imp > 200:
                scale *= 0.9
                no_imp = 0
                if scale < 1e-7:
                    break
    return h, c0

def _block_search(h, n, rng, t_end):
    """Block perturbation search."""
    h = h.copy()
    c0 = _eval(h)
    for _ in range(20000):
        if time.time() > t_end:
            break
        blk = rng.randint(2, 6)
        i0 = rng.randint(0, n - blk + 1)
        idx = range(i0, i0 + blk)
        old = h[idx].copy()
        # Sinusoidal perturbation within block
        t = np.linspace(0, 1, blk)
        d = 0.02 * rng.randn() * np.sin(np.pi * t)
        nv = old + d
        # Project to [0,1]
        nv = np.clip(nv, 0, 1)
        # Fix sum: adjust one outside element
        diff = h[idx].sum() - nv.sum()
        j = rng.randint(n)
        while j in idx:
            j = rng.randint(n)
        newj = h[j] - diff
        if 0 <= newj <= 1:
            h[idx] = nv
            h[j] = newj
            c1 = _eval(h)
            if c1 < c0 - 1e-15:
                c0 = c1
            else:
                h[idx] = old
                h[j] = newj + diff
    return h, c0

**Idea: Use the structure of the problem**

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-13:
                    break
                if diff > 0:
                    h[idx] = min(1.0, h[idx] + diff)
                else:
                    h[idx] = max(0.0, h[idx] + diff)
                diff = target - h.sum()
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for p in range(2 * n - 1):
        wk = w[p]
        if wk < 1e-15:
            continue
        lag = p - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - x[ll:])
            g[ll:] -= wk * x[:n - ll]
    return g

def _ones(n):
    return np.ones(n)

def _sumx(x, n):
    return np.sum(x) - n / 2.0

def _slsqp(h, n, alpha, maxit):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sumx, 'jac': _ones, 'args': (n,)}]
    try:
        r = minimize(_sf, h.copy(), jac=_gs, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _interp(hs, ns, nd):
    xs = np.linspace(0, 2, ns, endpoint=False)
    xd = np.linspace(0, 2, nd, endpoint=False)
    return np.interp(xd, xs, hs)

def _subgrad(h, n, t_end, max_it=80000):
    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    step = 0.003
    for it in range(max_it):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        kp = c.argmax()
        lag = kp - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            ll = -lag
            g[:n - ll] += (1.0 - h[ll:])
            g[ll:] -= h[:n - ll]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        ok = False
        for s in [step, step * 0.5, step * 0.25]:
            ht = _repair(h - s * g, n)
            ct = _eval(ht)
            if ct < best_c - 1e-16:
                h = ht
                best_h = h.copy()
                best_c = ct
                step = min(step * 1.08, 0.015)
                ok = True
                break
        if not ok:
            step *= 0.88
            if step < 5e-7:
                break
    return best_h, best_c

def _local(h, n, rng, t_end):
    h = h.copy()
    c0 = _eval(h)
    scale = 0.008
    no_imp = 0
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        a, b = h[i] + d, h[j] - d
        if not (0 <= a <= 1 and 0 <= b <= 1):
            continue
        h[i], h[j] = a, b
        c1 = _eval(h)
        if c1 < c0 - 1e-15:
            c0 = c1
            no_imp = 0
            scale = min(scale * 1.04, 0.025)
        else:
            h[i], h[j] = a - d, b + d
            no_imp += 1
            if no_imp > 300:
                scale *= 0.87
                no_imp = 0
                if scale < 5e-7:
                    break
    return h, c0

def _triple_local(h, n, rng, t_end):
    """Triple-move local search for more diversity."""
    h = h.copy()
    c0 = _eval(h)
    scale = 0.006
    no_imp = 0
    while time.time() < t_end:
        i, j, k = rng.randint(n), rng.randint(n), rng.randint(n)
        if len(set([i, j, k])) < 3:
            continue
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        d3 = -(d1 + d2)
        a, b, c_ = h[i] + d1, h[j] + d2, h[k] + d3
        if not all(0 <= v <= 1 for v in [a, b, c_]):
            continue
        h[i], h[j], h[k] = a, b, c_
        c1 = _eval(h)
        if c1 < c0 - 1e-15:
            c0 = c1
            no_imp = 0
            scale = min(scale * 1.03, 0.02)
        else:
            h[i], h[j], h[k] = a - d1, b - d2, c_ - d3
            no_imp += 1
            if no_imp > 400:
                scale *= 0.9
                no_imp = 0
                if scale < 5e-7:
                    break
    return h, c0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    n_init = len(initial_h_values)

        c0 = _eval(h0)
        if c0 > best_c + 0.03:

        for alpha in [500, 5000, 50000, 500000]:

            c = _eval(h)

                best_c = c
                best_h = h.copy()
                best_n = n
                h0 = h.copy()

        te = time.time() + min(60, _tl() - 20)
        h2, c2 = _subgrad(best_h.copy(), best_n, te)

        c = _eval(h)

            best_c = c

    if _tl() > 30:

        h3, c3 = _local(best_h.copy(), best_n, rng, te)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)

c[p] = sum_{j=k}^{n-1} h[j] * (1-h[j-k])

dc[p]/dh[i]:
- i appears as h[j] when j=i, valid if k <= i <= n-1: contribution (1-h[i-k])
- i appears as h[j-k] when j-k=i, j=i+k, valid if k <= i+k <= n-1, i.e., 0 <= i <= n-1-k: contribution -h[i+k]

So:
g[i] += w[p] * (1-h[i-k]) for i in [k, n-1]
g[i] -= w[p] * h[i+k] for i in [0, n-1-k]

In code:
for i in range(k, n): g[i] += w[p] * (1 - h[i-k])
for i in range(0, n-k): g[i] -= w[p] * h[i+k]

Vectorized:
g[k:] += w[p] * (1 - h[:n-k])    # h[i-k] for i in [k, n-1] → h[0], h[1], ..., h[n-1-k]
g[:n-k] -= w[p] * h[k:]          # h[i+k] for i in [0, n-1-k] → h[k], h[k+1], ..., h[n-1]

For k < 0 (let ell = -k):
c[p] = sum_{j=0}^{n-1-ell} h[j] * (1-h[j+ell])

dc[p]/dh[i]:
- i appears as h[j] when j=i, valid if 0 <= i <= n-1-ell: contribution (1-h[i+ell])

In code:
g[:n-ell] += w[p] * (1 - h[ell:])    # h[i+ell] for i in [0, n-1-ell] → h[ell], ..., h[n-1]

Actually, I want to add one more thing: a "grid shift" operation. Since the function is defined on [0,2] with n samples at positions 0, 2/n, 4/n, ..., 2(n-1)/n, shifting the grid by 1/(2n) and re-sampling might find a better alignment. This is like subpixel optimization.

def _grid_shift(h, n, shift):
    """Shift grid by 'shift' sample positions (can be fractional)."""
    s = shift % 1.0
    if s < 0:
        s += 1.0
    if s < 1e-10:
        return h.copy()
    idx0 = np.floor(shift).astype(int) % n
    h0 = np.roll(h, -idx0)
    h1 = np.roll(h, -idx0 - 1)
    return (1 - s) * h0 + s * h1

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    # Only compute for significant weights
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    g = np.zeros(n)
    for p in active:
        wk = w[p]
        lag = p - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - x[ll:])
            g[ll:] -= wk * x[:n - ll]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-13:
                    break
                if diff > 0:
                    st = min(diff, 1.0 - h[idx])
                    h[idx] += st
                    diff -= st
                else:
                    st = min(-diff, h[idx])
                    h[idx] -= st
                    diff += st
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    active = np.where(w > 1e-12)[0]
    g = np.zeros(n)
    for p in active:
        wk = w[p]
        lag = p - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - x[ll:])
            g[ll:] -= wk * x[:n - ll]
    return g

def _ones(n):
    return np.ones(n)

def _sumx(x, n):
    return np.sum(x) - n / 2.0

def _slsqp(h, n, alpha, maxit):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sumx, 'jac': _ones, 'args': (n,)}]
    try:
        r = minimize(_sf, h.copy(), jac=_gs, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-22})
        return _repair(r.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _interp(hs, ns, nd):
    xs = np.linspace(0, 2, ns, endpoint=False)
    xd = np.linspace(0, 2, nd, endpoint=False)
    return np.interp(xd, xs, hs)

def _subgrad(h, n, t_end, max_it=100000):
    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    step = 0.004
    stall = 0
    for _ in range(max_it):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        kp = c.argmax()
        lag = kp - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            ll = -lag
            g[:n - ll] += (1.0 - h[ll:])
            g[ll:] -= h[:n - ll]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        ok = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            ht = _repair(h - s * g, n)
            ct = _eval(ht)
            if ct < best_c - 1e-16:
                h = ht
                best_h = h.copy()
                best_c = ct
                step = min(step * 1.06, 0.012)
                stall = 0
                ok = True
                break
        if not ok:
            stall += 1
            step *= 0.9
            if stall > 500 or step < 3e-7:
                break
    return best_h, best_c

def _local(h, n, rng, t_end):
    h = h.copy()
    c0 = _eval(h)
    scale = 0.008
    no_imp = 0
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        a, b = h[i] + d, h[j] - d
        if not (0 <= a <= 1 and 0 <= b <= 1):
            continue
        h[i], h[j] = a, b
        c1 = _eval(h)
        if c1 < c0 - 1e-15:
            c0 = c1
            no_imp = 0
            scale = min(scale * 1.04, 0.025)
        else:
            h[i], h[j] = a - d, b + d
            no_imp += 1
            if no_imp > 300:
                scale *= 0.87
                no_imp = 0
                if scale < 4e-7:
                    break
    return h, c0

def _triple(h, n, rng, t_end):
    h = h.copy()
    c0 = _eval(h)
    scale = 0.005
    no_imp = 0
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        k = rng.randint(n)
        if len({i, j, k}) < 3:
            continue
        d1, d2 = scale * rng.randn(), scale * rng.randn()
        d3 = -(d1 + d2)
        a, b, c_ = h[i] + d1, h[j] + d2, h[k] + d3
        if not all(0 <= v <= 1 for v in (a, b, c_)):
            continue
        h[i], h[j], h[k] = a, b, c_
        c1 = _eval(h)
        if c1 < c0 - 1e-15:
            c0 = c1
            no_imp = 0
            scale = min(scale * 1.03, 0.018)
        else:
            h[i], h[j], h[k] = a - d1, b - d2, c_ - d3
            no_imp += 1
            if no_imp > 400:
                scale *= 0.88
                no_imp = 0
                if scale < 3e-7:
                    break
    return h, c0

def _shift_try(h, n):
    best = (h.copy(), _eval(h))
    for sh in np.linspace(-0.5, 0.5, 11):
        if abs(sh) < 1e-10:
            continue
        s = sh % 1.0
        i0 = int(np.floor(sh)) % n
        h0 = np.roll(h, -i0)
        h1 = np.roll(h, -i0 - 1)
        hs = (1 - s) * h0 + s * h1
        hs = _repair(hs, n)
        cs = _eval(hs)
        if cs < best[1] - 1e-14:
            best = (hs, cs)
    return best

        if cs < best_c:
            best_c, best_h = cs, hs

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
