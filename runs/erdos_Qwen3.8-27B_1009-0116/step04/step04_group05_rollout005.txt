        g = np.zeros(n)
        for ki in idx:
            L = ki - (n - 1)
            if L == 0:
                g += (1.0 - 2.0 * h)
            elif L > 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                mm = -L
                g[:n - mm] += (1.0 - h[mm:])
                g[mm:] -= h[:n - mm]
        g /= len(idx)
        g -= g.mean()
        gn = np.dot(g, g)
        if gn < 1e-20:
            break
        st = min(step / np.sqrt(gn), 0.3)
        ok = False
        for f in [1.0, 0.6, 0.35, 0.15, 0.05]:
            hn = repair(h - st * f * g, n)
            cn = c5v(hn, dx)
            if cn < bc - 1e-18:
                h = hn
                bc, bh = cn, h.copy()
                ok = True
                break
        if ok:
            step = min(step * 1.08, 0.5)
            stall = 0
        else:
            step *= 0.7
            stall += 1
            if stall > 15:
                break
    return bh, bc

def cd_phase(h, n, dx, rng, t0, lim, rounds=400):
    bh, bc = h.copy(), c5v(h, dx)
    h = h.copy()
    cc = bc
    halt = 0
    for rd in range(rounds):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(300):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in (0.008, 0.003, 0.001, 0.0005, -0.008, -0.003, -0.001, -0.0005):
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        if imp:
            halt = 0
        else:
            halt += 1
            if halt >= 5:
                break
    return bh, bc

def spec_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.12, 0.2, 0.35, 0.5, 0.65, 0.5, 0.3, 0.15]:
        h = bh.copy()
        for _ in range(200):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, nm.shape[0])
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def nelder_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    h = h.copy()
    # Powell's method (derivative-free, handles bounds via repair)
    def pow_obj(x):
        xr = repair(x, n)
        return c5v(xr, dx)
    try:
        r = minimize(pow_obj, h, method='Powell',
                     options={'maxiter': 300, 'ftol': 1e-16, 'maxfev': 5000})
        hr = repair(r.x, n)
        c = c5v(hr, dx)
        if c < bc:
            bh, bc = hr, c
    except Exception:
        pass
    return bh, bc

def multi_start_candidates(n, dx, seed):
    rng = np.random.RandomState(seed)
    cands = []
    # Block: 1 on first half, 0 on second
    h1 = np.zeros(n)
    h1[:n // 2] = 1.0
    cands.append(repair(h1, n))
    # Alternating blocks
    blk = max(n // 20, 2)
    h2 = np.zeros(n)
    for i in range(0, n, 2 * blk):
        h2[i:i + blk] = 1.0
    cands.append(repair(h2, n))
    # Sine
    x = np.linspace(0, 2, n, endpoint=False)
    h3 = 0.5 + 0.45 * np.sin(2 * np.pi * x / 2.0)
    cands.append(repair(h3, n))
    # Random smooth
    freqs = rng.randint(1, 8, size=5)
    amps = rng.uniform(0.1, 0.4, size=5)
    phs = rng.uniform(0, 2 * np.pi, size=5)
    h4 = 0.5 + sum(amps[k] * np.sin(2 * np.pi * freqs[k] * x / 2.0 + phs[k]) for k in range(5))
    cands.append(repair(h4, n))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h = repair(np.array(initial_h_values, dtype=float).copy(), n)

    # Quick multi-start to find a good basin
    best_h, best_c = h.copy(), c5v(h, dx)
    for i, cand in enumerate(multi_start_candidates(n, dx, seed + i * 100)):
        c = c5v(cand, dx)
        if c < best_c:
            best_c, best_h = c, cand.copy()

    # Phase A: L-BFGS-B with increasing smoothness
    for alpha in [50, 300, 2000, 15000, 100000, 500000]:
        if time.time() - t0 > budget_s - 350:
            break
        hc = lbfgs_run(best_h, n, alpha, t0, budget_s - 350, maxit=500)
        c = c5v(hc, dx)
        if c < best_c:
            best_c, best_h = c, hc

    # Phase B: Powell (derivative-free)
    if time.time() - t0 < budget_s - 250:
        r, c = nelder_phase(best_h, n, dx, t0, budget_s - 250)
        if c < best_c:
            best_c, best_h = c, r

    # Phase C: Subgradient with many iterations
    r, c = subgrad_phase(best_h, n, dx, t0, budget_s - 200, iters=4000)
    if c < best_c:
        best_c, best_h = c, r

    # Phase D: L-BFGS-B again (higher smoothness)
    for alpha in [200000, 1000000]:
        if time.time() - t0 > budget_s - 120:
            break
        hc = lbfgs_run(best_h, n, alpha, t0, budget_s - 120, maxit=600)
        c = c5v(hc, dx)
        if c < best_c:
            best_c, best_h = c, hc

    # Phase E: Spectral equalization
    r, c = spec_phase(best_h, n, dx, t0, budget_s - 80)
    if c < best_c:
        best_c, best_h = c, r

    # Phase F: Coordinate descent (long)
    r, c = cd_phase(best_h, n, dx, rng, t0, budget_s - 40, rounds=500)
    if c < best_c:
        best_c, best_h = c, r

    # Phase G: Final subgradient polish (very small steps)
    r, c = subgrad_phase(best_h, n, dx, t0, budget_s - 10, iters=5000)
    if c < best_c:
        best_c, best_h = c, r

    # Phase H: Try 2x resolution if time and size allow
    n2 = n * 2
    if n2 <= 1000 and time.time() - t0 < budget_s - 60:
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, best_h, period=2.0), n2)
        d2 = 2.0 / n2
        c2 = c5v(h2, d2)
        bh2, bc2 = h2.copy(), c2
        for alpha in [1000, 20000, 200000]:
            if time.time() - t0 > budget_s - 20:
                break
            hc = lbfgs_run(bh2, n2, alpha, t0, budget_s - 20, maxit=300)
            cc = c5v(hc, d2)
            if cc < bc2:
                bc2, bh2 = cc, hc
        r2, c2f = cd_phase(bh2, n2, d2, rng, t0, budget_s - 5, rounds=200)
        if c2f < bc2:
            bc2, bh2 = c2f, r2
        if bc2 < best_c:
            best_c, best_h = bc2, bh2
            n = n2

    best_h = repair(best_h, n)
    best_c = c5v(best_h, 2.0 / n)
    return (best_h, best_c, n)
