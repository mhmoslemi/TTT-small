```python
def _subgrad_step(h, n, step_sz):
    """Take one projected subgradient step using the active lag."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    lag = k_star - (n - 1)
    # Gradient of c_{k_star} w.r.t. h
    g = np.zeros(n)
    i1, j1 = max(0, lag), min(n, n + lag)
    if i1 < j1:
        g[i1:j1] += (1.0 - h[i1 - lag:j1 - lag])
    i2, j2 = max(0, -lag), min(n, n - lag)
    if i2 < j2:
        g[i2:j2] -= h[i2 + lag:j2 + lag]
    # Projected gradient step
    h_new = h - step_sz * g
    h_new = _repair(h_new, n)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=np.float64).copy(), 0., 1.)
    t = 0.5 * n; s = h.sum()
    if s < 1e-14: return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s; np.clip(h, 0., 1., out=h)
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1. - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1. - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12: break
    return h

def _ev(h):
    return float(np.max(np.correlate(h, 1. - h, mode='full')) * 2. / len(h))

def _sf(h, al):
    c = np.correlate(h, 1. - h, mode='full'); m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(al * (c - m), -500, 500)))) / al

def _sg(h, al):
    n = len(h); c = np.correlate(h, 1. - h, mode='full'); m = c.max()
    v = np.clip(al * (c - m), -500, 500)
    w = np.exp(v - v.max()); w /= w.sum(); g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14: continue
        lg = k - (n - 1)
        a, b = max(0, lg), min(n, n + lg)
        if a < b: g[a:b] += wk * (1. - h[a - lg:b - lg])
        a, b = max(0, -lg), min(n, n - lg)
        if a < b: g[a:b] -= wk * h[a + lg:b + lg]
    return g

def _se(x, n): return np.sum(x) - 0.5 * n
def _sej(x, n): return np.ones(len(x))

def _pf(h, al, p, n): return _sf(h, al) + p * (np.sum(h) - 0.5 * n) ** 2
def _pg(h, al, p, n):
    g = _sg(h, al); return g + 2. * p * (np.sum(h) - 0.5 * n)

def _slsqp(h, n, al, mi=400):
    b = [(0., 1.)] * n
    cs = [{'type': 'eq', 'fun': _se, 'jac': _sej, 'args': (n,)}]
    try:
        r = minimize(_sf, h.copy(), method='SLSQP', jac=_sg, args=(al,),
                     bounds=b, constraints=cs, options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except: return _repair(h, n)

def _lbfgs(h, n, al, p, mi=5000):
    b = [(0., 1.)] * n
    try:
        r = minimize(_pf, h.copy(), method='L-BFGS-B', jac=_pg,
                     args=(al, p, n), bounds=b, options={'maxiter': mi, 'ftol': 1e-20})
        return _repair(r.x, n)
    except: return _repair(h, n)
```

```python
def _lpair(h, n, rng, te):
    c = _ev(h); sc = 0.005; ni = 0
    for _ in range(10**7):
        if time.time() > te: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = sc * rng.randn()
        nv, ov = h[i] + d, h[j] - d
        if not (0 <= nv <= 1 and 0 <= ov <= 1): continue
        h[i], h[j] = nv, ov
        cn = _ev(h)
        if cn < c - 1e-15:
            c = cn; ni = 0; sc = min(sc * 1.02, 0.015)
        else:
            h[i], h[j] = nv - d, ov + d; ni += 1
            if ni > 400:
                sc *= 0.88; ni = 0
                if sc < 1e-7: break
    return h, c

def _bpush(h, n, rng, te):
    c = _ev(h)
    for _ in range(5 * 10**5):
        if time.time() > te: break
        i = rng.randint(n); ov = h[i]
        h[i] = 0. if rng.rand() < 0.5 else 1.
        d = h[i] - ov
        if abs(d) < 1e-8: h[i] = ov; continue
        ok = False
        for _a in range(50):
            j = rng.randint(n)
            if j != i and 0. <= h[j] - d <= 1.: ok = True; break
        if not ok: h[i] = ov; continue
        h[j] -= d; cn = _ev(h)
        if cn < c - 1e-15: c = cn
        else: h[i] = ov; h[j] += d
    return h, c

def _fine(h, n, rng, te):
    c = _ev(h)
    for m in [50, 20, 10, 5, 2, 1]:
        s = m * 1e-6
        for _ in range(3 * 10**6):
            if time.time() > te: return h, c
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = (1. if rng.rand() < .5 else -1.) * s
            nv, ov = h[i] + d, h[j] - d
            if 0 <= nv <= 1 and 0 <= ov <= 1:
                h[i], h[j] = nv, ov
                if _ev(h) < c - 1e-17: c = _ev(h)
                else: h[i], h[j] = nv - d, ov + d
    return h, c
```
