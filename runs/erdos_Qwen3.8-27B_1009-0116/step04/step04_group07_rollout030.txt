def fit_fourier(h, n, n_harmonics):
    """Fit Fourier coefficients to a given h (least squares on f = logit(h/2))."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.clip(h, 1e-6, 1-1e-6)
    f = 0.5 * np.log(h / (1.0 - h))  # f = 0.5 * logit(h)
    # Actually: h = sigmoid(2f), so f = 0.5 * logit(h) = 0.5 * ln(h/(1-h))
    
    A = np.zeros((n, 2 * n_harmonics))
    for j in range(n_harmonics):
        k = 2 * j + 1
        A[:, 2*j] = np.cos(np.pi * k * x)
        A[:, 2*j+1] = np.sin(np.pi * k * x)
    coeffs, _, _, _ = np.linalg.lstsq(A, f, rcond=None)
    return coeffs
