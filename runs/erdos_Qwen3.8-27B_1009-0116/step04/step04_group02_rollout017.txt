def project(h, target):
    """Project h onto {sum(x)=target, 0<=x<=1}"""
    n = len(h)
    # Binary search on lambda
    lo, hi = h.min() - target, h.max()  # rough bounds
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Final adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Adjust the free variables
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if free.sum() > 0:
            adj = diff / free.sum()
            h_proj[free] += adj
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

    lo, hi = h.min() - target / n - 1, h.max() + 1
    for _ in range(80):

        hp = np.clip(h - mid, 0.0, 1.0)

            lo = mid

            hi = mid

    if abs(diff) > 1e-13:

        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
        else:

            if diff > 0:

                    diff -= add
                else:
                    sub = min(-diff, h_proj[idx])
                    h_proj[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:

    return h_proj

def lbfgs_smooth(h, n, alpha, target, maxiter=300):

        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        val = m + np.log(np.sum(np.exp(s))) / alpha
        return val + 1000.0 * (np.sum(x) - target) ** 2

                      bounds=[(0.0, 1.0)] * n,

        return hc, compute_c5(hc, dx)
    except Exception:

from scipy.optimize import minimize

def project_budget(h, target):
    n = len(h)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        free = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if free.sum() > 0:
            hp[free] += diff / free.sum()
            hp = np.clip(hp, 0.0, 1.0)
        else:
            idx = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
            for ii in idx:
                if diff > 0:
                    a = min(diff, 1.0 - hp[ii]); hp[ii] += a; diff -= a
                else:
                    a = min(-diff, hp[ii]); hp[ii] -= a; diff += a
                if abs(diff) < 1e-14:
                    break
    return hp

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    n = len(h)
    h = h.copy()

                    hi = m2
                else:
                    lo = m1

            break
    return h, best_c5

def lbfgs_smooth(h, n, alpha, target, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha + 500.0 * (np.sum(x) - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        hc = project_budget(res.x, target)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def subgradient_opt(h, target, n_iters=5000, deadline=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = compute_c5(h, dx)
    lr = 1.0
    for it in range(n_iters):
        if deadline is not None and time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        v1 = (np.arange(n) - lag >= 0) & (np.arange(n) - lag < n)
        g[v1] += (1.0 - h[np.arange(n)[v1] - lag])
        v2 = (np.arange(n) + lag >= 0) & (np.arange(n) + lag < n)
        g[v2] -= h[np.arange(n)[v2] + lag]
        h_new = project_budget(h - lr * g, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5 - 1e-15:
            h, c5 = h_new, c5_new
            lr = min(lr * 1.02, 3.0)
        else:
            lr *= 0.7
            if lr < 0.01:
                lr = 0.01
    return h, c5

def perturb_diverse(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(10, n - 10))
    elif kind == 1:
        f = rng.randint(1, 8); ph = rng.uniform(0, 6.28); amp = rng.uniform(0.04, 0.15)
        hp += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, rng.uniform(0.02, 0.07), n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 7)
        t = np.arange(n) / n
        hp += rng.uniform(0.04, 0.12) * np.sin(2*np.pi*f1*t) * np.cos(2*np.pi*f2*t)
    elif kind == 4:
        bs = rng.randint(0, n - 15); bl = rng.randint(5, 18)
        hp[bs:bs+bl] = 1.0 - hp[bs:bs+bl]
    elif kind == 5:
        sigma = rng.uniform(0.02, 0.06)
        for k in range(3):
            hp = np.concatenate([hp[-1:], hp, hp[0]])
            hp[1:-1] = 0.25*hp[:-2] + 0.5*hp[1:-1] + 0.25*hp[2:]
            hp = np.clip(hp[1:-1], 0, 1)
    else:
        hp *= rng.uniform(0.8, 1.2)
    return project_budget(hp, n / 2.0)

def pair_search(h, n, dx, rng, deadline, scales, iters_per_scale=40000):
    h = h.copy()
    c5 = compute_c5(h, dx)
    for scale in scales:
        for _ in range(iters_per_scale):
            if time.time() >= deadline:
                break
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:
                continue
            h[i], h[j] = ni, nj
            c5n = float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)
            if c5n < c5 - 1e-15:
                c5 = c5n
            else:
                h[i] -= d; h[j] += d
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_budget(h_init.copy(), target)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    def rem():
        return budget_s - (time.time() - t_start)

    # Phase 1: Coordinate descent warmup (50s)
    t_end = t_start + 50
    while time.time() < t_end:
        h, c5 = coord_descent(h, target, n_sweeps=40, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: L-BFGS multi-alpha (70s)
    t_end = t_start + 120
    for alpha in [200, 1000, 5000, 20000]:
        if time.time() >= t_end:
            break
        hc, cc = lbfgs_smooth(h, n, alpha, target, 250)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()
            h = hc
        h, c5 = coord_descent(h, target, n_sweeps=30, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: Multi-restart diverse (250s)
    t_end = t_start + 370
    for r in range(40):
        if time.time() >= t_end or rem() < 250:
            break
        hp = perturb_diverse(best_h, n, rng, r % 7)
        hc, cc = coord_descent(hp, target, n_sweeps=30, rng=rng)
        if cc < best_c5:
            hc, cc = lbfgs_smooth(hc, n, 3000, target, 200)
            hc, cc = coord_descent(hc, target, n_sweeps=30, rng=rng)
            if cc < best_c5:
                best_c5, best_h, h = cc, hc.copy(), hc

    # Phase 4: Subgradient + coord descent (120s)
    if rem() > 200:
        t_end = time.time() + min(120, rem() - 120)
        hc, cc = subgradient_opt(best_h, target, n_iters=5000, deadline=t_end)
        hc, cc = coord_descent(hc, target, n_sweeps=40, rng=rng)
        if cc < best_c5:
            best_c5, best_h, h = cc, hc.copy(), hc

    # Phase 5: More multi-restart (150s)
    t_end = t_start + 700
    for r in range(20):
        if time.time() >= t_end or rem() < 150:
            break
        hp = perturb_diverse(best_h, n, rng, (r + 3) % 7)
        hc, cc = coord_descent(hp, target, n_sweeps=25, rng=rng)
        hc, cc = lbfgs_smooth(hc, n, 8000, target, 200)
        hc, cc = coord_descent(hc, target, n_sweeps=25, rng=rng)
        if cc < best_c5:
            best_c5, best_h, h = cc, hc.copy(), hc

    # Phase 6: Ultra-fine coordinate descent (120s)
    if rem() > 100:
        t_end = time.time() + min(120, rem() - 60)
        h = best_h.copy()
        while time.time() < t_end:
            h, c5 = coord_descent(h, target, n_sweeps=60, rng=rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 7: Final L-BFGS high-alpha (30s)
    if rem() > 40:
        t_end = time.time() + min(25, rem() - 15)
        for alpha in [50000, 200000]:
            if time.time() >= t_end:
                break
            hc, cc = lbfgs_smooth(best_h, n, alpha, target, 200)
            if cc < best_c5:
                best_c5, best_h = cc, hc.copy()
        best_h, best_c5 = coord_descent(best_h, target, n_sweeps=80, rng=rng)

    # Phase 8: Pair local search (remaining)
    if rem() > 15:
        t_end = time.time() + min(rem() - 5, 40)
        best_h, best_c5 = pair_search(best_h, n, dx, rng, t_end,
                                      [0.008, 0.003, 0.001, 0.0003], 30000)
        best_h, best_c5 = coord_descent(best_h, target, n_sweeps=30, rng=rng)

    best_h = project_budget(best_h, best_n if 'best_n' in dir() else n)
    best_c5 = compute_c5(best_h, 2.0 / n)
    return (best_h, best_c5, n)
