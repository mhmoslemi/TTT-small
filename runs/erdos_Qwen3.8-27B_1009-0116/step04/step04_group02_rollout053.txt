def fast_swap_eval(C, h, i, j, delta, n):
    """Compute max of C after h[i]+=delta, h[j]-=delta, without full recompute.
    Returns (new_max, changed_indices) or just new_max."""
    # ΔC[m] = delta*(1-2h[m-i]) for m-i in [0,n-1] (i.e., m in [i, i+n-1])
    #         - delta*(1-2h[m-j]) for m-j in [0,n-1] (i.e., m in [j, j+n-1])
    
    # We only need to check max at the affected positions
    # Unaffected positions haven't changed, so their max is the old max excluding affected positions
    
    # For n=140, the affected range is at most 2n-2 = 278 positions out of 279.
    # So we basically need to recompute everything. Not much gain.
    
    # Actually, for n=140, affected range is:
    # [i, i+n-1] ∪ [j, j+n-1] which is almost the entire [0, 2n-2]
    # So we can't avoid computing most of the new C.
    
    # But we CAN compute it in O(n) instead of O(n²)!
    # Because ΔC[m] is just a lookup into h.
    
    nh = h.copy()
    nh[i] += delta
    nh[j] -= delta
    
    # Compute full new C = C + delta*(1-2h[m-i])_valid - delta*(1-2h[m-j])_valid
    new_C = C.copy()

    lo_i = max(0, i)  # m-i >= 0 => m >= i
    hi_i = min(2*n-2, i+n-1)  # m-i <= n-1 => m <= i+n-1
    if lo_i <= hi_i:
        k_range = np.arange(lo_i - i, hi_i - i + 1)  # m-i values
        new_C[lo_i:hi_i+1] += delta * (1.0 - 2.0 * h[k_range])
    
    # Subtract contribution from j
    lo_j = max(0, j)
    hi_j = min(2*n-2, j+n-1)
    if lo_j <= hi_j:
        k_range = np.arange(lo_j - j, hi_j - j + 1)
        new_C[lo_j:hi_j+1] -= delta * (1.0 - 2.0 * h[k_range])
    
    return np.max(new_C)
