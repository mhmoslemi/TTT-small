    h = h.copy()

For the top-k equalization:
- Find top 3 lags: k1, k2, k3
- The gradient of corr(k_i) at h is g_i
- We want to find δh such that corr(k_i + δh) ≈ corr(k_j + δh) for all i,j
- First-order: c_i + g_i · δh = c_j + g_j · δh → (g_i - g_j) · δh = c_j - c_i
- Minimize ||δh||^2 subject to these constraints → least squares
- Project onto {sum(δh)=0, 0≤h+δh≤1}

This is a nice linear algebra approach! Let me implement it.

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _actgrad(h, n, frac=0.02):
    c = _corr(h)
    cm = c.max()
    spread = cm - c.min()
    thresh = cm - (frac * spread if spread > 1e-12 else 0.0)
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)

        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g, c

def _sobj(x, alpha, n):

    m = c.max()
    s_raw = alpha * (c - m)
    s = np.clip(s_raw - s_raw.max(), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 8.0 * (x.sum() - n / 2.0) ** 2

        else:

            g[M:] -= pw[k] * x[:n - M]
    g /= alpha
    g += 16.0 * (x.sum() - n / 2.0)
