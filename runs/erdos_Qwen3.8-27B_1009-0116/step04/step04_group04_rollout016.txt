def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)

        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)

    if abs(d) > 1e-12:

        for i in idx:

    return h

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _grad_for_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        return g
    if L > 0:
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    else:
        L2 = -L
        g[L2:] += (1.0 - h[:n - L2])
        g[:n - L2] -= h[L2:]
    return g

def _subgrad_optimize(h, dx, t_end, rng, beta_start=30, beta_max=500, n_active=30):
    n = len(h)
    target = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    beta = beta_start
    lr = 0.008
    patience = 0
    iter_count = 0

    while time.time() < t_end:
        c = _corr(h)
        m = c.max()
        cur = m * dx
        iter_count += 1

        if cur < best - 1e-16:
            best = cur
            patience = 0
        else:
            patience += 1

        if patience > 3000:
            amp = 0.02 * (1 + patience // 5000)
            h += amp * (rng.random(n) - 0.5)
            h = _repair(h, target)
            patience = 0
            lr = 0.008
            continue

        s = beta * (c - m)
        w = np.exp(s)
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
        else:
            w /= wsum

        active = np.where(w > 1e-8)[0]
        if len(active) > n_active:
            top_idx = np.argsort(w)[::-1][:n_active]
            w = np.zeros(len(c))
            w[top_idx] = np.exp(beta * (c[top_idx] - m))
            ts = w[top_idx].sum()
            w[top_idx] /= ts
            active = top_idx

        grad = np.zeros(n)
        for k_i in active:
            wi = w[k_i]
            L = int(k_i) - (n - 1)
            if L == 0:
                continue
            grad += wi * _grad_for_lag(h, L)

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            h -= lr * grad / gnorm
            h = _repair(h, target)

        lr *= 0.9997
        if patience % 500 == 0:
            beta = min(beta * 1.3, beta_max)

    return h, best

def _targeted_search(h, dx, rng, t_end, n_iter=150000):
    n = len(h)
    target = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for _ in range(n_iter):
        if time.time() > t_end:
            return h, best
        c = _corr(h)
        k_max = int(np.argmax(c))
        L = k_max - (n - 1)
        if L == 0:
            i, j = rng.randint(n), rng.randint(n)
        elif L > 0:
            i = rng.randint(n - L)
            j = i + L
        else:
            L2 = -L
            i = rng.randint(L2, n)
            j = i - L2
        d = 0.004 * rng.choice([-1.0, 1.0])
        h[i] += d; h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            cv = _c5(h, dx)
            if cv < best:
                best = cv
                h = _repair(h, target)
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return h, best

def _local_search(h, dx, rng, t_end, n_iter=400000):
    n = len(h)
    target = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per = max(50, n_iter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return h, best
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                cv = _c5(h, dx)
                if cv < best:
                    best = cv
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _binary_round(h, n):
    target = n // 2
    h = np.clip(h, 0.0, 1.0)
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb
