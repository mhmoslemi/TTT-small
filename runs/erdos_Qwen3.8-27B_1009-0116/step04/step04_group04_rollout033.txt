```python
import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, dx):
    return float(np.max(corr_full(h)) * dx)

def grad_c5_smooth(h, alpha):
    n = len(h)
    c = corr_full(h)
    cmax = c.max()
    w = np.exp(alpha * (c - cmax))
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n), cmax
    w = w / wsum
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    active = np.where(w > 1e-8)[0]
    for idx in active:
        wi = w[idx]
        lag = idx - (n - 1)
        if lag > 0:
            upper = n - lag
            grad[:upper] += wi * (1.0 - h[lag:lag+upper])
            grad[lag:] -= wi * h[:upper]
        elif lag < 0:
            s = -lag
            upper = n - s
            grad[s:] += wi * (1.0 - h[:upper])
            grad[:upper] -= wi * h[s:]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad, cmax

def project(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on shift t: find t such that sum(clip(h-t, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0, 1)

def repair(h, target):
    """Repair h to satisfy sum(h) = target and 0<=h<=1"""
    h = np.clip(h, 0, 1).copy()
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    # Water-filling
    if d > 0:
        # Need to add d: add to elements with most room
        order = np.argsort(1.0 - h)
        for i in order:
            add = min(d, 1.0 - h[i])
            h[i] += add
            d -= add
            if d < 1e-14:
                break
    else:
        order = np.argsort(h)
        for i in order:
            sub = min(-d, h[i])
            h[i] -= sub
            d += sub
            if d > -1e-14:
                break
    return h

def gradient_descent(h, dx, t_end, alpha_init=100, alpha_max=50000, 
                     lr_init=0.1, lr_decay=0.999, iters_per_alpha=200):
    n = len(h)
    target = n / 2.0
    best_c5 = c5_val(h, dx)
    best_h = h.copy()
    h = h.copy()
    alpha = alpha_init
    lr = lr_init
    it = 0
    
    while time.time() < t_end:
        for _ in range(iters_per_alpha):
            if time.time() > t_end:
                break
            grad, cmax = grad_c5_smooth(h, alpha)
            h_new = h - lr * grad
            h_new = project(h_new, target)
            c = c5_val(h_new, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
                h = h_new
            else:
                # Reject, reduce lr
                h = best_h.copy()  # go back to best
                lr *= 0.9
            it += 1
        alpha *= 2.0
        if alpha > alpha_max:
            alpha = alpha_max
        lr *= 0.8
        if lr < 1e-6:
            lr = 1e-6
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, dx):
    return float(np.max(corr_full(h)) * dx)

def grad_logsumexp(h, alpha):
    n = len(h)
    c = corr_full(h)
    cmax = c.max()
    shifted = alpha * (c - cmax)
    # Avoid overflow
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    wsum = w.sum()
    w = w / wsum
    
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        wi = w[idx]
        lag = int(idx - (n - 1))
        if lag > 0:
            upper = n - lag
            grad[:upper] += wi * (1.0 - h[lag:lag + upper])
            grad[lag:] -= wi * h[:upper]
        elif lag < 0:
            s = -lag
            upper = n - s
            grad[s:] += wi * (1.0 - h[:upper])
            grad[:upper] -= wi * h[s:]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad, cmax + np.log(wsum) / alpha

def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection on shift"""
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for i in order:
        if d > 0:
            a = min(d, 1.0 - h[i])
            h[i] += a
            d -= a
        else:
            a = min(-d, h[i])
            h[i] -= a
            d += a
        if abs(d) < 1e-14:
            break
    return h

def grad_descent(h, dx, t_end, alpha_start=200, alpha_end=100000, n_alpha_steps=30):
    n = len(h)
    target = n / 2.0

                    h = h_try
                    best_c5 = c_try
                    best_h = h.copy()
```

```python
def pair_search(h, dx, rng, t_end, n_iter=50000):
    n = len(h)
    target = n / 2.0
    best = c5_val(h, dx)
    h = h.copy()
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        # Pick worst lag
        c = corr_full(h)
        k_max = int(np.argmax(c) - (n - 1))
        
        # Try to fix it
        if k_max > 0:
            upper = n - k_max
            # Find pair (i, i+k_max) where reducing h[i]*(1-h[i+k_max]) helps most
            vals = h[:upper] * (1.0 - h[k_max:])
            i = int(np.argmax(vals))
            # Try moving weight from i to i+k_max or vice versa
            d = rng.uniform(0.01, 0.05)
            h[i] -= d
            h[i + k_max] += d
            if 0 <= h[i] and h[i + k_max] <= 1:
                c_new = c5_val(h, dx)
                if c_new < best:
                    best = c_new
                else:
                    h[i] += d
                    h[i + k_max] -= d
        elif k_max < 0:
            s = -k_max
            vals = h[s:] * (1.0 - h[:n-s])
            i = int(np.argmax(vals)) + s
            d = rng.uniform(0.01, 0.05)
            h[i] -= d
            h[i - s] += d

                c_new = c5_val(h, dx)
                if c_new < best:
                    best = c_new
                else:
                    h[i] += d
                    h[i - s] -= d
        else:
            # Lag 0: reduce h[i]*(1-h[i]) by pushing toward 0 or 1
            vals = h * (1.0 - h)
            i = int(np.argmax(vals))
            if h[i] > 0.5:
                d = min(h[i] - 0.5, 0.05)
                h[i] -= d
                j = int(np.argmin(h))
                h[j] += d
            else:
                d = min(0.5 - h[i], 0.05)
                h[i] += d
                j = int(np.argmax(h))
                h[j] -= d
            if 0 <= h[i] and 0 <= h[j] <= 1:

                        h[i] -= d
                        h[j] += d
                    else:
                        h[i] += d
                        h[j] -= d
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, dx):
    return float(np.max(corr_full(h)) * dx)

def grad_lse(h, alpha):
    n = len(h)
    c = corr_full(h)
    cmax = c.max()
    sh = np.clip(alpha * (c - cmax), -500, 500)
    w = np.exp(sh)
    ws = w.sum()
    w /= ws
    obj = cmax + np.log(ws) / alpha
    grad = np.zeros(n)
    for idx in np.where(w > 1e-8)[0]:
        wi = w[idx]
        lag = int(idx - (n - 1))
        if lag > 0:
            u = n - lag
            grad[:u] += wi * (1.0 - h[lag:lag + u])
            grad[lag:] -= wi * h[:u]
        elif lag < 0:
            s = -lag
            u = n - s
            grad[s:] += wi * (1.0 - h[:u])
            grad[:u] -= wi * h[s:]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return obj, grad

def proj(h, tgt):
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def fix(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
    for i in idx:
        if d > 0:
            a = min(d, 1.0 - h[i]); h[i] += a; d -= a
        else:
            a = min(-d, h[i]); h[i] -= a; d += a
        if abs(d) < 1e-14:
            break
    return h

def interp(h, m):
    return np.interp(np.linspace(0, 1, m), np.linspace(0, 1, len(h)), h)

def gd(h, dx, t_end, alphas, base_lr=0.5):
    n = len(h)
    tgt = n / 2.0
    bc5 = c5_val(h, dx)
    bh = h.copy()
    h = h.copy()
    for a in alphas:
        if time.time() > t_end:
            break
        lr = base_lr / (1.0 + a * 0.001)
        for _ in range(3000):
            if time.time() > t_end:
                break
            _, g = grad_lse(h, a)
            s = lr
            ok = False
            for _ in range(50):
                ht = proj(h - s * g, tgt)
                ct = c5_val(ht, dx)
                if ct < bc5:
                    h = ht; bc5 = ct; bh = h.copy(); ok = True; break
                s *= 0.5
            if not ok:
                h = bh.copy()
    return bh, bc5

def lbfgs(h, dx, a, mi):
    n = len(h)
    tgt = n / 2.0
    bds = [(0, 1)] * n
    def f(x):
        o, g = grad_lse(x, a)
        return o, g
    try:
        r = minimize(f, h, jac=True, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-12})
        hc = fix(r.x, tgt)
        return hc, c5_val(hc, dx)
    except:
        return h, c5_val(h, dx)

def slsqp(h, dx, a, mi):
    n = len(h)
    tgt = n / 2.0
    bds = [(0, 1)] * n
    cn = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n)}]
    def f(x):
        o, g = grad_lse(x, a)
        return o, g
    try:
        r = minimize(f, h, jac=True, method='SLSQP', bounds=bds,
                     constraints=cn, options={'maxiter': mi, 'ftol': 1e-16})
        hc = fix(r.x, tgt)
        c = c5_val(hc, dx)
        if c < c5_val(h, dx):
            return hc, c
    except:
        pass
    return h, c5_val(h, dx)

def pair(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    bc = c5_val(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = corr_full(h)
        km = int(np.argmax(c))
        lag = km - (n - 1)
        if lag > 0:
            u = n - lag
            v = h[:u] * (1.0 - h[lag:])
            i = int(np.argmax(v))
            i2 = i + lag
        elif lag < 0:
            s = -lag
            u = n - s
            v = h[s:] * (1.0 - h[:u])
            i = int(np.argmax(v))
            i2 = i
        else:
            v = h * (1.0 - h)
            i = int(np.argmax(v))
            i2 = int(np.argmin(h)) if h[i] > 0.5 else int(np.argmax(h))
        if i == i2:
            continue
        d = rng.uniform(0.003, 0.03)
        for _ in range(8):
            h[i] -= d; h[i2] += d
            if 0 <= h[i] and 0 <= h[i2] <= 1:
                cn = c5_val(h, dx)
                if cn < bc:
                    bc = cn; h = fix(h, tgt); break
                else:
                    h[i] += d; h[i2] -= d; d *= 0.5
            else:
                h[i] += d; h[i2] -= d; d *= 0.5
            if d < 1e-6:
                break
    return h, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = fix(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    bh = h0.copy()

    bn = n0
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: GD at n0
    t_end = min(t0 + 120, t0 + budget_s - 500)
    bh, bc5 = gd(bh, dx0, t_end, [50, 200, 800, 3000, 10000, 40000, 150000], 1.0)
    
    # Phase 2: L-BFGS at n0
    t_end = min(t0 + 250, t0 + budget_s - 400)
    for a in [3000, 15000, 60000]:
        if rem() < 350:
            break
        bh, bc5 = lbfgs(bh, dx0, a, 800)
    
    # Phase 3: SLSQP at n0
    t_end = min(t0 + 400, t0 + budget_s - 300)
    for a in [10000, 40000, 150000]:
        if rem() < 250:
            break
        bh, bc5 = slsqp(bh, dx0, a, 500)
    
    # Phase 4: Pair search at n0
    t_end = min(t0 + 550, t0 + budget_s - 200)
    bh, bc5 = pair(bh, dx0, rng, t_end)
    
    # Phase 5: Go to 2n
    if rem() > 250:
        n1 = 2 * n0
        dx1 = 2.0 / n1
        h1 = fix(interp(bh, n1), n1 / 2.0)
        c1 = c5_val(h1, dx1)
        t_end = min(t0 + 750, t0 + budget_s - 200)
        h1, c1 = gd(h1, dx1, t_end, [500, 2000, 8000, 30000], 0.5)
        if rem() > 120:
            t_end2 = min(t0 + 850, t0 + budget_s - 120)

    # Final

    return (bh, bc5, bn)
```
