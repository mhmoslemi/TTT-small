            w = 1.0 + 0.3 * (c[k] - c[top_lags[0]])
            if L == 0:
                importance += w * np.abs(h - 0.5)
            elif L > 0:
                importance[L:] += w * (1.0 - h[:n - L]) * h[L:]
                importance[:n - L] += w * h[:n - L] * (1.0 - h[L:])
            else:
                M = -L
                importance[:n - M] += w * (1.0 - h[M:]) * h[:n - M]
                importance[M:] += w * h[M:] * (1.0 - h[:n - M])
        idx_order = np.argsort(importance)[::-1]
        for ii in range(min(50, n)):
            i = idx_order[ii]
            for jj in range(ii + 1, min(ii + 12, n)):
                j = idx_order[jj]
                best_d, best_c = 0.0, c5c
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        cv = _c5(h, dx)
                        if cv < best_c:
                            best_c = cv
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h.copy()
                    break
        if not improved and sweep > 30:
            break

    # Phase 4: Random pair moves
    h = bh.copy()
    c5c = _c5(h, dx)
    while time.time() < t_end - 3:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.015, 0.015)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c - 1e-15:
                c5c = c
                if c5c < bc5:
                    bc5 = c5c
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)

    # Main optimization on parent size
    t_alloc1 = t0 + budget_s * 0.40
    best_h, best_c5 = _optimize(h0, n, dx, t_alloc1, rng)

    # Try n=100 (faster per iteration)
    if time.time() < t0 + budget_s * 0.65:
        n2 = 100
        dx2 = 2.0 / n2
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        t_alloc2 = min(t0 + budget_s * 0.78, time.time() + 150)
        h2r, c5_2 = _optimize(h2, n2, dx2, t_alloc2, rng)
        if c5_2 < best_c5:
            best_h, best_c5, n, dx = h2r, c5_2, n2, dx2

    # Try n=200
    if time.time() < t0 + budget_s * 0.75:
        n3 = 200
        dx3 = 2.0 / n3
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = _repair(h3, n3)
        t_alloc3 = min(t0 + budget_s * 0.92, time.time() + 120)
        h3r, c5_3 = _optimize(h3, n3, dx3, t_alloc3, rng)
        if c5_3 < best_c5:
            best_h, best_c5, n, dx = h3r, c5_3, n3, dx3

    # Final micro-polish on best
    h = best_h.copy()
    c5c = _c5(h, dx)
    t_final_end = t0 + budget_s - 1.5
    tiny = [0.0003, 0.001, 0.003, -0.0003, -0.001, -0.003,
            0.006, -0.006, 0.012, -0.012]
    while time.time() < t_final_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                best_c5 = c
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
