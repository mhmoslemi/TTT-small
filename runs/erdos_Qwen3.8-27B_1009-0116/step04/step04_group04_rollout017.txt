import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_soft(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    # w has length 2n-1, indexed by lag: lag = i - (n-1)
    v = 1.0 - h
    # First term: convolve(w, v) at positions (n-1) to (n-1)+n-1
    cv1 = np.convolve(w, v, 'full')
    # Second term: convolve(w_rev, h)
    w_rev = w[::-1]
    cv2 = np.convolve(w_rev, h, 'full')
    return cv1[(n-1):(2*n-1)] - cv2[(n-1):(2*n-1)]

def _pgd_step(h, grad, lr):
    return _project(h - lr * grad, len(h) / 2.0)

def _optimize_pgd(h, t_end, alpha_start=50, alpha_end=50000, n_iter=5000):
    n = len(h)
    h = h.copy()
    best_c = _c5(h, 2.0/n)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        # Anneal alpha
        t = it / max(1, n_iter - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        grad = _grad_soft(h, alpha)
        # Adaptive learning rate
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr = min(0.5, 0.01 * (1 + 0.1 * alpha_start) / (1.0 + alpha * gnorm))
        h_new = _pgd_step(h, grad, lr)
        c_new = _c5(h_new, 2.0/n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

    lo, hi = h.min() - 2.0, h.max() + 2.0

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level, no closures)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _soft_obj(h, alpha, tgt, pen):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - tgt) ** 2

def _lbfgs(h, dx, alpha, n_it, tgt, pens=(100, 2000, 20000)):
    bds = [(0.0, 1.0)] * len(h)
    best, bc = h.copy(), _c5(h, dx)
    for p in pens:
        try:
            r = minimize(lambda x: _soft_obj(x, alpha, tgt, p), best,
                         method='L-BFGS-B', bounds=bds,
                         options={'maxiter': n_it, 'ftol': 1e-14})
            hc = _project(r.x, tgt)
            cv = _c5(hc, dx)
            if cv < bc:
                best, bc = hc, cv
        except:
            pass
    return best, bc

def _slsqp(h, dx, alpha, n_it, tgt):
    bds = [(0.0, 1.0)] * len(h)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, tgt, 0), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': n_it, 'ftol': 1e-14})
        hc = _project(r.x, tgt)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _interp(h, n_new):
    old = np.linspace(0, 1, len(h), endpoint=False)
    new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(new, old, h)

def _downsample(h, n_new):
    return _interp(h, n_new)

def _anneal_local(h, dx, rng, t_end, n_total=200000):
    """Local search with adaptive scale and occasional larger jumps."""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scale = 0.03
    stall = 0
    for it in range(n_total):
        if time.time() > t_end:
            break
        i, j = rng.randint(n, size=2)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        h[i] += d; h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c = _c5(h, dx)
            if c < best:
                best = c
                stall = 0
            else:
                h[i] -= d; h[j] += d
                stall += 1
            if stall > 500:
                scale *= 0.7
                stall = 0
        else:
            h[i] -= d; h[j] += d
    return _project(h, tgt), best

def _focus_lag(h, dx, rng, t_end, rounds=150):
