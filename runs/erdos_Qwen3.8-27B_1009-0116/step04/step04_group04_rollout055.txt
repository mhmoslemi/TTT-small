def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:

            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break

    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / a + p * (h.sum() - tgt) ** 2

def _lbfgs(h, dx, a, mi, te):
    n = len(h); tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best, bc = h, _c5(h, dx)
    for p in [100, 2000, 20000]:
        if time.time() > te:
            break
        try:
            r = minimize(_soft, best, args=(a, tgt, p), method='L-BFGS-B',
                         bounds=bds, options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-10})
            hc = _repair(r.x, tgt); c = _c5(hc, dx)
            if c < bc:
                best, bc = hc, c
        except Exception:
            pass
    return best, bc

def _slsqp(h, dx, a, mi, te):
    n = len(h); tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    fn = lambda x: _soft(x, a, tgt, 0)
    con = lambda x: x.sum() - tgt
    try:
        r = minimize(fn, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': con}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _anneal(h, dx, rng, te, nit, t0=0.04, cool=0.99998):
    n = len(h); tgt = n / 2.0
    bh, bc = h.copy(), _c5(h, dx)
    h = h.copy(); cc = bc; temp = t0
    for _ in range(nit):
        if time.time() > te:
            break
        temp *= cool
        c = _corr(h); ks = c.argmax(); sh = ks - (n - 1)
        if sh != 0 and rng.random() < 0.7:
            ai = abs(sh)
            ii = np.arange(n - ai)
            i = int(ii[rng.randint(len(ii))]); j = i + ai
        else:
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
        d = 0.008 * (2.0 * rng.random() - 1.0)
        h[i] += d; h[j] -= d
        if -1e-10 <= h[i] <= 1.0 + 1e-10 and -1e-10 <= h[j] <= 1.0 + 1e-10:
            nc = _c5(h, dx); dl = nc - cc
            if dl < 0 or rng.random() < np.exp(-dl / max(temp, 1e-12)):
                cc = nc
                if nc < bc:
                    bc, bh = nc, _repair(h.copy(), tgt)
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return bh, bc

def _fine_local(h, dx, rng, te, nit):
    n = len(h); tgt = n / 2.0
    bh, bc = h.copy(), _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001, 0.00005]
    per = max(500, nit // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > te:
                return bh, bc
            c = _corr(h); ks = c.argmax(); sh = ks - (n - 1)
            if sh == 0:
                i = rng.randint(n); j = (i + 1 + rng.randint(n - 1)) % n
                d = sc * (1.0 if h[i] < 0.5 else -1.0)
            else:
                ai = abs(sh)
                ii = np.arange(n - ai)
                i = int(ii[rng.randint(len(ii))]); j = i + ai
                d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                nc = _c5(h, dx)
                if nc < bc:
                    bc = nc; bh = _repair(h.copy(), tgt); h = _repair(h.copy(), tgt)
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return bh, bc

def _coord_descent(h, dx, te, max_pass=8):
    n = len(h); tgt = n / 2.0
    bh, bc = h.copy(), _c5(h, dx)
    h = h.copy()
    for _ in range(max_pass):
        if time.time() > te:
            break
        for i in range(n):
            if time.time() > te:
                return bh, bc
            orig = h[i]
            best_v, best_c = orig, bc
            for trial in (0.0, 0.2, 0.5, 0.8, 1.0):
                h[i] = trial
                c_val = _c5(h, dx)
                if c_val < best_c - 1e-14:
                    best_c = c_val; best_v = trial
            h[i] = best_v
            if best_v != orig:
                h = _repair(h, tgt)
                c_now = _c5(h, dx)
                if c_now < bc:
                    bc = c_now; bh = h.copy()
        if time.time() > te:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0); dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0, _c5(h0, dx0), n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Quick SLSQP at n0 (50s)
    te = t0 + 50
    for a in [1000, 5000]:
        if time.time() > te or rem() < budget_s - 400:
            break
        h0, c = _slsqp(h0, dx0, a, 200, te)
        if c < best_c5:
            best_h, best_c5 = h0, c

    # Phase 2: Annealing at n0 (130s)
    te = t0 + 180
    best_h, best_c5 = _anneal(best_h, dx0, rng, te, 500000, 0.04, 0.99998)

    # Phase 3: L-BFGS refinement at n0 (60s)
    te = t0 + 240
    best_h, best_c5 = _lbfgs(best_h, dx0, 2000, 300, te)
    best_h, best_c5 = _slsqp(best_h, dx0, 5000, 200, te)

    # Phase 4: Multi-restart (80s)
    te = t0 + 320
    for r in range(5):
        if time.time() > te or rem() < budget_s - 350:
            break
        amp = 0.12 * (1.0 - r / 6.0)
        hp = best_h + amp * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, n0 / 2.0)
        hp, c = _lbfgs(hp, dx0, 1500, 200, te)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 5: Fine local + coordinate descent at n0 (80s)
    te = t0 + 400
    best_h, best_c5 = _fine_local(best_h, dx0, rng, te, 150000)
    te2 = min(te + 30, t0 + budget_s - 400)
    best_h, best_c5 = _coord_descent(best_h, dx0, te2, 5)

    # Phase 6: Interpolate to 2*n0 and optimize
    n1 = 2 * n0; dx1 = 2.0 / n1
    h1 = _repair(_interp(best_h, n1), n1 / 2.0)
    c1 = _c5(h1, dx1)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n1
    else:
        h1 = best_h

    # Phase 7: L-BFGS at n1 (180s)
    te = min(t0 + 620, t0 + budget_s - 280)
    for a in [500, 3000, 15000]:
        if time.time() > te or rem() < 280:
            break
        h1, c1 = _lbfgs(h1, dx1, a, 400, te)
        if c1 < best_c5:
            best_h, best_c5, best_n = h1, c1, n1

    # Phase 8: SLSQP at n1 (100s)
    te = min(t0 + 720, t0 + budget_s - 180)
    for a in [5000, 20000]:
        if time.time() > te or rem() < 180:
            break
        best_h, best_c5 = _slsqp(best_h, 2.0 / best_n, a, 300, te)

    # Phase 9: Fine local at n1 (200s)
    te = t0 + budget_s - 80
    best_h, best_c5 = _fine_local(best_h, 2.0 / best_n, rng, te, 400000)

    # Phase 10: Coordinate descent
    te2 = min(te + 40, t0 + budget_s - 30)
    best_h, best_c5 = _coord_descent(best_h, 2.0 / best_n, te2, 4)

    # Phase 11: Final tiny anneal
    te3 = t0 + budget_s - 5
    best_h, best_c5 = _anneal(best_h, 2.0 / best_n, rng, te3, 50000, 0.008, 0.999995)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
