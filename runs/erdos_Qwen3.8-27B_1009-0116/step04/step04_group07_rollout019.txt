    try:

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8, 'maxls': 30})
        return repair(res.x, n)
    except Exception:
        return h

def _coord_grad(corr, c, alpha):
    v = corr + 0 * c  # base
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    return float(np.dot(w, c))

def coord_sweep(h, n, alpha, t_end):
    dx = 2.0 / n
    n_lag = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    for j in range(n):
        if time.time() > t_end:
            return h
        t0 = h[j]
        c = np.zeros(n_lag)
        for k in range(n_lag):
            sh = k - (n - 1)
            val = 0.0
            if 0 <= j + sh < n:
                val += (1.0 - h[j + sh])
            if 0 <= j - sh < n:
                val -= h[j - sh]
            c[k] = val
        # Bisection on derivative
        def fgrad(t):
            v = corr + (t - t0) * c
            m = np.max(v)
            s = alpha * (v - m)
            s = np.clip(s, -500, 500)
            wt = np.exp(s) / np.sum(np.exp(s))
            return float(np.dot(wt, c))
        g0 = fgrad(0.0); g1 = fgrad(1.0)
        if g0 >= 0:
            t_new = 0.0
        elif g1 <= 0:
            t_new = 1.0
        else:
            lo, hi = 0.0, 1.0
            for _ in range(35):
                mid = (lo + hi) * 0.5
                if fgrad(mid) < 0:
                    lo = mid
                else:
                    hi = mid
            t_new = (lo + hi) * 0.5
        if abs(t_new - t0) > 1e-12:
            corr = corr + (t_new - t0) * c
            h[j] = t_new
    return h

def fourier_init(n, m, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    coeffs = rng.randn(2 * m) * 0.1
    h = np.full(n, 0.5)
    for k in range(1, m + 1):
        h += coeffs[2*(k-1)] * np.cos(k * np.pi * x) + coeffs[2*(k-1)+1] * np.sin(k * np.pi * x)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed + 555)
    best_h = None; best_c5 = np.inf; best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        if len(init_h) > 0:
            best_n = len(init_h)
            best_h = repair(init_h, best_n)
            best_c5 = compute_c5(best_h, best_n)
    except NameError:
        pass
    if best_h is None:
        best_h = np.full(64, 0.5)
        best_h[:32] = 0.7; best_h[32:] = 0.3
        best_h = repair(best_h, 64)
        best_c5 = compute_c5(best_h, 64); best_n = 64

    def elapsed():
        return time.time() - t0

    # Phase 1: Gradient L-BFGS-B
    for alpha in [200, 1000, 5000, 20000, 80000]:
        if elapsed() > budget_s * 0.25:
            break
        for pen in [50, 500, 3000]:
            if elapsed() > budget_s * 0.25:
                break
            h_new = lbfgs_run(best_h.copy(), best_n, alpha, pen, t0 + budget_s * 0.25, 2000)
            c5_new = compute_c5(h_new, best_n)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new; best_h = h_new.copy()

    # Phase 2: Coordinate descent with 1D bisection
    n_cd = 800
    for i in range(n_cd):
        if time.time() > t_end - budget_s * 0.30:
            break
        best_h = coord_sweep(best_h.copy(), best_n, 50000, t_end - budget_s * 0.30)
        if i % 10 == 0:
            best_h = repair(best_h, best_n)
            c5_new = compute_c5(best_h, best_n)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new

    # Phase 3: Try different n values
    if elapsed() < budget_s * 0.65:
        for n_alt in [48, 52, 56, 60, 64, 72, 80, 96, 112, 128, 160]:
            if elapsed() > budget_s * 0.70:
                break
            x_new = np.linspace(0, 2, n_alt, endpoint=False)
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_alt = lbfgs_run(h_seed, n_alt, 5000, 500, t0 + budget_s * 0.68, 1500)
            for _ in range(80):
                if time.time() > t0 + budget_s * 0.68:
                    break
                h_alt = coord_sweep(h_alt, n_alt, 30000, t0 + budget_s * 0.68)
            h_alt = repair(h_alt, n_alt)
            c5_alt = compute_c5(h_alt, n_alt)
            if c5_alt < best_c5 - 1e-14:
                best_c5 = c5_alt; best_h = h_alt.copy(); best_n = n_alt

    # Phase 4: Fourier multi-start + local refine
    if elapsed() < budget_s * 0.85:
        for m_four in [4, 6, 8]:
            if elapsed() > budget_s * 0.84:
                break
            for trial in range(3):
                if elapsed() > budget_s * 0.84:
                    break
                h_f = fourier_init(best_n, m_four, rng)
                h_f = lbfgs_run(h_f, best_n, 5000, 500, t0 + budget_s * 0.82, 1000)
                for _ in range(40):
                    if time.time() > t0 + budget_s * 0.82:
                        break
                    h_f = coord_sweep(h_f, best_n, 30000, t0 + budget_s * 0.82)
                h_f = repair(h_f, best_n)
                c5_f = compute_c5(h_f, best_n)
                if c5_f < best_c5 - 1e-14:
                    best_c5 = c5_f; best_h = h_f.copy()

    # Phase 5: Perturbation restarts
    if elapsed() < budget_s * 0.93:
        for trial in range(4):
            if elapsed() > budget_s * 0.92:
                break
            sigma = 0.015 * (trial + 1)
            h_p = repair(best_h + sigma * rng.randn(best_n), best_n)
            h_p = lbfgs_run(h_p, best_n, 10000, 1000, t0 + budget_s * 0.90, 1500)
            c5_p = compute_c5(h_p, best_n)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p; best_h = h_p.copy()

    # Phase 6: Final ultra-fine coordinate polish
    if elapsed() < t_end - 10:
        for _ in range(300):
            if time.time() > t_end - 10:
                break
            best_h = coord_sweep(best_h, best_n, 200000, t_end - 10)
        best_h = repair(best_h, best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
