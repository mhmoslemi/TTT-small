    target = n / 2.0

    x_avg = h.copy()
    best_local = h.copy()
    best_c5_local = _eval(h)

import numpy as np
from scipy.optimize import minimize
import time

def _project(h, target_sum):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -h.max() - 0.5, 1.0 - h.min() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = _project(h, target)
    elif (h < 0).any() or (h > 1).any():
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) > 1e-10:
            h = _project(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - x[lag:])
            g[lag:] -= wk * x[:n - lag]
        else:
            m2 = -lag
            g[m2:] += wk * (1.0 - x[:n - m2])
            g[:n - m2] -= wk * x[m2:]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _opt_slsqp(h, n, alpha, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _proj_subgrad(h, n, t_end, max_iter=200000):

    x = h.copy()
    x_prev = h.copy()
    x_avg = h.copy()
    best_x = h.copy()
    best_c5 = _eval(h)
    npassive = 0
    for t in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(x, 1.0 - x, mode='full')
        kmax = np.argmax(c)
        lag = kmax - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[:n - lag] += (1.0 - x[lag:])
            g[lag:] -= x[:n - lag]
        else:
            m2 = -lag
            g[m2:] += (1.0 - x[:n - m2])
            g[:n - m2] -= x[m2:]
        alpha = 0.01 / (1.0 + 0.001 * t)
        x_new = x_prev - alpha * g
        x_new = _project(x_new, target)
        x_avg = x_avg + (x_new - x_avg) / (t + 1.0)
        if t % 200 == 0:
            cv = _eval(x_avg)
            if cv < best_c5:

                best_x = x_avg.copy()
                npassive = 0
            else:
                npassive += 1
                if npassive > 200:
                    break
        x_prev = x.copy()
        x = x_new
    return best_x, best_c5

def _block_local_search(h, n, rng, t_end, max_iter=50000):

    target = n / 2.0

        if i0 == j0:
            continue
        shape = rng.randn(block_len)
        shape -= shape.mean()
        shape *= scale / (np.std(shape) + 1e-12)
        h[i0:i0 + block_len] += shape
        h[j0:j0 + block_len] -= shape
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5c - 1e-14:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.08, 0.05)
        else:
            h[i0:i0 + block_len] -= shape
            h[j0:j0 + block_len] += shape
            h = _repair(h, n)
            no_imp += 1
            if no_imp > 200:
                scale *= 0.8
                no_imp = 0
                if scale < 2e-6:
                    break
    return h, c5c

def _fine_local_search(h, n, rng, t_end):
    c5c = _eval(h)
    target = n / 2.0
    steps = [1, 2, 3, 5, 7, 11]
    for _ in range(200000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice(steps) * rng.choice([1, -1]) * 1e-5
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _balance_top_k(h, n, K=5, t_end=None):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[-K:]
    g = np.zeros(n)
    for k in top_k:
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            m2 = -lag
            g[m2:] += (1.0 - h[:n - m2])
            g[:n - m2] -= h[m2:]
    g *= -1.0 / np.linalg.norm(g)
    target = n / 2.0
    best_c5 = _eval(h)
    best_h = h.copy()
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        h_new = _project(h + step * g, target)
        cv = _eval(h_new)
        if cv < best_c5 - 1e-15:
            best_c5 = cv
            best_h = h_new.copy()
    return best_h, best_c5
