    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_con(x, target):
    return np.sum(x) - target

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9, 'maxls': 50})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj_eq, h, args=(alpha, target),
                      method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def simulated_anneal(h, n, dx, rng, t_deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    curr_c5 = best_c5
    it = 0
    stuck = 0
    T_cur = 0.004
    T_min = 0.000005
    reheat_counter = 0

    while time.time() < t_deadline:
        it += 1
        T_cur = max(T_min, T_cur * 0.9997)
        r = rng.random()
        if r < 0.72:
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            d = np.clip(0.03 * T_cur * 100 * (2.0 * rng.random() - 1.0), -0.08, 0.08)
            d = np.clip(d, -0.1 * (0.01 + T_cur), 0.1 * (0.01 + T_cur))
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i], h[j] = ni, nj
        elif r < 0.90:
            bs = rng.randint(2, 6)
            i = rng.randint(0, n - bs)
            j = rng.randint(0, max(1, n - bs))
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(-0.02, 0.02)
            if np.any(h[i:i+bs] + eps < -1e-9) or np.any(h[i:i+bs] + eps > 1.0 + 1e-9) or \
               np.any(h[j:j+bs] - eps < -1e-9) or np.any(h[j:j+bs] - eps > 1.0 + 1e-9):
                continue
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
        else:
            i, j, k = rng.choice(n, 3, replace=False)
            d1 = 0.012 * (2.0 * rng.random() - 1.0)
            d2 = 0.012 * (2.0 * rng.random() - 1.0)
            if not (0 <= h[i]+d1 <= 1 and 0 <= h[j]+d2 <= 1 and 0 <= h[k]-d1-d2 <= 1):
                continue
            h[i], h[j], h[k] = h[i]+d1, h[j]+d2, h[k]-d1-d2

        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        delta = c5n - curr_c5
        accept = False
        if delta < -1e-15:
            accept = True
        elif T_cur > T_min:
            ratio = -delta / T_cur
            if ratio < 500 and np.exp(-ratio) > rng.random():
                accept = True

        if accept:
            curr_c5 = c5n
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            if r < 0.72:
                h[i] -= d; h[j] += d
            elif r < 0.90:
                h[i:i+bs] -= eps; h[j:j+bs] += eps
            else:
                h[i] -= d1; h[j] -= d2; h[k] += d1 + d2
            stuck += 1

        if stuck > 4000 and T_cur <= T_min * 10:
            T_cur = min(0.003, T_cur * 8 + 0.0005)
            stuck = 0
            reheat_counter += 1
            h = repair(h + rng.normal(0, 0.015, n), n)
            curr_c5 = compute_c5(h, dx)
            if reheat_counter > 6:
                break
    return best_h, best_c5

def fine_local(h, n, dx, rng, deadline, n_iters, base_scale):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    imp_count = 0
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        scale = base_scale * (0.5 if imp_count > 30 else (2.0 if imp_count == 0 and it > 50 else 1.0))
        imp_count = max(0, imp_count - 1)
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i], h[j] = ni, nj
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            imp_count += 1
        else:
            h[i] -= d; h[j] += d
    return h, best_c5

def perturb_restart(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n // 7, 6 * n // 7))
    elif kind == 1:
        f = rng.randint(1, 7)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.04, 0.15)
        hp += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.05, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 6)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        t = np.arange(n) / n
        hp += rng.uniform(0.03, 0.12) * np.sin(2*np.pi*f1*t+p1) * np.sin(2*np.pi*f2*t+p2)
    else:
        hp *= rng.uniform(0.8, 1.2)
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t_start)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = repair(h0.copy(), n0)
    c5_0 = compute_c5(h0, dx0)
    best_h, best_c5, best_n = h0.copy(), c5_0, n0

    # Phase 1: L-BFGS at n=140 (50s)
    t_end = t_start + 50
    cur_h, cur_c5 = h0.copy(), c5_0
    for alpha in [100, 500, 2000, 8000]:
        if time.time() >= t_end:
            break
        for pen in [500, 5000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_opt(cur_h, n0, alpha, n0/2.0, pen, 250)
            if cc < cur_c5:
                cur_h, cur_c5 = hh, cc
    if cur_c5 < best_c5:
        best_h, best_c5, best_n = cur_h.copy(), cur_c5, n0

    # Phase 2: Simulated annealing (250s) - main new component
    if rem() > 300:
        t_end_sa = time.time() + min(250, rem() - 200)
        cur_h, cur_c5 = simulated_anneal(cur_h, n0, dx0, rng, t_end_sa)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n0

    # Phase 3: Fine local search (150s)
    if rem() > 250:
        t_end_ls = time.time() + min(150, rem() - 120)
        cur_h, cur_c5 = fine_local(cur_h, n0, dx0, rng, t_end_ls, 200000, 0.003)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n0

    # Phase 4: Multi-restart with annealing (120s)
    if rem() > 250:
        t_end_mr = time.time() + min(120, rem() - 150)
        for r in range(8):
            if time.time() >= t_end_mr:
                break
            hp = perturb_restart(cur_h, n0, rng, r % 5)
            hh, cc = lbfgs_opt(hp, n0, 3000, n0/2.0, 3000, 150)
            if cc < cur_c5:
                hh, cc = simulated_anneal(hh, n0, dx0, rng, min(time.time()+20, t_end_mr))
                hh, cc = fine_local(hh, n0, dx0, rng, time.time()+10, 30000, 0.002)
                if cc < cur_c5:
                    cur_h, cur_c5 = hh, cc
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n0

    # Phase 5: SLSQP polish (80s)
    if rem() > 130:
        t_end_sl = time.time() + min(80, rem() - 60)
        for alpha in [2000, 10000, 40000]:
            if time.time() >= t_end_sl:
                break
            hh, cc = slsqp_opt(cur_h, n0, alpha, n0/2.0, 600)
            if cc < cur_c5:
                cur_h, cur_c5 = hh, cc
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n0

    # Phase 6: Ultra-fine local search (remaining)
    if rem() > 30:
        t_end_fin = time.time() + max(5, rem() - 15)
        cur_h, cur_c5 = fine_local(cur_h, n0, dx0, rng, t_end_fin, 500000, 0.001)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur_h.copy(), cur_c5, n0

    # Final: one more SLSQP if time
    if rem() > 20:
        hh, cc = slsqp_opt(best_h, best_n, 50000, best_n/2.0, 400)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
