from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _soft_obj(h, alpha, target, pen):
    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2

def _project_sum(v, target):
    """Project v onto {h: 0<=h<=1, sum(h)=target} via bisection on shift."""
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        h = np.clip(v - lam, 0.0, 1.0)
        if h.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_max(h, n_top):
    """Subgradient of max c[k] using top n_top shifts."""
    n = len(h)
    c = _corr_full(h)
    top = np.argsort(c)[-n_top:]
    g = np.zeros(n)
    for idx in top:
        k = idx - (n - 1)
        if k == 0:
            g += 1.0 - 2.0 * h
        elif 0 < k < n:
            a = np.zeros(n); a[:n-k] = 1.0 - h[k:]
            b = np.zeros(n); b[k:] = -h[:n-k]
            g += a + b
        elif -n < k < 0:
            kk = -k
            a = np.zeros(n); a[kk:] = 1.0 - h[:n-kk]
            b = np.zeros(n); b[:n-kk] = -h[kk:]
            g += a + b
    return g / n_top

def _subgrad(h, tgt, t_end, iters):

    h = h.copy()
    eta = 0.015

        if time.time() > t_end:
            break
        g = _grad_max(h, 5)
        h_new = _project_sum(h - eta * g, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            h, best = h_new, c_new
            if it % 1000 == 0 and it > 0:
                eta *= 0.97
        else:
            eta *= 0.995
    return h, best

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, tgt, 0), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
