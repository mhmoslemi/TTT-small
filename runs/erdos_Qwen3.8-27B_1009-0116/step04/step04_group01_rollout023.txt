import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _proj_box_sum(h, target):
    """Project onto {0≤x≤1, sum(x)=target} using bisection on water level."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad_max(h, top_k=10):
    """Compute subgradient of max_k f_k(h) using top-k active constraints."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    # Get top-k lags
    idx = np.argsort(c)[-top_k:]
    # Weight by how close they are to max (soft max weighting)
    w = np.exp((c[idx] - c.max()) / 1.0)  # temperature 1 in correlation units
    w /= w.sum()
    
    g = np.zeros(n)
    for j, k in enumerate(idx):
        if w[j] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[j] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g, c.max() * dx

def _nesterov_subgrad(h, budget, top_k=10, lr0=0.01, momentum=0.9):
    """Nesterov-accelerated projected subgradient on smooth approximation of max."""
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    best_x = h.copy()
    best_f = _eval(h)
    
    # Adaptive learning rate
    lr = lr0
    for t in range(2000):
        # Nesterov step
        x_n = x + momentum * v
        g, f_val = _subgrad_max(x_n, top_k)
        v = momentum * v - lr * g
        x_new = _proj_box_sum(x_n + v, n * 0.5)
        
        f_new = _eval(x_new)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        
        # Adaptive lr
        if f_new > f_val:
            lr *= 0.95
        else:
            lr = min(lr * 1.01, lr0)
        
        x = x_new
    
    return best_x, best_f

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_box_sum(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _lsq_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lsq_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400.0, 400.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _red_obj(x):
    n = _sv_n
    h = np.empty(n)
    h[:-1] = x
    h[-1] = n * 0.5 - x.sum()
    if h[-1] < -0.1 or h[-1] > 1.1:
        return 1e6 + (h[-1] - 0.5) ** 2
    return _lsq_obj(h, _sv_alpha)

def _red_grad(x):
    n = _sv_n
    h = np.empty(n)
    h[:-1] = x
    h[-1] = n * 0.5 - x.sum()
    g = _lsq_grad(h, _sv_alpha)
    return g[:-1] - g[-1]

def _lbfgs(h, alpha, maxiter, gtol):
    global _sv_n, _sv_alpha
    n = len(h)
    _sv_n = n
    _sv_alpha = alpha
    bnds = [(0.0, 1.0)] * (n - 1)
    try:
        res = minimize(_red_obj, h[:-1].copy(), jac=_red_grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-18})
        h_new = np.empty(n)
        h_new[:-1] = res.x
        h_new[-1] = n * 0.5 - res.x.sum()
        return _repair(h_new, n)
    except Exception:
        return h

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    global _sv_n
    _sv_n = n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jc}]
    try:
        res = minimize(_lsq_obj, h, _lsq_fn_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _lsq_fn_jac(x, alpha):
    return _lsq_obj(x, alpha), _lsq_grad(x, alpha)

def _sum_fn(x):
    return x.sum() - _sv_n * 0.5

def _sum_jc(x):
    return np.ones(len(x))

def _mom_grad(h, alpha, steps, lr0, momentum):
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    lr = lr0
    best_x = h.copy()
    best_f = _eval(h)
    for t in range(steps):
        g = _lsq_grad(x, alpha)
        v = momentum * v - lr * g
        x_new = _proj_box_sum(x + v, n * 0.5)
        f_new = _eval(x_new)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        if f_new > _eval(x) + 1e-4:
            lr *= 0.85
            v *= 0.5
        else:
            lr = min(lr * 1.02, lr0)
        x = x_new
    return best_x, best_f

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _sa_pairwise(h, nc, n_iter, t_start, t_end, t0):
    rng = np.random.RandomState(t0 + 7)
    best_c5 = _eval(h)
    best_h = h.copy()
    c5 = best_c5
    cur_t = t_start
    for it in range(n_iter):
        if time.time() - _t0 > t_end:
            break
        cur_t *= 0.9995
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = 0.003 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        delta = c5n - c5
        if delta < 0:
            c5 = c5n
        else:
            if rng.random() > np.exp(-delta / max(cur_t, 1e-8)):
                h[i], h[j] = ni - d, nj + d
                continue
            c5 = c5n
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _sv_n
    _t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP polish at n0
    for alpha in [5000, 50000, 200000]:
        if time.time() - _t0 > 40:
            break
        _sv_n = n0
        h = _slsqp(best_h, alpha, 1500, 1e-18)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n0

    # Phase 2: Upscale to 2x
    if time.time() - _t0 < budget_s - 500:
        n1 = n0 * 2
        _sv_n = n1
        h1 = _upsample(best_h, 2)
        h1 = _repair(h1, n1)
        alpha = 2000
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h1.copy()
            best_n = n1

        # L-BFGS-B with annealed alpha at n1
        for alpha in [2000, 10000, 50000, 200000, 500000]:
            if time.time() - _t0 > budget_s - 350:
                break
            h1 = _lbfgs(h1, alpha, 300, 1e-13)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h1.copy()
                best_n = n1

        # Momentum gradient at n1
        if time.time() - _t0 < budget_s - 250:
            h1, f1 = _mom_grad(h1, 100000, 2000, 0.005, 0.7)
            h1 = _repair(h1, n1)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h1.copy()
                best_n = n1

    # Phase 3: Multi-start at best_n
    nc = best_n
    _sv_n = nc
    if time.time() - _t0 < budget_s - 250:
        n_restarts = 12
        t_left = budget_s - 250 - (time.time() - _t0)
        t_per = t_left / max(n_restarts, 1)
        for r in range(n_restarts):
            if time.time() - _t0 > budget_s - 250:
                break
            strat = r % 5
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h, nc)
            elif strat == 2:
                hp = best_h.copy()
                nblocks = rng.randint(3, 7)
                for _ in range(nblocks):
                    bs = rng.randint(5, max(6, nc // 15))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.05 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = best_h.copy()
                freqs = rng.randint(1, nc // 3 + 1, size=4)
                t = np.arange(nc)
                for f in freqs:
                    hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * t / nc)
                    hp += 0.008 * rng.randn() * np.cos(2 * np.pi * f * t / nc)
                hp = _repair(hp, nc)
            else:
                hp = _repair(np.clip(best_h + 0.5 * (rng.rand(nc) - 0.5) * (1.0 - np.abs(2 * best_h - 1)), 0, 1), nc)
            hp = _lbfgs(hp, 100000, 200, 1e-12)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()
                best_n = nc

    # Phase 4: SLSQP final polish
    if time.time() - _t0 < budget_s - 100:
        _sv_n = best_n
        h = _slsqp(best_h, 500000, 2000, 1e-19)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Simulated annealing pairwise
    if time.time() - _t0 < budget_s - 40:
        best_h, best_c5 = _sa_pairwise(best_h.copy(), best_n, 12000, 0.002, 0.0001, best_n)
        best_h = _repair(best_h, best_n)
        best_c5 = _eval(best_h)

    # Phase 6: Ultra-fine deterministic pairwise
    if time.time() - _t0 < budget_s - 5:
        nc = best_n
        h = best_h.copy()
        c5c = _eval(h)
        for scale in [2e-4, 5e-5, 1e-5]:
            if time.time() - _t0 > budget_s - 5:
                break
            for _it in range(20000):
                if time.time() - _t0 > budget_s - 5:
                    break
                i = rng.randint(nc)
                j = (i + rng.randint(1, nc)) % nc
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
