```python
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _corr_vals(h, n):

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Proportional scaling then clip, then fix
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]  # most room first
        else:
            order = np.argsort(h)  # smallest first
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

    w has length 2n-1, indexed 0..2n-2 corresponding to lags -(n-1)..(n-1)."""
```

```python
def _pair_move_delta(h, n, i, j, d):
    """Compute the change in all correlation values when h[i]+=d, h[j]-=d."""
    # For each lag k-(n-1), the change is:
    # d * (1-h[i+lag]) * valid_i(lag) - d * (1-h[j+lag]) * valid_j(lag)
    lags = np.arange(-(n-1), n)  # lag values
    # valid_i(lag): 0 <= i+lag < n, i.e., -i <= lag < n-i
    # valid_j(lag): 0 <= j+lag < n, i.e., -j <= lag < n-j
    valid_i = (lags >= -i) & (lags < n - i)
    valid_j = (lags >= -j) & (lags < n - j)
    
    # h at shifted positions (with padding for invalid)
    # For i+lag valid: h[i+lag]
    idx_i = np.where(valid_i, i + lags, 0)
    idx_j = np.where(valid_j, j + lags, 0)
    
    contrib_i = np.where(valid_i, d * (1 - h[idx_i]), 0)
    contrib_j = np.where(valid_j, -d * (1 - h[idx_j]), 0)
    
    return contrib_i + contrib_j
```

```python
        h[j] += d
```

```python
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
```

```python
            if abs(diff) < 1e-12:
                break
            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a

def _project_sum(x, n):

    x = np.clip(x, 0.0, 1.0)

def _grad_w(h, n, w):

        if abs(wk) < 1e-14:

def _slsqp_obj(h, n, alpha):
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = float(np.max(cv))
    s = np.clip(alpha * (cv - m), -400, 400)
    return m + np.log1p(np.sum(np.exp(s)) - 1.0) / alpha

def _slsqp_obj_grad(h, n, alpha):
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _slsqp_run(h, n, alpha, maxiter, budget_end):
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0
    def cfun(x):
        return x.sum() - target_sum
    def cjac(x):
        return np.ones(n)
    res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                   jac=_slsqp_obj_grad, bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   options={'maxiter': maxiter, 'ftol': 1e-19})
    if time.time() > budget_end:
        return h, _c5_val(h, n)
    hc = _repair(res.x, n)
    return hc, _c5_val(hc, n)

def _fast_pair_search(h, n, budget_end, rng):

def _sweep_search(h, n, budget_end):

    h = h.copy()
    corr = _corr_arr(h, n)
    dx = 2.0 / n
    c5_cur = float(np.max(corr)) * dx
    lags = np.arange(-(n - 1), n)
    # For each i, find j that maximizes benefit (using gradient info)
    cv = corr * dx
    m = float(np.max(cv))
    alpha = 5000
    s = np.clip(alpha * (cv - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = _grad_w(h, n, w)
    # For pair (i,j) with move d: benefit ~ d*(g[i] - g[j])
    # Try sweeping i from 0 to n-1, and for each, pick best j
    for sweep in range(200):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n):
            hi = h[i]
            if hi <= 0.01 or hi >= 0.99:
                continue
            # Best j is opposite gradient direction
            if g[i] > 0:
                j_best = np.argmax(g[:i]) if i > 0 else 0
                if i < n:
                    j_best2 = np.argmax(g[i+1:])
                    if g[j_best2] > g[j_best] if i > 0 else True:
                        j_best = i + 1 + j_best2
            else:
                j_best = np.argmin(g[:i]) if i > 0 else 0
                if i < n:
                    j_best2 = np.argmin(g[i+1:])
                    if g[j_best2] < g[j_best] if i > 0 else True:

            if j_best == i:

            for d in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                ni, nj = hi + d, h[j_best] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                vi = (lags >= -i) & (lags < n - i)
                vj = (lags >= -j_best) & (lags < n - j_best)
                delta = np.zeros(2 * n - 1)
                if vi.any():
                    delta[vi] += d * (1.0 - h[i + lags[vi]])
                if vj.any():
                    delta[vj] -= d * (1.0 - h[j_best + lags[vj]])
                corr += delta
                c5_new = float(np.max(corr)) * dx
                if c5_new < c5_cur - 1e-14:
                    h[i] = ni
                    h[j_best] = nj
                    c5_cur = c5_new
                    improved = True
                    break
                else:
                    corr -= delta
        if not improved:
            break
    return h, c5_cur

    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
```
