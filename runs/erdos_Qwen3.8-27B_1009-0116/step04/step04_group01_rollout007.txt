import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _power_obj(h, p):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_pos = np.clip(c, 0, None)
    return np.sum(c_pos ** p) ** (1.0 / p)

def _smooth_grad(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()

    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:

        sh = (n - 1) - k

        if s > target:

        else:
            hi = mid

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _c_fun(x):
        return x.sum() - n * 0.5
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, steps, step0):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.9
    step = step0

        g = _smooth_grad(h, alpha)
        h_prev = h.copy()

            f = f_new

    return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _sa_block(h, n, t0, t1, n_iter, rng):

    best_c = c_cur

        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-8)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c
