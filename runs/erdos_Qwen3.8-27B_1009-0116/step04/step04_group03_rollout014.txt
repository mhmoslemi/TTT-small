import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _sm_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha + 80.0 * (np.sum(x) - n / 2.0) ** 2

def _sm_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    g += 160.0 * (np.sum(x) - n / 2.0)
    return g

def _lbfgs_opt(h, n, alpha, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, n),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    c5c = _eval(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(200000):

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-8:]
        g = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            i1, j1 = max(0, lag), min(n, n + lag)
            if i1 < j1:
                g[i1:j1] += (1.0 - h[i1 - lag:j1 - lag])
            i2, j2 = max(0, -lag), min(n, n - lag)
            if i2 < j2:
                g[i2:j2] -= h[i2 + lag:j2 + lag]
