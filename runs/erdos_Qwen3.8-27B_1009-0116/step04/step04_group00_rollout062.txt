import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_w(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-18:
            continue
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _obj_sm(x, alpha, target, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    pen = 200.0 * (np.sum(x) - target) ** 2
    return m + np.sum(np.exp(s)) / (alpha * n) + pen

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / (ew.sum() * alpha * n)
    g = _grad_w(x, pw)
    g += 400.0 * (np.sum(x) - target)
    return g

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_dx)
    return (best_h, best_c5, best_n)
