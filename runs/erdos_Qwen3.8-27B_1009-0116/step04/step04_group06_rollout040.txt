    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _PairSearch(h, n, end_t, rng):
    h = h.copy(); bc = _c5(h, n); best = h.copy()
    deltas = np.array([0.0002, 0.0005, 0.001, 0.003, 0.01,
                       -0.0002, -0.0005, -0.001, -0.003, -0.01])
    improved = True
    while improved and time.time() < end_t:
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for i in idx:
            if time.time() > end_t:
                return best, bc
            hi = h[i]
            for j in idx:
                if j == i:
                    continue
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if not (0 <= ni <= 1 and 0 <= nj <= 1):
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < bc - 1e-15:
                        bc = c; hi = ni; hj = nj; best = h.copy(); improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
    return best, bc

def _SingleMoves(h, n, end_t, rng):
    h = h.copy(); bc = _c5(h, n); best = h.copy()
    smalls = np.array([5e-5, 1e-4, 3e-4, 1e-3, -5e-5, -1e-4, -3e-4, -1e-3])
    while time.time() < end_t:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = smalls[rng.randint(len(smalls))]
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < bc:
                bc = c; best = h.copy()
            else:
                h[i], h[j] = hi, hj
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_n, best_c5 = h0.copy(), n0, _c5(h0, n0)

    # Phase 1: Active-set PGD (fast, focused)
    end1 = t0 + budget_s * 0.22
    for n_act in [2, 3, 5]:
        if time.time() > end1:
            break
        hw, cw = _ActivePGD(best_h, best_n, end1, 0.04, 4000, n_act)
        if cw < best_c5:
            best_c5, best_h = cw, hw

    # Phase 2: Smooth PGD with momentum, multiple alphas
    end2 = t0 + budget_s * 0.38
    for alpha in [20000, 80000, 200000, 500000]:
        if time.time() > end2:
            break
        hw, cw = _MomentumPGD(best_h, best_n, alpha, end2, 0.015, 3000)
        if cw < best_c5:
            best_c5, best_h = cw, hw

    # Phase 3: SLSQP local polish (multiple alpha, restarts)
    end3 = t0 + budget_s * 0.52
    for alpha in [100000, 300000, 800000]:
        if time.time() > end3:
            break
        hc = _SLSQP(best_h, best_n, alpha, 500)
        c = _c5(hc, best_n)
        if c < best_c5:
            best_c5, best_h = c, hc
    # Small perturbation + SLSQP to escape
    if time.time() < end3:
        for _ in range(4):
            if time.time() > end3:
                break
            hp = _repair(best_h + 0.03 * rng.randn(best_n), best_n)
            hc = _SLSQP(hp, best_n, 200000, 400)
            c = _c5(hc, best_n)
            if c < best_c5:
                best_c5, best_h = c, hc

    # Phase 4: Multi-resolution exploration
    end4 = t0 + budget_s * 0.72
    for n_try in [100, 120, 160, 180, 200, 280]:
        if time.time() > end4:
            break
        if n_try == best_n:
            continue
        xo = np.linspace(0, 2.0, best_n, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hn = _repair(np.interp(xn, xo, best_h, period=2.0), n_try)
        hw, cw = _ActivePGD(hn, n_try, min(time.time() + 25, end4), 0.03, 1500, 3)
        if cw < best_c5 - 1e-7:
            best_c5, best_h, best_n = cw, hw, n_try
        elif cw < best_c5 + 0.002 and time.time() < end4 - 10:
            hc = _SLSQP(hw, n_try, 200000, 350)
            c = _c5(hc, n_try)
            if c < best_c5:
                best_c5, best_h, best_n = c, hc, n_try

    # Phase 5: Extensive pair search
    end5 = t0 + budget_s * 0.91
    best_h, best_c5 = _PairSearch(best_h, best_n, end5, rng)

    # Phase 6: Ultra-fine single moves
    end6 = t0 + budget_s - 8
    if time.time() < end6:
        best_h, best_c5 = _SingleMoves(best_h, best_n, end6, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
