def fast_local_search(h, n, dx, rng, deadline, n_scales):
    """Ultra-fast local search using incremental correlation updates."""
    h = h.copy()
    target = n / 2.0
    C = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(C) * dx)
    S = compute_S(h, n)
    rev_h = h[::-1]
    
    # Pre-allocated work array
    work = np.empty(2*n-1)
    
    for scale_idx in range(n_scales):
        scale = 0.05 / (1 + scale_idx ** 1.5)
        n_trials = 500000  # Many trials per scale
        improved = 0
        
        for _ in range(n_trials):
            if time.time() >= deadline:
                return h, c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = scale * (2.0 * rng.random() - 1.0)
            if abs(delta) < 1e-10:
                continue
            ni = h[i] + delta
            nj = h[j] - delta
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            
            # Compute new max correlation
            np.add(C, delta * (S[i] - S[j]), out=work)
            new_c5 = float(np.max(work) * dx)
            if new_c5 < c5 - 1e-14:
                # Accept
                h[i] = ni
                h[j] = nj
                C = work.copy()  # or work[:] 
                c5 = new_c5
                improved += 1
                # Update S
                S[i, n-1] -= 2*delta
                S[j, n-1] += 2*delta
                for m in range(n):
                    if m == i or m == j:
                        continue
                    # S[m, n-1+i-m] -= delta (from 1-h[m+s] where m+s=i)
                    k1 = n-1+i-m
                    if 0 <= k1 < 2*n-1:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_S(h, n):
    S = np.zeros((n, 2 * n - 1))
    rev_h = h[::-1]
    one_minus_h = 1.0 - h
    for i in range(n):
        S[i, n-1-i : 2*n-1-i] += one_minus_h
        S[i, i : i+n] -= rev_h
    return S

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, 0), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def fast_local_search(h, n, dx, rng, deadline):
    h = h.copy()
    C = np.correlate(h, 1.0 - h, mode='full').copy()
    c5 = float(np.max(C) * dx)
    S = compute_S(h, n)
    
    m_idx = np.arange(n)
    work = np.empty(2 * n - 1)
    
    scales = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for scale_idx, scale in enumerate(scales):
        if time.time() >= deadline:
            break
        improved = 0
        total = 0
        n_max = 2000000  # max iterations per scale
        
        for _ in range(n_max):
            if time.time() >= deadline:
                break
            total += 1
            # 70% random moves, 30% gradient-guided
            if rng.random() < 0.7:
                i = rng.randint(n)
                j = rng.randint(n)
            else:
                k_star = int(np.argmax(C))
                # Gradient: S[:, k_star]
                # Want to increase h where gradient is most negative, decrease where most positive
                i = int(np.argmin(S[:, k_star]))
                j = int(np.argmax(S[:, k_star]))
                if S[i, k_star] >= S[j, k_star]:
                    i, j = j, i
            if i == j:
                continue
            delta = scale * (2.0 * rng.random() - 1.0)
            if abs(delta) < 1e-12:
                continue
            ni = h[i] + delta
            nj = h[j] - delta
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            
            np.add(C, delta * (S[i] - S[j]), out=work)
            new_c5 = float(np.max(work)) * dx
            if new_c5 < c5 - 1e-15:
                h[i] = ni
                h[j] = nj
                C[:] = work  # C = work.copy()
                c5 = new_c5
                improved += 1
                # Update S
                S[:, n-1 + i - m_idx] -= delta
                S[:, n-1 + j - m_idx] += delta
                S[:, n-1 + m_idx - i] -= delta
                S[:, n-1 + m_idx - j] += delta
        if improved == 0 and scale_idx > 2:
            break
    return h, c5

        hh, cc = slsqp_opt(h, n, alpha, n/2.0, 300)
        if cc < c5:
            h, c5 = hh, cc
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5
    
    # Phase 3: Fast local search with incremental updates (main phase)
    deadline = t_start + budget_s - 10
    h, c5 = fast_local_search(h, n, dx, rng, deadline)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
