import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_ov(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def obj_g(x):

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + lam * (x.sum() - n * 0.5) ** 2

def jac_g(x):
    n = _P['n']; a = _P['a']; lam = _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)

            g += pw[k] * grad_ov(x, k, n)
    g /= a
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

                     bounds=bds, options={'maxiter': 350, 'ftol': 1e-19, 'gtol': 1e-16})
        hc = repair(r.x, n)
        return hc, c5v(hc, dx)
    except Exception:
        return h, c5v(h, dx)

    _P['n'], _P['a'], _P['lam'] = n, float(a), 0.0
    bds = [(0.0, 1.0)] * n

        r = minimize(obj_g, h, jac=jac_g, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq_f, 'jac': eq_j}],
                     options={'maxiter': 500, 'ftol': 1e-19})
        hc = repair(r.x, n)
        return hc, c5v(hc, dx)
    except Exception:
