def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = h.copy()
    target = n / 2.0
    # Simple projection: clip then adjust
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    # Binary search for shift
    # We need sum(clip(h + offset, 0, 1)) = target
    # But this changes which are clipped...
    # Simpler: iterative water-filling
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # need to increase some entries
            can_inc = h < 1.0
            frac = can_inc.sum()
            if frac == 0:
                break
            adj = diff / frac
            adj = min(adj, 1.0 - h[can_inc].min())
            h[can_inc] += adj
        else:
            can_dec = h > 0.0
            frac = can_dec.sum()
            if frac == 0:
                break
            adj = -diff / frac
            adj = min(adj, h[can_dec].min())
            h[can_dec] -= adj
    return np.clip(h, 0, 1)

    def sum_at_mu(mu):
        return np.clip(v - mu, 0, 1).sum()
    lo, hi = v.min() - 1, v.max() + 1

        s = sum_at_mu(mid)
        if s > target:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h))) * dx

def _project_exact(v, n):
    target = n / 2.0
    lo = v.min() - 1.5
    hi = v.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _subgrad_lag(h, L, n):
    g = np.zeros(n)
    if L == 0:
        return 1.0 - 2.0 * h
    elif L > 0:
        m = n - L
        g[:m] += (1.0 - h[L:])
        g[L:] -= h[:m]
    else:
        M = -L
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g

def _softobj(x, alpha, lam):
    c = _corr_full(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr_full(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        g += pk * _subgrad_lag(x, L, n)
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _lag_cycling(h, n, dx, time_budget, n_iter):
    h = h.copy()
    c5c = _c5_val(h, dx)
    for it in range(n_iter):
        if time.time() > time_budget:
            break
        c = _corr_full(h)
        cm = float(np.max(c))
        spread = cm - np.min(c)
        if spread < 1e-12:
            spread = 1.0
        thresh = cm - 0.08 * spread
        active = np.where(c >= thresh)[0]
        for k in active:
            L = k - (n - 1)
            g = _subgrad_lag(h, L, n)
            g_c = g - g.mean()
            ng = np.linalg.norm(g_c)
            if ng < 1e-14:
                continue
            g_c /= ng
            step = 0.02
            for _bt in range(20):
                h_new = _project_exact(h - step * g_c, n)
                c_new = _c5_val(h_new, dx)
                if c_new < c5c - 1e-15:
                    h = h_new
                    c5c = c_new
                    break
                step *= 0.6
        if it % 20 == 19:
            h = _project_exact(h, n)
    return h, c5c

def _pair_search(h, n, dx, time_budget, n_rounds):
    h = h.copy()
    c5c = _c5_val(h, dx)
    deltas = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                       -0.001, -0.003, -0.008, -0.02, -0.05])
    for rnd in range(n_rounds):
        if time.time() > time_budget:
            break
        c = _corr_full(h)
        n_act = max(5, n // 7)
        top_lags = np.argsort(c)[::-1][:n_act]
        g = np.zeros(n)
        for k in top_lags:
            L = k - (n - 1)
            g += _subgrad_lag(h, L, n)
        g_c = g - g.mean()
        order = np.argsort(np.abs(g_c))[::-1][:40]
        improved = False
        for ii in range(len(order)):
            if time.time() > time_budget:
                break
            i = order[ii]
            for jj in range(ii + 1, min(ii + 15, len(order))):
                j = order[jj]
                best_d = 0.0
                best_c = c5c
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5_val(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _project_exact(h, n)
                    c5c = _c5_val(h, dx)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            for d in deltas:
                cnt = 0
                if time.time() > time_budget:
                    break
                for ii in range(len(order)):
                    i = order[ii]
                    for jj in range(ii + 1, min(ii + 8, len(order))):
                        j = order[jj]
                        ni, nj = h[i] + d * 0.3, h[j] - d * 0.3
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5_val(h, dx)
                            if c < c5c:
                                c5c = c
                                cnt += 1
                            else:
                                h[i], h[j] = ni - d * 0.3, nj + d * 0.3
    return h, c5c

def _random_polish(h, n, dx, time_budget, rng, n_steps):
    h = h.copy()
    c5c = _c5_val(h, dx)
    tiny = np.array([0.0003, 0.001, 0.003, 0.008,
                     -0.0003, -0.001, -0.003, -0.008])
    for _ in range(n_steps):
        if time.time() > time_budget:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5_val(h, dx)
            if c < c5c:
                c5c = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _optimize_n(h0, n, dx, t_end, rng):
    best_h = h0.copy()
    best_c5 = _c5_val(best_h, dx)
    target_pts = n / 2.0

    t1 = t_end - (t_end - time.time()) * 0.35
    for a in [200, 1000, 5000, 25000, 100000, 500000]:
        if time.time() > t1:
            break
        try:
            res = minimize(lambda x: _softobj(x, a, 80.0), best_h.copy(),
                           jac=lambda x: _softgrad(x, a, 80.0),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-15})
            hc = _project_exact(res.x, n)
            c5v = _c5_val(hc, dx)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    t2 = t_end - (t_end - time.time()) * 0.35
    hc, c5v = _lag_cycling(best_h, n, dx, t2, 400)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    t3 = t_end - (t_end - time.time()) * 0.4
    hc, c5v = _pair_search(best_h, n, dx, t3, 200)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    hc, c5v = _lag_cycling(best_h, n, dx, t_end - 30, 300)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    hc, c5v = _random_polish(best_h, n, dx, t_end - 5, rng, 20000)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    best_h = _project_exact(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return best_h, best_c5
