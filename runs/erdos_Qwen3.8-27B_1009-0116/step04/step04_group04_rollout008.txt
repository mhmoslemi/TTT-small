```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _obj_grad(h, alpha, target, pen):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    e_sum = e.sum()
    w = e / e_sum
    obj = m + np.log(e_sum) / alpha + pen * (h.sum() - target) ** 2
    
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        i_lo = max(0, nm1 - k)
        i_hi = min(n - 1, nm1 + k)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        j = idx + k - nm1
        grad[idx] += wk * (1.0 - h[j])
        m_lo = max(0, i_lo + k - nm1)
        m_hi = min(n - 1, i_hi + k - nm1)
        if m_lo <= m_hi:
            idx_m = np.arange(m_lo, m_hi + 1)
            grad[idx_m] -= wk * h[idx_m - k + nm1]
    grad += 2.0 * pen * (h.sum() - target)
    return obj, grad

def _lbfgs(h, dx, alpha, maxiter, t_end):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best_h, best_c = h.copy(), _c5(h, dx)
    
    def objfun(x):
        o, _ = _obj_grad(x, alpha, tgt, 0)
        return o
    
    def gradfun(x):
        _, g = _obj_grad(x, alpha, tgt, 0)
        return g
    
    for pen in [10, 100, 1000, 10000]:
        if time.time() > t_end:
            break
        def objfun_p(x):
            o, _ = _obj_grad(x, alpha, tgt, pen)
            return o
        def gradfun_p(x):
            _, g = _obj_grad(x, alpha, tgt, pen)
            return g
        try:
            r = minimize(objfun_p, best_h, jac=gradfun_p, method='L-BFGS-B',
                        bounds=bds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
            hc = _repair(r.x, tgt)
            c = _c5(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
        except:
            pass
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G = {'alpha': 1.0, 'target': 0.0, 'pen': 0.0}

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    if np.isscalar(h) or len(h) == 1:
        return np.full(n_new, float(h[0]) if hasattr(h, '__getitem__') else h)
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _obj_grad(h, alpha, target, pen):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    e_sum = e.sum()
    w = e / e_sum
    obj = m + np.log(e_sum) / alpha + pen * (h.sum() - target) ** 2
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        i_lo = max(0, nm1 - k)
        i_hi = min(n - 1, nm1 + k)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        j = idx + k - nm1
        grad[idx] += wk * (1.0 - h[j])
        m_lo = max(0, i_lo + k - nm1)
        m_hi = min(n - 1, i_hi + k - nm1)
        if m_lo <= m_hi:
            idx_m = np.arange(m_lo, m_hi + 1)
            grad[idx_m] -= wk * h[idx_m - k + nm1]
    grad += 2.0 * pen * (h.sum() - target)
    return obj, grad

def _obj_cb(x):
    o, _ = _obj_grad(x, _G['alpha'], _G['target'], _G['pen'])
    return o

def _grad_cb(x):
    _, g = _obj_grad(x, _G['alpha'], _G['target'], _G['pen'])
    return g

def _lbfgs_phase(h, dx, alpha_list, maxiter_each, t_end, pen_list):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best_h, best_c = h.copy(), _c5(h, dx)
    for alpha in alpha_list:
        if time.time() > t_end:
            break
        for pen in pen_list:
            if time.time() > t_end:
                break
            _G['alpha'] = alpha
            _G['target'] = tgt
            _G['pen'] = pen
            try:
                r = minimize(_obj_cb, best_h, jac=_grad_cb, method='L-BFGS-B',
                             bounds=bds, options={'maxiter': maxiter_each,
                                                  'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
                hc = _repair(r.x, tgt)
                c = _c5(hc, dx)
                if c < best_c:
                    best_h, best_c = hc, c
            except Exception:
                pass
    return best_h, best_c

def _local_search(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    per_scale = max(50, n_iter // len(scales))
    for sc in scales:
        for _ in range(per_scale):
            if time.time() > t_end:
                return h, best
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            hi1, hj1 = h[i] + d, h[j] - d
            if 0 <= hi1 <= 1 and 0 <= hj1 <= 1:
                h[i], h[j] = hi1, hj1
                c = _c5(h, dx)
                if c < best:
                    best = c
                else:
                    h[i], h[j] = hi1 - d, hj1 + d
    return h, best

def _grad_local(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    alpha = 20000.0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        m = c.max()
        active = np.where(c > m - 1e-4)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        k = active[rng.randint(len(active))]
        _, g = _obj_grad(h, alpha, tgt, 0.0)
        i = int(np.argmax(np.abs(g)))
        j = int(np.argmin(g * np.sign(g[i])))
        if i == j:
            j = (i + 1) % n
        lr = 0.005 / max(1.0, np.abs(g[i]) + 1e-10)
        d = -np.sign(g[i] - g[j]) * lr
        hi1, hj1 = h[i] + d, h[j] - d
        if 0 <= hi1 <= 1 and 0 <= hj1 <= 1:
            h[i], h[j] = hi1, hj1
            c_new = _c5(h, dx)
            if c_new < best:
                best = c_new
            else:
                h[i], h[j] = hi1 - d, hj1 + d
        else:
            h[i], h[j] = hi1 - d, hj1 + d
    return h, best

def _snap_binary(h, dx, tgt, thresh=0.05):
    h2 = h.copy()
    h2[h2 < thresh] = 0.0
    h2[h2 > 1.0 - thresh] = 1.0
    h2 = _repair(h2, tgt)
    return h2
```
