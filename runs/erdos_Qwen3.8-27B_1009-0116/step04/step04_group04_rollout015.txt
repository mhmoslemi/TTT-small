```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _project(h, target):
    """Project onto {0<=h<=1, sum(h)=target}."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    # Fix tiny residual
    d = target - h_p.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.where(h_p < 1.0)[0]
            if len(idx) > 0:
                per = min(d / len(idx), 1e-14)
                h_p[idx] = np.minimum(h_p[idx] + per, 1.0)
        else:
            idx = np.where(h_p > 0.0)[0]
            if len(idx) > 0:
                per = max(d / len(idx), -1e-14)
                h_p[idx] = np.maximum(h_p[idx] + per, 0.0)
    return h_p

def _compute_grad(h, w):
    """Gradient of weighted sum of correlations. w has length 2n-1."""
    n = len(h)
    grad = np.zeros(n)
    for s in range(n):
        wt = w[n - 1 + s]
        if wt < 1e-15:
            continue
        if s == 0:
            grad += wt * (1.0 - 2.0 * h)
        else:
            grad[:n-s] += wt * (1.0 - h[s:])
            grad[s:] -= wt * h[:n-s]
    for t in range(1, n):
        wt = w[n - 1 - t]
        if wt < 1e-15:
            continue
        grad[:n-t] += wt * (1.0 - h[t:])
        grad[t:] -= wt * h[:n-t]
    return grad

def _nesterov_opt(h, tgt, dx, alpha, n_iter, lr0, decay, t_end):
    """Nesterov accelerated projected gradient with soft-max smoothing."""
    n = len(h)
    best_h = h.copy()
    best_c = _c5(h, dx)
    x = h.copy()
    y = h.copy()
    t = 1.0
    lr = lr0
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        # Compute soft-max weights
        c = _corr_full(y)
        c_max = c.max()
        w = np.exp(alpha * (c - c_max))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        g = _compute_grad(y, w)
        
        # Nesterov step
        beta = (t - 1.0) / (t + 2.0)
        x_new = y - lr * g
        x_new = _project(x_new, tgt)
        
        # Check improvement
        c_new = _c5(x_new, dx)
        if c_new < best_c:
            best_h = x_new.copy()
            best_c = c_new
        
        y = x_new + beta * (x_new - x)
        y = _project(y, tgt)
        x = x_new
        t += 1.0
        lr *= decay
    
    return best_h, best_c

def _sinusoidal_perturb(h, rng, n_modes=5, amp=0.05):
    """Add sinusoidal perturbations."""
    n = len(h)
    x = np.linspace(0, 2, n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.uniform(1, 5)
```
