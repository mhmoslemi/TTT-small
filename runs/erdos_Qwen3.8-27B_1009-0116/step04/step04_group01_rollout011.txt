            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _mom_proj(h, alpha, nsteps, step0, mu):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        g = _smooth_grad(h, alpha)
        v = mu * v - step * g
        lam = 1.0
        improved = False
        for _bt in range(25):
            h_try = _proj_to_sum(h - lam * v, n * 0.5)
            fn = _smooth_obj(h_try, alpha)
            if fn < f - 1e-4 * lam * np.dot(v, h_try - h):
                h, f = h_try, fn
                improved = True
                break
            lam *= 0.5
        if not improved:
            h = _proj_to_sum(h - lam * v, n * 0.5)
            f = _smooth_obj(h, alpha)
        step = min(step * 1.01, step0 * 3.0)
    return h

def _subgrad_descent(h, nsteps, step0, topk):
    n = len(h)
    c5c = _eval(h)
    step = step0
    for _ in range(nsteps):
        g = _topk_grad(h, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        for _bt in range(40):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c = h_new, c5n
                break
            step *= 0.5
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase A: SLSQP with increasing alpha at base resolution
    for alpha in [500, 3000, 15000, 80000]:
        if _el() > 180:
            break
        h_new = _slsqp_opt(best_h, alpha, 2500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase B: Momentum projected gradient at base resolution
    if _el() < budget_s - 500:
        h_new = _mom_proj(best_h, 50000, 600, 0.003, 0.85)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase C: Upsample to 2x and optimize
    if _el() < budget_s - 450:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [3000, 30000, 150000]:
            if _el() > budget_s - 300:
                break
            h_hi = _slsqp_opt(h_hi, alpha, 1500, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 250:
            h_hi = _mom_proj(best_h, 80000, 600, 0.001, 0.9)
            h_hi = _repair(h_hi, n1)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase D: Unsmoothed top-k subgradient
    if _el() < budget_s - 200:
        nc = best_n
        h_new, _ = _subgrad_descent(best_h, 400, 0.008, 5)
        h_new = _repair(h_new, nc)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase E: Multi-start with diverse perturbations
    nc = best_n
    if _el() < budget_s - 300:
        for _r in range(25):
            if _el() > budget_s - 300:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.015 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(3, max(4, nc // 8))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.05 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                shift = rng.randint(1, max(2, nc // 12))
                hp = _repair(np.roll(best_h, shift), nc)
            elif strat == 4:
                hp = best_h.copy()
                freqs = rng.randint(1, nc // 2 + 1, size=4)
                t_arr = np.arange(nc)
                for f in freqs:
                    hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * t_arr / nc)
                hp = _repair(hp, nc)
            else:
                hp = best_h.copy()
                mask = rng.rand(nc) < 0.3
                hp[mask] = rng.rand(mask.sum())
                hp = _repair(hp, nc)
            alpha_pick = [10000, 50000, 200000][rng.randint(3)]
            h_new = _slsqp_opt(hp, alpha_pick, 800, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase F: Simulated-annealing pairwise swaps
    if _el() < budget_s - 60:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        temp = 0.002
        for scale in [0.004, 0.0015, 0.0004, 0.0001]:
            if _el() > budget_s - 60:
                break
            for _it in range(6000):
                if _el() > budget_s - 60:
                    break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h, best_n = c5n, h.copy(), nc
                elif rng.rand() < np.exp(-(c5n - c5c) / max(temp, 1e-8)):
                    c5c = c5n
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
            temp *= 0.4

    # Phase G: Final SLSQP polish
    if _el() < budget_s - 12:
        h = _slsqp_opt(best_h, 300000, 2000, 1e-19)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase H: Ultra-fine local search
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(12000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h, best_n = c5n, h.copy(), nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
