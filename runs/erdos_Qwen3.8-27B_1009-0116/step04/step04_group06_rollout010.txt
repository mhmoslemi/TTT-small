def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, n):
    return float(np.max(_corr_full(h))) * (2.0 / n)

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff += -a if diff > 0 else a
    return h

def _project_box(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        m = (hp > 1e-9) & (hp < 1.0 - 1e-9)
        if m.sum() > 0:
            hp[m] += diff / m.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _grad_corr(h, n, w):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _smoothmax_opt(h, n, alpha, maxiter, budget_end):
    dx = 2.0 / n
    def obj(x):
        c = _corr_full(x) * dx
        m = np.max(c)
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        c = _corr_full(x) * dx
        m = np.max(c)
        s = np.clip(alpha * (c - m), -300, 300)
        e = np.exp(s)
        w = e / e.sum()
        return _grad_corr(x, n, w)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 50})
    return _repair(res.x, n)

def _powersum_opt(h, n, p, maxiter, budget_end):
    dx = 2.0 / n
    def obj(x):
        c = np.maximum(_corr_full(x) * dx, 0.0)
        return np.sum(c ** p) ** (1.0 / p)
    def grad(x):
        c = np.maximum(_corr_full(x) * dx, 0.0)
        sp = np.sum(c ** p)
        if sp < 1e-30:
            w = np.full(len(c), 1.0 / len(c))
        else:
            w = c ** (p - 1) / sp ** ((p - 1.0) / p)
        return _grad_corr(x, n, w)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 50})
    return _repair(res.x, n)

def _threshold_cycle(h, n, budget_end):
    best_h, best_c5 = h.copy(), _c5_val(h, n)
    for tt in np.linspace(0.35, 0.65, 7):
        if time.time() > budget_end:
            break
        h_t = (h > tt).astype(float)
        h_t = _repair(h_t, n)
        hc = _smoothmax_opt(h_t, n, 100000, 800, budget_end)
        c5 = _c5_val(hc, n)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, hc
    return best_h, best_c5

def _local_search(h, n, budget_end, rng, n_trials=100000):
    best_c5 = _c5_val(h, n)
    h = h.copy()
    end_t = min(t0 + budget_s, time.time() + (budget_end - time.time()))
    steps = [0.005, 0.01, 0.03, 0.1]
    cnt = 0
    while time.time() < end_t and cnt < n_trials:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        cnt += 1
        found = False
        for d in steps:
            for sgn in (1, -1):
                ni, nj = hi + sgn * d, hj - sgn * d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = _c5_val(h, n)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        hi, hj = ni, nj
                        found = True
                        break
                    h[i], h[j] = hi, hj
            if found:
                break
        h[i], h[j] = hi, hj
    return h, best_c5

def _basin_hop(h, n, budget_end, rng):
    best_h, best_c5 = h.copy(), _c5_val(h, n)
    end_t = min(t0 + budget_s, time.time() + (budget_end - time.time()))
    for _ in range(30):
        if time.time() > end_t:
            break
        eps = 0.03 * (1.0 - 0.5 * (time.time() - t0) / max(1, budget_s - 30))
        h_pert = _repair(best_h + eps * rng.randn(n), n)
        hc = _smoothmax_opt(h_pert, n, 100000, 600, end_t)
        c5 = _c5_val(hc, n)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, hc
    return best_h, best_c5

    budget_s = budget_s

    except NameError:

    n0 = len(h0)
    h0 = _repair(h0, n0)
    best_h, best_n, best_c5 = h0, n0, _c5_val(h0, n0)

    # Phase 1: L-BFGS-B smooth max, increasing alpha
    p1_end = t0 + budget_s * 0.22
    for alpha in [3000, 15000, 60000, 200000, 600000]:
        if time.time() > p1_end:
            break
        hc = _smoothmax_opt(best_h, best_n, alpha, 1500, p1_end)
        c5 = _c5_val(hc, best_n)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 2: Power-sum (different geometry)
    p2_end = t0 + budget_s * 0.35
    for p in [30, 80, 150]:
        if time.time() > p2_end:
            break
        hc = _powersum_opt(best_h, best_n, p, 1200, p2_end)
        c5 = _c5_val(hc, best_n)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 3: Threshold-and-refine
    p3_end = t0 + budget_s * 0.48
    best_h, best_c5 = _threshold_cycle(best_h, best_n, p3_end)
    if time.time() < p3_end:
        hc = _smoothmax_opt(best_h, best_n, 500000, 1000, p3_end)
        c5 = _c5_val(hc, best_n)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 4: Multi-resolution
    p4_end = t0 + budget_s * 0.65
    for n_try in [70, 100, 200, 280]:
        if time.time() > p4_end:
            break
        if n_try == best_n:
            continue
        xo = np.linspace(0, 2.0, best_n, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        h_new = _repair(np.interp(xn, xo, best_h), n_try)
        hc = _smoothmax_opt(h_new, n_try, 80000, 800, p4_end)
        c5 = _c5_val(hc, n_try)
        if c5 < best_c5 - 1e-6:
            best_c5, best_h, best_n = c5, hc, n_try
        if hc is not None and c5 < best_c5 + 0.001:
            hc2 = _powersum_opt(hc, n_try, 100, 600, p4_end)
            c52 = _c5_val(hc2, n_try)
            if c52 < best_c5:
                best_c5, best_h, best_n = c52, hc2, n_try

    # Phase 5: Basin hopping for exploration
    if time.time() < budget_s * 0.78:
        best_h, best_c5 = _basin_hop(best_h, best_n, budget_s * 0.78, rng)

    # Phase 6: Final polish
    p6_end = t0 + budget_s * 0.84
    for alpha in [200000, 800000, 2000000]:
        if time.time() > p6_end:
            break
        hc = _smoothmax_opt(best_h, best_n, alpha, 1000, p6_end)
        c5 = _c5_val(hc, best_n)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 7: Intensive local search
    if time.time() < budget_s - 20:
        best_h, best_c5 = _local_search(best_h, best_n, budget_s - 10, rng, 200000)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
