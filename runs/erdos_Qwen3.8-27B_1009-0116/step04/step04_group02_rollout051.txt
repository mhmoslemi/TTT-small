import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, 0), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sa_phase(h, n, dx, rng, t_end, T_init, T_min, cool_rate):
    target = n / 2.0
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    T = T_init
    stagnation = 0
    while time.time() < t_end:
        h_save = h.copy()
        mt = rng.randint(4)
        if mt == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.05, 0.05)
            if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                continue
            h[i] += d
            h[j] -= d
        elif mt == 1:
            bs = rng.randint(2, 6)
            i = rng.randint(0, max(1, n - bs))
            j = rng.randint(0, max(1, n - bs))
            if abs(i - j) < bs:
                continue
            d = rng.uniform(-0.03, 0.03)
            if np.any(h[i:i+bs] + d < -1e-9) or np.any(h[i:i+bs] + d > 1 + 1e-9):
                continue
            if np.any(h[j:j+bs] - d < -1e-9) or np.any(h[j:j+bs] - d > 1 + 1e-9):
                continue
            h[i:i+bs] += d
            h[j:j+bs] -= d
        elif mt == 2:
            f = rng.randint(1, 10)
            ph = rng.uniform(0, 6.283)
            amp = rng.uniform(-0.03, 0.03)
            h += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
            h = repair(h, n)
        else:
            k = rng.randint(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            eps = 0.015 * rng.randn(k)
            h[idx] += eps
            h = repair(h, n)
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - current_c5
        if delta < -1e-14 or (T > 1e-9 and rng.random() < np.exp(-max(0, delta) / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            h = h_save
            stagnation += 1
        T = max(T_min, T * cool_rate)
        if stagnation > 30000:
            T = min(T_init * 0.4, T * 4.0)
            stagnation = 0
    return best_h, best_c5

def pair_local(h, n, dx, rng, t_end, scales, per_scale):
    target = n / 2.0
    best_c5 = compute_c5(h, dx)
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.005:
            break
    return h, best_c5

def block_local(h, n, dx, rng, t_end, n_iters):
    best_c5 = compute_c5(h, dx)
    for _ in range(n_iters):
        if time.time() >= t_end:
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        if np.any(h[i:i+bs] + eps < -1e-9) or np.any(h[i:i+bs] + eps > 1 + 1e-9):
            continue
        if np.any(h[j:j+bs] - eps < -1e-9) or np.any(h[j:j+bs] - eps > 1 + 1e-9):
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = compute_c5(h, dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5
