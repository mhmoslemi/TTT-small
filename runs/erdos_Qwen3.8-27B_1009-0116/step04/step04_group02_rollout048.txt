import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) + pen * (np.sum(x) - target) ** 2

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lbfgs_opt(h, n, alpha, maxiter, pen):
    target = n / 2.0
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = repair(res.x, n)
        return hc, compute_c5(hc)
    except Exception:
        return h, compute_c5(h)

def slsqp_opt(h, n, alpha, maxiter):
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def con(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = repair(res.x, n)
        return hc, compute_c5(hc)
    except Exception:
        return h, compute_c5(h)

def sa_anneal(h, n, dx, rng, t_start, t_end):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    total_dur = max(t_end - t_start, 1.0)
    patience = 0
    max_pat = 30000
    T0, T1 = 1.5, 0.0005
    while time.time() < t_end:
        frac = min(1.0, (time.time() - t_start) / total_dur)
        T = T0 * (T1 / T0) ** frac
        h_new = h_cur.copy()
        r = rng.random()
        if r < 0.25:
            i, j = rng.randint(0, n), rng.randint(0, n)
            if i != j:
                d = rng.uniform(-0.12, 0.12)
                h_new[i] += d
                h_new[j] -= d
        elif r < 0.45:
            i, j = rng.randint(0, n), rng.randint(0, n)
            if i != j:
                push = rng.uniform(0.04, 0.35) * (1.0 if h_cur[i] < 0.5 else -1.0)
                h_new[i] += push
                h_new[j] -= push
        elif r < 0.65:
            bs = rng.randint(1, 6)
            i = rng.randint(0, max(1, n - bs))
            j = rng.randint(0, max(1, n - bs))
            if abs(i - j) >= bs:
                eps = rng.uniform(-0.07, 0.07)
                h_new[i:i+bs] += eps
                h_new[j:j+bs] -= eps
        elif r < 0.82:
            k = rng.randint(1, 12)
            idx = rng.choice(n, k, replace=False)
            h_new[idx] += rng.normal(0, 0.04, k)
        else:
            h_new = h_cur + rng.normal(0, 0.08, n)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or (T > 1e-7 and rng.random() < np.exp(-delta / T)):
            h_cur = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_h = h_cur.copy()
                best_c5 = cur_c5
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
        if patience > max_pat and T > 0.005:
            h_cur = best_h + rng.normal(0, 0.08, n)
            h_cur = repair(h_cur, n)
            cur_c5 = compute_c5(h_cur)
            patience = 0
    return best_h, best_c5

def fine_local(h, n, dx, rng, t_end):
    best_c5 = compute_c5(h)
    h = h.copy()
    for scale in [0.008, 0.003, 0.001, 0.0004, 0.0001, 0.00003]:
        improved = False
        for _ in range(80000):
            if time.time() >= t_end:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c5n = compute_c5(h)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and scale < 0.001:
            break
    return h, best_c5

def block_local(h, n, dx, rng, t_end, n_iters):
    best_c5 = compute_c5(h)
    h = h.copy()
    for _ in range(n_iters):
        if time.time() >= t_end:
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, max(1, n - bs + 1))
        j = rng.randint(0, max(1, n - bs + 1))
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (-1e-9 <= h[i+di] + eps <= 1.0+1e-9) or not (-1e-9 <= h[j+di] - eps <= 1.0+1e-9):
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = compute_c5(h)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def perturb_sol(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(3, n - 3))
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        hp += rng.uniform(0.04, 0.15) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.05, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 4), rng.randint(1, 6)
        hp += rng.uniform(0.04, 0.12) * np.sin(2*np.pi*f1*np.arange(n)/n) * np.cos(2*np.pi*f2*np.arange(n)/n)
    else:
        hp *= rng.uniform(0.8, 1.2)
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    h = repair(h0.copy(), n)
    best_h, best_c5 = h.copy(), compute_c5(h)

    def rem():
        return budget_s - (time.time() - t_start)

    # Phase 1: L-BFGS warmup (50s)
    t_end = t_start + 50
    for alpha in [300, 1500, 6000]:
        if time.time() >= t_end or rem() < 500:
            break
        hh, cc = lbfgs_opt(h, n, alpha, 300, 3000)
        if cc < best_c5:
            h, best_h, best_c5 = hh, hh.copy(), cc

    h_sa, c5_sa = sa_anneal(h, n, dx, rng, t_start + 50, t_end)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa.copy(), c5_sa
    h = best_h

    # Phase 3: L-BFGS + SLSQP refinement after SA (100s)
    t_end = time.time() + min(100, rem() - 250)
    for alpha in [5000, 20000, 80000]:
        if time.time() >= t_end:
            break
        hh, cc = lbfgs_opt(h, n, alpha, 400, 8000)
        if cc < best_c5:
            h, best_h, best_c5 = hh, hh.copy(), cc
        hh, cc = slsqp_opt(h, n, alpha * 2, 300)
        if cc < best_c5:
            h, best_h, best_c5 = hh, hh.copy(), cc

    # Phase 4: Block local search (80s)
    if rem() > 200:
        t_end = time.time() + min(80, rem() - 130)
        h, c5_bl = block_local(h, n, dx, rng, t_end, 20000)
        if c5_bl < best_c5:
            best_h, best_c5 = h.copy(), c5_bl
        h = best_h

    # Phase 5: Multi-restart (150s)
    t_end = time.time() + min(150, rem() - 100)
    for r in range(15):
        if time.time() >= t_end:
            break
        hp = perturb_sol(best_h, n, rng, r % 5)
        hh, cc = lbfgs_opt(hp, n, 4000, 200, 5000)
        hh, cc = slsqp_opt(hh, n, 20000, 250)
        if cc < best_c5:
            best_h, best_c5 = hh.copy(), cc

    # Phase 6: Fine local search (120s)
    if rem() > 60:
        t_end = time.time() + min(120, rem() - 30)
        h, c5_fs = fine_local(best_h, n, dx, rng, t_end)
        if c5_fs < best_c5:
            best_h, best_c5 = h.copy(), c5_fs
        h = best_h

    # Phase 7: Ultra-fine + final polish
    if rem() > 30:
        t_end = time.time() + min(20, rem() - 10)
        for alpha in [200000, 500000]:
            if time.time() >= t_end:

            hh, cc = lbfgs_opt(h, n, alpha, 500, 20000)
            if cc < best_c5:
                best_h, best_c5 = hh.copy(), cc

    if rem() > 10:
        t_end = time.time() + max(3, rem() - 3)
        h, c5_f = fine_local(best_h, n, dx, rng, t_end)
        if c5_f < best_c5:
            best_h, best_c5 = h.copy(), c5_f

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
