```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_val(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - hp.sum()
    if abs(d) > 1e-14:
        m = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if m.sum() > 0:
            hp[m] += d / m.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _topk_grad(h, n, topk):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = min(topk, 2 * n - 1)
    idx = np.argsort(c)[-m:]
    cv = c[idx]
    w = np.maximum(cv - cv[0] + 1e-9, 1e-9)
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(m):
        lag = idx[ki] - (n - 1)
        wk = w[ki]
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _pgd(h, n, end_t, topk, lr_init):
    h = h.copy()
    best = _c5_val(h, n)
    lr = lr_init
    stall = 0
    while time.time() < end_t:
        g = _topk_grad(h, n, topk)
        h_new = _project(h - lr * g, n)
        c = _c5_val(h_new, n)
        if c < best - 1e-17:
            h = h_new
            best = c
            stall = 0
            lr = min(lr * 1.04, 0.4)
        else:
            stall += 1
            lr *= 0.93
            if stall > 400:
                lr = max(0.001, lr_init * 0.6)
                stall = 0
    return h, best

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad_f(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -350, 350)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _perturb(h, n, rng, scale):
    x = np.arange(n)
    freq = rng.randint(2, max(3, n // 4))
    phase = rng.rand() * 2 * np.pi
    amp = rng.uniform(0.02, scale)
    h_new = h + amp * np.sin(2 * np.pi * freq * x / n + phase)
    return _repair(h_new, n)

def _pair_search(h, n, end_t, rng):
    h = h.copy()
    best = _c5_val(h, n)
    deltas = np.array([0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02])
    while time.time() < end_t:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        if rng.rand() < 0.5:
            d = -d
        hi_save, hj_save = h[i], h[j]
        h[i], h[j] = hi_save + d, hj_save - d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c = _c5_val(h, n)
            if c < best - 1e-16:
                best = c
            else:
                h[i], h[j] = hi_save, hj_save
    return h, best

def _block_search(h, n, end_t, rng):
    h = h.copy()
    best = _c5_val(h, n)
    while time.time() < end_t:
        bs = rng.choice([2, 3, 4, 5])
        i = rng.randint(0, max(1, n - bs))
        j = rng.randint(0, max(1, n - bs))
        if abs(i - j) < bs:
            continue
        d = rng.uniform(0.002, 0.04)
        if rng.rand() < 0.5:
            d = -d
        per = d / bs
        mi = np.arange(i, i + bs)
        mj = np.arange(j, j + bs)
        if np.all(h[mi] + per >= 0) and np.all(h[mi] + per <= 1) and \
           np.all(h[mj] - per >= 0) and np.all(h[mj] - per <= 1):
            h[mi] += per
            h[mj] -= per
            c = _c5_val(h, n)
            if c < best - 1e-16:
                best = c
            else:
                h[mi] -= per
                h[mj] += per
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        n = len(initial_h_values)
        h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    except Exception:
        n = 140
        h = _repair(np.full(n, 0.5), n)

    best_h = h.copy()
    best_n = n
    best_c5 = _c5_val(best_h, best_n)

    # Phase 1: Aggressive top-2 PGD
    end1 = t0 + budget_s * 0.22
    h_w, c5_w = _pgd(best_h, best_n, end1, 2, 0.05)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w.copy()

    # Phase 2: Top-4 PGD
    end2 = t0 + budget_s * 0.40
    h_w, c5_w = _pgd(best_h, best_n, end2, 4, 0.04)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w.copy()

    # Phase 3: Restart with perturbation + top-5
    for trial in range(8):
        end_trial = min(t0 + budget_s * (0.40 + 0.03 * (trial + 1)), t0 + budget_s * 0.64)
        h_p = _perturb(best_h, best_n, rng, 0.08)
        h_w, c5_w = _pgd(h_p, best_n, end_trial, 5, 0.03)
        if c5_w < best_c5:
            best_c5, best_h = c5_w, h_w.copy()

    # Phase 4: L-BFGS-B with smooth max
    end4 = t0 + budget_s * 0.72
    for alpha in [80000, 200000]:
        if time.time() > end4:
            break
        try:
            res = minimize(_smooth_obj, best_h.copy(), args=(best_n, alpha),
                           jac=_smooth_grad_f, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-14})
            hc = _repair(res.x, best_n)
            c5 = _c5_val(hc, best_n)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 5: Top-3 PGD polish
    end5 = t0 + budget_s * 0.80
    h_w, c5_w = _pgd(best_h, best_n, end5, 3, 0.02)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w.copy()

    # Phase 6: Block search
    end6 = t0 + budget_s * 0.90
    best_h, best_c5 = _block_search(best_h, best_n, end6, rng)

    # Phase 7: Fine pair search
    end7 = t0 + budget_s - 5
    best_h, best_c5 = _pair_search(best_h, best_n, end7, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)
    return (best_h, best_c5, best_n)
```
