Actually, let me vectorize the 1D search. For a fixed i and varying j:

For a given d, f_j(d) = max_k (c_k + d * a_j(k)) where a_j(k) = dx*(G_i(k) - G_j(k)).

This is still O(n) per eval, O(n) evals for ternary search. Total per j: O(n²). Per i: O(n³).

Actually, I just realized: for the fast pairwise search, I don't need to recompute the full correlation. I only need to check if the max decreases. The current max is at k*. The new value at k* is:
c_new[k*] = c[k*] + d*dx*(G_i(k*) - G_j(k*))

If this doesn't decrease, we reject immediately (O(1) check). If it does, we need to check all other k to make sure the max doesn't increase. This is O(n).

So most rejections are O(1), and acceptances are O(n). This makes the search much faster on average.

```python
def _fast_pairwise_v2(h, n_iter, scale, rng):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    c_max = c.max()
    c5c = c_max * dx
    shifts = np.arange(2*n-1) - (n-1)
    
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        
        # Quick check at argmax
        s1_i = i + shifts
        s2_i = i - shifts
        s1_j = j + shifts
        s2_j = j - shifts
        
        m1i = (s1_i >= 0) & (s1_i < n)
        m2i = (s2_i >= 0) & (s2_i < n)
        m1j = (s1_j >= 0) & (s1_j < n)
        m2j = (s2_j >= 0) & (s2_j < n)
        
        Ah = 1.0 - h
        Gi = np.zeros(2*n-1)
        Gi[m1i] = Ah[s1_i[m1i]]
        Gi[m2i] -= h[s2_i[m2i]]
        Gj = np.zeros(2*n-1)
        Gj[m1j] = Ah[s1_j[m1j]]
        Gj[m2j] -= h[s2_j[m2j]]
        
        dc = d * dx * (Gi - Gj)
        
        k_w = c.argmax()
        if c[k_w] + dc[k_w] >= c_max - 1e-15:
            continue  # Kmax doesn't decrease, reject
        
        c_new = c + dc
        c_max_new = c_new.max()
        if c_max_new < c_max - 1e-15:
            h[i], h[j] = ni, nj
            c = c_new
            c_max = c_max_new
            c5c = c_max * dx
    
    return h, c5c
```

This is O(n) per step (dominated by computing Gi, Gj), with a fast O(n) rejection at the argmax.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _proj_to_sum(h, target):
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    if abs(h.sum() - target) > 1e-10:
        h = _proj_to_sum(h, target)
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(_corr(h).max() * 2.0 / n)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500.0, 500.0)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter=2000, ftol=1e-19):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(np.asarray(res.x), n)
    except Exception:
        return _repair(h, n)

def _lbfgs(h, alpha, maxiter=5000, ftol=1e-19):
    n = len(h)
    def obj(x):
        xp = np.clip(x, 0.0, 1.0)
        return _smooth_obj(xp, alpha)
    def jac(x):
        xp = np.clip(x, 0.0, 1.0)
        return _smooth_grad(xp, alpha)
    try:
        from scipy.optimize import minimize as min2
        res = min2(obj, h, jac=jac, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol})
        best = _repair(res.x, n)
    except Exception:
        best = _repair(h, n)
    # Also try constrained via SLSQP
    return best

def _pairwise_search(h, n_iter, scale, rng):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifts = np.arange(2 * n - 1) - (n - 1)
    Ah = 1.0 - h
    h_work = h.copy()
    
    def _Gi(t):
        s1 = t + shifts
        s2 = t - shifts
        m1 = (s1 >= 0) & (s1 < n)
        m2 = (s2 >= 0) & (s2 < n)
        v = np.zeros(2 * n - 1)
        v[m1] = Ah[s1[m1]]
        v[m2] -= h_work[s2[m2]]
        return v
    
    n_acc = 0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h_work[i] + d
        nj = h_work[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        Gi = _Gi(i)
        Gj = _Gi(j)
        dc = d * dx * (Gi - Gj)
        kw = c.argmax()
        if c[kw] + dc[kw] >= cmax:
            continue
        c_new = c + dc
        cmax_new = c_new.max()
        if cmax_new < cmax - 1e-15:
            h_work[i] = ni
            h_work[j] = nj
            Ah[i] = 1.0 - ni
            Ah[j] = 1.0 - nj
            c = c_new
            cmax = cmax_new
            n_acc += 1
    return h_work, cmax * dx

def _targeted_descent(h, max_rounds=300):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_rounds):
        c = _corr(h)
        kw = c.argmax()
        sh = kw - (n - 1)
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n-sp] -= h[sp:]
        g *= dx
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002,
                     0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj_to_sum(h - step * g, 0.5 * n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _repeat(h, factor):
    return np.repeat(h, factor)
```

**Key new idea: Shift-Specific Optimization**

Instead of the standard log-sum-exp smoothing, I'll use an "iterative reweighting" approach:
1. Start with equal weights w_k = 1/K
2. Compute h* = argmin sum_k w_k * c_k(h) (approximately, via SLSQP on the weighted sum)
3. Update w_k ∝ exp(β * c_k(h*)) (softmax reweighting)
4. Repeat with increasing β

This is essentially the multiplicative weights / mirror descent approach to the max problem. At convergence, w concentrates on the argmax shift, and h is optimized for that specific shift.

The advantage: at each step, the objective is a quadratic (sum of weighted c_k's), which is easier to optimize than the log-sum-exp.

But the quadratic is h^T M(h) h where M depends on h... no, c_k = sum h_i(1-h_{i+sh}) dx is bilinear in h, not quadratic. The sum with weights is still bilinear:
sum_k w_k * c_k = sum_k w_k * dx * sum_i h_i - sum_k w_k * dx * sum_i h_i * h_{i+sh_k}

The first term is linear in h (and fixed since sum h = n/2). The second term is quadratic:
h^T * (sum_k w_k * P_k) * h

where P_k is the shift matrix for shift sh_k. This is a quadratic form! So the objective is:
const - dx * h^T * Q * h

where Q = sum_k w_k * P_k is a symmetric matrix (if the shifts include both k and -k).

Wait, is P_k symmetric? P_k represents the shift by sh_k: (P_k h)_i = h_{i+sh_k}. This is NOT symmetric in general. But h^T P_k h = sum_i h_i h_{i+sh_k} = sum_i h_{i+sh_k} h_i = h^T P_{-k} h.

So the quadratic form is:
sum_k w_k * h^T P_k h = h^T (sum_k w_k P_k) h

This is h^T S h where S = sum_k w_k P_k. S is a symmetric matrix if w_k = w_{-k}.

So the inner optimization is:
minimize: -h^T S h (equivalently, maximize h^T S h)
subject to: 0 ≤ h ≤ 1, sum h = n/2

This is a MAXIMIZATION of a quadratic form with bounds and one equality constraint. If S is positive semidefinite, this is a concave maximization, i.e., a convex minimization of -h^T S h. But S might not be PSD.

In any case, this can be solved by a QP solver. But scipy's minimize doesn't handle QPs directly. I could use cvxpy if S is PSD, or use a second-order method.

Actually, since the objective is quadratic and the constraints are linear (bounds + one equality), I can use a projected gradient descent or Newton's method in 2-3 iterations per reweighting step.

```python
def _pairwise_sweep(h, scale, rng, n_passes=10):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifts = np.arange(2 * n - 1) - (n - 1)
    Ah = 1.0 - h
    
    for _pass in range(n_passes):
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                # Try best d for (i, j)
                d_lo = max(-h[i], h[j] - 1.0) + 1e-12
                d_hi = min(1.0 - h[i], h[j]) - 1e-12
                if d_lo >= d_hi:
                    continue
                # Try a few d values
                for d in [0.5*(d_lo+d_hi), d_lo + 0.25*(d_hi-d_lo), d_lo + 0.75*(d_hi-d_lo)]:
                    ni = h[i] + d
                    nj = h[j] - d
                    # Fast check
                    s1i = i + shifts; s2i = i - shifts
                    s1j = j + shifts; s2j = j - shifts
                    m1i = (s1i >= 0) & (s1i < n)
                    m2i = (s2i >= 0) & (s2i < n)
                    m1j = (s1j >= 0) & (s1j < n)
                    m2j = (s2j >= 0) & (s2j < n)
                    Gi = np.zeros(2*n-1)
                    Gi[m1i] = Ah[s1i[m1i]]
                    Gi[m2i] -= h[s2i[m2i]]
                    Gj = np.zeros(2*n-1)
                    Gj[m1j] = Ah[s1j[m1j]]
                    Gj[m2j] -= h[s2j[m2j]]
                    dc = d * dx * (Gi - Gj)
                    kw = c.argmax()
                    if c[kw] + dc[kw] >= cmax:
                        continue
                    c_new = c + dc
                    if c_new.max() < cmax - 1e-15:
                        h[i] = ni; h[j] = nj
                        Ah[i] = 1-ni; Ah[j] = 1-nj
                        c = c_new
                        cmax = c_new.max()
                        break
    return h, cmax * dx
```

import numpy as np
from scipy.optimize import minimize
import time

_NG = [140]

def _cfn(x):
    return x.sum() - 0.5 * _NG[0]

def _cjn(x):
    return np.ones(len(x))

def _pts(h, tgt):
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _rep(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = 0.5 * n
    if abs(h.sum() - t) > 1e-10:
        h = _pts(h, t)
    return h

def _cr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _ev(h):
    return float(_cr(h).max() * 2.0 / len(h))

def _sobj(h, a):
    c = _cr(h)
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / a

def _sgrad(h, a):
    n = len(h)
    c = _cr(h)
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slq(h, a, mi=3000, ft=1e-19):
    n = len(h)
    _NG[0] = n
    b = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfn, 'jac': _cjn}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(a,), method='SLSQP',
                     bounds=b, constraints=cs,
                     options={'maxiter': mi, 'ftol': ft})
        return _rep(np.asarray(r.x), n)
    except Exception:
        return _rep(h, n)

def _targ(h, mx=400):
    n = len(h)
    dx = 2.0 / n
    c5 = _ev(h)
    for _ in range(mx):
        c = _cr(h)
        kw = c.argmax()
        sh = kw - (n - 1)
        g = np.zeros(n)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
        g *= dx
        ok = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
            hn = _pts(h - st * g, 0.5 * n)
            cn = _ev(hn)
            if cn < c5 - 1e-14:
                h = hn
                c5 = cn
                ok = True
                break
        if not ok:
            break
    return h, c5

def _pw(h, ni, sc, rng):
    n = len(h)
    dx = 2.0 / n
    c = _cr(h)
    cm = c.max()
    sh = np.arange(2 * n - 1) - (n - 1)
    Ah = 1.0 - h
    hw = h.copy()

        j = rng.randint(n)
        if i == j:

        m1j = (s1j >= 0) & (s1j < n)

        Gi = np.zeros(2 * n - 1)
        Gi[m1i] = Ah[s1i[m1i]]
        Gi[m2i] -= hw[s2i[m2i]]
        Gj = np.zeros(2 * n - 1)
        Gj[m1j] = Ah[s1j[m1j]]
        Gj[m2j] -= hw[s2j[m2j]]
