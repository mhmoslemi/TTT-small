import numpy as np

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_c5(h):
    return float(np.max(get_corr(h))) * 2.0 / len(h)

def grad_active(h, active):
    n = len(h)
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += 1.0 - h[M:]
            g[M:] -= h[:n - M]
    return g / len(active)

def softmax_obj(h, alpha, lam):
    c = get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (h.sum() - len(h) / 2.0) ** 2

def softmax_grad(h, alpha, lam):
    c = get_corr(h)
    n = len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
