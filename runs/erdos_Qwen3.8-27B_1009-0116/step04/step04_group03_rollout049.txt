        if i == j: continue
        d = d_scale * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if not (0 <= ni <= 1 and 0 <= nj <= 1): continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n <= c5 or rng.rand() < np.exp(-(c5n - c5) / max(T, 1e-10)):
            c5 = c5n
            if c5n < bc5: bc5, bh = c5n, h.copy()
        else:
            h[i], h[j] = ni-d, nj+d
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def tl(): return budget_s - (time.time() - t0)

    hp = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    npar = len(hp)
    best_h, best_c5, best_n = hp.copy(), _eval(hp), npar

    ns = sorted(set(x for x in [50,60,70,80,90,100,120,140,npar,160,180] if 40<=x<=200))
    inits = {}
    for n in ns:
        cands = []
        if n != npar: cands.append(_interp(hp, npar, n))
        else: cands.append(hp.copy())
        for f in [1,2,3,4,5,6]:
            for a in [0.15, 0.35, 0.55]:
                x = np.arange(n)
                cands.append(_repair(0.5 + a*np.sin(2*np.pi*f*x/n), n))
        x = np.arange(n)
        cands.append(_repair(0.5+0.3*np.sin(2*np.pi*x/n)+0.2*np.cos(4*np.pi*x/n), n))
        cands.append(_repair(0.5+0.25*np.sin(2*np.pi*x/n)*np.sin(3*np.pi*x/n), n))
        cands.append(_repair(rng.rand(n), n))
        bc, bh = 1.0, None
        for c in cands:
            v = _eval(c)
            if v < bc: bc, bh = v, c
        inits[n] = bh

    # Phase 1: SLSQP multi-alpha (55% budget)
    t1 = t0 + budget_s * 0.55
    for n in ns:
        if time.time() > t1: break
        h0 = inits[n].copy()
        c0 = _eval(h0)
        if c0 > best_c5 + 0.04: continue
        for a in [50, 200, 600, 2000, 8000, 30000, 100000]:
            if time.time() > t1: break
            mi = 350 if n <= 80 else (200 if n <= 120 else 120)
            hn = _slsqp(h0, n, a, mi=mi)
            cn = _eval(hn)
            if cn < c0 - 1e-14: h0, c0 = hn, cn
        if c0 < best_c5:
            best_c5, best_h, best_n = c0, h0.copy(), n

    # Phase 2: Simulated annealing for exploration (15% budget)
    if tl() > budget_s * 0.15:
        t2 = t0 + budget_s * 0.70
        h2, c2 = _sim_anneal(best_h.copy(), best_n, rng, t2, T0=0.02)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), best_n

    # Phase 3: SLSQP refinement after SA (10% budget)
    if tl() > budget_s * 0.10:
        t3 = t0 + budget_s * 0.80
        h3 = _slsqp(best_h.copy(), best_n, 50000, mi=400)
        c3 = _eval(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Long local search (20% budget)
    if tl() > budget_s * 0.18:
        t4 = t0 + budget_s * 0.97
        h4, c4 = _local_search(best_h.copy(), best_n, rng, t4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()

    # Phase 5: Final SLSQP polish
    if tl() > 10:
        h5 = _slsqp(best_h.copy(), best_n, 300000, mi=250, ft=1e-22)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Final tiny perturbations
    if tl() > 3:
        ht = best_h.copy()
        ct = _eval(ht)
        for _ in range(30000):
            if time.time() > t0 + budget_s - 1: break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j: continue
            d = rng.choice([1,-1]) * 1e-5 * rng.choice([1,2,5])
            ni, nj = ht[i]+d, ht[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                ht[i], ht[j] = ni, nj
                cn = _eval(ht)
                if cn < ct - 1e-16:
                    ct = cn
                    if ct < best_c5: best_c5, best_h = ct, ht.copy()
                else:
                    ht[i], ht[j] = ni-d, nj+d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
