def _repair(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)

    if abs(diff) > 1e-10:

import numpy as np
from scipy.optimize import minimize
import time

_gn = 140

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - hp))
            hp[idx] = min(1.0, hp[idx] + diff)
        else:
            idx = int(np.argmin(hp))
            hp[idx] = max(0.0, hp[idx] + diff)
    return hp

def _pgrad(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.maximum(c, 1e-25)
    sp = np.sum(c ** p)
    if sp < 1e-25:
        return np.zeros(n)
    pref = sp ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = pref * c[k] ** (p - 1)
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += w * (1.0 - h[l2:])
            g[l2:] -= w * h[:n - l2]
    return g

def _pgd_backtrack(h, n, p, lr, t_end):
    h = h.copy()
    best = h.copy()
    bc = _c5(h, n)
    while time.time() < t_end:
        g = _pgrad(h, n, p)
        fnorm = np.sum(g * g)
        if fnorm < 1e-20:
            break
        step = lr
        y = h
        for _ in range(20):
            y_new = _repair(y - step * g, n)
            if _c5(y_new, n) <= bc + 1e-14:
                break
            step *= 0.5
            y = y_new
        h = y_new
        c = _c5(h, n)
        if c < bc:
            bc = c
            best = h.copy()
    return best, bc

def _nesterov(h, n, p, lr, t_end):
    h = h.copy()
    y = h.copy()
    t = 1.0
    best = h.copy()
    bc = _c5(h, n)
    while time.time() < t_end:
        tn = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (1.0 - t) / tn
        y = h + beta * (h - y)
        g = _pgrad(y, n, p)
        h_new = _repair(y - lr * g, n)
        c = _c5(h_new, n)
        if c < bc:
            bc = c
            best = h_new.copy()
        h = h_new
        t = tn
    return best, bc

def _sa_search(h, n, t_end, rng):
    h = h.copy()
    best = h.copy()
    bc = _c5(h, n)
    cc = bc
    t_start = time.time()
    dur = max(t_end - t_start, 1.0)
    ds = [0.001, 0.003, 0.01, 0.03, 0.08, 0.15,
          -0.001, -0.003, -0.01, -0.03, -0.08, -0.15]
    nd = len(ds)
    while time.time() < t_end:
        frac = min((time.time() - t_start) / dur, 1.0)
        T = 0.005 * (0.005) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = ds[rng.randint(nd)]
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        hc = _c5(h, n)
        if hc <= cc:
            cc = hc
            if hc < bc:
                bc = hc
                best = h.copy()
        else:
            if rng.random() < np.exp(-(hc - cc) / max(T, 1e-10)):
                cc = hc
            else:
                h[i], h[j] = hi, hj
    return best, bc

def _slsqp_obj(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c = np.maximum(c, 1e-25)
    return float((np.sum(c ** p)) ** (1.0 / p))

def _slsqp_grad(x, n, p):
    return _pgrad(x, n, p)

def _slsqp_con(x):
    return np.sum(x) - _gn / 2.0

def _slsqp_conjac(x):
    return np.ones(_gn)

def _do_slsqp(h, n, p, t_end, rng, n_starts=3):
    global _gn
    _gn = n
    best = h.copy()
    bc = _c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_conjac}]
    for r in range(n_starts):
        if time.time() > t_end:
            break
        if r == 0:
            xi = best.copy()
        else:
            xi = _repair(best + 0.03 * rng.randn(n), n)
        try:
            res = minimize(_slsqp_obj, xi, args=(n, p),
                           jac=_slsqp_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, n)
            if c5 < bc:
                bc = c5
                best = hc
        except Exception:
            pass
    return best, bc

def _interp_to(h, n_src, n_dst):
    x_o = np.linspace(0, 2.0, n_src, endpoint=False)
    x_n = np.linspace(0, 2.0, n_dst, endpoint=False)
    return np.interp(x_n, x_o, h, period=2.0)

        elif c5_w < best_c5 + 0.003 and _el() < budget_s * 0.70:
