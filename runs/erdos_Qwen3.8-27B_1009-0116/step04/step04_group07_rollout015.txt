def slsqp_soft(h, n, alpha, maxiter=2000):
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)

        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, n, alpha, maxiter=1500):
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -450, 450)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def subgrad_step(h, n, step_size):
    """One subgradient step on the true max objective with projection."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = max_val * 1e-4 + 1e-14
    active = np.where(np.abs(corr - max_val) < tol)[0]
    grad = np.zeros(n)
    for m in active:
        d = m - (n - 1)
        ad = abs(d)
        if ad == 0:
            grad[:] += 1.0 - 2.0 * h
        elif ad < n:
            if d > 0:
                grad[:n - ad] += (1.0 - h[ad:])
                grad[ad:] += -h[:n - ad]
            else:
                grad[ad:] += (1.0 - h[:-ad])
                grad[:n - ad] += -h[ad:]
    grad /= len(active)
    h_new = h - step_size * grad
    return repair(h_new, n)

def cyclic_cd(h, n, t_end, rng, n_passes=200):
    """Cyclic coordinate descent: sweep through all positions, optimize each given others."""
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    for p in range(n_passes):
        if time.time() > t_end:
            break
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            if time.time() > t_end:
                break
            hi = h[i]
            for j in range(n):
                if j == i:
                    continue
                hj = h[j]
                if abs(hi - hj) < 1e-4:
                    continue
                # Move mass from higher to lower (or vice versa) to reduce peak
                direction = -1 if hi > hj else 1
                for s in steps:
                    delta = direction * s
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        hi = h[i]
                        break
                    h[i], h[j] = hi, hj
            h[i] = hi
    return h, best

def random_cd(h, n, t_end, rng, n_trials=200000):
    """Random pair coordinate descent with multiple step sizes."""
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0003]
    trials = 0
    stagnant = 0
    while trials < n_trials and time.time() < t_end:
        trials += 1
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        hi, hj = h[i], h[j]
        improved = False
        for s in steps:
            for sign in (1, -1):
                delta = sign * s
                ni, nj = hi + delta, hj - delta
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, n)
                if c5 < best - 1e-15:
                    best = c5
                    hi, hj = ni, nj
                    improved = True
                    break
                h[i], h[j] = hi, hj
            if improved:
                break
        if not improved:
            stagnant += 1
            if stagnant > 5000:
                break
        else:
            stagnant = 0
    h[i], h[j] = hi, hj
    return h, best

def multi_n_trial(h_best, c5_best, n_best, n_target, t_end, rng):
    """Resample to n_target, optimize, return result."""
    if n_target == n_best:
        return h_best, c5_best
    x_old = np.linspace(0.0, 1.0, n_best)
    x_new = np.linspace(0.0, 1.0, n_target)
    h_seed = np.interp(x_new, x_old, h_best)
    h_seed = repair(h_seed, n_target)
    # SLSQP
    h_opt = slsqp_opt(h_seed, n_target, 200000, 800)
    c5_opt = compute_c5(h_opt, n_target)
    # Coordinate descent
    h_cd, c5_cd = random_cd(h_opt, n_target, t_end, rng, 30000)
    if c5_cd < c5_opt:
        h_opt, c5_opt = h_cd, c5_cd
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):

    if init_h is not None:

    def elapsed():

    # Phase 1: SLSQP with escalating alpha (30% time)

        h_new = slsqp_opt(h_work.copy(), best_n, alpha, 1200)

        elif c5_new < compute_c5(h_work, best_n) - 1e-14:
