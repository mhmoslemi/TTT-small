import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, s, n):
    """Gradient of C(s) = sum h[i+s]*(1-h[i]) w.r.t. h"""
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n-sp]
    return g

def _subgrad_step(h, n, lr, top_m=3):
    c = _corr(h)
    idx = np.argsort(c[::-1])  # top m from the end... 
    # Actually, let's just find top m values
    top_idx = np.argpartition(c, -top_m)[-top_m:]
    weights = np.zeros(len(c))
    for i, idx_i in enumerate(top_idx):
        w = (i + 1)  # weight the max more
        weights[idx_i] = w
    # Normalize
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for k in range(len(c)):
        if weights[k] == 0:
            continue
        s = k - (n - 1)
        g += weights[k] * _grad_lag(h, s, n)
    
    h_new = h - lr * g
    return h_new

def _subgrad_opt(h, n, t_end, max_rounds=500):
    h = h.copy()
    c5 = _eval(h)
    lr = 0.02
    stuck = 0
    
    for it in range(max_rounds):

        h_raw = _subgrad_step(h, n, lr, top_m=5)

    s = np.clip(alpha * (c - m), -500, 500)

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sm_valgrad, h, jac=True, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_polish(h, n, t_end, max_iters=20000):

    c5 = _eval(h)
    lr = 0.05
    stuck = 0
