        return h, compute_c5(h, dx)

def local_search(h, n, dx, rng, deadline, fine):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    if fine:
        scales = [0.0008, 0.0004, 0.0002, 0.0001, 0.00004]
        per_scale = 50000
    else:
        scales = [0.015, 0.006, 0.0025, 0.001, 0.0004, 0.00015]
        per_scale = 35000
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.003:
            break
    return h, best_c5

def block_local(h, n, dx, rng, deadline, n_iters):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 5)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (0.0 <= h[i+di] + eps <= 1.0 and 0.0 <= h[j+di] - eps <= 1.0):
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-14:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def simulated_anneal(h_init, n, dx, rng, duration):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    current_c5 = best_c5
    target_iters = 12000000
    T0 = 4.0e-5
    T1 = 1.5e-9
    t_deadline = time.time() + duration
    step_scale = 1.0
    accept_count = 0
    total_count = 0
    acc_window = 20000

    for it in range(40000000):
        if time.time() >= t_deadline:
            break
        frac = min(1.0, it / float(target_iters))
        T = T1 + (T0 - T1) * (1.0 - frac) ** 2.5
        adaptive_scale = 0.4 + 0.6 * (1.0 - frac)

        h_old = h.copy()
        r = rng.random()
        made_change = False

        if r < 0.50:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = rng.normal(0, (0.010 + 0.008 * adaptive_scale) * step_scale)
                if 0.0 <= h[i] + d <= 1.0 and 0.0 <= h[j] - d <= 1.0:
                    h[i] += d
                    h[j] -= d
                    made_change = True
        elif r < 0.72:
            bs = rng.randint(2, 5)
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) >= bs:
                d = rng.normal(0, (0.007 + 0.005 * adaptive_scale) * step_scale)
                bi = h[i:i+bs] + d
                bj = h[j:j+bs] - d
                if np.all(bi >= 0) and np.all(bi <= 1) and np.all(bj >= 0) and np.all(bj <= 1):
                    h[i:i+bs] = bi
                    h[j:j+bs] = bj
                    made_change = True
        elif r < 0.88:
            npp = rng.randint(2, 4)
            ok = True
            for _ in range(npp):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = rng.normal(0, 0.005 * step_scale)
                if not (0.0 <= h[i]+d <= 1.0 and 0.0 <= h[j]-d <= 1.0):
                    ok = False
                    break
                h[i] += d
                h[j] -= d
            if not ok:
                h = h_old
                continue
            made_change = True
        else:
            bs = rng.randint(3, 9)
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) >= bs:
                h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
                made_change = True

        if not made_change:
            continue

        total_count += 1
        new_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        delta = new_c5 - current_c5

        if delta <= 0:
            current_c5 = new_c5
            accept_count += 1
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if T > 1e-14 and np.log(rng.uniform(1e-300, 1.0)) > -delta / T:
                current_c5 = new_c5
                accept_count += 1
            else:
                h = h_old

        if total_count % acc_window == 0:
            acc_rate = accept_count / float(acc_window)
            if acc_rate > 0.75:
                step_scale = min(step_scale * 1.3, 3.0)
            elif acc_rate < 0.15:
                step_scale = max(step_scale * 0.7, 0.1)
            accept_count = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    h0 = repair(h0.copy(), n)
    c5_0 = compute_c5(h0, dx)
    best_h, best_c5 = h0.copy(), c5_0

    # Phase 1: Quick L-BFGS (40s)
    t_end = t_start + 40
    hc = h0.copy()
    c5c = c5_0
    for alpha in [200, 1000, 5000]:
        if time.time() >= t_end:
            break
        for pen in [500, 5000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_opt(hc, n, alpha, n/2.0, pen, 150)
            if cc < c5c:
                hc, c5c = hh, cc
    if c5c < best_c5:
        best_h, best_c5 = hc.copy(), c5c
    work_h = hc.copy()

    # Phase 2: Simulated annealing (main workhorse, 700s)
    if remaining() > 700:
        sa_dur = min(700, remaining() - 200)
        work_h, c5_sa = simulated_anneal(work_h, n, dx, rng, sa_dur)
        if c5_sa < best_c5:
            best_h, best_c5 = work_h.copy(), c5_sa

    # Phase 3: Short second annealing pass at lower temp (100s)
    if remaining() > 150:
        work_h = best_h.copy()
        sa_dur2 = min(100, remaining() - 80)
        work_h, c5_sa2 = simulated_anneal(work_h, n, dx, rng, sa_dur2)
        if c5_sa2 < best_c5:
            best_h, best_c5 = work_h.copy(), c5_sa2

    # Phase 4: Block local search (60s)
    if remaining() > 100:
        dur = min(60, remaining() - 50)
        work_h, c5_bl = block_local(best_h, n, dx, rng, time.time() + dur, 8000)
        if c5_bl < best_c5:
            best_h, best_c5 = work_h.copy(), c5_bl

    # Phase 5: Fine local search (120s)
    if remaining() > 130:
        dur = min(120, remaining() - 50)
        work_h, c5_ls = local_search(best_h, n, dx, rng, time.time() + dur, fine=False)
        if c5_ls < best_c5:
            best_h, best_c5 = work_h.copy(), c5_ls

    # Phase 6: SLSQP polish (40s)
    if remaining() > 60:
        hh, cc = slsqp_opt(best_h, n, 5000, n/2.0, 300)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 7: Ultra-fine local search (40s)
    if remaining() > 50:
        dur = min(40, remaining() - 10)
        work_h, c5_f = local_search(best_h, n, dx, rng, time.time() + dur, fine=True)
        if c5_f < best_c5:
            best_h, best_c5 = work_h.copy(), c5_f

    # Phase 8: Final L-BFGS high-alpha (20s)
    if remaining() > 25:
        hh, cc = lbfgs_opt(best_h, n, 50000, n/2.0, 5000, 200)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
