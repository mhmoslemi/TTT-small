                    for d1 in [0.002, -0.002]:
                        for d2 in [0.002, -0.002]:
                            ni, nj, nj2 = h[i] + d1, h[j] + d2, h[j2] - d1 - d2
                            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nj2 <= 1:
                                h[i], h[j], h[j2] = ni, nj, nj2
                                c = _c5(h, dx)
                                if c < c5c - 1e-16:
                                    c5c = c
                                    moved = True
                                    if c5c < best_c5:
                                        best_c5, best_h = c5c, h.copy()
                                    break
                                else:
                                    h[i], h[j], h[j2] = h[i] - d1, h[j] - d2, h[j2] + d1 + d2
                        if moved:
                            break
            if not moved:
                break

    # Phase 4: Ultra-fine random polish
    h = best_h.copy()
    c5c = _c5(h, dx)
    deltas = np.array([0.0002, 0.0005, 0.001, 0.002,
                       -0.0002, -0.0005, -0.001, -0.002])

    while time.time() - t0 < budget_s - 2:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c - 1e-16:
                c5c = c
                best_c5 = c5c
                best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
