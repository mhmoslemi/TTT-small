    # Project to binary solution with exact sum
    n = len(h)
    target = n // 2
    # Sort by h value, keep top n/2 as 1
    idx = np.argsort(h)[::-1][:target]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _slsqp_opt(h, method='smooth', alpha=50000, maxiter=2000):
    n = len(h)
    
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - h[sh:])  # BUG: should use x not h
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _worst_shift_descent(h, n_rounds):
    nc = len(h)
    dx = 2.0 / nc
    c5 = _eval(h)
    for _ in range(n_rounds):
        c = _corr(h)
        k_w = c.argmax()
        sh = (nc-1) - k_w
        g = np.zeros(nc)
        if sh >= 0:
            g[:nc-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:nc-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:nc-sp])
            g[:nc-sp] -= h[sp:]
        g *= dx
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj_sum(h - step*g, nc*0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-14:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_max_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_max_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-18: continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh > 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        elif sh < 0:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
        else:
            g += wk * (1.0 - 2.0*h)
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _worst_descent(h, rounds):

    dx = 2.0/nc
    c5 = _eval(h)
    for _ in range(rounds):
        c = np.correlate(h, 1.0-h, mode='full')
        k_w = c.argmax()
        sh = (nc-1) - k_w
        g = np.zeros(nc)

            g[:nc-sh] += (1.0 - h[sh:])
            g[sh:] -= h[:nc-sh]
        elif sh < 0:
            sp = -sh
            g[sp:] += (1.0 - h[:nc-sp])
            g[:nc-sp] -= h[sp:]
        else:
            g = 1.0 - 2.0*h
        g *= dx

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-18:
            continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh > 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        elif sh < 0:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
        else:
            g += wk * (1.0 - 2.0*h)
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    dx = 2.0 / nc
    c5 = _eval(h)
    for _ in range(rounds):

        top_k = np.argsort(c)[-top_m:]

            if sh > 0:

                g[sh:] -= h[:nc-sh]
            elif sh < 0:

def _periodic_start(n, p, seed):
    rng = np.random.RandomState(seed)
    n_rep = n // p
    base = np.full(p, 0.5)
    base[:p//2] = 1.0
    base[p//2:] = 0.0
    rng.shuffle(base)

def _fine_local(h, n_iter, seed):

    rng = np.random.RandomState(seed)
    c5 = _eval(h)
    for scale in [0.003, 0.001, 0.0003, 0.00005]:
        per_scale = max(1000, n_iter // 4)
        for _ in range(per_scale):

            if i == j:

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP sweep at original resolution
    for alpha in [2000, 8000, 30000, 100000, 300000]:
        if _el() > 120:
            break
        hn = _slsqp(best_h, alpha, 2500)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Multi-worst descent
    if _el() < budget_s - 500:
        hn, c5 = _multi_worst_descent(best_h, 3, 500)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
        hn, c5 = _multi_worst_descent(best_h, 5, 400)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
        hn = _slsqp(best_h, 200000, 2000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 3: Try periodic and binary starts
    nc = best_n
    candidates = []
    for p in [7, 14, 20, 35]:
        if nc % p == 0:
            candidates.append(_periodic_start(nc, p, seed + p))
    candidates.append(_binary_start(nc, seed + 99))
    candidates.append(_repair(1.0 - best_h, nc))  # complement

    if _el() < budget_s - 400:
        for ci, cand in enumerate(candidates):
            if _el() > budget_s - 400:
                break
            hn = _slsqp(cand, 50000, 1500)
            hn, c5 = _multi_worst_descent(hn, 3, 200)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), nc

    # Phase 4: Upsample to 2x
    if _el() < budget_s - 500:
        n1 = n0 * 2
        hh = _repair(np.repeat(best_h, 2), n1)
        for alpha in [20000, 100000, 400000]:
            if _el() > budget_s - 250:
                break
            hn = _slsqp(hh, alpha, 1200)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), n1
        if _el() < budget_s - 180:
            hn, c5 = _multi_worst_descent(best_h, 5, 400)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), len(hn)
            hn = _slsqp(best_h, 500000, 2000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), len(hn)

    # Phase 5: Multi-start refinement at best resolution
    nc = best_n
    if _el() < budget_s - 200:
        for r in range(12):
            if _el() > budget_s - 200:
                break
            if r % 4 == 0:
                hp = _repair(best_h + 0.012*rng.randn(nc), nc)
            elif r % 4 == 1:
                hp = _repair(best_h, nc)
                idx = rng.choice(nc, size=nc//4, replace=False)
                hp[idx] += 0.05*rng.randn(len(idx))
                hp = _repair(hp, nc)
            elif r % 4 == 2:
                hp = _binary_start(nc, seed + 500 + r)
                hp = _repair(hp + 0.3*best_h, nc)
            else:
                hp = best_h.copy()
                f = rng.randint(3, nc//4 + 1)
                ph = 2*np.pi*rng.random()
                hp += 0.008*np.sin(2*np.pi*f*np.arange(nc)/nc + ph)
                hp = _repair(hp, nc)
            hn = _slsqp(hp, 150000, 800)
            hn, c5 = _multi_worst_descent(hn, 3, 100)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), nc

    # Phase 6: Final polish SLSQP
    if _el() < budget_s - 80:
        hn = _slsqp(best_h, 500000, 4000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 7: Multi-worst descent final
    if _el() < budget_s - 30:
        hn, c5 = _multi_worst_descent(best_h, 5, 800)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 8: Fine local search
    if _el() < budget_s - 3:
        hn, c5 = _fine_local(best_h, 40000, seed + 7777)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
        hn = _slsqp(hn, 800000, 2000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()
        hn, c5 = _fine_local(hn, 20000, seed + 8888)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
