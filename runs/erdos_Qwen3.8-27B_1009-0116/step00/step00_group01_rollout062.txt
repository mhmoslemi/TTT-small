import numpy as np

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def project(h, n):
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        err = n / 2.0 - h.sum()
        if abs(err) < 1e-14:
            break
        h += err / n
    # Final exact fix
    err = n / 2.0 - h.sum()
    if abs(err) > 1e-12:
        idx = np.argmax(h * (1 - h))  # most "interior" element
        h[idx] += err
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_gradient(h, dx, eps=0.02):
    n = len(h)
    one_h = 1 - h
    corr = np.correlate(h, one_h, mode='full') * dx
    c_max = corr.max()
    c_shifted = corr - c_max
    weights = np.exp(c_shifted / eps)
    w_sum = weights.sum()
    if w_sum == 0:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
        w_sum = 1.0
    weights /= w_sum
    
    grad = np.zeros(n)
    t = np.arange(n)
    for m in range(len(corr)):
        if weights[m] < 1e-12:
            continue
        k = m - (n - 1)
        v1 = (t - k >= 0) & (t - k < n)
        v2 = (t + k >= 0) & (t + k < n)
        grad[v1] += weights[m] * dx * (1 - h[t[v1] - k])
        grad[v2] -= weights[m] * dx * h[t[v2] + k]
    return grad, c_max + eps * np.log(w_sum)

        actual = _compute_c5(h)
        if actual < best_c5:
            best_c5 = actual
            best_h = h.copy()

        if step % 400 == 0:
            eps_cur = max(eps_cur * 0.7, 0.0005)
            lr = max(lr * 0.85, 0.0005)

    return _project(best_h, n), _compute_c5(best_h)


def _subgrad_polish(h, n, t_limit):
    t0 = _time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    lr = 0.002
    while _time.time() - t0 < t_limit:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = corr.max()
        active = np.where(corr >= cmax - 1e-9)[0]
        grad = np.zeros(n)
        nm1 = n - 1
        t_arr = np.arange(n)
        for m in active:
            k = m - nm1
            lo1 = max(0, k)
            hi1 = min(n, n + k)
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if hi1 > lo1:
                grad[lo1:hi1] += dx * (1.0 - h[t_arr[lo1:hi1] - k])
            if hi2 > lo2:
                grad[lo2:hi2] -= dx * h[t_arr[lo2:hi2] + k]
        grad /= float(len(active))
        h_try = _project(h - lr * grad, n)
        c5_try = _compute_c5(h_try)
        if c5_try < cmax:
            h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
            lr = min(lr * 1.02, 0.01)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = _time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
        h0 = _project(h0, n0)
        best_h, best_n = h0.copy(), n0
        best_c5 = _compute_c5(h0)
    else:
        best_n = 47
        best_h = np.zeros(best_n)
        best_h[0::2] = 1.0
        best_h = _project(best_h, best_n)
        best_c5 = _compute_c5(best_h)

    remaining = budget_s - (_time.time() - t_start)
    if remaining > 40:
        alloc = min(remaining * 0.25, 200.0)
        h_opt, c5 = _optimize(best_h.copy(), best_n, 2.0 / best_n, alloc)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt
        remaining = budget_s - (_time.time() - t_start)
        if remaining > 10:
            h_pol, c5_pol = _subgrad_polish(best_h, best_n, min(remaining * 0.1, 20.0))
            if c5_pol < best_c5:
                best_c5, best_h = c5_pol, h_pol

    for target_n in [94, 188, 376, 752]:
        remaining = budget_s - (_time.time() - t_start)
        if remaining < 50:
            break
        h_up = _project(_upsample(best_h, target_n), target_n)
        dx_new = 2.0 / target_n
        alloc = min(remaining * 0.35, 200.0)
        if alloc < 25:
            break
        h_opt, c5 = _optimize(h_up, target_n, dx_new, alloc)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, target_n
        remaining = budget_s - (_time.time() - t_start)
        if remaining > 15 and best_n == target_n:
            h_pol, c5_pol = _subgrad_polish(best_h, best_n, min(remaining * 0.15, 25.0))
            if c5_pol < best_c5:
                best_c5, best_h = c5_pol, h_pol

    remaining = budget_s - (_time.time() - t_start)
    if remaining > 25 and best_n > 0:
        rng = np.random.RandomState(seed)
        for _ in range(4):
            remaining = budget_s - (_time.time() - t_start)
            if remaining < 20:
                break
            h_pert = _project(best_h + rng.normal(0.0, 0.02, best_n), best_n)
            h_opt, c5 = _optimize(h_pert, best_n, 2.0 / best_n, min(remaining * 0.2, 45.0))
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt

    best_h = _project(best_h, best_n)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, best_n
