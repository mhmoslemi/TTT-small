    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    corr_scaled = corr * tau
    max_val = np.max(corr_scaled)
    m = max_val + np.log(np.sum(np.exp(corr_scaled - max_val)))

def smooth_max_c5_grad(h, tau):

    w = np.exp(corr_scaled - max_val)
