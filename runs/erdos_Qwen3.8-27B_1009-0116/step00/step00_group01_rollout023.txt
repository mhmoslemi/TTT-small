    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

    return compute_c5(x)

    return np.sum(x) - _N / 2.0

    return np.ones(_N)

        else:
            m = h > 1e-9
            c = int(m.sum())
