import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return dx * (m + temp * np.log(np.mean(np.exp((corr - m) / temp))))

def _slsqp_optimize(h0, n, temp, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_c5(x, temp)
    def con(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except Exception:
        return h0

def run(seed=42, budget_s=1100):
    t0 = time.time()
    n = len(initial_h_values)
    best_h = np.array(initial_h_values, dtype=float).copy()
    best_c5 = _c5(best_h)
    rng = np.random.RandomState(seed)

    for temp in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - t0 > budget_s - 60:
            break
        best_h = _slsqp_optimize(best_h, n, temp, 300)
        best_h = _repair(best_h, n)
        c5 = _c5(best_h)

    s = np.sum(h)
    if s > 1e-14:
