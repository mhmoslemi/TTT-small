                if res.fun < best_o:
                    best_v, best_o = res.x, res.fun
            except Exception:
                pass
            h[i] = best_v
            if best_v != old_v:
                improved = True
                old_obj = best_o
        if not improved:
            break
    diff = ts - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h, compute_c5(h, dx)

def nm_optimize(h0, dx, ts, max_iter=30000):
    def obj(v):
        vc = np.clip(v, 0, 1)
        return compute_c5(vc, dx) + 50.0 * (np.sum(vc) - ts) ** 2
    res = minimize(obj, h0, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-12})
    return np.clip(res.x, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h, best_c5 = None, float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float)
        c5_init = compute_c5(h_init, 2.0 / len(h_init))
        best_h, best_c5 = h_init.copy(), c5_init
    except NameError:
        n = 48
        h_init = np.zeros(n)
        phi = (1.0 + 5.0 ** 0.5) / 2.0
        pos = sorted(int((k * phi) % 48) for k in range(24))
        h_init[pos] = 1.0
        best_h, best_c5 = h_init.copy(), compute_c5(h_init, 2.0 / n)

    def elapsed():
        return time.time() - start_time

    def refine(h_in):
        nonlocal best_c5, best_h
        n = len(h_in)
        dx = 2.0 / n
        ts = n / 2.0
        # NM
        h1 = nm_optimize(h_in, dx, ts, max_iter=30000)
        c1 = compute_c5(h1, dx)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        # CD
        h2, c2 = coord_descent(h1, dx, ts, max_rounds=300)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        # NM again
        if elapsed() < budget_s * 0.9:
            h3 = nm_optimize(h2, dx, ts, max_iter=50000)
            c3 = compute_c5(h3, dx)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
            h4, c4 = coord_descent(h3, dx, ts, max_rounds=200)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()

    # Phase 1: Refine initial
    refine(best_h)

    # Phase 2: Try different n values (prefer small)
    for n_try in [48, 47, 49, 50, 51, 54, 63, 64, 96, 99, 100, 101, 128, 129]:
        if elapsed() > budget_s * 0.82:
            break
        if n_try == len(best_h):
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_start = np.interp(np.linspace(0, 1, n_try),
                            np.linspace(0, 1, len(best_h)), best_h)
        h_start = np.clip(h_start, 0, 1)
        # NM + CD
        dx = 2.0 / n_try
        ts = n_try / 2.0
        h_r = nm_optimize(h_start, dx, ts, max_iter=20000)
        c_r = compute_c5(h_r, dx)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()
        h_cd, c_cd = coord_descent(h_r, dx, ts, max_rounds=200)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 3: Random restarts at best n
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    ts_best = n_best / 2.0
    for _ in range(30):
        if elapsed() > budget_s * 0.9:
            break
        h_rand = np.zeros(n_best)
        perm = np.random.permutation(n_best)
        h_rand[perm[:int(ts_best)]] = 1.0
        if ts_best % 1 > 0.01:
            h_rand[perm[int(ts_best)]] = ts_best % 1
        h_r, c_r = coord_descent(h_rand, dx_best, ts_best, max_rounds=150)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()
            # Quick NM from this
            if elapsed() < budget_s * 0.9:
                h_nm = nm_optimize(h_r, dx_best, ts_best, max_iter=20000)
                c_nm = compute_c5(h_nm, dx_best)
                if c_nm < best_c5:
                    best_c5, best_h = c_nm, h_nm.copy()

    # Final polish
    if elapsed() < budget_s * 0.95:
        refine(best_h)

    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
