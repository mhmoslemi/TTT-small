    # Find worst lag
    k_star = np.argmax(corr)
    # k_star corresponds to lag = k_star - (n-1)
    lag = k_star - (n - 1)

    grad = np.zeros(n)
    if lag >= 0:
        # corr at lag: sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[j]: (1-h[j+lag]) for j in [0, n-1-lag]
        #           -h[j-lag] for j in [lag, n-1]
        for j in range(n - lag):

            grad[j] -= h[j - lag]

        l = -lag
        for j in range(n - l):
            grad[j] += (1 - h[j + l])

            grad[j] -= h[j - l]

    h_new = np.clip(h_new, 0, 1)

import numpy as np
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            a = min(diff, room[i])
            h[i] += a
            diff -= a
            if diff <= 1e-12:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            a = min(-diff, h[i])
            h[i] -= a
            diff += a
            if diff >= -1e-12:
                break
    return h

def to_binary(h, n):
    out = np.zeros(n)
    idx = np.argsort(-h)
    out[idx[:n // 2]] = 1.0
    return out

def binary_search(h_init, time_limit=30.0, seed=42):
    n = len(h_init)
    h = to_binary(h_init, n).astype(float)
    rng = np.random.RandomState(seed)
    best = evaluate_c5(h)
    best_h = h.copy()
    stalls = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c = evaluate_c5(h)
        if c <= best + 1e-15:
            best = c
            best_h = h.copy()
            stalls = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stalls += 1
        if stalls > 2000:
            break
    return best_h, best

    best = evaluate_c5(h)
    best_h = h.copy()
    stalls = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        n_flips = rng.choice([1, 2, 2, 3])
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < n_flips or len(zeros) < n_flips:
            break
        flip_o = rng.choice(ones, size=n_flips, replace=False)
        flip_z = rng.choice(zeros, size=n_flips, replace=False)
        for idx in flip_o:
            h[idx] = 0.0
        for idx in flip_z:
            h[idx] = 1.0
        c = evaluate_c5(h)
        if c <= best + 1e-15:
            best = c
            best_h = h.copy()
            stalls = 0
        else:
            for idx in flip_o:
                h[idx] = 1.0
            for idx in flip_z:
                h[idx] = 0.0
            stalls += 1
        if stalls > 1500:
            break
    return best_h, best

def continuous_search(h_init, time_limit=20.0, seed=42):
    n = len(h_init)
    h = project_feasible(h_init.copy(), n)
    rng = np.random.RandomState(seed)
    best = evaluate_c5(h)
    best_h = h.copy()
    step = 0.08
    stalls = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        h_new = h.copy()
        for _ in range(rng.randint(1, 4)):
            j = rng.randint(n)
            h_new[j] += rng.uniform(-step, step)
        h_new = project_feasible(h_new, n)
        c = evaluate_c5(h_new)
        if c <= best + 1e-15:
            best = c
            best_h = h_new.copy()
            h = h_new
            stalls = 0
        else:
            stalls += 1
        if stalls > 400:
            step *= 0.92
            stalls = 0
            if step < 0.002:
                break
    return best_h, best

def upsample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_ext = np.concatenate([h, h])
    x_ext = np.arange(2 * n_old) / n_old
    return np.interp(x_new, x_ext, h_ext)

def downsample_h(h, n_new):
    n_old = len(h)
    if n_old <= n_new:
        return upsample_h(h, n_new)
    blocks = n_old // n_new
    if blocks < 1:
        x_new = np.arange(n_new) / n_new
        x_old = np.arange(n_old) / n_old
        return np.interp(x_new, x_old, h)
    return h[:blocks * n_new].reshape(n_new, blocks).mean(axis=1)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def structured_binary(n, period, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for p in range(n // period):
        start = p * period
        positions = np.sort(rng.choice(period, size=period // 2, replace=False)) + start
        h[positions] = 1.0
    rem = n - (n // period) * period
    if rem > 0:
        s = (n // period) * period
        h[s + rng.choice(rem, size=rem // 2, replace=False)] = 1.0
    return to_binary(h, n)
