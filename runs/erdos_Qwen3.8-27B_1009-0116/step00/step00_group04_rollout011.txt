def compute_c5(h):

from scipy.optimize import minimize
import time

_g_n = 0
_g_alpha = 0.0
_g_target = 0.0

def _objective(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(_g_alpha * (corr - m)))) / _g_alpha)

def _constraint_fun(x):
    return np.sum(x) - _g_target

def _constraint_jac(x):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h_start, n, alpha, maxiter, ftol):
    global _g_n, _g_alpha, _g_target
    _g_n = n
    _g_alpha = alpha
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac},)
    try:
        res = minimize(_objective, h_start.copy(), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h_start.copy()

        hj = best[j]

        nj = hj - delta
