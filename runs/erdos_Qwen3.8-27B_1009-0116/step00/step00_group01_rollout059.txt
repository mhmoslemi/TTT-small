Actually, let me reconsider the problem. The evaluation uses:
C₅ = max(np.correlate(h, 1-h, mode="full") * dx)

Note: this is max of the ENTIRE correlation array, not just positive lags. The full correlation has 2n-1 elements. The element at position n-1 is the autocorrelation (lag 0), and elements on either side are positive and negative lags.

For our purpose, we want min of max over ALL lags.

Let me verify: np.correlate(a, v, 'full') for a=v of length n:
- Result has length 2n-1
- Result[n-1] = sum(a[i]*v[i]) = sum(a[i]²) (for self-correlation)
- Result[n-1+k] = sum valid a[i+k]*v[i] (positive lag)
- Result[n-1-k] = sum valid a[i]*v[i+k] (negative lag)

With a=h, v=1-h:
- Result[n-1] = sum(h[i]*(1-h[i])) = sum(h) - sum(h²)
- Result[n-1+k] = sum valid h[i+k]*(1-h[i])
- Result[n-1-k] = sum valid h[i]*(1-h[i+k])

So we're looking at both h*(1-h shifted) AND (1-h)*h shifted. Due to the structure, these might give different values.

Actually, by substitution, sum h[i+k]*(1-h[i]) for lag k and sum h[i]*(1-h[i+k]) for lag k are related but not identical (different ranges of summation due to boundary effects).

Anyway, the point is we take the max over ALL 2n-1 values.

import numpy as np
from scipy.optimize import minimize
import time

_dn = [0]  # mutable global for current problem size

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = h + (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(h - 0.5))
        idx = order[0]
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h

def interp_to_res(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return repair_h(h_new, n_new)

def penalty_objective(x):
    x = np.clip(x, 0.0, 1.0)
    n = _dn[0]
    c5 = compute_c5(x)
    s = np.sum(x) - n / 2.0
    return c5 + 50.0 * s * s

def coord_descent(h, n, max_rounds=50, budget_fn=None):
    """Local coordinate descent: adjust pairs to reduce max correlation."""
    best = compute_c5(h)
    for _ in range(max_rounds):
        if budget_fn is not None and budget_fn() < 10:
            break
        improved = False
        # Try transferring mass between pairs
        for i in range(0, n, 2):
            j = (i + n // 2) % n
            if i == j:
                continue
            delta = 0.05
            h_try = h.copy()
            h_try[i] = np.clip(h_try[i] - delta, 0, 1)
            h_try[j] = np.clip(h_try[j] + delta, 0, 1)
            c_try = compute_c5(h_try)
            if c_try < best - 1e-10:
                h = h_try
                best = c_try
                improved = True
            delta = -0.05
            h_try[i] = np.clip(h.copy()[i] - delta, 0, 1)
            h_try[j] = np.clip(h.copy()[j] + delta, 0, 1)
            c_try = compute_c5(h_try)
            if c_try < best - 1e-10:
                h = h_try
                best = c_try
                improved = True
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - 15 - (time.time() - t0)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 47
        h = np.zeros(n_init)
        h[::2] = 1.0
        h = repair_h(h, n_init)

    n = len(h)
    h = repair_h(h, n)
    c5 = compute_c5(h)
    best_h, best_c5, best_n = h.copy(), float(c5), n

    # Phase 1: Optimize at starting resolution
    if time_left() > 40:
        _dn[0] = n
        bounds = [(0.0, 1.0)] * n
        res = minimize(penalty_objective, h, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 300, 'ftol': 1e-14})
        h_opt = repair_h(res.x, n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < c5:
            h, c5 = h_opt, float(c5_opt)
            best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 2: Multi-resolution optimization
    resolutions = []
    cur = n
    while cur < 500 and len(resolutions) < 4:
        cur *= 2
        resolutions.append(cur)

    for rn in resolutions:
        if time_left() < 80:
            break
        h_cur = interp_to_res(best_h, best_n, rn)
        c5_cur = compute_c5(h_cur)
        if c5_cur >= best_c5:
            h_cur = best_h.copy()
            # subsample/upsample carefully
            h_cur = interp_to_res(best_h, best_n, rn)
            c5_cur = compute_c5(h_cur)

        _dn[0] = rn
        bounds_r = [(0.0, 1.0)] * rn
        max_iter = int(min(time_left() * 0.5, 600))

        res = minimize(penalty_objective, h_cur, method='L-BFGS-B',
                       bounds=bounds_r, options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = repair_h(res.x, rn)
        c5_opt = compute_c5(h_opt)
        if c5_opt < c5_cur:
            h_cur, c5_cur = h_opt, float(c5_opt)

        # Second refinement pass
        if time_left() > 80 and c5_cur < best_c5:
            max_iter2 = int(min(time_left() * 0.3, 400))
            res2 = minimize(penalty_objective, h_cur, method='L-BFGS-B',
                            bounds=bounds_r, options={'maxiter': max_iter2, 'ftol': 1e-15})
            h_opt2 = repair_h(res2.x, rn)
            c5_opt2 = compute_c5(h_opt2)

        if c5_cur < best_c5:

    # Final coordinate descent at best resolution
    if time_left() > 10:

        best_h = repair_h(best_h, best_n)
        best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), best_n
