```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_box_sum(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Simple: add diff to interior points
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    if np.sum(interior) == 0:
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        h[interior] += diff / np.sum(interior)
        h = np.clip(h, 0, 1)
    return h

def run_slsqp(h0, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    n_full = 2 * n - 1
    
    x0 = np.concatenate([h0, [compute_c5(h0) + 0.05]])
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    def corr_ineq(x):
        h = x[:n]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr * dx
    
    def corr_ineq_jac(x):
        h = x[:n]
        t = x[-1]
        J = np.zeros((n_full, n + 1))
        for k in range(n_full):
            c = k - (n - 1)
            # d corr[k] / d h[m] = (1-h[m-c]) * I[0<=m-c<n] - h[m+c] * I[0<=m+c<n]
            for m in range(n):
                jac_val = 0.0
                if 0 <= m - c < n:
                    jac_val += (1.0 - h[m - c])
                if 0 <= m + c < n:
                    jac_val -= h[m + c]
                J[k, m] = jac_val * dx
            J[k, n] = -1.0  # d/dt of (t - corr*dx) = 1, but constraint is t - corr*dx >= 0
            # Wait, constraint is corr_ineq >= 0, i.e., t - corr[k]*dx >= 0
            # d/dt = 1
            J[k, n] = 1.0
        return J
    
    eq_fun = lambda x: np.sum(x[:n]) - target_sum
    eq_jac = lambda x: np.concatenate([np.ones(n), [0.0]])
    
    constraints = [
        {'type': 'ineq', 'fun': corr_ineq, 'jac': corr_ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(-0.5, 5.0)]
    
    result = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      constraints=constraints, bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-10})
    
    h_opt = np.clip(result.x[:n], 0.0, 1.0)
    h_opt = project_to_box_sum(h_opt, target_sum)
    return h_opt, compute_c5(h_opt)
```
