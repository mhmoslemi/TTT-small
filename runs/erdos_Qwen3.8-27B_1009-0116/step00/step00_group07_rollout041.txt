        h_b[ones_idx[order]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_b == 0)[0]
        order = np.argsort(h_ref[zeros_idx])[::-1][:target - n_ones]
        h_b[zeros_idx[order]] = 1
    return h_b

def steepest_descent_binary(h, n_points, max_rounds=200, t0=None, budget_frac=1.0):
    target = n_points // 2
    for _ in range(max_rounds):
        if t0 is not None and (time.time() - t0) > budget_frac:
            break
        c5_cur = compute_c5(h, n_points)
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_c5 = c5_cur
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n_points)
                h[j] = 0.0
                h[i] = 1.0
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
    return h

def simulated_annealing_binary(h, n_points, t0, end_time, rng, T_init=0.05):
    T = T_init
    T_min = 0.002
    cool = 0.99995
    best_c5 = compute_c5(h, n_points)
    best_h = h.copy()
    target = n_points // 2
    while time.time() < end_time:
        c5_cur = compute_c5(h, n_points)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h, n_points)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / T):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cool, T_min)
    return best_h, best_c5

def subgradient_continuous(h, n_points, steps, lr_init, t0, end_time):
    target = n_points / 2.0
    lr = lr_init
    best_c5 = compute_c5(h, n_points)
    best_h = h.copy()
    for step in range(steps):
        if time.time() > end_time:
            break
        if step % 500 == 0 and step > 0:
            lr *= 0.85
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        offset = k_star - (n_points - 1)
        grad = np.zeros(n_points)
        idx1 = np.arange(n_points)
        i1 = idx1 + offset
        valid1 = (i1 >= 0) & (i1 < n_points)
        grad[valid1] += (1.0 - h[i1[valid1]])
        i2 = idx1 - offset
        valid2 = (i2 >= 0) & (i2 < n_points)
        grad[valid2] -= h[i2[valid2]]
        h = np.clip(h - lr * grad, 0.0, 1.0)
        h = project_constraint(h, target)
        c5_val = compute_c5(h, n_points)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n_pts = 80
        h0 = np.zeros(n_pts)
        for k in range(n_pts // 2):
            h0[k * 2] = 1.0
    n_points = len(h0)
    target = n_points // 2
    dx = 2.0 / n_points

    # Ensure initial is valid binary
    h_init = binary_to_target(h0, h0, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_points)

    # Phase 1: Steepest descent on binary
    h1 = h_init.copy()
    h1 = steepest_descent_binary(h1, n_points, 300, t0, (deadline - t0) * 0.35)
    c5_1 = compute_c5(h1, n_points)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multiple SA restarts
    sa_deadline = t0 + (deadline - t0) * 0.70
    n_restarts = 4
    for r in range(n_restarts):
        if time.time() > sa_deadline:
            break
        if r == 0:
            h_sa = h1.copy()
        else:
            h_sa = h_init.copy()
            # Perturb randomly
            for _ in range(10 + rng.randint(20)):
                ones = np.where(h_sa == 1)[0]
                zeros = np.where(h_sa == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    h_sa[rng.integers(len(ones)) if False else int(rng.choice(ones))] = 0.0
                    h_sa[int(rng.choice(zeros))] = 1.0
        h_sa, c5_sa = simulated_annealing_binary(h_sa, n_points, t0, sa_deadline, rng, T_init=0.08)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Steepest descent after SA
        h_sa2 = steepest_descent_binary(h_sa.copy(), n_points, 200, time.time(), sa_deadline)
        c5_sa2 = compute_c5(h_sa2, n_points)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 3: Continuous subgradient from best binary
    continuous_deadline = t0 + (deadline - t0) * 0.92
    h_cont = best_h.astype(float)
    h_cont, c5_cont = subgradient_continuous(h_cont, n_points, 8000, 0.15, time.time(), continuous_deadline)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()

    # Phase 4: Linearize and solve with scipy for tightening
    final_deadline = t0 + budget_s * 0.97
    # Try a few more subgradient runs with different starts
    for attempt in range(3):
        if time.time() > final_deadline:
            break
        h_start = best_h.copy()
        h_start, c5_start = subgradient_continuous(h_start, n_points, 3000, 0.05, time.time(), final_deadline)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()

    # Final: round to ensure clean output, verify constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if abs(s - target) > 1e-6:
        best_h = project_constraint(best_h, float(target))
    final_c5 = compute_c5(best_h, n_points)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n_points)
