            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def random_walk_ls(h, iters, rng, patience=3000):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    no_improve = 0
    for it in range(iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if new_c5 < c5 - 1e-12:
            c5 = new_c5
            no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
            if no_improve >= patience:
                break
    return h, c5

def periodic_candidate(n, period):
    p = np.zeros(period)
    ones_per_period = period // 2
    spread = period // ones_per_period
    for k in range(ones_per_period):
        p[k * spread] = 1.0
    h = np.tile(p, n // period + 1)[:n]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    n = 80
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        h0 = None

    if h0 is not None and len(h0) >= 4:
        h = make_binary(h0, n)
    else:
        n = 80
        h = random_binary(n, rng)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Exhaustive local search
    dl1 = t0 + min(180.0, budget_s * 0.2)
    h_ls, c5_ls = exhaustive_ls(h, dl1)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls.copy(), c5_ls

    # Phase 2: SA + random restarts
    dl2 = t0 + budget_s * 0.82
    sa_count = 0
    while time.time() < dl2:
        T_s = 0.004 + 0.002 * (sa_count % 3)
        h_sa, c5_sa = sa_optimize(best_h, 25000, T_s, 0.00005, rng)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa.copy(), c5_sa
        sa_count += 1
        if rng.random() < 0.3:
            h_r = random_binary(n, rng)
            h_rl, c5_rl = exhaustive_ls(h_r, time.time() + 8.0)
            if c5_rl < best_c5:
                best_h, best_c5 = h_rl.copy(), c5_rl
        if rng.random() < 0.1:
            for period in [4, 6, 8, 10, 12, 16]:
                if n % period == 0:
                    h_p = periodic_candidate(n, period)
                    h_pl, c5_pl = exhaustive_ls(h_p, time.time() + 3.0)
                    if c5_pl < best_c5:
                        best_h, best_c5 = h_pl.copy(), c5_pl

    # Phase 3: Final exhaustive polish
    dl3 = t0 + budget_s - 5.0
    if time.time() < dl3:
        h_f, c5_f = exhaustive_ls(best_h, dl3)
        if c5_f < best_c5:
            best_h, best_c5 = h_f.copy(), c5_f

    # Validate
    dx = 2.0 / n
    integral = float(np.sum(best_h)) * dx
    if abs(integral - 1.0) > 1e-6:
        ones_cnt = int(round(np.sum(best_h)))
        target = n // 2
        if ones_cnt > target:
            oi = np.where(best_h == 1)[0]
            best_h[oi[:ones_cnt - target]] = 0.0
        elif ones_cnt < target:
            zi = np.where(best_h == 0)[0]
            best_h[zi[:target - ones_cnt]] = 1.0
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
