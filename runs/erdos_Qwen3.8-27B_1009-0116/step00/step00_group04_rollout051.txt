import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vmax = np.max(corr)
    return float((vmax + temp * np.log(np.sum(np.exp((corr - vmax) / temp)))) * dx)


def eq_constraint(x):
    return np.sum(x) - len(x) / 2.0


def obj_t05(x):
    return smooth_c5(x, 0.05)


def obj_t01(x):
    return smooth_c5(x, 0.01)


def obj_t003(x):
    return smooth_c5(x, 0.003)


def obj_t001(x):

def project_constraint(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-13:

        if np.any(free):
            h[free] += diff2 / np.count_nonzero(free)
            h = np.clip(h, 0.0, 1.0)
    return h
