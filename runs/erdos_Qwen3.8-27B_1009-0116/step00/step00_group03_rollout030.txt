    h = np.clip(h, 0, 1)

        return h

    if current_sum > 0 and current_sum < len(h):

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project(h, ts):
    h = np.clip(h, 0, 1)
    d = ts - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
    else:
        r = h.copy()
    sr = r.sum()
    if sr < 1e-14:
        h[:] = ts / len(h)
        return h
    h = h + r * (d / sr)
    h = np.clip(h, 0, 1)
    d = ts - h.sum()
    for i in range(len(h)):
        if d > 0 and h[i] < 1:
            a = min(d, 1 - h[i]); h[i] += a; d -= a
        elif d < 0 and h[i] > 0:
            a = min(-d, h[i]); h[i] -= a; d += a
        if abs(d) < 1e-14:
            break
    return h

def ls_binary(h, dx, ts, tlim):
    b = h.astype(float).copy()
    bc = compute_c5(b, dx)

                    break
                b[i] = 1; b[j] = 0

                break

            break
    return b, bc

import numpy as np
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    d = ts - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
    else:
        r = h.copy()
    sr = r.sum()
    if sr < 1e-14:
        h[:] = ts / len(h)
        return h
    h = h + r * (d / sr)
    h = np.clip(h, 0.0, 1.0)
    d = ts - h.sum()
    for i in range(len(h)):
        if d > 0 and h[i] < 1.0:
            a = min(d, 1.0 - h[i]); h[i] += a; d -= a
        elif d < 0 and h[i] > 0.0:
            a = min(-d, h[i]); h[i] -= a; d += a
        if abs(d) < 1e-14:
            break
    return h

def ls_binary(h, dx, ts, tlim):
    b = h.astype(float).copy()
    bc = compute_c5(b, dx)
    t0 = time.time()
    n = len(b)
    while time.time() - t0 < tlim:
        ow = np.where(b == 1)[0]
        zw = np.where(b == 0)[0]
        imp = False
        for i in ow:
            for j in zw:
                b[i] = 0.0; b[j] = 1.0
                nc = compute_c5(b, dx)
                if nc < bc - 1e-12:
                    bc = nc; imp = True
                    break
                b[i] = 1.0; b[j] = 0.0
            if imp:
                break
        if not imp:
            break
    return b, bc

def sa_binary(h, dx, ts, tlim, sd):

    best = cur.copy()
