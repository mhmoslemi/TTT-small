from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, beta):
    scaled = values * beta
    m = np.max(scaled)
    if m > 500:
        scaled = scaled - m
    return (m + np.log(np.sum(np.exp(scaled)))) / beta

def obj_smoothed(x, n, beta):
    x = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    return smooth_max(corr, beta)

def obj_exact(x):
    return compute_c5(x)

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:

        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            h[interior] += diff / int(np.sum(interior))
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upscale_step(h, n_new):
    """Upscale using step function (repeat each value)"""
    n_old = len(h)
    ratio = n_new // n_old
    h_new = np.repeat(h, ratio)
    if len(h_new) < n_new:
        pad = n_new - len(h_new)
        h_new = np.concatenate([h_new, h[-pad:]])
    return h_new[:n_new]

import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(values, beta):
    scaled = values * beta
    m = float(np.max(scaled))
    if m > 500.0:
        scaled = scaled - m
    return (m + float(np.log(np.sum(np.exp(scaled))))) / beta


def obj_smoothed(x, n, beta):
    x = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    return smooth_max(corr, beta)


def sum_eq(x):
    return float(np.sum(x)) - len(x) / 2.0


def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def upscale_step(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    ratio = max(1, n_new // n_old)
    h_new = np.repeat(h, ratio)
    if len(h_new) < n_new:
        pad = n_new - len(h_new)
        h_new = np.concatenate([h_new, h[-pad:]])
    elif len(h_new) > n_new:
        h_new = h_new[:n_new]
    return h_new


def interp_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_ext = np.append(x_old, 2.0)
    h_ext = np.append(h_old, h_old[0])
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_ext, h_ext)


def optimize_slsqp(h_init, n, maxiter, beta):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq}]
    h_init = repair(h_init, n)
    try:
        res = minimize(obj_smoothed, h_init, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       args=(n, beta),
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return h_init


def optimize_trust(h_init, n, maxiter, beta):
    A = np.ones((1, n))
    lin = LinearConstraint(A, n / 2.0, n / 2.0)
    bounds = [(0.0, 1.0)] * n
    h_init = repair(h_init, n)
    try:
        res = minimize(obj_smoothed, h_init, method='trust-constr',
                       bounds=bounds, constraints=lin,
                       args=(n, beta),
                       options={'maxiter': maxiter, 'gtol': 1e-7,
                                'xtol': 1e-10, 'barrier_tol': 1e-10})
        return repair(res.x, n)
    except Exception:
        return h_init


def make_cosine_start(n, mode):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if mode == 0:
        h = 0.5 + 0.3 * np.cos(np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.25 * np.cos(2.0 * np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.2 * np.cos(np.pi * x) + 0.1 * np.cos(2.0 * np.pi * x)
    else:
        h = np.random.rand(n) * 0.5 + 0.25
    return repair(np.clip(h, 0.0, 1.0), n)

    # Phase 2: Multi-resolution upscaling
    for mult in [2, 3, 4, 6, 8, 12]:

    return best_h, final_c5, len(best_h)
