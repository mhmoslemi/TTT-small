    best_c5 = c5_cur
    best_h = h_bin.copy()
    for it in range(max_iters):
        if time.time() - t0 > budget_s * budget_frac:
            break
        k = rng.randint(1, 4)
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < k or len(zeros) < k:
            break
        f0 = ones[rng.randint(0, len(ones), size=k)]
        f1 = zeros[rng.randint(0, len(zeros), size=k)]
        h_bin[f0] = 0.0
        h_bin[f1] = 1.0
        c5_new = evaluate_c5(h_bin)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
        else:
            h_bin[f0] = 1.0
            h_bin[f1] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project_to_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Stochastic local search at initial resolution
    best_h, best_c5 = stochastic_search(
        best_h, n0, rng, 0.25, t0, budget_s, 400000, 0.15, 3)

    # Phase 2: Coarser step search
    best_h, best_c5 = stochastic_search(
        best_h, n0, rng, 0.40, t0, budget_s, 300000, 0.4, 4)

    # Phase 3: Fine-tuning with small steps
    best_h, best_c5 = stochastic_search(
        best_h, n0, rng, 0.55, t0, budget_s, 500000, 0.05, 2)

    # Phase 4: Binary optimization
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    while h_bin.sum() > n0 / 2.0:
        idx = np.where(h_bin == 1.0)[0]
        h_bin[idx[rng.randint(len(idx))]] = 0.0
    while h_bin.sum() < n0 / 2.0:
        idx = np.where(h_bin == 0.0)[0]
        h_bin[idx[rng.randint(len(idx))]] = 1.0
    h_bin, c5_b = binary_flip_search(h_bin, n0, rng, 0.72, t0, budget_s, 300000)
    if c5_b < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_b

    # Phase 5: Try upscaling to 2x resolution
    if time.time() - t0 < budget_s * 0.75:
        n2 = n0 * 2
        h2 = np.zeros(n2)
        for i in range(n2):
            h2[i] = best_h[min(int(i * n0 / n2), n0 - 1)]
        h2 = project_to_feasible(h2, n2)
        c5_2 = evaluate_c5(h2)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
        # Optimize at n2
        h2, c5_2 = stochastic_search(h2, n2, rng, 0.87, t0, budget_s, 200000, 0.15, 3)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2

    # Phase 6: Final polish at whatever resolution we have
    n_final = len(best_h)
    best_h, best_c5 = stochastic_search(
        best_h, n_final, rng, 0.97, t0, budget_s, 200000, 0.03, 2)

    # Ensure exact feasibility
    best_h = project_to_feasible(best_h, n_final)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_final
