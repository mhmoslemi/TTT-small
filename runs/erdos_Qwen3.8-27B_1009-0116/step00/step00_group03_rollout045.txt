    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s < 1e-10:
        return np.ones(n) * 0.5
    h = h * (target / s)
    h = np.clip(h, 0, 1)
    # Fine adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)[::-1]  # largest first, but we want to add to smallest
            idx = np.argsort(h)  # smallest first

        for i in idx:

                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-10:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.ones(n) * 0.5
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            idx = int(room.argmax())
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(h.argmax())
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
