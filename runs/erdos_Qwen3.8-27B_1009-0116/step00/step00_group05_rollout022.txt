```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def make_binary(h, n, rng):
    indices = np.argsort(-h)
    bits = np.zeros(n)
    bits[indices[:n // 2]] = 1.0
    return bits

def swap_local_search(h, n, max_rounds=300, rng=None, deadline=None):
    if rng is None:
        rng = np.random.default_rng()
    best = h.copy()
    best_c5 = compute_c5(best)
    ones_pos = np.where(best > 0.5)[0]
    zeros_pos = np.where(best <= 0.5)[0]
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        n Attempt = n * 8
        for _ in range(n Attempt):
            if len(ones_pos) == 0 or len(zeros_pos) == 0:
                break
            i = ones_pos[rng.integers(len(ones_pos))]
            j = zeros_pos[rng.integers(len(zeros_pos))]
            trial = best.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c5 = compute_c5(trial)
            if c5 < best_c5 - 1e-14:
                best = trial
                best_c5 = c5
                ones_pos = np.where(best > 0.5)[0]
                zeros_pos = np.where(best <= 0.5)[0]
                improved = True
                break
        if not improved:
            break
    return best, best_c5

def simulated_anneal(h, n, t_start=1.0, t_end=0.005, n_iter=20000, rng=None, deadline=None):
    if rng is None:
        rng = np.random.default_rng()
    current = h.copy()
    current_c5 = compute_c5(current)
    best = current.copy()
    best_c5 = current_c5
    best_iter = 0
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / float(n_iter))
        ones_pos = np.where(current > 0.5)[0]
        zeros_pos = np.where(current <= 0.5)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        i = ones_pos[rng.integers(len(ones_pos))]
        j = zeros_pos[rng.integers(len(zeros_pos))]
        trial = current.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        trial_c5 = compute_c5(trial)
        delta = trial_c5 - current_c5
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            current = trial
            current_c5 = trial_c5
            if current_c5 < best_c5 - 1e-14:
                best = trial.copy()
                best_c5 = current_c5
                best_iter = it
    return best, best_c5

def continuous_refine(h, n, max_iters=800, step=0.05, rng=None, deadline=None):
    if rng is None:
        rng = np.random.default_rng()
    best = h.copy()
    best_c5 = compute_c5(best)
    target = n / 2.0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(n * 4):
            i = rng.integers(n)
            delta = (rng.random() - 0.5) * 2.0 * step
            trial = best.copy()
            trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
            diff = target - trial.sum()
            if abs(diff) > 1e-9:
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                trial[mask] = np.clip(trial[mask] + diff / (n - 1), 0.0, 1.0)
                if abs(trial.sum() - target) > 1e-6:
                    continue
            c5 = compute_c5(trial)
            if c5 < best_c5 - 1e-13:
                best = trial
                best_c5 = c5
                improved = True
        if not improved:
            break
    return best, best_c5

def resample_h(h, n_new, n_old):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h, left=0.0, right=0.0)

def random_binary(n, rng):
    bits = np.zeros(n)
    idx = rng.permutation(n)
    bits[idx[:n // 2]] = 1.0
    return bits

def structured_binary(n, rng):
    bits = np.zeros(n)
    half = n // 2
    spacing = n / half
    positions = np.floor(np.arange(half) * spacing + rng.random(half) * 0.8).astype(int) % n
    seen = set()
    count = 0
    for p in positions:
        if p not in seen:
            bits[p] = 1.0
            seen.add(p)
            count += 1
    if count < half:
        remaining = np.array([i for i in range(n) if bits[i] == 0])
        for p in remaining[:half - count]:
            bits[p] = 1.0
    return bits

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 5.0

    # Initialize from parent
    global initial_h_values
    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    if h0 is None:
        h0 = np.zeros(70)
        h0[17:53] = 1.0
    n0 = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    target0 = n0 / 2.0
    if abs(h0.sum() - target0) > 1e-8:
        h0 *= target0 / max(h0.sum(), 1e-10)
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search on parent
    h_bin = make_binary(h0, n0, rng)
    h_bin, c5_bin = swap_local_search(h_bin, n0, max_rounds=200, rng=rng, deadline=deadline)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    # Phase 2: Simulated annealing on best binary
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    sa_time = min(remaining * 0.35, 250.0)
    sa_deadline = time.time() + sa_time
    n_iter_sa = int(min(50000, sa_time * 50000))
    h_sa, c5_sa = simulated_anneal(best_h.copy(), n0, t_start=0.8, t_end=0.003,
                                   n_iter=n_iter_sa, rng=rng, deadline=sa_deadline)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    # Phase 3: Multiple restarts at n0
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    n_restarts = min(8, int(remaining / 20))
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        roll = rng.random()
        if roll < 0.4:
            h_start = random_binary(n0, rng)
        elif roll < 0.7:
            h_start = structured_binary(n0, rng)
        else:
            h_start = make_binary(h0, n0, rng)
            h_start = h_start.copy()
            # Perturb
            for _ in range(n0 // 3):
                ones_p = np.where(h_start > 0.5)[0]
                zeros_p = np.where(h_start <= 0.5)[0]
                if len(ones_p) > 0 and len(zeros_p) > 0:
                    a = ones_p[rng.integers(len(ones_p))]
                    b = zeros_p[rng.integers(len(zeros_p))]
                    h_start[a], h_start[b] = 0.0, 1.0
        rd = min(time.time() + 15.0, deadline)
        h_r, c5_r = swap_local_search(h_start, n0, max_rounds=100, rng=rng, deadline=rd)
        h_r2, c5_r2 = simulated_anneal(h_r, n0, t_start=0.5, t_end=0.01,
                                       n_iter=5000, rng=rng, deadline=rd)
        c5_r = min(c5_r, c5_r2)
        h_r = h_r if c5_r2 >= c5_r else h_r2
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 4: Try other n values
    for n_try in [50, 60, 80, 90, 100, 110, 120]:
        if time.time() > deadline:
            break
        n_try = n_try if n_try % 2 == 0 else n_try + 1
        budget_phase = min(remaining * 0.15, 60.0)
        phase_deadline = min(time.time() + budget_phase, deadline)
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break

        h_try = make_binary(resample_h(best_h, n_try, len(best_h)), n_try, rng)
        h_try, c5_try = swap_local_search(h_try, n_try, max_rounds=80, rng=rng, deadline=phase_deadline)
        h_try2, c5_try2 = simulated_anneal(h_try, n_try, t_start=0.5, t_end=0.01,
                                           n_iter=3000, rng=rng, deadline=phase_deadline)
        if c5_try2 < c5_try:
            h_try, c5_try = h_try2, c5_try2
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 5: Continuous refinement
    if time.time() < deadline - 10:
        h_cont = best_h.copy()
        cd = min(time.time() + (deadline - time.time()) * 0.5, deadline)
        h_cont, c5_cont = continuous_refine(h_cont, len(h_cont), max_iters=600,
                                            step=0.04, rng=rng, deadline=cd)
        if c5_cont < best_c5:
            best_h, best_c5 = h_cont, c5_cont

    # Phase 6: Final local polish
    if time.time() < deadline - 5:
        n_best = len(best_h)
        if np.allclose(best_h, (best_h > 0.5).astype(float)):
            h_pol, c5_pol = swap_local_search(best_h, n_best, max_rounds=50, rng=rng, deadline=deadline)
        else:
            h_pol, c5_pol = continuous_refine(best_h, n_best, max_iters=300,
                                              step=0.02, rng=rng, deadline=deadline)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target_final = n_final / 2.0
    if abs(best_h.sum() - target_final) > 1e-8:
        best_h += (target_final - best_h.sum()) / n_final
        best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
```
