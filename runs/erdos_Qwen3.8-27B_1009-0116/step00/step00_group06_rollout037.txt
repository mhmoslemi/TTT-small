Let me also think about the increment for the correlation update. When we swap h[i]: 1→0 and h[j]: 0→1:

The correlation C[k] = sum_m h[m] * (1-h[m+n-1-k]) (as I derived earlier)

Change in C[k] from changing h[i] 1→0:
- C[k] has terms involving h[i]: 
  - As the "first factor": h[i]*(1-h[i+n-1-k]) → 0, so C[k] decreases by (1-h[i+n-1-k]) if i+n-1-k is in [0,n-1]
  - As the "second factor's complement": for any m where m+n-1-k = i, i.e., m = i-k+n-1, the term h[m]*(1-h[i]) → h[m]*1, so C[k] increases by h[i-k+n-1] if i-k+n-1 is in [0,n-1]

Total change from h[i]: 1→0:
ΔC_i[k] = -(1-h[i+n-1-k]) + h[i-k+n-1] (where terms are 0 if index out of range)

Similarly for h[j]: 0→1:
ΔC_j[k] = (1-h[j+n-1-k]) - h[j-k+n-1]

Total: ΔC[k] = ΔC_i[k] + ΔC_j[k]

This is O(n) to update for all k, compared to O(n^2) to recompute. But it's more complex to implement correctly.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    return float(np.max(corr) * dx)

def slsqp_optimize(h0, n_iters=200):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(h):
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode="full")
        return float(np.max(corr) * dx)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                  constraints=cons, options={'maxiter': n_iters, 'ftol': 1e-12})
    return res.x, res.fun

def binary_sa_search(h_init, n, max_iters, rng, t0, budget):
    dx = 2.0 / n
    target = n // 2
    
    h = h_init.copy()
    # Ensure exactly target ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    while len(ones) > target:
        idx = ones[rng.randint(len(ones))]
        h[idx] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    while len(ones) < target:
        idx = zeros[rng.randint(len(zeros))]
        h[idx] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_h = h.copy()
    
    temperature = 0.5
    cooldown = 0.9995
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - t0 > budget - 15:
            break
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0; h[j] = 1
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_corr))
        
        delta = new_max - curr_max
        if delta <= 0 or rng.rand() < np.exp(-delta / max(temperature, 1e-8)):
            curr_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1; h[j] = 0
        
        temperature *= cooldown
        if temperature < 0.005:
            break
    
    return best_h, best_max * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial

        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None
    
    best_c5 = float('inf')
    best_h = np.tile([1.0, 0.0], 50)  # fallback n=100
    best_n = 100
    
    if h_init is not None:
        best_h = h_init.copy()
        best_n = len(h_init)
        best_c5 = compute_c5(h_init)
    
    # Phase 1: SLSQP on initial grid
    if h_init is not None and budget_s - (time.time() - t0) > 60:
        try:
            h_opt, c5_opt = slsqp_optimize(h_init, 300)
            # Clip to [0,1]
            h_opt = np.clip(h_opt, 0, 1)
            # Fix sum
            diff = len(h_opt)/2.0 - h_opt.sum()
            h_opt += diff / len(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        except Exception:
            pass
    
    # Phase 2: Binary SA on various grid sizes
    grid_sizes = [100, 150, 200, 300, 400]
    for n in grid_sizes:
        if time.time() - t0 > budget_s - 60:
            break
        
        # Create initial binary
        if h_init is not None:
            x_old = np.arange(len(h_init)) + 0.5
            x_new = np.arange(n) + 0.5
            h_interp = np.interp(x_new, x_old, h_init)
            h_bin = (h_interp > 0.5).astype(float)
        else:
            h_bin = np.zeros(n)
            h_bin[rng.choice(n, n//2, replace=False)] = 1.0
        
        # Also try a few random starts
        n_restarts = 3
        for restart in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            if restart == 0 and h_init is not None:
                h_start = h_bin
            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def soft_max_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = corr.max()
    s = np.clip(beta * (corr - mc), -50, 50)
    return float(mc + np.log(np.exp(s).sum()) / beta)

def make_binary_from_cont(h_cont, n, rng):
    if len(h_cont) != n:
        xo = np.arange(len(h_cont)) + 0.5
        xn = np.arange(n) + 0.5
        h_cont = np.interp(xn, xo, h_cont)
    target = n // 2
    order = np.argsort(h_cont)[::-1]
    h = np.zeros(n)
    h[order[:target]] = 1.0
    return h

def sa_binary(h0, n, iters, rng, t0, budget):
    h = h0.copy()
    target = n // 2
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    curr = float(corr.max())
    best = curr
    best_h = h.copy()
    temp = 0.8
    for it in range(iters):
        if time.time() - t0 > budget - 12:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        if h[i] == 0:
            i, j = j, i
        h[i] = 0.0
        h[j] = 1.0
        nc = np.correlate(h, 1.0 - h, mode="full").astype(float)
        nm = float(nc.max())
        d = nm - curr
        if d <= 0.0 or rng.rand() < np.exp(-d / max(temp, 1e-10)):
            curr = nm
            corr = nc
            if nm < best:
                best = nm
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        temp *= 0.99997
        if temp < 0.004:
            break
    return best_h, best

def greedy_refine(h0, n, t0, budget, max_passes=3):
    h = h0.copy()
    dx = 2.0 / n
    target = n // 2
    for _pass in range(max_passes):
        if time.time() - t0 > budget - 15:
            break
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_gain = 0
        best_pair = None
        for ii in ones:
            if time.time() - t0 > budget - 15:
                break
            h[ii] = 0
            for jj in zeros:
                h[jj] = 1
                nc = np.correlate(h, 1.0 - h, mode="full")
                nm = float(nc.max())
                h[jj] = 0
                gain = best - nm
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (ii, jj)
            h[ii] = 1
        if best_pair is not None and best_gain > 0:
            ii, jj = best_pair
            h[ii] = 0
            h[jj] = 1
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            best = float(corr.max())
            improved = True
        else:
            corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
            best = float(corr.max())
            if not improved:
                break
    return h, best

def continuous_refine(h0, n, t0, budget):
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    if n % 2 == 1:
        target = n / 2.0
    else:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for beta in [60, 200]:
        if time.time() - t0 > budget - 20:
            break
        def obj(hv):
            return soft_max_c5(hv, dx, beta) + 5000.0 * (hv.sum() - target) ** 2
        try:
            r = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 200, 'ftol': 1e-15})
            h_new = np.clip(r.x, 0.0, 1.0)
            h_new += (target - h_new.sum()) / n
            h_new = np.clip(h_new, 0.0, 1.0)
            c5 = compute_c5(h_new)
            if c5 < compute_c5(h):
                h = h_new
        except Exception:
            break
    return h, compute_c5(h)
