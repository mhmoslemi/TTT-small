    return np.ones(len(x))


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)

    res_list = []
    for mult in [1, 2, 4, 8]:
        nr = n_init * mult
        if nr <= 400:
            res_list.append(nr)

    for n in res_list:
        if time.time() - t0 > budget_s * 0.88:
            break
        h0 = upsample_arr(best_h, n)
        bounds = [(0.0, 1.0)] * n
        cons = ({'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac},)

        # SLSQP phase with analytical gradient
        if time.time() - t0 < budget_s * 0.6:
            try:
                res = minimize(evaluate_c5, h0, method='SLSQP', jac=analytic_grad,
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-13})
                h_opt = project_feasible(res.x, n)
                c5 = evaluate_c5(h_opt)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_opt
            except Exception:
                pass

        # Nelder-Mead with multiple restarts and varied perturbations
        n_restarts = 5 if n <= 90 else 3
        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.93:
                break
            if r == 0:
                h_r = upsample_arr(best_h, n)
            elif r == 1:
                h_r = upsample_arr(best_h, n) + np.random.randn(n) * 0.03
                h_r = project_feasible(h_r, n)
            elif r == 2:
                h_r = upsample_arr(best_h, n) + np.random.randn(n) * 0.1
                h_r = project_feasible(h_r, n)
            else:
                h_r = upsample_arr(best_h, n)
                freq = np.random.randint(1, n // 2 + 1)
                phase = np.random.uniform(0, 2.0 * np.pi)
                h_r += 0.05 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
                h_r = project_feasible(h_r, n)

            max_it = 10000 if n <= 90 else 4000
            try:
                res = minimize(evaluate_c5, h_r, method='Nelder-Mead',
                               options={'maxiter': max_it, 'xatol': 1e-9, 'fatol': 1e-12})
                h_opt = project_feasible(res.x, n)
                c5 = evaluate_c5(h_opt)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_opt
            except Exception:
                pass

    # Final SLSQP polish at best resolution
    if time.time() - t0 < budget_s * 0.96:
        n_best = len(best_h)
        bounds = [(0.0, 1.0)] * n_best
        cons = ({'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac},)
        try:
            res = minimize(evaluate_c5, best_h, method='SLSQP', jac=analytic_grad,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_opt = project_feasible(res.x, n_best)
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt
        except Exception:
            pass

    return (best_h.tolist(), best_c5, len(best_h))
