import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def adjust_sum(h, n):
    """Adjust h to satisfy sum(h) = n/2 while staying in [0,1]."""
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # Redistribute
    mean_diff = diff / n
    h = h + mean_diff
    h = np.clip(h, 0, 1)
    # If still not exact, do iterative adjustment
    for _ in range(100):
        current = np.sum(h)
        diff = target - current
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1) if diff > 0 else (h > 0) & (h < 1)
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            break
    return h

def make_objective(n, alpha=50.0):
    def obj(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        a = alpha * corr * dx
        m = np.max(a)
        return (m + np.log(np.sum(np.exp(a - m)))) / (alpha * dx)
    return obj

def optimize_step(h, n, max_iter=200, alpha=50.0):
    """Run SLSQP to minimize C5."""
    obj = make_objective(n, alpha)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(
        obj, h, method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-10}
    )
    
    if result.success or result.x is not None:
        h_opt = np.clip(result.x, 0, 1)
