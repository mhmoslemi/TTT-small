import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_h(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Iterative water-filling
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Add/subtract uniformly from non-bound elements
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def grad_max_lag(h, n):
    """Compute gradient of C5 (max lag correlation)."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(c)
    # Use lags within tolerance of max
    tol = 1e-8
    active = np.where(c >= max_val - tol)[0]
    
    g = np.zeros(n)
    for idx in active:
        m = idx - (n - 1)
        if m >= 0:
            g[:n-m] += dx * (1.0 - h[m:])
            g[m:] += -dx * h[:n-m]
        else:
            k = -m
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
    return g

    return compute_c5(h)

def project_h(h, n):

    for _ in range(200):

        diff = target - s

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, n, idx):
    dx = 2.0 / n
    m = idx - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        if n - m > 0:
            g[:n-m] += dx * (1.0 - h[m:])
            g[m:] += -dx * h[:n-m]
    else:
        k = -m
        if n - k > 0:
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
    return g

def get_active_lags(h, n, tol=1e-7):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(c)
    return np.where(c >= mx - tol)[0]

def slsqp_opt(h0, n, maxiter=300):
    h0 = project_h(h0, n)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return compute_c5(x)

    def jac(x):
        active = get_active_lags(x, n)
        g = np.zeros(n)
        for idx in active:
            g += grad_lag(x, n, idx)
        return g / max(1, len(active))

    def eq(x):
        return np.sum(x) - n / 2.0

    res = minimize(fun, h0, jac=jac, bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq}],
                   method='SLSQP',
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_h(res.x, n)
    return h0

def upsample(h, new_n):
    n = len(h)
    if new_n <= n:
        return np.interp(np.arange(new_n) * n / new_n, np.arange(n), h)
    old_t = np.linspace(0.0, 1.0, n)
    new_t = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_t, old_t, h)

def local_refine(h, n, max_rounds=20):
    h = project_h(h.copy(), n)
    best = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in (0.02, 0.06, 0.1):
                    nt = h.copy()
                    hi = min(1.0, h[i] + d)
                    nj = h[j] - (hi - h[i])
                    if nj < 0.0:
                        continue
                    nt[i] = hi
                    nt[j] = nj
                    c = compute_c5(nt)
                    if c < best - 1e-12:
                        h = nt
                        best = c
                        improved = True
                        break
        if not improved:
            break
    return h, best

def smooth_max_c5(h, n, T=30.0):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = c - np.max(c)
    return (np.max(c) + np.log(np.sum(np.exp(T * cs))) / T) * dx

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)

        c5 = compute_c5(h_o)
        if c5 < best_c5:

    # Phase 2: Random restarts at base

        if tl() < 50:

        sigma = 0.03 + 0.02 * r

        h_r = project_h(h_r, n0)

        c5 = compute_c5(h_o)
        if c5 < best_c5:

    # Phase 3: Smooth-max guided optimization (helps escape local optima)
    if tl() > 60:
        for T_val in (10.0, 30.0, 80.0):

                return smooth_max_c5(x, n0, T_val)

                cs = c - np.max(c)
                w = np.exp(T_val * cs) / np.sum(np.exp(T_val * cs))
                g = np.zeros(n0)

                for idx in range(len(c)):

                return g

            res = minimize(sfun, h_s, jac=sfil, bounds=bounds,

    return best_h, best_c5, best_n
