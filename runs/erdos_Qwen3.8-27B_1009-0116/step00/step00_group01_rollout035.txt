import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_res(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.arange(n_old, dtype=float) / n_old
    xn = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(xn, xo, h_old)
    return project_feasible(h_new, n_new)

class SumCon:
    def __init__(self, t):
        self.t = t
    def fun(self, x):
        return np.sum(x) - self.t
    def jac(self, x):
        return np.ones(len(x))

def do_slsqp(h0, n, maxiter, ftol):
    h0 = project_feasible(h0, n)
    sc = SumCon(n / 2.0)
    cons = [{'type': 'eq', 'fun': sc.fun, 'jac': sc.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_r = project_feasible(res.x, n)
        return h_r, compute_c5(h_r)
    except Exception:
        return None, float('inf')

def do_cobyla(h0, n, maxiter):
    h0 = project_feasible(h0, n)
    sc = SumCon(n / 2.0)
    cons = [{'type': 'eq', 'fun': sc.fun}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(compute_c5, h0, method='COBYLA', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'rhobeg': 0.005})
        h_r = project_feasible(res.x, n)
        return h_r, compute_c5(h_r)
    except Exception:
        return None, float('inf')

Now, there's a subtlety: the gradient I computed is the gradient of c_{k*} only. The objective is max_k c_k, so the true subgradient is the gradient of the active (maximizing) c_k. import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    js = int(np.argmax(corr))
    m = js - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]
    return g * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def interp_res(ho, nn):
    no = len(ho)
    if no == nn:
        return ho.copy()
    xo = np.arange(no, dtype=float) / no
    xn = np.arange(nn, dtype=float) / nn
    return project_feasible(np.interp(xn, xo, ho), nn)

class SC:
    def __init__(self, t):
        self.t = t
    def fun(self, x):
        return np.sum(x) - self.t
    def jac(self, x):
        return np.ones(len(x))

def do_slsqp(h0, n, mi, ft):
    h0 = project_feasible(h0, n)
    sc = SC(n / 2.0)
    try:
        r = minimize(compute_c5, h0, jac=c5_grad, method='SLSQP',
                     bounds=[(0., 1.)] * n,
                     constraints=[{'type': 'eq', 'fun': sc.fun, 'jac': sc.jac}],
                     options={'maxiter': mi, 'ftol': ft})
        hr = project_feasible(r.x, n)
        return hr, compute_c5(hr)
    except Exception:
        return None, float('inf')

        return budget_s - (time.time() - t0)

    # Stage 1: parent resolution

        if tl() < budget_s * 0.80:
