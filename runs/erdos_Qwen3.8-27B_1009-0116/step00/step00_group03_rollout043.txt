def project_onto_simplex_box(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Use bisection on lambda for: sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def subgradient_step(h, n, lr, momentum=0.9):
    """One step of projected subgradient descent with momentum."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    m_star = np.argmax(corr)
    k = m_star - (n - 1)
    # Subgradient
    grad = np.zeros(n)
    js = np.arange(n)
    # (1-h[j-k]) for valid j-k
    valid1 = (js - k >= 0) & (js - k < n)
    grad[valid1] += (1 - h[js[valid1] - k])

    valid2 = (js + k >= 0) & (js + k < n)
    grad[valid2] -= h[js[valid2] + k]
    grad *= dx
    return grad

def project_onto_constraints(h, n):
    target = n / 2.0
