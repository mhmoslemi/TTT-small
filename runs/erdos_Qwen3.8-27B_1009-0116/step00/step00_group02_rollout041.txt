```python
    # Adjust to satisfy sum constraint
    target = n / 2.0

    if current > target:
        # Need to reduce: subtract from largest values

            reduce = min(h[idx], excess)
            h[idx] -= reduce
            excess -= reduce
```

```python
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if mask.sum() == 0: break
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    return h

def coord_desc(h, max_sweeps=200, n_candidates=20):
    n = len(h)
    best = compute_c5(h)
    h = h.copy()
    candidates = np.linspace(0, 1, n_candidates)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            best_val = old_val
            best_c = best
            for v in candidates:
                h[i] = v
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_val = v
            if best_val != old_val:
                h[i] = best_val
                best = best_c
                improved = True
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        n_perturb = rng.randint(5, 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        h_r[idx] = 1.0 - h_r[idx]
        h_r = project_h(h_r)
        
        # Coordinate descent
        h_t, c5_t = coord_desc(h_r, max_sweeps=150)
        if c5_t < best_c5:
```

Actually, let me try a different approach. Since the problem is small, I can use scipy.optimize.minimize with BFGS or L-BFGS-B, using a soft-max of the correlations as the objective:

f(h) = (1/T) * log(∑ exp(T * c_k)) ≈ max(c_k) for large T

This is smooth and differentiable! The gradient is:
∂f/∂h[i] = (1/T) * ∂/∂h[i] log(∑ exp(T * c_k))
= (1/T) * (∑ T * c'_k * exp(T * c_k)) / (∑ exp(T * c_k))
= ∑ (c'_k * exp(T * c_k)) / (∑ exp(T * c_k))
where c'_k = ∂c_k/∂h[i]

This is a convex combination of the gradients of the individual c_k's, weighted by their "importance" (softmax).

For the constraint sum(h) = n/2, I can use a Lagrange multiplier or handle it in projection.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    return np.clip(h, 0.0, 1.0)

def coord_desc(h, max_sweeps=300, n_cand=25):
    n = len(h)
    best = compute_c5(h)[0]
    h = h.copy()
    cands = np.linspace(0.0, 1.0, n_cand)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            bv, bc = old, best
            for v in cands:
                if v == old:
                    continue
                h[i] = v
                c = compute_c5(h)[0]
                if c < bc:
                    bc, bv = c, v
            if bv != old:
                h[i] = bv
                best = bc
                improved = True
        if not improved:
            break
    return h, best

def soft_max_obj(h, T=50.0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    val = m + np.log(np.mean(np.exp(T * (corr - m)))) / T
    pen = 100.0 * (np.sum(h) - len(h) / 2.0) ** 2
    return val + pen

def gradient_step(h, lr=0.02, T=50.0, n_iter=500):
    h = h.copy()
    best = compute_c5(h)[0]
    for it in range(n_iter):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        weights = np.exp(T * (corr - m))
        weights /= weights.sum()
        # Gradient of each c_k w.r.t. h
        n = len(h)
        grad = np.zeros(n)
        for k in range(len(corr)):
            # c_k = sum_i h[i] * (1-h[i+k-offset]) * dx for valid range
            # offset for np.correlate full mode: result index k corresponds to lag k-(n-1)
            lag = k - (n - 1)
            for i in range(max(0, -lag), min(n, n - lag)):
                j = i + lag
                # d/dh[i] of h[i]*(1-h[j]) = (1-h[j])
                # d/dh[i] of h[j]*(1-h[i]) = -h[j]
                if i == j:
                    grad[i] += weights[k] * dx * (1 - 2 * h[i])
                else:
                    # h[i] appears as h[i]*(1-h[j]) contribution
                    if j >= 0 and j < n:
                        # check: is this term h[i]*(1-h[j]) or h[j]*(1-h[i]) in c_k?
                        # In correlate(h, 1-h): c[k] = sum h[a] * (1-h)[a+lag]
                        # so term is h[a]*(1-h[a+lag])
                        # d/dh[i]: if a==i: (1-h[i+lag]); if a+lag==i: -h[a]
                        if lag >= 0:
                            # a ranges 0..n-1-lag, a+lag ranges lag..n-1
                            # h[i] appears when a==i (if i < n-lag): contrib (1-h[i+lag])
                            # h[i] appears when a+lag==i (if i >= lag): contrib -h[i-lag]
                            if i < n - lag:
                                grad[i] += weights[k] * dx * (1.0 - h[i + lag])
                            if i >= lag:
                                grad[i] += weights[k] * dx * (-h[i - lag])
                        else:
                            # a ranges -lag..n-1, a+lag ranges 0..n-1+lag-1
                            # h[i] appears when a==i (if -lag <= i <= n-1): contrib (1-h[i+lag])
                            # h[i] appears when a+lag==i (if 0 <= i <= n-1+lag-1): contrib -h[i+(-lag)]
                            if -lag <= i:
                                grad[i] += weights[k] * dx * (1.0 - h[i + lag]) if i + lag < n else 0
                            if 0 <= i - lag:
                                grad[i] += weights[k] * dx * (-h[i - lag]) if i - lag >= 0 else 0
        # Add constraint gradient
        grad += 200.0 * (np.sum(h) - n / 2.0)
        h -= lr * grad
        h = project_h(h)
        c = compute_c5(h)[0]
        if c < best:
            best = c
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 45
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0
        if n % 2 == 1:
            h0[n // 2] = 0.5
    
    h0 = project_h(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)[0]
    
    # Phase 1: Coordinate descent from initial
    h_cd, c5_cd = coord_desc(h0, max_sweeps=500, n_cand=30)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd.copy(), c5_cd
    
    # Phase 2: Multi-start with perturbations
    pool = [best_h.copy()]
    while time.time() - t0 < budget_s * 0.75:
        base = pool[rng.randint(len(pool))]
        # Random perturbation
        h_r = base.copy()
        npert = rng.randint(4, min(15, n))
        idx = rng.choice(n, size=npert, replace=False)
        h_r[idx] = rng.uniform(0, 1, size=npert)
        h_r = project_h(h_r)
        
        # Coordinate descent
        h_t, c5_t = coord_desc(h_r, max_sweeps=200, n_cand=25)
        if c5_t < best_c5:
            best_h, best_c5 = h_t.copy(), c5_t
        if c5_t < best_c5 + 0.01:
            pool.append(h_t.copy())
            if len(pool) > 20:
                pool.pop(0)
    
    # Phase 3: Gradient refinement on best
    if time.time() - t0 < budget_s * 0.85:
        for T in [30.0, 100.0, 300.0]:
            if time.time() - t0 > budget_s * 0.9:
                break
            h_g, c5_g = gradient_step(best_h, lr=0.03, T=T, n_iter=2000)
            if c5_g < best_c5:
                best_h, best_c5 = h_g.copy(), c5_g
        # Final coordinate descent polish
        h_cd, c5_cd = coord_desc(best_h, max_sweeps=500, n_cand=50)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd.copy(), c5_cd
    
    # Final check
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)[0]
    
    return best_h, best_c5, n
```
