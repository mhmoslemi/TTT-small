    rng = np.random.RandomState(seed)
    h = np.copy(h)
    # Ensure binary
    h = (h >= 0.5).astype(float)
    n = len(h)

    zeros = np.where(h == 0)[0]

    current_c5 = best_c5

        for _ in range(n):  # n swaps per sweep
            if len(ones) < target or len(zeros) < target:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:

            h[i] = 0
            h[j] = 1
            new_c5 = compute_c5(h)
            if new_c5 < current_c5 or rng.rand() < np.exp(-(new_c5 - current_c5) / max(t, 1e-10)):

                    best_c5 = current_c5
                    best_h = np.copy(h)

                h[i] = 1
                h[j] = 0

def swap_delta(h, i, j, n):
    """Compute change in full correlation when h[i]:1→0 and h[j]:0→1."""
    # s ranges from -(n-1) to (n-1), corresponding to m = s + (n-1)
    s_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    for idx, s in enumerate(s_range):
        d = 0
        # Term 1: l=i, i+s valid?
        if 0 <= i + s < n:
            d -= (1 - h[i+s])
        # Term 2: l=i-s, i-s valid?
        if 0 <= i - s < n:
            d += h[i-s]
        # Term 3: l=j, j+s valid?
        if 0 <= j + s < n:
            d += (1 - h[j+s])
        # Term 4: l=j-s, j-s valid?
        if 0 <= j - s < n:
            d -= h[j-s]
        delta[idx] = d
    
    return delta

def _coord_descent(h_init, time_limit):

    h = np.clip(h_init.astype(float), 0.0, 1.0)
    target = n / 2.0
    h = _project_sum(h, target)
    cur = _c5(h)
    best_h = np.copy(h)
    best = cur

    cands = np.array([0.0, 0.125, 0.25, 0.375, 0.5, 0.625, 0.75, 0.875, 1.0])

        for i in range(n):
            if time.time() - t_start >= time_limit:

            h[i] = 0.0
            c0 = _corr_full(h)
            h[i] = 1.0
            c1 = _corr_full(h)
            h[i] = old

                if v < best_val:

                improved = True

        cur = _c5(h)
        if cur < best:
            best = cur
            best_h = np.copy(h)
        if not improved:
