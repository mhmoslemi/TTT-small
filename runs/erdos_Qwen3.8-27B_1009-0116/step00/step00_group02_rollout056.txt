            best_h = h_o

    # Phase 3: Upscale to higher resolution (remaining budget)
    rem = budget_s - (time.time() - t_start)
    for mult in [2, 3]:
        if rem < 40:
            break
        tn = len(best_h) * mult
        if tn > 400:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, tn, endpoint=False)
        h_up = np.interp(xn, xo, best_h)
        h_up = project_feasible(h_up, tn)
        alloc = max(20.0, rem * 0.55)
        h_o, c5_o = optimize_from(h_up, tn, alloc, seed + 7000)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o
        rem = budget_s - (time.time() - t_start)

    # Final polish
    final_rem = budget_s - (time.time() - t_start) - 1.0
    if final_rem > 8:
        best_h, best_c5 = optimize_from(best_h, len(best_h), final_rem, seed + 99999)

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
