    """Gradient of f_{shift}(h) = sum_i h[i+shift]*(1-h[i]) w.r.t. h.
    shift can be negative. 0 <= index < n for all accesses."""
    n = len(h)

    # f = sum_i h[i+shift] * (1-h[i])

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h, shift):
    """Gradient of f_shift(h) = sum_i h[i+shift]*(1-h[i]) w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # h[i+shift]*(1-h[i]): 
    # d/dh[j]: appears as h[j] in "h[i+shift]" when j = i+shift => i = j-shift
    #          appears as h[j] in "(1-h[i])" when j = i
    # Term 1: (1-h[j-shift]) for 0<=j-shift<n => shift<=j<n+shift
    lo = max(0, shift)
    hi = min(n, n + shift)
    if lo < hi:
        grad[lo:hi] += (1.0 - h[lo-shift:hi-shift])
    # Term 2: -h[j+shift] for 0<=j+shift<n => -shift<=j<n-shift
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2+shift:hi2+shift]
    return grad

def top_k_directions(h, k=5):
    """Compute weighted gradient combining top-k shifts"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    top_idx = np.argsort(corr)[-k:][::-1]
    # Weight by how close to max
    vals = corr[top_idx]
    max_val = corr.max()
    floor_val = corr.min()
    # Soft weight: exp(beta * (val - max))
    beta = 5.0
    weights = np.exp(beta * (vals - max_val))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, ki in zip(weights, top_idx):
        shift = ki - (n - 1)
        grad += w * compute_grad(h, shift)
    return grad, corr

def project_feasible(h):
    """Project h to [0,1]^n with sum = n/2"""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Adjust sum
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase sum: add to elements with most room (h close to 1)
        # Simple: add diff/n to all
        h += diff / n
        h = np.clip(h, 0, 1)
    elif diff < 0:
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

def optimize(h, time_limit, lr=0.1, k_shifts=5):
    """Gradient descent with projection"""
    n = len(h)
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    velocity = np.zeros(n)
    beta_mom = 0.9
    
    while time.time() - t0 < time_limit:
        grad, corr = top_k_directions(h, k=k_shifts)
        # Center gradient to preserve sum
        grad -= np.mean(grad)
        # Momentum
        velocity = beta_mom * velocity - lr * grad
        h_new = h + velocity
        h_new = project_feasible(h_new)
        
        h = h_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

Let me also think about whether the gradient descent will actually help. The function we're minimizing is:
F(h) = max_k f_k(h)

The subgradient at h is ∇f_{k*}(h) where k* = argmax. Gradient descent on this will push h in the direction that decreases f_{k*}. But it might increase other f_k's. The "top-k directions" approach helps by considering multiple k's simultaneously.

However, there's a risk: if we're at a binary point, any small perturbation that's not exactly in the right direction will increase some other f_k. The non-convexity makes this tricky.

I think a good additional strategy is to also try:
- Coordinate descent: for each coordinate j, find the optimal h[j] given all others
- The 1D subproblem is: minimize max_k (a_k + b_k * t) subject to 0 ≤ t ≤ 1, where b_k = ∂f_k/∂h[j]

For the 1D minimax of linear functions: the optimal t is where the "upper envelope" of the lines is minimized. This occurs at:
- An intersection of two lines (where two f_k's are equal and both at the max)
- Or at a boundary (t=0 or t=1)

We can find this by:
1. For each pair of k's, find the intersection t_k1_k2 = (a_k2 - a_k1) / (b_k1 - b_k2)
2. Check which intersections are in [0,1]
3. Evaluate the max at each candidate
4. Pick the best

For n=80, there are 159 shifts, so ~12700 pairs. For each coordinate, this is O(n²) = 159² ≈ 25000 operations. For all 80 coordinates: 2M operations per sweep. Very fast.

But we also need the sum constraint. If we change h[j] by δ, we need to adjust another coordinate by -δ to maintain the sum. This makes it more complex.

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h, shift):
    n = len(h)
    grad = np.zeros(n)
    lo = max(0, shift)
    hi = min(n, n + shift)
    if lo < hi:
        grad[lo:hi] += (1.0 - h[lo - shift:hi - shift])
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + shift:hi2 + shift]
    return grad

def weighted_grad(h, k_shifts=5, beta=4.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_shifts:][::-1]
    vals = corr[top_idx]
    mx = vals[0]
    weights = np.exp(-beta * (mx - vals))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, ki in zip(weights, top_idx):
        shift = ki - (n - 1)
        grad += w * compute_grad(h, shift)
    return grad, corr

def project_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    diff = n / 2.0 - np.sum(h)
    h += diff / n
    return np.clip(h, 0.0, 1.0)

def grad_optimize(h, time_limit, lr=0.05, k_shifts=5, beta=4.0, momentum=0.85):
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    vel = np.zeros_like(h)
    lr_cur = lr
    while time.time() - t0 < time_limit:
        grad, corr = weighted_grad(h, k_shifts, beta)
        grad -= np.mean(grad)
        vel = momentum * vel - lr_cur * grad
        h = project_feasible(h + vel)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            lr_cur *= 0.95
        if time.time() - t0 > time_limit * 0.8 and lr_cur < lr * 0.01:
            break
    return best_h, best_c5

def coord_descent(h, time_limit):
    n = len(h)
    t0 = time.time()
    h = h.copy()
    best_c5 = eval_c5(h)
    while time.time() - t0 < time_limit:
        improved = False
        for j in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            # For coordinate j, compute a_k + b_k * t for all k
            # f_k = sum_i h[i+shift_k]*(1-h[i])
            # df_k/dh[j] = (1-h[j-shift_k])*[valid] - h[j+shift_k]*[valid]
            shifts = np.arange(2 * n - 1) - (n - 1)
            bk = np.zeros(2 * n - 1)
            for idx, s in enumerate(shifts):
                i1 = j - int(s)
                i2 = j + int(s)
                if 0 <= i1 < n:
                    bk[idx] += 1.0 - h[i1]
                if 0 <= i2 < n:
                    bk[idx] -= h[i2]
            ak = corr - h[j] * bk
            # min_t max_k (ak[k] + bk[k]*t) for t in [0,1]
            # Try boundaries and intersections
            candidates = [0.0, 1.0, h[j]]
            for ii in range(2 * n - 1):
                for jj in range(ii + 1, 2 * n - 1):
                    denom = bk[ii] - bk[jj]
                    if abs(denom) > 1e-12:
                        t_int = (ak[jj] - ak[ii]) / denom
                        if 0.0 <= t_int <= 1.0:
                            candidates.append(t_int)
            best_t = h[j]
            best_val = np.max(ak + bk * h[j])
            for t_c in candidates:
                t_c = float(np.clip(t_c, 0.0, 1.0))
                val = np.max(ak + bk * t_c)
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t_c
            new_h = h.copy()
            new_h[j] = best_t
            diff = n / 2.0 - np.sum(new_h)
            if abs(diff) > 1e-10:
                # Distribute to a different coordinate
                j2 = (j + 1) % n
                new_h[j2] = np.clip(new_h[j2] - diff, 0.0, 1.0)
                new_h[j] = np.clip(new_h[j], 0.0, 1.0)
            new_c5 = float(np.max(np.correlate(new_h, 1.0 - new_h, mode='full')) * 2.0 / n)
            if new_c5 < eval_c5(h) - 1e-14:
                h = new_h
                improved = True
        if not improved:
            break
    return h, eval_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(80)
        h0[:40] = 1.0

    best_h = h0.copy()
    best_c5 = eval_c5(best_h)

    def tl():
        return budget_s - (time.time() - t_start)

    # Phase 1: Gradient optimization at base resolution
    if tl() > 100:
        h, c5 = grad_optimize(best_h, min(150, tl() * 0.25), lr=0.1, k_shifts=5, beta=3.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Finer gradient with more shifts
    if tl() > 80:
        h, c5 = grad_optimize(best_h, min(120, tl() * 0.2), lr=0.02, k_shifts=10, beta=6.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Coordinate descent
    if tl() > 60:
        h, c5 = coord_descent(best_h, min(80, tl() * 0.12))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Random perturbations + re-optimize
    if tl() > 60:
        n = len(best_h)
        for _ in range(30):
            if tl() < 20:
                break
            h_p = best_h + np.random.randn(n) * 0.03
            h_p = project_feasible(h_p)
            h, c5 = grad_optimize(h_p, min(tl() * 0.03, 10), lr=0.03, k_shifts=5)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 5: Upsample to 2x resolution
    if tl() > 150:
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up)
        c5_up = eval_c5(h_up)
        h, c5 = grad_optimize(h_up, min(200, tl() * 0.3), lr=0.03, k_shifts=5, beta=4.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        elif c5 < c5_up:
            h2, c52 = coord_descent(h, min(40, tl() * 0.08))
            if c52 < best_c5:
                best_h, best_c5 = h2, c52

    # Phase 6: Fine-tune best with very small steps
    if tl() > 30:
        h, c5 = grad_optimize(best_h, min(60, tl() * 0.2), lr=0.005, k_shifts=20, beta=10.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Final coord descent polish
    if tl() > 20:
        h, c5 = coord_descent(best_h, min(30, tl() * 0.3))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return best_h, best_c5, len(best_h)
