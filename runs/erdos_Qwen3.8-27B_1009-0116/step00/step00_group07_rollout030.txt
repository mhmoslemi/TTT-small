def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    n = len(h)
    # Bisection for lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Small correction for numerical accuracy
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        mask = (projected > 0) & (projected < 1)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0, 1)
    return projected

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj[mask] = np.clip(proj[mask], 0, 1)
    return proj

def obj_grad_fast(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = corr.max()

    w = exps / Z
    fft_len = 4 * n
    H = np.fft.rfft(h, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    conv1 = np.fft.irfft(H * W, n=fft_len)
    H_rev = np.fft.rfft(h[::-1], n=fft_len)
    conv2 = np.fft.irfft(H_rev * W, n=fft_len)
    S1 = conv1[n - 1:2 * n - 1]
    S2 = conv2[n - 1:2 * n - 1][::-1]
    grad = dx * (1.0 - S1 - S2)
    return obj, grad
