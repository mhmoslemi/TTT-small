    return np.max(corr)

    m = np.max(x)
    if T < 1e-6:
        return m
    return m + np.log(np.sum(np.exp(T * (x - m)))) / T

def grad_smooth_max(x, T, smax):
    # gradient of smooth_max with respect to x
    shifted = T * (x - smax)
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    return weights  # d smooth_max / d x_k = weights_k

    s = np.sum(h)

import numpy as np
import time
from scipy.optimize import minimize

def _smooth_max(x, T):
    m = x.max()
    if T < 1e-8:
        e = np.zeros_like(x)
        e[np.argmax(x)] = 1.0
        return m, e
    ex = np.exp(T * (x - m))
    s = ex.sum()
    return m + np.log(s) / T, ex / s

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            rs = r.sum()
            if rs < 1e-15:
                break
            h = h + d * r / rs
        else:
            r = h.copy()
            rs = r.sum()
            if rs < 1e-15:
                break
            h = h + d * r / rs
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def _grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    smax, w = _smooth_max(corr, T)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        jmin = max(0, k)
        jmax = min(n - 1, n - 1 + k)
        for j in range(jmin, jmax + 1):
            g = 1.0 - h[j - k]
            jk = j + k
            if jmin <= jk <= jmax:
                g -= h[jk]
            grad[j] += wm * g
    return smax, grad

def _optimize(h, n, T, max_iter, t0, budget):
    dx = 2.0 / n
    best_c5 = _c5(h)
    best_h = h.copy()
    lr = 1.0 / (n * T) * 4
    prev_c5 = best_c5
    stall = 0
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 30:
            break
        smax, grad = _grad(h, T, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Step with backtracking
        step = lr / gn
        h_new = _project(h - step * grad, n / 2.0)
        c5_new = _c5(h_new)

            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
            lr = min(lr * 1.1, 2.0 / n)
        else:
            stall += 1
            lr *= 0.7
            if stall > 50:

        h = h_new
    
    return best_h, best_c5

    n2 = n * factor
    # Linear interpolation
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n2, endpoint=False)

        return m, e
    ex = np.exp(T * (x - m))
    s = ex.sum()
    return m + np.log(s) / T, ex / s

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - h
            rs = r.sum()
            if rs < 1e-15:
                break
            h = h + d * r / rs
        else:
            r = h.copy()
            rs = r.sum()
            if rs < 1e-15:
                break
            h = h + d * r / rs
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    smax, w = _smooth_max(corr, T)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        jmin = max(0, k)
        jmax = min(n - 1, n - 1 + k)
        for j in range(jmin, jmax + 1):
            g = 1.0 - h[j - k]
            jk = j + k
            if jmin <= jk <= jmax:
                g -= h[jk]
            grad[j] += wm * g
    return smax, grad

def _pgd(h, n, T, max_iter, t0, budget):
    best_c5 = _c5(h)
    best_h = h.copy()
    h_cur = h.copy()
    lr = 4.0 / (n * T)
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > budget - 30:
            break
        smax, grad = _grad(h_cur, T, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = _project(h_cur - lr * grad / gn, n / 2.0)
        c5_new = _c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
            lr = min(lr * 1.05, 3.0 / n)
        else:
            stall += 1
            lr *= 0.75
            if stall > 80:
                break
        h_cur = h_new
    return best_h, best_c5

def _slsqp(h, n, T, t0, budget, max_iter=200):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        sm, _ = _smooth_max(corr, T)
        return sm
    def og(x):
        _, g = _grad(x, T, n)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=og, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.fun < _c5(h) + 1e-8:
            return _project(res.x, n / 2.0)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except (NameError, UnboundLocalError):
        n = 45
        h = np.full(n, 0.5)

    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)

    # Phase 1: Multi-T projected gradient
    for T in [3, 5, 10, 20, 50, 100, 200, 400]:
        if time.time() - t0 > budget_s - 120:
            break
        h, c5 = _pgd(h, n, T, max_iter=2000, t0=t0, budget=budget_s)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: SLSQP polish
    for T in [20, 100, 500]:
        if time.time() - t0 > budget_s - 60:
            break
        h2 = _slsqp(best_h, n, T, t0, budget_s, max_iter=300)
        c5_2 = _c5(h2)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            best_h, best_c5 = _pgd(best_h, n, 100, 500, t0, budget_s)

    # Phase 3: Random restarts
    for attempt in range(10):
        if time.time() - t0 > budget_s - 60:
            break
        scale = 0.05 + 0.15 * attempt / 10.0
        h_p = best_h + rng.randn(n) * scale
        h_p = _project(h_p, n / 2.0)
        h_p, c5_p = _pgd(h_p, n, 20, 500, t0, budget_s)
        if c5_p < best_c5:
            best_h = h_p.copy()
            best_c5 = c5_p
            best_h, best_c5 = _pgd(best_h, n, 200, 500, t0, budget_s)

    # Phase 4: Try structured initializations
    x = np.linspace(0, 2, n, endpoint=False)
    structured = [
        np.clip(0.5 + 0.5 * np.cos(np.pi * (x - 1.0)), 0, 1),
        np.clip(0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x), 0, 1),
        np.where(x < 1.0, np.clip(1.0 - 0.5 * x, 0, 1), np.clip(0.5 * (2.0 - x), 0, 1)),
    ]
    for h_s in structured:
        if time.time() - t0 > budget_s - 60:
            break
        h_s = _project(h_s, n / 2.0)
        h_s, c5_s = _pgd(h_s, n, 10, 1000, t0, budget_s)
        h_s, c5_s = _pgd(h_s, n, 100, 500, t0, budget_s)
        if c5_s < best_c5:
            best_h = h_s.copy()
            best_c5 = c5_s

    # Phase 5: Final polish with SLSQP + PGD
    if time.time() - t0 < budget_s - 30:
        best_h = _slsqp(best_h, n, 200, t0, budget_s, max_iter=500)
        best_h, best_c5 = _pgd(best_h, n, 500, 500, t0, budget_s)

    # Phase 6: Upsample to 2n if time permits
    if time.time() - t0 < budget_s - 90:
        n2 = n * 2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = _project(h2, n2 / 2.0)
        c5_2 = _c5(h2)
        if c5_2 <= best_c5 + 0.02:
            h2, c5_2 = _pgd(h2, n2, 20, 2000, t0, budget_s)
            h2, c5_2 = _pgd(h2, n2, 200, 1000, t0, budget_s)
            if c5_2 < best_c5:
                best_h = h2
                best_c5 = c5_2
                n = n2

    best_c5 = _c5(best_h)
    return best_h.tolist(), best_c5, len(best_h)
