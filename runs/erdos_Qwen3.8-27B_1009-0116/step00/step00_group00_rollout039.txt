Now let me write the truly final version. Let me also make the code a bit more robust by adding a time check inside the inner loops.

Actually, I want to reconsider the learning rate strategy for PGD. The standard approach is:
- If the objective decreases: keep or increase lr
- If the objective increases: decrease lr

But for non-smooth objectives (max of smooth functions), the standard step-size strategy might not work well. A better approach is backtracking line search:
- Start with lr
- If f(h - lr*g) > f(h) + c*lr*||g||², reduce lr and try again

Let me implement a proper backtracking line search:

```python
def do_pgd(h, tgt, max_iters, t0, frac, budget_s):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    lr = 0.1
    for _ in range(max_iters):
        if time.time() - t0 > frac * budget_s:
            break
        _, g = c5_gradient(cur)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-14:
            break
        # Backtracking line search
        trial_lr = lr
        improved = False
        for bt in range(20):
            new_h = project_cv(cur - trial_lr * g, tgt)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-12:
                cur = new_h
                cur_c5 = new_c5
                lr = min(trial_lr * 1.1, 0.5)
                improved = True
                break
            trial_lr *= 0.5
        if not improved:
            break
    return cur, cur_c5
```

This is more robust but potentially slower (up to 20 evaluations per iteration). In practice, the backtracking should converge in 1-3 steps.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_star = int(np.argmax(corr))
    k = m_star - (n - 1)
    grad = np.zeros(n)
    sA = max(0, k)
    eA = min(n, n + k)
    if eA > sA:
        grad[sA:eA] += 1.0 - h[sA - k:eA - k]
    sB = max(0, -k)
    eB = min(n, n - k)
    if eB > sB:
        grad[sB:eB] -= h[sB + k:eB + k]
    return float(corr[m_star] * dx), grad * dx


def project_cv(v, target):
    lo = float(np.min(v)) - 1.0
    hi = 1.0 - float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)


def interp_h(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def slsqp_obj(xv):
    return compute_c5(xv)


def slsqp_grad(xv):
    return c5_gradient(xv)[1]


def do_pgd(h, tgt, max_iters, t0, frac, budget_s):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    lr = 0.1
    for _ in range(max_iters):
        if time.time() - t0 > frac * budget_s:
            break
        _, g = c5_gradient(cur)
        if np.dot(g, g) < 1e-16:
            break
        step = lr
        ok = False
        for _ in range(15):
            new_h = project_cv(cur - step * g, tgt)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-13:
                cur, cur_c5 = new_h, new_c5
                lr = min(step * 1.05, 0.4)
                ok = True
                break
            step *= 0.5
        if not ok:
            break
    return cur, cur_c5
