import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    s = h.sum()
    target = n / 2.0
    if abs(s - target) < 1e-12:
        return h
    # Simple adjustment
    diff = target - s
    # Distribute evenly
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If still not feasible, do iterative adjustment
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1
        else:
            mask = h > 0
        if mask.sum() == 0:
            break
        adj = diff / mask.sum()
        h[mask] += adj
        h = np.clip(h, 0, 1)
    return h

def optimize_subgradient(h, n, max_iter=5000, lr_init=0.01):
    """Subgradient optimization for min max_k c_k(h)"""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    lr = lr_init
    for it in range(max_iter):
        # Compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        c5_vals = corr * dx
        k_max = np.argmax(c5_vals)
        current_c5 = c5_vals[k_max]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute subgradient of c_{k_max}
        # c_k = sum_i h[i] * (1-h[i+k]) for valid indices
        # dc_k/dh[j] = (1-h[j-k]) for j in [k, n-1] (if k>=0)
        #              + (-h[j+k]) for j in [0, n-1-k] (if k>=0)
        # Need to handle np.correlate convention
        
        # np.correlate(h, 1-h, mode='full')[k] where k is 0..2n-2
        # corresponds to lag k-(n-1)
        lag = k_max - (n - 1)
        
        grad = np.zeros(n)
        if lag >= 0:
            # c = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            # d/dh[j]: if j >= lag and j <= n-1: (1-h[j-lag])
            #          if j >= 0 and j <= n-1-lag: -h[j+lag]
            for j in range(lag, n):
                grad[j] += (1 - h[j - lag])
            for j in range(0, n - lag):
                grad[j] += (-h[j + lag])
        else:
            ll = -lag
            # c = sum_{i=ll}^{n-1} h[i] * (1-h[i-ll])

            for j in range(ll, n):
                grad[j] += (1 - h[j - ll])
            for j in range(0, n - ll):
                grad[j] += (-h[j + ll])
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm * gnorm  # keep as is
        
        # Subgradient step
        h = h - lr * grad * dx  # scale by dx
        h = project_feasible(h, n)
        
        # Decrease learning rate
        lr *= 0.999
        if lr < 1e-5:
            lr = 1e-5
    
    return best_h, best_c5

from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(c_vals, eps=0.01):
    """Smooth maximum approximation"""
    c_scaled = c_vals / eps
    c_max = np.max(c_scaled)
    return eps * (c_max + np.log(np.sum(np.exp(c_scaled - c_max))))

def compute_c5_smooth(h, eps=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    return smooth_max(corr, eps)

def compute_grad_smooth(h, eps=0.01):
    """Gradient of smooth max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    
    # Weights from softmax
    c_scaled = corr / eps
    c_max = np.max(c_scaled)
    weights = np.exp(c_scaled - c_max)
    weights /= weights.sum()

    # c_k = dx * sum_i h[i] * (1-h[i+lag_k]) (for appropriate lag)

    for k in range(len(corr)):

            continue
        lag = k - (n - 1)
        if lag >= 0:
            # c_k = dx * sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            for j in range(lag, n):

            for j in range(0, n-lag):

This is O(n) per swap (vectorized), and we have n²/4 swaps, so O(n³) total per sweep. For n=80, that's about 128K operations per sweep (vectorized). Very fast.
