                    grad[:m] += wk * (1.0 - hv[s:s + m])
                    grad[s:s + m] -= wk * hv[:m]
            else:
                t_s = -s
                m = n - t_s
                if m > 0:
                    grad[t_s:t_s + m] += wk * (1.0 - hv[:m])
                    grad[:m] -= wk * hv[t_s:t_s + m]
        grad *= dx_p
        approx_c5 = (max_c + np.log(exps.sum()) / beta) * dx_p
        return grad, float(approx_c5)

    def worst_k_grad(hv, n_pts):
        dx_p = 2.0 / n_pts
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        k_max = int(np.argmax(corr))
        s = k_max - (n_pts - 1)
        g = np.zeros(n_pts)
        if s >= 0:
            m = n_pts - s
            if m > 0:
                g[:m] = (1.0 - hv[s:s + m])
                g[s:s + m] -= hv[:m]
        else:
            t_s = -s
            m = n_pts - t_s
            if m > 0:
                g[t_s:t_s + m] = (1.0 - hv[:m])
                g[:m] -= hv[t_s:t_s + m]
        g *= dx_p
        return g

    h = project(h_init.copy(), n0)
    best_c5 = c5_val(h, n0)
    best_h = h.copy()

    def optimize(hv, n_pts, lr_init, max_iters, deadline, beta=40.0, use_smooth=True):
        c5_cur = c5_val(hv, n_pts)
        best_l = c5_cur
        best_lh = hv.copy()
        lr = lr_init
        mom = np.zeros(n_pts)
        beta_m = 0.8
        stall = 0
        best_lr = lr_init

        for it in range(max_iters):
            if time.time() > deadline:
                break
            if use_smooth:
                g = smooth_grad(hv, n_pts, beta)[0]
            else:
                g = worst_k_grad(hv, n_pts)

            mom = beta_m * mom + (1.0 - beta_m) * g
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-10:
                break
            hv_new = hv - lr * mom
            hv_new = project(hv_new, n_pts)

            c5_new = c5_val(hv_new, n_pts)
            if c5_new < c5_cur - 1e-12:
                hv = hv_new
                c5_cur = c5_new
                stall = 0
                if c5_cur < best_l:
                    best_l = c5_cur
                    best_lh = hv.copy()
                    best_lr = lr
                lr = min(lr * 1.003, 0.4)
            else:
                stall += 1
                lr *= 0.97
                if lr < 0.0002:
                    break
                if stall > 150:
                    hv = best_lh + rng.normal(0, 0.04, n_pts)
                    hv = project(hv, n_pts)
                    lr = max(best_lr, 0.03)
                    mom = np.zeros(n_pts)
                    stall = 0

        return best_l, best_lh

    # Phase 1: Main optimization at original resolution
    d1 = t0 + 0.20 * budget_s
    best_c5, best_h = optimize(h, n0, 0.06, 200000, d1, beta=40)

    # Phase 2: Multiple perturbation restarts
    d2 = t0 + 0.55 * budget_s
    n_restarts = 12
    for r in range(n_restarts):
        if time.time() > d2:
            break
        noise_scale = 0.05 + 0.03 * r / max(n_restarts - 1, 1)
        pert = best_h + rng.normal(0, noise_scale, n0)
        pert = project(pert, n0)
        dl = min(time.time() + 0.03 * budget_s, d2)
        c5_r, h_r = optimize(pert, n0, 0.04, 50000, dl, beta=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 3: Higher resolution interpolation + optimization
    d3 = t0 + 0.82 * budget_s
    for mult in [2, 3, 4]:
        if time.time() > d3:
            break
        n2 = n0 * mult
        if n2 > 400:
            continue
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False) + 1.0 / n0
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False) + 1.0 / n2
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project(h2, n2)
        dl = min(time.time() + 0.18 * budget_s, d3)
        c5_2, h_2 = optimize(h2, n2, 0.025, 40000, dl, beta=25)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_2

    # Phase 4: Final fine-tuning
    d4 = t0 + budget_s - 5
    n_f = len(best_h)
    if time.time() < d4:
        best_c5, best_h = optimize(best_h, n_f, 0.008, 80000, d4, beta=60, use_smooth=True)

    # Final validation
    n_final = len(best_h)
    best_h = project(best_h, n_final)
    final_c5 = c5_val(best_h, n_final)

    return best_h, final_c5, n_final
