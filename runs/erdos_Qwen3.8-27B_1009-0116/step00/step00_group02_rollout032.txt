```python
import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def _smooth_max(v, t):
    if t < 1e-14:
        return float(np.max(v))
    m = np.max(v)
    return float(m + t * np.log(np.sum(np.exp((v - m) / t))))

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _optimize_n(h_init, n, t_end, time_limit):
    dx = 2.0 / n
    h = _project(h_init, n)
    best_c5 = _compute_c5(h)
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
```

```python
    max_val = np.max(corr)

            if 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] += g
    grad /= len(top_idx)
    return grad, float(max_val)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h + m, 0, 1)) > tgt:
            hi = m
        else:
            lo = m
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def _smax(v, t):
    if t < 1e-14:
        return float(np.max(v))
    m = np.max(v)
    return float(m + t * np.log(np.sum(np.exp((v - m) / t))))

def _slsqp(h0, n, temps, dl):
    dx = 2.0 / n
    h = _proj(h0, n)
    bc = _c5(h)
    for tp in temps:
        if time.time() > dl:
            break
        def f(x):
            return _smax(np.correlate(x, 1.0 - x, mode="full") * dx, tp)
        try:
            r = minimize(f, h, method='SLSQP', bounds=[(0,1)]*n,
                         constraints=[{'type':'eq','fun':lambda x: np.sum(x)-n*0.5}],
                         options={'maxiter':300,'ftol':1e-12})
            hc = _proj(r.x, n)
            c = _c5(hc)
            if c < bc:
                bc, h = c, hc
        except Exception:
            pass
    return h, bc

def _coord(h, n, dl, sw=30, gn=30):
    dx = 2.0 / n
    bc = _c5(h)
    grid = np.linspace(0, 1, gn)
    for s in range(sw):
        if time.time() > dl:
            break
        imp = False
        for j in range(n):
            old = h[j]
            bh = old
            bv = np.inf
            for gv in grid:
                if abs(gv - old) < 1e-8:
                    continue
                h[j] = gv
                c = _c5(h)
                if c < bv:
                    bv = c
                    bh = gv
            h[j] = old  # Reset before trying old value
            # Also check old value
            h[j] = old
            c = _c5(h)
            if c < bv:
                bv = c
                bh = old
            h[j] = bh
            if bh != old:
                imp = True
        h = _proj(h, n)
        c = _c5(h)
        if c < bc:
            bc = c
        if not imp:
            break
    return h, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + budget_s
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)
        h0[:22] = 1.0
        h0[22] = 0.5
    h0 = _proj(h0, len(h0))
    
    bh = h0.copy()
    bn = len(bh)
    bc = _c5(bh)
    
    # SLSQP optimization at n
    h, c = _slsqp(bh, bn, [0.05,0.02,0.01,0.005,0.002,0.001,0.0005,0.0001], end-200)
    if c < bc: bh, bc = h, c
    # Coordinate descent
    h, c = _coord(bh, bn, end-180)
    if c < bc: bh, bc = h, c
    
    # Try larger n
    for nt in [60, 90, 120, 180, 270, 400]:
        if time.time() > end - 60:
            break
        xo = np.linspace(0, 2, bn)
        xn = np.linspace(0, 2, nt)
        hi = np.interp(xn, xo, bh)
        h, c = _slsqp(hi, nt, [0.02,0.01,0.005,0.002,0.001,0.0001], end-60)
        if c < bc:
            bh, bc, bn = h, c, nt
        h, c = _coord(bh, bn, end-50)
        if c < bc:
            bh, bc = h, c
    
    # Random restarts
    for _ in range(10):
        if time.time() > end - 15:
            break
        hr = _proj(bh + 0.08*rng.randn(bn), bn)
        h, c = _slsqp(hr, bn, [0.01,0.005,0.002,0.001,0.0001], end-15)
        if c < bc: bh, bc = h, c
        h, c = _coord(bh, bn, end-10)
        if c < bc: bh, bc = h, c
    
    bh = _proj(bh, bn)
    bc = _c5(bh)
    return (bh, bc, bn)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h + m, 0, 1)) > tgt:
            hi = m
        else:
            lo = m
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def _smax(v, t):
    if t < 1e-14:
        return float(np.max(v))
    m = np.max(v)
    return float(m + t * np.log(np.sum(np.exp((v - m) / t))))

def _slsqp(h0, n, temps, dl):
    dx = 2.0 / n
    h = _proj(h0, n)
    bc = _c5(h)
    bnds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n * 0.5}]
    for tp in temps:
        if time.time() > dl:
            break

            return _smax(np.correlate(x, 1.0 - x, mode="full") * dx, tp)
        try:
            r = minimize(f, h, method='SLSQP', bounds=bnds, constraints=con,
                         options={'maxiter': 300, 'ftol': 1e-10})
            hc = _proj(r.x, n)
            c = _c5(hc)
            if c < bc:
                bc, h = c, hc
        except Exception:
            pass
    return h, bc
```
