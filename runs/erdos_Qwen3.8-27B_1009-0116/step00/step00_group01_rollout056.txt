            h_new = project_feasible(res.x, _TS)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-10:
                best_c5, best_h = c5, h_new.copy()
        except Exception:
            pass

    # Phase 4: Random restarts with local optimization
    for it in range(40):
        if remaining() < 30:
            break
        h_rand = project_feasible(rng.rand(n), _TS)
        c5r = compute_c5(h_rand)
        if c5r < best_c5:
            best_c5, best_h = c5r, h_rand.copy()
        try:
            res = minimize(compute_c5, h_rand, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 400, 'ftol': 1e-12})
            h_new = project_feasible(res.x, _TS)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-10:
                best_c5, best_h = c5, h_new.copy()
        except Exception:
            pass

    # Phase 5: Upsample and refine
    for factor in [2, 4, 8]:
        if remaining() < 120:
            break
        n2 = len(best_h) * factor
        if n2 > 500:
            break
        _TS = n2 / 2.0
        x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
        x_new = np.arange(n2) * (2.0 / n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n2 / 2.0)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        try:
            iters = 2000 if n2 <= 100 else 1000 if n2 <= 200 else 500
            res = minimize(compute_c5, h_up, method='SLSQP', bounds=bounds2,
                          constraints=cons2, options={'maxiter': iters, 'ftol': 1e-14})
            h_new = project_feasible(res.x, _TS)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-10:
                best_c5, best_h = c5, h_new.copy()
        except Exception:
            pass
        # Subgradient pass at higher res
        if remaining() > 30 and len(best_h) == n2:
            step_init = 0.03
            for it in range(1000):
                if remaining() < 10:
                    break
                step = step_init / (1.0 + it * 0.02)
                best_h = subgradient_step(best_h, step)
                best_h = project_feasible(best_h, _TS)
                c5 = compute_c5(best_h)
                if c5 < best_h[0] * 1e10:  # sanity check
                    pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
