from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(x, n, alpha, mu):
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full") * alpha
    m = np.max(corr)
    base = (m + np.log(np.sum(np.exp(corr - m))) / alpha) * (2.0 / n)
    penalty = mu * (np.sum(x) - n / 2.0) ** 2
    return base + penalty

def smooth_grad(x, n, alpha, mu):
    h = x
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * alpha
    m = np.max(corr)
    w = np.exp(corr - m)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] += wk * (-h[lo2 + lag:hi2 + lag])
    return grad * dx + 2.0 * mu * (np.sum(x) - n / 2.0) * np.ones(n)

def normalize_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 0:
        h *= (n / 2.0) / s
    return np.clip(h, 0.0, 1.0)

def exact_normalize(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-10
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    else:
        low = h
        mask = low > 1e-10
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return normalize_h(h_new, n_new)

    def try_update(h, n):
        nonlocal best_h, best_c5, best_n
        h = exact_normalize(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    mu = 500.0

    # Phase 1: Multi-alpha L-BFGS-B at initial n
    for alpha in [30, 80, 200]:
        if remaining() < 40:
            break
        bounds = [(0.0, 1.0)] * best_n
        for restart in range(3):
            if remaining() < 20:
                break
            if restart == 0:
                h_start = best_h.copy()
            else:
                h_start = best_h + rng.normal(0, 0.02, best_n)
                h_start = normalize_h(h_start, best_n)
            try:
                res = minimize(
                    smooth_obj, h_start,
                    args=(best_n, alpha, mu),
                    jac=smooth_grad,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10}
                )
                h_opt = exact_normalize(res.x, best_n)
                try_update(h_opt, best_n)
            except Exception:
                pass

    # Phase 2: 2-opt local search (preserves sum exactly)
    if remaining() > 30:
        h_ls = best_h.copy()
        n_ls = best_n
        c5_cur = compute_c5(h_ls, n_ls)
        steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
        for base_step in steps:

                break
            n_iter = 8000 if base_step > 0.01 else 4000
            for it in range(n_iter):
                if remaining() < 5:
                    break
                i = rng.integers(0, n_ls)
                j = rng.integers(0, n_ls)
                if i == j:
                    continue
                amt = base_step * (0.5 + 0.5 * rng.random()) * rng.choice([-1.0, 1.0])
                ni = h_ls[i] + amt
                nj = h_ls[j] - amt
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h_ls[i] = ni
                    h_ls[j] = nj
                    c5_try = compute_c5(h_ls, n_ls)
                    if c5_try < c5_cur:
                        c5_cur = c5_try
                        if c5_cur < best_c5:

                            best_h = h_ls.copy()
                    else:
                        h_ls[i] = ni - amt
                        h_ls[j] = nj + amt

            for alpha in [50, 150]:

                    break
                try:
                    res = minimize(
                        smooth_obj, h_try,
                        args=(n_try, alpha, mu),
                        jac=smooth_grad,
                        method='L-BFGS-B',
                        bounds=bounds_try,
                        options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-9}
                    )
                    h_opt = exact_normalize(res.x, n_try)
                    try_update(h_opt, n_try)
                    h_try = best_h  # restart from best
                except Exception:
                    pass

    # Phase 4: Final polish local search on best
    if remaining() > 20:
        h_polish = best_h.copy()
        n_pol = best_n
        c5_pol = compute_c5(h_polish, n_pol)
        for base_step in [0.003, 0.001, 0.0005]:
            if remaining() < 10:
                break
            for it in range(10000):
                if remaining() < 5:
                    break
                i = rng.integers(0, n_pol)
                j = rng.integers(0, n_pol)
                if i == j:
                    continue
                amt = base_step * rng.choice([-1.0, 1.0])
                ni = h_polish[i] + amt
                nj = h_polish[j] - amt
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h_polish[i] = ni
                    h_polish[j] = nj
                    c5_try = compute_c5(h_polish, n_pol)
                    if c5_try < c5_pol:
                        c5_pol = c5_try
                        if c5_pol < best_c5:
                            best_c5 = c5_pol
                            best_h = h_polish.copy()
                    else:
                        h_polish[i] -= amt
                        h_polish[j] += amt

    # Final exact normalization
    best_h = exact_normalize(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return (best_h.tolist(), best_c5, best_n)
