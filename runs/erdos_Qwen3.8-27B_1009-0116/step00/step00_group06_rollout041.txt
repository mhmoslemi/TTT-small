        dx2 = 2.0 / n2
        x_coarse = np.linspace(0.0, 2.0, n)
        x_fine = np.linspace(0.0, 2.0, n2)
        h_fine = np.interp(x_fine, x_coarse, h_best)
        h_fine = repair(h_fine, n2)

        bounds2 = [(0.0, 1.0)] * n2
        target2 = n2 / 2.0

        for attempt in range(3):
            if time.time() - t0 >= budget_s * 0.80:
                break
            h_start = h_fine.copy()
            if attempt > 0:
                h_start += rng.uniform(-0.05, 0.05, n2)
                h_start = repair(h_start, n2)

            try:
                res2 = minimize(
                    lambda x, a=0.5: smooth_obj(x, dx2, a),
                    h_start,
                    method='SLSQP',
                    bounds=bounds2,
                    constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target2}],
                    options={'maxiter': 200, 'ftol': 1e-9}
                )
                if res2.x is not None and np.all(np.isfinite(res2.x)):
                    c5_fine = compute_c5(res2.x, dx2)
                    if c5_fine < c5_best:
                        c5_best = c5_fine
                        h_best = res2.x.copy()
                        n = n2
                        dx = dx2
                        h_fine = res2.x.copy()
            except Exception:
                pass

    # Phase 3: Final polish on best grid
    if time.time() - t0 < budget_s * 0.95:
        bounds_f = [(0.0, 1.0)] * n
        target_f = n / 2.0
        try:
            res_f = minimize(
                lambda x: smooth_obj(x, dx, 0.3),
                h_best.copy(),
                method='SLSQP',
                bounds=bounds_f,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_f}],
                options={'maxiter': 300, 'ftol': 1e-11}
            )
            if res_f.x is not None and np.all(np.isfinite(res_f.x)):
                c5_f = compute_c5(res_f.x, dx)
                if c5_f < c5_best:
                    c5_best = c5_f
                    h_best = res_f.x.copy()
        except Exception:
            pass

    # Final repair and evaluation
    h_best = repair(h_best, n)
    dx_final = 2.0 / n
    c5_final = compute_c5(h_best, dx_final)

    return h_best, c5_final, n
