from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = n / 2.0 - h.sum()
        if abs(diff2) > 1e-10:
            if diff2 > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0)[0]
            if len(idx) > 0:
                h[idx] += diff2 / len(idx)
                h = np.clip(h, 0.0, 1.0)
    return h

def binarize_continuous(h, n):
    """Convert continuous h to near-binary maintaining sum=n/2"""
    target = n / 2.0
    # Sort indices by h value descending
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    # Take top floor(target) as 1
    full = int(target)
    frac = target - full
    h_bin[idx[:full]] = 1.0
    if frac > 1e-10:
        h_bin[idx[full]] = frac
    return h_bin

    """Greedy pair-flip local search"""
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if max_time is not None and time.time() - t0 > max_time:
            break
        ones = list(np.where(h > 0.999)[0])
        zeros = list(np.where(h < 0.001)[0])
        if not ones or not zeros:
            break
        
        improved = False
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    improved = True
                    ones = list(np.where(h > 0.999)[0])
                    zeros = list(np.where(h < 0.001)[0])
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break

            break

    h = h.copy()

        zeros = np.where(h < 0.001)[0]

        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        ii, jj = ones[i], zeros[j]

        h[jj] = 1.0
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - cur_c5

            if new_c5 < best_c5:

                best_h = h.copy()

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = n / 2.0 - h.sum()
        if abs(diff2) > 1e-10:
            if diff2 > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff2 / len(idx)
                h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    target = n / 2.0
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    full = int(target)
    frac = target - full
    h_bin[idx[:full]] = 1.0
    if frac > 1e-10:
        h_bin[idx[full]] = frac
    return h_bin

def local_search(h, n, dx, max_sweeps, deadline):

        if time.time() >= deadline:
            break
        ones = list(np.where(h > 0.999)[0])

            break

        for i in ones:
            for j in zeros:

                h[j] = 1.0
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
