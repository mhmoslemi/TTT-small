from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute among non-bound values
        mask = (h > 1e-6) & (h < 1 - 1e-6)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # All at bounds, do a more careful distribution
            if diff > 0:
                # Need to increase some
                idx = np.argsort(1.0 - h)[:int(np.ceil(diff))]
                h[idx] = np.minimum(1.0, h[idx] + diff / len(idx))
            else:
                idx = np.argsort(h)[:int(np.ceil(-diff))]
                h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return h

def binarize(h, n_ones):
    """Convert to 0/1 vector with exactly n_ones ones."""
    h_sorted = np.sort(h)[::-1]

    if n_actual > n_ones:

        # Remove ones with lowest h values
        remove = np.array(sorted(ones, key=lambda i: h[i]))[:n_actual - n_ones]
        h_bin[remove] = 0
    elif n_actual < n_ones:
        zeros = np.where(h_bin == 0)[0]
        add = np.array(sorted(zeros, key=lambda i: -h[i]))[:n_ones - n_actual]
        h_bin[add] = 1
    return h_bin

However, for the 0/1 case, we can reformulate:
overlap(k) = sum_j h[j]*(1-h[j+k]) = sum_j h[j] - sum_j h[j]*h[j+k]

For 0/1 h: sum_j h[j] = n/2 (fixed), so minimizing max_k overlap(k) is equivalent to maximizing min_k sum_j h[j]*h[j+k] (autocorrelation).

The autocorrelation sum_j h[j]*h[j+k] for 0/1 h is just the Hamming correlation of the binary sequence.

This is a well-studied problem in sequence design. The optimal binary sequences for minimizing maximum autocorrelation are related to:
- M-sequences (maximum length sequences)
- Barker sequences
- Costas arrays
- Golay complementary sequences

For our specific problem (minimizing max cross-correlation with complement = maximizing min autocorrelation at non-zero lags), the optimal sequences are those with the most "balanced" autocorrelation spectrum.

def binarize(h, n_ones):
    h_bin = (h >= 0.5).astype(float)
    n_actual = int(h_bin.sum())
    if n_actual > n_ones:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[order][:n_actual - n_ones]] = 0
    elif n_actual < n_ones:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order][:n_ones - n_actual]] = 1
    return h_bin

def local_search_01(h, rng, max_steps=3000, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    steps = 0
    while steps < max_steps:
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        improved = False
        for _ in range(100):
            oi = ones[rng.randint(len(ones))]
            zi = zeros[rng.randint(len(zeros))]
            if oi == zi:
                continue
            h[oi], h[zi] = h[zi], h[oi]
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                h[oi], h[zi] = h[zi], h[oi]
        steps += 100
        if not improved:
            break
    return best_h, best_c5

def sa_01(h, T0, T1, n_iters, rng, time_limit=None, start_time=None):
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for it in range(n_iters):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iters - 1))
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        if oi == zi:
            continue
        h[oi], h[zi] = h[zi], h[oi]
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            c5 = c5_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[oi], h[zi] = h[zi], h[oi]
    return best_h, best_c5

def perturb(h, k, rng):
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(k):
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        if oi != zi:
            h[oi], h[zi] = h[zi], h[oi]
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    n0 = len(initial_h_values)
    h0 = initial_h_values.astype(float).copy()
    target0 = n0 / 2.0
    n_ones = n0 // 2

    # Ensure feasibility of initial
    h0 = project_feasible(h0, target0)
    c5_init = compute_c5(h0)
    best_c5 = c5_init
    best_h = h0.copy()

    # Binarize
    h_bin = binarize(h0, n_ones)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Iterated local search (ILS) on binary solution
    t_ils = budget_s * 0.5
    h_cur = h_bin.copy()
    for iteration in range(100):
        if (time.time() - start_time) > t_ils:
            break
        h_ls, c5_ls = local_search_01(h_cur, rng, max_steps=2000,
                                       time_limit=t_ils, start_time=start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        k_pert = rng.randint(2, 7)
        h_cur = perturb(h_ls, k_pert, rng)

    # Phase 2: Simulated annealing (multiple restarts)
    t_sa = budget_s * 0.72
    for restart in range(4):
        if (time.time() - start_time) > t_sa:
            break
        n_iters = 150000
        h_start_sa = h_bin.copy() if restart == 0 else perturb(h_bin, rng.randint(3, 10), rng)
        h_sa, c5_sa = sa_01(h_start_sa, 0.05, 0.00002, n_iters, rng,
                            time_limit=t_sa, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Polish after SA
        t_polish = min(budget_s * 0.78, (time.time() - start_time) + 30)
        h_pol, c5_pol = local_search_01(h_sa, rng, max_steps=3000,
                                         time_limit=t_polish, start_time=start_time)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Continuous Nelder-Mead refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.85:
        def obj_func(x):
            return compute_c5(project_feasible(x, target0))

        # Multiple starting points for NM
        for nm_restart in range(3):
            if (time.time() - start_time) > budget_s * 0.88:
                break
            if nm_restart == 0:
                h_try = np.clip(best_h + (rng.rand(n0) - 0.5) * 0.1, 0, 1)
            elif nm_restart == 1:
                h_try = np.clip(best_h + (rng.rand(n0) - 0.5) * 0.3, 0, 1)
            else:
                h_try = binarize(best_h, n_ones).astype(float)
                h_try = np.clip(h_try + rng.rand(n0) * 0.2, 0, 1)
            h_try = project_feasible(h_try, target0)
            res = minimize(obj_func, h_try, method='Nelder-Mead',
                           options={'maxiter': 15000, 'xatol': 1e-10, 'fatol': 1e-12})
            h_res = project_feasible(res.x, target0)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    # Phase 4: Try other n values
    for n_try in [50, 60, 80, 100]:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.92:
            break
        h_init = np.interp(np.linspace(0, len(best_h) - 1, n_try),
                           np.arange(len(best_h)), best_h)
        h_init = project_feasible(h_init, n_try / 2.0)
        h_b = binarize(h_init, n_try // 2)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            continue
        h_ls, c5_ls = local_search_01(h_b, rng, max_steps=800,
                                       time_limit=budget_s * 0.95, start_time=start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final: ensure feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Verify sum
    diff = target_final - best_h.sum()
    if abs(diff) > 1e-9:
        mask = (best_h > 1e-6) & (best_h < 1 - 1e-6)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
