import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = int(mask.sum())
        if cnt == 0:
            return h
        h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
    else:
        mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            return h
        h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h
