import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        h.fill(0.5)
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt == 0:
            if diff > 0:
                mask = (h < 1.0)
            else:
                mask = (h > 0.0)
            cnt = np.sum(mask)
            if cnt == 0:
                break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = vals - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def make_cons(n):
    def cs(x):
        return np.sum(x) - n / 2.0
    return ({'type': 'eq', 'fun': cs},)

def slsqp_solve(h0, n, beta, bounds, cons, maxiter=500):
    try:
        res = minimize(smooth_max_obj, h0, args=(n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = repair(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return None, None

def coordinate_descent(h, n, max_time):
    best_c5 = compute_c5(h, n)
    grid = np.concatenate(([0.0, 1.0], np.linspace(0.01, 0.99, 30)))
    improved = True
    while improved:
        improved = False
        for i in range(n):
            if time.time() > max_time:
                return h, best_c5
            for v in grid:
                h_test = h.copy()
                h_test[i] = v
                h_test = repair(h_test, n)
                c5_t = compute_c5(h_test, n)
                if c5_t < best_c5 - 1e-11:
                    best_c5 = c5_t
                    h = h_test.copy()
                    improved = True
    return h, best_c5
