def softmax_objective(h, dx, eps=0.01):
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Scale for numerical stability
    shifted = (corr - np.max(corr)) / eps
    log_sum_exp = np.max(shifted) + np.log(np.sum(np.exp(shifted)))
    val = eps * (np.max(corr) / eps + log_sum_exp)  # = max(corr) + eps*log(sum(exp(...)))
    # Actually: eps * log(sum(exp(corr/eps))) = eps * (max(corr)/eps + log(sum(exp((corr-max)/eps))))
    # = max(corr) + eps * log(sum(exp((corr-max)/eps)))
    val = np.max(corr) + eps * np.log(np.sum(np.exp((corr - np.max(corr)) / eps)))
    return val * dx

def softmax_gradient(h, dx, eps=0.01):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    shifted = (corr - np.max(corr)) / eps
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient of corr[k] w.r.t. h[j]
    # corr[n-1+s] = sum_{i valid} h[i] * (1-h[i+s])
    # d/dh[j] = (1-h[j+s]) * [j valid as i] - h[j-s] * [j-s valid as i]
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-8:

        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        for i in range(i_lo, i_hi + 1):

                # (1-h[i+s]) factor: d/dh[j] (1-h[i+s]) = -1 if j==i+s
                grad[i + s] += -w * h[i]
    return grad * dx

def corr_gradient(h, weights):
    """Compute gradient of weighted sum of correlations.
    weights: array of length 2n-1, same as corr in 'full' mode.
    Returns gradient w.r.t. h of sum_k weights[k] * corr_full[k].
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if abs(w) < 1e-10:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        # First term: (1 - h[i+s]) for i in [i_lo, i_hi]
        # These contribute to grad[i]
        indices1 = np.arange(i_lo, i_hi + 1)
        if len(indices1) > 0:
            grad[indices1] += w * (1 - h[indices1 + s])
        # Second term: -h[i] for i in [i_lo, i_hi], contributing to grad[i+s]
        indices2 = indices1 + s
        if len(indices2) > 0:
            valid = (indices2 >= 0) & (indices2 < n)
            grad[indices2[valid]] += -w * h[indices1[valid]]
    return grad

def _project(h, n):
    target = n / 2.0
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny numerical error
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        # Adjust an element that can absorb it
        if diff > 0:
            idx = int(np.argmax(proj))
        else:
            idx = int(np.argmin(proj))
        proj[idx] += diff
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def _project(h, n):
    target = n / 2.0
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(proj))
        else:
            idx = int(np.argmin(proj))
        proj[idx] += diff
        proj = np.clip(proj, 0.0, 1.0)
    return proj


def _sm_obj(x, dx, eps):
    corr = np.correlate(x, 1.0 - x, mode="full")
    cmax = np.max(corr)
    return (cmax + eps * np.log(np.sum(np.exp((corr - cmax) / eps)))) * dx


def _sm_grad(x, dx, eps):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    cmax = np.max(corr)
    sh = (corr - cmax) / eps
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-9:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        idx = np.arange(lo, hi + 1)
        grad[idx] += wk * (1.0 - x[idx + s])
        idx2 = idx + s
        v = (idx2 >= 0) & (idx2 < n)
        grad[idx2[v]] -= wk * x[idx[v]]
    return grad * dx

    return np.sum(x) - target

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h = _project(np.clip(res.x, 0, 1), n)
        except Exception:
            pass
    return h
