import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    # Adjust to exactly n/2 ones
    n_ones = int(np.sum(h == 1.0))
    target = n // 2
    if n_ones < target:
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[:target - n_ones]] = 1.0
    elif n_ones > target:
        one_idx = np.where(h == 1)[0]
        h[one_idx[:n_ones - target]] = 0.0
    return h

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

def make_blocks(n, block_sizes, seed):
    """Create a sequence with blocks of 1s and 0s of given sizes."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = 0
    val = 1
    while pos < n:
        bs = min(block_sizes[rng.randint(len(block_sizes))], n - pos)
        h[pos:pos+bs] = val
        pos += bs
        val = 1 - val
    # Fix to have exactly n/2 ones
    n_ones = int(np.sum(h == 1.0))
    target = n // 2
    if n_ones < target:
        zero_idx = np.where(h == 0)[0]
        # Flip some zeros to ones
        for i in range(target - n_ones):
            h[zero_idx[i]] = 1.0
    elif n_ones > target:
        one_idx = np.where(h == 1)[0]
        for i in range(n_ones - target):
            h[one_idx[i]] = 0.0
    return h

def sa_optimize(h, n, time_limit, seed=42):
    """Simulated annealing with swap moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    ones_pos = np.where(h == 1)[0].tolist()
    zeros_pos = np.where(h == 0)[0].tolist()
    start_time = time.time()
    n_iter = 0
    T = 0.5
    
    while time.time() - start_time < time_limit:
        n_iter += 1
        # Cooling
        T = 0.5 * (0.9999 ** n_iter)
        i_idx = rng.randint(len(ones_pos))
        j_idx = rng.randint(len(zeros_pos))
        i, j = ones_pos[i_idx], zeros_pos[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones_pos[i_idx] = j
            zeros_pos[j_idx] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def scale_up(h_small, n_small, n_large):
    """Scale up a binary solution to higher resolution."""
    ratio = n_large / n_small
    h_large = np.zeros(n_large)
