import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s2 = np.sum(h)
        if abs(s2 - target) > 1e-8:
            diff = target - s2
            mask = (h > 0.02) & (h < 0.98)
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, target, max_sweeps=5):
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            best_obj = compute_c5(h)
            best_trial = None
            for v in np.linspace(0.0, 1.0, 15):
                if abs(v - old) < 0.03:
                    continue
                trial = h.copy()
                trial[j] = v
                diff = target - np.sum(trial)
                others = np.where(np.arange(n) != j)[0]
                adj = others[np.argmin(np.abs(trial[others] - 0.5))]
                trial[adj] += diff
                if trial[adj] < -1e-6 or trial[adj] > 1.0 + 1e-6:
                    continue
                trial = np.clip(trial, 0.0, 1.0)
                c5v = compute_c5(trial)
                if c5v < best_obj - 1e-10:
                    best_obj = c5v
                    best_trial = trial
            if best_trial is not None:
                h = best_trial
                improved = True
        h = repair(h, n)
        if not improved:
            break
    return h
