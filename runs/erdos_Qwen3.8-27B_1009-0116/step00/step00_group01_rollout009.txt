                          options={'maxiter': 500, 'ftol': 1e-12})
            h_try = fix_sum(res.x, n)
            val = compute_c5(h_try)[0]
            if val < best_val - 1e-12:
                best_val = val
                best = h_try
        except Exception:
            pass

    # Random perturbations
    n_pert = 8
    for _ in range(n_pert):
        if time.time() - t0 > time_limit:
            break
        scale = 0.03 + 0.03 * rng.rand()
        h_pert = best + rng.randn(n) * scale
        h_pert = fix_sum(h_pert, n)
        if not (np.all(h_pert >= -1e-12) and np.all(h_pert <= 1.0 + 1e-12)):
            continue
        try:
            res = minimize(lambda x: obj_func(x, 12.0), h_pert,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12})
            h_try = fix_sum(res.x, n)
            val = compute_c5(h_try)[0]
            if val < best_val - 1e-12:
                best_val = val
                best = h_try
        except Exception:
            pass

    # Subgradient-style coordinate refinement
    if time.time() - t0 < time_limit:
        for _iter in range(200):
            if time.time() - t0 > time_limit:
                break
            c_vals = np.correlate(best, 1.0 - best, mode="full")
            k_max = np.argmax(c_vals)
            # Determine which indices are affected
            n_shift = n - 1
            shift = k_max - n_shift
            affected = set()
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    affected.add(i)
            # Nudge: move pairs in opposite directions
            count = 0
            for i in affected:
                j = i + shift
                if 0 <= j < n and best[i] < 1.0 and best[j] > 0.0:
                    delta = 0.005
                    best[i] = min(1.0, best[i] + delta)
                    best[j] = max(0.0, best[j] - delta)
                    count += 1
            if count == 0:
                break
            best = fix_sum(best, n)
            val = compute_c5(best)[0]
            if val > best_val + 1e-9:
                best = fix_sum(best - np.sign(best - 0.5) * 0.01, n)
                val = compute_c5(best)[0]
                if val > best_val + 1e-9:
                    break

    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    best_h = fix_sum(initial_h_values.astype(float).copy(), n0)
    best_c5 = compute_c5(best_h)[0]

    # Build resolution ladder
    resolutions = []
    for factor in [1, 2, 4, 8, 16]:
        n = n0 * factor
        if n < 1000 and n not in resolutions:
            resolutions.append(n)

    # Multi-resolution optimization
    for idx, n in enumerate(resolutions):
        remaining = budget_s - (time.time() - t_start)
        if remaining < 15:
            break
        n_remaining_resolutions = len(resolutions) - idx
        t_alloc = min(remaining * 0.7, max(10, remaining / n_remaining_resolutions))

        # Upsample from current best
        src_n = len(best_h)
        h = upsample(best_h, n, src_n)
        h = fix_sum(h, n)

        h_opt, c5_opt = optimize_h(h, n, t_alloc, rng)

        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final polish at best resolution
    remaining = budget_s - (time.time() - t_start)
    if remaining > 8:
        n_best = len(best_h)
        h_opt, c5_opt = optimize_h(best_h.copy(), n_best, remaining * 0.6, rng)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Ensure validity
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)[0]

    return best_h, best_c5, len(best_h)
