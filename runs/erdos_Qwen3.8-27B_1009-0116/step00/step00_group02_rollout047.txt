import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        if s > target:
            # Need to decrease
            diff = s - target
            mask = h > 0
            count = np.sum(mask)
            step = min(diff / count, np.min(h[mask]) if count > 0 else diff)
            h[mask] -= step
        else:
            # Need to increase
            diff = target - s
            mask = h < 1
            count = np.sum(mask)
            step = min(diff / count, 1 - np.max(h[mask]) if count > 0 else diff)
            h[mask] += step
