import numpy as np
from scipy.ndimage import gaussian_filter1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        if np.abs(s - target) < 1e-14:
            break
        if s == 0:
            h[:] = 0.5
            break
        h = np.clip(h * (target / s), 0, 1)
    return h

def subgrad_max(h, n, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    thresh = max_val - 1e-12
    idxs = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for mi in idxs:
        t = abs(mi - (n - 1))
        if t < n:
            grad[:n-t] += (1 - h[t:])
            grad[t:] -= h[:n-t]
    grad *= dx / len(idxs)
    return grad, max_val * dx

def smooth_max_obj(h, n, dx, tau):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Smooth max
    m = np.max(corr)
    shifted = corr - m
    return m + (1.0/tau) * np.log(np.sum(np.exp(tau * shifted)))

def smooth_max_grad(h, n, dx, tau):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    weights = np.exp(tau * shifted) / np.sum(np.exp(tau * shifted))
    # Gradient of smooth max = weighted average of gradients of each component
    grad = np.zeros(n)
    for i, w in enumerate(weights):
        if w < 1e-10:
            continue
        t = abs(i - (n - 1))
        if t < n:
            grad[:n-t] += w * (1 - h[t:])
            grad[t:] -= w * h[:n-t]
    grad *= dx
    return grad, m + (1.0/tau)*np.log(np.sum(np.exp(tau * shifted)))

            h_r, c5 = optimize_anneal(h_r, n, dx, 1500, t0, budget_s, rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h_r

    # Phase D: final fine polish
    if time_left() > 15:
        lr = 0.005
        h = best_h.copy()
        c5 = compute_c5(h, dx)
        for _ in range(10000):
            if time_left() < 5:
                break
            grad, _ = subgrad_max(h, n, dx)
            h_new = project_feasible(h - lr * grad, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5, best_h = c5, h
            else:
                lr *= 0.97

    # Final smooth polish
    if time_left() > 10:
        h, c5 = optimize_anneal(best_h, best_n, 2.0 / best_n, 2000, t0, budget_s, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    final_c5 = compute_c5(best_h, 2.0 / best_n)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
