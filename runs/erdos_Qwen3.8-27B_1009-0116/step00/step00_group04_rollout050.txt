    c = np.correlate(h, 1 - h, mode="full") * dx

    shifted = beta * (c - max_c)
    lse = max_c * beta + np.log(np.sum(np.exp(shifted)))
    return lse / beta

def soft_max_grad(h, n_points, beta=50):
    dx = 2.0 / n_points
    n = n_points
    c = np.correlate(h, 1 - h, mode="full") * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of c[k] w.r.t. h[i]
    # c[k] = sum_j h[j]*(1-h[j+s]) where s = k-(n-1)
    # grad_c[k,i] = (1-h[i+s])*1[i+s<n] - h[i-s]*1[i-s>=0]
    
    # We need: sum_k weights[k] * grad_c[k,i] * dx  for each i
    # = dx * sum_k weights[k] * ((1-h[i+s_k])*1[i+s_k<n] - h[i-s_k]*1[i-s_k>=0])
    # where s_k = k-(n-1)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        # (1-h[i+s]) for i+s in [0,n): i in [-s, n-s)
        # -h[i-s] for i-s in [0,n): i in [s, n+s)
        # 
        # For the first term: valid i are max(0,-s) to min(n-1, n-1-s)
        # For the second term: valid i are max(0,s) to min(n-1, n-1+s)
        
        w = weights[k]
        # First term: (1-h[i+s]) for valid i
        i_min1 = max(0, -s)
        i_max1 = min(n-1, n-1-s)
        if i_min1 <= i_max1:
            idx = np.arange(i_min1, i_max1+1)
            grad[idx] += w * (1 - h[idx + s])
        
        # Second term: -h[i-s] for valid i
        i_min2 = max(0, s)
        i_max2 = min(n-1, n-1+s)
        if i_min2 <= i_max2:
            idx = np.arange(i_min2, i_max2+1)
            grad[idx] += -w * h[idx - s]
    
    grad *= dx
    return grad

from scipy.optimize import minimize
import time

_N = 0
_BETA = 0

def compute_c5(h, n):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(vals) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def soft_max_c5(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(c)
    s = beta * (c - mc)
    return mc + np.log(np.sum(np.exp(s))) / beta

def soft_max_grad(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(c)
    s = beta * (c - mc)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-9:
            continue
        sh = k - (n - 1)
        i1_lo = max(0, -sh)
        i1_hi = min(n - 1, n - 1 - sh)
        if i1_lo <= i1_hi:
            idx1 = np.arange(i1_lo, i1_hi + 1)
            grad[idx1] += w[k] * (1.0 - x[idx1 + sh])
        i2_lo = max(0, sh)
        i2_hi = min(n - 1, n - 1 + sh)
        if i2_lo <= i2_hi:
            idx2 = np.arange(i2_lo, i2_hi + 1)
            grad[idx2] += -w[k] * x[idx2 - sh]
    return grad * dx
