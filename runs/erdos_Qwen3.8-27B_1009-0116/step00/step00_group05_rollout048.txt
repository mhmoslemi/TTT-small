        return m, grad
    scaled = (overlaps - m) / temp
    scaled = np.clip(scaled, -400, 0)
    w = np.exp(scaled)
    w_sum = np.sum(w)
    w /= w_sum
    val = m + temp * np.log(w_sum)
    grad = np.zeros_like(h)
    for k in range(len(overlaps)):
        if w[k] > 1e-7:
            grad += w[k] * subgradient_at(h, k)
    grad *= temp
    return val, grad

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        test = np.clip(h + mid, 0.0, 1.0)
        if np.sum(test) < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    h_proj += (target_sum - np.sum(h_proj)) / len(h_proj)
    return np.clip(h_proj, 0.0, 1.0)

def initial_h_values_fallback(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * 3 * x / 2) + 0.1 * rng.normal(0, 1, n)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    n = 70
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
    else:
        init = initial_h_values_fallback(n, rng)

    target_sum = n / 2.0
    init = project_feasible(init, target_sum)
    best_c5 = get_c5(init)
    best_h = init.copy()

    bounds = [(0.0, 1.0)] * n

    def eq_con_fun(x):
        return np.sum(x) - target_sum

    def eq_con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]

    def time_left_frac():
        return max(0.0, 1.0 - (time.time() - t0) / budget_s)

    n_restarts = 60
    n_anneal = 8
    base_temp = 0.04

    for restart in range(n_restarts):
        if time_left_frac() < 0.12:
            break

        if restart == 0:
            h_curr = best_h.copy()
        elif restart % 5 == 1:
            x = np.linspace(0, 2, n, endpoint=False)
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            h_curr = 0.5 + rng.uniform(0.1, 0.35) * np.sin(2 * np.pi * freq * x / 2 + phase)
            h_curr = project_feasible(h_curr, target_sum)
        elif restart % 5 == 2:
            h_curr = (rng.random(n) > 0.4).astype(float)
            h_curr = project_feasible(h_curr, target_sum)
        else:
            size = min(0.3, 0.1 + 0.2 * restart / n_restarts)
            h_curr = best_h + rng.normal(0, size, n)
            h_curr = project_feasible(h_curr, target_sum)

        for anneal_i in range(n_anneal):
            if time_left_frac() < 0.12:
                break
            temp = base_temp * (0.75 ** anneal_i)
            max_iter = int(50 + 50 * (1 - anneal_i / n_anneal))

            def obj_func(hv):
                return smooth_max_grad(hv, temp)[0]

            def obj_grad_hv(hv):
                return smooth_max_grad(hv, temp)[1]

            try:
                res = minimize(obj_func, h_curr, jac=obj_grad_hv, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': max_iter, 'ftol': 1e-9})
                if res.x is not None:
                    h_new = project_feasible(res.x, target_sum)
                    c5_new = get_c5(h_new)
                    if c5_new < best_c5 - 1e-10:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h_curr = project_feasible(res.x, target_sum)
                else:
                    break
            except Exception:
                try:
                    val, grad = smooth_max_grad(h_curr, temp)
                    lr = 0.01 / (1 + np.linalg.norm(grad))
                    h_curr = project_feasible(h_curr - lr * grad, target_sum)
                except Exception:
                    pass

    # Final polish: a few more SLSQP iterations on best_h
    for _ in range(3):
        if time_left_frac() < 0.05:
            break
        try:
            def obj_final(hv):
                return smooth_max_grad(hv, 0.01)[0]

            def grad_final(hv):
                return smooth_max_grad(hv, 0.01)[1]

            res = minimize(obj_final, best_h, jac=grad_final, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 80, 'ftol': 1e-10})
            if res.x is not None:
                h_pol = project_feasible(res.x, target_sum)
                if get_c5(h_pol) < best_c5:
                    best_c5 = get_c5(h_pol)
                    best_h = h_pol
        except Exception:
            pass

    best_c5 = get_c5(best_h)
    best_h = project_feasible(best_h, target_sum)
    best_c5 = get_c5(best_h)
    return best_h, best_c5, n
