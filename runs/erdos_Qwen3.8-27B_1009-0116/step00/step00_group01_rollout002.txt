    best_n = n_p

    # Try binary projection of parent
    h_bin = binary_project(h_p, n_p)
    c5_bin = compute_c5(h_bin, n_p)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    # Stage 1: SLSQP at parent resolution
    if time.time() - t0 < budget_s * 0.30:
        h1, c5_1 = slsqp_optimize(best_h, best_n, 500, 40.0, budget_s * 0.22)
        if c5_1 < best_c5:
            best_h = h1.copy()
            best_c5 = c5_1
        # Binary project the result
        h1b = binary_project(h1, best_n)
        c5_1b = compute_c5(h1b, best_n)
        if c5_1b < best_c5:
            best_h = h1b.copy()
            best_c5 = c5_1b

    # Stage 2: Multi-start at parent resolution
    for k in range(5):
        if time.time() - t0 > budget_s * 0.50:
            break
        pert = best_h + np.random.randn(best_n) * 0.04
        pert = project_feasible(pert, best_n)
        h2, c5_2 = slsqp_optimize(pert, best_n, 250, 60.0, budget_s * 0.07)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2

    # Also try hat and fejer starts at parent resolution
    for init_fn in (hat_start, fejer_start):
        if time.time() - t0 > budget_s * 0.60:
            break
        h_init = init_fn(best_n)
        h3, c5_3 = slsqp_optimize(h_init, best_n, 300, 50.0, budget_s * 0.06)
        if c5_3 < best_c5:
            best_h = h3.copy()
            best_c5 = c5_3

    # Stage 3: Higher resolutions
    for n_try in (94, 188, 376):
        if time.time() - t0 > budget_s * 0.86:
            break
        h_int = interp_to_resolution(best_h, n_try)
        c5_int = compute_c5(h_int, n_try)

        if n_try <= 100:
            h4, c5_4 = slsqp_optimize(h_int, n_try, 300, 40.0, budget_s * 0.10)
        elif n_try <= 200:
            h4, c5_4 = slsqp_optimize(h_int, n_try, 120, 40.0, budget_s * 0.07)
        else:
            h4, c5_4 = coord_descent(h_int, n_try, 30000, budget_s * 0.06)

        if c5_4 < best_c5:
            best_h = h4.copy()
            best_c5 = c5_4
            best_n = n_try

    # Stage 4: Final coordinate descent refinement
    remaining = budget_s * 0.96 - (time.time() - t0)
    if remaining > 3:
        h5, c5_5 = coord_descent(best_h, best_n, 80000, remaining, rng_seed=seed)
        if c5_5 < best_c5:
            best_h = h5.copy()
            best_c5 = c5_5

    # Final binary projection check
    h6 = binary_project(best_h, best_n)
    c5_6 = compute_c5(h6, best_n)
    if c5_6 < best_c5:
        best_h = h6.copy()
        best_c5 = c5_6

    # Validate and return
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
