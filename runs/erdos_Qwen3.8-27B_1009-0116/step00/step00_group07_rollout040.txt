    h_c = best_h.astype(float)
    lr = 0.008
    while time.time() < t_end3:
        np_ = np.random.randint(1, 3)
        idx = np.random.choice(n, np_, replace=False)
        pert = np.random.uniform(-lr, lr, np_)
        h_t = h_c.copy()
        h_t[idx] += pert
        h_t = np.clip(h_t, 0.0, 1.0)
        h_t = fix_sum_cont(h_t, target_sum)
        nv = compute_c5(h_t)
        if nv < best_c5:
            best_c5 = nv
            best_h = h_t.copy()
            h_c = h_t

    # Final: try a few deterministic improvements
    for _ in range(50):
        h_t = best_h.copy()
        # Nudge toward 0/1 where clearly one side
        for k in range(n):
            if h_t[k] > 0.7:
                h_t[k] = 1.0
            elif h_t[k] < 0.3:
                h_t[k] = 0.0
        h_t = fix_sum_cont(h_t, target_sum)
        nv = compute_c5(h_t)
        if nv < best_c5:
            best_c5 = nv
            best_h = h_t.copy()
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum_cont(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
