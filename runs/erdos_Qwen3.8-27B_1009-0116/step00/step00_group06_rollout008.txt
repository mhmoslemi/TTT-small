            if rem > 0:
                G[k_idx, m:] += (1.0 - h[:rem])
                G[k_idx, :rem] += (-h[m:])
    return G


def project_to_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = float(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = float(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def solve_outer_approx(h0, n, max_iter=200, time_limit=None, t0=None):
    h = h0.copy()
    best = h.copy()
    best_c5 = c5_max(c5_values(h))
    n_total = 2 * n - 1
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0

    for it in range(max_iter):
        if time_limit and t0 and (time.time() - t0) > time_limit:
            break
        cvals = c5_values(h)
        cur = c5_max(cvals)
        if cur < best_c5:
            best_c5 = cur
            best = h.copy()
        G = compute_all_grads(h, n)
        b_const = cvals - G @ h
        A_ub = np.zeros((n_total, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -b_const
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
        if not res.success:
            break
        h_new = project_to_feasible(res.x[:n], n)
        new_c5 = c5_max(c5_values(h_new))
        if new_c5 <= cur + 1e-9:
            h = h_new
        else:
            h = project_to_feasible(0.5 * h + 0.5 * h_new, n)
    return best, best_c5


def gradient_phase(h0, n, t0, time_limit):
    h = h0.copy()
    best = h.copy()
    best_val = c5_max(c5_values(h))
    lr = 0.005
    temp = 0.001
    stagnant = 0
    while (time.time() - t0) < time_limit:
        cvals = c5_values(h)
        cur = float(np.max(cvals))
        if cur < best_val - 1e-12:
            best_val = cur
            best = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 500:
            h = best.copy()
            h += np.random.RandomState(int(time.time() * 1000) % 10000).normal(0, 0.008, n)
            h = project_to_feasible(h, n)
            lr = 0.008
            stagnant = 0
            continue
        shifted = (cvals - cur) / temp
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros_like(cvals)
            weights[np.argmax(cvals)] = 1.0
        else:
            weights /= ws
        G = compute_all_grads(h, n)
        grad = G.T @ weights
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, n)
        new_c5 = c5_max(c5_values(h_new))
        if new_c5 < cur + 1e-10:
            h = h_new
            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.85
        temp = max(0.0003, temp * 0.9995)
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h0 = project_to_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = c5_max(c5_values(best_h))

    # Phase 1: LP outer approximation
    rem = budget_s - (time.time() - t0) - 30
    h_opt, c5_opt = solve_outer_approx(h0, n, max_iter=300, time_limit=rem, t0=t0)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Gradient refinement
    if (time.time() - t0) < budget_s - 30:
        rem2 = budget_s - (time.time() - t0) - 20
        h_opt, c5_opt = gradient_phase(best_h, n, t0, rem2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Another LP polish
    if (time.time() - t0) < budget_s - 10:
        rem3 = budget_s - (time.time() - t0) - 5
        h_opt, c5_opt = solve_outer_approx(best_h, n, max_iter=150, time_limit=rem3, t0=t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Random restarts (short LP)
    if (time.time() - t0) < budget_s - 10:
        for _ in range(8):
            if (time.time() - t0) > budget_s - 3:
                break
            h_rand = rng.random(n)
            h_rand = project_to_feasible(h_rand, n)
            h_opt, c5_opt = solve_outer_approx(h_rand, n, max_iter=60,
                                               time_limit=budget_s - (time.time() - t0) - 3, t0=t0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    return best_h, best_c5, n
