```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def project_onto_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    # Iterative projection
    for _ in range(100):
        s = np.sum(h)
        target = n / 2
        if abs(s - target) < 1e-10:
            break
        # Water-filling: add/subtract uniform, clip
        if s < target:
            mask = h < 1
            if not np.any(mask):
                break
            add = (target - s) / np.count_nonzero(mask)
            h[mask] = np.minimum(h[mask] + add, 1)
        else:
            mask = h > 0
            if not np.any(mask):
                break
            sub = (s - target) / np.count_nonzero(mask)
            h[mask] = np.maximum(h[mask] - sub, 0)
    return h

def slsqp_optimize(h0, n, maxiter=200):
    dx = 2.0 / n
    m = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:n]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def eq(x):
        return np.sum(x[:n]) - n / 2
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    t0 = compute_c5(h0, n) + 0.01
    x0 = np.concatenate([h0, [t0]])
    
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    
    h_opt = project_onto_feasible(res.x[:n], n)
    val = compute_c5(h_opt, n)
    return h_opt, val
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] = np.minimum(h[mask] + diff / np.count_nonzero(mask), 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] = np.maximum(h[mask] + diff / np.count_nonzero(mask), 0.0)
    return h

def slsqp_optimize(h0, n, maxiter=300, ftol=1e-10):
    dx = 2.0 / n
    m = 2 * n - 1

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:n]
        return x[-1] - np.correlate(h, 1.0 - h, mode="full") * dx

    def eq(x):
        return np.sum(x[:n]) - n / 2.0

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    t0 = compute_c5(h0, n) + 0.01
    x0 = np.concatenate([h0, [t0]])
    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq}]

    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    h_opt = project_feasible(res.x[:n], n)
    val = compute_c5(h_opt, n)
    return h_opt, val

def lsq_bfgs_optimize(h0, n, alpha=200, maxiter=500):
    dx = 2.0 / n
    m = 2 * n - 1

    def smooth_obj(h):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        shifted = c - c.max()
        logsum = np.log(np.sum(np.exp(alpha * shifted)) / m) + c.max()
        return logsum / alpha

    def smooth_grad(h):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        shifted = c - c.max()
        w = np.exp(alpha * shifted)
        w = w / w.sum()
        grad = np.zeros(n)
        for k in range(m):
            lag = k - (n - 1)
            if abs(w[k]) < 1e-15:
                continue
            for i in range(max(0, -lag), min(n, n - lag)):
                j = i + lag
                grad[j] += w[k] * (1.0 - h[i]) * dx
                if i != j:
                    grad[i] -= w[k] * h[j] * dx
                else:
                    grad[i] -= w[k] * h[i] * dx
        return grad

    res = minimize(smooth_obj, h0, jac=smooth_grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                            'factr': 1e8})
    h_opt = project_feasible(res.x, n)
    return h_opt, compute_c5(h_opt, n)

def try_binary_patterns(n, seed=0):
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = 1.0
    for _ in range(200):
        idx = np.arange(n)

        h = np.zeros(n)
        h[idx[:n // 2]] = 1.0
        val = compute_c5(h, n)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val
