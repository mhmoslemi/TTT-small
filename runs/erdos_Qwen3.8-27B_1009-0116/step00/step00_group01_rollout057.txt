import numpy as np
import time

def project_box_sum(x, S):
    n = len(x)
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c.max()

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    z = alpha * dx * vals
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    p = exp_z / exp_z.sum()
    obj = (z.max() + np.log(exp_z.sum() + 1e-300)) / alpha
    p_rev = p[::-1]
    c1 = np.convolve(1.0 - h, p, mode='full')
    c2 = np.convolve(h, p_rev, mode='full')
    grad = dx * (c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1])
    grad = grad - grad.mean()
    return obj, grad

def proj_grad_momentum(h, alpha, n_steps, lr, momentum, t_start, t_limit):
    h = h.copy().astype(np.float64)
    v = np.zeros_like(h)
    S = len(h) / 2.0
    for _ in range(n_steps):
        if time.time() - t_start > t_limit:
            break
        obj, grad = smooth_obj_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        v = momentum * v - lr * grad
        moved = False
        scale = 1.0
        for _ in range(25):
            h_try = project_box_sum(h + scale * v, S)
            obj_try, _ = smooth_obj_grad(h_try, alpha)
            if obj_try < obj + 1e-15:
                h = h_try
                moved = True
                break
            scale *= 0.5
        if not moved:
            v = np.zeros_like(h)
            break
    return h

def interp_to_res(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old))
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new / 2.0)
