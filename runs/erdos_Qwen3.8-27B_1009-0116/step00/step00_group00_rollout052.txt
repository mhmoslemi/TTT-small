        if c5_ls < c5_best:
            h_best, c5_best = h_ls, c5_ls

        # Multi-start perturbations
        for i in range(4):
            if time.time() > t_end - 10:
                break
            h_p = perturb(h_best, 0.04, rng)
            h_opt2 = slsqp_opt(h_p, maxiter=150)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < c5_best:
                h_best, c5_best = h_opt2, c5_opt2

    # Phase 3: Final refinement
    if h_best is not None and time.time() < t_end - 15:
        h_final = slsqp_opt(h_best, maxiter=600, ftol=1e-15)
        c5_final = compute_c5(h_final)
        if c5_final < c5_best:
            h_best, c5_best = h_final, c5_final
        h_ls, c5_ls = local_search(h_best, rng, max_iters=200)
        if c5_ls < c5_best:
            h_best, c5_best = h_ls, c5_ls

    # Ensure valid output
    h_best = np.clip(h_best, 0.0, 1.0)
    n = len(h_best)
    h_best += (n / 2.0 - h_best.sum()) / n
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_best = compute_c5(h_best)

    return (h_best, c5_best, len(h_best))
