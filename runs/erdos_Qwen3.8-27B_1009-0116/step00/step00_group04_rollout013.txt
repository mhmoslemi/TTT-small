                    bo, bz = oi, oz
        if bo >= 0:
            h[bo] = 0.0
            h[bz] = 1.0
            c5 = best_nc5
        else:
            break
    return h, c5

def sa_search(h, c5, rng, t0, time_limit, T_init=0.3):
    n = len(h)
    T = T_init
    cur_c5 = c5
    best_L = c5
    h_best = h.copy()
    stall = 0
    while T > 1e-8:
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        oi = ones[rng.randint(len(ones))]
        oz = zeros[rng.randint(len(zeros))]
        if oi == oz:
            T *= 0.99999
            continue
        h[oi] = 0.0
        h[oz] = 1.0
        nc5 = compute_c5(h)
        delta = nc5 - cur_c5
        accept = delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T))
        if accept:
            cur_c5 = nc5
            if nc5 < best_L:
                best_L = nc5
                h_best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[oi] = 1.0
            h[oz] = 0.0
            stall += 1
        T *= 0.99999
        if stall > 1500:
            T *= 0.6
            stall = 0
            # Kick: random perturbation of 3-5 bits
            n_flip = rng.randint(3, 6)
            ones_k = np.where(h == 1.0)[0]
            zeros_k = np.where(h == 0.0)[0]
            for _ in range(n_flip):
                a = ones_k[rng.randint(len(ones_k))]
                b = zeros_k[rng.randint(len(zeros_k))]
                if a != b:
                    h[a] = 0.0
                    h[b] = 1.0
                    ones_k = np.where(h == 1.0)[0]
                    zeros_k = np.where(h == 0.0)[0]
            cur_c5 = compute_c5(h)
    return h_best, best_L

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.zeros(42)
        h0[:21] = 1.0

    n = len(h0)
    half_n = n // 2

    h = project_binary(h0, rng)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    # Phase 1: Exhaustive from initial
    h, c5 = exhaustive_improve(h, c5, t0, budget_s * 0.08, max_rounds=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SA exploration from improved
    h_sa, c5_sa = sa_search(h, c5, rng, t0, budget_s * 0.4, T_init=0.3)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Exhaustive after SA
    h3 = best_h.copy()
    h3, c5_3 = exhaustive_improve(h3, best_c5, t0, budget_s * 0.55, max_rounds=1000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Repeat SA + exhaustive cycles
    for cycle in range(4):
        tl1 = 0.55 + (0.15 - 0.03 * cycle) * (cycle + 1)
        tl1 = min(tl1, 0.85)
        tl2 = min(tl1 + 0.05, 0.92)
        if time.time() - t0 > tl1 * budget_s:
            break
        h_c = best_h.copy()
        h_c, c5_c = sa_search(h_c, best_c5, rng, t0, tl1 * budget_s, T_init=0.25)
        h_c, c5_c = exhaustive_improve(h_c, c5_c, t0, tl2 * budget_s, max_rounds=500)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 5: Multi-start at current n
    for attempt in range(6):
        if time.time() - t0 > budget_s * 0.88:
            break
        h_m = np.zeros(n)
        h_m[rng.choice(n, size=half_n, replace=False)] = 1.0
        c5_m = compute_c5(h_m)
        h_m, c5_m = exhaustive_improve(h_m, c5_m, t0, budget_s * 0.9, max_rounds=300)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Phase 6: Try other resolutions
    for n_try in [36, 48, 60, 80, 100]:
        if time.time() - t0 > budget_s * 0.96:
            break
        h_t = np.zeros(n_try)
        h_t[rng.choice(n_try, size=n_try // 2, replace=False)] = 1.0
        c5_t = compute_c5(h_t)
        h_t, c5_t = exhaustive_improve(h_t, c5_t, t0, budget_s * 0.98, max_rounds=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
