        if rng.rand() < 0.5:
            delta = -delta
        h[i] -= delta
        h[j] += delta
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            new_c5 = compute_c5(h)
            if not (new_c5 < c5):
                h[i] += delta
                h[j] -= delta
            else:
                c5 = new_c5
        else:
            h[i] += delta
            h[j] -= delta
    return h, c5


def optimize_grid(h0, n, rng, t0, deadline):
    global best_h, best_c5
    h0 = fix_sum(h0, n)
    c5_cur = compute_c5(h0)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h0.copy()

    max_iter_slq = 150 if n > 150 else 250
    for beta in [30, 100, 300]:
        if time.time() - t0 > deadline:
            break
        for pen in [50.0, 200.0]:
            if time.time() - t0 > deadline:
                break
            res = do_slsqp(h0, n, beta, pen, max_iter_slq)
            if res.x is not None:
                h_new = fix_sum(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    h0 = h_new
                    c5_cur = c5_new
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h0.copy()
    if time.time() - t0 < deadline:
        ls_iter = 300 if n <= 200 else 200
        h0, c5_ls = local_search(h0, n, rng, ls_iter)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h0.copy()
    return h0


best_h = None
best_c5 = 1e9


def run(seed=42, budget_s=1100, **kwargs):
    global best_h, best_c5
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = np.full(95, 0.5)

    n_init = len(h_init)
    best_h = fix_sum(h_init, n_init)
    best_c5 = compute_c5(best_h)

    # Phase 1: optimize on original grid
    h_work = best_h.copy()
    h_work = optimize_grid(h_work, n_init, rng, t0, t0 + budget_s * 0.3)

    # Phase 2: project to finer grids and optimize
    finer_grids = []
    if n_init <= 100:
        finer_grids = [199, 299]
    elif n_init <= 200:
        finer_grids = [399]
    else:
        finer_grids = []

    for n_f in finer_grids:
        deadline = t0 + budget_s - 60
        if time.time() > deadline:
            break

        h_proj = project_grid(best_h, n_f)
        h_proj = optimize_grid(h_proj, n_f, rng, t0, deadline)

        # Try binary-like perturbed start
        if time.time() < t0 + budget_s - 50:
            h_bin = np.zeros(n_f)
            h_bin[:n_f // 2] = 1.0
            if n_f % 2 == 1:
                h_bin[n_f // 2] = 0.5
            h_bin += rng.normal(0, 0.04, n_f)
            h_bin = fix_sum(h_bin, n_f)
            h_bin = optimize_grid(h_bin, n_f, rng, t0, t0 + budget_s - 30)

    # Phase 3: if we found a good solution on finer grid, try to refine back
    n_best = len(best_h)
    if n_best > n_init and time.time() < t0 + budget_s - 20:
        h_back = project_grid(best_h, n_init)
        h_back = optimize_grid(h_back, n_init, rng, t0, t0 + budget_s - 5)

    # Final fix and return
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
