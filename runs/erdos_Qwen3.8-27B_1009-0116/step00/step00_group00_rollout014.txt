import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.interpolate import interp1d
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff * room / s, 0, 1)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff * h / s, 0, 1)
    return h

def grad_for_lag(v, L, n):
    g = np.ones(n)
    if L > 0:
        g[:n - L] -= v[L:]
        g[L:] -= v[:n - L]
    elif L < 0:
        g[-L:] -= v[:n + L]
        g[:n + L] -= v[-L:]
    else:
        g -= 2.0 * v
    return g

def smooth_obj_and_grad(v, n, T):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    vals = corr * dx
    vshift = vals - vals.max()
    exps = np.exp(T * vshift)
    weights = exps / exps.sum()
    logsum = vals.max() + np.log(exps.sum()) / T
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        L = m - (n - 1)
        g += w * grad_for_lag(v, L, n) * dx
    return logsum, g

def slsqp_optimize(h0, maxiter=400, T=40.0, ftol=1e-12):
    n = len(h0)
    dx = 2.0 / n

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        return float(np.max(corr) * dx)

    def obj_jac(v):
        corr = np.correlate(v, 1.0 - v, mode="full")
        max_idx = int(np.argmax(corr))
        L = max_idx - (n - 1)
        return grad_for_lag(v, L, n) * dx

    constraints = [{'type': 'eq', 'fun': lambda v: np.sum(v) - n / 2.0,
                   'jac': lambda v: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=obj_jac,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    x = np.clip(res.x, 0, 1)
    return make_feasible(x, n), eval_c5(x)

def slsqp_smooth(h0, maxiter=400, T=30.0, ftol=1e-12):
    n = len(h0)
    dx = 2.0 / n

    def obj(v):
        return smooth_obj_and_grad(v, n, T)[0]

    def obj_jac(v):
        return smooth_obj_and_grad(v, n, T)[1]

    constraints = [{'type': 'eq', 'fun': lambda v: np.sum(v) - n / 2.0,
                   'jac': lambda v: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', jac=obj_jac,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    x = np.clip(res.x, 0, 1)
    return make_feasible(x, n), eval_c5(x)

def perturb(h, n, scale, rng):
    noise = rng.normal(0, scale, size=n)
    return make_feasible(h + noise, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    if c5_1 < best_c5:

    if time.time() - t0 > budget_s * 0.15:
        return (best_h, best_c5, n)

    # Phase 2: Smooth SLSQP from best
    h2, c5_2 = slsqp_smooth(h0, maxiter=400, T=25.0)
    if c5_2 < best_c5:

    h2b, c5_2b = slsqp_smooth(h0, maxiter=400, T=50.0)
    if c5_2b < best_c5:
        best_h, best_c5 = h2b, c5_2b
    if time.time() - t0 > budget_s * 0.3:

    # Phase 3: Random restarts with SLSQP
    for r in range(30):
        if time.time() - t0 > budget_s * 0.7:
            break
        scale = 0.03 + 0.12 * rng.random()
        src = best_h if r % 2 == 0 else h0
        hp = perturb(src, n, scale, rng)
        hc, c5c = slsqp_optimize(hp, maxiter=300)
        if c5c < best_c5:

    # Phase 5: Try higher resolution
    for n_try in [90, 180, 360]:
        if time.time() - t0 > budget_s * 0.9:

        x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        f = interp1d(x_old, best_h, kind='cubic', fill_value="extrapolate")
        h_up = make_feasible(f(x_new), n_try)
        hc, c5c = slsqp_optimize(h_up, maxiter=200)
        if c5c < best_c5:

        if c5c < best_c5:

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = eval_c5(best_h)
    return (best_h, final_c5, len(best_h))
