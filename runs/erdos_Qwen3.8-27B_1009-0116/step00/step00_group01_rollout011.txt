import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, -10, 10)  # prevent overflow
    lo, hi = x.min() - 2, x.max() + 2
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(x - lam, 0, 1)

def c5_subgradient(h, n):
    """Return (c5_value, subgradient) of the C5 function."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corr)
    c5_val = corr[k_star] * dx
    m = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jm = j + m
        if 0 <= jm < n:
            grad[j] += 1.0 - h[jm]
        jm2 = j - m
        if 0 <= jm2 < n:
            grad[j] -= h[jm2]
    return c5_val, grad * dx

def projected_gradient(h0, n, max_iter=5000, tol=1e-10):
    """Projected gradient descent with backtracking."""
    h = project_onto_feasible(h0.copy(), n)
    alpha = 1.0
    c5_prev = compute_c5(h)
    
    for it in range(max_iter):
        c5_val, grad = c5_subgradient(h, n)
        if np.abs(c5_prev - c5_val) < tol:
            break
        # Try step
        h_new = project_onto_feasible(h - alpha * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_val + 1e-15:
            h = h_new
            # Try to increase alpha
            if alpha < 10.0:
                alpha *= 1.2
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                break
    
    return h, compute_c5(h)

def smooth_c5(h, n, beta=50):
    """Smooth approximation of C5 using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # log-sum-exp for stability
    max_c = np.max(corr)
    return (max_c + np.log(np.sum(np.exp(corr - max_c)))) / beta * dx

def smooth_c5_grad(h, n, beta=50):
    """Gradient of smooth C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(corr - max_c) / np.sum(np.exp(corr - max_c))
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        m = k - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            jm = j + m
            if 0 <= jm < n:
                g[j] += 1.0 - h[jm]
            jm2 = j - m
            if 0 <= jm2 < n:
                g[j] -= h[jm2]
        grad += weights[k] * g
    return (max_c + np.log(np.sum(weights))) / beta * dx, grad * dx / beta

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, -5, 5)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def c5_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corr)
    c5 = corr[k_star] * dx
    m = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jm = j + m
        if 0 <= jm < n:
            grad[j] += 1.0 - h[jm]
        jm2 = j - m
        if 0 <= jm2 < n:
            grad[j] -= h[jm2]
    return c5, dx * grad

def pg_descent(h0, n, max_iter=3000, time_limit=60.0):

    t0 = time.time()

def topk_c5_grad(h, n, k_top=3):
    """Subgradient using average of top-k lags for stability."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    topk = np.argsort(corr)[-k_top:]
    c5 = corr[topk[0]] * dx  # max
    grad = np.zeros(n)
    for k in topk:
        m = k - (n - 1)
        for j in range(n):
            jm = j + m
            if 0 <= jm < n:
                grad[j] += 1.0 - h[jm]
            jm2 = j - m
            if 0 <= jm2 < n:
                grad[j] -= h[jm2]
    return c5, dx * grad / k_top

                break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, -5.0, 5.0)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def c5_subgrad(h, n, k_top=1):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    topk = np.argsort(corr)[-k_top:]
    c5 = corr[topk[0]] * dx
    grad = np.zeros(n)
    for k in topk:
        m = k - (n - 1)
        for j in range(n):
            jm = j + m
            if 0 <= jm < n:
                grad[j] += 1.0 - h[jm]
            jm2 = j - m
            if 0 <= jm2 < n:
                grad[j] -= h[jm2]
    return c5, dx * grad / k_top

def pg_descent(h0, n, max_iter=5000, time_limit=80.0, k_top=3):
    h = project_feasible(h0.astype(float).copy(), n)
    t0 = time.time()
    alpha = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        c5, grad = c5_subgrad(h, n, k_top)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        h_new = project_feasible(h - alpha * grad / grad_norm * 0.5, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:

                alpha *= 1.1

                break
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                break
    return best_h, best_c5

def smooth_obj(h, n, beta=30.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = np.max(corr)
    lse = mc + np.log(np.sum(np.exp(corr - mc)))
    return lse / beta * dx

def smooth_obj_wrap(x, n, beta, penalty):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1 - h, mode='full')
    mc = np.max(corr)
    lse = mc + np.log(np.sum(np.exp(corr - mc)))
    s = np.sum(h) - n / 2.0
    return (lse / beta) * dx + penalty * s * s

def lbgfbs_optimize(h0, n, time_limit=90.0, beta=30.0):
    t0 = time.time()
    h_start = project_feasible(h0.astype(float).copy(), n)
    for penalty in [10.0, 100.0, 1000.0]:
        if time.time() - t0 > time_limit:

                       method='L-BFGS-B', bounds=[(0, 1)] * n,
                       options={'maxiter': 200, 'ftol': 1e-12})
        h_start = np.clip(res.x, 0, 1)
        h_start = project_feasible(h_start, n)
    h_final = project_feasible(res.x, n)
    return h_final, compute_c5(h_final)
