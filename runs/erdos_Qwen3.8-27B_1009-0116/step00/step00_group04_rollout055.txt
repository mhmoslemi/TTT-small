from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def obj_func(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def eq_jac(h):
    return np.ones(len(h)) / len(h)

def slsqp(h0, max_iter=300):
    n = len(h0)
    res = minimize(obj_func, h0, method='SLSQP',
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':eq_fun,'jac':eq_jac}],
                   options={'maxiter':max_iter,'ftol':1e-12})
    return res.x
