import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def local_search_single(h, n, max_passes, time_limit):
    best_h = h.copy().astype(float)
    best_c5 = compute_c5(best_h, n)
    start = time.time()
    for _ in range(max_passes):
        if time.time() - start > time_limit:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        best_delta = 0.0
        best_swap = None
        for i in ones:
            best_h[i] = 0
            for j in zeros:
                if j == i:
                    continue
                best_h[j] = 1
                val = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * (2.0 / n)
                delta = best_c5 - val
                best_h[j] = 0
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
            best_h[i] = 1
        if best_swap is None:
            break
        best_h[best_swap[0]] = 0
        best_h[best_swap[1]] = 1
        best_c5 -= best_delta
    return best_h, best_c5

def local_search_multi(h, n, k, rng, max_iters, time_limit):
    best_h = h.copy().astype(float)
    best_c5 = compute_c5(best_h, n)
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) < k or len(zeros) < k:
            break
        pick_o = rng.choice(len(ones), size=k, replace=False)
        pick_z = rng.choice(len(zeros), size=k, replace=False)
        new_h = best_h.copy()
        new_h[ones[pick_o]] = 0.0
        new_h[zeros[pick_z]] = 1.0
        val = compute_c5(new_h, n)
        if val < best_c5 - 1e-12:
            best_c5 = val
            best_h = new_h
    return best_h, best_c5

def sa_search(h, n, rng, max_iters, T0, T1, time_limit):
    cur_h = h.copy().astype(float)
    cur_c5 = compute_c5(cur_h, n)
    best_c5 = cur_c5
    best_h = cur_h.copy()
    start = time.time()
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        T = T0 * (T1 / T0) ** (t / max(float(max_iters), 1))
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        npick = 1 if rng.random() < 0.7 else 2
        if npick == 1:
            i = int(rng.integers(len(ones)))
            j = int(rng.integers(len(zeros)))
            new_h = cur_h.copy()
            new_h[ones[i]] = 0.0
            new_h[zeros[j]] = 1.0
        else:
            i1 = int(rng.integers(len(ones)))
            i2 = int(rng.integers(len(ones)))
            j1 = int(rng.integers(len(zeros)))
            j2 = int(rng.integers(len(zeros)))
            if i1 == i2 or j1 == j2:
                continue
            new_h = cur_h.copy()
            new_h[ones[i1]] = 0.0
            new_h[ones[i2]] = 0.0
            new_h[zeros[j1]] = 1.0
            new_h[zeros[j2]] = 1.0
        new_c5 = compute_c5(new_h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    return best_h, best_c5

def upscale_binary(h, factor):
    n_new = len(h) * factor
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    def remaining():

    # Phase 1: Intensive single-swap LS at n0

    for mult in [2, 2]:

        # SA

        # Multi-point

            h_mp, c_mp = local_search_multi(h_up, n_new, 2, rng, 50000, t_mp)

    final_c5 = compute_c5(best_h, len(best_h))
    return (best_h, final_c5, len(best_h))
