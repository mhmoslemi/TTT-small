                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': 500, 'ftol': 1e-14})
        h_new = project_to_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 3: Systematic pair exchanges with 1D line search
    if time.time() - t0 < budget_s * 0.75:
        eps_list = [0.05, 0.02, 0.01, 0.005]
        for eps in eps_list:
            if time.time() - t0 > budget_s * 0.75:
                break
            stagnant = 0
            while time.time() - t0 < budget_s * 0.75 and stagnant < 8000:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    stagnant += 1
                    continue
                h_trial = best_h.copy()
                h_trial[i] += eps
                h_trial[j] -= eps
                h_trial = project_to_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-14:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                    stagnant = 0
                else:
                    stagnant += 1

    # Phase 4: Very fine local search
    if time.time() - t0 < budget_s * 0.9:
        eps = 0.003
        while time.time() - t0 < budget_s * 0.95:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            h_trial = best_h.copy()
            h_trial[i] += eps
            h_trial[j] -= eps
            h_trial = project_to_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                eps = max(0.0005, eps * 0.97)
            else:
                eps = min(0.01, eps * 1.003)

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
