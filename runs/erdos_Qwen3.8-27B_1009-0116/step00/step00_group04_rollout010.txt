import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr


def smooth_c5(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    return m + np.log(np.sum(np.exp((corr - m) * beta))) / beta


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final correction on the least-saturated entry
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.minimum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h
