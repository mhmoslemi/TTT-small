import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    d = target_sum - h.sum()
    if abs(d) > 1e-12:
        h = h + d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h
