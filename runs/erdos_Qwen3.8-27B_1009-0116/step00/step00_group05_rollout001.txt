import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-11
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-11
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(0.0, h[mask] + step)
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))[::-1]
        idx = [i for i in order if (diff > 0 and h[i] < 1.0) or (diff < 0 and h[i] > 0.0)]
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def slsqp_run(h_start, n, maxiter=600, ftol=1e-11):
    def obj(hv):
        return evaluate_c5(hv)
    cons = [
        {'type': 'eq', 'fun': lambda hv: float(np.sum(hv) - n / 2.0),
         'jac': lambda hv: np.ones(n)}
    ]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x, res.fun
    except Exception:
        return h_start, np.inf
