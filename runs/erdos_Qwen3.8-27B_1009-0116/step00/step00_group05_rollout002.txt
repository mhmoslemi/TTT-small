from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr), np.argmax(corr)

def compute_subgradient(h, dx, max_idx):
    n = len(h)
    L = max_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + L < n:
            g += (1.0 - h[j + L])
        if 0 <= j - L < n:
            g -= h[j - L]
        grad[j] = g
    return dx * grad

def project_constraint(h, n):
    # Project to sum(h) = n/2 and 0 <= h <= 1
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - n/2.0) < 1e-10:
            break
        h = h + (n/2.0 - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_descent(h_init, dx, n, max_iter=5000, lr_init=0.05, tol=1e-8):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)[0]
    best_h = h.copy()
    
    for it in range(max_iter):
        c5, max_idx = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        grad = compute_subgradient(h, dx, max_idx)
        # Project gradient onto constraint manifold
        grad = grad - np.mean(grad)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5_exact(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_smooth(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return float(dx * (m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau))

def corr_grad_single(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if n - L > 0:
            g[:n - L] += (1.0 - h[L:])
            g[L:] += (-h[:n - L])
    else:
        M = -L
        if n - M > 0:
            g[M:] += (1.0 - h[:n - M])
            g[:n - M] += (-h[M:])
    return g

def compute_c5_smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(tau * (corr - m))
    sw = np.sum(w)
    w /= sw
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        L = k - (n - 1)
        grad += w[k] * corr_grad_single(h, L)
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = n / 2.0 - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_obj(h, dx, n, tau, mu):
    v = compute_c5_smooth(h, dx, tau)
    v += mu * (np.sum(h) - n / 2.0) ** 2
    return v

def smoothed_grad(h, dx, n, tau, mu):
    g = compute_c5_smooth_grad(h, dx, tau)
    g += 2.0 * mu * (np.sum(h) - n / 2.0) * np.ones(n)
    return g

def run_lbfgs(h_init, dx, n, tau=50.0, mu=200.0, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smoothed_obj, h_init, jac=smoothed_grad,
                       args=(dx, n, tau, mu), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        h_opt = project_feasible(res.x, n)
    except Exception:
        h_opt = project_feasible(h_init.copy(), n)
    c5 = compute_c5_exact(h_opt, dx)
    return h_opt, c5

def gen_random_binary(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, size=n // 2, replace=False)
    h[pos] = 1.0
    return h

def gen_random_cont(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return project_feasible(h, n)

def gen_block(n, rng):
    h = np.zeros(n)
    i = 0
    while i < n:
        blen = max(1, int(rng.exponential(3.0)))
        val = rng.randint(0, 2)
        h[i:i + blen] = val
        i += blen
    return project_feasible(h, n)

def perturb(h, sigma, rng):
    return project_feasible(h + sigma * rng.randn(len(h)), len(h))

def resample_to(h_src, n_tgt):
    n_src = len(h_src)
    idx = np.linspace(0, n_src - 1, n_tgt).astype(int)
    return project_feasible(h_src[idx], n_tgt)
