def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:

        h[idx] = np.clip(h[idx] + diff, 0, 1)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def grad_c5(h, temp=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if temp > 1e-8:
        cm = np.max(corr)
        w = np.exp((corr - cm) / temp)
        w /= np.sum(w)
    else:
        w = np.zeros(len(corr))
        mk = corr == np.max(corr)
        w[mk] = 1.0 / np.sum(mk)
    grad = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        wj = w[j]
        if wj < 1e-14:
            continue
        k = j - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, k + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wj * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wj * h[lo2 + k:hi2 + k]
    grad *= dx
    return grad


def pgd_solve(h_init, n, duration, t0, seed=42):
    rng = np.random.RandomState(seed)
    ts = n / 2.0
    h = project_feasible(h_init, ts)
    best = h.copy()
    best_c = compute_c5(h)
    lr = 0.02
    temp = 3.0
    stall = 0
    deadline = t0 + duration
    while time.time() < deadline and stall < 200:
        g = grad_c5(h, temp)
        g -= np.mean(g)
        hn = project_feasible(h - lr * g, ts)
        cn = compute_c5(hn)
        if cn < best_c - 1e-13:
            h = hn
            best = hn.copy()
            best_c = cn
            stall = 0
            lr = min(lr * 1.05, 0.08)
            temp = max(0.2, temp * 0.97)
        else:
            if cn < best_c + 0.003 and rng.rand() < 0.25:
                h = hn
            stall += 1
            lr *= 0.82
            temp = max(0.2, temp * 0.92)
        if lr < 1e-5:
            break
    return best, best_c


def local_search_binary(h_init, duration, t0, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    ts = n / 2.0
    h = project_feasible(h_init, ts)
    order = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[order[:int(round(ts))]] = 1.0
    best_c = compute_c5(h)
    best = h.copy()
    deadline = t0 + duration
    stall = 0
    while time.time() < deadline and stall < 40:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        ntries = min(300, len(ones) * len(zeros))
        for _ in range(ntries):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best = h.copy()
                best_c = c
                stall = 0
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = h[i], h[j]
        if not improved:
            stall += 1
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)

    n_best = len(best_h)
    best_h = project_feasible(best_h, n_best / 2.0)
    best_c5 = compute_c5(best_h)

    def remaining(frac=1.0):
        return t0 + budget_s * frac - time.time()

    # Phase 1: PGD at original n
    dur = min(150, budget_s * 0.2)
    h1, c1 = pgd_solve(best_h, n_best, dur, t0, seed)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Binary local search
    if remaining(0.95) > 40:
        dur = min(120, remaining(0.95) * 0.2)
        h2, c2 = local_search_binary(best_h, dur, t0, seed + 1)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: PGD from current best (possibly binary)
    if remaining(0.95) > 40:
        dur = min(120, remaining(0.95) * 0.25)
        h3, c3 = pgd_solve(best_h, n_best, dur, t0, seed + 2)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Try other n values
    candidates_n = [50, 100, 140, 200]
    for n_try in candidates_n:
        if remaining(0.85) < 25:
            break
        if n_try == n_best:
            continue
        x_src = np.linspace(0, 2, n_best, endpoint=False)
        x_dst = np.linspace(0, 2, n_try, endpoint=False)
        ht = np.interp(x_dst, x_src, best_h)
        ht = project_feasible(ht, n_try / 2.0)
        # Also try a random init
        hr = np.clip(rng.rand(n_try) * 1.4 - 0.2, 0, 1)
        hr = project_feasible(hr, n_try / 2.0)
        for h_init in [ht, hr]:
            if remaining(0.85) < 15:
                break
            dur = min(40, remaining(0.85) * 0.3)
            if dur < 8:
                continue
            h5, c5v = pgd_solve(h_init, n_try, dur, t0, seed + n_try)
            if c5v < best_c5:
                best_h, best_c5, n_best = h5, c5v, n_try

    # Phase 5: Binary local search at best n
    if remaining(0.95) > 30:
        dur = min(60, remaining(0.95) * 0.3)
        h6, c6 = local_search_binary(best_h, dur, t0, seed + 10)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    # Phase 6: Final PGD polish
    if remaining(0.97) > 15:
        dur = min(remaining(0.97), 60)
        h7, c7 = pgd_solve(best_h, len(best_h), dur, t0, seed + 50)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    # Ensure feasibility and recompute
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
