import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_max_obj(h, n, dx, p):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    scaled = p * (corr - m)
    scaled = np.clip(scaled, -400, 400)
    return m + float(np.log(np.sum(np.exp(scaled)))) / p

def eq_con(h, n):
    return float(np.sum(h)) - 0.5 * n

def repair(x, n):
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    tgt = 0.5 * n
    s = float(x.sum())
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    d = tgt - float(x.sum())
    if abs(d) > 1e-10:
        k = min(n, max(1, int(abs(d)) + 1))
        if d > 0:
            idx = np.argsort(x)[-k:]
            x[idx] = np.minimum(1.0, x[idx] + d / k)
        else:
            idx = np.argsort(x)[:k]
            x[idx] = np.maximum(0.0, x[idx] + d / k)
    return x

def slsqp_opt(h_init, n, dx, p, maxiter):
    res = minimize(
        smooth_max_obj, h_init.copy(), args=(n, dx, p),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': eq_con, 'args': (n,)}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    h_new = repair(res.x, n)
    c5_new = eval_c5(h_new, dx)
    return h_new, c5_new

def coord_descent(h, n, dx, max_time):
    hc = h.copy()
    cc = eval_c5(hc, dx)
    t_end = time.time() + max_time
    for _ in range(5):
        changed = False
        for i in range(n):
            if time.time() > t_end:
                return hc, cc
            ov = hc[i]
            cands = [0.0, 1.0, 0.5, ov * 0.5, ov * 1.5,
                     min(1.0, ov + 0.15), max(0.0, ov - 0.15),
                     min(1.0, ov + 0.05), max(0.0, ov - 0.05)]
            for c in cands:
                if abs(c - ov) < 1e-6:
                    continue
                ht = hc.copy()
                ht[i] = c
                ht = repair(ht, n)
                ct = eval_c5(ht, dx)
                if ct < cc - 1e-12:
                    cc = ct
                    hc = ht
                    ov = c
                    changed = True
                    break
        if not changed:
            break
    return hc, cc
