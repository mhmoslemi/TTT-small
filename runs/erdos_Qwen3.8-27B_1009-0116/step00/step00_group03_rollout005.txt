import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_c5(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.mean(np.exp(corr - m))) / tau)

def repair_h(h, N):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = N / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
        cnt = np.sum(mask)
        if cnt > 0:
            per = diff / cnt
            h[mask] = np.minimum(h[mask] + per, 1.0)
            s2 = np.sum(h)
            diff2 = target - s2
            if diff2 > 1e-12:
                h[mask] = np.minimum(h[mask] + diff2 / cnt, 1.0)
    else:
        mask = h > 1e-9
        cnt = np.sum(mask)
        if cnt > 0:
            per = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - per, 0.0)
            s2 = np.sum(h)
            diff2 = s2 - target
            if diff2 > 1e-12:
                h[mask] = np.maximum(h[mask] - diff2 / cnt, 0.0)
    s = np.sum(h)
    if abs(s - target) > 1e-9:
        idx = int(np.argmax(1.0 - h)) if s < target else int(np.argmax(h))
        h[idx] = h[idx] + (target - s)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(x, N, tau):
    h = np.clip(x, 0.0, 1.0)
    return soft_max_c5(h, tau)

def sum_constraint(x, N):
    return np.sum(x) - N / 2.0

def local_search(h_in, c5_in, N, t_start, budget_s, n_iter, ls_label):
    h_curr = h_in.copy()
    best_c5 = c5_in
    best_h = h_in.copy()
    noise_scale = 0.08
    for i in range(n_iter):
        if (i % 200 == 0) and (time.time() - t_start > budget_s * 0.88):
            break
        n_pert = int(np.random.randint(2, 5))
        idxs = np.random.choice(N, n_pert, replace=False)
        h_new = h_curr.copy()
        h_new[idxs] += np.random.normal(0, noise_scale, n_pert)
        h_new = repair_h(h_new, N)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new.copy()
            noise_scale = max(0.005, noise_scale * 0.998)
        else:
            noise_scale = min(0.25, noise_scale * 1.002)
    return best_h, best_c5

def slsqp_optimize(h_in, N, t_start, budget_s, maxiter):
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - N / 2.0}]
    tau = 100.0
    try:
        res = minimize(soft_obj, h_in, args=(N, tau), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.fun < 1e6:
            h_out = repair_h(res.x, N)
            return h_out
    except Exception:
        pass
    return h_in

def run(seed=42, budget_s=1100, **kwargs):
