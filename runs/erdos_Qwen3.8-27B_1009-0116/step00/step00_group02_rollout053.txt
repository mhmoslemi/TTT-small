        if time.time() - t0 > budget_s - 40:
            break

        if r == 0:
            h_start = best_h.copy()
        elif r == 1:
            h_start = best_h + 0.02 * rng.randn(n)
            h_start = project_feasible(h_start, n)
        elif r == 2:
            kern = np.array([0.25, 0.5, 0.25])
            h_s = np.convolve(best_h, kern, mode='same')
            h_start = h_s + 0.04 * rng.randn(n)
            h_start = project_feasible(h_start, n)
        elif r == 3:
            h_start = make_block_init(n, rng)
        elif r == 4:
            h_start = make_block_init(n, rng)
        elif r == 5:
            h_start = best_h + 0.1 * rng.randn(n)
            h_start = project_feasible(h_start, n)
        else:
            scale = 0.015 + 0.06 * rng.random()
            h_start = best_h + scale * rng.randn(n)
            h_start = project_feasible(h_start, n)

        try:
            res = minimize(obj_penalty, h_start, method='Nelder-Mead',
                          args=(n, target),
                          options={'maxiter': 12000, 'xatol': 1e-10,
                                   'fatol': 1e-13, 'adaptive': True})
            h_cand = project_feasible(res.x, n)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
        except Exception:
            pass

        # Powell refinement every 4 rounds
        if r % 4 == 0 and time.time() - t0 < budget_s - 40:
            try:
                res2 = minimize(obj_penalty, best_h, method='Powell',
                               args=(n, target),
                               options={'maxiter': 400, 'ftol': 1e-14,
                                        'xtol': 1e-11})
                h_cand = project_feasible(res2.x, n)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

    # Binary snapping phase
    if time.time() - t0 < budget_s - 20:
        best_h, best_c5 = binary_snap(best_h, n)

    # Fine random perturbation search
    if time.time() - t0 < budget_s - 5:
        step = 0.002
        for i in range(500000):
            if time.time() - t0 > budget_s - 5:
                break
            if rng.random() < 0.4:
                h_try = best_h.copy()
                j = rng.randint(n)
                h_try[j] = np.clip(h_try[j] + rng.uniform(-0.08, 0.08), 0, 1)
            else:
                h_try = best_h + step * rng.randn(n)
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                step = max(0.0002, step * 0.97)

    # Final binary snap
    if time.time() - t0 < budget_s - 2:
        best_h, best_c5 = binary_snap(best_h, n)

    return (best_h, float(best_c5), n)
