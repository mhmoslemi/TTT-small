def project_box_sum(v, target_sum, n):
    """Project v onto {h : 0<=h<=1, sum(h)=target_sum}"""
    # Bisection on lambda: h_i = clip(v_i - lambda, 0, 1)
    lo, hi = -max(v) - 1, max(v) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust free variables
        free = (h > 0) & (h < 1)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

import numpy as np
import time

def compute_all_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def compute_c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def corr_grad_vec(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        k = -lag
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    return grad

def project_box_sum(v, target_sum):
    n = len(v)
    lo, hi = -3.0, 3.0
    for _ in range(90):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

    np.random.seed(seed)

        n = 70
        h = np.zeros(n)
        h[:n // 2] = 1.0

    target_sum = float(n) / 2.0
    h = project_box_sum(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Projected gradient descent with smooth-max
    alpha = 20.0
    lr = 0.03
    stall = 0
    n_lags = 2 * n - 1

    while time.time() < deadline - 400:
        corrs = compute_all_corrs(h)
        c5 = float(corrs.max())

        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break

        top_k = max(15, n_lags // 3)
        idx = np.argsort(corrs)[-top_k:]
        vals = corrs[idx]
        w = np.exp(alpha * (vals - vals.max()))
        w /= w.sum()

        grad = np.zeros(n)
        for ii in range(len(idx)):
            lag = int(idx[ii]) - (n - 1)
            grad += w[ii] * corr_grad_vec(h, lag)

        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new <= c5 + 1e-12:
            h = h_new
            stall = max(0, stall - 1)
            if stall > 1000:
                lr = max(lr * 0.97, 0.002)

            if lr < 0.002:

                alpha = max(alpha * 0.8, 5.0)
