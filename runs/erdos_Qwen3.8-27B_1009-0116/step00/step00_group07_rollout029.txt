import numpy as np
import time

def project_to_simplex_box(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection to find lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Adjust for any remaining error
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        interior = (h_proj > 0) & (h_proj < 1)
        if interior.sum() > 0:
            h_proj[interior] += diff / interior.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5_and_corr(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(corr)
    return c5, corr

def compute_gradient(h, n, dx, weights):
    """Compute gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:

        d = k - (n - 1)
        if d >= 0:
            m = n - d
            grad[d:] += w * (1.0 - h[:m])
            grad[:m] -= w * h[d:]
        else:
            d2 = -d
            m = n - d2
            grad[:m] += w * (1.0 - h[d2:])
            grad[d2:] -= w * h[:m]

def project_h(h, n):
    """Project h to {0<=h[i]<=1, sum(h)=n/2} via bisection on shift."""
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        interior = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if interior.sum() > 0:
            h_p[interior] += diff / interior.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p
