import time

def compute_c5(h):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx  # shape (2n-1,)
    
    if mu <= 0:
        # Hard max
        max_k = np.argmax(vals)
        L = max_k - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 1.0
            if 0 <= j - L < n:
                g -= h[j - L]
            if 0 <= j + L < n:
                g -= h[j + L]
            grad[j] = g * dx
        return float(vals[max_k]), grad
    else:
        # Smooth max (log-sum-exp)
        max_val = np.max(vals)
        exp_vals = np.exp((vals - max_val) / mu)
        sum_exp = np.sum(exp_vals)
        obj = max_val + mu * np.log(sum_exp)
        
        # p_k = exp(vals[k]/mu) / sum(exp(vals[k']/mu))
        p = exp_vals / sum_exp  # shape (2n-1,)

        grad = np.zeros(n)
        # For each k, grad_corr[k] at j is:
        # (1-h[j-L_k]) * 1_{valid} - h[j+L_k] * 1_{valid} where L_k = k-(n-1)
        # Vectorized:

                    g += (1 - h[j - L])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, mu=0.004):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    v_max = vals.max()
    exps = np.exp((vals - v_max) / mu)
    sum_exp = exps.sum()
    obj = v_max + mu * np.log(sum_exp)
    p = exps / sum_exp

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-11:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += pk * dx * (1.0 - 2.0 * h)
        elif L > 0:
            j1 = np.arange(L, n)
            grad[j1] += pk * dx * (1.0 - h[j1 - L])
            j2 = np.arange(0, n - L)
            grad[j2] -= pk * dx * h[j2 + L]
        else:
            M = -L
            j1 = np.arange(0, n - M)
            grad[j1] += pk * dx * (1.0 - h[j1 + M])
            j2 = np.arange(M, n)
            grad[j2] -= pk * dx * h[j2 - M]
    return obj, grad

def constraint_sum(h, n):
    return np.sum(h) - n / 2.0

def constraint_grad(h, n):
    return np.ones(n)

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def parent_to_n(parent, n):
    m = len(parent)
    x_old = np.arange(m) * (2.0 / m) + 1.0 / m
    x_new = np.arange(n) * (2.0 / n) + 1.0 / n
    h_new = np.interp(x_new, x_old, parent)
    return make_feasible(h_new, n)

def simple_initial(n, rng):
    t = rng.random(n)
    h = (t > 0.5).astype(float)
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        remove = rng.choice(idx, ones - target, replace=False)
        h[remove] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(idx, target - ones, replace=False)
        h[add] = 1
    return h

def block_initial(n, rng):
    h = np.zeros(n)
    blocks = rng.integers(2, 6)
    sizes = rng.integers(1, max(2, n // blocks + 1), size=blocks)
    pos = 0
    val = rng.random() > 0.3
    for sz in sizes:
        if pos >= n:
            break
        h[pos:pos+sz] = 1.0 if val else 0.0
        pos += sz
        val = not val
    h = make_feasible(h, n)
    return h

    rng = np.random.default_rng(seed)

    parent = None

    best_h = None

    resolutions = [45, 55, 70, 90, 120, 160]
    if parent is not None:
        pn = len(parent)
        if pn not in resolutions:
            resolutions.insert(0, pn)

    starts_per_n = 6
    slsqp_maxiter = 300
    mu_val = 0.004

    for n in resolutions:
        elapsed = time.time() - t0
        if elapsed > budget_s - 80:
            break

        inits = []
        # Start from parent
        if parent is not None:
            inits.append(parent_to_n(parent, n))
            inits.append(parent_to_n(parent, n) + rng.normal(0, 0.05, n))

        # Structured starts
        for _ in range(2):
            inits.append(simple_initial(n, rng))
        for _ in range(1):

        # Random smooth starts
        for _ in range(2):
            t = np.linspace(0, 2*np.pi, n)
            h0 = 0.5 + 0.3 * np.sin(rng.random() * t + rng.random())
            inits.append(make_feasible(h0, n))

        inits = [make_feasible(x, n) for x in inits]

        for h0 in inits:
            elapsed = time.time() - t0
            if elapsed > budget_s - 40:
                break

            h0 = np.clip(h0, 0.0, 1.0)

            res = minimize(
                lambda x: smooth_obj_grad(x, mu_val)[0],
                h0,
                jac=lambda x: smooth_obj_grad(x, mu_val)[1],
                method='SLSQP',
                bounds=[(0, 1)] * n,
