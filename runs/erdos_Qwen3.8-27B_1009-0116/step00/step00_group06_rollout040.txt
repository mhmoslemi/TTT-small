        h_new, c5_new = slsqp_optimize(h0, n3, dx3, maxiter=150)
        h_new, c5_new = coord_descent_step(h_new, n3, dx3, max_iters=2)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
            n_best = n3
    
    # Phase 4: Even finer grid if time permits (15% remaining)
    if time.time() < t_start + budget_s * 0.85:
        n4 = 337
        dx4 = 2.0 / n4
        h4 = interp_to_n(h_best, n4)
        h4 = repair(h4, n4)
        h_new, c5_new = slsqp_optimize(h4, n4, dx4, maxiter=100)
        h_new, c5_new = coord_descent_step(h_new, n4, dx4, max_iters=2)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
            n_best = n4
    
    # Final: one more pass on best grid
    if time.time() < t_start + budget_s * 0.95:
        dx_f = 2.0 / n_best
        h_best, c5_best = slsqp_optimize(h_best, n_best, dx_f, maxiter=200)
        h_best, c5_best = coord_descent_step(h_best, n_best, dx_f, max_iters=5)
    
    # Final repair and evaluate
    h_best = repair(h_best, n_best)
    dx_final = 2.0 / n_best
    c5_final = compute_c5(h_best, dx_final)
    
    return h_best, c5_final, n_best
