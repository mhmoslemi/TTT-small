import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def flip_pair(h, i, j, n):
    """Flip h[i] from 1 to 0 and h[j] from 0 to 1 (or vice versa)"""
    h = h.copy()
    if h[i] > 0.5 and h[j] < 0.5:
        h[i] = 0.0
        h[j] = 1.0
    elif h[i] < 0.5 and h[j] > 0.5:
        h[i] = 1.0
        h[j] = 0.0
    return h

def bit_flip_optimize(h0, n, max_iters, rng):
    """Simulated annealing with pair flips to maintain sum constraint"""
    h = h0.copy().astype(float)
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Fix sum constraint
    ones = np.sum(h)
    target = n / 2
    if ones > target:
        one_indices = np.where(h == 1)[0]
        to_flip = rng.choice(one_indices, int(ones - target), replace=False)
        h[to_flip] = 0
    elif ones < target:
        zero_indices = np.where(h == 0)[0]
        to_flip = rng.choice(zero_indices, int(target - ones), replace=False)
        h[to_flip] = 1
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 0.1
    cooling = 0.9995
    
    for it in range(max_iters):
        if rng.random() < 0.5:
            # Pick a random 1 and a random 0
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
            i = rng.integers(len(one_idx))
            j = rng.integers(len(zero_idx))
            i = one_idx[i]
            j = zero_idx[j]
        else:
            # Pick adjacent-ish pair
            i = rng.integers(n)
            # Find nearest 1 or 0
            if h[i] == 1:
                j_candidates = np.where(h == 0)[0]
            else:
                j_candidates = np.where(h == 1)[0]
            if len(j_candidates) == 0:
                continue
            j = j_candidates[rng.integers(len(j_candidates))]
        
        h_new = h.copy()
        h_new[i] = 1.0 - h_new[i]
        h_new[j] = 1.0 - h_new[j]
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5 or rng.random() < np.exp((current_c5 - new_c5) / T):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        T *= cooling
        if T < 0.005:
            break
    
    return best_h, best_c5

        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        current = best_new
        if current < best_val:
            best_val = current
            best_h = h.copy()
    return best_h, best_val

def sa_search(h0, n, rng, time_limit=20):
    h = fix_binary(h0, n, rng).copy()
    current = compute_c5(h)
    best_h = h.copy()
    best_val = current
    t0 = time.time()
    T = 0.01
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < current or rng.random() < np.exp((current - new_c5) / T):
            current = new_c5
            if new_c5 < best_val:
                best_val = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= 0.998
    return best_h, best_val

def continuous_refine(h0, n, time_limit=20):
    h0 = np.clip(h0, 0.05, 0.95)
    h0 = h0 / h0.sum() * (n / 2)
    
    def obj(x):
        x = np.clip(x, 0, 1)
        s = x.sum()
        if s < 1e-6:
            return 1.0
        x = x * (n / 2) / s
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': 300, 'ftol': 1e-9})
        h_opt = np.clip(res.x, 0, 1)
        h_opt = h_opt / h_opt.sum() * (n / 2)
        h_opt = np.clip(h_opt, 0, 1)
        c5_opt = compute_c5(h_opt)
        if c5_opt < compute_c5(h0):
            return h_opt, c5_opt
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    def remaining():
        return budget_s - (time.time() - t_start) - 5
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    h_bin = fix_binary(h_init, n, rng)
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Thorough local search from initial
    if remaining() > 10:
        h_ls, c5_ls = full_local_search(h_bin, n, rng, max_rounds=500,
                                        time_limit=min(120, remaining() * 0.25))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 2: SA to explore neighborhood
    if remaining() > 20:
        h_sa, c5_sa = sa_search(best_h, n, rng, time_limit=min(30, remaining() * 0.15))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Local search after SA
        if remaining() > 20:
            h_ls2, c5_ls2 = full_local_search(best_h, n, rng, max_rounds=200,
                                              time_limit=min(40, remaining() * 0.2))
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()
    
    # Phase 3: Random restarts at base resolution
    while remaining() > 40:
        h_r = best_h.copy()
        n_flip = int(rng.integers(3, max(4, n // 6)))
        for _ in range(n_flip):
            ones = np.where(h_r == 1)[0]
            zeros = np.where(h_r == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            h_r[ones[rng.integers(len(ones))]] = 0.0
            h_r[zeros[rng.integers(len(zeros))]] = 1.0
        h_r = fix_binary(h_r, n, rng)
        h_rs, c5_rs = full_local_search(h_r, n, rng, max_rounds=150,
                                        time_limit=min(15, max(3, remaining() - 15)))
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
    
    # Phase 4: Try higher resolution (upsample + search)
    for mult in [2, 3]:
        if remaining() < 60:
            break
        n_hi = n * mult
        h_hi = np.repeat(best_h, mult)
        h_hi = fix_binary(h_hi, n_hi, rng)
        h_hs, c5_hs = full_local_search(h_hi, n_hi, rng, max_rounds=100,
                                        time_limit=min(40, max(10, remaining() * 0.3)))
        if c5_hs < best_c5:
            best_c5 = c5_hs
            best_h = h_hs.copy()
    
    # Phase 5: Continuous refinement at best binary resolution
    if remaining() > 25:
        n_best = len(best_h)
        h_cr, c5_cr = continuous_refine(best_h, n_best, time_limit=min(20, remaining() - 10))
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = h_cr.copy()
    
    # Phase 6: Final random restarts at best resolution
    n_final = len(best_h)
    while remaining() > 25:
        # Random binary perturbation
        h_r = (best_h > 0.5).astype(float)
        h_r = fix_binary(h_r, n_final, rng)
        n_pert = int(rng.integers(2, max(3, n_final // 8)))
        for _ in range(n_pert):
            ones = np.where(h_r == 1)[0]
            zeros = np.where(h_r == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            h_r[ones[rng.integers(len(ones))]] = 0.0
            h_r[zeros[rng.integers(len(zeros))]] = 1.0
        h_r = fix_binary(h_r, n_final, rng)
        h_rs, c5_rs = full_local_search(h_r, n_final, rng, max_rounds=80,
                                        time_limit=min(10, max(3, remaining() - 10)))
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
        # Also try a mini-SA
        if remaining() > 30:
            h_sa2, c5_sa2 = sa_search(h_rs, n_final, rng, time_limit=min(8, remaining() - 15))
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
    
    return best_h, best_c5, len(best_h)
