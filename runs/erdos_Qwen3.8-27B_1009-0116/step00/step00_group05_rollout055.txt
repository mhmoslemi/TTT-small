        h_new = project_bsimplex(res.x, s)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    return best_h, best_c5


def nm_polish(h, n, s, deadline):
    if time.time() > deadline:
        return h, compute_c5(h)
    best_c5 = compute_c5(h)
    res = minimize(obj_nm, h, args=(n,), method='Nelder-Mead',
                   options={'xatol': 1e-9, 'fatol': 1e-12, 'maxiter': 1500,
                            'maxfev': 10000})
    h_new = project_bsimplex(res.x, s)
    c5 = compute_c5(h_new)
    if c5 < best_c5:
        return h_new, c5
    return h, best_c5


def try_binary_round(h, s):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in np.linspace(0.15, 0.85, 15):
        h_bin = (h >= alpha).astype(float)
        excess = int(round(h_bin.sum() - s))
        if excess > 0:
            idx = np.where(h_bin == 1)[0]
            order = idx[np.argsort(h[idx])][0:excess]
            h_bin[order] = 0.0
        elif excess < 0:
            idx = np.where(h_bin == 0)[0]
            order = idx[np.argsort(-h[idx])][0:-excess]
            h_bin[order] = 1.0
        c5 = compute_c5(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(70) * 0.5
    n = len(h0)
    s = n / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Refine initial with increasing temperature
    d1 = t0 + budget_s * 0.22
    best_h, best_c5 = local_opt(h0, n, s, [3, 5, 10, 20, 50, 100, 200, 500], d1)
    h_b, c5_b = try_binary_round(best_h, s)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b
    best_h, best_c5 = local_opt(best_h, n, s, [50, 100, 200, 500], t0 + budget_s * 0.30)

    # Phase 2: Random restarts
    d2 = t0 + budget_s * 0.84
    while time.time() < d2:
        r = rng.random()
        if r < 0.4:
            scale = 0.03 + 0.12 * rng.random()
            h_p = best_h + rng.randn(n) * scale
        elif r < 0.7:
            h_p = best_h + rng.randn(n) * 0.02
        elif r < 0.85:
            # Flip a random subset
            h_p = best_h.copy()
            n_flip = max(2, n // 7)
            flips = rng.choice(n, size=n_flip, replace=False)
            h_p[flips] = 1.0 - h_p[flips]
        else:
            h_p = rng.uniform(0, 1, n)
        h_p = project_bsimplex(h_p, s)
        h_new, c5_new = local_opt(h_p, n, s, [30, 80, 150], d2)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_b2, c5_b2 = try_binary_round(h_new, s)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 3: Final refinement with high temperature and Nelder-Mead
    d3 = t0 + budget_s * 0.99
    best_h, best_c5 = local_opt(best_h, n, s, [100, 200, 500, 1000], d3)
    best_h, best_c5 = nm_polish(best_h, n, s, d3)
    h_b3, c5_b3 = try_binary_round(best_h, s)
    if c5_b3 < best_c5:
        best_c5 = c5_b3
        best_h = h_b3

    return best_h, best_c5, n
