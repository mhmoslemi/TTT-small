import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def try_swap(h, i, j):
    """Try swapping h[i] and h[j], return new c5."""
    h[i], h[j] = h[j], h[i]
    c5, _ = compute_c5(h)
    h[i], h[j] = h[j], h[i]
    return c5

def local_search(h, n_swaps=100000, deadline=None):
    """Hill climbing with random restarts."""
    n = len(h)
    c5, corr = compute_c5(h)
    for _ in range(n_swaps):
        if deadline and time.time() > deadline:
            break
        # Pick random positions to swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h[i] == h[j]:
            continue
        new_c5 = try_swap(h, i, j)
        if new_c5 < c5 - 1e-12:
            c5 = new_c5
            compute_c5(h)  # Update corr
    return h, c5

def simulated_annealing(h, deadline):
    n = len(h)
    c5, corr = compute_c5(h)
    temp = 0.05
    cooling = 0.9999
    for iteration in range(500000):
        if time.time() > deadline:
            break
        if temp < 0.001:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h[i] == h[j]:
            continue
        new_c5 = try_swap(h, i, j)
        delta = new_c5 - c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            c5 = new_c5
        temp *= cooling
    return h, c5

```python
def swap_effect(h, i, j):
    """Compute delta c(L) for all L when swapping h[i] (1→0) and h[j] (0→1)."""
    n = len(h)
    # L ranges from -(n-1) to n-1, indexed as 0 to 2n-2 in the correlate output
    # c[L_idx] where L_idx = L + (n-1)
    delta = np.zeros(2*n-1)
    for L_idx in range(2*n-1):
        L = L_idx - (n - 1)
        dc = 0.0
        # Term: -(1-h[i-L]) if 0 <= i-L < n
        if 0 <= i - L < n:
            dc -= (1.0 - h[i - L])
        # Term: +h[i+L] if 0 <= i+L < n
        if 0 <= i + L < n:
            dc += h[i + L]
        # Term: +(1-h[j-L]) if 0 <= j-L < n
        if 0 <= j - L < n:
            dc += (1.0 - h[j - L])
        # Term: -h[j+L] if 0 <= j+L < n
        if 0 <= j + L < n:
            dc -= h[j + L]
        delta[L_idx] = dc
    return delta
```

This is O(n) per swap evaluation (for all lags). Then we can find the max of (c + delta) in O(n). Total O(n) per swap evaluation.

def swap_effect_vec(h, i, j):
    n = len(h)
    L = np.arange(-(n-1), n)  # all lags
    # For each L, compute the four terms
    # Term 1: -(1-h[i-L]) if valid
    # Term 2: +h[i+L] if valid
    # Term 3: +(1-h[j-L]) if valid
    # Term 4: -h[j+L] if valid

    idx2 = i + L
    idx3 = j - L
    idx4 = j + L

    h4 = np.where((idx4 >= 0) & (idx4 < n), h[idx4.clip(0, n-1)], 0.0)
