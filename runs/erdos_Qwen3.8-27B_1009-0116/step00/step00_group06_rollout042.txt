def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(corr - m, -40.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def project_fine(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    x_c = np.linspace(0.0, 2.0, n_coarse)
    x_f = np.linspace(0.0, 2.0, n_fine)
    h_fine = interp1d(x_c, h_coarse, kind='linear', bounds_error=False,
                      fill_value='extrapolate')(x_f)
    return repair(h_fine, n_fine)


def slsqp_optimize(h0, n, alpha, maxiter=300, ftol=1e-10):
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj(x, alpha)

    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            return repair(res.x, n)
    except Exception:
        pass
    return h0.copy()


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)

    best_h = repair(h_init.copy(), n)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - 30.0 - (time.time() - t0)

    # Phase 1: Gradually sharpen the max approximation via increasing alpha
    alphas_phase1 = [15, 30, 60, 120, 200]
    for alpha in alphas_phase1:
        if time_left() < 50:
            break
        result = slsqp_optimize(best_h, n, alpha, maxiter=500)
        c5 = compute_c5(result)
        if c5 < best_c5:
            best_c5 = c5
            best_h = result

    # Phase 2: Random perturbations + local SLSQP refinement
    n_pert = 0
    max_perts = 60
    while time_left() > 80 and n_pert < max_perts:
        n_pert += 1
        strategy = n_pert % 5
        h_p = best_h.copy()

        if strategy == 0:
            h_p += np.random.normal(0, 0.02, n)
        elif strategy == 1:
            mask = np.random.random(n) < 0.15
            idx = np.where(mask)[0]
            if len(idx) > 0:
                h_p[idx] += np.random.normal(0, 0.12, len(idx))
        elif strategy == 2:
            shift = np.random.randint(-4, 5)
            h_p = np.roll(best_h, shift)
        elif strategy == 3:
            frac = np.random.uniform(0.5, 0.9)
            h_p = frac * best_h + (1.0 - frac) * (best_h[::-1])
        else:
            # Gaussian bump perturbation
            center = np.random.randint(5, n - 5)
            width = np.random.uniform(3, 15)
            xs = np.arange(n)
            bump = np.random.uniform(-0.1, 0.1) * np.exp(-((xs - center) ** 2) / (2 * width ** 2))
            h_p = best_h + bump

        h_p = repair(h_p, n)
        result = slsqp_optimize(h_p, n, 100, maxiter=200)
        c5 = compute_c5(result)
        if c5 < best_c5:
            best_c5 = c5
            best_h = result

    # Phase 3: Project to finer grids and optimize
    for n_fine in [149, 199, 249]:
        if time_left() < 120:
            break
        h_fine = project_fine(best_h, n_fine)
        cur_n = n_fine
        for alpha in [40, 80, 150]:
            if time_left() < 50:
                break
            result = slsqp_optimize(h_fine, cur_n, alpha, maxiter=300)
            c5 = compute_c5(result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result
                n = cur_n
            h_fine = result

    # Phase 4: Final polish with very sharp max approximation
    cur_n = len(best_h)
    for alpha in [300, 500]:
        if time_left() < 20:
            break
        result = slsqp_optimize(best_h, cur_n, alpha, maxiter=300)
        c5 = compute_c5(result)
        if c5 < best_c5:
            best_c5 = c5
            best_h = result

    # Ensure final feasibility
    best_h = repair(best_h, len(best_h))

    # Use the provided evaluator for final score
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = compute_c5(best_h)

    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, float(best_c5), len(best_h)
