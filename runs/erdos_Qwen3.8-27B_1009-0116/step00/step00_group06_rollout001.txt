def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        if remaining() < 60:
            break
        n_try = int(rng.choice(n_values[:3]))
        h_rand = random_binary(n_try, seed + 100 + attempt)
        h_opt, c5_opt = optimize_smooth(h_rand, 2.0 / n_try, max_iters=3000, time_limit=min(120, remaining() * 0.25))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 4: Fine-tune best solution
    if remaining() > 40:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        h_tuned, c5_tuned = optimize_smooth(best_h, dx_best, max_iters=3000, time_limit=min(150, remaining() * 0.4))
        if c5_tuned < best_c5:
            best_c5 = c5_tuned
            best_h = h_tuned
        # Final binary polish
        if remaining() > 30:
            h_bin, c5_bin = coordinate_descent_binary(best_h, dx_best, max_iters=8000, time_limit=min(60, remaining() * 0.3))
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin
    
    # Final projection to ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h.copy(), n_final / 2.0)
    best_c5, _ = compute_c5(best_h, 2.0 / n_final)
    
    return best_h, best_c5, n_final
