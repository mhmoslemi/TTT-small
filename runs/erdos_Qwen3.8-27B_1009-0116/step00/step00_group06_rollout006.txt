import numpy as np
import time


def project_feasible(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def subgrad_active(h, n, n_active=3):
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    idx = np.argsort(corr)[-n_active:]
    g = np.zeros(n)
    for k in idx:
        t = k - (n - 1)
        if t >= 0:
            m = n - t
            g[:m] += (1.0 - h[t:t + m])
            g[t:] -= h[:m]
        else:
            u = -t
            m = n - u
            g[u:] += (1.0 - h[:m])
            g[:m] -= h[u:]
    return g / n_active


def opt_subgrad(h0, n, deadline, t0):
    target = n / 2.0
    h = project_feasible(h0.astype(float).copy(), n, target)
    best = h.copy()
    best_c = c5_of(h, n)
    lr = 0.03
    stuck = 0
    while time.time() < deadline:
        g = subgrad_active(h, n)
        h_new = project_feasible(h - lr * g, n, target)
        c = c5_of(h_new, n)
        if c < best_c - 1e-12:
            best, best_c = h_new.copy(), c
            h = h_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 80:
                lr *= 0.82
                stuck = 0
                if lr < 5e-7:
                    break
    return best, best_c


def opt_sa(h0, n, deadline, t0, rng_seed=0):
    rng = np.random.RandomState(rng_seed)
    target = n / 2.0
    h = project_feasible(h0.astype(float).copy(), n, target)
    best = h.copy()
    best_c = c5_of(h, n)
    cur_c = best_c
    temp = 0.002
    cool = 0.9997
    no_improve = 0
    restarts = 0
    while time.time() < deadline:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.07, 0.07)
        hi, hj = h[i], h[j]
        if hi + d < 0 or hi + d > 1 or hj - d < 0 or hj - d > 1:
            continue
        h[i] = hi + d
        h[j] = hj - d
        nc = c5_of(h, n)
        if nc <= cur_c or (nc - cur_c) / max(temp, 1e-9) < np.log(rng.random() + 1e-15):
            cur_c = nc
            if nc < best_c:
                best_c, best = nc, h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = hi
            h[j] = hj
            no_improve += 1
        temp = max(temp * cool, 2e-6)
        if no_improve > 8000:

            cur_c = c5_of(h, n)
            no_improve = 0
            restarts += 1
            if restarts > 15:
                break
    return best, best_c


def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)
