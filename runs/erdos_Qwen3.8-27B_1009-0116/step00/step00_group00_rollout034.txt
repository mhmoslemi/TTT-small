    n = len(v)
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    c5 = float(np.max(corr) * dx)
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n-shift] += (1.0 - v[shift:])
        grad[shift:] += -v[:n-shift]
    else:

        grad[s:] += (1.0 - v[:n-s])
        grad[:n-s] -= v[s:]

            n_list.append(n_try)

        if time.time() - t0 > budget_s * 0.88:
            break
        remaining = budget_s * 0.88 - (time.time() - t0)
        if remaining < 20:
            break

        if n == n_curr:
            h_start = h_best.copy()
        else:
            h_start = _upsample(h_best, n)

        h_opt, c5_opt = _optimize_slqp(h_start, n, maxiter=400)
        if c5_opt < c5_best - 1e-10:
            h_best = h_opt
            c5_best = c5_opt
            n_curr = n

        # Structured restarts
        for sm in range(4):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_s = _struct_start(n, sm)
            h_opt, c5_opt = _optimize_slqp(h_s, n, maxiter=200)
            if c5_opt < c5_best - 1e-10:
                h_best = h_opt
                c5_best = c5_opt
                n_curr = n

        # Random restarts
        max_restarts = 10 if n <= 90 else 6
        for r in range(max_restarts):
            if time.time() - t0 > budget_s * 0.88:
                break
            scale = 0.015 * (1.0 + 0.4 * r)
            h_rand = h_best + scale * np.random.randn(n)
            h_rand = _project(h_rand, n)
            h_opt, c5_opt = _optimize_slqp(h_rand, n, maxiter=200)
            if c5_opt < c5_best - 1e-10:
                h_best = h_opt
                c5_best = c5_opt
                n_curr = n

    h_best = _project(h_best, len(h_best))
    c5_final = _compute_c5(h_best)
    return h_best, c5_final, len(h_best)
