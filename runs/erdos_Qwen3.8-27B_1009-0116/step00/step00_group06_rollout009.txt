def fix_sum(h, target):

            break

            if cnt == 0: break

            h[mask] = np.minimum(1.0, h[mask] + step)
        else:

            cnt = np.sum(mask)
            if cnt == 0: break
            step = (-diff) / cnt
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

    return np.clip(np.interp(x_f, x_c, h_coarse), 0.0, 1.0)

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(penalty_obj, h_init.copy(), args=(n_pts, alpha, pw),
                  method='SLSQP', bounds=bounds,
                  options={'maxiter': maxiter, 'ftol': 1e-12})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = fix_sum(h_opt, target)
    return h_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    shifted = np.clip((vals - m) * alpha, -400.0, 400.0)
    return float(m + np.log(np.mean(np.exp(shifted))) / alpha)

def penalty_obj(x, n_pts, alpha, pw):
    dx = 2.0 / n_pts
    target = n_pts / 2.0
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    sm = smooth_max(corr, alpha)
    pen = pw * (np.sum(x) - target) ** 2
    return sm + pen

def fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):

        if abs(diff) < 1e-12:

        if diff > 0:

            cnt = np.sum(mask)
            if cnt == 0:

            step = (-diff) / cnt
            h[mask] = np.maximum(0.0, h[mask] - step)
