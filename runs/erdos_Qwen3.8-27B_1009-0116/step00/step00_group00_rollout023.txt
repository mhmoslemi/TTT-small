    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    # Use a small tolerance for active set
    tol = 1e-10
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)

        lag = k - (n - 1)
        if lag >= 0:
            # term1: (1-h[j-lag]) for j >= lag

            if n - lag > 0:
                term1[lag:] = 1.0 - h[:n - lag]
            # term2: h[j+lag] for j < n-lag
            term2 = np.zeros(n)

        else:

            # term2: h[j-a] for j >= a
            term2 = np.zeros(n)
            term2[a:] = h[:n - a]
        grad += dx * (term1 - term2)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 45

def _obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.max(corr))

def _grad(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = float(np.max(corr))
    active = np.where(corr >= mx - 1e-10)[0]
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                g[lag:] += dx * (1.0 - x[:n - lag])
                g[:n - lag] -= dx * x[lag:]
        else:
            a = -lag
            if n - a > 0:
                g[:n - a] += dx * (1.0 - x[a:])
                g[a:] -= dx * x[:n - a]
    return g

def _confun(x):
    return np.sum(x) - _g_n * 0.5

def _conjac(x):
    return np.ones(_g_n)

def c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

            break

        if s > 0:

        else:
            s = max(s, -h[m].min())
