                break
            for freq in [2, 3, 4, 5]:
                h_sin = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * np.arange(n_try) / n_try)
                h_sin = make_feasible(h_sin, n_try)
                h_opt, c5_opt = opt_slsqp(h_sin, n_try, maxiter=2000)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try

    # Phase 5: Coordinate descent refinement
    if time.time() - t0 < budget_s * 0.95:
        h_cd, c5_cd = coord_descent(best_h, best_n, n_rounds=15)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final safety: ensure feasibility and compute exact score
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
