    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    max_c = np.max(corr)
    exp_terms = np.exp(tau * (corr - max_c))
    exp_sum = np.sum(exp_terms)
    obj = max_c + np.log(exp_sum) / tau

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_terms = np.exp(tau * (corr - max_c))
    exp_sum = np.sum(exp_terms)
    obj = max_c + np.log(exp_sum) / tau
    weights = exp_terms / exp_sum
    grad = np.zeros(n)
    ar = np.arange(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-9:
            continue
        shift = k_idx - (n - 1)
        vp = (ar + shift >= 0) & (ar + shift < n)
        vm = (ar - shift >= 0) & (ar - shift < n)
        contrib = np.zeros(n)
        contrib[vp] += 1.0 - h[ar[vp] + shift]
        contrib[vm] -= h[ar[vm] - shift]
        grad += w * dx * contrib
    return obj, grad


def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize_push(h, n):
    """Push values toward 0 or 1 while maintaining sum constraint."""
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        # Identify values close to 0 or 1
        close_to_1 = (h > 0.9)
        close_to_0 = (h < 0.1)
        mid = (~close_to_1) & (~close_to_0)
        if not np.any(mid):
            break
        # Try pushing mid values toward 0 or 1
        excess = np.sum(h) - target
        if abs(excess) < 1e-12:
            # Try to push anyway for quality
            h[close_to_1] = 1.0
            h[close_to_0] = 0.0
            # Fix sum
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
            # Adjust mid values
            if np.sum(mid) > 0:
                h[mid] += diff / np.sum(mid)
                h = np.clip(h, 0.0, 1.0)
            else:
                break
        else:
            break
    return make_feasible(h, n)

    try:
        res = minimize(
            lambda x: lse_obj_grad(x, tau)[0],
            h_start,
            jac=lambda x: lse_obj_grad(x, tau)[1],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if res.success or res.fun < compute_c5(h_start):
            h_new = make_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h_start
