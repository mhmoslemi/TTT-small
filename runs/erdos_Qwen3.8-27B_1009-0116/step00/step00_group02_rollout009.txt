import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(corr, t):
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    shifted = np.clip(t * (corr - m), -500.0, 0.0)
    return m + np.log(np.mean(np.exp(shifted))) / t


def objective_fn(h, dx, t):
    corr = np.correlate(h, 1 - h, mode="full")
    return smooth_max_val(corr, t) * dx


def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)


def constraint_jac(h):
    return np.ones(len(h))


def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h


def build_bounds(n):
    return [(0.0, 1.0)] * n


def build_constraints():
    return [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]


def subgradient_step(h, dx, t, lr, target_sum):
    corr = np.correlate(h, 1 - h, mode="full")
    weights = np.exp(np.clip(t * (corr - np.max(corr)), -500, 0))
    weights /= np.sum(weights)
    grad = np.zeros_like(h)
    for k in range(len(corr)):
        idx = k - (len(h) - 1)
        if 0 <= idx < len(h):
            grad[:len(h) - idx] += weights[k] * (1 - h[idx:len(h)])
            grad[idx + len(h):] += -weights[k] * (1 - h[idx:])
        elif idx > 0:
            grad[:len(h) - idx] += weights[k] * (1 - h[idx:])
        elif idx < -len(h) + 1:
            grad[idx + 1:] += -weights[k] * (1 - h[:idx + 1 + len(h)])
    h_new = h - lr * grad
    h_new = fix_sum(np.clip(h_new, 0, 1), target_sum)
    return h_new
