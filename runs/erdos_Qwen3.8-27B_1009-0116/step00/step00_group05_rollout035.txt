        # Need to add mass: add to elements with room
        mask = h < 1.0
        if np.any(mask):
            room = (1.0 - h[mask]).sum()
            add = min(diff, room)
            h[mask] += add * (1.0 - h[mask]) / room
        # If still not enough, redistribute
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-10:

            h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    val = corr.max() * dx
    grad = np.zeros(n)
    for j in range(n):
        jm = j - shift
        if 0 <= jm < n:
            grad[j] += (1.0 - h[jm])
        jp = j + shift
        if 0 <= jp < n:
            grad[j] -= h[jp]
    grad *= dx
    return val, grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = (1.0 - h[mask]).sum()
            if room > 0:
                h[mask] += (diff / room) * (1.0 - h[mask])
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            mass = h[mask].sum()
            if mass > 0:
                h[mask] += (diff / mass) * h[mask]
    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h[int(n/2)] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    cons = [{'type': 'eq',

        return h
