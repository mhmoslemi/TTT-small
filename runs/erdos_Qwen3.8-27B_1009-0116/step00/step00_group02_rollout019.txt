where const_l is the sum over all other i≠j and i≠j-l terms.

So overlap_l as a function of h[j] is:
= dx * [h[j] - h[j]*h[j+l] - h[j-l]*h[j] + h[j-l]*h[j+l] + const_l] (for the terms involving j)
= dx * [-h[j+l]*h[j] - h[j-l]*h[j] + h[j] + ...]
= dx * [h[j]*(1 - h[j+l] - h[j-l]) + ...]

So it's linear in h[j]! (The -h[j]*h[j] term doesn't appear because j and j+l are different indices unless l=0, and for l=0, h[j]*(1-h[j]) which IS quadratic.)

For l=0: overlap_0 contains h[j]*(1-h[j]) which is -h[j]^2 + h[j], a concave quadratic.
For l≠0: overlap_l is linear in h[j].

So the max of these functions is: max of {concave quadratic, linear functions}. This is a concave function of h[j]!

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    W = np.exp(alpha * (overlaps - m))
    ws = W.sum()
    if ws < 1e-30:
        W = np.ones(2*n-1) / (2*n-1)
    else:
        W /= ws
    grad = np.zeros(n)
    for idx in range(2*n-1):
        w = W[idx]
        if w < 1e-12:
            continue
        lag = idx - (n-1)
        if lag >= 0:
            nv = n - lag
            if nv > 0:
                grad[:nv] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:nv]
        else:
            al = -lag
            nv = n - al
            if nv > 0:
                grad[al:] += w * (1.0 - h[:nv])
                grad[:nv] -= w * h[al:]
    return dx * grad

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff/n, 0.0, 1.0)
    return h

    """Coordinate descent: for each j, try values 0..1 in steps, pick best."""
    values = np.linspace(0, 1, 11)

                best_val = v
