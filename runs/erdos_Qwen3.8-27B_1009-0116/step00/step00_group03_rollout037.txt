def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    return np.max(vals), vals

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    # Final fix for numerical issues
    s = np.sum(h)
    if abs(s - target_sum) < 1e-6:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)

    c5 = np.max(vals)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            c = float(np.sum(m))
            if c == 0:
                break
            h[m] = np.minimum(1.0, h[m] + d / c)
        else:
            m = h > 0.0
            c = float(np.sum(m))
            if c == 0:
                break
            h[m] = np.maximum(0.0, h[m] + d / c)
    return h

def binary_search(h, dx, target, t0, t_limit, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    corr = compute_corr(h)
    max_c = np.max(corr)
    c5_best = max_c * dx
    best_h = h.copy()
    stall = 0
    while time.time() - t0 < t_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in ones:
            if improved:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_corr = compute_corr(h)
                nc = np.max(new_corr)
                if nc < max_c:
                    corr = new_corr
                    max_c = nc
                    c5_best = nc * dx
                    best_h = h.copy()
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
        if improved:
            stall = 0
            continue
        found = False
        npairs = min(3, max(1, len(ones) // 2))
        ntries = 2000
        for _ in range(ntries):
            if time.time() - t0 > t_limit:
                break
            idx1 = rng.choice(ones, npairs, replace=False)
            idx2 = rng.choice(zeros, npairs, replace=False)
            for ii in idx1:
                h[ii] = 0.0
            for jj in idx2:
                h[jj] = 1.0
            nc = np.max(compute_corr(h))
            if nc < max_c:
                corr = compute_corr(h)
                max_c = nc
                c5_best = nc * dx
                best_h = h.copy()
                found = True
                stall = 0
            else:
                for ii in idx1:
                    h[ii] = 1.0
                for jj in idx2:
                    h[jj] = 0.0
            if found:
                break
        if not found:
            stall += 1
            if stall % 8 == 0:
                h = best_h.copy()
                corr = compute_corr(h)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(min(stall, 10)):
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h[i] = 0.0
                    h[j] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
            if stall > 60:
                break
    return best_h, c5_best

def cont_refine(h, dx, target, t0, t_limit, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    corr = compute_corr(h)
    vals = corr * dx
    c5_best = np.max(vals)
    best_h = h.copy()
    lr = 0.01
    for it in range(200000):
        if time.time() - t0 > t_limit:
            break
        c5_cur = np.max(vals)
        if c5_cur < c5_best:
            c5_best = c5_cur
            best_h = h.copy()
        active = np.where(vals >= c5_cur - 0.003)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        grad = np.zeros(n)
        for idx_a in active:
            k = idx_a - (n - 1)
            i_min = max(0, -k)
            i_max = min(n, n - k)
            if i_max <= i_min:
                continue
            ia = np.arange(i_min, i_max)
            ja = ia + k
            grad[ja] += (1.0 - h[ia])
            grad[ia] += (-h[ja])
        grad /= len(active)
        h_new = project_feasible(h - lr * grad, target)
        corr_new = compute_corr(h_new)
        c5_new = np.max(corr_new) * dx
        if c5_new < c5_cur:
            h = h_new
            corr = corr_new
            vals = corr * dx
        else:
            lr *= 0.85
            if lr < 0.0005:
                lr = 0.01
                h = project_feasible(h + rng.normal(0, 0.02, n), target)
                corr = compute_corr(h)
                vals = corr * dx
        if it % 2000 == 0:
            hr = (h > 0.5).astype(float)
            s = int(np.sum(hr))
            tc = int(target)
            if s > tc:
                os_ = np.where(hr == 1)[0]
                hr[os_[np.argsort(h[os_])[:s - tc]]] = 0.0
            elif s < tc:
                oz = np.where(hr == 0)[0]
                hr[oz[np.argsort(-h[oz])[:tc - s]]] = 1.0
            c5r = np.max(compute_corr(hr)) * dx
            if c5r < c5_best:
                c5_best = c5r
                best_h = hr.copy()
                h = hr.copy()
                corr = compute_corr(h)
                vals = corr * dx
    return best_h, c5_best

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
