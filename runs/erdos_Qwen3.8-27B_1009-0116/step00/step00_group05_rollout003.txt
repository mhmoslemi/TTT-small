import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(vals, T):
    m = np.max(vals)
    return m + float(np.log(np.sum(np.exp(T * (vals - m))))) / T

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h, n, max_rounds=40):
    h = (h > 0.5).astype(float)
    s = int(np.round(np.sum(h)))
    if s > n // 2:
        idx = np.where(h == 1)[0]
        for i in range(s - n // 2):
            h[idx[i]] = 0
    elif s < n // 2:
        idx = np.where(h == 0)[0]
        for i in range(n // 2 - s):
            h[idx[i]] = 1
    best = compute_c5(h, n)
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bd = 1e-15
        ba, bb = -1, -1
        for a in ones:
            for b in zeros:
                h[a], h[b] = 0.0, 1.0
                c = compute_c5(h, n)
                if c < best - bd:
                    bd = best - c
                    ba, bb = a, b
                h[a], h[b] = 1.0, 0.0
        if ba >= 0:
            h[ba], h[bb] = 0.0, 1.0
            best -= bd
        else:
            break
    return h, best
