import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = beta * corr * dx
    m = float(np.max(vals))
    if m > 500.0:
        return m / beta
    return (m + float(np.log(np.sum(np.exp(vals - m))))) / beta

def sum_eq_fun(x, target):
    return float(np.sum(x) - target)
