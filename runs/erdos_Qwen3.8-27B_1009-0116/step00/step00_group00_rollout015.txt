1. When i = m: the term is h[m] * (1 - h[k-m]). Derivative wrt h[m]: (1 - h[k-m]) [treating h[k-m] as constant if k-m ≠ m]
   If k-m = m (i.e., k=2m): the term is h[m]*(1-h[m]) = h[m] - h[m]^2. Derivative: 1 - 2*h[m].

2. When k-i = m, i.e., i = k-m: the term is h[k-m] * (1 - h[m]). Derivative wrt h[m]: -h[k-m].
   If k-m = m: this is the same term as above.

Case k ≠ 2m (so m ≠ k-m):
d/dh[m] = (1 - h[k-m]) - h[k-m] = 1 - 2*h[k-m]

Case k = 2m (so m = k-m):
d/dh[m] = 1 - 2*h[m]

In both cases, the answer is 1 - 2*h[k-m] (when 0<=k-m<n), since in the second case k-m=m.

So: d(corr[k])/d(h[m]) = 1 - 2*h[(k-m) mod... ]

d(corr[k])/d(h[m]) = { 1 - 2*h[k-m] if 0 <= k-m < n, else 0 }

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

def project_feasible(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj_grad(h, dx, s=8.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = s * (corr - max_c)
    log_sum = np.logaddexp.reduce(shifted)
    obj = max_c + (log_sum - np.log(len(corr))) / s
    weights = np.exp(shifted - log_sum)
    active = np.where(weights > 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        m = k
        jlo = max(0, m - n + 1)
        jhi = min(n - 1, m)
        if jlo > jhi:
            continue
        h_idx = m - np.arange(jlo, jhi + 1)
        grad[jlo:jhi+1] += weights[k] * (1.0 - 2.0 * h[h_idx]) * dx
    return obj, grad

def grad_opt(h0, target_sum, dx, max_iter=3000, t0=None, budget=1100):

    h = h0.copy()

def slsqp_opt(h0, target_sum, dx, max_iter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x)
    def obj_jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m_star = np.argmax(corr)
        jlo = max(0, m_star - n + 1)
        jhi = min(n - 1, m_star)
        g = np.zeros(n)
        if jlo <= jhi:
            h_idx = m_star - np.arange(jlo, jhi + 1)
            g[jlo:jhi+1] = (1.0 - 2.0 * x[h_idx]) * dx
        return obj(x), g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success:
            h_res = project_feasible(res.x, target_sum)
            return h_res
    except Exception:
        pass
    return None

                best_h = h_pol
    
    return best_h, best_c5, len(best_h)
