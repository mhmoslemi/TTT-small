def subgradient_step(h, n, step_size, beta=10):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    k_star = np.argmax(c)
    
    # Compute gradient of c_{k_star} with respect to h
    # c_{k_star} = sum_i h[i] * (1-h[i+k']) * dx
    # where k' = k_star - (n-1) (the actual shift)
    shift = k_star - (n - 1)
    
    grad = np.zeros(n)
    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            # h[i] appears: grad += (1-h[j]) * dx
            # h[j] appears (j = i+shift): grad += -h[i] * dx
            grad[i] += (1 - h[j]) * dx
            grad[j] += (-h[i]) * dx
    
    # Wait, h[i] can appear in both terms when shift = 0
    # But in general, for a given k*, each h[m] appears in at most 2 pairs
    # Actually for a fixed shift, h[m] appears as the first element (pair (m, m+shift))
    # and as the second element (pair (m-shift, m))
    
    # Let me redo this:
    # c = sum_{i: 0<=i<n and 0<=i+shift<n} h[i] * (1-h[i+shift]) * dx
    # dc/dh[m] = dx * [(1-h[m+shift]) if 0<=m+shift<n - (h[m-shift]) if 0<=m-shift<n - (if m = m+shift, i.e., shift=0: extra -h[m])
    # Wait, let me be more careful.
    # 
    # c = sum_i h[i]*(1-h[i+s]) * dx where s = shift, sum over valid i
    # = sum_i h[i] * dx - sum_i h[i]*h[i+s] * dx
    #
    # dc/dh[m] = dx - (h[m+s] * dx + h[m-s] * dx) if both valid
    #           + extra term if shift=0 (since h[m]*h[m] = h[m]^2, derivative is 2h[m])
    
    # Let me just compute it directly:
    grad = np.zeros(n)
    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            # Term: h[i] * (1 - h[j]) * dx
            # d/dh[i]: (1-h[j]) * dx
            # d/dh[j]: -h[i] * dx
            grad[i] += (1 - h[j]) * dx
            grad[j] += (-h[i]) * dx
    
    # Apply projected subgradient step
    h_new = h - step_size * grad
    h_new = project_onto_constraint(h_new, n)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_constraint(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fix_sum(h, n):
    return project_constraint(h, n)

def upsample(init_arr, n):
    n0 = len(init_arr)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x1, x0, init_arr)
    return fix_sum(h, n)

def lag_gradient(h, n, shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    i_start = max(0, -shift)
    i_end = min(n, n - shift)
    for i in range(i_start, i_end):
        j = i + shift
        grad[i] += (1.0 - h[j]) * dx
        grad[j] += (-h[i]) * dx
    return grad

    decay = 0.9997

        c_max = np.max(c)
        shifted = np.clip(beta * (c - c_max), -40, 40)
