def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

    h = np.zeros(n)
    idx = np.random.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def simulate_annealing(h, steps=100000, T_start=1.0, T_end=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(h)
    h = np.copy(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    c5 = best_c5
    T = T_start
    
    for s in range(steps):
        T = T_start * (T_end / T_start) ** (s / steps)
        
        # Pick a random 1 and a random 0 to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        new_c5 = compute_c5(h)

            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
        else:
            h[i], h[j] = h[j], h[i]
    
    return h, best_c5

        if c5_up < best_c5:

            best_h = h_up
    
    elapsed = time.time() - start_time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def steepest_descent(h, max_rounds=5):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        ones = list(np.flatnonzero(h))
        zeros = list(np.flatnonzero(1.0 - h))

                h[j] = 1.0
