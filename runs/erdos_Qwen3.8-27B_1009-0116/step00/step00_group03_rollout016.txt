import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_smooth(h, dx, beta=40.0):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def repair(h, target_sum):
    h = np.array(h, dtype=float).copy()
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, target_sum / n)
    h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        rem = diff
        for idx in order:
            if rem > 0:
                step = min(rem, 1.0 - h[idx])
            else:
                step = max(rem, -h[idx])
            h[idx] += step
            rem -= step
            if abs(rem) < 1e-14:
                break
    return h
