            h_ls, c5_ls = greedy_ls(h_sa, min(20.0, deadline - time.time() - 20))
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h_bin_best = binarize(best_h)

    # Phase 4: Continuous refinement
    t_rem = deadline - time.time()
    if t_rem > 40:
        h_c, c5_c = cont_refine(best_h, min(120.0, t_rem * 0.5))
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 5: More binary search from best
    t_rem = deadline - time.time()
    if t_rem > 60:
        h_b = binarize(best_h)
        h_sa, c5_sa = sa_binary(h_b, min(200.0, t_rem * 0.5), T0=0.35, T1=0.004)
        h_ls, c5_ls = greedy_ls(h_sa, min(50.0, max(deadline - time.time() - 30, 10)))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 6: Final continuous polish
    t_rem = deadline - time.time()
    if t_rem > 30:
        h_c, c5_c = cont_refine(best_h, min(80.0, t_rem - 5))
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Final: also check binary version
    h_b_final = binarize(best_h)
    c5_b_final = compute_c5(h_b_final)
    if c5_b_final < best_c5:
        best_h = h_b_final
        best_c5 = c5_b_final

    # One last greedy polish
    t_rem = deadline - time.time()
    if t_rem > 5:
        h_ls, c5_ls = greedy_ls(best_h, t_rem)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # Final validation
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
