import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def compute_c5(h, n):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

def smooth_max(vals, alpha):
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -500.0, 500.0)
    return float(m + (1.0 / alpha) * np.log(np.sum(np.exp(shifted))))

def objective_fn(h, params):
    n, alpha = params
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(vals, alpha)

def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def time_exceeded(start_time, budget_s, frac=0.85):
    return (time.time() - start_time) > budget_s * frac

def binary_local_search(h, n, max_flips=5000):
    """Flip bits to reduce C5. Only useful for near-binary solutions."""
    dx = 2.0 / n
    target = n / 2.0
    cur = h.copy().round().astype(float)
    # Ensure exactly n//2 ones
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur < 0.5)[0]
    while len(ones) > n // 2:
        cur[ones[-1]] = 0.0
        ones = ones[:-1]
    while len(ones) < n // 2:
        cur[zeros[-1]] = 1.0
        zeros = zeros[:-1]
    cur_c5 = compute_c5(cur, n)
    improved = True
    flips = 0
    while improved and flips < max_flips:
        improved = False
        ones_idx = np.where(cur > 0.5)[0]
        zeros_idx = np.where(cur < 0.5)[0]
        for oi in ones_idx:
            for zi in zeros_idx:
                if oi == zi:
                    continue
                cur[oi], cur[zi] = 0.0, 1.0
                new_c5 = compute_c5(cur, n)
                if new_c5 < cur_c5 - 1e-12:
                    cur_c5 = new_c5
                    improved = True
                    flips += 1
                    break
            if improved:
                continue
            cur[oi], cur[zi] = 1.0, 0.0
    return cur, cur_c5
