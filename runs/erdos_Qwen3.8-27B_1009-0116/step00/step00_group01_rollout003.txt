        lags = np.array([m_star - (n - 1)])
    else:
        top_indices = np.argsort(corr)[-topk:]
        lags = top_indices - (n - 1)
    grad = np.zeros(n)
    j_arr = np.arange(n)
    for lag in lags:
        idx_a = j_arr - lag
        valid_a = (idx_a >= 0) & (idx_a < n)
        grad[valid_a] += (1.0 - h[idx_a[valid_a]]) * dx
        idx_b = j_arr + lag
        valid_b = (idx_b >= 0) & (idx_b < n)
        grad[valid_b] -= h[idx_b[valid_b]] * dx
    if topk > 1:
        grad /= topk
    return grad

def obj_pen(h, n, mu, topk):
    s = np.sum(h) - n / 2.0
    return compute_c5(h) + mu * s * s

def obj_pen_grad(h, n, mu, topk):
    g = c5_grad(h, topk)
    g += 2.0 * mu * (np.sum(h) - n / 2.0)
    return g

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def try_optimize(h_start, n, bounds, mu, topk, maxiter, t0, budget_s, frac):
    if time.time() - t0 > budget_s * frac:
        return h_start, compute_c5(h_start)
    try:
        res = minimize(obj_pen, h_start, args=(n, mu, topk), jac=lambda x: obj_pen_grad(x, n, mu, topk),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
        h_new = project(res.x, n)
    except Exception:
        h_new = project(h_start, n)
    return h_new, compute_c5(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with increasing penalty (topk=1 for sharp gradient)
    h_curr = h0.copy()
    for mu in [10.0, 100.0, 1000.0, 1e5, 1e6]:
        if time.time() - t0 > budget_s * 0.45:
            break
        h_new, c5_new = try_optimize(h_curr, n, bounds, mu, 1, 300, t0, budget_s, 0.45)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new.copy()

    # Phase 2: Smooth gradient (topk=3) to escape flat regions
    if time.time() - t0 < budget_s * 0.55:
        h_curr = best_h.copy()
        for mu in [1e5, 1e6]:
            if time.time() - t0 > budget_s * 0.55:
                break
            h_new, c5_new = try_optimize(h_curr, n, bounds, mu, 3, 200, t0, budget_s, 0.55)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_curr = h_new.copy()

    # Phase 3: Nelder-Mead local refinement
    if time.time() - t0 < budget_s * 0.70:
        h_curr = best_h.copy()
        try:
            res = minimize(obj_pen, h_curr, args=(n, 1e6, 1),
                           method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-14,
                                    'adaptive': True, 'maxls': 50})
            h_new = project(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 4: Random restarts with local optimization
    if time.time() - t0 < budget_s * 0.88:
        for trial in range(50):
            if time.time() - t0 > budget_s * 0.88:
                break
            scale = 0.05 + 0.1 * (trial % 5)
            h_trial = best_h + rng.randn(n) * scale
            h_trial = project(h_trial, n)
            h_new, c5_new = try_optimize(h_trial, n, bounds, 1e6, 1, 150, t0, budget_s, 0.88)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Fine-tune with very small steps
    if time.time() - t0 < budget_s * 0.95:
        for trial in range(30):
            if time.time() - t0 > budget_s * 0.95:
                break
            idx = rng.randint(n)
            delta = rng.uniform(-0.02, 0.02)
            h_trial = best_h.copy()
            h_trial[idx] = np.clip(h_trial[idx] + delta, 0.0, 1.0)
            h_trial = project(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()

    # Final evaluation
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
