from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute proportionally
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        h = np.clip(h, 0, 1)
    return h

def local_search_binary(h, n, max_iter=10000, rng=None):
    """Local search on 0-1 sequences with n/2 ones"""
    if rng is None:
        rng = np.random.RandomState(42)
    h = np.round(h).astype(float)
    # Ensure exactly n/2 ones
    ones = np.sum(h > 0.5)
    if ones != n // 2:
        # Fix
        idx = np.argsort(h)
        h = np.zeros(n)
        h[idx[n - n//2:]] = 1.0
    else:
        h = (h > 0.5).astype(float)
    
    best_c5 = compute_c5(h)

    zeros_idx = np.where(h == 0)[0]

def slsqp_optimize(h0, n, maxiter=200):

    def objective(h):

        h_opt = h_opt + diff / n
        h_opt = np.clip(h_opt, 0, 1)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha=100):
    """Smooth approximation of max"""
    return np.log(np.sum(np.exp(alpha * vals))) / alpha

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Water filling
        for _ in range(100):
            if diff > 0:
                mask = h < 1.0 - 1e-12
            else:
                mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] += step
            diff = target - np.sum(h)
            if abs(diff) < 1e-10:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(h0, n, maxiter=300):
    def obj(h):
        return compute_c5(h)
    def con(h):
        return np.sum(h) - n / 2.0

                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':con}],
                   options={'maxiter':maxiter, 'ftol':1e-8})

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        for _ in range(200):
            if diff > 0:
                mask = h < 1.0 - 1e-14
            else:
                mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] += step
            diff = n / 2.0 - np.sum(h)
            if abs(diff) < 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(h0, n, maxiter=300, ftol=1e-10):
    def obj(h):
        return compute_c5(h)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    try:
        res = minimize(obj, h0.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_feasible(res.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0.copy(), compute_c5(h0)
