from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def max_corr_val(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))


def project_constraint(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 0.9999
        if not mask.any():
            return h
        room = np.zeros_like(h)
        room[mask] = 1.0 - h[mask]
        h += room / room[mask].sum() * diff
    else:
        mask = h > 1e-8
        if not mask.any():
            return h
        room = h.copy()
        h -= room / room[mask].sum() * (-diff)
    return np.clip(h, 0.0, 1.0)


def make_binary(h, n):
    order = np.argsort(-h)
    h_b = np.zeros(n)
    k = n // 2
    h_b[order[:k]] = 1.0
    if n % 2 == 1:
        h_b[order[k]] = 0.5
    return h_b


def thue_morse(n):
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = bin(i).count('1') % 2
    return tm


def sa_search(h_init, n, max_iters, seed, t_init=1.0, cool=0.99995):
    rng = np.random.RandomState(seed)
    h_cur = make_binary(h_init, n)
    mc_cur = max_corr_val(h_cur)
    h_best = h_cur.copy()
    mc_best = mc_cur
    T = t_init
    for it in range(max_iters):
        if T < 1e-4:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        mc_new = max_corr_val(h_cur)
        delta = mc_new - mc_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            mc_cur = mc_new
            if mc_new < mc_best:
                mc_best = mc_new
                h_best = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        T *= cool
    return h_best, mc_best * (2.0 / n)

            best_h = h_s

def nelder_refine(h_init, n, maxiter, t0, deadline):
    dx = 2.0 / n

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        p = 200.0 * (np.sum(h) - n / 2.0) ** 2
        return max_corr_val(h) * dx + p

    try:

                       options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
        h_res = project_constraint(res.x, n)
        return h_res, compute_c5(h_res, dx)
    except Exception:
        return h_init, compute_c5(h_init, dx)
