```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C5 value"""
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return corr.max()

def smooth_max_obj_and_grad(h, dx, T=20.0):
    """Compute smooth max of correlation and its gradient"""
    n = len(h)
    corr_raw = np.correlate(h, 1 - h, mode="full")
    corr = corr_raw * dx
    
    cm = corr.max()
    # Log-sum-exp for numerical stability
    shifted = (corr - cm) * T
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    S = cm + log_sum / T
    
    # Weights p_k
    p = np.exp(shifted - log_sum)
    
    # Gradient computation
    grad = np.zeros(n)
    # Only iterate over k where p[k] is significant
    threshold = 1e-8
    active = np.where(p > threshold)[0]
    
    for k in active:
        # First term: (1 - h[m+k-n+1]) for valid m+k-n+1 in [0,n-1]
        j_range_start = max(0, n - 1 - k)
        j_range_end = min(n - 1, 2*n - 2 - k)
        if j_range_start <= j_range_end:
            indices = np.arange(j_range_start, j_range_end + 1)
            second_idx = indices + k - n + 1
            grad[indices] += p[k] * dx * (1 - h[second_idx])
            # Second term: -h[m-k+n-1] for valid m-k+n-1 in [0,n-1]
            # m-k+n-1 in [0,n-1] => 1-n+k ≤ m ≤ n-1-k+n-1... 
            # m in [k-n+1, k+n-1] intersected with [0, n-1]
            s_start = max(0, k - n + 1)
            s_end = min(n - 1, k + n - 1)
            if s_start <= s_end:
                m_indices = np.arange(s_start, s_end + 1)
                first_idx = m_indices - k + n - 1
```

```python
import numpy as np
```

```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def c5_smooth_obj_grad(h, dx, T=50.0):
    N = len(h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    cm = corr.max()
    shifted = np.clip((corr - cm) * T, -400.0, 400.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    S = cm + log_sum / T
    p = np.exp(shifted - log_sum)
    grad = np.zeros(N)
    active = np.where(p > 1e-10)[0]
    for j in active:
        L = max(0, N - 1 - j)
        U = min(N - 1, 2 * N - 2 - j)
        pj = p[j] * dx
        m1s = max(0, L + j - N + 1)
        m1e = min(N - 1, U + j - N + 1)
        if m1s <= m1e:
            ms = np.arange(m1s, m1e + 1)
            ns = ms - j + N - 1
            grad[ms] += pj * (1.0 - h[ns])
        if L <= U:
            ms2 = np.arange(L, U + 1)
            hs = ms2 + j - N + 1
            grad[ms2] -= pj * h[hs]
    return S, grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)

            break

                break
            d = diff / len(idx)
            d = min(d, room[idx].min() * 0.99)
            h[idx] += d
        else:
            room = h
            idx = np.where(room > 1e-12)[0]
            if len(idx) == 0:
                break
            d = (-diff) / len(idx)
            d = min(d, room[idx].min() * 0.99)

        h_new = project_sum(h_new, target)
```
