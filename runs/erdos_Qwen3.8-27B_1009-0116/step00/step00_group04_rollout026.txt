def _compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _sum_eq(h):
    return float(np.sum(h) - _N / 2.0)

def _sum_jac(h):
    return np.ones(_N)

def _slsqp_optimize(h0):
    h0 = np.clip(h0.copy(), 0.0, 1.0)
    diff = _N / 2.0 - np.sum(h0)
    h0 = h0 + diff / _N
    h0 = np.clip(h0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * _N
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    res = minimize(_compute_c5, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = np.clip(res.x, 0.0, 1.0)
        diff = _N / 2.0 - np.sum(h_new)
        h_new = h_new + diff / _N
        h_new = np.clip(h_new, 0.0, 1.0)
        if _compute_c5(h_new) < _compute_c5(h0):
            return h_new
    return h0

def _local_search(h, n_iter, rng):
    h = h.copy()
    best_val = _compute_c5(h)
    n = _N
    step_size = 0.1
    stall = 0
    
    for _ in range(n_iter):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(-step_size, step_size)
        h_new = h.copy()
        h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h[j] - delta, 0.0, 1.0)
        # Repair integral
        diff = n / 2.0 - np.sum(h_new)
        if abs(diff) > 1e-10:
            h_new = h_new + diff / n
            h_new = np.clip(h_new, 0.0, 1.0)
        
        val = _compute_c5(h_new)
        if val < best_val:
            best_val = val
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                step_size *= 0.9
                stall = 0
    
    return h, best_val

def _interpolate_up(h, new_n):

    new_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    target = new_n / 2.0
    for _ in range(10):
        diff = target - np.sum(h_new)
        h_new = h_new + diff / new_n
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

_N = 42

def _compute_c5(h):
    dx = 2.0 / _N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _sum_eq(h):
    return float(np.sum(h) - _N / 2.0)

def _sum_jac(h):
    return np.ones(_N)

def _repair(h):
    target = _N / 2.0
    for _ in range(20):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        h = h + (target - s) / _N
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_optimize(h0, max_iter=5000):
    h0 = _repair(h0.copy())
    bounds = [(0.0, 1.0)] * _N
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    best = _compute_c5(h0)
    res = minimize(_compute_c5, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = _repair(res.x)
        val = _compute_c5(h_new)
        if val < best:
            return h_new, val
    return h0, best

def _local_search(h, n_iter, rng, step_init=0.05):
    h = _repair(h.copy())
    best_val = _compute_c5(h)
    n = _N
    step = step_init
    stall = 0

    for _ in range(n_iter):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = rng.uniform(-step, step)
        h_new = h.copy()
        h_new[i] = min(1.0, max(0.0, h[i] + delta))
        h_new[j] = min(1.0, max(0.0, h[j] - delta))
        h_new = _repair(h_new)
        val = _compute_c5(h_new)
        if val < best_val - 1e-12:
            best_val = val
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                step *= 0.92
                stall = 0
    return h, best_val

def _interpolate_up(h, new_n):
    old_n = len(h)
    if new_n <= old_n:
        return h[:new_n].copy()
    old_x = (np.arange(old_n) + 0.5) * 2.0 / old_n
    new_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    _N_save = _N
    _N = new_n
    h_new = _repair(h_new)
    _N = _N_save
    return h_new

def _random_perturb(h, rng, scale=0.1):
    h_new = h + rng.uniform(-scale, scale, size=len(h))
    h_new = np.clip(h_new, 0.0, 1.0)
    return _repair(h_new)
