    return float(np.max(corr) * dx)

        if cnt == 0:

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0

def slsqp_step(h, n, timeout=10):
    """One SLSQP optimization attempt."""
    target = n / 2.0
    def objective(x):
        return general_compute_c5(x)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bounds = [(0.0, 1.0)] * n
    
    t0 = time.time()
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})
        if time.time() - t0 > timeout:
            return h
        x = repair_h(res.x, n)
        if general_compute_c5(x) < general_compute_c5(h):
            return x
    except Exception:
        pass
    return h

    if rng is None:
        rng = np.random.default_rng()

def sa_binary(h_bin, n, t_start, t_end, rng):

    target = n // 2
    h_bin = h_bin.copy()
    best_h = h_bin.copy()
    best_c5 = general_compute_c5(h_bin)
    cur_c5 = best_c5
    total_time = t_end - t_start
    T = 0.01
    T_min = 0.001
    
    while time.time() < t_end:
        elapsed = time.time() - t_start
        frac = min(1.0, elapsed / total_time)
        T_cur = T * (T_min / T) ** frac
        
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        i = int(ones[rng.integers(len(ones))])

            hb[i] = 0.0
            for j in zeros:
                hb[j] = 1.0
                v = general_compute_c5(hb)
                if v < best_c5 - 1e-15:
                    best_c5 = v
                    best_ij = (i, j)
                hb[j] = 0.0
            hb[i] = 1.0
        if best_ij is None:
            break
        hb[best_ij[0]] = 0.0
        hb[best_ij[1]] = 1.0
        best = best_c5
    return hb, best

def random_swap_search(hb, n, rng, max_rounds=2000, time_limit=30.0):
    hb = hb.copy()
    t_start = time.time()
    best = general_compute_c5(hb)
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        for _ in range(150):
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            hb[i] = 0.0
            hb[j] = 1.0
            v = general_compute_c5(hb)
            if v < best - 1e-15:
                best = v
                improved = True
            else:
                hb[i] = 1.0
                hb[j] = 0.0
        if not improved:
            break
    return hb, best

def sa_search(hb, n, rng, t_end):
    hb = hb.copy()
    best_h = hb.copy()
    best_c5 = general_compute_c5(hb)
    cur_c5 = best_c5
    t0 = time.time()
    total = max(t_end - t0, 1.0)
    while time.time() < t_end:
        frac = min(1.0, (time.time() - t0) / total)
        T = 0.015 * (0.05) ** frac
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        hb[i] = 0.0
        hb[j] = 1.0
        v = general_compute_c5(hb)
        d = v - cur_c5
        if d < 0 or rng.random() < math.exp(-d / max(T, 1e-12)):
            cur_c5 = v
            if v < best_c5:
                best_c5 = v
                best_h = hb.copy()
        else:
            hb[i] = 1.0
            hb[j] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(70, 0.5)
    n = len(h0)

    best_h = repair_h(h0, n)
    best_c5 = general_compute_c5(best_h)

    # Phase 1: SLSQP continuous optimization (130s)
    t_p1 = t0 + 130
    attempt = 0
    while time.time() < t_p1:
        start = best_h.copy()
        if attempt > 0:
            sigma = 0.1 + 0.1 * (attempt % 3)
            start = repair_h(start + rng.normal(0, sigma, n), n)
        new_h = slsqp_optimize(start, n, maxiter=60)
        new_c5 = general_compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        attempt += 1

    # Phase 2: Binary local search (500s)
    t_p2 = t0 + 630
    hb = binary_project(best_h, n)
    hb_c5 = general_compute_c5(hb)
    if hb_c5 < best_c5:
        best_h = hb.copy()
        best_c5 = hb_c5

    stall_count = 0
    while time.time() < t_p2:
        # Exhaustive single-swap refinement
        hb, c5 = full_swap_refine(hb, n, max_rounds=4000, time_limit=80.0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hb.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Perturbed restart
        sigma = 0.3 + 0.2 * stall_count
        hp = repair_h(hb + rng.normal(0, sigma, n), n)
        hb2 = binary_project(hp, n)
        hb2, c52 = full_swap_refine(hb2, n, max_rounds=2000, time_limit=50.0)
        if c52 < best_c5:
            best_c5 = c52
            best_h = hb2.copy()
            hb = hb2.copy()
            stall_count = 0
        else:
            # Random swap search as additional escape
            hb3 = binary_project(repair_h(best_h + rng.normal(0, 0.5, n), n), n)
            hb3, c53 = random_swap_search(hb3, n, rng, max_rounds=1500, time_limit=30.0)
            if c53 < best_c5:
                best_c5 = c53
                best_h = hb3.copy()
                hb = hb3.copy()
                stall_count = 0

    # Phase 3: Simulated annealing (remaining time minus buffer)
    if time.time() < t_end - 20:
        hb_sa, c5_sa = sa_search(hb, n, rng, t_end - 20)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = hb_sa.copy()
            hb = hb_sa.copy()

    # Phase 4: Final SLSQP polish
    if t_end - time.time() > 15:
        new_h = slsqp_optimize(repair_h(best_h + rng.normal(0, 0.05, n), n), n, maxiter=40)
        new_c5 = general_compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h

    best_h = repair_h(best_h, n)
    best_c5 = general_compute_c5(best_h)
    return best_h, best_c5, n
