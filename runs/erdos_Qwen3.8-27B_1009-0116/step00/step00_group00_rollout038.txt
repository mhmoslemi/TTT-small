    return np.max(c), c

    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(m * (c - c_max)))) / m

def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    h = np.clip(h, 0, 1)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-12:
        return h
    # Bisection on alpha: find alpha such that sum(clip(h - alpha * sign, 0, 1)) = target_sum
    # If diff > 0, need to decrease sum → subtract alpha
    # If diff < 0, need to increase sum → add alpha (i.e., subtract negative alpha)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if diff > 0:

        else:

        if diff > 0:
            if np.sum(h_test) >= target_sum:

            else:
                hi = mid
        else:
            if np.sum(h_test) <= target_sum:
                lo = mid
            else:
                hi = mid
        mid = (lo + hi) / 2

In terms of convolution: (h ⋆ q)[m] = sum_j q[j] * h[m+j] = (q_rev ⊛ h)[m] where q_rev = q[::-1] and ⊛ is convolution.

So A = convolve(p_shifted[::-1], h, 'full')[some_offset]

This is getting too complicated for the benefit. Let me just use the Python loop with early termination and limit the number of effective lags.

For temp=20, the weights decay as exp(-20*Δc). If the max c is 0.38 and the next is 0.37, the ratio is exp(-0.2)≈0.82. If the next is 0.35, ratio is exp(-0.6)≈0.55. So many lags might still be active.

But for the gradient, I can just take the top 10 lags (by p[k] value) and ignore the rest. This gives a good approximation of the gradient and is fast.

Let me restructure:

```python
def compute_grad_topk(h, dx, p, topk=10):
    n = len(h)
    grad = np.ones(n)  # the "1" in (1 - h[m+s] - h[m-s])
    # Find top-k lags
    idx = np.argsort(p)[::-1][:topk]
    for k in idx:
        w = p[k]
        if w < 1e-8:
            break
        s = k - (n - 1)
        if s == 0:
            grad -= 2.0 * w * h
        elif s > 0:
            grad[:n-s] -= w * h[s:]
            grad[s:] -= w * h[:n-s]
        else:
            sp = -s
            grad[sp:] -= w * h[:n-sp]
            grad[:n-sp] -= w * h[sp:]
    return grad * dx
```

With topk=10, this is 10 iterations of O(n) numpy ops. Very fast!

But wait, I need to normalize p so that sum of top-k weights is accounted for. Actually, for the smooth max gradient, the full gradient is sum_k p[k] * dc_k/dh. If I only use top-k, I get an approximation. The missing part is small (the tail of p). This should be fine for optimization.

Let me finalize the code with this approach. I'll use topk=15 for a good balance.

2. For n=90: SLSQP (still manageable)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_max(c, temp):
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(temp * (c - m)))) / temp)

def project_feasible(h, S):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        a = (lo + hi) / 2.0
        if np.sum(np.clip(h + a, 0.0, 1.0)) < S:
            lo = a
        else:
            hi = a
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.clip(np.interp(x_new, x_old, h), 0.0, 1.0)

def optimize_slsqp(h0, dx, max_iter=300, temp=40.0):
    n = len(h0)
    S = n / 2.0
    h0 = project_feasible(h0.copy(), S)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max(c, temp)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    h_opt = project_feasible(res.x, S)
    return h_opt, compute_c5(h_opt, dx)

def compute_grad_topk(h, dx, p, topk=15):
    n = len(h)
    grad = np.ones(n)
    idx = np.argsort(p)[::-1][:topk]
    for k in idx:
        w = p[k]
        if w < 1e-9:
            break
        s = k - (n - 1)
        if s == 0:
            grad -= 2.0 * w * h
        elif s > 0:
            grad[:n - s] -= w * h[s:]
            grad[s:] -= w * h[:n - s]
        else:
            sp = -s
            grad[sp:] -= w * h[:n - sp]
            grad[:n - sp] -= w * h[sp:]
    return grad * dx

def proj_grad(h0, dx, max_iters=5000, temp=25.0):
    n = len(h0)
    S = n / 2.0
    h = project_feasible(h0.copy(), S)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step = 0.5
    stale = 0
    for it in range(max_iters):
        c_all = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(c_all)
        exp_c = np.exp(temp * (c_all - c_max))
        p = exp_c / np.sum(exp_c)
        grad = compute_grad_topk(h, dx, p, topk=15)
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - step * grad, S)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_new
                improved = True
                break
            step *= 0.5
        if improved:
            stale = 0
            step = min(step * 1.3, 3.0)
        else:
            stale += 1
            if stale > 300:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_init = np.zeros(45)
        h_init[:22] = 1.0
        h_init[22] = 0.5
    best_h = h_init.copy()
    dx = 2.0 / len(best_h)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: SLSQP at initial resolution
    if time.time() - t0 < budget_s * 0.08:
        h_o, c5_o = optimize_slsqp(best_h, dx, max_iter=500, temp=50.0)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 2: Nelder-Mead for local refinement
    if time.time() - t0 < budget_s * 0.15:
        S = len(best_h) / 2.0
        def nm_obj(x):
            hp = project_feasible(x, S)
            c = np.correlate(hp, 1.0 - hp, mode='full') * dx
            return smooth_max(c, 30.0)
        res = minimize(nm_obj, best_h, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14})
        h_nm = project_feasible(res.x, S)
        c5_nm = compute_c5(h_nm, dx)
        if c5_nm < best_c5:
            best_h, best_c5 = h_nm, c5_nm

    # Phase 3: Upsample and optimize at higher resolution
    for target_n in [90, 180, 360]:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.72:
            break
        h_up = upsample(best_h, target_n)
        h_up = project_feasible(h_up, target_n / 2.0)
        dx_t = 2.0 / target_n
        if target_n <= 120:
            h_o, c5_o = optimize_slsqp(h_up, dx_t, max_iter=400, temp=50.0)
        elif elapsed < budget_s * 0.55:
            h_o, c5_o = proj_grad(h_up, dx_t, max_iters=8000, temp=30.0)
        else:
            h_o, c5_o = proj_grad(h_up, dx_t, max_iters=3000, temp=30.0)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 4: Random restarts at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        n_restarts = min(int(remaining / 25), 8)
        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.93:
                break
            scale = 0.03 + 0.02 * r
            h_r = best_h + rng.normal(0, scale, n_b)
            h_r = project_feasible(h_r, n_b / 2.0)
            if n_b <= 100:
                h_o, c5_o = optimize_slsqp(h_r, dx_b, max_iter=300, temp=50.0)
            else:
                h_o, c5_o = proj_grad(h_r, dx_b, max_iters=3000, temp=30.0)
            if c5_o < best_c5:
                best_h, best_c5 = h_o, c5_o

    # Final polish with tight SLSQP if time allows
    if time.time() - t0 < budget_s * 0.96 and len(best_h) <= 200:
        h_o, c5_o = optimize_slsqp(best_h, 2.0 / len(best_h), max_iter=200, temp=60.0)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
