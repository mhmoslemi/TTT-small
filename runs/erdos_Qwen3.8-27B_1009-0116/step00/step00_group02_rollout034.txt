def project_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        # Fix sum first
        diff = target - np.sum(h)
        h = h + diff / n
        # Then clip
        h = np.clip(h, 0, 1)
        if abs(np.sum(h) - target) < 1e-12:
            break
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among non-bound entries
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)

    target = n / 2.0
    def sum_clipped(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    # Bisection
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_clipped(mid) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(y - lam, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

_NG = 45

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def compute_c5_grad(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(c))
    d = k - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] -= h[d:]
    else:
        g[:n + d] += (1.0 - h[-d:])
        g[-d:] -= h[:n + d]
    return g * dx

def obj_pen(h, n):
    c5 = compute_c5_raw(h)
    pen = 1e5 * (np.sum(h) - n / 2.0) ** 2
    return c5 + pen

def grad_pen(h, n):
    g_c5 = compute_c5_grad(h, n)
    g_p = 2.0e5 * (np.sum(h) - n / 2.0) * np.ones(n)
    return g_c5 + g_p

def slsqp_optimize(h0, n, max_iter=200, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h0, jac=grad_pen, args=(n,),
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': ftol})
        cand = project_feasible(res.x, n)
        return cand
    except Exception:
        return h0

def projected_grad_loop(h0, n, n_steps=500):
    h = h0.copy()
    best = h.copy()
    best_val = compute_c5_raw(h)
    step = 0.5
    for i in range(n_steps):
        g = compute_c5_grad(h, n)
        h_new = project_feasible(h - step * g, n)
        v = compute_c5_raw(h_new)
        if v < best_val:
            best_val = v
            best = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.7
        if step < 1e-7:
            break
    return best, best_val

def make_binary_start(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    else:
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
        h[rng.choice(n)] += 0.5
    return h

def make_smooth_start(n, rng):
    h = 0.5 * np.ones(n)
    for freq in range(1, 6):
        amp = 0.15 / freq
        phase = rng.rand() * 2.0 * np.pi
        h += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return h

def make_block_start(n, rng):
    h = np.zeros(n)
    bs = int(rng.choice([3, 4, 5, 7]))
    i = 0
    on = rng.rand() < 0.5
    while i < n:
        if on:
            h[i:min(i + bs, n)] = 1.0
        i += bs
        on = not on
    return h
