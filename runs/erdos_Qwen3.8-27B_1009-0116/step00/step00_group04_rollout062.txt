def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def fix_integral(h, target, tol=1e-12):
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < tol:
            return h
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.sum(mask) == 0:
            # Try shifting all
            h += diff / len(h)
            h = np.clip(h, 0, 1)
        else:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def interpolate_h(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return fix_integral(h_new, n_new / 2.0)

def slsqp_optimize(h0, maxiter=3000):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,

        return compute_c5(h)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
