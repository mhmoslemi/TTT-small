    cool = 0.99993
    stall = 0
    stall_max = 12000

    while time.time() < deadline - 30:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        cur[i] = 0.0
        cur[j] = 1.0
        nc = compute_c5(cur)
        if nc < cur_c5:
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = cur.copy()
                stall = 0
            else:
                stall += 1
        else:
            d = nc - cur_c5
            if d < 50.0 * T and np.log(np.random.random()) < -d / T:
                cur_c5 = nc
                stall += 1
            else:
                cur[i] = 1.0
                cur[j] = 0.0
                stall += 1
        T = max(T * cool, T_min)
        if stall > stall_max:
            T = 0.015
            stall = 0
            cur = best_h.copy()
            cur_c5 = best_c5

    # Greedy polish
    if time.time() < deadline - 10:
        hp, cp = greedy_polish(best_h, deadline)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Continuous refinement via SA
    if time.time() < deadline - 5:
        h_c = best_h.copy()
        cc5 = best_c5
        Tc = 0.004
        while time.time() < deadline:
            a, b = np.random.choice(n, 2, replace=False)
            d = np.random.uniform(-0.12, 0.12)
            h_c[a] = min(1.0, max(0.0, h_c[a] + d))
            h_c[b] = min(1.0, max(0.0, h_c[b] - d))
            h_c = project_sum(h_c, target)
            nc = compute_c5(h_c)
            if nc < cc5:
                cc5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h_c.copy()
            elif d * 1000 < Tc and np.log(np.random.random()) < -(nc - cc5) / Tc:
                cc5 = nc
            Tc = max(Tc * 0.99997, 0.0003)

    # Final binary polish on best result
    if time.time() < deadline - 1:
        best_h_b = to_binary(best_h, n)
        cb = compute_c5(best_h_b)
        if cb < best_c5:
            best_c5 = cb
            best_h = best_h_b

    return (best_h, best_c5, n)
