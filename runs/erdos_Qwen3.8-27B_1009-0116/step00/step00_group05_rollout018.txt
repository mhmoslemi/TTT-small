def smooth_max_gradient(h, n, alpha):

    max_c = np.max(corr)
    exps = np.exp(alpha * (corr - max_c))
    total = np.sum(exps)

    weights = exps / total
    grad = np.zeros(n)

import numpy as np

def project_feasible(x, n):
    """Project x onto {h: 0<=h<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def c5_value(h, n):
    """Compute C5 for h with n points."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_grad(h, n, alpha):
    """Gradient of smooth max of correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(alpha * (corr - mc))
    sw = np.sum(e)
    if sw < 1e-300:
        return np.zeros(n)
    w = e / sw
    grad = np.zeros(n)
    omh = 1.0 - h
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-9:
            continue
        s = idx - (n - 1)
        # First term: sum over valid j of w * (1-h[j+s])
        j0, j1 = max(0, -s), min(n - 1, n - 1 - s)
        if j0 <= j1:
            grad[j0:j1+1] += wi * omh[j0+s:j1+s+1]
        # Second term: sum over valid j of w * h[j-s]
        j0, j1 = max(0, s), min(n - 1, n - 1 + s)
        if j0 <= j1:
            grad[j0:j1+1] -= wi * h[j0-s:j1-s+1]
    return grad

def local_refine(h, n, iters=200):
    """Local search: try small adjustments to individual coordinates."""
    best = h.copy()
    best_c5 = c5_value(best, n)
    target = n / 2.0
    for _ in range(iters):
        moved = False
        j = np.random.randint(n)
        for delta in [0.5, 0.1, 0.05, -0.5, -0.1, -0.05, 0.2, -0.2]:
            trial = best.copy()
            trial[j] = np.clip(best[j] + delta, 0.0, 1.0)
            # Adjust to maintain sum
            diff = best[j] - trial[j]
            if abs(diff) < 1e-10:
                continue
            # Spread adjustment among free coordinates
            if diff > 0:
                mask = (best < 1.0 - 1e-8) & (np.arange(n) != j)
            else:
                mask = (best > 1e-8) & (np.arange(n) != j)
            if not np.any(mask):
                continue
            nm = np.sum(mask)
            adj = -diff / nm
            trial[mask] = np.clip(best[mask] + adj, 0.0, 1.0)
            trial = project_feasible(trial, n)
            c5_trial = c5_value(trial, n)
            if c5_trial < best_c5 - 1e-12:
                best = trial
                best_c5 = c5_trial
                moved = True
                break
        # If no move, try a pair swap
        if not moved:
            i = np.random.randint(n)
            k = np.random.randint(n)
            if i == k:

            trial = best.copy()

                if c5_trial < best_c5 - 1e-12:

                    moved = True

import numpy as np
import time

def project_feasible(x, n):
    """Project x onto {h: 0<=h<=1, sum(h)=n/2} via bisection on Lagrange multiplier."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def c5_value(h, n):
    """Compute C5 = max_k corr(k) * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_grad(h, n, alpha):
    """Gradient of smooth max (log-sum-exp) of all correlation values."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(alpha * (corr - mc))
    sw = np.sum(e)
    if sw < 1e-300:
        return np.zeros(n)
    w = e / sw
    grad = np.zeros(n)
    omh = 1.0 - h
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-9:
            continue
        s = idx - (n - 1)
        # (1-h[j-s]) term: j-s in [0,n-1]
        j0 = max(0, s)
        j1 = min(n - 1, n - 1 + s)
        if j0 <= j1:
            grad[j0:j1 + 1] += wi * omh[j0 - s:j1 - s + 1]
        # h[j+s] term: j+s in [0,n-1]
        j0 = max(0, -s)
        j1 = min(n - 1, n - 1 - s)
        if j0 <= j1:
            grad[j0:j1 + 1] -= wi * h[j0 + s:j1 + s + 1]
    return grad

def local_refine(h, n, rng, iters=500):
    """Local search: random coordinate adjustments maintaining feasibility."""
    best = h.copy()
    best_c5 = c5_value(best, n)
    for _ in range(iters):
        j = rng.randint(n)
        deltas = [0.5, -0.5, 0.2, -0.2, 0.1, -0.1, 0.05, -0.05]
        rng.shuffle(deltas)
        for delta in deltas:
            trial = best.copy()
            new_val = best[j] + delta
            if new_val < 0 or new_val > 1:
                continue
            trial[j] = new_val
            diff = best[j] - new_val
            if abs(diff) < 1e-12:
                continue
            if diff > 0:
                mask = (best < 1.0 - 1e-8) & (np.arange(n) != j)
            else:
                mask = (best > 1e-8) & (np.arange(n) != j)
            if not np.any(mask):
                continue
            trial[mask] = best[mask] - diff / np.sum(mask)
            trial = project_feasible(trial, n)
            c5_t = c5_value(trial, n)
            if c5_t < best_c5 - 1e-12:
                best = trial
                best_c5 = c5_t
                break
        # Pair swap
        i = rng.randint(n)
        k = rng.randint(n)
        if i != k:
            trial = best.copy()
            d = 0.3 * min(best[i], 1.0 - best[k])
            if d > 1e-8 and best[i] > 0:
                trial[i] = best[i] - d
                trial[k] = best[k] + d
                trial = project_feasible(trial, n)
                c5_t = c5_value(trial, n)
                if c5_t < best_c5 - 1e-12:
                    best = trial
                    best_c5 = c5_t
    return best, best_c5

def grad_descent(h, n, avg_time, alpha, max_iter=300, lr=0.003):
    """Projected gradient descent with smooth max."""
    h = h.copy()
    best_local = h.copy()
    best_c5_local = c5_value(h, n)
    for it in range(max_iter):
        if time.time() > avg_time:
            break
        grad = smooth_grad(h, n, alpha)
        grad -= np.mean(grad)
        h = project_feasible(h - lr * grad, n)
        c5_cur = c5_value(h, n)
        if c5_cur < best_c5_local:
            best_local = h.copy()
            best_c5_local = c5_cur
        if it > 0 and it % 100 == 0:
            lr *= 0.95
    return best_local, best_c5_local
