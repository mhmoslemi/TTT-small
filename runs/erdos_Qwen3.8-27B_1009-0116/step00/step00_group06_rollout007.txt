    d = target - np.sum(h)

            bnds = [(0.0, 1.0)] * n
            try:
                r = minimize(smooth_obj, hs, args=(n, beta), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0})
                ht = project(r.x, target)
                ct = compute_c5(ht)
                if ct < best_c5:
                    best_h, best_c5 = ht, ct
            except Exception:
                pass

    # Pairwise local search: main optimization phase
    while time.time() - t0 < budget_s - 80:
        batch = 30000 if n <= 120 else 10000
        best_h, best_c5 = pairwise_refine(best_h, target, rng, batch)

    # Nelder-Mead final polish
    if time.time() - t0 < budget_s - 15:
        try:
            r = minimize(neld_obj, best_h, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-12})
            ht = project(r.x, target)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_h, best_c5 = ht, ct
        except Exception:
            pass

    # Final pairwise pass
    if time.time() - t0 < budget_s - 2:
        best_h, best_c5 = pairwise_refine(best_h, target, rng, 10000)

    return best_h, best_c5, n
