        h_init = np.array(initial_h_values, dtype=float)

    # Phase 1: Optimize initial at its native resolution
    if h_init is not None:
        n0 = len(h_init)
        h_opt, val_opt = opt_slsqp(h_init, n0, min(120, budget_s * 0.25))
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        h_sg, val_sg = opt_subgrad(h_opt, n0, min(60, budget_s * 0.1))
        if val_sg < best_val:
            best_val = val_sg
            best_h = h_sg.copy()

    # Phase 2: Multi-resolution search
    n_list = [42, 48, 56, 64, 72, 84, 96, 112, 128, 144, 168, 200]
    if h_init is not None:
        n0 = len(h_init)
        if n0 not in n_list:
            n_list.insert(0, n0)

    for n in n_list:
        if time.time() - t_global > budget_s * 0.85:
            break
        # Build starting points
        starts = []
        if best_h is not None:
            starts.append(resample_h(best_h, n))
        if h_init is not None:
            starts.append(resample_h(h_init, n))
        starts.append(make_random_start(n, rng))
        starts.append(make_block_start(n, rng))

        for sp in starts:
            if time.time() - t_global > budget_s * 0.85:
                break
            t_rem = budget_s * 0.85 - (time.time() - t_global)
            if t_rem < 5:
                break
            alloc = min(60, t_rem * 0.4)
            h_opt, val_opt = opt_slsqp(sp, n, alloc)
            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
            # Subgradient polish
            if time.time() - t_global < budget_s * 0.88:
                h_sg, val_sg = opt_subgrad(h_opt, n, min(30, budget_s * 0.03))
                if val_sg < best_val:
                    best_val = val_sg
                    best_h = h_sg.copy()

    # Phase 3: Final refinement with tight budget
    if best_h is not None and time.time() - t_global < budget_s * 0.95:
        n_best = len(best_h)
        t_rem = budget_s * 0.95 - (time.time() - t_global)
        h_fin, val_fin = opt_slsqp(best_h, n_best, max(10, min(120, t_rem)))
        if val_fin < best_val:
            best_val = val_fin
            best_h = h_fin.copy()

    # Fallback
    if best_h is None:
        best_h = np.ones(42) * 0.5
        best_val = compute_c5(best_h, 2.0 / 42)

    n_final = len(best_h)
    dx_final = 2.0 / n_final
    final_val = compute_c5(best_h, dx_final)
    return (best_h.tolist(), final_val, n_final)
