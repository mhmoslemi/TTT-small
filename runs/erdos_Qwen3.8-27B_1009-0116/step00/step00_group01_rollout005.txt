import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_max_w(corr, beta):
    m = corr.max()
    s = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(s)
    return e / e.sum()

def grad_c5_smooth(w, h, dx):
    n = len(h)
    rp = np.zeros(n)
    rm = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        mk = n - 1 - k
        if mk >= 0:
            rp[:n - mk] += wk * h[mk:]
            rm[mk:] += wk * h[:n - mk]
        else:
            mm = -mk
            rp[mm:] += wk * h[:n - mm]
            rm[:n - mm] += wk * h[mm:]
    return dx * (1.0 - rp - rm)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] += d / c
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_res(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    xo = np.arange(n, dtype=float)
    xn = np.linspace(0, n - 1, n_new)
    return np.interp(xn, xo, h)

    dx = 2.0 / n
    target = n / 2.0
    h = project(h, target)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta = 30.0
    lr = 0.5
    mom = 0.8
    vel = np.zeros(n)
    t_start = time.time()
    no_improve = 0

                    no_improve = 0

import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_max_w(corr, beta):
    m = corr.max()
    s = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(s)
    return e / e.sum()

def grad_c5_smooth(w, h, dx):
    n = len(h)
    rp = np.zeros(n)
    rm = np.zeros(n)
    active = w > 1e-13
    for k in np.where(active)[0]:
        wk = w[k]
        mk = n - 1 - k
        if mk >= 0:
            rp[:n - mk] += wk * h[mk:]
            rm[mk:] += wk * h[:n - mk]
        else:
            mm = -mk
            rp[mm:] += wk * h[:n - mm]
            rm[:n - mm] += wk * h[mm:]
    return dx * (1.0 - rp - rm)

def project(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] += d / c
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_res(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    xo = np.arange(n, dtype=float)
    xn = np.linspace(0, n - 1, n_new)
    return np.interp(xn, xo, h)

def opt_loop(h, time_limit, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 25.0
    lr = 0.6
    mom = 0.85
    vel = np.zeros(n)
    t_start = time.time()
    no_imp = 0
    while time.time() - t_start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        w = smooth_max_w(corr, beta)
        g = grad_c5_smooth(w, h, dx)
        vel = mom * vel - lr * g
        h_new = project(h + vel, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_imp = 0
            beta = min(beta * 1.03, 150.0)
        else:
            no_imp += 1
            if no_imp > 80:
                h_new = project(h + 0.15 * rng.standard_normal(n), target)
                vel[:] = 0.0
                lr = 0.6
                beta = max(10.0, beta * 0.5)
                no_imp = 0
            elif no_imp > 30:
                lr *= 0.85
                if lr < 0.02:
                    lr = 0.4
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float)

        h0 = np.clip(h0, 0, 1)
        h0 = project(h0, n_s / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    res_list = []
    for r in [len(h0), 94, 188, 376, 752]:
        if r > 0 and (not res_list or r > res_list[-1]):
            res_list.append(r)
    n_stages = len(res_list)
    t_avail = budget_s * 0.82
    wts = np.array([1.0, 1.4, 2.0, 3.0, 4.0])[:n_stages]
    wts /= wts.sum()
    times = wts * t_avail
    for i, nr in enumerate(res_list):
        elapsed = time.time() - t0
        remain = budget_s * 0.88 - elapsed
        if remain < 20:
            break
        t_alloc = min(times[i], remain - 10)
        h = interp_res(best_h, nr)
        h = project(h, nr / 2.0)
        h_opt, c5_opt = opt_loop(h, t_alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    # Final polish: short local search on best
    elapsed = time.time() - t0
    remaining = budget_s * 0.95 - elapsed
    if remaining > 30:
        nb = len(best_h)
        ntrials = max(2, int(remaining / 25))
        tper = remaining / (ntrials + 1)
        for _ in range(ntrials):
            if time.time() - t0 > budget_s * 0.97:

                best_c5 = c5_try
                best_h = h_try.copy()
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
