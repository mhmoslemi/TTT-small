from scipy.special import logsumexp
import time

_g_alpha = 40.0

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    return dx * logsumexp(_g_alpha * c) / _g_alpha

def _ceq(x):
    return np.sum(x) - len(x) * 0.5

def _cjac(x):
    return np.ones(len(x))

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp_up(h, n1, n2):
    x1 = (np.arange(n1) + 0.5) * 2.0 / n1
    x2 = (np.arange(n2) + 0.5) * 2.0 / n2
    return np.interp(x2, x1, h)

def _slsqp(h0, n, alpha, iters, tl):
    global _g_alpha
    _g_alpha = alpha
    t_s = time.time()
    res = minimize(_obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _ceq, 'jac': _cjac}],
                   options={'maxiter': iters, 'ftol': 1e-14})
    el = time.time() - t_s
    if el > tl or not np.isfinite(res.fun):
        return h0, _c5(h0)
    h1 = _repair(res.x, n)
    v1 = _c5(h1)
    return h1, v1

def _local_search(h0, n, tl, s0=0):
    rng = np.random.RandomState(s0)
    h = h0.copy()
    c5 = _c5(h)
    t_s = time.time()
    ss = 0.15
    while time.time() - t_s < tl and ss > 0.002:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = rng.uniform(-ss, ss)
        h2 = h.copy()
        h2[i] = min(1.0, max(0.0, h[i] + d))
        h2[j] = min(1.0, max(0.0, h[j] - d))

        v2 = _c5(h2)
        if v2 < c5:
            h, c5 = h2, v2
        ss *= 0.9985
