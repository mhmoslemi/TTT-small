        x_f = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_f, x_c, best_h)
        h_try = project_feasible(h_try, n_try)

        # Also try complement as starting point
        h_comp = project_feasible(1.0 - best_h, n_try)
        x_f2 = x_f
        h_try2 = np.interp(x_f2, x_c, 1.0 - best_h)
        h_try2 = project_feasible(h_try2, n_try)

        for h_init in [h_try, h_try2]:
            if time.time() > t_end - 30:
                break
            h_opt = slsqp_optimize(h_init, n_try, maxiter=150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, n_best = c_opt, h_opt, n_try

    # Phase 6: Final coordinate descent polish
    h_cd, c_cd = coordinate_descent(best_h, n_best, 20, t_end)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, h_cd

    # Final SLSQP
    h_final = slsqp_optimize(best_h, n_best, maxiter=300)
    c_final = compute_c5(h_final)
    if c_final < best_c5:
        best_c5, best_h = c_final, h_final

    return best_h, best_c5, len(best_h)
