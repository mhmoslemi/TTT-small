import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        if s < target:
            # Need to add (target - s) total, spread over non-saturated elements
            # Simple approach: add uniformly, then clip, repeat
            h += (target - s) / n
        else:
            h -= (s - target) / n
    return h

def smooth_obj_and_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = max_c + np.log(sum_exp) / beta
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        j_min = max(0, -offset)
        j_max = min(n - 1, n - 1 - offset)
        for j in range(j_min, j_max + 1):
            grad[j] += w * (1.0 - h[j + offset])
            grad[j + offset] += w * (-h[j])

    target = n / 2.0
    s = np.sum(h)
    obj += lam * (s - target) ** 2
    grad += 2.0 * lam * (s - target) * np.ones(n)
    
    return obj, grad

    best_h = h0.copy()

        nonlocal best_h, best_obj

        o, g = smooth_obj_and_grad(x, beta, lam)
