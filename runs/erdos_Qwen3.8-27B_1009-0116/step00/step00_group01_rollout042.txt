import numpy as np
from scipy.optimize import minimize
import time

_G_N = 47
_G_TARGET = 23.5
_G_ALPHA = 20.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def soft_max(arr, alpha):
    m = arr.max()
    shifted = (arr - m) * alpha
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def repair(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_soft(x):
    n = _G_N
    dx = 2.0 / n
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full") * dx
    return soft_max(corr, _G_ALPHA)

def constraint_sum(x):
    return x.sum() - _G_TARGET

def constraint_sum_jac(x):
    return np.ones(_G_N)

def one_slsqp(x0, bounds, cons, maxiter=300):
    res = minimize(obj_soft, x0.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return repair(res.x, _G_TARGET)
    return x0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    global _G_N, _G_TARGET, _G_ALPHA
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    h0 = repair(h0, _G_TARGET)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n

        if time.time() - t_start > budget_s - 20:

        _G_ALPHA = alpha_val
        cand = one_slsqp(best_h, bounds, cons, 400)
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_h = cand
            best_c5 = c5

    # Phase 2: Many random restarts
