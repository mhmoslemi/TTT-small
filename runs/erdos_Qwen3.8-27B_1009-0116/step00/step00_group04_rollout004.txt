import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    """Project to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):  # Iterate to convergence
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Add to positions not at 1
            free = h < 1.0 - 1e-12
            if free.sum() == 0:
                break
            adj = diff / free.sum()
            h[free] = np.clip(h[free] + adj, 0, 1)
        else:
            free = h > 1e-12
            if free.sum() == 0:
                break
            adj = diff / free.sum()
            h[free] = np.clip(h[free] + adj, 0, 1)
    # Final fix
    h = np.clip(h, 0, 1)
    return h

def to_binary(h, n):
    """Round to binary with exactly n/2 ones"""
    target = n // 2
    h_sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:target]] = 1.0
    return h_bin

def project_simplex(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_simplex(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            step = diff / mask.sum()
            h[mask] = np.clip(h[mask] + step, 0, 1)
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            step = (-diff) / mask.sum()
