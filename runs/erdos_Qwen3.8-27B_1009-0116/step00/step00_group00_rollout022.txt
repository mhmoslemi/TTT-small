import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    lo, hi = -1.0, 2.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_smooth_max(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = corr * temp
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    p = exps / np.sum(exps)
    offsets = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx] * dx
        if w < 1e-14:
            continue
        k = offsets[idx]
        if k >= 0:
            rem = n - k
            if rem > 0:
                grad[:rem] += w * (1.0 - h[k:])
                grad[k:] += w * (-h[:rem])
        else:
            ka = -k
            sh = n - ka
            if sh > 0:
                grad[ka:] += w * (1.0 - h[:sh])
                grad[:sh] += w * (-h[ka:])
    return grad

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def optimize_res(h, max_time):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    scale = 200.0 / n
    phases = [(10, 1.0*scale), (20, 0.8*scale), (40, 0.5*scale),
              (80, 0.3*scale), (150, 0.2*scale), (300, 0.1*scale)]
    for temp, lr in phases:
        if time.time() - t0 > max_time:
            break
        h_cur = best.copy()
        stale = 0
        iters = 400
        for it in range(iters):
            if time.time() - t0 > max_time:
                break
            g = grad_smooth_max(h_cur, temp)
            step = lr * g
            max_step = np.max(np.abs(step))
            if max_step > 0.3:
                step *= 0.3 / max_step
            h_new = h_cur - step
            h_cur = project_feasible(h_new, target)
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best = h_cur.copy()
                stale = 0
            else:
                stale += 1
                if stale > 150:
                    break
    return best, best_c5
