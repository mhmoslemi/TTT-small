import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_max(corr, tau):
    m = np.max(corr)
    if tau < 1e-6:
        return m
    shifted = tau * (corr - m)
    # For numerical stability
    log_sum = m + np.log(np.sum(np.exp(shifted))) / tau
    return log_sum

def project_to_feasible(h, target):
    """Project h to {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)

        return h
