        p = np.ones_like(f) / len(f)
    else:
        p /= p_sum
    
    grad = np.zeros(n)

        pk = p[k_idx]
        if pk < 1e-18:
            continue
        s = k_idx - (n - 1)

        lo1 = max(0, s)
        hi1 = min(n - 1, s + n - 1)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += dx * pk * h_complement[lo1 - s : hi1 - s + 1]
        
        # Term 2: -h[j+s] for j+s in [0, n-1]
        lo2 = max(0, -s)
        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= dx * pk * h[lo2 + s : hi2 + s + 1]
    
    return grad, f * dx

            break

            T = 50.0 + 200.0 * (it / max_iter)

        g_norm = np.sqrt(np.sum(grad**2))
        if g_norm < 1e-15:
            break

        h_new = project_onto_constraint(h_new, target_sum)

            best_h = h_new.copy()
            # Increase step slightly if improving
            step *= 1.05
        else:
            step *= 0.8
        
        h = h_new
    
    return best_h, best_c5

def coordinate_descent(h, dx, target_sum, time_limit=120, max_rounds=100):
    """Local search: try setting each coordinate to 0 or 1"""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break

                break

        if not improved:

def project_onto_constraint(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def compute_gradient(h, dx, T):

    f = np.correlate(h, 1.0 - h, mode="full")

    if p_sum < 1e-300:

            h_opt, c5_opt = coordinate_descent(h_opt, dx_t, target_t, time_limit=TL2)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 6: Final polish
    if rem() > 30:
        n_fin = len(best_h)
        dx_fin = 2.0 / n_fin
        target_fin = n_fin / 2.0
        TL = min(60, rem() * 0.5)
        h_opt, c5_opt = coordinate_descent(best_h, dx_fin, target_fin, time_limit=TL)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_bin, c5_bin = try_binary_01(best_h, dx_fin, target_fin, n_fin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Ensure final constraint
    n_fin = len(best_h)
    dx_fin = 2.0 / n_fin
    best_h = project_onto_constraint(best_h, n_fin / 2.0)
    best_c5 = compute_c5(best_h, dx_fin)

    return (best_h, best_c5, n_fin)
