        starts.append(make_binary_start(n, seed + 100 * s + 1))

    ramp = np.sort(np.linspace(0, 1, n))[::-1]
    ramp = project_to_feasible(ramp, target)
    starts.append(ramp)
    # Block structure
    blk = np.zeros(n)
    blk[:int(n * 0.4)] = 1.0
    blk[int(n * 0.4) + 1:int(n * 0.4) + 2] = 0.5
    blk = project_to_feasible(blk, target)
    starts.append(blk)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}]

    temps = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    for h_start in starts:
        if time.time() > t_end - 30:
            break
        for temp in temps:
            if time.time() > t_end - 30:
                break
            for restart in range(2):
                if time.time() > t_end - 30:
                    break
                x0 = h_start + rng.randn(n) * 0.005
                x0 = project_to_feasible(x0, target)
                try:
                    res = minimize(
                        lambda x, tp=temp: obj_smooth(x, tp),
                        x0, method='SLSQP', bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 2000, 'ftol': 1e-14}
                    )
                    h_new = project_to_feasible(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_start = h_new.copy()
                except Exception:
                    pass

    # Coordinate descent polish
    if time.time() < t_end - 15:
        best_h, best_c5 = coordinate_descent(best_h, target, max_iter=800)

    # Try other n values using interpolation
    for n2 in [n + 1, n + 2, n - 1, n + 5, n - 5]:
        if time.time() > t_end - 20:
            break
        if n2 < 10:
            continue
        target2 = n2 / 2.0
        xs_old = np.linspace(1, n, n)
        xs_new = np.linspace(1, n2, n2)
        h_interp = np.interp(xs_new, xs_old, best_h)
        h_interp = project_to_feasible(h_interp, target2)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': lambda x, t=target2: np.sum(x) - t}]
        for temp in [0.5, 0.2, 0.05]:
            if time.time() > t_end - 20:
                break
            try:
                res = minimize(
                    lambda x, tp=temp: obj_smooth(x, tp),
                    h_interp, method='SLSQP', bounds=bounds2,
                    constraints=cons2,
                    options={'maxiter': 1500, 'ftol': 1e-14}
                )
                h_new = project_to_feasible(res.x, target2)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_interp = h_new.copy()
            except Exception:
                pass
        if time.time() < t_end - 15:
            best_h, c5_cd = coordinate_descent(best_h, target2, max_iter=300)
            if c5_cd < best_c5:
                best_c5 = c5_cd

    return (best_h, best_c5, len(best_h))
