import numpy as np
from scipy.optimize import minimize
import time

_g_n = 95

def _obj_smooth(x, temp):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if temp < 1e-6:
        return m
    return m + temp * np.log(np.mean(np.exp((corr - m) / temp)))

def _eq_constraint(x):
    return np.sum(x) - 0.5 * _g_n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask_u = h < 1.0
        mask_d = h > 0.0
        cnt = np.sum(mask_u) + np.sum(mask_d)
        if cnt == 0:
            break
        step = diff / cnt
        if step > 0:
            h[mask_u] = np.minimum(h[mask_u] + step, 1.0)
        else:
            h[mask_d] = np.maximum(h[mask_d] + step, 0.0)
    return h

    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.arange(n_dst, dtype=float) * (n_src / n_dst)
    h_dst = np.interp(xd, xs, h_src)
    return project_feasible(h_dst, n_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 95

def _obj_smooth(x, temp, n, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if temp < 1e-7:
        f = m
    else:
        shifted = (corr - m) / temp
        shifted = np.clip(shifted, -50, 50)
        f = m + temp * np.log(np.mean(np.exp(shifted)))
    cv = np.sum(x) - 0.5 * n
    return f + penalty * cv * cv

    dx = 2.0 / n
