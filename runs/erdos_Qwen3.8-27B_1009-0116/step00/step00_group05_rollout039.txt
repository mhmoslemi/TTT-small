def to_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h = np.sort(h)[::-1]  # descending
    if n % 2 == 0:
        binary = (h >= h[n//2]).astype(float)
    # Adjust count
    ones = np.sum(binary)
    idx_sorted = np.argsort(-h)  # indices in descending order of h
    binary = np.zeros(n)
    binary[idx_sorted[:n//2]] = 1.0
    return binary

def golden_ratio_pattern(n, rng):
    """Generate binary pattern based on golden ratio."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    idx = rng.permutation(n)
    frac = (np.arange(n) * (1.0/phi)) % 1.0
    # Assign top n/2 to be 1
    frac[idx[:]] = frac[idx[:]]  # shuffle
    sorted_idx = np.argsort(-frac)
    h = np.zeros(n)
    h[sorted_idx[:n//2]] = 1.0
    return h

def anneal_binary(h_init, n, rng, n_iter=50000):
    """Simulated annealing on binary pattern with fixed Hamming weight."""
    h = to_binary(h_init, n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best = h.copy()
    
    T = 0.05
    T_min = 1e-7
    cool = (T_min / T) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        T *= cool
        # Pick random one and zero to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best, best_c5

def steepest_descent_binary(h_init, n, max_rounds=100):
    """Try all possible single swaps, take best improvement."""
    h = to_binary(h_init, n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    cur_c5 = compute_c5(h)
    best = h.copy()
    best_c5 = cur_c5
    
    for rnd in range(max_rounds):
        best_delta = 0
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                delta = c5 - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        cur_c5 += best_delta
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        best = h.copy()
        best_c5 = cur_c5
    
    return best, best_c5

        T *= lr
        i = on[r.randint(len(on))]
        j = zf[r.randint(len(zf))]
        h[i], h[j] = 0.0, 1.0
        nv = compute_c5(h)
        d = nv - cv
        if d < 0 or r.rand() < np.exp(-d / T):
            cv = nv
            on = np.where(h == 1)[0]
            zf = np.where(h == 0)[0]
            if cv < bc:
                bc, bh = cv, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, bc

def _ns(h, n, rd=300):
    h = h.astype(float).copy()
    cv = compute_c5(h)
    for _ in range(rd):
        ok = False
        for i in range(n):
            if h[i] != 1.0:
                continue
            for j in (i - 1, i + 1):
                if 0 <= j < n and h[j] == 0.0:
                    h[i], h[j] = 0.0, 1.0
                    nv = compute_c5(h)
                    if nv < cv - 1e-14:
                        cv = nv
                        ok = True
                    else:
                        h[i], h[j] = 1.0, 0.0
        if not ok:
            break
    return h, cv
