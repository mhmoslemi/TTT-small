def compute_scaled_corr(h):

def is_feasible(h):
    h_np = np.array(h)
    if len(h_np) == 0:
        return False
    s = np.sum(h_np)
    if abs(s - len(h_np)/2) > 1e-5:

    if any(x < 0 or x > 1 for x in h_np):

    return True

def clamp_vector(v):
