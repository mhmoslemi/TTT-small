            h[i] += a * (1 if d > 0 else -1)
    return h

def _agrad(h):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    tol = max(1e-9, 3.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    # FFT-based gradient
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gw(h, wP, wN)

def _gw(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _fista(h0, a, tl):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    fy = _sval(h, a)
    fh = fy
    bh = h.copy()
    bc = _c5(h)
    stale = 0
    Lc = 8.0
    for it in range(300000):
        if time.time() - t0 > tl:
            break
        if it:

            y = h + ((tv - 1.0) / tn) * (h - hp)

        else:

            fy = fh

            fx = _sval(xn, a)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                ok = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp = h
            h = xn
            fh = fx
            stale = 0
            Lc = max(Lc * 0.8, 1.0)
            if it % 8 == 0:

                if c < bc:
                    bc = c
                    bh = h.copy()
        else:
            stale += 1
            hp, y, tv = h, h, 1.0
            fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 20:
                break
    c = _c5(h)
    if c < bc:

def _sa_local(h0, rng, dl, s0=5e-5, T0=0.05, Tmin=1e-8):
    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    g = _agrad(h)
    order = np.argsort(g)
    q = max(10, n // 5)
    lo = order[:q]
    hi = order[-q:]
    recomp = 300
    it = 0
    while time.time() < dl:
        if it >= recomp:
            c = _c5(h)
            if c < bc:
                bc = c
                bh = h.copy()
            g = _agrad(h)
            order = np.argsort(g)
            lo = order[:q]
            hi = order[-q:]
            recomp = it + 300
        fr = min(1.0, (time.time() - t0) / max(1e-9, dl - t0))
        T = T0 * (Tmin / T0) ** fr
        base = max(1e-12, s0 * (1e-6) ** fr)
        r = rng.random()
        if r < 0.6:
            # Gradient-guided pair swap
            i = int(lo[rng.randint(len(lo))])
            j = int(hi[rng.randint(len(hi))])
        elif r < 0.8:
            # Random pair
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        elif r < 0.92:
            # Block swap (2-4 points)
            blen = int(rng.randint(2, 5))
            i = int(rng.randint(0, max(1, n - blen - 1)))
            j = i + blen + int(rng.randint(0, max(1, n - 2 * blen - 1)))
            if j + blen > n:
                j = n - blen
            old_i = h[i:i+blen].copy()
            old_j = h[j:j+blen].copy()
            h[i:i+blen] = old_j
            h[j:j+blen] = old_i
            cn = _c5(h)
            delta = cn - c
            if delta < -1e-14 or (T > 1e-12 and rng.random() < np.exp(-max(0.0, delta) / T)):
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i:i+blen] = old_i
                h[j:j+blen] = old_j
            it += 1
            continue
        else:
            # Large random perturbation
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        
        if i == j:
            it += 1
            continue
        sgn = 1 if g[i] <= g[j] else -1
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.3 + 1.7 * rng.random()
            if rng.random() < 0.03:
                mult *= 10.0
            d_val = sgn * min(avail, base * mult)
            h[i] += d_val
            h[j] -= d_val
            cn = _c5(h)
            delta = cn - c
            if delta < -1e-14 or (T > 1e-12 and rng.random() < np.exp(-max(0.0, delta) / T)):
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i] -= d_val
                h[j] += d_val
        it += 1
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

Actually, for the coordinate descent: for each coordinate i, we can compute the optimal value of h[i] (given all other coordinates fixed) to minimize C₅. This is a 1D optimization that can be done analytically.

Given h with h[i] = v, the overlap at lag k is:
- For k=0: v(1-v) + const
- For k>0, i < n-k: v(1-h[i+k]) + v*h[i-k] + const...

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
    mx = float(np.max(np.concatenate([P, Nv[1:]])))
    return P, Nv, mx

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(40):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        msk = (x < 1.0) if d > 0 else (x > 0.0)
        c = msk.sum()
        if c == 0:
            break
        x[msk] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _gw(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _agrad(h):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    tol = max(1e-9, 3.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1; wP[pos] = 1.0; cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1; wN[pos] = 1.0; cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt; wN /= cnt
    return _gw(h, wP, wN)

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gw(h, wP, wN)

def _fista(h0, a, tl):
    t0 = time.time(); n = len(h0); S = n * 0.5
    h = _repair(h0, n); hp = h.copy(); y = h.copy(); tv = 1.0
    fy = _sval(h, a); fh = fy; bh = h.copy(); bc = _c5(h)
    stale = 0; Lc = 8.0
    for it in range(300000):
        if time.time() - t0 > tl: break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp); tv = tn
            if y.min() < -1e-5 or y.max() > 1+1e-5 or abs(y.sum()-S) > 1e-8:
                y = _proj(y, S)
            fy = _sval(y, a)
        else:
            y = h; fy = fh
        g = _sg(y, a); step = 1.0 / max(Lc, 1e-12); ok = False
        for _ in range(12):
            xn = _proj(y - step * g, S); fx = _sval(xn, a)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5*(1.0/step)*float(np.dot(d, d))
            if fx <= rhs + 1e-10: ok = True; break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp = h; h = xn; fh = fx; stale = 0; Lc = max(Lc*0.8, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < bc: bc = c; bh = h.copy()
        else:
            stale += 1; hp, y, tv = h, h, 1.0; fy = fh
            Lc = max(Lc*0.5, 1.0)
            if stale > 20: break
    c = _c5(h)
    if c < bc: bc = c; bh = h.copy()
    return h, bh, bc

def _sa_local(h0, rng, dl, s0=8e-5, T0=0.05, Tmin=1e-9):
    t0 = time.time(); n = len(h0)
    h = h0.copy(); c = _c5(h); bc = c; bh = h.copy()
    g = _agrad(h); order = np.argsort(g)
    q = max(10, n // 5); lo = order[:q]; hi = order[-q:]
    recomp = 250; it = 0; kick_every = 5000
    while time.time() < dl:
        if it >= recomp:
            c = _c5(h)
            if c < bc: bc = c; bh = h.copy()
            g = _agrad(h); order = np.argsort(g)
            lo = order[:q]; hi = order[-q:]; recomp = it + 250
        fr = min(1.0, (time.time() - t0) / max(1e-9, dl - t0))
        T = T0 * (Tmin / T0) ** fr
        base = max(1e-12, s0 * (2e-6) ** fr)
        r = rng.random()
        if r < 0.55:
            i = int(lo[rng.randint(len(lo))]); j = int(hi[rng.randint(len(hi))])
        elif r < 0.75:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        elif r < 0.90:
            blen = int(rng.randint(2, 5))
            i = int(rng.randint(0, max(1, n - blen - 2)))
            j = i + blen + int(rng.randint(0, max(1, n - i - 2*blen - 1)))
            if j + blen > n: j = n - blen
            oi = h[i:i+blen].copy(); oj = h[j:j+blen].copy()
            h[i:i+blen] = oj; h[j:j+blen] = oi
            cn = _c5(h); delta = cn - c
            if delta < -1e-14 or (T > 1e-12 and rng.random() < np.exp(-max(0.0, delta)/T)):
                c = cn
                if c < bc: bc = c; bh = h.copy()
            else:
                h[i:i+blen] = oi; h[j:j+blen] = oj
            it += 1; continue
        elif it % kick_every < 50 and T > 1e-10:
            nkick = int(rng.randint(5, 20))
            idx = rng.choice(n, nkick, replace=False)
            h[idx] += (rng.rand(nkick) - 0.5) * 0.15
            h = _proj(h, n * 0.5)
            it += 1; continue
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j: it += 1; continue
        sgn = 1 if g[i] <= g[j] else -1
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.3 + 1.7 * rng.random()
            if rng.random() < 0.02: mult *= 12.0
            dv = sgn * min(avail, base * mult)
            h[i] += dv; h[j] -= dv
            cn = _c5(h); delta = cn - c
            if delta < -1e-14 or (T > 1e-12 and rng.random() < np.exp(-max(0.0, delta)/T)):
                c = cn
                if c < bc: bc = c; bh = h.copy()
            else:
                h[i] -= dv; h[j] += dv
        it += 1
    c = _c5(h)
    if c < bc: bc = c; bh = h.copy()
    return bh, bc

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():

    else:

    best = _repair(h0, n0)
    best_c = _c5(best)
    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n0 * 0.5); cv = _c5(hv)
        if cv < best_c: best, best_c = hv, cv
    # Phase 1: FISTA warm-up
    end1 = t0 + 50.0
    hc = best.copy()

        if bc < best_c: best, best_c = bh, bc
        hc = hf
    # Phase 2: Higher-alpha FISTA
    end2 = t0 + 130.0
    for a in [8000, 30000, 100000, 400000]:
        rem = end2 - time.time()
        if rem < 5: break
        hf, bh, bc = _fista(hc, a, rem * 0.24)
        if bc < best_c: best, best_c = bh, bc
        hc = hf
    # Phase 3: Main SA local search
    end3 = t0 + 580.0

    best = _repair(best, len(best)); best_c = _c5(best)
    # Phase 4: Multi-resolution attempt
    end4 = t0 + 820.0

    n = len(best)
    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n * 0.5); cv = _c5(hv)
        if cv < best_c: best, best_c = hv, cv
    best = _repair(best, n); best_c = _c5(best)
    return best, best_c, len(best)
