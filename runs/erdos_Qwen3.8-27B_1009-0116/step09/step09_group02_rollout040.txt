import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(100):
        if abs(s - target) < 1e-13:
            break
        h = np.clip(h * target / s, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _grad_lags(h, lags):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g * dx

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lsq_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    top = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[top] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _lsq_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    top = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[top] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(top)):
        if w[ii] < 1e-15:
            continue
        k = int(top[ii])
        L = k - (n - 1)
        if L >= 0:
            g[L:] += w[ii] * (1.0 - x[:n - L])
            g[:n - L] -= w[ii] * x[L:]
        else:
            M = -L
            g[:n - M] += w[ii] * (1.0 - x[M:])
            g[M:] -= w[ii] * x[:n - M]
    return g

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_lsq_obj, h, jac=_lsq_grad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _pgrad(h, m_lags, steps, step0, mu):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur = _eval(h)
    best, best_h = cur, h.copy()
    stall = 0
    for it in range(steps):
        c = _corr(h)
        top = np.argsort(c)[-m_lags:]
        g = _grad_lags(h, top)
        g -= g.mean()
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-15:
            break
        v /= ng
        taken = False
        for _ in range(14):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur - 1e-16:
                h = h_new; cur = c_new
                if cur < best:
                    best = cur; best_h = h.copy(); stall = 0
                else:
                    stall += 1
                step = min(step * 1.06, step0 * 60.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(max(step * 1.5, step0), step0 * 60.0)
            stall += 1
        if stall > 80:
            h = best_h + 1e-4 * np.random.randn(n)
            h = _repair(h, n)
            cur = _eval(h)
            v = np.zeros(n)
            stall = 0
    return best_h, best

def _anneal(h, n_iter, rng, step0, T0, T1):
    n = len(h)
    cur = _eval(h)
    best, best_h = cur, h.copy()
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(1, n_iter))
        n_p = int(rng.choice([2, 3, 4, 5]))
        idx = rng.choice(n, size=n_p, replace=False)
        d = step0 * (0.5 + rng.rand()) * rng.randn(n_p)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.1 or h_new.max() > 1.1:
            continue
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        delta = c_new - cur
        if delta < -1e-16 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            h = h_new; cur = c_new
            if cur < best:
                best = cur; best_h = h.copy()
    return best_h, best

def _fine(h, n_iter, rng, lo, hi):
    n = len(h)
    cur = _eval(h)
    best, best_h = cur, h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = (lo + (hi - lo) * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cur - 1e-16:
                cur = c
                if c < best:
                    best = c; best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best
