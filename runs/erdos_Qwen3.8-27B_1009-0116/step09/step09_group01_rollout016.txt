def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.clip(np.array(v, dtype=float, copy=True), 0.0, 1.0)
    for _ in range(50):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-12:
                break
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(-d, h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
    return h

def _pn(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    P = corr[n - 1:]
    Nv = corr[n - 2::-1] if n > 1 else np.zeros(1)
    mx = float(np.max(P))
    return P, Nv, mx

def _grad(h, frac=0.04):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    tol = mx * frac
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[np.where(mP)[0] + 1] = 1.0; cnt += np.count_nonzero(mP)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[np.where(mN)[0] + 1] = 1.0; cnt += np.count_nonzero(mN)
    if cnt == 0:
        return np.zeros(n), mx
    wP /= cnt; wN /= cnt
    g = np.zeros(n)
    for k in range(n):
        wp = wP[k]
        wn = wN[k] if k > 0 else 0.0
        if wp == 0.0 and wn == 0.0:
            continue
        nk = n - k
        if nk <= 0:
            continue
        if wp != 0.0:
            g[:nk] += wp * (1.0 - h[k:])
            g[k:] += -wp * h[:nk]
        if wn != 0.0:
            g[:nk] += -wn * h[k:]
            g[k:] += wn * (1.0 - h[:nk])
    g *= dx
    return g, mx
