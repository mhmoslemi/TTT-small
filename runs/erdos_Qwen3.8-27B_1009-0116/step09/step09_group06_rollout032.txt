def _repair(h, n):

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
    return h

def _proj(h, t):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj(x, a, m, n):
    c = _corr(x, n)
    idx = np.argsort(c)[-m:]
    cm = c.max()
    s = a * (c[idx] - cm)
    s = np.clip(s - s.max(), -500, 500)
    return cm + np.log(np.exp(s).sum()) / a

def _grad(x, a, m, n):
    dx = 2.0 / n
    c = _corr(x, n)
    idx = np.argsort(c)[-m:]
    cm = c.max()
    s = a * (c[idx] - cm)
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

            continue

            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cf(x, n):
    return x.sum() - n * 0.5

    try:

                     bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h
