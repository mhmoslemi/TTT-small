```python
    """Gradient of f_L(h) = sum_i h[i]*(1-h[i-L]) for valid i, times 2/n."""
    g = np.zeros(n)
    dx = 2.0 / n
```

```python
def _repair(h, n):
    # Project to [0,1]^n with sum = n/2
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Binary search for offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
```

```python
def _fista(h, t0, t_end, alpha, m):
    n = len(h)
    x = h.copy()
    y = h.copy()
    t = 1.0
    L = alpha * 2.0 / n  # Lipschitz constant estimate (rough)
    lr = 1.0 / L
    best_h, best_c = x.copy(), _eval(x)
    
    while time.time() < t_end:
        # Compute gradient at y
        g = _grad_topm(y, alpha, m, n)
        # Gradient step
        x_new = _proj(y - lr * g, n * 0.5)
        # Update t
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        # Update y
        beta = (t - 1.0) / t_new
        y = x_new + beta * (x_new - x)
        y = np.clip(y, 0.0, 1.0)  # keep in box
        x = x_new
        t = t_new
        
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_h = x.copy()
    
    return best_h, best_c
```
