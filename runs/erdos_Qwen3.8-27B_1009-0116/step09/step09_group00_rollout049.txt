def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _corr(h):
    return np.correlate(h, 1.0-h, mode='full')

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    d = S - s
    if abs(d) > 1e-12:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj_simplex(v, S):
    """Project v onto {x: 0<=x_i<=1, sum(x)=S}"""
    x = np.array(v, dtype=float, copy=True)

        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-13:
            break
    return x

        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)

    # Phase 1: Structured initializations (25s)
    end1 = t0 + 25
    x = np.arange(n0) * (2.0 / n0)
    for k in range(2, 16):
        if time.time() > end1:
            break
        for amp in [0.15, 0.25, 0.35, 0.45, 0.5]:
            if time.time() > end1:
                break
            hs = _proj(0.5 + amp * np.sin(np.pi * k * x), n0 * 0.5)
            cv = _c5(hs)
            if cv < best_c:
                best = hs.copy()
                best_c = cv
    for ns in [2, 3, 4, 5, 6, 7, 8, 10, 12, 14]:
        if time.time() > end1:
            break
        step = n0 // ns
        hs = np.zeros(n0)
        for i in range(ns):
            hs[i * step:(i + 1) * step] = 0.2 if i % 2 == 0 else 0.8
        hs = _proj(hs, n0 * 0.5)
        cv = _c5(hs)
        if cv < best_c:
            best = hs.copy()
            best_c = cv
    # Asymmetric patterns
    for frac in [0.3, 0.35, 0.4, 0.45, 0.55, 0.6]:
        if time.time() > end1:
            break
        hs = np.zeros(n0)
        cut = int(n0 * frac)
        hs[:cut] = 0.1
        hs[cut:] = 0.9
        hs = _proj(hs, n0 * 0.5)
        cv = _c5(hs)
        if cv < best_c:
            best = hs.copy()
            best_c = cv

    # Phase 2: Active-set gradient (300s)
    end2 = t0 + 325
    slot = end2 - time.time()
    if slot > 10:
        h2, c2 = _ps_opt(best.copy(), slot * 0.6, top_k=6, tol=0.015, rng=rng)
        if c2 < best_c:
            best = h2
            best_c = c2
        slot2 = end2 - time.time() - 1
        if slot2 > 10:
            h2b, c2b = _ps_opt(best.copy(), slot2, top_k=12, tol=0.03, rng=rng)
            if c2b < best_c:
                best = h2b
                best_c = c2b

    # Phase 3: Gradient pair optimization (120s)
    end3 = t0 + 445
    slot = end3 - time.time()
    if slot > 5:
        h3, c3 = _pair_opt(best.copy(), slot, rng)
        if c3 < best_c:
            best = h3
            best_c = c3

    # Phase 4: Multi-resolution cycle (180s)
    end4 = t0 + 625
    slot = end4 - time.time()
    if slot > 30:
        for m in [280, 224, 448]:
            if time.time() >= end4 - 20:
                break
            hm = _resample(best, m)
            per = min(55.0, (end4 - time.time() - 10) / 1.0)
            hd, cd = _ps_opt(hm, per * 0.7, top_k=6, tol=0.02, rng=rng)
            hb = _resample(hd, n0)
            cv = _c5(hb)
            if cv < best_c:
                best = hb
                best_c = cv
            slot2 = min(25.0, end4 - time.time() - 5)
            if slot2 > 5:
                hb2, c2 = _pair_opt(best.copy(), slot2, rng)
                if c2 < best_c:
                    best = hb2
                    best_c = c2

    # Phase 5: Coordinated active-set + pair (200s)
    end5 = t0 + 825
    hcur = best.copy()
    while time.time() < end5:
        slot = end5 - time.time()
        if slot < 10:
            break
        per = min(70.0, slot * 0.5)
        h5, c5 = _ps_opt(hcur, per, top_k=8, tol=0.01, rng=rng)
        if c5 < best_c:
            best = h5
            best_c = c5
        hcur = h5
        slot2 = end5 - time.time()
        if slot2 < 10:
            break
        per2 = min(50.0, slot2 * 0.5)
        h6, c6 = _pair_opt(hcur, per2, rng)
        if c6 < best_c:
            best = h6
            best_c = c6
        hcur = h6
        # Perturb occasionally
        if rng.random() < 0.3:
            hcur = _repair(hcur + 0.03 * rng.randn(n0), n0)

    # Phase 6: Final coordinate polish (remaining)
    end6 = t0 + budget_s - 3
    slot = end6 - time.time()
    if slot > 5:
        h7, c7 = _coord_opt(best.copy(), slot, rng)
        if c7 < best_c:
            best = h7
            best_c = c7

    # Symmetry variants
    for hh in [1.0 - best.copy(), best.copy()[::-1]]:
        h2 = _proj(hh, len(hh) * 0.5)
        cv = _c5(h2)
        if cv < best_c:
            best = h2
            best_c = cv

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
