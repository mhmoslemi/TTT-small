    n = len(h)
    dx = 2.0 / n

    #   term where j+k'=i: h[j] * (1-h[i]) → slope = -h[j] * dx

    # c[m] = sum_j a[j] * v[j + m - (n-1)] for valid indices

    # h[i] appears as v[...] when i = j + m - n + 1, i.e., m = n - 1 + i - j:
    #   contributes -h[j] to c[m]

    h_old = h[i]

    c_plus = np.correlate(h, 1.0 - h, mode='full') * dx

    idx = np.argpartition(-A, min(top_m, m_full)-1)[:top_m]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

    # valid when 0 <= i + m_idx - n + 1 <= n-1
    # i.e., n-1-i <= m_idx <= 2n-2-i
    lo1 = max(0, n - 1 - i)
    hi1 = min(m - 1, 2 * n - 2 - i)
    for m_idx in range(lo1, hi1 + 1):
        v_idx = i + m_idx - n + 1
        slopes[m_idx] += dx * (1.0 - h[v_idx])
    # h[i] appears as second factor (inside 1-...): v_idx = i, so j = i - m_idx + n - 1
    # valid when 0 <= i - m_idx + n - 1 <= n-1
    # i.e., i <= m_idx <= i + n - 1
    lo2 = max(0, i)
    hi2 = min(m - 1, i + n - 1)
    for m_idx in range(lo2, hi2 + 1):
        a_idx = i - m_idx + n - 1
        slopes[m_idx] -= dx * h[a_idx]
    return slopes

def _solve_1d_minimax(A, B, t0):
    """Solve min_{t in [0,1]} max_k (A[k] + B[k]*t). Returns optimal t."""
    m = len(A)
    best_t = t0
    best_val = np.max(A + B * t0)
    
    # Check boundaries
    for t in [0.0, 1.0]:
        val = np.max(A + B * t)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Check intersection points of top lines
    # Sort by B to reduce candidates
    order = np.argsort(A)
    # Take top m lines by A (most active)
    top = order[:min(m, 60)]
    
    for p in range(len(top)):
        for q in range(p + 1, len(top)):
            dB = B[top[p]] - B[top[q]]
            if abs(dB) < 1e-15:
                continue
            t = (A[top[q]] - A[top[p]]) / dB
            if -0.1 <= t <= 1.1:
                t = np.clip(t, 0.0, 1.0)
                val = np.max(A[top] + B[top] * t)
                if val < best_val:
                    best_val = val
                    best_t = t
    return best_t

    best_c = _eval(h)
    best_h = h.copy()

            best_h = h.copy()

            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _slopes(h, i):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    sl = np.zeros(m)
    lo1 = max(0, n - 1 - i)
    hi1 = min(m - 1, 2 * n - 2 - i)
    if lo1 <= hi1:
        vi = np.arange(lo1, hi1 + 1) + i - n + 1
        sl[lo1:hi1 + 1] += dx * (1.0 - h[vi])
    lo2 = max(0, i)
    hi2 = min(m - 1, i + n - 1)
    if lo2 <= hi2:
        ai = i - np.arange(lo2, hi2 + 1) + n - 1
        sl[lo2:hi2 + 1] -= dx * h[ai]
    return sl

def _minimax_1d(A, B, t0, tm=40):
    m = min(tm, len(A))
    idx = np.argpartition(-A, m - 1)[:m]
    At = A[idx].copy()
    Bt = B[idx].copy()
    lo, hi = 0.0, 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) * 0.3333
        m2 = hi - (hi - lo) * 0.3333
        if np.max(At + Bt * m1) < np.max(At + Bt * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def _coord_descent(h, n_passes, rng, tm=40):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c = c.max()
    best_h = h.copy()
    for _ in range(n_passes):
        order = rng.permutation(n)
        moved = 0
        for i in order:
            sl = _slopes(h, i)
            ho = h[i]
            A = c - sl * ho
            t_new = _minimax_1d(A, sl, ho, tm)
            if abs(t_new - ho) > 1e-10:
                c += sl * (t_new - ho)
                h[i] = t_new
                moved += 1
        h = _repair(h, n)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cv = c.max()
        if cv < best_c - 1e-14:
            best_c = cv
            best_h = h.copy()
        elif moved == 0:
            break
    return best_h, best_c

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': lambda x: np.ones(len(x)), 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        if res.success or _eval(res.x) < _eval(h):
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _pg(h, alpha, ml, steps, lr0, mom):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    lr = lr0
    for _ in range(steps):
        g = _smooth_grad(h, alpha, ml)
        v = mom * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        if _eval(h_new) < best_c:
            best_c = _eval(h_new)
            best_h = h_new.copy()
        step = lr
        taken = False
        for _ in range(12):
            h_try = _proj(h - step * g, n * 0.5)
            if _eval(h_try) < _eval(h):
                h = h_try
                v = mom * v
                lr = min(step * 1.1, lr0 * 20.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            h = h_new
            lr = min(lr * 1.05, lr0 * 20.0)
    return best_h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _down(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]

            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                cv = _eval(h2)
                if cv < c5c:
                    c5c = cv
                    if cv < best_c5:
                        best_c5, best_h = cv, h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
