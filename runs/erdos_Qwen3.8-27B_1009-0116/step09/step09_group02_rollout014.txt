                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(_corr(h).max())

def _lag_grad(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g - g.mean()

def _active_subgrad(h, k_list):
    g = np.zeros(len(h))
    for k in k_list:
        L = k - (len(h) - 1)
        g += _lag_grad(h, L)
    g /= max(len(k_list), 1)
    return g - g.mean()

def _smooth_fn(x, alpha, mk):
    c = _corr(x)
    idx = np.argsort(c)[-mk:]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -350, 350)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_jac(x, alpha, mk):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    idx = np.argsort(c)[-mk:]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -350, 350)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        wk = w[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    g *= dx
    return g - g.mean()

def _proj_thresh(h, target):
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad_opt(h, alpha, mk, steps, s0, mu0):
    n = len(h)
    v = np.zeros(n)
    step = s0
    mu = mu0
    cur_c = _eval(h)
    for it in range(steps):
        c = _corr(h)
        idx = np.argsort(c)[-mk:]
        m = c[idx].max()
        s = np.clip(alpha * (c[idx] - m), -350, 350)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-14:
                continue
            L = k - (n - 1)
            if L >= 0:
                g[L:] += w[ii] * (1.0 - h[:n - L])
                g[:n - L] -= w[ii] * h[L:]
            else:
                M = -L
                g[:n - M] += w[ii] * (1.0 - h[M:])
                g[M:] -= w[ii] * h[:n - M]
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = mu * v + (1.0 - mu) * g / gn
        vn = np.linalg.norm(v)
        if vn < 1e-16:
            break
        v /= vn
        mu = max(0.5, mu * 0.999)
        taken = False
        for _t in range(10):
            h_new = _proj_thresh(h - step * v, n / 2.0)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-17:
                h = h_new
                cur_c = c_new
                step = min(step * 1.04, s0 * 50.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(max(step * 1.6, s0), s0 * 50.0)
    return _repair(h, n)

def _slsqp_run(h, alpha, mk, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_fn, h, jac=_smooth_jac, args=(alpha, mk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    c_cache = _corr(h)
    top_k = np.argsort(c_cache)[-15:]
    for it in range(n_iter):
        if it % 500 == 0:
            c_cache = _corr(h)
            top_k = np.argsort(c_cache)[-15:]
        r = rng.rand()
        if r < 0.25:
            sz = rng.randint(2, 5)
            scale = base_step * (0.5 + rng.rand())
        elif r < 0.6:
            sz = rng.randint(5, 15)
            scale = base_step * (2.0 + rng.rand())
        else:
            sz = rng.randint(15, 40)
            scale = base_step * (5.0 + 3.0 * rng.rand())
        idx = rng.randint(0, n, size=sz)
        d = scale * rng.randn(sz)
        if sz >= 10:
            g = _active_subgrad(h, list(top_k))
            d -= 0.4 * scale * np.abs(g[idx]) / max(1e-15, np.linalg.norm(g))
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.15 or h_new.max() > 1.15:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h_new.copy()
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 30)) % n
        d = (3e-9 + 4e-8 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_opt(h, n_iter, rng, gs):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 40, size=gs)) % n)
        d = (5e-9 + 8e-9 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-17:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Progressive SLSQP - increase alpha, increase mk
    alphas = [10000, 50000, 200000, 1000000, 5000000, 50000000, 500000000]
    mks = [15, 25, 50, 100, 200, 400, 600]
    for i in range(len(alphas)):
        if _el() > 250:
            break
        h_new = _slsqp_run(best_h.copy(), alphas[i], min(mks[i], n), 800)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Projected gradient with multiple configs
    pg_configs = [
        (50000, 30, 1000, 0.004, 0.82),
        (300000, 50, 800, 0.002, 0.80),
        (2000000, 70, 600, 0.001, 0.78),
        (10000000, 100, 500, 0.0006, 0.75),
        (50000000, 150, 400, 0.0003, 0.70),
    ]
    for alpha, mk, steps, s0, mu in pg_configs:
        if _el() > budget_s - 250:
            break
        h_new = _proj_grad_opt(best_h.copy(), alpha, mk, steps, s0, mu)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Second SLSQP pass with high alpha
    if _el() < budget_s - 200:
        for alpha, mk in [(1000000000, 300), (5000000000, 500)]:
            if _el() > budget_s - 100:
                break
            h_new = _slsqp_run(best_h.copy(), alpha, min(mk, n), 400)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Local search with variable neighborhood
    if _el() < budget_s - 60:
        rem = budget_s - 60 - _el()
        ntries = int(min(rem * 180, 200000))
        best_h, best_c5 = _local_search(best_h, ntries, rng, 3e-5)

    # Phase 5: Fine 2-point polish
    if _el() < budget_s - 20:
        rem = budget_s - 20 - _el()
        ntries = int(min(rem * 700, 800000))
        best_h, best_c5 = _fine_polish(best_h, ntries, rng)

    # Phase 6: Group coordination
    if _el() < budget_s - 10:
        rem = budget_s - 10 - _el()
        ntries = int(min(rem * 350, 300000))
        best_h, best_c5 = _group_opt(best_h, ntries, rng, 5)

    # Phase 7: Final fine polish
    if _el() < budget_s - 2:
        rem = budget_s - 2 - _el()
        ntries = int(min(rem * 900, 500000))
        best_h, best_c5 = _fine_polish(best_h, ntries, rng)

    # Phase 8: One more projected gradient sweep with best settings
    if _el() < budget_s - 1:
        h_new = _proj_grad_opt(best_h.copy(), 10000000, 80, 200, 0.0005, 0.72)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new
        best_h, best_c5 = _fine_polish(best_h, 20000, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
