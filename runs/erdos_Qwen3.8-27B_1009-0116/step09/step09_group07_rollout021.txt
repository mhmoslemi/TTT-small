def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):

Hmm wait, I made an error. Let me redo the objective:
f(x) = (1/α) * log(Σ_k exp(α * c[k]))

Let M = max_k c[k]. Then:
f = (1/α) * log(exp(αM) * Σ_k exp(α(c[k]-M)))
  = M + (1/α) * log(Σ_k exp(α(c[k]-M)))

Let u_k = α(c[k]-M), so max(u_k) = 0.
Σ_k exp(u_k) ≥ 1.

For numerical stability, if α is large, u_k for non-max k will be very negative, and exp(u_k) ≈ 0. This is fine.

The gradient:
∂f/∂x[j] = (1/α) * Σ_k exp(α*c[k]) * α * ∂c[k]/∂x[j] / Σ_m exp(α*c[m])
          = Σ_k w_k * ∂c[k]/∂x[j]
where w_k = exp(α*c[k]) / Σ_m exp(α*c[m])

And ∂c[k]/∂x[j] = dx * beta[j,k] (as computed in CD).

So grad[j] = dx * Σ_k w_k * beta[j,k]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _beta_j(h, j):
    n = len(h)
    nm1 = n - 1
    cmax = 2 * n - 1
    beta = np.zeros(cmax)
    k1_lo, k1_hi = j, min(cmax - 1, j + nm1)
    beta[k1_lo:k1_hi + 1] += (1.0 - h[j - (np.arange(k1_lo, k1_hi + 1) - nm1)])
    k2_lo, k2_hi = max(0, nm1 - j), min(cmax - 1, cmax - 1 - j)
    beta[k2_lo:k2_hi + 1] -= h[j + (np.arange(k2_lo, k2_hi + 1) - nm1)]
    return beta

def _cd_pass(h, rng, n_top=18):
    n = len(h)
    c = _corr(h)
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:

        beta = _beta_j(h, j)
        g0 = c - beta * h[j]
        vals = g0 + beta * h[j]
        top_idx = np.argpartition(vals, -n_top)[-n_top:]
        top_idx = top_idx[np.argsort(vals[top_idx])[::-1]]
        candidates = np.array([0.0, 1.0])

            for jj in range(ii + 1, len(top_idx)):

        if len(candidates) < 5:
            candidates = np.append(candidates, rng.uniform(0, 1, 8))
        best_t = h[j]
        best_v = float('inf')
        for t in candidates:

            if v < best_v:

                best_t = t
        lo_t, hi_t = max(0.0, best_t - 0.05), min(1.0, best_t + 0.05)

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    nm1 = n - 1
    cmax = 2 * n - 1
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    wfull = np.zeros(cmax)
    wfull[idx] = w
    for j in range(n):
        beta_j = _beta_j(x, j)
        grad[j] = np.dot(wfull, beta_j) * dx
    return grad

def _slsqp(h, alpha, m_lags, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'disp': 0})
        return _repair(res.x, n)
    except Exception:
        return h
