    if s < 1e-12:
        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:

        for idx in order:

            lo = mid
        else:

def _coord_polish(h, n, iters, rng, step_sizes):
    """Fine coordinate descent with multiple step sizes."""
    target = n * 0.5
    cc = _eval(h)
    for _ in range(iters):
        improved = False
        for ss in step_sizes:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            if i == j:
                j = (i + 1) % n
            d = ss * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < cc:
                    cc = cn
                    improved = True
                else:
                    h[i], h[j] = ni - d, nj + d
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sgrad(h, alpha, ml):
    n = len(h)
    c = _eval_corr(h)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        L = k - (n - 1)
        if L >= 0:
            g[:n-L] += w[ii] * (1.0 - h[L:n])
            g[L:] -= w[ii] * h[:n-L]
        else:
            M = -L
            g[M:] += w[ii] * (1.0 - h[:n-M])
            g[:n-M] -= w[ii] * h[M:]
    return g

def _sobj(x, alpha, ml):
    n = len(x)
    c = _eval_corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n*0.5,)}]
    try:
        r = minimize(_sobj, h, jac=lambda x, *a: _sgrad(x, a[0], a[1]),
                     args=(alpha, ml), method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _momentum_phase(h, alpha, ml, steps, s0, mu, noise):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    step = s0
    cur = _eval(h)
    for it in range(steps):
        g = _sgrad(h, alpha, ml)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn
        noise_vec = noise * np.random.randn(n)
        noise_vec -= noise_vec.mean()
        v = mu * v + (1.0 - mu) * g + noise_vec
        vn = np.linalg.norm(v)
        if vn > 1e-15:
            v = v / vn
        taken = False
        for _ in range(14):
            hn = _proj(h - step * v, target)
            cn = _eval(hn)
            if cn < cur - 1e-18:
                h, cur, taken = hn, cn, True
                step = min(step * 1.07, s0 * 25.0)
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, s0 * 25.0)
    return _repair(h, n), cur

def _sweep_cd(h, n, iters, step_sizes, stride):
    """Structured cyclic coordinate descent."""
    target = n * 0.5
    cc = _eval(h)
    rng_local = np.random.RandomState(12345)
    for it in range(iters):
        improved = False
        i = (it * stride) % n
        for ss in step_sizes:
            for dj in [-1, 1, 2, -2, 3]:
                j = (i + dj) % n
                if i == j: continue
                d = ss * rng_local.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                    h[i], h[j] = ni, nj
                    cn = _eval(h)
                    if cn < cc:
                        cc = cn; improved = True
                    else:
                        h[i], h[j] = ni - d, nj + d
        if not improved:
            break
    return h, cc
