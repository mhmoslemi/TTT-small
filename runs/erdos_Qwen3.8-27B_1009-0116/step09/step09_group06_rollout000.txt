def _fill_grad(h, k, n, out):

import numpy as np
import time
from scipy.optimize import linprog, minimize

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval_max(h):
    return float(np.max(_eval_vec(h)))

def _repair(h):
    n = len(h)
    t = 0.5 * n
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(h - m, 0.0, 1.0)) > t:
            lo = m
        else:
            hi = m
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(6):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            idx = np.flatnonzero(h < 1.0)
            if not idx.size:
                break
            w = 1.0 - h[idx]
            h[idx] += d * w / w.sum()
        else:
            idx = np.flatnonzero(h > 0.0)
            if not idx.size:
                break
            w = h[idx]
            h[idx] -= (-d) * w / w.sum()
    d = t - h.sum()
    if abs(d) > 1e-15:
        if d > 0:
            j = int(np.argmax(1.0 - h))
            h[j] = min(1.0, h[j] + d)
        else:
            j = int(np.argmin(h))
            h[j] = max(0.0, h[j] - (-d))
    return np.clip(h, 0.0, 1.0)

def _quick(h, t):
    h = h + (t - h.sum()) / float(len(h))
    if h.min() < -1e-13 or h.max() > 1.0 + 1e-13:
        return _repair(h)
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - t) > 1e-10:
        return _repair(h)
    return h

def _active(c, maxm, near):
    m = min(maxm, len(c))
    if m <= 0:
        return np.array([], dtype=int)
    C = float(np.max(c))
    top = np.argsort(c)[::-1][:m]
    if near <= 0.0:
        return top
    nr = np.flatnonzero(c >= C - near)
    if nr.size <= m:
        arr = np.unique(np.concatenate((top, nr)))
        return arr if arr.size <= maxm + 32 else top
    return top

def _fill_grad(h, k, n, out):
    out.fill(0.0)
    dx = 2.0 / n
    L = k - (n - 1)
    if L >= 0:
        out[L:] += dx * (1.0 - h[:n - L])
        out[:n - L] -= dx * h[L:]
    else:
        M = -L
        out[:n - M] += dx * (1.0 - h[M:])
        out[M:] -= dx * h[:n - M]

def _lp_dir(h, c, act, rho):
    n = len(h)
    m = len(act)
    if m == 0 or rho <= 0.0:
        return None, float(np.max(c))
    C = float(np.max(c))
    A = np.zeros((m, n + 1))
    b = np.empty(m)
    for r, k in enumerate(act):
        _fill_grad(h, k, n, A[r, :n])
        A[r, n] = -1.0
        b[r] = -c[k]
    Aeq = np.concatenate((np.ones(n), [0.0])).reshape(1, -1)
    lb = np.maximum(-rho, -h)
    ub = np.minimum(rho, 1.0 - h)
    bounds = list(zip(lb.tolist(), ub.tolist()))
    bounds.append((0.0, 1.0))
    ce = np.zeros(n + 1)
    ce[n] = 1.0
    res = None
    for meth in ('highs', 'interior-point', 'simplex'):
        try:
            r = linprog(ce, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0], bounds=bounds, method=meth)
            if r.status == 0:
                res = r
                break
        except Exception:
            pass
    if res is None:
        return None, C
    d = np.asarray(res.x[:n], float).copy()
    tau = float(res.x[n])
    d -= d.mean()
    if float(np.abs(d).max()) < 1e-10:
        return None, tau
    return d, tau

def _grad_dir(h, c, act, alpha=500.0):
    n = len(h)
    v = np.zeros(n)
    g = np.zeros(n)
    vals = c[act]
    ordv = np.argsort(vals)[::-1]
    s = alpha * (vals[ordv] - vals[ordv[0]])
    s = np.clip(s - s.max(), -500.0, 0.0)
    w = np.exp(s)
    w /= w.sum()
    for wk, k in zip(w, act[ordv]):
        _fill_grad(h, k, n, g)
        v += wk * g
    v -= v.mean()
    return -v

def _random_dir(h, rho, rng):
    n = len(h)
    if n < 2:
        return None
    if rng.rand() < 0.65:
        m = max(8, n // 12)
        x = np.linspace(0.0, 1.0, m)
        y = np.linspace(0.0, 1.0, n, endpoint=False)
        r = np.interp(y, x, rng.randn(m))
    else:
        r = np.zeros(n)
        k = min(n, max(2, int(rng.randint(4, 13))))
        idx = rng.choice(n, k, replace=False)
        vals = rng.randn(k)
        vals -= vals.mean()
        r[idx] = vals
    r -= r.mean()
    norm = np.sqrt(np.mean(r * r))
    if norm < 1e-14:
        return None
    r *= rho / norm
    r -= r.mean()
    return r

def _line_search(h, d, curc):
    if d is None or float(np.abs(d).max()) < 1e-14:
        return None
    n = len(h)
    t = 0.5 * n
    amax = 1.0
    pos = d > 1e-14
    if np.any(pos):
        lim = (1.0 - h[pos]) / d[pos]
        amax = min(amax, float(np.min(lim)))
    neg = d < -1e-14
    if np.any(neg):
        lim = h[neg] / (-d[neg])
        amax = min(amax, float(np.min(lim)))
    amax = min(amax, 1.0 - 1e-12)
    if amax <= 1e-12:
        return None
    best = None
    best_c = curc
    for f in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
        a = f * amax
        if a <= 0.0:
            continue
        hn = h + a * d
        hn = _quick(hn, t)
        cv = _eval_vec(hn)
        cm = float(np.max(cv))
        if cm < best_c - 1e-13:
            best = (hn, cm, cv, a)
            best_c = cm
    return best

def _perturb(h, size, rng):
    d = _random_dir(h, size, rng)
    if d is None:
        d = rng.randn(len(h)) * size
        d -= d.mean()
    return _repair(h + d)

def _down(h, f):
    n = len(h)
    f = max(1, f)
    m = max(1, n // f)
    f = max(1, n // m)
    return _repair(h[:m * f].reshape(m, f).mean(axis=1))

def _up(h, f):
    f = max(1, f)
    return _repair(np.repeat(h, f))

def _s_obj(x):
    return float(x[-1])

def _s_obj_jac(x):
    out = np.zeros(len(x))
    out[-1] = 1.0
    return out

def _s_eq(x):
    return float(np.sum(x[:-1]) - 0.5 * len(x[:-1]))

def _s_eq_jac(x):
    out = np.zeros(len(x))
    out[:-1] = 1.0
    return out

def _s_neq(x, act):
    h = x[:-1]
    return x[-1] - _eval_vec(h)[act]

def _s_neq_jac(x, act):
    n = len(x) - 1
    m = len(act)
    J = np.zeros((m, n + 1))
    J[:, n] = 1.0
    g = np.zeros(n)
    for r, k in enumerate(act):
        _fill_grad(x[:-1], k, n, g)
        J[r, :n] = -g
    return J

def _slsqp_active(h, maxiter):
    cv = _eval_vec(h)
    c0 = float(np.max(cv))
    act = np.argsort(cv)[::-1][:max(8, min(24, len(cv)))]
    x0 = np.concatenate([h, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * len(h)
    bounds.append((0.0, 1.0))
    cons = [
        {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_jac},
        {'type': 'ineq', 'fun': _s_neq, 'jac': _s_neq_jac, 'args': (act,)}
    ]
    try:
        res = minimize(_s_obj, x0, jac=_s_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            h2 = _repair(res.x[:-1])
            return h2, _eval_max(h2)
    except Exception:
        pass
    return h, c0

def _local_loop(h, end_time, rng, maxm, near, rho):
    h = _repair(h)
    c = _eval_vec(h)
    cur = float(np.max(c))
    best_h = h.copy()
    best = cur
    no = 0
    att = 0
    m = maxm
    tol = near
    while time.time() < end_time - 3.0:
        if cur < best - 1e-15:
            best = cur
            best_h = h.copy()
        act = _active(c, m, tol)
        improved = False

        dp, tau = _lp_dir(h, c, act, rho)
        if dp is not None and tau < cur - 1e-12:
            ls = _line_search(h, dp, cur)
            if ls is not None:
                h = _repair(ls[0])
                c = _eval_vec(h)
                cur = float(np.max(c))
                improved = True
                if float(np.abs(dp).max()) > 0.9 * rho and rho < 0.08:
                    rho = min(0.08, rho * 1.15)

        if not improved:
            gd = _grad_dir(h, c, act)
            ls = _line_search(h, gd, cur)
            if ls is not None:
                h = _repair(ls[0])
                c = _eval_vec(h)
                cur = float(np.max(c))
                improved = True
                rho = max(0.004, rho * 0.95)

        if not improved and att < 3:
            rd = _random_dir(h, max(rho, 0.012), rng)
            ls = _line_search(h, rd, cur)
            if ls is not None:
                h = _repair(ls[0])
                c = _eval_vec(h)
                cur = float(np.max(c))
                improved = True
                rho = max(0.004, rho * 0.9)

        if improved:
            if cur < best - 1e-14:
                best = cur
                best_h = h.copy()
            no = 0
            att = 0
            continue

        no += 1
        att += 1

        if no >= 30 and time.time() < end_time - 150.0:
            h2, c2 = _slsqp_active(h, 40)
            if c2 < best - 1e-12:
                h = _repair(h2)
                c = _eval_vec(h)
                cur = float(np.max(c))
                best = cur
                best_h = h.copy()
                no = 0
                att = 0
                rho = 0.03
                continue

        if att >= 4:
            size = min(0.06, 0.004 + 0.0015 * min(no, 30))
            src = best_h if rng.rand() < 0.85 else h
            h = _perturb(src, size, rng)
            c = _eval_vec(h)
            cur = float(np.max(c))
            att = 0
            no = 0
            if rng.rand() < 0.25 and m < 180:
                m += 16
            tol = min(1e-3, tol * 1.2)
            rho = 0.03
        else:
            rho = max(0.003, rho * 0.6)
            if no > 10 and m < 180:
                m += 8
            if no > 20 and tol < 1e-3:
                tol = min(1e-3, tol * 1.3)

    best_h = _repair(best_h)
    return best_h, _eval_max(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(560, 0.5)

    while len(h0) > 1200:
        h0 = _down(h0, len(h0) // 1200 + 1)

    h = _repair(h0)
    best_h = h.copy()
    best = _eval_max(h)

    base = best_h.copy()
    for tw in range(4):
        x = base.copy()
        if tw & 1:
            x = 1.0 - x
        if tw & 2:
            x = x[::-1]
        x = _repair(x)
        cx = _eval_max(x)
        if cx < best - 1e-15:
            best = cx
            best_h = x

    if budget_s > 700 and len(best_h) >= 400:
        f = max(2, len(best_h) // 128)
        hc = _down(best_h, f)
        coarse_s = min(120.0, float(budget_s) / 6.0)
        if coarse_s > 35.0:
            hc, _ = _local_loop(hc, t0 + coarse_s, rng, 48, 4e-4, 0.05)
            hu = _up(hc, max(1, len(best_h) // len(hc)))
            basin_s = min(150.0, float(budget_s) / 8.0)
            end2 = min(t0 + budget_s - 450, time.time() + basin_s)
            if end2 > time.time() + 20.0:
                hu, chu = _local_loop(hu, end2, rng, 110, 2e-4, 0.03)
                if chu < best:
                    best = chu
                    best_h = hu

    end_main = t0 + budget_s - 15.0
    best_h, best = _local_loop(best_h, end_main, rng, 110, 2e-4, 0.03)

    while time.time() < t0 + budget_s - 30.0:
        h2, c2 = _slsqp_active(best_h, 25)
        if c2 < best - 1e-12:
            best_h = h2
            best = c2
        else:
            break

    best_h = _repair(best_h)
    best = _eval_max(best_h)
    return best_h, best, len(best_h)
