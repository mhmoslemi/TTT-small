def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(100):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _block_cd(h, n, rng, corr, block_size, n_blocks):
    nm1 = n - 1
    for _ in range(n_blocks):
        idx = rng.choice(n, block_size, replace=False)
        for p in idx:
            A = np.zeros(2 * n - 1)
            L_lo = max(-nm1, -p)
            L_hi = min(nm1, nm1 - p)
            if L_lo <= L_hi:
                Ls = np.arange(L_lo, L_hi + 1)
                A[Ls + nm1] += 1.0 - h[p + Ls]
            L_lo2 = max(-nm1, p - nm1)
            L_hi2 = min(nm1, p)
            if L_lo2 <= L_hi2:
                Ls2 = np.arange(L_lo2, L_hi2 + 1)
                A[Ls2 + nm1] -= h[p - Ls2]
            B = corr - A * h[p]
            new_val = _coord_opt_1d(A, B)
            delta = new_val - h[p]
            if abs(delta) > 1e-15:
                corr += A * delta
                h[p] = new_val
    return h, corr

def _fine_polish(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = (1e-8 + 3e-7 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_polish(h, n_iter, rng, gs=4):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 25, size=gs)) % n)
        d = (1e-8 + 5e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _elapsed():
        return time.time() - t0

    # Phase 1: Pure coordinate descent sweeps
    sweep_count = 0
    no_improve = 0
    corr = _corr_full(h)
    while _elapsed() < budget_s - 250:
        h, corr = _coord_sweep(h, n, rng, corr)
        sweep_count += 1
        if sweep_count % 8 == 0:
            h = _proj_sum(h, n / 2.0)
            h = _repair(h, n)
            corr = _corr_full(h)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200 and _elapsed() < budget_s - 250:
                h = np.array(best_h, dtype=float)
                n_pert = 40
                pert_idx = rng.randint(0, n, n_pert)
                h[pert_idx] += rng.randn(n_pert) * 0.08
                h = _repair(h, n)
                corr = _corr_full(h)
                no_improve = 0

    # Phase 2: Block coordinate descent (2-5 coords at a time)
    if _elapsed() < budget_s - 150:
        h2 = np.array(best_h, dtype=float)
        corr2 = _corr_full(h2)
        for bs in [2, 3, 5]:
            if _elapsed() > budget_s - 150:
                break
            for _ in range(300):
                if _elapsed() > budget_s - 155:
                    break
                h2, corr2 = _block_cd(h2, n, rng, corr2, bs, 50)
                h2 = _proj_sum(h2, n / 2.0)
                h2 = _repair(h2, n)
                c5n = _eval(h2)
                if c5n < best_c5 - 1e-12:
                    best_c5 = c5n
                    best_h = h2.copy()

    # Phase 3: Fine polish
    rem = budget_s - 20 - _elapsed()
    if rem > 5:
        best_h, best_c5 = _fine_polish(best_h, int(rem * 400), rng)

    # Phase 4: Group polish
    rem = budget_s - 5 - _elapsed()
    if rem > 3:
        best_h, best_c5 = _group_polish(best_h, int(rem * 250), rng)

    # Phase 5: Final ultra-fine
    rem = budget_s - 1 - _elapsed()
    if rem > 2:
        best_h, best_c5 = _fine_polish(best_h, int(rem * 500), rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
