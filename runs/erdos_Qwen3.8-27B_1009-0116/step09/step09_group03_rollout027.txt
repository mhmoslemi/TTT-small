    target_ones = n // 2
    idx = np.argsort(-h)  # largest values first
    h_bin = np.zeros(n)
    h_bin[idx[:target_ones]] = 1.0
    return h_bin

def _slsqp_opt(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    
    def _sum_fn(x):
        return x.sum() - n * 0.5
    
    def _sum_jac(x):
        return np.ones(n)
    
    def _obj(x):
        c = _eval_vec(x)
        m = c.max()
        idx = np.argpartition(-c, ml)[:ml]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def _grad(x):
        c = _eval_vec(x)
        m = c.max()
        idx = np.argpartition(-c, ml)[:ml]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * 2.0 / n
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n-L])
                g[:n-L] -= wk * x[L:]
            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        return g
    
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        return _repair(res.x, n)

    if n_ones > n//2:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

    m4 = (i4 >= 0) & (i4 < n) & (i4 != a) & (i4 != b)
    d[m4] -= h[i4[m4]]
    return d
