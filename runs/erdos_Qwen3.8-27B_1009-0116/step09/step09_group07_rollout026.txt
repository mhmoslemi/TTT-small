- min{b_k : active} if we can decrease t
- max{b_k : active} if we can increase t

            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-13:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _lsq_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _lsq_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, target):
    return x.sum() - target

def _sj(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        res = minimize(_lsq_obj, h, jac=_lsq_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_refine(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: CD passes (35% budget)
    p1_end = t0 + budget_s * 0.35
    stall = 0
    while time.time() < p1_end:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            stall = 0

    # Phase 2: SLSQP alpha sweep (25% budget)
    p2_end = t0 + budget_s * 0.60
    alphas = [2000, 8000, 30000, 100000, 400000, 1500000, 5000000]
    mls = [30, 50, 50, 80, 80, 100, 120]
    for i, a in enumerate(alphas):
        if time.time() > p2_end:
            break
        h_new = _slsqp(best_h.copy(), a, mls[i], 500)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-9:
            best_c5 = c5n
            best_h = h_new
        # Quick CD cleanup after SLSQP
        for _ in range(3):
            h_new = _cd_pass(h_new, rng)
            h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-10:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: More CD (15% budget)
    p3_end = t0 + budget_s * 0.75
    h2 = best_h.copy()
    while time.time() < p3_end:
        h2 = _cd_pass(h2, rng)
        h2 = _repair(h2, n)
        c5 = _eval(h2)
        if c5 < best_c5 - 5e-11:
            best_c5 = c5
            best_h = h2.copy()

    # Phase 4: Perturbation restarts (15% budget)
    p4_end = t0 + budget_s * 0.90
    perts = [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
    for ps in perts:
        if time.time() > p4_end:
            break
        h_p = best_h + ps * rng.randn(n)
        h_p = _repair(h_p, n)
        for _ in range(4):
            h_p = _cd_pass(h_p, rng)
            h_p = _repair(h_p, n)
        c5p = _eval(h_p)
        if c5p < best_c5 - 1e-9:
            best_c5 = c5p
            best_h = h_p.copy()
        # Also try partial perturbation (only 20% of coords)
        if time.time() > p4_end:
            break
        mask = rng.rand(n) < 0.2
        h_p2 = best_h.copy()
        h_p2[mask] += ps * 2 * rng.randn(mask.sum())
        h_p2 = _repair(h_p2, n)
        for _ in range(4):
            h_p2 = _cd_pass(h_p2, rng)
            h_p2 = _repair(h_p2, n)
        c5p2 = _eval(h_p2)
        if c5p2 < best_c5 - 1e-9:
            best_c5 = c5p2
            best_h = h_p2.copy()

    # Phase 5: Final SLSQP + CD (7% budget)
    if time.time() < t0 + budget_s - 30:
        h_new = _slsqp(best_h.copy(), 8000000, 100, 400)
        for _ in range(5):
            h_new = _cd_pass(h_new, rng)
            h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 6: Local search (3% budget)
    if time.time() < t0 + budget_s - 2:
        remain = t0 + budget_s - 2 - time.time()
        ntr = int(remain / 0.004)
        ntr = min(ntr, 150000)
        best_h, best_c5 = _local_refine(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
