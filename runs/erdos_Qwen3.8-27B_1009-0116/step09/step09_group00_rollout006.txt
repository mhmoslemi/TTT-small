import numpy as np

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    n = x.size
    x += (S - x.sum()) / n
    for _ in range(20):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13: break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0: break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += 1.0 - h[k:]
        g[k:] += -h[:n-k]
    else:
        kp = -k
        g[kp:] += 1.0 - h[:n-kp]
        g[:n-kp] += -h[kp:]
    return g

def _active_grad(h, tol_frac):
    corr = _corrs(h)
    mx = float(corr.max())
    active = np.where(corr >= mx - tol_frac * mx)[0]
    g = np.zeros(len(h))
    n = len(h)
    for idx in active:
        g += _grad_lag(h, int(idx) - (n - 1))
    return g / len(active), mx, active, corr

def _sa_informed(h0, rng, deadline):
    n = len(h0)

    mx = float(corr.max())

    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    while time.time() < deadline:

            d = min(d, avail)
            if d < 1e-12: continue
            h[i] += s * d; h[j] -= s * d
            c2 = _corrs(h)
            mx2 = float(c2.max())
            delta = mx2 - mx

                if cn < best_c - 1e-14: best_c = cn; best_h = h.copy()
            else:
                h[i] -= s * d; h[j] += s * d

            d = (rng.random() - 0.5) * 0.04

def _coord_descent(h0, deadline):
    """Coordinate descent focusing on active lags"""
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    best_c = _c5(h)
    best_h = h.copy()
    t0 = time.time()
    while time.time() < deadline:
        corr = _corrs(h)
        mx = float(corr.max())
        c_cur = mx * (2.0 / n)
        if c_cur < best_c - 1e-14:
            best_c = c_cur; best_h = h.copy()
        active = np.where(corr >= mx - 0.02 * mx)[0]
        # For each active lag, do a few coordinate updates
        rng_local = np.random.RandomState(int(time.time() * 1000) % 100000)
        for idx in active[:8]:
            k = int(idx) - (n - 1)
            g = _grad_lag(h, k)
            # Find best coordinate to update
            # Try to decrease the most positive gradient component
            order = np.argsort(g)
            for _ in range(3):
                i = int(order[rng_local.randint(len(order))])
                j = int(order[-1 - rng_local.randint(len(order))])
                if i == j: continue
                s = 1 if g[i] < g[j] else -1
                avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
                if avail < 1e-14: continue
                for mult in [0.3, 0.1, 0.05, 0.02]:
                    d = s * mult * avail
                    h[i] += d; h[j] -= d
                    cn = _c5(h)
                    if cn < c_cur - 1e-14:
                        c_cur = cn
                        break
                    else:
                        h[i] -= d; h[j] += d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if m == n: return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)
