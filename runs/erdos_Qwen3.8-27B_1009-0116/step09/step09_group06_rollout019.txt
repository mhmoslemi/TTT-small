            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - r[idx])
                r[idx] += a
                diff -= a
            else:
                a = min(-diff, r[idx])
                r[idx] -= a
                diff += a
    return r

def _soft_grad(h, alpha, m, n):
    dx = 2.0 / n
    c = _correl(h)
    idx = np.argsort(c)[-m:]
    cmax = c.max()
    s = alpha * (c[idx] - cmax)
    s = s - s.max()
    s = np.clip(s, -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(m):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        k = idx[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _nesterov_pg(h0, alpha_list, n):
    """Nesterov accelerated projected gradient"""
    h = h0.copy()
    y = h.copy()
    best_h = h.copy()
    best_c = _eval(h)
    t = 1
    for alpha in alpha_list:
        m = min(80, n // 7)
        for _ in range(150):
            t1 = (1 + np.sqrt(1 + 4.0 * t * t)) / 2.0
            theta = (t - 1.0) / t1
            y = h + theta * (y - h)  # Wait, Nesterov: y = h + (t-1)/t1 * (h - y_old)
            # Actually standard Nesterov: y_new = h + (t-1)/t1 * (h - y)
            # Then step: h_new = proj(y_new - step * grad(y_new))
            # Then t = t1
            g = _soft_grad(y, alpha, m, n)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-12:
                break
            step = 1.0 / (alpha * gnorm) * 0.1  # adaptive step
            # Backtracking
            for _bt in range(10):
                h_new = _proj(y - step * g, n * 0.5)
                c_new = _eval(h_new)
                if c_new < best_c - 1e-15 or c_new < _eval(y) - 1e-15:
                    break
                step *= 0.4
            h = h_new
            c_new = _eval(h)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            t = t1
    return _repair(best_h, n), best_c

Let me also reconsider the Nesterov method. The standard Nesterov acceleration is:
- y_0 = x_0
- For t = 1, 2, ...:
  - t_{k+1} = (1 + sqrt(1 + 4t_k²))/2
  - y = x + (t_k - 1)/t_{k+1} * (x - y)  [extrapolation]
  - x = proj(y - step * grad(y))

But for projected methods, Nesterov acceleration is trickier because the projection is non-smooth. The FISTA variant is:
- y = x + (t_k - 1)/(t_k + 1) * (x - y)
- z = y - (1/L) * grad(y)
- x = proj(z)
- t_{k+1} = (1 + sqrt(1 + 4t_k²))/2

where L is the Lipschitz constant of the gradient. For our problem, L depends on the alpha and the problem structure.

Actually, for practical purposes, let me just use a simple momentum method (heavy ball) which is easier to implement with projections:
- v = beta * v + (1-beta) * g
- x = proj(x - step * v)

    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _correl(h):

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
