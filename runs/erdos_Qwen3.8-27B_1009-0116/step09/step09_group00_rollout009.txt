    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full')
    mx = float(corr.max())

            g[kk:] += 1.0 - h[:n-kk]
            g[:n-kk] -= h[kk:]
