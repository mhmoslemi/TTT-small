import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    for _ in range(60):
        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-13:
            break
    return x

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _fft_next(n):
    N = 1
    while N < n:
        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_next(2 * n - 1)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    return P, Nv

def _soft_obj(h, a):
    P, Nv = _pn(h)
    n = len(h)
    dx = 2.0 / n
    mx = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
    e = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400, 0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a), P, Nv, mx

def _grad_soft(h, a, P, Nv, mx):
    n = len(h)
    dx = 2.0 / n
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
    en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
    tot = e0 + ep.sum() + en.sum()
    if tot <= 0:
        tot = 1.0; e0 = 1.0
    wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    u = np.zeros(n); u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cp[1:] = np.cumsum(wP[1:])
    cn = np.zeros(n); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    Nf = _fft_next(2 * n - 1)
    Hf = np.fft.rfft(h, Nf)
    Rf = np.fft.irfft(Hf * np.fft.rfft(u[::-1], Nf), Nf)
    Lf = np.fft.irfft(Hf * np.fft.rfft(u, Nf), Nf)
    q = -(Rf[n - 1:2 * n - 1].real + Lf[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _fista(h0, alpha, max_t):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fx, P, Nv, mx = _soft_obj(h, alpha)
    best_h = h.copy(); best_c = _c5(h)
    stale = 0; Lc = 10.0
    for it in range(80000):
        if time.time() - t0 > max_t:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum() - S) > 0.05:
                y = _proj(y, S)
            fy, Py, Ny, mxy = _soft_obj(y, alpha)
        else:
            y = h; fy = fx; Py, Ny, mxy = P, Nv, mx
        g = _grad_soft(y, alpha, Py, Ny, mxy)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fxn, _, _, _ = _soft_obj(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 / step * float(np.dot(dv, dv))
            if fxn <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fx = fxn; stale = 0
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1
            if stale > 20:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _sa_targeted(h0, rng, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    total_t = max(1.0, deadline - t0)
    P, Nv = _pn(h)
    max_all = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
    refresh_cnt = 0
    while time.time() < deadline:
        refresh_cnt += 1
        if refresh_cnt % 5000 == 0:
            P, Nv = _pn(h)
            max_all = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
        frac = min(1.0, (time.time() - t0) / total_t)
        T = 0.008 * (1e-4) ** frac
        tol = max(1e-9, 0.08 * dx)
        wP = np.zeros(n); wN = np.zeros(n)
        if P[0] > max_all - tol:
            wP[0] = 1.0
        if n > 1:
            mp = P[1:] > max_all - tol
            if np.any(mp):
                wP[1:][mp] = 1.0
            mn = Nv[1:] > max_all - tol
            if np.any(mn):
                wN[1:][mn] = 1.0
        cnt = wP.sum() + wN.sum()
        if cnt > 0:
            wP /= cnt; wN /= cnt
        if rng.random() < 0.7:
            # Targeted: pick lag from active set
            active = np.concatenate(([0 if wP[0] > 0 else -1],
                                     np.where(wP[1:] > 0)[0] + 1,
                                     np.where(wN[1:] > 0)[0] + 1))
            active = active[active >= 0]
            if len(active) == 0:
                active = np.array([int(rng.randint(1, min(50, n)))])
            k = int(active[rng.randint(len(active))])
            if k == 0:
                k = int(rng.randint(1, min(40, n)))
            if k >= n:
                k = n - 1
            # Find position i to swap with i+k
            if k_act_pos := (n - k):
                w = np.zeros(k_act_pos)
                # Contribute to autocorr: want to increase h[i]*h[i+k]
                # d(product)/dh[i] = h[i+k], so increasing h where h[i+k] is high helps
                w = h[k:n] + h[0:n - k]  # proxy for impact
                if w.sum() > 0:
                    w /= w.sum()
                    i = int(rng.choice(n - k, p=w))
                else:
                    i = int(rng.randint(n - k))
            else:
                i = int(rng.randint(max(1, n - 1)))
            if i + k < n:
                j = i + k
            elif i >= k:
                j = i - k
                k_use = k
            else:
                j = int(rng.randint(n))
                if j == i:
                    j = (j + 1) % n
                k_use = abs(j - i) if j != i else 1
            # Determine direction: move to increase product h[i]*h[j]
            if h[i] > h[j]:
                d = min(0.01 + 0.03 * rng.random(), h[i] - h[j], 1.0 - h[j])
                if d > 1e-9:
                    h[i] -= d; h[j] += d
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    mx_new = float(new_corr.max())
                    delta_c = (mx_new - mx) * dx
                    if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-15)):
                        mx = mx_new; corr = new_corr
                        cn = mx * dx
                        if cn < best_c:
                            best_c = cn; best_h = h.copy()
                    else:
                        h[i] += d; h[j] -= d
            elif h[j] > h[i]:
                d = min(0.01 + 0.03 * rng.random(), h[j] - h[i], 1.0 - h[i])
                if d > 1e-9:
                    h[j] -= d; h[i] += d
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    mx_new = float(new_corr.max())
                    delta_c = (mx_new - mx) * dx
                    if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-15)):
                        mx = mx_new; corr = new_corr
                        cn = mx * dx
                        if cn < best_c:
                            best_c = cn; best_h = h.copy()
                    else:
                        h[j] += d; h[i] -= d
        else:
            # Random exploratory move
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = (rng.random() - 0.5) * 0.05
            if h[i] + d < 0: d = -h[i]
            if h[i] + d > 1: d = 1.0 - h[i]
            if h[j] - d < 0: d = h[j]
            if h[j] - d > 1: d = h[j] - 1.0
            if abs(d) < 1e-10:
                continue
            h[i] += d; h[j] -= d
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            mx_new = float(new_corr.max())
            delta_c = (mx_new - mx) * dx
            if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-15)):
                mx = mx_new; corr = new_corr
                cn = mx * dx
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
    return best_h, best_c

def _grad_polish(h0, rng, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    c = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    P, Nv = _pn(h)
    max_all = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 0.06 * dx)
    wP = np.zeros(n); wN = np.zeros(n)
    if P[0] > max_all - tol: wP[0] = 1.0
    if n > 1:
        mp = P[1:] > max_all - tol
        if np.any(mp): wP[1:][mp] = 1.0
        mn = Nv[1:] > max_all - tol
        if np.any(mn): wN[1:][mn] = 1.0
    cnt = wP.sum() + wN.sum()
    if cnt > 0: wP /= cnt; wN /= cnt
    a = 500.0
    g = _grad_soft(h, a, P, Nv, max_all)
    base_lr = 0.005
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        lr = base_lr * (0.1) ** frac
        # Pick a pair along gradient
        idx_sorted = np.argsort(g)
        i = int(idx_sorted[rng.randint(min(30, n))])
        j = int(idx_sorted[-(rng.randint(min(30, n)) + 1)])
        if i == j:
            continue
        s = 1.0 if g[i] < g[j] else -1.0
        avail = min(1.0 - h[i] if s > 0 else h[i], h[j] if s > 0 else 1.0 - h[j])
        if avail < 1e-12:
            continue
        d = s * min(avail * 0.8, lr * (0.5 + rng.random()))
        if abs(d) < 1e-12:
            continue
        h[i] += d; h[j] -= d
        cn = _c5(h)
        if cn < c:
            c = cn
            if c < best_c:
                best_c = c; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
        if rng.random() < 0.005:
            P, Nv = _pn(h)
            max_all = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
            g = _grad_soft(h, a, P, Nv, max_all)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    # Try symmetric variants
    for hh in (1.0 - h0.copy(), 0.5 + 0.5 * (h0 - h0[::-1]), np.interp(np.linspace(0, 1, n0), np.linspace(0, 1, n0), h0[::-1])):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    # Phase 1: FISTA warmup (90s)
    end1 = t0 + min(90.0, 0.09 * budget_s)
    if end1 > time.time():
        hcur = best.copy()
        for a in [200, 2000]:
            if time.time() >= end1 - 1:
                break
            tt = (end1 - time.time()) / 2.0
            hf, bh, bc = _fista(hcur, a, tt)
            cf = _c5(hf)
            if bc < best_c:
                best = bh; best_c = bc
            if cf < best_c:
                best = hf; best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 2: Coarse resolution exploration (80s)
    end2 = t0 + min(170.0, 0.17 * budget_s)
    if end2 > time.time():
        for m in [140, 280]:
            if time.time() >= end2 - 5:
                break
            slot = (end2 - time.time()) / 2.0 - 3
            if slot < 5:
                break
            hm = _resample(best, m)
            hf, bh, bc = _fista(hm, 5000, slot)
            cf = _c5(hf)
            for cand in (hf, bh):
                hc = _resample(cand, n0)
                cc = _c5(hc)
                if cc < best_c:
                    best = hc; best_c = cc
    # Phase 3: Long targeted SA (600s)
    end3 = t0 + min(800.0, 0.80 * budget_s)
    if end3 > time.time():
        best, best_c = _sa_targeted(best, rng, end3)
    # Phase 4: Gradient polish (120s)
    end4 = t0 + budget_s - 60.0
    if end4 > time.time():
        best, best_c = _grad_polish(best, rng, end4)
    # Phase 5: Final SA polish (40s)
    end5 = t0 + budget_s - 10.0
    if end5 > time.time():
        best, best_c = _sa_targeted(best, rng, end5)
    # Final variants
    for hh in (1.0 - best, best[::-1]):
        h2 = _proj(hh.copy(), len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
