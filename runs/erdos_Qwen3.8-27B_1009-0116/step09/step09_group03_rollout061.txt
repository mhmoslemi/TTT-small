from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 1e-15:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            elif diff < -1e-15:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_c(h, k, n):
    """Gradient of c_k with respect to h, for a single lag k (0 to 2n-2)."""
    dx = 2.0 / n
    L = k - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        g[:n-L] += (1.0 - h[L:n])
        g[L:] += -h[:n-L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n-M])
        g[:n-M] += -h[M:]
    return g * dx

def _grad_softmax(h, alpha, top_m):
    """Smoothed gradient of max(c_k)."""
    n = len(h)
    c = _correl(h)
    m_val = c.max()
    idx = np.argpartition(-c, top_m - 1)[:top_m]
    s = alpha * (c[idx] - m_val)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= wsum
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        g += w[ii] * _grad_c(h, k, n)
    return g

def _softmax_val(c, alpha, top_m):
    m_val = c.max()
    idx = np.argpartition(-c, top_m - 1)[:top_m]
    s = alpha * (c[idx] - m_val)
    s -= s.max()
    return m_val + np.log(np.exp(s).sum()) / alpha

def _fista(h_init, alpha, top_m, steps, lr_init, budget_check=None):

    x = h_init.copy()
    y = h_init.copy()
    t_mom = 1.0
    best_c = _eval(x)
    best_h = x.copy()
    lr = lr_init
    
    for it in range(steps):
        if budget_check is not None and budget_check():
            break
        g = _grad_softmax(y, alpha, top_m)
        
        # Backtracking
        step = lr
        c_y = _eval(y)

            c_new = _eval(x_new)

        y = _proj(y, target)

def _slsqp(h0, alpha, top_m, maxiter, ftol):
    n = len(h0)
    target = n * 0.5
    
    def obj(x):
        return _softmax_val(_correl(x), alpha, top_m)
    
    def grad(x):
        return _grad_softmax(x, alpha, top_m)
    
    def sum_fn(x):
        return x.sum() - target
    
    def sum_jac(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        return _repair(res.x, n)
    except:
        return h0

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(axis=1).copy(), m)

    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n * f)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 1e-15:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            elif diff < -1e-15:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_single(h, k, n):
    dx = 2.0 / n
    L = k - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        g[:n-L] += (1.0 - h[L:n])
        g[L:] += -h[:n-L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n-M])
        g[:n-M] += -h[M:]
    return g * dx

def _grad_sm(h, alpha, top_m):
    n = len(h)
    c = _correl(h)
    mv = c.max()
    idx = np.argpartition(-c, top_m - 1)[:top_m]
    s = alpha * (c[idx] - mv)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= ws
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_single(h, idx[ii], n)
    return g

def _fista(h0, alpha, top_m, steps, lr0, deadline):
    n = len(h0)
    tgt = n * 0.5
    x = h0.copy()
    y = h0.copy()
    tm = 1.0
    lr = lr0
    best_c = _eval(x)
    best_h = x.copy()
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_sm(y, alpha, top_m)
        c_y = _eval(y)
        step = lr
        for _ in range(25):
            x_new = _proj(y - step * g, tgt)
            if _eval(x_new) < c_y:
                break
            step *= 0.5
        if step < 1e-12:

        tn = (1.0 + np.sqrt(1.0 + 4.0 * tm * tm)) / 2.0
        y = x_new + (tm / tn - 1.0) * (x_new - x)
        y = _proj(y, tgt)
        x = x_new
        tm = tn
        lr = max(step * 1.06, lr0 * 0.4)
        cx = _eval(x)
        if cx < best_c:

            best_h = x.copy()
    return best_h, best_c

def _slsqp(h0, alpha, top_m, maxiter, ftol, deadline):
    n = len(h0)
    tgt = n * 0.5
    def obj(x):
        c = _correl(x)
        mv = c.max()
        idx = np.argpartition(-c, top_m - 1)[:top_m]
        s = alpha * (c[idx] - mv); s -= s.max()
        return mv + np.log(np.exp(s).sum()) / alpha
    def gr(x):
        return _grad_sm(x, alpha, top_m)
    def sf(x):
        return x.sum() - tgt
    def sj(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sf, 'jac': sj}]
    try:
        res = minimize(obj, h0, jac=gr, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        if res.x is not None and time.time() < deadline:
            return _repair(res.x, n)
    except Exception:
        pass
    return h0

def _downsample(h, f):
    n = len(h); m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h); out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n * f)
