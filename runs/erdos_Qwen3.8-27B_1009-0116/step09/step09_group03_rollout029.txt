    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sg(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    if ml >= c.size:
        idx = np.arange(c.size)
    else:
        idx = np.argpartition(-c, ml - 1)[:ml]
    sc = np.clip(alpha * (c[idx] - m), -400, 400)
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    obj = m + np.log(w.sum()) / alpha
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * 2.0 / n
        L = k - (n - 1)
        M = abs(L)
        if M == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            g[:n - M] -= wk * x[M:]
            g[M:] += wk * (1.0 - x[:n - M])
    return obj, g

def _adam_pg(h, alpha, ml, steps, lr0, b1, b2, eps):
    n = len(h)
    mm = np.zeros(n)
    vv = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    for t in range(1, steps + 1):
        obj, g = _sg(h, alpha, ml)
        bt1 = 1.0 - (1.0 - b1) ** t
        bt2 = 1.0 - (1.0 - b2) ** t
        mm = b1 * mm + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        lr = lr0 / (1.0 + 0.1 * np.log1p(t / 100.0))
        h_new = _proj(h - lr * (mm / bt1) / (np.sqrt(vv / bt2) + eps), n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= obj + 1e-12:
            h = h_new
        if t % 300 == 0:
            mm *= 0.3
            vv *= 0.3
    return best_h

def _obj_slsqp(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argpartition(-c, ml - 1)[:ml]
    sc = np.clip(alpha * (c[idx] - m), -400, 400)
    sc -= sc.max()
    return m + np.log(np.exp(sc).sum()) / alpha

def _grad_slsqp(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argpartition(-c, ml - 1)[:ml]
    sc = np.clip(alpha * (c[idx] - m), -400, 400)
    sc -= sc.max()
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * 2.0 / n
        L = k - (n - 1)
        M = abs(L)
        if M == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            g[:n - M] -= wk * x[M:]
            g[M:] += wk * (1.0 - x[:n - M])
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_slsqp, h, jac=_grad_slsqp, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _lag_search(h, n_iter, rng, ml):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    for _ in range(n_iter):
        c = _corr(best_h)
        top = np.argsort(-c)[:ml]
        for tk in top[:5]:
            L = tk - (n - 1)
            M = abs(L)
            if M == 0:
                contrib = best_h * (1.0 - best_h)
            else:
                a = np.arange(n - M)
                contrib = best_h[a + M] * (1.0 - best_h[a])
            tc_top = np.argsort(-contrib)[:3]
            for tc in tc_top:
                d = 3e-5 * rng.randn()
                hn = best_h.copy()
                if M == 0:
                    hn[tc] += d
                    j2 = (tc + 1 + rng.randint(0, n - 1)) % n
                    hn[j2] -= d
                else:
                    i1 = tc
                    i2 = tc + M
                    if d > 0:
                        hn[i2] -= d
                        hn[i1] += d
                    else:
                        hn[i1] -= d
                        hn[i2] += d
                if hn.min() < -1e-10 or hn.max() > 1 + 1e-10:
                    continue
                hn = _repair(hn, n)
                c5n = _eval(hn)
                if c5n < best_c:
                    best_h = hn
                    best_c = c5n
                    break
            if best_c < _eval(h):
                break
    return best_h, best_c

def _slsqp_wrapper(x, alpha, ml):
    return _obj_slsqp(x, alpha, ml)

def _grad_wrapper(x, alpha, ml):
    return _grad_slsqp(x, alpha, ml)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Adam PG with progressive alpha
    alphas = [5000, 20000, 80000, 300000, 1200000, 5000000, 20000000, 80000000]
    for alpha in alphas:
        if _el() > budget_s - 380:
            break
        ml = min(120, max(25, int(15 * np.log2(alpha / 5000.0) + 25)))
        steps = max(150, min(600, int((budget_s - 380 - _el()) / 0.015)))
        lr0 = 0.004 / (1.0 + 0.3 * np.log10(alpha / 5000.0 + 1))
        h_new = _adam_pg(best_h.copy(), alpha, ml, steps, lr0, 0.9, 0.999, 1e-8)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Lag-targeted perturbation
    if _el() < budget_s - 280:
        nt = min(int((budget_s - 280 - _el()) / 0.003), 400000)
        best_h, best_c5 = _lag_search(best_h, nt, rng, 40)

    # Phase 3: SLSQP polish with high alpha
    if _el() < budget_s - 160:
        for alpha in [50000000, 200000000, 1000000000]:
            if _el() > budget_s - 120:
                break
            ml = min(200, max(50, int(30 * np.log10(alpha / 5e7 + 1) + 50)))
            h_new = _slsqp(best_h, alpha, ml, 600, 1e-15)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Reheat and Adam re-optimization
    if _el() < budget_s - 80:
        for att in range(4):
            if _el() > budget_s - 60:
                break
            ns = 8e-5 * (4 - att)
            hp = best_h + ns * rng.randn(n)
            hp = _repair(hp, n)
            h_new = _adam_pg(hp, 30000000, 80, 250, 0.003, 0.9, 0.999, 1e-8)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        n_it = min(int((budget_s - 2 - _el()) / 0.0005), 1000000)
        for _ in range(n_it):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 16)) % nc2
            d = 5e-8 * rng.randn()
            h2[i] += d
            h2[j] -= d
            if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                h2[i] -= d
                h2[j] += d
                continue
            c5n = _eval(h2)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h2.copy()
            else:
                h2[i] -= d
                h2[j] += d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
