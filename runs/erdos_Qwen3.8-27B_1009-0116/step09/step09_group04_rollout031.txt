            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 5: Multi-start diversification (limited)
    if _el() < budget_s - 100:
        bl = min(budget_s - 100 - _el(), 60)
        best_h, best_c5 = _multistart_polish(best_h, bl, rng)

    # Phase 6: Fine 2-point hill climbing
    if _el() < budget_s - 30:
        nt = min(int((budget_s - 30 - _el()) * 50), 300000)
        best_h, best_c5 = _hill(best_h, nt, rng, 5e-7, 6)

    # Phase 7: Ultra-fine polish
    if _el() < budget_s - 3:
        npol = min(int((budget_s - 3 - _el()) * 800), 1000000)
        best_h, best_c5 = _hill(best_h, npol, rng, 2e-8, 4)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
