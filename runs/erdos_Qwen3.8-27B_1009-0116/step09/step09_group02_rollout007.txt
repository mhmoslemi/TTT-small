- This gives a linear objective for each lag's correlation
- Solve the resulting convex problem (minimize max of linear functions, subject to constraints)

def solve_qp(h_curr, n, dx):

    t = cp.Variable()
    target = n / 2.0
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]

    for L in range(-n+1, n):

        # c_L = sum_i h[i] * (1 - h[i+L]) * dx (where indices are valid)
        # Linearized: h[i] * (1 - h_curr[i+L]) + (1 - h[i]) * ...

        # ≈ h_curr[i]*h[i+L] + h[i]*h_curr[i+L] - h_curr[i]*h_curr[i+L]

    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-12:
            break
        ratio = target / s if s > 1e-15 else 1.0
        h *= ratio
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - tgt) < 1e-12:
            break
        if s < 1e-14:
            h[:] = 0.5
            break
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = tgt - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-13:
                break
            amt = min(abs(diff), 1.0 - h[i]) if diff > 0 else min(-diff, h[i])
            h[i] += amt if diff > 0 else -amt
            diff -= amt if diff > 0 else -amt
    return h

    t = cp.Variable()
