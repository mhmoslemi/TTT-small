        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Gradient-informed search
    dl_p3 = t0 + budget_s * 0.72
    rem3 = max(1, dl_p3 - time.time())
    n_it3 = int(rem3 * 400)
    best_h, best_c5 = _grad_search(best_h, rng, dl_p3, n_it3, 8)

    # Phase 4: Block search
    dl_p4 = t0 + budget_s * 0.85
    rem4 = max(1, dl_p4 - time.time())
    n_it4 = int(rem4 * 300)
    best_h, best_c5 = _block_search(best_h, n_it4, rng, dl_p4, 5)

    # Phase 5: Pair polish (medium)
    dl_p5 = t0 + budget_s * 0.94
    rem5 = max(1, dl_p5 - time.time())
    n_it5 = int(rem5 * 800)
    best_h, best_c5 = _pair_polish(best_h, n_it5, rng, dl_p5)

    # Phase 6: Ultra-fine polish
    dl_p6 = t0 + budget_s * 0.995
    rem6 = max(1, dl_p6 - time.time())
    n_it6 = int(rem6 * 1000)
    best_h, best_c5 = _pair_polish(best_h, n_it6, rng, dl_p6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
