import numpy as np
import time
from scipy.optimize import minimize as _scipy_min

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
        h *= S / (s + 1e-30)
        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-15:
                break
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _proj_box_sum(v, S):
    x = v.copy()
    for _ in range(60):
        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-14:
            break
    d = S - x.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - x) if d > 0 else np.argsort(x)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - x[i]); x[i] += a; d -= a
            else:
                a = min(-d, x[i]); x[i] -= a; d += a
            if abs(d) < 1e-15:
                break
    return x

def _soft_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = corr.max()
    e = np.exp(np.clip(alpha * (corr - mx), -400, 0))
    return mx + np.log(e.sum()) / alpha

def _soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = corr.max()
    e = np.exp(np.clip(alpha * (corr - mx), -400, 0))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if abs(wk) < 1e-15:
            continue
        if k == 0:
            g += wk * (1.0 - 2.0 * h)
        else:
            g[:n - k] += wk * (1.0 - h)
            g[k:] -= wk * h
    return g * dx

def _fista(h0, alpha, max_t, L0=10.0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0.copy(), n)
    hp = h.copy(); y = h.copy(); t = 1.0
    best_h = h.copy(); best_c = _c5(h)
    fy = _soft_max_val(h, alpha)
    Lc = L0
    for it in range(200000):
        if time.time() - t0 > max_t:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            y = _proj_box_sum(y, S)
            fy = _soft_max_val(y, alpha)
        g = _soft_max_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-10)
        ok = False
        for _ in range(14):
            xn = _proj_box_sum(y - step * g, S)
            fx = _soft_max_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(g @ dv) + 0.5 / step * float(dv @ dv)
            if fx <= rhs + 1e-10:
                ok = True; break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-10)
        if ok:
            hp = h; h = xn; Lc = max(Lc * 0.8, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            hp = h; y = h; t = 1.0
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _lfbgssolve(h0, alpha, penalty, max_t):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    x0 = _repair(h0.copy(), n)
    def obj_and_grad(x):
        c = _soft_max_val(x, alpha)
        g = _soft_max_grad(x, alpha)
        s = x.sum()
        c += penalty * (s - S) ** 2
        g = g + 2.0 * penalty * (s - S)
        return c, g
    bounds = [(0.0, 1.0)] * n
    res = _scipy_min(obj_and_grad, x0, jac=True, method='L-BFGS-B',
                     bounds=bounds, maxiter=500, maxfun=2000,
                     options={'maxls': 30, 'ftol': 1e-14, 'gtol': 1e-10,
                              'iprint': -1})
    if time.time() - t0 > max_t:
        return x0, _c5(x0)
    x_out = _repair(np.asarray(res.x), n)
    c_out = _c5(x_out)
    return x_out, c_out

def _pairwise_refine(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    mx = float(corr.max())
    best_c = mx * (2.0 / n); best_h = h.copy()
    t0 = time.time()
    total_t = max(1.0, deadline - t0)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total_t)
        T = 1.0 * (1e-6) ** frac
        # Find top-3 active lags
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:3]
        # Pick a random active lag
        ai = active[rng.randint(len(active))]
        k = int(abs(ai - (n - 1)))
        if k == 0:
            i = rng.randint(n)
            j = (i + 1) % n
        else:
            valid_i = np.where((h[:n-k] > 0.01) & (h[k:] < 0.99))[0]
            if len(valid_i) < 2:
                i = rng.randint(n); j = (i + k) % n
            else:
                i = int(valid_i[rng.randint(len(valid_i))])
            j = i + k
        # Gradient-guided direction
        # For constraint k: decrease h at position where it increases overlap
        if k > 0 and k < n:
            gi = (1.0 - h[i+k]) - (h[i-k] if i >= k else 0.0)
            gj = (1.0 - h[j+k]) - (h[j-k] if j >= k else 0.0)
        else:
            gi = 1.0 - 2.0 * h[i]
            gj = 1.0 - 2.0 * h[j]
        sgn = 1.0 if gi > gj else (-1.0 if gi < gj else (1.0 if rng.random() < 0.5 else -1.0))
        base = max(1e-5, 2e-3 * (1e-8) ** frac)
        d = sgn * base * (0.5 + rng.random())
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        d = max(-avail, min(avail, d))
        if abs(d) < 1e-12:
            continue
        h[i] += d; h[j] -= d
        c2 = np.correlate(h, 1.0 - h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx
        if delta < -1e-15 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-16))):
            mx = mx2; corr = c2
            cn = mx * (2.0 / n)
            if cn < best_c - 1e-15:
                best_c = cn; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj_box_sum(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():

    # Phase 1: FISTA warm-up (150s)
