def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _rep(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    for _ in range(60):
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
        h *= t / (s + 1e-30)
        h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _bin(h, n):
    b = (h > 0.5).astype(np.float64)
    t = n // 2
    s = int(b.sum())
    if s == t:
        return b
    if s > t:
        ones = np.where(b == 1)[0]
        idx = np.argsort(h[ones])[:s - t]
        b[ones[idx]] = 0.0
    else:
        zeros = np.where(b == 0)[0]
        idx = np.argsort(-h[zeros])[:t - s]
        b[zeros[idx]] = 1.0
    return b

def _sobj(x, a, ml):

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = a * (c[idx] - m); s -= s.max()
    return m + np.log(np.exp(s).sum()) / a

def _sgrad(x, a, ml):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = a * (c[idx] - m); s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):

            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, tn):
    return x.sum() - tn

def _sj(x, tn):
    return np.ones(len(x))

def _slsqp(h, a, ml, mi):
    n = len(h)
    try:

                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}],
                     options={'maxiter': mi, 'ftol': 1e-16})
        return _rep(r.x, n)
    except:
        return h
