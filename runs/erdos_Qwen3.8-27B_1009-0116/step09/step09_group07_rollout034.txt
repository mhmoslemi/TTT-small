import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-14:
            break
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h); c = _corr(h); cm = 2*n - 1
    jl = np.arange(n); rng.shuffle(jl); nm1 = n - 1
    for j in jl:
        j = int(j); beta = np.zeros(cm)
        lo, hi = j, min(cm-1, j+nm1)
        L1 = np.arange(lo, hi+1) - nm1
        beta[lo:hi+1] += 1.0 - h[j - L1]
        lo, hi = max(0, nm1-j), min(cm-1, cm-1-j)
        L2 = np.arange(lo, hi+1) - nm1
        beta[lo:hi+1] -= h[j + L2]
        g = c - beta * h[j]
        a, b = 0.0, 1.0
        for _ in range(35):
            t1 = a + (b-a)*0.333; t2 = b - (b-a)*0.333
            if np.max(g + beta*t1) <= np.max(g + beta*t2): b = t2
            else: a = t1
        tn = (a+b)*0.5
        if abs(tn - h[j]) > 1e-14:
            c += beta*(tn - h[j]); h[j] = tn
    return h

    n = len(x); c = np.correlate(x, 1-x, mode='full')*(2.0/n)
    m = c.max(); s = np.clip(alpha*(c-m), -500, 500); s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha):
    n = len(x); dx = 2.0/n; nm1 = n-1
    c = np.correlate(x, 1-x, mode='full')*dx
    m = c.max(); s = np.clip(alpha*(c-m), -500, 500); s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for k in np.where(w > 1e-12)[0]:
        wk = w[k]*dx; L = int(k) - nm1
        if L >= 0:
            g[:n-L] += wk*(1-x[L:]); g[L:] -= wk*x[:n-L]
        else:
            M = -L; g[M:] += wk*(1-x[:n-M]); g[:n-M] -= wk*x[M:]
    return g

def _sf(x, t): return x.sum() - t
def _sj(x, t): return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    try:
        r = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha,),
                     method='SLSQP', bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':_sf,'jac':_sj,'args':(n*0.5,)}],
                     options={'maxiter':maxiter,'ftol':1e-16})
        return _repair(r.x, n)
    except:
        return h

def _smooth(h, sig):
    n = len(h); k = max(3, int(sig*6)); k |= 1
    w = np.exp(-0.5*(np.arange(k)-k//2)**2/sig**2); w /= w.sum()
    hp = np.pad(h, k//2, mode='wrap')
    return np.convolve(hp, w, 'valid')[:n]

def _subgrad_pass(h, rng, n_iter, lr0):
    n = len(h); nm1 = n-1
    for t in range(n_iter):
        lr = lr0 / (1.0 + 0.05*t)
