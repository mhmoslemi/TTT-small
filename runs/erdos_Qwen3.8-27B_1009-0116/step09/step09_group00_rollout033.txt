def _subgrad_step(h, k_star, alpha):
    """One projected subgradient step for the active lag k_star."""
    n = len(h)
    p = k_star - n + 1
    g = np.zeros(n)
    # First term: (1 - h[j-p]) for valid j-p
    lo = max(0, -p)  # j >= -p means j-p >= 0
    hi = min(n, n - p)  # j-p < n means j < n+p
    for j in range(lo, hi):
        g[j] += (1.0 - h[j - p])
    # Second term: -h[j+p] for valid j+p
    lo2 = max(0, -p)  # j+p >= 0 means j >= -p
    hi2 = min(n, n + p)  # j+p < n
    for j in range(lo2, hi2):
        g[j] -= h[j + p]
    # Projected step
    h_new = h - alpha * g
    return h_new

I need to be smarter. Instead of recomputing the full correlation each time, I can compute the change incrementally.

For a pair (a, b) where b = a+p, the change in c[k] when moving δ from b to a is:
Δc[k] = (h[b]-δ)*(1-(h[a]+δ)) - h[b]*(1-h[a]) for the k-th term
       = (h[b]-δ)*(1-h[a]-δ) - h[b]*(1-h[a])
       = h[b]*(1-h[a]) - h[b]*δ - (1-h[a])*δ + δ² - h[b]*(1-h[a])
       = -h[b]*δ - (1-h[a])*δ + δ²
       = δ*(-h[b] - 1 + h[a]) + δ²
       = δ*(h[a] - h[b] - 1) + δ²

For other lags, the change is more complex but can be computed similarly.

            if len(active) > 1:
                g_avg = np.zeros(n)
                for k2 in active:
                    g_avg += _lag_grad(h, k2)
                g_avg /= len(active)
                ga = float(np.dot(g_avg, g_avg))
                a2 = min(0.1, max(0.005, (c - lb) / (ga + 1e-12)))
                for _ in range(12):
                    h_new = _proj(h - a2 * g_avg, n * 0.5)
                    c_new = _c5(h_new)
                    if c_new < c - 1e-14:
                        h = h_new; c = c_new
                        if c < best_c:
                            best_c = c; best_h = h.copy()
                        stall = 0
                        accepted = True
                        break
                    a2 *= 0.5
            if not accepted:
                stall += 1
                alpha *= 0.85
                if stall > 300:
                    break
    return best_h, best_c

def _pair_polish(h0, deadline):
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    rounds = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = float(corr.max())
        tol = max(1e-9, 0.008 / (2.0/n))
        active = np.where(corr >= mx - tol)[0]
        improved = False
        for k in active[:6]:
            p = k - n + 1
            if p >= 0:
                npair = n - p
                gain = h[p:] + h[:npair] - 1.0
                top = np.argsort(gain)[::-1][:80]
                for idx in top:
                    if gain[idx] < 0.005:
                        break
                    i, j = int(idx), int(idx) + p
                    max_d = min(1.0 - h[i], h[j])
                    for frac in [0.003, 0.008, 0.02, 0.04]:
                        d = min(max_d, frac)
                        if d < 1e-7:
                            break
                        h[i] += d; h[j] -= d
                        cn = _c5(h)
                        if cn < c - 1e-13:
                            c = cn; improved = True
                            if c < best_c:
                                best_c = c; best_h = h.copy()
                        else:
                            h[i] -= d; h[j] += d
                            break
            else:
                q = -p
                npair = n - q
                gain = h[q:] + h[:npair] - 1.0
                top = np.argsort(gain)[::-1][:80]
                for idx in top:
                    if gain[idx] < 0.005:
                        break
                    i, j = int(idx) + q, int(idx)
                    max_d = min(1.0 - h[j], h[i])
                    for frac in [0.003, 0.008, 0.02, 0.04]:
                        d = min(max_d, frac)
                        if d < 1e-7:
                            break
                        h[i] -= d; h[j] += d
                        cn = _c5(h)
                        if cn < c - 1e-13:
                            c = cn; improved = True
                            if c < best_c:
                                best_c = c; best_h = h.copy()
                        else:
                            h[i] += d; h[j] -= d
                            break
        if not improved:
            break
        rounds += 1
        if rounds > 5000:
            break
    return best_h, best_c

def _fista_phase(h0, deadline):
    n = len(h0)
    dx = 2.0 / n
    S = n * 0.5
    best_h = h0.copy()
    best_c = _c5(h0)
    a_list = [40, 150, 600, 2500, 10000]
    hcur = h0.copy()
    for a in a_list:
        if time.time() >= deadline - 3:
            break
        tt = min(40.0, (deadline - time.time()) / (a_list.index(a) + 1))
        if tt < 2:
            break
        h = _repair(hcur.copy(), n)
        hp = h.copy(); y = h.copy(); t_mom = 1.0
        fy = _soft_val(y, a, dx)
        t0f = time.time()
        for it in range(200000):
            if time.time() - t0f > tt:
                break
            if it > 0:
                tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
                y = h + ((t_mom - 1.0) / tn) * (h - hp)
                y = _proj(y, S)
                t_mom = tn
                fy = _soft_val(y, a, dx)
            g = _soft_grad(y, a, dx, n)
            step = 1.0
            accepted = False
            for _ in range(14):
                xn = _proj(y - step * g, S)
                fx = _soft_val(xn, a, dx)
                dv = xn - y
                rhs = fy + float(np.dot(g, dv)) + 0.5 * step * float(np.dot(dv, dv))
                if fx <= rhs + 1e-11:
                    accepted = True; break
                step *= 0.5
            if accepted:
                hp = h; h = xn
                if it % 20 == 0:
                    cc = _c5(h)
                    if cc < best_c:
                        best_c = cc; best_h = h.copy()
        cc = _c5(h)
        if cc < best_c:
            best_c = cc; best_h = h.copy()
        hcur = best_h.copy()
    return best_h, best_c

def _soft_val(h, a, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    sh = np.clip(a * (corr - mx), -400, 0)
    return mx + np.log(np.exp(sh).sum()) / a

def _soft_grad(h, a, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    sh = np.clip(a * (corr - mx), -400, 0)
    w = np.exp(sh); w /= w.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for k in active:
        g += w[k] * _lag_grad(h, k)
    return g

def _perturb_reopt(h0, rng, deadline):
    n = len(h0)
    best_h = h0.copy()
    best_c = _c5(h0)
    t0 = time.time()
    for trial in range(30):
        if time.time() >= deadline - 30:
            break
        h = h0.copy()
        mask = rng.random(n) < 0.3
        h[mask] += (rng.random(mask.sum()) - 0.5) * 0.15
        h = _proj(h, n * 0.5)
        t_sub = 10.0
        hf, bc = _exact_descent(h, time.time() + t_sub)
        if bc < best_c:
            best_h = hf; best_c = bc
        hf2, bc2 = _pair_polish(fif := hf, time.time() + 5)
        if bc2 < best_c:
            best_h = hf2; best_c = bc2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    h0 = _repair(h0, n)
    best = h0.copy()
    best_c = _c5(best)
    for hh in (1.0 - h0, h0[::-1], 0.5 + 0.5 * (h0 - h0[::-1])):
        h2 = _proj(hh, n * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c

    end1 = t0 + min(250.0, 0.23 * budget_s)
    if time.time() < end1:
        best, best_c = _fista_phase(best, end1)

    end2 = t0 + min(500.0, 0.46 * budget_s)
    if time.time() < end2:
        best, best_c = _exact_descent(best, end2)

    end3 = t0 + min(750.0, 0.68 * budget_s)
    if time.time() < end3:
        best, best_c = _pair_polish(best, end3)

    end4 = t0 + min(900.0, 0.82 * budget_s)
    if time.time() < end4:
        best, best_c = _perturb_reopt(best, rng, end4)

    end5 = t0 + budget_s - 30.0
    if time.time() < end5:
        best, best_c = _exact_descent(best, end5)

    end6 = t0 + budget_s - 3.0
    if time.time() < end6:
        best, best_c = _pair_polish(best, end6)

    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
