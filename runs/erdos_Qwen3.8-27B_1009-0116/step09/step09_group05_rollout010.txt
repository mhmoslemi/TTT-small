import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _slsqp_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = idx[ii] - (n - 1)
        if L >= 0:
            j = np.arange(n - L)
            g[j + L] += wk * (1.0 - x[j])
            g[j] -= wk * x[j + L]
        else:
            M = -L
            j = np.arange(M, n)
            g[j - M] += wk * (1.0 - x[j])
            g[j] -= wk * x[j - M]
    return g

def _sum_fn(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _active_set_opt(h, n_iter, K, rng):
    """Targeted active-set 2-point optimization with line search."""
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c = c5
    dx = 2.0 / n
    for it in range(n_iter):
        c = _corr(h)
        m = c.max()
        active = c > m - 0.0005
        r = np.where(active)[0]
        if len(r) == 0:
            r = np.argsort(c)[-K:]
        r = r[:K]
        g = np.zeros(n)
        for L in r:
            ml = L - (n - 1)
            if ml >= 0:
                j = np.arange(n - ml)
                g[j + ml] += (1.0 - h[j])
                g[j] -= h[j + ml]
            else:
                M = -ml
                j = np.arange(M, n)
                g[j - M] += (1.0 - h[j])
                g[j] -= h[j - M]
        g *= dx / (len(r) + 1e-15)
        gp = g - g.mean()
        i = int(np.argmax(gp))
        j2 = int(np.argmin(gp))
        if i == j2:
            continue
        spread = gp[i] - gp[j2]
        if spread < 1e-14:
            continue
        best_d = 0.0
        best_c5 = c5
        for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
            d = frac * spread * 0.005
            h[i] += d
            h[j2] -= d
            if h[i] < 0 or h[i] > 1 or h[j2] < 0 or h[j2] > 1:
                h[i] -= d
                h[j2] += d
                continue
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_d = d
            else:
                h[i] -= d
                h[j2] += d
                break
        if best_d != 0.0:
            c5 = best_c5
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
    return best_h, best_c

def _quad_2pt(h, n_iter, rng, K):

                best_h = h.copy()
        else:

            h[j2] = nj + d
    return best_h, best_c

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
