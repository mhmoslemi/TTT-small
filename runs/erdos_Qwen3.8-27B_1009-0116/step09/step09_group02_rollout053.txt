```python
def subgradient_step(h, n, lr, lags_to_use):
    # Compute all correlations
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    # Get top lags
    top_idx = np.argsort(c)[-lags_to_use:]
    # Average gradient over top lags
    g = np.zeros(n)
    for idx in top_idx:
        L = idx - (n-1)  # actual lag
        if L >= 0:
            # c[idx] = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # d/dh[i] for i in [0, n-1-L]: (1-h[i+L])
            # d/dh[i+L] for i in [0, n-1-L]: -h[i]
            g[:n-L] += (1-h[L:])
            g[L:] -= h[:n-L]
        else:
            M = -L
            # c[idx] = sum_{i=M}^{n-1} h[i]*(1-h[i-M])
            # d/dh[i] for i in [M, n-1]: (1-h[i-M])
            # d/dh[i-M] for i in [M, n-1]: -h[i]
            g[M:] += (1-h[:n-M])
            g[:n-M] -= h[M:]
    g /= len(top_idx)
    # Normalize for the integral constraint: subtract mean
    h_new = h - lr * (g - g.mean())
    return h_new
```

```python
        c5 = c5 - best_improvement

    return best_h, best_c5
```
