def _resample(h,m):

import numpy as np
import time
import math

def _proj(v, S):
    x = np.array(v, float, copy=True)
    if x.size == 0:
        return x
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    d = S - x.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - x) if d > 0 else np.argsort(x)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - x[i]); x[i] += a; d -= a
            else:
                a = min(-d, x[i]); x[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return x

def _pn(h):
    n = len(h)
    S = n * 0.5
    ac = np.correlate(h, h, 'full')[n - 1:]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    P = np.empty(n)
    N = np.empty(n)
    P[0] = S - ac[0]
    N[0] = P[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        N[idx] = S - pref[idx] - ac[idx]
    mx = float(max(P.max(), N.max()))
    return P, N, mx

def _c5(h):
    n = len(h)
    _, _, mx = _pn(h)
    return mx * (2.0 / n)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, m * 0.5)
    if n > m and n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _apply(P, N, h, i, j, d):
    n = len(h)
    if i == j or abs(d) < 1e-15:
        return
    oi = h[i]
    oj = h[j]
    P[0] += 2.0 * d * (oj - oi) - 2.0 * d * d
    N[0] = P[0]
    if i + 1 < n:
        P[1:n - i] += d * (1.0 - h[i + 1:i + n - i])
    if i > 0:
        P[1:i + 1] -= d * h[i - 1::-1]
    if j + 1 < n:
        P[1:n - j] -= d * (1.0 - h[j + 1:j + n - j])
    if j > 0:
        P[1:j + 1] += d * h[j - 1::-1]
    dd = abs(i - j)
    P[dd] += d * d
    if i > 0:
        N[1:i + 1] += d * (1.0 - h[i - 1::-1])
    if i + 1 < n:
        N[1:n - i] -= d * h[i + 1:i + n - i]
    if j > 0:
        N[1:j + 1] -= d * (1.0 - h[j - 1::-1])
    if j + 1 < n:
        N[1:n - j] += d * h[j + 1:j + n - j]
    N[dd] += d * d
    h[i] += d
    h[j] -= d

def _add_grad(G, h, n, k, pos, w):
    m = n - k
    if m <= 0:
        return
    if pos:
        G[:m] += w * (1.0 - h[k:k + m])
        G[k:] -= w * h[:m]
    else:
        G[k:] += w * (1.0 - h[:m])
        G[:m] -= w * h[k:k + m]

def _edge_refine(h, P, N, mx, rng, deadline, k, pos):
    n = len(h)
    m = n - k
    if m <= 0:
        return mx, 0
    if pos:
        diff = h[:m] - h[k:]
    else:
        diff = h[k:] - h[:m]
    mask = diff > 1e-4
    if not np.any(mask):
        return mx, 0
    idx = np.nonzero(mask)[0]
    r = min(24, idx.size)
    if idx.size > 2 * r:
        top = idx[np.argpartition(diff[idx], -r)[:r]]
        rnd = idx[rng.randint(idx.size, size=r)]
        cand = np.unique(np.concatenate((top, rnd)))
    else:
        cand = idx
    acc = 0
    for r0 in cand:
        if time.time() > deadline or acc >= 8:
            break
        if pos:
            base = h[r0] - h[r0 + k]
        else:
            base = h[r0 + k] - h[r0]
        if base <= 1e-4:
            continue
        for alpha in (1.0, 0.5, 0.25):
            d = base * alpha
            if pos:
                _apply(P, N, h, r0 + k, r0, d)
            else:
                _apply(P, N, h, r0, r0 + k, d)
            nm = float(max(P.max(), N.max()))
            if nm < mx - 1e-12:
                mx = nm
                acc += 1
                break
            else:
                if pos:
                    _apply(P, N, h, r0 + k, r0, -d)
                else:
                    _apply(P, N, h, r0, r0 + k, -d)
    return mx, acc

def _grad_refine(h, P, N, mx, rng, deadline):
    n = len(h)
    if n < 4:
        return mx, 0
    G = np.zeros(n)
    q = max(4, n // 12)
    kth = min(max(1, q), n - 1)
    acc = 0
    last = time.time()
    prep = True
    while time.time() < deadline and time.time() - last < 5.0:
        if prep:
            A = np.maximum(P, N)
            mx = float(A.max())
            tol = max(1e-10, 0.0004 * n)
            topk = min(32, n)
            idx = np.argpartition(A, -topk)[-topk:] if topk < n else np.arange(n)
            G.fill(0.0)
            for kk in idx:
                kk = int(kk)
                lo = mx - tol
                wp = P[kk] - lo if P[kk] > lo else 0.0
                wn = N[kk] - lo if N[kk] > lo else 0.0
                if wp <= 0.0 and wn <= 0.0:
                    continue
                if kk == 0:
                    G += (wp + wn) * (1.0 - 2.0 * h)
                else:
                    if wp > 0.0:
                        _add_grad(G, h, n, kk, 1, wp)
                    if wn > 0.0:
                        _add_grad(G, h, n, kk, 0, wn)
            if float(np.max(np.abs(G))) < 1e-14:
                break
            low = np.argpartition(G, kth)[:q]
            high = np.argpartition(-G, kth)[:q]
            prep = False
        chosen = False
        for _ in range(20):
            ai = int(low[rng.randint(len(low))])
            bi = int(high[rng.randint(len(high))])
            if ai == bi:
                ai = int(rng.randint(n))
            av = min(h[bi], 1.0 - h[ai])
            if av > 1e-11:
                chosen = True
                break
        if not chosen:
            break
        cap = 0.03
        d = min(av, cap * (0.3 + 1.7 * rng.random()))
        if d <= 1e-12:
            continue
        _apply(P, N, h, ai, bi, d)
        nm = float(max(P.max(), N.max()))
        if nm < mx - 1e-12:
            mx = nm
            acc += 1
            last = time.time()
            if acc % 6 == 0:
                prep = True
        else:
            _apply(P, N, h, ai, bi, -d)
    return mx, acc

def _polish(h, P, N, mx, rng, deadline):
    n = len(h)
    last = time.time()
    while time.time() < deadline and time.time() - last < 6.0:
        A = np.maximum(P, N)
        mx = float(A.max())
        topk = min(12, n)
        idx = np.argpartition(A, -topk)[-topk:] if topk < n else np.arange(n)
        acc = 0
        for kk in idx:
            if time.time() > deadline or time.time() - last > 6.0:
                break
            kk = int(kk)
            mx, a = _edge_refine(h, P, N, mx, rng, deadline, kk, 1)
            acc += a
            if time.time() > deadline or time.time() - last > 6.0:
                break
            mx, a = _edge_refine(h, P, N, mx, rng, deadline, kk, 0)
            acc += a
        mx, a = _grad_refine(h, P, N, mx, rng, deadline)
        acc += a
        if acc:
            last = time.time()
        else:
            break
    return h, P, N, mx

def _anneal(h, P, N, mx, rng, deadline):
    n = len(h)
    S = n * 0.5
    t0 = time.time()
    total = max(1e-6, deadline - t0)
    it = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total)
        T = 0.5 * n * (0.02 * math.exp(-1.8 * frac) + 1e-5)
        chosen = False
        for _ in range(12):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            if h[i] > h[j]:
                hi, lo = i, j
            else:
                hi, lo = j, i
            if rng.random() < 0.75:
                add, sub = lo, hi
            else:
                add, sub = (lo, hi) if rng.random() < 0.5 else (hi, lo)
            av = min(h[sub], 1.0 - h[add])
            if av > 1e-12:
                chosen = True
                break
        if not chosen:
            continue
        cap = 0.02 + 0.10 * math.exp(-2.0 * frac)
        d = min(av, cap * (0.2 + 1.8 * rng.random()))
        if d <= 1e-12:
            continue
        _apply(P, N, h, add, sub, d)
        nm = float(max(P.max(), N.max()))
        delta = nm - mx
        if delta <= 0.0 or (delta < T and rng.random() < math.exp(-delta / T)):
            mx = nm
        else:
            _apply(P, N, h, add, sub, -d)
        it += 1
        if it % 3000 == 0:
            h = _proj(h, S)
            P, N, mx = _pn(h)
    return h, P, N, mx

def _local_search(start, n, rng, deadline):
    h = _proj(start, n * 0.5)
    P, N, mx = _pn(h)
    best = h.copy()
    bestm = mx
    t0 = time.time()
    last = t0
    cyc = 0
    while time.time() < deadline:
        rem = deadline - time.time()
        if rem <= 0.0:
            break
        cyc += 1
        noimp = time.time() - last
        if rem < 15.0 or (cyc % 3 != 0 and noimp <= 45.0):
            typ_p = True
        else:
            typ_p = False
        span = max(1.0, deadline - t0)
        if typ_p:
            block = max(5.0, min(rem, 0.35 * span + 10.0))
        else:
            block = max(5.0, min(rem, 0.12 * span + 10.0))
        end = time.time() + block
        if typ_p:
            h, P, N, mx = _polish(h, P, N, mx, rng, end)
        else:
            h, P, N, mx = _anneal(h, P, N, mx, rng, end)
        mx = float(max(P.max(), N.max()))
        if mx < bestm - 1e-13:
            bestm = mx
            best = h.copy()
            last = time.time()
        if mx > bestm + 0.0025 * n:
            h = best.copy()
            P, N, mx = _pn(h)
        if cyc % 5 == 0:
            h2 = _proj(h, n * 0.5)

                bestm = mx2
                best = h2.copy()

def _make_start(h0, n, rng):
    cands = []
    if h0 is not None:
        r = _resample(h0, n)
        cands.append(r)
        cands.append(_proj(1.0 - r, n * 0.5))
    x = (np.arange(n) + 0.5) * (2.0 / n)
    f = rng.uniform(0.5, 5.0)
    ph = rng.uniform(0.0, 6.283)
    hs = 0.5 + 0.6 * np.sin(2.0 * np.pi * f * x + ph) + 0.25 * np.sin(4.0 * np.pi * f * x + 2.0 * ph)
    cands.append(_proj(hs, n * 0.5))
    a = np.zeros(n)
    a[:n // 2] = 1.0
    a = rng.permutation(a) + 0.2 * (2.0 * rng.rand(n) - 1.0)
    cands.append(_proj(a, n * 0.5))
    best = None
    bc = 1e100
    for hh in cands:
        _, _, mx = _pn(hh)
        c = mx * (2.0 / n)
        if c < bc:
            best = hh
            bc = c
    return best
