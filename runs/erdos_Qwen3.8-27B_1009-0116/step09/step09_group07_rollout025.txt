        j = (i + 1 + rng.randint(0, 15)) % n
        d = 8e-6 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = float(_eval_c(h).max())
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5
