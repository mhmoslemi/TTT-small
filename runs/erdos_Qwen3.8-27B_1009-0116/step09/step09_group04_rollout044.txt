import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _con_fun(x, hs):
    return x.sum() - hs

def _con_jac(x, hs):

def _proj(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad_vec(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _fine_polish(h, n_iters, rng, base_amp):
    n = len(h)
    h2 = h.copy()
    c2 = _eval(h2)
    bh, bc = h2.copy(), c2
    for _ in range(n_iters):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = base_amp * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c2:
                c2 = cn
                if cn < bc:
                    bc, bh = cn, h2.copy()
            else:
                h2[i], h2[j] = ni - d, nj + d
    return bh, bc

def _slsqp_pass(h, alpha, ml, maxit, n_hs):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n_hs,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-15})
        return _repair(r.x, n)
    except:
        return h
