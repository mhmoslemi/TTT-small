def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(100):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _all_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, m_lags):
    c = _all_corr(x)
    m = c.max()
    top = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[top] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _all_corr(x)
    m = c.max()
    top = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[top] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(top):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g - g.mean()

2. The step sizes might not be well-tuned.
3. The search is purely greedy (only accepts improvements).

My improvements:
1. **Exact coordinate descent**: Instead of random perturbations, find the exact optimal δ for each pair. This is more efficient per "step".
2. **Simulated annealing**: Allow some worsening moves to escape local minima.
3. **Multi-scale approach**: Start with large steps (structural changes) and gradually refine.
4. **Better gradient weighting**: Use a smoother weighting over all lags, not just the top M.

def _grad_matrix(h):
    """Compute G[i,k] = d(c_k)/d(h[i]) for all i, k."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((n, 2 * n - 1))
    for L in range(n):
        k = L + n - 1
        if L == 0:
            G[:, k] = (1.0 - 2.0 * h) * dx
        else:
            G[:n - L, k] += (1.0 - h[L:]) * dx
            G[L:, k] -= h[:n - L] * dx
            k2 = n - 1 - L
            G[L:, k2] += (1.0 - h[:n - L]) * dx
            G[:n - L, k2] -= h[L:] * dx
    return G

def _coord_step(h, i, j, G, c):
    """Find optimal delta for transferring from j to i."""
    d = G[i] - G[j]
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0
    # Minimize max(c + d*delta)
    # Quick: evaluate at 20 points
    deltas = np.linspace(lo, hi, 20)
    # c + d*delta: shape (20, 2n-1)
    vals = np.max(c[None, :] + d[None, :] * deltas[:, None], axis=1)
    bi = np.argmin(vals)
    if bi == 0:
        return lo
    if bi == 19:
        return hi
    # Refine
    lo_r = deltas[bi - 1]
    hi_r = deltas[bi + 1]
    ref_d = np.linspace(lo_r, hi_r, 15)
    ref_v = np.max(c[None, :] + d[None, :] * ref_d[:, None], axis=1)
    return ref_d[np.argmin(ref_v)]

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target_sum = n * 0.5
    c_fun = lambda x: x.sum() - target_sum
    c_jac = lambda x: np.ones(n)
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': c_fun, 'jac': c_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _projected_momentum(h, alpha, m_lags, steps, step0, mu=0.9):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _smooth_grad(h, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        taken = False
        for _t in range(8):
            h_new = h - step * v
            h_new = np.clip(h_new, 0, 1)
            h_new = _repair(h_new, n)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 30.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(max(step * 1.3, step0 * 0.5), step0 * 30.0)
    return h, cur_c

def _coord_descent(h, n_iter, rng, G_refresh=50):
    """Coordinate descent with exact 1D optimization."""
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    G = _grad_matrix(h)
    c = _all_corr(h)
    
    for it in range(n_iter):
        # Refresh gradient matrix periodically
        if it % G_refresh == 0:
            G = _grad_matrix(h)
            c = _all_corr(h)
        
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        
        delta = _coord_step(h, i, j, G, c)
        if abs(delta) < 1e-14:
            continue
        
        h[i] += delta
        h[j] -= delta
        h = _repair(h, n)
        c5n = _eval(h)
        
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            # Revert
            h[i] -= delta
            h[j] += delta
            h = _repair(h, n)
    
    return best_h, best_c

def _sim_anneal(h, n_iter, rng, T0, T1, step_scale):
    """Simulated annealing with gradient bias."""
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    G = _grad_matrix(h)
    
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(1, n_iter - 1))
        # Gradient-biased random move
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        
        # Step size with gradient bias
        g_diff = G[i] @ _all_corr(h) - G[j] @ _all_corr(h)  # rough gradient
        # Actually let's use the smooth gradient
        # Simpler: random step with gradient bias
        step_mag = step_scale * (0.1 + 0.9 * rng.rand())
        direction = rng.randn()
        # Bias toward gradient descent
        grad_i = _smooth_grad(h, 10000, 30)[i]
        grad_j = _smooth_grad(h, 10000, 30)[j]
        bias = -0.5 * (grad_i - grad_j)
        direction += bias / (abs(bias) + 1e-10) * 0.3
        
        delta = step_mag * direction
        if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
            continue
        
        h[i] += delta
        h[j] -= delta
        h = _repair(h, n)
        c5n = _eval(h)
        delta_c = c5n - c5cur
        
        if delta_c < 0 or rng.rand() < np.exp(-delta_c / T):
            c5cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
            h = _repair(h, n)
    
    return best_h, best_c
