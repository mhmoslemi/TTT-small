    # Phase 6: Fine 2-point polish
    if time.time() - t0 < budget_s - 8:
        rem = budget_s - 8 - (time.time() - t0)
        best_h, best_c5 = _fine_polish(best_h, int(rem * 500), rng)

    # Phase 7: Group polish
    if time.time() - t0 < budget_s - 2:
        rem = budget_s - 2 - (time.time() - t0)
        best_h, best_c5 = _group_polish(best_h, int(rem * 250), rng, 4)

    # Final fine polish
    if time.time() - t0 < budget_s - 0.3:
        rem = budget_s - 0.3 - (time.time() - t0)
        best_h, best_c5 = _fine_polish(best_h, int(rem * 700), rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
