import numpy as np
import time

def _c5_val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else a
            if abs(d) < 1e-14:
                break
    return h

def _proj_arr(h, S):
    h = h.astype(float).copy()
    for _ in range(60):
        h += (S - h.sum()) / len(h)
        np.clip(h, 0.0, 1.0, out=h)
        if abs(h.sum() - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else a
            if abs(d) < 1e-14:
                break
    return h

def _pn_corr(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(corr.max())

def _targeted_swap(h, rng, deadline):

        improved = False

def _fourier_kick(h, rng, strength=0.15):
    n = len(h)
    N = 1
    while N < n:
        N <<= 1
    H = np.fft.rfft(h, N)
    n_modes = max(2, n // 20)
    modes = rng.choice(range(1, n_modes + 1), size=min(3, n_modes), replace=False)
    for m in modes:
        phase = rng.random() * 2.0 * np.pi
        amp = strength * rng.random()
        H[m] += amp * np.exp(1j * phase)
        if m < N // 2:
            H[N - m] += amp * np.exp(-1j * phase)
    h_new = np.fft.irfft(H, N).real[:n]
    return _proj_arr(h_new, n * 0.5)

def _gradient_step(h, corr, rng, step=0.01):
    n = len(h)
    k_act = int(np.argmax(corr)) - (n - 1)
    ka = abs(k_act)
    if ka == 0 or ka >= n:
        return h
    if k_act > 0:
        for i in range(n - ka):
            g = 1.0 - 2.0 * h[i + ka]
            h[i] -= step * g * 0.5
            h[i + ka] += step * g * 0.5
    else:
        for i in range(ka):
            g = 1.0 - 2.0 * h[i - ka]
            h[i] -= step * g * 0.5
            h[i - ka] += step * g * 0.5
    return _proj_arr(h, n * 0.5)
