        # Find top candidate pairs
        i = int(np.argmax(g_c))
        j = int(np.argmin(g_c))
        if i == j:
            continue
        # Try multiple directions and steps
        d_mag = g_c[i] - g_c[j]
        if d_mag < 1e-12:

        best_c5 = cur_c
        for frac in [0.05, 0.1, 0.2, 0.5, 1.0]:

        # Random 2-point move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 2e-5 * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c_new = _eval(h_cur)
        if c_new < cur_c or rng.rand() < np.exp(min(0, -(c_new - cur_c) / T)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
    return best_h, best_c
