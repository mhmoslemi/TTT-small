    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    return P, Nv

def _grad_w(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 4 * n: Nf <<= 1
        Hf = np.fft.rfft(h, Nf)
        R = np.fft.irfft(Hf * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(Hf * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _soft_val(h, a):
    P, Nv, _ = _pn(h)
    n = len(h)
    dx = 2.0 / n
    mx = max(float(P[0]), float(np.max(P[1:])) if n > 1 else 0.0, float(np.max(Nv[1:])) if n > 1 else 0.0)
    e = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400, 0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a)

def _soft_grad(h, a):
    P, Nv, _ = _pn(h)
    n = len(h)
    mx = max(float(P[0]), float(np.max(P[1:])) if n > 1 else 0.0, float(np.max(Nv[1:])) if n > 1 else 0.0)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0: tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_w(h, wP, wN)

def _fista(h0, alpha, tmax, L0=10.0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    stale = 0; Lc = L0
    for it in range(500000):
        if time.time() - t0 > tmax: break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        else:
            y = h; fy = fh
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-11:
                accepted = True; break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.9, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c: best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 15: break
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _active_weights(P, Nv, top_k=5):
    n = len(P)
    vals = [(float(P[0]), 0, 'P')]
    if n > 1:
        vals += [(float(P[k]), k, 'P') for k in range(1, n)]
        vals += [(float(Nv[k]), k, 'N') for k in range(1, n)]
    vals.sort(key=lambda x: -x[0])
    mx = vals[0][0]
    tol = max(5e-5, 30.0 * (2.0 / n))
    sel = [v for v in vals if v[0] >= mx - tol][:max(top_k, 3)]
    wP = np.zeros(n)
    wN = np.zeros(n)
    for val, k, typ in sel:
        w = 0.5 + 0.5 * (val - (mx - tol)) / max(tol, 1e-15)
        if typ == 'P': wP[k] += w
        else: wN[k] += w
    s = wP.sum() + wN.sum()
    if s > 0:
        wP /= s; wN /= s
    return wP, wN

def _fine_local(h0, rng, deadline):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    it = 0
    step_base = 5e-5
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / max(deadline - t0, 1e-9))
        step_base = max(1e-9, 5e-5 * (0.01) ** frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = float(corr.max())
        c = mx * (2.0 / n)
        if c < best_c - 1e-15:
            best_c = c; best_h = h.copy()
        P, Nv = _pn(h)
        wP, wN = _active_weights(P, Nv, top_k=7)
        g = _grad_w(h, wP, wN)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i != j:
                d = (rng.random() - 0.5) * step_base
                h2 = h.copy(); h2[i] += d; h2[j] -= d
                h2 = _proj(h2, S)
                c2 = _c5(h2)
                if c2 < c - 1e-15:
                    h = h2; it += 1
                    continue
            it += 1
            continue
        direction = -g / gn
        tried = 0
        while tried < 4:
            step = step_base * (1.0 + 0.5 * rng.random()) * min(gn * 5.0, 0.1)
            cand = _proj(h + step * direction, S)
            c2 = _c5(cand)
            if c2 < c - 1e-16:
                h = cand; c = c2
                if c < best_c - 1e-15:
                    best_c = c; best_h = h.copy()
                tried = 4; break
            step *= 0.5
            tried += 1
        if tried >= 4:
            for _ in range(3):
                i = int(low_idx[rng.randint(len(low_idx))]) if rng.random() < 0.7 else int(rng.randint(n))
                j = int(high_idx[rng.randint(len(high_idx))]) if rng.random() < 0.7 else int(rng.randint(n))
                if i == j: continue
                sgn = 1 if g[i] <= g[j] else -1
                avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
                if avail < 1e-13: continue
                d = sgn * min(avail, step_base * (0.5 + rng.random()))
                h2 = h.copy(); h2[i] += d; h2[j] -= d
                h2 = _proj(h2, S)
                c2 = _c5(h2)
                if c2 < c - 1e-16:
                    h = h2; c = c2
                    if c < best_c - 1e-15:
                        best_c = c; best_h = h.copy()
                    break
        gsort = np.argsort(g)
        q = max(8, n // 10)
        low_idx = gsort[:q]
        high_idx = gsort[-q:]
        it += 1
    return best_h, best_c

def _fine_sa(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * (2.0 / n)
    best_h = h.copy()
    t0 = time.time()
    total_t = max(deadline - t0, 1.0)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total_t)
        T = 0.002 * (0.001) ** frac
        dx = 2.0 / n
        k_act = int(np.argmax(corr)) - (n - 1)
        if rng.random() < 0.6 and 0 < abs(k_act) < n:
            k = abs(k_act)
            if k_act > 0:
                valid = (h[:n - k] > 0.01) & (h[k:] < 0.99)
            else:
                valid = (h[k:] > 0.01) & (h[:n - k] < 0.99)
            if valid.sum() > 2:
                pos = np.nonzero(valid)[0]
                i = int(pos[rng.randint(len(pos))])
                j = i + k if k_act > 0 else i - k
                d = (rng.random() - 0.5) * 0.01
                h2 = h.copy(); h2[i] += d; h2[j] -= d
                h2 = _proj(h2, n * 0.5)
                c2_arr = np.correlate(h2, 1.0 - h2, mode='full')
                mx2 = float(c2_arr.max())
                delta = mx2 - mx
                if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                    h = h2; mx = mx2; corr = c2_arr
                    cn = mx * dx
                    if cn < best_c - 1e-15: best_c = cn; best_h = h.copy()
                continue
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j: continue
        d = (rng.random() - 0.5) * 0.008
        h2 = h.copy()
        h2[i] += d; h2[j] -= d
        h2 = _proj(h2, n * 0.5)
        c2_arr = np.correlate(h2, 1.0 - h2, mode='full')
        mx2 = float(c2_arr.max())
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            h = h2; mx = mx2; corr = c2_arr
            cn = mx * dx
            if cn < best_c - 1e-15: best_c = cn; best_h = h.copy()
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    for hh in (1.0 - h0.copy(), 0.5 + 0.5 * (h0 - h0[::-1]), 0.5 + 0.5 * (h0 + h0[::-1] - 1.0)):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c: best = h2; best_c = c
    end1 = t0 + min(80.0, 0.08 * budget_s)
    if end1 > time.time():
        hcur = best.copy()
        for a in [50, 300, 1500, 8000]:
            if time.time() >= end1 - 1: break
            slot = (end1 - time.time()) / 3.0
            hf, bh, bc = _fista(hcur, a, slot, 10.0)
            cf = _c5(hf)
            if bc < best_c: best = bh; best_c = bc
            if cf < best_c: best = hf; best_c = cf
            hcur = hf if cf <= bc else bh
    end2 = t0 + min(180.0, 0.18 * budget_s)
    if end2 > time.time():
        for m in [280, 640]:
            if time.time() >= end2 - 5: break
            slot = min(40.0, end2 - time.time() - 4)
            if slot < 8: break
            hm = _resample(best, m)
            for a in [500, 5000]:
                if time.time() >= end2 - 3: break
                hf, bh, bc = _fista(hm, a, slot / 2.0, 10.0)
                cf = _c5(hf)
                for cand, cv in ((bh, bc), (hf, cf)):
                    if cv < best_c:
                        hc = _resample(cand, n0) if len(cand) != n0 else cand
                        best = hc; best_c = _c5(hc)
    hmain = best if len(best) == n0 else _resample(best, n0)
    end3 = t0 + min(300.0, 0.30 * budget_s)
    if end3 > time.time():
        for a in [5000, 30000, 150000]:
            if time.time() >= end3 - 2: break
            slot = (end3 - time.time()) / 2.0
            hf, bh, bc = _fista(hmain, a, slot, 10.0)
            cf = _c5(hf)
            if bc < best_c: best = bh; best_c = bc
            if cf < best_c: best = hf; best_c = cf
            hmain = hf if cf <= bc else bh
    end4 = t0 + min(780.0, 0.78 * budget_s)
    if end4 > time.time():
        best, best_c = _fine_local(best, rng, end4)
    end5 = t0 + budget_s - 30.0
    if end5 > time.time():
        best, best_c = _fine_sa(best, rng, end5)
    end6 = t0 + budget_s - 3.0
    if end6 > time.time():
        best, best_c = _fine_local(best, rng, end6)
    for hh in (1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1]), 0.5 + 0.5 * (best + best[::-1] - 1.0)):
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c: best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
