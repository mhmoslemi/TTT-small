import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:

            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(abs(diff), h[idx])

    s -= s.max()

            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur - 1e-16:
                cur = cn
                if cn < best:
                    best = cn
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam with ramping alpha
    alphas = [1000, 5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]
    for a in alphas:
        if time.time() - t0 > 300:
            break
        steps = max(60, min(400, int(300000 / max(a, 1))))
        lr = 0.005 * (1000.0 / a) ** 0.25
        h_new, c_new = _adam_opt(best_h.copy(), a, 50, steps, lr)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: SLSQP refinement
    if time.time() - t0 < budget_s - 350:
        for a in [50000, 500000, 5000000, 50000000]:
            if time.time() - t0 > budget_s - 300:
                break
            h_new = _slsqp_opt(best_h.copy(), a, 60, 300)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 3: Projected gradient with Nesterov
    if time.time() - t0 < budget_s - 220:
        cfgs = [(200000, 45, 1200, 0.004), (2000000, 60, 800, 0.002),
                (20000000, 80, 600, 0.0008), (200000000, 100, 400, 0.0003)]
        for a, ml, st, s0 in cfgs:
            if time.time() - t0 > budget_s - 160:
                break
            h_new, c_new = _proj_grad(best_h.copy(), a, ml, st, s0)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 4: Local search with gradient guidance
    if time.time() - t0 < budget_s - 100:
        rem = budget_s - 100 - (time.time() - t0)
        best_h, best_c5 = _local_search(best_h, int(rem * 350), rng, 5e-5)

    # Phase 5: Block swap polish
    if time.time() - t0 < budget_s - 50:
        rem = budget_s - 50 - (time.time() - t0)
        best_h, best_c5 = _block_swap(best_h, int(rem * 500), rng, 5)

    # Phase 6: Fine polish
    if time.time() - t0 < budget_s - 10:
        rem = budget_s - 10 - (time.time() - t0)
        best_h, best_c5 = _fine_polish(best_h, int(rem * 800), rng)

    # Phase 7: Final ultra-fine
    if time.time() - t0 < budget_s - 2:
        rem = budget_s - 2 - (time.time() - t0)
        best_h, best_c5 = _fine_polish(best_h, int(rem * 900), rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
