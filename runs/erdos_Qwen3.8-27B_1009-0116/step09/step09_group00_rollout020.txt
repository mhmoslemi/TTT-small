import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(60):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    return _proj(np.clip(h, 0.0, 1.0), n * 0.5)

def _act_grad(h, tol):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    mx = c.max()
    active = np.nonzero(c >= mx - tol)[0]
    g = np.zeros(n)
    for idx in active:
        k = int(idx) - (n - 1)
        if k == 0:
            g += (1.0 - 2.0 * h)
        elif k > 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            kk = -k
            g[:n-kk] -= h[kk:]
            g[kk:] += (1.0 - h[:n-kk])
    g /= max(len(active), 1)
    return g * dx, mx * dx

def _fista(h0, max_time, tol=0.02, L0=5.0):
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    best_h = h.copy()
    best_c = _c5(h)
    t0 = time.time()
    t_mom = 1.0
    y = h.copy()
    hp = h.copy()
    L = L0
    it = 0
    stale = 0
    while time.time() - t0 < max_time:
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = h + ((t_mom - 1.0) / tn) * (h - hp)
            t_mom = tn
            if y.min() < -0.05 or y.max() > 1.05:
                y = _proj(y, S)
        g, _ = _act_grad(y, tol)
        step = 1.0 / L
        gy = _c5(y)
        acc = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            gx = _c5(xn)
            if gx <= gy + np.dot(g, xn - y) + 0.5 * step * np.dot(xn - y, xn - y) + 1e-12:
                acc = True
                break
            step *= 0.5
        L = max(1.0 / max(step, 1e-12), 1.0)
        if acc:
            hp = h
            h = xn
            stale = 0
            L = max(L * 0.82, 1.0)
            if it % 4 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            t_mom = 1.0
            y = h.copy()
            hp = h
            L = max(L * 0.45, 1.0)
            if stale > 25:
                break
        it += 1
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def _sa(h0, rng, max_time):
    n = len(h0)
    h = _repair(h0, n)
    c = _c5(h)
    best_h = h.copy()
    best_c = c
    t0 = time.time()
    total = max_time
    corr = _corr(h)
    while time.time() - t0 < total:
        frac = min(1.0, (time.time() - t0) / total)
        T = 0.25 * (0.0003) ** frac
        k_act = abs(int(np.argmax(corr)) - (n - 1))
        moved = False
        if rng.random() < 0.68 and 0 < k_act < n:
            if rng.random() < 0.5:
                idx = np.nonzero((h[:n-k_act] > 0.004) & (h[k_act:] < 0.996))[0]
            else:
                idx = np.nonzero((h[:n-k_act] < 0.996) & (h[k_act:] > 0.004))[0]
            if len(idx) > 2:
                i = int(idx[rng.randint(len(idx))])
                j = i + k_act
                sc = 0.004 + 0.025 * (1.0 - frac)
                d = (rng.random() - 0.5) * 2.0 * sc
                h[i] += d
                h[j] -= d
                if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                    cn = _c5(h)
                    if cn <= c + 1e-16 or rng.random() < np.exp(-(cn - c) / max(T, 1e-16)):
                        c = cn
                        corr = _corr(h)
                        if c < best_c:
                            best_c = c
                            best_h = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
                    moved = True
        if not moved:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            sc = 0.003 + 0.018 * (1.0 - frac)
            d = (rng.random() - 0.5) * 2.0 * sc
            h[i] += d
            h[j] -= d
            if not (0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0):
                h[i] -= d
                h[j] += d
                continue
            cn = _c5(h)
            if cn <= c + 1e-16 or rng.random() < np.exp(-(cn - c) / max(T, 1e-16)):
                c = cn
                corr = _corr(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def _micro_polish(h0, rng, max_time):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_h = h.copy()
    best_c = c
    t0 = time.time()
    it = 0
    while time.time() - t0 < max_time:
        g, _ = _act_grad(h, 0.015)
        order = np.argsort(g)
        for i in order[:40]:
            if time.time() - t0 > max_time - 0.3:
                break
            if g[i] >= 0:
                break
            d_base = max(0.001, 0.008 * (1.0 - (time.time() - t0) / max_time))
            d = d_base * (0.4 + 0.8 * rng.random())
            j_pool = order[-40:]
            j = int(j_pool[rng.randint(len(j_pool))])
            if j == i:
                continue
            avail = min(1.0 - h[i], h[j])
            if avail < 1e-9:
                continue
            dd = min(d, avail)
            h[i] += dd
            h[j] -= dd
            cn = _c5(h)
            if cn < c - 1e-16:
                c = cn
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i] -= dd
                h[j] += dd
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    for hh in (1.0 - h0.copy(), h0[::-1]):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    # Phase 1: FISTA warm-up (100s)
    end1 = t0 + 100.0
    if end1 > time.time() + 8:
        hcur = best.copy()
        for tol in [0.06, 0.025, 0.01]:
            slot = (end1 - time.time()) / 3.0

                break
            hf, bc = _fista(hcur, slot, tol, 5.0)
            if bc < best_c:
                best = hf
                best_c = bc
            hcur = hf
    # Phase 2: Multi-res FISTA (100s)
    end2 = t0 + 200.0
    if end2 > time.time() + 8:
        for m in [320, 448, n0, 640]:
            slot = (end2 - time.time()) / 2.0
            if slot < 12:
                break
            hm = _resample(best, m)
            hf, bc = _fista(hm, slot, 0.02, 8.0)
            if len(hf) != n0:
                hf = _resample(hf, n0)
            cf = _c5(hf)
            if cf < best_c:
                best = hf
                best_c = cf
    # Phase 3: Long SA (280s)
    end3 = t0 + 480.0
    if end3 > time.time() + 8:
        best, best_c = _sa(best, rng, end3 - time.time())
    # Phase 4: Micro polish (200s)
    end4 = t0 + 680.0
    if end4 > time.time() + 8:
        best, best_c = _micro_polish(best, rng, end4 - time.time())
    # Phase 5: FISTA tight tolerance (140s)
    end5 = t0 + 820.0
    if end5 > time.time() + 8:
        hf, bc = _fista(best, end5 - time.time(), 0.005, 12.0)
        if bc < best_c:
            best = hf
            best_c = bc
    # Phase 6: Final SA (remaining)
    end6 = t0 + budget_s - 4.0
    if end6 > time.time() + 3:
        best, best_c = _sa(best, rng, end6 - time.time())
    # Final micro-polish
    if time.time() < t0 + budget_s - 2.0:
        best, best_c = _micro_polish(best, rng, min(30.0, budget_s - (time.time() - t0) - 1.0))

    for hh in (1.0 - best, best[::-1]):
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
