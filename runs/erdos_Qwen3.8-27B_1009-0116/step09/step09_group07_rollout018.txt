from scipy.optimize import minimize
import time

def _repair(h, n):

                h[idx] -= a
                diff += a

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):

def _cd_pass(h, rng):
    n = len(h)
    dx = 2.0 / n
    c = _corr_full(h)  # this already includes dx
    # Actually, for CD we work with the raw correlation (without dx)
    # since dx is a constant factor
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    improved = 0
    for j in jlist:
        j = int(j)
        # Compute beta: how c changes when h[j] changes by dt
        # c[k] = sum_i h[i] * (1 - h[i+k-nm1]) for k in range(cmax)
        # If h[j] changes by dt:
        #   For pairs where i=j: contribution is dt*(1-h[j+k-nm1]) for valid k
        #   For pairs where i+k-nm1=j: contribution is h[i]*(-dt) for valid k
        beta = np.zeros(cmax)

        # In correlate(h, 1-h, 'full'), lag L (L = k-nm1) means:
        # c_raw[k] = sum_{valid i} h[i] * (1-h[i-L])

        # Actually: correlate(a, v)[k] = sum_n a[n+k] * v[n] for k in -(len(v)-1)..len(a)-1
        # With mode='full', output has length len(a)+len(v)-1
        # correlate(h, 1-h, 'full')[k] = sum_n h[n] * (1-h)[n + k - (len(1-h)-1)]
        # = sum_n h[n] * (1-h[n + k - nm1])
        # where n ranges so that 0 ≤ n < n and 0 ≤ n+k-nm1 < n
        
        # So c_raw[k] = sum_{n=max(0,nm1-k)}^{min(n-1, n-1+k-nm1)} h[n] * (1-h[n+k-nm1])
        
        # If h[j] changes by dt:
        # Term 1: n=j, so 1-h[j+k-nm1] must be valid: 0 ≤ j+k-nm1 < n
        #   → nm1-j ≤ k ≤ nm1-j+n-1 = 2*nm1-j
        #   c_raw[k] gets dt * (1-h[j+k-nm1])
        # Term 2: n+k-nm1=j, so n = j+k-nm1 must be valid: 0 ≤ j+k-nm1 < n
        #   → nm1-j ≤ k ≤ nm1-j+n-1
        #   c_raw[k] gets h[j+k-nm1] * (-dt)

        k_hi_nm1 = nm1 - j + n - 1

        k1_hi = min(cmax - 1, k_hi_nm1)
        if k1_lo <= k1_hi:
            lags = np.arange(k1_lo, k1_hi + 1) - nm1
            beta[k1_lo:k1_hi+1] += (1.0 - h[j - lags])

        #   k from nm1-j to nm1-j+n-1
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, nm1 - j + n - 1)
        if k2_lo <= k2_hi:

            beta[k2_lo:k2_hi+1] -= h[j + lags2]

        # Minimize max(c_new) over t ∈ [0,1]

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-18:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter, max_dist=12):

    best_h = h.copy()
    best_c5 = c5c
    for it in range(n_iter):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _cd_pass(h, rng, nit=50):
    n = len(h)
    nm1 = n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        klo = nm1 - j
        khi = klo + n - 1
        a1 = max(0, klo)
        b1 = min(cmax - 1, khi)
        if a1 <= b1:
            lags = np.arange(a1, b1 + 1) - nm1
            beta[a1:b1+1] += 1.0 - h[j - lags]
            beta[a1:b1+1] -= h[j + lags]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(nit):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _slsqp_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-18:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _s_eq(x, tgt):
    return x.sum() - tgt

def _s_eq_j(x, tgt):
    return np.ones(len(x))

def _slsqp_solve(h, alpha, ml, maxit):
    n = len(h)
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, ml),
                       method='SLSQP', bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':_s_eq,'jac':_s_eq_j,'args':(n*0.5,)}],
                       options={'maxiter': maxit, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, niter, step=2e-6, maxd=15):
    c5c = _eval(h)
    best = h.copy()
    bestc = c5c
    for _ in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, maxd)) % n
        d = step * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c:
                c5c = cn
                if cn < bestc:
                    bestc = cn
                    best = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best, bestc
