    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    obj = m + np.log(np.exp(s).sum()) / alpha + mu * (x.sum() - n * 0.5) ** 2
    return obj

def _perturb_and_reopt(h, n, rng, budget_frac, t_deadline):
    c5_orig = _eval(h)
    for eps in [0.03, 0.015, 0.008]:
        if time.time() > t_deadline:
            break
        h_p = h + eps * rng.randn(n)
        h_p = _repair(h_p, n)
        # Quick CD (2 passes)
        for _ in range(2):
            h_p = _cd_pass(h_p, rng)
            h_p = _repair(h_p, n)
        c5_p = _eval(h_p)
        if c5_p < c5_orig:
            h = h_p
            c5_orig = c5_p
    return h

def _fine_search(h, n, rng, deadline):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(int(1e7)):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = 3e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: CD (48% budget)
    p1_end = t0 + budget_s * 0.48
    stalls = 0
    while time.time() < p1_end:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls > 40:
                break
    # Ensure we're at best
    h = best_h.copy()

    # Phase 2: Perturbation + reoptimize (10% budget)
    p2_end = t0 + budget_s * 0.58
    if time.time() < p2_end:
        h = _perturb_and_reopt(best_h, n, rng, 0.1, p2_end)
        h = _repair(h, n)
        c5 = _eval(h)
        # A few more CD passes from perturbed
        while time.time() < p2_end:
            h = _cd_pass(h, rng)
            h = _repair(h, n)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: L-BFGS-B (20% budget)
    p3_end = t0 + budget_s * 0.78
    for alpha, ml, mu in [(8000, 40, 50.0), (40000, 40, 200.0),
                           (200000, 60, 500.0), (1000000, 80, 2000.0)]:
        if time.time() > p3_end:
            break
        sub_dl = min(p3_end, time.time() + (p3_end - t0 - (time.time() - t0)) / max(1, 4))
        h_new = _lbfgsb_solve(best_h.copy(), alpha, ml, mu, n, 300, p3_end)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: More CD (12% budget)
    p4_end = t0 + budget_s * 0.90
    h = best_h.copy()
    while time.time() < p4_end:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Final L-BFGS-B high alpha (5% budget)
    p5_end = t0 + budget_s * 0.95
    if time.time() < p5_end:
        h_new = _lbfgsb_solve(best_h.copy(), 5000000, 100, 5000.0, n, 200, p5_end)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 6: Fine local search (5% budget)
    deadline = t0 + budget_s - 3
    if time.time() < deadline:
        best_h, best_c5 = _fine_search(best_h, n, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
