                quad[:n-k2] += h[k2:]
                quad[k2:] += h[:n-k2]
                g += lin - quad
        
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        
        moved = False
        for a in [alpha, alpha*0.5, alpha*0.2, 1e-4, 5e-5]:
            h_try = _proj(h - a * g, S)
            c_try = _c5(h_try)
            if c_try <= c_val + 1e-14:
                h = h_try; alpha = min(alpha*1.4, 0.08); moved = True
                break
        if not moved:
            alpha *= 0.5
            if alpha < 1e-7:
                break
    
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h.copy(), n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _soft_max_c5(h, a):
    """Soft-max approximation of C5 using FFT-based autocorrelation"""
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    # autocorrelation of h
    ac = np.fft.irfft(H * np.conj(H), N).real
    # total sum
    ps = np.concatenate([[0], np.cumsum(h)])
    P = np.zeros(n)
    P[0] = ps[n] - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = ps[n - idx] - ac[idx]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        Nv[idx] = ps[n] - ps[idx] - ac[idx]
    all_vals = np.concatenate([P, Nv[1:]])
    mx = float(np.max(all_vals))
    # Log-sum-exp for soft max
    shifted = a * (all_vals - mx)
    shifted = np.clip(shifted, -400, 0)
    e = np.exp(shifted).sum()
    return (2.0/n) * (mx + np.log(e) / a)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    
    best = h0.copy()
    best_c = _c5(best)
    
    # Quick variant check
    for hh in (1.0 - h0, h0[::-1].copy()):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    
    # Phase 1: Coarse-to-fine (first 100s)
    end1 = t0 + min(100.0, 0.10 * budget_s)
    for nc in [64, 128, 256]:
        remaining = end1 - time.time()
        if remaining < 15:
            break
        hc = _resample(best, nc)
        hc, cc = _sa_opt(hc, rng, time.time() + min(30, remaining * 0.5), scale=2.0)
        hu = _resample(hc, n0)
        cu = _c5(hu)
        if cu < best_c:
            best = hu; best_c = cu
    
    # Phase 2: Subgradient refinement (next 100s)
    end2 = t0 + min(250.0, 0.25 * budget_s)
    if time.time() < end2:
        hs, cs = _subgrad_opt(best, end2 - time.time(), max_active=5, tol_frac=0.02)
        if cs < best_c:
            best = hs; best_c = cs
    
    # Phase 3: Main SA at full resolution (next 250s)
    end3 = t0 + min(550.0, 0.55 * budget_s)
    if time.time() < end3:
        n_rounds = 3
        slot = (end3 - time.time()) / n_rounds
        for r in range(n_rounds):
            if time.time() >= end3 - 5:
                break
            # Escape: add structured noise
            noise_scale = 0.015 / (r + 1)
            noise = (rng.random(n0) - 0.5) * noise_scale
            h_try = _proj(best + noise, n0 * 0.5)
            hs, cs = _sa_opt(h_try, rng, time.time() + max(10, slot * 0.85))
            if cs < best_c:
                best = hs; best_c = cs
    
    # Phase 4: Subgradient + SA alternate (next 200s)
    end4 = t0 + min(800.0, 0.80 * budget_s)
    if time.time() < end4:
        n_pairs = 2
        slot = (end4 - time.time()) / n_pairs
        for r in range(n_pairs):
            if time.time() >= end4 - 10:
                break
            hs, cs = _subgrad_opt(best, time.time() + slot * 0.5, max_active=3, tol_frac=0.015)
            if cs < best_c:
                best = hs; best_c = cs
            hs2, cs2 = _sa_opt(best, rng, time.time() + max(5, slot * 0.45))
            if cs2 < best_c:
                best = hs2; best_c = cs2
    
    # Phase 5: Final polish SA
    end5 = t0 + budget_s - 15.0
    if time.time() < end5:
        best, best_c = _sa_opt(best, rng, end5, scale=0.5)
    
    # Try different n for final answer
    final_n = n0
    for alt_n in [n0, n0 - 8, n0 + 8]:
        if alt_n < 40 or alt_n > 1200:
            continue
        ha = _resample(best, alt_n)
        ca = _c5(ha)
        if ca < best_c:
            best = ha; best_c = ca; final_n = alt_n
    
    # Final variant
    for hh in (1.0 - best, best[::-1].copy()):
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
