    return _repair(h, n), _eval(h)

def _subgrad(h, base_step, top_k, n_steps, rng):
    n = len(h); target = n / 2.0
    cur = _eval(h); step = base_step
    for it in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        idx = np.argsort(c)[-top_k:]
        w = c[idx] + 0.005; w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, idx):
            L = ki - (n-1)
            if L >= 0:
                g[:n-L] += wi*(1.0-h[L:]); g[L:] -= wi*h[:n-L]
            else:
                M = -L
                g[:n-M] += wi*(1.0-h[M:]); g[M:] -= wi*h[:n-M]
        h_new = _proj_c(h - step * g, target)
        c_new = _eval(h_new)
        if c_new < cur - 1e-16:
            h = h_new; cur = c_new
            step = min(step * 1.015, base_step * 15)
        else:
            step *= 0.82
    return _repair(h, n), _eval(h)

def _fine_2pt(h, n_iter, rng, start_scale):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    scale = start_scale
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + rng.randint(1, 6)) % n
        if i == j: continue
        t = scale * rng.randn()
        ni, nj = h[i]+t, h[j]-t
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                scale = min(scale * 1.1, start_scale * 5)
            else:
                h[i], h[j] = ni - t, nj + t
                scale *= 0.9
    return best_h, best_c

def _sa_large(h, n_iter, rng, n_blocks, block_len):
    """Simulated annealing with block swaps"""
    n = len(h); target = n / 2.0
    best_h, best_c = h.copy(), _eval(h)
    cur_c = best_c
    temp = 1e-4
    for it in range(n_iter):
        temp *= 0.9998
        nb = rng.randint(n_blocks)
        sz = min(block_len + rng.randint(-2, 3), n)
        sz = max(2, sz)
        s1 = rng.randint(0, n - sz)
        s2 = rng.randint(0, n - sz)
        if abs(s1 - s2) < sz:
            s2 = (s2 + sz) % max(1, n - sz)
        if s1 == s2: continue
        save1 = h[s1:s1+sz].copy()
        save2 = h[s2:s2+sz].copy()
        h[s1:s1+sz] = save2
        h[s2:s2+sz] = save1
        h = _proj_c(h, target)
        c_new = _eval(h)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            cur_c = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            h[s1:s1+sz] = save1
            h[s2:s2+sz] = save2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c = h.copy(), _eval(h)

    # Phase 1: SLSQP warm-up with multiple alphas
    for alpha, k in [(3000, 8), (30000, 12), (300000, 15), (3000000, 20)]:
        if _el() > 100: break
        hn = _slsqp(best_h, alpha, k, 1000)
        cn = _eval(hn)
        if cn < best_c:
            best_c, best_h = cn, hn

    # Phase 2: Block coordinate descent (multi-scale)
    if _el() < budget_s - 300:
        for B, ns in [(28, 3), (14, 5), (7, 8)]:
            if _el() > budget_s - 280: break
            ab = [5000, 50000, 500000][0 if B >= 20 else (1 if B >= 10 else 2)]
            kb = min(8, 3 + int(2 * np.log10(max(ab, 100) / 5000)) + 3)
            best_h, best_c = _block_cd(best_h, B, ns, rng, ab*100, kb)

    # Phase 3: Subgradient descent
    if _el() < budget_s - 200:
        stk = min(int((budget_s - 200 - _el()) * 300), 15000)
        best_h, best_c = _subgrad(best_h, 0.003, 3, stk, rng)

    # Phase 4: Block CD again (refinement)
    if _el() < budget_s - 150:
        best_h, best_c = _block_cd(best_h, 5, 5, rng, 200000, 6)

    # Phase 5: Simulated annealing with block swaps
    if _el() < budget_s - 100:
        sa_iter = min(int((budget_s - 100 - _el()) * 200), 50000)
        best_h, best_c = _sa_large(best_h, sa_iter, rng, max(2, n//12), 8)
        # Quick SLSQP after SA
        if _el() < budget_s - 20:
            hn = _slsqp(best_h, 500000, 10, 500)
            cn = _eval(hn)
            if cn < best_c:
                best_c, best_h = cn, hn

    # Phase 6: Fine 2-point polish
    if _el() < budget_s - 5:
        fnter = min(int((budget_s - 5 - _el()) * 800), 400000)
        best_h, best_c = _fine_2pt(best_h, fnter, rng, 2e-6)

    # Phase 7: Final SLSQP
    if _el() < budget_s - 3:
        for a2 in [5e6, 5e7]:
            if _el() > budget_s - 4: break
            hn = _slsqp(best_h, a2, 8, 300)
            cn = _eval(hn)
            if cn < best_c:
                best_c, best_h = cn, hn

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
