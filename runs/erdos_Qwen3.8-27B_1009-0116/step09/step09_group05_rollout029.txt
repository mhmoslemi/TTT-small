def _sum_repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:

        else:

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

    """Project to binary with exactly n//2 ones"""
    target = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:target]] = 1.0
    return h_bin

def _pair_swap_search(h, c5, n, n_iter, rng, temp=0.01):
    """Simulated annealing on binary solutions via pair swaps"""
    best_h = h.copy()
    best_c5 = c5
    cur_c5 = c5
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for it in range(n_iter):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        c5_new = _eval_val(h)
        # Accept if better or with SA prob
        if c5_new < cur_c5 or rng.rand() < np.exp(-(c5_new - cur_c5) / temp):

            ones.remove(i)
