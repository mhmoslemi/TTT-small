import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s_new = h.sum()
        if abs(s_new - target) < 1e-12:
            break
        s = s_new
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _cvector(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _profiles(h, n):
    j = np.arange(n)[:, None]
    L = np.arange(-(n - 1), n)[None, :]
    jp = j + L
    jm = j - L
    P = np.zeros((n, 2 * n - 1))
    mp = (jp >= 0) & (jp < n)
    mm = (jm >= 0) & (jm < n)

    h[k] -= t_opt
    c += t_opt * delta
    return mv - f_new

def _smooth_grad(h, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_m = np.argsort(c)[-m:]
    s = alpha * (c[top_m] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, ko in enumerate(top_m):
        if w[ii] < 1e-15:
            continue
        L = ko - (n - 1)
        wk = w[ii] * dx
        jp = np.arange(n) + L
        jm = np.arange(n) - L
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        contrib = np.zeros(n)
        contrib[mp] += (1.0 - h[jp[mp]])
        contrib[mm] -= h[jm[mm]]
        g += wk * contrib
    return g

def _proj_onto(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
