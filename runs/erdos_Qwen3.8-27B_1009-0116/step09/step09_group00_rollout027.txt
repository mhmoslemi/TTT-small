   Parameterize: x = S/2 + d, y = S/2 - d. Back to 1D!
   
   For 3 free variables with sum constraint (2 DOF):
   Parameterize: x1 = S/3 + d1, x2 = S/3 + d2, x3 = S/3 - d1 - d2.
   Now f(d1, d2) = max_k (c_k + d1*a1_k + d2*a2_k) is a 2D convex piecewise-linear function.
   The minimum can be found by:
   - LP: min t s.t. c_k + d1*a1_k + d2*a2_k ≤ t, with bounds on d1, d2
   - This is a small LP with 2 variables + t = 3 variables, and ~559 constraints

   - Sum constraint: sum x_i = S (fixed by the other n-m values)
   - Bounds: 0 ≤ x_i ≤ 1
   - Objective: min t
   - Constraints: for each lag ℓ (n-1 of them):
     sum_{j=1}^{m} x_j * coef_j_ℓ + const_ℓ ≤ t
     
   Where coef_j_ℓ is the slope of corr[ℓ] w.r.t. h[p_j] (computed at current h, ignoring in-block bilinear terms), and const_ℓ = corr[ℓ] - sum_j x_j_current * coef_j_ℓ.

   New corr[ℓ] = corr[ℓ] + sum_j (x_j_new - x_j_old) * slope_j_ℓ
   = corr[ℓ] - sum_j x_j_old * slope_j_ℓ + sum_j x_j_new * slope_j_ℓ
   
   So in the LP:
   const_ℓ = corr[ℓ] - sum_j x_j_old * slope_j_ℓ

   Constraint: sum_j slope_j_ℓ * x_j + const_ℓ ≤ t

import numpy as np
import time
from scipy.optimize import linprog

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _slopes_all(h, p):
    n = len(h)
    lags = np.arange(-(n-1), n)
    idx1 = p + lags
    valid1 = (idx1 >= 0) & (idx1 < n)
    f1 = np.zeros(2*n-1)
    f1[valid1] = 1.0 - h[idx1[valid1]]
    idx2 = p - lags
    valid2 = (idx2 >= 0) & (idx2 < n)
    f2 = np.zeros(2*n-1)
    f2[valid2] = h[idx2[valid2]]
    return f1 - f2

def _min_1d(corr, a, d_lo, d_hi):
    if d_hi - d_lo < 1e-14:
        return 0.5*(d_lo + d_hi)
    for _ in range(60):
        dm = 0.5*(d_lo + d_hi)
        vals = corr + dm * a
        fm = vals.max()
        active = np.abs(vals - fm) < 1e-12
        if not active.any():
            ai = int(np.argmax(vals))
            if a[ai] > 0: d_hi = dm
            elif a[ai] < 0: d_lo = dm
            else: return dm
            continue
        mn = a[active].min()
        mx = a[active].max()
        if mx <= 1e-14: d_lo = dm
        elif mn >= -1e-14: d_hi = dm
        else: return dm
    return 0.5*(d_lo + d_hi)

def _pair_update(h, i, j):
    n = len(h)
    si = _slopes_all(h, i)
    sj = _slopes_all(h, j)
    a = si - sj
    if np.max(np.abs(a)) < 1e-15:
        return 0.0
    corr = _corr(h)
    d_lo = max(-h[i], h[j]-1.0)
    d_hi = min(1.0-h[i], h[j])
    if d_hi - d_lo < 1e-14:
        return 0.0
    d_opt = _min_1d(corr, a, d_lo, d_hi)
    return d_opt

def _pair_cd(h, n, deadline, rng, n_evals=50):
    t0 = time.time()
    best_c = _c5(h)
    best_h = h.copy()
    corr = _corr(h)
    cmax = corr.max()
    tol = 12.0 * (2.0/n)
    active_lags = np.where(corr >= cmax - tol)[0]
    it = 0
    while time.time() < deadline:
        if it % 400 == 0:
            c = _c5(h)
            if c < best_c - 1e-15:
                best_c = c; best_h = h.copy()
            corr = _corr(h)
            cmax = corr.max()
            active_lags = np.where(corr >= cmax - tol)[0]
        positions = set()
        for k in active_lags:
            ell = k - (n-1)
            lo = max(0, -ell)
            hi = min(n-1, n-1-ell)
            for j in range(lo, hi+1):
                positions.add(j)
                positions.add(j+ell)
        positions = list(positions)
        rng.shuffle(positions)
        use = positions[:n_evals]
        for i in use:
            if time.time() >= deadline:
                break
            for j in positions:
                if j == i:
                    continue
                d = _pair_update(h, i, j)
                if abs(d) > 1e-12:
                    h[i] += d
                    h[j] -= d
                    c2 = _c5(h)
                    if c2 <= cmax * (2.0/n) + 1e-10:
                        cmax = c2 * n / (2.0)
                    else:
                        h[i] -= d
                        h[j] += d
        it += 1
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _block_lp(h, block, n):
    m = len(block)
    if m < 2:
        return h
    S = n*0.5 - h[np.setdiff1d(np.arange(n), block)].sum()
    if S < 0 or S > m:
        return h
    lags = np.arange(-(n-1), n)
    slopes = np.zeros((m, 2*n-1))
    for idx, p in enumerate(block):
        slopes[idx] = _slopes_all(h, int(p))
    corr = _corr(h)
    x_old = h[block].astype(float)
    const = corr - slopes @ x_old
    A_ub = np.zeros((2*n-1, m+1))
    for r in range(2*n-1):
        A_ub[r, :m] = slopes[:, r]
        A_ub[r, m] = -1.0
    b_ub = -const
    A_eq = np.zeros((1, m+1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([S])
    bounds = [(0.0, 1.0)]*m + [(None, None)]
    res = linprog(c=np.zeros(m+1), method='highs',
                  A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds)
    if res.status == 0:
        h_new = h.copy()
        new_vals = res.x[:m]
        t_val = res.x[m]
        for idx, p in enumerate(block):
            h_new[int(p)] = new_vals[idx]
        c_new = _c5(h_new)
        c_old = _c5(h)
        if c_new <= c_old + 1e-12:
            return h_new
    return h

def _block_cd(h, n, deadline, rng, block_size=8):
    t0 = time.time()
    best_c = _c5(h)
    best_h = h.copy()
    tol = 14.0 * (2.0/n)
    while time.time() < deadline:
        corr = _corr(h)
        cmax = corr.max()
        active_lags = np.where(corr >= cmax - tol)[0]
        positions = set()
        for k in active_lags:
            ell = k - (n-1)
            lo = max(0, -ell)
            hi = min(n-1, n-1-ell)
            for j in range(lo, hi+1):
                positions.add(j); positions.add(j+ell)
        positions = sorted(positions)
        rng.shuffle(positions)
        n_blocks = max(1, len(positions)//block_size)
        for bi in range(n_blocks):
            if time.time() >= deadline:
                break
            start = (bi * block_size) % max(1, len(positions) - block_size)
            block = np.array(positions[start:start+block_size])
            if len(block) < 2:
                continue
            h = _block_lp(h, block, n)
        c = _c5(h)
        if c < best_c - 1e-14:
            best_c = c; best_h = h.copy()
    return best_h, best_c

def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _soft_val(h, a):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n+1); pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n-idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = max(float(P.max()), float(Nv.max()))
    else:
        mx = float(P[0])
    dx = 2.0/n
    e = np.exp(np.clip(a*(P[0]-mx), -400, 0))
    if n > 1:
        e += np.exp(np.clip(a*(P[1:]-mx), -400, 0)).sum()
        e += np.exp(np.clip(a*(Nv[1:]-mx), -400, 0)).sum()
    return dx*(mx + np.log(max(e,1e-300))/a)

def _fista_simple(h0, alpha, max_t, L0=10.0):
    n = len(h0)

            y = h + ((t-1.0)/tn)*(h-hp); t = tn

        g = np.zeros(n)

        cmax = corr.max()

        active = np.where(corr >= cmax - tol_g)[0]

            ell = k - (n-1)

                g[j+ell] -= h[j] / len(active)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        fy = _soft_val(y, alpha)
        for _ in range(10):
            xn = _proj(y - step*g, S)
