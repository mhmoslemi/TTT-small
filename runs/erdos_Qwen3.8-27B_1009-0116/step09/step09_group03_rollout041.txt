For lag n-1 (maximum positive lag): c_{n-1} = h[0](1-h[n-1])

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, L, n):
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[:n-L] = 1.0 - h[L:]
            g[L:] -= h[:n-L]
    else:
        M = -L
        if M < n:
            g[:n-M] = 1.0 - h[M:]
            g[M:] -= h[:n-M]
    return g

def _lp_step(h, active_lags, n, damping):
    c_corr = _corr(h)
    M = len(active_lags)
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)
    for m, lag_idx in enumerate(active_lags):
        L = lag_idx - (n - 1)
        g = _grad_lag(h, L, n)
        A_ub[m, :n] = g
        A_ub[m, n] = -1.0
        b_ub[m] = c_corr[lag_idx] - g @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    c_obj = np.zeros(n + 1); c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        h_new = (1.0 - damping) * h + damping * res.x[:n]
        return _repair(h_new, n)
    return h.copy()

def _active_lp(h, n, n_iter, t_end, M=80, lam_start=0.4, update_freq=8):
    best_c = _eval(h); best_h = h.copy()
    lam = lam_start
    for it in range(n_iter):
        if time.time() - t_end > 0:
            break
        c_corr = _corr(h)
        top = np.argpartition(-c_corr, M)[:M]
        spread = np.linspace(0, 2*n-2, max(4, M//8)).astype(int)
        top = np.unique(np.concatenate([top, spread]))[:min(M+20, 2*n-1)]
        h_new = _lp_step(h, top, n, lam)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new; best_h = h_new.copy(); lam = min(lam*1.15, 0.95)
        else:
            lam = max(lam*0.7, 0.05)
        h = h_new
    return best_h, best_c

def _downsample(h, f):
    n = len(h); m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    n = len(h); out = np.empty(n*f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n*f)

def _grad_search(h, n_iter, rng, t_end):
    n = len(h); c5 = _eval(h); best_h = h.copy(); best_c = c5
    for _ in range(n_iter):
        if time.time() - t_end > 0:
            break
        i = rng.randint(n); j = (i + 1 + rng.randint(0, 12)) % n

        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < c5:
                c5 = c
                if c < best_c: best_c = c; best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c
