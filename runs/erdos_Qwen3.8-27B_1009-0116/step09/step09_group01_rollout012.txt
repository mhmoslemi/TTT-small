    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol_eff:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol_eff
        if np.any(mP):
            wP[1:][mP] = 1.0
            cnt += mP.sum()
        mN = Nv[1:] >= mx - tol_eff
        if np.any(mN):
            wN[1:][mN] = 1.0
            cnt += mN.sum()
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    return _gW(h, wP, wN)

def _loc_search(h0, rng, dl, s0=1e-4):

    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    no_improve = 0
    max_no_improve = 2000
    it = 0
    while time.time() - t0 < dl:
        P, Nv, mx = _pn(h)
        g = _agrad(h)
        # Find best move: decrease h at high-g location, increase at low-g location
        order = np.argsort(g)
        q = max(5, n//10)
        lo_idx = order[:q]
        hi_idx = order[-q:]
        # Temperature for SA

        T = s0 * (0.1)**fr

            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            it += 1
            continue
        sgn = 1 if g[i] <= g[j] else -1
        avail = min(1.0-h[i], h[j]) if sgn>0 else min(h[i], 1.0-h[j])
        if avail < 1e-14:

            continue

        h[j] -= d

            c = cn
            no_improve = 0
            if c < bc:
                bc = c
                bh = h.copy()
        else:

            if T < 1e-16 or rng.random() > np.exp(-max(cn-c, 0)/max(T, 1e-30)):
                h[i] -= d
                h[j] += d
            no_improve += 1

                        h[p1] += 0.1
                        h[p2] -= 0.1
                h = _repair(h, n)
                c = _c5(h)
                no_improve = 0
        it += 1
    c = _c5(h)
    if c < bc: bc, bh = c, h.copy()
    return bh, bc

def _block_move_search(h0, rng, dl):

    return bh, bc

- Increase h[i+k] where h[i] is large

import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
        mx = float(max(P.max(), Nv.max()))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _gW(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n - 1:2 * n - 1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sg(h, a):

    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))

        if tot <= 0:

        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)

    c = _c5(h)

def _agrad(h):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    tol = max(1e-8, 3.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1

        mN = Nv[1:] >= mx - tol

            wN[1:][mN] = 1.0
            cnt += int(mN.sum())

        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    return _gW(h, wP, wN)

                h[i] -= d
                h[j] += d
            no_imp += 1
            if no_imp > 5000:
                for _ in range(20):
                    p1, p2 = int(rng.randint(n)), int(rng.randint(n))
                    if p1 != p2:
                        dd = 0.05 * rng.random()
                        h[p1] += dd
                        h[p2] -= dd
                h = _repair(h, n)
                c = _c5(h)
                no_imp = 0
    c = _c5(h)
    if c < bc:
        bc, bh = c, h.copy()
    return bh, bc

def _block_swap(h0, rng, dl):
    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    bc = _c5(h)
    bh = h.copy()
    sizes = [1, 2, 3, 4, 5, 7, 10, 14, 20]
    while time.time() - t0 < dl:
        L = int(sizes[rng.randint(len(sizes))])
        if L >= n // 2:
            continue
        a = int(rng.randint(n - L))
        b = int(rng.randint(n - L))
        if abs(a - b) < L:
            continue
        blk = h[a:a + L].copy()
        h[a:a + L] = h[b:b + L]
        h[b:b + L] = blk
        cn = _c5(h)
        if cn < bc - 1e-14:
            bc = cn
            bh = h.copy()
        else:
            h[b:b + L] = h[a:a + L]
            h[a:a + L] = blk
    return bh, bc

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)

    def elapsed():
        return time.time() - t0

    def time_left():
        return max(1.0, budget_s - 5.0 - elapsed())

    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n0 * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    end1 = t0 + 45.0
    hc = best.copy()
    for a in [40, 150, 600, 2000]:
        rem = end1 - time.time()
        if rem < 4:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.25)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    end2 = t0 + 110.0
    for a in [8000, 40000, 200000]:
        rem = end2 - time.time()
        if rem < 5:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.3)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    end3 = t0 + 420.0
    best, best_c = _loc_search(best, rng, end3 - time.time(), s0=8e-5)

    end4 = t0 + 580.0
    bh2, bc2 = _block_swap(best, rng, end4 - time.time())
    if bc2 < best_c:
        best, best_c = bh2, bc2

    end5 = t0 + 700.0
    best, best_c = _loc_search(best, rng, end5 - time.time(), s0=3e-5)

    end6 = t0 + 850.0
    if time_left() > 100 and best_c > 0.3800:
        for m_try in [640, 720, 800]:
            if time_left() < 40:
                break
            h_up = _resample(best, m_try)
            hu, bu = _fista(h_up, 2000, min(25.0, time_left() * 0.2))
            hu, bu = _loc_search(hu, rng, min(30.0, time_left() * 0.3), s0=2e-5)
            h_back = _resample(hu, n0)
            cb = _c5(h_back)
            if cb < best_c:
                best, best_c = h_back, cb
            hb = _loc_search(h_back, rng, min(15.0, time_left() * 0.15), s0=1e-5)
            if hb[1] < best_c:
                best, best_c = hb[0], hb[1]

    end7 = t0 + budget_s - 10.0
    best, best_c = _loc_search(best, rng, end7 - time.time(), s0=1e-5)

    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, len(best) * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
