    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _self_corr(h, n):
    """Compute self-correlation h*h using FFT"""
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    return ac

def _pn(h):
    """Compute P (positive shifts) and Nv (negative shifts) contributions"""
    n = len(h)
    ac = _self_corr(h, n)
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    Nv = np.zeros(n)
    P[0] = tot - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
    mx = float(max(P[0], (np.max(P[1:]) if n > 1 else 0), (np.max(Nv[1:]) if n > 1 else 0)))
    return P, Nv, mx

    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _sgrad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)

def _gW(h, wP, wN):
    n = len(h); dx = 2.0 / n
    u = np.zeros(n)
    if n > 1: u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n-1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0]*(1.0-2.0*h) + linP + linN + q)

def _fista(h0, a, tl):
    t0 = time.time(); n = len(h0); S = n*0.5
    h = _proj(h0, S); hp = h.copy(); y = h.copy(); tv = 1.0
    fy = _sval(h, a); bh = h.copy(); bc = _c5(h); stale = 0; Lc = 8.0
    for it in range(600000):
        if time.time() - t0 > tl: break
        if it > 0:
            tn = 0.5*(1.0+np.sqrt(1.0+4.0*tv*tv))
            y = h + ((tv-1.0)/tn)*(h-hp)
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum()-S) > 1e-6:
                y = _proj(y, S)
            tv = tn
        g = _sgrad(y, a)
        step = 1.0/max(Lc, 1e-12); ok = False
        for _ in range(12):
            xn = _proj(y - step*g, S)
            fx = _sval(xn, a)
            d = xn - y
            rhs = fy + float(np.dot(g,d)) + 0.5*(1.0/step)*float(np.dot(d,d))
            if fx <= rhs + 1e-10: ok = True; break
            step *= 0.5
        Lc = 1.0/max(step, 1e-12)
        if ok:
            hp = h; h = xn; stale = 0; Lc = max(Lc*0.92, 1.0)
        else:
            stale += 1; hp,y,tv = h,h,1.0; fy = _sval(h,a); Lc = max(Lc*0.5, 1.0)
            if stale > 30: break
        if it % 12 == 0:
            c = _c5(h)
            if c < bc: bc = c; bh = h.copy()
    c = _c5(h)
    if c < bc: bc = c; bh = h.copy()
    return h, bh, bc

def _subgrad_local(h0, rng, deadline, s0=5e-5):
    t0 = time.time(); n = len(h0); S = n*0.5
    h = h0.copy(); c = _c5(h); bc = c; bh = h.copy()
    no_impr = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
        mx = np.max(corr)
        tol = max(1e-10, 0.002*mx)
        crit = np.where(corr >= mx - tol)[0]
        grad = np.zeros(n)
        for ki in crit:
            k = int(ki - (n-1))
            if k == 0:
                grad += (1.0 - 2.0*h)
            elif k > 0:
                grad[:n-k] += (1.0-h[k:]); grad[k:] -= h[:n-k]
            else:
                k2 = -k; grad[k2:] += (1.0-h[:n-k2]); grad[:n-k2] -= h[k2:]
        grad /= max(1, len(crit))
        fr = (time.time()-t0)/max(1e-9, deadline-t0)
        step = s0*(1e-4)**fr
        order = np.argsort(grad)
        q = max(10, n//8)
        improved = False
        nattempts = 40
        for ai in range(nattempts):
            if rng.random() < 0.75:
                i = int(order[rng.randint(q)])
                j = int(order[rng.randint(n-q, n)])
            else:
                i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j: continue
            sgn = 1.0 if grad[i] <= grad[j] else -1.0
            h[i] += sgn*step; h[j] -= sgn*step
            h = _proj(h, S)
            cn = _c5(h)
            if cn < c - 1e-15:
                c = cn; improved = True; no_impr = 0
                if c < bc - 1e-14: bc = c; bh = h.copy()
            else:
                h[i] -= sgn*step; h[j] += sgn*step; h = _proj(h, S)
        if not improved:
            no_impr += 1
            if no_impr > 300 and rng.random() < 0.02:
                i1 = int(rng.randint(n)); i2 = int(rng.randint(n))
                d = step*30*(1+2*rng.random())
                h[i1] += d; h[i2] -= d; h = _proj(h, S)
                c = _c5(h); no_impr = 0
    return bh, bc

def _nelder(h0, rng, deadline):
    """Simple coordinate-descent style refinement"""
    t0 = time.time(); n = len(h0); S = n*0.5
    h = h0.copy(); c = _c5(h); bc = c; bh = h.copy()
    while time.time() < deadline:
        fr = (time.time()-t0)/max(1e-9, deadline-t0)
        step = 1e-4*(0.1)**fr
        corr = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
        mx = np.max(corr)
        tol = max(1e-10, 0.003*mx)
        crit = np.where(corr >= mx-tol)[0]
        for ki in crit[:8]:
            k = int(ki-(n-1))
            for _ in range(200):
                if k > 0:
                    vi = int(rng.randint(n-k))
                elif k < 0:
                    vi = int(rng.randint(-k, n))
                else:
                    vi = int(rng.randint(n))
                if k != 0:
                    vj = vi + k
                else:
                    vj = int(rng.randint(n))
                if vj < 0 or vj >= n or vi == vj: continue
                sgn = 1.0
                h[vi] += sgn*step; h[vj] -= sgn*step
                h = _proj(h, S)
                cn = _c5(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < bc - 1e-14: bc = c; bh = h.copy()
                else:
                    h[vi] -= sgn*step; h[vj] += sgn*step; h = _proj(h, S)
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _proj(h0, n0*0.5)
    best_c = _c5(best)

    # Try diverse starting points
    cnt = np.arange(n0, dtype=float)
    starts = [best,
              _proj(0.5 + 0.4*np.cos(2*np.pi*cnt/n0), n0*0.5),
              _proj(0.5 + 0.3*np.cos(4*np.pi*cnt/n0) + 0.2*np.sin(2*np.pi*cnt/n0), n0*0.5),
              _proj(1.0 - best, n0*0.5),
              _proj(best[::-1], n0*0.5)]
    for s in starts:
        c = _c5(s)
        if c < best_c:
            best, best_c = s.copy(), c

    # Try different n
    for n_try in [n0, max(200, n0//2), int(n0*1.5//1)]:
        if n_try != n0 and 100 <= n_try <= 2000:
            h_n = _resample(best, n_try)
            rf, bh, bc = _fista(h_n, 500, min(8, max(1, budget_s*0.01)))
            if bc < best_c:
                best, best_c = bh, bc
                n0 = n_try

    # Phase 1: FISTA alpha sweep (180s)
    end1 = t0 + 180
    hc = best.copy()
    for a in [15, 40, 100, 300, 800, 2500, 8000, 30000]:
        rem = end1 - time.time()
        if rem < 2: break
        ta = min(rem*0.14, 22.0)
        hc, bh, bc = _fista(hc, a, ta)
        if bc < best_c: best, best_c = bh, bc

    # Phase 2: High-alpha FISTA (130s)
    end2 = t0 + 310
    for a in [80000, 300000, 1000000, 5000000]:
        rem = end2 - time.time()
        if rem < 3: break
        hc, bh, bc = _fista(hc, a, rem*0.24)
        if bc < best_c: best, best_c = bh, bc

    # Phase 3: Subgradient local (350s)
    end3 = t0 + 660
    best, best_c = _subgrad_local(best, rng, end3, s0=5e-5)

    # Phase 4: Nelder-style (150s)
    end4 = t0 + 810
    best, best_c = _nelder(best, rng, end4)

    # Phase 5: Fine subgradient (150s)
    end5 = t0 + 960
    best, best_c = _subgrad_local(best, rng, end5, s0=1e-5)

    # Phase 6: Ultra-fine (remaining)
    end6 = t0 + budget_s - 3
    if end6 > time.time() + 5:
        best, best_c = _subgrad_local(best, rng, end6, s0=3e-6)

    # Final symmetry
    n = len(best)
    alt = _proj(1.0 - best, n*0.5)
    ca = _c5(alt)
    if ca < best_c:
        best = alt; best_c = ca
    alt2 = _proj(best[::-1], n*0.5)
    c2 = _c5(alt2)
    if c2 < best_c:
        best = alt2; best_c = c2

    best = _proj(best, n*0.5)
    best_c = _c5(best)
    return best, best_c, n
