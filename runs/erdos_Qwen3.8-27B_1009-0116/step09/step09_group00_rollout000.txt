    if abs(s - target) < 1e-14:
        return h
    # Iterative water-filling
    for _ in range(50):

        h *= frac

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h *= target / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            a = min(abs(d), max(0, d) + (1.0 - h[i]) if d > 0 else h[i])
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a

def _proj(v, S):
    x = np.array(v, float, copy=True)
    x += (S - x.sum()) / x.size

    n = len(h0)
    S = n * 0.5
    h = _repair(h0.copy(), n)
    t0 = time.time()
    best_h = h.copy()
    best_c = _c5(h)
    step = 1.0
    for _ in range(100000):
        if time.time() - t0 > max_time:
            break
        g = _grad_soft(h, alpha, top_k)
        # Backtracking line search
        y = h
        fy = _soft_c5(h, alpha, top_k)
        accepted = False
        for _ls in range(30):
            h_new = _proj(y - step * g, S)
            f_new = _soft_c5(h_new, alpha, top_k)
            dy = h_new - y
            if f_new <= fy + float(np.dot(g, dy)) + 0.5 * step * float(np.dot(dy, dy)) + 1e-12:
                accepted = True
                break
            step *= 0.5

            if c < best_c:
                best_c = c
                best_h = h_new.copy()
            h = h_new
            step = min(step * 1.5, 10.0)
