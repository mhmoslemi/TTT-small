import numpy as np
import time

def _project(v, S):
    n = len(v)
    x = v.copy()
    for _ in range(60):
        x += (S - x.sum()) / n
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _pn(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = max(P[0], float(P[1:].max()) if n > 1 else P[0])
    if n > 1:
        mx = max(mx, float(Nv[1:].max()))
    return P, Nv, mx

def _soft_val_grad(h, alpha):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    # Build soft-max value
    a = alpha
    e0 = np.exp(np.clip(a * (P[0] - mx), -500, 0.0))
    tot = e0
    wP = np.zeros(n)
    wN = np.zeros(n)
    wP[0] = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -500, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -500, 0.0))
        tot += ep.sum() + en.sum()
        wP[1:] = ep
        wN[1:] = en
    if tot < 1e-300:
        tot = 1.0
        wP[0] = 1.0
    wP /= tot
    wN /= tot
    val = dx * (mx + np.log(tot) / a)
    
    # Gradient
    grad = np.zeros(n)
    # Positive lags
    for k in range(n):
        wk = wP[k]
        if wk < 1e-14:
            continue
        m = n - k
        grad[:m] += wk * (1.0 - h[k:k+m])
        grad[k:k+m] += wk * (-h[:m])
    # Negative lags
    for k in range(1, n):
        wk = wN[k]
        if wk < 1e-14:
            continue
        m = n - k
        grad[k:k+m] += wk * (1.0 - h[:m])
        grad[:m] += wk * (-h[k:k+m])
    return val, dx * grad

def _fista_run(h0, alpha, max_t):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _project(h0, S)
    hp = h.copy()
    y = h.copy()
    t_val = 1.0
    fy, _ = _soft_val_grad(h, alpha)
    best_h = h.copy()
    best_c = _c5(h)
    L = 5.0
    stale = 0
    for it in range(200000):
        if time.time() - t0 > max_t * 0.95:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val**2))
            y = h + ((t_val - 1.0) / tn) * (h - hp)
            t_val = tn
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum() - S) > 0.01:
                y = _project(y, S)
            fy, _ = _soft_val_grad(y, alpha)
        g = _soft_val_grad(y, alpha)[1]
        step = 1.0 / max(L, 1e-10)
        ok = False
        for _ in range(12):
            xn = _project(y - step * g, S)
            fx, _ = _soft_val_grad(xn, alpha)
            dv = xn - y
            rhs = fy + np.dot(g, dv) + 0.5 * dv.dot(dv) / step
            if fx <= rhs + 1e-10:
                ok = True
                break
            step *= 0.5
        if ok:
            hp = h.copy()
            h = xn
            fy = fx
            stale = 0
            L = max(L * 0.9, 1.0)
            if it % 3 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h.copy()
            y = h.copy()
            t_val = 1.0
            fy = np.nan  # will recompute
            L = max(L * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c
