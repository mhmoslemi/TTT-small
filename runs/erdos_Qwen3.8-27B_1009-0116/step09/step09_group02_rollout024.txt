    idx = np.argsort(c)[-k:]
    s = alpha * c[idx]
    s -= s.max()

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            break
    diff = tf - h.sum()
    if abs(diff) > 1e-13:

                diff -= a

    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 14)) % n
        d = (1e-9 + 8e-8 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _coord_grp(h, n_iter, rng, target, gs=4):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 22, size=gs)) % n)
        d = (1e-9 + 4e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, target)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _perturb_restart(h, rng, target, deadline, n_trials):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_try = best_h.copy()
        n_pert = rng.randint(8, 25)
        idx = rng.choice(n, n_pert, replace=False)
        d = 0.015 * rng.randn(n_pert)
        d -= d.mean()
        h_try[idx] += d
        h_try = _repair(h_try, target)
        h_opt, c_opt = _nesterov(h_try, 8, 300, 0.003, 0.82, deadline, target)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c
