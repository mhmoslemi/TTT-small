def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:

    return h

    return m + np.exp(s).sum() / alpha

import numpy as np

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), h[i]) if d < 0 else min(abs(d), 1.0 - h[i])
            h[i] += a if d > 0 else -a
            d = target - h.sum()
            if abs(d) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return x.sum() - t

    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[ix] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[ix] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm = n - 1
    for ii in range(len(ix)):

        if wk < 1e-15:

        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

    try:

                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cd_pass(h, rng, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cm = 2 * n - 1
    ol = np.arange(n)
    rng.shuffle(ol)
    nm = n - 1
    for j in ol:
        j = int(j)
        b = np.zeros(cm)
        l1 = j
        h1 = min(cm - 1, j + nm)
        lg = np.arange(l1, h1 + 1) - nm
        b[l1:h1 + 1] += 1.0 - h[j - lg]
        l2 = max(0, nm - j)
        h2 = min(cm - 1, cm - 1 - j)
        lg = np.arange(l2, h2 + 1) - nm
        b[l2:h2 + 1] -= h[j + lg]
        g = c - b * h[j]
        a_, b_ = 0.0, 1.0
        for _ in range(35):
            t1 = a_ + (b_ - a_) / 3.0
            t2 = b_ - (b_ - a_) / 3.0
            if np.max(g + b * t1) <= np.max(g + b * t2):
                b_ = t2
            else:
                a_ = t1
        t = 0.5 * (a_ + b_)
        if abs(t - h[j]) > 1e-14:
            c += b * (t - h[j])
            h[j] = t
    return h

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _obj(x, al, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(al * (c[ix] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / al

def _grad(x, al, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(al * (c[ix] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    nm = n - 1
    for ii in range(len(ix)):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        k = int(ix[ii]); L = k - nm
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, al, ml, mi, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (0.5 * n,)}]
    try:
        r = minimize(_obj, h.copy(), jac=_grad, args=(al, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

def _lbfgs(h, al, ml, n):

                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': 200, 'gtol': 1e-10})
        return _repair(r.x, n)
    except Exception:
        return h

def _local(h, n, rng, ni, te):

                bc = nc
                if nc < _eval(bh):
                    bh = h.copy()
            else:
                h[a], h[b] = va - d, vb + d

        g = c - b * h[j]
        a_, b_ = 0.0, 1.0
        for _ in range(35):
            t1 = a_ + (b_ - a_) / 3.0
            t2 = b_ - (b_ - a_) / 3.0
            if np.max(g + b * t1) <= np.max(g + b * t2):
                b_ = t2
            else:
                a_ = t1
        t = 0.5 * (a_ + b_)
        if abs(t - h[j]) > 1e-14:
            c += b * (t - h[j])
            h[j] = t
    return h

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _obj(x, al, ml):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(al * (c[ix] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / al
