        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp = h
            h = xn
            fh = fx
            stale = 0
            Lc = max(Lc * 0.8, 1.0)
            if it % 12 == 0:
                c = _c5(h)
                if c < bc:
                    bc = c
                    bh = h.copy()
        else:
            stale += 1
            hp, y, tv = h, h, 1.0
            fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 20:
                break
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return h, bh, bc

def _loc(h0, rng, deadline, s0=5e-5, use_blocks=True):
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    t0 = time.time()
    step = s0
    no_imp = 0
    total_imp = 0
    it = 0
    while time.time() < deadline - 1.0:
        fr = (time.time() - t0) / max(1e-9, deadline - t0 - 1.0)
        base = max(1e-12, s0 * (3e-6) ** fr)
        r = rng.random()
        did_something = False
        if use_blocks and r < 0.25:
            bl = rng.randint(2, 5)
            i = rng.randint(0, n - bl)
            j = rng.randint(0, n - bl)
            if j < i:
                j += bl
                if j + bl > n:
                    j = rng.randint(0, n - bl)
            if j >= i + bl:
                sgn = 1 if h[i] >= 1.0 - h[j] else -1
                if sgn > 0:
                    avail = min((1.0 - h[i:i+bl]).min(), h[j:j+bl].min())
                else:
                    avail = min(h[i:i+bl].min(), (1.0 - h[j:j+bl]).min())
                if avail > 1e-14:
                    d = sgn * min(avail * 0.7, base * (1.0 + rng.random()))
                    h[i:i+bl] += d
                    h[j:j+bl] -= d
                    cn = _c5(h)
                    if cn < c - 1e-15:
                        c = cn
                        no_imp = 0
                        step = min(step * 1.05, 0.05)
                        if c < bc:
                            bc = c
                            bh = h.copy()
                    else:
                        h[i:i+bl] -= d
                        h[j:j+bl] += d
                        no_imp += 1
                        step *= 0.97
                    did_something = True
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                sgn = 1 if h[i] >= 1.0 - h[j] else -1
                avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
                if avail > 1e-14:
                    mult = 0.3 + 1.7 * rng.random()
                    if rng.random() < 0.03:
                        mult *= 5.0
                    d = sgn * min(avail, base * mult)
                    h[i] += d
                    h[j] -= d
                    cn = _c5(h)
                    if cn < c - 1e-15:
                        c = cn
                        no_imp = 0
                        step = min(step * 1.05, 0.05)
                        if c < bc:
                            bc = c
                            bh = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
                        no_imp += 1
                        step *= 0.97
                    did_something = True
        if no_imp > 5000:
            for _ in range(rng.randint(2, 6)):
                k1 = rng.randint(n)
                k2 = rng.randint(n)
                if k1 != k2:
                    h[k1] += base * 3.0 * rng.randn()
                    h[k2] -= base * 3.0 * rng.randn()
            h = _repair(h, n)
            c = _c5(h)
            no_imp = 0
        it += 1
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def _loc_fine(h0, rng, deadline, s0=2e-6):
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    t0 = time.time()
    it = 0
    while time.time() < deadline - 1.0:
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            sgn = 1 if h[i] >= 1.0 - h[j] else -1
            avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                d = sgn * min(avail, s0 * (0.5 + rng.random()))
                h[i] += d
                h[j] -= d
                cn = _c5(h)
                if cn < c - 1e-16:
                    c = cn
                    if c < bc:
                        bc = c
                        bh = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
        it += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)

    for var in [best[::-1], 0.5 + 0.5 * (best - best[::-1]), 0.5 + 0.5 * (best + best[::-1] - 1.0)]:
        hv = _proj(var, n0 * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    end1 = t0 + 80.0
    hc = best.copy()
    for a in [50, 200, 800]:
        rem = end1 - time.time()
        if rem < 3:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.3)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    end2 = t0 + 140.0
    for a in [3000, 8000]:
        rem = end2 - time.time()
        if rem < 5:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.45)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    end3 = t0 + 700.0
    best, best_c = _loc(best, rng, end3, s0=4e-5, use_blocks=True)

    end4 = t0 + 900.0
    best, best_c = _loc(best, rng, end4, s0=8e-6, use_blocks=True)

    end5 = t0 + budget_s - 10.0
    best, best_c = _loc_fine(best, rng, end5, s0=1.5e-6)

    n = len(best)
    for var in [best[::-1], 0.5 + 0.5 * (best - best[::-1]), 0.5 + 0.5 * (best + best[::-1] - 1.0)]:
        hv = _proj(var, n * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
