    # Phase 1: Multi-resolution FISTA (first 30%)
    end1 = t0 + 0.30 * budget_s
    ns_try = [120, 160, 200, 240, 320, 400, 480, 560]
    ns_try = [x for x in ns_try if abs(x - n0) > 40 or x == n0]
    if n0 not in ns_try: ns_try.append(n0)
    ns_try.sort()
    slot = max(2.0, (end1 - time.time()) / len(ns_try))
    for m in ns_try:
        if time.time() >= end1 - 2: break
        hm = _resample(best, m)
        hcur = hm.copy()
        for a in [50, 500, 5000, 50000]:
            if time.time() >= end1 - 1: break
            hf, bh, bc = _fista(hcur, a, slot / 4.0, 10.0)
            cf = _c5(hf)
            if bc < best_c:
                hb = _resample(bh, n0); best = hb; best_c = _c5(hb)
            if cf < best_c:
                hb = _resample(hf, n0); best = hb; best_c = _c5(hb)
            hcur = hf if cf <= bc else bh
    # Phase 2: Intensive FISTA at n0 (next 30%)
    end2 = t0 + 0.60 * budget_s
    hmain = best if len(best) == n0 else _resample(best, n0)
    alphas = [100, 500, 2000, 8000, 30000, 100000, 300000, 1000000]
    tt_per = max(1.0, (end2 - time.time()) / len(alphas))
    hcur = hmain.copy()
    for a in alphas:
        if time.time() >= end2 - 2: break
        hf, bh, bc = _fista(hcur, a, tt_per, 10.0)
        cf = _c5(hf)
        if bc < best_c: best = bh; best_c = bc
        if cf < best_c: best = hf; best_c = cf
        hcur = hf if cf <= bc else bh
    # Phase 3: Greedy lag optimization (next 20%)
    end3 = t0 + 0.80 * budget_s
    if end3 > time.time():
        best, best_c = _greedy_lag(best, end3, rng)
    # Phase 4: Block SA (next 15%)
    end4 = t0 + 0.95 * budget_s
    if end4 > time.time():
        best, best_c = _sa_blocks(best, rng, end4)
    # Phase 5: Final greedy polish
    end5 = t0 + budget_s - 1.0
    if end5 > time.time():
        best, best_c = _greedy_lag(best, end5, rng)
    # Final variants
    for hh in (1.0 - best, best[::-1].copy()):
        h2 = _proj(hh, len(hh) * 0.5); c = _c5(h2)
        if c < best_c: best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
