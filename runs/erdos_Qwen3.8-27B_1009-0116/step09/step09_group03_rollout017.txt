import numpy as np
import time
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    n = len(h)
    lo, hi = -4.0, 4.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha, ml):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-16:
            continue
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _pg_loop(h, alpha, ml, n_steps, lr0, mom, t0, t_end):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    lr = lr0
    stall = 0
    for it in range(n_steps):
        if time.time() - t0 > t_end:
            break
        g = _grad_smooth(h, alpha, ml)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = mom * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        c_cur = _eval(h)
        c_new = _eval(h_new)
        if c_new < c_cur - 1e-14:
            h = h_new
            stall = 0
            lr = min(lr * 1.03, 0.05)
        else:
            step = 1.0
            accepted = False
            for _ in range(8):
                h_try = _proj(h - step * lr * g / gn * gn, n * 0.5)
                c_try = _eval(h_try)
                if c_try < c_cur:
                    h = h_try
                    v = mom * v
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                h = h_new
                lr = max(lr * 0.6, 1e-7)
                stall += 1
            else:
                stall = 0
        c_now = _eval(h)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
        if stall > 80:
            break
    return best_h, best_c

def _local_search(h, n_iter, alpha, ml, t0, t_end):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    c_cur = best_c
    g = _grad_smooth(h, alpha, ml)
    for it in range(n_iter):
        if time.time() - t0 > t_end:
            break
        if it % 20 == 0:
            g = _grad_smooth(h, alpha, ml)
        ip = np.argmin(g)
        im = np.argmax(g)
        if ip == im:
            continue
        dmax = min(1.0 - h[ip], h[im])
        if dmax < 1e-14:
            continue
        for frac in [1.0, 0.5, 0.25, 0.1, 0.04, 0.01]:
            d = dmax * frac
            h[ip] += d
            h[im] -= d
            c_try = _eval(h)
            if c_try < c_cur - 1e-15:
                c_cur = c_try
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
                break
            else:
                h[ip] -= d
                h[im] += d
    return best_h, best_c

def _objs(x, active_lags, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, len(active_lags))[:len(active_lags)]
    vals = c[idx]
    s = 50000.0 * (vals - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / 50000.0

    def _rem():
