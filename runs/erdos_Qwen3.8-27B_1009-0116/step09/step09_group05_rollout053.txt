def _adam_proj_grad(h, alpha, m_lags, steps, lr):
    """Adam with projection onto feasible set."""
    n = len(h)
    target = n / 2.0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_h = h.copy()
    best_c = _eval(h)
    for t in range(1, steps + 1):
        g = _smooth_grad(h, alpha, m_lags)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        # Project step direction onto sum=0 subspace
        step -= step.mean()
        # Line search
        taken = False

                    best_h = h.copy()
                taken = True
                break

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        shift = k - (n - 1)
        if shift >= 0:
            L = shift
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -shift
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _coef_slope(k, j, n, h):
    """Slope of corr[k] w.r.t. h[j], excluding the k==2j quadratic term."""
    if k < 0 or k >= 2 * n - 1:
        return 0.0
    diff = k - j
    if diff < 0 or diff >= n:
        return 0.0
    if k == 2 * j:
        return 0.0
    return (1.0 - 2.0 * h[diff]) * (2.0 / n)

def _coord_sweep(h, n_sweeps, m_lags, topk_frac=1.0):
    n = len(h)
    ml = min(m_lags, int(2 * n - 1))
    best_h = h.copy()
    best_c = _eval(h)
    for sweep in range(n_sweeps):
        c = _corr(h)
        top_idx = np.argsort(c)[-ml:]
        improved = False
        for j in range(n):
            t0 = h[j]
            slopes = np.array([_coef_slope(int(k), j, n, h) for k in top_idx])
            quad = np.array([1.0 if int(k) == 2 * j else 0.0 for k in top_idx]) * (2.0 / n)
            base = c[top_idx].copy()
            best_t = t0
            best_val = base.max()
            for t_try in [0.0, 0.25, 0.5, 0.75, 1.0, t0]:
                new_vals = base + slopes * (t_try - t0) + quad * (t_try * (1 - t_try) - t0 * (1 - t0))
                mv = new_vals.max()
                if mv < best_val - 1e-15:
                    best_val = mv
                    best_t = t_try
            if best_t != t0:
                lo, hi2 = 0.0, 1.0
                if abs(t0) < 1e-9:
                    lo = 0.0
                if abs(t0 - 1.0) < 1e-9:
                    hi2 = 1.0
                for _ in range(30):
                    mid = 0.5 * (lo + hi2)
                    new_vals = base + slopes * (mid - t0) + quad * (mid * (1 - mid) - t0 * (1 - t0))
                    mv = new_vals.max()
                    if mv < best_val:
                        best_val = mv
                        best_t = mid
                    left = 0.5 * (lo + mid)
                    right = 0.5 * (mid + hi2)
                    vleft = (base + slopes * (left - t0) + quad * (left * (1 - left) - t0 * (1 - t0))).max()
                    vright = (base + slopes * (right - t0) + quad * (right * (1 - right) - t0 * (1 - t0))).max()
                    if vleft < vright:
                        hi2 = mid
                    else:
                        lo = mid
                if best_t != t0:
                    h[j] = best_t
                    improved = True
        if not improved:
            break
        cv = _eval(h)
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
    return best_h, best_c

def _adam_proj(h, alpha, m_lags, steps, lr):
    n = len(h)
    target = n / 2.0
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_h = h.copy()
    best_c = _eval(h)
    for t in range(1, steps + 1):
        g = _smooth_grad(h, alpha, m_lags)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        mh = m1 / (1 - b1 ** t)
        vh = m2 / (1 - b2 ** t)
        step = -lr * mh / (np.sqrt(vh) + eps)
        step -= step.mean()
        taken = False
        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            h_new = _proj(h + mult * step, target)
            c_new = _eval(h_new)
            if c_new < best_c + 1e-14:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                taken = True
                break
    return best_h, best_c

def _slsqp_opt(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _grad_2pt(h, n_iter, alpha, m_lags, t_end):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha, m_lags)
        gp = g - g.mean()
        i = int(np.argmax(gp))
        j = int(np.argmin(gp))
        if i == j:
            continue
        base = 0.01 * (gp[i] - gp[j])
        for frac in [0.1, 0.3, 0.5, 0.8, 1.0, 1.5]:
            d = base * frac
            hi, hj = h[i] + d, h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h[i], h[j] = hi, hj
            cn = _eval(h)
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
                break
        if h[i] != hi:
            h[i], h[j] = hi - (h[i] - (hi - (h[i] - (hi - d)))), hj + (h[j] - (hj - (h[j] - (hj - d))))
    return best_h, best_c

def _fine_2pt(h, t_end, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    h2 = h.copy()
    c5c = best_c
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 3e-7 * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c5c:
                c5c = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h2.copy()
            else:
                h2[i], h2[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)
