import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(v, target):
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0, 1)

def _obj_smooth(x, alpha, m_lags):
    c = _corr_all(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_smooth(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_all(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            if L == 0:
                g[:] += wk * (1.0 - 2.0 * x)
            else:
                g[:n-L] += wk * (1.0 - x[L:])
                g[L:] -= wk * x[:n-L]
        else:
            M = -L
            g[M:] += wk * (1.0 - x[:n-M])
            g[:n-M] -= wk * x[M:]
    return g

def _con_fun(x, target):
    return x.sum() - target

def _con_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _pg_nesterov(h, alpha, m_lags, steps, step0):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    x = h.copy()
    mu = 0.7
    step = step0
    cur_c = _eval(x)
    best_x = x.copy()
    best_c = cur_c
    for it in range(steps):
        g = _grad_smooth(x, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-18:
            break
        v /= ng
        taken = False
        for _t in range(12):
            x_new = _proj(x - step * v, target)
            c_new = _eval(x_new)
            if c_new < cur_c - 1e-17:
                x = x_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 30.0)
                taken = True
                if c_new < best_c:
                    best_c = c_new
                    best_x = x.copy()
                break
            step *= 0.3
        if not taken:
            step = min(max(step * 1.3, step0 * 0.5), step0 * 30.0)
            if step < step0 * 0.01:
                break
    return best_x, best_c

def _subgrad_hard(h, steps, step0):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    cur_c = _eval(x)
    best_x = x.copy()
    best_c = cur_c
    dx = 2.0 / n
    for it in range(steps):
        c = _corr_all(x)
        m = c.max()
        thresh = m - max(5e-6, 0.005 * m)
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([c.argmax()])
        g = np.zeros(n)
        for k in active:
            L = k - (n - 1)
            if L >= 0:
                if L == 0:
                    g[:] += dx * (1.0 - 2.0 * x)
                else:
                    g[:n-L] += dx * (1.0 - x[L:])
                    g[L:] -= dx * x[:n-L]
            else:
                M = -L
                g[M:] += dx * (1.0 - x[:n-M])
                g[:n-M] -= dx * x[M:]
        g /= len(active)
        ng = np.linalg.norm(g)
        if ng < 1e-18:
            break
        g /= ng
        step = step0 / (1.0 + 0.005 * it)
        x_new = _proj(x - step * g, target)
        c_new = _eval(x_new)
        if c_new < cur_c:
            cur_c = c_new
            x = x_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
    return best_x, best_c

def _med_search(h, n_iter, rng, step_lo, step_hi, m_lags):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    cur_c = _eval(x)
    best_x = x.copy()
    best_c = cur_c
    dx = 2.0 / n
    for _ in range(n_iter):
        step = step_lo + (step_hi - step_lo) * rng.rand()
        c = _corr_all(x)
        idx = np.argsort(c)[-m_lags:]
        g = np.zeros(n)
        for k in idx:
            L = k - (n - 1)
            if L >= 0:
                if L == 0:
                    g[:] += dx * (1.0 - 2.0 * x)
                else:
                    g[:n-L] += dx * (1.0 - x[L:])
                    g[L:] -= dx * x[:n-L]
            else:
                M = -L
                g[M:] += dx * (1.0 - x[:n-M])
                g[:n-M] -= dx * x[M:]
        g /= m_lags
        nn = rng.choice([4, 6, 8, 10, 14])
        pos = rng.choice(n, size=nn, replace=False)
        d = step * rng.randn(nn)
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-15:
            for i, p in enumerate(pos):
                d[i] -= 0.5 * step * g[p] / gnorm
        d -= d.mean()
        x_new = x.copy()
        x_new[pos] += d
        if x_new.min() < -0.15 or x_new.max() > 1.15:
            continue
        x_new = _repair(x_new, n)
        c_new = _eval(x_new)
        if c_new < cur_c - 1e-16:
            cur_c = c_new
            x = x_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
    return best_x, best_c

def _fine_polish(h, n_iter, rng, scale_lo=1e-9, scale_hi=8e-8):
    n = len(h)
    x = h.copy()
    cur_c = _eval(x)
    best_x = x.copy()
    best_c = cur_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 24)) % n
        if i == j:
            j = (j + 1) % n
        scale = scale_lo + (scale_hi - scale_lo) * rng.rand()
        d = scale * rng.randn()
        ni, nj = x[i] + d, x[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            x[i], x[j] = ni, nj
            c_new = _eval(x)
            if c_new < cur_c - 1e-17:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_x = x.copy()
            else:
                x[i], x[j] = x[i] - d, x[j] + d
    return best_x, best_c
