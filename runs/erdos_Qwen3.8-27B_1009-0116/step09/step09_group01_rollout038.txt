def _fft_corr(h):
    n = int(len(h))
    M = 2 * n - 1
    N = 1
    while N < M:
        N <<= 1
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    R = np.fft.irfft(H * np.conj(B), N)
    c = np.empty(M)
    c[n - 1] = R[0]
    if n > 1:
        c[:n - 1] = R[N - (n - 1):N]
        c[n:] = R[1:n]
    return c

    if abs(d) > 1e-15:
        if d > 0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, x[idx] + d)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + d)
    return x


def _active_lags(c, limit=48, tol=1.5):
    cmax = float(c.max())
    idx = np.nonzero(c >= cmax - tol)[0]
    if len(idx) > limit:
        sub = c[idx]
        idx = idx[np.argpartition(sub, -limit)[-limit:]]
    return idx


def _grad_active(h, active, center):
    n = len(h)
    g = np.zeros(n)
    for idx in active:
        t = int(idx - center)
        if t >= 0:
            L = n - t
            g[:L] += 1.0 - h[t:]
            g[t:] -= h[:L]
        else:
            s = -t
            L = n - s
            g[s:] += 1.0 - h[:L]
            g[:L] -= h[s:]
    return g


def _fill_K(om, h, K, i, j):
    n = len(h)
    K.fill(0.0)
    a0 = n - 1 - i
    b0 = n - 1 - j
    K[a0:a0 + n] += om
    K[b0:b0 + n] -= om
    K[i:i + n] -= h[::-1]
    K[j:j + n] += h[::-1]


def _eval_move(c, K, d, center, ip, im, work, delta):
    d2 = d * d
    np.multiply(K, d, out=delta)
    np.add(c, delta, out=work)
    work[center] -= 2.0 * d2
    work[ip] += d2
    work[im] += d2
    return float(work.max())


def _binarize(h):
    n = len(h)
    out = np.zeros(n)
    if n:
        k = n // 2
        idx = np.argpartition(h, -k)[-k:]
        out[idx] = 1.0
    return out


def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, 0.5 * m)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), 0.5 * m)


def _local_cont(hin, end, temp0=0.0, base0=0.25, seed=0):
    t0 = time.time()
    total = end - t0
    if total <= 0.0:
        h = _proj(np.asarray(hin, dtype=float).copy(), 0.5 * len(hin))
        return h, float(_fft_corr(h).max())
    h = _proj(np.asarray(hin, dtype=float).copy(), 0.5 * len(hin))
    n = len(h)
    om = 1.0 - h
    c = _fft_corr(h)
    cm = float(c.max())
    best_h = h.copy()
    best_c = cm
    work = np.empty_like(c)
    K = np.empty_like(c)
    delta = np.empty_like(c)
    center = n - 1
    rng = np.random.RandomState(seed)
    it = 0
    noimp = 0
    acc = 0
    g = None
    lo = hi = None

    while time.time() < end - 0.2:
        if it and (it % 4096 == 0 or noimp > 40000):
            c[:] = _fft_corr(h)
            cm = float(c.max())
            if cm < best_c:
                best_c = cm
                best_h = h.copy()
            if noimp > 40000:
                h[:] = best_h
                om[:] = 1.0 - h
                c[:] = _fft_corr(h)
                cm = float(c.max())
                for _ in range(4):
                    a = int(rng.randint(n))
                    b = int(rng.randint(n))
                    if h[b] <= h[a]:
                        a, b = b, a
                    d = min(om[a], h[b], 0.02 + 0.08 * rng.random())
                    if d > 1e-8:
                        h[a] += d
                        h[b] -= d
                        om[a] -= d
                        om[b] += d
                cn = _fft_corr(h)
                mc = float(cn.max())
                if mc < best_c + 3.0:
                    c[:] = cn
                    cm = mc
                else:
                    h[:] = best_h
                    om[:] = 1.0 - h
                    c[:] = _fft_corr(h)
                    cm = float(c.max())
                noimp = 0
                it = 0
                g = None

        if g is None or acc >= 12:
            active = _active_lags(c, 48, max(1.5, 0.02 * cm))
            g = _grad_active(h, active, center)
            q = max(16, n // 12)
            glo = g.copy()
            glo[om <= 1e-7] = np.inf
            lo = np.argsort(glo)[:q]
            ghi = g.copy()
            ghi[h <= 1e-7] = -np.inf
            hi = np.argsort(ghi)[-q:][::-1]
            acc = 0

        if lo is None or len(lo) == 0:
            break

        if rng.random() < 0.18:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                i = int(lo[rng.randint(len(lo))])
                j = int(hi[rng.randint(len(hi))])
            elif h[j] <= h[i]:
                i, j = j, i
        else:
            i = int(lo[rng.randint(len(lo))])
            j = int(hi[rng.randint(len(hi))])

        if i == j or h[j] <= h[i] + 1e-9:
            noimp += 1
            it += 1
            continue

        dmax = float(om[i] if om[i] < h[j] else h[j])
        if dmax <= 1e-10:
            noimp += 1
            it += 1
            continue

        _fill_K(om, h, K, i, j)
        ip = center + j - i
        im = center + i - j
        frac = max(0.0, (end - time.time()) / total)
        base = max(1e-6, base0 * (0.05 + 0.95 * frac))
        s = abs(g[i] - g[j]) if g is not None else 1.0
        ds = (dmax, 0.5 * dmax, 0.25 * dmax, min(base, dmax), min(0.05 / (s + 0.1), dmax))

        bd = 0.0
        bnm = 1e100
        for d in ds:
            if d <= 1e-11 or d > dmax + 1e-12:
                continue
            _eval_move(c, K, d, center, ip, im, work, delta)
            nm = float(work.max())
            if nm < bnm:
                bnm = nm
                bd = d

        accept = bd > 0.0 and bnm < cm - 1e-12
        if not accept and bd > 0.0 and temp0 > 0.0:
            temp = temp0 * frac * frac
            if temp > 1e-6 and bnm < cm + max(0.5, temp):
                if rng.random() < math.exp(-(bnm - cm) / temp):
                    accept = True

        if accept:
            _eval_move(c, K, bd, center, ip, im, work, delta)
            h[i] += bd
            h[j] -= bd
            om[i] -= bd
            om[j] += bd
            c, work = work, c
            cm = bnm
            acc += 1
            noimp = 0
            if cm < best_c:
                best_c = cm
                best_h = h.copy()
        else:
            noimp += 1
        it += 1

    c[:] = _fft_corr(h)
    cm = float(c.max())
    if cm < best_c:
        best_c = cm
        best_h = h.copy()
    return best_h, best_c


def _subgrad(hin, end, seed=0):
    h = _proj(np.asarray(hin, dtype=float).copy(), 0.5 * len(hin))
    n = len(h)
    c = _fft_corr(h)
    cm = float(c.max())
    best_h = h.copy()
    best_c = cm
    rng = np.random.RandomState(seed)
    S = 0.5 * n
    center = n - 1
    stale = 0

    while time.time() < end - 0.3:
        active = _active_lags(c, 32, max(1.0, 0.02 * cm))
        g = _grad_active(h, active, center)
        if active.size:
            g = g / float(active.size)
        nrm = float(np.dot(g, g)) ** 0.5
        if nrm < 1e-9:
            break

        best_eta = -1.0
        best_cm = cm - 1e-12
        best_hv = None
        best_cnv = None
        for eta in (0.3, 0.15, 0.07, 0.03, 0.015, 0.007):
            x = _proj(h - eta * g, S)
            cn = _fft_corr(x)
            ncm = float(cn.max())
            if ncm < best_cm:
                best_cm = ncm
                best_eta = eta
                best_hv = x
                best_cnv = cn

        if best_eta > 0.0:
            h = best_hv
            c = best_cnv
            cm = best_cm
            stale = 0
            if cm < best_c:
                best_c = cm
                best_h = h.copy()
        else:
            improved = False
            for _ in range(6):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if h[j] <= h[i]:
                    i, j = j, i
                d = min(1.0 - h[i], h[j], 0.002 + 0.01 * rng.random())
                if d <= 1e-8:
                    continue
                h[i] += d
                h[j] -= d
                cn = _fft_corr(h)
                ncm = float(cn.max())
                if ncm < cm - 1e-12:
                    c = cn
                    cm = ncm
                    improved = True
                    break
                h[i] -= d
                h[j] += d
            if improved:
                stale = 0
                if cm < best_c:
                    best_c = cm
                    best_h = h.copy()
            else:
                stale += 1
                if stale > 25:
                    break

    cn = _fft_corr(h)
    ncm = float(cn.max())
    if ncm < best_c:
        best_h = h
        best_c = ncm
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    deadline = t0 + B - 4.0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n = 560
        a = np.arange(n)
        v = np.sin(2.0 * np.pi * 0.6180339887 * a) + 0.5 * np.sin(2.0 * np.pi * 0.3819660113 * a + 0.7)
        h0 = np.zeros(n)
        h0[np.argsort(v)[-n // 2:]] = 1.0

    n = len(h0)
    if n % 2:
        n -= 1
        h0 = h0[:n]
    if n < 10:
        n = 560
        h0 = np.zeros(n)
        h0[rng.permutation(n)[:n // 2]] = 1.0

    best_h = _proj(h0, 0.5 * n)
    best_n = n
    best_cmax = float(_fft_corr(best_h).max())
    best_c5 = best_cmax * 2.0 / best_n

    for v in (1.0 - best_h, best_h[::-1]):
        vv = _proj(v, 0.5 * n)
        cv = float(_fft_corr(vv).max()) * 2.0 / n
        if _better(cv, n, best_c5, best_n):
            best_h = vv
            best_c5 = cv
            best_n = n

    end = t0 + 0.32 * B
    if time.time() < end:
        hv, cv = _local_cont(best_h, end, 0.0, 0.30, seed)
        c5v = cv * 2.0 / len(hv)
        if _better(c5v, len(hv), best_c5, best_n):
            best_h, best_c5, best_n = hv, c5v, len(hv)

    end = t0 + 0.42 * B
    if time.time() < end:
        hv, cv = _subgrad(best_h, end, seed + 55)
        c5v = cv * 2.0 / len(hv)
        if _better(c5v, len(hv), best_c5, best_n):
            best_h, best_c5, best_n = hv, c5v, len(hv)

    end = t0 + 0.62 * B
    if time.time() < end:
        hb = _binarize(best_h)
        hv, cv = _local_cont(hb, end, 2.0, 0.25, seed + 101)
        c5v = cv * 2.0 / len(hv)
        if _better(c5v, len(hv), best_c5, best_n):
            best_h, best_c5, best_n = hv, c5v, len(hv)

    end = t0 + 0.80 * B
    if time.time() < end:
        hv, cv = _local_cont(best_h, end, 0.0, 0.06, seed + 202)
        c5v = cv * 2.0 / len(hv)
        if _better(c5v, len(hv), best_c5, best_n):
            best_h, best_c5, best_n = hv, c5v, len(hv)

    sizes = (280, 420)
    if B > 900:
        sizes = (280, 420, 700)
    for m in sizes:
        if m == len(best_h):
            continue
        if time.time() > deadline - 150:
            break
        rem = deadline - time.time()
        alloc = min(50.0, rem * 0.35)
        if alloc < 25.0:
            continue
        hm = _resample(best_h, m)
        endm = time.time() + alloc
        hv, cv = _local_cont(hm, endm, 0.0, 0.15, seed + m)
        c5v = cv * 2.0 / m
        if _better(c5v, m, best_c5, best_n):
            best_h, best_c5, best_n = hv, c5v, m

    end = deadline - 3.0
    if time.time() < end:
        hv, cv = _local_cont(best_h, end, 0.0, 0.02, seed + 999)
        c5v = cv * 2.0 / len(hv)
        if _better(c5v, len(hv), best_c5, best_n):
            best_h, best_c5, best_n = hv, c5v, len(hv)

    best_h = _proj(best_h, 0.5 * len(best_h))
    c5v = float(_fft_corr(best_h).max()) * 2.0 / len(best_h)
    return best_h, c5v, len(best_h)
