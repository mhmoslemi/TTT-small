import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a if d > 0 else -a
    return h

def _lag_gradient(h, shift, n):
    """d(corr_shift)/dh, vector of length n"""
    g = np.ones(n)
    if shift == 0:
        g -= 2.0 * h
    elif shift > 0:
        g[shift:] -= h[:-shift]
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] -= h[s:]
        g[s:] -= h[:-s]
    return g

def _subgradient(h, n, k_active=5, tol_frac=0.02):
    """Average subgradient of max_k corr_k"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    C = corr * dx
    Cmax = C.max()
    active = np.where(C >= Cmax - Cmax * tol_frac)[0]
    if len(active) == 0:
        active = np.array([np.argmax(C)])
    # Cap active set size
    if len(active) > k_active:
        # Keep the top k_active
        top = np.argsort(C)[::-1][:k_active]
        active = top
    grad = np.zeros(n)
    for idx in active:
        shift = idx - (n - 1)
        grad += _lag_gradient(h, shift, n)
    grad *= dx / len(active)
    return grad, Cmax, active

def _cutting_plane(h0, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    h = h0.copy()
    S = n * 0.5
    h = _proj(h, S)
    best_c = _c5(h)
    best_h = h.copy()
    
    momentum = np.zeros(n)
    beta = 0.9
    alpha = 0.5
    stall = 0
    
    while time.time() - t0 < time_limit:
        grad, Cmax, active = _subgradient(h, n, k_active=5)
        
        if Cmax < best_c - 1e-12:
            best_c = Cmax
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Momentum step
        momentum = beta * momentum - alpha * grad
        h_try = _proj(h + momentum, S)
        c_try = _c5(h_try)
        
        if c_try < Cmax:
            h = h_try
            alpha = min(alpha * 1.05, 5.0)
        else:
            # Backtracking
            a = alpha
            found = False
            for _ in range(10):
                a *= 0.5
                h_try = _proj(h - a * grad, S)
                if _c5(h_try) < Cmax:
                    h = h_try
                    found = True
                    break
            if found:
                alpha = a
            else:
                alpha = a
                h = _proj(h - a * grad, S)
        
        # Random kick every 500 iterations or on long stall
        if stall > 500 and rng.random() < 0.01:
            h = _proj(h + 0.01 * rng.randn(n), S)
            stall = 0
    
    h = _repair(h, n)
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def _coord_descent(h0, n, time_limit, seed):
    """Coordinate descent: optimize one coordinate at a time"""
    rng = np.random.RandomState(seed)
    t0 = time.time()
    h = h0.copy()
    S = n * 0.5
    best_c = _c5(h)
    best_h = h.copy()
    order = rng.permutation(n)
    
    while time.time() - t0 < time_limit:
        for _ in range(n):
            i = order[_ % n]
            # Optimal h[i] given others: we want to minimize max_k corr_k
            # For a 1D search, try a few values
            orig = h[i]
            # Try: 0, 0.25, 0.5, 0.75, 1, and values near orig
            candidates = [0.0, 1.0, orig + 0.1, orig - 0.1, orig + 0.05, orig - 0.05]
            candidates = [c for c in candidates if 0 <= c <= 1]
            best_val = orig
            best_c_i = _c5(h)
            for cv in candidates:
                h[i] = cv
                # Fix sum
                deficit = S - h.sum()
                if abs(deficit) > 1e-14:
                    # Distribute deficit to another coordinate
                    j = (i + 1) % n
                    adj = np.clip(deficit, -h[j], 1.0 - h[j])
                    h[j] += adj
                    # If adj != deficit, the constraint isn't met exactly
                    # This is a problem...

                    # Let's skip this and use a different approach
                    pass
                c_i = _c5(h)
                if c_i < best_c_i:
                    best_c_i = c_i
                    best_val = cv
                h[i] = cv if cv != orig else orig
            h[i] = best_val
            # Fix sum
            deficit = S - h.sum()
            if abs(deficit) > 1e-14:
                h[i] = best_val - deficit
                h[i] = np.clip(h[i], 0, 1)
                # Fix the rest
                h = _proj(h, S)
        
        c = _c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
