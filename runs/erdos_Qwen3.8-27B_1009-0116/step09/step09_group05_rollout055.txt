    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sm_max(c, m_lags):
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    return m, idx

def _obj(x, alpha, m_lags, mu):
    c = _corr_full(x)
    m, idx = _sm_max(c, m_lags)
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    sm = m + np.log(np.exp(s).sum()) / alpha
    ssum = x.sum() - len(x) * 0.5
    return sm + mu * ssum * ssum

def _grad(x, alpha, m_lags, mu):
    n = len(x)
    c = _corr_full(x)
    m, idx = _sm_max(c, m_lags)
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(len(idx), 1.0 / len(idx))
    else:
        w /= ws
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    ssum = x.sum() - n * 0.5
    g += 2.0 * mu * ssum
    return g

def _lbfgs(h, alpha, m_lags, mu, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha, m_lags, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, steps, step0):

def _coord2(h, n_iter, rng, base_step):

    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
