    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    # Get top K lags
    idx = np.argsort(c)[-K:]

            g[:n-L] -= wk * h[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - h[M:])

        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _csum_fun(x, target):
    return x.sum() - target

def _csum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csum_fun, 'jac': _csum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slexp_obj, h, jac=_slexp_grad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    cur_c5 = best_c5
    steps = np.array([3e-7, 1e-6, 3e-6, 1e-5, 3e-5, 1e-4])
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        sz = steps[rng.randint(len(steps))]
        d = sz * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < cur_c5:
                cur_c5 = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _cd_block(h, n, rng, nb, n_passes, t_deadline, perturb_every=20, repair_every=10):
    passes = 0
    for _ in range(n_passes):
        if time.time() > t_deadline:
            break
        h = _cd_pass(h, rng, nb)
        passes += 1
        if passes % repair_every == 0:
            h = _repair(h, n)
        if passes % perturb_every == 0 and time.time() < t_deadline - 5:
            idx_p = rng.choice(n, size=5, replace=False)
            h[idx_p] += 0.004 * rng.randn(5)
            h = _repair(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)

    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: CD with perturbations (35% time)
    p1_end = t0 + budget_s * 0.35
    h = _cd_block(h, n, rng, 50, 500, p1_end, 20, 10)
    h = _repair(h, n)
    c5 = _eval(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Multi-start perturbation test (10% time)
    p2_end = t0 + budget_s * 0.45
    for sigma in [0.01, 0.02, 0.04]:
        if time.time() > p2_end - 10:
            break
        hp = best_h + sigma * rng.randn(n)
        hp = _repair(hp, n)
        hp = _cd_block(hp, n, rng, 40, 40, p2_end - 10, 15, 8)
        hp = _repair(hp, n)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 3: SLSQP sweep (30% time)
    p3_end = t0 + budget_s * 0.75
    alphas = [5000, 20000, 80000, 300000, 1000000, 4000000, 15000000]
    for alpha in alphas:
        if time.time() > p3_end - 15:
            break
        h_new = _slsqp(best_h.copy(), alpha, 90, 1200)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
        if time.time() < p3_end - 25:
            h_new = _cd_pass(best_h, rng, 40)
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Extended CD (15% time)
    p4_end = t0 + budget_s * 0.90
    h2 = best_h.copy()
    h2 = _cd_block(h2, n, rng, 50, 400, p4_end, 25, 12)
    h2 = _repair(h2, n)
    c5 = _eval(h2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h2.copy()

    # Phase 5: Final SLSQP (5% time)
    if time.time() < t0 + budget_s * 0.95:
        h_new = _slsqp(best_h.copy(), 50000000, 120, 600)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
        h_new = _cd_pass(best_h, rng, 50)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 6: Local search (5% time)
    remaining = t0 + budget_s - 3 - time.time()
    if remaining > 1:
        ntr = int(remaining / 0.006)
        ntr = min(ntr, 150000)
        best_h, best_c5 = _local_search(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
