import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

        hi = h[i]
        for m in range(2 * n - 1):
            L = m - (n - 1)
            coeff = 0.0
            # h[i] as first factor: p=i, need i+L in [0,n-1]
            if 0 <= i + L < n:
                coeff += (1.0 - h[i + L])
                c_rest[m] -= hi * (1.0 - h[i + L])
            # h[i] in second factor: p=i-L, need i-L in [0,n-1], L!=0
            if L != 0 and 0 <= i - L < n:
                coeff += (-h[i - L])
                c_rest[m] -= h[i - L] * (1.0 - hi)
            a[m] = coeff
        # Find candidate t values
        # For L=0 (m=n-1): c_0(t) = c_rest[n-1] + a[n-1]*t - t^2
        # For other L: c_L(t) = c_rest[m] + a[m]*t
        # Candidates: 0, 1, current, parabola vertex, quadratic roots with top lags
        cands = [0.0, 1.0, hi]
        m0 = n - 1
        # Parabola vertex
        if abs(a[m0]) > 1e-15:
            tv = a[m0] / 2.0
            if 0 <= tv <= 1:
                cands.append(tv)
        # Top lags by current value
        top_idx = np.argsort(c_rest + a * hi)[-n_top:]
        for m in top_idx:
            if m == m0:
                continue
            # Solve: c_rest[0]+a[0]*t - t^2 = c_rest[m]+a[m]*t
            # t^2 + (a[m]-a[m0])*t + (c_rest[m]-c_rest[m0]) = 0
            A2 = 1.0
            B2 = a[m] - a[m0]
            C2 = c_rest[m] - c_rest[m0]
            disc = B2 * B2 - 4 * C2
            if disc >= 0:
                sq = math.sqrt(disc)
                for t in [(-B2 + sq) / 2, (-B2 - sq) / 2]:
                    if -0.01 <= t <= 1.01:
                        cands.append(max(0.0, min(1.0, t)))
        cands = sorted(set(np.clip(cands, 0, 1).round(14)))
        best_t = hi
        best_c = np.inf
        for t in cands:
            # Evaluate g(t) = max_m (c_rest[m] + a[m]*t - [m==m0]*t^2)
            vals = c_rest + a * t
            vals[m0] -= t * t
            c_val = vals.max() * 2.0 / n
            if c_val < best_c:
                best_c = c_val
                best_t = t
        if abs(best_t - hi) > 1e-15:
            h[i] = best_t
            h = _repair(h, n)
            if _eval(h) < c5c:
                c5c = _eval(h)
        elif abs(best_t - hi) > 1e-15:
            h[i] = hi  # revert
            h = _repair(h, n)
    return h, _eval(h)

def _proj_grad(h, alpha, m_lags, nsteps, step0):
    n = len(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(nsteps):
        g = _smooth_grad(h, alpha, m_lags)
        g -= g.mean()
        v = 0.88 * v + 0.12 * g
        sq = 0.95 * sq + 0.05 * g * g
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h, cur_c = h_new, c_new
                step = min(step * 1.1, step0 * 25)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 25)
    return _repair(h, n)

def _sa_2pt(h, n_iters, rng, t_start, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        d = 0.015 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        delta = c5n - c5c
        if delta < 0 or rng.rand() < math.exp(-delta / max(t, 1e-12)):
            c5c = c5n
            if c5n < best_c:
                best_c, best_h = c5n, h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _fine_2pt(h, n_iters, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iters):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, min(n - 1, 12))) % n
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _expired(s=45):
        return time.time() - t0 > budget_s - s

    # Phase 1: ICM coordinate descent (fast, pushes toward structure)
    for p in range(30):
        if _expired(80):
            break
        h, c5v = _icm_pass(h, rng, n_top=50)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: SLSQP smooth optimization
    alpha_seq = [5000, 20000, 80000, 300000, 1200000, 5000000, 20000000, 80000000]
    for alpha in alpha_seq:
        if _expired(400):
            break
        lift = min(100, max(25, int(30 * math.log10(alpha / 5000) + 25)))
        h_new = _slsqp(best_h.copy(), alpha, lift, 800, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new

    # Phase 3: ICM + projected gradient interleaved
    h = best_h.copy()
    for round_i in range(6):
        if _expired(250):
            break
        h, _ = _icm_pass(h, rng, n_top=60)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()
        alpha = 50000 * (10 ** round_i)
        h = _proj_grad(h.copy(), alpha, min(80, 30 + 10 * round_i), 300, 0.003 / (1 + round_i))
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 4: Simulated annealing
    if not _expired(150):
        n_sa = min(int((budget_s - time.time() + t0 - 80) / 0.004), 120000)
        best_h, best_c5 = _sa_2pt(best_h.copy(), n_sa, rng, 0.008, 1e-7)

    # Phase 5: More ICM passes on best
    h = best_h.copy()
    for p in range(40):
        if _expired(60):
            break
        h, c5v = _icm_pass(h, rng, n_top=70)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 6: Fine 2-point polish
    if not _expired(30):
        n_fine = min(int((budget_s - time.time() + t0 - 5) / 0.0008), 400000)
        best_h, best_c5 = _fine_2pt(best_h.copy(), n_fine, rng)

    # Phase 7: Final SLSQP touch
    if not _expired(15):
        h_new = _slsqp(best_h.copy(), 500000000, 120, 400, 1e-17)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
