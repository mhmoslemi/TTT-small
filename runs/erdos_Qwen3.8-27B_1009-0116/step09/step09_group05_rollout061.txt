def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lse_obj(x, alpha, mlags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _lse_grad(x, alpha, mlags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)

    g = np.zeros(n)
    for ii, k in enumerate(idx):

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lse_obj(x, alpha, mlags, lam):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    pen = lam * (x.sum() - len(x) / 2.0) ** 2
    return m + np.log(np.exp(s).sum()) / alpha + pen

def _lse_grad(x, alpha, mlags, lam):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    g += 2.0 * lam * (x.sum() - n / 2.0) * np.ones(n)
    return g

def _slsqp(h, alpha, mlags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, mlags, 0.0), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _lbfgs(h, alpha, mlags, maxiter, lam):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, mlags, lam), method='L-BFGS-B',
                       bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': maxiter * 10})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _grad_2pt(h, n_iter, alpha, mlags, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        g = _lse_grad(h, alpha, mlags, 0.0)
        g_proj = g - g.mean()
        i = int(np.argmax(g_proj))
        j = int(np.argmin(g_proj))
        if i == j:
            continue
        best_d = 0.0
        best_c5 = c5c
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            d = frac * (g_proj[i] - g_proj[j]) * 0.01
            h[i] += d
            h[j] -= d
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= d
                h[j] += d
                continue
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_d = d
            else:
                h[i] -= d
                h[j] += d
                break
        if best_d != 0.0:
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c
                best_h = h.copy()
    return best_h, best_c

def _micro_search(h, n_iter, rng, step_range):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        step = step_range[0] + rng.rand() * (step_range[1] - step_range[0])
        idx = rng.randint(0, n, size=2)
        d = step * rng.randn(2)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx[0]] += d[0]
        h_new[idx[1]] += d[1]
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = -min(-diff, h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-13:
                break
    return h

    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    pen = lam * (x.sum() - len(x) * 0.5) ** 2
    return m + np.log(np.exp(s).sum()) / alpha + pen

def _lse_grad(x, alpha, mlags, lam):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    if lam > 0:
        g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, mlags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, mlags, 0.0), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, alpha, mlags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, mlags, 1e5), method='L-BFGS-B',
                       bounds=bnds,

                                'gtol': 1e-14, 'maxls': 40,
                                'maxfun': maxiter * 5})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _proj_grad(h, alpha, mlags, t_end):
    n = len(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = 0.01
    cur_c = _eval(h)
    while time.time() < t_end:
        g = _lse_grad(h, alpha, mlags, 0.0)
        v = 0.8 * v + 0.2 * g
        sq = 0.95 * sq + 0.05 * g * g
        d = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _ in range(12):
            h_new = _proj(h - step * d, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, 0.1)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, 0.1)
    return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _grad_2pt(h, t_end, alpha, mlags):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() < t_end:
        g = _lse_grad(h, alpha, mlags, 0.0)
        gp = g - g.mean()
        i = int(np.argmax(gp))
        j = int(np.argmin(gp))
        if i == j:
            break
        best_d = 0.0
        best_c5 = c5c
        for frac in (0.1, 0.3, 0.5, 0.7, 0.9):
            d = frac * (gp[i] - gp[j]) * 0.01
            old_i, old_j = h[i], h[j]
            h[i] = old_i + d
            h[j] = old_j - d
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i], h[j] = old_i, old_j
                continue
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_d = d
            else:
                h[i], h[j] = old_i, old_j
                break
        if best_d != 0.0:
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c
                best_h = h.copy()

def _micro3(h, t_end, rng, smin, smax):

def _restarts(h_best, n, rng, t_end, n_restarts):

    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, mlags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, mlags, 0.0), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, alpha, mlags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, mlags, 1e5), method='L-BFGS-B',
                       bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxls': 40,
                                'maxfun': maxiter * 5})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _quick_opt(h, alpha, mlags, n_steps):
    n = len(h)
    step = 0.005
    cur_c = _eval(h)
    for _ in range(n_steps):
        g = _lse_grad(h, alpha, mlags, 0.0)
        h_new = _proj(h - step * g, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.5
    return _repair(h, n)

def _proj_grad(h, alpha, mlags, t_end):
    n = len(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = 0.01
    cur_c = _eval(h)
    while time.time() < t_end:
        g = _lse_grad(h, alpha, mlags, 0.0)
        v = 0.8 * v + 0.2 * g
        sq = 0.95 * sq + 0.05 * g * g
        d = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _ in range(12):
            h_new = _proj(h - step * d, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, 0.1)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, 0.1)
    return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _restarts(h_best, n, rng, t_end, n_r):
    best_h = h_best.copy()
    best_c = _eval(h_best)
    for r in range(n_r):
        if time.time() > t_end:
            break
        if r < n_r // 2:
            pert = 0.06 * rng.randn(n)
            pert -= pert.mean()
            cand = _repair(np.clip(h_best + pert, 0, 1), n)
        else:
            base = 0.5 + 0.15 * rng.randn(n)
            cand = _repair(np.clip(0.4 * h_best + 0.6 * base, 0, 1), n)
        cand = _quick_opt(cand, 5000, 25, 40)
        cand = _slsqp(cand, 10000, 30, 200, 1e-14)
        c5c = _eval(cand)
        if c5c < best_c:
            best_c = c5c
            best_h = cand.copy()
    return best_h, best_c

def _sa(h_start, n, rng, t_end):
    ps = 5
    pop = [h_start.copy()]
    for i in range(1, ps):
        pert = 0.05 * rng.randn(n)
        pert -= pert.mean()
        cand = _repair(np.clip(h_start + pert, 0, 1), n)
        cand = _quick_opt(cand, 5000, 25, 30)
        cand = _slsqp(cand, 5000, 25, 80, 1e-13)
        pop.append(cand)
    best_h = min(pop, key=_eval)
    best_c = _eval(best_h)
    T0, Tf = 4e-4, 1e-8
    gen = 0
    stall = 0
    while time.time() < t_end:
        T = max(T0 * (Tf / T0) ** (gen / 300.0), Tf)
        new_pop = [best_h.copy()]
        for i in range(ps - 1):
            if time.time() > t_end:
                new_pop.append(pop[i].copy())
                continue
            if rng.random() < 0.15 and len(pop) > 1:
                j = rng.randint(0, min(len(pop), 3))
                child = 0.5 * (pop[i] + pop[j])
            else:
                sig = max(T * 300, 3e-5)
                d = sig * rng.randn(n)
                d -= d.mean()
                child = pop[i] + d
            child = _repair(np.clip(child, 0, 1), n)
            child = _quick_opt(child, 5000, 25, 30)
            child = _slsqp(child, 3000, 20, 60, 1e-13)
            cc = _eval(child)
            cp = _eval(pop[i])
            if cc < cp or (T > 1e-9 and rng.random() < np.exp(-(cc - cp) / T)):
                new_pop.append(child)
            else:
                new_pop.append(pop[i].copy())
        pop = new_pop
        gb = min(pop, key=_eval)
        gc = _eval(gb)
        if gc < best_c - 1e-10:
            best_c = gc
            best_h = gb.copy()
            stall = 0
        else:
            stall += 1
            if stall > 35:
                d = 0.03 * rng.randn(n)
                d -= d.mean()
                pt = _repair(np.clip(best_h + d, 0, 1), n)
                pt = _slsqp(pt, 5000, 25, 100, 1e-13)
                pop[-1] = pt
                stall = 0
        gen += 1
    return best_h, best_c

def _grad_2pt(h, t_end, alpha, mlags):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() < t_end:
        g = _lse_grad(h, alpha, mlags, 0.0)
        gp = g - g.mean()
        i = int(np.argmax(gp))
        j = int(np.argmin(gp))
        if i == j:
            break
        bd = 0.0
        bc5 = c5c
        for frac in (0.1, 0.3, 0.5, 0.7, 0.9):
            d = frac * (gp[i] - gp[j]) * 0.01
            oi, oj = h[i], h[j]
            h[i] = oi + d
            h[j] = oj - d
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i], h[j] = oi, oj
                continue
            c5n = _eval(h)
            if c5n < bc5:
                bc5 = c5n
                bd = d
            else:
                h[i], h[j] = oi, oj
                break
        if bd != 0.0:
            c5c = bc5
            if c5c < best_c:
                best_c = c5c
                best_h = h.copy()
    return best_h, best_c

def _micro3(h, t_end, rng, smin, smax):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() < t_end:
        st = smin + rng.random() * (smax - smin)
        i, j, k = rng.randint(0, n, 3)
        if i == j or j == k or i == k:
            continue
        d = st * rng.randn(3)
        d -= d.mean()
        oi, oj, ok = h[i], h[j], h[k]
        h[i] += d[0]
        h[j] += d[1]
        h[k] += d[2]
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1 or h[k] < 0 or h[k] > 1:
            h[i], h[j], h[k] = oi, oj, ok
            continue
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j], h[k] = oi, oj, ok
            h = _repair(h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution coarse (40s)
    for ratio in (4, 8):
        if time.time() - t0 > 38:
            break
        nc = max(40, n // ratio)
        f = n // nc
        hc = _downsample(h, f)
        for alpha in (1000, 10000, 100000):
            if time.time() - t0 > 45:
                break
            hc = _slsqp(hc, alpha, 20, 300, 1e-14)
        h_up = _upsample(hc, f)
        if _eval(h_up) < best_c5:
            best_c5 = _eval(h_up)
            best_h = h_up.copy()

    # Phase 2: SLSQP alpha cascade (150s)
    for alpha in (2000, 10000, 50000, 200000, 1000000, 5000000, 30000000):
        if time.time() - t0 > 190:
            break
        lift = min(100, max(15, int(30 * np.log10(alpha / 2000) + 15)))
        hn = _slsqp(best_h.copy(), alpha, lift, 700, 1e-16)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = hn

    # Phase 3: L-BFGS-B (80s)
    if time.time() - t0 < 270:
        for alpha in (5000, 50000, 500000):
            if time.time() - t0 > 270:
                break
            hn = _lbfgs(best_h.copy(), alpha, 40, 3000)
            c5n = _eval(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn

    # Phase 4: Random restarts (60s)
    if time.time() - t0 < 330:
        best_h, best_c5 = _restarts(best_h, n, rng, t0 + 340, 8)

    # Phase 5: Simulated annealing population (200s)
    if time.time() - t0 < 530:
        best_h, best_c5 = _sa(best_h, n, rng, t0 + 540)

    # Phase 6: Projected gradient (100s)
    if time.time() - t0 < 630:
        for alpha, ml in ((50000, 40), (500000, 60), (5000000, 80)):
            if time.time() - t0 > 630:
                break
            hn = _proj_grad(best_h.copy(), alpha, ml, t0 + 640)
            c5n = _eval(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn

    # Phase 7: High-alpha SLSQP (50s)
    if time.time() - t0 < 690:
        for alpha in (100000000, 1000000000, 5000000000):
            if time.time() - t0 > 695:
                break
            hn = _slsqp(best_h.copy(), alpha, 80, 500, 1e-17)
            c5n = _eval(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn

    # Phase 8: SLSQP after SA improvements
    if time.time() - t0 < budget_s - 120:
        for alpha in (50000, 500000, 5000000):
            if time.time() - t0 > budget_s - 115:
                break
            hn = _slsqp(best_h.copy(), alpha, 50, 500, 1e-16)
            c5n = _eval(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn

    # Phase 9: Gradient 2-point (budget-60)
    if time.time() - t0 < budget_s - 60:
        best_h, best_c5 = _grad_2pt(best_h, t0 + budget_s - 30, 1000000, 60)

    # Phase 10: Micro 3-point search (budget-10)
    if time.time() - t0 < budget_s - 10:
        best_h, best_c5 = _micro3(best_h, t0 + budget_s - 3, rng, 5e-8, 2e-6)

    # Phase 11: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 0.5:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        while time.time() < t0 + budget_s - 0.3:
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 6)) % nc2
            d = 1e-8 * rng.randn()
            oi, oj = h2[i], h2[j]
            h2[i] = oi + d
            h2[j] = oj - d
            if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = oi, oj
            else:
                h2[i], h2[j] = oi, oj

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
