def _project(h, step, g, S):
    """Project h - step*g onto {0<=x<=1, sum(x)=S}"""
    x = h - step * g
    # Find offset t such that sum(clip(x+t, 0, 1)) = S
    # clip(x+t, 0, 1) = min(max(x+t, 0), 1)
    # Let f(t) = sum(min(max(x_i+t, 0), 1))
    # f is piecewise linear, increasing in t
    # Find t such that f(t) = S
    
    lo = -x.max()  # f(lo) = 0
    hi = 1 - x.min()  # f(hi) = n
    for _ in range(60):
        mid = (lo + hi) / 2
        val = np.clip(x + mid, 0, 1).sum()
        if val < S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    x = np.clip(x + t, 0, 1)
    # Fine adjustment
    d = S - x.sum()
    if abs(d) > 1e-12:
        if d > 0:
            free = (x < 1.0)
            if free.sum() > 0:
                x[free] += d / free.sum()
        else:
            free = (x > 0.0)
            if free.sum() > 0:
                x[free] += d / free.sum()
        np.clip(x, 0, 1, out=x)
    return x
