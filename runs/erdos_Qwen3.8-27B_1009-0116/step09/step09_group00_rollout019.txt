        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn(h):
    """Compute all overlap values (positive and negative lags and zero)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # length 2n-1

def _soft_val(h, a):
    corr = _pn(h)
    n = len(h)
    dx = 2.0 / n
    mx = float(corr.max())
    e = np.exp(np.clip(a * (corr - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a)
