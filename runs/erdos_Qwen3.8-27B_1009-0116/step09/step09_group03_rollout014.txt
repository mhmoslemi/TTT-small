        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _lbfgs_opt(h, alpha, topk, maxiter):
    n = len(h)
    rho = 1.0
    for _ in range(12):
        args = (alpha, topk, rho, n)
        res = minimize(_aug_obj, h, jac=_aug_grad, args=args,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        h = _repair(res.x, n)
        if abs(h.sum() - n * 0.5) < 1e-10:
            break
        rho *= 10.0
    return h

def _pair_opt(h, npairs, alpha):
    n = len(h)
    for _ in range(30):
        g = _sm_grad(h, alpha, 60)
        order = np.argsort(g)
        moved = False
        for p in range(min(npairs, n // 2)):
            i, j = order[p], order[n - 1 - p]
            if g[i] - g[j] < 1e-12:
                break
            ci = h[i] + h[j]
            h[i] = max(0.0, min(1.0, ci - 1.0))
            h[j] = max(0.0, min(1.0, ci))
            moved = True
        if not moved:
            break
    return _repair(h, n)

def _nesterov(h, alpha, topk, steps, lr0, mom):
    n = len(h)
    v = np.zeros(n)
    bh_c = _eval(h); bh_h = h.copy()
    for _ in range(steps):
        g = _sm_grad(h, alpha, topk)
        v = mom * v - lr0 * g
        h_try = _proj_sum(h + v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < bh_c:
            bh_c = c_try; bh_h = h_try.copy()
        step = lr0; taken = False
        for _ in range(12):
            h_t = _proj_sum(h - step * g, n * 0.5)
            if _eval(h_t) < _eval(h):
                h = h_t; v = mom * v; taken = True; break
            step *= 0.5
        if not taken:
            h = h_try
    return bh_h

def _downsample(h, f):
    n = len(h); m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h); out = np.repeat(h, f)
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse multi-resolution optimization
    if _el() < 80:
        for ratio in [4, 7]:
            if _el() > 60:
                break
            nc = max(32, n // ratio)
            f = n // nc
            hc = _downsample(h, f)
            for ac in [1e5, 1e6, 1e7]:
                if _el() > 70:
                    break
                hc = _lbfgs_opt(hc, ac, 30, 400)
            h_up = _upsample(hc, f)
            c5u = _eval(h_up)
            if c5u < best_c5:
                best_c5 = c5u; best_h = h_up.copy()

    # Phase 2: L-BFGS-B alpha sweep at full resolution
    alpha_seq = [1e4, 5e4, 2e5, 1e6, 5e6, 2e7, 1e8, 5e8, 2e9, 1e10]
    for alpha in alpha_seq:
        if _el() > budget_s - 250:
            break
        topk = min(120, max(30, int(15 + 25 * np.log10(max(alpha, 1)))))
        h_new = _lbfgs_opt(best_h.copy(), alpha, topk, 400)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new.copy()

    # Phase 3: Pair optimization passes
    if _el() < budget_s - 120:
        for alpha_p in [1e6, 1e8, 1e10]:
            if _el() > budget_s - 100:
                break
            for _ in range(30):
                if _el() > budget_s - 90:
                    break
                h_new = _pair_opt(best_h.copy(), 15, alpha_p)
                c5n = _eval(h_new)
                if c5n < best_c5:
                    best_c5 = c5n; best_h = h_new.copy()
                else:
                    break

    # Phase 4: Nesterov projected gradient
    if _el() < budget_s - 180:
        pg_cfgs = [
            (1e6, 60, 1000, 0.004, 0.7),
            (1e7, 80, 800, 0.003, 0.8),
            (1e8, 100, 600, 0.002, 0.85),
            (1e9, 150, 500, 0.001, 0.9),
            (1e10, 200, 400, 0.0005, 0.92),
        ]
        for alpha, topk, steps, lr0, mom in pg_cfgs:
            if _el() > budget_s - 100:
                break
            h_new = _nesterov(best_h.copy(), alpha, topk, steps, lr0, mom)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new.copy()

    # Phase 5: Reheat + re-optimize
    if _el() < budget_s - 100:
        for attempt in range(10):
            if _el() > budget_s - 50:
                break
            ns = 5e-5 * (10 - attempt)
            h_p = best_h + ns * rng.randn(n)
            h_p = _repair(h_p, n)
            alpha_r = [1e6, 1e8, 1e10][attempt % 3]
            h_new = _lbfgs_opt(h_p, alpha_r, 80, 300)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new.copy()

    # Phase 6: High-alpha L-BFGS squeeze
    if _el() < budget_s - 60:
        for alpha_h in [5e9, 5e10, 5e11]:
            if _el() > budget_s - 30:
                break
            h_new = _lbfgs_opt(best_h.copy(), alpha_h, 200, 600)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new.copy()

    # Phase 7: Systematic 2-point sweep
    if _el() < budget_s - 10:
        h2 = best_h.copy()
        c5c = _eval(h2)
        while _el() < budget_s - 2:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            d = 5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5 = c5c; best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
