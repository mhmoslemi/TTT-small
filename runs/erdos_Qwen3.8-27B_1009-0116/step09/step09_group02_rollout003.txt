import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            a = min(abs(diff), min(h[idx] if diff < 0 else 1.0 - h[idx], abs(diff)))
            h[idx] += a if diff > 0 else -a
            diff += a if diff > 0 else -a
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def sse_obj(x, alpha, mL):
    c = corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-mL:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def sse_grad(x, alpha, mL):
    n = len(x)
    dx = 2.0 / n
    c = corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-mL:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def con_fun(x, target):
    return x.sum() - target

def con_jac(x, target):
    return np.ones(len(x))

def slsqp_solve(h, alpha, mL, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    con = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (tgt,)}]
    try:
        res = minimize(sse_obj, h, jac=sse_grad, args=(alpha, mL),
                       method='SLSQP', bounds=bnds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return h

def proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def proj_gradient(h, alpha, mL, nsteps, s0):
    n = len(h)
    v = np.zeros(n)
    mu = 0.82
    step = s0
    cur = eval_c5(h)
    for _ in range(nsteps):
        g = sse_grad(h, alpha, mL)
        g = (g - g.mean())
        ng = np.linalg.norm(g)
        if ng < 1e-16:
            break
        g /= ng
        v = mu * v + (1 - mu) * g
        v /= (np.linalg.norm(v) + 1e-16)
        ok = False
        for _ in range(15):
            hn = proj(h - step * v, n * 0.5)
            cn = eval_c5(hn)
