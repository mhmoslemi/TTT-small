def _eval(h):

    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _repair(h, n):

    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    # Find positions with room
    if diff > 0:
        room = 1.0 - h
    else:
        room = h
    total_room = room.sum()
    if total_room < abs(diff):

    h += diff * room / total_room

        else:

This suggests that the optimal h should be close to a {0,1}-valued function (a "hard" threshold), not a smooth function. But with the smoothness of the correlate, there's a tradeoff.

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    room = (1.0 - h) if diff > 0 else h.copy()
    tr = room.sum()
    if tr < abs(diff) * 0.999:
        return _project(h, target)
    h += diff * room / tr
    return np.clip(h, 0, 1)

def _grad(h, m, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-m:]
    ct = c[idx] - c[idx].max()
    s = np.clip(alpha * ct, -400, 400)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(m, 1.0 / m)
    else:
        w /= ws
    g = np.zeros(n)
    for ii in range(m):
        wi = w[ii]
        if wi < 1e-14:
            continue
        L = idx[ii] - (n - 1)
        if L >= 0:
            if L < n:
                g[L:] += wi * (1.0 - h[:n - L])
                g[:n - L] -= wi * h[L:]
        else:
            M = -L
            if M < n:
                g[:n - M] += wi * (1.0 - h[M:])
                g[M:] -= wi * h[:n - M]
    return g * dx, c.max()
