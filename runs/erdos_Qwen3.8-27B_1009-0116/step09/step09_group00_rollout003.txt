def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

    s = h.sum()

        np.clip(h, 0.0, 1.0, out=h)

        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:

            break

        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * (2.0 / n), corr

    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]

    pref[0] = 0.0

    Nv = np.zeros(n)

        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]

        cn[1:] = np.cumsum(wN[1:])

    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):

        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2*n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _grad_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _active_grad(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-8, 8.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0; cnt += int(pos.size)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0; cnt += int(pos.size)
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv; wN *= inv
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t, L0=20.0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha)
    best_h = h.copy(); best_c = _c5(h)[0]
    stale = 0; Lc = L0
    for it in range(800000):
        if time.time() - t0 > max_t:
            break
        if it % 400 == 399:
            y = h.copy(); t = 1.0; fy = _soft_val(h, alpha)
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum() - S) > 0.5:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(16):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn
            Lc = max(Lc * 0.8, 1.0)
            stale = 0
            if it % 20 == 0:
                c = _c5(h)[0]
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0
            Lc = max(Lc * 0.5, 1.0)
            if stale > 20:
                break
    c = _c5(h)[0]
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _subgrad_desc(h0, max_t, n_act=8):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c, _ = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    lr = 0.02
    for it in range(50000):
        if time.time() - t0 > max_t:
            break
        P, Nv, mx = _pn(h)
        g = _active_grad(h, P, Nv)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = 0
        for trial in range(12):
            xn = _proj(h - lr * g / (gn + 1e-15) * lr * 100, S)
            cn, _ = _c5(xn)
            if cn <= c + 1e-12:
                d = trial
                h = xn
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
                c = cn
                break
            lr *= 0.6
        else:
            break
        if it % 50 == 49:
            lr = min(lr * 1.1, 0.08)
    return h, best_h, best_c

def _local_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    c, _ = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    it = 0
    recompute_at = 512
    P = Nv = None
    g = np.zeros(n)
    low = high = None
    q = max(8, n // 6)
    while time.time() < deadline:
        if it % recompute_at == 0:
            P, Nv, _ = _pn(h)
            g = _active_grad(h, P, Nv)
            order = np.argsort(g)
            low = order[:q]; high = order[-q:]
            recompute_at = it + 512
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        base = max(1e-10, 2e-4 * (5e-4) ** frac)
        if rng.random() < 0.75:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            it += 1; continue
        s = 1 if g[i] <= g[j] else (-1 if g[j] < g[i] else (1 if rng.random() < 0.5 else -1))
        avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.5 + 2.0 * rng.random()
            if rng.random() < 0.03:
                mult *= 8.0
            d = s * min(avail, base * mult)
            h[i] += d; h[j] -= d
            cn, _ = _c5(h)
            if cn < c - 1e-15:
                c = cn
                if c < best_c:
                    best_c = c; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _sa_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    _, corr = _c5(h)
    mx = float(corr.max())
    best_c = mx * (2.0 / n); best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 1.5 * (0.003) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        if rng.random() < 0.8 and abs(k_act) >= 1 and abs(k_act) < n:
            k = abs(k_act)
            if k_act > 0:
                valid = (h[:n - k] > 0.005) & (h[k:] < 0.995)
            else:
                valid = (h[k:] > 0.005) & (h[:n - k] < 0.995)
            nvalid = int(valid.sum())
            if nvalid > 2:
                pos = np.nonzero(valid)[0]
                i = int(pos[rng.randint(len(pos))])
                j = i + k if k_act > 0 else i - k
                if 0 <= j < n:
                    d = (rng.random() - 0.5) * 0.05
                    if h[i] + d < 0: d = -h[i]
                    if h[i] + d > 1: d = 1.0 - h[i]
                    if h[j] - d < 0: d = h[j]
                    if h[j] - d > 1: d = h[j] - 1.0
                    if abs(d) > 1e-10:
                        h[i] += d; h[j] -= d
                        c2 = np.correlate(h, 1.0 - h, mode='full')
                        mx2 = float(c2.max())
                        delta = mx2 - mx
                        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                            mx = mx2; corr = c2
                            cn = mx * (2.0 / n)
                            if cn < best_c:
                                best_c = cn; best_h = h.copy()
                        else:
                            h[i] -= d; h[j] += d
                    continue
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            continue
        d = (rng.random() - 0.5) * 0.06
        if h[i] + d < 0: d = -h[i]
        if h[i] + d > 1: d = 1.0 - h[i]
        if h[j] - d < 0: d = h[j]
        if h[j] - d > 1: d = h[j] - 1.0
        if abs(d) < 1e-10:
            continue
        h[i] += d; h[j] -= d
        c2 = np.correlate(h, 1.0 - h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            mx = mx2; corr = c2
            cn = mx * (2.0 / n)
            if cn < best_c:
                best_c = cn; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c, _ = _c5(best)
    for hh in (1.0 - h0.copy(), 0.5 + 0.5 * (h0 - h0[::-1])):
        h2 = _proj(hh, n0 * 0.5)
        c2, _ = _c5(h2)
        if c2 < best_c:
            best = h2; best_c = c2
    # Phase 1: FISTA moderate alpha (0-100s)
    end1 = t0 + min(100.0, 0.10 * budget_s)
    if end1 > time.time():
        hcur = best.copy()
        alphas_p1 = [50, 200, 1000, 5000]
        tt = (end1 - time.time()) / len(alphas_p1)
        for a in alphas_p1:
            if time.time() >= end1 - 1:
                break
            hf, bh, bc = _fista(hcur, a, tt, 20.0)
            cf, _ = _c5(hf)
            if bc < best_c:
                best = bh; best_c = bc
            if cf < best_c:
                best = hf; best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 2: FISTA high alpha (100-380s)
    end2 = t0 + min(380.0, 0.36 * budget_s)
    if end2 > time.time():
        hcur = best.copy()
        alphas_p2 = [10000, 50000, 200000, 600000, 1500000]
        tt = (end2 - time.time()) / len(alphas_p2)
        for a in alphas_p2:
            if time.time() >= end2 - 2:
                break
            hf, bh, bc = _fista(hcur, a, tt, 20.0)
            cf, _ = _c5(hf)
            if bc < best_c:
                best = bh; best_c = bc
            if cf < best_c:
                best = hf; best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 3: Subgradient descent (380-580s)
    end3 = t0 + min(580.0, 0.55 * budget_s)
    if end3 > time.time():
        hf, bh, bc = _subgrad_desc(best, end3 - time.time() - 2)
        cf, _ = _c5(hf)
        if bc < best_c:
            best = bh; best_c = bc
        if cf < best_c:
            best = hf; best_c = cf
    # Phase 4: Local search (580-850s)
    end4 = t0 + min(850.0, 0.80 * budget_s)
    if end4 > time.time():
        best, best_c = _local_search(best, rng, end4)
    # Phase 5: SA (850-1050s)
    end5 = t0 + min(1050.0, 0.96 * budget_s)
    if end5 > time.time():
        best, best_c = _sa_search(best, rng, end5)
    # Final: quick local + repair
    end6 = t0 + budget_s - 3.0
    if end6 > time.time():
        best, best_c = _local_search(best, rng, end6)
    best = _repair(best, len(best))
    best_c, _ = _c5(best)
    return best, best_c, len(best)
