    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:

                diff -= a

                h[idx] -= a
                diff += a

Actually, I want to add one more innovation: **exact active-set method**. 

After the smooth optimization converges, I'll:
1. Identify the exact active set A = {k : O_k = C₅} (within machine precision)
2. For each active k, compute grad(O_k)
3. The descent direction is the minimum-norm subgradient: d = (1/|A|) * sum_{k in A} grad(O_k)
4. Project d to be orthogonal to the constraint (sum = n/2): d -= mean(d) * ones
5. Take a small step: h_new = proj(h - ε*d, n/2)
6. Check if C₅ decreased; if not, halve ε
7. Repeat

This is cleaner than the log-sum-exp approach for the final fine-tuning.

Let me also add: **simulated annealing restarts**. Every so often, add a small random perturbation, re-optimize with SLSQP for a few iterations, and keep the best. This helps escape local minima.

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def get_overlaps(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def shift_grad(h, k, n):
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def active_subgrad(h, tol=1e-10):
    n = len(h)
    c = get_overlaps(h, n)
    thresh = c.max() - tol
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        g += shift_grad(h, k, n)
    g /= len(active)
    g -= g.mean()
    return g, len(active)

def smooth_obj(x, alpha, ml):
    n = len(x)
    c = get_overlaps(x, n)
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def smooth_grad(x, alpha, ml):
    n = len(x)
    c = get_overlaps(x, n)
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * shift_grad(x, k, n)
    g *= (2.0 / n)
    return g

def sum_fn(x, half):
    return x.sum() - half

def sum_jac(x, half):
    return np.ones(len(x))

def slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac, 'args': (n*0.5,)}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 100})
        return repair(res.x, n)
    except Exception:
        return h

def proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def active_descent(h, n_iter, tol, step0):
    n = len(h)
    best_h = h.copy()
    best_c = eval_c5(h)
    c_cur = best_c
    step = step0
    for _ in range(n_iter):
        g, _ = active_subgrad(h, tol)
        if np.linalg.norm(g) < 1e-15:
            break
        s = step
        improved = False
        for _ in range(20):
            h_try = proj(h - s * g, n * 0.5)
            c_try = eval_c5(h_try)
            if c_try < c_cur - 1e-16:
                h = h_try
                c_cur = c_try
                step = min(s * 1.15, step0 * 30)
                improved = True
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
                break
            s *= 0.5
        if not improved:
            step = max(step * 0.6, 1e-12)
    return best_h, best_c

def fine_perturb(h, n_iter, rng, alpha, ml):
    n = len(h)
    best_h = h.copy()
    best_c = eval_c5(h)
    c_cur = best_c
    g = smooth_grad(h, alpha, ml)
    g -= g.mean()
    for _ in range(n_iter):
        top = np.argsort(-np.abs(g))[:30]
        i = top[rng.randint(len(top))]
        j = top[(i + 1 + rng.randint(0, 20)) % len(top)]
        if i == j:
            j = (i + 3) % n
        d = 5e-6 * rng.randn()
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = repair(h_new, n)
        c5n = eval_c5(h_new)
        if c5n < c_cur:
            h = h_new
            c_cur = c5n
            g = smooth_grad(h, alpha, ml)
            g -= g.mean()
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def downsample(h, f):
    n = len(h)
    m = n // f
    return repair(h[:m*f].reshape(m, f).mean(axis=1).copy(), m)
