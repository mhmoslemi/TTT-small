def _c5(h):

    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    for _ in range(60):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14: break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else a
            if abs(d) < 1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)

        if abs(d) < 1e-13: break

        if c == 0: break

    return x

    while N < m: N <<= 1
    return N
