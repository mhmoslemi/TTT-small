def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n)), int(np.argmax(corr)) - (n - 1)

def _repair(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = S - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return h

def _subgrad(h, active_lag):
    """Compute subgradient of c_k at the given lag."""
    n = len(h)
    k = abs(active_lag)
    if k == 0:
        g = np.zeros(n)
        # c_0 = sum h[i]*(1-h[i]) - sum h[i] = sum h[i] - 2*sum h[i]^2
        # dc_0/dh[i] = 1 - 2*h[i]
        g[:] = 1.0 - 2.0 * h
        return g
    # For lag k > 0 (positive): c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # dc_k/dh[j] = (1-h[j+k]) * 1_{j < n-k} + (-h[j-k]) * 1_{k <= j < n}
    # For lag -k: c_{-k} = sum_{j=k}^{n-1} h[j]*(1-h[j-k])
    # dc_{-k}/dh[j] = (1-h[j-k]) * 1_{j >= k} + (-h[j+k]) * 1_{j < n-k}
    g = np.zeros(n)
    if active_lag > 0:
        # c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]

    g = np.zeros(len(h))
    for k in active_lags:
        g += _subgrad(h, k)
    g /= len(active_lags)
    return g

    n = len(h0)
    h = _repair(h0.copy(), n)
    best_h = h.copy()
    best_c, best_k = _c5(h)
    t0 = time.time()
    step = alpha_init / np.sqrt(1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    tol = 1e-6
    active = np.where(corr >= mx - tol)[0] - (n - 1)
    
    it = 0
    stagnation = 0
    while time.time() < deadline:
        # Compute objective and active set
        c, _ = _c5(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Recompute active set every 10 iters

            tol = max(1e-6, 0.01 * mx)

        h_new = _repair(h - step * g, n)
        h = h_new
        step *= 0.999  # Slow decay
        it += 1
    
    return best_h, best_c

    n = len(h0)
    h = _repair(h0.copy(), n)
    best_h = h.copy()
    best_c = _c5(h)
    t0 = time.time()

        if c < best_c:

        # Active set
        mx = float(corr.max())
        tol = max(1e-8, 0.05 * (2.0/n))
        active = np.where(corr >= mx - tol/n)[0] - (n-1)
        if len(active) == 0:

            if c_new < c - 1e-15:

                c = c_new
                break
            step *= 0.5

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _c5_with_lag(h):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * (2.0/n)), idx - (n-1), corr

def _proj(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = S - h.sum()
        if abs(d) < 1e-15:
            return h
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            return h
        h[m] += d / c
    return h

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0*h
    kk = abs(k)
    if k > 0:
        g[kk:] += 1.0 - h[:n-kk]
        g[:n-kk] -= h[kk:]
    else:
        g[:n-kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n-kk]
    return g

def _pd_search(h0, deadline, name=""):
    """Projected descent with active-set line search."""
    n = len(h0)
    h = _proj(h0.copy(), n)
    best_h = h.copy()
    best_c = _c5(h)
    t0 = time.time()
    it = 0
    while time.time() < deadline - 0.5:
        c, ka, corr = _c5_with_lag(h)
        if c < best_c - 1e-15:
            best_c = c; best_h = h.copy()
        mx = float(corr.max())
        tol = max(1e-7, 0.02 * 2.0/n)
        act = np.where(corr >= mx - tol*n/2.0)[0] - (n-1)
        if len(act) == 0:
            act = np.array([ka])
        g = np.zeros(n)
        for ak in act:
            g += _grad_lag(h, int(ak))
        g /= len(act)
        ng = np.linalg.norm(g)
        if ng < 1e-8:
            break
        # Backtracking line search
        step = min(1.0, 0.5/ng)
        improved = False
        for _ in range(30):
            h_new = _proj(h - step * g, n)
            c_new = _c5(h_new)
            if c_new < c - 1e-14:
                h = h_new; c = c_new; improved = True
                break
            step *= 0.5
        if not improved:
            break
        it += 1
    if _c5(h) < best_c:
        best_h = h.copy(); best_c = _c5(h)
    return best_h, best_c

        T = 0.003 * (0.002)**frac  # Temperature in "correlation" units

        kk = abs(ka)

                if 0 <= j < n:

                        # c_k = sum h[i+kk](1-h[i]), reduce by decreasing h[i+kk] or increasing h[i]
                        d = -(1 if rng.random() < 0.6 else -1) * rng.random() * 0.025
                    else:
                        # c_-kk = sum h[i](1-h[i+kk]), reduce by decreasing h[i] or increasing h[i+kk]

                        continue
                    h[i] += d; h[j] -= d
                    c2 = np.correlate(h, 1.0-h, mode='full')
                    mx2 = float(c2.max())

                    if delta < -1e-14 or rng.random() < np.exp(-delta/max(T,1e-15)):
                        corr = c2; mx = mx2
                        cn = mx*2.0/n
                        if cn < best_c:
                            best_c = cn; best_h = h.copy()
                    else:
                        h[i] -= d; h[j] += d
                    continue
        
        # Random move
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = (rng.random()-0.5) * 0.03
        hi_ = h[i]+d; hj_ = h[j]-d
        if hi_ < 0 or hi_ > 1 or hj_ < 0 or hj_ > 1:
            i2 = min(1.0, max(0.0, h[i]+d)) - h[i]
            d = i2
            if abs(d) < 1e-12: continue
        h[i] += d; h[j] -= d
        c2 = np.correlate(h, 1.0-h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx

        else:
            h[i] -= d; h[j] += d
    
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m: return _proj(h.copy(), n)
    x0 = (np.arange(n)+0.5)*2.0/n
    x1 = (np.arange(m)+0.5)*2.0/m
    return _proj(np.interp(x1, x0, h), m)
