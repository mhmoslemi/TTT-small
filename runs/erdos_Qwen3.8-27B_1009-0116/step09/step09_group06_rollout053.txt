def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _cfull(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a

def _proj(h, target):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    cmax = c.max()
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - cmax)
    s = np.clip(s - s.max(), -500, 500)
    return cmax + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m, n):

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = c.max()
    idx = np.argsort(c)[-m:]

        L = k - (n - 1)

            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g
