def _swap_delta_fast(h, a, b):
    n = len(h)
    k = np.arange(-(n-1), n)  # all lags
    # a+k
    ak = (a + k)
    val_ak = np.where((ak >= 0) & (ak < n), h[ak[(ak >= 0) & (ak < n)]], 0.0)
    # a-k
    ak2 = (a - k)
    val_ak2 = np.where((ak2 >= 0) & (ak2 < n), 1.0 - h[ak2[(ak2 >= 0) & (ak2 < n)]], 0.0)
    # b-k
    bk = (b - k)
    val_bk = np.where((bk >= 0) & (bk < n), 1.0 - h[bk[(bk >= 0) & (bk < n)]], 0.0)
    # b+k
    bk2 = (b + k)
    val_bk2 = np.where((bk2 >= 0) & (bk2 < n), h[bk2[(bk2 >= 0) & (bk2 < n)]], 0.0)
    return val_ak - val_ak2 + val_bk - val_bk2

import time

def _corr(h):
    return np.correlate(h, 1.0-h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h))) * (2.0/n)

def _binarize(h, n):
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:n//2]] = 1.0
    return out

def _swap_delta(h, a, b):
    """Return delta array for swapping h[a](1->0) and h[b](0->1)"""
    n = len(h)
    k = np.arange(-(n-1), n)
    ak = a + k
    ak2 = a - k
    bk = b - k
    bk2 = b + k
    d = np.zeros(2*n-1)
    m = (ak >= 0) & (ak < n)
    d[m] += h[ak[m]]
    m = (ak2 >= 0) & (ak2 < n)
    d[m] -= (1.0 - h[ak2[m]])
    m = (bk >= 0) & (bk < n)
    d[m] += (1.0 - h[bk[m]])
    m = (bk2 >= 0) & (bk2 < n)
    d[m] -= h[bk2[m]]
    return d

    zeros = np.where(h == 0)[0]
    
    stall = 0
    it = 0
    while time.time() - t0 < time_limit:
        # Find worst lags
        top_lags = np.argsort(-corr)[:n_top]
        worst_val = corr[top_lags[0]]
        
        # Generate candidate swaps
        found = False
        for attempt in range(200):
            if rng.random() < 0.7:
                # Focus on worst lag
                k = int(top_lags[rng.randint(n_top)])
            else:
                k = int(rng.randint(-(n-1), n))

            # Pick b from zeros where h[b-k]+h[b+k] is large

                a_scores[j] = s

                s = 0.0
                if 0 <= pos-k < n: s += h[pos-k]
                if 0 <= pos+k < n: s += h[pos+k]
                b_scores[j] = s

            a = ones[np.argmin(a_scores)]

                        best_max = current_max

        it += 1

Let me also double-check the _swap_delta formula. The correlation at lag k (index n-1+k in the full array) is:
C(k) = sum_i h[i+k] * (1-h[i]) for valid i

If I change h[a] from 1 to 0:
- For terms where a is the "i" position (second index): C(k) has term h[i+k]*(1-h[i]). When i=a: h[a+k]*(1-h[a]). Before (h[a]=1): h[a+k]*0 = 0. After (h[a]=0): h[a+k]*1 = h[a+k]. Change: +h[a+k].
- For terms where a is the "i+k" position (first index): C(k) has term h[i+k]*(1-h[i]). When i+k=a, i=a-k. Term: h[a]*(1-h[a-k]). Before: 1*(1-h[a-k]) = 1-h[a-k].
