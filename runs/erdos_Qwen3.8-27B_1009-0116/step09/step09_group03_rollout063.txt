def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

        h = np.clip(h, 0.0, 1.0)

def _dir_vec(h, i, j):
    n = len(h)
    L = np.arange(-(n-1), n)  # length 2n-1
    # i - L: valid when 0 <= i-L < n → i-n < L <= i → L in (i-n, i]
    # i + L: valid when 0 <= i+L < n → -i <= L < n-i → L in [-i, n-i)
    term1 = np.zeros(2*n-1)
    mask1 = (i - L >= 0) & (i - L < n)
    term1[mask1] = 1.0 - h[i - L][mask1]
    
    term2 = np.zeros(2*n-1)
    mask2 = (i + L >= 0) & (i + L < n)
    term2[mask2] = h[i + L][mask2]
    
    term3 = np.zeros(2*n-1)
    mask3 = (j - L >= 0) & (j - L < n)
    term3[mask3] = 1.0 - h[j - L][mask3]
    
    term4 = np.zeros(2*n-1)
    mask4 = (j + L >= 0) & (j + L < n)
    term4[mask4] = h[j + L][mask4]
    
    return term1 - term2 - term3 + term4

def _solve_1d(c, d, h_i, h_j):
    """Find δ that minimizes max(c + δ*d) subject to bounds."""
    lo = max(-h_i, h_j - 1.0) + 1e-15
    hi = min(1.0 - h_i, h_j) - 1e-15
    if lo >= hi:
        return 0.0
    for _ in range(40):

        c_new = c + mid * d
        mx = c_new.max()
        active = c_new >= mx - 1e-15
        if active.sum() == 0:
            active = (c_new == mx)
        d_active = d[active]

            hi = mid
        elif d_active.max() < 0:
            # f is decreasing, go right
            lo = mid
        else:
            # 0 is in subgradient, optimal
            return mid
    return (lo + hi) * 0.5
