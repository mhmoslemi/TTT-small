    dx = 2.0 / n
    # Smooth max
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()

Let me now think about the local search phase more carefully.

At a near-optimal point, the active set (lags at the maximum) is small, maybe 5-20 lags. The optimal direction to decrease the max is:
- Find the direction d that minimizes max_{L∈A} ∇c_L · d
- This is a linear programming problem: min max_{L∈A} ∇c_L · d s.t. ||d|| ≤ 1
- The solution is the minimum of the maximum of linear functions, which can be found by bisection on t: find d such that ∇c_L · d ≤ t for all L ∈ A

But this is overkill. A simpler approach:
- Average the gradients over the active set: d = -mean_{L∈A} ∇c_L
- This is the negative subgradient of the max function
- Project d onto the feasible set tangent space
- Take a step in direction d

Even simpler: for 2-point changes (h[i] += δ, h[j] -= δ), the change in c_L is:
δc_L = δ * [(1-h[i+L])-h[i-L] - (1-h[j+L])+h[j-L]] (approximately, for the relevant terms)

We want to find (i, j, δ) that most reduces the max c_L.

Let me think about this differently. The most effective 2-point change is one that:
- Increases correlations for all active lags simultaneously
- For lag L > 0: increasing h[i]*h[i+L] helps. So if h[i] is small and h[i+L] is large, increasing h[i] (and decreasing some h[j]) helps.

Actually, let me just go with a practical approach:

1. Use L-BFGS-B with the smooth max + penalty (fast, handles bounds)
2. Use projected subgradient for the max (handles the non-smoothness)
3. Use targeted 2-point local search with gradient information
4. Multiple alpha values and restarts

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(x, alpha, ml, mu_sum):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, 2*n-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    obj = m + np.log(np.exp(s).sum()) / alpha
    obj += mu_sum * (x.sum() - n * 0.5) ** 2
    return obj

def _smax_grad(x, alpha, ml, mu_sum):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ml_eff = min(ml, 2 * n - 1)
    idx = np.argpartition(-c, ml_eff)[:ml_eff]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= ws
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    g += 2.0 * mu_sum * (x.sum() - n * 0.5)
    return g

def _lbfgs_opt(h, alpha, ml, mu_sum, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad,
                       args=(alpha, ml, mu_sum), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h

def _fista(h, alpha, ml, mu_sum, steps, lr):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    t_val = 1.0
    best_c = _eval(x)
    best_x = x.copy()
    for k in range(steps):
        g = _smax_grad(y, alpha, ml, mu_sum)
        z = y - lr * g
        x_new = _proj(z, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + (t_val - 1.0) / (t_new + 1.0) * (x_new - x)
        y = np.clip(y, -1.0, 2.0)  # keep reasonable
        x = x_new
        t_val = t_new
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x

def _local_search(h, budget, rng, alpha, ml):
    n = len(h)
    c5_cur = _eval(h)
    best_c = c5_cur
    best_h = h.copy()
    t_start = time.time()
    c_arr = _corr(h)
    # Find active lags
    threshold = c5_cur - 0.002
    active = np.where(c_arr > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_arr)])
    
    iters = 0
    while time.time() - t_start < budget:
        iters += 1
        # Recompute active set every 50 iterations
        if iters % 50 == 1:
            c_arr = _corr(h)
            c5_cur = _eval(h)  # actually this is redundant, use c_arr.max()
            c5_cur = float(c_arr.max())
            threshold = c5_cur - 0.003
            active = np.where(c_arr > threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c_arr)])
        
        # Try random 2-point perturbation
        i = rng.randint(n)
        offset = 1 + rng.randint(0, 12)
        j = (i + offset) % n
        d = rng.randn() * 2e-5
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            d *= -0.5
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
        h[i], h[j] = hi, hj
        c_new = _eval(h)
        if c_new < c5_cur:
            c5_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i], h[j] = hi - d, hj + d
    
    return best_h, best_c
