import numpy as np

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    for _ in range(50):
        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-12:
            break
    d = S - x.sum()
    if abs(d) > 1e-10:
        if d > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - x[i])
                x[i] += a; d -= a
            else:
                a = min(-d, x[i])
                x[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return x

        h = np.clip(h, 0, 1)
        h = _proj(h, S)
        c2 = np.correlate(h, 1.0-h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta/max(T,1e-15))):
            mx = mx2; corr = c2
            if mx*(2.0/n) < best_c:
                best_c = mx*(2.0/n); best_h = h.copy()
        else:
            h = h.copy()  # keep (already projected)
        it += 1
    return best_h, best_c

def _fista_pass(h0, alpha, max_t):
    n = len(h0)
    S = n * 0.5
    dx = 2.0 / n
    t0 = time.time()
    h = _proj(h0, S)
    hp = h.copy(); yk = h.copy(); tk = 1.0
    best_h = h.copy(); best_c = _c5(h)
    L = 8.0
    for it in range(200000):
        if time.time() - t0 > max_t:
            break
        if it > 0:
            tn = 0.5*(1.0+np.sqrt(1.0+4.0*tk*tk))
            yk = h + ((tk-1.0)/tn)*(h-hp)
            tk = tn
            if yk.min() < -0.5 or yk.max() > 1.5:
                yk = _proj(yk, S)
        corr = np.correlate(yk, 1.0-yk, mode='full')
        shift = alpha*(corr - corr.max())
        np.clip(shift, -400, 0, out=shift)
        w = np.exp(shift); w /= w.sum()
        grad = np.zeros(n)
        for ko in range(n):
            wi = w[ko]
            if wi < 1e-8: continue
            k = ko - (n-1)
            if k == 0:
                grad += wi*(1.0-2.0*yk)
            elif k > 0:
                grad[:n-k] += wi*(1.0-yk[k:])
                grad[k:] -= wi*yk[:n-k]
            else:
                k2 = -k
                grad[k2:] += wi*(1.0-yk[:n-k2])
                grad[:n-k2] -= wi*yk[k2:]
        grad *= dx
        grad -= np.mean(grad)
        step = 1.0/L
        acc = False
        for _ in range(12):
            xn = _proj(yk - step*grad, S)
            dv = xn - yk
            fy = dx*(float(corr.max()) + np.log(np.sum(np.exp(shift+1e-15)))/alpha)
            fx = dx*(float(np.max(np.correlate(xn,1.0-xn,mode='full'))) + 0.0)
            rhs = fy + float(np.dot(grad, dv)) + 0.5*(1.0/step)*float(np.dot(dv,dv))
            if fx <= rhs + 1e-9:
                hp = h; h = xn; acc = True
                break
            step *= 0.5
        if acc:
            L = max(L*0.9, 2.0)
            if it % 20 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            L *= 2.0
            if L > 1e6: break
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return h, best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    S = n * 0.5
    h0 = _repair(h0, n)
    best = h0.copy()
    best_c = _c5(best)
    for v in (1.0-h0, h0[::-1]):
        v2 = _proj(v, S)
        c = _c5(v2)
        if c < best_c:
            best = v2; best_c = c

    # Phase 1: FISTA quick pass (40s)
    e1 = t0 + min(40.0, 0.04*budget_s)
    if time.time() < e1 - 2:
        hc = best.copy()
        for a in [100, 500, 2000]:
            if time.time() >= e1 - 1: break
            slot = min(12.0, e1 - time.time())
            if slot < 3: break
            hf, bh, bc = _fista_pass(hc, a, slot)
            cf = _c5(hf)
            if bc < best_c: best = bh.copy(); best_c = bc
            if cf < best_c: best = hf.copy(); best_c = cf
            hc = hf if cf < bc else bh

    # Phase 2: Fast local search (450s) - main workhorse
    e2 = t0 + min(490.0, 0.49*budget_s)
    if time.time() < e2 - 5:
        best, best_c = _fast_local(best, rng, e2, top_k=5)

    # Phase 3: SA exploration (200s)
    e3 = t0 + min(690.0, 0.69*budget_s)
    if time.time() < e3 - 5:
        best, best_c = _sa_local(best, rng, e3)

    # Phase 4: Multi-start (150s)
    e4 = t0 + min(840.0, 0.84*budget_s)
    if time.time() < e4 - 10:
        ns = 3
        sl = (e4 - time.time()) / ns
        for si in range(ns):
            if time.time() >= e4 - 2: break
            hs = best.copy()
            noise = rng.randn(n) * 0.04
            hs = _proj(hs + noise, S)
            if si == 1: hs = _proj(1.0 - best + noise*0.5, S)
            elif si == 2: hs = _proj(best[::-1] + noise, S)
            hf, cf = _fast_local(hs, rng, time.time()+sl, top_k=3)
            if cf < best_c: best = hf; best_c = cf

    # Phase 5: Final local polish (200s)
    e5 = t0 + budget_s - 20.0
    if time.time() < e5 - 5:
        best, best_c = _fast_local(best, rng, e5, top_k=7)

    # Phase 6: Tiny SA polish (15s)
    e6 = t0 + budget_s - 3.0
    if time.time() < e6:
        best, best_c = _sa_local(best, rng, e6)

    for v in (1.0-best, best[::-1]):
        v2 = _proj(v, n*0.5)
        c = _c5(v2)
        if c < best_c: best = v2; best_c = c
    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
