    # where (K_plus_k @ h)[i] = h[i+k]*[valid] and (K_minus_k @ h)[i] = h[i-k]*[valid]

    # (K_plus_k @ h)[i] = h[i+k] if 0<=i+k<n, else 0
    # This is just h shifted: (K_plus_k @ h) = np.roll trick or direct slice
    # (K_plus_k @ h)[i] = h[i+k] for i in [0, n-1-k]
    # (K_minus_k @ h)[i] = h[i-k] for i in [k, n-1]
    
    # So: grad = sum_k w_k - sum_k w_k * (h shifted by k) - sum_k w_k * (h shifted by -k)
    
    # sum_k w_k * h[i+k]: this is like a weighted sum of shifted h
    # = (weighted_correlation of w with h)
    
    # I can compute this as: np.correlate(h, w_arr, 'full') where w_arr is w laid out
    # Let's define w_arr of length 2n-1: w_arr[j] = w_full[j] (indexed by k=j-(n-1))
    
    # sum_k w[k] * h[i+k] for valid: 
    # Let m = i+k, k = m-i. Sum_m h[m] * w[m-i] for valid m and k.
    # = np.correlate(w_full_padded, h, 'full')[...]
    
    # Ugh, let me just do the O(n²) approach directly for n=560.
    # 560 iterations, each O(n) → O(n²) = 313k. Very fast in numpy.
    
    grad = np.ones(n)  # sum_k w_k * 1 = 1 (since w is normalized)
    # Subtract sum_k w_k * h[i-k]
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        if k >= 0:
            # h[i-k] valid for i in [k, n-1]
            grad[k:] -= wk * h[:-k] if k > 0 else wk * h
        else:
            kk = -k
            # h[i+kk] valid for i in [0, n-1-kk]
            grad[:n-kk] -= wk * h[kk:]
    # Subtract sum_k w_k * h[i+k]  
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        if k >= 0:
            # h[i+k] valid for i in [0, n-1-k]
            grad[:n-k] -= wk * h[k:]
        else:
            kk = -k
            # h[i-kk] valid for i in [kk, n-1]
            grad[kk:] -= wk * h[:-kk]
    
    return grad

    for a in alphas:
        rem = end1 - time.time()
        if rem < 10: break
        hf, bh_c = _fista_top(hc, a, rem*0.18, top_k=5, L0=12.0)
        if bh_c < best_c: best, best_c = hf, bh_c
        hc = hf

    # Phase 2: Local search from best (200s)
    end2 = t0 + 320.0
    best, best_c = _local_search(best, rng, end2, s0=5e-5)

    # Phase 3: Binary SA (100s)
    end3 = t0 + 420.0
    bh, bc = _binary_sa(best, rng, min(100.0, end3-time.time()))
    if bc < best_c: best, best_c = bh, bc

    # Phase 4: Re-FISTA from binary (60s)
    end4 = t0 + 480.0
    for a in [200, 1000, 5000]:
        rem = end4 - time.time()
        if rem < 10: break
        hf, bh_c = _fista_top(best, a, rem*0.3, top_k=5, L0=15.0)
        if bh_c < best_c: best, best_c = hf, bh_c
        hc = hf

    # Phase 5: Fine local search (300s)
    end5 = t0 + 780.0
    best, best_c = _local_search(best, rng, end5, s0=2e-5)

    # Phase 6: Another binary SA pass (100s)
    end6 = t0 + 880.0
    bh, bc = _binary_sa(best, rng, min(100.0, end6-time.time()))
    if bc < best_c: best, best_c = bh, bc

    # Phase 7: Very fine local (215s)
    end7 = t0 + budget_s - 10.0
    best, best_c = _local_search(best, rng, end7, s0=5e-6)

    # Final: symmetry and repair
    n = len(best)
    best, best_c = _symmetry_improve(best, n)
    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
