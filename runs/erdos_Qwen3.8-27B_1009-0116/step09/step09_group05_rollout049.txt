    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_run(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_proj(h, steps, step0, alpha, m_lags, t0, tlimit):
    n = len(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    mom = 0.8
    for it in range(steps):
        if time.time() - t0 > tlimit:
            break
        g = _smooth_grad(h, alpha, m_lags)
        v = mom * v + (1.0 - mom) * g
        sq = 0.9 * sq + 0.1 * g * g
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(16):
            h_new = _proj_sum(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 50.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.5, step0 * 50.0)
            if step < 1e-10:
                break
        it += 1
    return _repair(h, n), cur_c

def _pair_bisect(h, i, j, n):
    """Bisect optimal transfer from j to i (or vice versa)"""
    best_c5 = _eval(h)
    best_val = 0.0
    lo, hi = -0.5, 0.5
    if h[i] <= 0:
        lo = 0.0
    if h[j] >= 1.0:
        hi = 0.0
    if lo >= hi:
        return best_c5, 0.0
    for _ in range(20):
        mid = (lo + hi) * 0.5
        if abs(mid) < 1e-12:
            break
        h[i] += mid
        h[j] -= mid
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= mid
            h[j] += mid
            if mid > 0:
                hi = mid
            else:
                lo = mid
            continue
        c = _eval(h)
        if c < best_c5:
            best_c5 = c
            best_val = mid
            lo = mid if mid > 0 else hi
            hi = mid if mid > 0 else lo
            # Re-center
            lo, hi = mid - 0.5 * abs(mid) * 0.5, mid + 0.5 * abs(mid) * 0.5
            # Actually just use standard bisection
            hi = mid + (1.0 - h[j]) * 0.5 if mid > 0 else mid - h[j] * 0.5
            lo = mid - (1.0 - h[i]) * 0.5 if mid > 0 else mid + h[i] * 0.5
        else:
            if mid > 0:
                hi = mid
            else:
                lo = mid
        h[i] -= mid
        h[j] += mid
    return best_c5, best_val

def _pair_search(h, n_max_iters, t0, tlimit):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    # Get gradient to guide pair selection
    g = _smooth_grad(h, 1000000, 80)
    g_proj = g - g.mean()
    # Sort by gradient magnitude
    order = np.argsort(np.abs(g_proj))[::-1]
    half_n = n * 0.5
    for it in range(n_max_iters):
        if time.time() - t0 > tlimit:
            break
        # Re-compute gradient every 200 iterations
        if it % 200 == 0 and it > 0:
            g = _smooth_grad(h, 1000000, 80)
            g_proj = g - g.mean()
            order = np.argsort(np.abs(g_proj))[::-1]
        i = order[it % len(order)]
        j = order[(it + len(order) // 2) % len(order)]
        if i == j:
            continue
        # Try both directions
        for direction in [1, -1]:
            # Try a range of step sizes
            best_c = _eval(h)
            for scale in [0.1, 0.3, 0.5, 0.7, 0.9]:
                d = direction * scale * 0.1
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < best_c:
                        best_c = c5n
                        # Try to improve with smaller steps around this
                        for s2 in [0.5, 2.0, 0.3, 3.0]:
                            d2 = d * s2
                            ni2, nj2 = h[i] + d2, h[j] - d2
                            if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
                                h[i], h[j] = ni2, nj2
                                c5n2 = _eval(h)
                                if c5n2 < best_c:
                                    best_c = c5n2
                        h = _repair(h, n)
                        if best_c < c5c:
                            c5c = best_c
                            if c5c < best_c if False else c5c < _eval(h):
                                pass
                            best_c = c5c
                            best_h = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
                        break
        h[i], h[j] = h[i], h[j]  # no-op
        # Re-repair after modifications
        h = _repair(h, n)
        c5c = _eval(h)
        if c5c < best_c:
            best_c = c5c
            best_h = h.copy()
    return best_h, best_c

def _pair_search2(h, n_max_iters, t0, tlimit):
    """More systematic pair search on top-gradient coordinates"""
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    g = _smooth_grad(h, 5000000, 60)
    g_proj = g - g.mean()
    top_k = min(60, n)
    top_idx = np.argsort(np.abs(g_proj))[-top_k:]
    for it in range(n_max_iters):
        if time.time() - t0 > tlimit:
            break
        if it % 500 == 0 and it > 0:
            g = _smooth_grad(h, 5000000, 60)
            g_proj = g - g.mean()
            top_idx = np.argsort(np.abs(g_proj))[-top_k:]
        i = top_idx[it % top_k]
        j = top_idx[(it * 7 + 3) % top_k]
        if i == j:
            continue
        d = 0.05 * (0.5 + np.random.RandomState(it).rand() if False else 0.5)
        # Deterministic: alternate sign
        sign = 1 if (it % 2 == 0) else -1
        for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 1.0]:
            dd = sign * frac * d
            ni, nj = h[i] + dd, h[j] - dd
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    best_h = h.copy()
                    best_c = c5c
                else:
                    h[i], h[j] = ni - dd, nj + dd
                    break
            else:
                break
    return best_h, best_c

def _gen_structured(n, seed):
    """Generate a structured initial: periodic with golden-ratio-like spacing"""
    rng = np.random.RandomState(seed)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.linspace(0, 2, n, endpoint=False)
    # Use a high-frequency oscillation to spread 1s and 0s
    thresh = 0.5 + 0.5 * np.sin(2.0 * np.pi * phi * x / 2.0 + rng.rand() * 2.0)
    h = (x < thresh).astype(float)
    # Add noise
    h += 0.3 * rng.randn(n)
    return _repair(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP warmup with a few alpha values
    alpha_list = [5000, 50000, 500000]
    for alpha in alpha_list:
        if time.time() - t0 > 80:
            break
        ml = min(100, max(30, int(20 * np.log10(alpha / 5000) + 30)))
        h_new = _slsqp_run(best_h, alpha, ml, 400, 1e-14)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Also try a structured start
    if time.time() - t0 < budget_s - 400:
        h_alt = _gen_structured(n, seed + 1)
        h_alt = _slsqp_run(h_alt, 50000, 50, 300, 1e-14)
        c5_alt = _eval(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt

    # Phase 2: Nesterov projected gradient
    if time.time() - t0 < budget_s - 300:
        configs = [
            (20000, 40, 3000, 0.01),
            (200000, 60, 2500, 0.005),
            (2000000, 80, 2000, 0.003),
            (20000000, 100, 1500, 0.001),
        ]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s - 250:
                break
            tlimit = min(t0 + 80, budget_s * 0.75)
            h_new, c_new = _nesterov_proj(best_h.copy(), steps, s0, alpha, ml, t0, tlimit)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More SLSQP at high alpha
    if time.time() - t0 < budget_s - 150:
        for alpha in [10000000, 100000000, 1000000000]:
            if time.time() - t0 > budget_s - 100:
                break
            ml = min(120, n)
            h_new = _slsqp_run(best_h, alpha, ml, 500, 1e-16)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Systematic pair search
    if time.time() - t0 < budget_s - 50:
        n_iter = min(int((budget_s - 50 - (time.time() - t0)) / 0.002), 100000)
        best_h, best_c5 = _pair_search2(best_h, n_iter, t0, t0 + budget_s - 30)

    # Phase 5: Random local polish
    if time.time() - t0 < budget_s - 5:
        h2 = best_h.copy()
        c5c = _eval(h2)
        nc2 = len(h2)
        npol = min(int((budget_s - 5 - (time.time() - t0)) / 0.001), 300000)
        for _ in range(npol):
            if time.time() - t0 > budget_s - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 10)) % nc2
            d = 2e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
