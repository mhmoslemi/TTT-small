    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

    return _sobj(x, alpha, ml), _sgrad(x, alpha, ml)

def _slsqp(h, alpha, ml, maxiter):

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n*0.5,)}]

        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _block_swap_search(h, nsteps, rng, temp):
    """Local search using block swaps - preserves sum automatically"""
    n = len(h)
    cur = _eval(h)
    best_h, best_c = h.copy(), cur
    for step in range(nsteps):
        l = rng.randint(3, 40)
        i1 = rng.randint(0, n - l)
        i2 = rng.randint(0, n - l)
        if abs(i1 - i2) <= l:
            continue
        # Save for undo
        b1 = h[i1:i1+l].copy()
        b2 = h[i2:i2+l].copy()
        h[i1:i1+l] = b2
        h[i2:i2+l] = b1
        cn = _eval(h)
        if cn < cur or (temp > 1e-12 and rng.rand() < np.exp(-(cn-cur)/temp)):
            cur = cn
            if cn < best_c:
                best_c, best_h = cn, h.copy()
                temp *= 0.995
        else:
            h[i1:i1+l] = b1
            h[i2:i2+l] = b2
        temp *= 0.9999
    return best_h, best_c

def _seg_perturb_search(h, nsteps, rng, temp, seg_len=15):
    """Perturb random segments with uniform shifts"""
    n = len(h)
    cur = _eval(h)
    best_h, best_c = h.copy(), cur
    for step in range(nsteps):
        l = rng.randint(5, seg_len * 3)
        i1 = rng.randint(0, n - l)
        i2 = rng.randint(0, n - l)
        if abs(i1 - i2) <= l:
            continue
        # Shift segment 1 up, segment 2 down (mass transfer)
        eps = rng.uniform(0.005, 0.03) * l / l  # uniform shift
        save1 = h[i1:i1+l].copy()
        save2 = h[i2:i2+l].copy()
        h[i1:i1+l] += eps
        h[i2:i2+l] -= eps
        if h[i1:i1+l].max() > 1.01 or h[i2:i2+l].min() < -0.01:
            h[i1:i1+l] = save1
            h[i2:i2+l] = save2
            continue
        h = np.clip(h, 0, 1)
        h = _repair(h, n)
        # Need to compare with saved state - but repair changed things
        # Let's just evaluate and decide
        cn = _eval(h)
        if cn < cur or (temp > 1e-12 and rng.rand() < np.exp(-(cn-cur)/temp)):
            cur = cn
            if cn < best_c:
                best_c, best_h = cn, h.copy()
                temp *= 0.995
        else:
            h = best_h.copy()  # revert to best
            cur = _eval(h)
        temp *= 0.9999
    return best_h, best_c

def _coarse_optimize(h, n_coarse, maxiter):
    """Downsample, optimize, upsample"""
    n = len(h)
    # Downsample by averaging pairs
    if n_coarse == n // 2:
        h_c = (h[0::2] + h[1::2]) / 2.0
    else:
        # General downsampling
        idx = np.linspace(0, n, n_coarse + 1).astype(int)[:-1]
        h_c = np.array([h[idx[i]:idx[i]+n//n_coarse].mean() for i in range(n_coarse)])

    h_c = _slsqp(h_c, 5000, 30, maxiter)
    
    # Upsample by repeating
    h_up = np.repeat(h_c, n // n_coarse)[:n]
    h_up = _repair(h_up, n)
    return h_up
