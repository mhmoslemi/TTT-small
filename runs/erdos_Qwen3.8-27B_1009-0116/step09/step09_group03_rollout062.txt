import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    """Project onto {x: sum(x) = target, 0<=x<=1}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _peak_status(h, K):
    """Return top K lags, their values, and combined gradient direction"""
    n = len(h)
    c = _corrs(h)
    idx = np.argpartition(-c, K)[:K]
    idx = idx[np.argsort(-c[idx])]
    top_vals = c[idx]
    # Compute combined gradient
    g_comb = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        if L >= 0:
            for i in range(L, n):
                g_comb[i] += (1.0 - h[i - L])
                g_comb[i - L] -= h[i]
        else:
            M = -L
            for i in range(n - M):
                g_comb[i] += (1.0 - h[i + M])
                g_comb[i + M] -= h[i]
    g_comb *= (2.0 / n) / K  # normalize
    return top_vals, g_comb, c

def _equalize_step(h, K, max_iter=200, tol=1e-10):
    """Iteratively equalize top K correlations"""
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    for inner in range(max_iter):
        top_vals, g, c = _peak_status(h, K)
        # Remove mean to enforce sum constraint
        g = g - g.mean()
        # Line search
        best_local = _eval(h)
        step = 0.1
        found = False
        for _ in range(20):
            h_try = _proj_sum(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < best_local:
                best_local = c_try
                best_h = h_try.copy()
                h = h_try
                found = True
                break
            step *= 0.5
        if not found:
            break
        if best_local < best_c:
            best_c = best_local
            best_h = h.copy()
        spread = top_vals[0] - top_vals[-1]
        if spread < tol:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _f_eq(x, half):
    return x.sum() - half

def _j_eq(x, half):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g * dx

def _combined_grad(h, top_idx, weights):
    n = len(h)
    g = np.zeros(n)
    for ii, k in enumerate(top_idx):
        g += weights[ii] * _grad_lag(h, k - (n - 1))
    return g

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _f_eq, 'jac': _j_eq, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _equalize(h, K, n_inner, step0, rng):
    """Peel the top K lags by projected gradient in their combined direction."""
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    cur_c = best_c
    step = step0
    for _ in range(n_inner):
        c = _corrs(h)
        top_idx = np.argpartition(-c, K)[:K]
        vals = c[top_idx]
        spread = vals.max() - vals.min()
        if spread < 1e-12:
            break
        w = np.maximum(vals - vals.min(), 0)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(K) / K
        else:
            w /= wsum
        g = _combined_grad(h, top_idx, w)
        g -= g.mean()
        improved = False
        s_try = step
        for _ in range(25):
            h_try = _proj(h - s_try * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < cur_c - 1e-15:
                h = h_try
                cur_c = c_try
                improved = True
                step = min(s_try * 1.05, step0 * 5)
                break
            s_try *= 0.6
        if not improved:
            h = h + 1e-6 * rng.randn(n)
            h = _repair(h, n)
            cur_c = _eval(h)
            step = step0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def _directed_polish(h, n_steps, rng, K=4):
    """Fine polish using gradient-directed coordinate moves."""
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    cur_c = best_c
    step_base = 2e-5
    for it in range(n_steps):
        c = _corrs(h)
        top_idx = np.argpartition(-c, K)[:K]
        g = _combined_grad(h, top_idx, np.ones(K) / K)
        # Pick 4 coordinates with largest |gradient|
        top_coords = np.argsort(-np.abs(g))[:4]
        i1, i2, i3, i4 = top_coords[0], top_coords[1], top_coords[2], top_coords[3]
        # Move along gradient: decrease where g>0, increase where g<0
        moves = []
        d = step_base * rng.rand() * 0.5
        if g[i1] > 0 and g[i2] < 0:
            moves = [(i1, -d), (i2, d)]
        elif g[i3] > 0 and g[i4] < 0:
            moves = [(i3, -d), (i4, d)]
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            moves = [(i, d), (j, -d)]
        h_new = h.copy()
        ok = True
        for (idx, delta) in moves:
            nv = h_new[idx] + delta
            if nv < -1e-12 or nv > 1 + 1e-12:
                ok = False
                break
            h_new[idx] = nv
        if not ok:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            d2 = 1e-6 * rng.randn()
            if 0 <= h[i] + d2 <= 1 and 0 <= h[j] - d2 <= 1:
                h_new = h.copy()
                h_new[i] += d2
                h_new[j] -= d2
            else:
                continue
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < cur_c - 1e-15:
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def _nesterov_pg(h, alpha, ml, steps, lr0, momentum):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    lr = lr0
    for _ in range(steps):
        g = _smooth_grad(h, alpha, ml)
        v = momentum * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = _proj(h - lr * g, n * 0.5)
    return best_h
