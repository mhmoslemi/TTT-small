```python
    vals = c[active_idx] - t_val
    pos = np.maximum(vals, 0)
    return np.sum(pos * pos)

    dx = 2.0 / n

    vals = c[active_idx] - t_val
    weights = np.maximum(2 * vals, 0)  # 2 * max(0, c_k - t)

            if 0 <= j + L < n:
                vp = 1.0 - x[j + L]
            if 0 <= j - L < n:
                vn = x[j - L]
            g[j] += wk * (vp - vn)
```

where g_a(k) = (1-h[a+k]) - h[a-k] (the gradient of c_k w.r.t. h[a]).

If d > 0: g_i(k*) < g_j(k*) for all active k*
If d < 0: g_i(k*) > g_j(k*) for all active k*

This is a nice condition! We want to find pairs (i,j) where g_i is componentwise less than g_j on the active lags.

For a point i, the gradient contribution to lag K is:
g_i(K) = (1 - h[i+K]) * 1{0≤i+K<n} - h[i-K] * 1{0≤i-K<n}

A move: h[i] += d, h[j] -= d
Change in c_K: d * g_i(K) - d * g_j(K) = d * (g_i(K) - g_j(K))

For this to reduce c_K: d * (g_i(K) - g_j(K)) < 0 for all K in active set.

If we want d > 0 (increase h[i], decrease h[j]): need g_i(K) < g_j(K) for all K.
If we want d < 0: need g_i(K) > g_j(K) for all K.

The maximum d for d > 0 case: limited by h[i]+d ≤ 1 and h[j]-d ≥ 0, so d ≤ min(1-h[i], h[j]).
But we also need ALL lags to not increase. For inactive lags K', c_K' might increase. To be safe, we should check all lags, not just active ones. But for a first pass, checking only active lags is fine (the inactive ones have slack).

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    diff = t - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), abs(1.0 - h[idx]) if diff > 0 else abs(h[idx]))
            h[idx] += a if diff > 0 else -a
            diff += a if diff < 0 else -a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sm_grad(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii] * 2.0 / n
        L = k - (n - 1)
        if L >= 0:
            g[:n-L] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:n-L]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sum_f(x, half):
    return x.sum() - half

def _sum_j(x):
    return np.ones(len(x))

def _slsqp_smooth(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _grad_matrix(h, active_lags):
    """Compute gradient of each correlation w.r.t. each h[i] for active lags.
    Returns array of shape (n, len(active_lags))"""
    n = len(h)
    G = np.zeros((n, len(active_lags)))
    for kk, idx in enumerate(active_lags):
        L = idx - (n - 1)
        for i in range(n):
            v = 0.0
            if 0 <= i + L < n:
                v += 1.0 - h[i + L]
            if 0 <= i - L < n:
                v -= h[i - L]
            G[i, kk] = v
    return G

def _structure_search(h, t_fn, t_end, rng, max_iters=500000):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    iters = 0
    while iters < max_iters and t_fn() < t_end:
        iters += 1
        c = _corr(best_h)
        cmax = c.max()
        # Active lags: top 8
        top_k = min(8, len(c))
        active = np.argpartition(-c, top_k)[:top_k]
        G = _grad_matrix(best_h, active)
        # Find a helpful pair: i with small G[i,:], j with large G[j,:]
        # Sort by first active lag gradient
        s1 = G[:, 0].argsort()
        i = s1[rng.randint(len(s1)) // 4]  # bottom quartile
        j = s1[rng.randint(len(s1) * 3 // 4, len(s1))]  # top quartile
        d_sign = 1.0 if G[i].mean() < G[j].mean() else -1.0
        d = d_sign * 1e-4 * rng.uniform(0.5, 1.5)
        ni, nj = best_h[i] + d, best_h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        best_h[i], best_h[j] = ni, nj
        c5n = _eval(best_h)
        if c5n < cmax - 1e-12:
            if c5n < best_c:
                best_c = c5n
        else:
            best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c
```

I need to be smarter. Let me use the incremental update:
- After a 2-point move, only the correlations involving i or j change
- But "involving" means any lag where i or j appears in the product, which is all lags! Because c_k = sum_i h[i](1-h[i+k]), and changing h[i] affects c_k for ALL k.

- h[j] appears in c_k if 0≤j<n and 0≤j+k<n

Since i and j are fixed indices, changing h[i] affects ALL c_k (for all k where i is in range, which is all k for i in (0,n)).

Alternative: use the gradient to PREDICT the change, and only verify with full eval occasionally.

Predicted change: Δc_k = d * (G[i,k] - G[j,k]) for each active lag k.
If the prediction says c_k will decrease for all k, make the move and verify.

This way, most moves are rejected by the prediction (cheap), and only promising ones are verified (expensive).

```python
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    
    while t_fn() < t_end:
        c = _corr(best_h)
        cmax = c.max()
        active = np.argpartition(-c, 10)[:10]
        G = _grad_matrix(best_h, active)  # shape (n, 10)
        
        # Gradient-based local search
        improved = False
        for _ in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j: continue
            # Predicted change for each active lag
            delta = G[i] - G[j]  # for d>0: Δc_k = d*delta[k]
            d_sign = 1.0 if delta.min() < 0 else -1.0
            # Choose d to reduce worst active lag
            worst_k = np.argmin(delta * d_sign)
            max_d = min(1.0 - best_h[i], best_h[j]) if d_sign > 0 else min(best_h[i], 1.0 - best_h[j])
            d = d_sign * min(1e-4, 0.5 * max_d)
            if abs(d) < 1e-8: continue
            # Predicted worst c after move
            pred_worst = cmax + abs(d) * max(0, (delta * d_sign).max())
            if pred_worst >= cmax - 1e-6:
                continue
            # Make the move
            best_h[i] += d
            best_h[j] -= d
            # Verify
            c5n = _eval(best_h)
            if c5n < best_c:
                best_c = c5n
                improved = True
            else:
                best_h[i] -= d
                best_h[j] += d
        
        if not improved and cmax == best_c:
            # Stuck, try random perturbation
            for _ in range(10):
                idx = rng.randint(n)
                d = 1e-3 * rng.randn()
                best_h[idx] += d
                best_h = _repair(best_h, n)
                c5n = _eval(best_h)
                if c5n < best_c:
                    best_c = c5n
                    break
    
    return best_h, best_c
```
