from scipy.optimize import linprog

def _slp_step(h, m_lags, n):
    """One Sequential LP step."""
    c = _eval_all(h)
    idx = np.argsort(c)[-m_lags:]
    target = n * 0.5
    dx = 2.0 / n
    
    # Build LP: minimize t
    # Variables: [h_0, h_1, ..., h_{n-1}, t]  (n+1 vars)
    # Objective: [0, 0, ..., 0, 1] (minimize t, index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Constraints: for each top lag k:
    # c_k_lin(h) = c_k(h_0) + sum_i grad_k[i] * (h[i] - h_0[i]) <= t
    # => sum_i grad_k[i] * h[i] - t <= c_k(h_0) - sum_i grad_k[i] * h_0[i]
    A_ub = np.zeros((m_lags, n + 1))
    b_ub = np.zeros(m_lags)
    
    for ii, k in enumerate(idx):
        L = k - (n - 1)
        gk = np.zeros(n)
        if L >= 0:
            gk[:n - L] += (1.0 - h[L:]) * dx
            gk[L:] -= h[:n - L] * dx
        else:
            M = -L
            gk[M:] += (1.0 - h[:n - M]) * dx
            gk[:n - M] -= h[M:] * dx
        A_ub[ii, :n] = gk
        A_ub[ii, n] = -1.0
        b_ub[ii] = c[ii] - gk @ h
    
    # Box constraints: 0 <= h[i] <= 1 (handled by bounds)
    # Sum constraint: sum(h) = target (equality)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [target]
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        h_new = res.x[:n]
        h_new = _repair(h_new, n)
        return h_new
    return h

    s = np.clip(s - s.max(), -400, 400)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(len(h))
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_lag(h, idx[ii], n)
    return g

def _obj_slsqp(x, alpha, m):
    c = _eval_all(x)
    idx = np.argsort(c)[-m:]
    cmax = c.max()
    s = alpha * (c[idx] - cmax)
    s = np.clip(s - s.max(), -400, 400)
    return cmax + np.log(np.exp(s).sum()) / alpha

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_slsqp, h, jac=lambda x: _grad_softmax(x, alpha, m),
                       args=(alpha, m), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _slp_step(h, m_lags, n, max_lp_iter):
    """Sequential LP step with line search."""
    c = _eval_all(h)
    idx = np.argsort(c)[-m_lags:]
    target = n * 0.5
    dx = 2.0 / n
    c_cur = c.max()
    
    # Build LP
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    b_ub = np.zeros(len(idx))
    
    for ii, k in enumerate(idx):
        gk = _grad_lag(h, k, n)
        A_ub[ii, :n] = gk
        A_ub[ii, n] = -1.0
        b_ub[ii] = c[ii] - gk @ h
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return h, c_cur
    h_lp = res.x[:n]
    
    # Line search
    for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        h_try = _repair((1.0 - t) * h + t * h_lp, n)
        c_try = _eval(h_try)
        if c_try < c_cur - 1e-15:
            return h_try, c_try
    return h, c_cur

def _momentum_pg(h, alpha, m, nsteps, step0, time_limit):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    x_prev = x.copy()
    cur_c = _eval(x)
    step = step0
    t_start = time.time()
    for it in range(1, nsteps + 1):
        if time.time() - t_start > time_limit:
            break
        beta = min(0.98, 0.7 + 0.01 * it)
        y = x + beta * (x - x_prev)
        g = _grad_softmax(y, alpha, m)
        taken = False
        for _ in range(12):
            x_new = _project(y - step * g, target)
            c_new = _eval(x_new)
            if c_new < cur_c - 1e-17:
                x_prev = x
                x = x_new
                cur_c = c_new
                step = min(step * 1.03, step0 * 100.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            x_prev = x.copy()
            step = min(step * 1.3, step0 * 100.0)
    return _repair(x, n), cur_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def _coord_polish(h, n_tries, rng, time_limit):
    n = len(h)
    c5 = _eval(h)
    t_start = time.time()
    for _ in range(n_tries):
        if time.time() - t_start > time_limit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, max(1, n // 10))) % n
        d = 1e-5 * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d
            h[j] -= d
            c_new = _eval(h)
            if c_new < c5:
                c5 = c_new
            else:
                h[i] -= d
                h[j] += d
    return h, c5
