def _grad_corr(h, k):
    """Gradient of c_k w.r.t. h, where k is lag index in [0, 2n-2], center at n-1."""
    n = len(h)
    g = np.zeros(n)
    L = k - (n - 1)  # actual lag
    dx = 2.0 / n
    if L >= 0:
        m = n - L
        g[:m] += dx * (1.0 - h[L:])
        g[L:] -= dx * h[:m]
    else:
        M = -L
        m = n - M
        g[:m] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:m]
    return g

    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c.max())
    s -= s.max()
    return c.max() + np.log(np.exp(s).sum()) / alpha

        g += w[ii] * _grad_corr(x, k)
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        g[:m] += dx * (1.0 - h[L:])
        g[L:] -= dx * h[:m]
    else:

        m = n - M
        g[M:] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[M:]
    return g

    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - c.max())
    s -= s.max()
    return c.max() + np.log(np.exp(s).sum()) / alpha

    w = np.exp(s)

                best_h = h.copy()
    return best_h, best_c

def _coord_descent(h, n_sweeps, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_sweeps):
        perm = rng.permutation(n)
        for j in perm:
            c_vec = _corr_vec(h)
            top3 = np.argsort(c_vec)[-3:]
            g = np.zeros(n)
            for k in top3:
                g += _grad_lag(h, k - (n - 1))
            g /= 3
            for delta in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]:
                h[j] += delta
                if h[j] < 0 or h[j] > 1:
                    h[j] -= delta
                    continue
                h[j] = np.clip(h[j], 0, 1)
                h = _repair(h, n)
                c_new = _eval(h)
                if c_new < best_c + 1e-15:
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    break
                else:
                    h[j] = np.clip(h[j] - delta, 0, 1)
                    h = _repair(h, n)
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 3e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with aggressive alpha
    for alpha in [10000, 100000, 1000000, 10000000, 100000000]:
        if elapsed() > 100:
            break
        ml = min(90, max(20, int(12 * np.log10(alpha / 1e4) + 20)))
        h_new = _slsqp_opt(best_h, alpha, ml, 600)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Multi-start perturbation + SLSQP
    for trial in range(4):
        if elapsed() > 220:
            break
        scale = [0.08, 0.04, 0.02, 0.15][trial]
        h_p = _proj(best_h + scale * rng.randn(n), n * 0.5)
        h_p = _repair(h_p, n)
        h_t = _slsqp_opt(h_p, 200000, 50, 400)
        c5t = _eval(h_t)
        if c5t < best_c5:
            best_c5 = c5t
            best_h = h_t

    # Phase 3: Nesterov cyclic gradient
    if elapsed() < budget_s - 450:
        for lr0, nact, steps in [(0.003, 30, 2500), (0.001, 50, 2000), (0.0005, 80, 1500)]:
            if elapsed() > budget_s - 380:
                break
            h_new, c_new = _nesterov_cyc(best_h.copy(), steps, lr0, nact)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 4: Binary push + swap
    if elapsed() < budget_s - 200:
        n_sw = min(int((budget_s - 200 - elapsed()) / 0.004), 80000)
        h_new, c_new = _swap_search(best_h.copy(), n_sw, rng)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: SLSQP refinement after swaps
    if elapsed() < budget_s - 150:
        for alpha in [5000000, 50000000]:
            if elapsed() > budget_s - 100:
                break
            h_new = _slsqp_opt(best_h, alpha, 60, 400)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 6: Two-point gradient search
    if elapsed() < budget_s - 80:
        n_it = min(int((budget_s - 80 - elapsed()) / 0.007), 40000)
        h_w = best_h.copy()
        c5c = _eval(h_w)
        for _ in range(n_it):
            c_v = _corr_vec(h_w)
            top3 = np.argsort(c_v)[-3:]
            g = np.zeros(n)
            for k in top3:
                g += _grad_lag(h_w, k - (n - 1))
            g /= 3
            gp = g - g.mean()
            i = int(np.argmax(gp))
            j = int(np.argmin(gp))
            if i == j:
                continue
            improved = False
            for frac in [0.5, 0.3, 0.15, 0.05]:
                d = frac * 0.008 * (gp[i] - gp[j])
                ni, nj = h_w[i] + d, h_w[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h_w[i], h_w[j] = ni, nj
                    h_w = _repair(h_w, n)
                    c5n = _eval(h_w)
                    if c5n < c5c:
                        c5c = c5n
                        improved = True
                        break
                    else:
                        h_w[i], h_w[j] = ni - d, nj + d
                        h_w = _repair(h_w, n)
                        break
            if not improved:
                h_w[i], h_w[j] = h_w[j], h_w[i]
                h_w = _repair(h_w, n)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_w.copy()

    # Phase 7: Ultra-fine polish
    if elapsed() < budget_s - 3:
        n_pol = min(int((budget_s - 3 - elapsed()) / 0.0008), 500000)
        best_h, best_c5 = _fine_polish(best_h, n_pol, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
