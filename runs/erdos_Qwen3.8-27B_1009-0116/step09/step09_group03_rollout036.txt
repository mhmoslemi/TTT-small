```python
    """Compute analytic gradient of smooth max and take a step."""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # Focus on top ml values
    idx = np.argpartition(-c, ml)[:ml]
    s = alpha * (c[idx] - c.max())
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
```

```python
    h = np.clip(h, 0.0, 1.0).copy()

        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(s - target) < 1e-12:
```

```python
    """Compute change in all correlations when h[i]+=d, h[j]-=d."""
    n = len(h)
    dx = 2.0 / n
    # Precompute effect of changing h[i] by d
    # c[k] += d * (1-h[i-(k-n+1)]) - d*h[i+(k-n+1)]
    # Let's compute for all k from 0 to 2n-2
    
    # Effect of h[i] += d on c[k]:
    # Term 1: sum includes h[i]*(1-h[i-k+n-1]) -> adds d*(1-h[i-k+n-1]) if 0<=i-k+n-1<n
    # Term 2: sum includes h[j']*(1-h[j'-k+n-1]) where j'-k+n-1=i -> j'=i+k-n+1
    #   -> adds -d*h[i+k-n+1] if 0<=i+k-n+1<n
    #   (because (1-h[i]) decreases by d, so the product decreases by h[j']*d)
    
    # For a 2-point swap h[i]+=d, h[j]-=d:
    # Net change in c[k] = d*(1-h[i-k+n-1]) - d*h[i+k-n+1] - d*(1-h[j-k+n-1]) + d*h[j+k-n+1]
    # (first two from i, last two from j with -d)
    
    # To find new max, we need c[k] + delta[k] for all k
    # This is O(n) per k, total O(n^2)
    # But we can vectorize it!
    
    k = np.arange(2*n-1)
    shift = k - (n-1)  # shift ranges from -(n-1) to n-1

    # Term: d*(1-h[i-shift]) if 0<=i-shift<n, else 0
    # Term: -d*h[i+shift] if 0<=i+shift<n, else 0

    eff_i[m2] += -d * h[idx2_i[m2]]

    idx2_j = j + shift
    
    eff_j = np.zeros(2*n-1)
    m1j = (idx1_j >= 0) & (idx1_j < n)
    m2j = (idx2_j >= 0) & (idx2_j < n)
    eff_j[m1j] += -d * (1 - h[idx1_j[m1j]])
    eff_j[m2j] += d * h[idx2_j[m2j]]
    
    return eff_i + eff_j
```

This is O(n) vectorized operations, much faster than recomputing the full correlation.

Now for the local search, I can:
1. Compute c = correlate(h, 1-h, 'full') * dx once
2. For each move, compute the delta (O(n) vectorized)
3. Find the new max: max(c + delta)
4. Accept if new_max < old_max

Let me recheck: c[k] = sum_p h[p] * (1-h[p-s]) where s = k-(n-1).

```python
            stall += 1
            if stall > 3:
                c_cur, c_full = _eval_corr(h)
                rc = 0
                for _ in range(8):
                    ii = rng.randint(n)
                    jj = (ii + 1 + rng.randint(0, 80)) % n
                    dd = 0.003 * rng.randn()
                    if 0 <= h[ii] + dd <= 1 and 0 <= h[jj] - dd <= 1:
                        h[ii] += dd; h[jj] -= dd
                        break
                c_cur, c_full = _eval_corr(h)
                stall = 0
    return best_h, best_c

def _pg_opt(h, alpha, ml, steps, lr0, mom):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr0
    for _ in range(steps):
        g = _smooth_grad(h, alpha, ml)
        ls = lr
        c_here = _eval(h)
        for _ in range(10):
            h_try = _proj(h - ls * g, n * 0.5)
            if _eval(h_try) < c_here:
                v = mom * v - ls * g
                h = h_try
                lr = min(ls * 1.04, lr0 * 40)
                break
            ls *= 0.5
        else:
            v = mom * v - lr * g
            h = _proj(h + v, n * 0.5)
            lr = max(lr * 0.93, lr0 * 0.01)
        cn = _eval(h)
        if cn < best_c:
            best_c = cn
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t_start

    h = _repair(np.array(initial_h_values, dtype=float), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP alpha ramp
    for alpha, ml, iters in [(5000, 25, 300), (50000, 40, 350), (500000, 60, 400), (5000000, 80, 400)]:
        if _el() > 70:
            break
        h_new = _slsqp(best_h, alpha, ml, iters, 1e-14)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Projected gradient with momentum
    if _el() < budget_s - 350:
        for alpha, ml, steps, lr0, mom in [
            (100000, 50, 300, 0.01, 0.75),
            (1000000, 70, 250, 0.005, 0.82),
            (10000000, 90, 200, 0.002, 0.88),
        ]:
            if _el() > budget_s - 300:
                break
            h_new, c5n = _pg_opt(best_h.copy(), alpha, ml, steps, lr0, mom)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: Fast local search with annealing
    remain = budget_s - 80 - _el()
    if remain > 30:
        t1 = min(remain * 0.45, 250)
        h_new, c5n = _local_search(best_h.copy(), t1, rng, 5000000, 80, True)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: Exploitation local search (no annealing)
    remain2 = budget_s - 40 - _el()
    if remain2 > 20:
        h_new, c5n = _local_search(best_h.copy(), remain2, rng, 100000000, 120, False)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Final SLSQP polish
    if _el() < budget_s - 10:
        h_new = _slsqp(best_h, 50000000, 100, 300, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 6: Last-second local search
    remain3 = budget_s - 2 - _el()
    if remain3 > 5:
        h_new, c5n = _local_search(best_h.copy(), remain3, rng, 200000000, 150, False)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
