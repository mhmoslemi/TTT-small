def _project_box_sum(x, target):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)

            lo = tau
        else:
            hi = tau
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _default_h(n):

def _cor(h):

    pref[0] = 0.0

    ln = np.arange(1, n)
    c[off - ln] = (hn - pref[ln]) - p[ln]
    return c

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, 'full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500.0, 500.0)
    s -= s.max()
    return float(m + np.log(np.exp(s).sum()) / alpha)

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, 'full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g
