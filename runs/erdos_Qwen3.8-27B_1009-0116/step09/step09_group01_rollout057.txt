```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _ov(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(v, S):
    x = np.clip(np.array(v, dtype=float), 0.0, 1.0).copy()
    for _ in range(40):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
        x = np.clip(x, 0.0, 1.0)
    return x

def _grad_active(h, M):
    """Gradient w.r.t. h of the max overlap, approximated by top-M average."""
    n = len(h)
    dx = 2.0 / n
    cov = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(cov)[::-1][:M]
    g = np.zeros(n)
    for idx in top:
        s = int(idx) - (n - 1)
        if s == 0:
            g += (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
        else:
            ss = -s
            g[ss:] += (1.0 - h[:n-ss])
            g[:n-ss] -= h[ss:]
    return g * dx / M

def _grad_descent(h0, budget, M_list=[5,10,20,50,100]):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _proj(h0, S)
    best_c = _c5(h)
    best_h = h.copy()
    rm = 0.9  # momentum
    v = np.zeros(n)
    lr = 5.0
    
    for M in M_list:
        it = 0
        while time.time() - t0 < budget:
            g = _grad_active(h, M)
            g -= g.mean()  # center to maintain sum
            v = rm * v - lr * g
            h_new = _proj(h + v, S)
            c_new = _c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
            lr *= 0.999
            it += 1
    return best_h, best_c
```

```python
    dp[:j+1] -= h[j::-1]
    
    dn = np.zeros(n-1)  # negative shifts s=1..n-1
    if i >= 1: dn[:i] -= (1.0 - h[i-1::-1])
    if j >= 1: dn[:j] += (1.0 - h[j-1::-1])
    if n-1-i >= 1: dn[:n-1-i] += h[i+1:]
    if n-1-j >= 1: dn[:n-1-j] -= h[j+1:]
    
    # Combine into full correlate index
    delta = np.empty(2*n-1)
    delta[n-1:] = dp  # positive shifts, index n-1+s for s=0..n-1
    delta[:n-1] = dn[::-1]  # negative shifts, index n-1-s for s=1..n-1
    # Wait, dn[0] corresponds to s=1, which is index n-2.
    # dn[k] corresponds to s=k+1, which is index n-1-(k+1) = n-2-k
    # So delta[n-2-k] = dn[k] for k=0..n-2
    # That means delta[:n-1] = dn[::-1]? Let me check:
    # k=0: delta[n-2] = dn[0]
    # k=n-2: delta[0] = dn[n-2]
    # So delta[0] = dn[n-2], delta[n-2] = dn[0] → delta[:n-1] = dn[::-1] ✓
    
    return delta * dx
```

This is O(n) with numpy operations, which should be fast.

```python
def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.clip(np.array(v, dtype=float), 0.0, 1.0).copy()
    for _ in range(40):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
        x = np.clip(x, 0.0, 1.0)
    return x

def _grad_top(h, M):
    n, dx = len(h), 2.0/len(h)
    cov = np.correlate(h, 1.0-h, mode='full')
    top = np.argsort(cov)[::-1][:M]
    g = np.zeros(n)
    for idx in top:
        s = int(idx) - (n-1)
        if s == 0:
            g += 1.0 - 2.0*h
        elif s > 0:
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
        else:
            ss = -s
            g[ss:] += 1.0 - h[:n-ss]
            g[:n-ss] -= h[ss:]
    return g * dx / M

def _grad_search(h0, budget_s, rng):
    t0 = time.time()
    n, S = len(h0), len(h0)*0.5
    h = _proj(h0, S)
    best_c, best_h = _c5(h), h.copy()
    v = np.zeros(n)
    for M, lr0 in [(5, 10.0), (10, 5.0), (20, 3.0), (50, 1.5)]:
        lr = lr0
        for _ in range(5000):

                break
            g = _grad_top(h, M)
            g -= g.mean()
            v = 0.9*v - lr*g
            h = _proj(h + v, S)
            c = _c5(h)
            if c < best_c:

            lr *= 0.998
```

```python
def _swap_delta(h, i, j, n):
    """Delta in raw correlation count when swapping h[i](1) and h[j](0)."""
    # Returns array of length 2n-1 (same indexing as np.correlate full)
    # For bang-bang h
    d = np.zeros(2*n-1)
    # Positive shifts s=0..n-1 (correlate index n-1+s)
    # Δ_s = -(1-h[i+s]) for s<n-i, +(1-h[j+s]) for s<n-j, +h[i-s] for s<=i, -h[j-s] for s<=j
    if n-i > 0:
        d[n-1:n-1+(n-i)] -= (1.0 - h[i:])
    if n-j > 0:
        d[n-1:n-1+(n-j)] += (1.0 - h[j:])
    if i >= 0:
        d[n-1:n-1+(i+1)] += h[i::-1]
    if j >= 0:
        d[n-1:n-1+(j+1)] -= h[j::-1]
    # Negative shifts s=1..n-1 (correlate index n-1-s)
    # Δ_{-s} = -(1-h[i-s]) for s<=i, +(1-h[j-s]) for s<=j, +h[i+s] for i+s<n, -h[j+s] for j+s<n
    if i >= 1:
        d[n-1-i:n-1] -= (1.0 - h[i-1::-1])
    if j >= 1:
        d[n-1-j:n-1] += (1.0 - h[j-1::-1])
    if n-1-i >= 1:
        d[n-1-(n-1-i):n-1] += h[i+1:]
    if n-1-j >= 1:
        d[n-1-(n-1-j):n-1] -= h[j+1:]
    return d
```

In the "full" correlate output, this is at index (n-1) - s.

The change when swapping h[i]:1→0, h[j]:0→1:

  Actually: was h[i]*(1-h[i-s]) = 1*(1-h[i-s]). If h[i-s]=0, this was 1. If h[i-s]=1, this was 0.
  Now: 0*(1-h_new[i-s]) = 0.
  Δ = -(1-h[i-s])

- k=j (if s≤j≤n-1): was 0*(...) = 0. Now: 1*(1-h[j-s]). Δ = +(1-h[j-s])

- k=i+s (if i+s ≤ n-1, i.e., s ≤ n-1-i): 
  Before: h[i+s]*(1-h[i]) = h[i+s]*0 = 0 (since h[i]=1)
  After: h[i+s]*(1-h_new[i]) = h[i+s]*1 = h[i+s]
  Δ = +h[i+s]

- k=j+s (if j+s ≤ n-1, i.e., s
