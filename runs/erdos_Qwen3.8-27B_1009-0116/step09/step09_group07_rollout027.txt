def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _cd_pass(h, order):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    nm1 = n - 1
    cmax = 2 * n - 1
    for j in order:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        # Minimize max(c + beta * t) over t in [0,1]
        # This is a convex PL function
        lo_t, hi_t = 0.0, 1.0
        for _ in range(40):
            t1 = lo_t + (hi_t - lo_t) / 3.0

            if np.max(c + beta * t1) <= np.max(c + beta * t2):

            h[j] = t_new

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c))

def _cd_pass(h, order, n_trisect=25):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    nm1 = n - 1
    cmax = 2 * n - 1
    for j in order:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(n_trisect):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(c + beta * t1) <= np.max(c + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.sum(np.exp(s)) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_con_fun(x, target):
    return x.sum() - target

def _sum_con_jac(x, target):
    return np.ones(len(x))

def _do_slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    return (best_h, best_c5, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c))

def _cd_pass(h, order, n_tri=25):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    nm1 = n - 1
    cmax = 2 * n - 1
    for j in order:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(n_tri):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(c + beta * t1) <= np.max(c + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _smax_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smax_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-16:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _csum_fun(x, target):
    return x.sum() - target

def _csum_jac(x, target):
    return np.ones(len(x))

def _do_slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csum_fun, 'jac': _csum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _cd_block(h, n, rng, n_passes):
    h = h.copy()
    for _ in range(n_passes):
        h = _cd_pass(h, rng.permutation(n))
        h = _repair(h, n)
    return h

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

        c5_ls = best_c5

            ni, nj = h_ls[i1] + d, h_ls[i2] - d

                if c5n < c5_ls:

                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h_ls.copy()
                else:
                    h_ls[i1], h_ls[i2] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
