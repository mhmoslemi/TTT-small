import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(25):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

    idx = np.argsort(c)[-topm:]
    cm = c[idx].max()
    s = np.clip(alpha * (c[idx] - cm), -500, 500)
    s -= s.max()
    return cm + np.log(np.exp(s).sum()) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-(1.0 - h), kind='stable')
        else:
            order = np.argsort(h, kind='stable')
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1

        lo_t, hi_t = 0.0, 1.0

            h[j] = t_new

    s -= s.max()

    dx = 2.0 / n

    return x.sum() - target

    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(28):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _sm_obj(x, alpha, topm):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-topm:]
    cm = c[idx].max()
    s = np.clip(alpha * (c[idx] - cm), -500, 500)
    s -= s.max()
    return cm + np.log(np.exp(s).sum()) / alpha

def _sm_grad(x, alpha, topm):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-topm:]
    cm = c[idx].max()
    s = np.clip(alpha * (c[idx] - cm), -500, 500)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, topm, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, topm),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h

def _pen_obj(x, alpha, topm, pen):
    n = len(x)
    base = _sm_obj(x, alpha, topm)
    s = x.sum() - n * 0.5
    return base + pen * s * s

def _pen_grad(x, alpha, topm, pen):
    n = len(x)
    g = _sm_grad(x, alpha, topm)
    s = x.sum() - n * 0.5
    g += 2.0 * pen * s
    return g

def _lbfgs(h, alpha, topm, pen, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pen_obj, h, jac=_pen_grad, args=(alpha, topm, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h

def _perturb_cd(h, n, rng, sigma, n_cd):
    hp = h + sigma * rng.randn(n)
    hp = _repair(hp, n)
    for _ in range(n_cd):
        hp = _cd_pass(hp, rng)
        hp = _repair(hp, n)
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: CD (30% budget)
    p1_end = t0 + budget_s * 0.30
    h = best_h.copy()
    while time.time() < p1_end:
        h = _cd_pass(h, rng)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = _repair(h, n)

    # Phase 2: SLSQP refinement (25% budget)
    p2_end = t0 + budget_s * 0.55
    cfg2 = [(2000, 20, 1200), (15000, 30, 1000), (100000, 40, 800),
            (500000, 50, 700), (2000000, 60, 600)]
    for alpha, topm, miter in cfg2:
        if time.time() > p2_end:
            break
        h_new = _slsqp(best_h.copy(), alpha, topm, miter)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: L-BFGS-B with penalty (15% budget)
    p3_end = t0 + budget_s * 0.70
    cfg3 = [(5000, 30, 50, 1500), (50000, 40, 40, 1200),
            (200000, 50, 30, 1000), (1000000, 60, 20, 800)]
    for alpha, topm, pen, miter in cfg3:
        if time.time() > p3_end:
            break
        h_new = _lbfgs(best_h.copy(), alpha, topm, pen, miter)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: Perturbation + CD (20% budget)
    p4_end = t0 + budget_s * 0.90
    sigmas = [0.025, 0.015, 0.035, 0.01, 0.04, 0.02, 0.05, 0.012]
    n_cd_pert = [5, 5, 4, 4, 3, 4, 3, 5]
    for i in range(len(sigmas)):
        if time.time() > p4_end:
            break
        hp = _perturb_cd(best_h, n, rng, sigmas[i], n_cd_pert[i])
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp

    # Phase 5: Final SLSQP + CD (10% budget)
    p5_end = t0 + budget_s * 0.99
    if time.time() < p5_end - 60:
        h_new = _slsqp(best_h.copy(), 5000000, 80, 800)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
    while time.time() < p5_end:
        best_h = _cd_pass(best_h.copy(), rng)
        c5 = _eval(best_h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            best_h = _repair(_cd_pass(best_h, rng), n)
        best_h = _repair(best_h, n)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
