def _repair(h, n):

    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)

            else:

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _to_binary(h, n):
    S = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:S]] = 1.0
    return b

    h = h0.copy()
    corr = _corr_arr(h)
    mx = float(corr.max())
    dx = 2.0 / n
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    total_t = max(1.0, deadline - t0)
    # Track top-3 lags for multi-lag targeting
    top_lags = []
    
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total_t)
        T = 0.008 * (0.00001) ** frac
        step = base_step * (1.0 - 0.7 * frac)
        
        # Update top lags every 500 iterations (approx)
        if len(top_lags) == 0 or rng.random() < 0.002:
            order = np.argsort(corr)[::-1]
            top_lags = [(int(o) - (n - 1)) for o in order[:min(5, len(order))]
                        if abs(int(o) - (n - 1)) > 0]
            if not top_lags:
                top_lags = [1]
        
        k_act = top_lags[rng.randint(len(top_lags))] if top_lags else 0
        
        if rng.random() < target_frac and 0 < abs(k_act) < n:
            k = abs(k_act)
            if k_act > 0:
                contrib = h[k:] * (1.0 - h[:n-k])
                n_valid = contrib.size
                if n_valid > 2 and contrib.max() > 1e-6:
                    # Weighted random pick from top contributors
                    w = contrib / max(contrib.sum(), 1e-15)
                    i = int(rng.choice(n_valid, p=w))
                    j = i + k
                    # Reduce: increase h[i] or decrease h[j]

                    elif h[j] > 0.05:

def _polish(h0, deadline):
    """Very fine-grained local search with tiny steps"""
    n = len(h0)
    h = h0.copy()
    corr = _corr_arr(h)
    mx = float(corr.max())
    dx = 2.0 / n
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    
    steps = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    cycle = 0
    
    while time.time() < deadline:
        step = steps[cycle % len(steps)]
        # Determine which direction: try to move toward binary
        for i in range(n):
            if time.time() > deadline:
                break
            # Determine preferred direction: push toward 0 or 1
            if h[i] < 0.5:
                direction = 1  # push up toward 1 (if it's a "1" region)
            else:
                direction = -1  # push down toward 0
            # But also try opposite sometimes
            if rng_local_rand() < 0.3:
                direction = -direction
            
            d = direction * step
            # Find partner j
            j = (i + 1) % n
            if j == i:
                j = (i + 7) % n
            
            if h[i] + d < 0 or h[i] + d > 1:
                continue
            if h[j] - d < 0 or h[j] - d > 1:
                # Try other j
                j = (i + n // 2) % n
                if j == i or h[j] - d < 0 or h[j] - d > 1:
                    continue
            
            h[i] += d; h[j] -= d
            new_corr = _corr_arr(h)
            new_mx = float(new_corr.max())
            if new_mx < mx - 1e-15:
                mx = new_mx
                corr = new_corr
                cn = mx * dx
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
        
        cycle += 1
    
    return best_h, best_c
