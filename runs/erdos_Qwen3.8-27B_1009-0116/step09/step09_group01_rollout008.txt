    return a_gain, b_gain

3. For each swap, compute the Δ for these top lags
4. Prioritize swaps that reduce the worst lag

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _binarize(h, n):
    S = n // 2
    idx = np.argsort(h)[::-1][:S]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def _overlap_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _swap_delta(h, n, a, b, lags):
    """Compute change in overlap at given lags for swap: h[a]1->0, h[b]0->1."""
    ds = np.zeros(len(lags))
    for li, k in enumerate(lags):
        d = 0.0
        if 0 <= a - k < n:
            d += h[a - k]
        if 0 <= a + k < n:
            d -= h[a + k]
        if 0 <= b + k < n:
            d += h[b + k]
        if 0 <= b - k < n:
            d -= h[b - k]
        ds[li] = d
    return ds

def _fista_quick(h0, n, deadline):
    S = n * 0.5
    a = 500.0
    t0 = time.time()
    h = _proj(h0.copy(), S)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    best_h = h.copy()
    best_c = _c5(h)
    L = 10.0
    for it in range(100000):
        if time.time() - t0 > deadline:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
        # Compute gradient of soft C5
        prof = _overlap_profile(y)
        mx = prof.max()
        lam = 1.0 / a
        w = np.exp((prof - mx) / lam)
        w /= w.sum()
        # Gradient: dw/dh * dh/dy for each component
        g = np.zeros(n)
        for k in range(n):
            wk = w[k]
            if wk < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):
                j = i + (k - i)
                # O_k = sum_i h[i](1-h[i+k]), grad wrt h[i] = (1-h[i+k]) for the i term, -h[i-k] for the i-k term
                if i + k < n:
                    g[i] += wk * (1.0 - y[i + k])
                if i > 0:
                    g[i - k + k] -= 0  # simplified, let's use finite diff instead
        # Use simple subgradient: pick argmax lag
        kmax = int(np.argmax(prof))
        g = np.zeros(n)
        for i in range(n - kmax):
            g[i] += (1.0 - h[i + kmax])
            if i > 0:
                g[i] -= h[i - kmax]
        # Actually let's just use a simple projection-based approach
        step = 0.01
        xn = _proj(y - step * g, S)
        hp = h
        h = xn
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _sa_lag_targeted(h, rng, deadline, T0=0.08):
    """Lag-targeted simulated annealing on binary h."""
    n = len(h)
    c = _c5(h)
    bc = c
    bh = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    t0 = time.time()
    it = 0
    T = T0
    best_no_improve = 0
    
    while time.time() - t0 < deadline:
        elapsed = time.time() - t0
        T = max(1e-12, T0 * (1e-6) ** (elapsed / deadline))
        
        prof = _overlap_profile(h)
        mx = prof.max()
        # Top lags: within 5% of max
        tol = 0.05 * mx
        top_lags = np.where(prof >= mx - tol)[0].tolist()
        if not top_lags:
            top_lags = [int(np.argmax(prof))]
        # Limit to first 10 for speed
        top_lags = top_lags[:10]
        
        # Choose swap
        r = rng.random()
        if r < 0.6:
            # Random pair
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            # Lag-targeted: find swap that reduces worst lag
            k0 = top_lags[0]
            # For lag k0: Δ = h[a-k] - h[a+k] + h[b+k] - h[b-k]
            # Want to minimize Δ (make negative)
            # So want: h[a-k] small, h[a+k] large, h[b+k] small, h[b-k] large
            best_score = 1e9
            best_a, best_b = -1, -1
            # Sample a few candidates for speed
            n_sample = min(20, len(ones))
            sample_a = [ones[rng.randint(len(ones))] for _ in range(n_sample)]
            sample_b = [zeros[rng.randint(len(zeros))] for _ in range(n_sample)]
            for a in sample_a:
                score_a = (h[a + k0] if 0 <= a + k0 < n else 0) - (h[a - k0] if 0 <= a - k0 < n else 0)
                for b in sample_b:
                    if b == a:
                        continue
                    score_b = (h[b - k0] if 0 <= b - k0 < n else 0) - (h[b + k0] if 0 <= b + k0 < n else 0)
                    s = score_a + score_b
                    if s < best_score:
                        best_score = s
                        best_a, best_b = a, b
            if best_a >= 0:
                i, j = best_a, best_b
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            # Adjacent swap
            i = ones[rng.randint(len(ones))]
            if i > 0 and h[i - 1] < 0.5:
                j = i - 1
            elif i < n - 1 and h[i + 1] < 0.5:
                j = i + 1
            else:
                j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        cn = _c5(h)
        if cn <= c or rng.random() < np.exp(-(cn - c) / T):
            c = cn
            if c < bc - 1e-15:
                bc = c
                bh = h.copy()
                best_no_improve = 0
            else:
                best_no_improve += 1
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return bh, bc

def _fine_local(h, rng, deadline):
    """Very fine local search: only accept improvements."""
    n = len(h)
    c = _c5(h)
    bc = c
    bh = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    t0 = time.time()
    while time.time() - t0 < deadline:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn
            if c < bc:
                bc = c
                bh = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
    return bh, bc

def _perturb(h, rng, frac=0.02):
    """Randomly swap a fraction of ones/zeros."""
    n = len(h)
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    n_swap = max(1, int(len(ones) * frac))
    for _ in range(n_swap):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
    return h
