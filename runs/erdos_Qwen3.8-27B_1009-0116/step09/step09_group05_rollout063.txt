def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

    diff = target - h.sum()
    if abs(diff) > 1e-12:

        for idx in order:

                h[idx] += a

                diff += a

    n = len(h)
    L = k - (n - 1)
    g = np.zeros(n)
    M = abs(L)
    if M == 0:
        g[:] = 1.0 - 2.0 * h

        g[M:] -= h[:n - M]
    else:
        M2 = -L
        g[M2:] += (1.0 - h[:n - M2])
        g[:n - M2] -= h[M2:]
    return g

    top_k = np.argsort(c)[-K:]

        g += s[ii] * _lag_grad(h, k)
    return g, m

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
