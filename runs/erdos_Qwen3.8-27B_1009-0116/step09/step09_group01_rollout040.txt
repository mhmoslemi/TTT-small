def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break

    if abs(d) > 1e-13:

        for i in order:
            if abs(d) < 1e-13:

            else:
                a = min(-d, h[i])
            h[i] += a

    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size

        np.clip(x, 0.0, 1.0, out=x)

        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):

def _active_lags(h, tol=0.001):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:

        cn[1:] = np.cumsum(wN[1:])

    linN = cn
    q = np.zeros(n)

        H = np.fft.rfft(h, Nf)

    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _agrad(h, tol_frac=0.002):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    tol = max(1e-9, tol_frac * dx)

        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    return _gW(h, wP, wN)

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0

        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)

    h = _repair(h0, n)

    y = h.copy()
    tv = 1.0

def _loc_sa(h0, rng, dl, s0=5e-5, T0=0.001):
    """Local search with simulated annealing."""
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    P, Nv, _ = _pn(h)
    g = _agrad(h)
    recomp = 500
    it = 0
    acc_count = 0
    tot_count = 0
    
    while time.time() < dl:
        if it >= recomp:
            recomp = it + 500
            P, Nv, _ = _pn(h)
            g = _agrad(h)
            # Track acceptance rate to adapt step size
            acc_rate = acc_count / max(1, tot_count)
            if acc_rate < 0.3:
                s0 *= 0.7
            elif acc_rate > 0.7:
                s0 *= 1.3
            acc_count = 0
            tot_count = 0
        
        fr = min(1.0, (time.time() - t0) / max(1e-9, dl - t0))
        # Temperature schedule: geometric cooling
        T = T0 * (0.3 ** fr)
        base = max(1e-12, s0 * (1e-3) ** fr)
        
        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Pair move guided by gradient
            order = np.argsort(g)
            q = max(5, n // 7)
            i = int(order[rng.randint(q)])
            j = int(order[rng.randint(len(order)-q, len(order))])
        elif r < 0.85:
            # Random pair
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        else:
            # Block move
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        
        if i == j:
            it += 1
            continue
        
        sgn = 1 if g[i] <= g[j] else -1
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        if avail < 1e-14:
            it += 1
            continue
        
        mult = 0.3 + 1.7 * rng.random()
        if rng.random() < 0.05:
            mult *= 5.0
        d = sgn * min(avail, base * mult)
        
        h[i] += d
        h[j] -= d
        tot_count += 1
        
        cn = _c5(h)
        delta = cn - c
        if delta < -1e-14:
            c = cn
            acc_count += 1
            if c < best_c:
                best_c = c
                best_h = h.copy()
        elif T > 1e-15 and rng.random() < np.exp(-delta / max(T, 1e-15)):
            c = cn
            acc_count += 1
        else:
            h[i] -= d
            h[j] += d
        
        it += 1
    
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c
