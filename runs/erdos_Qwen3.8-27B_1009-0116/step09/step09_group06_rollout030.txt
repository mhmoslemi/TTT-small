    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for i in range(nsteps):

        taken = False

                taken = True
                break

def _pair_opt(h, i, j, n):
    """Find optimal shift t for pair (i,j) to minimize max correlation"""
    dx = 2.0 / n
    # Current correlations
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    # Coefficient of t for each lag
    # Δc[k] = dx * t * [(1-h[i+L]) - h[i-L] - (1-h[j+L]) + h[j-L]] where L = lag
    # But need to be careful with boundaries
    b = np.zeros(2*n - 1)
    for L in range(-(n-1), n):
        # L is the actual lag (k - (n-1))
        coeff = 0.0
        # h[i] contributes to c at lag L via p=i: h[i]*(1-h[i+L]) -> (1-h[i+L])*t
        ip = i + L
        if 0 <= ip < n:
            coeff += (1.0 - h[ip])
        # h[i] also appears as the "1-h" factor: (1-h[i-L])*h[i-L]... wait
        # c[k] = dx * sum_p h[p] * (1-h[p+L])
        # h[i] appears when p=i: h[i]*(1-h[i+L])
        # h[i] appears when p+L=i, i.e., p=i-L: h[i-L]*(1-h[i])
        # So Δc[k] from h[i] -> h[i]+t:
        #   p=i: t*(1-h[i+L])
        #   p=i-L: -t*h[i-L]
        
        ip = i + L
        if 0 <= ip < n:
            coeff += (1.0 - h[ip])  # from h[i] in first factor
        ip2 = i - L
        if 0 <= ip2 < n:
            coeff -= h[ip2]  # from (1-h[i]) in second factor

        jp = j + L
        if 0 <= jp < n:
            coeff -= (1.0 - h[jp])  # -t*(1-h[j+L])
        jp2 = j - L
        if 0 <= jp2 < n:
            coeff += h[jp2]  # -(-t)*h[j-L] = t*h[j-L]

    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi:
        return 0.0
    
    # The max of linear functions is convex, so minimize by finding the point
    # where the two largest lines cross
    # Simple approach: ternary search on [t_lo, t_hi]
    best_t = 0.0
    best_val = c0.max()
    for _ in range(100):

        if val < best_val:

        v2 = (c0 + b * t2).max()
        if v1 < v2:
            t_hi = t2
        else:
            t_lo = t1
    return best_t

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad_fast(h, k_idx, n):
    dx = 2.0 / n
    L = k_idx - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _sobj(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c[idx].max())
    s = np.clip(s - s.max(), -400, 400)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _sjac(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c[idx].max())
    s = np.clip(s - s.max(), -400, 400)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] > 1e-15:
            g += w[ii] * _subgrad_fast(x, k, n)
    return g

def _sfn(x, t):
    return x.sum() - t

def _sjac_s(x, t):
    return np.ones(len(x))

def _do_slsqp(h, alpha, m, mi, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sfn, 'jac': _sjac_s, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sobj, h, jac=_sjac, args=(alpha, m, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-16})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _do_proj_grad(h, alpha, m, steps, s0, n):

def _pair_b(h, i, j, n):
    dx = 2.0 / n
    L = np.arange(-(n - 1), n)
    iL1 = i + L
    iL2 = i - L
    jL1 = j + L
    jL2 = j - L
    m1 = (iL1 >= 0) & (iL1 < n)
    m2 = (iL2 >= 0) & (iL2 < n)
    m3 = (jL1 >= 0) & (jL1 < n)
    m4 = (jL2 >= 0) & (jL2 < n)
    b = np.zeros(2 * n - 1)
    if np.any(m1):
        b[m1] += (1.0 - h[np.clip(iL1[m1], 0, n - 1)])

    if np.any(m4):
        b[m4] += h[np.clip(jL2[m4], 0, n - 1)]
    return b * dx

def _pair_optimize(h, i, j, n, c0, ntern):
    if i == j:
        return 0.0
    b = _pair_b(h, i, j, n)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi:
        return 0.0
    for _ in range(ntern):
        w = t_hi - t_lo
        t1 = t_lo + w / 3.0
        t2 = t_lo + 2.0 * w / 3.0
        v1 = (c0 + b * t1).max()
        v2 = (c0 + b * t2).max()
        if v1 < v2:
            t_hi = t2
        else:
            t_lo = t1
    return (t_lo + t_hi) * 0.5

def _pair_search(h, n_iter, rng, n):

    best_c = _eval(h)
    cur_c = best_c
    c0 = _corrs(h)
    accepted = 0
    for it in range(n_iter):

        j = (i + 1 + rng.randint(0, max(1, n // 20))) % n
        if i == j:
            continue
        t = _pair_optimize(h, i, j, n, c0, 60)
        if abs(t) < 1e-15:
            continue
        hi, hj = h[i] + t, h[j] - t
        if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
            continue
        h[i] = min(max(hi, 0.0), 1.0)
        h[j] = min(max(hj, 0.0), 1.0)
        cx = _eval(h)
        if cx < cur_c - 1e-15:
            cur_c = cx
            c0 = _corrs(h)
            accepted += 1
            if cx < best_c:
                best_c = cx
                best_h = h.copy()
        else:
            h[i] -= t
            h[j] += t
    return best_h, best_c

def _lag_search(h, n_iter, rng, n):
