def _binary_local_search(h, n_iter, rng):
    n = len(h)
    # First, round to binary
    h = (h >= 0.5).astype(float)
    # Repair to ensure exactly n/2 ones
    n_ones = int(h.sum())
    if n_ones > n // 2:
        # Need to remove n_ones - n/2 ones
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        for k in range(n_ones - n // 2):
            h[ones[k]] = 0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for k in range(n // 2 - n_ones):
            h[zeros[k]] = 1
    
    best_c5 = _eval(h)
    best_h = h.copy()
    
    for _ in range(n_iter):
        # Best-improvement: try all swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = None
        best_c = best_c5
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c = _eval(h)
                if c < best_c:
                    best_c = c
                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        best_c5 = best_c
        best_h = h.copy()
    
    return best_h, best_c5
