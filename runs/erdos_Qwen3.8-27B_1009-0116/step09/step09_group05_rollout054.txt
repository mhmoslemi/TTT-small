                diff -= a

                diff += a

    # Smoothed max + penalty for sum constraint
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    smax = m + np.log(np.exp(s).sum()) / alpha
    # Sum penalty
    sm = x.sum() - n/2.0
    return smax + mu * sm * sm

    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()

        best_d = 0.0; best_c5 = c5c
        for frac in [0.15, 0.4, 0.65, 0.85, 1.0]:
            d = frac * (gp[i] - gp[j]) * 0.006
            hi, hj = h[i] + d, h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n; best_d = d
            else:
                h[i], h[j] = hi - d, hj + d
                break
        if best_d != 0.0:
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c; best_h = h.copy()
    return best_h, best_c

def _correlated_search(h, ntr, rng, base_step):
    n = len(h)
    c5cur = _eval(h); best_h = h.copy(); best_c = c5cur
    for _ in range(ntr):
        step = base_step * (0.4 + rng.rand() * 0.6)
        center = rng.randint(0, n)
        width = rng.randint(2, 12)
        lo = (center - width // 2) % n
        idx = np.array([(lo + k) % n for k in range(width)])
        d = step * rng.randn(width)
        d -= d.mean()
        hn = h.copy()
        hn[idx] += d
        if hn.min() < -0.2 or hn.max() > 1.2:
            continue
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5cur:
            c5cur = c5n; h = hn
            if c5n < best_c:
                best_c = c5n; best_h = h.copy()
    return best_h, best_c

def _polish(h, ntr, rng, scale):
    n = len(h); c5c = _eval(h)
    best_h = h.copy(); best_c = c5c
    for _ in range(ntr):
        i = rng.randint(n); j = (i + 1 + rng.randint(0, 12)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5c < best_c:
                    best_c = c5c; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _t():
        return time.time() - t0

    # Phase 1: Quick SLSQP polish with moderate alpha
    for alpha in [8000, 80000]:
        if _t() > 40:
            break
        h_new = _slsqp(best_h, alpha, 35, 250, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 2: L-BFGS-B with sum penalty (faster per iteration)
    for alpha, mu, iters, ml in [(50000, 5e4, 150, 40),
                                  (500000, 5e5, 120, 60),
                                  (5e6, 5e6, 100, 80)]:
        if _t() > 200:
            break
        h_new = _lbfgs(best_h, alpha, ml, mu, iters)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 3: Projected gradient descent (different from RMSProp)
    if _t() < 450:
        h_new = _proj_grad_step(best_h.copy(), 200000, 70, 400, 0.003, rng)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new
    if _t() < 600:
        h_new = _proj_grad_step(best_h.copy(), 2e6, 100, 300, 0.001, rng)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 4: SLSQP at high alpha for precision
    for alpha in [5e6, 5e7, 5e8]:
        if _t() > 700:
            break
        h_new = _slsqp(best_h, alpha, 70, 600, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 5: Small perturbation restarts
    for restart in range(3):
        if _t() > budget_s - 250:
            break
        hp = best_h.copy()
        d = 3e-5 * rng.randn(n)
        d -= d.mean()
        hp = _repair(np.clip(hp + d, 0, 1), n)
        h_new = _slsqp(hp, 1e6, 60, 200, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 6: Gradient-guided 2-point search
    if _t() < budget_s - 150:
        ntr = min(int((budget_s - 150 - _t()) / 0.006), 60000)
        best_h, best_c5 = _grad_2pt(best_h, ntr, 1e6, 60)

    # Phase 7: Correlated local search
    if _t() < budget_s - 60:
        ntr = min(int((budget_s - 60 - _t()) / 0.003), 80000)
        best_h, best_c5 = _correlated_search(best_h, ntr, rng, 4e-6)

    # Phase 8: Ultra-fine 2-point polish (multi-scale)
    if _t() < budget_s - 4:
        for scale in [8e-7, 3e-7, 1.5e-7, 8e-8]:
            if _t() > budget_s - 2:
                break
            ntr_s = min(int((budget_s - 2 - _t()) / 0.0008), 100000)
            best_h, best_c5 = _polish(best_h, ntr_s, rng, scale)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
