import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, K):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-K:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, K):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-K:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wki = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wki * (1.0 - x[:n - L])
            g[:n - L] -= wki * x[L:]
        else:
            M = -L
            g[:n - M] += wki * (1.0 - x[M:])
            g[M:] -= wki * x[:n - M]
    return g

def _sum_fn(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, K, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, K),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_grad(h, K, steps, step0):
    n = len(h)
    c = _corr_full(h)
    top = np.argsort(c)[-K:]
    wk = 2.0 / n
    g = np.zeros(n)
    for k in top:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _nesterov_loop(h, K, steps, step0):
    n = len(h)
    v = np.zeros(n)
    M = h.copy()
    cur_c = _eval(h)
    best = h.copy()
    best_c = cur_c
    step = step0
    for it in range(steps):
        g = _nesterov_grad(M, K, steps, step0)
        g -= g.mean()
        v = 0.85 * v + 0.15 * g
        if it % 50 == 0:
            step *= 0.97
        h_new = _proj(h - step * v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            M = h_new + 0.8 * (h_new - h)
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best = h.copy()
            step = min(step * 1.06, step0 * 20)
        else:
            step *= 0.5
            M = h.copy()
    return best, best_c

def _targeted_ls(h, niter, rng):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    for _ in range(niter):
        c = _corr_full(h)
        m = c.max()
        tol = m - 1e-6
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.argsort(c)[-5:]
        ka = active[rng.randint(len(active))]
        L = ka - (n - 1)
        if L >= 0:
            pairs = np.arange(n - L)
            contrib = h[L:L + n - L] * (1.0 - h[:n - L])
        else:
            M = -L
            pairs = np.arange(n - M)
            contrib = h[:n - M] * (1.0 - h[M:M + n - M])
        top_pairs = pairs[np.argsort(contrib)[-8:]]
        if len(top_pairs) < 2:
            continue
        i = int(top_pairs[rng.randint(len(top_pairs))])
        j = (i + L if L >= 0 else i - L) % n
        if j == i or j < 0 or j >= n:
            j = (i + 1) % n
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5c < best_c:
                    best_c = c5c
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, best_c

def _fine_2pt(h, niter, rng, sscale):
    n = len(h)
    c5c = _eval(h)

    best_c = c5c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.repeat(h, f)
    return _repair(out, n * f)
