def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
            else:
                a = min(-diff, h[i])
            h[i] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-15:
                break
    return h

    delta = np.zeros(2 * n - 1)

        iq2 = q - k + nm1
        if 0 <= iq2 < n:
            v += h[iq2]

    lo, hi = d_lo, d_hi
    for _ in range(44):
        mid = 0.5 * (lo + hi)
        vals = c + mid * delta
        km = np.argmax(vals)
        if delta[km] > 1e-14:
            hi = mid
        elif delta[km] < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    m = len(h)
    return _repair(np.repeat(h, f), m * f)

def _two_point_loop(h, n, c_arr, alpha, ml, n_rounds, rng, t0, budget):
    """Main optimization loop using optimal 2-point swaps."""
    best_c5 = _eval(h)
    g = _smooth_grad(h, alpha, ml)
    improved = 0
    stagnation = 0
    for rnd in range(n_rounds):
        if time.time() > budget - 30:
            break
        # Refresh gradient every 500 rounds
        if rnd % 500 == 0:
            g = _smooth_grad(h, alpha, ml)
            c_arr = np.correlate(h, 1.0 - h, mode='full')
        # Pick top gradient indices
        K = min(24, n)
        top = np.argpartition(-np.abs(g), K)[:K]
        n_try = 0
        for ii in range(K):
            p = int(top[ii])
            # Choose partner: nearby or low-gradient
            j_off = 1 + rng.randint(0, 30)
            q = (p + j_off) % n
            if q == p:
                continue
            # Feasible d range
            d_lo = max(-h[p], h[q] - 1.0)
            d_hi = min(1.0 - h[p], h[q])
            if d_lo >= d_hi - 1e-16:
                continue
            n_try += 1
            delta = _pair_delta(h, p, q, n)
            c_reduced = np.clip(c_arr.copy(), -1e10, 1e10)
            d_opt = _optimal_d(c_arr, delta, d_lo, d_hi)
            if abs(d_opt) < 1e-14:
                continue
            h[p] += d_opt
            h[q] -= d_opt
            h[p] = np.clip(h[p], 0.0, 1.0)
            h[q] = np.clip(h[q], 0.0, 1.0)
        # Periodic evaluation and repair
        if rnd % 200 == 199:
            h = _repair(h, n)
            c_new = _eval(h)
            if c_new < best_c5 - 1e-15:
                best_c5 = c_new
                improved += 1
                stagnation = 0
            else:
                stagnation += 1
            c_arr = np.correlate(h, 1.0 - h, mode='full')
            if stagnation > 4:
                # Small random reheat
                ns = 5e-5 * stagnation
                h = np.clip(h + ns * rng.randn(n), 0, 1)
                h = _repair(h, n)
                stagnation = 0
            alpha = min(alpha * 1.5, 5e9)
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse warm-up
    if _el() < 40:
        nc = max(40, n // 7)
        hc = _downsample(h, n // nc)
        for ac in [50000, 500000]:
            hc = _slsqp(hc, ac, 20, 150, 1e-14)
        hu = _upsample(hc, n // nc)
        if _eval(hu) < best_c5:
            best_c5 = _eval(hu)
            best_h = hu.copy()
            h = hu.copy()

    # Phase 2: SLSQP at full res (quick)
    if _el() < budget_s - 800:
        for alpha in [50000, 500000, 5000000]:
            if _el() > budget_s - 800:
                break
            h_new = _slsqp(best_h, alpha, 30, 800, 1e-15)
            c = _eval(h_new)
            if c < best_c5:
                best_c5 = c
                best_h = h_new

    # Phase 3: Main 2-point swap optimization
    h = best_h.copy()
    n_rounds = int(min(500000, (budget_s - 100 - _el()) * 80))
    h, best_c5 = _two_point_loop(h, n, np.correlate(h, 1.0-h, mode='full'),
                                  5000000, 40, n_rounds, rng, t0, budget_s)
    best_h = h.copy()

    # Phase 4: Multi-retry with perturbation
    if _el() < budget_s - 60:
        for attempt in range(3):
            if _el() > budget_s - 40:
                break
            ns = 8e-5 * (attempt + 1)
            hp = np.clip(best_h + ns * rng.randn(n), 0, 1)
            hp = _repair(hp, n)
            c = _eval(hp)
            if c < best_c5:
                best_c5 = c
                best_h = hp.copy()
            hp = _slsqp(hp, 10000000, 40, 400, 1e-15)
            c = _eval(hp)
            if c < best_c5:
                best_c5 = c
                best_h = hp.copy()
            # Quick 2-point polish on new best
            if _el() < budget_s - 20:
                hp, c2 = _two_point_loop(hp, n, np.correlate(hp, 1.0-hp,
                              mode='full'), 20000000, 30, 5000, rng, t0, budget_s)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = hp.copy()

    # Phase 5: Ultra-fine final pass
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        c5c = _eval(h2)
        n_iter = int(min(200000, (budget_s - 2 - _el()) * 2000))
        for _ in range(n_iter):
            if _el() > budget_s - 1:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 40)) % n
            d = 5e-7 * rng.randn()
            ni = h2[i] + d
            nj = h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c = _eval(h2)
                if c < c5c:
                    c5c = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
