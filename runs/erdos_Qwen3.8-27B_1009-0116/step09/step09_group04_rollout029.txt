import numpy as np
from scipy.optimize import minimize, linprog
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _corr_grad(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        m = n - 1 - L
        g[:m+1] += (1.0 - h[L:])
        g[L:] -= h[:m+1]
    else:
        M = -L
        m = n - 1 - M
        g[M:] += (1.0 - h[:m+1])
        g[:m+1] -= h[M:]
    return g * dx

def _pmax_obj(x, p, k):
    c = _corr(x)
    idx = np.argsort(c)[-k:]
    top = c[idx]
    return np.mean(top**p)**(1.0/p)

def _pmax_grad(x, p, k):
    n = len(x)
    c = _corr(x)
    idx = np.argsort(c)[-k:]
    top = c[idx]
    s = np.mean(top**p)
    base = s**(1.0/p - 1.0) / k
    g = np.zeros(n)
    for ii, kk in enumerate(idx):
        w = base * top[ii]**(p-1)
        g += w * _corr_grad(x, kk - (n-1))
    return g

def _cf(x, hn):
    return x.sum() - hn

def _cj(x, hn):
    return np.ones(len(x))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_solve(h, p, k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_pmax_obj, h, jac=_pmax_grad, args=(p, k),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _active_lp(h, k, step_frac):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-k:]
    lags = idx - (n - 1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ineq = np.zeros((k, n + 1))
    b_ineq = np.zeros(k)
    for ii, kk in enumerate(idx):
        g = _corr_grad(h, lags[ii])
        A_ineq[ii, :n] = g
        A_ineq[ii, n] = -1.0
        b_ineq[ii] = -c[ii]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, 1.0))
    try:
        r = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if r.status == 0:
            dh = r.x[:n]
            return _repair(h + step_frac * dh, n)
    except Exception:
        pass
    return h

def _grad_local(h, nit, rng, p, k, base_step):
    n = len(h)
    cc = _eval(h)
    bh, bc = h.copy(), cc
    for _ in range(nit):

        step = base_step * (0.5 + rng.rand())
        hn = _proj(h - step * g, n * 0.5)
        cn = _eval(hn)
        if cn < cc:
            h, cc = hn, cn
            if cn < bc:
                bc, bh = cn, h.copy()
    return bh, bc

def _polish(h, c5, t_end, rng):

    h2, c2 = h.copy(), c5

            else:
