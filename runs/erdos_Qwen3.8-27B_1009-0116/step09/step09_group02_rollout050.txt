        best_h, best_c5 = _pair_polish(best_h, nf, rng)

    # Final SLSQP micro-polish at very high alpha
    if _el() < budget_s - 1:
        h_new = _slsqp(best_h.copy(), 1e9, 15, 200, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
        best_h, best_c5 = _pair_polish(best_h, 5000, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
