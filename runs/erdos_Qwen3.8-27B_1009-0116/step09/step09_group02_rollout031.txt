        v -= v.mean()
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        taken = False
        for _ in range(12):
            h_new = _proj_box_sum(h - step * v, n * 0.5)
            c_new = _eval_c5(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 40.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(max(step * 1.4, step0 * 0.5), step0 * 40.0)
    return _repair(h, n), cur_c

def _anneal_search(h, n_iter, rng, temp0, base_step, topk):
    n = len(h)
    cur_c = _eval_c5(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(n_iter):
        T = max(temp0 * (0.99995 ** it), 1e-12)
        step = base_step * (0.4 + 0.8 * rng.rand())
        g = _smooth_grad(h, 50000.0, topk)
        n_pert = int(rng.choice([2, 3, 4, 5, 7]))
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        ng = np.linalg.norm(g)
        if ng > 1e-15:
            d -= 0.4 * step * g[idx] / ng
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.1 or h_new.max() > 1.1:
            continue
        h_new = _repair(h_new, n)
        c_new = _eval_c5(h_new)
        if c_new < cur_c or (c_new > cur_c and rng.rand() < np.exp(-(c_new - cur_c) / max(T, 1e-12))):
            cur_c = c_new
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    cur_c = _eval_c5(h)
    best_h = h.copy()
    best_c = cur_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n
        d = (5e-9 + 4e-8 * rng.rand()) * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d
            h[j] -= d
            c_new = _eval_c5(h)
            if c_new < cur_c - 1e-16:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def _group_polish(h, n_iter, rng, gs=5):
    n = len(h)
    cur_c = _eval_c5(h)
    best_h = h.copy()
    best_c = cur_c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 35, size=gs)) % n)
        d = (5e-9 + 6e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c_new = _eval_c5(h_new)
        if c_new < cur_c - 1e-16:
            cur_c = c_new
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    return best_h, best_c

def _alt_start_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return _repair(h, n)

def _alt_start_alt(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return _repair(h, n)

def _alt_start_sine(n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * x)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    h = _repair(h_init, n)
    best_h = h.copy()
    best_c5 = _eval_c5(best_h)

    # Check alternative starts
    for mk in [_alt_start_block, _alt_start_alt, _alt_start_sine]:
        hs = mk(n)
        c = _eval_c5(hs)
        if c < best_c5:
            best_c5 = c
            best_h = hs

    # Phase 1: L-BFGS-B with penalty, increasing alpha
    alpha_seqs = [
        (30000, 20, 1e6, 600),
        (100000, 25, 1e5, 600),
        (500000, 30, 1e5, 500),
        (2000000, 35, 1e4, 500),
        (10000000, 40, 1e4, 400),
        (50000000, 50, 1e3, 400),
    ]
    for alpha, tk, pen, mi in alpha_seqs:
        if _el() > 220:
            break
        h_new = _lbfgsb_solve(best_h.copy(), alpha, tk, pen, mi)
        c5n = _eval_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: SLSQP with higher alpha (constraint-based)
    for alpha, tk in [(200000000, 25), (1000000000, 20), (5000000000, 15)]:
        if _el() > 350:
            break
        h_new = _slsqp_solve(best_h.copy(), alpha, tk, 500)
        c5n = _eval_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Momentum projected gradient (multiple configs)
    if _el() < budget_s - 200:
        cfgs = [
            (50000, 30, 1000, 0.005, 0.90),
            (300000, 40, 800, 0.003, 0.92),
            (1000000, 50, 700, 0.002, 0.93),
            (5000000, 60, 500, 0.001, 0.94),
            (20000000, 80, 400, 0.0005, 0.95),
            (100000000, 100, 350, 0.0003, 0.96),
        ]
        for alpha, tk, steps, s0, mu in cfgs:
            if _el() > budget_s - 150:
                break
            h_new, _ = _mom_grad(best_h.copy(), alpha, tk, steps, s0, mu)
            c5n = _eval_c5(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Another L-BFGS-B pass at high alpha
    if _el() < budget_s - 100:
        h_new = _lbfgsb_solve(best_h.copy(), 500000000, 15, 1e3, 400)
        c5n = _eval_c5(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Simulated annealing
    if _el() < budget_s - 50:
        rem = budget_s - 50 - _el()
        ntries = int(rem * 350)
        best_h, best_c5 = _anneal_search(best_h, ntries, rng, 8e-5, 4e-5, 12)

    # Phase 6: Fine 2-point polish
    if _el() < budget_s - 20:
        rem = budget_s - 20 - _el()
        ntries = int(rem * 800)
        best_h, best_c5 = _fine_polish(best_h, ntries, rng)

    # Phase 7: Group polish
    if _el() < budget_s - 5:
        rem = budget_s - 5 - _el()
        ntries = int(rem * 450)
        best_h, best_c5 = _group_polish(best_h, ntries, rng, 5)

    # Phase 8: Final ultra-fine
    if _el() < budget_s - 1:
        rem = max(budget_s - 1 - _el(), 0.5)
        ntries = int(rem * 1000)
        best_h, best_c5 = _fine_polish(best_h, ntries, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, n)
