        for pert_scale in [0.01, 0.005]:
            if _el() > budget_s - 60:
                break
            h_pert = _repair(best_h + pert_scale * rng.randn(n) * 0.01, n)
            h_pert = _slsqp(h_pert, 5000000, 80, 500)
            c5n = _eval(h_pert)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_pert
                h_pert = _slsqp(h_pert, 500000000, 150, 600)
                c5n2 = _eval(h_pert)
                if c5n2 < best_c5:
                    best_c5 = c5n2
                    best_h = h_pert

    # Phase 5: Fine coordinate descent
    if _el() < budget_s - 25:
        rem = budget_s - 25 - _el()
        best_h, best_c5 = _fine_cd(best_h, int(rem * 350), rng, 1e-7)

    # Phase 6: Top-k equalization
    if _el() < budget_s - 12:
        rem = budget_s - 12 - _el()
        best_h, best_c5 = _topk_eq(best_h, int(rem * 200), rng, 10)

    # Phase 7: Group polish
    if _el() < budget_s - 6:
        rem = budget_s - 6 - _el()
        best_h, best_c5 = _grp_cd(best_h, int(rem * 250), rng, 4, 3e-8)

    # Phase 8: Ultra-fine final
    if _el() < budget_s - 1:
        rem = budget_s - 1 - _el()
        best_h, best_c5 = _fine_cd(best_h, int(rem * 600), rng, 2e-9)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
