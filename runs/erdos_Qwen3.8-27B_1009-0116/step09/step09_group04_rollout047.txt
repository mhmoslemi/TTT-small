import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_single(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    L = k_idx - (n - 1)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _sobj(x, alpha, ml):
    c = _eval_corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    c = _eval_corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(len(x))
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_single(x, k)
    return g

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _cycle_descent(h, n, nit, top_m, step_init, rng):
    c = _eval_corr(h)
    best_h = h.copy()
    best_c = c.max()
    step = step_init
    vel = np.zeros(n)
    mu = 0.8
    for it in range(nit):
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_single(h, k)
        g /= top_m
        vel = mu * vel + (1.0 - mu) * g
        gn = np.linalg.norm(vel)
        if gn < 1e-14:
            break
        vel /= gn
        taken = False
        for _ in range(10):
            hn = _proj_sum(h - step * vel, n * 0.5)
            cn = _eval_corr(hn).max()
            if cn < c.max():

                c = _eval_corr(h)
                if c.max() < best_c:
                    best_c, best_h = c.max(), h.copy()
                taken = True
                step = min(step * 1.1, step_init * 200.0)
                break
            step *= 0.4
        if not taken:
            step *= 0.7
            vel = 0.5 * vel
    return best_h, best_c

def _active_set_qp(h, n, T, top_m):
    c = _eval_corr(h)
    active = np.argsort(c)[-top_m:]
    h_var = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(h_var - h))

    except Exception:

def _local_swaps(h, n, nit, rng, max_shift, scale):
    c = _eval(h)
    best_h = h.copy()
    best_c = c
    for _ in range(nit):
        i = rng.randint(n)

            h[j] += d
        else:

            if cn < best_c:

Let me also double-check: the `_eval` function computes `np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n`. The correlate has length 2n-1, indexed 0 to 2n-2. The center (lag 0) is at index n-1. For h with n elements:

- correlate[n-1] = sum h[i]*(1-h[i]) (all n terms, lag 0)
- correlate[n-1+k] = sum h[i+k]*(1-h[i]) for valid i (positive lag k)
- correlate[n-1-k] = sum h[i]*(1-h[i+k]) for valid i (negative lag k)

For binary h: ∫h(1-h) = 0 at lag 0.
For h = 0.5 everywhere: ∫h(1-h) = 0.25 * 2 = 0.5.

from scipy.optimize import minimize
import cvxpy as cp
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_one(h, ki):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    L = ki - (n - 1)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _proj(h, tgt):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    return x.sum() - t

def _sobj(x, a, ml):

    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = a * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / a

    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = a * (c[idx] - m)
    s -= s.max()

    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        g += w[ii] * _grad_one(x, k)
    return g

def _qp(h, n, T, tm):
    c = _eval_corr(h)
    act = np.argsort(c)[-tm:]
    v = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(v - h))
    con = [v >= 0, v <= 1, cp.sum(v) == n / 2.0]
    for k in act:
        g = _grad_one(h, k)
        con.append(g @ v <= T - c[k] + float(g @ h))
    p = cp.Problem(obj, con)
    for solver in [cp.SCS, cp.ECOS, cp.OSQP]:

            if v.value is not None:
