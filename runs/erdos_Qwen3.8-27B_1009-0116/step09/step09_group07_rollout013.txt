Let me also consider: **can we exploit the near-convexity?**

At the optimum, the top few lags are all at (or very near) the maximum value M. If we enforce c_k = M for all active k (equalization), we get a system of equations. The solution would be a "saddle point" where the gradient of the active constraints is non-negative (for minimization).

This is like a KKT condition: at the optimum, there exist multipliers λ_k ≥ 0 for the active constraints such that:
∇_h (sum_k λ_k c_k(h)) + ∇_h (penalty for bounds) + μ * 1 = 0

This is a semidefinite program (SDP) relaxation, but for our purposes, we can approximate it with:
1. Identify active lags A = {k : c_k > M - ε}
2. Set λ_k = 1/|A| for k ∈ A
3. Minimize sum_{k∈A} (1/|A|) c_k(h) with projected gradient
4. Update A, repeat

This "equalization" approach might help because it distributes the optimization effort across the active lags rather than just chasing the single maximum.

def _proj_constraint(h, n):
    target = n / 2.0

        d = target - s

            break

            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:

        else:

            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + d / cnt, 0.0, 1.0)
    return h

def _equalize_obj(x, w):
    dx = 2.0 / len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.sum(w * c))

def _equalize_grad(x, w):
    n = len(x)
    nm1 = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for k_idx in range(len(w)):
        wv = w[k_idx]
        if wv < 1e-15:

        L = k - nm1
        wv *= dx
        if L >= 0:

            g[:n - L] -= wv * x[L:]
        else:
            M = -L
            g[:n - M] += wv * (1.0 - x[M:])
            g[M:] -= wv * x[:n - M]
    return g

def _nesterov_eq(h, n, alpha, budget_s):

    c = _corr_dx(h)

def _slsqp_obj(x, alpha, m_lags):

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx

def _slsqp_grad(x, alpha, m_lags):

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h
