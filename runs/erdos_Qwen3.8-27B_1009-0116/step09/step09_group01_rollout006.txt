    """Project h onto {0<=h<=1, sum(h)=S}"""
    x = h.copy()
    for _ in range(50):
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = x < 0.999999
        else:
            mask = x > 1e-7
        if mask.sum() == 0:

        np.clip(x, 0, 1, out=x)

    n = len(h)

    # Find top lags
    idx = np.argsort(corr)[::-1][:n_active]
    # For each active lag, compute gradient

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(h, S):
    x = h.copy()
    for _ in range(100):
        d = S - x.sum()
        if abs(d) < 1e-15:
            break
        m = (x < 1.0 - 1e-12) if d > 0 else (x > 1e-12)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _grad_top(h, top_k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:top_k]
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in idx:
        lag = int(k) - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if hi > lo:
            grad[lo:hi] += (1.0 - h[lo+lag:hi+lag])
            grad[lo+lag:hi+lag] -= h[lo:hi]
    return dx * grad

def _grad_smooth(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(corr.max())
    sh = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    w = np.exp(sh)
    tot = w.sum()
    if tot < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        tot = 1.0
    w /= tot
    thresh = 0.005
    active = w > thresh
    if not active.any():
        active = w >= w.max()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(corr)):
        if not active[k]:
            continue
        lag = int(k) - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if hi > lo:
            ww = w[k]
            grad[lo:hi] += ww * (1.0 - h[lo+lag:hi+lag])
            grad[lo+lag:hi+lag] -= ww * h[lo:hi]
    return dx * grad

    t0 = time.time()
    n = len(h0)
    S = 0.5 * n
    h = _proj(h0, S).copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    lr = lr_init
    stuck = 0
    mode_top = top_k > 0
    while time.time() < t_end:
        if mode_top:
            g = _grad_top(h, top_k)
        else:
            g = _grad_smooth(h, alpha)
        gnorm = np.sqrt(np.dot(g, g))
        if gnorm < 1e-15:
            break
        g = g / gnorm
        for _ in range(15):
            h_new = _proj(h - lr * gnorm * g, S)
            c_new = _c5(h_new)
            if c_new < c - 1e-16:
                c = c_new
                h = h_new
                break
            lr *= 0.5
        else:
            lr = lr_init * 0.1
            stuck += 1
            if stuck > 30:
                h = _proj(h + 0.02 * rng.randn(n), S)
                c = _c5(h)
                lr = lr_init
                stuck = 0
            continue
        stuck = 0
        lr = min(lr * 1.05, 0.5)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    if _c5(h) < best_c:

        best_h = h.copy()
    return best_h, best_c
