def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

import numpy as np
import time

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _minimax_1d(C, D):
    n = len(C)
    # Quick exits
    if np.all(D <= 0):
        return 0.0
    if np.all(D >= 0):
        return 1.0
    
    lo = np.max(np.minimum(C, C + D))
    hi = lo + max(1e-14, 1.0)  # use hi = upper envelope at boundaries
    hi = min(np.max(C), np.max(C + D))
    
    if hi - lo < 1e-14:
        return 0.0
    
    pos = D > 1e-15
    neg = D < -1e-15
    
    best_t = 0.0
    for _ in range(35):
        v = 0.5 * (lo + hi)
        
        # Check D_s = 0: need C_s <= v
        zero = ~(pos | neg)
        if np.any(zero) and np.max(C[zero]) > v:
            lo = v
            continue
        
        feasible = True
        t_lo = 0.0
        t_hi = 1.0
        
        if np.any(neg):
            # t >= (v-C_s)/D_s for D_s < 0, i.e., t >= (C_s-v)/|D_s|
            b = (C[neg] - v) / (-D[neg])
            ml = np.max(b)
            if ml > t_lo:
                t_lo = ml
        
        if np.any(pos):
            # t <= (v-C_s)/D_s for D_s > 0
            b = (v - C[pos]) / D[pos]
            mh = np.min(b)
            if mh < t_hi:
                t_hi = mh
        
        if t_lo <= t_hi:
            feasible = True
            best_t = 0.5 * (t_lo + t_hi)
        else:
            feasible = False
        
        if feasible:
            hi = v
        else:
            lo = v
    
    return best_t

def _coord_sweep(h, n, L, order):
    """One sweep of coordinate descent with incremental L. Returns (h, L, improved)."""
    dx = 2.0 / n
    s_range = np.arange(-(n-1), n)
    m = 2 * n - 1
    improved = False
    
    for j in order:
        js = j + s_range  # j + s for each s
        jm = j - s_range  # j - s for each s
        
        I1 = (jm >= 0) & (jm < n)  # j-s valid
        I2 = (js >= 0) & (js < n)  # j+s valid
        
        h_jm = np.zeros(m)
        h_js = np.zeros(m)
        if np.any(I1):
            h_jm[I1] = h[jm[I1]]
        if np.any(I2):
            h_js[I2] = h[js[I2]]
        
        D = dx * ((1.0 - h_jm) * I1.astype(np.float64) - h_js * I2.astype(np.float64))
        
        if np.all(np.abs(D) < 1e-18):
            continue
        
        C = L - D * h[j]
        t_opt = _minimax_1d(C, D)
        
        if abs(t_opt - h[j]) > 1e-14:
            delta = t_opt - h[j]
            h[j] = t_opt
            L += delta * D
            improved = True
    
    return h, L, improved
