    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        d = 3e-5 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Fast coordinate descent (main optimization)
    stall = 0
    threshold = best_c5 - 1e-7
    p1_end = t0 + budget_s * 0.60
    while time.time() < p1_end:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            threshold = best_c5 - 1e-7
        else:
            stall += 1
        if stall > 60 and time.time() < p1_end - 30:
            h = best_h.copy()
            stall = 0
            rng.shuffle(h)
            h = _repair(h, n)

    # Phase 2: SLSQP with increasing sharpness
    if time.time() < t0 + budget_s * 0.82:
        for alpha in [5000, 50000, 500000, 3000000]:
            if time.time() > t0 + budget_s * 0.80:
                break
            h_new = _slsqp(best_h.copy(), alpha, 60, 600)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More CD from best solution
    if time.time() < t0 + budget_s * 0.93:
        h2 = best_h.copy()
        while time.time() < t0 + budget_s * 0.91:
            h2 = _cd_pass(h2, rng)
            h2 = _repair(h2, n)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 4: Final SLSQP polish
    if time.time() < t0 + budget_s - 30:
        h_new = _slsqp(best_h.copy(), 10000000, 100, 400)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Pair local search for final polish
    if time.time() < t0 + budget_s - 3:
        remaining = budget_s - 5 - (time.time() - t0)
        ntr = int(remaining / 0.0005)
        ntr = min(ntr, 500000)
        best_h, best_c5 = _pair_search(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
