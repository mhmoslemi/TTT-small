from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _eval_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _proj(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _lag_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g

def _con_fun(x, tgt):
    return x.sum() - tgt

def _con_jac(x, tgt):
    return np.ones(len(x))

def _soft_obj(x, alpha, ml):
    c = _eval_c(x)
    idx = np.argsort(c)[-ml:]
    m = c[idx].max()
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _soft_grad(x, alpha, ml):
    c = _eval_c(x)
    n = len(x)
    idx = np.argsort(c)[-ml:]
    m = c[idx].max()
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] > 1e-15:
            g += w[ii] * _lag_grad(x, k - (n - 1))
    return g

def _slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (tgt,)}]
    try:
        r = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return h

def _lbfgs(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (tgt,)}]
    try:
        r = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': maxiter * 5})
        return _repair(r.x, n)
    except:
        return h

def _proj_subgrad(h, nsteps, step0, ml, rng):
    n = len(h)
    cur = _eval(h)
    bc = cur
    step = step0
    for _ in range(nsteps):
        c = _eval_c(h)
        idx = np.argsort(c)[-ml:]
        k = idx[np.argmax(c[idx])]
        g = _lag_grad(h, k - (n - 1))
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        taken = False
        for _ in range(12):
            hn = _proj(h - step * g, n * 0.5)
            cn = _eval(hn)
            if cn < cur - 1e-16:
                h, cur = hn, cn
                if cur < bc:
                    bc = cur
                step = min(step * 1.05, 0.08)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 12.0)
    return h, bc

def _local_search(h, n_iter, rng):
    n = len(h)
    cur = _eval(h)
    best = h.copy()
    bc = cur
    stall = 0
    for it in range(n_iter):
        c = _eval_c(h)
        kmax = np.argmax(c)
        L = kmax - (n - 1)
        g = _lag_grad(h, L)
        top_i = np.argsort(g)[-8:]
        bot_j = np.argsort(g)[:8]
        improved = False
        for _ in range(15):
            i = int(rng.choice(top_i))
            j = int(rng.choice(bot_j))
            if i == j:
                continue
            for d in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                ni, nj = h[i] - d, h[j] + d
                if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                    h[i], h[j] = ni, nj
                    cn = _eval(h)
                    if cn < cur:
                        cur = cn
                        if cn < bc:
                            bc, best = cn, h.copy()
                        stall = 0
                        improved = True
                        break
                    else:
                        h[i], h[j] = ni + d, nj - d
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 200:
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    d = 0.005 * (1.0 if rng.rand() < 0.5 else -1.0)
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        h[i] += d; h[j] -= d
                        cn = _eval(h)
                        if cn < cur:
                            cur = cn
                            if cn < bc:
                                bc, best = cn, h.copy()
                            stall = 0
    return best, bc

def _fine_polish(h, n_iter, rng):
    n = len(h)
    cur = _eval(h)
    best = h.copy()
    bc = cur
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        for d in [5e-7, -5e-7, 2e-7, -2e-7, 1e-7, -1e-7, 5e-8]:
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < cur:
                    cur = cn
                    if cn < bc:
                        bc, best = cn, h.copy()
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
    return best, bc
