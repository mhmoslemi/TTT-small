import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_h(h, alpha, ktop):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ktop, 2 * n - 2))[:ktop]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[:n - L] += w[ii] * (1.0 - h[L:])
            g[L:] -= w[ii] * h[:n - L]
        else:
            M = -L
            g[M:] += w[ii] * (1.0 - h[:n - M])
            g[:n - M] -= w[ii] * h[M:]
    return g * dx

    return _repair(h - lr * g, n)

        if c_cur <= best_c + 1e-8:

        else:
            lr *= 0.75
    return best_h, best_c

def _topk_smooth_and_grad(h, alpha, ktop):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()

    s -= s.max()

def _sum_fun(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ktop, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_topk_smooth_and_grad, h, jac=None, method='SLSQP',
                       args=(alpha, ktop), bounds=bnds, constraints=cns,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    n = len(h)
    return _repair(np.repeat(h, f), n * f)

    for i in range(n):
