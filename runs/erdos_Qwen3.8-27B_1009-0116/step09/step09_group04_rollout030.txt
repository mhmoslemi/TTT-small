    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]

    s -= s.max()

    w /= w.sum()
    g = np.zeros(n)

        else:
            M = -L
            g[:n-M] += wk * (1.0 - h[M:])

        else:

    bnds = [(0.0, 1.0)] * n

def _adam_opt(h, alpha, ml, steps, lr):
    n = len(h)
    b1, b2 = 0.9, 0.999
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    best = h.copy()
    best_c = _eval(h)
    for t in range(1, steps+1):
        g = _smooth_grad(h, alpha, ml)
        m_arr = b1*m_arr + (1-b1)*g
        v_arr = b2*v_arr + (1-b2)*g*g
        mhat = m_arr / (1 - b1**t)
        vhat = v_arr / (1 - b2**t)
        h = _proj_sum(h - lr * mhat / (np.sqrt(vhat) + 1e-8), n*0.5)
        c = _eval(h)
        if c < best_c:
            best_c = c
            best = h.copy()
    return best, best_c
