import numpy as np
import time

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * (2.0 / n)

def _c5_k(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * (2.0 / n), int(np.argmax(corr)) - (n - 1)

def _repair(h, n):
    S = float(n * 0.5)
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-12:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = int(m.sum())
        if c == 0:
            break
        x[m] += d / c
    return x

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) / n
    x1 = (np.arange(m) + 0.5) / m
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _subgrad_step(h, k_act, n_active=8, tol=0.02):
    """Compute subgradient from top active lags"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    thresh = m - max(tol * m, 0.5)
    active = np.where(corr >= thresh)[0]
    # Sort by correlation value descending, take top n_active
    if len(active) > n_active:
        vals = corr[active]
        top = active[np.argsort(vals)[-n_active:]]
    else:
        top = active
    g = np.zeros(n)
    for k_idx in top:
        lag = int(k_idx) - (n - 1)
        if lag == 0:
            g += 2.0 * (1.0 - 2.0 * h)
        elif lag > 0 and lag < n:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
        elif lag < 0 and -lag < n:
            la = -lag
            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
    if len(top) > 0:
        g /= len(top)
    return g

def _subgrad_opt(h0, deadline, base_step=0.03):
    """Projected subgradient with line search"""
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c, _ = _c5_k(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    step = base_step
    stuck = 0
    while time.time() < deadline:
        g = _subgrad_step(h, 0)
        if np.all(g == 0):
            break
        improved = False
        for _ in range(6):
            hn = _proj(h - step * g, S)
            cn, _ = _c5_k(hn)
            if cn < c - 1e-13:
                h = hn; c = cn; improved = True
                if c < best_c - 1e-13:
                    best_c = c; best_h = h.copy()
                break
            step *= 0.6
        if not improved:
            stuck += 1
            # Random perturbation
            if stuck > 20:
                idx = np.random.randint(0, n, size=4)
                for i in range(2):
                    d = (np.random.random() - 0.5) * 0.04
                    h[idx[i]] += d
                    h[3-i] -= d
                h = _proj(h, S)
                stuck = 0
            step = min(step * 1.1, 0.05)
        else:
            step = min(step * 1.02, 0.05)
    return best_h, best_c

def _sa_opt(h0, rng, deadline):
    """Targeted SA with active constraint awareness"""
    n = len(h0)
    h = h0.copy()

    mx = float(corr.max())
    dx = 2.0 / n
    best_c = mx * dx
    best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    best_it = 0
    it = 0
    
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_t)
        # Temperature: start higher, cool down
        T = 0.008 * (0.01) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        ka = abs(k_act)
        
        # 70% targeted moves, 30% random
        if rng.random() < 0.7 and 0 < ka < n:
            if k_act > 0:
                mask = (h[:n-ka] > 0.01) & (h[ka:] < 0.99)
                pos = np.nonzero(mask)[0]
            else:
                mask = (h[ka:] > 0.01) & (h[:n-ka] < 0.99)
                pos = np.nonzero(mask)[0]
            if len(pos) < 3:
                pos = np.arange(n)
            i = int(pos[rng.randint(len(pos))])
            j = i + k_act  # j = i + k_act (works for both signs)
            if j < 0 or j >= n:
                j = rng.randint(n)
            # Decide direction: try to reduce h[i]*(1-h[j])
            # If h[i] > 0.5 and h[j] < 0.5, moving mass from i to j helps
            if h[i] > 0.5 and h[j] < 0.5:
                d = rng.random() * min(0.04, h[i] - 0.5)
            elif h[i] < 0.5 and h[j] > 0.5:
                d = -rng.random() * min(0.04, 0.5 - h[i])
            else:
                d = (rng.random() - 0.5) * 0.04
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                it += 1
                continue
            d = (rng.random() - 0.5) * 0.05
        
        # Apply and check bounds
        if h[i] + d < 0: d = -h[i]
        if h[i] + d > 1: d = 1.0 - h[i]
        if h[j] - d < 0: d = h[j]
        if h[j] - d > 1: d = h[j] - 1.0
        if abs(d) < 1e-10:
            it += 1
            continue
        
        h[i] += d; h[j] -= d
        corr2 = np.correlate(h, 1.0 - h, mode='full')
        mx2 = float(corr2.max())
        delta = mx2 - mx
        if delta < -1e-14:
            h[i] += 0;  # noop, already applied
            mx = mx2; corr = corr2
            cn = mx * dx
            if cn < best_c - 1e-14:
                best_c = cn; best_h = h.copy(); best_it = it
        elif delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15)):
            mx = mx2; corr = corr2
        else:
            h[i] -= d; h[j] += d
        it += 1
        
        # Restart if stuck for too long
        if it - best_it > 50000 and frac < 0.9:

            corr = np.correlate(h, 1.0 - h, mode='full')
            mx = float(corr.max())
            # Perturb

                h[a] += dd; h[b] -= dd
            h = _proj(h, n * 0.5)
            corr = np.correlate(h, 1.0 - h, mode='full')
            mx = float(corr.max())
            best_it = it
    
    return best_h, best_c

def _fista_sm(h0, alpha, deadline, L0=5.0):
    """FISTA with soft-max objective"""
    n = len(h0)
    S = n * 0.5
    dx = 2.0 / n
    t0 = time.time()
    
    def sval(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        a = alpha
        e = np.sum(np.exp(np.clip(a*(c - m), -300, 0)))
        return m + np.log(e) / a
    
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = sval(h); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    Lc = L0
    stale = 0
    
    for it in range(100000):
        if time.time() > deadline:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            y = _proj(y, S)
            fy = sval(y)
        else:
            y = h; fy = fh
        
        # Gradient of soft-max
        c = np.correlate(y, 1.0 - y, mode='full') * dx
        m = c.max()
        a = alpha
        w = np.exp(np.clip(a*(c - m), -300, 0))
        w /= w.sum()
        
        # Compute gradient (vectorized over active lags)
        g = np.zeros(n)
        for k_idx in range(len(w)):
            wi = w[k_idx]
            if wi < 1e-8:
                continue
            lag = k_idx - (n - 1)
            if lag == 0:
                g += 2.0 * (1.0 - 2.0 * y) * wi
            elif 0 < lag < n:
                g[:n-lag] += (1.0 - y[lag:]) * wi
                g[lag:] -= y[:n-lag] * wi
            elif -n < lag < 0:
                la = -lag
                g[la:] += (1.0 - y[:n-la]) * wi
                g[:n-la] -= y[la:] * wi
        # Note: this gradient is for the soft-max of (corr*dx), so already includes dx
        
        step = 1.0 / max(Lc, 1e-10)
        accepted = False
        for _ in range(10):
            xn = _proj(y - step * g, S)
            fx = sval(xn)
            dv = xn - y
            rhs = fy + np.dot(g, dv) + 0.5 * (1.0/step) * np.dot(dv, dv)
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-10)
        
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.9, 1.0)
            if it % 8 == 0:
                cc = _c5(h)
                if cc < best_c:
                    best_c = cc; best_h = h.copy()
