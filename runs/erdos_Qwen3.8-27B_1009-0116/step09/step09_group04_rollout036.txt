            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, hn):
    return x.sum() - hn

def _sj(x, hn):
    return np.ones(len(x))

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _do_lbfgs(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': maxiter * 3})
        return _repair(r.x, n)
    except:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _bobj(x, h_full, start, end, alpha, ml):
    n = len(h_full)
    hh = h_full.copy()
    hh[start:end] = np.clip(x, 0, 1)
    c = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha + 1e10 * (hh.sum() - n * 0.5) ** 2

def _bgrad(x, h_full, start, end, alpha, ml):
    n = len(h_full)
    hh = h_full.copy()
    hh[start:end] = np.clip(x, 0, 1)
    return _sgrad(hh, alpha, ml)[start:end]

def _block_opt(h, start, end, alpha, ml, maxiter):
    n = len(h)
    sz = end - start
    if sz < 3:
        return h.copy()
    x0 = h[start:end].copy()
    bnds = [(0.0, 1.0)] * sz
    try:
        r = minimize(_bobj, x0, jac=_bgrad, args=(h, start, end, alpha, ml),
                    method='SLSQP', bounds=bnds,
                    options={'maxiter': maxiter, 'ftol': 1e-14})
        xn = np.clip(r.x, 0, 1)
        hh = h.copy()
        hh[start:end] = xn
        if abs(hh.sum() - n * 0.5) > 0.1:
            return h.copy()
        return hh
    except:
        return h.copy()

def _bcd(h, n, block_size, n_sweeps, rng, alpha, ml, t0, budget_s):
    c5 = _eval(h)
    for sweep in range(n_sweeps):
        if time.time() - t0 > budget_s - 60:
            break
        starts = list(range(0, n, block_size))
        rng.shuffle(starts)
        for start in starts:
            if time.time() - t0 > budget_s - 60:
                break
            end = min(start + block_size, n)
            if end - start < 3:
                continue
            hh = _block_opt(h, start, end, alpha, ml, 60)
            cn = _eval(hh)
            if cn < c5:
                h = hh.copy()
                c5 = cn
    return h, c5

def _grad_descent(work_h, n, rng, alpha, ml, steps, t0, tlimit):
    work_c5 = _eval(work_h)
    s0 = 0.004
    for it in range(steps):
        if time.time() - t0 > tlimit:
            break
        g = _sgrad(work_h, alpha, ml)
        step = s0 * (0.6 + 0.8 * rng.rand())
        hn = _proj(work_h - step * g, n * 0.5)
        cn = _eval(hn)
        if cn < work_c5:
            work_h, work_c5 = hn, cn
            s0 = min(s0 * 1.05, 0.04)
        else:
            s0 *= 0.7
    return work_h, work_c5

def _multi_polish(best_h, best_c5, n, rng, t0, tlimit):
    h2 = best_h.copy()
    c2 = best_c5
    step_scale = 3e-7
    while time.time() - t0 < tlimit:
        k = 2 + rng.randint(3)
        idx = rng.choice(n, size=k, replace=False)
        d = step_scale * rng.randn(k)
        d -= d.mean()
        hn = h2.copy()
        ok = True
        for ii in range(k):
            v = hn[idx[ii]] + d[ii]
            if v < 0 or v > 1:
                ok = False
                break
            hn[idx[ii]] = v
        if not ok:
            continue
        cn = _eval(hn)
        if cn < c2:
            c2, h2 = cn, hn
            step_scale = min(step_scale * 1.02, 5e-6)
            if cn < best_c5:
                best_c5, best_h = cn, hn.copy()
        else:
            for ii in range(k):
                h2[idx[ii]] -= d[ii]
            step_scale = max(step_scale * 0.95, 1e-8)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP + L-BFGS-B with geometric alpha schedule
    for a in [3000, 15000, 80000, 400000, 2000000]:
        if time.time() - t0 > 100:
            break
        ml = min(80, max(18, int(12 * np.log2(a / 3000.0) + 18)))
        hn = _do_slsqp(best_h, a, ml, 800)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn.copy()
        if time.time() - t0 > 160:
            break
        hn = _do_lbfgs(best_h, a, ml, 1200)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn.copy()

    # Phase 2: Projected gradient descent (multiple alpha configs)
    if time.time() - t0 < budget_s - 280:
        for a, ml in [(30000, 40), (300000, 60), (3000000, 80)]:
            if time.time() - t0 > budget_s - 230:
                break
            work_h = best_h.copy()
            work_h, work_c5 = _grad_descent(work_h, n, rng, a, ml, 400, t0, budget_s - 230)
            if work_c5 < best_c5:
                best_c5, best_h = work_c5, work_h.copy()

    # Phase 3: Block coordinate descent (key differentiator)
    if time.time() - t0 < budget_s - 200:
        for bs, a, ml, ns in [(24, 8000, 30, 5), (16, 40000, 45, 4), (12, 200000, 55, 3)]:
            if time.time() - t0 > budget_s - 140:
                break
            best_h, best_c5 = _bcd(best_h, n, bs, ns, rng, a, ml, t0, budget_s)

    # Phase 4: High-alpha SLSQP for convergence
    if time.time() - t0 < budget_s - 100:
        for a, ml in [(5000000, 50), (50000000, 70), (500000000, 90)]:
            if time.time() - t0 > budget_s - 60:
                break
            hn = _do_slsqp(best_h, a, ml, 500)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn.copy()

    # Phase 5: Second BCD pass with smaller blocks
    if time.time() - t0 < budget_s - 100:
        best_h, best_c5 = _bcd(best_h, n, 10, 4, rng, 2000000, 60, t0, budget_s)

    # Phase 6: Multi-point adaptive polish
    if time.time() - t0 < budget_s - 4:
        best_h, best_c5 = _multi_polish(best_h, best_c5, n, rng, t0, budget_s - 4)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
