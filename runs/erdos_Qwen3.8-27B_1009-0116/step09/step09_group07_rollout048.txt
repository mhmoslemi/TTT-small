import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break

    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smo_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

    return x.sum() - target

    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):

                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,

    except Exception:

                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Pure coordinate descent (35% budget)
    p1_end = t0 + budget_s * 0.35
    while time.time() < p1_end:
        h = _cd_pass(h, rng)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() > p1_end + 5:
            break

    # Phase 2: Iterated perturbation + CD restarts (35% budget)
    p2_end = t0 + budget_s * 0.70
    amp = 0.025
    min_amp = 0.003
    while time.time() < p2_end - 5:
        h_try = best_h + amp * rng.randn(n)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = _repair(h_try, n)
        sub_t0 = time.time()
        max_sub = min(12.0, p2_end - 15 - time.time())
        prev_c5 = _eval(h_try)
        stagnant = 0
        while time.time() - sub_t0 < max_sub:
            h_try = _cd_pass(h_try, rng)
            c5s = _eval(h_try)
            if c5s < prev_c5 - 1e-8:
                prev_c5 = c5s
                stagnant = 0
            else:
                stagnant += 1
            if stagnant > 4:
                break
        h_try = _repair(h_try, n)
        c5_final = _eval(h_try)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_try.copy()
            amp = min(amp * 1.15, 0.06)
        else:
            amp = max(amp * 0.82, min_amp)

    # Phase 3: SLSQP refinement (20% budget)
    if time.time() < t0 + budget_s - 180:
        slsqp_configs = [(5000, 25), (50000, 40), (500000, 60), (5000000, 80)]
        for alpha, ml in slsqp_configs:
            if time.time() > t0 + budget_s - 130:
                break
            h_new = _slsqp(best_h.copy(), alpha, ml, 350)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Final CD + local search (10% budget)
    if time.time() < t0 + budget_s - 5:
        h = best_h.copy()
        while time.time() < t0 + budget_s - 3:
            h = _cd_pass(h, rng)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Fine local search with decaying steps
    if time.time() < t0 + budget_s - 1:
        remaining = max(1.0, t0 + budget_s - 1 - time.time())
        ntr = int(remaining / 0.003)
        ntr = min(ntr, 300000)
        steps = [1e-4, 5e-5, 1e-5, 5e-6, 1e-6, 5e-7, 1e-7]
        best_h, best_c5 = _local_search(best_h, n, rng, ntr, steps)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
