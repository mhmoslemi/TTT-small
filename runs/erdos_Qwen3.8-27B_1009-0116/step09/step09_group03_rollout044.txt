import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_eq(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _pg_step(h, g, lr, target):
    h_new = _proj(h - lr * g, target)
    return h_new

def _pure_subgrad(h, n_iter, top_k, lr0, momentum):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    lr = lr0
    mom = momentum
    for it in range(n_iter):
        c_vec = _eval_vec(h)
        m = c_vec.max()
        sorted_idx = np.argsort(-c_vec)[:top_k]
        thresh = c_vec[sorted_idx[-1]]
        active = c_vec >= thresh * 0.999
        g = np.zeros(n)
        wts = c_vec[active]
        wts = wts - wts.min() + 1e-12
        wts /= wts.sum()
        for ii, k in enumerate(np.where(active)[0]):
            L = k - (n - 1)
            wk = wts[ii]
            if L >= 0:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - h[M:])
                g[M:] -= wk * h[:n - M]
        g = g * (2.0 / n)
        v = mom * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if it % 50 == 0:
            lr = min(lr * 1.02, lr0 * 50.0)
            mom = min(mom + 0.001, 0.95)
        else:
            step = lr
            for _ in range(20):
                h_try = _proj(h - step * g, n * 0.5)
                c_try = _eval(h_try)
                if c_try < _eval(h):
                    h = h_try
                    break
                step *= 0.7
            else:
                h = h_new
    return best_h, best_c

def _fourier_restart(h, n_pert, amp, rng):
    n = len(h)
    x = np.arange(n)
    for _ in range(n_pert):
        freq = rng.randint(1, n // 3)
        ph = rng.rand() * 2.0 * np.pi
        h = h + amp * rng.rand() * np.sin(2.0 * np.pi * freq * x / n + ph)
    return _repair(h, n)

def _pair_polish(h, n_iter, rng, window):
    n = len(h)
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, window)) % n
        d = 1e-6 * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return best_h, best_c

def _grad_pair_search(h, n_iter, rng, alpha, ml):
    n = len(h)
    c5_cur = _eval(h)
    best_c = c5_cur
    best_h = h.copy()
    g = _smooth_grad(h, alpha, ml)
    ag = np.abs(g)
    for _ in range(n_iter):
        top3 = np.argpartition(-ag, 3)[:3]
        i = top3[rng.randint(3)]
        j = top3[(rng.randint(3) + 1) % 3]
        d = 3e-5 * rng.randn()
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        if h_new.min() < 0 or h_new.max() > 1:
            d *= -0.5
            h_new[i] += d
            h_new[j] -= d
            if h_new.min() < 0 or h_new.max() > 1:
                continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5_cur:
            h = h_new
            c5_cur = c5n
            g = _smooth_grad(h, alpha, ml)
            ag = np.abs(g)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c
