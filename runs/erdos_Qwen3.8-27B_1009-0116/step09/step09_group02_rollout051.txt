        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, top_m):
    c = _corr_dx(x)
    top_idx = np.argsort(c)[-top_m:]
    m = c[top_idx].max()
    e = alpha * (c[top_idx] - m)
    e -= e.max()
    return m + np.log(np.exp(e).sum()) / alpha

def _smooth_grad(x, alpha, top_m):
    n = len(x)
    dx = 2.0 / n
    c = _corr_dx(x)
    top_idx = np.argsort(c)[-top_m:]
    m = c[top_idx].max()
    e = alpha * (c[top_idx] - m)
    e -= e.max()
    w = np.exp(e)
    w /= w.sum()
    g = np.zeros(n)
    om2 = 1.0 - 2.0 * x
    for ii in range(len(top_idx)):
        wi = w[ii] * dx
        if wi < 1e-15:
            continue
        j = int(top_idx[ii])
        p_lo = max(0, j - n + 1)
        p_hi = min(n - 1, j)
        g[p_lo:p_hi+1] += wi * om2[j-p_hi : j-p_lo+1][::-1]
    return g

def _lbfgs_opt(h, alpha, top_m, penalty, maxiter, ftol):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        f = _smooth_obj(x, alpha, top_m)
        s = x.sum() - target
        return f + penalty * s * s
    
    def grad(x):
        g = _smooth_grad(x, alpha, top_m)
        s = x.sum() - target
        return g + 2.0 * penalty * s
    
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-14, 'maxls': 50, 'maxfun': maxiter * 10})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_box_sum(h_raw, target):
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h_raw - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_raw - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, top_m, steps, lr0, mu):
    n = len(h)
    target = n / 2.0
    v = np.zeros(n)
    lr = lr0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _smooth_grad(h, alpha, top_m)
        g -= g.mean()
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        taken = False
        for _ in range(15):
            h_new = _proj_box_sum(h - lr * v, target)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                lr = min(lr * 1.05, lr0 * 50)
                taken = True
                break
            lr *= 0.35
        if not taken:
            lr = min(max(lr * 1.3, lr0), lr0 * 50)
    return _repair(h, n)

def _sim_anneal(h, n_iter, rng, t0, t1):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_h = h.copy()
    cur_c = _eval(cur_h)
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        T = t0 * (t1 / t0) ** frac
        k = 2 + rng.randint(0, 4)
        idx = rng.choice(n, k, replace=False)
        delta = (rng.rand(k) * 2 - 1) * 0.015
        delta -= delta.mean()
        h_new = cur_h.copy()
        h_new[idx] += delta
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        dc = c_new - cur_c
        if dc < 0 or (T > 1e-18 and rng.rand() < np.exp(-dc / T)):
            cur_h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        d = (1e-9 + 3e-7 * rng.rand()) * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            oi, oj = h[i], h[j]
            h[i], h[j] = hi, hj
            c = _eval(h)
            if c < cur_c - 1e-16:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i], h[j] = oi, oj
    return best_h, best_c

def _multi_point_polish(h, n_iter, rng, pts=3):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, n, size=pts)) % n)
        d = (1e-8 + 3e-7 * rng.rand()) * rng.randn(pts)
        d -= d.mean()
        if np.abs(d).max() > 0.01:
            d *= 0.01 / np.abs(d).max()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            h_new = _repair(h_new, n)
        c = _eval(h_new)
        if c < cur_c - 1e-16:
            cur_c = c
            h = h_new
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
    return best_h, best_c
