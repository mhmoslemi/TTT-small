def _soft_max(corr, alpha):
    """Smooth max approximation"""
    m = corr.max()
    return m + np.log(np.exp(alpha * (corr - m)).sum()) / alpha

def _soft_max_grad(corr, alpha):
    """Gradient of smooth max w.r.t. the array (for backprop)"""
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    return w  # These are the weights for the subgradient

import numpy as np
import time

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-11:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    for _ in range(60):
        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-13:
            break
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _grad_k(h, k):
    n = len(h)
    if k < 0:
        k = -k
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += (-h[k:])
    return g

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = max(float(np.max(P)), float(np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _active_grad(h, P, Nv, topK=8):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 20.0 * dx)
    cands = []
    if P[0] >= mx - tol:
        cands.append((P[0], 0, 0))
    if n > 1:
        for kk in range(1, n):
            if P[kk] >= mx - tol:
                cands.append((P[kk], kk, 0))
            if Nv[kk] >= mx - tol:
                cands.append((Nv[kk], kk, 1))
    if not cands:
        return np.zeros(n)
    cands.sort(key=lambda x: -x[0])
    cands = cands[:topK]
    wts = np.array([c[0] for c in cands])
    wts = np.exp(5.0 * (wts - wts.max()))
    wts /= wts.sum()
    g = np.zeros(n)
    for wt, (val, kk, typ) in zip(wts, cands):
        g += wt * _grad_k(h, kk if typ == 0 else -kk)
    return g

def _resample(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _gen_harmonic(m, freq, phase, amp, rng):

    return _proj(np.clip(h, 0.0, 1.0), m * 0.5)
