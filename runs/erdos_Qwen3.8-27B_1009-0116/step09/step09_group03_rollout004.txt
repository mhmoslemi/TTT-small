import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    return float(c.max()), c

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n*0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                      args=(alpha, ml), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        return _repair(res.x, n)
    except:
        return h

def _local_search(h, rng, budget_fn, max_time_frac):
    """Aggressive pairwise local search with simulated annealing."""
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    
    T0 = 0.002  # initial temperature
    T_end = 0.00001
    t_start = time.time()
    t_end = t_start + max_time_frac
    
    # Pre-compute gradient for pair selection
    g = _smooth_grad(h, 100000, 50)
    # Sort by |gradient| for weighted selection
    abs_g = np.abs(g)
    
    n_eval = 0
    while time.time() < t_end:
        # Adaptive step size
        step = T0 * (T_end / T0) ** ((time.time() - t_start) / max(1.0, t_end - t_start))
        step = max(step, 1e-6)
        
        # Weighted pair selection based on gradient
        # Pick i from top gradient indices, j from full range
        top_k = min(100, n // 5)
        top_idx = np.argpartition(-abs_g, top_k)[:top_k]
        i = top_idx[rng.randint(top_k)]
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        
        d = step * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        
        h[i], h[j] = hi, hj
        c5n = _eval(h)
        n_eval += 1

        if c5n < c5c:

            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:

            delta = c5n - c5c
            T = max(T0 * 0.999 ** (n_eval // 1000), T_end)
            if delta < 0.01 and rng.rand() < np.exp(-delta / max(T, 1e-10)):

        if accept:

            # Update gradient less frequently
            if n_eval % 5000 == 0:
                g = _smooth_grad(h, 100000, 50)
                abs_g = np.abs(g)
        else:

    one_minus_h = 1.0 - h
    for k, wk in enumerate(w):

        g[:n-L] += wk * one_minus_h[L:]
        g[L:] -= wk * h[:n-L]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-11:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, len(c)-1))[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n*0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                      args=(alpha, ml), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 25})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0, 1)

def _local_search(h, rng, t_end):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    # Gradient for pair selection
    g = _smooth_grad(h, 200000, 50)
    abs_g = np.abs(g)
    top_k = min(80, n // 4)
    top_idx = np.argpartition(-abs_g, top_k)[:top_k]
    n_eval = 0
    step = 0.008
    t0 = time.time()
    total_time = max(1.0, t_end - t0)
    while time.time() < t_end:
        frac = (time.time() - t0) / total_time
        T = 0.003 * np.exp(-3.0 * frac) + 0.00001
        step = 0.008 * np.exp(-1.5 * frac) + 0.0005
        # Weighted pair selection
        if n_eval % 8 == 0:
            i = top_idx[rng.randint(top_k)]
        else:
            i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        d = step * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i], h[j] = hi, hj
        c5n = _eval(h)
        n_eval += 1
        accept = False
        if c5n < c5c:
            accept = True
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            delta = c5n - c5c
            if delta < T * 5 and rng.rand() < np.exp(-delta / T):
                accept = True
        if accept:
            c5c = c5n
            if n_eval % 20000 == 0:
                g = _smooth_grad(h, 200000, 50)
                abs_g = np.abs(g)
                top_idx = np.argpartition(-abs_g, top_k)[:top_k]
        else:
            h[i], h[j] = hi - d, hj + d
    return best_h, best_c

def _fine_polish(h, rng, t_end):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    while time.time() < t_end:
        i = rng.randint(n)
        offset = 1 + rng.randint(0, 12)

        hi, hj = h[i] + d, h[j] - d
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:

        h[i], h[j] = hi, hj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = hi - d, hj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    out = h[:m*f].reshape(m, f).mean(axis=1)
    return _repair(out, m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n * f)
