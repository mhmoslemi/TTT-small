import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

    dx = 2.0 / n

        for k_idx in worst_k_idx[:5]:

            else:

                contrib = h[M:] * (1 - h[:n-M])

            i_max = np.argmax(contrib)
            if L >= 0:
                ia, ib = i_max, i_max + L

            # Try to reduce: increase h[ia] if h[ib] is small, or decrease h[ia]

def _alt_linear_step(h, n, top_m=20):
    """One pass of alternating linearization"""
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    # Find active shifts (those with c[k] > m * 0.95)
    active = np.where(c > m * 0.99)[0]
    
    # For each j, compute the gradient of max(c) w.r.t. h[j]
    # This is the weighted sum over active k of ∂c[k]/∂h[j]
    g = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)  # actual shift
        w = c[k_idx]  # weight (proportional to how "active" this constraint is)
        if L >= 0:
            # c[k] = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # ∂c[k]/∂h[j] = (1-h[j+L]) for j < n-L (forward)
            #              = -h[j-L] for j >= L (backward)
            for j in range(n - L):
                g[j] += w * (1 - h[j + L])
            for j in range(L, n):
                g[j] -= w * h[j - L]

    g /= np.abs(g).max() + 1e-30
    
    # Take a step
    lr = 0.01
    h_new = _repair(h - lr * g, n)

    block = h[start:end].copy()
    m = end - start
    rest = h.copy()
    rest[start:end] = 0  # temporarily zero out

        rest[start:end] = x
        return _eval(rest)

        c = np.correlate(rest, 1-rest, mode='full') * (2.0/n)

            w = c[k_idx]
            if L >= 0:
                g_full[:n-L] += w * (1 - rest[L:])
                g_full[L:] -= w * rest[:n-L]
            else:

                g_full[M:] += w * (1 - rest[:n-M])
                g_full[:n-M] -= w * rest[M:]

    res = minimize(obj, block, jac=grad, method='L-BFGS-B', bounds=bnds,

def _shift_targeted_local(h, n, n_iter, rng):
    """Local search targeting the worst correlation shift"""
    best_h = h.copy()
    best_c = _eval(h)
    cur = h.copy()
    cur_c = _eval(cur)
    
    for it in range(n_iter):
        # Compute worst shift
        c_arr = np.correlate(cur, 1.0 - cur, mode='full') * (2.0/n)
        worst_idx = np.argmax(c_arr)
        L = worst_idx - (n - 1)  # actual shift
        
        if L >= 0 and n - L > 1:
            # contrib[i] = cur[i] * (1 - cur[i+L]) for i in 0..n-1-L
            contrib = cur[:n-L] * (1.0 - cur[L:])
            # Pick a random top contributor
            top_n = min(20, n - L)
            top_i = np.argpartition(-contrib, top_n)[:top_n]
            i = top_i[rng.randint(top_n)]
            j = i + L
        elif L < 0 and n + L > 1:
            M = -L
            # contrib[i] = cur[i+M] * (1 - cur[i]) for i in 0..n-1-M
            contrib = cur[M:] * (1.0 - cur[:n-M])
            top_n = min(20, n - M)
            top_i = np.argpartition(-contrib, top_n)[:top_n]
            i = top_i[rng.randint(top_n)] + M
            j = top_i[rng.randint(top_n)]
        else:
            continue
        
        # Try to reduce cur[i]*(1-cur[j])
        # Increase cur[j] and/or decrease cur[i]
        step = rng.choice([1e-4, 5e-5, 1e-5, 5e-6, 1e-6])
        cur_new = cur.copy()
        cur_new[i] -= step
        cur_new[j] += step
        if cur_new.min() < 0 or cur_new.max() > 1:
            # Try smaller step
            step *= 0.1
            cur_new = cur.copy()
            cur_new[i] -= step
            cur_new[j] += step
            if cur_new.min() < 0 or cur_new.max() > 1:
                continue
        cur_new = _repair(cur_new, n)
        c_new = _eval(cur_new)
        
        if c_new < cur_c:
            cur = cur_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = cur_new.copy()
        elif rng.random() < 0.005:
            # Occasional acceptance of worse (annealing)
            cur = cur_new
            cur_c = c_new
    
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-13:
            break
    d = target - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(d), (1.0 - h[idx]) if d > 0 else h[idx])
            h[idx] += np.sign(d) * a
            d -= np.sign(d) * a
            if abs(d) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, al, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argpartition(-c, min(ml, 2 * n))[:ml]
    s = np.clip(al * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / al

def _sm_grad(x, al, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argpartition(-c, min(ml, 2 * n))[:ml]

    s -= s.max()

    for ii, k in enumerate(idx):

        else:

            g[M:] -= wk * x[:n - M]
    return g
