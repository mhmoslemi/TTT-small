import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = min(-d, h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn(h):
    """Compute per-lag penalty values."""
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:

        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

    P, Nv, mx = _pn(h)
    n = len(h)

        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _gW(h, wP, wN):
    """Weighted gradient."""
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sg(h, a):
    """Smooth gradient (softmax-weighted)."""
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)

def _agrad(h):
    """Active-set gradient (equal weight on active lags)."""
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    tol = max(1e-9, 2.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    return _gW(h, wP, wN)

    wN /= cnt
    return _gW(h, wP, wN)

def _fista(h0, a, tl, L=8.0):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    fy = _sval(h, a)
    fh = fy
    bh = h.copy()
    bc = _c5(h)
    stale = 0
    Lc = L
    for it in range(200000):
        if time.time() - t0 > tl:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            if y.min() < -1e-5 or y.max() > 1 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _sval(y, a)
        else:
            y = h
            fy = fh
        g = _sg(y, a)
        step = 1.0 / max(Lc, 1e-12)
        ok = False
        for _ in range(10):
            xn = _proj(y - step * g, S)
            fx = _sval(xn, a)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                ok = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp = h
            h = xn
            fh = fx
            stale = 0
            if it % 10 == 0:
                c = _c5(h)
                if c < bc:
                    bc = c
                    bh = h.copy()
        else:
            stale += 1
            hp, y, tv = h, h, 1.0
            fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 30:
                break
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return h, bh, bc

def _loc_sa(h0, rng, deadline, s0=5e-5, T0=0.05):
    h = h0.copy()
    n = len(h)
    c = _c5(h)
    bc = c
    bh = h.copy()
    P, Nv, _ = _pn(h)
    g = _agrad(h)
    order = np.argsort(g)
    q = max(8, n // 5)
    lo = order[:q]
    hi = order[-q:]
    recomp = 200
    it = 0
    t_start = time.time()
    while time.time() < deadline:
        if it >= recomp:
            c = _c5(h)
            if c < bc:
                bc = c
                bh = h.copy()
            P, Nv, _ = _pn(h)
            g = _agrad(h)
            order = np.argsort(g)
            lo = order[:q]
            hi = order[-q:]
            recomp = it + 200
        fr = min(1.0, (time.time() - t_start) / max(1e-9, deadline - t_start))
        base = max(1e-10, s0 * (1e-8) ** fr)
        temp = T0 * (1.0 - fr) ** 2
        r = rng.random()
        if r < 0.70:
            i = int(lo[rng.randint(len(lo))])
            j = int(hi[rng.randint(len(hi))])
        elif r < 0.85:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        else:
            mid = order[q:2*q] if len(order) > 2*q else order[:q]
            i = int(mid[rng.randint(len(mid))])
            j = int(rng.randint(n))
        if i != j:
            sgn = 1 if g[i] <= g[j] else -1
            avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                mult = 0.3 + 1.7 * rng.random()
                if rng.random() < 0.04:
                    mult *= 6.0
                d = sgn * min(avail, base * mult)
                h[i] += d
                h[j] -= d
                cn = _c5(h)
                if cn < c - 1e-14:
                    c = cn
                    if c < bc:
                        bc = c
                        bh = h.copy()
                elif cn - c < temp and temp > 0:
                    c = cn
                else:
                    h[i] -= d
                    h[j] += d
        it += 1
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def _bin_loc(h0, rng, deadline):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    s = int(h.sum())
    target = n // 2
    if s > target:
        idx = np.where(h == 1.0)[0]
        order = np.argsort(h0[idx])
        for k in range(s - target):
            h[idx[order[k]]] = 0.0
    elif s < target:
        idx = np.where(h == 0.0)[0]
        order = np.argsort(-h0[idx])
        for k in range(target - s):
            h[idx[order[k]]] = 1.0
    c = _c5(h)
    bc = c
    bh = h.copy()
    ones = [i for i in range(n) if h[i] > 0.5]
    zeros = [i for i in range(n) if h[i] < 0.5]
    o2i = {v: k for k, v in enumerate(ones)}
    z2i = {v: k for k, v in enumerate(zeros)}
    stagn = 0
    while time.time() < deadline:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        cn = _c5(h)
        if cn <= bc + 1e-14:
            if cn < bc:
                bc = cn
                bh = h.copy()
                stagn = 0
            else:
                stagn += 1
            ki = o2i[i]
            kj = z2i[j]
            o2i[i] = kj
            z2i[j] = ki
            ones[ki], zeros[kj] = j, i
        else:
            h[i] = 1.0
            h[j] = 0.0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)

    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n0 * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    # Phase 1: FISTA moderate alpha
    end1 = t0 + 60.0
    hc = best.copy()
    for a in [15, 60, 250]:
        rem = end1 - time.time()
        if rem < 3:
            break
        hf, bh, bc = _fista(hc, a, min(rem, 20.0))
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    # Phase 2: FISTA high alpha (push to binary)
    end2 = t0 + 150.0
    for a in [1500, 8000, 40000]:
        rem = end2 - time.time()
        if rem < 4:
            break
        hf, bh, bc = _fista(hc, a, min(rem, 28.0))
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    # Phase 3: Binary local search
    end3 = t0 + 220.0
    bb, cb = _bin_loc(hc, rng, end3)
    if cb < best_c:
        best, best_c = bb, cb
    bb2, cb2 = _bin_loc(best, rng, min(end3 + 20, t0 + 240.0))
    if cb2 < best_c:
        best, best_c = bb2, cb2

    # Phase 4: SA continuous local search (main optimization)
    end4 = t0 + 600.0
    best, best_c = _loc_sa(best, rng, end4, s0=5e-5, T0=0.04)

    # Phase 5: Finer SA
    end5 = t0 + 820.0
    best, best_c = _loc_sa(best, rng, end5, s0=1.5e-5, T0=0.012)

    # Phase 6: Very fine polish
    end6 = t0 + budget_s - 15.0
    best, best_c = _loc_sa(best, rng, end6, s0=4e-6, T0=0.0)

    # Final binary attempt
    if time.time() < t0 + budget_s - 5:
        bb3, cb3 = _bin_loc(best, rng, t0 + budget_s - 5)
        if cb3 < best_c:
            best, best_c = bb3, cb3

    # Final repair and symmetry
    best = _repair(best, len(best))
    best_c = _c5(best)
    n = len(best)
    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    return best, best_c, len(best)
