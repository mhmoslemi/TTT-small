    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2*n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _soft_grad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    stale = 0; Lc = 50.0
    for it in range(1000000):
        if time.time() - t0 > max_t:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        else:
            y = h; fy = fh
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(14):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.8, 1.0)
            if it % 10 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _subgrad_step(h, step_size):
    """Take a subgradient step on the exact max of correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    m = k_star - (n - 1)  # effective lag
    # Compute gradient of c_k with respect to h
    g = np.zeros(n)
    if m >= 0:
        # c = sum_{i=0}^{n-1-m} h[i](1-h[i+m])
        # dc/dh[j] = (1-h[j+m]) for j in [0, n-1-m] + (-h[j-m]) for j in [m, n-1]
        valid1 = np.arange(n - m)
        g[valid1] += (1.0 - h[m:2*n-m])[0:len(valid1)]  # (1-h[j+m]) for j=0..n-1-m
        # Wait, h[j+m] for j in [0, n-1-m] means h[m .. n-1]
        g[:n-m] += (1.0 - h[m:])
        valid2 = np.arange(m, n)
        g[m:] -= h[:n-m]  # -h[j-m] for j=m..n-1 means -h[0..n-1-m]
    else:
        p = -m
        # c = sum_{i=p}^{n-1} h[i](1-h[i-p])
        # dc/dh[j] = (1-h[j-p]) for j in [p, n-1] + (-h[j+p]) for j in [0, n-1-p]
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return h - step_size * g, k_star

def _subgrad_optimize(h0, max_t):
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    best_h = h.copy()
    best_c = _c5(h)
    t0 = time.time()
    step_size = 0.01
    decay = 0.9997
    while time.time() - t0 < max_t:
        h_new, _ = _subgrad_step(h, step_size)
        h_new = _proj(h_new, S)
        c = _c5(h_new)
        if c < best_c:
            best_c = c; best_h = h_new.copy()
            h = h_new
        else:
            step_size *= decay
            if step_size < 1e-10:
                break
    return best_h, best_c

def _targeted_polish(h0, rng, max_t):
    """Very targeted local polish using active constraints"""
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    step = 0.002
    while time.time() - t0 < max_t:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        m = k_star - (n - 1)
        # Find positions that contribute most to the active constraint
        if m >= 0:
            # c = sum_{i=0}^{n-1-m} h[i](1-h[i+m])
            contrib = h[:n-m] * (1.0 - h[m:])
            # To decrease c: increase h[m:] or decrease h[:n-m] where product is high
            top_idx = np.argsort(-contrib)[:20]
            low_idx = np.argsort(contrib)[-20:]
        else:
            p = -m
            contrib = h[p:] * (1.0 - h[:n-p])
            top_idx = np.argsort(-contrib)[:20] + p
            low_idx = np.argsort(contrib)[-20:]
        # Try moves
        for _ in range(50):
            if time.time() - t0 < max_t:
                if rng.random() < 0.5:
                    i = int(top_idx[rng.randint(len(top_idx))])
                    j = int(low_idx[rng.randint(len(low_idx))])
                    if m >= 0:
                        d = rng.random() * step  # increase h[i] (in h[m:]) or decrease
                    else:
                        d = rng.random() * step
                else:
                    i = rng.randint(n)
                    j = rng.randint(n)
                    d = (rng.random() - 0.5) * step
                if i == j:
                    continue
                # Move: h[i] += d, h[j] -= d
                if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                    continue
                h[i] += d; h[j] -= d
                cn = _c5(h)
                if cn < c:
                    c = cn
                    if c < best_c:
                        best_c = c; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
        step *= 0.999
        if step < 1e-8:
            break
    return best_h, best_c

def _sa_polish(h0, rng, max_t):
    """Simulated annealing with very small moves"""
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * (2.0 / n)
    best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, max_t - (time.time() - t_start))
    while time.time() - t_start < max_t:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 0.001 * (0.01) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        if rng.random() < 0.7 and abs(k_act) < n:
            k = abs(k_act)
            if k_act > 0:
                valid = (h[:n-k] + h[k:] > 0.01) & (h[:n-k] < 0.99) & (h[k:] < 0.99)
            else:
                valid = (h[k:] + h[:n-k] > 0.01) & (h[k:] < 0.99) & (h[:n-k] < 0.99)
            if valid.sum() > 4:
                pos = np.nonzero(valid)[0]
                i = int(pos[rng.randint(len(pos))])
                j = i + k if k_act > 0 else i - k
                if 0 <= j < n:
                    d = (rng.random() - 0.5) * 0.01
                    h[i] += d; h[j] -= d
                    c2 = np.correlate(h, 1.0 - h, mode='full')
                    mx2 = float(c2.max())
                    delta = mx2 - mx
                    if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                        mx = mx2; corr = c2
                        cn = mx * (2.0 / n)
                        if cn < best_c:
                            best_c = cn; best_h = h.copy()
                    else:
                        h[i] -= d; h[j] += d
                    continue
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            d = (rng.random() - 0.5) * 0.008
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c2 = np.correlate(h, 1.0 - h, mode='full')
                mx2 = float(c2.max())
                delta = mx2 - mx
                if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                    mx = mx2; corr = c2
                    cn = mx * (2.0 / n)
                    if cn < best_c:
                        best_c = cn; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)
