def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        if abs(s - S) < 1e-14: break
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0-h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float)
    x += (S - x.sum()) / x.size

        if c == 0: break

    return x
