import numpy as np

import time

def _eval_c(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval_max(h):
    return float(_eval_c(h).max())

def _pnorm(h, p):
    c = _eval_c(h)
    c = np.maximum(c, 1e-30)
    return (np.sum(c**p))**(1.0/p)

def _pnorm_grad(h, p, lam, mu):
    n = len(h)
    c = _eval_c(h)
    c = np.maximum(c, 1e-30)
    sp = np.sum(c**p)
    w = (sp**(1.0/p - 1.0)) * c**(p-1)
    
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-20:
            continue
        lag = k - (n-1)
        if lag >= 0:
            if lag < n:
                grad[:n-lag] += wk * omh[lag:]
                grad[lag:] -= wk * h[:n-lag]
        else:
            m = -lag
            if m < n:
                grad[m:] += wk * omh[:n-m]
                grad[:n-m] -= wk * h[m:]
    
    sum_err = h.sum() - n/2.0
    grad += (lam + mu * sum_err)
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def _lgbfsopt(h, p, n, maxiter=500):

    lam = 0.0
    mu = 1.0
    for al_iter in range(8):

def _eval_c(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval_max(h):
    return float(_eval_c(h).max())

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-13:
                break
            st = min(abs(diff), 1.0 - h[i]) if diff > 0 else min(-diff, h[i])
            h[i] += st if diff > 0 else -st
            diff -= st if diff > 0 else -st
    return h

def _sm_obj(x, alpha, lam, mu, n):
    c = _eval_c(x)
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500.0, 0.0)
    ws = np.exp(s).sum()
    se = x.sum() - n * 0.5
    return cm + np.log(ws) / alpha + lam * se + 0.5 * mu * se * se

    se = x.sum() - n * 0.5
