def _corr_raw(h):

def _repair(h, n):

    target = n * 0.5
    s = h.sum()

        order = np.argsort(-h if diff < 0 else 1.0 - h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            cap = (1.0 - h[idx]) if diff > 0 else h[idx]
            a = min(abs(diff), cap)
            h[idx] += a if diff > 0 else -a

def _proj(h, target):

        else:

        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
