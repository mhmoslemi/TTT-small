def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a

def _slsqp_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 2: SLSQP refinement
    if _el() < budget_s * 0.45:
        for alpha, ml in [(5000, 20), (50000, 30), (500000, 40)]:
            if _el() > budget_s * 0.45: break
            h2 = _do_slsqp(best_h, alpha, ml, 800)
            c2 = _eval(h2)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 3: Nesterov PG again (tighter)
    if _el() < budget_s * 0.55:
        h2, c2 = _nesterov_pg(best_h, n, 5000, 1e-5, 0.001)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 4: Multi-scale local search (main improvement)
    if _el() < budget_s * 0.80:
        for s0 in [5e-4, 1.5e-4, 5e-5, 1.5e-5]:
            if _el() > budget_s * 0.80: break
            nt = min(int((budget_s * 0.80 - _el()) * 400), 80000)
            if nt < 500: break
            h2, c2 = _local_search(best_h, n, nt, rng, s0, 8000, True)
            if c2 < best_c:
                best_c, best_h = c2, h2

    # Phase 5: SLSQP final touch
    if _el() < budget_s * 0.90:
        h2 = _do_slsqp(best_h, 2000000, 50, 600)
        c2 = _eval(h2)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 6: Ultra-fine 2-point and 3-point polish
    if _el() < budget_s - 5:
        h2 = best_h.copy()
        c2 = best_c
        step = 3e-7
        it_total = 0
        max_iter = int((budget_s - 5 - _el()) * 800)
        max_iter = min(max_iter, 500000)
        for it in range(max_iter):
            if _el() > budget_s - 3:
                break
            it_total += 1
            if it_total % 20000 == 0:
                step *= 0.85
            if it % 5 == 4:
                idx = rng.randint(0, n, 3)
                i, j, k = idx[0], idx[1], idx[2]
                d1 = step * rng.randn()
                d2 = step * rng.randn()
                ni = h2[i] + d1; nj = h2[j] - d1 * 0.5; nk = h2[k] - d2 * 0.5
                if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1 and 0 <= nk and nk <= 1:
                    h2[i], h2[j], h2[k] = ni, nj, nk
                    cn = _eval(h2)
                    if cn < c2:
                        c2 = cn
                        if cn < best_c:
                            best_c, best_h = cn, h2.copy()
                    else:
                        h2[i], h2[j], h2[k] = ni - d1, nj + d1 * 0.5, nk + d2 * 0.5
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 5)) % n
                d = step * rng.randn()
                ni = h2[i] + d; nj = h2[j] - d
                if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                    h2[i], h2[j] = ni, nj
                    cn = _eval(h2)
                    if cn < c2:
                        c2 = cn
                        if cn < best_c:
                            best_c, best_h = cn, h2.copy()
                    else:
                        h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
