import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fn(x, t):
    return x.sum() - t

def _sum_jac(x, t):
    return np.ones(len(x))

def _smobj(x, alpha, m_lags):
    c = _corr_vec(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smgrad(x, alpha, m_lags):
    n = len(x)
    c = _corr_vec(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp_solve(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smobj, h, jac=_smgrad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, m_lags, steps, step0):
    n = len(h)
    v = np.zeros(n)
    cur_c = _eval(h)
    step = step0
    for it in range(steps):
        g = _smgrad(h, alpha, m_lags)
        v = 0.9 * v + g
        direction = v / (np.linalg.norm(v) + 1e-12)
        taken = False
        for _t in range(16):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.18, step0 * 60.0)
                taken = True
                break
            step *= 0.35
        if not taken:
            step = min(step * 1.5, step0 * 60.0)
    return _repair(h, n)

def _coord_cycle(h, n_passes, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    ds = [1e-6, 5e-6, 1e-5, 2e-5, 5e-5]
    for p in range(n_passes):
        for i in range(n):
            for off in [1, 2, 3, 5, 8]:
                j = (i + off) % n
                for d in ds:
                    for sign in [1, -1]:
                        ni, nj = h[i] + sign * d, h[j] - sign * d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c5n = _eval(h)
                            if c5n < best_c:
                                best_c = c5n
                                best_h = h.copy()
                            else:
                                h[i], h[j] = ni - sign * d, nj + sign * d
                                if c5n > c5c * 1.005:
                                    break
                    c5c = best_c
    return best_h, best_c

def _random_fine(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        for off in range(1, 11):
            j = (i + off) % n
            j2 = (i - off) % n
            for jj in [j, j2]:
                d = rng.choice([-1, 1]) * rng.choice([1e-6, 3e-6, 1e-5, 3e-5])
                ni, nj = h[i] + d, h[jj] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[jj] = ni, nj
                    c5n = _eval(h)
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                        c5c = c5n
                    else:
                        h[i], h[jj] = ni - d, nj + d
                        if c5n > c5c * 1.003:
                            break
    return best_h, best_c
