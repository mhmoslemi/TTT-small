        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(ti):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _pgd(h, alpha, m_lags, n_iter, lr_init, lr_end, rng, explore_frac=0.3):
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    explore_until = int(n_iter * explore_frac)
    for it in range(n_iter):
        t_frac = it / max(1, n_iter - 1)
        lr = lr_init * (lr_end / lr_init) ** t_frac
        if it < explore_until:
            h = h + lr * 0.1 * rng.randn(n)
            h = np.clip(h, 0, 1)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-m_lags:]
        cm = c.max()
        s = alpha * (c[top_idx] - cm)
        ew = np.exp(s)
        w = ew / ew.sum()
        g = np.zeros(n)
        for ii in range(len(top_idx)):
            wk = w[ii]
            if wk < 1e-15:
                continue
            wk *= alpha * dx
            k = int(top_idx[ii])
            L = k - nm1
            if L >= 0:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - h[M:])
                g[M:] -= wk * h[:n - M]
        h = h - lr * g
        h = _repair(h, n)
    return h

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                best_h = h.copy()
                best_c5 = c5c
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def remaining():
        return t0 + budget_s - time.time()

    # Phase 1: Coordinate descent (primary optimizer)
    p1_end = t0 + budget_s * 0.28
    while time.time() < p1_end:
        h = _cd_pass(h, rng, ti=18)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Projected gradient with moderate alpha
    if remaining() > budget_s * 0.35:
        h2 = best_h.copy()
        h2 = _pgd(h2, 3000, 60, 350, 0.04, 0.0005, rng, 0.25)
        h2 = _repair(h2, n)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2

    # Phase 3: More coordinate descent from best
    p3_end = t0 + budget_s * 0.55
    if time.time() < p3_end:
        h3 = best_h.copy()
        while time.time() < p3_end:
            h3 = _cd_pass(h3, rng, ti=18)
            h3 = _repair(h3, n)
            c5 = _eval(h3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()

    # Phase 4: Projection gradient with high alpha
    if remaining() > budget_s * 0.25:
        h4 = best_h.copy()
        h4 = _pgd(h4, 15000, 100, 300, 0.015, 0.0001, rng, 0.15)
        h4 = _repair(h4, n)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h4

    # Phase 5: Perturbation restarts to escape local optima
    p5_end = t0 + budget_s * 0.78
    if time.time() < p5_end:
        for attempt in range(8):
            if time.time() > p5_end:
                break
            h5 = best_h.copy()
            mask = rng.rand(n) < 0.08
            h5[mask] += 0.04 * rng.randn(int(mask.sum()))
            h5 = _repair(h5, n)
            for _ in range(4):
                if time.time() > p5_end:
                    break
                h5 = _cd_pass(h5, rng, ti=15)
                h5 = _repair(h5, n)
            c5 = _eval(h5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h5.copy()

    # Phase 6: Final CD polish
    p6_end = t0 + budget_s * 0.88
    if time.time() < p6_end:
        h6 = best_h.copy()
        while time.time() < p6_end:
            h6 = _cd_pass(h6, rng, ti=22)
            h6 = _repair(h6, n)
            c5 = _eval(h6)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h6.copy()

    # Phase 7: High-alpha projected gradient final push
    if remaining() > 30:
        h7 = best_h.copy()
        h7 = _pgd(h7, 50000, 120, 200, 0.005, 0.00005, rng, 0.05)
        h7 = _repair(h7, n)
        c5 = _eval(h7)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h7

    # Phase 8: Final local search
    time_left = t0 + budget_s - 3 - time.time()
    if time_left > 5:
        ntr = int(time_left / 0.004)
        ntr = min(ntr, 150000)
        if ntr > 50:
            best_h, best_c5 = _local_search(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
