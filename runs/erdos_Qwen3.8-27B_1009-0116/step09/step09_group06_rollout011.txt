def _systematic_sweep(h, rng, n_passes=3):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for p in range(n_passes):
        c5c = best_c
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, 10)) % n
            for d in [5e-8, -5e-8, 2e-8, -2e-8, 1e-8, -1e-8]:
                ni, nj = best_h[i] + d, best_h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    best_h[i], best_h[j] = ni, nj
                    c5n = _eval(best_h)
                    if c5n < c5c - 1e-18:
                        c5c = c5n
                        if c5c < best_c:
                            best_c = c5c
                    else:
                        best_h[i], best_h[j] = ni - d, nj + d
                        break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Quick SLSQP warm-up
    for alpha, ml, mi in [(20000, 40, 600), (200000, 60, 500), (2000000, 80, 400)]:
        if _el() > 120:
            break
        h_new = _slsqp(best_h, alpha, ml, mi)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: Nesterov projected gradient
    if _el() < budget_s - 350:
        configs = [(20000, 50, 1200, 0.004), (200000, 70, 1000, 0.0025),
                   (2000000, 90, 800, 0.0015), (20000000, 110, 600, 0.0008)]
        for alpha, ml, steps, s0 in configs:
            if _el() > budget_s - 300:
                break
            h_new = _nesterov_grad(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 3: SLSQP at high alpha
    if _el() < budget_s - 250:
        for alpha, ml in [(20000000, 100), (200000000, 120), (2000000000, 140)]:
            if _el() > budget_s - 200:
                break
            h_new = _slsqp(best_h, alpha, ml, 350)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Smart local search targeting critical coords
    if _el() < budget_s - 100:
        rem = budget_s - 100 - _el()
        ntries = int(rem / 0.0025)
        best_h, best_c5 = _smart_local(best_h, ntries, rng, 2.5e-5)

    # Phase 5: Second round of SLSQP + gradient
    if _el() < budget_s - 80:
        for alpha, ml in [(5000000000, 150)]:
            if _el() > budget_s - 60:
                break
            h_new = _slsqp(best_h, alpha, ml, 200)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
        if _el() < budget_s - 50:
            h_new = _nesterov_grad(best_h.copy(), 50000000, 120, 500, 0.0003)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 6: Intensive local search
    if _el() < budget_s - 40:
        rem = budget_s - 40 - _el()
        ntries = int(rem / 0.0015)
        best_h, best_c5 = _smart_local(best_h, ntries, rng, 1e-5)

    # Phase 7: Systematic sweep
    if _el() < budget_s - 5:
        best_h, best_c5 = _systematic_sweep(best_h, rng, n_passes=3)

    # Phase 8: Ultra-fine polish
    if _el() < budget_s - 1:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = int((budget_s - 1 - _el()) / 0.0005)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 20)) % nc2
            d = 3e-8 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
