def _c5(h):

def _c5_and_lags(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def _binarize(h, n, rng):
    h = np.where(h >= 0.5, 1.0, 0.0).copy()
    target = n // 2
    ones = int(h.sum())
    if ones < target:
        idx = rng.choice(np.where(h == 0)[0], target - ones, replace=False)
        h[idx] = 1.0
    elif ones > target:
        idx = rng.choice(np.where(h == 1)[0], ones - target, replace=False)
        h[idx] = 0.0
    return h

def _greedy(h, deadline, rng, max_iter=200000):
    """Accept-only improvements with random single swaps."""
    n = len(h)
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    all_ones = np.where(h == 1)[0]
    all_zeros = np.where(h == 0)[0]
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = int(rng.choice(all_ones))
        j = int(rng.choice(all_zeros))
        h[i], h[j] = 0.0, 1.0
        c_new = _c5(h)
        if c_new < c - 1e-15:
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
            stall = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if stall > 5000:

    return best_h, best_c

- Change h[i] from 1 to 0 where h[i+k*]=0 (reduces count by 1)
- Or change h[i+k*] from 0 to 1 where h[i]=1 (reduces count by 1)

But both require a compensating change that might increase other counts.

Let me implement a targeted steepest descent:
1. Find k* (the max lag)
2. For the max lag, identify "offender" positions
3. For each offender, try all compensating moves and keep the best
4. Apply the best move
5. Repeat

This is O(n) per iteration instead of O(n²), making it much faster.

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _c5_arr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _binarize(h, n, rng):
    h = np.where(h >= 0.5, 1.0, 0.0).copy()
    t = n // 2
    s = int(h.sum())
    if s != t:
        if s < t:
            idx = rng.choice(np.where(h == 0)[0], t - s, replace=False)
            h[idx] = 1.0
        else:
            idx = rng.choice(np.where(h == 1)[0], s - t, replace=False)
            h[idx] = 0.0
    return h

def _greedy(h, deadline, rng, max_iter=800000):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = 0.0, 1.0
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
            stall = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if stall > 8000:
            break
    return bh, bc

def _steepest(h, deadline, max_passes=50):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > deadline:
                break
            if h[i] != 1:
                continue
            best_j = -1
            best_cn = c
            for j in zeros:
                if h[j] != 0:
                    continue
                h[i], h[j] = 0.0, 1.0
                cn = _c5(h)
                h[i], h[j] = 1.0, 0.0
                if cn < best_cn - 1e-15:
                    best_cn = cn
                    best_j = j
            if best_j >= 0:
                h[i], h[best_j] = 0.0, 1.0
                c = best_cn
                improved = True
                if c < bc:
                    bc, bh = c, h.copy()
        if not improved:
            break
    return bh, bc

def _sa(h, deadline, seed, t_scale=1.0, lag_target=True):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    T0 = 0.004 * t_scale
    t_start = time.time()
    total = max(deadline - t_start, 1.0)
    k_max = n - 1
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total)
        T = T0 * (1e-5) ** frac
        if lag_target and (time.time() - t_start) % 0.5 < 1e-9:
            pass
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if lag_target and rng.random() < 0.55:
            arr = _c5_arr(h)
            k_max = int(np.argmax(arr))
            lag = k_max - (n - 1)
            if lag > 0 and lag < n:
                valid = np.arange(n - lag)
                off = valid[(h[valid] == 1) & (h[valid + lag] == 0)]
                if len(off) > 0:
                    i = int(off[int(rng.randint(len(off)))])
                    j = int(rng.choice(zeros))
                else:
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
            elif lag < 0 and abs(lag) < n:
                l = abs(lag)
                valid = np.arange(n - l)
                off = valid[(h[valid + l] == 1) & (h[valid] == 0)]
                if len(off) > 0:
                    i = int(off[int(rng.randint(len(off)))])
                    j = int(rng.choice(zeros))
                else:
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
            else:
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
        else:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        cn = _c5(h)
        if cn < c or (T > 1e-12 and rng.random() < np.exp(-(cn - c) / T)):
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, bc

def _targeted_steepest(h, deadline, max_rounds=200):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        arr = _c5_arr(h)
        k_max = int(np.argmax(arr))
        lag = k_max - (n - 1)
        if lag == 0:
            break
        found = False
        if lag > 0:
            valid = np.arange(n - lag)
            idx_i = valid[(h[valid] == 1) & (h[valid + lag] == 0)]
            idx_j = valid[(h[valid + lag] == 0) & (h[valid] == 1)]
        else:
            l = abs(lag)
            valid = np.arange(n - l)
            idx_i = valid[(h[valid + l] == 1) & (h[valid] == 0)]
            idx_j = valid[(h[valid] == 0) & (h[valid + l] == 1)]
        candidates = list(idx_i) + list(idx_j)
        if len(candidates) == 0:
            break
        best_delta = 0
        best_ij = None
        others = np.setdiff1d(np.arange(n), candidates)
        for pos in candidates[:60]:
            if time.time() > deadline:
                break
            if h[pos] == 1:
                comp = others[(h[others] == 0)]
            else:
                comp = others[(h[others] == 1)]
            for comp_pos in comp:
                h[pos] = 1.0 - h[pos]
                h[comp_pos] = 1.0 - h[comp_pos]
                cn = _c5(h)
                h[pos] = 1.0 - h[pos]
                h[comp_pos] = 1.0 - h[comp_pos]
                if cn < c - 1e-15:
                    delta = cn - c
                    if delta < best_delta:
                        best_delta = delta
                        best_ij = (pos, comp_pos)
        if best_ij is not None:
            p, q = best_ij
            h[p] = 1.0 - h[p]
            h[q] = 1.0 - h[q]
            c += best_delta
            found = True
            if c < bc:
                bc, bh = c, h.copy()
        if not found:
            break
    return bh, bc

def _multi_restart_sa(h, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    chunk = 60.0
    chunk_start = time.time()
    round_id = 0
    while time.time() < deadline:
        end = min(chunk_start + chunk, deadline)
        sh, sc = _sa(h, end, seed + round_id * 7, t_scale=1.0)
        if sc < bc:
            bc, bh = sc, sh
            h = sh.copy()
        else:
            pert = bh.copy()
            n_flips = int(n * 0.02) + 2
            ones = np.where(pert == 1)[0]
            zeros = np.where(pert == 0)[0]
            swap_from = rng.choice(ones, min(n_flips, len(ones)), replace=False)
            swap_to = rng.choice(zeros, min(n_flips, len(zeros)), replace=False)
            pert[swap_from] = 0.0
            pert[swap_to] = 1.0
            h = pert
        chunk_start = time.time()
        round_id += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
