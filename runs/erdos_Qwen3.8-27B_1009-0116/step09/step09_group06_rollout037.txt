            hi = mid

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_sm(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - cm), -500, 0)
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _obj_pen(x, alpha, m, n, pen):
    c = _corr_vec(x)
    cm = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - cm), -500, 0)
    return cm + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - n * 0.5) ** 2

def _grad_pen(x, alpha, m, n, pen):
    g = _grad_sm(x, alpha, m, n)
    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g

def _obj_nop(x, alpha, m, n):
    c = _corr_vec(x)
    cm = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - cm), -500, 0)
    return cm + np.log(np.exp(s).sum()) / alpha

def _se(x, t): return x.sum() - t
def _sj(x, t): return np.ones(len(x))

def _lbfgs(h, alpha, m, mi, pen):

    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj_pen, h, jac=_grad_pen, args=(alpha, m, n, pen),
