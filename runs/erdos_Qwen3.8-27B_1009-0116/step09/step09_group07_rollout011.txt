```python
def _repair(h, n):
    """Clip to [0,1] and adjust sum to n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    m = c.max()
    idx = np.argsort(c)[-n_active:]

    s = k_top * (c[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    logsumexp = s.max() + np.log(np.exp(s).sum())
    obj = m + logsumexp / k_top
    
    # Gradient

    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1

def _slsqp_refine(h, alpha, n_active, maxiter):

        return x.sum() - target

        return np.ones(len(x))
```

```python
        k = int(k)
        L = k - nm1
        # Compute gradient for this lag
        if L >= 0:
            # c_k = dx * sum_{i=0}^{n-1-L} h[i] * (1-h[i+L])
            # d/dh[j] = dx * (1-h[j+L]) for j < n-L
            #          = dx * (-h[j-L]) for j >= L
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
        else:
            M = -L
            # c_k = dx * sum_{i=M}^{n-1} h[i] * (1-h[i-M])
            # d/dh[j] = dx * (1-h[j-M]) for j >= M
            #          = dx * (-h[j+M]) for j < n-M
            g[M:] += (1.0 - h[:n-M])
            g[:n-M] -= h[M:]
    g *= (2.0 / n)  # dx
    h_new = h - eta * g
    return _repair(h_new, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(24):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _subgrad_dir(h, n_active):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-n_active:]
    g = np.zeros(n)
    nm1 = n - 1
    for k in idx:
        k = int(k)
        L = k - nm1
        if L >= 0:
            g[:n - L] += (1.0 - h[L:])
            g[L:] -= h[:n - L]
        else:
            M = -L
            g[M:] += (1.0 - h[:n - M])
            g[:n - M] -= h[M:]
    g *= (2.0 / n)
    return g

def _subgrad_phase(h, n, budget_frac, t0, budget_s, rng):
    end_t = t0 + budget_s * budget_frac
    best_h = h.copy()
    best_c5 = _eval(best_h)
    eta = 0.08
    n_act = 30
    momentum = np.zeros(n)
    mu = 0.7
    c5_prev = best_c5
    steps = 0
    while time.time() < end_t:
        g = _subgrad_dir(best_h, n_act)
        momentum = mu * momentum + g
        h_new = _repair(best_h - eta * momentum, n)
        c5_new = _eval(h_new)
        if c5_new < c5_prev:
            best_h = h_new
            c5_prev = c5_new
            eta = min(eta * 1.05, 0.3)
            if c5_new < best_c5:
                best_c5 = c5_new
        else:
            h_new = _repair(best_h - 0.3 * eta * momentum, n)
            c5_new = _eval(h_new)
            if c5_new < c5_prev:
                best_h = h_new
                c5_prev = c5_new
            else:
                eta *= 0.7
        steps += 1
        if steps % 100 == 0:
            n_act = min(80, 20 + steps // 20)
    return best_h, best_c5

def _slsqp_obj(x, alpha, n_act):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-n_act:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, n_act):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-n_act:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, n_act, maxiter):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, n_act), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 25})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 3e-6 * rng.randn()
        if abs(d) < 1e-9:
            continue
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

    rng = np.random.RandomState(seed)
```
