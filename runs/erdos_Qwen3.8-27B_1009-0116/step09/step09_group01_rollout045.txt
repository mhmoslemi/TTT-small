    n = len(h)
    N = 1
    m = 2 * n - 1
    while N < m:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    Nv = np.empty(n)

    Nv[0] = 0.0

        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
    mx = float(max(np.max(P), np.max(Nv)))
    return P, Nv, mx

    n = len(h)

        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _gW(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    Nf = 1
    m = 2 * n - 1
    while Nf < m:
        Nf <<= 1
    H = np.fft.rfft(h, Nf)
    R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
    L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
    q = -(R + L)

    if n > 1:

import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _c5_arr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = min(-d, h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _pn(h):
    n = len(h)
    N = 1
    m = 2 * n - 1
    while N < m:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    Nv = np.empty(n)
    P[0] = tot - ac[0]
    Nv[0] = 0.0
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
    mx = float(max(np.max(P), np.max(Nv)))
    return P, Nv, mx

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _gW(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    Nf = 1
    m = 2 * n - 1
    while Nf < m:
        Nf <<= 1
    H = np.fft.rfft(h, Nf)
    R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
    L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
    q = -(R + L)
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)
