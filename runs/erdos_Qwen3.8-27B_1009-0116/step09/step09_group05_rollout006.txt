            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a

def _eval(h):
    n = len(h)

def _corr_full(h):

    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    base = m + np.log(np.exp(s).sum()) / alpha
    sd = x.sum() - len(x) * 0.5
    return base + pen * sd * sd

def _smooth_grad(x, alpha, m_lags, pen):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)

    w = np.exp(s); w /= w.sum()

        if w[ii] < 1e-15:

        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    sd = x.sum() - n * 0.5

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _lbfgsb(h, alpha, m_lags, pen, maxiter, ftol):

def _nesterov_proj(h, alpha, m_lags, steps, step0):

        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 3: Nesterov projected gradient
    if time.time() - t0 < budget_s - 350:
        configs = [
            (30000, 40, 500, 0.004),
            (200000, 60, 400, 0.002),
            (1e6, 80, 300, 0.001),
            (5e6, 100, 250, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s - 300:
                break
            h_new = _nesterov_proj(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 4: Another L-BFGS-B pass with very high alpha
    if time.time() - t0 < budget_s - 200:
        for alpha in [3e9, 1e10, 5e10]:
            if time.time() - t0 > budget_s - 160:
                break
            h_new = _lbfgsb(best_h, alpha, 200, alpha * 5, 400, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 5: Gradient-guided 2-point search
    if time.time() - t0 < budget_s - 80:
        remaining = budget_s - 90 - (time.time() - t0)
        nitr = int(remaining / 0.015)
        best_h, best_c5 = _two_point(best_h, min(nitr, 80000), 1e7, 100, rng, t0, budget_s - 80)

    # Phase 6: Random perturbation search
    if time.time() - t0 < budget_s - 40:
        remaining = budget_s - 50 - (time.time() - t0)
        ntries = int(remaining / 0.004)
        best_h, best_c5 = _random_perturb(best_h, min(ntries, 100000), rng, 3e-6, 4, t0, budget_s - 40)

    # Phase 7: Fine polish
    if time.time() - t0 < budget_s - 3:
        best_h, best_c5 = _fine_polish(best_h, rng, t0, budget_s - 2)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
