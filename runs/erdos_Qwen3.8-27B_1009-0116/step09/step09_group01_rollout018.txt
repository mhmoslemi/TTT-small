def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = min(-d, h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _pn(h):

    N = 1
    m = 2 * n - 1
    while N < m:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]

        mx = float(P[0])

        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]

        m = 2 * n - 1

        q = -(R + L)

def _sg(h, a):

    n = len(h)

        if rem < 3:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.28)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    end2 = t0 + 260.0
    if best_c > 0.3806:
        bh, bc = _sa_cont(best, rng, min(180.0, end2 - time.time()))
        if bc < best_c:
            best, best_c = bh, bc

    end3 = t0 + 380.0
    bh, bc = _sa_bin(best, rng, min(120.0, end3 - time.time()))
    if bc < best_c + 0.002:
        bh_c = _repair(bh.astype(float), n0)
        rem = end3 - time.time()
        if rem > 5:
            _, bh2, bc2 = _fista(bh_c, 5000, rem)
            if bc2 < best_c:
                best, best_c = bh2, bc2

    end4 = t0 + 720.0
    best, best_c = _loc(best, rng, end4 - time.time(), s0=4e-5)

    end5 = t0 + 960.0
    best, best_c = _loc(best, rng, end5 - time.time(), s0=5e-6)

    end6 = t0 + budget_s - 5.0
    best, best_c = _loc(best, rng, end6 - time.time(), s0=1e-6)

    n = len(best)
    for var in [1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1])]:
        hv = _proj(var, n * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
