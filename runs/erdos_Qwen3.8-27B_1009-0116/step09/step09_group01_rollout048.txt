    """Combinatorial local search on binary sequences."""
    n = len(h0)
    S = n * 0.5
    # Threshold to binary
    h = np.zeros(n)
    # Choose top S values
    thresh = np.sort(h0)[::-1][int(S)]
    h[h0 >= thresh] = 1.0
    # Ensure exactly S ones
    n_ones = h.sum()
    if n_ones != S:
        # Fix
        idx = np.argsort(h0)[::-1]
        h = np.zeros(n)
        h[idx[:int(S)]] = 1.0
    
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    
    ones = set(np.nonzero(h)[0].tolist())
    zeros = set(range(n)) - ones
    
    t0 = time.time()
    it = 0
    # Simulated annealing
    T = 0.01  # Initial temperature
    T_end = 0.0001
    total_iters = 100000
    
    while time.time() < dl:
        # Temperature schedule
        fr = min(1.0, it / total_iters)
        T = T_end + (T - T_end) * (1 - fr)**2
        
        # Choose move
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        
        # Try swap
        h[i] = 0.0
        h[j] = 1.0
        cn = _c5(h)
        
        # Accept
        if cn < c or rng.random() < np.exp(-(cn - c) / T):
            c = cn
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)

                best_h = h.copy()

            h[i] = 1.0
            h[j] = 0.0

    t0 = time.time()

            y = h + ((tv-1.0)/tn)*(h-hp)
