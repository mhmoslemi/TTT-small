import numpy as np
import time

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / n))

def _c5_all(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(corr.max())

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _subgrad(h, corr, n, M=50, tol_frac=0.02):
    """Compute subgradient of max_k O(k) using top-M lags."""
    mx = float(corr.max())
    threshold = mx * (1.0 - tol_frac)
    active = corr >= threshold
    n_active = int(active.sum())
    if n_active == 0:
        active[np.argmax(corr)] = True
        n_active = 1
    # Weight lags by how close they are to max (soft weighting)
    weights = np.zeros_like(corr)
    weights[active] = (corr[active] - threshold) / (mx - threshold + 1e-30)
    weights[active] += 0.5  # base weight for all active
    weights[active] /= (weights[active].sum() + 1e-30)
    
    # For each active lag k (indexed as k - (n-1) in corr), compute gradient contribution
    # corr[m] corresponds to lag k = m - (n-1)
    g = np.zeros(n)
    for m in np.nonzero(active)[0]:
        k = m - (n - 1)
        w = weights[m]
        if k == 0:
            # O(0) = Σ h[i](1-h[i]), ∂/∂h[j] = 1 - 2h[j]
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            # O(k) = Σ_{i=0}^{n-1-k} h[i](1-h[i+k])
            # ∂/∂h[j] = (1-h[j+k]) for j ≤ n-1-k, plus -h[j-k] for j ≥ k
            upper = n - k  # indices j where j+k ≤ n-1
            g[:upper] += w * (1.0 - h[k:])  # (1-h[j+k]) for j=0..n-1-k
            g[k:] += w * (-h[:n-k])  # -h[j-k] for j=k..n-1
        else:
            # k < 0, let k2 = -k
            k2 = -k
            # O(-k2) = Σ_{i=k2}^{n-1} h[i](1-h[i-k2])
            # ∂/∂h[j] = (1-h[j-k2]) for j ≥ k2, plus -h[j+k2] for j ≤ n-1-k2
            g[k2:] += w * (1.0 - h[:n-k2])  # (1-h[j-k2]) for j=k2..n-1
            g[:n-k2] += w * (-h[k2:])  # -h[j+k2] for j=0..n-1-k2
    return g, mx

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _c5_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(60):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _subgrad_step(h, corr, n, M=40):
    mx = float(corr.max())
    threshold = mx - 0.06 * mx
    active_mask = corr >= threshold
    n_act = int(active_mask.sum())
    if n_act == 0:
        active_mask[argmax(int, corr)] = True
        n_act = 1
    idx = np.nonzero(active_mask)[0]
    wts = (corr[idx] - threshold) / (mx - threshold + 1e-30)
    wts += 1.0
    wts /= wts.sum()
    g = np.zeros(n)
    for mi in range(len(idx)):
        m = idx[mi]
        k = m - (n - 1)
        w = wts[mi]
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            upper = n - k
            g[:upper] += w * (1.0 - h[k:])
            g[k:] += -w * h[:upper]
        else:
            k2 = -k
            lower = k2
            g[lower:] += w * (1.0 - h[:n - k2])
            g[:n - k2] += -w * h[k2:]
    return g, mx

def _sg_optimize(h0, deadline, best_c_in):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    best = h0.copy()
    best_c = best_c_in
    corr = _c5_raw(h)
    lr = 0.02
    momentum = 0.0
    mom_vec = np.zeros(n)
    t_start = time.time()
    while time.time() < deadline:
        g, mx = _subgrad_step(h, corr, n)
        momentum = 0.9 * momentum
        mom_vec = momentum * mom_vec + g
        step = lr / (1.0 - momentum)
        h_new = _proj(h - step * mom_vec, S)
        corr_new = _c5_raw(h_new)
        c_new = float(corr_new.max()) * (2.0 / n)
        if c_new < best_c - 1e-15:
            h = h_new; corr = corr_new
            best = h.copy(); best_c = c_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best, best_c

def _targeted_search(h0, rng, deadline):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    corr = _c5_raw(h)
    mx = float(corr.max())
    best_c = mx * (2.0 / n)
    best = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    step_scale = 0.05
    while time.time() < deadline:
        frac = (time.time() - t_start) / total_t
        step_scale = 0.05 * (0.01) ** (frac ** 1.5)
        if step_scale < 1e-6:
            break
        corr = _c5_raw(h)
        mx = float(corr.max())
        k_worst = int(np.argmax(corr)) - (n - 1)
        # Build candidate pairs for worst few lags
        top_lags = np.argsort(corr)[-8:]
        candidates = []
        for m in top_lags:
            k = m - (n - 1)
            if k == 0:
                # For k=0, reduce variance: move extreme values toward 0.5
                over = np.where(h > 0.5 + 0.05)[0]
                under = np.where(h < 0.5 - 0.05)[0]
                if len(over) > 0 and len(under) > 0:
                    for _ in range(5):
                        i = int(over[rng.randint(len(over))])
                        j = int(under[rng.randint(len(under))])
                        candidates.append((i, j, 1.0))
                        candidates.append((j, i, 1.0))
            elif k > 0:
                overlap = n - k
                contrib = h[:overlap] * (1.0 - h[k:])
                top_idx = np.argsort(contrib)[-12:]
                for ti in top_idx:
                    i = int(ti); j = int(ti + k)
                    if h[i] > h[j] + 0.01:
                        candidates.append((i, j, 1.0))
            else:
                k2 = -k
                overlap = n - k2
                contrib = h[k2:] * (1.0 - h[:overlap])
                top_idx = np.argsort(contrib)[-12:]
                for ti in top_idx:
                    j = int(ti + k2); i = int(ti)
                    if h[j] > h[i] + 0.01:
                        candidates.append((j, i, 1.0))
        # Also add random candidates
        for _ in range(4):
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i != j:
                candidates.append((i, j, 1.0))
        accepted = False
        rng.shuffle(candidates)
        for (i, j, _) in candidates:
            if i == j:
                continue
            d = step_scale * rng.random()
            if rng.random() < 0.5:
                delta = -d
            else:
                delta = d
            if h[i] + delta < 0 or h[i] + delta > 1:
                continue
            if h[j] - delta < 0 or h[j] - delta > 1:
                continue
            h[i] += delta; h[j] -= delta
            corr_new = _c5_raw(h)
            mx_new = float(corr_new.max())
            if mx_new < mx - 1e-14:
                corr = corr_new; mx = mx_new
                c = mx * (2.0 / n)
                if c < best_c - 1e-14:
                    best_c = c; best = h.copy()
                accepted = True
                break
            else:
                h[i] -= delta; h[j] += delta
        if not accepted and rng.random() < 0.01:
            # Large random perturbation
            for _ in range(3):
                i = int(rng.randint(n)); j = int(rng.randint(n))
                if i != j:
                    d = (rng.random() - 0.5) * 0.15
                    if 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1:
                        h[i] += d; h[j] -= d
            corr = _c5_raw(h)
            mx = float(corr.max())
    return best, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _multi_res(h0, rng, deadline, n_orig):
    best = h0.copy()
    best_c = _c5(best)
    sizes = [n_orig, 280, 384, 560, 640]
    sizes = [s for s in sizes if s <= 1000]
    per_slot = max(2.0, (deadline - time.time()) / len(sizes))
    for m in sizes:
        if time.time() >= deadline - 2:
            break
        slot = min(per_slot, deadline - time.time() - 1)
        if slot < 3:
            break
        if m == n_orig:
            hm = best.copy()
        else:
            hm = _resample(best, m)
            # Add small random perturbation
            hm += rng.normal(0, 0.02, m)
            hm = _proj(hm, m * 0.5)
        # Quick optimize at this resolution
        h_opt, c_opt = _targeted_search(hm, rng, time.time() + slot * 0.7)
        c_opt = _c5(h_opt)
        if m != n_orig:
            h_back = _resample(h_opt, n_orig)
            c_back = _c5(h_back)
            if c_back < best_c:
                best = h_back; best_c = c_back
    return best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:

    for hh in (1.0 - h0, h0[::-1], 0.5 + 0.5*(h0 - h0[::-1])):
        h2 = _proj(hh.copy(), S)
        c2 = _c5(h2)
        if c2 < best_c:

    # Phase 1: Subgradient descent
    end1 = t0 + min(250.0, 0.25 * budget_s)
    if time.time() < end1:
        h_s, c_s = _sg_optimize(best, end1, best_c)
        if c_s < best_c:

    # Phase 2: Targeted search (main workhorse)
    end2 = t0 + min(700.0, 0.7 * budget_s)
    if time.time() < end2:
        best, best_c = _targeted_search(best, rng, end2)
    # Phase 3: Multi-resolution restarts
    end3 = t0 + budget_s - 120.0
    if time.time() < end3:
        best = _multi_res(best, rng, end3, n0)
    # Phase 4: Final targeted polish
    end4 = t0 + budget_s - 3.0
    if time.time() < end4:
        best, best_c = _targeted_search(best, rng, end4)
    # Final variants
    for hh in (1.0 - best, best[::-1], 0.5 + 0.5*(best - best[::-1])):
        h2 = _proj(hh.copy(), len(hh) * 0.5)
        c2 = _c5(h2)
        if c2 < best_c:
            best = h2; best_c = c2
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
