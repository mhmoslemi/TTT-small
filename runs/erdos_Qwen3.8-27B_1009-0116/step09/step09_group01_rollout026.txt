import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), h[i] if d < 0 else 1.0 - h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _binarize(h, n):
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:n // 2]] = 1.0
    return out

def _single_swap(h, rng, deadline, target_lag=False):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    stall = 0
    while time.time() < deadline:
        if len(ones) < 1 or len(zeros) < 1:
            break
        if it % 500 == 0 and it > 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if stall > 3000:
                for _ in range(8):
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    h[ii], h[jj] = 0, 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                c = _c5(h)
                stall = 0
        if target_lag and it % 100 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            im = np.argmax(corr)
            k = im - (n - 1)
            if 0 < k < n:
                bo = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
                sz = np.where((h[:n-k] == 0) & (h[k:] == 1))[0]
            elif k < 0 and -k < n:
                k2 = -k
                bo = np.where((h[k2:] == 1) & (h[:n-k2] == 0))[0] + k2
                sz = np.where((h[k2:] == 0) & (h[:n-k2] == 1))[0] + k2
            else:
                bo, sz = ones, zeros
            if len(bo) > 0 and len(sz) > 0:
                i = bo[rng.randint(len(bo))]
                j = sz[rng.randint(len(sz))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1

            c = cn
            stall = 0
            if c < bc:

        else:
            h[i] = 1
            h[j] = 0

def _double_swap(h, rng, deadline):

    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    ones = np.where(h == 1)[0]

                    j2 = zeros[rng.randint(len(zeros))]
                    if i1 != i2 and j1 != j2:

                c = _c5(h)

        i2 = ones[rng.randint(len(ones))]

        j2 = zeros[rng.randint(len(zeros))]
        if i1 == i2 or j1 == j2:
            continue

                bc, bh = c, h.copy()
        else:
            h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
            stall += 1
        it += 1
    return bh, bc

def _block_move(h, rng, deadline):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    it = 0
    while time.time() < deadline:
        L = rng.randint(3, max(4, n // 10))
        start = rng.randint(0, n - L)
        shift = rng.randint(1, n // 4)
        new_start = (start + shift) % n
        block = h[start:start + L].copy()
        h[start:start + L] = 0
        for s in range(L):
            idx = (new_start + s) % n
            h[idx] = block[s]
        # Rebalance: need exactly n/2 ones. Block preserves count since we move it.
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[start:start + L] = block  # undo
        it += 1
    return bh, bc

def _symtry(h, n):
    best, bc = h.copy(), _c5(h)
    for var in [1.0 - h, h[::-1], np.roll(h, n // 2)]:
        v = _proj(var, n * 0.5)
        cv = _c5(v)
        if cv < bc:
            best, bc = v, cv
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
