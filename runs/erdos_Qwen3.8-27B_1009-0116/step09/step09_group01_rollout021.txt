            break
        h[mask] += d / cm
        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-12:
                break
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
    return h

def _sa_bin(h, rng, deadline, T0=0.004, T1=1e-6, double=False):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    t0 = time.time()
    tot = max(deadline - t0, 0.1)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    it = 0
    while time.time() < deadline:
        fr = min(1.0, (time.time() - t0) / tot)
        T = T0 * (T1 / T0) ** fr
        it += 1
        if double and len(ones) >= 2 and len(zeros) >= 2:
            iz = rng.choice(len(zeros), 2, replace=False)
            io = rng.choice(len(ones), 2, replace=False)
            i1, i2 = zeros[iz[0]], zeros[iz[1]]
            j1, j2 = ones[io[0]], ones[io[1]]
            h[i1] = h[i2] = 1.0
            h[j1] = h[j2] = 0.0
            cn = _c5(h)
            d = cn - c
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
                c = cn
                zeros.remove(i1); zeros.remove(i2)
                zeros.append(j1); zeros.append(j2)
                ones.remove(j1); ones.remove(j2)
                ones.append(i1); ones.append(i2)
                if c < bc:
                    bc, bh = c, h.copy()
            else:
                h[i1] = h[i2] = 0.0
                h[j1] = h[j2] = 1.0
        else:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            cn = _c5(h)
            d = cn - c
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
                c = cn
                zeros.remove(i); zeros.append(j)
                ones.remove(j); ones.append(i)
                if c < bc:
                    bc, bh = c, h.copy()
            else:
                h[i], h[j] = 0.0, 1.0
    return bh, bc

def _greedy_bin(h, rng, deadline):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    t0 = time.time()
    while time.time() < deadline:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        cn = _c5(h)
        if cn < c - 1e-14:
            c = cn
            zeros.remove(i); zeros.append(j)
            ones.remove(j); ones.append(i)
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    return bh, bc

def _ils_bin(h, rng, deadline, n_kicks=30):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    t0 = time.time()
    kick_budget = max(0.1, (deadline - t0) / n_kicks)
    for _ in range(n_kicks):
        end_kick = time.time() + kick_budget
        if end_kick > deadline:
            end_kick = deadline
        if time.time() >= deadline:
            break
        hg, cg = _greedy_bin(h, rng, end_kick)
        if cg < bc:
            bc, bh = cg, hg
        h = hg.copy()
        # Kick: random 5-15 swaps
        nk = rng.randint(3, 12)
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        for _ in range(nk):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            zeros.remove(i); zeros.append(j)
            ones.remove(j); ones.append(i)
    return bh, bc

def _nonbin_2pair(h, rng, deadline, S):
    n = len(h)
    h = h.copy()
    c = _c5(h)
    bc, bh = c, h.copy()
    t0 = time.time()
    tot = max(deadline - t0, 0.1)
    T0, T1 = 0.001, 1e-8
    base = 0.04
    while time.time() < deadline:
        fr = min(1.0, (time.time() - t0) / tot)
        T = T0 * (T1 / T0) ** fr
        step = base * (0.1 + 0.9 * T / T0)
        idx = rng.choice(n, 4, replace=False)
        i, j, k, l = idx
        s = step * rng.uniform(0.2, 1.0)
        t = step * rng.uniform(0.2, 1.0)
        h[i] += s; h[j] -= s; h[k] += t; h[l] -= t
        if h.min() < 0 or h.max() > 1:
            h[i] -= s; h[j] += s; h[k] -= t; h[l] += t
            continue
        cn = _c5(h)
        d = cn - c
        if d < -1e-14 or rng.random() < np.exp(-d / max(T, 1e-15)):
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
                bh = _proj_sum(bh, S)
        else:
            h[i] -= s; h[j] += s; h[k] -= t; h[l] += t
    return bh, bc

def _fista_light(h0, a, timeout, S):
    n = len(h0)
    h = _proj_sum(h0.copy(), S)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    t0 = time.time()
    bc = _c5(h)
    bh = h.copy()
    L = 8.0
    for it in range(50000):
        if time.time() - t0 > timeout:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            if y.min() < -0.1 or y.max() > 1.1:
                y = _proj_sum(y, S)
        else:
            y = h
        corr = np.correlate(y, 1.0 - y, mode='full')
        mx = corr.max()
        tol = max(mx * 0.01, 1e-8)
        near = np.where(corr >= mx - tol)[0]
        g = np.zeros(n)
        for kidx in near:
            lag = kidx - (n - 1)
            if lag >= 0:
                g[:n - lag] += (1.0 - y[lag:])
                g[lag:] -= y[:n - lag]
            else:
                g[-lag:] += (1.0 - y[:n + lag])
                g[:n + lag] -= y[-lag:]
        g *= 2.0 / n
        step = 1.0 / max(L, 1e-10)
        for _ in range(10):
            xn = _proj_sum(y - step * g, S)
            cn = _c5(xn)
            d = xn - y
            rhs = _c5(y) + float(np.dot(g, d)) + 0.5 / step * float(np.dot(d, d))
            if cn <= rhs + 1e-9:
                break
            step *= 0.4
        L = max(1.0 / max(step, 1e-10) * 0.9, 1.0)
        hp = h
        h = xn
        c = _c5(h)
        if c < bc:
            bc, bh = c, h.copy()
    return h, bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    S = n0 * 0.5
    h0 = _proj_sum(h0, S)
    c0 = _c5(h0)
    best, best_c = h0.copy(), c0

    # Binarize initial
    hb = _to_bin(h0, rng)
    cb = _c5(hb)
    if cb < best_c:
        best, best_c = hb, cb

    # Phase 1: Binary SA single (250s)
    end1 = t0 + 250
    h1, c1 = _sa_bin(hb, rng, end1, T0=0.005, T1=1e-6, double=False)
    if c1 < best_c:
        best, best_c = h1, c1

    # Phase 2: Binary SA double (200s)
    end2 = t0 + 450
    h2, c2 = _sa_bin(h1, rng, end2, T0=0.006, T1=1e-6, double=True)
    if c2 < best_c:
        best, best_c = h2, c2
    h2b, c2b = _sa_bin(hb, rng, end2, T0=0.004, T1=1e-7, double=True)
    if c2b < best_c:
        best, best_c = h2b, c2b

    # Phase 3: ILS binary (200s)
    end3 = t0 + 650
    h3, c3 = _ils_bin(h1 if c1 < cb else hb, rng, end3, n_kicks=40)
    if c3 < best_c:
        best, best_c = h3, c3

    # Phase 4: Greedy polish (80s)
    end4 = t0 + 730
    h4, c4 = _greedy_bin(best, rng, end4)
    if c4 < best_c:
        best, best_c = h4, c4

    # Phase 5: Non-binary FISTA light (120s)
    end5 = t0 + 850
    hn = best.astype(float)
    for a in [200, 800, 3000]:
        rem = end5 - time.time()
        if rem < 10:
            break
        hf, bh, bc = _fista_light(hn, a, rem * 0.3, S)
        if bc < best_c:
            best, best_c = bh, bc
        hn = hf

    # Phase 6: Non-binary 2-pair SA (150s)
    end6 = t0 + 1000
    h6, c6 = _nonbin_2pair(best, rng, end6, S)
    if c6 < best_c:
        best, best_c = h6, c6

    # Phase 7: Final binary polish (70s)
    end7 = t0 + budget_s - 10
    h7b = _to_bin(best, rng)
    h7, c7 = _sa_bin(h7b, rng, end7, T0=0.003, T1=1e-7, double=False)
    if c7 < best_c:
        best, best_c = h7, c7
    h7g, c7g = _greedy_bin(h7, rng, end7)
    if c7g < best_c:
        best, best_c = h7g, c7g

    # Final: ensure constraints
    n = len(best)
    best = _proj_sum(best, n * 0.5)
    best_c = _c5(best)

    # Try evaluation helper if available
    if 'evaluate_erdos_solution' in globals():
        try:
            eval_c = evaluate_erdos_solution(best)
            if eval_c is not None and eval_c < best_c:
                best_c = float(eval_c)
        except Exception:
            pass

    return best, best_c, len(best)
