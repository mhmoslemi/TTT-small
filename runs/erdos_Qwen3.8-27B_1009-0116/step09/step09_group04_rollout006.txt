    if s < 1e-12:

            else:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cf(x, hn):
    return x.sum() - hn

def _cj(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pgrad(h, alpha, ml, steps, s0, mu):
    n = len(h)
    v = np.zeros(n)
    step = s0
    cur = _eval(h)
    for _ in range(steps):
        g = _sgrad(h, alpha, ml)
        v = mu * v + (1.0 - mu) * g
        gn = np.linalg.norm(v)
        if gn > 1e-15:
            v /= gn
        taken = False
        for _ in range(15):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cur:
                h, cur, taken = hn, cn, True
                step = min(step * 1.08, 0.3)
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, 0.3)
    return _repair(h, n)

def _local_sa(h, rng, nit, ml, temp0, decay, moves_per_iter):
    """Multi-move subgradient-guided simulated annealing."""
    n = len(h)
    cc = _eval(h)
    bh, bc = h.copy(), cc
    t = temp0
    no_improve = 0
    for it in range(nit):
        t *= decay
        c = _eval_corr(h)
        top = np.argsort(c)[-ml:]
        improved = False
        for _ in range(moves_per_iter):
            k = top[rng.randint(ml)]
            L = k - (n - 1)
            if L >= 0:
                contrib = h[L:] * (1.0 - h[:n-L])
                if contrib.max() < 1e-4:
                    continue
                j = np.argmax(contrib * rng.rand(n-L))
                i1, i2 = j + L, j
            else:
                M = -L
                contrib = h[:n-M] * (1.0 - h[M:])
                if contrib.max() < 1e-4:
                    continue
                j = np.argmax(contrib * rng.rand(n-M))
                i1, i2 = j, j + M
            d = 0.003 * (0.5 + rng.rand())
            ni1, ni2 = h[i1] - d, h[i2] + d
            if ni1 < -0.1 or ni2 > 1.1:
                continue
            h[i1], h[i2] = max(0, ni1), min(1, ni2)
            cn = _eval(h)
            delta = cn - cc
            if delta < 0 or (t > 1e-10 and rng.rand() < np.exp(-delta / max(t, 1e-12))):
                cc = cn
                improved = True
                if cn < bc:
                    bc, bh = cn, h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i1], h[i2] = ni1, ni2
        if not improved and t < 1e-6:
            # Random restart within small neighborhood
            idx = rng.randint(n, 8)
            d = 0.002 * rng.randn(8)
            d -= d.mean()
            h[idx] += d
            h = _repair(h, n)
            cc = _eval(h)
    return bh, bc

def _fine_polish(h, rng, npol):
    """Ultra-fine 2-3 point polish."""
    n = len(h)
    h2 = h.copy()
    c2 = _eval(h2)
    for _ in range(npol):
        r = rng.rand()
        if r < 0.7:

            ni, nj = h2[i] + d, h2[j] - d

                cn = _eval(h2)

                    c2 = cn

                    h2[i], h2[j] = ni - d, nj + d
        else:

            d1, d2 = 2e-8 * rng.randn(2)
