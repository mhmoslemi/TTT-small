def _eval_corr(h):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:

                h[idx] += a
                diff -= a

    cmax = c.max()
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - cmax)
    s -= s.max()
    return cmax + np.log(np.exp(s).sum()) / alpha

def _soft_grad(h, alpha, m, n):
    dx = 2.0 / n
    c = _eval_corr(h)
    cmax = c.max()
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - cmax)
    s -= s.max()
    w = np.exp(s)

    # Phase 4: Kick + re-optimize to escape local minima
    if time.time() < t_end - 120:
        for kick in [0.01, 0.005, 0.002]:
            if time.time() > t_end - 80:
                break
            h_kick = best_h + kick * rng.randn(n)
            h_kick = _repair(h_kick, n)
            h_new = _pgd(h_kick, 5000000, 90, 50, 0.0005, t_end - 50)
            c_new = _eval(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    # Phase 5: Smart local search
    if time.time() < t_end - 50:
        rem = t_end - 50 - time.time()
        n_iter = int(rem / 0.0012)
        best_h, best_c = _smart_local(best_h, rng, n_iter, t_end - 20, 4e-5)

    # Phase 6: Two-point ultra-fine polish
    if time.time() < t_end - 3:
        rem = t_end - 3 - time.time()
        npol = int(rem / 0.0006)
        best_h, best_c = _two_point_polish(best_h, rng, t_end - 1, npol)

    # Final SLSQP micro-polish if time remains
    if time.time() < t_end - 10:
        h_new = _slsqp_opt(best_h, 500000000, 110, n, 150, t_end - 3)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
