For each pair (i, j), consider moving h[i] → h[i]+t and h[j] → h[j]-t for t ∈ [t_min, t_max].
The objective as a function of t is:
f(t) = max_k (c_k + t * (a_k[i] - a_k[j]))

where a_k[i] is the coefficient of h[i] in correlation k.

This is the max of linear functions of t, so f(t) is convex piecewise linear.
The minimum is either at the endpoints or at a breakpoint.

For a given pair (i,j), the function f(t) is convex, so we can find its minimum by:
1. Evaluating at t=0 (current)
2. Evaluating at t=t_min and t=t_max
3. If f is decreasing at t=0, go to the right; if increasing, go left
4. Binary search for the minimum

But computing f(t) for arbitrary t requires updating the correlations, which is O(n) per update (since only pairs involving i or j change).

Actually, for a fixed pair (i,j), changing h[i] by t and h[j] by -t:
- Correlation at shift k changes by: t * (coeff_i(k) - coeff_j(k))
- where coeff_i(k) is the coefficient of h[i] in c(k)

For shift s > 0: c(s) = sum_{m} h[m+s](1-h[m])
Coeff of h[i] in c(s): (1-h[i-s]) if i≥s, plus -h[i+s] if i+s<n
Coeff is: (1-h[i-s])*[i≥s] - h[i+s]*[i+s<n]

This changes as we modify h! So it's not purely linear. Wait, but for a SINGLE step (changing h[i] by t), the coefficient IS fixed (it depends on current h values of other coordinates). So for a single pair move, it IS linear in t.

So: f(t) = max_k (c_k + t * d_k) where d_k = coeff_i(k) - coeff_j(k) (computed at current h).

This is a convex piecewise-linear function of t. To minimize:
- d_k > 0 means c_k increases with t → we want t < 0 for those
- d_k < 0 means c_k decreases with t → we want t > 0 for those
- d_k = 0 means c_k unchanged

The minimum of max_k (c_k + t*d_k) over t is achieved where:
- There exist k₁ with d_{k₁} > 0 and k₂ with d_{k₂} < 0 such that c_{k₁} + t*d_{k₁} = c_{k₂} + t*d_{k₂}
- i.e., t = (c_{k₂} - c_{k₁}) / (d_{k₁} - d_{k₂})

And we need this t to be in [t_min, t_max].

This is essentially a linear programming problem in 1D! We can solve it by:
1. Computing all d_k
2. For positive d_k: t ≤ (T - c_k) / d_k where T is the target max
3. For negative d_k: t ≥ (T - c_k) / d_k
4. Feasible interval for t is the intersection

But we're minimizing, not just checking feasibility. The minimum value of max(c_k + t*d_k) is:
- If all d_k ≤ 0: go as far as t_max
- If all d_k ≥ 0: go as far as t_min
- Otherwise: the minimum is at the crossing point of the "upper envelope"

For practical purposes, we can:
1. Compute d_k for all k (or just the top few active ones)
2. Find t* that minimizes the max
3. Move to t*

def _pair_optimal(h, i, j, c, n):
    """Find optimal t for h[i]+=t, h[j]-=t, given current correlations c"""
    dx = 2.0/n
    # Compute d_k = coeff of h[i] in c_k - coeff of h[j] in c_k
    # For shift s (where s = k_idx - (n-1)):
    # coeff of h[pos] in c at shift s:
    #   if s > 0: (1-h[pos-s]) if pos>=s, plus -h[pos+s] if pos+s<n
    #   if s < 0: (1-h[pos+|s|]) if pos<n-|s|, plus -h[pos-|s|] if pos>=|s|
    #   if s = 0: 1-2*h[pos]
    d = np.zeros(2*n-1)
    for pos in [i, j]:
        sign = 1 if pos == i else -1
        for s in range(-(n-1), n):
            if s > 0:
                coeff = 0.0
                if pos >= s:
                    coeff += (1.0 - h[pos-s])
                if pos + s < n:
                    coeff -= h[pos+s]
            elif s < 0:
                ss = -s
                coeff = 0.0
                if pos + ss < n:
                    coeff += (1.0 - h[pos+ss])
                if pos >= ss:
                    coeff -= h[pos-ss]
            else:
                coeff = 1.0 - 2.0*h[pos]
            d[s + n - 1] += sign * coeff
    # Now minimize max_k (c[k] + t*d[k]) subject to t in [t_min, t_max]
    # Also need h[i]+t in [0,1] and h[j]-t in [0,1]
    t_min = max(-h[i], -(1.0-h[j]))  # h[i]+t >= 0, h[j]-t <= 1
    t_max = min(1.0-h[i], h[j])  # h[i]+t <= 1, h[j]-t >= 0
    
    # Find t that minimizes max_k (c[k] + t*d[k])
    # Binary search or direct computation
    # The function is convex in t, so we can use ternary search or:
    # Find where the "slope" changes sign
    # At t, the slope is: max of d[k] for k where c[k]+t*d[k] = max
    
    # Simpler: evaluate at a few candidate points
    # 1. t=0, t=t_min, t=t_max
    # 2. Crossing points of top active constraints
    
    # For efficiency, just do a small grid + refinement
    candidates = [t_min, t_max, 0.0]
    # Add crossing points

    if len(active_idx) > 1:

        top_active = np.argsort(-c)[-5:]

                ka, kb = top_active[a], top_active[b]

                    t_cross = (c[kb] - c[ka]) / (d[ka] - d[kb])
                    if t_min <= t_cross <= t_max:
                        candidates.append(t_cross)
    
    best_t = 0.0
    best_val = _c5(h)
    for t in candidates:
        if t < t_min or t > t_max:
            continue
        h[i] += t; h[j] -= t
        val = _c5(h)
        h[i] -= t; h[j] += t
        if val < best_val:
            best_val = val
            best_t = t
    return best_t, best_val

    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total_t)
        T = 0.002 * (0.001) ** frac
        active = np.where(c >= c.max() - tol / dx)[0]
        k_act = int(active[rng.randint(len(active))])
        k = k_act - (n - 1)
        if rng.random() < 0.7 and abs(k) > 0 and abs(k) < n:
            if k > 0:
                valid = (h[:n - k] < 0.999) & (h[k:] > 0.001)
                valid2 = (h[:n - k] > 0.001) & (h[k:] < 0.999)
            else:
                valid = (h[abs(k):] < 0.999) & (h[:n - abs(k)] > 0.001)
                valid2 = (h[abs(k):] > 0.001) & (h[:n - abs(k)] < 0.999)
            use_up = valid if rng.random() < 0.5 else valid2
            if use_up.sum() < 2:
                i = rng.randint(n)
                j = rng.randint(n)
            else:
                pos = np.nonzero(use_up)[0]
                i = int(pos[rng.randint(len(pos))])
                j = i + k
                if j < 0 or j >= n:
                    j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d = (rng.random() - 0.5) * 0.015
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i] = hi
        h[j] = hj
        c2 = _corr(h)
        mx2 = float(np.max(c2))
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            c = c2
            mx = mx2
            cn = mx * dx
            if cn < best_c - 1e-15:
                best_c = cn
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c

def _subgrad_step(h, c, n, top_k=15, lr=0.01):
    dx = 2.0 / n
    idx = np.argpartition(c, -top_k)[-top_k:]
    active = idx[c[idx] >= c.max() - 1e-9]
    if len(active) == 0:
        active = idx[:1]
    g = np.zeros(n)
    for k_idx in active:
        s = k_idx - (n - 1)
        if s > 0:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
        elif s < 0:
            ss = -s
            g[:n - ss] += (1.0 - h[ss:])
            g[ss:] -= h[:n - ss]
        else:
            g[:] += (1.0 - 2.0 * h)
    g /= len(active)
    g *= dx
    return g

def _sgd_momentum(h0, rng, deadline):
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    c = _corr(h)
    best_c = float(np.max(c)) * (2.0 / n)
    best_h = h.copy()
    v = np.zeros(n)
    t0 = time.time()
    total_t = max(1.0, deadline - t0)
    it = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total_t)
        lr = 0.02 * (0.2 ** frac) + 0.002
        g = _subgrad_step(h, c, n, top_k=20)
        v = 0.85 * v + 0.15 * g
        h_new = _proj(h - lr * v, S)
        c_new = _corr(h_new)
        cn = float(np.max(c_new)) * (2.0 / n)
        h = h_new
        c = c_new
        if cn < best_c:
            best_c = cn
            best_h = h.copy()
        it += 1
        if it % 200 == 0:
            h += rng.normal(0, 0.005, n)
            h = _repair(h, n)
            c = _corr(h)
    return h, best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
