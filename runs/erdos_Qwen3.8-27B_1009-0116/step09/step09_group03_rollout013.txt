        g_proj = g - g.mean()
        y_new = _proj(y - lr * g_proj, target)
        # Backtracking
        c_y = _eval(y)
        c_yn = _eval(y_new)
        step = lr
        for _ in range(20):
            if c_yn < c_y:
                break
            step *= 0.5
            y_new = _proj(y - step * g_proj, target)
            c_yn = _eval(y_new)
        v_new = y_new
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = v_new + (t/t_new - 1) * (v_new - v)
        y = np.clip(y, 0, 1)
        v = v_new
        t = t_new
        lr = step
        c = _eval(v)
        if c < best_c:
            best_c = c
            best_h = v.copy()
    return best_h, best_c
