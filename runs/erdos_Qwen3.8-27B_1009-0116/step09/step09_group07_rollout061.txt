        h *= target / s
        h = np.clip(h, 0, 1)

        if abs(s - target) < 1e-13:

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _cd_pass(h, rng, nbits=25):

    c = _corr(h)
    nm1 = n - 1
    cmax = 2 * n - 1
    order = np.arange(n)

            beta[k2lo:k2hi+1] -= h[j + lags2]
        
        hj = h[j]
        a = c - beta * hj

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.exp(s).sum() / (alpha * len(s))

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):

    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:

                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            oo = np.argsort(1.0 - h)
        else:
            oo = np.argsort(h)
        for idx in oo:
            if diff > 0:
                a = min(diff, 1.0 - float(h[idx]))
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, float(h[idx]))
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    nm1 = n - 1
    cmax = 2 * n - 1
    c = _corr(h)
    order = np.arange(n)
    rng.shuffle(order)
    for jj in order:
        j = int(jj)
        beta = np.zeros(cmax)
        k1hi = j + nm1
        if k1hi > cmax - 1:
            k1hi = cmax - 1
        ks = np.arange(j, k1hi + 1)
        L = ks - nm1
        beta[j:k1hi + 1] += 1.0 - h[j - L]
        k2lo = nm1 - j
        k2hi = cmax - 1 - j
        if k2lo < 0:
            k2lo = 0
        if k2lo <= k2hi:
            ks2 = np.arange(k2lo, k2hi + 1)
            L2 = ks2 - nm1
            beta[k2lo:k2hi + 1] -= h[j + L2]

def _sobj(x, al, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(al * (c[ix] - m), -500, 500)
    s -= s.max()
    return m + float(np.exp(s).sum()) / (al * len(s))

def _sgrad(x, al, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()

            g[M:] -= wk * x[:n - M]
    return g
