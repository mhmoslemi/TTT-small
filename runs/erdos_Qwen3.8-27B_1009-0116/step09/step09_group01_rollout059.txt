        if time.time() - t0 > t_limit:
            break
        if it > 0:

            y = h + ((tv-1.0)/tn)*(h - hp)
            tv = tn
            y = _proj(y, S)
        
        # Compute P and Nv
        H = np.fft.rfft(y, Nf)
        ac = np.fft.irfft(H * np.conj(H), Nf).real[:n]
        pref = np.empty(n+1); pref[0]=0; np.cumsum(y, out=pref[1:])
        tot = pref[n]
        P = tot - ac[0]
        Nv = np.zeros(n)
        Ps = np.zeros(n); Nvs = np.zeros(n)
        Ps[0] = P
        if n > 1:
            idx = np.arange(1, n)
            Ps[idx] = pref[n-idx] - ac[idx]
            Nvs[idx] = tot - pref[idx] - ac[idx]
            mx = max(Ps.max(), Nvs.max())
        else:
            mx = P
        
        # Weights
        wP = np.exp(np.clip(alpha*(Ps-mx), -400, 0))
        wN = np.exp(np.clip(alpha*(Nvs-mx), -400, 0))
        tot_w = wP.sum() + wN.sum()
        if tot_w < 1e-300:
            wP[:] = 1.0/n
            wN[:] = 0.0
            tot_w = 1.0
        wP /= tot_w
        wN /= tot_w
        
        # Gradient
        dx = 2.0/n
        u = np.zeros(n)
        if n > 1: u[1:] = wP[1:] + wN[1:]
        cp = np.zeros(n); cn = np.zeros(n)
        if n > 1:
            cp[1:] = np.cumsum(wP[1:])
            cn[1:] = np.cumsum(wN[1:])

        linN = cn
        q = np.zeros(n)
        if n > 1 and np.any(u[1:]):
            Hu = np.fft.rfft(y, Nf)
            R = np.fft.irfft(Hu * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
            L2 = np.fft.irfft(Hu * np.fft.rfft(u, Nf), Nf)[:n].real
            q = -(R + L2)
        g = dx * (wP[0]*(1.0-2.0*y) + linP + linN + q)
        
        # Step with backtracking
        step = 1.0/L
        hp = h
        for _ in range(10):
            xn = _proj(y - step*g, S)
            # Simple acceptance (skip full line search for speed)
            h = xn
            break
        
        if it % 50 == 0:
            c = _c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            it += 1
            continue

        h[i] = 0; h[j] = 1
        
        # Quick evaluation
        corr_new = _c5_all(h)
        c_new = float(corr_new.max())

            c = c_new
            corr = corr_new

                no_improve += 1

    return best_h, best_c

import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True).clip(0.0, 1.0)
    for _ in range(40):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        x[m] += d / cnt
        x.clip(0.0, 1.0, out=x)
    return x

def _resample(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    x0 = (np.arange(n) + 0.5) / n
    x1 = (np.arange(m) + 0.5) / m
    return np.interp(x1, x0, h)

def _to_binary(h):
    n = len(h)
    t = n // 2
    idx = np.argsort(-h)[:t]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb
