A better approach: **Exact 2-point optimization**. For a pair (i, j), find the optimal transfer δ that minimizes the max lags. This is a 1D problem (given i, j, find best δ). We can solve it by bisection or by computing the piecewise-linear structure of the max.

For a given pair (i,j) and transfer δ:
- h[i] → h[i] + δ, h[j] → h[j] - δ
- For each lag k, c_k changes linearly in δ
- The new max is max_k (c_k + a_k * δ) where a_k is the slope
- We want to find δ that minimizes this max

This is a "minimize a convex piecewise-linear function of one variable" problem! We can solve it in O(n) by finding the breakpoints.

Let me implement this. For each candidate pair (i,j):
1. Compute the slope vector a_k for all lags k
2. Find the δ that minimizes max_k (c_k + a_k * δ) subject to 0 ≤ h[i]+δ ≤ 1 and 0 ≤ h[j]-δ ≤ 1
3. Apply the optimal δ

This is exact (up to the choice of which pair to optimize) and can make significant improvements in few iterations.

The slope a_k for lag k when transferring δ from h[j] to h[i]:
- c_k changes by: δ * (dc_k/dh[i] - dc_k/dh[j])
- dc_k/dh[i] and dc_k/dh[j] are computed as above

So a_k = dc_k/dh[i] - dc_k/dh[j] (times dx, already included)

Then new c_k(δ) = c_k + a_k * δ

We want to minimize max_k (c_k + a_k * δ) over δ in [δ_min, δ_max] where:
δ_min = max(-h[i], h[j]-1) (so that h[i]+δ ≥ 0 and h[j]-δ ≤ 1)
δ_max = min(1-h[i], h[j]) (so that h[i]+δ ≤ 1 and h[j]-δ ≥ 0)

The function f(δ) = max_k (c_k + a_k * δ) is convex piecewise-linear. Its minimum is where the "upper envelope" is minimized, which occurs at a point where two or more lines cross, or at a boundary.

To find the minimum:
- Evaluate f at all breakpoints (where c_k + a_k*δ = c_l + a_l*δ for some k,l)
- The breakpoints are at δ = (c_l - c_k)/(a_k - a_l) for pairs where a_k ≠ a_l
- Also check the boundaries δ_min and δ_max

    n = len(h)

    # L ∈ [-j, n-1-j], intersect with [-(n-1), n-1]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(80):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _dc_all(h, j):
    n = len(h)
    dx = 2.0 / n
    slopes = np.zeros(2 * n - 1)
    L_min = max(-(n - 1), -j)
    L_max = min(n - 1, n - 1 - j)
    Ls = np.arange(L_min, L_max + 1)
    slopes[Ls + (n - 1)] += dx * (1.0 - h[j + Ls])
    L_min2 = max(-(n - 1), j - (n - 1))
    L_max2 = min(n - 1, j)
    Ls2 = np.arange(L_min2, L_max2 + 1)
    Ls2 = Ls2[Ls2 != 0]
    slopes[Ls2 + (n - 1)] -= dx * h[j - Ls2]
    return slopes

def _lag_grad_s(h, k_idx, i, j):
    n = len(h)
    dx = 2.0 / n
    L = k_idx - (n - 1)
    gi = 0.0
    gj = 0.0
    if L >= 0:
        if i < n - L:
            gi += dx * (1.0 - h[i + L])
        if i >= L:
            gi -= dx * h[i - L]
        if j < n - L:
            gj += dx * (1.0 - h[j + L])
        if j >= L:
            gj -= dx * h[j - L]
    else:
        M = -L
        if i >= M:
            gi += dx * (1.0 - h[i - M])
        if i <= n - 1 - M:
            gi -= dx * h[i + M]
        if j >= M:
            gj += dx * (1.0 - h[j - M])
        if j <= n - 1 - M:
            gj -= dx * h[j + M]
    return gi - gj

def _pobj(x, p, K):
    c = _corr(x)
    idx = np.argsort(c)[-K:]
    top = np.clip(c[idx], 1e-12, None)
    return (top ** p).sum() ** (1.0 / p)

def _pgrad(x, p, K):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    idx = np.argsort(c)[-K:]
    top = np.clip(c[idx], 1e-12, None)
    S = (top ** p).sum()
    factor = S ** (1.0 / p - 1.0)
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        w = factor * top[ii] ** (p - 1) * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += w * (1.0 - x[:n - L])
            g[:n - L] -= w * x[L:]
        else:
            M = -L
            g[:n - M] += w * (1.0 - x[M:])
            g[M:] -= w * x[:n - M]
    return g

def _slsqp_p(h, p, K, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_pobj, h, jac=_pgrad, args=(p, K), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _pg_p(h, p, K, steps):
    n = len(h)
    cur_c = _eval(h)
    step = 0.01
    for _ in range(steps):
        g = _pgrad(h, p, K)
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            break
        g /= gn
        for _t in range(8):
            h_new = _proj_sum(h - step * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, 0.5)
                break
            step *= 0.4
        else:
            step = min(max(step * 1.5, 0.001), 0.5)
    return _repair(h, n)

def _opt2(h, i, j, c, K):
    n = len(h)
    si = _dc_all(h, i)
    sj = _dc_all(h, j)
    sd = si - sj
    dmin = max(-h[i], h[j] - 1.0)
    dmax = min(1.0 - h[i], h[j])
    if dmin >= dmax - 1e-14:
        return 0.0
    top_idx = np.argsort(c)[-K:]
    candidates = [dmin, dmax, 0.0]
    for a in range(min(6, K)):
        for b in range(a + 1, min(6, K)):
            sa = sd[top_idx[a]]
            sb = sd[top_idx[b]]
            if abs(sa - sb) > 1e-14:
                d = (c[top_idx[b]] - c[top_idx[a]]) / (sa - sb)
                if dmin <= d <= dmax:
                    candidates.append(d)
    best_d = 0.0
    best_f = np.inf
    for d in candidates:
        d = float(np.clip(d, dmin, dmax))
        f = (c + sd * d).max()
        if f < best_f:
            best_f = f
            best_d = d
    return best_d

def _local2(h, n_iter, rng, K):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    c = _corr(best_h)
    for _ in range(n_iter):
        c = _corr(best_h)
        top_idx = np.argsort(c)[-K:]
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        if i == j:
            continue
        d = _opt2(best_h, i, j, c, K)
        if abs(d) > 1e-14:
            best_h[i] += d
            best_h[j] -= d
            best_h = _repair(best_h, n)
            c5n = _eval(best_h)
            if c5n < best_c:
                best_c = c5n
            else:
                best_h[i] -= d
                best_h[j] += d
                best_h = _repair(best_h, n)
    return best_h, best_c

def _local2_grad(h, n_iter, rng, K):

    best_h = h.copy()
    best_c = _eval(best_h)
    c = _corr(best_h)

        top_idx = np.argsort(c)[-K:]
        g = np.zeros(n)

            if L >= 0:
                g[L:] += (1.0 - best_h[:n - L])

            else:

                g[:n - M] += (1.0 - best_h[M:])

        neg = np.where(g < 0)[0]
        pos = np.where(g > 0)[0]
        if len(neg) == 0 or len(pos) == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n - 1)) % n
        else:
