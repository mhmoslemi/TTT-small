```python
    # This is O(n²) which for n=560 is fine.
    
    g = np.zeros(n)
    for m in range(n):
        wm = w[n - 1 + m] * dx  # weight for shift +m
        if m > 0:
            wn = w[n - 1 - m] * dx  # weight for shift -m
        else:
            wn = 0.0
        # shift +m: ∂/∂h[j] = (1-h[j-m])*1[j≥m] - h[j+m]*1[j≤n-1-m]
        if m == 0:
            g += wm * (1.0 - 2.0 * h)
        else:
            # (1-h[j-m]) for j ≥ m
            g[m:] += wm * (1.0 - h[:n-m])
            # -h[j+m] for j ≤ n-1-m
            g[:n-m] -= wm * h[m:]
            # shift -m: ∂/∂h[j] = (1-h[j+m])*1[j≤n-1-m] - h[j-m]*1[j≥m]
            g[:n-m] += wn * (1.0 - h[m:])
            g[m:] -= wn * h[:n-m]
    return g
```

In the previous code, `_sg` (subgradient of the smooth max) computes:

The key insight was:
- The overlap at lag k (positive) is: ∫h(x)(1-h(x+k))dx
  = ∫₀^{2-k} h(x)(1-h(x+k))dx (in continuous)
  In discrete: (1/dx) * sum_{i=0}^{n-1-k} h[i](1-h[i+k]) * dx... wait no.

  = (2/n) * sum_{i=0}^{n-1-k} h[i](1-h[i+k])

The gradient of this w.r.t. h[j]:
∂/∂h[j] sum_{i=0}^{n-1-k} h[i](1-h[i+k]) * dx
= dx * [(1-h[j-k]) * 1[j≥k, j-k≤n-1-k] - h[j+k] * 1[j≤n-1-k]]
= dx * [(1-h[j-k]) * 1[k≤j≤n-1] - h[j+k] * 1[j≤n-1-k]]

```python
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)
```

This computes the weighted sum of gradients across all lags using:
- `linP` and `linN`: the "linear" (cumulative sum) parts from the (1-h[j-k]) terms
- `q`: the "quadratic" (h * h) parts computed via FFT cross-correlation

def _loc(h0, rng, dl, s0=5e-5):
    """Lag-targeted local search with adaptive step and block moves."""
    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    # Identify active lags
    lags = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(lags)[-10:]
    
    it = 0
    while time.time() < dl:
        fr = (time.time() - t0) / max(1e-9, dl - t0)
        base = max(1e-12, s0 * (1e-5) ** fr)
        
        # Recompute active lags periodically
        if it % 500 == 0:
            c = _c5(h)
            if c < bc:
                bc = c
                bh = h.copy()
            lags = np.correlate(h, 1.0 - h, mode='full')
            top_k = np.argsort(lags)[-8:]
        
        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Single coordinate: pick position where gradient is most negative
            # (approximate by picking random position near high h values for lags in top_k)
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            # Direction: reduce h[i] if h[i] > h[j], else increase
            sgn = 1 if h[i] <= h[j] else -1
            avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
            if avail < 1e-14:
                continue
            d = sgn * min(avail, base * (0.5 + 2.0 * rng.random()))
            h[i] += d
            h[j] -= d
            cn = _c5(h)
            if cn < c - 1e-15:
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i] -= d
                h[j] += d
        elif r < 0.85:
            # Block move: swap a block of b elements
            b = int(rng.choice([2, 3, 4, 6, 8]))
            if n < 2 * b:
                continue
            i = int(rng.randint(0, n - b))
            j = int(rng.randint(0, n - b))
            if abs(i - j) < b:
                continue
            save_i = h[i:i+b].copy()
            save_j = h[j:j+b].copy()
            h[i:i+b] = save_j
            h[j:j+b] = save_i
            cn = _c5(h)
            if cn < c - 1e-15:
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i:i+b] = save_i
                h[j:j+b] = save_j
        else:
            # Gradient step
            g = _sg(h, 200.0)
            step = base * 0.1
            hn = _proj(h - step * g, n * 0.5)
            cn = _c5(hn)
            if cn < c - 1e-15:
                h = hn
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
        it += 1
    
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def _anneal(h0, rng, dl, t_init=0.3, t_final=0.001):
    """Simulated annealing with coordinate moves."""
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    it_total = max(1, int((dl) * 3000))  # estimate iterations
    for it in range(it_total):
        if time.time() - t0 > dl:
            break
        fr = it / it_total
        T = t_init * (t_final / t_init) ** fr
        # Pick two random positions
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        sgn = 1 if h[i] <= h[j] else -1
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        if avail < 1e-14:
            continue
        d = sgn * min(avail, T * rng.random())
