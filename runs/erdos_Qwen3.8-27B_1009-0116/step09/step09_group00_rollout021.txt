```python
def _active_grad_fast(h, corr, tol_frac=0.01):
    n = len(h)
    dx = 2.0/n
    mx = float(corr.max())
    tol = tol_frac * mx
    # Active lags: where corr[p] >= mx - tol
    active = corr >= mx - tol
    cnt = active.sum()
    if cnt == 0:
        return np.zeros(n)
    w = np.where(active, 1.0/cnt, 0.0)
    # grad[i] = dx * sum_p w[p] * [(1-h[i-(p-n+1)])*[0<=i-(p-n+1)<n] - h[i+(p-n+1)]*[0<=i+(p-n+1)<n]]
    # = dx * sum_p w[p] * dcorr_p[i] where dcorr_p[i] = ∂corr[p]/∂h[i]
    # Use matrix-free: for each active p, add w[p] * ∂corr[p]/∂h
    # ∂corr[p]/∂h[i] = (1-h[i-(p-n+1)]) - h[i+(p-n+1)] (with validity)
    # Let k = p-(n-1). Then:
    # ∂corr[n-1+k]/∂h[i] = (1-h[i-k])*[0≤i-k<n] - h[i+k]*[0≤i+k<n]
    
    grad = np.zeros(n)
    active_idx = np.nonzero(active)[0]
    for p in active_idx:
        k = p - (n - 1)
        # (1-h[i-k]) for valid i-k: i from k to k+n-1 (clipped to [0,n-1])
        i_lo = max(0, k)
        i_hi = min(n, k + n)
        if i_lo < i_hi:
            # (1-h[i-k]) for i in [i_lo, i_hi)
            j = np.arange(i_lo, i_hi) - k  # = i-k, ranges over valid h indices
            grad[i_lo:i_hi] += w[p] * (1.0 - h[j])
        # -h[i+k] for valid i+k: i from -k to n-1-k (clipped to [0,n-1])
        i_lo2 = max(0, -k)
        i_hi2 = min(n, n - k)
        if i_lo2 < i_hi2:
            j2 = np.arange(i_lo2, i_hi2) + k
            grad[i_lo2:i_hi2] -= w[p] * h[j2]
    return dx * grad
```
