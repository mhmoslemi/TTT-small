                d = d_sign * d_mag
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c_new = _eval(h)
                    if c_new < cur_c:
                        cur_c = c_new
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
                        break
    return h, best_c

def _twopoint_polish(h, n, t_limit, rng):
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    while time.time() < t_limit:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 5e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return h, best_c

def _basin_jump(h, n, rng, strength):
    target = n * 0.5
    h_new = h.copy()
    n_kick = max(10, int(0.1 * n))
    idx = rng.choice(n, size=n_kick, replace=False)
    h_new[idx] += strength * rng.randn(n_kick)
    return _repair(_project(h_new, target), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: PGD with low alpha (smooth landscape)
    t1 = t0 + min(120, budget_s * 0.1)
    h, c1 = _pgd_optimize(h, n, t1, 500, 30, 0.85, 0.01, rng, kick_every=400, kick_std=0.01)
    if c1 < best_c5:
        best_c5, best_h = c1, h.copy()

    # Phase 2: PGD with medium alpha
    t2 = t0 + min(350, budget_s * 0.3)
    h, c2 = _pgd_optimize(best_h, n, t2, 2000, 50, 0.88, 0.005, rng, kick_every=300, kick_std=0.015)
    if c2 < best_c5:
        best_c5, best_h = c2, h.copy()

    # Phase 3: Basin jumps + re-optimize
    t3 = t0 + min(600, budget_s * 0.55)
    for attempt in range(8):
        if _el() > t3:
            break
        h_j = _basin_jump(best_h, n, rng, 0.05 + 0.03 * attempt)
        t_sub = min(t3, t0 + _el() + 30)
        h_j, c_j = _pgd_optimize(h_j, n, t_sub, 5000, 60, 0.9, 0.003, rng, kick_every=500, kick_std=0.01)
        if c_j < best_c5:
            best_c5, best_h = c_j, h_j.copy()

    # Phase 4: PGD with high alpha (sharp max approximation)
    t4 = t0 + min(budget_s - 150, budget_s * 0.75)
    h, c4 = _pgd_optimize(best_h, n, t4, 10000, 80, 0.91, 0.002, rng, kick_every=600, kick_std=0.008)
    if c4 < best_c5:
        best_c5, best_h = c4, h.copy()

    # Phase 5: Coordinate descent
    t5 = t0 + min(budget_s - 60, budget_s * 0.9)
    h, c5 = _coord_descent(best_h, n, t5, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 6: Two-point polish
    h, c6 = _twopoint_polish(best_h, n, t0 + budget_s - 3, rng)
    if c6 < best_c5:
        best_c5, best_h = c6, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
