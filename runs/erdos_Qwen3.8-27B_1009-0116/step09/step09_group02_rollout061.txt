        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 20, size=3)) % n)
        d = (1e-9 + 3e-8 * rng.rand()) * rng.randn(3)
        d -= d.mean()
        hn = h.copy()
        hn[idx] += d
        if hn.min() < 0 or hn.max() > 1:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < cur - 1e-16:
            cur = cn
            h = hn
            if cn < best_c:
                best_c = cn
                best_h = hn.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    best_h = _repair(h0, n)
    best_c = _eval(best_h)

    # Phase 1: Gradient descent with exp-tilted active set
    if el() < budget_s - 500:
        h1, c1 = _grad_descent(best_h.copy(), n, 4000, 0.002, 25, 1000)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: SLSQP with increasing alpha
    for alpha in [50000, 500000, 5000000, 50000000]:
        if el() > budget_s - 350:
            break
        h2 = _slsqp_run(best_h.copy(), n, alpha, 40, 400)
        c2 = _eval(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Another gradient pass after SLSQP
    if el() < budget_s - 250:
        h3, c3 = _grad_descent(best_h.copy(), n, 2000, 0.001, 35, 5000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Push-pull targeted search
    if el() < budget_s - 150:
        rem = budget_s - 150 - el()
        h4, c4 = _pushpull(best_h, n, int(rem * 250), rng)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Swap search with moderate temperature
    if el() < budget_s - 80:
        rem = budget_s - 80 - el()
        h5, c5 = _swap_search(best_h, n, int(rem * 700), rng, 0.015)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine polish
    if el() < budget_s - 20:
        rem = budget_s - 20 - el()
        h6, c6 = _fine_polish(best_h, n, int(rem * 1200), rng)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Group polish
    if el() < budget_s - 5:
        rem = budget_s - 5 - el()
        h7, c7 = _group_polish(best_h, n, int(rem * 400), rng)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final ultra-fine
    if el() < budget_s - 1:
        rem = budget_s - 1 - el()
        h8, c8 = _fine_polish(best_h, n, int(rem * 2000), rng)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
