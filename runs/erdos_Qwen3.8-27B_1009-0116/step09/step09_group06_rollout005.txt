    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argpartition(c, -m)[-m:]
    vals = c[idx]
    mx = vals.max()
    s = alpha * (vals - mx)
    s = np.clip(s, -500, 500)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx

            g[M:] -= wk * x[:n-M]
    g += 2.0 * mu * (x.sum() - 0.5 * n)
    return g

def _so2(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argpartition(c, -m)[-m:]
    v = c[idx]
    mx = v.max()
    s = alpha * (v - mx)
    s = np.clip(s, -500, 500)
    return mx + np.log(np.exp(s).sum()) / alpha

def _sg2(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argpartition(c, -m)[-m:]
    v = c[idx]
    mx = v.max()
    s = alpha * (v - mx)
    s = np.clip(s, -500, 500)
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(m):
        if w[ii] < 1e-14: continue
        wk = w[ii] * dx
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _lbfgs(h, alpha, m, n, mu, mi):
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_so, h, jac=_sg, args=(alpha, m, n, mu),
                    method='L-BFGS-B', bounds=bnds,
                    options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

def _slsqp(h, alpha, m, n, mi):
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_so2, h, jac=_sg2, args=(alpha, m, n),
                    method='SLSQP', bounds=bnds,
                    constraints=[{'type':'eq','fun':_cf,'jac':_cj,'args':(n,)}],
                    options={'maxiter': mi, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _anti_grad(h, n):
    """Gradient of the single worst lag (subgradient of max)."""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    k = np.argmax(c)
    L = k - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n-L])
        g[:n-L] -= dx * h[L:]
    else:
        M = -L
        g[:n-M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n-M]
    return g, k

def _local(h, nt, rng, s0):
    n = len(h)
    bc = _eval(h)
    for i in range(nt):
        f = 1.0 - i / max(nt, 1)
        if rng.rand() < 0.3:
            # Anti-correlation move on worst lag
            g, _ = _anti_grad(h, n)
            st = s0 * f * (0.5 + rng.rand())
            hn = _repair(h - st * g, n)
        else:
            k = rng.randint(2, 12)
            idx = rng.choice(n, size=k, replace=False)
            d = s0 * f * (0.5 + rng.rand()) * rng.randn(k)
            d -= d.mean()
            hn = h.copy()
            hn[idx] += d
            if hn.min() < -0.05 or hn.max() > 1.05: continue
            hn = _repair(hn, n)
        c = _eval(hn)
        if c < bc:
            bc = c
            h = hn
    return h, bc

def _polish(h, ns, rng):
    n = len(h)
    cc = _eval(h)
    for _ in range(ns):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 3e-8 * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cc: cc = c
            else: h[i], h[j] = ni-d, nj+d
    return h, cc
