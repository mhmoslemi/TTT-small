def _min_convex_pl(cor, d, tmin, tmax):
    # Find t in [tmin, tmax] minimizing max_m (cor[m] + t*d[m])
    # All 2n-1 constraints
    n = len(cor)
    # At t, the max is achieved by argmax_m (cor[m] + t*d[m])
    # The function is convex. Minimum is where 0 is in subgradient.
    # Subgradient at t: {d[m] : m in argmax_m (cor[m]+t*d[m])}
    # 0 in subgradient iff min active slope ≤ 0 ≤ max active slope
    
    # Binary search approach:
    # f'(t-) = min{d[m] : cor[m]+t*d[m] = max}
    # f'(t+) = max{d[m] : cor[m]+t*d[m] = max}
    # f is decreasing for t < t*, increasing for t > t*
    
    lo, hi = tmin, tmax
    for _ in range(50):
        mid = 0.5*(lo+hi)
        vals = cor + mid*d
        mx = vals.max()
        active_min_slope = d[vals >= mx - 1e-12].min()
        active_max_slope = d[vals >= mx - 1e-12].max()
        if active_min_slope > 1e-12:
            # f increasing at mid, go left
            hi = mid
        elif active_max_slope < -1e-12:
            # f decreasing at mid, go right
            lo = mid
        else:
            # 0 in subgradient
            return mid
        if hi - lo < 1e-12:
            break
    return 0.5*(lo+hi)

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size

        if abs(d) < 1e-12: break

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    S = n * 0.5

        if s < 1e-15: return np.full(n, 0.5)

        mx = max(float(P.max()), float(Nv.max()))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _derive(h, i, j, n):
    d = np.zeros(2 * n - 1)
    d[n - 1] = 2.0 * (h[j] - h[i])
    k = np.arange(1, n)
    v1 = np.zeros(n - 1)
    m1 = k <= i
    if m1.any(): v1[m1] += 1.0 - h[i - k[m1]]
    m2 = k <= j
    if m2.any(): v1[m2] -= 1.0 - h[j - k[m2]]
    m3 = (i + k) < n
    if m3.any(): v1[m3] -= h[i + k[m3]]
    m4 = (j + k) < n
    if m4.any(): v1[m4] += h[j + k[m4]]
    d[n:] = v1
    v2 = np.zeros(n - 1)
    m5 = (i + k) < n
    if m5.any(): v2[m5] += 1.0 - h[i + k[m5]]
    m6 = (j + k) < n
    if m6.any(): v2[m6] -= 1.0 - h[j + k[m6]]
    m7 = k <= i
    if m7.any(): v2[m7] -= h[i - k[m7]]
    m8 = k <= j
    if m8.any(): v2[m8] += h[j - k[m8]]
    d[:n - 1] = v2[::-1]
    return d

def _min_convex(cor, d, tmin, tmax):
    if tmin >= tmax - 1e-15: return 0.0
    lo, hi = tmin, tmax
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = cor + mid * d
        mx = vals.max()
        mask = vals >= mx - 1e-11
        sm = d[mask].min(); sx = d[mask].max()
        if sm > 1e-13: hi = mid
        elif sx < -1e-13: lo = mid
        else: return mid
        if hi - lo < 1e-14: break
    return 0.5 * (lo + hi)

def _grad_soft(h, n, alpha):
    P, Nv, mx = _pn(h, n)
    dx = 2.0 / n
    wP = np.exp(np.clip(alpha * (P - mx), -400, 0))
    wN = np.zeros(n)
    if n > 1: wN[1:] = np.exp(np.clip(alpha * (Nv[1:] - mx), -400, 0))
    tot = wP.sum() + wN.sum()
    if tot < 1e-300: return np.zeros(n)
    wP /= tot; wN /= tot
    u = np.zeros(n)
    if n > 1: u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_next(2 * n - 1)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1 - 2 * h) + linP + linN + q)

def _soft_val(h, n, alpha):
    P, Nv, mx = _pn(h, n)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P - mx), -400, 0)).sum()
    if n > 1: e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / alpha)

def _fista(h0, alpha, max_t, n):
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    Lc = 10.0
    best_h = h.copy(); best_c = _c5(h)
    for it in range(500000):
        if time.time() - t0 > max_t: break
        if it:
            tn = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
            y = h + ((t - 1) / tn) * (h - hp)
            t = tn
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum() - S) > 0.01:
                y = _proj(y, S)
        g = _grad_soft(y, n, alpha)
        step = 1.0 / max(Lc, 1e-12)
        fy = _soft_val(y, n, alpha)
        ok = False
        for _ in range(14):
            xn = _proj(y - step * g, S)
            dv = xn - y
            fx = _soft_val(xn, n, alpha)
            if fx <= fy + float(dv @ g) + 0.5 * (1.0 / step) * float(dv @ dv) + 1e-10:
                ok = True; break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp = h; h = xn
            if it % 40 == 0:
                c = _c5(h)
                if c < best_c: best_c = c; best_h = h.copy()
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return best_h

def _pair_opt(h, rng, deadline, max_stall=400):
    n = len(h)
    S = n * 0.5
    cor = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(cor.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    order = np.argsort(h)
    low = order[:n // 2]
    high = order[n // 2:]
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 2000 == 0:
            order = np.argsort(h)
            low = order[:n // 2]
            high = order[n // 2:]
            # Re-project to fix drift
            if abs(h.sum() - S) > 1e-10:
                h = _proj(h, S)
                cor = np.correlate(h, 1.0 - h, mode='full')
        if rng.random() < 0.75:
            i = int(high[rng.randint(n // 2)])
            j = int(low[rng.randint(n // 2)])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j: continue
        tmin = max(-h[i], h[j] - 1.0)
        tmax = min(1.0 - h[i], h[j])
        if tmax - tmin < 1e-12: continue
        d = _derive(h, i, j, n)
        t_opt = _min_convex(cor, d, tmin, tmax)
        new_mx = (cor + t_opt * d).max()
        cur_mx = cor.max()
        improved = False
        if new_mx < cur_mx - 1e-14:
            h[i] += t_opt; h[j] -= t_opt
            cor += t_opt * d
            cn = cor.max() * (2.0 / n)
            if cn < best_c - 1e-15: best_c = cn; best_h = h.copy()
            improved = True
        else:
            d2 = -d
            t2 = _min_convex(cor, d2, -tmax, -tmin)
            nm2 = (cor + t2 * d2).max()
            if nm2 < cur_mx - 1e-14:
                h[j] += t2; h[i] -= t2
                cor += t2 * d2
                cn = cor.max() * (2.0 / n)
                if cn < best_c - 1e-15: best_c = cn; best_h = h.copy()
                improved = True
        if improved: stall = 0
        else:
            stall += 1
            if stall > max_stall:
                for _ in range(8):
                    a = int(rng.randint(n)); b = int(rng.randint(n))
                    if a != b:
                        dt = (rng.random() - 0.5) * 0.03
                        if 0 <= h[a] + dt <= 1 and 0 <= h[b] - dt <= 1:
                            h[a] += dt; h[b] -= dt
                            cn2 = np.correlate(h, 1.0 - h, mode='full')
                            if cn2.max() <= cor.max() + 5e-9:
                                cor = cn2
                            else:
                                h[a] -= dt; h[b] += dt
                stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    h0 = _repair(h0, n)
    best = h0.copy()
    best_c = _c5(best)
    for hh in (1.0 - h0.copy(), h0[::-1].copy()):
        c = _c5(hh)
        if c < best_c: best = hh; best_c = c
    # Phase 1: FISTA warmup
    end1 = t0 + min(70.0, 0.07 * budget_s)
    if time.time() < end1:
        hcur = best.copy()
        for a in [30, 200, 1000, 5000]:
            if time.time() >= end1 - 2: break
            slot = max(1.0, (end1 - time.time()) / 4.0)
            hf = _fista(hcur, a, slot, n)
            cf = _c5(hf)
            if cf < best_c: best = hf; best_c = cf
            hcur = hf
    # Phase 2: Main pair optimization
    end2 = t0 + budget_s - 80.0
    if time.time() < end2:
        best, best_c = _pair_opt(best, rng, end2)
    # Phase 3: FISTA polish
    end3 = t0 + budget_s - 40.0
    if time.time() < end3:
        hf = _fista(best, 50000, max(1.0, end3 - time.time()), n)
        cf = _c5(hf)
        if cf < best_c: best = hf; best_c = cf
    # Phase 4: Final pair polish
    end4 = t0 + budget_s - 3.0
    if time.time() < end4:
        best, best_c = _pair_opt(best, rng, end4, max_stall=200)
    # Final variants
    for hh in (1.0 - best, best[::-1]):
        h2 = _proj(hh.copy(), n * 0.5)
        c = _c5(h2)
        if c < best_c: best = h2; best_c = c
    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
