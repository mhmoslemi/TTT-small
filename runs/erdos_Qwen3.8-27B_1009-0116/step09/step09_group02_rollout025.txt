    d = tgt - h.sum()

        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-16:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def _grad(h, alpha, ma):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-ma:]
    s = np.clip(alpha * (c[idx] - c[idx].max()), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(ma):
        wk = w[ii]
        if wk < 1e-14:
            continue
        k = idx[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

        else:

    m = c.max()
    idx = np.argsort(c)[-ma:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-18})
