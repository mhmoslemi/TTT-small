import numpy as np
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = 0.5 * n
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        ind = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ind:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h


def _build_C(h, n):
    M = 2 * n - 1
    m = n - 1
    omh = 1.0 - h
    C = np.empty(M)
    C[m] = float(np.dot(h, omh))
    for L in range(1, n):
        ell = n - L
        C[m + L] = float(np.dot(h[:ell], omh[L:]))
        C[m - L] = float(np.dot(h[L:], omh[:ell]))
    return C


def _upd_single(p, d, h, C, n):
    m = n - 1
    C[m - p:m + n - p] += d * (1.0 - h)
    C[m + p - (n - 1):m + p + 1] -= d * h[::-1]
    h[p] += d


def _delta_grid(dm):
    if dm <= 1e-14:
        return np.empty(0)
    fr = np.geomspace(1e-8, 1.0, 10)
    fr = np.concatenate((fr, np.linspace(0.0, 1.0, 16)))
    fr = np.unique(fr[fr > 0.0])
    return dm * fr


def _try_pair(h, C, Cmax, a, b, n, m, A, B):
    if a == b:
        return False
    dmax = min(1.0 - h[a], h[b])
    if dmax <= 1e-13:
        return False
    dlimit = dmax * (1.0 - 1e-12)
    dvals = _delta_grid(dlimit)
    if dvals.size == 0:
        return False

    A[:] = 0.0
    B[:] = 0.0
    omh = 1.0 - h
    rev = h[::-1]
    A[m - a:m + n - a] += omh
    A[m + a - (n - 1):m + a + 1] -= rev
    A[m - b:m + n - b] -= omh
    A[m + b - (n - 1):m + b + 1] += rev
    B[m] = -2.0
    dl = abs(a - b)
    if dl:
        B[m + dl] += 1.0
        B[m - dl] += 1.0

    dv = dvals[:, None]
    F = (C[None, :] + A[None, :] * dv + B[None, :] * (dv * dv)).max(axis=1)
    imin = int(np.argmin(F))
    if F[imin] >= Cmax - 1e-10:
        return False

    bestF = F[imin]
    bestD = dvals[imin]
    lo = dvals[imin - 1] if imin > 0 else 0.0
    hi = dvals[imin + 1] if imin + 1 < dvals.size else dlimit
    if hi > lo:
        dref = np.linspace(lo, hi, 12)
        rd = dref[:, None]
        Fr = (C[None, :] + A[None, :] * rd + B[None, :] * (rd * rd)).max(axis=1)
        j = int(np.argmin(Fr))
        if Fr[j] < bestF:
            bestF = Fr[j]
            bestD = dref[j]

    if bestD <= 1e-14:
        return False

    _upd_single(a, bestD, h, C, n)
    _upd_single(b, -bestD, h, C, n)
    newmax = float(C.max())
    if newmax < Cmax - 1e-12:
        return True

    _upd_single(b, bestD, h, C, n)
    _upd_single(a, -bestD, h, C, n)
    return False


def _compute_g_top(h, C, n, ml, beta):
    M = 2 * n - 1
    m = n - 1
    cmax = float(C.max())
    if ml >= M:
        idx = np.arange(M)
        w = np.exp(np.clip(beta * (C - cmax), -50.0, 0.0))
    else:
        if ml <= 1:
            idx = np.array([int(np.argmax(C))])
            w = np.array([1.0])
        else:
            idx = np.argpartition(-C, ml)[:ml]
            w = np.exp(np.clip(beta * (C[idx] - cmax), -50.0, 0.0))
    sw = w.sum()
    if sw <= 0.0:
        w = np.ones_like(w, dtype=float)
        sw = w.sum()
    w = w / sw

    g = np.zeros(n)
    for ii in range(idx.size):
        wi = w[ii]
        if wi <= 0.0:
            continue
        L = int(idx[ii]) - m
        if L >= 0:
            g[:n - L] += wi * (1.0 - h[L:])
            g[L:] -= wi * h[:n - L]
        else:
            M2 = -L
            g[M2:] += wi * (1.0 - h[:n - M2])
            g[:n - M2] -= wi * h[M2:]
    return g


def _proj(y, n):

    y = np.clip(y, 0.0, 1.0)
    s = y.sum()
    if s < 1e-15:
        return np.full(n, 0.5)

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    return _repair(np.clip(y - 0.5 * (lo + hi), 0.0, 1.0), n)

    gm = g.mean()
    d = g - gm
    scale = float(np.std(d))
    if scale <= 1e-14:
        return False, h, C, Cmax
    bases = (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2)

        y = _proj(h - lr * d, n)
        Cn = _build_C(y, n)
        mx = float(Cn.max())
        if mx < Cmax - 1e-12:
            return True, y, Cn, mx
    return False, h, C, Cmax


def _gradient_pairs(g, h, K, ntop, rng):

    D[low[:, None] == high[None, :]] = -1e30
    flat = D.ravel()
    take = min(ntop, flat.size)
    idx = np.argpartition(flat, -take)[-take:]
    idx = idx[np.argsort(flat[idx])[::-1]]
    pairs = []
    for t in idx:
        r = int(t // K)
        c = int(t - r * K)
        a = int(low[r])
        b = int(high[c])
