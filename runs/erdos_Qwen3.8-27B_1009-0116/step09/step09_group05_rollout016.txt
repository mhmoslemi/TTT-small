def _subgrad_opt(h, nsteps, K):
    """Projected subgradient with top-K active set."""
    n = len(h)
    cur_c = _eval(h)
    best = h.copy()
    best_c = cur_c
    h_avg = h.copy()
    gamma = 0.1
    for t in range(nsteps):
        c = _corr(h)
        idx = np.argsort(c)[-K:]
        m = c[idx].max()
        # Weighted subgradient
        s = c[idx] - m
        w = np.exp(s / 0.01)
        w /= w.sum()
        g = np.zeros(n)
        sc = 2.0 / n
        for ii in range(K):
            wk = w[ii] * sc
            L = idx[ii] - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - h[M:])
                g[M:] -= wk * h[:n - M]
        # Remove mean (orthogonal to sum constraint)
        g -= g.mean()
        step = gamma / np.sqrt(t + 1)
        h_new = _proj(h - step * g, n * 0.5)
        h_avg = (t * h_avg + h_new) / (t + 1)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
        c_avg = _eval(_repair(h_avg, n))
        if c_avg < best_c:
            best_c = c_avg
            best = _repair(h_avg, n).copy()
    return best, best_c

def _pos_delta(h, j, k):
    """Compute delta[L] = dc(L)/dh[j] - dc(L)/dh[k] for all lags."""
    n = len(h)
    d = np.zeros(2 * n - 1)
    # For position j:
    for L in range(n):
        v = 0.0
        if L <= j:
            v += 1.0 - h[j - L]
        if L <= n - 1 - j:
            v -= h[j + L]
        d[L + n - 1] += v
    for M in range(1, n):
        v = 0.0
        if M <= n - 1 - j:
            v += 1.0 - h[j + M]
        if M <= j:
            v -= h[j - M]
        d[n - 1 - M] += v
    # For position k (subtract):
    for L in range(n):
        v = 0.0
        if L <= k:
            v += 1.0 - h[k - L]
        if L <= n - 1 - k:
            v -= h[k + L]
        d[L + n - 1] -= v
    for M in range(1, n):
        v = 0.0
        if M <= n - 1 - k:
            v += 1.0 - h[k + M]
        if M <= k:
            v -= h[k - M]
        d[n - 1 - M] -= v
    return d * (2.0 / n)
