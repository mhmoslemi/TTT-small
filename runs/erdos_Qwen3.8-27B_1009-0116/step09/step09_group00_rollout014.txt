            if h[i] + d < 0: d = -h[i]
            if h[i] + d > 1: d = 1.0 - h[i]
            if h[j] - d < 0: d = h[j]
            if h[j] - d > 1: d = h[j] - 1.0
            if abs(d) > 1e-13:
                h[i] += d; h[j] -= d
                c2 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                delta = c2 - mx
                if delta < -1e-14 or (delta > 0 and rng.random() < math.exp(-delta / max(T, 1e-15))):
                    mx = c2
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    cn = mx * (2.0 / n)
                    if cn < best_c:
                        best_c = cn; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h_init = _repair(np.asarray(initial_h_values, dtype=float).copy(), len(initial_h_values))
    else:
        h_init = np.full(280, 0.5)
    n0 = len(h_init)
    best = h_init.copy()
    best_c = _c5(best)

    for hh in (1.0 - h_init, h_init[::-1]):
        h2 = _proj(hh.copy(), n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c

    # Phase 1: FISTA sweep
    end1 = t0 + min(180.0, 0.16 * budget_s)
    if end1 > time.time():
        hcur = best.copy()
        for a in [40, 200, 800, 4000]:
            if time.time() >= end1 - 1:
                break
            tt = min(45.0, end1 - time.time())
            bh, bc = _fista(hcur, a, tt, 5.0)
            if bc < best_c:
                best = bh; best_c = bc
            hcur = bh

    # Phase 2: Subgradient refinement
    end2 = t0 + min(480.0, 0.44 * budget_s)
    if end2 > time.time():
        for mk, tl in [(3, 0.05), (2, 0.02)]:
            if time.time() >= end2 - 1:
                break
            tt = min(120.0, end2 - time.time())
            bh, bc = _subgrad_opt(best, end2, momentum=0.88, topk=mk, tol=tl)
            if bc < best_c:
                best = bh; best_c = bc

    # Phase 3: High-alpha FISTA
    end3 = t0 + min(680.0, 0.62 * budget_s)
    if end3 > time.time():
        for a in [20000, 100000, 500000]:
            if time.time() >= end3 - 1:
                break
            tt = min(65.0, end3 - time.time())
            bh, bc = _fista(best, a, tt, 5.0)
            if bc < best_c:
                best = bh; best_c = bc

    # Phase 4: Fine SA
    end4 = t0 + budget_s - 8.0
    if end4 > time.time():
        best, best_c = _sa_polish(best, rng, end4, max_shift=0.018)

    # Final variants
    for hh in (1.0 - best, best[::-1]):
        h2 = _proj(hh.copy(), len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
