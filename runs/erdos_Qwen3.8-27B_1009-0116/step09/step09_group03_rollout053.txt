            # dc/dh[j] for the first sum: 1 if j >= L, else 0
            # dc/dh[j] for the second sum (sum h[i]*h[i-L]): 
            #   j appears as i (if j>=L): contributes h[j-L]
            #   j appears as i-L (if j+L >= L and j+L <= n-1, i.e., 0<=j<=n-1-L): contributes h[j+L]
            
            # So dc/dh[j] = [j>=L] - [j>=L]*h[j-L] - [j<=n-1-L]*h[j+L]
            
            if L == 0:
                # dc/dh[j] = 1 - h[j] - h[j] = 1 - 2*h[j]
                g += wk * (1.0 - 2.0 * h)
            else:
                # j in [0, L-1]: dc/dh[j] = -h[j+L]
                g[:L] += wk * (-h[L:])
                if L <= n - 1 - L:
                    # j in [L, n-1-L]: dc/dh[j] = 1 - h[j-L] - h[j+L]
                    seg = slice(L, n - L)
                    g[seg] += wk * (1.0 - h[:n - 2 * L] - h[2 * L:n])
                # j in [n-L, n-1] (if n-L > n-1-L, i.e., always for L>0): dc/dh[j] = 1 - h[j-L]
                if n - L > n - 1 - L:
                    seg = slice(n - L, n)
                    g[seg] += wk * (1.0 - h[:L])
        else:
            M = -L  # M > 0
            # c = sum_{i=0}^{n-1-M} h[i]*(1-h[i+M])
            # dc/dh[j]:
            # First sum: 1 if j <= n-1-M
            # Second sum (sum h[i]*h[i+M]):
            #   j appears as i (if j <= n-1-M): contributes h[j+M]
            #   j appears as i+M (if j-M >= 0 and j-M <= n-1-M, i.e., M<=j<=n-1): contributes h[j-M]
            
            if M == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                # j in [0, n-1-M]: dc/dh[j] = 1 - h[j+M]
                g[:n - M] += wk * (1.0 - h[M:])
                # j in [n-M, n-1]: dc/dh[j] = -h[j-M]
                # j in [M, n-1-M]: also -h[j-M]
                # So for j in [M, n-1]: additional -h[j-M]
                # Total for j in [M, n-1-M]: 1 - h[j+M] - h[j-M]
                # Total for j in [n-M+1, n-1]: -h[j-M]
                # But I already added 1-h[j+M] for j in [0, n-1-M]
                # Now add -h[j-M] for j in [M, n-1]
                g[M:n] += wk * (-h[:n - M])
    
    g *= dx
    return obj, g

def _pg_optimize(h0, alpha, top_m, n_steps, lr, momentum, time_limit):
    t_start = time.time()
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    
    for it in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        obj, g = _smooth_obj_and_grad(h, alpha, top_m)
        
        # Backtracking line search on exact objective
        c_cur = _eval(h)
        step = lr
        found = False
        for _ in range(20):
            h_try = _proj_sum(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < c_cur - 1e-18:
                h = h_try
                v = momentum * v - step * g
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
                found = True
                break
