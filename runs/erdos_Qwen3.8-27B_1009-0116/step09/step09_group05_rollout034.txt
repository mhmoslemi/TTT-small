import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _corr_full(x)
        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):
        n = len(x)
        c = _corr_full(x)
        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):

                g[:n-L] -= wk * x[L:]
            else:

                g[:n-M] += wk * (1.0 - x[M:])

        best_h, best_c5 = _fine_sweep(best_h, n_trials, 20, rng, 12)

    # Phase 6: Ultra-fine final polish
    if time.time() - t0 < budget_s - 2:
        remaining = budget_s - 2 - (time.time() - t0)
        n_trials = min(int(remaining / 0.001), 200000)
        best_h, best_c5 = _fine_sweep(best_h, n_trials, 15, rng, 6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
