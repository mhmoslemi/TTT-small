def _rmsprop(h, alpha, ml, steps, lr0, decay=0.995):

        h = h_new
        lr *= 0.9998
    return _repair(h, n)

def _nest_grad(h, alpha, ml, steps, lr):

        y = y_new + beta * (y_new - h)

        h = y_new
        t = t_new
    return _repair(h, n)
