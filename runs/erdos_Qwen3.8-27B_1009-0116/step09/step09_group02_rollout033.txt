import time
import numpy as np


def _repair(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    n = len(h)
    s = float(h.sum())
    if s <= 1e-14:
        h.fill(target / max(1, n))
        return h
    for _ in range(80):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        if s <= 1e-14:
            h.fill(target / max(1, n))
            s = target
            break
    d = target - float(h.sum())
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0.0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0.0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h


def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _eval(h):
    return float(np.max(_corr_dx(h)))


def _grad_lag_into(g, h, L, dx):
    n = len(h)
    if L >= 0:
        m = n - L
        g[:m] += dx * (1.0 - h[L:n])
        g[L:n] -= dx * h[:m]
    else:
        M = -L
        m = n - M
        g[M:n] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[M:n]


def _grad_lags(h, idxs):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    off = n - 1
    for k in idxs:
        k = int(k)
        _grad_lag_into(g, h, off - k, dx)
    return g


def _pair_AB(lags, h, n, i, j):
    ml = len(lags)
    A = np.empty(ml, dtype=float)
    B = np.empty(ml, dtype=float)
    for r in range(ml):
        L = int(lags[r])
        at = 0.0
        bt = 0.0
        if L == 0:
            for p in (i, j):
                e = 1.0 if p == i else (-1.0 if p == j else 0.0)
                hp = h[p]
                at += e * (1.0 - 2.0 * hp)
                bt -= 1.0
        elif L > 0:
            m = n - L
            pv = []
            if i < m:
                pv.append(i)
            if i >= L:
                pv.append(i - L)
            if j < m:
                pv.append(j)
            if j >= L:
                pv.append(j - L)
            uniq = []
            for p in pv:
                if p not in uniq:
                    uniq.append(p)
            for p in uniq:
                q = p + L
                a = 1.0 if p == i else (-1.0 if p == j else 0.0)
                b = 1.0 if q == i else (-1.0 if q == j else 0.0)
                if a != 0.0 or b != 0.0:
                    hp = h[p]
                    hq = h[q]
                    at += a * (1.0 - hq) - b * hp
                    bt -= a * b
        else:
            M = -L
            m = n - M
            pv = []
            if i >= M:
                pv.append(i)
            if i + M < n:
                pv.append(i + M)
            if j >= M:
                pv.append(j)
            if j + M < n:
                pv.append(j + M)
            uniq = []
            for p in pv:
                if p not in uniq:
                    uniq.append(p)
            for p in uniq:
                q = p - M
                a = 1.0 if p == i else (-1.0 if p == j else 0.0)
                b = 1.0 if q == i else (-1.0 if q == j else 0.0)
                if a != 0.0 or b != 0.0:
                    hp = h[p]
                    hq = h[q]
                    at += a * (1.0 - hq) - b * hp
                    bt -= a * b
        A[r] = at
        B[r] = bt
    return A, B


def _search_pair(h, c, lags, i, j, grid):
    n = len(h)
    lo = float(max(-h[i], h[j] - 1.0))
    hi = float(min(1.0 - h[i], h[j]))
    lo += 1e-12
    hi -= 1e-12
    if hi <= lo + 1e-15:
        return None
    A, B = _pair_AB(lags, h, n, i, j)
    base = c[(n - 1) - lags]
    ds = np.linspace(lo, hi, int(max(3, grid)))
    if lo < 0.0 < hi:
        ds = np.concatenate((ds, np.array([0.0])))
    vals = base[None, :] + A[None, :] * ds[:, None] + B[None, :] * ds[:, None] * ds[:, None]
    am = vals.max(axis=1)
    bi = int(np.argmin(am))
    bd = float(ds[bi])
    ba = float(am[bi])
    span = hi - lo
    if span > 0.0:
        half = max(span * 0.2, 1e-7)
        rlo = max(lo, bd - half)
        rhi = min(hi, bd + half)
        if rhi - rlo > 1e-15:
            ds2 = np.linspace(rlo, rhi, 21)
            vals2 = base[None, :] + A[None, :] * ds2[:, None] + B[None, :] * ds2[:, None] * ds2[:, None]
            am2 = vals2.max(axis=1)
            bi2 = int(np.argmin(am2))
            if float(am2[bi2]) < ba:
                bd = float(ds2[bi2])
                ba = float(am2[bi2])
    return bd, ba


def _candidate_pairs(g, n, rng, p, nrand):
    cands = []
    seen = set()
    if g is not None and n >= 2:
        p = int(max(1, p))
        low = np.argsort(g)[:p]
        high = np.argsort(g)[-p:]
        for a in range(p):
            i = int(low[a])
            for b in range(p):
                j = int(high[b])
                if i != j and (i, j) not in seen:
                    seen.add((i, j))
                    cands.append((i, j))
    total = max(8, int(p * p + nrand))
    maxpairs = max(1, n * (n - 1))
    while len(cands) < total and len(cands) < maxpairs:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        if (i, j) in seen:
            continue
        seen.add((i, j))
        cands.append((i, j))
    return cands


def _add_noise(h, scale, rng):
    n = len(h)
    noise = rng.randn(n)
    if n > 9:
        noise = np.convolve(noise, np.ones(7.0) / 7.0, mode="same")
    noise -= noise.mean()
    st = float(noise.std())
    if st > 1e-15:
        noise *= scale / st
    h2 = h + noise
    return _repair(h2, n / 2.0)


def _resample(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    xf = np.linspace(0.0, 2.0, n, endpoint=False)
    xm = np.linspace(0.0, 2.0, m, endpoint=False)
    return np.interp(xm, xf, h, left=h[0], right=h[-1])


def _initial_guess(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = (
        0.5
        + 0.25 * np.sin(2.0 * np.pi * x)
        + 0.12 * np.cos(4.0 * np.pi * x)
        + 0.08 * rng.randn(n)
    )
    return _repair(h, n / 2.0)


def _pair_polish(h, stop, rng, K=100, grid=37, stall=350, noise_max=0.03):
