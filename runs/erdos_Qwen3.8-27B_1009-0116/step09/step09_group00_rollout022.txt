        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    elif m % n == 0:

    else:

        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _repair(out, m)

def _c5(h):

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(60):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _subgrad_step(h, top_n, alpha_sm):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    top_idx = np.argsort(corr)[-top_n:]
    top_vals = corr[top_idx]
    w = np.exp(alpha_sm * (top_vals - top_vals[-1]))
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(top_n); w[-1] = 1.0; wsum = 1.0
    w /= wsum
    grad = np.zeros(n)
    for idx, wk in zip(top_idx, w):
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            m = -k
            grad[m:] += wk * (1.0 - h[:n-m])
            grad[:n-m] -= wk * h[m:]
    return grad, mx, corr

def _phase_subgrad(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    mom = np.zeros(n)
    best_c = _c5(h)
    best_h = h.copy()
    lr = 0.015
    mu = 0.85
    t0 = time.time()
    stall = 0
    top_n = 6
    alpha_sm = 30.0
    while time.time() < deadline:
        grad, mx, _ = _subgrad_step(h, top_n, alpha_sm)
        c = mx * (2.0 / n)
        mom = mu * mom - lr * grad
        h_new = _proj(h + mom, n * 0.5)
        c_new = _c5(h_new)
        if c_new < c - 1e-12:
            h = h_new
            if c_new < best_c - 1e-10:
                best_c = c_new; best_h = h.copy(); stall = 0
            else:
                stall += 1
            lr = min(lr * 1.03, 0.08)
        else:
            h_try = _proj(h + 0.1 * mom, n * 0.5)
            c_try = _c5(h_try)
            if c_try < c:
                h = h_try
                if c_try < best_c:
                    best_c = c_try; best_h = h.copy(); stall = 0
            else:
                stall += 1
                lr *= 0.75
                mom *= 0.4
                if stall > 80:
                    h = best_h.copy(); mom = np.zeros(n); lr = 0.015; stall = 0
                    top_n = int(top_n * 1.5) + 1
                    if top_n > 20: top_n = 20
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        alpha_sm = 30.0 + 150.0 * frac
    return best_h, best_c

def _phase_coord(h0, rng, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    step = 0.012
    no_imp = 0
    while time.time() < deadline:
        k_act = int(np.argmax(corr)) - (n - 1)
        k_abs = abs(k_act)
        if 0 < k_abs < n:
            if k_act > 0:
                prod = h[:n-k_abs] * (1.0 - h[k_abs:])
            else:
                prod = h[k_abs:] * (1.0 - h[:n-k_abs])
            total = prod.sum()
            if total > 1e-8:
                p = prod / total
                i = int(rng.choice(len(p), p=p))
                j = i + k_abs
                d = step * (0.3 + 0.7 * rng.random())
                if k_act > 0:
                    h[i] = np.clip(h[i] - d, 0, 1)
                    h[j] = np.clip(h[j] + d, 0, 1)
                else:
                    h[j] = np.clip(h[j] - d, 0, 1)
                    h[i] = np.clip(h[i] + d, 0, 1)
            else:
                i, j = int(rng.randint(n)), int(rng.randint(n))
        elif k_abs == 0:
            mid = np.where((h > 0.3) & (h < 0.7))[0]
            if len(mid) > 0:
                i = int(mid[rng.randint(len(mid))])
                ext = np.where((h < 0.25) | (h > 0.75))[0]
                if len(ext) > 0:
                    j = int(ext[rng.randint(len(ext))])
                    d = step * (0.3 + 0.7 * rng.random())
                    if h[i] > 0.5:
                        h[i] = np.clip(h[i]-d, 0, 1); h[j] = np.clip(h[j]+d, 0, 1)
                    else:
                        h[i] = np.clip(h[i]+d, 0, 1); h[j] = np.clip(h[j]-d, 0, 1)
                    i = j = -1
                else:
                    i = j = -1
            else:
                i = j = -1
            if i == -1:
                i, j = int(rng.randint(n)), int(rng.randint(n))
        else:
            i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            i = (i + 1) % n
        if i == j:
            continue
        diff = n * 0.5 - h.sum()
        if abs(diff) > 1e-10:
            h += diff / n
            np.clip(h, 0, 1, out=h)
            diff = n * 0.5 - h.sum()
            if abs(diff) > 1e-10:
                msk = (h < 0.999) if diff > 0 else (h > 0.001)
                if msk.sum() > 0:
                    h[msk] += diff / msk.sum()
                    np.clip(h, 0, 1, out=h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx_new = float(corr.max())
        c_new = mx_new * dx
        if c_new < best_c - 1e-13:
            best_c = c_new; best_h = h.copy(); no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5000:
                h = best_h.copy()
                corr = np.correlate(h, 1.0 - h, mode='full')
                mx = float(corr.max())
                no_imp = 0
        mx = float(corr.max())
        frac = min(1.0, (time.time()-t0) / max(1e-9, deadline - t0))
        step = 0.012 * (1.0 - 0.85 * frac)
    return best_h, best_c

def _phase_sa(h0, rng, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    total_t = max(1.0, deadline - t0)
    while time.time() < deadline:
        frac = min(1.0, (time.time()-t0) / total_t)
        T = 0.008 * (5e-5) ** frac
        k_act = int(np.argmax(corr)) - (n-1)
        k_abs = abs(k_act)
        if 0 < k_abs < n:
            if k_act > 0:
                vi = np.where(h[:n-k_abs] > 0.002)[0]
                vj = np.where(h[k_abs:] < 0.998)[0] + k_abs
            else:
                vi = np.where(h[:n-k_abs] > 0.002)[0]
                vj = np.where(h[k_abs:] < 0.998)[0] + k_abs
            if len(vi) > 0 and len(vj) > 0:
                i = int(vi[rng.randint(len(vi))])
                j = int(vj[rng.randint(len(vj))])
            else:
                i, j = int(rng.randint(n)), int(rng.randint(n))
        else:
            i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        d = (rng.random() - 0.5) * 0.015
        h[i] = np.clip(h[i]+d, 0, 1)
        h[j] = np.clip(h[j]-d, 0, 1)
        corr2 = np.correlate(h, 1.0-h, mode='full')
        mx2 = float(corr2.max())
        delta = mx2 - mx
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            mx = mx2; corr = corr2
            c = mx * dx
            if c < best_c:
                best_c = c; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    # Try simple variants
    for hh in (1.0 - h0.copy(), h0[::-1].copy()):
        h2 = _repair(hh, n0)
        c = _c5(h2)
        if c < best_c:
            best = h2.copy(); best_c = c
    # Phase 1: Subgradient (38%)
    end1 = t0 + 0.38 * budget_s
    if time.time() < end1:
        best, best_c = _phase_subgrad(best, rng, end1)
    # Phase 2: Coordinate descent (24%)
    end2 = t0 + 0.62 * budget_s
    if time.time() < end2:
        best, best_c = _phase_coord(best, rng, end2)
    # Phase 3: SA polish (20%)
    end3 = t0 + 0.82 * budget_s
    if time.time() < end3:
        best, best_c = _phase_sa(best, rng, end3)
    # Phase 4: Second subgradient with lower lr (10%)
    end4 = t0 + 0.94 * budget_s
    if time.time() < end4:
        best, best_c = _phase_subgrad(best, rng, end4)
    # Phase 5: Final coordinate polish (6%)
    end5 = t0 + budget_s - 3.0
    if time.time() < end5:
        best, best_c = _phase_coord(best, rng, end5)
    # Final repair
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
