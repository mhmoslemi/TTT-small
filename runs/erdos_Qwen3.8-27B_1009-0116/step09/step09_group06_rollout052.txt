def _grad_lag(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        if L == 0:
            g[:] += (1.0 - h)
            g[:] -= 0.0
        else:

            g[:n - L] -= h[L:]

        else:
            g[:n - m] += (1.0 - h[m:])

    for _ in range(nblocks):
        start = rng.randint(0, max(1, n - bs + 1))
        idx = np.arange(start, min(start + bs, n))
        for _a in range(15):
            d = lr * rng.randn(len(idx))
            d -= d.mean()
            h_new = h.copy()
            h_new[idx] += d
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < c5cur - 1e-15:
                c5cur = c5n
                h = h_new
                lr = min(lr * 1.1, lr0 * 10)
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                break
            lr *= 0.65
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Quick SLSQP polish
    for alpha, ml, mi in [(5000, 30, 250), (30000, 40, 250)]:
        if _el() > 80:
            break
        h_new = _slsqp(best_h, alpha, ml, mi, n)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: Nesterov subgradient (key difference from previous)
    if _el() < budget_s - 500:
        configs = [(1, 0.03, 25), (5, 0.015, 40), (10, 0.008, 60), (20, 0.004, 80)]
        for t_init, lr_init, ml in configs:
            if _el() > budget_s - 450:
                break
            h_new, c_new = _nesterov(best_h.copy(), n, 400, t_init, lr_init, ml)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 3: Projected gradient with momentum
    if _el() < budget_s - 350:
        for alpha, ml, steps, lr0 in [(60, 0.004), (80, 0.003), (100, 0.002)]:
            if _el() > budget_s - 300:
                break
            h_new, c_new = _proj_grad(best_h.copy(), n, 800, alpha, ml, lr0)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 4: SLSQP at higher alphas
    alpha_seq = [(100000, 50, 350), (500000, 60, 300), (2000000, 70, 250),
                 (10000000, 80, 200), (100000000, 90, 150)]
    for alpha, ml, mi in alpha_seq:
        if _el() > budget_s - 200:
            break
        h_new = _slsqp(best_h, alpha, ml, mi, n)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 5: Block coordinate descent
    if _el() < budget_s - 120:
        rem = budget_s - 120 - _el()
        nb = int(rem / 0.08)
        h_new, c_new = _block_cd(best_h, n, nb, 12, rng, 4e-4)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 6: Fine local search with annealing
    if _el() < budget_s - 30:
        rem = budget_s - 30 - _el()
        ntries = int(rem / 0.0012)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 2.5e-5)

    # Phase 7: Ultra-fine 2-point polish
    if _el() < budget_s - 1:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = int((budget_s - 1 - _el()) / 0.0006)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 4)) % nc2
            d = 5e-9 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
