from scipy.optimize import linprog, minimize
import time

def _eval_correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval_c5(h):
    return float(np.max(_eval_correl(h)))

def _grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        L = lag
        if L < n:
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
    else:
        L = -lag
        if L < n:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        return h
    h_lp = res.x[:n]
    
    # Backtracking line search
    c_cur = _eval_c5(h)
    best = h
    best_c = c_cur
    for alpha in [1.0, 0.8, 0.5, 0.3, 0.1]:
        h_try = _repair(h + alpha * (h_lp - h), n)
        c_try = _eval_c5(h_try)
        if c_try < best_c:
            best_c = c_try
            best = h_try
    return best

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 40)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 40)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * x)
        elif L > 0:
            g[:n-L] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:n-L]
        else:
            M = -L
            g[M:] += wk * (1.0 - x[:n-M])
            g[:n-M] -= wk * x[M:]
    return g

def _slsqp(h, alpha, ml, maxiter, t_limit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cfun(x):
        return x.sum() - n * 0.5

        return np.ones(n)

                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
        return _repair(res.x, n)
    except:
        return h

from scipy.optimize import linprog, minimize
import time

def _eval_correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval_c5(h):
    return float(np.max(_eval_correl(h)))

def _grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0 and lag < n:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    elif lag < 0 and -lag < n:
        L = -lag
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lp_majorize(h, m, n, deadline):
    g = _eval_correl(h)
    if m >= len(g):
        top_idx = np.arange(len(g))
    else:
        top_idx = np.argpartition(-g, m)[:m]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for ii, ci in enumerate(top_idx):
        lag = ci - (n - 1)
        grad = _grad_lag(h, lag, n)
        A_ub[ii, :n] = grad
        A_ub[ii, n] = -1.0
        b_ub[ii] = -g[ci] + grad @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return h
    except Exception:
        return h
    h_lp = res.x[:n]

        if c_try < best_c:

            best = h_try

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 40)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 40)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * x)
        elif L > 0:
            g[:n - L] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:n - L]
        else:
            M = -L
            g[M:] += wk * (1.0 - x[:n - M])
            g[:n - M] -= wk * x[M:]
    return g

def _sum_con_fun(x, half):
    return x.sum() - half

def _sum_con_jac(x, half):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, ml, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n_iter, rng, deadline):

    c5_cur = _eval_c5(h)
    best_c = c5_cur

        h[i] += delta
        h[j] -= delta
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i], h[j] = hi, hj

        if c5n < c5_cur:
            c5_cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f, target_n):
    out = np.repeat(h, f)
    if len(out) > target_n:
        out = out[:target_n]
    elif len(out) < target_n:
        out = np.concatenate([out, np.zeros(target_n - len(out))])
    return _repair(out, target_n)
