        if s > target_sum:
            lo = mid
        else:

So: g[j] += (1-h[j+M]) for j ∈ [0, n-1-M]
    g[j] -= h[j-M] for j ∈ [M, n-1]

  g[:n-L] -= h[L:]

For L < 0 (M = -L):

  g[M:] -= h[:n-M]

    dx = 2.0 / n

def _project(x, target):

    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    h = _project(h, target)

    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

                     method='SLSQP', bounds=bnds, constraints=cons,

        return _repair(r.x, n)
    except:

    v_hat = np.zeros(n)

        g = _topk_grad(h, top_k)
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * g * g
        m_c = m_hat / (1.0 - beta1 ** t)
        v_c = v_hat / (1.0 - beta2 ** t)

        if c_new < bc:
            h = h_new
            if c_new < _eval(bh):
                bc = c_new
                bh = h.copy()
    return bh
