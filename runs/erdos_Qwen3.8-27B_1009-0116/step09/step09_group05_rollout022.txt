from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h = np.clip(h * target / s, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-11:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _pnorm_smooth(c, p):
    m = c.max()
    active = c > m - 1e-3
    if not active.any():
        return m, np.zeros_like(c)
    s = c[active]
    return m + np.sum((s - m + 1e-10)**p) ** (1.0/p) / max(1, int(active.sum())) ** (1.0/p), c

def _pnorm_grad(h, p, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    idx = np.argsort(c)[-topk:]
    active = c[idx] > m - 0.01
    if not active.any():
        idx = np.array([np.argmax(c)])
    g = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        val = c[k]
        w = (val - m + 1e-6) ** (p - 1) if p > 1 else 1.0
        w *= 2.0 / n
        if L >= 0:
            g[L:] += w * (1.0 - h[:n - L])
            g[:n - L] -= w * h[L:]
        else:
            M = -L
            g[:n - M] += w * (1.0 - h[M:])
            g[M:] -= w * h[:n - M]
    return g

def _proj_onto_simplex(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam_proj_grad(h, p, topk, steps, lr, t0_budget):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    cur_c = _eval(h)
    t_rel = 0
    for t in range(steps):
        if time.time() > t0_budget:
            break
        g = _pnorm_grad(h, p, topk)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** (t + 1))
        vh = v / (1 - b2 ** (t + 1))
        direction = mh / (np.sqrt(vh) + eps)
        h_new = _proj_onto_simplex(h - lr * direction, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
    return _repair(h, n), cur_c

def _coord_descent(h, n_iter, t_budget):
    n = len(h)
    c5 = _eval(h)
    c = _corr(h)
    best = h.copy()
    best_c = c5
    for it in range(n_iter):
        if time.time() > t_budget:
            break
        top_lags = np.argsort(c)[-15:]
        best_gain = 0.0
        best_ij = None
        best_frac = 0.0
        rng_sample = np.random.choice(n, size=min(200, n), replace=False)
        for _ in range(30):
            i = rng_sample[np.random.randint(len(rng_sample))]
            j = (i + 1 + np.random.randint(0, 10)) % n
            if i == j:
                continue
            for frac in [0.05, 0.15, 0.3, 0.5]:
                d = frac / n
                h[i] += d
                h[j] -= d
                if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                    h[i] -= d
                    h[j] += d
                    continue
                c5n = _eval(h)
                if c5n < c5 - best_gain:
                    best_gain = c5 - c5n
                    best_ij = (i, j)
                    best_frac = d
                    c5 = c5n
                    c = _corr(h)
                else:
                    h[i] -= d
                    h[j] += d
                    break
        if best_ij is None:
            # Try reverse: decrease i, increase j
            i = rng_sample[np.random.randint(len(rng_sample))]
            j = (i + 1 + np.random.randint(0, 10)) % n
            if i != j:
                d = 0.1 / n
                h[i] -= d
                h[j] += d
                if 0 <= h[i] and h[i] <= 1 and 0 <= h[j] and h[j] <= 1:
                    c5n = _eval(h)
                    if c5n < c5:
                        c5 = c5n
                    else:
                        h[i] += d
                        h[j] -= d
        if c5 < best_c:
            best_c = c5
            best = h.copy()
    return best, best_c

def _slsqp_opt(h, p, topk, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    half_n = n * 0.5

    def _obj(x):
        c = _corr(x)
        m = c.max()
        idx = np.argsort(c)[-topk:]
        s = c[idx] - m + 1e-10
        if p <= 1:
            return m + s.mean() * 0.1
        return m + (np.sum(s**p))**(1.0/p) / len(idx)**(1.0/p)

    def _grad(x):
        return _pnorm_grad(x, p, topk)

    def _sum_f(x):
        return x.sum() - half_n

    def _sum_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution (down to 140, optimize, upsample)
    for ratio in [4]:
        if time.time() > t0 + 40:
            break
        nc = n // ratio
        hc = _downsample(h, ratio)
        for p in [4, 16, 64]:
            if time.time() > t0 + 60:
                break
            hc = _slsqp_opt(hc, p, 20, 200, 1e-14)
        hu = _upsample(hc, ratio)
        cu = _eval(hu)
        if cu < best_c5:
            best_c5 = cu
            best_h = hu.copy()

    # Phase 2: SLSQP with increasing p at full res
    for p, topk, mit in [(8, 30, 400), (32, 50, 400), (128, 80, 400), (512, 120, 300), (2048, 160, 200)]:
        if time.time() > t0 + 320:
            break
        h_new = _slsqp_opt(best_h.copy(), p, topk, mit, 1e-15)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: Adam projected gradient
    for p, topk, steps, lr in [(16, 40, 800, 0.02), (64, 60, 600, 0.01), (256, 80, 400, 0.005)]:
        if time.time() > t0 + 550:
            break
        h_new, c_new = _adam_proj_grad(best_h.copy(), p, topk, steps, lr, t0 + 550)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 4: High-p SLSQP polish
    for p, topk, mit in [(4096, 200, 300), (16384, 250, 200)]:
        if time.time() > t0 + 750:
            break
        h_new = _slsqp_opt(best_h.copy(), p, topk, mit, 1e-16)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Coordinate descent (targeted)
    if time.time() < t_end - 120:
        n_iter = min(int((t_end - 120 - time.time()) / 0.005), 80000)
        best_h, best_c5 = _coord_descent(best_h, n_iter, t_end - 120)

    # Phase 6: Ultra-fine 2-point polish
    if time.time() < t_end - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        while time.time() < t_end - 1:
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 6)) % nc2
            d = 8e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
