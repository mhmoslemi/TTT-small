A better approach: instead of ternary search on max_k c[k], we can:

3. Find the minimum of f over [0, 1]

- Evaluate f at t=0 and t=1
- Find all intersection points of pairs of lines a_L*t + b_L = a_L'*t + b_L' that lie in [0,1]

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, n_top=40):
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    cmax = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    jlist = np.arange(n)
    rng.shuffle(jlist)
    
    for j in jlist:
        j = int(j)
        M_arr = np.arange(cmax) - nm1
        jp = j + M_arr
        jm = j - M_arr
        slope = np.zeros(cmax)
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        slope[mp] += (1.0 - h[jp[mp]])
        slope[mm] -= h[jm[mm]]
        base = c - slope * h[j]
        
        # Find top lags at t=0, t=1, t=h[j]
        f0 = base.copy()
        f1 = base + slope
        fc = base + slope * h[j]
        top0 = np.argsort(f0)[-n_top:]
        top1 = np.argsort(f1)[-n_top:]
        topc = np.argsort(fc)[-n_top:]
        union = np.unique(np.concatenate([top0, top1, topc]))
        
        best_t = h[j]
        best_f = float(np.max(fc))
        
        for tc in (0.0, 1.0):
            ft = float(np.max(base + slope * tc))
            if ft < best_f - 1e-15:
                best_f = ft
                best_t = tc
        
        ku = len(union)
        su = slope[union]
        bu = base[union]
        for a in range(ku):
            sa, ba = su[a], bu[a]
            for b in range(a+1, ku):
                sb, bb = su[b], bu[b]
                ds = sa - sb
                if abs(ds) < 1e-15:
                    continue
                t_int = (bb - ba) / ds
                if -1e-12 <= t_int <= 1.0 + 1e-12:
                    t_int = min(1.0, max(0.0, t_int))
                    ft = float(np.max(bu + su * t_int))
                    if ft < best_f - 1e-15:
                        best_f = ft
                        best_t = t_int
        
        if abs(best_t - h[j]) > 1e-12:
            h[j] = best_t
            c = base + slope * best_t

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _fine_search(h, n, rng, n_iter):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    c = _corr(best_h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        # Try small transfer from i to j
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5
```

```python
def _repair(h, n):

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
```
