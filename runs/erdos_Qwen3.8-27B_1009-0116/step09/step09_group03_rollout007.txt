            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            h_try = _repair(h_try, n)
            c5 = _eval(h_try)
            if c5 < best_c:
                best_h = h_try; best_c = c5
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick SLSQP with smooth top-m (warm-up)
    for alpha, m in [(50000, 10), (500000, 15), (5000000, 20), (50000000, 25)]:
        if _el() > 120:
            break
        def _obj_s(x):
            return _smooth_obj_grad(x, alpha, m)[0]
        def _jac_s(x):
            return _smooth_obj_grad(x, alpha, m)[1]
        h_new = _slsqp(best_h.copy(), _obj_s, _jac_s, n, 2000, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 2: Bundle method (exact active-sum optimization)
    if _el() < budget_s - 350:
        best_h, best_c5 = _bundle_opt(best_h, n, budget_s - 300)

    # Phase 3: More SLSQP with very high alpha (squeeze max further)
    if _el() < budget_s - 200:
        for alpha, m in [(100000000, 30), (1000000000, 40)]:
            if _el() > budget_s - 180:
                break
            def _obj_h(x):
                return _smooth_obj_grad(x, alpha, m)[0]
            def _jac_h(x):
                return _smooth_obj_grad(x, alpha, m)[1]
            h_new = _slsqp(best_h.copy(), _obj_h, _jac_h, n, 3000, 1e-18)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 4: Bundle method again (refined active set)
    if _el() < budget_s - 150:
        best_h, best_c5 = _bundle_opt(best_h, n, budget_s - 120)

    # Phase 5: Multi-restart local search with annealing
    if _el() < budget_s - 80:
        for attempt in range(4):
            if _el() > budget_s - 80 + 20:
                break
            noise = 2e-5 * (4 - attempt)
            h_pert = best_h + noise * rng.randn(n)
            h_pert = _repair(h_pert, n)
            c5p = _eval(h_pert)
            if c5p > best_c5 + 0.01:
                continue
            t_sub = min(budget_s - 80, time.time() + 25)
            h_new, c5n = _local_search(h_pert, n, t_sub, rng)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 6: Final bundle + polish
    if _el() < budget_s - 40:
        best_h, best_c5 = _bundle_opt(best_h, n, budget_s - 30)

    # Phase 7: Ultra-fine polish
    if _el() < budget_s - 2:
        best_h, best_c5 = _ultra_polish(best_h, n, budget_s - 1, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
