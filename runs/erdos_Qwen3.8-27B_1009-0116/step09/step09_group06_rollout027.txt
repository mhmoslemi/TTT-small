def _logsumexp_grad(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _grad_topm(x, n, top_m, sharpness):
    """Weighted gradient of top-m lags with exponential weighting."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_m:]
    w = np.exp(sharpness * (c[idx] - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _proj_grad(h, steps, step0, top_m, sharpness):
    """Projected gradient descent with top-m weighted gradient."""
    n = len(h)
    mu = 0.5
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _grad_topm(h, n, top_m, sharpness)
        v = mu * v + (1.0 - mu) * g
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v = v / vn
        taken = False
        st = step
        for _t in range(15):
            h_new = _proj(h - st * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(st * 1.2, step0 * 30.0)
                taken = True
                break
            st *= 0.4
        if not taken:
            step = min(step * 1.5, step0 * 30.0)
    return _repair(h, n), cur_c
