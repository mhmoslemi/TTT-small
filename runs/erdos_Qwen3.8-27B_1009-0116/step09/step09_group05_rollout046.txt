            if c5n < best_c5:
                best_c5 = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B warm-up (fast, ~80s)
    for alpha, ml in [(500, 40), (3000, 70), (15000, 120), (80000, 200)]:
        if elapsed() > 75:
            break
        res = _lbfgsb(best_h, alpha, ml, 500, 5000)
        c5n = _eval(res)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = res

    # Phase 2: Progressive high-alpha L-BFGS-B (~300s)
    alpha_seq = [(300000, 300), (1000000, 400), (5000000, 500),
                 (20000000, 700), (100000000, 900)]
    for alpha, ml in alpha_seq:
        if elapsed() > 380:
            break
        res = _lbfgsb(best_h, alpha, ml, 2000, 8000)
        c5n = _eval(res)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = res

    # Phase 3: Subspace optimization (~200s)
    if elapsed() < budget_s - 350:
        K = 16
        r1, c1 = _subspace_opt(best_h.copy(), 500000, 300, K, 30, rng)
        if c1 < best_c5:
            best_c5 = c1
            best_h = r1
        r2, c2 = _subspace_opt(best_h.copy(), 2000000, 500, 24, 20, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = r2

    # Phase 4: More L-BFGS-B at very high alpha (~120s)
    if elapsed() < budget_s - 200:
        for alpha, ml in [(500000000, 1000), (2000000000, 1500)]:
            if elapsed() > budget_s - 180:
                break
            res = _lbfgsb(best_h, alpha, ml, 5000, 3000)
            c5n = _eval(res)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = res

    # Phase 5: Gradient-guided 2-point search (~100s)
    if elapsed() < budget_s - 80:
        nit = min(int((budget_s - 80 - elapsed()) / 0.015), 80000)
        best_h, best_c5 = _twopoint(best_h, nit, 1000000, 800, rng, 5e-5, 2e-3)

    # Phase 6: Random pair perturbations (~60s)
    if elapsed() < budget_s - 30:
        nit = min(int((budget_s - 30 - elapsed()) / 0.006), 200000)
        best_h, best_c5 = _random_pair(best_h, nit, rng, 1e-5)

    # Phase 7: Ultra-fine polish
    if elapsed() < budget_s - 3:
        c5c = _eval(best_h)
        nit = min(int((budget_s - 3 - elapsed()) / 0.003), 300000)
        for _ in range(nit):
            if time.time() - t0 > budget_s - 1:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            d = 8e-7 * rng.randn()
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                c5n = _eval(best_h)
                if c5n < best_c5:
                    best_c5 = c5n
                    c5c = c5n
                else:
                    best_h[i], best_h[j] = ni - d, nj + d
                    c5c = best_c5

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
