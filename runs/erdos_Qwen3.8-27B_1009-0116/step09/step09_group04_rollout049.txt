        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argsort(c)[-ml:]
        s = alpha * (c[idx] - m)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def _sf(x):
        return x.sum() - 0.5 * n

        return np.ones(n)

        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()

        s -= s.max()

    bstart = rng.randint(0, n - blen)
    h2[bstart:bstart+blen] = 1.0 - h2[bstart:bstart+blen]

    """Swap two random blocks of the same size."""
    h2 = h.copy()
    blen = rng.randint(4, 16)
    i = rng.randint(0, n - blen)
    j = rng.randint(0, n - blen)
    if abs(i - j) < blen:
        return h2
    h2[i:i+blen], h2[j:j+blen] = h2[j:j+blen].copy(), h2[i:i+blen].copy()
    return _repair(h2, n)

    def grad(x):
        return _soft_grad(x, alpha, ml, dx)
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'maxfun': maxiter * 4})
        return _repair(r.x, n)
    except:
        return h

def _perturb_block(h, rng, n):
    h2 = h.copy()
    blen = rng.randint(6, 35)
    bstart = rng.randint(0, max(1, n - blen))
    h2[bstart:bstart + blen] = 1.0 - h2[bstart:bstart + blen]
    return _repair(h2, n)

def _perturb_swap(h, rng, n):
    h2 = h.copy()
    blen = rng.randint(5, 20)
    i = rng.randint(0, max(1, n - blen))
    j = rng.randint(0, max(1, n - blen))
    if abs(i - j) < blen:
        return h2
    h2[i:i+blen], h2[j:j+blen] = h2[j:j+blen].copy(), h2[i:i+blen].copy()
    return _repair(h2, n)

def _perturb_coarse(h, rng, n):
    cm = 4
    nc = n // cm
    hc = np.zeros(nc)
    for i in range(nc):
        hc[i] = h[i*cm:(i+1)*cm].mean()
    for _ in range(rng.randint(2, 8)):
        idx = rng.randint(nc)
        hc[idx] = 1.0 - hc[idx]
    h2 = np.repeat(hc, cm)[:n]
    h2 += 0.04 * rng.randn(n)
    return _repair(h2, n)

def _perturb_noise(h, rng, n):
    h2 = h + 0.15 * rng.randn(n)
    return _repair(h2, n)

def _binary_round(h):
    n = len(h)
    hr = np.round(h)
    diff = int(round(n / 2.0)) - int(hr.sum())
    if diff > 0:
        idx = np.argsort(h - hr)[::-1][:diff]
        for i in idx:
            hr[i] = 1.0
    elif diff < 0:
        idx = np.argsort(h - hr)[:(-diff)]
        for i in idx:
            hr[i] = 0.0
    return hr

def _binary_il(h, rng, n, niter):
    hb = _binary_round(h)
    cb = _eval(hb)
    best_b, best_cb = hb.copy(), cb
    temp = 0.02
    for it in range(niter):
        temp *= 0.99985
        k = rng.randint(1, 5)
        idx = rng.randint(0, n, size=k)
        hb2 = hb.copy()
        for i in idx:
            hb2[i] = 1.0 - hb2[i]
        _repair(hb2, n)
        cb2 = _eval(hb2)
        if cb2 < cb or (temp > 1e-8 and rng.rand() < np.exp(-(cb2 - cb) / temp)):
            hb, cb = hb2, cb2
            if cb2 < best_cb:
                best_cb, best_b = cb2, hb2.copy()
    return best_b, best_cb

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, ml, steps, s0, mu):
    n = len(h)
    dx = 2.0 / n
    v = np.zeros(n)
    step = s0
    cur = _eval(h)
    for _ in range(steps):
        g = _soft_grad(h, alpha, ml, dx)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(10):
            hn = _proj(h - step * v, 0.5 * n)
            cn = _eval(hn)
            if cn < cur:
                h, cur, taken = hn, cn, True
                step = min(step * 1.08, s0 * 20.0)
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, s0 * 20.0)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)

    def _el():
        return time.time() - t0

    # Phase 1: Initial SLSQP + L-BFGS-B from start
    for alpha, ml in [(3000, 25), (30000, 40), (300000, 70)]:
        if _el() > 80:
            break
        hn = _slsqp(best_h, alpha, ml, 250)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn
        if _el() > 120:
            break
        hn = _lbfgs(best_h, alpha, ml, 300)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 2: ILS with diverse perturbations + SLSQP polish
    perturb_fns = [_perturb_block, _perturb_swap, _perturb_coarse, _perturb_noise]
    while _el() < budget_s - 250:
        pfn = perturb_fns[rng.randint(len(perturb_fns))]
        hn = pfn(best_h, rng, n)
        # Quick SLSQP polish
        hn = _slsqp(hn, 30000, 45, 120)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn
            # Deepen from new best
            if _el() < budget_s - 260:
                hn2 = _slsqp(best_h, 300000, 70, 200)
                cn2 = _eval(hn2)
                if cn2 < best_c5:
                    best_c5, best_h = cn2, hn2
        # Occasionally try L-BFGS from perturbed point
        if rng.rand() < 0.3 and _el() < budget_s - 260:
            hn3 = _lbfgs(hn, 50000, 55, 250)
            cn3 = _eval(hn3)
            if cn3 < best_c5:
                best_c5, best_h = cn3, hn3

    # Phase 3: Binary ILS to find structural improvements
    if _el() < budget_s - 120:
        niter = min(int((budget_s - 120 - _el()) / 0.03), 80000)
        hb, cb = _binary_il(best_h, rng, n, niter)
        if cb < best_c5:
            # Polish the binary solution back to continuous
            hn = _slsqp(hb, 50000, 60, 300)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn
            elif cb < best_c5:
                best_c5, best_h = cb, hb

    # Phase 4: Projected gradient with momentum
    if _el() < budget_s - 100:
        for a, ml, st, s0, mu in [(50000, 50, 400, 0.005, 0.88),
                                   (500000, 80, 300, 0.002, 0.92),
                                   (5000000, 100, 250, 0.001, 0.94)]:
            if _el() > budget_s - 105:
                break
            hn = _proj_grad(best_h.copy(), a, ml, st, s0, mu)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 5: High-alpha SLSQP for final tightening
    if _el() < budget_s - 60:
        for alpha, ml in [(2000000, 90), (20000000, 120), (200000000, 150)]:
            if _el() > budget_s - 65:
                break
            hn = _slsqp(best_h, alpha, ml, 350)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 3:
        h2 = best_h.copy()
        c2 = best_c5
        npol = min(int((budget_s - 3 - _el()) / 0.0008), 500000)
        for _ in range(npol):
            if _el() > budget_s - 3:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
            d = 2.5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                cn = _eval(h2)
                if cn < c2:
                    c2 = cn
                    if cn < best_c5:
                        best_c5, best_h = cn, h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
