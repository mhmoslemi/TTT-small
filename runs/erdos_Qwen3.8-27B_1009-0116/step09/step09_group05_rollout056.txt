def _swap_search(h, n_iter, t0, tlimit, focus_lags=12):
    """Targeted pairwise swap: find pairs that reduce max correlation."""
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        if time.time() - t0 > tlimit:
            break
        c = _corr(h)
        m = c.max()
        # Top contributing lags
        top = np.argsort(c)[-focus_lags:]
        # For the top lag, find the pair with max contribution
        # c[k] = sum_i h[i+L](1-h[i]) where L = k-(n-1)
        # Gradient at k: for position j, ∂c/∂h[j] = (1-h[j-L]) - h[j+L] (roughly)
        # We want to find i where moving mass helps most
        g = np.zeros(n)
        for k in top:
            L = k - (n - 1)
            if L >= 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                M = -L
                g[:n - M] += (1.0 - h[M:])
                g[M:] -= h[:n - M]
        g -= g.mean()
        # Best pair: max gradient position (decrease) and min gradient position (increase)
        i_dec = int(np.argmax(g))  # decrease h here
        i_inc = int(np.argmin(g))  # increase h here
        if i_dec == i_inc:
            continue
        # Try different step sizes
        base = min(h[i_dec], 1.0 - h[i_inc])
        best_d = 0
        best_c5 = c5c
        for frac in [0.02, 0.05, 0.1, 0.2, 0.4, 0.7]:
            d = frac * base
            h[i_dec] -= d
            h[i_inc] += d
            if h[i_dec] < 0 or h[i_inc] > 1:
                h[i_dec] += d
                h[i_inc] -= d
                continue
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_d = d
            else:
                h[i_dec] += d
                h[i_inc] -= d
                break
        if best_d > 0:
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c
                best_h = h.copy()
        else:
            h[i_dec] += best_d
            h[i_inc] -= best_d
    return best_h, best_c

def _random_polish(h, n_iter, rng, t0, tlimit):

                g[M:] -= w[ii] * x[:n - M]
        return g * (2.0 / n)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_gfun, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _fine_polish(h, n_iter, rng, t0, tlim):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() - t0 > tlim:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 1.5e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                h = _repair(h, n)
    return best_h, best_c

def _coarse_cycle(h, nc, t0, tlim):
    n = len(h)
    f = n // nc
    m = n // f
    hc = h[:m * f].reshape(m, f).mean(axis=1)
    hc = _repair(hc, m)
    hc, _ = _adam_grad(hc, 2000, 12, 300, 0.02, t0, tlim)
    hc = _slsqp_smooth(hc, 8000, 15, 400, 1e-14)
    out = np.repeat(hc, f)
    return _repair(out, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam projected gradient with increasing alpha
    for alpha, ml, steps, s0 in [(1000, 12, 700, 0.012), (5000, 25, 500, 0.006),
                                   (25000, 45, 350, 0.003), (100000, 65, 250, 0.0015)]:
        if time.time() - t0 > 180:
            break
        h_new, _ = _adam_grad(best_h.copy(), alpha, ml, steps, s0, t0, 180)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: SLSQP smooth-max (moderate alpha)
    if time.time() - t0 < budget_s - 250:
        for alpha in [8000, 60000, 400000]:
            if time.time() - t0 > budget_s - 230:
                break
            ml = min(50, max(12, int(12 * np.log10(alpha / 8000) + 18)))
            h_new = _slsqp_smooth(best_h, alpha, ml, 600, 1e-14)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: Coarse refinement (downsample, optimize, upsample)
    if time.time() - t0 < budget_s - 200:
        for nc in [140, 200, 280]:
            if time.time() - t0 > budget_s - 180:
                break
            h_up = _coarse_cycle(best_h, nc, t0, budget_s - 180)
            c5u = _eval(h_up)
            if c5u < best_c5:
                best_c5 = c5u
                best_h = h_up.copy()

    # Phase 4: Another Adam pass after coarse
    if time.time() - t0 < budget_s - 100:
        h_new, _ = _adam_grad(best_h.copy(), 50000, 50, 300, 0.002, t0, budget_s - 100)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Targeted swap search (focus on top-contributing pairs per lag)
    if time.time() - t0 < budget_s - 50:
        niter = min(int((budget_s - 50 - (time.time() - t0)) / 0.004), 150000)
        best_h, best_c5 = _targeted_swap(best_h, niter, t0, budget_s - 50, focus=8)

    # Phase 6: Ultra-fine random polish
    if time.time() - t0 < budget_s - 3:
        remaining = budget_s - 3 - (time.time() - t0)
        ntries = min(int(remaining / 0.0012), 600000)
        best_h, best_c5 = _fine_polish(best_h, ntries, rng, t0, budget_s - 1)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
