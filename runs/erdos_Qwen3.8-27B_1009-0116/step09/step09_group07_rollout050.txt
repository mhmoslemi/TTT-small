    best_h = h.copy(); best_c5 = _eval(best_h); c5c = best_c5
    for _ in range(n_iter):
        i = rng.randint(n); j = (i + 1 + rng.randint(0, max_dist)) % n
        d = step * rng.randn()
        h[i] += d; h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Pure CD (35% budget)
    t_end1 = t0 + budget_s * 0.35
    stall = 0
    prev_c5 = best_c5
    while time.time() < t_end1:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5; best_h = h.copy(); stall = 0
        else:
            stall += 1
            if stall > 120:
                break
    h = best_h.copy()

    # Phase 2: CD with noise restarts (25% budget)
    t_end2 = t0 + budget_s * 0.60
    while time.time() < t_end2:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if (int((time.time() - t0) * 20) % 37) == 0:
            h[ rng.randint(0, n, size=30) ] += 0.008 * rng.randn(30)
            h = _repair(h, n)
        if (int((time.time() - t0) * 10) % 53) == 0:
            h = best_h + 0.003 * rng.randn(n)
            h = _repair(h, n)
    h = best_h.copy()

    # Phase 3: SLSQP refinement (20% budget)
    t_end3 = t0 + budget_s * 0.80
    for alpha in [2000, 20000, 200000, 2000000, 8000000]:
        remaining = t_end3 - time.time()
        if remaining < 40:
            break
        mi = min(1500, max(300, int(remaining * 8)))
        h_s = _run_slsqp(best_h.copy(), alpha, 50, mi)
        c5n = _eval(h_s)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_s
    h = best_h.copy()
    h = _cd_pass(h, rng)
    h = _repair(h, n)
    c5 = _eval(h)
    if c5 < best_c5:
        best_c5 = c5; best_h = h.copy()

    # Phase 4: Simulated annealing (10% budget)
    t_end4 = t0 + budget_s * 0.93
    if time.time() < t_end4:
        n_itt = int((t_end4 - time.time()) / 0.004)
        n_itt = min(n_itt, 300000)
        best_h, best_c5 = _anneal(best_h.copy(), n, rng, n_itt, 5e-4, 1e-8)

    # Phase 5: Fine local search (remaining)
    if time.time() < budget_s - 3:
        n_itt = int((budget_s - 4 - time.time()) / 0.003)
        n_itt = min(n_itt, 250000)
        best_h, best_c5 = _local_search(best_h.copy(), n, rng, n_itt, 3e-6, 20)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
