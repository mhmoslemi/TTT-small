import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_step(h, alpha, m_lags, step, n):
    """One subgradient step with projection"""
    g = _smooth_grad(h, alpha, m_lags)
    g_c = g - g.mean()  # center to respect sum constraint
    h_new = _proj(h - step * g_c, n * 0.5)
    return h_new, _eval(h_new)

def _subgrad_loop(h, alpha, m_lags, n_iter, step0, n):
    """Subgradient loop with adaptive step size"""
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    step = step0
    for it in range(n_iter):
        h_new, c_new = _subgrad_step(h, alpha, m_lags, step, n)
        if c_new < cur:
            cur = c_new
            h = h_new
            step = min(step * 1.1, step0 * 20)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            step *= 0.5
            if step < step0 * 1e-3:
                step = step0 * 1e-3
    return best_h, best_c

def _twopoint(h, n_iter, rng, n, mode='grad'):
    """2-point hill climbing"""
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if mode == 'grad':
            g = _smooth_grad(h, 1000000, 80)
            g_c = g - g.mean()
            i = int(np.argmax(g_c))
            j = int(np.argmin(g_c))
            if i == j:
                continue
            d = 0.01 * (g_c[i] - g_c[j])
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = 1e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _multi_start_slsqp(h, n, alpha, m_lags, maxiter, rng, n_starts):

        if c < best_c:

            best_h = res.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _rmsprop(h, alpha, m_lags, steps, step0, n):
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(steps):
        g = _smooth_grad(h, alpha, m_lags)
        g_c = g - g.mean()
        v = 0.9 * v + 0.1 * g_c
        sq = 0.95 * sq + 0.05 * g_c * g_c
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(16):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.15, step0 * 40.0)
                taken = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.5, step0 * 40.0)
    return best_h, best_c

def _grad_2pt(h, n_iter, rng, n, alpha, m_lags):
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        g = _smooth_grad(h, alpha, m_lags)
        g_c = g - g.mean()
        i = int(np.argmax(g_c))
        j = int(np.argmin(g_c))
        if i == j:
            continue
        d = 0.008 * (g_c[i] - g_c[j])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _rand_2pt(h, n_iter, rng, n, step_scale):

def _multi_start(h, n, alpha, m_lags, maxiter, rng, n_starts):
    best_h = h.copy()
    best_c = _eval(best_h)
    for _ in range(n_starts):

        c = _eval(res)
        if c < best_c:
            best_c = c
            best_h = res.copy()
    return best_h, best_c
