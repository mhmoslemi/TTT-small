import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(120):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _slsqp_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _do_slqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_sum(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)
