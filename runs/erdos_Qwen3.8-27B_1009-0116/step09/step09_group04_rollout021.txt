import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h = np.clip(h * (target / s), 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), h[idx] if diff < 0 else 1.0 - h[idx])
            h[idx] += a if diff > 0 else -a
            diff += a if diff < 0 else -a
            if abs(diff) < 1e-14:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sf(x, t): return x.sum() - t
def _sj(x, t): return np.ones(len(x))

def _slsqp(h, a, ml, mi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(a, ml), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(r.x, n)
    except:
        return h

def _to_bin(h, n, rng):
    h = (h >= 0.5).astype(float)
    t = n // 2
    s = int(h.sum())
    if s > t:
        for _ in range(s - t): h[rng.randint(n)] = 0.0
    elif s < t:
        for _ in range(t - s): h[rng.randint(n)] = 1.0
    return h

def _bin_sa(h, n, rng, deadline, T0=0.004):
    h = _to_bin(h, n, rng)
    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
    cur = _eval(h); bh, bc = h.copy(), cur
    T = T0
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0: break
        i = ones[rng.randint(len(ones))]; j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc = _eval(h)
        d = nc - cur
        if d < 0 or (T > 1e-10 and rng.rand() < np.exp(-d / T)):

        else:
