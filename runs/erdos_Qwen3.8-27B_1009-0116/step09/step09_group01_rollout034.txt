    for _ in range(60):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = int(m.sum())
        if c == 0:
            break
        x[m] += d / c
    return x

def _grad_max(h):
    n = len(h)
    corr = _corr(h)
    dx = 2.0 / n
    vals = corr * dx
    mx = vals.max()
    tol = max(1e-8, 3.0 * dx)
    active = np.where(vals >= mx - tol)[0]
    g = np.zeros(n)
    wsum = 0.0
    for m in active:
        k = int(m) - (n - 1)
        w = vals[m]
        wsum += w
        if k >= 0:
            n1 = n - k
            g[k:] += w * (1.0 - h[:n1])
            if k > 0:
                g[:n1] += -w * h[k:]
        else:
            l = -k
            n1 = n - l
            g[:n1] += w * (1.0 - h[l:])
            if l > 0:
                g[l:] += -w * h[:n1]
    if wsum > 0:
        g /= wsum
    g *= dx
    return g

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h.copy(), m * 0.5)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _local_search(h0, rng, tl, s0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _proj(h0.copy(), S)
    c = _c5(h)
    bc = c
    bh = h.copy()
    s = s0
    no_imp = 0
    g = _grad_max(h)
    order = np.argsort(g)
    q = max(30, n // 4)
    lo = order[:q]
    hi = order[-q:]
    recompute = 0
    it = 0
    while time.time() - t0 < tl:
        if it >= recompute:
            g = _grad_max(h)
            order = np.argsort(g)
            lo = order[:q]
            hi = order[-q:]
            recompute = it + 200
        # Pick direction
        r = rng.random()
        if r < 0.85:
            i = int(hi[rng.randint(len(hi))])
            j = int(lo[rng.randint(len(lo))])
        elif r < 0.95:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        else:
            # 2-swap
            i1 = int(hi[rng.randint(len(hi))])
            j1 = int(lo[rng.randint(len(lo))])
            i2 = int(hi[rng.randint(len(hi))])
            j2 = int(lo[rng.randint(len(lo))])
            if len(set([i1, j1, i2, j2])) < 4:
                i, j = i1, j1
                i2, j2 = -1, -1
            else:
                i, j = i1, j1
                pass  # handle 2-swap
        # Single swap
        if g[i] >= g[j]:
            sgn = 1
            avail = min(h[i], 1.0 - h[j])
        else:
            sgn = -1
            avail = min(1.0 - h[i], h[j])
        if avail < 1e-13:
            it += 1
            continue
        # Line search: try several step sizes
        best_c = c
        best_d = 0.0
        base = min(avail, s * (0.3 + 2.0 * rng.random()))
        for mult in (0.25, 0.5, 1.0, 2.0, 4.0):
            d = sgn * min(avail, base * mult)
            if abs(d) < 1e-14:
                continue
            h[i] += d
            h[j] -= d
            cn = _c5(h)
            h[i] -= d
            h[j] += d
            if cn < best_c - 1e-14:
                best_c = cn
                best_d = d
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            c = best_c
            if c < bc:
                bc = c
                bh = h.copy()
            no_imp = 0
            s = min(s * 1.005, 0.1)
        else:
            no_imp += 1
            s = max(s * 0.95, 1e-9)
        # Kick
        if no_imp > 3000:
            nk = max(3, n // 50)
            idx = rng.choice(n, nk, replace=False)
            for ii in idx:
                h[ii] += (rng.random() - 0.5) * s * 5.0
            h = _proj(h, S)
            c = _c5(h)
            no_imp = 0
        it += 1
    if c < bc:
        bc = c
    bh = _proj(bh, S)
    return bh, _c5(bh)

def _fista_sm(h0, a, tl, S):
    """Simplified FISTA with log-sum-exp."""
    t0 = time.time()
    n = len(h0)
    N = 1
    m = 2 * n - 1
    while N < m:
        N <<= 1
    h = _proj(h0.copy(), S)
    bh = h.copy()
    bc = _c5(h)
    y = h.copy()
    hp = h.copy()
    tv = 1.0
    fy = 0.0
    Lc = 8.0
    for it in range(100000):
        if time.time() - t0 > tl:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            if y.min() < -0.01 or y.max() > 1.01:
                y = _proj(y, S)
        # Compute smooth gradient
        Hy = np.fft.rfft(y, N)
        ac = np.fft.irfft(Hy * np.conj(Hy), N).real[:n]
        pref = np.empty(n + 1)
        pref[0] = 0.0
        np.cumsum(y, out=pref[1:])
        tot = pref[n]
        P = np.empty(n)
        Nv = np.zeros(n)
        P[0] = tot - ac[0]
        if n > 1:
            idx = np.arange(1, n)
            P[idx] = pref[n - idx] - ac[idx]
            Nv[idx] = tot - pref[idx] - ac[idx]
            mx = max(P.max(), Nv.max())
        else:
            mx = P[0]
        e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
        e = e0
        if n > 1:
            ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
            en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
            e += ep.sum() + en.sum()
        sv = (2.0/n) * (mx + np.log(max(e, 1e-300)) / a)
        ty = max(e, 1e-300)
        wP = np.zeros(n)
        wN = np.zeros(n)
        wP[0] = e0 / ty
        if n > 1:
            wP[1:] = ep / ty
            wN[1:] = en / ty
        u = np.zeros(n)
        if n > 1:
            u[1:] = wP[1:] + wN[1:]
        cp = np.cumsum(wP)
        cn = np.cumsum(wN)
        linP = cp[n-1::-1]
        linN = cn
        q = np.zeros(n)
        if n > 1 and np.any(u[1:]):
            R = np.fft.irfft(Hy * np.fft.rfft(u[::-1], N), N)[n-1:2*n-1].real
            L = np.fft.irfft(Hy * np.fft.rfft(u, N), N)[:n].real
            q = -(R + L)
        g = (2.0/n) * (wP[0] * (1.0 - 2.0*y) + linP + linN + q)
        # Step
        step = 1.0 / max(Lc, 1e-12)
        hn = _proj(y - step * g, S)
        Lc = 1.0 / max(step, 1e-12)
        hp = h
        h = hn
        fy = sv
        if it % 20 == 0:
            c = _c5(h)
            if c < bc:
                bc = c
                bh = h.copy()
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return h, bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    S = n * 0.5
    best = _proj(h0, S)
    best_c = _c5(best)
    
    # Symmetry
    for var in [1.0-best, best[::-1], 0.5+0.5*(best-best[::-1])]:
        hv = _proj(var, S)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv
    
    # Phase 1: FISTA smooth (90s)
    end1 = t0 + 90.0
    hc = best.copy()
    for a in [50, 200, 800, 4000, 20000]:
        rem = end1 - time.time()
        if rem < 5:
            break
