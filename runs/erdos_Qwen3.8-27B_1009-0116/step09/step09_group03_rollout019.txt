import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _sobj(x, alpha, k):
    c = _corr(x)
    idx = np.argpartition(-c, k)[:k]
    vals = c[idx]
    m = vals.max()
    s = np.clip(alpha * (vals - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argpartition(-c, k)[:k]
    vals = c[idx]
    m = vals.max()
    s = np.clip(alpha * (vals - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii] * dx
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, k, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sobj, h.copy(), jac=_sgrad, args=(alpha, k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 25})
        return _repair(res.x, n)
    except Exception:
        return h

def _pg(h, alpha, k, steps, lr0, mom):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    for it in range(steps):
        g = _sgrad(h, alpha, k)
        v = mom * v - lr0 * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        step = lr0
        taken = False
        for _t in range(10):
            h_try = _proj(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < _eval(h) * (1.0 - 1e-6):
                h = h_try
                v = mom * v
                taken = True
                break
            step *= 0.5
        if not taken:
            h = h_new
    return best_h, best_c

def _down(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _up(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _gsearch(h, niter, rng, alpha, k, scale):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    gc = _sgrad(h, alpha, k)
    for _ in range(niter):
        top = np.argpartition(-np.abs(gc), 5)[:5]
        i = top[rng.randint(5)]
        j = top[(rng.randint(5) + 1) % 5]
        if i == j:
            continue
        d = scale * rng.randn()
        h_new = best_h.copy()
        h_new[i] += d
        h_new[j] -= d
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            gc = _sgrad(h_new, alpha, k)
    return best_h, best_c

def _fine_polish(h, niter, rng):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    for _ in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 3e-8 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            c = _eval(best_h)
            if c < best_c:
                best_c = c
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c
