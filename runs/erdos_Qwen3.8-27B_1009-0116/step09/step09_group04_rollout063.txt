import numpy as np
from scipy.optimize import minimize
import time

def _eval_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _fft_corr_raw(h, N):
    n = len(h)
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    R = np.fft.irfft(H * G.conj(), N)
    return np.roll(R, n - 1)[:2 * n - 1]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _sf(x, hn):
    return x.sum() - hn

def _sj(x, hn):
    return np.ones(len(x))

def _sobj(x, alpha, ml, N):
    n = len(x)
    dx = 2.0 / n
    c = _fft_corr_raw(x, N) * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml, N):
    n = len(x)
    dx = 2.0 / n
    c = _fft_corr_raw(x, N) * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        k = idx[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pdf_correlation(i, j, h, n):
    """Compute delta vector for pairwise change h[i]+=d, h[j]-=d."""
    Ls = np.arange(-(n - 1), n)
    idx_a = j - Ls
    idx_b = j + Ls
    idx_c = i - Ls
    idx_d = i + Ls
    delta = np.zeros(2 * n - 1)
    m1 = (idx_a >= 0) & (idx_a < n)
    m2 = (idx_b >= 0) & (idx_b < n)
    m3 = (idx_c >= 0) & (idx_c < n)
    m4 = (idx_d >= 0) & (idx_d < n)
    delta[m1] += h[idx_a[m1]]
    delta[m2] += h[idx_b[m2]]
    delta[m3] -= h[idx_c[m3]]
    delta[m4] -= h[idx_d[m4]]
    return delta

def _pairwise_random(h, c, rng, n_iters, n, dx, pair_range=7, ts_iters=20):
    """Random pairwise coordinate descent with incremental correlation update."""
    best_c5 = c.max()
    bh = h.copy()
    for _ in range(n_iters):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, pair_range)) % n
        if i == j:
            continue
        hi, hj = h[i], h[j]
        delta = _pdf_correlation(i, j, h, n)
        d_lo = max(-hi, hj - 1.0)
        d_hi = min(1.0 - hi, hj)
        if d_hi - d_lo < 1e-14:
            continue
        lo, hi_d = d_lo, d_hi
        for _ in range(ts_iters):
            m1 = lo + (hi_d - lo) * 0.4
            m2 = lo + (hi_d - lo) * 0.6
            if (c + m1 * delta).max() < (c + m2 * delta).max():
                hi_d = m1
            else:
                lo = m2
        d_opt = (lo + hi_d) * 0.5
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * delta
        cm = c.max()
        if cm < best_c5:
            best_c5 = cm
            bh = h.copy()
    return bh, best_c5, c

def _pairwise_sweep(h, c, n_rounds, n, dx, max_k=4):
    """Structured sweep through pairs."""
    best_c5 = c.max()
    bh = h.copy()
    for _ in range(n_rounds):
        for k in range(1, max_k + 1):
            for i in range(n):
                j = (i + k) % n
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                delta = _pdf_correlation(i, j, h, n)
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-14:
                    continue
                lo, hi_d = d_lo, d_hi
                for _ in range(15):
                    m1 = lo + (hi_d - lo) * 0.4
                    m2 = lo + (hi_d - lo) * 0.6
                    if (c + m1 * delta).max() < (c + m2 * delta).max():
                        hi_d = m1
                    else:
                        lo = m2
                d_opt = (lo + hi_d) * 0.5
                if abs(d_opt) < 1e-14:
                    continue
                h[i] += d_opt
                h[j] -= d_opt
                c += d_opt * delta
                cm = c.max()
                if cm < best_c5:
                    best_c5 = cm
                    bh = h.copy()
    return bh, best_c5, c
