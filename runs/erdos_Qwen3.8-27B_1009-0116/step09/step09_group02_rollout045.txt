def _fft_corr(h):
    """Compute correlation of h with (1-h) using FFT."""
    n = len(h)
    g = 1.0 - h
    # Full correlation via FFT
    N = 2 * n - 1
    H = fft(h, N)
    G = fft(g, N)
    # correlation[n-1+k] = sum h[i] * g[i+k] for valid k
    # Using circular correlation trick
    corr = ifft(H * np.conj(G)).real
    # Rearrange to get full correlation
    corr_full = np.zeros(2*n-1)
    corr_full[:n] = corr[n-1:]  # lags 0 to n-1 (positive)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):

    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-18:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    def _cfun(x):
        return x.sum() - tgt
    def _cjac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-18:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    tgt = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - tgt
    def _cj(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, m_lags, steps, step0, deadline):

    v = np.zeros(n)
    mu = 0.82
    step = step0

    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    stall = 0
    total = 0
    T = 5e-6
    it = 0
    while time.time() < deadline:
        it += 1
        # Reduce temperature
        T = max(1e-12, 5e-6 * np.exp(-it / 30000.0))
        
        # Choose scale based on phase and stall
        if phase_scale == 0:  # coarse
            kp = rng.choice([5, 8, 12, 16])
            step_mag = 5e-4 * (1.0 + 0.5 * min(stall / 200.0, 3.0))
        elif phase_scale == 1:  # medium
            kp = rng.choice([3, 4, 5, 6])
            step_mag = 3e-5 * (1.0 + 0.3 * min(stall / 100.0, 2.0))
        elif phase_scale == 2:  # fine
            kp = rng.choice([2, 3, 4, 5])
            step_mag = 5e-7 * (1.0 + 0.2 * min(stall / 50.0, 2.0))
        else:  # ultra-fine
            kp = rng.choice([2, 3])
            step_mag = 2e-8 * (1.0 + 0.1 * min(stall / 20.0, 2.0))
        
        step = step_mag * (0.4 + 0.6 * rng.rand())
        idx = rng.randint(0, n, size=kp)
        d = step * rng.randn(kp)
        d -= d.mean()
        
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            stall += 1
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        total += 1
        
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            stall = 0
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            # Annealing: accept with probability
            dC = c5n - c5c
            if dC < T and rng.rand() < np.exp(-dC / max(T, 1e-15)):
                h = h_new
                c5c = c5n
            stall += 1
            # Wake: larger kick if very stuck
            if stall > 300 and rng.rand() < 0.005:

                kidx = rng.randint(0, n, size=10)

                kd -= kd.mean()
                h_new2 = h.copy()
                h_new2[kidx] += kd
                h_new2 = _repair(h_new2, n)
                c5k = _eval(h_new2)
                if c5k < best_c + 5e-5:
                    h = h_new2
                    c5c = c5k
                    stall = 0
    return best_h, best_c

def _pair_polish(h, deadline, rng, scale):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() < deadline:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        if i == j:
            continue
        d = scale * (0.2 + 0.8 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c
