import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            nt = n - L
            g[L:] += wk * (1.0 - x[:nt])
            g[:nt] -= wk * x[L:]
        else:
            M = -L
            nt = n - M
            g[:nt] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:nt]
    return g

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _nesterov(h, alpha, m_lags, steps, step0, time_budget):
    n = len(h)
    v = np.zeros(n)
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    step = step0
    t_start = time.time()
    for it in range(steps):
        if time.time() - t_start > time_budget:
            break
        g = _smooth_grad(h, alpha, m_lags)
        v = 0.9 * v + g
        nv = np.linalg.norm(v)
        if nv < 1e-14:
            break
        direction = v / nv
        s = step
        taken = False
        for _t in range(14):
            h_new = _proj(h - s * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(s * 1.18, step0 * 40.0)
                taken = True
                break
            s *= 0.45
        if not taken:
            step = min(step * 1.5, step0 * 40.0)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return _repair(best_h, n), best_c

def _slsqp_h(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def sfn(x):
        return x.sum() - n * 0.5
    def sjac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': sfn, 'jac': sjac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _multi_pair_2pt(h, n_iter, alpha, m_lags, time_budget, rng):

        g_proj = g - g.mean()
        top_i = np.argsort(g_proj)[-3:]
        top_j = np.argsort(g_proj)[:3]
        found_move = False
        for ii in range(3):
            for jj in range(3):
                i, j = int(top_i[ii]), int(top_j[jj])
                if i == j:
                    continue
                for frac in [0.1, 0.3, 0.6, 1.0]:
                    d = frac * (g_proj[i] - g_proj[j]) * 0.004
                    h[i] += d
                    h[j] -= d
                    if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                        h[i] -= d
                        h[j] += d

                    c5n = _eval(h)

                        c5c = c5n
                        found_move = True
                        if c5c < best_c:

                            best_h = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
                        break
                if found_move:
                    break
            if found_move:
                break
    return best_h, best_c

def _anneal_local(h, n_iter, rng, base_step, n_pert, time_budget):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    t_start = time.time()
    temp = 0.002
    for _ in range(n_iter):
        if time.time() - t_start > time_budget:
            break
        step = base_step * (0.5 + rng.rand() * 1.5)
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur or rng.rand() < np.exp(-(c5n - c5cur) / max(temp, 1e-8)):
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        temp *= 0.9997
    return best_h, best_c

def _segment_shift(h, rng, n_segments, time_budget):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    t_start = time.time()
    for _ in range(500):
        if time.time() - t_start > time_budget:
            break
        seg_len = rng.randint(n // 20, n // 4)
        start = rng.randint(0, n - seg_len)
        shift = rng.randint(-seg_len, seg_len + 1)
        h_new = h.copy()
        seg = h_new[start:start + seg_len].copy()
        ns = start + shift
        if ns < 0 or ns + seg_len > n:
            continue
        h_new[ns:ns + seg_len] = seg
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < best_c:
            best_c = c5n
            best_h = h_new.copy()
            h = h_new
            c5c = best_c
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)
