The sensitivity of corr_k to h[i]:

Δcorr_k ≈ d * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]

This is a linear approximation! And we can compute it efficiently.

def _grad_guided_search(h, n, n_iters, rng, refinement):
    dx = 2.0 / n
    c5 = _eval(h)
    best_c5 = c5
    best_h = h.copy()
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argpartition(-corr, refinement)[:refinement]
        c_max = corr.max()
        
        # Compute sensitivity for top lags
        # For lag index li (in full corr output), actual shift = li - (n-1)
        # ∂corr_shift/∂h[i] = (1-h[i+shift]) - h[i-shift] (with boundary handling)
        
        # For the worst lag, try moves that reduce it
        worst = top_k[0]
        shift = worst - (n - 1)
        
        # Sensitivity at each position
        sens = np.zeros(n)
        for i in range(n):
            s = shift
            val = 0.0
            if 0 <= i + s < n:
                val += (1.0 - h[i + s])
            if 0 <= i - s < n:
                val -= h[i - s]
            sens[i] = val
        
        # We want to increase h where sens is negative (reduces corr)
        # and decrease h where sens is positive
        # A 2-point move (i, j, d) changes corr by d * (sens[i] - sens[j])
        # We want d * (sens[i] - sens[j]) < 0
        
        # Find i with most negative sens and j with most positive sens
        i_cand = np.argmin(sens)
        j_cand = np.argmax(sens)
        
        # Try a few moves
        for _ in range(5):
            i = i_cand if rng.rand() < 0.5 else np.argpartition(sens, 5)[rng.randint(5)]
            j = j_cand if rng.rand() < 0.5 else np.argpartition(-sens, 5)[rng.randint(5)]
            d = rng.uniform(1e-5, 1e-3) * (1 if sens[i] < sens[j] else -1)
            hi, hj = h[i] + d, h[j] - d
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h[i], h[j] = hi, hj
                c_new = _eval(h)
                if c_new < c5:
                    c5 = c_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    h[i], h[j] = hi - d, hj + d
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -350, 350)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _obj_grad_lse(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -350, 350)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

            continue
        wk = w[ii] * dx
        L = k - (n - 1)

            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argpartition(-c, ml)[:ml]
    ct = c[idx]
    return np.sum(ct**p) ** (1.0/p)

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(_obj_lse, h, jac=_obj_grad_lse, args=(alpha,ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':ftol,'maxls':30})
        return _repair(res.x, n)
    except:
        return h

def _lbfgs_proj(h, alpha, ml, steps, lr0):

    target = n * 0.5
    best_c = _eval(h)
    best_h = h.copy()

        if c < best_c:

            best_h = x.copy()
        if it % 50 == 0:

    return best_h, best_c

        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sf(x, t): return x.sum() - t
def _sj(x, t): return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type':'eq','fun':_sf,'jac':_sj,'args':(n*0.5,)}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, args=(alpha,ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':ftol,'maxls':30})
        return _repair(res.x, n)
    except:
        return h

def _proj_grad(h, alpha, ml, steps, lr0):
    n = len(h); target = n * 0.5
    best_c = _eval(h); best_h = h.copy()
    x = h.copy(); lr = lr0
    for it in range(steps):
        g = _obj_grad(x, alpha, ml)
        c_cur = _eval(x)
        step = lr
        taken = False
        for _bt in range(25):
            x_try = _proj(x - step * g, target)
            c_try = _eval(x_try)
            if c_try < c_cur:
                x = x_try; taken = True
                lr = min(step * 1.05, lr0 * 100)
                break
            step *= 0.5
        if not taken:
            x = _proj(x - step * g, target)
            lr = max(step * 0.9, 1e-10)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_h = x.copy()
    return best_h, best_c

def _sens_lag(h, n, idx):
    k = idx - (n - 1)
    sens = np.zeros(n)
    if k >= 0:
        sens[k:] += 1.0 - h[:n-k]
        sens[:n-k] -= h[k:]
    else:
        s = -k
        sens[:n-s] += 1.0 - h[s:]
        sens[s:] -= h[:n-s]
    return sens

def _grad_search(h, n, n_iters, rng, alpha, ml):
    dx = 2.0 / n
    c5 = _eval(h); best_c = c5; best_h = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_m = min(ml, 2*n-1)
        top_idx = np.argpartition(-corr, top_m-1)[:top_m]
        
        # Compute combined sensitivity for top lags
        sens = np.zeros(n)
        for idx in top_idx[:min(10, len(top_idx))]:
            s = _sens_lag(h, n, idx)
            w = max(corr[idx] - corr[top_idx].max() + 0.1, 0.01)
            sens += w * s
        
        neg = np.where(sens < -1e-4)[0]
        pos = np.where(sens > 1e-4)[0]
        
        for _ in range(15):
            if len(neg) > 2 and len(pos) > 2 and rng.rand() < 0.7:
                i = neg[rng.randint(len(neg))]
                j = pos[rng.randint(len(pos))]
                d = abs(rng.uniform(1e-5, 1e-3))
            else:
                i = rng.randint(n); j = rng.randint(n)
                if i == j: continue
                d = rng.uniform(1e-5, 1e-4) * rng.choice([-1,1])
            hi, hj = h[i]+d, h[j]-d
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h[i], h[j] = hi, hj
                cn = _eval(h)
                if cn < c5:
                    c5 = cn
                    if cn < best_c:
                        best_c = cn; best_h = h.copy()
                else:
                    h[i], h[j] = hi-d, hj+d
    return best_h, best_c

def _polish(h, n, n_iters, rng):
    c5 = _eval(h); best_c = c5; best_h = h.copy()
    for _ in range(n_iters):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 1e-7 * rng.randn() * 5
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5:
                c5 = cn
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c

def _downsample(h, f):
    n = len(h); m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    n = len(h); out = np.empty(n*f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n*f)
