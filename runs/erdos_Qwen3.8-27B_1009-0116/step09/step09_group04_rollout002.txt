def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _eval_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):

    s = h.sum()

        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a

def _grad_lag(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[:n - L] += (1.0 - h[L:]) * dx
        g[L:] -= h[:n - L] * dx
    else:
        M = -L
        g[M:] += (1.0 - h[:n - M]) * dx
        g[:n - M] -= h[M:] * dx
    return g

def _smooth_obj(x, alpha, ml):

    idx = np.argsort(c)[-ml:]

    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _proj(h, target):

        if np.clip(h - mid, 0, 1).sum() > target:

        else:

def _make_block(n, m, rng):
