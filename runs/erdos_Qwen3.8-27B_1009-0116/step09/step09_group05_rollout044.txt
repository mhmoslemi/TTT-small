import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_full(h):
    n = len(h)
    return _corr_raw(h) * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _topk_grad(h, k):
    n = len(h)
    c_raw = _corr_raw(h)
    idx = np.argsort(c_raw)[-k:]
    g = np.zeros(n)
    for kk in idx:
        L = kk - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g / (float(k))

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _proj_grad(h, k_top, steps, step0, t_end):
    n = len(h)
    cur_c = _eval(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    for it in range(steps):
        if time.time() > t_end:
            break
        g = _topk_grad(h, k_top)
        v = 0.8 * v + 0.2 * g
        sq = 0.9 * sq + 0.1 * g * g
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(16):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.15, step0 * 50.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.5, step0 * 50.0)
    return h, cur_c

def _golden_transfer(h, inv, src, n_iter=12):
    """Transfer: h[inv] += t, h[src] -= t. Golden section on t."""
    n = len(h)
    tlo = max(-h[inv], h[src] - 1.0)
    thi = min(1.0 - h[inv], h[src])
    if thi - tlo < 1e-15:
        return _eval(h)
    phi = (1.0 + 5.0**0.5) / 2.0
    c1 = 1.0 - 1.0 / phi
    c2 = 1.0 / phi
    a1 = tlo + c2 * (thi - tlo)
    a2 = tlo + c1 * (thi - tlo)
    h[inv] += a1; h[src] -= a1
    f1 = _eval(h)
    h[inv] += a2 - a1; h[src] -= a2 - a1
    f2 = _eval(h)
    h[inv] -= a2; h[src] += a2
    for _ in range(n_iter):
        if f1 < f2:
            thi = a2; a2 = a1; f2 = f1
            a1 = tlo + c2 * (thi - tlo)
            h[inv] += a1; h[src] -= a1
            f1 = _eval(h)
            h[inv] -= a1; h[src] += a1
        else:
            tlo = a1; a1 = a2; f1 = f2
            a2 = tlo + c1 * (thi - tlo)
            h[inv] += a2; h[src] -= a2
            f2 = _eval(h)
            h[inv] -= a2; h[src] += a2
    t_opt = (tlo + thi) / 2.0
    h[inv] += t_opt; h[src] -= t_opt
    return _eval(h)

def _grad_2pt(h, t_end, alpha, m_lags):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    while time.time() < t_end:
        g = _smooth_grad(h, alpha, m_lags)
        gp = g - g.mean()
        i = int(np.argmax(gp))
        j = int(np.argmin(gp))
        if i == j:
            break
        c_after = _golden_transfer(h, j, i, 10)
        if c_after < best_c:
            best_c = c_after
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best_c

def _fine_polish(h, rng, t_end, step_scale):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    stuck = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n // 2)) % n
        t = step_scale * rng.randn()
        hug = h[i] + t
        hug2 = h[j] - t
        if 0.0 <= hug <= 1.0 and 0.0 <= hug2 <= 1.0:
            h[i] = hug; h[j] = hug2
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                stuck = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i] = hug - t; h[j] = hug2 + t
                stuck += 1
            if stuck > 200:
                step_scale *= 0.7
                stuck = 0
    return best_h, best_c
