def _repair(h, n):

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _do_slsqp(h, alpha, ml, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    def _of(x): return x.sum() - n * 0.5
    def _oj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _of, 'jac': _oj}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return h

def _do_lbfgs(h, alpha, ml, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    def _of(x): return x.sum() - n * 0.5
    def _oj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _of, 'jac': _oj}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfun': maxiter * 4})
        return _repair(r.x, n)
    except:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

                if cn < bc:

            else:

        hn = _repair(hn, n)
        cn = _eval(hn)
        delta = cn - c
        if delta < 0 or (t > 1e-12 and rng.rand() < np.exp(-delta / t)):
            h, c = hn, cn
            if cn < bc:
                bc, bh = cn, hn.copy()
    return bh

def _grad_descent(h, n, rng, n_steps, alpha, ml, base_step):
    """Projected gradient with momentum and adaptive step."""
    cc = _eval(h)
    bh, bc = h.copy(), cc
    v = np.zeros(n)
    step = base_step
    mu = 0.85
    for _ in range(n_steps):
        g = _sgrad(h, alpha, ml)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(10):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cc:
                h, cc, taken = hn, cn, True
                step = min(step * 1.05, base_step * 20)
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.3, base_step * 20)
        if cc < bc:
            bc, bh = cc, h.copy()
    return bh

def _fine_polish(h, n, n_iters, rng):
    """Very fine 2-point and 3-point polish with adaptive steps."""
    h2 = h.copy()
    c2 = _eval(h2)
    step_scale = 1e-5
    for _ in range(n_iters):
        r = rng.rand()
        if r < 0.6:
            i = rng.randint(n)
            j = (i + rng.randint(1, 4)) % n
            d = step_scale * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                cn = _eval(h2)
                if cn < c2:
                    c2 = cn
                    step_scale = min(step_scale * 1.02, 5e-5)
                else:
                    h2[i], h2[j] = ni - d, nj + d
                    step_scale = max(step_scale * 0.9, 1e-9)
        else:
            i = rng.randint(n)
            j = (i + rng.randint(1, 3)) % n
            k = (i + rng.randint(4, 12)) % n
            if k == i or k == j:
                continue
            d1 = step_scale * rng.randn()
            d2 = step_scale * rng.randn()
            ni, nj, nk = h2[i] + d1, h2[j] + d2, h2[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h2[i], h2[j], h2[k] = ni, nj, nk
                cn = _eval(h2)
                if cn < c2:
                    c2 = cn
                    step_scale = min(step_scale * 1.02, 5e-5)
                else:
                    h2[i], h2[j], h2[k] = ni - d1, nj - d2, nk + d1 + d2
                    step_scale = max(step_scale * 0.9, 1e-9)
    return h2, c2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP warm-up with moderate alphas
    if _el() < budget_s - 500:
        for a, ml, mi in [(30000, 25, 600), (300000, 40, 800), (3000000, 60, 1000)]:
            if _el() > budget_s - 500:
                break
            hn = _do_slsqp(best_h, a, ml, mi, n)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 2: High-alpha SLSQP + L-BFGS-B
    if _el() < budget_s - 400:
        for a, ml, mi in [(3e7, 80, 1200), (3e8, 100, 1500)]:
            if _el() > budget_s - 400:
                break
            hn = _do_slsqp(best_h, a, ml, mi, n)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        if _el() < budget_s - 350:
            hn = _do_lbfgs(best_h, 3e8, 100, 3000, n)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 3: Projected gradient with momentum
    if _el() < budget_s - 280:
        for a, ml, st, s0 in [(50000, 50, 400, 0.003), (5e6, 80, 350, 0.001), (5e7, 100, 300, 0.0005)]:
            if _el() > budget_s - 280:
                break
            hn = _grad_descent(best_h.copy(), n, rng, st, a, ml, s0)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 4: Structured 2-point search (many candidates per unit time)
    if _el() < budget_s - 200:
        nt = min(int((budget_s - 200 - _el()) / 0.0005), 500000)
        best_h = _structured_2pt(best_h, n, nt, rng, (1e-5, 5e-4))
        cn = _eval(best_h)
        if cn < best_c5:
            best_c5 = cn

    # Phase 5: Multi-scale SA
    if _el() < budget_s - 120:
        nt = min(int((budget_s - 120 - _el()) / 0.003), 100000)
        best_h = _block_sa(best_h, n, rng, nt, 5e-5)
        cn = _eval(best_h)
        if cn < best_c5:
            best_c5 = cn

    # Phase 6: Another round of SLSQP at very high alpha
    if _el() < budget_s - 80:
        for a, ml, mi in [(1e9, 100, 1000), (5e9, 120, 800)]:
            if _el() > budget_s - 80:
                break
            hn = _do_slsqp(best_h, a, ml, mi, n)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 7: Fine polish
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.0015), 400000)
        best_h, best_c5 = _fine_polish(best_h, n, nt, rng)

    # Phase 8: Final SLSQP + polish
    if _el() < budget_s - 3:
        hn = _do_slsqp(best_h, 1e9, 120, 500, n)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
