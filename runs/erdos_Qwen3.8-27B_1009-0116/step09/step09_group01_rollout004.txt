import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _subgrad(h, topk=8, tol=1e-10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    active = np.where(corr >= mx - tol)[0]
    if len(active) > topk:
        active = active[np.argsort(corr[active])[::-1][:topk]]
    g = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        if shift == 0:
            g += (1.0 - 2.0 * h)
        elif shift > 0:
            g[:n - shift] += (1.0 - h[shift:n])
            g[shift:n] += -h[:n - shift]
        else:
            ss = -shift
            g[ss:n] += (1.0 - h[:n - ss])
            g[:n - ss] += -h[ss:n]
    return g / max(len(active), 1) * (2.0 / n)

    n = len(h0)

    hp = h.copy()
    y = h.copy()
    tv = 1.0
    L = 8.0
    # Precompute for objective
    def _pn(h):
        N = 1
        m = 2 * n - 1
        while N < m:
            N <<= 1
        H = np.fft.rfft(h, N)
        ac = np.fft.irfft(H * np.conj(H), N).real[:n]
        pref = np.empty(n + 1)
        pref[0] = 0.0
        np.cumsum(h, out=pref[1:])
        tot = pref[n]
        P = np.empty(n)
        P[0] = tot - ac[0]
        Nv = np.zeros(n)
        if n > 1:
            idx = np.arange(1, n)
            P[idx] = pref[n - idx] - ac[idx]
            Nv[idx] = tot - pref[idx] - ac[idx]
            mx = float(max(np.max(P), np.max(Nv)))
        else:
            mx = float(P[0])
        return P, Nv, mx

    def _sval(h, a):

        dx = 2.0 / n
        e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
        if n > 1:
            e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
            e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
        return dx * (mx + np.log(max(e, 1e-300)) / a)

    def _sg(h, a):

        wN = np.zeros(n)

            cn_[1:] = np.cumsum(wN[1:])
        linP = cp[n - 1::-1]

        q = np.zeros(n)

        tv = tn
        if y.min() < -0.1 or y.max() > 1.1:
            y = _proj(y, S)
        g = _sg(y, a)
        step = 1.0 / L
        ok = False
        for _ in range(8):

            if _sval(xn, a) <= rhs + 1e-10:
                ok = True
                break
            step *= 0.5
        if ok:
            hp = h.copy()
            h = xn
            L = max(L * 0.9, 1.0)
        else:
            hp = h.copy()
            y = h.copy()
            tv = 1.0
            L = max(L * 0.5, 1.0)
