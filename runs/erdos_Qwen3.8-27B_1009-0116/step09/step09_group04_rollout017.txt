            cn = _eval(hn)
            if cn < cur:
                h, cur = hn, cn
                if cn < bc:
                    bc, bh = cn, h.copy()
                st = min(st * 1.05, s0 * 12.0)
                ok = True
                break
            st *= 0.4
        if not ok:
            st = min(st * 1.2, s0 * 12.0)
    return bh, bc

def _cdesc(h, steps, rng, al, ml, bd):
    n = len(h)
    cur = _eval(h)
    bh, bc = h.copy(), cur
    g = _sgrad(h, al, ml)
    for it in range(steps):
        if it % 100 == 0:
            g = _sgrad(h, al, ml)
        gi, gj = g.argmax(), g.argmin()
        if gi != gj and g[gi] * g[gj] < 0:
            i, j, d = gi, gj, bd
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = bd * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _polish(h, remaining, rng):
    n = len(h)
    cur = _eval(h)
    bh, bc = h.copy(), cur
    g = _sgrad(h, 200000, 60)
    npol = min(int(remaining * 700), 800000)
    for it in range(npol):
        if time.time() > h[0] * 0:
            break
        if it % 400 == 0:
            g = _sgrad(h, 200000, 60)
        if it % 3 == 0:
            gi, gj = g.argmax(), g.argmin()
            if gi != gj and g[gi] * g[gj] < 0:
                i, j = gi, gj
                d = 4e-7 if g[gi] > 0 else -4e-7
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 3)) % n
                d = 4e-7 * rng.randn()
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            d = 4e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    bh, bc5 = h.copy(), _eval(h)
    # Symmetry transform as alternative start
    hsym = _repair(1.0 - np.roll(h0, n // 2), n)
    if _eval(hsym) < bc5:
        bh, bc5 = hsym.copy(), _eval(hsym)

    # Phase 1: SLSQP with increasing alpha
    for al, ml in [(5000, 30), (50000, 50), (500000, 80), (5000000, 100)]:
        if time.time() - t0 > 150:
            break
        hn = _slsqp(bh, al, ml, 800)
        cn = _eval(hn)
        if cn < bc5:
            bc5, bh = cn, hn

    # Phase 2: Multi-resolution at n//2
    if time.time() - t0 < budget_s - 420:
        n2 = n // 2
        h2 = bh[:n2 * 2].reshape(n2, 2).mean(axis=1)
        h2 = _repair(h2, n2)
        for al in [5000, 50000, 500000]:
            if time.time() - t0 > budget_s - 350:
                break
            h2 = _slsqp(h2, al, 30, 500)
        hu = np.repeat(h2, 2)
        if len(hu) >= n:
            hu = hu[:n]
        for t in [0.3, 0.5, 0.7, 1.0]:
            hb = _repair(t * hu + (1.0 - t) * bh, n)
            cn = _eval(hb)
            if cn < bc5:
                bc5, bh = cn, hb
        # Also try n//4
        if time.time() - t0 < budget_s - 300:
            n4 = n // 4
            h4 = bh[:n4 * 4].reshape(n4, 4).mean(axis=1)
            h4 = _repair(h4, n4)
            for al in [5000, 50000]:
                if time.time() - t0 > budget_s - 260:
                    break
                h4 = _slsqp(h4, al, 20, 300)
            hu4 = np.repeat(h4, 4)
            if len(hu4) >= n:
                hu4 = hu4[:n]
            for t in [0.5, 0.8, 1.0]:
                hb = _repair(t * hu4 + (1.0 - t) * bh, n)
                cn = _eval(hb)
                if cn < bc5:
                    bc5, bh = cn, hb

    # Phase 3: Top-k gradient descent (equal-weight top-k, tighter than soft-max)
    if time.time() - t0 < budget_s - 280:
        for k, s0, st, dm in [(3, 0.005, 500, 0.78), (5, 0.004, 450, 0.82),
                               (10, 0.003, 400, 0.86), (20, 0.002, 350, 0.89),
                               (40, 0.001, 300, 0.92), (100, 0.0005, 250, 0.94)]:
            if time.time() - t0 > budget_s - 180:
                break
            hn, cn = _gdesc(bh.copy(), k, st, s0, dm)
            if cn < bc5:
                bc5, bh = cn, hn

    # Phase 4: Gradient-guided coordinate descent
    if time.time() - t0 < budget_s - 100:
        for al, ml, bd in [(50000, 40, 2e-4), (500000, 70, 1e-4),
                            (5000000, 100, 5e-5)]:
            if time.time() - t0 > budget_s - 60:
                break
            st = min(int((budget_s - 60 - (time.time() - t0)) / 0.0025), 200000)
            hn, cn = _cdesc(bh.copy(), st, rng, al, ml, bd)
            if cn < bc5:
                bc5, bh = cn, hn

    # Phase 5: Ultra-fine gradient-guided polish
    rem = budget_s - 2 - (time.time() - t0)
    if rem > 1:
        bh, bc5 = _polish(bh, rem, rng)

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
