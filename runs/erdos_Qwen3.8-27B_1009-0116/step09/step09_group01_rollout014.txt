import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _c5_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    for _ in range(100):
        if abs(s - target) < 1e-12:
            break
        h = np.clip(h * (target / max(s, 1e-30)), 0.0, 1.0)
        s = h.sum()
    d = target - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-12:
                break
            step = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += step if d > 0 else -step
    return h

def _to_bin(h, n):
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:n // 2]] = 1.0
    return h_b

def _targeted(h, rng, tl):
    t0 = time.time()
    n = len(h)
    c = _c5(h)
    best_c = c
    bh = h.copy()
    stall = 0
    while time.time() - t0 < tl:
        arr = _c5_arr(h)
        mv = arr.max()
        thresh = mv - 0.04
        hot = np.where(arr >= thresh)[0]
        moved = False
        for hi in hot:
            k = hi - (n - 1)
            if k == 0:
                continue
            if k > 0:
                v = np.arange(n - k)
                ci = v[(h[v + k] > 0.5) & (h[v] < 0.5)]
            else:
                kk = -k
                v = np.arange(n - kk)
                ci = v[(h[v] > 0.5) & (h[v + kk] < 0.5)]
            if len(ci) == 0:
                continue
            idx = ci[rng.randint(len(ci))]
            if k > 0:
                i, j = idx, idx + k
            else:
                i, j = idx, idx - k
            h[i], h[j] = h[j], h[i]
            nc = _c5(h)
            if nc < c:
                c = nc
                if nc < best_c:
                    best_c = nc
                    bh = h.copy()
                moved = True
                stall = 0
                break
            else:
                h[i], h[j] = h[j], h[i]
        if not moved:
            esc = 0
            while esc < 40 and time.time() - t0 < tl:
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    esc += 1
                    continue
                h[i], h[j] = h[j], h[i]
                nc = _c5(h)
                if nc < c * 1.008:
                    c = nc
                    esc = 0
                else:
                    h[i], h[j] = h[j], h[i]
                esc += 1
            stall += 1
            if stall > 80:
                break
    return bh, best_c

def _blocks(h, rng, tl):
    t0 = time.time()
    n = len(h)
    c = _c5(h)
    best_c = c
    bh = h.copy()
    while time.time() - t0 < tl:
        r = rng.random()
        if r < 0.4:
            L = rng.randint(3, min(20, n // 12))
            i = rng.randint(0, n - L)
            h[i:i+L] = h[i:i+L][::-1].copy()
            nc = _c5(h)
            if nc < best_c:
                best_c, bh, c = nc, h.copy(), nc
            else:
                h[i:i+L] = h[i:i+L][::-1].copy()
        elif r < 0.7:
            L = rng.randint(5, min(25, n // 10))
            i = rng.randint(0, n - L)
            s = rng.randint(1, min(3, L // 2))
            bl = h[i:i+L].copy()
            h[i:i+L] = np.roll(bl, s)
            nc = _c5(h)
            if nc < best_c:
                best_c, bh, c = nc, h.copy(), nc
            else:
                h[i:i+L] = np.roll(h[i:i+L], -s)
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            nc = _c5(h)
            if nc < best_c:
                best_c, bh, c = nc, h.copy(), nc
            else:
                h[i], h[j] = h[j], h[i]
    return bh, best_c

def _anneal(h, rng, tl):
    t0 = time.time()
    n = len(h)
    c = _c5(h)
    best_c = c
    bh = h.copy()
    no_improve = 0
    while time.time() - t0 < tl:
        el = (time.time() - t0) / max(tl, 1e-9)
        T = 0.03 * (1e-4) ** el
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        nc = _c5(h)
        if nc <= c - 1e-15:
            c = nc
            no_improve = 0
            if nc < best_c:
                best_c, bh = nc, h.copy()
        elif rng.random() < np.exp(-(nc - c) / max(T, 1e-12)):
            c = nc
            no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if no_improve > 3000:
            break
    return bh, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    h = _repair(h0, n)
    hb = _to_bin(h, n)
    # Try symmetric variants
    best_h, best_c = hb, _c5(hb)
    sym_variants = [1.0 - hb, hb[::-1]]
    for sv in sym_variants:
        sv = _to_bin(sv, n)
        cv = _c5(sv)
        if cv < best_c:
            best_h, best_c = sv, cv
    # Niche target: focus on the current best
    cur = best_h.copy()
    # Phase 1: Targeted lag repair (200s)
    e1 = t0 + 200
    cur, c1 = _targeted(cur, rng, e1 - time.time())
    if c1 < best_c:
        best_h, best_c = cur, c1
    # Phase 2: Block ops + swaps (150s)
    e2 = t0 + 350
    hb2, c2 = _blocks(cur, rng, e2 - time.time())
    if c2 < best_c:
        best_h, best_c = hb2, c2
    cur = hb2 if c2 < c1 else cur
    # Phase 3: Simulated annealing (200s)
    e3 = t0 + 550
    ha, c3 = _anneal(cur, rng, e3 - time.time())
    if c3 < best_c:
        best_h, best_c = ha, c3
    cur = ha if c3 < (c2 if c2 < c1 else c1) else cur
    # Phase 4: Targeted again (150s)
    e4 = t0 + 700
    ht, c4 = _targeted(cur, rng, e4 - time.time())
    if c4 < best_c:
        best_h, best_c = ht, c4
    cur = ht if c4 < best_c else cur
    # Phase 5: Multi-grid exploration (120s)
    e5 = t0 + 820
    for tn in [280, 420, 840]:
        rem = e5 - time.time()
        if rem < 15:
            break
        hr = _to_bin(_resample(best_h, tn), tn)
        hr2, cr2 = _targeted(hr, rng, min(45, rem // 3))
        if cr2 < best_c:
            best_h, best_c = hr2, cr2
    # Phase 6: Final fine-tuning (200s)
    e6 = t0 + budget_s - 10
    nf = len(best_h)
    cur = best_h.copy()
    hf, cf = _targeted(cur, rng, (e6 - time.time()) * 0.6)

    hf2, cf2 = _blocks(cur, rng, e6 - time.time())
    if cf2 < best_c:

    # Final repair
    best_h = _repair(best_h, len(best_h))
    best_c = _c5(best_h)
    return best_h, best_c, len(best_h)
