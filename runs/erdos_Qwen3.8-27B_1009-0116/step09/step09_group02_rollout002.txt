def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):

def _corr(h):

    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mtop:]

    s -= s.max()
    feas = (x.sum() - n / 2.0) * lam
    return m + np.log(np.exp(s).sum()) / alpha + feas * feas

def _smooth_max_grad(x, alpha, mtop, lam):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_corr(h):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    idx = np.argsort(c)
    return c, idx

def _smooth_obj(x, alpha, mtop):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mtop:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, mtop):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mtop:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam_opt(h, alpha, mtop, nsteps, lr0):
    n = len(h)
    m = np.zeros(n); v = np.zeros(n)
    b1, b2, eps = 0.82, 0.999, 1e-8
    cur_c = _eval(h)
    best_h = h.copy(); best_c = cur_c
    for t in range(1, nsteps + 1):
        g = _smooth_grad(h, alpha, mtop)
        g -= g.mean()
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        step = lr0 / (1.0 + 0.05 * t)
        h_new = h - step * mh / (np.sqrt(vh) + eps)

            h = h_new; cur_c = c_new
            if c_new < best_c:
                best_c = c_new; best_h = h.copy()
        else:
            m = b1 * m  # forget the bad step
    return best_h, best_c

def _pdf_descent(h, alpha, mtop, nsteps, step0):
    n = len(h)
    mu = 0.75
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)

        g -= g.mean()
        v = mu * v + (1 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        taken = False
        for _t in range(12):

def _slsqp_opt(h, alpha, mtop, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, mtop),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h
