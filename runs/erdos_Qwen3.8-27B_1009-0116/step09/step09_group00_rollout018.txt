    H = np.fft.rfft(h, N)

                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-10)
        if accepted:
            hp = h
            h = xn
            fy = fx
            Lc = max(Lc * 0.9, 1.0)
            if it % 10 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                if fy < best_soft:
                    best_soft = fy
        else:
            h = y
            hp = y
            t = 1.0
            fy = _soft_max(*_pn(y), alpha)
            Lc = max(Lc * 0.5, 1.0)
        it += 1
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def _parametric_opt(n, M, deadline, rng):
    noise = rng.randn(n)
    noise -= noise.mean()
    noise /= np.std(noise)
    best_params = np.zeros(2 * M + 1)
    best_c = _c5(np.full(n, 0.5))
    x = np.arange(n)
    
    def eval_params(p):
        f = np.zeros(n)
        for m in range(M):
            f += p[2*m] * np.sin(np.pi * (m+1) * x / n + p[2*m+1])
        f += p[-1] * noise
        h = np.clip(0.5 + f, 0, 1)
        h = _proj(h, n * 0.5)
        return h
    
    # Random restarts in parameter space
    for restart in range(20):
        if time.time() > deadline:
            break
        p = np.zeros(2 * M + 1)
        for m in range(M):
            p[2*m] = rng.uniform(-0.3, 0.3)
            p[2*m+1] = rng.uniform(0, 2*np.pi)
        p[-1] = rng.uniform(0, 0.15)
        # Local optimization with Nelder-Mead style
        h = eval_params(p)
        c = _c5(h)
        step = 0.02
        dim = len(p)
        for it in range(3000):
            if time.time() > deadline:
                break
            best_p = p.copy()
            best_c_local = c
            for d in range(dim):
                for sgn in [1, -1]:
                    p[d] = best_p[d] + sgn * step
                    h2 = eval_params(p)
                    c2 = _c5(h2)
                    if c2 < best_c_local:
                        best_c_local = c2
                        p[d] = best_p[d] + sgn * step
                        best_p = p.copy()
                        p[d] = best_p[d]
            p = best_p
            c = best_c_local
            step *= 0.97
            if step < 1e-6:
                break
        h = eval_params(p)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_params = p.copy()
    h = eval_params(best_params)
    return h, best_c

def _sa_swap(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * (2.0 / n)
    best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 0.003 * (0.3) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        k = abs(k_act)
        if k > 0 and k < n:
            if k_act > 0:
                valid = (h[:n-k] > 0.01) & (h[k:] < 0.99)
                if valid.sum() > 1:
                    pos = np.nonzero(valid)[0]
                    i = int(pos[rng.randint(len(pos))])
                    j = i + k
                    d = rng.uniform(-0.05, 0.05)
                    if h[i]+d < 0 or h[j]-d < 0 or h[i]+d > 1 or h[j]-d > 1:
                        d = min(d, min(h[i],1-h[i],1-h[j],h[j])) * 0.5
                    if abs(d) > 1e-10:
                        h[i] += d; h[j] -= d
                        c2 = np.correlate(h, 1.0-h, mode='full')
                        mx2 = float(c2.max())
                        delta = mx2 - mx
                        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta/max(T,1e-15))):
                            mx = mx2; corr = c2
                            cn = mx * (2.0/n)
                            if cn < best_c:
                                best_c = cn; best_h = h.copy()
                        else:
                            h[i] -= d; h[j] += d
                        continue
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            continue
        d = rng.uniform(-0.04, 0.04)
        d = max(-h[i], min(d, 1-h[i], h[j], 1-h[j])) if d>0 else max(-h[i], min(d, h[i]-1, -h[j], h[j]-1))
        # Simpler: clamp
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            hl = max(0, min(1, hi))
            d = hl - h[i]
            h[i] += d; h[j] -= d
        else:
            h[i] += d; h[j] -= d
        c2 = np.correlate(h, 1.0-h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta/max(T,1e-15))):
            mx = mx2; corr = c2
            cn = mx * (2.0/n)
            if cn < best_c:
                best_c = cn; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

def _fine_polish(h0, deadline):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    step = 0.002
    while time.time() < deadline:
        P, Nv = _pn(h)
        vals = np.concatenate([P[:1], P[1:], Nv[1:]])
        mx = float(vals.max())
        tol = max(2e-6, 15.0 * (2.0/n))
        active = np.where(vals >= mx - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(vals))])
        for a in active:
            if time.time() > deadline:
                break
            lo = max(0, a - n + 1)
            hi = min(n - 1, a)
            m_idx = a - np.arange(lo, hi + 1)
            contrib = 1.0 - 2.0 * h[m_idx]
            # Move mass from high-contribution to low-contribution positions
            opts = np.where(np.abs(contrib) > 1e-4)[0]
            if len(opts) < 2:
                continue
            src = opts[np.argmax(contrib)] if contrib[opts].max() > 0 else opts[np.argmin(contrib)]
            dst = opts[np.argmin(contrib)] if contrib[opts].max() > 0 else opts[np.argmax(contrib)]
            if src == dst:
                continue
            si = lo + src
            di = lo + dst
            if si == di:
                continue
            amt = step * np.sign(contrib[src] - contrib[dst])
            if h[si] + amt < 0 or h[si] + amt > 1 or h[di] - amt < 0 or h[di] - amt > 1:
                amt *= 0.3
                if h[si] + amt < 0 or h[si] + amt > 1 or h[di] - amt < 0 or h[di] - amt > 1:
                    continue
            h[si] += amt
            h[di] -= amt
            c2 = _c5(h)
            if c2 < c - 1e-15:
                c = c2
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[si] -= amt
                h[di] += amt
        step *= 0.998
        if step < 1e-7:
            step = 1e-7
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, m * 0.5)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _proj(h0, n0 * 0.5)
    best = h0.copy()
    best_c = _c5(best)
    
    # Try symmetry variants
    for hh in (1.0 - h0.copy(),):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    
    # Phase 1: Parametric optimization (150s)
    end1 = t0 + min(150.0, 0.14 * budget_s)
    if time.time() < end1 - 2:
        h_p, c_p = _parametric_opt(n0, 8, end1, rng)
        if c_p < best_c:
            best = h_p
            best_c = c_p
    
    # Phase 2: FISTA from best (250s)
    end2 = end1 + min(250.0, 0.23 * budget_s)
    if time.time() < end2 - 2:
        alphas = [50, 300, 1500, 8000, 40000]
        hcur = best.copy()
        tt_per = max(3.0, (end2 - time.time()) / len(alphas))
        for a in alphas:
            if time.time() >= end2 - 1:
                break
            hf, cf = _fista_v2(hcur, a, time.time() + tt_per, 5.0)
            if cf < best_c:
                best = hf
                best_c = cf
            hcf = _c5(hf)
            if hcf < best_c:
                best = hf
                best_c = hcf
            hcur = hf
    
    # Phase 3: Multi-resolution (150s)
    end3 = end2 + min(150.0, 0.14 * budget_s)
    if time.time() < end3 - 3:
        for m in [280, 480, 640]:
            if time.time() >= end3 - 5:
                break
            hm = _resample(best, m)
            hf, cf = _fista_v2(hm, 2000, time.time() + min(30, end3 - time.time() - 2), 5.0)
            hb = _resample(hf, n0)
            cb = _c5(hb)
            if cb < best_c:
                best = hb
                best_c = cb
    
    # Phase 4: SA with swap moves (300s)
    end4 = end3 + min(300.0, 0.28 * budget_s)
    if time.time() < end4 - 2:
        best, best_c = _sa_swap(best, rng, end4)
    
    # Phase 5: Fine polish (150s)
    end5 = end4 + min(150.0, 0.14 * budget_s)
    if time.time() < end5 - 2:
        best, best_c = _fine_polish(best, end5)
    
    # Phase 6: Final tiny SA
    end6 = t0 + budget_s - 3.0
    if time.time() < end6:
        best, best_c = _sa_swap(best, rng, end6)
    
    # Final symmetry check
    for hh in (1.0 - best,):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    
    best = _proj(best, n0 * 0.5)
    best_c = _c5(best)
    return best, best_c, n0
