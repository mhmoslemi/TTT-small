    h = np.clip(h, 0.0, 1.0).copy()

        h *= target / s

            else:
                k2 = -k
                g[:n - k2] += wk * (1.0 - h[k2:])
                g[k2:] -= wk * h[:n - k2]
        v = 0.8 * v + 0.2 * g
        sq = 0.9 * sq + 0.1 * g * g
        direction = -(v / (np.sqrt(sq) + 1e-8))
        taken = False
        for _t in range(10):
            h_new = _proj(h + step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 20.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.3, step0 * 20.0)
    return _repair(h, n)

def _two_point_search(h, best_c5, t_start, t_end):
    n = len(h)
    c = _corr(h)
    c5c = float(c.max())
    best_h = h.copy()
    best_c = best_c5
    rng = np.random.RandomState(77)
    no_improve = 0
    n_active = 12

    while time.time() < t_end:
        # Recompute gradient periodically or when changed
        active = np.argsort(c)[::-1][:n_active]
        g = _grad_active(h, active)
        gp = g - g.mean()
        top_i = np.argsort(gp)[::-1][:10]
        bot_j = np.argsort(gp)[:10]

        found = False
        for _attempt in range(30):
            if rng.rand() < 0.25:
                ii = int(rng.randint(n))
                jj = int(rng.randint(n))
            else:
                ii = int(top_i[rng.randint(len(top_i))])
                jj = int(bot_j[rng.randint(len(bot_j))])
            if ii == jj:
                continue
            d_max = min(1.0 - h[ii], h[jj])
            if d_max < 1e-9:
                continue
            # Try multiple step sizes
            gd = gp[ii] - gp[jj]
            if gd <= 0 and rng.rand() > 0.1:
                continue
            base_d = d_max * 0.3
            if abs(gd) > 1e-6:
                base_d = min(base_d, 0.5 * d_max / max(1.0, abs(gd) * n * 0.5))
            for d in [base_d * 0.3, base_d, base_d * 0.6, base_d * 1.5]:
                if d <= 0 or d > d_max:
                    continue
                h[ii] += d
                h[jj] -= d
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    c = _corr(h)
                    found = True
                    break
                else:
                    h[ii] -= d
                    h[jj] += d
                    break
            if found:
                no_improve = 0
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
                break

        if not found:
            no_improve += 1
            if no_improve > 5000:
                # Escape: random perturbation
                idx = rng.randint(n, size=4)
                d = 0.003 * rng.randn(4)
                d -= d.mean()
                tmp = h.copy()
                tmp[idx] += d
                if tmp.min() >= 0 and tmp.max() <= 1:
                    h = _repair(tmp, n)
                    c = _corr(h)
                    c5c = float(c.max())
                no_improve = 0

    return best_h, best_c

def _fine_polish(h, best_c5, t_start, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = min(best_c5, c5c)
    rng = np.random.RandomState(31337)

    while time.time() < t_end:
        i = int(rng.randint(n))
        j = (i + 1 + int(rng.randint(0, 12))) % n
        d_max = 0.5 * min(1.0 - h[i], h[j])
        if d_max < 1e-10:
            continue
        for d in [d_max * 0.1, d_max * 0.5, d_max]:
            h[i] += d
            h[j] -= d
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with increasing alpha
    dl1 = t0 + 150
    for alpha, ml in [(5000, 35), (50000, 60), (500000, 90), (5000000, 120)]:
        if time.time() > dl1:
            break
        h_new = _slsqp_opt(best_h, alpha, ml, 600)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Projected gradient
    dl2 = t0 + 220
    if time.time() < dl2:
        for alpha, ml, st, s0 in [(10000, 40, 300, 0.008), (100000, 60, 200, 0.004)]:
            if time.time() > dl2:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, st, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: SLSQP with high alpha
    dl3 = t0 + 300
    if time.time() < dl3:
        for alpha, ml in [(20000000, 100), (100000000, 130)]:
            if time.time() > dl3:
                break
            h_new = _slsqp_opt(best_h, alpha, ml, 400)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Main 2-point search
    dl4 = t0 + budget_s - 40
    best_h, best_c5 = _two_point_search(best_h, best_c5, t0, dl4)

    # Phase 5: Fine polish
    best_h, best_c5 = _fine_polish(best_h, best_c5, t0, t0 + budget_s - 1)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
