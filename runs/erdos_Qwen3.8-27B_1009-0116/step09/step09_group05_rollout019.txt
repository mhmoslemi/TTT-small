        if np.clip(h - mid, 0, 1).sum() > target:

def _smooth_obj(x, alpha, m_lags):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / len(x))
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp_fn(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP',
                       args=(alpha, m_lags), bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    p = int(np.argmax(c_full))
    L = p - (n - 1)

        if 0 <= x + L < n:

        if 0 <= x - L < n:
