    h=np.asarray(h,dtype=float).copy()
    for _ in range(3):
       diff = target - h.sum()
       if abs(diff) < 1e-12: break
       if diff > 0:

              diff -= add

def _exact_sum(h, t):
    h = np.asarray(h, float).copy()
    for _ in range(3):
        d = t - h.sum()
        if abs(d) < 1e-12:
            break
        o = np.argsort(1.0 - h if d > 0 else h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a > 0:
                    h[i] += a
                    d -= a
            else:
                a = min(-d, h[i])
                if a > 0:
                    h[i] -= a
                    d += a
            if abs(d) < 1e-12:
                break
    return h


def _clip_sum(v, t):
    v = np.clip(np.asarray(v, float), 0.0, 1.0).copy()
    if abs(v.sum() - t) < 1e-12:
        return _exact_sum(v, t)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        m = 0.5 * (lo + hi)
        if np.clip(v - m, 0.0, 1.0).sum() > t:
            lo = m
        else:
            hi = m
    return _exact_sum(np.clip(v - 0.5 * (lo + hi), 0.0, 1.0), t)


def _corr_full(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _evalmax(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))


def _deltas(md):
    out = []
    for d in (5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3, 5e-3,
              1e-2, 2e-2, 5e-2, 1e-1, 2e-1, 0.5 * md, md):

            d = md
        if not out or abs(d - out[-1]) > 1e-7:
            out.append(float(d))
    out.sort()
    return out

        try:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=t,
                          bounds=bnds, method="highs", options={"presolve": True})
        except Exception:
            res = None

        if res is None or not res.success:
            stk += 1
        else:
            xl = np.clip(res.x[:n], 0.0, 1.0)
            bv = cm
            nh = None
            for th in (0.05, 0.1, 0.2, 0.35, 0.6, 1.0):
                hyp = _clip_sum(h + th * (xl - h), t)
                v = _evalmax(hyp)
                if v < bv - 1e-14:
                    bv, nh = v, hyp
            if nh is not None:
                h = nh.copy()
                c = _corr_full(h)
                cm = float(c.max())
                if cm < bc - 1e-13:
                    bc, bh = cm, h.copy()
                stk = 0
            else:
                stk += 1

        if stk > 12:
            amp = 0.015 + 0.01 * min(1.0, float(stk) / 30.0)
            h = _clip_sum(bh + amp * rng.randn(n), t)
            c = _corr_full(h)
            cm = float(c.max())
            stk = 0

    bh = _exact_sum(bh, t)
    bc = _evalmax(bh)
    return bh, bc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        base = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        base = _fallback_h(560)

    n = base.size
    if n < 16:
        n = 560
        base = _fallback_h(n)
    target = 0.5 * n

    best_h = _clip_sum(base, target)
    best_c = _evalmax(best_h)

    step = max(1, n // 20)
    for s in range(0, n, step):
        hyps = (np.roll(best_h, s),) if s == 0 else (np.roll(best_h, s), np.roll(best_h, -s))
        for hyp in hyps:
            v = _evalmax(hyp)
            if v < best_c - 1e-14:
                best_c, best_h = v, hyp.copy()

    for g in (0.9, 0.95, 1.05, 1.1, 1.2, 1.35):
        hyp = _clip_sum(0.5 + g * (best_h - 0.5), target)
        v = _evalmax(hyp)
        if v < best_c - 1e-14:
            best_c, best_h = v, hyp.copy()

    for g in (0.6, 0.8, 1.3, 1.6):
        hyp = _clip_sum(0.5 + g * np.tanh(2.0 * (best_h - 0.5)), target)
        v = _evalmax(hyp)
        if v < best_c - 1e-14:
            best_c, best_h = v, hyp.copy()

    kw = max(2, n // 40)
    for amp in (0.02, 0.05):
        noise = np.convolve(rng.randn(n), np.ones(kw) / kw, mode="same")
        hyp = _clip_sum(best_h + amp * noise, target)
        v = _evalmax(hyp)
        if v < best_c - 1e-14:
            best_c, best_h = v, hyp.copy()

    total = max(1.0, budget_s - 2.0)
    sc = max(0.02, min(1.0, budget_s / 1100.0))
    p3 = t0 + budget_s - 0.5
    p1 = min(t0 + 120.0 * sc, t0 + 0.15 * total, p3 - 1.0)
    l1 = min(t0 + 250.0 * sc, t0 + 0.35 * total, p3 - 1.0)
    p2 = min(t0 + 600.0 * sc, t0 + 0.58 * total, p3 - 1.0)
    l2 = min(t0 + 760.0 * sc, t0 + 0.72 * total, p3 - 1.0)

    p1 = min(p1, l1 - 0.1)
    l1 = min(l1, p2 - 0.1)
    p2 = min(p2, l2 - 0.1)
    l2 = min(l2, p3 - 0.1)

    if p1 > time.time():
        best_h, best_c = _pair_polish(best_h, best_c, p1, rng, 16, 3)
    if l1 > time.time():
        best_h, best_c = _lp_stage(best_h, best_c, l1, rng)
    if p2 > time.time():
        best_h, best_c = _pair_polish(best_h, best_c, p2, rng, 24, 3)
    if l2 > time.time():
        best_h, best_c = _lp_stage(best_h, best_c, l2, rng)
    best_h, best_c = _pair_polish(best_h, best_c, p3, rng, 36, 4)

    best_h = np.clip(np.asarray(best_h, dtype=float), 0.0, 1.0)
    best_h = _exact_sum(best_h, 0.5 * best_h.size)
    best_c = _evalmax(best_h)
    return best_h, float(best_c), int(best_h.size)
