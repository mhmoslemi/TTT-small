    return best_h, best_c5

def _down(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _up(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _elapsed():
        return time.time() - t0

    # Phase 1: Multi-resolution warm-up
    for ratio in [4, 2]:
        if _elapsed() > 40:
            break
        nc = max(40, n // ratio)
        hc = _down(h, n // nc)
        for p_c in [8, 64, 512]:
            if _elapsed() > 55:
                break
            hc, _ = _pg_opt(hc, p_c, min(20, nc // 4), 400, 0.03, 0.75, t0 + _elapsed() + 15)
        hu = _up(hc, n // nc)
        cu = _eval(hu)
        if cu < best_c5:
            best_c5 = cu
            best_h = hu.copy()
            h = hu.copy()

    # Phase 2: Power-mean PG with increasing p and m
    pm_configs = [
        (4, 20, 1200, 0.02, 0.7),
        (8, 40, 1000, 0.018, 0.7),
        (16, 60, 800, 0.015, 0.72),
        (32, 80, 700, 0.012, 0.74),
        (128, 100, 600, 0.009, 0.76),
        (512, 120, 500, 0.006, 0.78),
        (2048, 140, 400, 0.004, 0.8),
        (8192, 160, 350, 0.003, 0.82),
        (32768, 180, 300, 0.002, 0.85),
    ]
    t_p2 = t0 + budget_s * 0.55
    for p, m, steps, lr0, mom in pm_configs:
        if _elapsed() > budget_s * 0.5:
            break
        h_new, c_new = _pg_opt(h, p, m, steps, lr0, mom, t_p2)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Active-set descent on top lags
    t_p3 = t0 + budget_s * 0.75
    active_configs = [
        (3, 800, 0.004),
        (5, 600, 0.003),
        (8, 500, 0.002),
        (12, 400, 0.0015),
        (20, 300, 0.001),
    ]
    for nl, steps, lr in active_configs:
        if _elapsed() > budget_s * 0.7:
            break
        h_new, c_new = _active_descent(h, nl, steps, lr, t_p3)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Coordinate descent polish (most important for last digits)
    t_p4 = t0 + budget_s * 0.95
    h = best_h.copy()
    h, best_c5 = _coord_ds(h, 2000000, rng, t_p4, 2e-5)
    best_h = h.copy()

    # Phase 5: Ultra-fine 2-point polish
    t_p5 = t0 + budget_s * 0.99
    h = best_h.copy()
    n5 = len(h)
    c5c = _eval(h)
    step_f = 1e-6
    for _ in range(500000):
        if time.time() > t_p5:
            break
        i = rng.randint(n5)
        j = (i + 1 + rng.randint(0, 12)) % n5
        d = step_f * rng.randn()
        hi = h[i] + d
        hj = h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c_new = _eval(h)
            if c_new < c5c - 1e-17:
                c5c = c_new
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h.copy()
                step_f = min(step_f * 1.01, 5e-5)
            else:
                h[i], h[j] = hi - d, hj + d
                step_f *= 0.85
    # Final repair and evaluate
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
