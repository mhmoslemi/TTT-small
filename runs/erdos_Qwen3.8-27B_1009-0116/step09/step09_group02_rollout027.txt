In the 'full' output, result is indexed from 0 to 2n-2, where:

So result[kk] corresponds to k' = kk - (n-1) = L.

If L >= 0: result[kk] = sum_{j=0}^{n-1-L} a[j+L] * v[j] = sum_{j=0}^{n-1-L} h[j+L] * (1-h[j])

For L >= 0:

        if h[i1] < 0 or h[i1] > 1 or h[i2] < 0 or h[i2] > 1:
            h[i1] -= d
            h[i2] += d
            continue
        c_new = _eval(h)
        if c_new < c - 1e-18:
            c = c_new
            best = h.copy()
            best_c = c
        else:
            h[i1] -= d
            h[i2] += d
    return best, best_c

def _group_polish(h, target, n_iter, gs, rng, t0, tlimit):
    n = len(h)
    best = h.copy()
    best_c = _eval(h)
    c = best_c
    for i in range(n_iter):
        if time.time() - t0 > tlimit:
            break
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 30, size=gs)) % n)
        d = (1e-9 + 3e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        c_new = _eval(h_new)
        if c_new < c - 1e-18:
            c = c_new
            h = h_new
            best = h.copy()
            best_c = c
    return best, best_c

def _perturb_restart(h, target, n, rng, t0, tlimit):
    x = h.copy()
    best = h.copy()
    best_c = _eval(h)
    n_rounds = 12
    for r in range(n_rounds):
        if time.time() - t0 > tlimit:
            break
        scale = 0.008 * (1.0 - r / n_rounds)
        x_p = x + scale * rng.randn(n)
        x_p -= x_p.mean()
        x_p = _repair(x_p, n)
        x_p, c_p = _nesterov_loop(x_p, target, 8000, 3, 0.002, 0.0003, 0.8, t0, tlimit)
        if c_p < best_c:
            best_c = c_p
            best = x_p.copy()
        x = x_p
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov top-3
    best_h, best_c5 = _nesterov_loop(best_h, target, 50000, 3, 0.004, 0.0004, 0.85, t0, 100)

    # Phase 2: Nesterov top-5
    best_h, best_c5 = _nesterov_loop(best_h, target, 40000, 5, 0.0025, 0.0003, 0.75, t0, 250)

    # Phase 3: Nesterov top-8
    best_h, best_c5 = _nesterov_loop(best_h, target, 30000, 8, 0.0015, 0.0002, 0.7, t0, 380)

    # Phase 4: Perturb and restart
    best_h, best_c5 = _perturb_restart(best_h, target, n, rng, t0, 600)

    # Phase 5: Nesterov top-2 (tight)
    best_h, best_c5 = _nesterov_loop(best_h, target, 30000, 2, 0.001, 0.0001, 0.8, t0, 720)

    # Phase 6: Gradient-guided polish
    best_h, best_c5 = _grad_guided_polish(best_h, target, 200000, 3, t0, budget_s - 120)

    # Phase 7: Random 2-point polish
    best_h, best_c5 = _two_point_polish(best_h, target, 800000, rng, t0, budget_s - 30)

    # Phase 8: Group polish
    best_h, best_c5 = _group_polish(best_h, target, 200000, 3, rng, t0, budget_s - 8)

    # Phase 9: Final ultra-fine 2-point
    best_h, best_c5 = _two_point_polish(best_h, target, 600000, rng, t0, budget_s - 2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
