def _c5(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:

            if d > 0:

        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _gW(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)

def _fista(h0, a, tl, L=8.0):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    fh = _sval(h, a)
    bh = h.copy()
    bc = _c5(h)
    Lc = L
    for it in range(500000):
        if time.time() - t0 > tl:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            if y.min() < -1e-5 or y.max() > 1 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
        else:
            y = h
        g = _sg(y, a)
        step = 1.0 / max(Lc, 1e-12)
        ok = False
        for _ in range(10):
            xn = _proj(y - step * g, S)
            fx = _sval(xn, a)
            d = xn - y
            rhs = fh + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                ok = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp, h = h, xn
            fh = fx
            Lc = max(Lc * 0.8, 1.0)
            if it % 5 == 0:
                c = _c5(h)
                if c < bc:
                    bc, bh = c, h.copy()
        else:
            hp, y, tv = h, h, 1.0
            fh = _sval(h, a)
            Lc = max(Lc * 0.5, 1.0)
    c = _c5(h)
    if c < bc:
        bc, bh = c, h.copy()
    return bh, bc

def _penalties(h, ovl, tol=2.0):
    n = len(h)
    mx = ovl.max()
    active = (ovl >= mx - tol).astype(float)
    N = 1
    while N < 4 * n:
        N <<= 1
    A = np.zeros(N)
    A[:2*n-1] = active
    B1 = np.zeros(N)
    B1[:n] = 1.0 - h
    B2 = np.zeros(N)
    B2[:n] = h
    FA = np.fft.rfft(A)
    xc1 = np.fft.irfft(np.fft.rfft(B1) * np.conj(FA), N)
    xc2 = np.fft.irfft(np.fft.rfft(B2) * np.conj(FA), N)
    idx = (np.arange(n) - (n - 1)) % N
    return xc1[idx], xc2[idx]

def _binary_ls(h, rng, t_limit, tol=2.0):
    t0 = time.time()
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    S = n // 2
    cnt = int(h.sum())
    if cnt > S:
        ones = np.where(h == 1.0)[0]
        h[rng.choice(ones, cnt - S, replace=False)] = 0
    elif cnt < S:
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(zeros, S - cnt, replace=False)] = 1
    c = _c5(h)
    bc = c
    bh = h.copy()
    st = 0
    while time.time() - t0 < t_limit:
        ovl = np.correlate(h, 1.0 - h, mode='full')
        p1, p0 = _penalties(h, ovl, tol)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 5 or len(zeros) < 5:
            break
        src_score = p1[ones]
        tgt_score = p0[zeros]
        top_k = max(5, len(ones) // 8)
        bot_k = max(5, len(zeros) // 8)
        top_src = ones[np.argsort(src_score)[-top_k:]]
        bot_tgt = zeros[np.argsort(tgt_score)[:bot_k]]
        i = int(rng.choice(top_src))
        j = int(rng.choice(bot_tgt))
        h[i], h[j] = 0.0, 1.0
        cn = _c5(h)
        if cn < c - 1e-12:
            c = cn
            st = 0
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            st += 1
            if st > 2000:
                break
    return bh, bc

def _sa_multi(h, rng, t_limit):
    t0 = time.time()
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    S = n // 2
    cnt = int(h.sum())
    if cnt > S:
        ones = np.where(h == 1.0)[0]
        h[rng.choice(ones, cnt - S, replace=False)] = 0
    elif cnt < S:
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(zeros, S - cnt, replace=False)] = 1
    c = _c5(h)
    bc = c
    bh = h.copy()
    total = t_limit
    while time.time() - t0 < t_limit:
        fr = (time.time() - t0) / total
        T = 0.005 * np.exp(-4.0 * fr)
        ns = 1 if rng.random() < 0.6 else (2 if rng.random() < 0.7 else 3)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < ns or len(zeros) < ns:
            break
        sw_o = rng.choice(ones, ns, replace=False)
        sw_z = rng.choice(zeros, ns, replace=False)
        h[sw_o] = 0.0
        h[sw_z] = 1.0
        cn = _c5(h)
        d = cn - c
        if d < -1e-12 or (d > 0 and rng.random() < np.exp(-d / max(T, 1e-9))):
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[sw_o] = 1.0
            h[sw_z] = 0.0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    best = _repair(h0, n)
    best_c = _c5(best)
    # Symmetries
    for v in [1.0 - best, best[::-1]]:
        hv = _proj(v, n * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv
    # Phase 1: Binary local search (300s)
    end1 = t0 + 300.0
    bh, bc = _binary_ls(best, rng, min(295.0, end1 - time.time()), tol=2.0)
    if bc < best_c:
        best, best_c = bh, bc
    # Phase 2: Simulated annealing multi-swap (250s)
    end2 = t0 + 550.0
    bh, bc = _sa_multi(best, rng, min(245.0, end2 - time.time()))
    if bc < best_c:
        best, best_c = bh, bc
    # Phase 3: FISTA smooth polish (250s)
    end3 = t0 + 800.0
    rem = end3 - time.time()
    if rem > 10:
        for a in [200, 1000, 5000]:
            if rem < 5:
                break
            bf, cf = _fista(best, a, rem * 0.3)
            if cf < best_c:
                best, best_c = bf, cf
            rem = end3 - time.time()
    # Phase 4: Final binary local search (200s)
    end4 = t0 + 1000.0
    bh, bc = _binary_ls(best, rng, min(195.0, end4 - time.time()), tol=1.0)
    if bc < best_c:
        best, best_c = bh, bc
    # Phase 5: Micro SA (50s)
    end5 = t0 + budget_s - 5.0
    bh, bc = _sa_multi(best, rng, min(45.0, end5 - time.time()))
    if bc < best_c:
        best, best_c = bh, bc
    # Final checks
    for v in [1.0 - best, best[::-1]]:
        hv = _proj(np.clip(v, 0, 1), n * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv
    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
