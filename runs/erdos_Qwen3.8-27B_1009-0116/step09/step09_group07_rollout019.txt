def _lp_solve(h, n, solver='CLARABEL'):
    import cvxpy as cp
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # Compute correlation coefficients
    nm1 = n - 1
    for k in range(2*n - 1):
        L = k - nm1  # lag
        if L >= 0:
            # c_L = sum_{i=0}^{n-1-L} h[i] * (1-h[i+L]) * dx
            coeff = np.zeros(n)
            coeff[:n-L] = (1.0 - h[L:]) * dx  # coefficients for h[0],...,h[n-1-L]
        else:
            M = -L
            # c_L = sum_{i=M}^{n-1} h[i] * (1-h[i-M]) * dx
            coeff = np.zeros(n)
            coeff[M:] = (1.0 - h[:-M]) * dx  # coefficients for h[M],...,h[n-1]
        constraints.append(coeff @ h_var <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver)
    return h_var.value, t.value

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    p1 = t0 + budget_s * 0.32
    while time.time() < p1:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    p2 = t0 + budget_s * 0.63
    h_lp = best_h.copy()
    no_imp = 0
    while time.time() < p2:
        h_new = _lp_step(h_lp, n, 200)
        improved = False
        for omega in [1.0, 0.7, 0.5, 0.3]:
            h_try = _repair((1.0 - omega) * best_h + omega * h_new, n)
            c5t = _eval(h_try)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = h_try.copy()
                h_lp = h_try.copy()
                improved = True
                no_imp = 0
                break
        if not improved:
            no_imp += 1
            scale = 0.05 if no_imp > 20 else 0.015
            h_lp = _repair(best_h + scale * rng.randn(n), n)
            if no_imp > 40:
                no_imp = 0
        if time.time() < p2 - 0.6:
            hc = _cd_pass(h_lp, rng)
            hc = _repair(hc, n)
            c5c = _eval(hc)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
                h_lp = hc.copy()
                no_imp = 0

    p3 = t0 + budget_s * 0.84
    h = best_h.copy()
    while time.time() < p3:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if time.time() < t0 + budget_s * 0.93:
        for _ in range(5):
            if time.time() > t0 + budget_s * 0.93:
                break
            h_new = _lp_step(best_h, n, 300)
            for omega in [1.0, 0.5, 0.3]:
                h_try = _repair((1.0 - omega) * best_h + omega * h_new, n)
                c5t = _eval(h_try)
                if c5t < best_c5:
                    best_c5 = c5t
                    best_h = h_try.copy()
                    break
            hcd = _cd_pass(best_h, rng)
            hcd = _repair(hcd, n)
            c5c = _eval(hcd)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hcd.copy()

    if time.time() < t0 + budget_s - 3:
        rem = t0 + budget_s - 5 - time.time()
        ntr = min(int(rem / 0.003), 200000)
        if ntr > 0:
            best_h, best_c5 = _local_search(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
