def _proj(h, target):

def _lse_grad(h, alpha, ntop, n):

def _sg(x, alpha, ntop, n):
    dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full') * dx
    idx = np.argpartition(c, -ntop)[-ntop:]

    m = v.max()

    w = np.exp(s)

    g = np.zeros(n)

            g[:n-L] -= wk*x[L:]

            g[M:] -= wk*x[:n-M]

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _proj(h, target):

        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:

        else:
            hi = mid

def _sg_calc(x, alpha, ntop, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argpartition(c, -ntop)[-ntop:]

    lo = max(0.0, s - 1.0)
