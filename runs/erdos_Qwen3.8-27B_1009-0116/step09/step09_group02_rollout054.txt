Actually, Adam with projection is tricky because the projection changes the effective step. Let me keep it simpler.

Let me just go with a clean, well-tuned version that builds on the previous code's strengths but with some key improvements:

1. Better alpha schedule (more values, finer control)
2. Better local search (gradient-directed, multiple scales)
3. A "reheat" step to escape local minima
4. More aggressive fine polishing

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp_inner(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    top = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - top), -400, 400)
    s -= s.max()
    return top + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_full(x)
    top = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - top), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        wk = w[ii] * dx
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp_run(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target_val = n * 0.5
    cons_fun = _cons_func
    cons_jac = _cons_jac
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (target_val,)}]
    try:
        res = minimize(_slsqp_inner, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _cons_func(x, target):
    return x.sum() - target

def _cons_jac(x, target):
    return np.ones(len(x))

def _pg_opt(h, alpha, m_lags, n_iter, step0):
    n = len(h)
    target = n * 0.5
    cur_c = _eval(h)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    for it in range(n_iter):
        g = _slsqp_grad(h, alpha, m_lags)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mu * v + (1.0 - mu) * (g / gn)
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        taken = False
        for _ in range(10):
            h_try = _proj(h - step * v, target)
            c_try = _eval(h_try)
            if c_try < cur_c - 1e-17:
                h = h_try
                cur_c = c_try
                step = min(step * 1.1, step0 * 40.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(max(step * 1.3, step0 * 0.3), step0 * 40.0)
    return _repair(h, n), cur_c

def _directed_search(h, n_iter, rng, top_k=6):
    n = len(h)
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    dx = 2.0 / n
    for _ in range(n_iter):
        c = _corr_full(h)
        top_idx = np.argsort(c)[-top_k:]
        # Weighted gradient over top lags
        top_vals = c[top_idx]
        top_max = top_vals.max()
        scale = 0.01
        weights = np.exp(-scale * (top_max - top_vals) / max(top_max, 1e-15))
        weights /= weights.sum()
        
        g = np.zeros(n)
        for ii in range(len(top_idx)):
            if weights[ii] < 1e-6:
                continue
            L = top_idx[ii] - (n - 1)
            wv = weights[ii] * dx
            if L >= 0:
                g[L:] += wv * (1.0 - h[:n - L])
                g[:n - L] -= wv * h[L:]
            else:
                M = -L
                g[:n - M] += wv * (1.0 - h[M:])
                g[M:] -= wv * h[:n - M]
        g -= g.mean()
        
        gs = np.max(np.abs(g))
        if gs < 1e-15:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 30)) % n
        else:
            thresh = 0.03 * gs
            i_c = np.where(g > thresh)[0]
            j_c = np.where(g < -thresh)[0]
            if len(i_c) < 1 or len(j_c) < 1:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 30)) % n
            else:
                i = i_c[rng.randint(len(i_c))]
                j = j_c[rng.randint(len(j_c))]
        
        step = 1e-6 + 4e-6 * rng.rand()
        oi, oj = h[i], h[j]
        ni, nj = oi - step, oj + step
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            ct = _eval(h)
            if ct < cur_c - 1e-17:
                cur_c = ct
                if ct < best_c:
                    best_c = ct
                    best_h = h.copy()
            else:
                h[i], h[j] = oi, oj
    return best_h, best_c
