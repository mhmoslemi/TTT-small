            p = -l
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g

def _slsqp_fn(x, n, alpha, k):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argpartition(-c, k)[:k]
    m = c[top].max()
    s = np.clip(alpha * (c[top] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, n, alpha, k):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argpartition(-c, k)[:k]
    m = c[top].max()
    s = np.clip(alpha * (c[top] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    idxs = top
    return _grad_w(x, n, idxs, w, dx)

def _eq_fn(x, half):
    return x.sum() - half

def _eq_jac(x, half):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, k, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fn, 'jac': _eq_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_fn, h, jac=_slsqp_grad, args=(len(h), alpha, k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 40})
        if res.success or res.fun < 1e6:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _pg_loop(h, n, rng, steps, lr0, mom, k, t0, deadline):
    dx = 2.0 / n
    best_c = _eval(h)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr0
    for it in range(steps):
        if time.time() > deadline:
            break
        c_arr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_val = c_arr.max()
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        top = np.argpartition(-c_arr, k)[:k]
        g = _grad_w(h, n, top, np.ones(k), dx)
        v = mom * v - lr * g
        h_new = _project(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new > c_val * 1.002 + 1e-8:
            lr *= 0.7
            h_new = _project(h - lr * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new > c_val:
                h_new = _project(h - 0.5 * lr * g, n * 0.5)
        else:
            lr = min(lr * 1.03, lr0 * 30.0)
        h = h_new
    return _repair(best_h, n), best_c

def _coord_descent(h, n, rng, n_iter, t0, deadline):
    c5_cur = _eval(h)
    best_c = c5_cur
    best_h = h.copy()
    dx = 2.0 / n
    c_arr = np.correlate(h, 1.0 - h, mode='full') * dx
    for it in range(n_iter):
        if time.time() > deadline:
            break
        top = np.argpartition(-c_arr, 10)[:10]
        lags = top - (n - 1)
        g = _grad_w(h, n, top, np.ones(10), dx)
        i = int(np.argmax(np.abs(g)))
        j = int(np.argmin(np.abs(g)))
        if i == j:
            j = (j + 1) % n
        d = 5e-5 * rng.randn()
        h2 = h.copy()
        h2[i] += d
        h2[j] -= d
        if h2.min() < -1e-12 or h2.max() > 1 + 1e-12:
            d *= -0.4
            h2 = h.copy()
            h2[i] += d
            h2[j] -= d
            if h2.min() < 0 or h2.max() > 1:
                continue
        h2 = _repair(h2, n)
        c5n = _eval(h2)
        if c5n < c5_cur:
            h = h2
            c5_cur = c5n
            c_arr = np.correlate(h, 1.0 - h, mode='full') * dx
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    deadline = t0 + budget_s

    # Phase 1: Coarse multi-resolution warm-up
    for ratio in [4, 3]:
        if _el() > 40:
            break
        nc = max(40, n // ratio)
        hc = _downsample(h, n // nc)
        for alpha_c in [20000, 200000, 2000000]:
            if _el() > 55:
                break
            hc = _slsqp_opt(hc, alpha_c, 30, 400, 1e-14)
        h_up = _upsample(hc, n // nc)
        c_new = _eval(h_up)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_up.copy()

    # Phase 2: SLSQP sweep at full resolution (moderate alpha)
    alpha_seq = [10000, 100000, 1000000, 10000000, 100000000]
    for alpha in alpha_seq:
        if _el() > budget_s * 0.30:
            break
        k = min(60, max(20, int(20 * np.log10(alpha / 10000.0) + 20)))
        h_new = _slsqp_opt(best_h, alpha, k, 1500, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient with momentum (adaptive active set)
    pg_configs = [
        (30, 0.005, 0.7, 3000),
        (50, 0.003, 0.8, 2500),
        (80, 0.002, 0.85, 2000),
        (120, 0.001, 0.9, 1500),
    ]
    for k, lr0, mom, steps in pg_configs:
        if _el() > budget_s * 0.55:
            break
        h_new, c5n = _pg_loop(best_h.copy(), n, rng, steps, lr0, mom, k, t0, t0 + budget_s * 0.55)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: High-alpha SLSQP (squeeze the max)
    for alpha in [500000000, 5000000000, 50000000000]:
        if _el() > budget_s * 0.70:
            break
        h_new = _slsqp_opt(best_h, alpha, 150, 1000, 1e-17)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Reheat + re-optimize
    for att in range(4):
        if _el() > budget_s * 0.80:
            break
        ns = 5e-5 * (4 - att)
        h_pert = best_h + ns * rng.randn(n)
        h_pert = _repair(h_pert, n)
        h_new = _slsqp_opt(h_pert, 100000000, 80, 600, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 6: Coordinate descent polish
    if _el() < budget_s - 30:
        nt = int(min((budget_s - 30 - _el()) / 0.002, 500000))
        best_h, best_c5 = _coord_descent(best_h, n, rng, nt, t0, t0 + budget_s - 5)

    # Phase 7: Ultra-fine 2-point random polish
    if _el() < budget_s - 3:
        h2 = best_h.copy()
        c5c = _eval(h2)
        niter = int(min((budget_s - 3 - _el()) / 0.0006, 400000))
        for _ in range(niter):
            if time.time() > deadline - 2:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = 3e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    # Try also at half resolution in case it's better
    if n > 200:
        nc = n // 2
        hc = _downsample(best_h, 2)
        for alpha_c in [100000000, 1000000000]:
            if time.time() > deadline - 5:
                break
            hc = _slsqp_opt(hc, alpha_c, 40, 500, 1e-15)
        hc = _repair(hc, nc)
        hc_c5 = _eval(hc)
        if hc_c5 < best_c5:
            best_h = hc
            best_c5 = hc_c5
            n = nc

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
