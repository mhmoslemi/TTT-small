from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_active(h, m_lags, alpha=1.0):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    if alpha > 1:
        m = c.max()
        top = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[top] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
    else:
        top = np.argsort(c)[-m_lags:]
        w = np.ones(len(top)) / len(top)
    g = np.zeros(n)
    for ii, k in enumerate(top):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    def obj(x):
        c = _corr(x)
        m = c.max()
        top = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[top] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        return _grad_active(x, m_lags, alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, steps, step0):
    n = len(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0

        v = mu * v + (1.0 - mu) * g

def _fine_polish(h, n_iter, rng):
