_SLP_ALPHA = 1.0
_SLP_MLAGS = 1
_SLP_N = 0

def _repair(h, n):

            order = np.argsort(1.0 - h)
        else:

        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(np.max(_corr(h)))

def _slsqp_obj(x):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-_SLP_MLAGS:]
    s = np.clip(_SLP_ALPHA * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / _SLP_ALPHA

def _slsqp_grad(x):
    n = _SLP_N
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-_SLP_MLAGS:]
    s = np.clip(_SLP_ALPHA * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)

        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_con_fun(x):
    return x.sum() - _SUM_N * 0.5

def _sum_con_jac(x):
    return np.ones(len(x))

def _do_slsqp(h, alpha, m_lags, maxiter, ftol):
    global _SLP_ALPHA, _SLP_MLAGS, _SLP_N, _SUM_N
    n = len(h)
    _SLP_ALPHA = alpha
    _SLP_MLAGS = m_lags
    _SLP_N = n
    _SUM_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _active_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-K:]

        else:
            M = -L
            g[:n - M] += dx * (1.0 - h[M:])
            g[M:] -= dx * h[:n - M]
    return g - g.mean()

def _proj_grad(h, K, n_iter, step0):

    v = np.zeros(n)
    step = step0

        ng = np.linalg.norm(g)
        if ng < 1e-18:
            break
