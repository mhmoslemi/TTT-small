def _coord_sweep(h, n, n_pts=12):
    """One full coordinate sweep. Returns new h and c5."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    S = n * 0.5
    for i in range(n):  # or random order
        old = h[i]
        # Precompute coeff for this i
        coeff = np.zeros(2*n-1)
        # For each lag k (from -(n-1) to n-1), compute coefficient of delta
        # Index in corr: m = n-1+lag
        for lag in range(-(n-1), n):
            m = n - 1 + lag
            c = 0.0
            if lag == 0:
                pass  # handled separately (quadratic)
            elif lag > 0:
                if i <= n - 1 - lag:
                    c += (1.0 - h[i + lag])
                if i - lag >= 0:
                    c -= h[i - lag]
            else:
                L = -lag
                if i >= L:
                    c += (1.0 - h[i - L])
                if i + L < n:
                    c -= h[i + L]
            coeff[m] = c
        # Evaluate at n_pts values
        deltas = np.linspace(-old, 1.0-old, n_pts)
        # Also include 0
        best_delta = 0.0
        best_val = float(corr.max())
        for d in deltas:
            new_corr = corr + coeff * d
            new_corr[n-1] -= (2.0*old + d) * d  # quadratic correction
            v = float(new_corr.max())
            if v < best_val:
                best_val = v
                best_delta = d
        h[i] = old + best_delta
        # Update corr
        h_new_i = h[i]
        actual_d = h_new_i - old
        corr += coeff * actual_d
        corr[n-1] -= (2.0*old + actual_d) * actual_d
    return h

    S = n * 0.5
    best_h = h.copy()
    best_c = _c5(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for i in order:
            old = h[i]
            # Compute coeff
            coeff = np.zeros(2*n-1)
            # Positive lags
            for lag in range(1, n):
                c = 0.0
                if i + lag < n:
                    c += (1.0 - h[i + lag])
                if i - lag >= 0:
                    c -= h[i - lag]
                coeff[n-1+lag] = c
            # Negative lags
            for L in range(1, n):
                c = 0.0
                if i - L >= 0:
                    c += (1.0 - h[i - L])
                if i + L < n:
                    c -= h[i + L]
                coeff[n-1-L] = c
            # Evaluate
            deltas = np.linspace(-old, 1.0-old, n_pts)
            # new_corr = corr + coeff*d + correction for m=n-1
            vals = corr + np.outer(deltas, coeff)  # (n_pts, 2n-1)
            vals[:, n-1] -= (2.0*old + deltas[:,np.newaxis]) * deltas[:,np.newaxis]
            max_vals = vals.max(axis=1)
            best_idx = int(np.argmin(max_vals))
            h[i] = old + deltas[best_idx]
        # Project
        h = _proj(h, S)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _prepare_coeff(h, i, n):
    """Compute coefficient vector for changing h[i] by delta"""
    lags = np.arange(1, n)
    coeff = np.zeros(2*n-1)
    # Positive lags: coeff[n-1+lag] for lag=1..n-1
    # = (1-h[i+lag])*I(i+lag<n) - h[i-lag]*I(i-lag>=0)
    pos_idx = i + lags
    valid_pos = pos_idx < n
    c = np.zeros(n-1)
    c[valid_pos] = 1.0 - h[pos_idx[valid_pos]]
    neg_idx = i - lags
    valid_neg = neg_idx >= 0
    c[valid_neg] -= h[neg_idx[valid_neg]]
    coeff[n:] = c  # indices n to 2n-2
    
    # Negative lags: coeff[n-1-L] for L=1..n-1
    # = (1-h[i-L])*I(i-L>=0) - h[i+L]*I(i+L<n)

    c2 = np.zeros(n-1)
    c2[valid2] = 1.0 - h[neg_idx2[valid2]]
    pos_idx2 = i + lags
    valid3 = pos_idx2 < n
    c2[valid3] -= h[pos_idx2[valid3]]
    coeff[:n-1] = c2
    return coeff

So: -h[i] < delta < 1-h[i] AND h[j]-1 < delta < h[j]

Combined: max(-h[i], h[j]-1) < delta < min(1-h[i], h[j])

This is a valid interval if max(-h[i], h[j]-1) < min(1-h[i], h[j]).
Since h[i], h[j] ∈ [0,1]: -h[i] ≤ 0 ≤ h[j] and h[j]-1 ≤ 0 ≤ 1-h[i].
So the interval is always non-empty (contains 0).

Great, so pair moves with the sum constraint are clean.

For the 1D optimization of delta in a pair move:
f(delta) = max over m of [old_corr[m] + coeff_diff[m]*delta + (m==n-1)*(2(h[j]-h[i])*delta - 2*delta²)]

This is max of one concave quadratic and (2n-2) linear functions.

To minimize: evaluate at a grid of points in the valid interval for delta.

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
        h *= S / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    for _ in range(30):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        x = np.clip(x, 0.0, 1.0)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _pair_sweep(h, n, n_pts=16, rng=None, deadline=None, max_sweeps=None):
    S = n * 0.5
    best_h = h.copy()
    best_c = _c5(h)
    t0 = time.time()
    n_sweeps_done = 0
    while True:
        if max_sweeps is not None and n_sweeps_done >= max_sweeps:
            break
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for ii in range(n):
            i = int(order[ii])
            old = h[i]
            lags = np.arange(1, n)
            coeff = np.zeros(2 * n - 1)
            idx_p = i + lags
            vp = idx_p < n
            c = np.zeros(n - 1)
            c[vp] = 1.0 - h[idx_p[vp]]
            idx_m = i - lags
            vm = idx_m >= 0
            c[vm] -= h[idx_m[vm]]
            coeff[n:] = c
            c2 = np.zeros(n - 1)
            c2[vm] = 1.0 - h[idx_m[vm]]
            c2[vp] -= h[idx_p[vp]]
            coeff[:n-1] = c2
            lo = -old * 0.99
            hi = (1.0 - old) * 0.99
            deltas = np.linspace(lo, hi, n_pts)
            vals = corr[np.newaxis, :] + np.outer(deltas, coeff)
            vals[:, n-1] -= (2.0 * old + deltas[:, np.newaxis]) * deltas[:, np.newaxis]
            mx = vals.max(axis=1)
            bidx = int(np.argmin(mx))
            h[i] = old + deltas[bidx]
        h = _proj(h, S)
        c = _c5(h)
        n_sweeps_done += 1
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    return P, Nv, max(float(P[0]), float(np.max(P)), float(np.max(Nv)))

def _soft_val(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400, 0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a)

def _grad_weights(h, wP, wN, n, dx):
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _soft_grad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_weights(h, wP, wN, n, dx)

def _fista(h0, alpha, max_t, L0=10.0):
    if max_t <= 0.5:
        return h0.copy(), h0.copy(), _c5(h0)
    n = len(h0)
    S = n * 0.5
    dx = 2.0 / n
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha)
    best_h = h.copy(); best_c = _c5(h)
    Lc = L0
    for it in range(200000):
        if time.time() - t0 > max_t:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; stale = 0
            Lc = max(Lc * 0.9, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            hp = h; y = h; t = 1.0
            Lc = max(Lc * 0.5, 1.0)
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)
