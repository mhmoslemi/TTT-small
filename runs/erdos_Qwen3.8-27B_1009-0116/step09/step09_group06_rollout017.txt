from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-15:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            a = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += a if diff > 0 else -a
            diff = target - h.sum()
    return h

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_for_shift(h, k_full, n):
    """Gradient of c_k with respect to h for a specific shift index k_full in the full correlation."""
    dx = 2.0 / n
    L = k_full - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _chase_opt(h, n, n_steps, alpha_peak, top_m, step0, rng):
    """Gradient descent chasing the worst shifts."""
    dx = 2.0 / n
    cur_c = _eval(h)
    step = step0
    v = np.zeros(n)
    mu = 0.4
    
    for it in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        order = np.argsort(c)
        top_idx = order[-top_m:]
        vals = c[top_idx]
        mx = vals.max()
        # Very peaked weighting
        s = alpha_peak * (mx - vals)
        s = np.clip(s, 0, 500)
        w = np.exp(s)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(top_m) / top_m
        else:
            w /= wsum
        
        # Weighted gradient
        g = np.zeros(n)
        for ii in range(len(top_idx)):
            if w[ii] < 1e-12:
                continue
            g += w[ii] * _grad_for_shift(h, top_idx[ii], n)
        
        # Momentum
        v = mu * v + (1.0 - mu) * g
        
        # Line search
        taken = False
        s_try = step
        for _ in range(12):
            h_new = _proj(h - s_try * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(s_try * 1.08, step0 * 20.0)
                taken = True
                break
            s_try *= 0.45
        if not taken:

            if rng.rand() < 0.05:
                # Random perturbation
                idx = rng.randint(n, size=5)
                h[idx] += 0.005 * rng.randn(5)
                h = _repair(h, n)
                cur_c = _eval(h)
        
        v *= 0.99  # slight decay
    
    return _repair(h, n), cur_c

def _slsqp_focused(h, n, alpha, m, maxiter):

    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m:]
        vals = c[idx]
        mx = vals.max()
        s = alpha * (mx - vals)
        s = np.clip(s, 0, 400)
        return mx + np.log(np.exp(s).sum()) / alpha

        idx = np.argsort(c)[-m:]
        vals = c[idx]
        mx = vals.max()
        s = alpha * (mx - vals)
        s = np.clip(s, 0, 400)
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)

            if w[ii] < 1e-14:
                continue
            wk = w[ii] * dx

            else:
                M = -L
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
        return g
    
    def cfun(x):
        return x.sum() - n * 0.5
    
    def cjac(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > 1e-15:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            a = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += a if diff > 0 else -a
            diff = target - h.sum()
    return h

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(h, k_full, n):
    dx = 2.0 / n
    L = k_full - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _chase_opt(h, n, n_steps, alpha_peak, top_m, step0, rng):
    dx = 2.0 / n
    cur_c = _eval(h)
    step = step0
    v = np.zeros(n)
    mu = 0.45
    for it in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        order = np.argsort(c)
        top_idx = order[-top_m:]
        vals = c[top_idx]
        mx = vals.max()
        s = alpha_peak * (mx - vals)
        s = np.clip(s, 0, 500)
        w = np.exp(s)
        ws = w.sum()
        w = w / ws if ws > 1e-30 else np.ones(top_m) / top_m
        g = np.zeros(n)
        for ii in range(len(top_idx)):
            if w[ii] < 1e-12:
                continue
            g += w[ii] * _grad_shift(h, top_idx[ii], n)
        v = mu * v + (1.0 - mu) * g
        taken = False
        st = step
        for _ in range(12):
            h_new = _proj(h - st * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(st * 1.08, step0 * 30.0)
                taken = True
                break
            st *= 0.4
        if not taken:
            step = min(st * 1.3, step0 * 30.0)
            if rng.rand() < 0.04:
                idx = rng.randint(n, size=5)
                h[idx] += 0.004 * rng.randn(5)
                h = _repair(h, n)
                cur_c = _eval(h)
        v *= 0.995
    return _repair(h, n), cur_c

def _slsqp_focused(h, n, alpha, m, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m:]
        vals = c[idx]
        mx = vals.max()
        s = alpha * (mx - vals)
        s = np.clip(s, 0, 400)
        return mx + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m:]
        vals = c[idx]
        mx = vals.max()
        s = alpha * (mx - vals)
        s = np.clip(s, 0, 400)
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-14:
                continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n - L])
                g[:n - L] -= wk * x[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
        return g
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        temp = (1.0 - frac) * 0.04
        step = base_step * (1.5 - 0.8 * frac) * (0.3 + rng.rand())
        k = int(rng.choice([2, 3, 3, 4, 5, 7, 9]))
        idx = rng.randint(0, n, size=k)
        d = step * rng.randn(k)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.02 or h_new.max() > 1.02:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur or (rng.rand() < temp and c5n < c5cur + 0.0005):
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _fine_polish(h, n, n_iter, rng):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    scales = [3e-7, 1e-7, 5e-8, 2e-8]
    per_scale = max(1, n_iter // len(scales))
    for scale in scales:
        for _ in range(per_scale):
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 25)) % n
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5cur:
                    c5cur = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
            if rng.rand() < 0.4:
                i2 = rng.randint(n)
                j2 = (i2 + 1 + rng.randint(0, 12)) % n
                d2 = scale * 0.6 * rng.randn()
                ni2, nj2 = h[i2] + d2, h[j2] - d2
                if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
                    h[i2], h[j2] = ni2, nj2
                    c5n = _eval(h)
                    if c5n < c5cur:
                        c5cur = c5n
                        if c5n < best_c:
                            best_c = c5n
                            best_h = h.copy()
                    else:
                        h[i2], h[j2] = ni2 - d2, nj2 + d2
    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return best_h, best_c
