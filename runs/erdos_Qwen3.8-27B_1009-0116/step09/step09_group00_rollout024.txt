import numpy as np
import time

def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)

Let me also think about a completely different approach: **coordinate descent**.

At each step, pick one coordinate j and one "partner" coordinate i (to maintain the constraint), and change h[j] by +δ and h[i] by -δ. This is a 1D subproblem.

The C₅ as a function of δ (with h[j]→h[j]+δ, h[i]→h[i]-δ) is:
C₅(δ) = max_k V_k(δ)

where V_k(δ) is the correlate value at lag k as a function of δ.

For a specific lag k:
V_k(δ) = V_k(0) + δ * (∂V_k/∂h[j] - ∂V_k/∂h[i])

So V_k is LINEAR in δ! Therefore C₅(δ) = max_k (a_k + b_k * δ) is a piecewise-linear (convex) function of δ. We can find its minimum analytically!

The minimum of a convex piecewise-linear function occurs where the subgradient contains 0. We can find this by:
- Computing all the breakpoints where the active set changes
- Or simply: the minimum is at δ* where the sign of b changes

Actually, for max of linear functions, the minimum is at the point where the maximum switches from one function to another. This can be found by:
- Computing V_k(0) and b_k for all k
- Finding δ* that minimizes max_k (V_k(0) + b_k * δ)

This is a 1D convex optimization. We can solve it by:
- The subgradient at δ is {b_k : V_k(δ) = max}. We want 0 in this set.
- If all b_k > 0 at the max, decrease δ; if all < 0, increase δ; otherwise δ is optimal.

But this is a lot of work for a 1D problem. A simpler approach: just evaluate C₅ at a few values of δ and pick the best. Since it's piecewise linear with O(n) breakpoints, bisection or ternary search would work.

Actually, the simplest: evaluate C₅ at δ=0, δ=±step, and pick the best. This is basically what a coordinate descent with line search does.

But the issue is: there are n(n-1)/2 possible pairs (i,j), and we can only try a few per iteration. So we need to be smart about which pairs to try.

This brings us back to the gradient-guided approach: use the subgradient to identify which pairs (i,j) would be most effective to move.

import numpy as np
import time

def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = max(float(np.max(P)), float(np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _soft_val(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400, 0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a)

def _soft_grad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_w(h, wP, wN)

def _grad_w(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2*n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _fista(h0, alpha, max_t, L0=10.0):
    if max_t <= 0:
        h = h0.copy(); return h, h.copy(), _c5(h)
    n = len(h0); S = n * 0.5; t0 = time.time()
    h = _repair(h0, n); hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h); stale = 0; Lc = L0
    for it in range(500000):
        if time.time() - t0 > max_t: break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp); t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        else:
            y = h; fy = fh
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12); accepted = False
        for _ in range(14):
            xn = _proj(y - step * g, S); fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-11:
                accepted = True; break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.85, 1.0)
            if it % 5 == 0:
                c = _c5(h)
                if c < best_c: best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 12: break
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _sg_lag(h, k):
    n = len(h); g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        kk = abs(k)
        if kk >= n: return g
        if k > 0:
            g[kk:] += (1.0 - h[:n-kk]); g[:n-kk] -= h[kk:]
        else:
            g[:n-kk] += (1.0 - h[kk:]); g[kk:] -= h[:n-kk]
    return g

def _sg_topk(h, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    tol = max(1e-8, 6.0 * 2.0 / n)
    top = np.where(corr >= mx - tol)[0]
    if len(top) == 0: top = np.array([int(np.argmax(corr))])
    if len(top) > K:
        vals = corr[top]; top = top[np.argsort(vals)[::-1][:K]]
    g = np.zeros(n)
    for pk in top:
        g += _sg_lag(h, int(pk) - (n - 1))
    return g / len(top)

def _sg_opt(h0, deadline, rng):
    n = len(h0); S = n * 0.5
    h = h0.copy(); c = _c5(h)
    best_c = c; best_h = h.copy()
    t_start = time.time(); total_t = max(1.0, deadline - t_start)
    eta = 0.08; stall = 0; it = 0
    while time.time() < deadline - 1:
        frac = (time.time() - t_start) / total_t
        K = 3 + int(frac * 6)
        g = _sg_topk(h, K)
        ng = np.linalg.norm(g)
        if ng < 1e-14: break
        g /= ng
        eta_cur = eta * (1.0 - 0.6 * frac)
        accepted = False; step = eta_cur
        for _ in range(20):
            h_new = _proj(h - step * g, S)
            c_new = _c5(h_new)
            if c_new < c - 1e-14:
                h = h_new; c = c_new; accepted = True; break
            step *= 0.4
        if accepted:
            if c < best_c: best_c = c; best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                sz = 0.005 + 0.02 * (1.0 - frac)
                h_p = _proj(h + rng.randn(n) * sz, S)
                c_p = _c5(h_p)
                if c_p < best_c + 0.005:
                    h = h_p; c = c_p
                stall = 0; eta *= 0.7
        it += 1
    return best_h, best_c

def _sa_search(h0, rng, deadline):
    n = len(h0); S = n * 0.5
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max()); best_c = mx * (2.0 / n); best_h = h.copy()
    t_start = time.time(); total_t = max(1.0, deadline - t_start)
    while time.time() < deadline - 0.5:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 1.5 * (0.003) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        ak = abs(k_act)
        if rng.random() < 0.7 and 0 < ak < n:
            if k_act > 0:
                valid = (h[:n-ak] > 0.01) & (h[ak:] < 0.99)
            else:
                valid = (h[ak:] > 0.01) & (h[:n-ak] < 0.99)
            if valid.sum() > 2:
                pos = np.nonzero(valid)[0]
                i = int(pos[rng.randint(len(pos))])
                j = i + ak if k_act > 0 else i - ak
                if 0 <= j < n:
                    d = (rng.random() - 0.45) * 0.05
                    h[i] += d; h[j] -= d
                    corr2 = np.correlate(h, 1.0 - h, mode='full')
                    mx2 = float(corr2.max())
                    if mx2 < mx - 1e-14 or (mx2 > mx and rng.random() < np.exp(-(mx2-mx)/max(T,1e-15))):
                        mx = mx2; corr = corr2
                        cn = mx * (2.0/n)
                        if cn < best_c: best_c = cn; best_h = h.copy()
                    else:
                        h[i] -= d; h[j] += d
                    continue
        i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j: continue
        d = (rng.random() - 0.5) * 0.04
        d = max(-h[i], min(d, 1.0 - h[i])); d = max(d, -h[j]); d = min(d, 1.0 - h[j])
        if abs(d) < 1e-12: continue
        h[i] += d; h[j] -= d
        corr2 = np.correlate(h, 1.0 - h, mode='full')
        mx2 = float(corr2.max())
        if mx2 < mx - 1e-14 or (mx2 > mx and rng.random() < np.exp(-(mx2-mx)/max(T,1e-15))):
            mx = mx2; corr = corr2
            cn = mx * (2.0/n)
            if cn < best_c: best_c = cn; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

def _fine_local(h0, rng, deadline):
    n = len(h0); S = n * 0.5
    h = h0.copy(); c = _c5(h); best_c = c; best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_act = int(np.argmax(corr)) - (n - 1); ak = abs(k_act)

    base_step = 0.03
    while time.time() < deadline - 0.5:
        frac = (time.time() - t_start) / total_t
        step_sz = base_step * (0.05 ** frac)
        # Recompute active lag periodically
        if rng.random() < 0.01:
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_act = int(np.argmax(corr)) - (n - 1); ak = abs(k_act)
        if 0 < ak < n:
            if k_act > 0:
                valid = np.where((h[:n-ak] > 1e-4) & (h[ak:] < 1.0 - 1e-4))[0]
            else:
                valid = np.where((h[ak:] > 1e-4) & (h[:n-ak] < 1.0 - 1e-4))[0]
            if len(valid) > 0:
                i = int(valid[rng.randint(len(valid))])
                j = i + ak if k_act > 0 else i - ak
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j: continue
        d = (rng.random() - 0.5) * step_sz
        d = max(-h[i], min(d, 1.0 - h[i])); d = max(d, min(d, 1.0 - h[j]))
        d = max(max(-h[i], min(-d, 1.0 - h[i])), -h[j])
        if abs(d) < 1e-14: continue
        h[i] += d; h[j] -= d
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn
            if c < best_c: best_c = c; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)
    for hh in (1.0 - best.copy(), best[::-1], 0.5 + 0.5*(best - best[::-1])):
        h2 = _proj(hh, n0 * 0.5); c = _c5(h2)
        if c < best_c: best = h2; best_c = c

    # Phase 1: FISTA moderate alpha (90s)
    end1 = t0 + min(90.0, 0.09 * budget_s)
    if end1 > time.time():
        hcur = best.copy()
        for a in [100, 500, 3000]:
            if time.time() >= end1 - 2: break
            hf, bh, bc = _fista(hcur, a, (end1 - time.time()) * 0.6, 10.0)
            cf = _c5(hf)
            if bc < best_c: best = bh; best_c = bc
            if cf < best_c: best = hf; best_c = cf
            hcur = hf if cf <= bc else bh

    # Phase 2: Subgradient with actual C5 line search (230s)
    end2 = t0 + min(320.0, 0.32 * budget_s)
    if end2 > time.time():
        best, best_c = _sg_opt(best, end2, rng)

    # Phase 3: Multi-resolution FISTA (150s)
    end3 = t0 + min(470.0, 0.47 * budget_s)
    if end3 > time.time():
        for m in [512, n0, 640]:
            if time.time() >= end3 - 3: break
            slot = min(40.0, end3 - time.time() - 2)
            if slot < 5: break
            hm = _resample(best, m) if m != n0 else best.copy()
            for a in [500, 5000]:
                if time.time() >= end3 - 2: break
                hf, bh, bc = _fista(hm, a, (end3 - time.time()) * 0.45, 10.0)
                cf = _c5(hf)
                if bc < best_c:
                    hb = _resample(bh, n0) if len(bh) != n0 else bh
                    best = hb; best_c = _c5(hb)
                if cf < best_c:
                    hb = _resample(hf, n0) if len(hf) != n0 else hf
                    best = hb; best_c = _c5(hb)
                hm = hf if cf <= bc else bh

    # Phase 4: SA with targeted moves (200s)
    end4 = t0 + min(670.0, 0.67 * budget_s)

        best, best_c = _sa_search(best, rng, end4)

        if c < best_c: best = h2; best_c = c

    return best, best_c, len(best)
