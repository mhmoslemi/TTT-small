def _newton_minimax(h, n, M, steps, step0):
    c5 = _eval(h)
    for it in range(steps):
        c = _corr_full(h)
        # Top M lags
        active = np.argsort(c)[-M:]
        c_act = c[active]
        # Compute gradients for active lags
        G = np.zeros((M-1, n))  # Differences
        for i in range(M-1):
            g_i = _grad_single(c, active[i], h, n)
            g_j = _grad_single(c, active[i+1], h, n)
            G[i] = g_i - g_j
        # Want: G @ d = -(c_act[:-1] - c_act[1:]) = c_act[1:] - c_act[:-1]
        # Min norm: d = G^T (G G^T)^{-1} b
        b = c_act[1:] - c_act[:-1]
        GGT = G @ G.T + 1e-8 * np.eye(M-1)
        rhs = np.linalg.solve(GGT, b)
        d = G.T @ rhs
        d -= d.mean()  # Ensure sum = 0
        # Line search
        best_step = 0
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_new = _proj(h + frac * d, n/2)
            c_new = _eval(h_new)
            if c_new < c5:
                h = h_new
                c5 = c_new
                best_step = frac
                break
        if best_step == 0:
            # Try negative direction
            for frac in [0.5, 0.25, 0.125]:
                h_new = _proj(h - frac * d, n/2)
                c_new = _eval(h_new)
                if c_new < c5:
                    h = h_new
                    c5 = c_new
                    break
    return _repair(h, n)
