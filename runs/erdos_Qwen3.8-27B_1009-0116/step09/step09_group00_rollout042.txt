def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):

        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = x < 1.0
        else:
            m = x > 0.0
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

    pref[0] = 0.0

    total = pref[n]

        Nv[idx] = total - pref[idx] - ac[idx]

    return P, Nv, mx

def _grad_weights(h, wP, wN):

    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        m = 2 * n - 1
        while Nf < m:

        H = np.fft.rfft(h, Nf)

        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2*n - 1].real + L[:n].real)

            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0; cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv; wN *= inv
    return _grad_weights(h, wP, wN)

import numpy as np
import time

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _grad_soft(h, a):
    n = len(h)
    dx = 2.0 / n
    P, Nv, mx = _pn(h)
    allv = np.concatenate([P, Nv[1:]])
    sh = allv - mx
    w = np.exp(np.clip(a * sh, -40, 0))
    w /= w.sum()
    wP = w[:n].copy()
    wN = w[n:].copy()
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

    hp = h.copy(); y = h.copy(); t = 1.0

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)
