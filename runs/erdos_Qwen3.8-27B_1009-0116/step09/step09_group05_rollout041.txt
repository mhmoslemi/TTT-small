        if np.clip(h - mid, 0, 1).sum() > target:

        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _binary_push(h, n, strength):
    h2 = h.copy()
    h2[h2 > 0.5] = np.minimum(1.0, h2[h2 > 0.5] + strength)
    h2[h2 <= 0.5] = np.maximum(0.0, h2[h2 <= 0.5] - strength)
    return _repair(h2, n)

def _proj_grad(h, alpha, m_lags, steps, step0, t_deadline):
    n = len(h)
    cur_c = _eval(h)
    step = step0
    best_local = h.copy()
    for it in range(steps):
        if time.time() > t_deadline:
            break
        g = _smooth_grad(h, alpha, m_lags)
        h_new = _proj(h - step * g, n * 0.5)
        c5n = _eval(h_new)
        if c5n < cur_c:
            h = h_new
            cur_c = c5n
            best_local = h.copy()
            step = min(step * 1.15, step0 * 80.0)
        else:
            step *= 0.45
        if it % 80 == 79:
            h = _binary_push(h, n, 0.03)
            cur_c = _eval(h)
            if cur_c < _eval(best_local):
                best_local = h.copy()
    return best_local, _eval(best_local)

def _sa_search(h, rng, n_iter, T0, T_floor, n_pert, t_deadline):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    cur_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        T = max(T0 * (T_floor / T0) ** (it / max(n_iter, 1)), T_floor)
        idx = rng.randint(0, n, size=n_pert)
        scale = 0.08 if T > 0.001 else 0.03
        d = scale * rng.randn(n_pert)
        d -= d.mean()
        cur_h[idx] += d
        if cur_h.min() < -0.01 or cur_h.max() > 1.01:
            cur_h = best_h.copy()
            continue
        cur_h = _repair(cur_h, n)
        c5n = _eval(cur_h)
        delta = c5n - c5cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            c5cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = cur_h.copy()
        else:
            cur_h = best_h.copy()
    return best_h, best_c

def _binary_sa(h, rng, n_iter, t_deadline):
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = _repair(h_bin, n)
    c5cur = _eval(h_bin)
    best_h = h_bin.copy()
    best_c = c5cur
    T = 0.02
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        T *= 0.9998
        k = rng.randint(1, 5)
        idx = rng.choice(n, k, replace=False)
        for i in idx:
            h_bin[i] = 1.0 - h_bin[i]
        h_bin = _repair(h_bin, n)
        c5n = _eval(h_bin)
        if c5n < c5cur or rng.rand() < np.exp(-(c5n - c5cur) / max(T, 1e-10)):
            c5cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h_bin.copy()
        else:
            h_bin = best_h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _fine_2pt(h, rng, n_iter, t_deadline):
    n = len(h)
    h2 = h.copy()
    c5c = _eval(h2)
    best_c = c5c
    best_h2 = h2.copy()
    for _ in range(n_iter):
        if time.time() > t_deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 2e-7 * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            c5n = _eval(h2)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h2 = h2.copy()
            else:
                h2[i], h2[j] = ni - d, nj + d
    return best_h2, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution with binary push
    for ratio in [4, 6, 8]:
        if time.time() - t0 > 50:
            break
        nc = max(32, n // ratio)
        f = n // nc
        hc = _downsample(h, f)
        for alpha in [200, 2000, 20000, 200000]:
            if time.time() - t0 > 70:
                break
            hc = _slsqp(hc, alpha, min(40, nc), 250, 1e-13)
            hc = _binary_push(hc, nc, 0.08)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: SLSQP alpha escalation at full res
    alpha_seq = [2000, 10000, 50000, 200000, 800000, 3000000, 12000000, 50000000, 200000000]
    for alpha in alpha_seq:
        if time.time() - t0 > 320:
            break
        lift = min(90, max(20, int(35 * np.log10(max(alpha / 2000, 1)) + 20)))
        h_new = _slsqp(best_h, alpha, lift, 1000, 1e-15)
        h_new = _binary_push(h_new, n, 0.02)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Adaptive projected gradient with binary push
    if time.time() - t0 < budget_s - 300:
        configs = [
            (5000, 40, 700, 0.004),
            (50000, 60, 550, 0.0025),
            (300000, 80, 400, 0.0015),
            (1500000, 100, 300, 0.0008),
            (8000000, 130, 250, 0.0004),
        ]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s - 250:
                break
            h_new, c5n = _proj_grad(best_h.copy(), alpha, ml, steps, s0, budget_s - 250)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Binary simulated annealing
    if time.time() - t0 < budget_s - 150:
        deadline = budget_s - 150
        n_iter_b = int(min((deadline - (time.time() - t0)) / 0.003, 60000))
        h_b, c5_b = _binary_sa(best_h, rng, n_iter_b, t0 + deadline)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        # Interpolate back to continuous for final polish
        best_h = _binary_push(best_h, n, -0.15)  # pull back slightly from binary

    # Phase 5: Continuous SA
    if time.time() - t0 < budget_s - 80:
        deadline = budget_s - 80
        n_iter_s = int(min((deadline - (time.time() - t0)) / 0.003, 50000))
        h_s, c5_s = _sa_search(best_h, rng, n_iter_s, 0.003, 1e-12, 6, t0 + deadline)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 6: Final SLSQP polish at very high alpha
    if time.time() - t0 < budget_s - 40:
        for alpha in [800000000, 3000000000]:
            if time.time() - t0 > budget_s - 30:
                break
            h_new = _slsqp(best_h, alpha, 100, 400, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 7: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 3:
        deadline = budget_s - 3
        n_pol = int(min((deadline - (time.time() - t0)) / 0.0008, 500000))
        best_h, best_c5 = _fine_2pt(best_h, rng, n_pol, t0 + deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
