import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _eval_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_top(h, alpha, m_lags, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top = np.argsort(c)[-m_lags:]
    s = alpha * (c[top] - c.max())
    s = np.clip(s, -500, 0)
    ew = np.exp(s)
    ew /= ew.sum()
    g = np.zeros(n)
    for ii, k_idx in enumerate(top):
        wk = ew[ii] * dx
        L = k_idx - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)

        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g, c.max()

def _block_obj(x, h_fix, start, bw, n, alpha, ml):
    h = h_fix.copy()
    h[start:start + bw] = np.clip(x, 0, 1)
    c = _eval_full(h)
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s, -500, 0)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _block_grad(x, h_fix, start, bw, n, alpha, ml):
    h = h_fix.copy()
    h[start:start + bw] = np.clip(x, 0, 1)
    g_full, _ = _grad_top(h, alpha, ml, n)
    return g_full[start:start + bw]

def _b_sum(x, tgt):
    return x.sum() - tgt

def _b_sum_j(x):
    return np.ones(len(x))

def _block_opt(h, start, bw, alpha, ml, mi, n):
    h_fix = h.copy()
    x0 = h_fix[start:start + bw].copy()
    tgt = n * 0.5 - (h_fix.sum() - x0.sum())
    tgt = max(0.0, min(bw, tgt))
    bnds = [(0.0, 1.0)] * bw
    cons = [{'type': 'eq', 'fun': _b_sum, 'jac': _b_sum_j, 'args': (tgt,)}]
    try:
        res = minimize(_block_obj, x0, jac=_block_grad,
                       args=(h_fix, start, bw, n, alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-15})
        h_fix[start:start + bw] = np.clip(res.x, 0, 1)
    except Exception:
        pass
    return _repair(h_fix, n)

def _pgd(h, alpha, ml, n, step):
    g, _ = _grad_top(h, alpha, ml, n)
    g -= g.mean()
    return _proj(h - step * g, n * 0.5)

def _local_ref(h, n_iter, rng, ss):
    n = len(h)
    c5 = _eval(h)
    best, bc = h.copy(), c5
    for _ in range(n_iter):
        k = rng.randint(2, 6)
        idx = rng.randint(0, n, size=k)
        d = ss * (0.5 + rng.rand()) * rng.randn(k)

        hn = h.copy()
        hn[idx] += d
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < c5:
            c5, h = cn, hn
            if cn < bc:
                bc, best = cn, h.copy()
    return best, bc
