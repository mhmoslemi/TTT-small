def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _corr_dx(h):

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

# Gradient of the k-th lag correlation w.r.t. h
def _lag_grad(h, k, n):
    """Gradient of c_k = sum_i h[i]*(1-h[i+k]) w.r.t. h (with dx factor)"""
    dx = 2.0 / n
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        # c_k = sum_{i=0}^{n-1-L} h[i] * (1 - h[i+L])
        g[:n-L] += dx * (1.0 - h[L:])
        g[L:] -= dx * h[:n-L]
    else:
        M = -L
        # c_k = sum_{i=0}^{n-1-M} h[i+M] * (1 - h[i])
        g[M:] += dx * (1.0 - h[:n-M])
        g[:n-M] -= dx * h[M:]
    return g

def _smooth_obj(x, beta, m_lags):

    c = _corr_dx(x)
    idx = np.argsort(c)[-m_lags:]

    m = c.max()
    vals = beta * (c[idx] - m)
    vals = np.clip(vals, -400, 400)
    vals -= vals.max()
    return m + np.log(np.exp(vals).sum()) / beta

def _smooth_grad(x, beta, m_lags):
    n = len(x)
    c = _corr_dx(x)
    idx = np.argsort(c)[-m_lags:]
    m = c.max()
    s = np.clip(beta * (c[idx] - m), -400, 400)

        g += wk * _lag_grad(x, k, n)

def _topk_obj(x, p, m_lags):
    """Power mean of top m_lags - smoother than log-sum-exp for large p"""
    n = len(x)
    c = _corr_dx(x)
    idx = np.argsort(c)[-m_lags:]
    vals = c[idx]
    # p-norm / m_lags^(1/p) approximates max as p -> inf
    # But use shifted version for stability

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _lag_grad(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        g[:n - L] += dx * (1.0 - h[L:])
        g[L:] -= dx * h[:n - L]
    else:
        M = -L
        g[M:] += dx * (1.0 - h[:n - M])
        g[:n - M] -= dx * h[M:]
    return g

def _smooth_obj(x, beta, m_lags):
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(beta * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / beta

def _smooth_grad(x, beta, m_lags):
    n = len(x)
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(beta * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        g += w[ii] * _lag_grad(x, k, n)
    return g

def _slsqp_opt(h0, beta, m_lags, maxiter, ftol):
    n = len(h0)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, t=target: x.sum() - t,
             'jac': lambda x, t=target: np.ones(len(x))}]
    try:
        res = minimize(_smooth_obj, h0, jac=_smooth_grad, args=(beta, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0

def _proj_box_sum(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slp_opt(h, n_slp, m_lags, lambda_start):
    """Successive Linear Programming on active lags"""
    n = len(h)
    target = n * 0.5
    cur_c = _eval(h)
    for _ in range(n_slp):
        c = _corr_dx(h)
        active = np.argsort(c)[-m_lags:]
        g_stack = np.array([_lag_grad(h, k, n) for k in active])
        c_vals = c[active]
        rhs = g_stack @ h - c_vals + 1e-12
        c_obj = np.zeros(n + 1); c_obj[-1] = 1.0
        A_ub = np.zeros((m_lags, n + 1))
        A_ub[:, :n] = g_stack; A_ub[:, -1] = -1.0
        A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
        bounds = [(0, 1)] * n + [(None, None)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=[target],
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_slp = res.x[:n]
        lam = lambda_start
        improved = False
        for _ in range(20):
            h_try = (1 - lam) * h + lam * h_slp
            h_try = _repair(h_try, n)
            c_try = _eval(h_try)
            if c_try < cur_c - 1e-16:
                h = h_try; cur_c = c_try; improved = True
                break
            lam *= 0.5
            if lam < 1e-6: break
        if not improved:
            break
    return h, cur_c

def _proj_subgrad(h, beta, m_lags, n_steps, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n); mu = 0.8; step = step0
    cur_c = _eval(h); best_h = h.copy(); best_c = cur_c
    for _ in range(n_steps):

        g -= g.mean()
        v = mu * v + (1 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-18: break
        v /= ng
        taken = False
        for _t in range(15):
            h_new = _proj_box_sum(h - step * v, target)

                step = min(step * 1.07, step0 * 40); taken = True; break
            step *= 0.4
        if not taken:

    return best_h, best_c

        n_pert = int(rng.choice([2, 3, 4, 5, 7]))
        idx_p = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert) - 0.5 * step * g[idx_p]
        d -= d.mean()
        h_new = h.copy(); h_new[idx_p] += d
        if h_new.min() < -0.15 or h_new.max() > 1.15: continue
        h_new = _repair(h_new, n)

        if c5n < cur_c - 1e-16:

            if i % 50 == 0:

    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    cur_c = _eval(h); best_h = h.copy(); best_c = cur_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n

    return best_h, best_c
