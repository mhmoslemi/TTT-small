        if wk < 1e-12:
            continue
        wk *= dx
        L = k - nm1
        if L >= 0:

            g[L:] -= wk * h[:n-L]
        else:
            M = -L
            g[M:] += wk * (1.0 - h[:n-M])
            g[:n-M] -= wk * h[M:]
    return val, g

def _fista(h, alpha, n_iters, lam_scale=1.0):
    """FISTA for minimizing smoothed max over sumbox constraint"""
    n = len(h)
    x_old = h.copy()
    x_curr = h.copy()
    c_curr = h.copy()
    t_k = 1.0
    # Estimate Lipschitz constant
    dx = 2.0 / n
    # L ≈ 2*n*dx*2 + alpha*4 (rough upper bound)
    # More carefully: ||∇²c_k||_op ≤ 2*n*dx (each c_k involves n terms, each with Hessian entry ±dx)
    # Actually c_k is bilinear: h^T B_k h type, Hessian is 2*B_k
    # ||∇c_k||_∞ ≤ 2*dx, so ||∇c_k||_2 ≤ 2*dx*sqrt(n) = 4/sqrt(n)
    # Hessian of c_k: entries are ±dx, at most 2 per row (one from each factor)
    # ||∇²c_k||_op ≤ 2*2*dx = 4*dx = 8/n
    # Hessian of smoothed max: sum w_k ∇²c_k + (1/alpha) sum w_k ∇c_k ∇c_k^T * alpha^2 / alpha
    # ≈ sum w_k ∇²c_k + alpha * sum w_k ∇c_k ∇c_k^T
    # ≤ 8/n + alpha * (4/sqrt(n))^2 = 8/n + 16*alpha/n
    L = (8 + 16 * alpha) / n * lam_scale
    step = 1.0 / L
    
    for it in range(n_iters):
        _, g = _smoothed_max_and_grad(x_curr, alpha)
        # Gradient step
        y = x_curr - step * g
        # Project
        x_new = _sumbox_project(y, n / 2.0)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        c_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x_old)
        c_new = np.clip(c_new, 0.0, 1.0)  # Keep in bounds for numerical stability
        x_old = x_curr
        x_curr = x_new
        c_curr = c_new
        t_k = t_new
    return x_curr

- H[i+L, i] = -dx
- Diagonal: H[i,i] = 0 (no h[i]^2 term)

Wait, c[m] = dx * sum_i h[i] - dx * sum_i h[i]*h[i+L]

The Hessian of -dx * h[i]*h[i+L] is:
- ∂²/-∂h[i]² = 0 (it's linear in h[i])
- ∂²/-∂h[i+L]² = 0 (linear in h[i+L])
- ∂²/-∂h[i]∂h[i+L] = -dx

The operator norm of H_m is at most 2*dx (since each row has at most 2 non-zero entries of magnitude dx).

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sumbox_project(x, t):
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s < t:
            hi = mid
        else:
            lo = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = t - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _grad_smoothed(h, alpha, m_lags=0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if m_lags > 0 and m_lags < len(c):
        top = np.argsort(c)[-m_lags:]
        w = np.zeros(len(c))
        cc = c[top]
        s = alpha * (cc - cc.max())
        e = np.exp(s)
        w[top] = e / e.sum()
    else:
        s = alpha * c
        s -= s.max()
        e = np.exp(s)
        w = e / e.sum()
    nm1 = n - 1
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        L = k - nm1
        if L >= 0:
            g[:n - L] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:n - L]
        else:
            M = -L
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
    return g

def _fista(h, alpha, n_iters, m_lags=0):
    n = len(h)
    x_old = h.copy()
    x_curr = h.copy()
    t_k = 1.0
    dx = 2.0 / n
    L = (8.0 + 32.0 * alpha) / n
    step = 1.0 / L
    for _ in range(n_iters):
        g = _grad_smoothed(x_curr, alpha, m_lags)
        y = x_curr - step * g
        x_new = _sumbox_project(y, n / 2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        c_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x_old)
        c_new = np.clip(c_new, 0.0, 1.0)
        x_old = x_curr
        x_curr = x_new
        t_k = t_new
    return _sumbox_project(x_curr, n / 2.0)

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax_len = 2 * n - 1
    nm1 = n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax_len)
        k1_lo = j
        k1_hi = min(cmax_len - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax_len - 1, cmax_len - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(22):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _sla_obj(x, alpha, m_lags, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m_lags:]
    cc = c[idx]
    s = np.clip(alpha * (cc - cc.max()), -500, 500)
    return cc.max() + np.log(np.exp(s).sum()) / alpha

def _sla_grad(x, alpha, m_lags, n):
    return _grad_smoothed(x, alpha, m_lags)

def _sum_con(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_sla_obj, h, jac=_sla_grad,
                       args=(alpha, m_lags, n), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 8e-6 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            c5n = _eval(best_h)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c5
