import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 2.5
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _fg(h, alpha, ml):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    idx = np.argpartition(-c, min(ml, 2 * n - 2))[:ml]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    obj = m + np.log(w.sum()) / alpha
    g = np.zeros(n)
    for ii in range(ml):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = idx[ii] - (n - 1)
        if L >= 0:
            nv = n - L
            g[:nv] += wk * (1.0 - h[L:])
            g[L:] += wk * (-h[:nv])
        else:
            M = -L
            nv = n - M
            g[M:] += wk * (1.0 - h[:nv])
            g[:nv] += wk * (-h[M:])
    return obj, g

def _proj_grad(h, alpha, ml, n_steps, t_start, t_limit):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    mu = 0.7
    lr = 0.5
    for it in range(n_steps):
        if time.time() - t_start > t_limit:
            break
        _, g = _fg(h, alpha, ml)
        grad_norm = np.linalg.norm(g)
        if grad_norm < 1e-14:
            break
        g_n = g / grad_norm
        step_dir = -g_n
        step_size = lr / grad_norm
        for _ in range(12):
            h_try = _proj(h + step_size * step_dir + mu * momentum, n / 2.0)
            c_try = _eval(h_try)
            if c_try < best_c + 1e-14:
                momentum = mu * momentum + step_size * step_dir
                h = h_try
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
                break
            step_size *= 0.5
        else:
            momentum = mu * momentum
        lr *= 1.002
        if lr > 5.0:
            lr = 5.0
    return best_h, best_c

def _slsqp_wrap(h, alpha, ml, maxiter, t_start, t_limit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        return _fg(x, alpha, ml)[0]
    def _jac(x):
        return _fg(x, alpha, ml)[1]
    def _sum_fun(x):
        return x.sum() - n / 2.0
    def _sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, rng, t_start, t_limit):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    c_full = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
    top_k = np.argsort(-c_full)[:80]
    top_set = set(top_k.tolist())
    pert_scale = 1e-4
    for it in range(20000000):
        if time.time() - t_start > t_limit:
            break
        if it % 5000 == 0:
            pert_scale *= 0.97
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = pert_scale * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        ni = np.clip(ni, 0, 1)
        nj = np.clip(nj, 0, 1)
        best_h[i] = ni
        best_h[j] = nj
        c_new = _eval(best_h)
        if c_new < best_c:
            best_c = c_new
        else:
            best_h[i] -= best_h[i] - (ni - d)
            best_h[j] -= best_h[j] - (nj + d)
            best_h[i] = np.clip(best_h[i], 0, 1)
            best_h[j] = np.clip(best_h[j], 0, 1)
    best_h = _repair(best_h, n)
    return best_h, _eval(best_h)
