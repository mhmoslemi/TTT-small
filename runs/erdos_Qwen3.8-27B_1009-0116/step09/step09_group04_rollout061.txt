def _repair_bin(h, n):
    """Repair binary h to have exactly n/2 ones."""
    h = (h > 0.5).astype(float)
    target = n // 2
    s = int(h.sum())
    diff = target - s
    if diff > 0:

        idx = np.random.permutation(zeros)[:diff]
        h[idx] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        idx = np.random.permutation(ones)[:abs(diff)]
        h[idx] = 0.0
    return h

    best_h = h.copy()

    ones = list(np.where(h == 1)[0])

import numpy as np
import time
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _round_bin(h, n, rng):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    s = int(h.sum())
    diff = target - s
    if diff > 0:
        z = np.where(h == 0)[0]
        h[rng.choice(z, min(diff, len(z)), replace=False)] = 1.0
    elif diff < 0:
        o = np.where(h == 1)[0]
        h[rng.choice(o, min(-diff, len(o)), replace=False)] = 0.0
    return h

def _random_bin(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _sa_binary(h, n, rng, t_end, T0, cool):
    h = h.copy()
    c = _eval(h)
    best_h, best_c = h.copy(), c
    T = T0

        o = np.where(h == 1)[0]
        z = np.where(h == 0)[0]
        i = o[rng.randint(len(o))]
        j = z[rng.randint(len(z))]
        h[i], h[j] = 0.0, 1.0

        d = cn - c

                best_c, best_h = c, h.copy()
                stall = 0
            else:
                stall += 1

            c = _eval(h)
            stall = 0
    return best_h, best_c

    c = _eval(h)
    best_h, best_c = h.copy(), c
    no_improve_rounds = 0

        z = np.where(h == 0)[0]
        for _ in range(n_try_per_round):
            i = o[rng.randint(len(o))]
            j = z[rng.randint(len(z))]
            h[i], h[j] = 0.0, 1.0
            cn = _eval(h)
            if cn < c + round_best_delta:
                round_best_delta = cn - c
                best_i, best_j = i, j
            else:
                h[i], h[j] = 1.0, 0.0
        if round_best_delta < 0:
            # Apply best swap found
            h[best_i], h[best_j] = 0.0, 1.0
            c += round_best_delta
            if c < best_c:
                best_c, best_h = c, h.copy()
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds > 3:
                break
    return best_h, best_c

def _repair_cont(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h = h + (target - s) / n
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1 - h)
        else:
            idx = np.argsort(h)
        for k in idx:
            a = min(abs(diff), 1 - h[k]) if diff > 0 else min(abs(diff), h[k])
            h[k] += np.sign(diff) * a
            diff -= np.sign(diff) * a
            if abs(diff) < 1e-13:
                break
    return h

def _softmax_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return float(m + np.log(np.exp(s).sum()) / alpha)

def _softmax_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(w)):
        if w[ii] < 1e-14:
            continue
        wk = w[ii] * dx
        L = ii - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

    return x.sum() - target

def _cjac(x, target):
    return np.ones(len(x))

                                     'args': (n * 0.5,)}],
                       options={'maxiter': 1000, 'ftol': 1e-15})
            hr = np.clip(r.x, 0, 1)
            hr = _repair_cont(hr, n)
            cr = _eval(hr)
            if cr < best_c:
                best_c, best_h = cr, hr
        except:
            pass
    return best_h, best_c

    s = a * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(w)):
        if w[ii] < 1e-12:
            continue
        wk = w[ii] * dx
        L = ii - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cf(x, t):
    return x.sum() - t

def _cj(x, t):
    return np.ones(len(x))

def _solsqp(h, n, t_end, alphas, mi=800):
    bh, bc = h.copy(), _eval(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    for a in alphas:
        if time.time() > t_end:
            break
        try:
            r = minimize(_sm_obj, bh, jac=_sm_grad, args=(a,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-15})
            hr = _repair_c(np.clip(r.x, 0, 1), n)
            cr = _eval(hr)
            if cr < bc:
                bc, bh = cr, hr
        except Exception:
            pass
    return bh, bc

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pgrad(h, n, t_end, a, steps, s0):
    h = h.copy()
    c = _eval(h)
    bh, bc = h.copy(), c
    v = np.zeros(n)
    step = s0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _sm_grad(h, a)
        v = 0.85 * v + 0.15 * g
        ok = False
        for _ in range(12):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < c:
                h, c, ok = hn, cn, True
                step = min(step * 1.05, s0 * 12)
                break
            step *= 0.5
        if not ok:
            step = min(step * 1.3, s0 * 12)
        if c < bc:
            bc, bh = c, h.copy()
    return bh, bc

import numpy as np
import time
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _rnd_bin(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _blk_bin(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h

def _to_bin(h, n, rng):
    h = (h > 0.5).astype(float).copy()
    t = n // 2
    d = t - int(h.sum())
    if d != 0:
        if d > 0:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, min(d, len(idx)), replace=False)] = 1.0
        else:
            idx = np.where(h == 1)[0]
            h[rng.choice(idx, min(-d, len(idx)), replace=False)] = 0.0
    return h

def _bin_sa(h, n, rng, t_end, T0=0.01):
    h = h.copy()
    c = _eval(h)
    bh, bc = h.copy(), c
    T = T0
    while time.time() < t_end:
        T = max(T * 0.999999, 1e-11)
        o = np.where(h == 1)[0]
        z = np.where(h == 0)[0]
        i = o[rng.randint(len(o))]
        j = z[rng.randint(len(z))]
        h[i], h[j] = 0.0, 1.0
        cn = _eval(h)
        d = cn - c
        if d < 0 or (T > 1e-11 and rng.rand() < np.exp(-d / T)):
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, bc

def _greedy(h, n, rng, t_end, nt=5000):
    h = h.copy()
    c = _eval(h)
    bh, bc = h.copy(), c
    st = 0
    while time.time() < t_end:
        o = np.where(h == 1)[0]
        z = np.where(h == 0)[0]
        bd, bi, bj = 0.0, -1, -1
        for _ in range(nt):
            i = o[rng.randint(len(o))]
            j = z[rng.randint(len(z))]
            h[i], h[j] = 0.0, 1.0
            cn = _eval(h)
            dd = cn - c
            h[i], h[j] = 1.0, 0.0
            if dd < bd:
                bd, bi, bj = dd, i, j
        if bi >= 0:
            h[bi], h[bj] = 0.0, 1.0
            c += bd
            if c < bc:
                bc, bh = c, h.copy()
            st = 0
        else:
            st += 1
            if st >= 5:
                break
    return bh, bc

def _repair_c(h, n):
    h = np.clip(h, 0, 1).copy()
    t = n * 0.5
    for _ in range(30):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for k in idx:
            a = min(abs(d), (1 - h[k]) if d > 0 else h[k])
            h[k] += (1 if d > 0 else -1) * a
            d -= (1 if d > 0 else -1) * a
            if abs(d) < 1e-14:
                break
    return h

def _sm_obj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = a * (c - m)
    s -= s.max()
    return float(m + np.log(np.exp(s).sum()) / a)

def _sm_grad(x, a):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = a * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(w)):
        if w[ii] < 1e-12:
            continue
        wk = w[ii] * dx
        L = ii - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cf(x, t):
    return x.sum() - t

def _cj(x, t):
    return np.ones(len(x))

def _solsqp(h, n, t_end, alphas, mi=800):
    bh, bc = h.copy(), _eval(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    for alpha in alphas:
        if time.time() > t_end:
            break
        try:
            r = minimize(_sm_obj, bh, jac=_sm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-15})
            hr = _repair_c(np.clip(r.x, 0, 1), n)
            cr = _eval(hr)
            if cr < bc:
                bc, bh = cr, hr
        except Exception:
            pass
    return bh, bc

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pgrad(h, n, t_end, a, steps, s0):
    h = h.copy()
    c = _eval(h)
    bh, bc = h.copy(), c
    v = np.zeros(n)
    step = s0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _sm_grad(h, a)
        v = 0.85 * v + 0.15 * g
        ok = False
        for _ in range(12):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < c:
                h, c, ok = hn, cn, True
                step = min(step * 1.05, s0 * 12)
                break
            step *= 0.5
        if not ok:
            step = min(step * 1.3, s0 * 12)
        if c < bc:
            bc, bh = c, h.copy()
    return bh, bc

def _two_pt(h, n, rng, t_end, npol):
    h = h.copy()
    c = _eval(h)
    bh, bc = h.copy(), c
    for _ in range(npol):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 2e-7 * rng.randn()
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d
        h[j] -= d
        cn = _eval(h)
        if cn < c:
            c = cn
            if c < bc:
                bc, bh = c, h.copy()
        else:
            h[i] -= d
            h[j] += d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bh = None
    bc = float('inf')
    bn = 0

    def rem():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    hB = _to_bin(h0, n0, rng)
    bc = _eval(hB)
    bh, bn = hB.copy(), n0

    # P1: SA from rounded initial (0-18%)
    t = t0 + budget_s * 0.18
    r, cr = _bin_sa(hB, n0, rng, t, 0.01)
    if cr < bc:
        bh, bc = r, cr

    # P2: Greedy local search (18-33%)
    t = t0 + budget_s * 0.33
    r, cr = _greedy(bh, bn, rng, t, 5000)
    if cr < bc:
        bh, bc = r, cr

    # P3: Lower-temperature SA (33-48%)
    t = t0 + budget_s * 0.48
    hB3 = _to_bin(bh, bn, rng)
    r, cr = _bin_sa(hB3, bn, rng, t, 0.004)
    if cr < bc:
        bh, bc = r, cr

    # P4: Random and block starts for n0 (48-60%)
    t4 = t0 + budget_s * 0.60
    starts = [_rnd_bin(n0, rng), _blk_bin(n0, rng), _rnd_bin(n0, rng)]
    per4 = max(3.0, (t4 - time.time()) / max(1, len(starts)))
    for s in starts:
        if rem() < 15:
            break
        r, cr = _bin_sa(s, n0, rng, time.time() + per4, 0.02)
        if cr < bc:
            bh, bc, bn = r, cr, n0

    # P5: Different n values (60-74%)
    t5 = t0 + budget_s * 0.74
    ns = [320, 400, 480, 640, 720, 800, 880]
    per5 = max(4.0, (t5 - time.time()) / max(1, len(ns)))
    for n in ns:
        if time.time() > t5 or rem() < 15:
            break
        hr = _rnd_bin(n, rng)
        r, cr = _bin_sa(hr, n, rng, time.time() + per5, 0.02)
        if cr < bc:
            bh, bc, bn = r, cr, n

    # P6: SLSQP continuous refinement (74-90%)
    if rem() > 100:
        t6 = t0 + budget_s * 0.90
        r, cr = _solsqp(bh.astype(float), bn, t6, [5000, 50000, 300000])
        if cr < bc:
            bh, bc = r, cr

    # P7: Projected gradient (90-95%)
    if rem() > 40:
        t7 = t0 + budget_s * 0.95
        for alpha, st in [(100000, 300), (200000, 250)]:
            if rem() < 15:
                break
            r, cr = _pgrad(bh, bn, t7, alpha, st, 0.003)
            if cr < bc:
                bh, bc = r, cr

    # P8: Final binary polish + 2-point continuous (95-99.5%)
    if rem() > 8:
        t8 = t0 + budget_s * 0.995
        hB8 = _to_bin(bh, bn, rng)
        r, cr = _bin_sa(hB8, bn, rng, t8, 0.001)
        if cr < bc:
            bh, bc = r, cr
        if rem() > 3:
            npol = min(int(rem() / 0.001), 300000)
            r, cr = _two_pt(bh.astype(float), bn, rng, t0 + budget_s - 1, npol)
            if cr < bc:
                bh, bc = r, cr

    bc = _eval(bh)
    return bh, bc, bn
