        c5n = _eval(h)
        if c5n < c5c - 1e-17:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _group_move(h, n_iter, rng, gs=5):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        offsets = rng.randint(0, 40, size=gs)
        idx = np.sort((base + offsets) % n)
        d = (3e-8 + 8e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-17:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _ga_refine(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    g = _grad_smooth(h, 5e7, 30)
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-18:
        return best_h, best_c
    for _ in range(n_iter):
        if _ % 500 == 0:
            g = _grad_smooth(h, 5e7, 30)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-18:
                break
        step = (3e-7 + 7e-7 * rng.rand())
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        mg = np.array([g[i], -g[j]])
        mn = np.linalg.norm(mg)
        if mn < 1e-18:
            continue
        d = step * mg / mn
        d += 1e-9 * rng.randn(2)
        d[0] -= d.mean()
        d[1] -= d.mean()
        ni, nj = h[i] + d[0], h[j] + d[1]
        if not (0 <= ni <= 1 and 0 <= nj <= 1):
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-17:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = h[i] - d[0], h[j] - d[1]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B with alpha continuation + Lagrangian dual
    alpha_schedule = [50000, 200000, 1000000, 5000000, 50000000]
    for ai, alpha in enumerate(alpha_schedule):
        if _el() > budget_s * 0.30:
            break
        m_lags = max(15, 60 - ai * 10)
        h_new = _lbfgsb_dual(best_h.copy(), alpha, m_lags, n_dual=8,
                             iters_per_phase=800)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Projected gradient with momentum, multiple configs
    if _el() < budget_s - 300:
        pg_configs = [
            (30000, 35, 1500, 0.003),
            (200000, 50, 1200, 0.002),
            (1000000, 65, 900, 0.001),
            (10000000, 80, 700, 0.0005),
            (100000000, 100, 500, 0.0003),
        ]
        for alpha, ml, steps, s0 in pg_configs:
            if _el() > budget_s - 250:
                break
            h_new = _proj_grad_step(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More L-BFGS-B at high alpha (polish the smooth max)
    if _el() < budget_s - 200:
        for alpha in [200000000, 1000000000]:
            if _el() > budget_s - 160:
                break
            h_new = _lbfgsb_step(best_h, alpha, 25, 0.0, 1500)
            h_new = _repair(h_new, n)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Gradient-aligned refine
    if _el() < budget_s - 100:
        rem = budget_s - 100 - _el()
        ntries = int(rem * 400)
        best_h, best_c5 = _ga_refine(best_h, ntries, rng)

    # Phase 5: Targeted 2-point polish
    if _el() < budget_s - 40:
        rem = budget_s - 40 - _el()
        ntries = int(rem * 800)
        best_h, best_c5 = _targeted_polish(best_h, ntries, rng, 8)

    # Phase 6: Group moves
    if _el() < budget_s - 10:
        rem = budget_s - 10 - _el()
        ntries = int(rem * 350)
        best_h, best_c5 = _group_move(best_h, ntries, rng, 5)

    # Final ultra-fine
    if _el() < budget_s - 1:
        rem = budget_s - 1 - _el()
        ntries = int(rem * 1000)
        best_h, best_c5 = _targeted_polish(best_h, ntries, rng, 5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
