import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_val(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g, idx

def _nesterov(h, alpha, topk, steps, step0, beta, deadline):
    n = len(h)
    f = _smax_val(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        g, _ = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_val(hn, alpha, topk)
        if fn < f - 1e-16:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 30.0)
        else:
            step *= 0.5
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax_val(hn, alpha, topk)
            if fn < f - 1e-16:
                h = hn
                f = fn
            else:
                step *= 0.5
                if step < 1e-12:
                    break
                hn = _proj(h - step * v, n * 0.5)
                fn = _smax_val(hn, alpha, topk)
                if fn < f - 1e-16:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_val(x, alpha, topk)
    def jac(x):
        g, _ = _smax_grad(x, alpha, topk)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _equalize_step(h, topk, n, eta, deadline):
    """Reduce correlation at active lags by equalizing mismatched pairs."""
    c = _correl(h)
    idx = np.argpartition(c, -topk)[-topk:]
    sh_list = [(n - 1) - k for k in idx]
    improved = False
    for sh in sh_list[:2]:
        if sh == 0:
            continue
        if sh > 0:
            pairs = np.arange(n - sh)
        else:
            pairs = np.arange(-sh)
        for p in pairs:
            if time.time() > deadline:
                return h, improved
            i = p
            j = p + abs(sh)
            if j >= n:
                continue
            diff = h[j] - h[i]
            if abs(diff) > 0.05:
                d = eta * diff
                ni = h[i] + d
                nj = h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
        h = _repair(h, n)
        cn = _eval(h)
        if cn > _eval(_repair(h.copy(), n)) + 1e-15:
            break
        improved = True
    return h, improved

def _active_optimize(h, n, deadline, niter, alpha, eta):
    """Iteratively: find active lags, equalize, then Nesterov polish."""
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        active = np.argsort(c)[-3:]
        sh_list = [(n - 1) - k for k in active]
        for sh in sh_list[:2]:
            if sh == 0 or time.time() > deadline:
                continue
            if sh > 0:
                rng_idx = np.arange(n - sh)
            else:
                rng_idx = np.arange(-sh)
            for p in rng_idx:
                if time.time() > deadline:
                    return bh, bc
                i = p
                j = p + abs(sh)
                if j >= n:
                    continue
                d = h[j] - h[i]
                if abs(d) > 0.03:
                    delta = eta * d
                    ni = h[i] + delta
                    nj = h[j] - delta
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h = bh.copy()
                c5c = bc
        if it % 5 == 0:
            h = _nesterov(h, alpha, 2, 500, 0.0003, 0.9, deadline)
            c5n = _eval(h)
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _ultra_fine(h, n, rng, deadline, step0):
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = step0
    for it in range(200000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        if i == j:
            continue
        d = step * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
                step = min(step * 1.01, step0)
            else:
                h[i], h[j] = ni - d, nj + d
                if it % 500 == 499:
                    step *= 0.92
    return bh, bc
