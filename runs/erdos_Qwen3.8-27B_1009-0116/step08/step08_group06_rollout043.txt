    # Or: h' = 0.5 + 0.5 * tanh((h - 0.5) / temperature)
    return np.clip(0.5 + 0.5 * np.tanh((h - 0.5) / temperature), 0, 1)

def _gradient_guided_local(h, n_steps, rng, t_end, step_size=1e-5):

    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        # Compute gradient to identify "bad" positions
        c = _corr_full(h)

        g = np.zeros(n)
        for k in top_10:

            if s >= 0:
                g[:n-s] += (1.0 - h[s:])
                g[s:] -= h[:n-s]
            else:
                sp = -s
                g[sp:] += (1.0 - h[:n-sp])
                g[:n-sp] -= h[sp:]
        
        # Pick position with highest |gradient| (most "profitable" to change)
        # and a nearby position to swap mass
        abs_g = np.abs(g)
        i = rng.choice(n, p=abs_g / abs_g.sum())
        j = (i + rng.randint(1, 5)) % n
        d = step_size * rng.randn()
        # Ensure sum is preserved
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lse_obj(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        return _lse_obj(x, alpha, n)

    def jac(x):
        return _lse_grad(x, alpha, n)

    def cfun(x):
        return x.sum() - target

    def cjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_topk(h, k_top, steps, step0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _s in range(steps):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-k_top:]
        c_vals = c[top_k]
        w = np.exp(50 * (c_vals - c_vals.max()))
        w /= w.sum()
        g = np.zeros(n)
        for idx_k in range(len(top_k)):
            k = top_k[idx_k]
            sh = (n - 1) - k
            wk = w[idx_k]
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n-sp])
                g[:n-sp] -= wk * h[sp:]
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 12.0)
        else:
            step *= 0.35
            if step < 1e-12:
                v = np.zeros(n)
                step = step0 * 0.5
            h = _repair(h_prev - step * v, n)
            f = _eval(h)
    return _repair(h, n)

def _soft_binary_push(h, n, temp):
    h2 = 0.5 + 0.5 * np.tanh((h - 0.5) / temp)
    return _repair(h2, n)

def _gradient_local(h, n_steps, rng, t_end, base_step=2e-5):

    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    c_cache = _corr(h)
    top_15 = np.argsort(c_cache)[-15:]
    for _it in range(n_steps):
        if time.time() > t_end:
            break
        if _it % 200 == 0:
            c_cache = _corr(h)
            top_15 = np.argsort(c_cache)[-15:]
            # Recompute gradient for guidance
            g_abs = np.zeros(n)
            for k in top_15:
                sh = (n - 1) - k
                if sh >= 0:
                    g_abs[:n-sh] += np.abs(1.0 - h[sh:])
                    if sh > 0:
                        g_abs[sh:] += np.abs(h[:n-sh])
                else:
                    sp = -sh
                    g_abs[sp:] += np.abs(1.0 - h[:n-sp])
                    g_abs[:n-sp] += np.abs(h[sp:])

        i = rng.choice(n, p=g_prob)
