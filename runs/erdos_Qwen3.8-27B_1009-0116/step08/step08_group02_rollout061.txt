Actually, I just realized something. Let me re-read the problem statement:

"The evaluation computes: C₅ = max(np.correlate(h, 1-h, mode="full") * dx)"

So C₅ is computed as max of the FULL correlation vector scaled by dx. The correlation has 2n-1 elements. The max could be at any of these.

For the full correlation of h and (1-h):
- At lag 0: Σ h_i(1-h_i)
- At lag k (positive): Σ_{i=0}^{n-1-k} h_i(1-h_{i+k})
- At lag -k: Σ_{i=0}^{n-1-k} h_i(1-h_{i-k}) = Σ_{i=k}^{n-1} h_{i-k}(1-h_i)

So the "overlap" at shift k is the sum of h(x)·(1-h(x+k)) over the valid range, times dx.

Now, the key: for binary h, at lag 0, c_0 = Σ h_i(1-h_i) = 0. So the max is always at lag ≠ 0 for binary h.

For continuous h, c_0 = Σ h_i(1-h_i) > 0, so the max might be at lag 0.

This means: if we push h towards binary, c_0 → 0, and the max shifts to other lags. The optimal solution might be exactly binary or very close to binary.

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

c_k = Σ h_i(1-h_{i+k}) dx = Σ h_i dx - Σ h_i h_{i+k} dx = (sum over valid range of h_i) dx - (autocorrelation at k) dx

For k=0: c_0 = Σ h_i(1-h_i) dx = (total - Σh_i²) dx
For binary h: c_0 = 0

So c_k = (Σ_{valid} h_i) dx - (autocorrelation of h at k) dx

For binary h: c_k = (number of 1s in valid range) dx - (number of pairs (i,i+k) both 1) dx

This is equivalent to: c_k = (number of 1s at positions that are NOT followed by 1 at position +k) dx

To minimize c_k for all k, we want the 1s to be arranged so that for every k, most 1s are followed by 1s. This is like a "clumping" problem.

But we can't have this for ALL k simultaneously. For k near 0, we want clumping. For k near N/2, we want spreading.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _pg(h, alpha, steps, step0, top_k, momentum, tlim):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    ts = time.time()
    for it in range(steps):
        if time.time() - ts > tlim:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if top_k > 0 and top_k < len(c):
            active = np.argsort(c)[-top_k:]
        else:
            active = np.arange(len(c))
        c_a = c[active]
        m_a = c_a.max()
        sv = np.clip(alpha * (c_a - m_a), -400, 400)
        sv -= sv.max()
        w = np.exp(sv)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(active[ii], h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _ in range(20):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-18:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 25.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = max(step * 1.5, step0 * 0.05)
            v *= 0.4
        if it % 400 == 0 and it > 0:
            v *= 0.25
    return _repair(h, n), cur_c

def _scan_worst(h, n_iter, tlim):
    n = len(h)
    bh = h.copy()
    bc, _ = _eval_worst(h)
    ts = time.time()
    dx = 2.0 / n
    for _ in range(n_iter):
        if time.time() - ts > tlim:
            break
        c5c, kw = _eval_worst(h)
        c5c = max(c5c, _eval(h))
        sh = (n - 1) - kw
        if not (0 < sh < n):
            break
        best_gain = 0.0
        best_move = None
        for i in range(n - sh):
            d1 = min(h[i] - 0.01, 1.0 - h[i + sh], 0.04)
            if d1 > 1e-4:
                g1 = d1 * (1.0 - h[i + sh]) * dx
                if g1 > best_gain:
                    best_gain = g1
                    best_move = (i, i + sh, d1, 1)
            d2 = min(h[i + sh] - 0.01, 1.0 - h[i], 0.04)
            if d2 > 1e-4:
                g2 = d2 * (1.0 - h[i]) * dx
                if g2 > best_gain:
                    best_gain = g2
                    best_move = (i, i + sh, d2, -1)
        if best_move is not None:
            i, j, d, direction = best_move
            if direction == 1:
                h[i] -= d
                h[j] += d
            else:
                h[i] += d
                h[j] -= d
            h = _repair(h, n)
            c_new = _eval(h)
            if c_new < bc:
                bc = c_new
                bh = h.copy()
    return bh, bc
