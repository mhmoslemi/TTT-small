    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    a = min(diff, 1.0 - h[idx])

    ones = np.ones(n)

    def obj(x):
        return _smax_obj(x, alpha, topk) + pen * (x.sum() - target) ** 2

    def jac(x):
        return _smax_grad(x, alpha, topk) + 2.0 * pen * (x.sum() - target) * ones

    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 400, 'ftol': 1e-18, 'maxfun': 5000})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _focused_grad(h, deadline):
    n = len(h)
    target = n / 2.0
    c5 = _eval(h)
    best = c5
    best_h = h.copy()
    step = 0.02
    momentum = 0.0
    v = np.zeros(n)
    for it in range(20000):
        if time.time() > deadline:
            break
        if it % 200 == 0:
            c = _all_corr(h)
            topk = max(3, min(12, int(12 * np.exp(-it / 3000.0))))
            alpha = 20000 + 30000 * (it / 5000.0)
            alpha = min(alpha, 500000)
            momentum = min(0.3 + 0.005 * (it / 1000.0), 0.7)
        else:
            topk = 5
            alpha = 100000
            momentum = 0.5
        g = _smax_grad(h, alpha, topk)
        v = momentum * v + (1.0 - momentum) * g
        found = False
        for _ in range(30):
            hn = _proj(h - step * v, target)
            c5n = _eval(hn)
            if c5n < c5 - 1e-16:
                h = hn
                c5 = c5n
                step = min(step * 1.15, 0.1)
                found = True
                if c5n < best - 1e-15:
                    best = c5n
                    best_h = h.copy()
                break
            step *= 0.55
        if not found:
            step *= 0.7
            if step < 1e-12:
                break
    return best_h, best

def _pair_opt(h, niter, rng, deadline):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    stall = 0
    steps = [2e-5, -2e-5, 1e-5, -1e-5, 5e-6, -5e-6, 2e-6, -2e-6, 1e-6, -1e-6]
    for it in range(niter):
        if time.time() > deadline:
            break
        if stall > 8000:
            steps = [1e-6, -1e-6, 5e-7, -5e-7, 2e-7, -2e-7]
            stall = 0
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, min(80, n - 2))) % n
        if i == j:
            continue
        for d in steps:
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5 - 1e-19:
                    c5 = c5n
                    stall = 0
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        else:
            stall += 1
    return bh, bc

def _micro(h, niter, rng, deadline):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    step = 5e-7
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n // 25)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-20:
                c5 = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        if it % 800 == 799:
            step *= 0.96
            step = max(step, 1e-10)
    return bh, bc

def _slsqp_refine(h, alpha, topk, deadline):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return _smax_obj(x, alpha, topk)

    def jac(x):
        return _smax_grad(x, alpha, topk)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': 200, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 6

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP + L-BFGS-B (0-250s)
    dl1 = min(deadline, t0 + 250)
    h = _slsqp_refine(h, 5000, 30, dl1)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h.copy()
    for alpha, tk in [(10000, 25), (50000, 12), (200000, 6)]:
        if time.time() > dl1:
            break
        h = _lbfgs_step(best_h.copy(), alpha, tk, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 2: Focused gradient with momentum (250-500s)
    dl2 = min(deadline, t0 + 500)
    hf, cf = _focused_grad(best_h.copy(), dl2)
    if cf < best_c5:
        best_c5 = cf; best_h = hf.copy()

    # Phase 3: L-BFGS-B high-alpha refinement (500-600s)
    dl3 = min(deadline, t0 + 600)
    for alpha, tk in [(500000, 5), (1500000, 3)]:
        if time.time() > dl3:
            break
        h = _lbfgs_step(best_h.copy(), alpha, tk, dl3)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 4: Pair optimization (600-870s)
    dl4 = min(deadline, t0 + 870)
    hp, cp = _pair_opt(best_h.copy(), 300000, rng, dl4)
    if cp < best_c5:
        best_c5 = cp; best_h = hp.copy()

    # Phase 5: Micro polish (870-end)
    dl5 = deadline
    hm, cm = _micro(best_h.copy(), 800000, rng, dl5)
    if cm < best_c5:
        best_c5 = cm; best_h = hm.copy()

    # Final: one more focused gradient with tiny steps if time remains
    if time.time() < dl5 - 30:
        hf2, cf2 = _focused_grad(best_h.copy(), dl5 - 5)
        if cf2 < best_c5:
            best_c5 = cf2; best_h = hf2.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
