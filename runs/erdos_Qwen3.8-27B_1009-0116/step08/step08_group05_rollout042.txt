import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def _lp_descent(h, n_steps, eps0, m_top, deadline, eps_min=0.0005, eps_decay=0.97):

    best_h, best_c5 = h.copy(), _eval(h)

                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-15:
            stall += 1
            if stall > 20:

                    if c5_cur < best_c5:

                stall = 0

                continue

                if eps < eps_min:

                continue

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        k, sh, wk = idx[ii], (n-1)-idx[ii], w[ii]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-h[sh:])
            if sh > 0: g[sh:] -= wk*h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta = 0.8; step = step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta*v + (1.0-beta)*g
        hn = _proj(h - step*v, n*0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn; step = min(step*1.03, step0*5.0)
        else:
            step *= 0.5
            if step < 1e-11: break
            h = _proj(hp - step*v, n*0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _sweep_polish(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    sss = [5e-7, 2e-7, 1e-7, 5e-8, 2e-8]
    for ss in sss:
        ni2 = max(1, niter // len(sss))
        for _ in range(ni2):
            if time.time() > deadline: return bh, bc
            i = rng.randint(nc); j = (i+1+rng.randint(0,4))%nc
            d = ss*(2*rng.random()-1)
            ni, nj = h[i]+d, h[j]-d

                    if c5n<bc: bc,bh=c5n,h.copy()
                else:

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax_obj(x, alpha, topk)
    def jac(x): return _grad_topk(x, alpha, topk)
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:

                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter':maxit,'ftol':1e-20,'maxls':50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

        t_lo = cor[idx_top[0]] - eps * n
        t_hi = cor[idx_top[0]] + eps * n
        bnds = [(-eps, eps)] * n + [(t_lo, t_hi)]
        try:

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k, sh, wk = idx[ii], (n - 1) - idx[ii], w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):

    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta = 0.82; step = step0
    for _ in range(steps):
        if time.time() > deadline:

        g = _grad_topk(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:

            step = min(step * 1.04, step0 * 6.0)

            step *= 0.5

                break
            h = _proj(h + step * v * beta, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_topk(x, alpha, topk)
    def con_fun(x):
        return x.sum() - n * 0.5
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _sweep_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sss = [5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]

        for _ in range(ni2):

            i = rng.randint(nc)

                    if c5n < bc:

    return bh, bc
