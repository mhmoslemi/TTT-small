def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx_val = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx_val)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline, beta=0.85):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    prev_grad = None
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        # BB step size if we have previous gradient
        if prev_grad is not None:
            dy = g - prev_grad
            y_to_x = np.dot(dy, g - prev_grad)
            y_to_y = np.dot(prev_grad - g, prev_grad - g)
            if y_to_y > 1e-14:

                step = 0.5 * step + 0.5 * np.clip(bb_step, 1e-6, step0 * 5)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx_val = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx_val)
    return g

def _nesterov_bb(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    beta = 0.82
    prev_g = None
    prev_x = None
    best_f = f
    best_h = h.copy()
    stuck = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        # BB step
        if prev_g is not None and np.dot(prev_g, prev_g) > 1e-16:
            s_vec = h - prev_x
            y_vec = g - prev_g
            ys = np.dot(y_vec, s_vec)
            yy = np.dot(y_vec, y_vec)
            if abs(ys) > 1e-14:
                bb = ys / yy
                step = np.clip(0.7 * step + 0.3 * bb, 1e-7, step0 * 8)
        prev_g = g.copy()
        prev_x = h.copy()
        
        v = beta * v + (1.0 - beta) * g
        x_trial = h - step * v
        hn = _proj(x_trial, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        
        if fn < f:
            if fn < best_f:
                best_f = fn
                best_h = hn.copy()
                stuck = 0
            else:
                stuck += 1
            h = hn
            f = fn
            step = min(step * 1.05, step0 * 10.0)
        else:
            v = np.zeros(n)
            step *= 0.5
            hn = _proj(h - step * g, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
                stuck += 1
            else:
                step *= 0.5
                hn = _proj(h - step * g, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    # Restart from best with perturbation
                    h = best_h + 0.001 * np.random.randn(n)
                    h = _repair(h, n)
                    v = np.zeros(n)
                    f = _smax_obj(h, alpha, topk)
                    step = step0 * 0.5
                    stuck = 0
    return _repair(best_h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c

            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))

            c5c = c5n

                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_base = 1e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 15)) % nc

        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
