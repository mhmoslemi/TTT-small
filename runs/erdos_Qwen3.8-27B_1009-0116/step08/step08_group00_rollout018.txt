```python
import cvxpy as cp

def _solve_lp(h, grad_list, f_list):
    """Solve linearized subproblem
    min t
    s.t. f_list[i] + grad_list[i] @ (x - h) <= t
         0 <= x <= 1
         sum(x) = n/2
    """
    n = len(h)
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [x >= 0, x <= 1, cp.sum(x) == n/2]
    for i in range(len(grad_list)):
        constraints.append(f_list[i] + grad_list[i] @ (x - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS)  # or GLOP, or SCS
    return x.value
```

    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sm_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, steps, step0, n):
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _sm_grad(h, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 8.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 8.0)
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step, n_pert=3):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5cur
    for _ in range(n_iter):
        step = base_step * (0.5 + rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
    return best_h, best_c5

def _two_point_polish(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    cur_c5 = best_c5

    # Phase 1: LP cutting-plane method
    m_active = 25
    stuck_count = 0
    max_iter_lp = 5000
    for it in range(max_iter_lp):
        if time.time() - t0 > budget_s * 0.25:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        cmax = corrs.max()
        thresh = cmax - 0.005
        active_mask = corrs >= thresh
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 60:
            active_idx = np.argsort(corrs)[-60:]
        elif len(active_idx) < 5:
            active_idx = np.argsort(corrs)[-m_active:]
        sol = _lp_step(h, active_idx, corrs, n)
        if sol is None:
            break
        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_try = (1.0 - alpha) * h + alpha * sol
            h_try = _repair(h_try, n)
            c5_try = _eval(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                accepted = True
                stuck_count = 0
                break
        if not accepted:
            stuck_count += 1
            if stuck_count > 80:
                h = _repair(h + 0.02 * rng.randn(n), n)
                cur_c5 = _eval(h)
                stuck_count = 0
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    # Phase 2: SLSQP refinement with increasing smoothness
    alpha_seq = [5000, 20000, 100000, 500000, 2000000, 10000000, 50000000, 200000000]
    for alpha in alpha_seq:
        if time.time() - t0 > budget_s * 0.55:
            break
        lift = min(80, max(20, int(25 * np.log10(max(alpha / 5000, 1)) + 20)))
        h_new = _slsqp(best_h, alpha, lift, 1200, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient with momentum
    if time.time() - t0 < budget_s * 0.80:
        configs = [(10000, 40, 600, 0.004), (50000, 50, 500, 0.003),
                   (200000, 60, 400, 0.002), (1000000, 80, 300, 0.001),
                   (5000000, 100, 200, 0.0005)]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s * 0.80:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0, n)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Another LP round from improved point
    if time.time() - t0 < budget_s * 0.90:
        h = best_h.copy()
        cur_c5 = best_c5
        stuck = 0
        for it in range(2000):
            if time.time() - t0 > budget_s * 0.90:
                break
            corrs = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            cmax = corrs.max()
            active_idx = np.where(corrs >= cmax - 0.003)[0]
            if len(active_idx) > 50:
                active_idx = np.argsort(corrs)[-50:]
            elif len(active_idx) < 5:
                active_idx = np.argsort(corrs)[-25:]
            sol = _lp_step(h, active_idx, corrs, n)
            if sol is None:
                break
            acc = False
            for a2 in [1.0, 0.5, 0.25]:
                ht = _repair((1.0 - a2) * h + a2 * sol, n)
                ct = _eval(ht)
                if ct < cur_c5 - 1e-15:
                    h = ht
                    cur_c5 = ct
                    acc = True
                    break
            if not acc:
                stuck += 1
                if stuck > 50:
                    break
            else:
                stuck = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    # Phase 5: Gradient-informed local search
    if time.time() - t0 < budget_s - 40:
        remaining = budget_s - 40 - (time.time() - t0)
        ntries = min(int(remaining * 400), 150000)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 4e-6, 3)

    # Phase 6: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 2:
        h2 = best_h.copy()
        remaining = budget_s - 2 - (time.time() - t0)
        npol = min(int(remaining * 800), 400000)
        best_h, best_c5 = _two_point_polish(h2, npol, rng, 2e-7)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
