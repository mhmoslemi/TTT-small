                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = h.copy()
    mu = 0.75
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.06, step0 * 5.0)
        else:
            step *= 0.5
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _local_adapt(h, n_iter, step0, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    step = step0
    stuck = 0
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hi, hj = h[i], h[j]
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                stuck = 0
                step = min(step * 1.008, step0 * 4.0)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = hi, hj
                stuck += 1
                if stuck > 300:
                    step *= 0.4
                    stuck = 0
                    step = max(step, 1e-9)
        else:
            stuck += 1
            if stuck > 300:
                step *= 0.5
                stuck = 0
                step = max(step, 1e-9)
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out

def _slsqp_run(h, alpha, maxiter, top_k, n):
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def objfn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def gradfn(x):
        return _grad_lse(x, alpha, top_k)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(objfn, h, jac=gradfn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution MGD at n/2
    nc = n // 2
    hc = _repair(_downsample(best_h, 2), nc)
    for alpha, steps, tk, s0 in [(300, 1500, 8, 0.006), (1500, 1200, 6, 0.004), (6000, 800, 5, 0.002)]:
        if time.time() - t0 > 100:
            break
        hc, _ = _mgd(hc.copy(), steps, alpha, tk, s0, rng)
    h_up = _repair(_upsample(hc, 2), n)
    c_up = _eval(h_up)
    if c_up < best_c5:
        best_c5 = c_up
        best_h = h_up.copy()

    # Phase 2: MGD at full resolution, geometric alpha
    alpha_vals = [500, 2000, 8000, 30000, 80000]
    for a in alpha_vals:
        if time.time() - t0 > budget_s - 500:
            break
        steps = max(200, min(2000, 8000 // (a // 100 + 1)))
        tk = max(4, 10 - a // 5000)
        s0 = max(0.0004, 0.005 * 500.0 / a)
        h_t, _ = _mgd(best_h.copy(), steps, a, tk, s0, rng)
        c_t = _eval(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()

    # Phase 3: Binary-inspired restart
    if time.time() - t0 < budget_s - 400:
        med = np.median(best_h)
        h_bin = (best_h > med).astype(float)
        h_bin = _repair(h_bin, n)
        for blend in [0.4, 0.6]:
            if time.time() - t0 > budget_s - 380:
                break
            h_blend = _repair((1 - blend) * best_h + blend * h_bin, n)
            h_t, _ = _mgd(h_blend, 600, 5000, 5, 0.003, rng)
            c_t = _eval(h_t)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 4: SLSQP
    if time.time() - t0 < budget_s - 300:
        for a, tk, mxi in [(8000, 8, 500), (40000, 6, 400), (150000, 4, 300)]:
            if time.time() - t0 > budget_s - 270:
                break
            h_t = _slsqp_run(best_h.copy(), a, mxi, tk, n)
            c_t = _eval(h_t)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 5: MGD micro-refinement
    if time.time() - t0 < budget_s - 200:
        h_t, _ = _mgd(best_h.copy(), 500, 50000, 4, 0.0008, rng)
        c_t = _eval(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()

    # Phase 6: Intensive adaptive local search (bulk of remaining time)
    if time.time() - t0 < budget_s - 60:
        n_iter1 = min(int((budget_s - 200 - (time.time() - t0)) * 2000), 400000)
        best_h, best_c5 = _local_adapt(best_h, n_iter1, 5e-5, rng)

    # Phase 7: Second local pass with smaller step
    if time.time() - t0 < budget_s - 40:
        n_iter2 = min(int((budget_s - 40 - (time.time() - t0)) * 3500), 300000)
        best_h, best_c5 = _local_adapt(best_h, n_iter2, 1e-5, rng)

    # Phase 8: Final ultra-fine polish
    if time.time() - t0 < budget_s - 5:
        n_fin = min(int((budget_s - 5 - (time.time() - t0)) * 5000), 200000)
        best_h, best_c5 = _local_adapt(best_h, n_fin, 2e-6, rng)

    # Phase 9: Tiny MGD nudge
    if time.time() - t0 < budget_s - 2:
        h_t, _ = _mgd(best_h, 100, 200000, 4, 0.0002, rng)
        c_t = _eval(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()
            n_fin2 = min(int((budget_s - 2 - (time.time() - t0)) * 5000), 50000)
            best_h, best_c5 = _local_adapt(best_h, n_fin2, 1e-6, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
