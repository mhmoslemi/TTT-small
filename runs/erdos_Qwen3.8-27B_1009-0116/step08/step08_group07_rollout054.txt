import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection on level-set."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_smax(h, alpha, topk):
    """Gradient of soft-max approximation of max of correlations."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _grad_exact(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _adam_smax(h, alpha0, alpha1, topk, lr, iters, deadline, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with annealed soft-max."""
    n = len(h)
    target = n * 0.5
    best_h, best_f = h.copy(), _eval(h)
    m_hat = np.zeros(n)  # First moment
    v_hat = np.zeros(n)  # Second moment
    t = 0
    for it in range(iters):
        if time.time() > deadline:
            break
        t += 1
        # Anneal alpha
        alpha = alpha0 * (alpha1 / alpha0) ** (it / max(iters, 1))
        g, _ = _grad_smax(h, alpha, topk)
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * g**2
        m_hat_hat = m_hat / (1 - beta1**t)
        v_hat_hat = v_hat / (1 - beta2**t)
        h_new = _proj_sum(h - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps), target)
        f_new = _eval(h_new)
        h = h_new
        if f_new < best_f:
            best_f, best_h = f_new, h.copy()
    return _repair(best_h, n), best_f

def _nesterov_smax(h, alpha, topk, step0, iters, deadline, beta=0.85):

    target = n * 0.5

def _block_anneal(h, niter, rng, deadline, T0=0.005, T1=1e-10):
    n = len(h)

    bh = h.copy()
    bc = c5c

            break

        bs = rng.choice([1, 2, 3, 4, 5, 7, 10, 15])
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        scale = 0.01 * max(T, 1e-10) ** 0.5
        d = scale * rng.randn(bs)
        d -= d.mean()
        seg += d

        hn = h.copy()
        hn[i0:i0+bs] = seg

        if rng.random() < 0.3:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            k = rng.randint(n)
            if i == j or i == k or j == k:
                continue
            d1 = scale * rng.randn()
            d2 = scale * rng.randn()
            d3 = -(d1 + d2)
            hi, hj, hk = h[i] + d1, h[j] + d2, h[k] + d3
            if 0 <= hi <= 1 and 0 <= hj <= 1 and 0 <= hk <= 1:
                h[i], h[j], h[k] = hi, hj, hk
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j], h[k] = hi - d1, hj - d2, hk - d3
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam with multi-stage annealing (fast)
    dl1 = min(deadline, t0 + 40)
    for a0, a1, tk, lr, ns in [(100, 30000, 25, 0.02, 8000),
                                (30000, 150000, 8, 0.005, 6000),
                                (150000, 500000, 3, 0.002, 5000)]:
        if time.time() > dl1:
            break
        h2, c5v = _adam(best_h.copy(), a0, a1, tk, lr, ns, dl1)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 2: Nesterov worst-k (focus on binding constraints)
    dl2 = min(deadline, t0 + 120)
    for tk, st, bm, ns in [(3, 0.004, 0.85, 5000),
                           (5, 0.003, 0.88, 4000),
                           (2, 0.005, 0.83, 4000)]:
        if time.time() > dl2:
            break
        h2, c5v = _nest(best_h.copy(), 'worst', ns, st, dl2, topk=tk, beta=bm)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Nesterov soft-max (smooth landscape)
    dl3 = min(deadline, t0 + 250)
    for al, tk, st, bm, ns in [(50000, 10, 0.003, 0.87, 4000),
                                (200000, 5, 0.002, 0.90, 3500),
                                (10000, 30, 0.004, 0.85, 3000)]:
        if time.time() > dl3:
            break
        h2, c5v = _nest(best_h.copy(), 'smax', ns, st, dl3, alpha=al, topk=tk, beta=bm)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Block annealing (explore basin)
    dl4 = min(deadline, t0 + 550)
    best_h, best_c5 = _anneal(best_h.copy(), 200000, rng, dl4, T0=0.008)

    # Phase 5: Jitter + Nesterov reoptimization (escape)
    dl5 = min(deadline, t0 + 800)
    for j_scale in [0.003, 0.001, 0.005, 0.0005]:
        if time.time() > dl5:
            break
        d = j_scale * rng.randn(n0)
        d -= d.mean()
        hj = _repair(np.clip(best_h + d, 0, 1), n0)
        h2, c5v = _nest(hj, 'smax', 3000, 0.001, dl5, alpha=100000, topk=8, beta=0.90)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 6: Fine Nesterov (last push)
    dl6 = min(deadline, t0 + budget_s - 120)
    h2, c5v = _nest(best_h.copy(), 'worst', 5000, 0.0008, dl6, topk=2, beta=0.92)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Extended fancy polish (main improvement phase)
    best_h, best_c5 = _fancy_polish(best_h, 2000000, rng, deadline, scale0=5e-6)

    # Phase 8: Final ultra-fine polish
    best_h, best_c5 = _fancy_polish(best_h, 1000000, rng, deadline, scale0=1e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
