```python
def _proj(h, n):
    """Project onto {x: 0<=x<=1, sum(x)=n/2}"""
    h = h.copy()
    target = n * 0.5
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_power(h, p):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    K = len(c)
    S = np.sum(c**p)
    if S < 1e-30:
        return np.zeros(n)
    factor = (S / K) ** (1.0/p - 1.0) / K
    weights = c ** (p - 1)
    
    g = np.zeros(n)
    for k in range(K):
        if weights[k] < 1e-18:
            continue
        wk = weights[k]
        sh = n - 1 - k
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]

def _adam_loop(h, n, n_iter, lr, p, beta1, beta2, eps):
    m = np.zeros(n)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    for t in range(1, n_iter+1):
        g = _grad_power(h, p)
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*g**2

        v_hat = v / (1 - beta2**t)
        h_new = _proj(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c = _eval(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c
```

```python
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()

def _a_vec(h, j, shifts):
    """Compute a_k = dc_k/dh[j] for all k."""
    n = len(h)
    K = 2*n - 1
    a = np.zeros(K)
    for k in range(K):
        s = shifts[k]
        if s == 0:
            a[k] = 1.0 - 2.0*h[j]
        elif s > 0:
            if j <= n-1-s:
                a[k] += 1.0 - h[j+s]
            if j >= s:
                a[k] -= h[j-s]
        else:
            sp = -s
            if j >= sp:
                a[k] += 1.0 - h[j-sp]
            if j <= n-1-sp:
                a[k] -= h[j+sp]
    return a
```

Wait, this inner loop over K=1119 values for _a_vec is going to be slow in Python. 1119 iterations with numpy scalar ops each... that's about 1119*2μs ≈ 2.2ms per call. For n=560 calls per sweep, that's 1.2s per sweep. For 500 sweeps, that's 10 minutes. Too slow!

I need to vectorize _a_vec. Let me think about how.

For shift s > 0 and coordinate j:
- a_k gets (1-h[j+s]) if j ≤ n-1-s, i.e., s ≤ n-1-j
- a_k gets -h[j-s] if j ≥ s, i.e., s ≤ j

So for a fixed j:
- Positive shifts contributing (1-h[j+s]): s = 1, 2, ..., min(n-1-j, n-1)
- Positive shifts contributing -h[j-s]: s = 1, 2, ..., min(j, n-1)
- Negative shifts sp > 0 contributing (1-h[j-sp]): sp = 1, 2, ..., min(j, n-1)
- Negative shifts sp > 0 contributing -h[j+sp]: sp = 1, 2, ..., min(n-1-j, n-1)
- Shift 0: 1-2h[j]

I can vectorize this using numpy:

```python
def _a_vec(h, j):
    n = len(h)
    K = 2*n - 1
    a = np.zeros(K)
    # shifts[k] = (n-1) - k
    # k for shift s: k = (n-1) - s
    # Positive shifts: s = 1..n-1, k = n-2 down to 0
    # Negative shifts: s = -(n-1)..-1, k = n down to 2n-2
    
    # Shift 0: k = n-1
    a[n-1] = 1.0 - 2.0*h[j]
    
    # For s > 0: k = n-1-s
    # Contribution (1-h[j+s]) for s <= n-1-j
    max_s1 = min(j, n-1)  # for -h[j-s]
    max_s2 = min(n-1-j, n-1)  # for (1-h[j+s])
    
    if max_s2 > 0:
        s_range = np.arange(1, max_s2+1)
        k_range = (n-1) - s_range
        a[k_range] += (1.0 - h[j+s_range])
    
    if max_s1 > 0:
        s_range = np.arange(1, max_s1+1)
        k_range = (n-1) - s_range
        a[k_range] -= h[j-s_range]
    
    # For s < 0 (sp > 0): k = n-1+sp
    # Contribution (1-h[j-sp]) for sp <= j
    if j > 0:
        sp_range = np.arange(1, j+1)
        k_range = (n-1) + sp_range
        a[k_range] += (1.0 - h[j-sp_range])
    
    max_sp2 = min(n-1-j, n-1)
    if max_sp2 > 0:
        sp_range = np.arange(1, max_sp2+1)
        k_range = (n-1) + sp_range
        a[k_range] -= h[j+sp_range]
    
    return a
```

This is O(n) with vectorized numpy ops. Should be fast (~10μs per call).

Now the coordinate descent step:
- Compute a_j = _a_vec(h, j): O(n)
- Compute a_jp = _a_vec(h, j_prime): O(n)
- F = a_j - a_jp: O(n)
- For p=2: D* = dot(c, F) / dot(F, F): O(n)
- Update c: c -= F * D: O(n)
- Update h[j], h[j_prime]: O(1)

Total per pair: ~4O(n) = 4n
Per sweep: n * 4n = 4n² = 4*313K ≈ 1.25M ops
Per 1000 sweeps: 1.25B ops → about 1-2 seconds in numpy!

This is way faster than the previous approach. Let me use this.

For the max case (p=∞), I need ternary search on D:
- t(D) = max_k (c_k - F_k * D)
- Minimize t(D) over D ∈ [D_lo, D_hi]
- Ternary search: ~40 evals, each O(n)
- Total per pair: 40n = 40*560 = 22K ops
- Per sweep: 560 * 22K = 12.3M ops
- Per 100 sweeps: 1.23B ops → about 1-2 seconds

Great, both are fast enough for many sweeps.

Let me now write the complete solution. I'll aim for:
1. Quick p=2 coordinate descent (1000 sweeps)
2. Max coordinate descent with ternary search (500 sweeps)
3. Adam with power-sum for further refinement
4. Fine local search
5. Ultra-fine polish

Let me also add the multi-start idea: try different partner offsets.

One concern: the coordinate descent with p=2 optimizes sum of squares, which might not be aligned with the max. But it's a great warm-start. The max coordinate descent (step 2) should then converge to the true optimum.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, n):
    target = n * 0.5
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _a_vec(h, j, n):
    K = 2 * n - 1
    a = np.zeros(K)
    a[n - 1] = 1.0 - 2.0 * h[j]
    max_s1 = min(j, n - 1)
    max_s2 = min(n - 1 - j, n - 1)
    if max_s2 > 0:
        sr = np.arange(1, max_s2 + 1)
        a[n - 1 - sr] += (1.0 - h[j + sr])
    if max_s1 > 0:
        sr = np.arange(1, max_s1 + 1)
        a[n - 1 - sr] -= h[j - sr]
    if j > 0:
        sp = np.arange(1, j + 1)
        a[n - 1 + sp] += (1.0 - h[j - sp])
    if max_s2 > 0:
        sp = np.arange(1, max_s2 + 1)
        a[n - 1 + sp] -= h[j + sp]
    return a

def _cd_quad(h, n, n_sweeps, offset):
    c = _corr(h)
    cur_c = c.max()
    for sw in range(n_sweeps):
        for j in range(n):
            jp = (j + offset) % n
            if jp == j:
                continue
            S = h[j] + h[jp]
            aj = _a_vec(h, j, n)
            ajp = _a_vec(h, jp, n)
            F = aj - ajp
            F2 = np.dot(F, F)
            if F2 < 1e-20:
                continue
            D = np.dot(c, F) / F2
            D_lo = max(0.0, S - 1.0) - h[j]
            D_hi = min(1.0, S) - h[j]
            D = max(D_lo, min(D_hi, D))
            if abs(D) < 1e-15:
                continue
            c -= F * D
            h[j] += D
            h[jp] -= D
        cm = c.max()
        if cm < cur_c:
            cur_c = cm
    _repair(h, n)
    return h, _eval(h)

def _cd_max(h, n, n_sweeps, offset, t3_iters=50):
    c = _corr(h)
    cur_c = c.max()
    for sw in range(n_sweeps):
        for j in range(n):
            jp = (j + offset) % n
            if jp == j:
                continue
            S = h[j] + h[jp]
            aj = _a_vec(h, j, n)
            ajp = _a_vec(h, jp, n)
            F = aj - ajp
            D_lo = max(0.0, S - 1.0) - h[j]
            D_hi = min(1.0, S) - h[j]
            if D_hi - D_lo < 1e-14:
                continue
            lo, hi = D_lo, D_hi
            for _ in range(t3_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                t1 = np.max(c - F * m1)
                t2 = np.max(c - F * m2)
                if t1 < t2:
                    hi = m2
                else:
                    lo = m1
            D = (lo + hi) * 0.5
            c -= F * D
            h[j] += D
            h[jp] -= D
        cm = c.max()
        if cm < cur_c:
            cur_c = cm
    _repair(h, n)
    return h, _eval(h)

def _adam_power(h, n, n_iter, lr, p, dec):
    m = np.zeros(n); v = np.zeros(n)
    best_h, best_c = h.copy(), _eval(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    c = _corr(h)
    for t in range(1, n_iter + 1):
        K = len(c)
        S = np.sum(c ** p)
        if S < 1e-30:
            break
        factor = (S / K) ** (1.0 / p - 1.0) / K
        w = c ** (p - 1)
        g = np.zeros(n)
        for k in range(K):
            wks = w[k]
            if wks < 1e-18:
                continue
            sh = (n - 1) - k
            if sh == 0:
                g += wks * (1.0 - 2.0 * h)
            elif sh > 0:
                g[:n - sh] += wks * (1.0 - h[sh:])
                g[sh:] -= wks * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wks * (1.0 - h[:n - sp])
                g[:n - sp] -= wks * h[sp:]
        g *= factor
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        lr_t = lr / (1.0 + t * dec)
        h_try = _proj(h - lr_t * m / ((t - b1**t) ** 0 + 1) / (np.sqrt(v / (t - b2**t) ** 0 + 1) + eps), n)
        c = _corr(h_try)
        cm = c.max()
        if cm < best_c:
            best_c = cm
            best_h = h_try.copy()
        h = h_try
    return best_h, best_c

def _local2(h, n_iter, step, rng):
    n = len(h); cur_c = _eval(h); best_c = cur_c; best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local3(h, n_iter, step, rng):
    n = len(h); cur_c = _eval(h); best_c = cur_c; best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 6 + rng.randint(0, 14)) % n
        d1 = step * rng.randn(); d2 = step * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[i], h[j], h[k] = s0, s1, s2
    return best_h, best_c

def _blob_move(h, n_iter, rng, max_len=8):
    n = len(h); cur_c = _eval(h); best_c = cur_c; best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        L = rng.randint(2, max_len + 1)
        idx = [(i + t) % n for t in range(L)]
        d = 0.002 * rng.randn()
        ok = True
        for ii in idx:
            if not (0 <= h[ii] + d <= 1):
                ok = False; break
        if not ok:
            partner_pool = [x for x in range(n) if x not in idx]
            if not partner_pool:
                continue
            comp_total = -d * L
            p = rng.choice(partner_pool)
            if not (0 <= h[p] - comp_total <= 1):
                continue
            for ii in idx:
                h[ii] += d
            h[p] -= comp_total
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                for ii in idx:
                    h[ii] -= d
                h[p] += comp_total
    return best_h, best_c

    def _el():
```
