import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(v, S):
    x = np.array(v, dtype=float)
    for _ in range(30):
        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-13:
            return x
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _active_grad(h, corr, T=12, tol_frac=0.008):
    n = len(h)
    mx = float(np.max(corr))
    tol = mx - tol_frac * mx
    active = np.where(corr >= tol)[0]
    if len(active) > T:
        active = np.argsort(corr)[-T:]
    g = np.zeros(n)
    for k in active:
        s = int(k) - (n - 1)
        if s >= 0:
            e = n - s
            if e > 0:
                g[:e] += (1.0 - h[s:s + e])
                g[s:] -= h[:e]
        else:
            s2 = -s
            g[s2:] += (1.0 - h[:n - s2])
            g[:n - s2] -= h[s2:]
    c = len(active)
    if c > 0:
        g /= c
    return g

def _peel(h, deadline, T_start=5, T_end=25, base_step=0.02):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    best_c = _c5(h)
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        frac = (time.time() - t0) / max(1e-9, deadline - t0)
        T = int(T_start + (T_end - T_start) * min(1.0, frac * 1.5))
        corr = _corr(h)
        g = _active_grad(h, corr, T=T, tol_frac=0.004 + 0.004 * frac)
        gsum = float(np.abs(g).sum())
        if gsum < 1e-14:
            break
        step = base_step * (0.3) ** frac
        accepted = False
        for _ in range(10):
            h_new = _proj(h - step * g, n * 0.5)
            c_new = _c5(h_new)
            if c_new < best_c * 0.99999 or c_new < _c5(h):
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted:
            # small random nudge
            idx = np.random.randint(n)
            h[idx] += 0.001
            h = _proj(h, n * 0.5)
        it += 1
    return best_h, best_c

def _grad_local(h, rng, deadline, top_k=15):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        frac = (time.time() - t0) / max(1e-9, deadline - t0)
        step = 5e-5 * (0.02) ** (frac * 0.7)
        corr = _corr(h)
        g = _active_grad(h, corr, T=top_k, tol_frac=0.005)
        order = np.argsort(g)
        q = max(5, n // 8)
        lows = order[:q]
        highs = order[-q:]
        # deterministic-ish: pick best pair
        if rng.random() < 0.6:
            i = int(lows[rng.randint(len(lows))])
            j = int(highs[rng.randint(len(highs))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            continue
        sgn = 1 if g[i] < g[j] else -1
        if sgn > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail < 1e-14:
            continue
        mult = 0.5 + 1.5 * rng.random()
        if rng.random() < 0.05:
            mult *= 5.0

        h[i] += d
        h[j] -= d
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn

                best_c = c
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        it += 1

            best_c = c
            best_h = h.copy()
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, m * 0.5)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _scramble(h, rng, amt=0.05):
    n = len(h)
    perm = rng.permutation(n)
    h2 = h[perm].copy()
    h2 = h + amt * (h2 - h)
    return _proj(h2, n * 0.5)
