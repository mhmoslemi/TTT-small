    n = len(h)
    dx = 2.0 / n

            best_c5, best_h = c2, h2

    # Phase 3: SLSQP with moderate alpha (15% time)
    t3 = t0 + budget_s * 0.60
    for alpha in [50000, 500000]:
        if time.time() > t3 - 10:
            break
        best_h = _slsqp(best_h, alpha, 500, 1e-14, target)
        best_c5 = _eval(best_h)

    # Phase 4: Semi-smooth with tighter lb (15% time)
    t4 = t0 + budget_s * 0.75
    lb2 = best_c5 - 0.003
    h4, c4 = _semi_smooth(best_h, t4, 8000, lb2)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: SLSQP high alpha (10% time)
    t5 = t0 + budget_s * 0.85
    best_h = _slsqp(best_h, 5000000, 400, 1e-15, target)
    best_c5 = _eval(best_h)

    # Phase 6: Local search medium (7% time)
    t6 = t0 + budget_s * 0.92
    ntr = max(int((t6 - time.time()) / 0.003), 2000)
    best_h, best_c5 = _local_search(best_h, ntr, rng, t6, 3e-6)

    # Phase 7: Ultra-fine local search (8% time)
    if time.time() < t_global - 5:
        ntr2 = max(int((t_global - 5 - time.time()) / 0.002), 2000)
        best_h, best_c5 = _local_search(best_h, ntr2, rng, t_global - 2, 1e-6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
