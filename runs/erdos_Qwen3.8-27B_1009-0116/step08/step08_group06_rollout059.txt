```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
from cvxpy.reducers.solvers import CLARABEL
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    # Bisection to find shift
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _compute_grads_topk(h, n, k_top):
    """Compute correlation values and top-k gradients."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-k_top:]
    grads = np.zeros((k_top, n))
    for ii, k in enumerate(top_idx):
        sh = (n - 1) - k
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:]) * dx
            if sh > 0:
                g[sh:] -= h[:n-sh] * dx
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp]) * dx
            g[:n-sp] -= h[sp:] * dx
        grads[ii] = g
    return c, top_idx, grads

def _slp_step(h, n, k_top, max_block):
    """One SLP iteration: linearize top-k constraints, solve LP."""
    c, top_idx, grads = _compute_grads_topk(h, n, k_top)
    c_max = c.max()
    
    # LP: minimize t
    # s.t. c_k + grads[i]·(h' - h) ≤ t for top k
    #      sum(h') = n/2
    #      0 ≤ h' ≤ 1
    hp = cp.Variable(n)
    t = cp.Variable()
    
    cons = [hp >= 0, hp <= 1, cp.sum(hp) == n/2.0]
    for i in range(k_top):
        cons.append(c[top_idx[i]] + grads[i] @ (hp - h) <= t)
    
    # Also add all constraints as before-prediction (for safety)
    # Actually just the top-k should be enough for direction
    
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return h.copy()
    
    if hp.value is None:
        return h.copy()
    
    h_new = np.clip(hp.value, 0, 1)
    h_new = _repair(h_new, n)
    
    # Line search
    best_h, best_c = h.copy(), _eval(h)
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
        h_try = _repair(h + alpha * (h_new - h), n)
        c_try = _eval(h_try)
        if c_try < best_c:
            best_h, best_c = h_try, c_try
        if c_try < c_max:  # definitely improving
            break
    return best_h
```

```python
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _all_grads(h, n):
    """Compute gradient of all correlation values w.r.t. h."""
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        sh = (n-1) - k
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:]) * dx
            if sh > 0:
                g[sh:] -= h[:n-sh] * dx
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp]) * dx
            g[:n-sp] -= h[sp:] * dx
        grads[k] = g
    return grads

def _slp_iter(h, n, use_all=True):
    """One SLP iteration using linprog."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    grads = _all_grads(h, n)
    
    if not use_all:
        top_k = 100
        # Find top k
        idx = np.argsort(c)[-top_k:]
        grads_use = grads[idx]
        c_use = c[idx]
    else:
        grads_use = grads
        c_use = c
    
    n_cons = len(c_use)
    # LP: min t s.t. grads_use[i]·h' - t ≤ grads_use[i]·h - c_use[i]
    #              sum(h') = n/2, 0≤h'≤1
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0
    
    A_ub = np.zeros((n_cons, n+1))
    A_ub[:, :n] = grads_use
    A_ub[:, -1] = -1.0
    b_ub = grads_use @ h - c_use
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)]*n + [(-0.01, 1.0)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return _repair(h, n)
    
    h_new = _repair(res.x[:n], n)
    return h_new

def _slp_phase(h, n, t_end, use_all=True):
    best_h, best_c = h.copy(), _eval(h)
    stagnant = 0
    while time.time() < t_end:
        h_new = _slp_iter(h, n, use_all)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-15:
            best_h, best_c = h_new.copy(), c_new
            stagnant = 0
        else:
            stagnant += 1
        h = h_new if c_new <= _eval(h) + 1e-15 else h
        if stagnant > 200:
            # Reset stagnation counter, try perturb
            break
    return best_h, best_c

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g
```

```python
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _local_2pt(h, n_tries, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLP with full linearization (40% budget)
    t1 = t0 + budget_s * 0.40
    best_h, best_c5 = _slp_phase(best_h, n, t1)

    # Phase 2: SLSQP log-sum-exp polishing (20%)
    t2 = t0 + budget_s * 0.60
    for alpha in [5000, 50000, 500000]:
        if time.time() > t2 - 5:
            break
        h_new = _slsqp_run(best_h.copy(), alpha, 600, 1e-15, target)
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_new

    # Phase 3: Second SLP round (12%)
    t3 = t0 + budget_s * 0.72
    if time.time() < t3:
        best_h, best_c5 = _slp_phase(best_h, n, t3)

    # Phase 4: Guided gradient local search (12%)
    t4 = t0 + budget_s * 0.84
    if time.time() < t4:
        ntr = min(int((t4 - time.time()) / 0.003), 15000)
        best_h, best_c5 = _guided_local(best_h, n, ntr, rng, t4)

    # Phase 5: 3-point random local search (7%)
    t5 = t0 + budget_s * 0.91
    if time.time() < t5:
        ntr = min(int((t5 - time.time()) / 0.004), 20000)
        best_h, best_c5 = _local_3pt(best_h, n, ntr, rng, t5)

    # Phase 6: 2-point ultra-fine (5%)
    t6 = t0 + budget_s * 0.96
    if time.time() < t6:
        ntr = min(int((t6 - time.time()) / 0.002), 35000)
        best_h, best_c5 = _local_2pt(best_h, n, ntr, rng, t6)

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
