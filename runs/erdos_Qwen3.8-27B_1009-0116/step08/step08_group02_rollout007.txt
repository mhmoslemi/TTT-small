def _repair(h, n):

        s = h.sum()

            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13: break
    return h

def _eval_full(h):

        g[sp:] -= h[:n-sp]

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cfull(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _ggrad(sh, h, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if sh == 0:
        return (1.0 - 2.0 * h) * dx
    if sh > 0:
        g[sh:] += 1.0 - h[:n - sh]
        g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += 1.0 - h[sp:]
        g[sp:] -= h[:n - sp]
    return g * dx

def _pg(h0, alpha, steps, s0, tk, mu, tl, t0):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    cc = _eval(h)
    bc = cc
    bh = h.copy()
    st = s0
    for it in range(steps):
        if time.time() - t0 > tl:
            break
        c = _cfull(h)
        if tk < len(c):
            idx = np.argpartition(c, -tk)[-tk:]
        else:
            idx = np.arange(len(c))
        ca = c[idx]
        m = ca.max()
        s = np.clip(alpha * (ca - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-15:
                g += w[ii] * _ggrad((n - 1) - idx[ii], h, n)
        v = mu * v + (1.0 - mu) * g
        sl = st
        taken = False
        for _ in range(20):
            hn = _proj(h - sl * v, n * 0.5)
            cn = _eval(hn)
            if cn < cc - 1e-15:

                cc = cn
                if cn < bc:
                    bc = cn
                    bh = h.copy()
                st = min(sl * 1.05, s0 * 50)
                taken = True
                break
            sl *= 0.5
        if not taken:
            st = min(st * 1.2, s0 * 50)
        if it % 100 == 99:
