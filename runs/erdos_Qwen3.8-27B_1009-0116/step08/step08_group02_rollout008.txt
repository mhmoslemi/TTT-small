```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(np.where(diff > 0, 1.0 - h, h))
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_fn(x, alpha, top_k, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cs = c[-top_k:] if top_k > 0 else c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_fn(x, alpha, top_k, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:] if top_k > 0 else np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, kk in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        sh = (n - 1) - kk
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[ii] * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= w[ii] * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[ii] * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[ii] * x[i2 + sp]
    return g * dx

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_top(h, alpha, maxiter, top_k):
    n = len(h)
    args = (alpha, max(1, top_k), n)
    res = minimize(_obj_fn, h, args=args, jac=_grad_fn, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return _repair(res.x, n)

def _proj_grad(h, alpha, steps, step0, top_k, momentum):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        c = _eval_all(h)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -300, 300)
        s -= s.max()
        w = np.exp(s)

            else:

                g[i1] += w[ii] * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[ii] * h[i2 + sp]
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(15):

            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:

                cur_c = c_new
                step = min(step * 1.15, step0 * 8.0)
                taken = True
                break
            step *= 0.35
        if not taken:

        if it % 300 == 0:
            v *= 0.2
    return _repair(h, n)

def _informed_search(h, n_iter, rng, step_size):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    c_all = _eval_all(h)
    for _ in range(n_iter):
        k_w = int(np.argmax(c_all))
        sh = (n - 1) - k_w
        if sh >= 0:
            i = rng.randint(n - sh)
            j = i + sh
        else:
            sp = -sh
            j = rng.randint(n - sp)
            i = j + sp
        d = step_size * rng.randn()
        ni, nj = h[i] - d, h[j] + d
        if 0.0 <= ni and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                c_all = _eval_all(h)
            else:
                h[i], h[j] = ni + d, nj - d
        if rng.rand() < 0.25:
            a = rng.randint(n)
            b = (a + 1 + rng.randint(0, 4)) % n
            d2 = step_size * 0.6 * rng.randn()
            na, nb = h[a] + d2, h[b] - d2
            if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:
                h[a], h[b] = na, nb
                c_new = _eval(h)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    c_all = _eval_all(h)
                else:
                    h[a], h[b] = na - d2, nb + d2
    return best_h, best_c

def _fft_kick(h, rng, amp):
    n = len(h)
    h_hat = np.fft.rfft(h).copy()
    nk = min(12, n // 2 - 1)
    for k in range(1, nk + 1):
        h_hat[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(h_hat, n)
    return _proj(h_new, n * 0.5)

def _fourier_opt(h, rng, n_K, n_iter, lr):
    n = len(h)
    h_hat = np.fft.rfft(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_iter):
        c_new = _eval(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        kk = rng.randint(1, min(n_K, n // 2 - 1))
        dd = lr * rng.randn()
        h_hat[kk] += dd
        h_try = np.fft.irfft(h_hat, n)
        h_try = _proj(h_try, n * 0.5)
        c_try = _eval(h_try)
        if c_try < c_new:
            h = h_try
            h_hat = np.fft.rfft(h)
        else:
            h_hat[kk] -= dd
    return best_h, best_c

    for _ in range(count):
        _eval(h)
    return 1.0 / max((time.time() - t) / count, 1e-8)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    best_h = _repair(h0, n)
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP quick refinement (50s)
    for alpha, tk, mit in [(3000, 0, 250), (20000, 4, 200), (150000, 2, 150)]:
        if _el() > 50:
            break
        h_new = _slsqp_top(best_h, alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: Projected gradient (100s)
    if _el() < 180:
        for alpha, steps, s0, tk, mu in [(5000, 400, 0.005, 6, 0.7),
                                         (50000, 300, 0.003, 4, 0.8),
                                         (300000, 200, 0.001, 3, 0.85)]:
            if _el() > 180:
                break
            h_new = _proj_grad(best_h.copy(), alpha, steps, s0, tk, mu)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 3: Fourier-domain optimization (80s)
    if _el() < budget_s - 420:
        time_left = budget_s - 420 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        evals_s = 1.0 / eps
        n_iter = min(int(time_left * evals_s * 0.4), 300000)
        best_h, best_c5 = _fourier_opt(best_h, rng, 15, n_iter, 0.02)

    # Phase 4: SLSQP after Fourier (60s)
    if _el() < budget_s - 300:
        for alpha, tk, mit in [(500000, 2, 200), (3000000, 1, 150)]:
            if _el() > budget_s - 300:
                break
            h_new = _slsqp_top(best_h, alpha, mit, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Informed local search targeting worst lag (main phase)
    if _el() < budget_s - 120:
        time_left = budget_s - 120 - _el()
        rate = _estimate_rate(best_h)
        ntries = min(int(time_left * rate * 0.7), 6000000)
        best_h, best_c5 = _informed_search(best_h, ntries, rng, 1.5e-5)

    # Phase 6: Fourier kicks to escape (if time)
    if _el() < budget_s - 60:
        for _ in range(5):
            if _el() > budget_s - 40:
                break
            h_kick = _fft_kick(best_h, rng, 0.03)
            h_kick = _slsqp_top(h_kick, 50000, 80, 2)
            c5k = _eval(h_kick)
            if c5k < best_c5:
                best_c5 = c5k
                best_h = h_kick.copy()
            h_short = _informed_search(h_kick, 20000, rng, 1e-5)
            if h_short[1] < best_c5:
                best_c5 = h_short[1]
                best_h = h_short[0].copy()

    # Phase 7: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        time_left = budget_s - 2 - _el()
        rate = _estimate_rate(best_h)
        ntries = min(int(time_left * rate * 0.9), 12000000)
        for _ in range(ntries):
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            d = 3e-7 * rng.randn()
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                best_h[i], best_h[j] = ni, nj
                c5n = _eval(best_h)
                if c5n < best_c5:
                    best_c5 = c5n
                else:
                    best_h[i], best_h[j] = ni - d, nj + d
            if _el() > budget_s - 1:
                break

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
