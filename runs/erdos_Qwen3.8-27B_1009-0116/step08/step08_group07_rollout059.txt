from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / max(s, 1e-12)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < c.size:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    ss = s - s.max()
    return m + np.log(np.sum(np.exp(ss))) / alpha

def _grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < c.size:
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(c.size); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(idx.size):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii]); sh = n - 1 - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, topk, maxiter, deadline):
    n = len(h)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_obj, h, jac=_grad, args=(alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return _repair(r.x, n)
    except Exception:
        return h

def _exact_grad(h):
    """Gradient of the exact max correlation (subgradient)"""
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - 1e-6)[0]
    g = np.zeros(n)
    for k in active:

        if sh >= 0:

        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / max(s, 1e-12)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < c.size:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    ss = s - s.max()
    return m + np.log(np.sum(np.exp(ss))) / alpha

def _grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < c.size:
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(c.size); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(idx.size):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii]); sh = n - 1 - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, topk, maxiter, deadline):
    n = len(h)
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        r = minimize(_obj, h, jac=_grad, args=(alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return _repair(r.x, n)
    except Exception:
        return h

def _exact_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - 1e-6)[0]
    g = np.zeros(n)
    for k in active:
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_descent(h, steps, deadline, step0=0.002):
    n = len(h)
    target = n * 0.5
    best_h, best_f = h.copy(), _eval(h)
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g, m = _exact_grad(h)
        improved = False
        for _ in range(30):
            cand = _proj(h - step * g, target)
            f_new = _eval(cand)
            if f_new < best_f - 1e-15:
                h = cand
                if f_new < best_f:
                    best_f, best_h = f_new, h.copy()
                step = min(step * 1.25, step0 * 30.0)
                improved = True
                break
            step *= 0.5
        if not improved:
            step = step0
    return best_h, best_f

def _adam_opt(h, steps, lr, alpha, topk, deadline):
    n = len(h)
    target = n * 0.5
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    tc = 0
    best_h, best_f = h.copy(), _eval(h)
    for _ in range(steps):
        if time.time() > deadline:
            break
        tc += 1
        g = _grad(h, alpha, topk)
        m_hat = 0.9 * m_hat + 0.1 * g
        v_hat = 0.999 * v_hat + 0.001 * g * g
        mh = m_hat / (1 - 0.9 ** tc)
        vh = v_hat / (1 - 0.999 ** tc)
        h_new = _proj(h - lr * mh / (np.sqrt(vh) + 1e-8), target)
        f_new = _eval(h_new)
        if f_new < best_f:
            h = h_new
            best_f, best_h = f_new, h.copy()
    return best_h, best_f

def _polish(h, niter, rng, deadline, scale=0.0003):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_restart(h, rng, scale, deadline):
    n = len(h)
    c5 = _eval(h)
    d = scale * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    h2 = _slsqp(hn, 50000, 10, 200, deadline)
    c5n = _eval(h2)
    if c5n < c5:
        return h2, c5n
    return h, c5
