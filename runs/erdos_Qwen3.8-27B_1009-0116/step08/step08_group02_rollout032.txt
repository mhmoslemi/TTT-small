        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k_idx, h, n):
    sh = k_idx - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += -h[sh:]
        g[sh:] += (1.0 - h[:n - sh])
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] += -h[sp:]
    return g * (2.0 / n)

def _grad_topk(h, n, top_k, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-10:
            g += w[ii] * _grad_lag(idx[ii], h, n)
    return g

def _nesterov_pg(h, n_steps, step0, top_k, alpha, momentum):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for it in range(n_steps):
        y = _proj(h + momentum * v, n * 0.5)
        g = _grad_topk(y, n, top_k, alpha)
        v_new = momentum * v + g
        taken = False
        for _t in range(18):
            h_new = _proj(h + v_new, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                v = v_new
                cur_c = c_new
                step = min(step * 1.06, step0 * 8)
                taken = True
                break
            v_new = v_new * 0.5
        if not taken:
            v = v * 0.3
        if it % 80 == 79:
            c_check = _eval(h)
            if c_check < best_c:
                best_c = c_check
                best_h = h.copy()
    c_final = _eval(h)
    if c_final < best_c:
        best_c, best_h = c_final, h.copy()
    return _repair(best_h, n), _eval(best_h)

def _slsqp_focus(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        return _grad_topk(x, n, top_k, alpha)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_descend(h, n_iter, rng):
    n = len(h)
    dx = 2.0 / n
    best_h, best_c = h.copy(), _eval(h)
    h = h.copy()
    for it in range(n_iter):
        c_all = np.correlate(h, 1.0 - h, mode='full') * dx
        k_w = int(np.argmax(c_all))
        sh = k_w - (n - 1)
        if sh >= 0:
            n_t = n - sh
            contrib = h[sh:sh + n_t] * (1.0 - h[:n_t])
            pairs = [(i + sh, i) for i in np.argsort(contrib)[-6:]]
        else:
            sp = -sh
            n_t = n - sp
            contrib = h[sp:sp + n_t] * (1.0 - h[:n_t])
            pairs = [(i, i + sp) for i in np.argsort(contrib)[-6:]]
        for i1, i2 in pairs:
            if h[i1] > 0.005 and (1.0 - h[i2]) > 0.005:
                d = rng.uniform(0.0005, 0.003)
                h[i1] -= d
                h[i2] += d
            elif h[i2] > 0.005 and (1.0 - h[i1]) > 0.005:
                d = rng.uniform(0.0005, 0.003)
                h[i2] -= d
                h[i1] += d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        elif c_new > best_c + 0.0005:
            h = best_h.copy()
    return best_h, best_c

def _anneal_2pt(h, n_iter, rng, step0, temp):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    cur_c = best_c
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = step0 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            t = temp * max(0.0, 1.0 - it / float(n_iter))
            if c_new <= cur_c or (t > 1e-14 and rng.random() < np.exp(-(c_new - cur_c) / t)):
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n, h0 = 560, np.full(560, 0.5)

    best_h = _repair(h0, n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine SLSQP
    for ratio in [7, 4, 2]:
        if _el() > 70:
            break
        nc = max(32, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        for alpha in [2000, 20000, 200000]:
            if _el() > 70:
                break
            hc = _slsqp_focus(hc, alpha, 250, 0)
            h_up = _repair(_upsample(hc, n // nc), n)
            if _eval(h_up) < best_c5:
                best_c5 = _eval(h_up)
                best_h = h_up.copy()

    # Phase 2: SLSQP full resolution
    for alpha, tk, mit in [(5000, 0, 500), (30000, 3, 400), (150000, 5, 350), (800000, 3, 300), (4000000, 2, 250)]:
        if _el() > 190:
            break
        h_new = _slsqp_focus(best_h.copy(), alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()

    # Phase 3: Nesterov projected gradient
    for alpha, steps, s0, tk, mu in [(5000, 500, 0.006, 1, 0.85), (30000, 400, 0.005, 2, 0.85),
                                      (100000, 350, 0.004, 3, 0.8), (500000, 300, 0.003, 4, 0.75),
                                      (2000000, 250, 0.002, 3, 0.7), (8000000, 200, 0.001, 2, 0.6)]:
        if _el() > 400:
            break
        h_new, c_new = _nesterov_pg(best_h.copy(), steps, s0, tk, alpha, mu)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 4: Second SLSQP
    if _el() < budget_s - 350:
        for alpha, tk, mit in [(3000000, 2, 200), (12000000, 1, 150)]:
            if _el() > budget_s - 300:
                break
            h_new = _slsqp_focus(best_h.copy(), alpha, mit, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5, best_h = c5n, h_new.copy()

    # Phase 5: Pair descent
    if _el() < budget_s - 220:
        _t = _el()
        for _ in range(20):
            _eval(best_h)
        eps = max((_el() - _t) / 20.0, 1e-6)
        ntries = min(int((budget_s - 220 - _el()) / (eps * 12)), 80000)
        best_h, best_c5 = _pair_descend(best_h, max(ntries, 200), rng)

    # Phase 6: Annealed 2-point
    if _el() < budget_s - 120:
        _t = _el()
        for _ in range(20):
            _eval(best_h)
        eps = max((_el() - _t) / 20.0, 1e-6)
        ntries = min(int((budget_s - 120 - _el()) / eps * 0.5), 200000)
        best_h, best_c5 = _anneal_2pt(best_h, ntries, rng, 2e-5, 5e-7)

    # Phase 7: Ultra-fine polish
    if _el() < budget_s - 10:
        t_left = budget_s - 10 - _el()
        _t = _el()
        for _ in range(20):
            _eval(best_h)
        eps = max((_el() - _t) / 20.0, 1e-6)
        npol = min(int(t_left / eps * 0.7), 8000000)
        best_h, best_c5 = _anneal_2pt(best_h, npol, rng, 1e-6, 1e-10)
        # Final micro-polish
        if _el() < budget_s - 3:
            np2 = min(int((budget_s - 3 - _el()) / eps * 0.7), 4000000)
            best_h, best_c5 = _anneal_2pt(best_h, np2, rng, 3e-7, 0.0)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
