def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

def _analytic_2pt(h, i, j, n):
    """Find optimal d for h[i]+=d, h[j]-=d using analytic minimax."""
    # Compute current correlation
    corr = np.correlate(h, 1.0-h, mode='full')
    # k range: 0 to 2n-2, shift s = k - (n-1)
    # Top shifts
    top_idx = np.argsort(corr)[-10:]
    
    # For shift s, the change in corr due to h[i]+=d is:
    # corr[k] gets d * (1-h[i-s]) if i-s in [0,n-1] [h[i] is in "a"]
    # corr[k] gets -d * h[i+s] if i+s in [0,n-1] [h[i] is in "b"=1-h]
    # Similarly for h[j]-=d with sign flip.
    
    # Actually let me just compute D_k numerically for the top shifts
    s_center = n - 1  # center of full correlation
    D_vals = []
    D_idxs = []
    for k in top_idx:
        s = k - s_center  # the actual shift in indices
        # Contribution of +d at position i to corr[k]:
        # In np.correlate(a,b,'full')[k] = sum_m a[m] * b[m - k + n - 1]
        # = sum_m h[m] * (1-h[m-k+n-1])
        # h[i] appears when m=i: contribution = h[i]*(1-h[i-k+n-1]) → derivative = (1-h[i-k+n-1]) if valid
        # h[i] appears in (1-h[...]) when m-k+n-1=i → m = i+k-n+1: contribution = h[m]*(-1) = -h[m]
        #   so derivative wrt h[i] = -h[i+k-n+1] if valid
        
        di = 0.0
        b_idx = i - k + n - 1  # for the (1-h[...]) term
        if 0 <= b_idx < n:
            di += (1.0 - h[b_idx])
        a_idx = i + k - n + 1  # for the h[m] term (m = i+k-n+1)
        if 0 <= a_idx < n:
            di -= h[a_idx]
        
        dj = 0.0
        b_idx_j = j - k + n - 1
        if 0 <= b_idx_j < n:
            dj += (1.0 - h[b_idx_j])
        a_idx_j = j + k - n + 1
        if 0 <= a_idx_j < n:
            dj -= h[a_idx_j]
        
        # corr[k] += d * di - d * dj = d * (di - dj)
        D_vals.append(di - dj)
        D_idxs.append(k)
    
    D_vals = np.array(D_vals)
    D_idxs = np.array(D_idxs)
    c_vals = corr[D_idxs]
    
    # Minimize max(c_vals + D_vals * d)
    # The minimum of max of lines is at the intersection of the top 2 lines
    # at the optimal point.
    
    # Feasible range for d
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    
    # Try the intersections of all pairs
    best_d = 0.0
    best_obj = corr.max()
    for a in range(len(c_vals)):
        for b in range(a+1, len(c_vals)):
            if abs(D_vals[a] - D_vals[b]) < 1e-15:
                continue
            d_star = (c_vals[b] - c_vals[a]) / (D_vals[a] - D_vals[b])
            d_star = np.clip(d_star, d_min, d_max)
            obj = (c_vals + D_vals * d_star).max()
            if obj < best_obj:
                best_obj = obj
                best_d = d_star
    
    # Also check boundaries
    for d_cand in [d_min, d_max, 0.0]:
        obj = (c_vals + D_vals * d_cand).max()
        if obj < best_obj:
            best_obj = obj
            best_d = d_cand
    
    return best_d

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _repair(y + mu * v, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
            step = min(step * 1.02, step0 * 3.0)
        else:
            step *= 0.7
            step = max(step, step0 * 0.05)
    return best_h, best_c

def _optimal_d(corr, h, i, j, top_idx, n):
    """Find optimal d for h[i]+=d, h[j]-=d using top-shift analysis."""
    sc = n - 1
    K = len(top_idx)
    D = np.zeros(K)
    for t, k in enumerate(top_idx):
        s = k - sc
        # h[i] contribution to corr[k]:
        bi = i - s
        di = 0.0
        if 0 <= bi < n:
            di += 1.0 - h[bi]
        ai = i + s
        if 0 <= ai < n:
            di -= h[ai]
        # h[j] contribution:
        bj = j - s
        dj = 0.0
        if 0 <= bj < n:
            dj += 1.0 - h[bj]
        aj = j + s
        if 0 <= aj < n:
            dj -= h[aj]
        D[t] = di - dj  # corr[k] += d*D[t]
    cv = corr[top_idx]
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    if d_min >= d_max:
        return 0.0
    best_d = 0.0
    best_obj = cv.max()
    for a in range(K):
        for b in range(a + 1, K):
            dd = D[a] - D[b]
            if abs(dd) < 1e-15:
                continue
            ds = (cv[b] - cv[a]) / dd
            ds = float(np.clip(ds, d_min, d_max))
            obj = (cv + D * ds).max()
            if obj < best_obj:
                best_obj = obj
                best_d = ds
    for dc in (d_min, d_max):
        obj = (cv + D * dc).max()
        if obj < best_obj:
            best_obj = obj
            best_d = dc
    return best_d

def _analytic_2pt(h, t_end, rng):
    """Main local search using analytic 2-point line search."""
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    improvements = 0
    total_iters = 0
    while time.time() < t_end:
        total_iters += 1
        # Choose (i, j) pair
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 40)) % n
        
        # Compute correlation and top shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-12:]
        
        # Find optimal d
        d = _optimal_d(corr, h, i, j, top_idx, n)
        if abs(d) < 1e-10:
            continue
        
        # Apply and verify
        h[i] += d
        h[j] -= d
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        c_new = _eval(h)
        if c_new < cur_c:
            cur_c = c_new
            improvements += 1
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
            # Restore exactly
        # Periodic repair
        if total_iters % 500 == 0:
            h = _repair(h, n)
            cur_c = _eval(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def _sweep_2pt(h, t_end, rng):
    """Cycling sweep through all i with nearby j."""
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    while time.time() < t_end:
        offset_range = [1+int(x) for x in rng.randint(1, 25, size=8)]
        for i in range(n):
            if time.time() > t_end:
                break
            for off in offset_range:
                j = (i + off) % n
                corr = np.correlate(h, 1.0 - h, mode='full')
                top_idx = np.argsort(corr)[-10:]
                d = _optimal_d(corr, h, i, j, top_idx, n)
                if abs(d) > 1e-10:
                    h[i] += d
                    h[j] -= d
                    h[i] = np.clip(h[i], 0, 1)
                    h[j] = np.clip(h[j], 0, 1)
                    c_new = _eval(h)
                    if c_new < cur_c:
                        cur_c = c_new
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
        h = _repair(h, n)
        cur_c = _eval(h)
    return best_h, best_c
