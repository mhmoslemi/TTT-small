def _downsample(h, m):
    n = len(h)
    if m >= n:
        return h.copy()
    ratio = n // m
    return h[:m * ratio].reshape(m, ratio).mean(axis=1).copy()

def _upsample(h, n_target):

    if n_target <= n:
        return h[:n_target].copy()

    out = np.repeat(h, ratio)
    if len(out) < n_target:
        out = np.concatenate([out, h[:n_target - len(out)]])
    return out[:n_target]
