Let me also think about the structured greedy more carefully:

For the worst lag k (with shift sh):
- Gradient g_i = ∂c_k/∂h[i]
- We want to decrease c_k, so we want to move in direction -g
- Projected: h_new = proj(h - step * g, target)
- But this changes ALL coordinates. For a 2-point move, we pick the most "beneficial" pair.

The most beneficial 2-point move: decrease h[i*] by δ where g[i*] is max, increase h[j*] by δ where g[j*] is min.
The improvement is approximately δ * (g[i*] - g[j*]).

But we need to check the actual improvement (nonlinearity). So:
1. Compute g for worst lag
2. Pick i* = argmax(g), j* = argmin(g)
3. Try δ = step_size
4. Evaluate, accept if better

        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _slq_obj(x):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / _st['n'])
    tk = _st['top_k']
    if 0 < tk < len(c):
        idx = np.argsort(c)[-tk:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(_st['alpha'] * (cs - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / _st['alpha']

def _slq_grad(x):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / _st['n'])
    tk = _st['top_k']
    n = _st['n']
    if 0 < tk < len(c):
        idx = np.argsort(c)[-tk:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(_st['alpha'] * (cs - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_lag(idx[ii], x, n)
    return g

def _slq_cfun(x):
    return x.sum() - _st['n'] / 2.0

def _slq_cjac(x):
    return np.ones(_st['n'])

def _slsqp(h0, alpha, maxiter, top_k):
    n = len(h0)
    _st['alpha'] = alpha
    _st['top_k'] = top_k
    _st['n'] = n
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _slq_cfun, 'jac': _slq_cjac}
    try:
        res = minimize(_slq_obj, h0, jac=_slq_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h0

def _smooth_grad(h, alpha, top_k):
    n = len(h)
    c = _corr(h)
    if 0 < top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_lag(idx[ii], h, n)
    return g

def _pg_nesterov(h0, alpha_sweep, steps_each, step0, top_k, mu):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    step = step0
    for alpha in alpha_sweep:
        for it in range(steps_each):
            g = _smooth_grad(h, alpha, top_k)
            v = mu * v + (1.0 - mu) * g
            vn = np.linalg.norm(v)
            if vn < 1e-20:
                break
            taken = False
            for _t in range(18):
                h_new = _proj(h - step * v, n * 0.5)
                c_new = _eval(h_new)
                if c_new < cur_c - 1e-17:
                    h = h_new
                    cur_c = c_new
                    step = min(step * 1.06, step0 * 40.0)
                    taken = True
                    break
                step *= 0.4
            if not taken:
                step = max(step * 1.4, step0 * 0.03)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            if it % 400 == 0:
                v *= 0.4
        mu = min(mu * 1.02, 0.92)
    return _repair(best_h, n), best_c

def _greedy_lag(h, n_iter, rng, lag_candidates):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    base_step = 4e-5
    for it in range(n_iter):
        k_w = lag_candidates[it % len(lag_candidates)]
        g = _grad_lag(k_w, h, n)
        i_s = int(np.argmax(g))
        j_s = int(np.argmin(g))
        if i_s == j_s:
            continue
        d = base_step * rng.randn()
        oi, oj = h[i_s], h[j_s]
        if g[i_s] > g[j_s]:
            ni, nj = oi - d, oj + d
        else:
            ni, nj = oi + d, oj - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i_s], h[j_s] = ni, nj
            c_new = _eval(h)
            if c_new < c5c:
                c5c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i_s], h[j_s] = oi, oj
    return best_h, best_c

def _fine_polish(h, n_iter, rng, step_size):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step_size * rng.randn()
        oi, oj = h[i], h[j]
        ni, nj = oi + d, oj - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < c5c:
                c5c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = oi, oj
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.tile(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0
    _t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse SLSQP (30s)
    for ratio in [4, 2]:
        nc = max(28, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        for alpha_val in [2000, 20000]:
            if _elapsed() > 30:
                break
            hc = _slsqp(hc, alpha_val, 150, 0)
        h_up = _repair(_upsample(hc, n // nc), n)
        c_up = _eval(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()

    # Phase 2: Full-grid SLSQP (70s)
    slsqp_cfgs = [(4000, 0, 400), (15000, 6, 300), (60000, 4, 250), (200000, 3, 200)]
    for alpha_val, tk, mit in slsqp_cfgs:
        if _elapsed() > 100:
            break
        h_new = _slsqp(best_h, alpha_val, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Nesterov projected gradient (400s)
    alphas_pg = [
        (500, 2000, 0.005, 8, 0.6),
        (2000, 2000, 0.004, 6, 0.7),
        (8000, 1500, 0.003, 5, 0.75),
        (30000, 1200, 0.002, 4, 0.8),
        (100000, 1000, 0.0015, 3, 0.82),
        (400000, 800, 0.001, 3, 0.85),
        (2000000, 600, 0.0006, 2, 0.88),
        (10000000, 400, 0.0003, 2, 0.9),
    ]
    for sv in alphas_pg:
        if _elapsed() > 500:
            break
        a, stp, s0, tk, mu = sv
        h_new, c5n = _pg_nesterov(best_h.copy(), [a], stp, s0, tk, mu)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: Greedy structured local search (400s)
    if _elapsed() < budget_s - 80:
        time_left = budget_s - 80 - _elapsed()
        # Estimate eval rate
        _t = time.time()
        for _ in range(30):
            _eval(best_h)
        eps = max((time.time() - _t) / 30.0, 1e-7)
        evals_per_s = 1.0 / eps
        ntries = min(int(time_left * evals_per_s * 0.7), 3000000)
        # Get top lags
        c_all = _corr(best_h)
        top_lags_list = np.argsort(c_all)[-5:].tolist()
        # Increase top_lags if they converge
        best_h, best_c5 = _greedy_lag(best_h, ntries, rng, top_lags_list)
        # Second pass with refined lags
        if _elapsed() < budget_s - 50:
            c_all = _corr(best_h)
            top_lags_list = np.argsort(c_all)[-3:].tolist()
            ntries2 = min(int((budget_s - 50 - _elapsed()) * evals_per_s * 0.7), 2000000)
            best_h, best_c5 = _greedy_lag(best_h, ntries2, rng, top_lags_list)

    # Phase 5: Fine 2-point polish (70s)
    if _elapsed() < budget_s - 3:
        time_left = budget_s - 3 - _elapsed()
        _t = time.time()
        for _ in range(30):
            _eval(best_h)
        eps = max((time.time() - _t) / 30.0, 1e-7)
        evals_per_s = 1.0 / eps
        npol = min(int(time_left * evals_per_s * 0.8), 8000000)
        best_h, best_c5 = _fine_polish(best_h, npol, rng, 1e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
