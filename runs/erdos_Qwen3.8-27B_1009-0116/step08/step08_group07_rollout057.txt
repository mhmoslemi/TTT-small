            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_fjac(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    ws = w.sum()
    obj = m + (sm + np.log(ws)) / alpha
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        wi = w[ii] / ws
        if wi < 1e-13:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = wi * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

def _slsqp(h, alpha, topk, maxiter, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _smax_fjac(x, alpha, topk)[0], h,
                       jac=lambda x: _smax_fjac(x, alpha, topk)[1],
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h

def _nesterov(h, steps, step0, deadline, beta, tol):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    best_h, best_f = h.copy(), f
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        dx = 2.0 / n
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:n - sp] -= h[sp:]
        g *= dx / len(active)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _subgrad_top(h, niter, deadline, ntop, step0):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h); bh = h.copy()
    step = step0
    for _ in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        top = np.argsort(c)[-ntop:]
        g = np.zeros(n)
        dx = 2.0 / n
        for k in top:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:n - sp] -= h[sp:]
        g *= dx / len(top)
        hn = _proj(h - step * g, target)
        fn = _eval(hn)
        if fn < c5c:
            h = hn; c5c = fn; step = min(step * 1.01, 0.02)
            if c5c < _eval(bh):
                bh = h.copy()
        else:
            step *= 0.75
    return _repair(bh, n), _eval(bh)

def _gen_inits(n, rng):
    inits = []
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 4, 5, 6, 7]:
        inits.append(_repair(0.5 + 0.3 * np.sin(np.pi * freq * x), n))
        inits.append(_repair(0.5 + 0.25 * np.cos(np.pi * freq * x + 0.3), n))
    for freq in [1, 2, 3]:
        t = (freq * x) % 2.0
        inits.append(_repair(0.5 + 0.35 * (1.0 - np.abs(t - 1.0)), n))
    for _ in range(5):
        noise = rng.randn(n)
        kern = np.exp(-0.5 * (np.arange(-15, 16) / 5.0) ** 2)
        kern /= kern.sum()
        sm = np.convolve(noise, kern, 'same')
        h = 0.5 + 0.35 * (sm - sm.mean()) / (np.std(sm) + 1e-12)
        inits.append(_repair(h, n))
    for _ in range(3):
        h = np.clip(0.5 + rng.randn(n) * 0.4, 0, 1)
        inits.append(_repair(h, n))
    return inits

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h); c5c = _eval(h); bh = h.copy()
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        h[i0:i0 + bs] = seg
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            c5c = c5n
            if c5n < _eval(bh):
                bh = h.copy()
    return bh, _eval(bh)

def _polish(h, niter, rng, deadline, scale):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 9)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, step0, tol, deadline):
    n = len(h)
    c5c = _eval(h)
    d = 0.002 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _nesterov(hn, 800, step0, deadline, 0.88, tol)
    if c5n < c5c:
        return hn2, c5n
    d = 0.005 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _nesterov(hn, 400, step0, deadline, 0.90, tol)
    if c5n < c5c:
        return hn2, c5n
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h)

    # Phase 1: Quick SLSQP
    dl = min(deadline, t0 + 60)
    for alpha, tk, mi in [(5000, 20, 200), (20000, 10, 200)]:
        if time.time() > dl:
            break
        h = _slsqp(h, alpha, tk, mi, dl)
        c = _eval(h)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Diverse starting points
    dl = min(deadline, t0 + 240)
    inits = _gen_inits(n0, rng)
    for h0 in inits:
        if time.time() > dl:
            break
        if time.time() > dl - 20:
            break
        h0 = _slsqp(h0, 8000, 10, 120, dl)
        c = _eval(h0)
        if c < best_c5:
            best_c5, best_h = c, h0.copy()

    # Phase 3: Nesterov exact subgradient
    dl = min(deadline, t0 + 450)
    for step0, tol, mom, ns in [(0.004, 0.004, 0.85, 3000),
                                 (0.002, 0.002, 0.88, 3000),
                                 (0.001, 0.001, 0.90, 3000),
                                 (0.0005, 0.0005, 0.92, 2000)]:
        if time.time() > dl:
            break
        h2, cv = _nesterov(best_h.copy(), ns, step0, dl, mom, tol)
        if cv < best_c5:
            best_c5, best_h = cv, h2.copy()

    # Phase 4: Focused subgradient
    dl = min(deadline, t0 + 580)
    for ntop, st, it in [(2, 0.004, 3000), (3, 0.003, 3000), (5, 0.002, 2000)]:
        if time.time() > dl:
            break
        h2, cv = _subgrad_top(best_h.copy(), it, dl, ntop, st)
        if cv < best_c5:
            best_c5, best_h = cv, h2.copy()

    # Phase 5: Jitter + reoptimize
    dl = min(deadline, t0 + 700)
    for _ in range(6):
        if time.time() > dl:
            break
        hj, cj = _jitter_reopt(best_h.copy(), rng, 0.002, 0.002, dl)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 6: Annealing
    dl = min(deadline, t0 + 920)
    blocks = [2, 3, 4, 5, 8, 12]
    best_h, best_c5 = _anneal(best_h.copy(), 80000, rng, dl, blocks, 0.003, 1e-10)

    # Phase 7: Final SLSQP + Nesterov
    dl = min(deadline, t0 + budget_s - 40)
    h = _slsqp(best_h.copy(), 50000, 5, 300, dl)
    c = _eval(h)
    if c < best_c5:
        best_c5, best_h = c, h.copy()
    h2, cv = _nesterov(best_h.copy(), 3000, 0.0005, dl, 0.92, 0.0003)
    if cv < best_c5:
        best_c5, best_h = cv, h2.copy()

    # Phase 8: Fine polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0012), 60000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 3e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
