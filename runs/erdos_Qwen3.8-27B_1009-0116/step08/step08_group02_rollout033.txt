```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_fk(k, h, n):
    dx = 2.0 / n
    shift = (n - 1) - k
    g = np.zeros(n)
    if shift >= 0:
        m = n - shift
        i1 = np.arange(m)
        g[i1] += 1.0 - h[i1 + shift]
        if shift > 0:
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:

        m = n - sp
        i1 = np.arange(sp, n)
        g[i1] += 1.0 - h[i1 - sp]
        i2 = np.arange(m)
        g[i2] -= h[i2 + sp]
    return g * dx

    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _ls_grad(x, alpha, top_k, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if top_k > 0 and top_k < len(c):

    else:

    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_fk(int(idx[ii]), x, n)
    return g

    return x.sum() - _G_N / 2.0

    return np.ones(_G_N)

        if res.fun < _ls_val(h, alpha, top_k, n):

    except Exception:
```

```python
            lo = mid
        else:

    s = (n - 1) - k
    g = np.zeros(n)

            g[i2] -= h[i2 + sp]

    s -= s.max()
    return m + np.log(np.exp(s).sum()) / a

    m = cs.max()
```
