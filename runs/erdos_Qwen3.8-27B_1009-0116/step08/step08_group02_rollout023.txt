            if diff > 0:

                a = min(-diff, h[idx])
                h[idx] -= a; diff += a

        _t = _el()
        for _ in range(30):
            _eval_worst(best_h)
        eps = max(_el() - _t, 1e-9) / 30
        ntries = min(int(time_left / eps * 0.6), 2000000)
        best_h, best_c5 = _local_informed(best_h, ntries, rng, 4e-6)

    # Phase 5: Second round of Adam worst-lag with smaller lr
    if _el() < budget_s - 40:
        end5 = t0 + budget_s - 40
        h_new = _adam_worst(best_h.copy(), n, 20000, 0.0003, end5)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new.copy()

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        time_left = budget_s - 2 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max(_el() - _t, 1e-9) / 30
        npol = min(int(time_left / eps * 0.7), 8000000)
        best_h, best_c5 = _local_2pt(best_h, npol, rng, 5e-8)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
