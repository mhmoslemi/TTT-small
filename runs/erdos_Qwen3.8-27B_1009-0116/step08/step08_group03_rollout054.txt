    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(h, target):

    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(50):

        if abs(d) < 1e-12:
            break
        m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()
    if abs(d) > 1e-10:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = min(-d, h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-14:
                break
    return h
