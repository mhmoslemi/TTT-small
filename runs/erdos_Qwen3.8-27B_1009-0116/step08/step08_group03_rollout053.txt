    return x

def _c5(h):
    n=len(h)
    return float(np.max(np.correlate(h,1.0-h,mode='full'))*(2.0/n))

def _pn(h):
    n=len(h); N=_fft_size(n); H=np.fft.rfft(h,N)
    ac=np.fft.irfft(H*np.conj(H),N).real[:n]
    pref=np.empty(n+1); pref[0]=0.0; np.cumsum(h,out=pref[1:])
    P=np.empty(n); P[0]=pref[n]-ac[0]; Nv=np.zeros(n)
    if n>1:
        idx=np.arange(1,n)
        P[idx]=pref[n-idx]-ac[idx]
        Nv[idx]=pref[n]-pref[idx]-ac[idx]
    mx=max(P[0],np.max(P[1:]),np.max(Nv[1:])) if n>1 else P[0]
    return P,Nv,float(mx)

def _sv(h,a):
    P,Nv,mx=_pn(h); n=len(h); dx=2.0/n
    e=np.exp(np.clip(a*(P[0]-mx),-400,0))
    if n>1:
        e+=np.exp(np.clip(a*(P[1:]-mx),-400,0)).sum()
        e+=np.exp(np.clip(a*(Nv[1:]-mx),-400,0)).sum()
    return dx*(mx+np.log(e)/a)

def _gw(h,wP,wN):
    n=len(h); dx=2.0/n; u=np.zeros(n)
    if n>1: u[1:]=wP[1:]+wN[1:]
    cp=np.zeros(n); cn=np.zeros(n)
    if n>1:
        cp[1:]=np.cumsum(wP[1:]); cn[1:]=np.cumsum(wN[1:])
    linP=cp[n-1::-1]; linN=cn; q=np.zeros(n)
    if n>1 and np.any(u[1:]):
        Nf=_fft_size(n); H=np.fft.rfft(h,Nf)
        R=np.fft.irfft(H*np.fft.rfft(u[::-1],Nf),Nf)[n-1:2*n-1].real
        L=np.fft.irfft(H*np.fft.rfft(u,Nf),Nf)[:n].real
        q=-(R+L)
    return dx*(wP[0]*(1.0-2.0*h)+linP+linN+q)

def _sg(h,a):
    P,Nv,mx=_pn(h); n=len(h); wP=np.zeros(n); wN=np.zeros(n)
    e0=np.exp(np.clip(a*(P[0]-mx),-400,0)); tot=e0
    if n>1:
        ep=np.exp(np.clip(a*(P[1:]-mx),-400,0))
        en=np.exp(np.clip(a*(Nv[1:]-mx),-400,0))
        tot+=ep.sum()+en.sum()
        wP[0]=e0/tot; wP[1:]=ep/tot; wN[1:]=en/tot
    else: wP[0]=1.0
    return _gw(h,wP,wN)

def _fista(h0,a,mt,L0=10.0):
    if mt<=0: return h0.copy(),_c5(h0)
    n=len(h0); S=n*0.5; t0=time.time()
    h=_repair(h0,n); hp=h.copy(); y=h.copy(); tm=1.0
    fy=_sv(h,a); fh=fy; bc=_c5(h); bh=h.copy(); Lc=L0; stale=0
    for it in range(300000):
        if time.time()-t0>mt: break
        if it:
            tn=0.5*(1.0+np.sqrt(1.0+4.0*tm*tm))
            y=h+((tm-1.0)/tn)*(h-hp); tm=tn
            if y.min()<-1e-4 or y.max()>1+1e-4 or abs(y.sum()-S)>1e-6:
                y=_proj(y,S)
            fy=_sv(y,a)
        else: y=h; fy=fh
        g=_sg(y,a); step=1.0/max(Lc,1e-12); ok=False
        for _ in range(14):
            xn=_proj(y-step*g,S); fx=_sv(xn,a); d=xn-y
            if fx<=fy+g@d+0.5*step*d@d+1e-10:
                ok=True; break
            step*=0.5
        Lc=1.0/max(step,1e-12)
        if ok:
            hp=h; h=xn; fh=fx; stale=0; Lc=max(Lc*0.85,1.0)
            if it%6==0:
                c=_c5(h)
                if c<bc: bc=c; bh=h.copy()
        else:
            stale+=1; hp=h; y=h; tm=1.0; fy=fh; Lc=max(Lc*0.5,1.0)
            if stale>10: break
    c=_c5(h)
    if c<bc: bc=c; bh=h.copy()
    return bh,bc

def _resample(h,m):
    n=len(h)
    if n==m: return _repair(h,n)
    if n%m==0: out=h.reshape(m,n//m).mean(axis=1)
    else:
        x0=(np.arange(n)+0.5)*(2.0/n); x1=(np.arange(m)+0.5)*(2.0/m)
        out=np.interp(x1,x0,h)
    return _proj(out,m*0.5)

def _act_grad(h,P,Nv):
    n=len(h)
    mx=max(P[0],np.max(P[1:]),np.max(Nv[1:])) if n>1 else P[0]
    tol=max(1e-9,6.0*(2.0/n))
    wP=np.zeros(n); wN=np.zeros(n); cnt=0
    if P[0]>=mx-tol: wP[0]=1.0; cnt+=1
    if n>1:
        mP=P[1:]>=mx-tol
        if np.any(mP): wP[np.nonzero(mP)[0]+1]=1.0; cnt+=int(mP.sum())
        mN=Nv[1:]>=mx-tol
        if np.any(mN): wN[np.nonzero(mN)[0]+1]=1.0; cnt+=int(mN.sum())
    if cnt==0: return np.zeros(n)
    inv=1.0/cnt
    wP*=inv; wN*=inv
    return _gw(h,wP,wN)

def _local(h0,rng,deadline):
    n=len(h0); h=h0.copy(); c=_c5(h); bc=c; bh=h.copy()
    tl0=time.time()
    P,Nv,_=_pn(h); g=_act_grad(h,P,Nv)
    q=max(8,n//6); it=0; stale=0
    while time.time()<deadline:
        if it%4096==0:
            c=_c5(h)
            if c<bc: bc=c; bh=h.copy()
            P,Nv,_=_pn(h); g=_act_grad(h,P,Nv)
        frac=(time.time()-tl0)/max(1e-9,deadline-tl0)
        step=max(1e-10,5e-5*(0.01)**(frac*0.6))
        if rng.random()<0.75:
            low_idx=np.argsort(g)[:q]; high_idx=np.argsort(g)[-q:]
            i=int(low_idx[rng.randint(len(low_idx))])
            j=int(high_idx[rng.randint(len(high_idx))])
        else:
            i=rng.randint(n); j=rng.randint(n)
        if i==j: continue
        s=1 if g[i]<=g[j] else (-1 if g[j]<g[i] else (1 if rng.random()<0.5 else -1))
        avail=min(1.0-h[i],h[j]) if s>0 else min(h[i],1.0-h[j])
        if avail<1e-13: continue
        d=s*min(avail,step*(0.4+rng.random()*1.6))
        h[i]+=d; h[j]-=d
        cn=_c5(h)
        if cn<c-1e-14:
            c=cn; stale=0
            # Try to extend with binary search
            lo=d; hi=min(s*avail,s*step*30.0)
            if s*hi>s*avail or s*hi>s*d: hi=s*min(avail,step*30.0)
            best_d=lo
            for _ in range(4):
                mid=s*0.5*(abs(lo)+abs(hi))
                if mid*s<=0: mid=lo
                # simplify: just try a few larger steps
                trial=s*(step*2.0*rng.uniform(1.0,4.0))
                if abs(trial)>avail: continue
                h2_i=h[i]+trial-d; h2_j=h[j]-trial+d
                if h2_i<0 or h2_i>1 or h2_j<0 or h2_j>1: continue
                h[i]=h2_i; h[j]=h2_j
                cm=_c5(h)
                if cm<c:
                    c=cn=cm; best_d=trial; lo=trial
                else:
                    h[i]=h[i]+d-trial  # revert, back to step d
                    h[j]=h[j]-d+trial
                    break
            if c<bc: bc=c; bh=h.copy()
        else:
            h[i]-=d; h[j]+=d; stale+=1
            if stale>50000:
                # Perturb to escape
                for _ in range(int(n*0.1)):
                    k=rng.randint(n); kk=rng.randint(n)
                    if k==kk: continue
                    dd=rng.uniform(-1,1)*step*0.5
                    ii,jj=k,kk
                    hv=min(1.0-h[ii],h[jj]) if dd>0 else min(h[ii],1.0-h[jj])
                    if hv>1e-13:
                        h[ii]+=dd; h[jj]-=dd
                h=_repair(h,n); c=_c5(h); stale=0
        it+=1
    c=_c5(h)
    if c<bc: bc=c; bh=h.copy()
    return bh,bc

def run(seed=42,budget_s=1100,**kwargs):
    t0=time.time(); rng=np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0=np.asarray(initial_h_values,dtype=float).copy()
    else:
        h0=np.full(560,0.5)
    n0=len(h0); h0=_repair(h0,n0)
    best=h0.copy(); best_c=_c5(best)
    for variant in (1.0-best, best[::-1]):
        v=_proj(variant,n0*0.5); c=_c5(v)
        if c<best_c: best=v; best_c=c
    # Phase 1: Coarse FISTA
    e1=t0+min(50.0,0.05*budget_s)
    if e1>time.time()+2:
        b,c=_fista(best,[5,20,80,300],e1-time.time(),10.0)
        # need to call with proper alpha list - let's do sequential
        hcur=best.copy()
        alphas1=[5,20,80,300]
        per1=min(12.0,(e1-time.time())/len(alphas1))
        for a in alphas1:
            if e1-time.time()<1: break
            hcur,c=_fista(hcur,a,per1,10.0)
            if c<best_c: best_c=c; best=hcur.copy()
    # Phase 2: Downsample and optimize
    e2=t0+min(160.0,0.15*budget_s)
    if e2>time.time()+5:
        hm=_resample(best,n0//2)
        hcur=hm
        per2=min(25.0,(e2-time.time())/4)
        for a in [10,60,300,1500]:
            if e2-time.time()<2: break
            hcur,c=_fista(hcur,a,per2,10.0)
        hu=_resample(hcur,n0); cu=_c5(hu)
        if cu<best_c: best=hu; best_c=cu
    # Phase 3: Main FISTA on full resolution
    e3=t0+budget_s-280.0
    if e3>time.time()+10:
        hcur=best.copy()
        alphas3=[15,60,200,700,2500,8000,25000,80000]
        per3=(e3-time.time())/len(alphas3)
        for a in alphas3:
            if e3-time.time()<2: break
            hcur,c=_fista(hcur,a,min(per3,e3-time.time()),10.0)
            if c<best_c: best_c=c; best=hcur.copy()
    # Phase 4: Local search
    e4=t0+budget_s-60.0
    if e4>time.time()+5:
        best,best_c=_local(best,rng,e4)
    # Phase 5: Final polish FISTA + local
    e5=t0+budget_s-2.0
    if e5>time.time()+10:
        best,c=_fista(best,500,e5-time.time()-30.0,10.0)
        if c<best_c: best_c=c
        if e5-time.time()>5:
            best,best_c=_local(best,rng,e5)
    # Final check variants
    for v in (1.0-best, best[::-1]):
        v2=_proj(v,len(v)*0.5); c=_c5(v2)
        if c<best_c: best=v2; best_c=c
    best=_repair(best,len(best)); best_c=_c5(best)
    return best,best_c,len(best)
