import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_for_lag(sh, h, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _nesterov_pg(h0, alpha0, alpha1, steps, step0, top_k, max_step):
    n = len(h0)
    h = h0.copy()
    y = h.copy()
    t_val = 1.0
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        frac = it / max(steps - 1, 1)
        alpha = alpha0 + (alpha1 - alpha0) * frac
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        order = np.argsort(c)[::-1]
        top = order[:top_k]
        c_top = c[top]
        m = c_top.max()
        s = np.clip(alpha * (c_top - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(top_k):
            if w[ii] < 1e-15:
                continue
            sh = (n - 1) - (order[ii] if ii < top_k else 0)
            sh = (n - 1) - top[ii]
            g += w[ii] * _grad_for_lag(sh, h, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new

def _slsqp_focus(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
        else:
            idx = np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
        else:
            idx = np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)

            if w[ii] < 1e-15:
                continue
            sh = (n - 1) - idx[ii]
            g += w[ii] * _grad_for_lag(sh, x, n)
        return g

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

        return h

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out
