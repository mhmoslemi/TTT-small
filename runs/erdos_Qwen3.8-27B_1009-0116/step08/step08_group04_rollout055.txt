def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad**2
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps), m, v

    """Project h onto {0<=h<=1, sum(h)=target} using bisection on Lagrange multiplier."""

        if h_p.sum() > target:

        else:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i])
                    h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i])
                    h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corr(h).max())

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _cntl1(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        p = -s
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]
    return g * dx

def _adam(h, alpha, topk, nsteps, lr, dl):
    n = len(h)
    ma = np.zeros(n); va = np.zeros(n); t = 0
    c = _corr(h); fb = c.max()
    b1, b2, ep = 0.85, 0.999, 1e-9
    for _ in range(nsteps):
        if time.time() > dl:
            break
        t += 1
        ic = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[ic]
        wm = np.exp(alpha * (ct - ct.max())); wm /= wm.sum()
        dxi = 2.0 / n
        g = np.zeros(n)
        for ji in range(len(ic)):
            if wm[ji] > 1e-13:
                g += wm[ji] * _cntl1(h, ic[ji], dxi)
        ma = b1 * ma + (1 - b1) * g
        va = b2 * va + (1 - b2) * g * g
        stp = lr * ma / (1 - b1**t) / (np.sqrt(va / (1 - b2**t)) + ep)
        hn = _proj(h - stp, n * 0.5)
        cn = _corr(hn)
        if cn.max() < fb - 1e-15:
            h, c, fb = hn, cn, cn.max()
            lr = min(lr * 1.015, 0.05)
        else:
            lr *= 0.6
            if lr < 1e-11:
                break
    return h

def _anneal(h, rng, dl, nt, bsl):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    T0, T1 = 0.004, 1e-13
    for it in range(nt):
        if time.time() > dl:
            break
        T = T0 * (T1 / T0) ** (it / max(nt, 1))
        bs = bsl[rng.randint(len(bsl))]
        i0 = rng.randint(0, n - bs + 1)
        seg = h[i0:i0 + bs] + 0.008 * max(T, 1e-14) ** 0.5 * rng.randn(bs)
        hn = h.copy()
        hn[i0:i0 + bs] = np.clip(seg, 0, 1)
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5 or rng.random() < np.exp(-max(c5n - c5, 0) / max(T, 1e-14)):
            h, c5 = hn, c5n
            if c5 < bc:
                bc, bh = c5, h.copy()
    return bh, bc

def _shift_fix(h, rng, dl):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for rnd in range(120):
        if time.time() > dl:
            break
        c = _corr(h)
        hot = np.argsort(c)[-4:]
        for km in hot:
            s = km - (n - 1)
            if s >= 0:
                pairs = [(i + s, i) for i in range(n - s)]
            else:
                p = -s
                pairs = [(i, i + p) for i in range(n - p)]
            rng.shuffle(pairs)
            for a, b in pairs[:25]:
                for d in [1.5e-4, -1.5e-4, 8e-5, -8e-5]:
                    na, nb = h[a] + d, h[b] - d
                    if 0 <= na <= 1 and 0 <= nb <= 1:
                        h[a], h[b] = na, nb
                        c5n = _eval(h)
                        if c5n < c5 - 1e-16:
                            c5 = c5n
                            if c5 < bc:
                                bc, bh = c5, h.copy()
                        else:
                            h[a], h[b] = na - d, nb + d
    return bh, bc

def _fine(h, rng, dl):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    step = 4e-5
    while time.time() < dl:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5 < bc:
                    bc, bh = c5, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                if step > 3e-7:
                    step *= 0.98
    return bh, bc

def _perturb_restart(h, rng, n_pert):
    n = len(h)
    hp = h.copy()
    mask = rng.rand(n) < n_pert / n
    hp[mask] = np.clip(hp[mask] + 0.3 * rng.randn(mask.sum()), 0, 1)
    return _repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl_end = t0 + budget_s - 8
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: ADAM with increasing sharpness (0-180s)
    d1 = min(dl_end, t0 + 180)
    for al, tk, ns, lr in [(3000, 16, 4000, 4e-4), (12000, 8, 3000, 3e-4), (50000, 4, 2500, 2e-4)]:
        if time.time() > d1:
            break
        h = _adam(h, al, tk, ns, lr, d1)
        cv = _eval(h)
        if cv < best_c5:
            best_c5, best_h = cv, h.copy()

    # Phase 2: Block annealing (180-460s)
    d2 = min(dl_end, t0 + 460)
    h2, c2 = _anneal(best_h.copy(), rng, d2, 120000, [1, 2, 3, 5, 8, 13, 21, 34])
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Perturb + re-optimize (460-580s)
    d3 = min(dl_end, t0 + 580)
    for frac in [0.05, 0.12, 0.25]:
        if time.time() > d3:
            break
        hp = _perturb_restart(best_h, rng, int(n * frac))
        hp = _adam(hp, 15000, 6, 2000, 2e-4, d3)
        cv = _eval(hp)
        if cv < best_c5:
            best_c5, best_h = cv, hp

    # Phase 4: Shift-aware coordinate fixing (580-800s)
    d4 = min(dl_end, t0 + 800)
    h4, c4 = _shift_fix(best_h.copy(), rng, d4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: High-alpha ADAM polish (800-910s)
    d5 = min(dl_end, t0 + 910)
    for al, tk, ns, lr in [(100000, 3, 3000, 8e-5), (300000, 2, 2000, 5e-5)]:
        if time.time() > d5:
            break
        h5 = _adam(best_h.copy(), al, tk, ns, lr, d5)
        cv = _eval(h5)
        if cv < best_c5:
            best_c5, best_h = cv, h5

    # Phase 6: Fine polish (910-end)
    best_h, best_c5 = _fine(best_h, rng, dl_end)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
