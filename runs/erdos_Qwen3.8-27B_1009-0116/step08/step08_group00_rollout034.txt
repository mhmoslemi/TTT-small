def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _golden_min(b, slope, n_iter=45):
    """Minimize max_k(b[k] + slope[k]*x) over x in [0,1] using golden section."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    r = 1.0 / gr
    lo, hi = 0.0, 1.0
    c = lo + (1.0 - r) * (hi - lo)
    d = lo + r * (hi - lo)
    fc = np.max(b + slope * c)
    fd = np.max(b + slope * d)
    for _ in range(n_iter):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = lo + (1.0 - r) * (hi - lo)
            fc = np.max(b + slope * c)
        else:
            lo, c, fc = c, d, fd
            d = lo + r * (hi - lo)
            fd = np.max(b + slope * d)
    return (lo + hi) * 0.5

def _coord_pass(h, out, max_iter=2000):
    """One pass of coordinate descent. Updates h in place, returns improvement."""
    n = len(h)
    k_idx = np.arange(2 * n - 1)
    total_improve = 0.0
    for it in range(max_iter):
        j = it % n
        j1 = j + k_idx - (n - 1)
        j2 = j - k_idx + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        
        slope = np.zeros(2 * n - 1)
        contrib = np.zeros(2 * n - 1)
        slope[v1] += (1.0 - h[j1[v1]])
        contrib[v1] += h[j] * (1.0 - h[j1[v1]])
        slope[v2] += (-h[j2[v2]])
        contrib[v2] += h[j2[v2]] * (1.0 - h[j])
        
        b = out - contrib
        new_val = _golden_min(b, slope)
        delta = new_val - h[j]
        if abs(delta) > 1e-12:
            h[j] = new_val
            out += delta * slope
            total_improve += abs(delta)
    return total_improve

def _binary_push(h, n):
    """Push values toward 0 or 1 while maintaining sum."""
    h = h.copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-6:
        return h
    # Try to push toward binary
    for _ in range(100):
        changes = 0
        for j in range(n):
            if h[j] < 0.01 and s > target:
                delta = min(h[j], s - target)
                if delta > 1e-9:
                    h[j] -= delta
                    s -= delta
                    changes += 1
            elif h[j] > 0.99 and s < target:
                delta = min(1.0 - h[j], target - s)
                if delta > 1e-9:
                    h[j] += delta
                    s += delta
                    changes += 1
        if changes == 0:
            break
    return _repair(h, n)

def _local_search_2pt(h, n_iter, rng, step0=1e-5):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    c5 = best_c5
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = step0 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5_new = _eval(h)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _optimize_from(h, t_start, t_limit):
    """Full optimization pipeline from a starting point."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    
    # Coordinate descent
    for _ in range(50):

        out = _corr(h)
        h_work = h.copy()
        _coord_pass(h_work, out)
        h_work = _repair(h_work, n)

        if c5 < best_c5 - 1e-12:

            best_h = h_work.copy()
            h = h_work
        else:

                best_c5 = c5
                best_h = h_work.copy()
                h = h_work
            # Try binary push

                if c5b < best_c5:
                    best_c5 = c5b
                    best_h = h_bin.copy()
                    h = h_bin
                continue
            break
    
    return best_h, best_c5
