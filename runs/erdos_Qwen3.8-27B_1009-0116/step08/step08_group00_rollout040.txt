def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Sort-based adjustment
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a

    return h

    target = n * 0.5
    # min g·z s.t. 0<=z<=1, sum(z)=target

def _pairwise_swap(h, i, j, n_top=5):
    """Find optimal d for h[i]+=d, h[j]-=d."""
    n = len(h)
    dx = 2.0 / n
    c = _get_corr(h)
    idx_top = np.argsort(c)[-n_top:]

        L = k - (n - 1)
        si = _subgrad_lag(h, L)
        slopes[ii] = si[i] - si[j]
    
    intercepts = c[idx_top].copy()
    
    # f(d) = max_ii (intercepts[ii] + d * slopes[ii])
    # Minimize over d ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    
    if d_min >= d_max:
        return h, 0.0
    
    # Convex PWL minimization: check boundaries and intersection points
    best_d = 0.0
    best_val = np.max(intercepts)
    
    # Evaluate at boundaries
    for d_cand in [d_min, d_max, 0.0]:
        val = np.max(intercepts + d_cand * slopes)
        if val < best_val:
            best_val = val
            best_d = d_cand
    
    # Check intersection of each pair of lines
    # Line ii: intercepts[ii] + d * slopes[ii]
    # Intersection: (intercepts[ii] - intercepts[jj]) / (slopes[jj] - slopes[ii])
    for a in range(n_top):
        for b in range(a+1, n_top):
            ds = slopes[b] - slopes[a]
            if abs(ds) < 1e-15:
                continue
            d_int = (intercepts[a] - intercepts[b]) / ds
            if d_min <= d_int <= d_max:
                val = np.max(intercepts + d_int * slopes)
                if val < best_val:
                    best_val = val
                    best_d = d_int
    
    if abs(best_d) > 1e-15:
        h[i] += best_d
        h[j] -= best_d
    return h, best_d
