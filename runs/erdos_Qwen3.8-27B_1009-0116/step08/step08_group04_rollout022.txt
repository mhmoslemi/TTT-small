    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sh = s - s.max()
    return m + (sh.max() + np.log(np.sum(np.exp(sh)))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    beta = 0.83
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        accepted = False
        for _ in range(6):
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
                step = min(step * 1.04, step0 * 30.0)
                accepted = True
                break
            step *= 0.4
        if not accepted:
            break
    return _repair(h, n)

def _resize(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.arange(old_n) / float(old_n) * 2.0
    x_new = np.arange(new_n) / float(new_n) * 2.0
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new, new_n)

def _active_opt(h, deadline, n_iters):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _eval(h)
    step = 0.003
    f_prev = _eval(h)
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        progress = min(it / max(n_iters, 1), 1.0)
        topk = max(2, int(8.0 * (1.0 - 0.8 * progress)))
        alpha = 5000.0 + 500000.0 * progress * progress
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _grad_lag(h, idx[ii], dx)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-13:
            break
        f = _smax_obj(h, alpha, topk)
        hp = h.copy()
        step_now = step * (1.0 - 0.4 * progress)
        accepted = False
        for _ in range(10):
            hn = _proj(hp - step_now * g, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
                c_new = _eval(h)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                elif c_new < f_prev:
                    stall = 0
                else:
                    stall += 1
                step = min(step * 1.02, 0.008)
                accepted = True
                break
            step_now *= 0.4
        if not accepted:
            if stall > 50:
                h = _repair(h + 0.005 * np.random.randn(n), n)
                stall = 0
            else:
                break
        if it % 200 == 0:
            f_prev = _eval(h)
    return best_h, best_c

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T + 1e-10, 1e-12)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T + 1e-14, 1e-16)):
            h = hn
            c5c = c5n
            if c5c < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP (0-100s)
    dl1 = min(deadline, t0 + 100)
    for alpha, tk, mi in [(3000, 40, 400), (20000, 15, 250)]:
        if time.time() > dl1:
            break
        try:
            def obj(x, a=alpha, k=tk):
                return _smax_obj(x, a, k)
            def jac(x, a=alpha, k=tk):
                return _grad_smax(x, a, k)
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n0 * 0.5,
                     'jac': lambda x: np.ones(n0)}]
            r = minimize(obj, h, jac=jac, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n0, constraints=cons,
                         options={'maxiter': mi, 'ftol': 1e-17})
            h = _repair(r.x, n0)
            cv = _eval(h)
            if cv < best_c5:
                best_c5 = cv
                best_h = h.copy()
        except Exception:
            pass

    # Phase 2: Multi-resolution (100-260s)
    dl2 = min(deadline, t0 + 260)
    for tn in [320, 480, 640, 800]:
        if time.time() > dl2:
            break
        hr = _resize(best_h, tn)
        hr = _nesterov(hr, 3000, 25, 800, 0.004, dl2)
        hr = _nesterov(hr, 30000, 8, 800, 0.0015, dl2)
        hb = _resize(hr, n0)
        hb = _nesterov(hb, 8000, 12, 400, 0.002, dl2)
        cv = _eval(hb)
        if cv < best_c5:
            best_c5 = cv
            best_h = hb.copy()

    # Phase 3: Long Nesterov with high alpha (260-520s)
    dl3 = min(deadline, t0 + 520)
    hw = best_h.copy()
    for a0, tk, ns, st in [(5000, 20, 2500, 0.004),
                           (30000, 8, 2500, 0.002),
                           (150000, 5, 2000, 0.001),
                           (800000, 3, 2000, 0.0005)]:
        if time.time() > dl3:
            break
        hw = _nesterov(hw, a0, tk, ns, st, dl3)
        cv = _eval(hw)
        if cv < best_c5:
            best_c5 = cv
            best_h = hw.copy()

    # Phase 4: Active-set optimization (520-760s)
    dl4 = min(deadline, t0 + 760)
    ha, ca = _active_opt(best_h.copy(), dl4, 12000)
    if ca < best_c5:
        best_c5 = ca
        best_h = ha.copy()

    # Phase 5: Block annealing (760-920s)
    dl5 = min(deadline, t0 + 920)
    blocks = [1, 2, 3, 5, 8, 13, 21, 34, 55]
    bh, bc = _block_anneal(best_h.copy(), 80000, rng, dl5, 0.004, 1e-11, blocks)
    if bc < best_c5:
        best_c5 = bc
        best_h = bh.copy()

    # Phase 6: Final very-high-alpha Nesterov (920-1040s)
    dl6 = min(deadline, t0 + 1040)
    hf = _nesterov(best_h.copy(), 3000000, 2, 4000, 0.0003, dl6)
    hf = _nesterov(hf, 10000000, 1, 2000, 0.0002, dl6)
    cv = _eval(hf)
    if cv < best_c5:
        best_c5 = cv
        best_h = hf.copy()

    # Phase 7: Fine polish (1040-end)
    dl7 = deadline
    c5c = _eval(best_h)
    bh = best_h.copy()
    bc = c5c
    nc = len(best_h)
    sbase = 1e-7
    for it in range(50000):
        if time.time() > dl7:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        d = sbase * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            best_h[i], best_h[j] = ni, nj
            c5n = _eval(best_h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5c < bc:
                    bc = c5c
                    bh = best_h.copy()
            else:
                best_h[i], best_h[j] = ni - d, nj + d
                sbase = max(sbase * 0.97, 1e-9)
    if bc < best_c5:
        best_c5 = bc
        best_h = bh.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
