import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        for _ in range(300):
            h *= target / max(s, 1e-15)
            h = np.clip(h, 0.0, 1.0)
            s = h.sum()
            if abs(s - target) < 1e-11:
                break
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_correl(h)))

def _topk_idx(c, k):
    if k >= c.size:
        return np.arange(c.size)
    return np.argpartition(c, -k)[-k:].copy()

def _smax_obj(h, alpha, k):
    c = _correl(h)
    idx = _topk_idx(c, k)
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)

def _smax_grad(h, alpha, k):
    n = len(h)
    c = _correl(h)
    idx = _topk_idx(c, k)
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        shift = idx[ii] - (n - 1)
        if shift >= 0:
            g[:n - shift] += (1.0 - h[shift:])
            if shift > 0:
                g[shift:] -= h[:-shift]
        else:
            sp = -shift
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
        g *= w[ii]
    return g * dx

def _lbfgs_run(h, alpha, k, maxit, deadline):
    n = len(h)
    def f(x):
        return _smax_obj(x, alpha, k)
    def g(x):
        return _smax_grad(x, alpha, k)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(f, h, jac=g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12,
                              'maxfun': maxit * 3})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha0, grow, k0, kmin, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, k0)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    alpha = alpha0
    k = k0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it % 500 == 0 and it > 0:
            alpha = min(alpha * grow, 5000000.0)
            k = max(kmin, int(k * 0.92))
        g = _smax_grad(h, alpha, k)
        v = beta * v + (1.0 - beta) * g
        for trial in range(10):
            step_try = step
            hn = _repair(np.clip(h - step_try * v, 0, 1), n)
            fn = _smax_obj(hn, alpha, k)
            if fn < f:
                h = hn
                f = fn
                step = min(step * 1.04, step0 * 20.0)
                break
            step_try *= 0.5
            step = step_try
        else:
            break
    return _repair(h, n)
