import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-13: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        if s > 0: g[s:] -= h[:n-s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g * dx

def _smax_obj(h, alpha, K):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    idx = np.argpartition(-c, K - 1)[:K] if K < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, K, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    idx = np.argpartition(-c, K - 1)[:K] if K < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-15: continue
        g += w[i] * _grad_lag(h, idx[i], dx)
    return g

def _coord_descend(h, max_sweeps, deadline, K=10):
    n = len(h)
    tgt = n * 0.5
    bh = h.copy(); bc = _eval(h)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for sw in range(max_sweeps):
        if time.time() > deadline: break
        cfr = np.correlate(h, 1.0 - h, mode='full')
        imp = False
        for j in range(n):
            old = h[j]
            i1 = j + shifts; v1 = (i1 >= 0) & (i1 < n)
            i2 = j - shifts; v2 = (i2 >= 0) & (i2 < n)
            cf = np.zeros(2 * n - 1)
            if v1.any(): cf[v1] += 1.0 - h[i1[v1]]
            if v2.any(): cf[v2] += -h[i2[v2]]
            ac = np.abs(cf)
            if ac.max() < 1e-12: continue
            ti = np.argpartition(-ac, K - 1)[:K]
            b = cfr[ti] - old * cf[ti]
            cv = cf[ti]
            bi_, bj_ = b[:, None], b[None, :]
            di = cv[:, None] - cv[None, :]
            nu = bj_ - bi_
            mk = np.abs(di) > 1e-10
            with np.errstate(divide='ignore', invalid='ignore'):
                ch = np.where(mk, nu / di, 99.0)
            ch = ch[(ch >= 0) & (ch <= 1)]
            acs = np.unique(np.round(np.concatenate(([0.0, 1.0, old], ch.ravel())), 10))
            dl = (acs[:, None] - old) * cf[None, :]
            mx = (cfr[None, :] + dl).max(axis=1)
            bv = acs[mx.argmin()]
            if abs(bv - old) > 1e-10:
                h[j] = bv; cfr += (bv - old) * cf; imp = True
        h = _repair(h, n)
        cv5 = _eval(h)
        if cv5 < bc:
            bc = cv5; bh = h.copy()
        if not imp: break
    return bh, bc

def _nesterov(h, alpha0, grow, K, steps, st0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, K)
    v = np.zeros(n); beta = 0.83; st = st0; a = alpha0
    for it in range(steps):
        if time.time() > deadline: break
        if it > 0 and it % 200 == 0:
            a = min(a * grow, 5e7)
        g = _smax_grad(h, a, K, 2.0 / n)
        v = beta * v + (1.0 - beta) * g
        hp = h.copy()
        hn = _proj(hp - st * v, n * 0.5)
        fn = _smax_obj(hn, a, K)
        if fn < f:
            h, f, st = hn, fn, min(st * 1.03, st0 * 8.0)
        else:
            st *= 0.5
            for _ in range(3):
                hn = _proj(hp - st * v, n * 0.5)
                fn = _smax_obj(hn, a, K)
                if fn < f: h, f = hn, fn; break
                st *= 0.5
            else: break
    return _repair(h, n)

def _slsqp(h, alpha, K, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    try:
        r = minimize(_smax_obj, h,
                     jac=lambda x: _smax_grad(x, alpha, K, 2.0 / n),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq',
                                   'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, nit, rng, deadline):
    n = len(h)
    fc = _eval(h); bh = h.copy(); bc = fc
    blks = [1, 2, 3, 5, 8, 13, 21]
    for it in range(nit):
        if time.time() > deadline: break
        T = 0.003 * (1e-10 / 0.003) ** (it / max(nit, 1))
        bs = blks[rng.randint(len(blks))]
        i0 = rng.randint(0, max(1, n - bs))
        sg = h[i0:i0+bs].copy()
        sg += 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        sg -= sg.mean()
        sg = np.clip(sg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = sg
        hn = _repair(hn, n)
        fn = _eval(hn)
        if fn < fc or rng.random() < np.exp(-(fn - fc) / max(T, 1e-12)):
            h, fc = hn, fn
            if fn < bc: bc = fn; bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):

    # P3: Nesterov

                                (3000, 1.4, 5, 0.002, 4000)]:

        h3 = _nesterov(bh.copy(), a0, gr, k, ns, st, d3)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a; d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a; d += a
                if abs(d) < 1e-13: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        if s > 0: g[s:] -= h[:n-s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g * dx

def _smax_obj(h, alpha, K):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    idx = np.argpartition(-c, K - 1)[:K] if K < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, K, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    idx = np.argpartition(-c, K - 1)[:K] if K < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-14:
            g += w[i] * _grad_lag(h, idx[i], dx)
    return g

def _coord_descend(h, max_sweeps, deadline, K=12):

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-13: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, k, dx):
    n = len(h); g = np.zeros(n); s = k - (n - 1)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        if s > 0: g[s:] -= h[:n-s]
    else:
        p = -s; g[p:] += (1.0 - h[:n-p]); g[:n-p] -= h[p:]
    return g * dx

def _smax_obj(h, alpha, K):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    idx = np.argpartition(-c, K-1)[:K] if K < len(c) else np.arange(len(c))
    ct = c[idx]; s = np.clip(alpha*(ct-m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, K, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    idx = np.argpartition(-c, K-1)[:K] if K < len(c) else np.arange(len(c))
    s = np.clip(alpha*(c[idx]-m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-14: g += w[i] * _grad_lag(h, idx[i], dx)
    return g

def _obj_wr(x, alpha, K, dx):
    return _smax_obj(x, alpha, K)

def _grad_wr(x, alpha, K, dx):
    return _smax_grad(x, alpha, K, dx)

def _csum_fun(x, n):
    return x.sum() - n * 0.5

def _csum_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, K, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    try:
        r = minimize(_obj_wr, h, args=(alpha, K, 2.0/n), jac=_grad_wr,
                     method='SLSQP', bounds=[(0.0, 1.0)]*n,
                     constraints=[{'type':'eq','fun':_csum_fun,'jac':_csum_jac,'args':(n,)}],
                     options={'maxiter':maxit,'ftol':1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descend(h, max_sweeps, deadline, K=12):

    shifts = np.arange(2*n-1) - (n-1)
