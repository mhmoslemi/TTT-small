def _fft_size(n):
    m = 2 * n - 1
    N = 1

import numpy as np

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]

    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    P[0] = total - ac[0]
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _grad_from_weights(h, wP, wN):
    n = len(h); dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0]*(1.0 - 2.0*h) + linP + linN + q)

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h); dx = 2.0 / n
    e = np.exp(np.clip(alpha*(P[0]-mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha*(P[1:]-mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha*(Nv[1:]-mx), -400.0, 0.0)).sum()
    return dx*(mx + np.log(max(e,1e-300))/alpha)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha*(P[0]-mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha*(P[1:]-mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha*(Nv[1:]-mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
    if tot <= 0.0:
        tot = 1.0; e0 = 1.0
    wP[0] = e0/tot
    if n > 1:
        wP[1:] = ep/tot; wN[1:] = en/tot
    return _grad_from_weights(h, wP, wN)

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n//m).mean(axis=1)
    else:
        x0 = (np.arange(n)+0.5)*(2.0/n)
        x1 = (np.arange(m)+0.5)*(2.0/m)
        out = np.interp(x1, x0, h)
    return _proj(out, m*0.5)

def _subgrad_opt(h0, max_time, tol=8e-7, step_scale=0.03):

def _active_grad(h, P, Nv):

        mN = Nv[1:]>=mx-tol
        if np.any(mN): pos=np.nonzero(mN)[0]+1; wN[pos]=1.0; cnt+=pos.size
    if cnt==0: return np.zeros(n)
    wP/=cnt; wN/=cnt
    return _grad_from_weights(h, wP, wN)

def run(seed=42, budget_s=1100, **kwargs):

        if bc < best_c: best_c=bc

    # Phase 2: Subgradient (~100s)

    if e2 > time.time():

                    for a in [50, 400, 3000]:

        for a in [100, 500, 2500, 10000, 50000]:
            if time.time() > e4-2: break

            hm = hf if _c5_direct(hf)<_c5_direct(bh) else bh

        if c < best_c: best=h2; best_c=c
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
