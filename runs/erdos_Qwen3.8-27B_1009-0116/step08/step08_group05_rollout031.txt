    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _lp_epigraph(h, n_iter, eps, m_top, deadline):
    n = len(h)
    target = n * 0.5
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        
        # Build gradient matrix
        G = np.zeros((m_top, n))
        for ii, k in enumerate(idx_top):
            sh = (n - 1) - k
            if sh >= 0:
                G[ii, :n-sh] += (1.0 - c[sh:])
                if sh > 0:
                    G[ii, sh:] -= c[:-sh]
            else:
                sp = -sh
                G[ii, sp:] += (1.0 - c[:-sp])
                G[ii, :n-sp] -= c[sp:]
        G *= 2.0 / n
        
        # LP: min t s.t. G@(d) <= cor[idx_top] - cor[idx_top[0]] (allow others to be at most equal to worst)
        # Actually: min t s.t. G@delta + cor[idx_top] <= t, sum(delta)=0, |delta|<=eps
        # Variables: [delta(1..n), t] -> n+1 variables
        # Let's use scipy linprog
        
        # Simpler: just use the previous formulation but with more peaks
        c_obj = -G[0]  # minimize decrease of worst peak... wait
        # Actually: minimize G[0]@delta (most negative = most decrease)
        # s.t. G[i]@delta <= 0 for i=1..m-1 (others don't increase)
        #      sum(delta) = 0
        #      |delta| <= eps
        
        c_obj = G[0]  # we want to minimize this (make it negative)
        A_ub = G[1:]  # G[i]@delta <= 0
        b_ub = np.zeros(m_top - 1)
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        d = res.x
        if np.max(np.abs(d)) < 1e-15:
            break
        
        # Backtracking line search
        step = 1.0
        for _ in range(10):

                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_peak(c, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - c[sh:])
        if sh > 0:
            g[sh:] -= c[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - c[:-sp])
        g[:n - sp] -= c[sp:]
    return g * (2.0 / n)

def _proj_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _grad_peak(h, idx[ii], n)
    return g

def _slsqp_phase(h, alpha, topk, maxit, deadline):
    n = len(h)
    target = n * 0.5
    def obj(x):
        return _smax(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
        return _repair(r.x, n, target)
    except Exception:
        return _repair(h, n, target)

def _lp_phase(h, n_iter, eps_start, eps_end, m_top, deadline):
    n = len(h)
    target = n * 0.5
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        eps = eps_start + (eps_end - eps_start) * (it / max(n_iter, 1))
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        G = np.array([_grad_peak(c, k, n) for k in idx_top])
        c_obj = G[0]
        A_ub = G[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        d = res.x
        if np.max(np.abs(d)) < 1e-15:
            break
        step = 1.0
        improved = False
        for _ in range(10):

def _nesterov_phase(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    target = n * 0.5
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta, step = 0.85, step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_box_sum(h - step * v, target)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.04, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_box_sum(h - step * v, target)
            f = _smax(h, alpha, topk)
    return _repair(h, n, target)

def _anneal_phase(h, n_iter, rng, deadline, T0, T1, block_sizes):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        bs = block_sizes[rng.randint(len(block_sizes))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        noise = 0.003 * max(T, 1e-9) ** 0.4 * rng.randn(bs)
        if T < T0 * 0.3:
            g = _smax_grad(h, 5000, 8)
            noise += 0.0005 * g[i0:i0 + bs]
        noise -= noise.mean()
        seg = np.clip(seg + noise, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n, target)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish_phase(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    scales = [3e-7, 1e-7, 5e-8, 2e-8]
    per_scale = max(n_iter // len(scales), 100)
    for ss in scales:
        for _ in range(per_scale):
            if time.time() > deadline:
                return bh, bc
            move = rng.random()
            if move < 0.45:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 4)) % n
                d = ss * (2 * rng.random() - 1)
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-18:
                        c5c = c5n
                        if c5n < bc:
                            bc, bh = c5n, h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
    return bh, bc

                best_c5, best_h = cv2, hv2.copy()

    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.0004), 60000)
    best_h, best_c5 = _polish_phase(best_h, npol, rng, deadline)

    best_h = _repair(best_h, n, target)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
