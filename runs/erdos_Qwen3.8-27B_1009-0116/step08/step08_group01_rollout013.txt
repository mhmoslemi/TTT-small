    a_plus_s = a + s  # index for h[a+s]
    b_plus_s = b + s
    
    dc = np.zeros(2*n-1)
    m1 = (a_minus_s >= 0) & (a_minus_s < n)

    m3 = (b_minus_s >= 0) & (b_minus_s < n)

    m4 = (b_plus_s >= 0) & (b_plus_s < n)
    dc[m4] += delta * h[b_plus_s[m4]]
    return dc

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = h.copy()
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    stall = 0
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c - 1e-15:
            h = h_try
            cur_c = c_try
            stall = 0
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            stall += 1
            if step < step0 * 0.02 or stall > 30:
                break
    return h, cur_c

def _dc(h, a, b, delta):
    n = len(h)
    s = np.arange(-(n - 1), n)
    ams = a - s
    aps = a + s
    bms = b - s
    bps = b + s
    dc = np.zeros(2 * n - 1)
    m1 = (ams >= 0) & (ams < n)
    dc[m1] += delta * (1.0 - h[ams[m1]])
    m2 = (aps >= 0) & (aps < n)
    dc[m2] -= delta * h[aps[m2]]
    m3 = (bms >= 0) & (bms < n)
    dc[m3] -= delta * (1.0 - h[bms[m3]])
    m4 = (bps >= 0) & (bps < n)
    dc[m4] += delta * h[bps[m4]]
    return dc

def _fast_local(h, c_full, n_iter, step0, rng, sa_temp=0.0, gap_range=8):
    n = len(h)
    cur_c = float(c_full.max())
    best_c = cur_c
    best_h = h.copy()
    step = step0
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, gap_range)) % n
        d = step * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        dc = _dc(h, i, j, d)
        cn = c_full + dc
        nm = float(cn.max())
        accept = nm < cur_c
        if not accept and sa_temp > 1e-10:
            if rng.rand() < np.exp(-(nm - cur_c) / sa_temp):
                accept = True
        if accept:
            h[i], h[j] = ni, nj
            c_full = cn
            cur_c = nm
            step = min(step * 1.01, step0 * 3.0)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            step = max(step * 0.95, step0 * 0.1)
    return best_h, best_c, _corr_full(best_h)

def _sweep_local(h, c_full, n_passes, step, rng, gap_range=6):
    n = len(h)
    cur_c = float(c_full.max())
    best_c = cur_c
    best_h = h.copy()
    for p in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, gap_range)) % n
            d = step * rng.randn()
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            dc = _dc(h, i, j, d)
            cn = c_full + dc
            nm = float(cn.max())
            if nm < cur_c:
                h[i], h[j] = ni, nj
                c_full = cn
                cur_c = nm
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
    return best_h, best_c, _corr_full(best_h)

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):
        return _grad_lse(x, alpha, top_k)
    def _sum_c(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
