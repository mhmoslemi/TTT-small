Actually, for a problem like min max_k c_k where c_k are quadratic, the KKT conditions at the optimum are:
- There exist λ_k ≥ 0 for active k, sum λ_k = 1, such that:
  sum_k λ_k * ∇c_k(h) + μ * 1 + ν = 0 (where μ is for sum constraint, ν for box constraints)
- Complementary slackness: ν_i * h_i = 0 for h_i = 0, ν_i = 0 for 0 < h_i < 1

1. Identify active set S (top m correlations)
2. Find λ ≥ 0, sum λ = 1, such that sum λ_k * ∇c_k is as "flat" as possible (minimum norm after projecting)
3. If the projected gradient is ~0, we're at a KKT point
4. Otherwise, take a step in the negative projected gradient direction

This is essentially a Frank-Wolfe or conditional gradient method applied to the dual.

def _kkt_descent(h, deadline, m_active=15):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    for it in range(10000):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_active)[-m_active:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        c_top = cor[idx_top]
        c_max = c_top[0]

        grads = np.array([_sg(c, k, n) for k in idx_top])  # shape (m, n)

        # This is: find λ that makes the weighted average gradient as flat as possible
        # Equivalently: max sum λ_k * (c_max - c_k) s.t. sum λ_k * grad_k ≈ 0

        # λ_k = exp(-c_k / T) / sum exp(-c_j / T)
        T = 1e-4
        w = np.exp(-(c_top - c_max) / T)
        w /= w.sum()
        
        # Weighted gradient
        g = w @ grads  # shape (n,)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lp_step(h, n, m_top, eps):
    cor = _correl(h)
    m_eff = min(m_top, len(cor))
    idx_top = np.argpartition(cor, -m_eff)[-m_eff:]
    idx_top = idx_top[np.argsort(-cor[idx_top])]
    grads = np.array([_sg(h, k, n) for k in idx_top])
    c_obj = -grads[0]
    A_ub = grads[1:] if m_eff > 1 else None
    b_ub = np.zeros(m_eff - 1) if m_eff > 1 else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    return res.x

def _lp_descent(h, n_steps, eps0, m_top, deadline, m_top_growth=None):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    eps = eps0
    mt = m_top
    stall = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        d = _lp_step(c, n, mt, eps)
        if d is None or np.max(np.abs(d)) < 1e-14:
            stall += 1
            if stall > 30:
                if m_top_growth and mt < m_top_growth:
                    mt += 2
                    stall = 0
                    continue
                eps *= 0.7
                if eps < 5e-7:
                    break
                continue
            rng = np.random.RandomState(it * 7 + 13)
            d = rng.randn(n)
            d -= d.mean()
            d *= eps / max(np.max(np.abs(d)), 1e-15)
        else:
            stall = 0
        step = 1.0
        improved = False
        for _ in range(12):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            eps *= 0.95
            if eps < 5e-7:
                break
    return best_h, best_c5

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 5)
        else:
            step *= 0.5
            if step < 1e-11:
                break
            h = _proj(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _sweep_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sss = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8]
    for ss in sss:
        ni2 = max(niter // len(sss), 10)
        for _ in range(ni2):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 5)) % nc
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc
