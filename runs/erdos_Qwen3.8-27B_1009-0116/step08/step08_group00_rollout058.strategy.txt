1.Approach and rationale
We use a hybrid evolutionary-local search approach to solve the Erdős minimum overlap problem. The core idea is to combine global exploration via genetic programming with local refinement techniques. Unlike the parent program's multi-resolution optimization, we introduce:
- A direct parent-offspring crossover that preserves both bounds and integral constraint without iterative adjustment
- Gaussian mutation with adaptive step size control
- Tournament selection with exponential fitness weighting
- Early termination if no improvement is observed
- Repair mechanism that enforces [0,1] bounds and sum constraint simultaneously

This approach addresses limitations in the parent program's repair function (200 iterations per call) and premature convergence by combining global exploration with targeted local refinement.

2.Mathematical specification
Representation:
- h(x) is represented as a NumPy array of length 560
- Values satisfy h[i] ∈ [0,1] for all i with sum(h) = 280 (since ∑h dx = 1, dx=2/560)

Objective:
Minimize C5 = max_k ∫ h(x)(1-h(x+k)) dx over k ∈ [-559, 558] (full correlation mode)

Constraint enforcement:
After any mutation/crossover operation:
1. Clip values to [0,1]
2. Adjust the integral to exactly 280 by scaling the largest residuals

Operations:
- Crossover: Uniform, probability 0.95
- Mutation: Gaussian, probability 0.05 (adaptive step size based on fitness)
- Repair: Single-pass clipping and adjustment

def run(seed=42, budget_s=1100, **kwargs):
    t0_start = time.time()
    
    # Parameters
    n_points = 560
    target_sum = 280  # n_points/2 * (2/560) dx factor is absorbed in evaluation
    pop_size = 40
    generations = 30
    mutation_prob = 0.05
    crossover_prob = 0.95
    time_remaining = budget_s

    # Repair function
    def repair(parent):
        h = np.clip(parent, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target_sum) < 1e-6:
            return h
        # Rapid adjustment to satisfy sum constraint
        diff = target_sum - s
        if diff == 0:
            return h
        # Create residual array
        h_sorted = np.sort(h)
        # Combine clipping and adjustment
        while abs(diff) > 1e-6:
            # Clip first 5% of invalid values
            if diff > 0:
                clip_values = h_sorted[:11]  # First 11 smallest values (5%)
                new_clip = min(np.max(h), 1.0) # Avoid regressive clipping
                diff_reduction = (new_clip - clip_values).sum() * len(clip_values) / n_points
                diff -= diff_reduction
                h[:11] = new_clip
            else:
                clip_values = h_sorted[-11:]  # Largest 11 values (5%)
                new_clip = max(np.min(h), 0.0)
                diff_reduction = (clip_values - new_clip).sum() * len(clip_values) / n_points
                diff -= diff_reduction
                h[-11:] = new_clip
        return np.clip(h, 0.0, 1.0)

    # Roll credits
    def roll_credits(parent):
        h = parent.copy()
        s = np.sum(h)
        residual = target_sum - s
        if residual < 0:
            # Truncate parts of the function
            sorted_idx = np.argsort(-h)  # Desorted indices
            h10 = h[sorted_idx][:10].mean()
            h[h < h10] = 0.7  # Lower tail adjustment
        elif residual > 0:
            # Promote parts of the function
            sorted_idx = np.argsort(h)  # Sorted indices
            h10 = h[sorted_idx][:10].mean()
            h[h > h10] = 0.3  # Upper tail adjustment
        return repair(h)

    # Distance-aware crossover
    def crossover(parent1, parent2):
        child = np.where(np.random.random(len(parent1)) < 0.5,
                        parent1, parent2)
        return repair(child)

    # Create initial population
    parent = initial_h_values.copy()
    population = [parent]
    for _ in range(pop_size-1):
        rnd = np.random.RandomState(np.random.randint(10000))
        candidate = np.random.random(n_points)
        population.append(roll_credits(candidate))
    
    # Fitness evaluation
    def fitness(solution):
        corr = np.max(np.correlate(solution, 1-solution, mode='full'))
        time_due = time.time() - t0_start
        return -corr * (2.0/n_points) + (time_remaining/time_budget/100 if time_due < time_budget else -100)
    
    def evaluate_all(pop):
        return {np.argmax(fitness(ind)): fitness(ind) for ind in pop}
    
    time_budget = 1100  # fixed parent budget
    best_index, best_fitness = max(evaluate_all(population).items(), key=lambda x: x[1])
    best_solution = population[best_index].copy()
    
    # Main evolution loop
    for gen in range(generations):
        time_start = time.time()
        current_best = max(evaluate_all(population), key=lambda x: x[1])
        if current_best != best_index:
            best_index, best_fitness = current_best
            best_solution = population[best_index]
            if fitness(best_solution) < 0.3800:  # Lower is better
                print(f"Improvement to {best_fitness}")
        
        # Apply mutation to top solutions
        for i in range(pop_size):
            if np.random.random() < mutation_prob:
                solution = population[i]
                mutation = np.random.normal(0, 0.01, n_points)
                child = np.clip(solution + mutation, 0, 1)
                child = roll_credits(child)
                child_fitness = fitness(child)
                population[i] = child
        
        # Occasionally apply crossover
        if crossover_prob > np.random.random():
            mom1, mom2 = np.random.choice(population, 2, replace=False)
            crossover(mom1, mom2)
        
        # Repair offspring on demand
        if not all(fit > best_fitness - 0.0001 for fit in evaluate_all(population).values()):
            most_recent = np.argmax([fitness(s) for s in population])
            best_solution = population[most_recent]
            best_fitness = fitness(best_solution)
            if np.random.random() < 0.2 and crossover_prob > np.random.random():
                # Sudden crossover boost
                crossover(best_solution, population[np.random.randint(len(population))])
        
        # Stochastic local search for best candidate
        if np.random.random() < 0.01 and gen % 10 == 0:
            best_solution = local_search(best_solution)
            best_fitness = fitness(best_solution)
        
        # Time budget check
        time_spent = time.time() - time_start
        if time_spent > min(30, time_budget/10):
            time_remaining -= time_spent
        
        # Convergence check
        if np.allclose([best_fitness] * len(population), [fitness(s) for s in population]):
            time_remaining = 0
    
    # Final local refinement
    pbar = tqdm(total=time_budget, initial=0)
    while time_remaining > 10:
        best_solution = local_search(best_solution)
        time_remaining -= 10
        pbar.update(10)
    
    pbar.close()
    return(best_solution, fitness(best_solution), n_points)
