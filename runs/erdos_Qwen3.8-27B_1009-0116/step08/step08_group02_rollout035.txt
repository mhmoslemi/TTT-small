import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c)), c

def _grad_lag(k, h, n):
    """Gradient of the k-th correlation term w.r.t. h, scaled by dx."""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        nvalid = n - sh
        g[:nvalid] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:nvalid]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _slsqp_smooth(h, alpha, maxiter, top_k, t0, tlimit):
    n = len(h)
    dx = 2.0 / n
    full_c = np.correlate(np.ones(n), np.ones(n), mode='full')
    n_lags = len(full_c)
    
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        if top_k > 0:
            idx = np.argsort(c)[-top_k:]
        else:
            idx = np.arange(n_lags)
        cs = c[idx]
        m = cs.max()
        sm = alpha * (cs - m)
        sm = np.clip(sm, -400, 400)
        sm -= sm.max()
        return m + np.log(np.exp(sm).sum()) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        if top_k > 0:
            idx = np.argsort(c)[-top_k:]
        else:
            idx = np.arange(n_lags)
        cs = c[idx]
        m = cs.max()
        sm = np.clip(alpha * (cs - m), -400, 400)
        sm -= sm.max()
        w = np.exp(sm)
        w /= w.sum()
        g = np.zeros(n)
        for ii, kk in enumerate(idx):
            if w[ii] > 1e-14:
                g += w[ii] * _grad_lag(kk, x, n)
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.zeros(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        nv = n - sh
        g[:nv] += (1.0 - h[sh:sh+nv])
        if sh > 0:
            g[sh:sh+nv] -= h[:nv]
    else:
        sp = -sh
        g[:n-sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n-sp]
    return g * dx

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    nlag = 2 * n - 1
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        if 0 < top_k < nlag:
            cs = c[np.argsort(c)[-top_k:]]
        else:
            cs = c
        m = cs.max()
        sm = np.clip(alpha*(cs-m), -400, 400) - np.clip(alpha*(cs-m), -400, 400).max()
        return m + np.log(np.exp(sm).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        if 0 < top_k < nlag:
            idx = np.argsort(c)[-top_k:]
        else:
            idx = np.arange(nlag)
        cs = c[idx]
        m = cs.max()
        sm = np.clip(alpha*(cs-m), -400, 400)
        sm -= sm.max()
        w = np.exp(sm); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-14:
                g += w[ii] * _grad_lag(int(idx[ii]), x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
