def _fft_corr(h):
    n = len(h)
    b = (1.0 - h)[::-1]
    size = 1
    while size < 2*n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    B = np.fft.rfft(b, size)
    corr = np.fft.irfft(H * B, size)[:2*n-1]
    return corr

import numpy as np
from scipy.optimize import minimize

def _fft_corr(h):
    """Fast correlation using FFT."""
    n = len(h)
    b = (1.0 - h)[::-1]
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    H = np.fft.rfft(h, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(H * B, size)[:target]

def _eval(h):
    n = len(h)
    return float(np.max(_fft_corr(h)) * 2.0 / n)

def _eval_all(h):
    n = len(h)

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:

    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _grad_at_lag(k, h, n):
    """Gradient of c_k with respect to h, where k is the index in the full correlation (0 to 2n-2)."""
    # Full correlation index k corresponds to lag m = k - (n-1)
    m = k - (n - 1)  # actual lag

        # d/dh[i] += -h[i-m] for i in [m, n-1]
        i1 = np.arange(n - m)
        g[i1] += (1.0 - h[i1 + m])
        if m > 0:
            i2 = np.arange(m, n)
            g[i2] -= h[i2 - m]
    else:

        g[i1] += (1.0 - h[i1 - p])
        i2 = np.arange(n - p)
        g[i2] -= h[i2 + p]
    return g * (2.0 / n)

def _binary_round_and_fix(h, n):
    """Round to nearest binary, then adjust to meet sum constraint, then refine."""
    b = (h > 0.5).astype(float)
    # Adjust count
    target = n // 2
    cnt = b.sum()
    # If too many ones, flip some 1s to 0s (those closest to 0.5)
    if cnt > target:
        ones = np.where(b == 1)[0]
        order = ones[np.argsort(h[ones])]  # ones with smallest h
        for idx in order[:int(cnt - target)]:
            b[idx] = 0
    elif cnt < target:
        zeros = np.where(b == 0)[0]
        order = zeros[np.argsort(-h[zeros])]  # zeros with largest h
        for idx in order[:int(target - cnt)]:
            b[idx] = 1
    # Now b is binary with correct sum. Refine around it.
    return _repair(b, n)

        # Second pass with stochastic selection
        if _el() < t_lim4 - 60:
            n_iter2 = min(int((t_lim4 - 60 - _el()) / eps * 0.7), 30000000)
            best_h, best_c5 = _directed_search(best_h, n_iter2, rng, t0, t_lim4, tag=2)

    # Phase 5: Ultra-fine polish (remaining)
    if _el() < budget_s - 2:
        t_lim5 = budget_s - 2
        _t = _el()
        for _ in range(20):
            _eval(best_h)
        eps = max((_el() - _t) / 20.0, 5e-6)
        n_iter = min(int((t_lim5 - _el()) / eps * 0.85), 50000000)
        best_h, best_c5 = _ultra_fine(best_h, n_iter, rng, t0, t_lim5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
