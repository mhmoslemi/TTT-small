    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:

                h[idx] += a

import numpy as np
import cvxpy as cp
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _grad_lag(k, h, n):
    dx = 2.0 / n
    shift = (n - 1) - k
    g = np.zeros(n)
    if shift >= 0:
        if shift == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            i1 = np.arange(n - shift)
            g[i1] += (1.0 - h[i1 + shift])
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:
        sp = -shift
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _solve_qp(h0, n, lags, mu, solver):
    dx = 2.0 / n
    c = np.correlate(h0, 1.0 - h0, mode='full') * dx
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
    for ki in lags:
        g = _grad_lag(int(ki), h0, n)
        constraints.append(c[int(ki)] + g @ (h_var - h0) <= t_var)
    obj = t_var + (1.0 / (2.0 * mu)) * cp.sum_squares(h_var - h0)
    prob = cp.Problem(cp.Minimize(obj), constraints)
    solved = False
    for s in [solver, 'SCIP', 'CVXOPT', 'ECOS']:
        try:
            prob.solve(solver=s, verbose=False)
            solved = True
            break
        except Exception:
            continue
    if solved and h_var.value is not None:
        return np.clip(np.asarray(h_var.value).flatten(), 0, 1)
    return h0.copy()

def _solve_lp(h0, n, lags, solver):
    dx = 2.0 / n
    c = np.correlate(h0, 1.0 - h0, mode='full') * dx
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
    for ki in lags:
        g = _grad_lag(int(ki), h0, n)
        constraints.append(c[int(ki)] + g @ (h_var - h0) <= t_var + 1e-12)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solved = False
    for s in [solver, 'ECOS', 'PDLP', 'SLS']:
        try:
            prob.solve(solver=s, verbose=False)
            solved = True
            break
        except Exception:
            continue
    if solved and h_var.value is not None:
        return np.clip(np.asarray(h_var.value).flatten(), 0, 1)
    return h0.copy()

def _sqp_round(h0, n, n_iter, tms, tme, mus, mue, mode):
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        top_m = int(tms + (tme - tms) * frac)
        mu = mus * (mue / mus) ** frac
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if top_m >= len(c):
            lags = np.arange(len(c))
        else:
            lags = np.argsort(c)[-top_m:]
        if mode == 'qp':
            h_new = _solve_qp(h, n, lags, mu, 'SCIP')
        else:
            h_lp = _solve_lp(h, n, lags, 'SLS')
            h_new = h.copy()
            best_try = best_c
            for alpha in [1.0, 0.75, 0.5, 0.3, 0.15]:
                ht = _repair(h + alpha * (h_lp - h), n)
                ct = _eval(ht)
                if ct < best_try:
                    best_try = ct
                    h_new = ht
                if ct < best_c:
                    best_c = ct
                    best_h = h_new.copy()
        h = _repair(h_new, n)
        cn = _eval(h)
        if cn < best_c:
            best_c = cn
            best_h = h.copy()
    return best_h, best_c

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, steps, step0, top_k, momentum):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s_arr = np.clip(alpha * (c_a - m_a), -400, 400)
        s_arr -= s_arr.max()
        w = np.exp(s_arr)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(int(active[ii]), h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 15.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.2, step0 * 15.0)
        if it % 150 == 0:
            v *= 0.4
    return _repair(h, n)

            h_alt, c_alt = _sqp_round(h_alt, n, 25, 20, 500, 0.2, 0.02, 'qp')
            if c_alt < best_c5:
                best_c5 = c_alt
                best_h = h_alt.copy()

    # Phase 4: Projected gradient with Nesterov-like momentum
    if time.time() - t0 < budget_s - 130:
        for alpha, steps, s0, tk, mu in [(3000, 200, 0.004, 5, 0.55),
                                          (30000, 150, 0.002, 3, 0.65)]:
            if time.time() - t0 > budget_s - 90:
                break
            h_new = _proj_grad(best_h.copy(), alpha, steps, s0, tk, mu)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 5: Informed local search (targets worst lag)
    if time.time() - t0 < budget_s - 40:
        time_left = budget_s - 40 - (time.time() - t0)
        t_est = time.time()
        for _ in range(20):
            _eval_worst(best_h)
        eps = max((time.time() - t_est) / 20.0, 1e-6)
        ntries = min(int(time_left / eps * 0.6), 500000)
        best_h, best_c5 = _local_informed(best_h, ntries, rng, 5e-6)

    # Phase 6: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 2:
        time_left = budget_s - 2 - (time.time() - t0)
        t_est = time.time()
        for _ in range(20):
            _eval(best_h)
        eps = max((time.time() - t_est) / 20.0, 1e-6)
        npol = min(int(time_left / eps * 0.75), 5000000)
        best_h, best_c5 = _local_2pt(best_h, npol, rng, 5e-7)

    # Final repair and evaluation
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    if best_c5 > 0.380891:
        best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
        best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
