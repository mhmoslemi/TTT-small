import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_soft(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp + cnt]
    return g

def _adam(h, n_iter, alpha, top_k, lr):
    n = len(h)
    tm = np.zeros(n)
    tv = np.zeros(n)
    b1, b2 = 0.85, 0.999
    best_h = h.copy()
    best_c = _eval(h)
    for t in range(1, n_iter + 1):
        g = _grad_soft(h, alpha, top_k)
        tm = b1 * tm + (1 - b1) * g
        tv = b2 * tv + (1 - b2) * g * g
        mh = tm / (1.0 - b1 ** t)
        vh = tv / (1.0 - b2 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _slsqp_obj(x, n, alpha, top_k):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx2 = np.argsort(c)[-top_k:]
    ct = c[idx2]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_con(x, n):
    return x.sum() - n * 0.5

def _slsqp_con_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = []
    entry = {}
    entry['type'] = 'eq'
    entry['fun'] = lambda x: x.sum() - target
    entry['jac'] = lambda x: np.ones(n)
    cons.append(entry)
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha, top_k),
                      jac=lambda x: _grad_soft(x, alpha, top_k),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _local2(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

                        best_c = c_new
                        best_h = h.copy()
                else:
