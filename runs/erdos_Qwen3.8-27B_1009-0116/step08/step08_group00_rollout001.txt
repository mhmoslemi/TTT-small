    lo, hi = -2.0, 2.0

            lo = mid

    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _anneal(h, n_iter, rng, t_start, t_end):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = 1e-5 * rng.randn()
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        c5n = _eval(h)
        if c5n < c5 or rng.rand() < np.exp(-(c5n - c5) / max(t, 1e-12)):
            c5 = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c5

def _fine_polish(h, n_iters, rng):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    h2 = h.copy()
    c5c = c5
    for it in range(n_iters):
        scale = max(1e-7, 5e-6 * (1.0 - it / max(n_iters, 1)))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        h2[i] += d
        h2[j] -= d
        if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
            c5n = _eval(h2)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h2.copy()
            else:
                h2[i] -= d
                h2[j] += d
        else:
            h2[i] -= d
            h2[j] += d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution SLSQP
    for ratio in [4, 8]:
        if time.time() - t0 > 40:
            break
        nc = max(40, n // ratio)
        hc = _downsample(h, n // nc)
        for alpha in [2000, 10000, 100000]:
            if time.time() - t0 > 55:
                break
            hc = _slsqp(hc, alpha, 30, 300, 1e-14)
        h_up = _upsample(hc, n // nc)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: FISTA with increasing sharpness
    alpha_seq = [3000, 10000, 40000, 150000, 600000, 2500000, 10000000, 40000000]
    for alpha in alpha_seq:
        if time.time() - t0 > 300:
            break
        ml = min(70, max(15, int(25 * np.log10(alpha / 3000.0) + 15)))
        h_new = _fista(best_h.copy(), alpha, ml, 300, 0.002)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: SLSQP with high alpha
    for alpha in [100000, 500000, 2000000, 10000000, 50000000]:
        if time.time() - t0 > 550:
            break
        ml = min(90, max(25, int(30 * np.log10(alpha / 100000.0) + 25)))
        h_new = _slsqp(best_h, alpha, ml, 600, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: Perturb + FISTA to escape local optima
    if time.time() - t0 < budget_s - 200:
        for trial in range(5):
            if time.time() - t0 > budget_s - 150:
                break
            h_pert = best_h + 0.01 * rng.randn(n)
            h_pert = _repair(h_pert, n)
            h_new = _fista(h_pert, 50000, 40, 200, 0.003)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Simulated annealing
    if time.time() - t0 < budget_s - 80:
        remaining = budget_s - 80 - (time.time() - t0)
        n_ann = min(int(remaining / 0.004), 150000)
        best_h, best_c5 = _anneal(best_h, n_ann, rng, 2e-4, 1e-9)

    # Phase 6: Fine coordinate polish
    if time.time() - t0 < budget_s - 5:
        remaining = budget_s - 5 - (time.time() - t0)
        n_pol = min(int(remaining / 0.0012), 600000)
        best_h, best_c5 = _fine_polish(best_h, n_pol, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
