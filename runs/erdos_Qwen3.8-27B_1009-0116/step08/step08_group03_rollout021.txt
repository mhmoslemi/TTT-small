def _repair(h, n):

    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):

        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:

                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    """Project v onto {x: 0<=x<=1, sum(x)=S}."""
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size

        c = m.sum()
        if c == 0:
            break

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_arrays(h):
    """Compute P (positive lags) and Nv (negative lags) overlap arrays."""
    n = len(h)
    N = 1
    m = 2 * n - 1
    while N < m:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    return P, Nv

def _smooth_value(h, alpha):
    P, Nv = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    mx = float(np.max(np.concatenate([P, Nv])))
    e = np.exp(np.clip(alpha * (P - mx), -400.0, 0.0)).sum()
    e += np.exp(np.clip(alpha * (Nv - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _smooth_grad(h, alpha):
    P, Nv = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    mx = float(np.max(np.concatenate([P, Nv])))
    wP = np.exp(np.clip(alpha * (P - mx), -400.0, 0.0))

    tot = wP.sum() + wN.sum()
    if tot <= 0:

    wP /= tot
    wN /= tot
    return _grad_from_weights(h, wP, wN)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        convR = np.fft.irfft(H * Urev, Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * Ur, Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

    return best, best_c, len(best)
