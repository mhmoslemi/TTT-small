        return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return h

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_pair(h, j, p, n):
    L = np.arange(-(n - 1), n)
    jp = j + L
    jm = j - L
    term_j = np.where((jp >= 0) & (jp < n), 1.0 - h[jp], 0.0) - np.where((jm >= 0) & (jm < n), h[jm], 0.0)
    pp = p + L
    pm = p - L
    term_p = np.where((pp >= 0) & (pp < n), 1.0 - h[pp], 0.0) - np.where((pm >= 0) & (pm < n), h[pm], 0.0)
    return term_j - term_p

def _bisect_min(c, g, lo, hi, niter=45):
    if lo >= hi:
        return lo
    for _ in range(niter):
        mid = (lo + hi) * 0.5
        km = np.argmax(c + mid * g)
        if g[km] > 1e-14:
            hi = mid
        elif g[km] < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        cc = _corr(x) * dx
        m = cc.max()
        idx = np.argsort(cc)[-m_lags:]
        s = np.clip(alpha * (cc[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _obj_g(x):
        cc = _corr(x) * dx
        m = cc.max()
        idx = np.argsort(cc)[-m_lags:]
        s = np.clip(alpha * (cc[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx
            L = int(k) - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n - L])
                g[:n - L] -= wk * x[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h
