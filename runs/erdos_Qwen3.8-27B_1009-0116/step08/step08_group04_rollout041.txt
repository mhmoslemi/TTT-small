    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            shift = idx[ii] - (n - 1)
            g += w[ii] * _lag_grad(h, shift, dx)
    return g

def _nesterov(h, alpha0, alpha1, topk0, topk1, steps, step0, beta0, beta1, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk0)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline: break
        t = it / max(steps - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** t
        topk = max(1, int(round(topk0 + (topk1 - topk0) * t)))
        beta = beta0 + (beta1 - beta0) * t
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        for _try in range(6):
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn; f = fn; step = min(step * 1.05, step0 * 20); break
            step *= 0.5
        else:
            step *= 0.5
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f: h = hn; f = fn
            else: break
        if f < best_f: best_f = f; best_h = h.copy()
    return _repair(best_h, n)

def _subgrad_minimax(h, n_iter, n_top, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    step = 0.01
    momentum = np.zeros(n)
    beta = 0.7
    for it in range(n_iter):
        if time.time() > deadline: break
        c = _correl(h)
        idx = np.argpartition(c, -n_top)[-n_top:]
        w = c[idx] - c.min(); w = np.maximum(w, 0); w /= (w.sum() + 1e-15)
        g = np.zeros(n)
        for ii in range(len(idx)):
            shift = idx[ii] - (n - 1)
            g += w[ii] * _lag_grad(h, shift, dx)
        momentum = beta * momentum + (1 - beta) * g
        hn = _proj(h - step * momentum, n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c:
            h = hn; c5c = c5n; step = min(step * 1.03, 0.05)
            if c5n < bc: bc = c5n; bh = h.copy()
        else:
            step *= 0.6
            hn = _proj(h - step * momentum, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5c: h = hn; c5c = c5n
            else: break
    return _repair(bh, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_reopt(h, block_sizes, n_rounds, rng, deadline):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(n_rounds):
        if time.time() > deadline: break
        bs = block_sizes[rng.randint(len(block_sizes))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * rng.randn(bs); d -= d.mean()
        seg_new = np.clip(seg + d, 0, 1)
        seg_new = _proj(seg_new, seg_new.sum())
        hn = h.copy(); hn[i0:i0 + bs] = seg_new
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c:
            h = hn; c5c = c5n
            if c5n < bc: bc = c5n; bh = h.copy()
    return _repair(bh, n)

def _fine_polish(h, n_iter, rng, deadline):
    nc = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    step_b = 5e-6
    for _ in range(n_iter):
        if time.time() > deadline: break
        i = rng.randint(nc); j = (i + 1 + rng.randint(0, 5)) % nc
        d = step_b * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20: c5c = c5n
            else: h[i], h[j] = ni - d, nj + d; step_b = max(step_b * 0.97, 1e-8)
            if c5n < bc: bc = c5n; bh = h.copy()
    return _repair(bh, nc)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h)

    # Phase 1: SLSQP warm-up (0-80s)
    dl1 = min(deadline, t0 + 80)
    for alpha, tk, mi in [(2000, 40, 300), (10000, 20, 250)]:
        if time.time() > dl1: break
        h = _slsqp(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5: best_c5 = c5v; best_h = h.copy()

    # Phase 2: Multi-scale Nesterov (80-350s)
    dl2 = min(deadline, t0 + 350)
    h_w = best_h.copy()
    for a0, a1, tk0, tk1, st, ns, b0, b1 in [
        (3000, 50000, 50, 10, 0.006, 4000, 0.85, 0.6),
        (20000, 300000, 20, 5, 0.003, 4000, 0.80, 0.5),
        (150000, 2000000, 10, 3, 0.001, 4000, 0.70, 0.4),
        (1000000, 10000000, 5, 1, 0.0005, 3000, 0.5, 0.2),
    ]:
        if time.time() > dl2: break
        h_w = _nesterov(h_w, a0, a1, tk0, tk1, ns, st, b0, b1, dl2)
        c5v = _eval(h_w)
        if c5v < best_c5: best_c5 = c5v; best_h = h_w.copy()

    # Phase 3: Subgradient minimax (350-550s)
    dl3 = min(deadline, t0 + 550)
    for nt, ni in [(5, 8000), (3, 8000), (8, 6000)]:
        if time.time() > dl3: break
        hs = _subgrad_minimax(best_h.copy(), ni, nt, dl3)
        c5v = _eval(hs)
        if c5v < best_c5: best_c5 = c5v; best_h = hs.copy()

    # Phase 4: Block re-optimization (550-750s)
    dl4 = min(deadline, t0 + 750)
    bs_list = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48]
    hb = _block_reopt(best_h.copy(), bs_list, 80000, rng, dl4)
    c5v = _eval(hb)
    if c5v < best_c5: best_c5 = c5v; best_h = hb.copy()

    # Phase 5: SLSQP fine-tune (750-950s)
    dl5 = min(deadline, t0 + 950)
    for alpha, tk, mi in [(100000, 5, 300), (500000, 3, 250), (2000000, 2, 200)]:
        if time.time() > dl5: break
        h_s = _slsqp(best_h.copy(), alpha, tk, mi, dl5)
        c5v = _eval(h_s)
        if c5v < best_c5: best_c5 = c5v; best_h = h_s.copy()

    # Phase 6: Final Nesterov + fine polish (950-end)
    dl6 = deadline
    if time.time() < dl6 - 120:
        h_n = _nesterov(best_h.copy(), 5000000, 50000000, 2, 1, 2000, 0.0003, 0.3, 0.0, dl6 - 120)
        c5v = _eval(h_n)
        if c5v < best_c5: best_c5 = c5v; best_h = h_n.copy()
    rem = max(0, dl6 - time.time())
    n_pol = min(int(rem / 0.0012), 80000)
    best_h, best_c5 = _fine_polish(best_h.copy(), n_pol, rng, dl6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
