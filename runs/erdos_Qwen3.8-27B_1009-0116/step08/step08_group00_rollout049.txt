    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    # Proportional adjustment
    if diff > 0:
        # Add to elements with room
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-14:
            return h
        h += diff * room / total_room
    else:
        h += diff * h / h.sum() if h.sum() > 1e-14 else 0
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:

        else:
            order = np.argsort(h)

                diff -= a

                diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-14:
            h += diff * room / tr
    else:
        hs = h.sum()
        if hs > 1e-14:
            h += diff * h / hs
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _eval(h):
    n = len(h)

def _corr(h):
