    lo, hi = -1.5, 1.5

        if np.clip(h - mid, 0, 1).sum() > target:

    # Positive lags k=1..n-1
    for k in range(1, n):
        A[n-1+k, :n-k] = q[k:]
    A[:, n] = -1.0
    c_obj = np.zeros(n + 1); c_obj[n] = 1.0
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    b_eq = np.array([n*0.5])
    bounds = [(0,1)]*n + [(0,None)]
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return _repair(res.x[:n], n)
    return h

def _eval(h):

def _correl(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    h = _proj(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _build_lp(q, n):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n + 1))
    A[n - 1, :n] = q
    for k in range(1, n):
        A[n - 1 - k, k:n] = q[:n - k]
        A[n - 1 + k, :n - k] = q[k:]
    A[:, n] = -1.0
    return A

    A = _build_lp(q, n)

    c_obj = np.zeros(n + 1); c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0

                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1, deadline - time.time())})
        if res.success:
            return _repair(res.x[:n], n)
    except Exception:
        pass
    return h

def _grad_shift(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _nesterov(h, n_iter, step0, deadline, tol=0.003, beta=0.87, topk=5):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        if topk < len(c):
            active = np.argpartition(c, -topk)[-topk:]
        else:
            active = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for k in active:
            g += _grad_shift(h, int(k), n)
        g /= max(len(active), 1)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f:

            step = min(step * 1.02, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step = max(step * 0.45, 1e-6)
    return _repair(best_h, n), best_f
