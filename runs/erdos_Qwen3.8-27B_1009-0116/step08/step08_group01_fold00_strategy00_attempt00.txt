def estimate_gradient(h):
    # Calculate gradient for all overlaps using analytical derivatives
    # C[k] = (1/n) * ∑_{i} h[i]*(1-h[i+|k|])
    n = len(h)
    k_max = n-1
    grad = np.zeros(n)
    k_ind = argmax_overlap(h)  # most active lag index
    k = k_ind

    if k >= 0:  # positive lag
        for i in range(n-k):
            # Derivative = (1 - h[i+k]) - h[i]
            grad[i] += (1 - h[i+k]) - h[i]
    else:  # negative lag
        for i in range(k, n):
            # Derivative = (1 - h[i]) - h[i+k]  (k negative)
            grad[i] += (1 - h[i]) - h[i+k]

    return grad

def argmax_overlap(h):
    c_all = np.correlate(h, 1-h, mode='full')
    return np.argmax(c_all)

def affine_project(h):
    current_sum = h.sum()
    target_sum = len(h)*0.5
    diff = target_sum - current_sum
    h += diff / len(h)
    return np.clip(h, 0, 1)

def repair_array(h):
    # Fix boundary conditions and normalize sum
    h[0] = 0
    h[-1] = 1
    s = h.sum()
    if abs(s - 0.5*len(h)) > 1e-4:
        h += (0.5*len(h) - s) / len(h)
    return np.clip(h, 0, 1)

def trust_region_search(h0):
    # Local optimization using trust-region
    n_dim = len(h0)
    x = h0.copy()
    r = 0.1  # Initial trust region radius
    max_iter = 100
    f = lambda x: np.max(np.correlate(x, 1-x, mode='full'))
    f0 = f(x)

    grad = estimate_gradient(x)
    norm_g = np.linalg.norm(grad)

    for it in range(max_iter):
        if norm_g < 1e-6:

            r = min(r * 1.5, 0.3*n_dim)

        norm_g = np.linalg.norm(grad)

    return x

- Random perturbation coefficient: [0.01, 0.05]
- Gradient descent step size: 0.005-0.02 (adaptive)
- Trust region radius: [0.05, 0.1] (depends on dimension)
- Global-to-local transition: triggered by both time and improvement rate

Adaptations include:
1. Coarser resolution experiments (n=80-200) as initial heuristics
