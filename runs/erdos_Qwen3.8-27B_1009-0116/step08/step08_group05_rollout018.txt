import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k, sh, wk = idx[ii], (n-1)-idx[ii], w[ii]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-h[sh:])
            if sh > 0: g[sh:] -= wk*h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x): return _smax_obj(x, alpha, topk)
    def jac(x): return _grad_topk(x, alpha, topk)
    def cfun(x): return x.sum() - n*0.5
    def cjac(x): return np.ones(n)
    cons = [{'type':'eq','fun':cfun,'jac':cjac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter':maxit,'ftol':1e-18,'maxls':80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stuck = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-15:
            stuck += 1
            if stuck > 50:
                break
            d = np.random.RandomState(it).randn(n)
            d -= d.mean(); d *= eps*0.1 / max(np.max(np.abs(d)),1e-15)
        else:
            stuck = 0
            d = res.x
        step = 1.0
        improved = False
        for _ in range(14):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            if best_c5 - c5_cur > 1e-12:
                c5_cur = _eval(_repair(c, n))
    return best_h, best_c5

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta, step = 0.85, step0
    best_h, best_c5 = h.copy(), _eval(h)
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_topk(h, alpha, topk)
        v = beta*v + (1.0-beta)*g
        h_new = _proj(h - step*v, n*0.5)
        f_new = _smax_obj(h_new, alpha, topk)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step*1.05, step0*8.0)
            c5_c = _eval(h)
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h.copy()
        else:
            step *= 0.5
            if step < 1e-12: break
            h = _proj(h - step*v, n*0.5)
            f = _smax_obj(h, alpha, topk)
    return best_h

def _block_cd(h, block_size, n_passes, deadline, alpha):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    n_blocks = max(1, n // block_size)
    for _pass in range(n_passes):
        if time.time() > deadline: break
        for _b in range(n_blocks):
            if time.time() > deadline: break
            i0 = _b * block_size
            i1 = min(i0 + block_size, n)
            bs = i1 - i0
            h_block = h[i0:i1].copy()
            def bo(x):
                ht = h.copy(); ht[i0:i1] = x
                return _smax_obj(ht, alpha, 20)
            def bj(x):
                ht = h.copy(); ht[i0:i1] = x
                return _grad_topk(ht, alpha, 20)[i0:i1]
            try:
                r = minimize(bo, h_block, jac=bj, method='SLSQP',
                             bounds=[(0,1)]*bs, options={'maxiter':40,'ftol':1e-14})
                if r.success and np.max(np.abs(r.x - h_block)) > 1e-10:
                    h_new = h.copy(); h_new[i0:i1] = r.x
                    h_new = _repair(h_new, n)
                    c5_new = _eval(h_new)
                    if c5_new < best_c5:
                        h, best_h, best_c5 = h_new, h_new.copy(), c5_new
            except Exception:
                pass
    return best_h, best_c5

def _coord_polish(h, n_iters, rng, deadline):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    scales = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    for ss in scales:
        ni2 = n_iters // len(scales)
        for _ in range(ni2):
            if time.time() > deadline: return bh, bc
            i = rng.randint(n)
            d = ss*(2*rng.random()-1)
            h[i] += d
            # find compensation
            j = rng.randint(n)
            while j == i: j = rng.randint(n)
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc: bc, bh = c5n, h.copy()
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return bh, bc

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0*(T1/T0)**(it/max(niter,1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n-bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002*max(T,1e-9)**0.5*rng.randn(bs); d -= d.mean()
        seg = np.clip(seg+d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n-c5c)/max(T,1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc
