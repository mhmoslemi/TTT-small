def _adaptive_topk_descent(h, n, t_end, step0=0.001, max_k=50, min_k=1):
    """Gradient descent with adaptive top-k focus."""
    best_h = h.copy()
    best_c = _eval(best_h)
    v = np.zeros(n)  # momentum
    mu = 0.7
    step = step0
    step_max = step0 * 20
    
    while time.time() < t_end:
        c = np.correlate(best_h, 1.0 - best_h, mode='full') * 2.0 / n
        m = c.max()
        # Adaptive k: how many correlations are within 1% of max?
        threshold = m - 0.01 * m
        n_above = np.sum(c >= threshold)
        k = max(min_k, min(max_k, n_above))
        
        top_k = np.argsort(c)[-k:]
        g = np.zeros(n)
        wk = 1.0 / k
        for ki in top_k:
            g += wk * _grad_lag(best_h, ki, n)
        
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(best_h - step * v, n)
        c_new = _eval(h_new)
        
        if c_new < best_c:
            best_c = c_new
            step = min(step * 1.05, step_max)
            v = mu * v  # keep momentum
        else:
            step *= 0.5
            v *= 0.5
            if step < 1e-12:

                v = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, n):
    g = np.zeros(n)
    dx2 = 2.0 / n
    sh = n - 1 - k
    if sh >= 0:
        g[:n-sh] += dx2 * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx2 * h[:n-sh]
    else:
        sp = -sh
        g[sp:] += dx2 * (1.0 - h[:n-sp])
        g[:n-sp] -= dx2 * h[sp:]
    return g

class _LSE:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        sm = s.max()
        return m + np.log(np.sum(np.exp(s - sm))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        sm = s.max()
        w = np.exp(s - sm)
        w /= w.sum()
        g = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-14:
                continue
            g += w[k] * _grad_lag(x, k, self.n)
        return g

class _SumCon:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return x.sum() - self.target
    def jac(self, x):
        return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _LSE(alpha, n)
    sc = _SumCon(target)
    cons = [{'type': 'eq', 'fun': sc.fun, 'jac': sc.jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_momentum(h, n, t_end, step0, mu, max_k):
    best_h = h.copy()
    best_c = _eval(best_h)
    v = np.zeros(n)
    step = step0
    step_max = step0 * 30.0
    stall = 0
    while time.time() < t_end:
        c = np.correlate(best_h, 1.0 - best_h, mode='full') * 2.0 / n
        m = c.max()
        thr = m - 0.005 * m
        k = max(1, min(max_k, int(np.sum(c >= thr))))
        top_k = np.argsort(c)[-k:]
        g = np.zeros(n)
        wk = 1.0 / k
        for ki in top_k:
            g += wk * _grad_lag(best_h, ki, n)
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(best_h - step * v, n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
            step = min(step * 1.04, step_max)
            stall = 0
        else:
            step *= 0.5
            v *= 0.3
            stall += 1
            if step < 1e-13:
                step = step0 * 0.1
                v = np.zeros(n)
            if stall > 200:
                break
    return best_h, best_c

def _guided_local(h, n, rng, t_end):
    best_h = h.copy()
    best_c = _eval(best_h)
    while time.time() < t_end:
        c = np.correlate(best_h, 1.0 - best_h, mode='full') * 2.0 / n
        k_w = int(np.argmax(c))
        g = _grad_lag(best_h, k_w, n)
        top_idx = np.argsort(np.abs(g))[-30:]
        i = top_idx[rng.randint(30)]
        j = top_idx[rng.randint(30)]
        if i == j:
            j = rng.randint(n)
        d = 5e-6 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            c_new = _eval(best_h)
            if c_new < best_c:
                best_c = c_new
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c

def _local_3pt(h, n, rng, t_end):
    best_h = h.copy()
    best_c = _eval(best_h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + rng.randint(2, 8)) % n
        k = (i + rng.randint(9, 20)) % n
        d1 = 1.5e-5 * rng.randn()
        d2 = 1.5e-5 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return best_h, best_c

def _local_2pt(h, n, rng, t_end):
    best_h = h.copy()
    best_c = _eval(best_h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n

        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _block_perturb(h, n, rng, t_end):
    best_h = h.copy()
    best_c = _eval(best_h)
    while time.time() < t_end:
        start = rng.randint(n - 10)
        length = rng.randint(3, 11)
        end = min(start + length, n)
        delta = 2e-4 * rng.randn()
        old = best_h[start:end].copy()
        best_h[start:end] = np.clip(old + delta, 0, 1)
        c_new = _eval(best_h)
        if c_new > best_c:
            best_h[start:end] = old
    return best_h, best_c
