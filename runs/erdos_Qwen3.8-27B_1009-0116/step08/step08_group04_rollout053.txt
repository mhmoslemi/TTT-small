def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

_NG = 0

def _con_fun(x):
    return x.sum() - _NG / 2.0

def _con_jac(x):
    return np.ones(_NG)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                a = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
                h[idx] += a if diff > 0 else -a
                diff -= a if diff > 0 else -a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, lag_idx, dx):
    n = len(h)
    g = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        if n - k > 0:
            g[:n-k] += (1.0 - h[k:]) * dx
            g[k:] -= h[:n-k] * dx
    else:
        s = -k
        g[s:] += (1.0 - h[:n-s]) * dx
        g[:n-s] -= h[s:] * dx
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, topk, sharpness):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    w = np.exp((c[idx] - c[idx].max()) * sharpness)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-12:
            g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _adam_opt(h, steps, deadline, lr, topk, sharpness):
    n = len(h)
    target = n / 2.0
    hf = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    for _ in range(steps):
        if time.time() > deadline:
            break
        t += 1
        g = _topk_grad(hf, topk, sharpness)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        hf = _proj_sum(hf - lr * mh / (np.sqrt(vh) + 1e-8), target)
    return _repair(hf, n)

def _nesterov_opt(h, steps, deadline, step0, topk, sharpness):

    target = n / 2.0

    v = np.zeros(n)
    beta = 0.82

    stall = 0

            break
        g = _topk_grad(hf, topk, sharpness)
        v = beta * v + (1.0 - beta) * g
        for _ in range(10):

            ft = _eval(ht)
            if ft < f:
                hf = ht
                f = ft
                stall = 0
                break
            step *= 0.5
            if step < 1e-10:

        else:

    return _repair(hf, n)

def _coord_descent(h, n_sweeps, deadline, rng):
    n = len(h)
    target = n / 2.0
    hf = h.copy()
    fc = _eval(hf)
    bc = fc
    bh = hf.copy()
    step = 2e-4
    for sw in range(n_sweeps):
        if time.time() > deadline:
            break
        idx = np.arange(n)
        for i in idx:
            if time.time() > deadline:
                break
            j = (i + 1 + rng.randint(0, 4)) % n
            d = step * rng.randn()
            ni, nj = hf[i] + d, hf[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                hf[i], hf[j] = ni, nj
                fn = _eval(hf)
                if fn < fc - 1e-16:
                    fc = fn
                    if fn < bc:
                        bc = fn
                        bh = hf.copy()
                    step = min(step * 1.01, 5e-3)
                else:
                    hf[i], hf[j] = hf[i] - hf[i] + hf[i] - d + d, hf[j] + d
                    hf[i], hf[j] = ni - d, nj + d
                    step = max(step * 0.95, 1e-7)
    return bh, bc

def _slsqp(h, maxit, deadline):
    global _NG
    n = len(h)
    _NG = n
    c = _correl(h)
    kmax = np.argmax(c)
    dx = 2.0 / n

    def _obj(x):
        cx = _correl(x)
        return cx.max()

    def _jac(x):
        cx = _correl(x)
        km = np.argmax(cx)
        return _grad_lag(x, km, dx)

    try:
        r = minimize(_obj, h, jac=_jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _ga_search(parent, pop_size, n_gen, deadline, rng):
    n = len(parent)
    pop = [parent.copy() + 0.02 * rng.randn(n) for _ in range(pop_size)]
    pop = [_repair(p, n) for p in pop]
    pop = [parent.copy()] + pop
    best = parent.copy()
    bf = _eval(best)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        fit = np.array([_eval(p) for p in pop])
        order = np.argsort(fit)
        if fit[order[0]] < bf:
            bf = fit[order[0]]
            best = pop[order[0]].copy()
        elite = [pop[order[i]].copy() for i in range(min(4, len(pop)))]
        new_pop = list(elite)
        while len(new_pop) < pop_size:
            c1 = order[rng.randint(0, len(pop))]
            c2 = order[rng.randint(0, len(pop))]
            mask = rng.random(n) < 0.5
            child = np.where(mask, pop[c1], pop[c2])
            fr = max(0.01, 0.05 * (1 - gen / max(n_gen, 1)))
            child += fr * rng.randn(n)
            new_pop.append(_repair(child, n))
        pop = new_pop
    return best

def _block_opt(h, n_iter, deadline, rng, bsize):
    n = len(h)
    target = n / 2.0
    hf = h.copy()
    fc = _eval(hf)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        bs = bsize[rng.randint(len(bsize))]
        i0 = rng.randint(0, max(1, n - bs))
        block = hf[i0:i0+bs].copy()
        best_block = block.copy()
        best_f = fc
        for attempt in range(5):
            pert = block + 0.01 * rng.randn(bs)
            pert = np.clip(pert, 0, 1)
            trial = hf.copy()
            trial[i0:i0+bs] = pert
            trial = _repair(trial, n)
            ft = _eval(trial)
            if ft < best_f:
                best_f = ft
                best_block = pert.copy()
        if best_f < fc:
            hf[i0:i0+bs] = best_block
            hf = _repair(hf, n)
            fc = _eval(hf)
    return _repair(hf, n)
