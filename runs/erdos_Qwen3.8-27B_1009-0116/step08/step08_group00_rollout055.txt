    v = np.zeros(n)
    for it in range(steps):
        g = _smooth_grad(h, alpha, ml)
        v = beta * v - step0 * g
        h_new = _proj(h + v, target)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            if c_new < best:
                best = c_new
                best_h = h.copy()
            if it % 100 == 0 and step0 < 0.005:
                step0 *= 1.05
        else:
            step0 *= 0.75
            v *= 0.5
    return _repair(best_h, n), best

def _fine_search(h, n_iter, rng, base_scale=2e-6, n_pert=3):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        step = base_scale * (0.3 + rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _two_point(h, n_iter, rng, scale):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
