- If all active k have a_k < 0, increase δ (move right)

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_exact(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov(h, steps, step0, deadline, tol=0.001, beta=0.88):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        g, _ = _grad_exact(h, tol)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _coord_a(h, i):
    """Compute a_k = dc_k/dh_i for all k, given h and coordinate i."""
    n = len(h)
    a = np.zeros(2 * n - 1)
    # pos: k = 0..n-1, sh = n-1-k >= 0
    kp = np.arange(n)
    shp = (n - 1) - kp
    c1 = kp >= i  # k >= i means i+sh valid
    c2 = kp >= (n - 1 - i)  # k >= n-1-i means i-sh valid
    a[kp[c1]] += (1.0 - h[i + shp[c1]])
    a[kp[c2]] -= h[i - shp[c2]]
    # neg: k = n..2n-2, sp = k-(n-1) > 0
    kn = np.arange(n, 2 * n - 1)
    spn = kn - (n - 1)
    c3 = spn <= i  # i >= spn means i-spn valid
    c4 = spn < (n - i)  # i < n-spn means spn+i valid
    a[kn[c3]] += (1.0 - h[i - spn[c3]])
    a[kn[c4]] -= h[spn[c4] + i]
    return a

def _solve_1d(c, a, lo, hi):
    """Solve min_{d in [lo,hi]} max_k(c_k + d*a_k) via bisection on subgradient."""
    if hi - lo < 1e-15:
        return 0.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c + mid * a
        mx = vals.max()
        mask = vals >= mx - 1e-13
        if mask.any():
            am = a[mask]
            if am.min() > 1e-13:
                hi = mid
            elif am.max() < -1e-13:
                lo = mid
            else:
                break
    return (lo + hi) * 0.5

def _coord_descent(h, maxpasses, deadline, rng):
    n = len(h)
    target = n * 0.5
    c = _correl(h)
    best_f = _eval(h)
    best_h = h.copy()
    stall = 0
    for p in range(maxpasses):
        if time.time() > deadline:
            break
        old_f = _eval(h)
        for i in range(n):
            if time.time() > deadline:
                break
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            a = _coord_a(h, i)
            delta = _solve_1d(c, a, lo, hi)
            if abs(delta) > 1e-15:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                c += delta * a
        # Periodic repair
        h = _repair(h, n)
        c = _correl(h)
        f = _eval(h)
        if f < best_f - 1e-12:
            best_f, best_h = f, h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3 and p > 10:
            # Perturb and restart
            h = h.copy()
            d = 0.005 * rng.randn(n)
            h = _repair(np.clip(h + d, 0, 1), n)
            c = _correl(h)
            stall = 0
    return best_h, best_f

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_exact(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov(h, steps, step0, deadline, tol=0.001, beta=0.88):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        g, _ = _grad_exact(h, tol)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _coord_a(h, i):
    n = len(h)
    a = np.zeros(2 * n - 1)
    kp = np.arange(n)
    shp = (n - 1) - kp
    c1 = kp >= i
    c2 = kp >= (n - 1 - i)
    if c1.any():
        a[kp[c1]] += (1.0 - h[i + shp[c1]])
    if c2.any():
        a[kp[c2]] -= h[i - shp[c2]]
    kn = np.arange(n, 2 * n - 1)
    spn = kn - (n - 1)
    c3 = spn <= i
    c4 = spn < (n - i)
    if c3.any():
        a[kn[c3]] += (1.0 - h[i - spn[c3]])
    if c4.any():
        a[kn[c4]] -= h[spn[c4] + i]
    return a

def _solve_1d(c, a, lo, hi):
    if hi - lo < 1e-15:
        return 0.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        mx = np.max(c + mid * a)
        mask = (c + mid * a) >= mx - 1e-13
        am = a[mask]
        if am.min() > 1e-13:
            hi = mid
        elif am.max() < -1e-13:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def _coord_descent(h, maxpasses, deadline, rng):
    n = len(h)
    c = _correl(h)
    best_f = _eval(h)
    best_h = h.copy()
    stall = 0
    order = np.arange(n)
    for p in range(maxpasses):
        if time.time() > deadline:
            break
        if p % 10 == 9:
            rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            lo, hi = -h[i], 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            a = _coord_a(h, i)
            delta = _solve_1d(c, a, lo, hi)
            if abs(delta) > 1e-15:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                c += delta * a
        h = _repair(h, n)
        c = _correl(h)
        f = _eval(h)
        if f < best_f - 1e-13:
            best_f, best_h = f, h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5 and p > 20:
            d = 0.003 * rng.randn(n)
            h = _repair(np.clip(h + d, 0, 1), n)
            c = _correl(h)
            stall = 0
    return best_h, best_f

    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
