def _repair(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Bisection on threshold
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

    best_h = h.copy()

        def obj(x):

            return np.dot(p, c)
        def grad(x):
            return _grad_weighted(x, p, n)
        cons = {'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
                'jac': lambda x: np.ones(n)}
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=[(0,1)]*n, constraints=cons,
                      options={'maxiter': inner_iters, 'ftol': 1e-16})
        h = _repair(res.x, n)
        c = _correl(h)
        c5 = c.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Update p: mirror descent
        k_star = c.argmax()
        p_new = (1 - eta) * p
        p_new[k_star] += eta
        p = p_new / p_new.sum()
    return best_h, best_c5

    else:

        return np.full(n, 0.5)

            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:

            for i in idx:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
                if abs(diff) < 1e-13:

                h[i] -= a
                diff += a

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    t = np.clip(alpha * (ct - m), -400, 400)
    return m + (t.max() + np.log(np.sum(np.exp(t - t.max())))) / alpha

def _grad_smax(h, alpha, topk):

    c = _correl(h)

        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-18,
                              'gtol': 1e-10, 'maxfun': maxiter * 3})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.83
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hp = h.copy()
        hn = _repair(h - step * v, n)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 12.0)
        else:
            step *= 0.5
            hn = _repair(hp - step * v, n)
            fn = _smax(hn, alpha, topk)
            if fn < f:
                h, f = hn, fn
            else:
                step *= 0.5
                hn = _repair(hp - step * v, n)
                fn = _smax(hn, alpha, topk)
                if fn < f:
                    h, f = hn, fn
                else:
                    break
    return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _pair_squeeze(h, deadline, nrounds):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for rnd in range(nrounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            j = (i + 1) % n
            mid = 0.5 * (h[i] + h[j])
            spread = 0.003 * (0.5 - abs(mid - 0.5)) ** 0.5
            for d in [spread, -spread, 0.5 * spread, -0.5 * spread]:
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-16:
                        c5c = c5n
                        improved = True
                        if c5n < bc:
                            bc, bh = c5n, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        if not improved:
            break
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_b = 5e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step_b * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step_b = max(step_b * 0.97, 1e-9)
    return bh, bc

def _basin_bounce(h, n, rng, deadline, n_bounces, alpha, topk, steps):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    cur = h.copy()
    for b in range(n_bounces):
        if time.time() > deadline:
            break
        cur = _nesterov(cur, alpha, topk, steps, 0.003, deadline)
        c5v = _eval(cur)
        if c5v < best_c5:
            best_c5, best_h = c5v, cur.copy()
        if time.time() > deadline:
            break
        pert = rng.randn(n) * 0.02
        pert -= pert.mean()
        cur = _repair(np.clip(cur + pert, 0, 1), n)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 6

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B smooth max, increasing alpha (0-200s)
    dl1 = min(deadline, t0 + 200)
    for alpha, tk, mi in [(1500, 50, 300), (8000, 30, 250),
                          (40000, 18, 200), (200000, 10, 150)]:
        if time.time() > dl1:
            break
        h = _lbfgs_opt(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Nesterov multi-alpha (200-450s)
    dl2 = min(deadline, t0 + 450)
    for a0, tk, st, ns in [(2000, 35, 0.006, 4000),
                           (20000, 20, 0.004, 4000),
                           (150000, 10, 0.002, 3500),
                           (1000000, 6, 0.001, 2500)]:
        if time.time() > dl2:
            break
        h_w = _nesterov(best_h.copy(), a0, tk, ns, st, dl2)
        c5v = _eval(h_w)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_w.copy()

    # Phase 3: Basin bouncing - perturb and re-optimize (450-680s)
    dl3 = min(deadline, t0 + 680)
    h_bb, c5_bb = _basin_bounce(best_h.copy(), n0, rng, dl3,
                                n_bounces=8, alpha=50000, topk=12, steps=800)
    if c5_bb < best_c5:
        best_c5, best_h = c5_bb, h_bb.copy()

    # Phase 4: Aggressive annealing (680-850s)
    dl4 = min(deadline, t0 + 850)
    blocks = [1, 2, 3, 5, 8, 13, 20, 33, 55, 89]
    h_a, c5_a = _anneal(best_h.copy(), 100000, rng, dl4, 0.012, 1e-12, blocks)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a.copy()

    # Phase 5: Pair squeeze (850-980s)
    dl5 = min(deadline, t0 + 980)
    h_p, c5_p = _pair_squeeze(best_h.copy(), dl5, nrounds=300)
    if c5_p < best_c5:
        best_c5, best_h = c5_p, h_p.copy()

    # Phase 6: Final high-alpha Nesterov (980-1060s)
    dl6 = min(deadline, t0 + 1060)
    h_n2 = _nesterov(best_h.copy(), 8000000, 3, 4000, 0.0004, dl6)
    c5v = _eval(h_n2)
    if c5v < best_c5:
        best_c5, best_h = c5v, h_n2.copy()

    # Phase 7: Fine polish (1060-end)
    dl7 = deadline
    rem = max(0.1, dl7 - time.time())
    n_pol = min(int(rem / 0.0012), 80000)
    h_f, c5_f = _fine_polish(best_h.copy(), n_pol, rng, dl7)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
