import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _lse_obj(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    a = alpha * (c - m)
    a = np.clip(a, -500, 500)
    return m + np.log(np.sum(np.exp(a - a.max()))) / alpha

def _lse_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    a = alpha * (c - m)
    a = np.clip(a, -500, 500)
    w = np.exp(a - a.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        g += w[k] * dx * _grad_shift(x, s, n)
    return g

class _Ctx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        return _lse_obj(x, self.alpha, self.n)
    def grad(self, x):
        return _lse_grad(x, self.alpha, self.n)
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    ctx = _Ctx(alpha, n, target)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_top1(h, steps, step0, mu, t_end):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        g = dx * _grad_shift(h, s, n)
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.01, step0 * 20.0)
        else:
            step *= 0.4
            if step < 1e-12:
                v = np.zeros(n)
                step = step0 * 0.3
                h = _repair(h_prev if 'h_prev' in dir() else h, n)
            h_prev = h
    return _repair(h, n)

def _nesterov_topk(h, k_top, steps, step0, mu, t_end):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in top_k:
            s = k - (n - 1)
            g += _grad_shift(h, s, n)
        g *= dx / k_top
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.015, step0 * 15.0)
        else:
            step *= 0.4
            if step < 1e-12:
                v = np.zeros(n)
                step = step0 * 0.4
            h = _repair(h_prev - step * v, n)
            f = _eval(h)
    return _repair(h, n)

def _coord_descent(h, n_passes, t_end, rng):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for _p in range(n_passes):
        if time.time() > t_end:
            break
        perm = rng.permutation(n)
        for idx in perm:
            c_best = best_c
            h_current = best_h.copy()
            base = h_current[idx]
            for frac in [0.15, 0.35, 0.55, 0.75, 0.95]:
                delta = (frac - base) * 0.3
                if abs(delta) < 1e-10:
                    continue
                j = (idx + 1 + rng.randint(n - 2)) % n
                h_current[idx] += delta
                h_current[j] -= delta
                h_current = _repair(h_current, n)
                c_try = _eval(h_current)
                if c_try < c_best:
                    best_h = h_current.copy()
                    best_c = c_try
                    c_best = c_try
                h_current = best_h.copy()
    return best_h, best_c

def _local_search(h, n_tries, rng, t_end, scale):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _basin_hopper(h, rng, t_end, n_rounds=3):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        noise_scale = 0.04
        h_pert = h + noise_scale * rng.randn(n)
        h_pert = _repair(h_pert, n)
        h_opt = _slsqp(h_pert, 3000, 400, 1e-12)
        h_opt = _nesterov_top1(h_opt, 1500, 0.003, 0.85, max(time.time() + 30, t_end))
        c_opt = _eval(h_opt)
        if c_opt < best_c:
            best_h, best_c = h_opt.copy(), c_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = _eval(best_h)

    def _try(h_new):

        c = _eval(h_new)

    # Phase 1: SLSQP broad (15%)
